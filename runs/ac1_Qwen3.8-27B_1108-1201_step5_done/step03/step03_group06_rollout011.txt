def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

"""
Active-set LP + exact subgradient hybrid for minimizing max autocorrelation peak.
Key improvements: active-set LP (much faster per iteration), momentum in updates,
target-M binary search for systematic tightening, diverse initializations,

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_row(a, k):
    """Build one row of the LP constraint matrix for constraint k: conv(a,b)[k] <= M."""
    n = len(a)
    row = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1)
    row[idx] = a[k - idx]
    return row


def solve_lp_active_set(a, M, active_indices, conv_ab=None, max_inner=30):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for k in active_indices.
    Returns (b, updated_active_indices, success).
    """
    n = len(a)
    
    for _ in range(max_inner):
        if len(active_indices) == 0:
            active_indices = np.array([np.argmax(conv_ab) if conv_ab is not None else n - 1])
            
        # Build constraint matrix
        A_ub = np.zeros((len(active_indices), n))
        for row, k in enumerate(active_indices):
            A_ub[row] = build_lp_row(a, k)
        
        b_ub = np.full(len(active_indices), M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
            if not res.success:
                return None, active_indices, False
            b = res.x
        except Exception:
            return None, active_indices, False
        
        # Check for violated constraints (only check a sample for speed)
        if n <= 512:
            conv_ab = fast_cross_convolve(a, b)
        else:
            conv_ab = fast_cross_convolve(a, b)
        
        violated = np.where(conv_ab > M + 1e-10)[0]
        
        if len(violated) == 0:
            return b, active_indices, True
        
        # Add violated constraints
        new_indices = np.setdiff1d(violated, active_indices)
        if len(new_indices) > 0:
            to_add = new_indices[:10]
            active_indices = np.unique(np.concatenate([active_indices, to_add]))
        else:
            return b, active_indices, True
        
        # Cap active set size
        if len(active_indices) > max(4 * n, 100):
            slacks = M - conv_ab[active_indices]
            # Keep the most binding constraints
            sort_idx = np.argsort(slacks)
            active_indices = active_indices[sort_idx[:max(2 * n, 50)]]
    
    return b, active_indices, True


def exact_subgradient(a, conv=None, tol=1e-6):
    """
    Compute a subgradient of max(conv(a,a)) at a.
    Uses uniform weighting over the near-argmax set.
    Returns (grad, conv).
    """
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    
    max_val = conv.max()
    # Use a small tolerance to include near-peaks
    active = conv >= max_val - tol * max_val
    num_active = max(1, active.sum())
    
    if num_active == 0:
        active = np.zeros(len(conv), dtype=bool)
        active[np.argmax(conv)] = True
        num_active = 1
    
    # grad[i] = (1/num_active) * sum_{k in active} [a[k-i] (if valid) + a[i-k] (if valid)]
    # = (1/num_active) * [cross_corr(a, active)[i] + conv(a, active)[i]]
    
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    F = np.fft.fft(active.astype(np.float64), fft_size)
    
    # cross_corr[i] = sum_j a[j] * active[j+i] = ifft(fft(a) * conj(fft(active)))[i]
    cross = np.fft.ifft(A * np.conj(F)).real[:n]
    # conv_term[i] = sum_j a[j] * active[i-j] = ifft(fft(a) * fft(active))[i]
    conv_term = np.fft.ifft(A * F).real[:n]
    
    grad = (cross + conv_term) / num_active
    return grad, conv


def softmax_gradient(a, beta=500.0):
    """Compute smoothed gradient using softmax over convolution peaks."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_term = np.fft.ifft(A * W).real[:n]
    
    grad = cross + conv_term
    return grad, conv

            if g_norm > 1e-15:
                grad /= g_norm
                v = 0.85 * v - 0.02 * grad
                a_new = project_simplex(a + v)
                
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1
                    v *= 0.8
        
        iter_count += 1
        if no_improve > 0 and no_improve % 50 == 0:
            step = max(0.2, step * 0.9)
        
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid iter {iter_count}: obj={best_obj:.6f}, t={elapsed:.1f}s")
    
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Bimodal (edges) - often good for this problem
    for frac in [0.2, 0.3, 0.4]:
        bi = np.zeros(n)
        edge = max(1, int(n * frac))
        bi[:edge] = np.linspace(1.0, 0.3, edge)
        bi[-edge:] = np.linspace(0.3, 1.0, edge)
        bi = np.maximum(bi, 0.05)
        bi /= bi.sum()
        inits[f'bimodal_{frac}'] = bi
    
    # Sharp edge peaks
    sharp = np.zeros(n)
    sharp[0] = 1.0
    sharp[-1] = 1.0
    for i in range(1, min(5, n)):
        sharp[i] = 0.5 ** i
        sharp[-1 - i] = 0.5 ** i
    sharp /= sharp.sum()
    inits['sharp_edge'] = sharp
    
    # Cosine window
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w
    
    # Gaussians
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g
    
    # Split Gaussian (two bumps)
    for gap in [0.2, 0.3]:
        g1 = np.exp(-0.5 * ((x - 0.5 * gap * n) / (0.1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - (1 - 0.5 * gap) * n) / (0.1 * n)) ** 2)
        g = g1 + g2
        g /= g.sum()
        inits[f'double_gauss_{gap}'] = g
    
    # Triangle (single peak center)
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * n) / n
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    inits['triangle'] = tri
    
    # Power-law decay from edges
    for p in [1.0, 2.0, 3.0]:
        d = np.minimum(x, n - 1 - x) + 1.0
        pl = d ** (-p)
        pl /= pl.sum()
        inits[f'powlaw_{p}'] = pl
    
    # Random
    inits['random_spiky'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['random_soft'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits


def evaluate_obj_normalized(a):
    """Evaluate objective assuming sum(a)=1. Returns 2n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with active-set LP, exact subgradient,
    momentum, and diverse initializations. Targets best score below 1.5027.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: length={len(p)}, sum={p.sum():.2f}, obj={evaluate_obj_normalized(p/ p.sum()):.6f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load failed: {e}")
    
    def try_update(a_opt, label=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  NEW BEST [{label}]: score={score:.6f} (n={len(a_opt)})")
            return True
        return False
    
    # ========== Phase 1: Quick Exploration (200s) ==========
    print("=== Phase 1: Quick Exploration ===")
    t1 = time.monotonic()
    phase1_time = 200.0
    
    lengths = [256, 320, 384, 448, 512]
    scan_results = []
    
    for n in lengths:
        if time.monotonic() - t1 > phase1_time - 30:
            break
        inits = generate_initializations(n, parent, rng)
        
        # Priority initializations
        priority = ['parent', 'parent_ds', 'parent_up', 'bimodal_0.3', 'sharp_edge',
                    'bimodal_0.2', 'cosine', 'gauss_0.25', 'random_spiky', 'powlaw_2.0']
        init_names = [nm for nm in priority if nm in inits]
        # Add a few more
        for nm in inits:
            if nm not in init_names and len(init_names) < 6:
                init_names.append(nm)
        
        time_per = max(8.0, (phase1_time - 30) / (len(lengths) * max(1, len(init_names))))
        
        for name in init_names:
            if time.monotonic() - t1 > phase1_time - 15:
                break
            t_alloc = min(time_per * 1.5, phase1_time - 15 - (time.monotonic() - t1))
            if t_alloc < 5:
                break
            
            a0 = inits[name].copy()
            a_opt, obj_opt = lp_optimize_momentum(a0, time_limit=t_alloc,
                                                  step_init=0.7, seed=n + hash(name) % 10000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                try_update(a_opt, f"scan n={n} {name}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best: {best_score:.6f}")
    for s, n, nm, _ in scan_results[:5]:
        print(f"  {s:.6f} n={n} {nm}")
    
    # ========== Phase 2: Deep LP Optimization (600s) ==========
    print("=== Phase 2: Deep LP Optimization ===")
    t2 = time.monotonic()
    phase2_time = 600.0
    
    # Collect diverse starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:10]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)
        if len(deep_starts) >= 6:
            break
    
    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy()))
    
    # Add parent-based starts at various lengths
    if parent is not None:
        for n in [320, 384, 448]:
            inits_n = generate_initializations(n, parent, rng)
            for k in ['parent', 'parent_ds', 'parent_up']:
                if k in inits_n:
                    deep_starts.append((n, inits_n[k].copy()))
                    break
            if len(deep_starts) > 8:
                break
    
    # Different LP configurations
    configs = [
        {'step_init': 0.7, 'momentum': 0.7, 'relax': 1.0},
        {'step_init': 0.5, 'momentum': 0.8, 'relax': 1.0},
        {'step_init': 1.0, 'momentum': 0.5, 'relax': 1.0},
        {'step_init': 0.8, 'momentum': 0.6, 'relax': 0.99},  # tighter
        {'step_init': 0.6, 'momentum': 0.75, 'relax': 1.01},  # looser
    ]
    
    total_runs = len(deep_starts) * len(configs)
    time_per_run = phase2_time / max(1, total_runs)
    time_per_run = max(15.0, time_per_run)
    
    run_count = 0
    for i, (n_start, a_start) in enumerate(deep_starts):
        for j, cfg in enumerate(configs):
            if time.monotonic() - t2 > phase2_time - 20:
                break
            t_alloc = min(time_per_run, phase2_time - 20 - (time.monotonic() - t2))
            if t_alloc < 10:
                break
            
            a_opt, obj_opt = lp_optimize_momentum(a_start, time_limit=t_alloc,
                                                  seed=run_count * 13 + 7, **cfg)
            try_update(a_opt, f"LP [{run_count}] n={n_start} cfg{j}")
            run_count += 1
    
    print(f"Phase 2 done ({time.monotonic()-t2:.0f}s). Best: {best_score:.6f}")
    
    # ========== Phase 3: Subgradient + LP Alternating (400s) ==========
    print("=== Phase 3: Hybrid Alternating ===")
    t3 = time.monotonic()
    phase3_time = 400.0
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        
        # Run hybrid on best sequence
        t_alloc = phase3_time * 0.4
        a_opt, obj_opt = hybrid_alternating(best_sequence, time_limit=t_alloc, seed=1)
        try_update(a_opt, "Hybrid-1")
        
        # Run pure subgradient (exact)
        t_alloc = phase3_time * 0.25
        a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                              lr_init=0.03, method='exact',
                                              momentum=0.92, seed=2)
        try_update(a_opt, "SG-exact")
        
        # Run soft gradient
        t_alloc = phase3_time * 0.15
        a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                              lr_init=0.04, method='soft',
                                              beta=800.0, momentum=0.9, seed=3)
        try_update(a_opt, "SG-soft")
        
        # Hybrid from perturbed start
        t_alloc = phase3_time * 0.2
        noise = rng.standard_normal(n_best) * 0.01
        a_pert = project_simplex(best_sequence + noise)
        a_opt, obj_opt = hybrid_alternating(a_pert, time_limit=t_alloc, seed=4)
        try_update(a_opt, "Hybrid-pert")
    
    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s). Best: {best_score:.6f}")
    
    # ========== Phase 4: Try different n (200s) ==========
    print("=== Phase 4: Different n exploration ===")
    t4 = time.monotonic()
    phase4_time = 200.0
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        # Try nearby n values
        for n_try in [n_best + 32, n_best - 32, n_best + 64, n_best - 64]:
            if n_try < 128 or n_try > 1024:
                continue
            if time.monotonic() - t4 > phase4_time - 30:
                break
            # Interpolate from best
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                np.linspace(0, 1, len(best_sequence)), best_sequence)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s > 1e-10:
                a_interp /= s
            
            t_alloc = min(phase4_time / 4, phase4_time - 30 - (time.monotonic() - t4))
            if t_alloc < 15:
                break
            
            a_opt, obj_opt = lp_optimize_momentum(a_interp, time_limit=t_alloc,
                                                  step_init=0.6, momentum=0.7, seed=n_try)
            try_update(a_opt, f"n-scan n={n_try}")
    
    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s). Best: {best_score:.6f}")
    
    # ========== Phase 5: Final Polish (remaining time) ==========
    print("=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    remaining = deadline - time.monotonic()
    
    if best_sequence is not None and remaining > 30:
        n_best = len(best_sequence)
        
        # Very fine LP polishing
        t_alloc = min(remaining * 0.3, 80.0)
        if t_alloc > 15:
            a_opt, obj_opt = lp_optimize_momentum(best_sequence, time_limit=t_alloc,
                                                  step_init=0.3, momentum=0.8,
                                                  relax=1.0, seed=999)
            try_update(a_opt, "Polish-LP")
        
        # Fine subgradient
        t_alloc = min(remaining * 0.25, 60.0)
        if t_alloc > 10:
            a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                                  lr_init=0.01, method='exact',
                                                  momentum=0.95, seed=777)
            try_update(a_opt, "Polish-SG")
        
        # Tiny perturbations
        t_sub = min(remaining * 0.2, 40.0)
        if t_sub > 5:
            t_pert_start = time.monotonic()
            for i in range(50):
                if time.monotonic() - t_pert_start > t_sub - 5:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb [{i}] s={scale}: {best_score:.6f}")
                        break
        
        # One more hybrid round if time
        t_remain = deadline - time.monotonic() - 10
        if t_remain > 30:
            a_opt, obj_opt = hybrid_alternating(best_sequence, time_limit=t_remain, seed=1234)
            try_update(a_opt, "Final-Hybrid")
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
