"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
frequency-domain initialized projected gradient descent, smooth L-BFGS-B with
beta annealing, peak-targeted refinement, and adaptive length selection.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve_full(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_gradient(a, w):
    """
    Gradient of sum_k w_k * conv_full[k] w.r.t. a.
    Uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    fft_size = 1 << (3 * n).bit_length()
    A = fft(a, fft_size)
    W = fft(w, fft_size)
    cross_corr = np.real(ifft(np.conj(A) * W, fft_size))
    return 2.0 * cross_corr[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(rho_candidates[cond])
    theta = (cssv[int(rho) - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum_one(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_objective(a):
    """Compute 2n * max(conv) for sum-normalized sequence."""
    n = len(a)
    conv = fast_convolve_full(a)
    return 2.0 * n * conv.max()


def pg_descent(a0, beta, lr, n_iter, time_limit=None, t_start=None):
    """
    Projected gradient descent on the simplex with smooth max objective.
    Returns (best_a, best_obj).
    """
    n = len(a0)
    a = normalize_sum_one(a0.copy())
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        conv = fast_convolve_full(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        grad = compute_gradient(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        t = lr
        for _ in range(20):
            a_new = project_simplex(a - t * grad)
            conv_new = fast_convolve_full(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5
        else:
            break
        
        a = a_new
        lr *= 0.999
    
    # Update with actual (non-smooth) objective
    actual_obj = compute_objective(best_a)
    if actual_obj < best_obj:
        best_obj = actual_obj
    
    return best_a, best_obj


def lbfgsb_smooth(a0, beta, maxiter):
    """
    L-BFGS-B optimization with softmax reparametrization on the simplex.
    """
    n = len(a0)
    a_norm = normalize_sum_one(a0.copy())
    a_safe = np.maximum(a_norm, 1e-30)
    x0 = np.log(a_safe)
    
    def obj_func(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve_full(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac_func(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve_full(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = compute_gradient(a, w)
        # Gradient through softmax: d obj / d x = a * (grad_a - <a, grad_a>)
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)
    
    result = minimize(obj_func, x0, jac=jac_func, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-12, 'gtol': 1e-8})
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun

            best_obj = obj
            best_a = a.copy()
    
    return best_a


def peak_targeted_refine(a, n_steps=100, time_limit=None, t_start=None):
    """
    Iteratively identify the peak and reduce it via gradient steps.
    """
    n = len(a)
    a = normalize_sum_one(a.copy())
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    lr = 0.1
    for step in range(n_steps):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        conv = fast_convolve_full(a)
        peak_val = conv.max()
        peak_idx = int(np.argmax(conv))
        
        # Compute the exact subgradient at the peak
        # grad[j] = 2 * a[peak_idx - j] (if valid) ... wait
        # From our derivation: grad[j] = 2 * sum_k w_k * a[k-j] (for valid k-j)
        # At the peak only: w has 1 at peak_idx
        # grad[j] = 2 * a[peak_idx - j] if 0 <= peak_idx - j < n
        #          = 2 * a[j - peak_idx] if 0 <= j - peak_idx < n (from the symmetric term)
        # Wait, I derived: grad[j] = 2 * sum_m a[m] * w[j+m]
        # With w concentrated at peak_idx: grad[j] = 2 * a[j] * w[2j + peak_idx - j]...
        
        # Actually from the derivation: v[j] = sum_m a[m] * w[j+m]
        # With w = e_{peak}: v[j] = a[peak - j] (if 0 <= peak-j < n)
        # So grad[j] = 2 * a[peak - j] (if 0 <= peak - j < n)
        
        # Wait, that's not right either. Let me re-derive.
        # We showed: grad[j] = 2 * IFFT(conj(FFT(a)) * FFT(w))[:n][j]
        # If w = e_{peak} (one-hot at position peak):
        # FFT(w)[k] = e^{-2πi*peak*k/N}
        # IFFT(conj(A) * W)[j] = (1/N) sum_k conj(A[k]) * e^{-2πi*peak*k/N} * e^{2πijk/N}
        # = (1/N) sum_k conj(A[k]) * e^{2πi(k(j-peak))/N}
        # = a[j - peak] (circular, but with enough padding it's linear)
        # So grad[j] = 2 * a[j - peak] (if 0 <= j - peak < n, else 0)
        # Hmm, that gives negative indices. Let me reconsider.
        
        # Actually with proper zero-padding, IFFT(conj(FFT(a)) * FFT(w)) gives:
        # result[j] = sum_k a[k] * w[j+k] (from our earlier derivation)
        # With w = e_{peak}: result[j] = a[j + peak - peak] ... no.
        # result[j] = sum_k a[k] * w[j+k] = a[j + (2n-2) - peak + peak - (2n-2)] 
        # I'm confusing myself. Let me just use the FFT formula directly.
        
        # From the derivation: IFFT(conj(FFT(a)) * FFT(w))[n] = sum_j a[j] * w[n+j]
        # With w = e_peak: = a[n + peak - peak]... no.
        # = sum_j a[j] * e_{peak}[n+j] = a[n+peak] if n+peak is a valid index... 
        # That doesn't seem right.
        
        # Let me just go back to basics.
        # conv[k] = sum_i a[i] * a[k-i] (where a[x]=0 for x outside [0,n-1])
        # f = sum_k w_k * conv[k]
        # df/da[j] = sum_k w_k * d(conv[k])/da[j]
        # d(conv[k])/da[j] = a[k-j] + a[j-k] (both terms, where defined)
        #   = a[k-j] if 0 <= k-j <= n-1, plus a[j-k] if 0 <= j-k <= n-1
        #   Note: a[k-j] and a[j-k] are the SAME term when we account for the sum over i.
        #   Actually: conv[k] = sum_i a[i]*a[k-i]
        #   d/da[j]: the term a[i] gives a[k-j] (when i=j), and a[k-i] gives a[j] (when k-i=j, i.e. i=k-j)
        #   So d(conv[k])/da[j] = a[k-j] + a[j] * [k-j in [0,n-1]]
        #   Wait no: a[k-i] with i = k-j gives a[j]. So:
        #   d(conv[k])/da[j] = a[k-j] (from i=j) + a[j] (from i=k-j)
        #   But these are the same when k-j = j, i.e., k = 2j.
        #   In general: d(conv[k])/da[j] = a[k-j] + a[j] * delta(k-j in [0,n-1])
        #   Hmm, that's not right. Let me be very careful.
        
        # conv[k] = sum_{i=0}^{n-1} a[i] * a[k-i]
        # where a[x] = 0 if x < 0 or x >= n.
        
        # Partial derivative w.r.t. a[j]:
        # d(conv[k])/da[j] = sum_i (delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j})
        # = a[k-j] + sum_i a[i] * delta_{k-i,j}
        # = a[k-j] + a[k-j] (the sum is 1 when k-j is in [0,n-1], giving a[k-j])
        # Wait: delta_{k-i,j} means k-i=j, so i=k-j. The sum gives a[k-j] * 1 (if k-j is in [0,n-1]).
        # So d(conv[k])/da[j] = a[k-j] + a[k-j] = 2*a[k-j] (when 0 <= k-j <= n-1)
        # And 0 otherwise.
        
        # So grad[j] = sum_k w_k * 2 * a[k-j] * [0 <= k-j <= n-1]
        # = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]  (where m = k-j, so k = j+m)
        
        # With w = e_peak: grad[j] = 2 * a[j + peak - j] ... no.
        # grad[j] = 2 * w[j + ... ] 
        # m = k - j, k = peak: m = peak - j
        # grad[j] = 2 * a[peak - j] (if 0 <= peak - j <= n-1, i.e., j <= peak <= j+n-1)
        #          = 2 * a[peak - j] (if peak - n + 1 <= j <= peak)
        
        # So for the peak at index p:
        # grad[j] = 2 * a[p - j] for j in [max(0, p-n+1), min(n-1, p)]
        #          = 0 otherwise
        
        p = peak_idx
        grad = np.zeros(n)
        j_start = max(0, p - n + 1)
        j_end = min(n - 1, p)
        grad[j_start:j_end + 1] = 2.0 * a[p - j_start:p - j_end + 1][::-1]
        
        # Projected gradient step
        for _ in range(5):
            a_new = project_simplex(a - lr * grad)
            if not np.allclose(a_new, a):
                break
            lr *= 0.5
        
        a = a_new
        obj = compute_objective(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        lr *= 0.98
    
    return best_a, best_obj


def adaptive_smooth_optimize(a0, time_budget, t_start):
    """
    Multi-stage smooth optimization with decreasing beta.
    """
    n = len(a0)
    a = normalize_sum_one(a0.copy())
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    betas = [200, 100, 50, 25, 12, 6, 3, 2, 1.5, 1]
    stages = len(betas)
    time_per_stage = time_budget / stages
    
    for stage_idx, beta in enumerate(betas):
        remaining = (t_start + time_budget) - time.monotonic()
        if remaining < 5:
            break
        
        t_alloc = min(time_per_stage, remaining - 5)
        if t_alloc < 3:
            break
        
        # Use L-BFGS-B for this stage
        maxiter = int(min(300, max(20, t_alloc * 20)))
        a_opt, _ = lbfgsb_smooth(a, beta, maxiter)
        obj = compute_objective(a_opt)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
        
        a = a_opt
    
    return best_a, best_obj


def random_perturb_and_optimize(a_best, n_perturbations, time_budget, rng, t_start):
    """
    Perturb the best solution and re-optimize.
    """
    n = len(a_best)
    best_a = a_best.copy()
    best_obj = compute_objective(a_best)
    
    perturbations = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]
    time_per_pert = time_budget / n_perturbations
    
    for i in range(n_perturbations):
        remaining = (t_start + time_budget) - time.monotonic()
        if remaining < 10:
            break
        
        scale = perturbations[i % len(perturbations)]
        noise = rng.standard_normal(n) * scale
        a_pert = project_simplex(normalize_sum_one(a_best) + noise)
        
        # Quick optimization
        t_alloc = min(time_per_pert, remaining - 5)
        if t_alloc < 5:
            break
        maxiter = int(min(200, max(20, t_alloc * 15)))
        a_opt, _ = lbfgsb_smooth(a_pert, beta=10, maxiter=maxiter)
        obj = compute_objective(a_opt)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search to minimize 2n*max(conv)/(sum)^2.
    Uses frequency-domain initialization, smooth L-BFGS-B with beta annealing,
    peak-targeted refinement, and adaptive length selection.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass
    
    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            elapsed = time.monotonic() - t_start
            print("  New best: score=%.6f, n=%d (t=%.0fs)" % (score, len(a), elapsed))
        return score if np.isfinite(score) else float('inf')
    
    # Phase 1: Quick exploration
    print("=== Phase 1: Quick Exploration ===")
    
    # Determine candidate lengths
    if parent is not None:
        parent_n = len(parent)
        candidate_lengths = sorted(set([
            max(64, parent_n // 2),
            parent_n,
            min(1024, parent_n + parent_n // 4),
            min(1024, parent_n * 2),
        ]))
    else:
        candidate_lengths = [128, 192, 256, 320, 384, 512]
    
    # Add some extra lengths
    for n in [160, 224, 288, 352, 448]:
        if n not in candidate_lengths:
            candidate_lengths.append(n)
    candidate_lengths = sorted(set([n for n in candidate_lengths if 64 <= n <= 1024]))
    
    phase1_time = min(400.0, (deadline - t_start) * 0.25)
    phase1_deadline = t_start + phase1_time
    
    scan_results = []
    
    for n in candidate_lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        
        # Generate initializations
        inits = []
        inits.append(('uniform', np.ones(n) / n))
        
        # Edge-weighted
        x = np.arange(1, n + 1, dtype=np.float64)
        for alpha in [0.3, 0.7]:
            a = (x * (n + 1 - x)) ** alpha
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            inits.append(('edge_%s' % alpha, a))
        
        # Flat spectrum
        a_flat = flat_spectrum_init(n, rng)
        inits.append(('flat_spec', a_flat))
        
        # Symmetric random
        a_sym = symmetric_init(n, rng)
        inits.append(('sym', a_sym))
        
        # From parent
        if parent is not None:
            if len(parent) == n:
                p_norm = np.maximum(parent, 0)
                p_norm /= p_norm.sum()
                inits.append(('parent', p_norm))
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(parent)),
                                      np.maximum(parent, 0))
                p_interp = np.maximum(p_interp, 0)
                p_interp /= p_interp.sum()
                inits.append(('parent_interp', p_interp))
        
        # Random Dirichlet
        a_rand = rng.dirichlet(np.ones(n) * 0.3)
        inits.append(('random', a_rand))
        
        # Quick optimize each
        time_per_init = max(10.0, (phase1_deadline - time.monotonic() - 10) / len(inits))
        
        for name, a0 in inits:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(time_per_init, phase1_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            
            # Quick gradient descent
            n_iters = int(min(200, max(50, t_alloc * 30)))
            a_opt, obj = pg_descent(a0, beta=50, lr=0.1, n_iter=n_iters,
                                     time_limit=t_alloc, t_start=time.monotonic())
            score = score_and_update(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
    
    scan_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best: %.6f (n=%d)" % (best_score, len(best_sequence) if best_sequence is not None else 0))
    
    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    
    # Select top candidates
    top_candidates = scan_results[:5] if len(scan_results) >= 5 else scan_results
    if best_sequence is not None:
        top_candidates.insert(0, (best_score, len(best_sequence), 'best', best_sequence.copy()))
    
    phase2_deadline = t_start + (deadline - t_start) * 0.85
    n_configs = max(1, len(top_candidates))
    time_per_config = max(30.0, (phase2_deadline - time.monotonic()) / n_configs)
    
    for score, n, name, a_init in top_candidates:
        if time.monotonic() > phase2_deadline - 30:
            break
        
        t_alloc = min(time_per_config, phase2_deadline - time.monotonic() - 20)
        if t_alloc < 20:
            break
        
        print("  Optimizing candidate n=%d, name=%s, init_score=%.6f, time_alloc=%.0fs" % (n, name, score, t_alloc))
        
        # Adaptive smooth optimization with beta annealing
        a_opt, obj = adaptive_smooth_optimize(a_init, t_alloc, t_start)
        score = score_and_update(a_opt)
    
    print("Phase 2 done. Best: %.6f" % best_score)
    
    # Phase 3: Fine-tuning
    print("=== Phase 3: Fine-tuning ===")
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)
        
        # 3a: Very low beta L-BFGS-B
        if remaining > 30:
            t_alloc = min(120.0, remaining * 0.3)
            maxiter = int(min(500, max(100, t_alloc * 5)))
            a_opt, _ = lbfgsb_smooth(best_sequence, beta=3, maxiter=maxiter)
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 20.0
        
        # 3b: Peak-targeted refinement
        if remaining > 30:
            t_alloc = min(120.0, remaining * 0.3)
            n_steps = int(min(200, t_alloc * 5))
            a_opt, obj = peak_targeted_refine(best_sequence, n_steps=n_steps,
                                               time_limit=t_alloc, t_start=time.monotonic())
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 20.0
        
        # 3c: Random perturbations
        if remaining > 20:
            n_pert = max(3, int(remaining / 30))
            a_opt, obj = random_perturb_and_optimize(best_sequence, n_pert,
                                                      remaining - 20, rng, t_start)
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 20.0
        
        # 3d: Final PG with very low beta
        if remaining > 15:
            n_iters = int(min(300, remaining * 10))
            a_opt, obj = pg_descent(best_sequence, beta=5, lr=0.05, n_iter=n_iters,
                                     time_limit=remaining - 15, t_start=time.monotonic())
            score_and_update(a_opt)
    
    # Phase 4: Try alternative lengths if time remains
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 60:
            alt_lengths = []
            n_best = len(best_sequence)
            for factor in [0.75, 1.25, 1.5, 2.0]:
                alt_n = int(n_best * factor)
                if 64 <= alt_n <= 1024 and alt_n != n_best:
                    alt_lengths.append(alt_n)
            
            for alt_n in alt_lengths[:2]:
                if remaining < 30:
                    break
                # Interpolate best to new length
                a_interp = np.interp(np.linspace(0, 1, alt_n),
                                      np.linspace(0, 1, n_best),
                                      best_sequence)
                a_interp = np.maximum(a_interp, 0)
                a_interp /= a_interp.sum()
                
                t_alloc = min(60.0, remaining - 10)
                a_opt, obj = adaptive_smooth_optimize(a_interp, t_alloc, t_start)
                score = score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 10.0
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
