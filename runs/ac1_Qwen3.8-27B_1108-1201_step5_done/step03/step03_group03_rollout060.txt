import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coordinate_descent(a, max_rounds=50, time_limit=None):
    """Coordinate descent: for each pair (i,j), try optimal transfer."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    for rnd in range(max_rounds):
        if time_limit and time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Try moving mass from high-contributing to low-contributing positions
        # Use gradient-based ordering
        # The subgradient at the peak: for peak at k*, grad[i] = a[k*-i] + a[i-k*]
        peaks = np.argsort(conv)[::-1][:min(5, 2*n-1)]

            if a[i] < 1e-9:
                continue
            # Compute contribution of a[i] to each convolution entry

            # So changing a[i] by delta changes conv[k] by delta*(a[k-i] + a[i-k]) for valid k

            # minimizes the increase in the peak.

            candidates = [j for j in range(n) if j != i][:20]
            # Prioritize j values that contribute less to the current peaks

                    if delta < 1e-10:

                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    if a_test[i] < 0:

                    st = a_test.sum()
                    if st < 1e-15:

                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-14:

                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        break
        if not improved:

    return a

def subgradient_descent(a, time_limit, lr_init=0.01, momentum=0.9, top_k=20):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:

        top_k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        # Compute subgradient (average of gradients at top peaks, weighted)
        # For peak at k, gradient of conv[k] w.r.t. a[i] is:

        sg = np.zeros(n)

            k = int(peaks[idx])

                ik = i - k
                g = 0.0

                if 0 <= ik < n:
                    g += a[ik]
                sg[i] += wi * g

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

            v *= 0.5
            no_improve = 0
            lr = max(0.001, lr * 0.5)
        
        lr *= 0.999
        
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Fast coordinate descent with controlled transfers
2. Subgradient-based optimization
3. Simulated annealing with structured moves
4. Multi-length and multi-initialization exploration

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def get_obj(a):
    """Compute objective for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_grad_at_k(k, a):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_end >= i_start:
        idx = k - np.arange(i_start, i_end + 1)
        grad[i_start:i_end + 1] = 2.0 * a[idx]
    return grad

def coordinate_descent(a, time_limit=60.0, nfracs=5):
    """Coordinate descent: transfer mass between pairs to reduce peak."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    fracs = np.linspace(0.05, 0.8, nfracs)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Get the peak and its gradient
        k_peak = int(np.argmax(conv))
        grad_peak = conv_grad_at_k(k_peak, a)
        
        # Sort positions by their contribution to the peak (descending)
        order_i = np.argsort(-grad_peak)  # positions that most affect the peak
        order_j = np.argsort(grad_peak)   # positions that least affect the peak
        
        # Try transferring from high-contribution to low-contribution
        for i in order_i[:min(30, n)]:
            if a[i] < 1e-8:
                continue
            if time.monotonic() - t0 > time_limit:
                break
            
            for j in order_j[:min(30, n)]:
                if j == i:
                    continue
                if grad_peak[j] >= grad_peak[i] * 0.5:
                    continue  # j also significantly contributes to peak
                
                # Try different transfer fractions
                for frac in fracs:
                    delta = a[i] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    if a_test[i] < -1e-12:
                        continue
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-10:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-14:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        break
                if improved:
                    # Recompute gradient after improvement
                    k_peak = int(np.argmax(conv))
                    grad_peak = conv_grad_at_k(k_peak, a)
                    order_i = np.argsort(-grad_peak)
                    order_j = np.argsort(grad_peak)
                    break
        
        if not improved:
            break
    
    a = a / a.sum()
    return a

def subgrad_descent(a, time_limit=60.0, lr_init=0.005, momentum=0.9, top_k=10):
    """Subgradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        top_k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        # Softmax weighting
        temp = 0.05 * pmax + 1e-10
        w = np.exp((peak_vals - pmax) / temp)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-8:
                continue
            k = int(peaks[idx])
            wi = w[idx]
            # grad[i] = 2*a[k-i] for valid i
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end >= i_start:
                idx_arr = k - np.arange(i_start, i_end + 1)
                sg[i_start:i_end + 1] += wi * 2.0 * a[idx_arr]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.0005, lr * 0.5)
        
        lr *= 0.998
    
    return best_a, best_obj

def simulated_annealing(a, time_limit=60.0, t_init=0.01, t_min=0.0001, cooling=0.999):
    """Simulated annealing with structured moves."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = t_init
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit and T > t_min:
        # Choose a random move
        move_type = rng.integers(0, 3)
        a_try = a.copy()
        
        if move_type == 0:
            # Transfer between two random positions
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.3)
            delta = a[i] * frac
            a_try[i] -= delta
            a_try[j] += delta
        elif move_type == 1:
            # Gaussian perturbation
            perturbation = rng.standard_normal(n) * 0.01
            a_try = project_simplex(a + perturbation)
        else:
            # Swap-like: reduce a random element, increase a neighbor
            i = rng.integers(0, n - 1)
            j = i + 1
            frac = rng.uniform(0.01, 0.2)
            delta = min(a[i], a[j]) * frac
            if rng.random() < 0.5:
                a_try[i] -= delta
                a_try[j] += delta
            else:
                a_try[i] += delta
                a_try[j] -= delta
        
        a_try = np.maximum(a_try, 0.0)
        st = a_try.sum()
        if st < 1e-10:
            continue
        a_try /= st
        
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        
        # Accept with simulated annealing probability
        if obj_try < best_obj:
            a = a_try
            conv = conv_try
            best_obj = obj_try
            best_a = a_try.copy()
        elif rng.random() < np.exp(-(obj_try - best_obj) / (T + 1e-15)):
            a = a_try
            conv = conv_try
        
        T *= cooling
    
    return best_a, best_obj

def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Inverse triangle (U-shaped, mass at edges)

    inits['inv_triangle'] = inv_tri / inv_tri.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.15 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Power law
    pl = (x + 1.0) ** 2
    inits['power'] = pl / pl.sum()
    
    # Sine
    sines = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine'] = sines / sines.sum()
    
    if parent is not None and len(parent) > 0:

                ps2 = p_res.sum()
                if ps2 > 1e-10:

    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def get_obj_normalized(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_peak_gradients(peaks, a):
    """Compute weighted sum of gradients at given peaks."""
    n = len(a)
    sg = np.zeros(n)
    for k in peaks:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            idx = k - np.arange(i_start, i_end + 1)
            sg[i_start:i_end + 1] += 2.0 * a[idx]
    return sg

def multi_peak_coord_descent(a, time_limit=60.0, npeaks=3, nfracs=7, max_i=40, max_j=40):
    """Coordinate descent considering multiple peaks simultaneously."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    fracs = np.linspace(0.03, 0.7, nfracs)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Get top peaks
        npeaks_actual = min(npeaks, 2 * n - 1)
        peaks = np.argpartition(conv, -npeaks_actual)[-npeaks_actual:]
        
        # Compute combined gradient (sum of gradients at all peaks)
        combined_grad = compute_peak_gradients(peaks, a)
        
        # Sort by combined gradient
        order_high = np.argsort(-combined_grad)[:max_i]
        order_low = np.argsort(combined_grad)[:max_j]
        
        for i in order_high:
            if a[i] < 1e-9:
                continue
            if time.monotonic() - t0 > time_limit:
                break
            
            for j in order_low:
                if j == i:
                    continue
                if combined_grad[j] > combined_grad[i] * 0.3:
                    continue
                
                best_delta = None
                best_local_obj = best_obj
                
                for frac in fracs:
                    delta = a[i] * frac
                    if delta < 1e-11:
                        continue
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < -1e-12:
                        continue
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st < 1e-10:
                        continue
                    a_try /= st
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_local_obj:
                        best_local_obj = obj_try
                        best_delta = (a_try, conv_try)
                
                if best_delta is not None and best_local_obj < best_obj - 1e-14:
                    a, conv = best_delta
                    best_obj = best_local_obj
                    improved = True
                    # Recompute gradients
                    npeaks_actual = min(npeaks, 2 * n - 1)
                    peaks = np.argpartition(conv, -npeaks_actual)[-npeaks_actual:]
                    combined_grad = compute_peak_gradients(peaks, a)
                    order_high = np.argsort(-combined_grad)[:max_i]
                    order_low = np.argsort(combined_grad)[:max_j]
                    break
        
        if not improved:
            break
    
    a = a / a.sum()
    return a

def subgrad_mwu(a, time_limit=60.0, lr_init=0.005, momentum=0.9, top_k=15, mode='subgrad'):
    """Subgradient or multiplicative weight update optimization."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        top_k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        # Weight peaks by softmax
        temp = 0.03 * pmax + 1e-12
        w = np.exp((peak_vals - pmax) / temp)
        w_sum = w.sum()
        w = w / w_sum if w_sum > 1e-15 else np.array([1.0] + [0.0] * (len(peaks) - 1))
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-8:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end >= i_start:
                idx_arr = k - np.arange(i_start, i_end + 1)
                sg[i_start:i_end + 1] += w[idx] * 2.0 * a[idx_arr]
        
        if mode == 'subgrad':
            v = momentum * v - lr * sg
            a = project_simplex(a + v)
        else:  # MWU
            a = a * np.exp(-lr * sg)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 80:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.0003, lr * 0.4)
        
        lr *= 0.997
    
    return best_a, best_obj

def simulated_annealing(a, time_limit=30.0, t_init=0.008, t_min=0.00005, cooling=0.9995):
    """Simulated annealing with structured moves."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = t_init
    rng = np.random.default_rng(555)
    
    while time.monotonic() - t0 < time_limit and T > t_min:
        move_type = rng.integers(0, 4)
        a_try = a.copy()
        
        if move_type == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                T *= cooling
                continue
            frac = rng.uniform(0.01, 0.25)
            delta = min(a[i], a[j]) * frac if a[i] < a[j] else a[i] * frac
            a_try[i] -= delta
            a_try[j] += delta
        elif move_type == 1:
            scale = rng.uniform(0.005, 0.02)
            a_try = project_simplex(a + rng.standard_normal(n) * scale)
        elif move_type == 2:
            # Block swap: transfer mass from a block to another block
            block_size = max(1, n // 16)
            i_start = rng.integers(0, n - 2 * block_size)
            j_start = rng.integers(0, n - 2 * block_size)
            if abs(i_start - j_start) < block_size:
                T *= cooling
                continue
            total_i = a_try[i_start:i_start + block_size].sum()
            total_j = a_try[j_start:j_start + block_size].sum()
            if total_i < 1e-10:
                T *= cooling
                continue
            frac = rng.uniform(0.05, 0.3)
            delta = total_i * frac
            a_try[i_start:i_start + block_size] -= delta / block_size
            a_try[j_start:j_start + block_size] += delta / block_size
        else:
            # Local smoothing: average with neighbors
            i = rng.integers(1, n - 1)
            window = rng.integers(1, 4)
            i_lo = max(0, i - window)
            i_hi = min(n, i + window + 1)
            local_mean = a_try[i_lo:i_hi].mean()
            a_try[i_lo:i_hi] = a_try[i_lo:i_hi] * 0.5 + local_mean * 0.5
        
        a_try = np.maximum(a_try, 0.0)
        st = a_try.sum()
        if st < 1e-10:
            T *= cooling
            continue
        a_try /= st
        
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        
        if obj_try <= best_obj:
            a = a_try
            conv = conv_try
            best_obj = obj_try
            best_a = a_try.copy()
        elif rng.random() < np.exp(-(obj_try - best_obj) / max(T, 1e-15)):
            a = a_try
            conv = conv_try
        
        T *= cooling
    
    return best_a, best_obj
