def project_simplex(v):
    n = len(v)
    if n == 1:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using asymmetric ramp-like initializations
with conservative smooth-gradient local search and LP refinement.
Key insight: asymmetric sequences (especially ramps) have much lower auto-convolution
peaks than symmetric ones, as the cross-pairing at the center reduces sum of products.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def objective_from_conv(a, c):
    """Compute objective given precomputed convolution."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(a) * c.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_weights(c, T):
    """Compute softmax weights for smooth max approximation."""
    c_shifted = c - c.max()
    exp_c = np.exp(T * c_shifted)
    total = exp_c.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return exp_c / total


def grad_smooth_max(a, c, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*c_k)) w.r.t. a.
    Uses the identity: grad[i] = 2 * sum_m a[m] * w[i+m]
    where w_k = T * exp(T*c_k) / sum_j exp(T*c_j)
    """
    n = len(a)
    w = smooth_max_weights(c, T)
    # grad[i] = 2 * sum_{m=0}^{n-1} a[m] * w[i+m] for i=0,...,n-1
    # This is 2 * correlate(w, a, 'full')[:n]
    grad = 2.0 * np.correlate(w, a, mode='full')[:n]
    return grad


def smooth_gradient_optimize(a0, time_limit, T_init=3.0, T_max=200.0,
                              T_growth=1.08, lr_init=0.005,
                              momentum=0.95, min_improve=1e-12):
    """
    Conservative smooth gradient optimization with temperature annealing.
    Only accepts steps that improve the objective (with a small tolerance).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = objective_from_conv(a, c)
    best_a = a.copy()

    v = np.zeros(n)
    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        grad = grad_smooth_max(a, T)

        # Momentum step
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        c_new = fast_convolve(a_new)
        obj_new = objective_from_conv(a_new, c_new)

        if obj_new < best_obj - min_improve:
            a = a_new
            c = c_new
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            # Don't accept; reduce momentum
            v *= 0.5
            stall += 1

        if stall > 50:
            # Reduce learning rate
            lr *= 0.85
            stall = 0
            v = np.zeros(n)
            # Slight perturbation to escape
            if lr < lr_init * 0.1:
                # Try a random restart from best
                a = project_simplex(best_a + np.random.default_rng(int(time.time()*1000) % 2**31).standard_normal(n) * 0.002)
                c = fast_convolve(a)
                lr = lr_init * 0.5
                v = np.zeros(n)

        # Increase temperature periodically
        if stall % 30 == 0 and T < T_max:
            T = min(T * T_growth, T_max)

        if lr < 1e-6:
            break

    return best_a, best_obj


def top_k_subgrad_optimize(a0, time_limit, lr_init=0.008, momentum=0.92,
                            top_k=30, T_param=0.001):
    """
    Subgradient method focusing on top-k peaks with temperature-based weighting.
    Conservative: only accepts improvements.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = objective_from_conv(a, c)
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        peaks = np.argpartition(c, -top_k)[-top_k:]
        peak_vals = c[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Temperature-based weighting
        scaled = peak_vals / (T_param + pmax * 0.03)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)
        c_new = fast_convolve(a_new)
        obj_new = objective_from_conv(a_new, c_new)

        if obj_new < best_obj - 1e-13:
            a = a_new
            c = c_new
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            v *= 0.5
            stall += 1

        if stall > 100:
            lr *= 0.8
            v = np.zeros(n)
            if lr < 1e-5:
                a = project_simplex(best_a + np.random.default_rng(42 + int(stall)).standard_normal(n) * 0.003)
                c = fast_convolve(a)
                lr = lr_init * 0.3
                v = np.zeros(n)
                stall = 0

    return best_a, best_obj


def build_lp_matrix(n):
    """Build the constraint matrix for LP direction finding."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = 1.0  # Will be scaled by f values
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 10})
        if result.success:
            g = np.maximum(result.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_refine(a0, time_limit, blend_init=0.05, blend_min=0.002):
    """
    LP-based directional optimization with very conservative blending.
    Only accepts improvements.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = objective_from_conv(a, c)
    best_a = a.copy()

    blend = blend_init
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = c.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, n)
        if g is None:
            break

        # Try the blend
        a_new = (1.0 - blend) * a + blend * g
        a_new = project_simplex(a_new)
        c_new = fast_convolve(a_new)
        obj_new = objective_from_conv(a_new, c_new)

        if obj_new < best_obj - 1e-13:
            a = a_new
            c = c_new
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 8:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break

    return best_a, best_obj


def coordinate_descent(a_in, max_rounds=50, time_limit=60.0):
    """
    Coordinate descent: try moving small amounts between pairs of coordinates.
    Only accepts improvements.
    """
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = objective_from_conv(a, c)
    best_a = a.copy()

    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Focus on elements near the edges and elements contributing to peaks
        peak_indices = np.argpartition(c, -20)[-20:]

        for i in range(0, n, max(1, n // 200)):
            if a[i] < 1e-8:
                continue
            # Try moving mass to nearby positions
            for dj in [-3, -2, -1, 1, 2, 3, -10, 10, -50, 50]:
                j = i + dj
                if j < 0 or j >= n:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = a[i] * frac
                    if delta < 1e-9:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    if a_test[i] < -1e-12:
                        continue
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    c_test = fast_convolve(a_test)
                    obj_test = objective_from_conv(a_test, c_test)
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        c = c_test
                        best_obj = obj_test
                        best_a = a_test.copy()
                        improved = True
        if not improved and rnd > 10:
            break

    return best_a


def gen_ramp_inits(n, rng):
    """Generate diverse ramp-like and asymmetric initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    xm = x / max(n - 1, 1)  # normalized to [0, 1]

    # Various power ramps
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        s = (xm + 0.01) ** alpha
        inits[f'ramp_{alpha}'] = s / s.sum()

    # Exponential ramps
    for rate in [0.5, 1.0, 2.0, 3.0, 5.0]:
        s = np.exp(rate * xm)
        inits[f'exp_{rate}'] = s / s.sum()

    # Reversed ramps
    for alpha in [0.5, 1.0, 2.0]:
        s = (1.0 - xm + 0.01) ** alpha
        inits[f'rev_ramp_{alpha}'] = s / s.sum()

    # Log-sine ramps (asymmetric bump)
    for shift in [0.2, 0.35, 0.5, 0.65, 0.8]:
        s = np.sin(np.pi * np.clip(xm, 0.01, 0.99)) ** 2
        s *= np.exp(2.0 * (xm - shift))
        inits[f'asym_bump_{shift}'] = s / s.sum()

    # Standard shapes
    inits['uniform'] = np.ones(n) / n
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge-concentrated
    e = max(1, n // 5)
    edge = np.zeros(n)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # One-side edge
    edge1 = np.zeros(n)
    edge1[:e] = 1.0
    inits['edge_left'] = edge1 / edge1.sum()
    edge1b = np.zeros(n)
    edge1b[-e:] = 1.0
    inits['edge_right'] = edge1b / edge1b.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)

    # Sqrt ramp (concave)
    s = np.sqrt(xm + 0.01)
    inits['sqrt_ramp'] = s / s.sum()

    # Inverse ramp (1/(x+1))
    s = 1.0 / (xm * (n - 1) + 1.0)
    inits['inv_ramp'] = s / s.sum()

    # Piecewise linear (trapezoid, asymmetric)
    for peak_pos in [0.3, 0.5, 0.7]:
        s = np.zeros(n)
        for i in range(n):
            pos = i / max(n - 1, 1)
            if pos < peak_pos:
                s[i] = pos / peak_pos
            else:
                s[i] = (1.0 - pos) / (1.0 - peak_pos)
        s = np.maximum(s, 0.01)
        inits[f'trap_{peak_pos}'] = s / s.sum()

    return inits

    """

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

                if ps > 1e-10:
                    inits['parent'] /= ps
            elif len(parent) > n:
                p_ds = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
                ps = p_ds.sum()
