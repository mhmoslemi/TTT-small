import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def safe_softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def compute_smoothed_obj_and_grad(x, beta):
    """
    Compute the smoothed objective value and its gradient w.r.t. x.
    x: log-parameters (unconstrained)
    Returns: (smoothed_objective, gradient)
    """
    p = safe_softmax(x)
    n = len(p)
    conv = fast_convolve(p)

    # q = softmax(beta * conv)
    scaled = beta * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    q_sum = exp_scaled.sum()
    if q_sum < 1e-300:
        q = np.zeros(len(conv))
        q[np.argmax(conv)] = 1.0
    else:
        q = exp_scaled / q_sum

    # Smoothed objective
    smooth_obj = (scaled.max() + np.log(q_sum)) / beta

    # c = q · conv
    c = np.dot(q, conv)

    # r_l = sum_m p_m * q_{m+l} for l=0..n-1 (cross-correlation)
    fft_size = 1 << (3 * n - 2).bit_length()
    P = np.fft.fft(p, fft_size)
    Q = np.fft.fft(q, fft_size)
    r = np.fft.ifft(P * np.conj(Q)).real[:n]

    # Gradient: grad_l = 2 * p_l * (r_l - c)
    grad = 2.0 * p * (r - c)

    return smooth_obj, grad


def compute_actual_obj(p):
    """Compute the actual objective (max of auto-convolution)."""
    conv = fast_convolve(p)
    return conv.max()


def lbfgs_anneal(x0, time_limit, beta_schedule=None, max_iter_per_beta=300,

        remaining = time_limit - elapsed
        if remaining < 1.0:
            break

            return compute_smoothed_obj_and_grad(x_curr, beta)

        try:
            result = minimize(
                lambda xc: compute_smoothed_obj_and_grad(xc, beta)[0],
                lambda xc: compute_smoothed_obj_and_grad(xc, beta)[1],

        except Exception:

        conv = fast_convolve(p)
        if conv.max() > best_obj * 1.001:

def lp_direction(a, M, n):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 20})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum

def coordinate_descent_polish(p, time_limit, seed=0):
    """Coordinate descent: try moving mass between pairs of coordinates."""
    rng = np.random.default_rng(seed)

            break

        max_val = conv.max()

        if max_val <= best_obj * (1 - 1e-10):
            no_improve += 100
            continue

        # For the peak, find contributing pairs and try to reduce
        peak_k = int(np.argmax(conv))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)

        if i_end < i_start:
            no_improve += 10
            continue

        # Try reducing the largest contributing element
        contributions = np.array([a[i] * a[peak_k - i] for i in range(i_start, i_end + 1)])
        if len(contributions) == 0:
            no_improve += 10
            continue

        # Try several random target positions
        improved = False
        for _ in range(20):
            i_max = i_start + int(np.argmax(contributions))

                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True

                    break

                break

            i = int(rng.integers(0, n))

        # Priority: parent first, then structured inits
        priority = ['parent', 'parent_ds', 'parent_up', 'gauss_center',
                    'gauss_wide', 'raised_cos', 'bimodal', 'uniform',
                    'dirichlet_3', 'dirichlet_10', 'gauss_narrow']
        names = [nm for nm in priority if nm in names] + \
                [nm for nm in names if nm not in priority]
        names = names[:5]  # Limit per length

        time_rem = phase1_budget - (time.monotonic() - t_phase1)
        per_init = max(8.0, time_rem / max(1, len(names) * len(lengths)))

        for name in names:
            if time.monotonic() > deadline - 280:
                break
            t_alloc = min(per_init, 40.0)
            if t_alloc < 5:
                break

            x0 = inits[name]
            x_opt, obj_opt = lbfgs_with_restarts(
                x0, t_alloc, n_restarts=1,
                seed=rng.integers(0, 100000))

            p_opt = safe_softmax(x_opt)
            score = float(evaluate_sequence(p_opt.tolist()))
            if np.isfinite(score):
                candidates.append((score, n, x_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = p_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}", flush=True)

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep optimization ===", flush=True)
    top_configs = []
    seen_n = set()
    for score, n, x_opt in candidates:
        if n not in seen_n:
            top_configs.append((n, x_opt))
            seen_n.add(n)
        if len(top_configs) >= 4:
            break

    # Always include current best
    if best_sequence is not None:
        best_n = len(best_sequence)
        x_best = np.log(np.maximum(best_sequence, 1e-10))
        top_configs.insert(0, (best_n, x_best))

    remaining = deadline - time.monotonic() - 120.0
    n_configs = len(top_configs)
    per_config = max(20.0, remaining / max(1, n_configs))

    for i, (n, x0) in enumerate(top_configs):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(per_config, 250.0)
        if t_alloc < 10:
            break

        print(f"  Config {i}: n={n}, t_alloc={t_alloc:.1f}s", flush=True)

        # Extended L-BFGS with more restarts
        t_lbfgs = t_alloc * 0.5
        x_opt, obj_opt = lbfgs_with_restarts(
            x0, t_lbfgs, n_restarts=3,
            seed=rng.integers(0, 1000000))

        p_opt = safe_softmax(x_opt)
        score = float(evaluate_sequence(p_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = p_opt.copy()
            print(f"    L-BFGS: {score:.6f}", flush=True)

        # LP refinement
        t_lp = t_alloc * 0.25
        if t_lp > 5:
            p_lp, obj_lp = lp_refine(p_opt, t_lp, seed=rng.integers(0, 100000))
            score_lp = float(evaluate_sequence(p_lp.tolist()))
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = p_lp.copy()
                print(f"    LP: {score_lp:.6f}", flush=True)
            p_opt = p_lp

        # Final L-BFGS pass on LP result
        t_final = t_alloc * 0.25
        if t_final > 5:
            x_from_lp = np.log(np.maximum(p_opt, 1e-10))
            x_fin, obj_fin = lbfgs_with_restarts(
                x_from_lp, t_final, n_restarts=2,
                seed=rng.integers(0, 1000000))
            p_fin = safe_softmax(x_fin)
            score_fin = float(evaluate_sequence(p_fin.tolist()))
            if np.isfinite(score_fin) and score_fin < best_score:
                best_score = score_fin
                best_sequence = p_fin.copy()
                print(f"    Final L-BFGS: {score_fin:.6f}", flush=True)

    # Phase 3: Final polish
    print("=== Phase 3: Final polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # Coordinate descent polish
        t_cd = remaining * 0.4
        if t_cd > 5:
            p_cd, obj_cd = coordinate_descent_polish(
                best_sequence, t_cd, seed=int(rng.integers(0, 100000)))
            score_cd = float(evaluate_sequence(p_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = p_cd.copy()
                print(f"  CoordDesc: {best_score:.6f}", flush=True)

        # One more L-BFGS with very high beta
        remaining = deadline - time.monotonic() - 10.0
        t_hi = remaining * 0.3
        if t_hi > 5:
            x_hi = np.log(np.maximum(best_sequence, 1e-10))
            x_hi_opt, _ = lbfgs_anneal(
                x_hi, t_hi,
                beta_schedule=[500, 1000, 2000, 5000],
                max_iter_per_beta=100, gtol=1e-9,
                seed=int(rng.integers(0, 100000)))
            p_hi = safe_softmax(x_hi_opt)
            score_hi = float(evaluate_sequence(p_hi.tolist()))
            if np.isfinite(score_hi) and score_hi < best_score:
                best_score = score_hi
                best_sequence = p_hi.copy()
                print(f"  HighBeta: {best_score:.6f}", flush=True)

        # Final LP + coordinate descent
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            p_final, _ = lp_refine(best_sequence, remaining * 0.5, seed=999)
            score_f = float(evaluate_sequence(p_final.tolist()))
            if np.isfinite(score_f) and score_f < best_score:
                best_score = score_f
                best_sequence = p_final.copy()
                print(f"  FinalLP: {best_score:.6f}", flush=True)

            remaining2 = deadline - time.monotonic() - 2.0
            if remaining2 > 3:
                p_f2, _ = coordinate_descent_polish(
                    best_sequence, remaining2, seed=7777)
                score_f2 = float(evaluate_sequence(p_f2.tolist()))
                if np.isfinite(score_f2) and score_f2 < best_score:
                    best_score = score_f2
                    best_sequence = p_f2.copy()
                    print(f"  FinalCD: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) / best_sequence.sum() for x in best_sequence]
