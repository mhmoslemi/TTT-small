import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_gradient(a, conv, beta):
    """
    Gradient of logsumexp(beta*conv)/beta w.r.t. a.
    Returns grad of same length as a.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p.sum()

    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def exact_subgradient(a, conv, tol=1e-8):
    """
    Exact subgradient of max_k conv[k] at a.
    Active set = {k : conv[k] >= max - tol * max}.
    Returns gradient vector of length n.
    """
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.ones(n) / n
    active_mask = conv >= max_val - tol * max_val
    active_indices = np.where(active_mask)[0]

    grad = np.zeros(n)
    for k in active_indices:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        partners = a[i_start:k + 1] if k < n else a[0:n - k]
        # For each i in [i_start, i_end], the partner is a[k-i]
        # grad[i] += 2 * a[k-i]
        for i in range(i_start, i_end + 1):
            partner_idx = k - i
            if 0 <= partner_idx < n:
                grad[i] += 2.0 * a[partner_idx]

    # Normalize by number of active constraints for stability
    n_active = len(active_indices)
    if n_active > 0:
        grad /= n_active
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def multiplicative_update(a, grad, eta):
    """Mirror descent update: a_i <- a_i * exp(-eta * g_i) / Z."""
    shifted = grad - grad.max()
    weights = a * np.exp(-eta * shifted)
    s = weights.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return weights / s


def normalize_simplex(a):
    """Project onto simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_direction(a, M, n):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns b on simplex or None.
    """
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 20})
    except Exception:
        return None
    if not result.success or result.x is None:
        return None
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum


def optimize_nesterov(a0, time_limit, beta_init=30.0, beta_end=300.0,
                      lr_init=0.012, momentum=0.92, seed=0):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed objective."""

    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)

    best_a = a.copy()

    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0

        iters += 1

            v *= 0.3
            no_improve = 0

def coordinate_descent(a0, time_limit, seed=0, top_k=3):

            break

        for k in peak_indices:

            if i_start > i_end:
                continue

            # Contribution of a[i] to conv[k] is proportional to a[k-i]

            indices = np.arange(i_start, i_end + 1)

            if len(partners) == 0:

                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True

                    if improved:
                        break
                if improved:
                    break

        if not improved:

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    inits['uniform'] = np.ones(n) / n

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    rc = np.maximum(rc, 1e-8)
    inits['raised_cos'] = rc / rc.sum()

    # Gaussian center
    g = np.exp(-0.5 * ((x - mid) / (0.3 * n)) ** 2)
    inits['gauss_center'] = g / g.sum()

    # Gaussian wide
    gw = np.exp(-0.5 * ((x - mid) / (0.5 * n)) ** 2)
    inits['gauss_wide'] = gw / gw.sum()

    # Bimodal edges
    bi = np.zeros(n)
    edge = max(3, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Triangular (peak center)
    tri = 1.0 - np.abs(x - mid) / (mid + 1)
    tri = np.maximum(tri, 1e-8)
    inits['triangle'] = tri / tri.sum()

    # Chirp-inspired (magnitude of linear chirp)
    phase = np.pi * x ** 2 / n
    chirp_mag = np.abs(np.cos(phase)) + 0.1
    inits['chirp'] = chirp_mag / chirp_mag.sum()

    # Quadratic
    quad = 1.0 / (1.0 + ((x - mid) / (0.2 * n)) ** 4)
    inits['quartic'] = quad / quad.sum()

    # Random Dirichlet
    inits['rand_a'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_b'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_c'] = rng.dirichlet(np.ones(n) * 3.0)

    # Sine wave
    sine = np.abs(np.sin(np.pi * x / n)) + 0.05
    inits['sine'] = sine / sine.sum()

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_trunc'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # === Phase 2: Deep optimization ===
    print("=== Phase 2: Deep Optimization ===", flush=True)
    t_phase2_start = time.monotonic()

    # Build candidate list
    candidates = []
    if best_sequence is not None:
        candidates.append(('best', len(best_sequence), best_sequence.copy()))

    seen_configs = set()
    for score, n, name, a_opt in scan_results[:6]:
        key = (n, name)
        if key not in seen_configs:
            candidates.append((name, n, a_opt.copy()))
            seen_configs.add(key)

    # Add random restarts at best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    candidates.append(('rand1', best_n, rng.dirichlet(np.ones(best_n) * 0.5)))
    candidates.append(('rand2', best_n, rng.dirichlet(np.ones(best_n) * 1.5)))

    remaining = deadline - time.monotonic() - 100.0
    n_cands = len(candidates)
    per_cand = max(20.0, remaining / n_cands)

    for ci, (name, n, a0) in enumerate(candidates):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(per_cand, 250.0)
        if t_alloc < 10:
            break

        print(f"  [{ci}/{n_cands}] {name}, n={n}, alloc={t_alloc:.0f}s", flush=True)
        a_cur = a0.copy()

        # Method A: Nesterov smoothed (40%)
        t_a = t_alloc * 0.35
        if t_a > 5:
            a_cur, _ = optimize_nesterov(a_cur, t_a, beta_init=30, beta_end=250,
                                         lr_init=0.01, momentum=0.92,
                                         seed=rng.integers(0, 99999))
            score = float(evaluate_sequence(a_cur.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cur.copy()
                print(f"    After Nesterov: {score:.6f}", flush=True)

        # Method B: Mirror descent (25%)
        t_b = t_alloc * 0.25
        if t_b > 5:
            a_cur, _ = optimize_mirror(a_cur, t_b, eta_init=0.4, eta_end=0.01,
                                       seed=rng.integers(0, 99999))
            score = float(evaluate_sequence(a_cur.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cur.copy()
                print(f"    After Mirror: {score:.6f}", flush=True)

        # Method C: Exact subgradient (20%)
        t_c = t_alloc * 0.20
        if t_c > 5:
            a_cur, _ = optimize_subgrad(a_cur, t_c, seed=rng.integers(0, 99999))
            score = float(evaluate_sequence(a_cur.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cur.copy()
                print(f"    After Subgrad: {score:.6f}", flush=True)

        # Method D: LP direction (20%)
        t_d = t_alloc * 0.20
        if t_d > 8:
            a_cur, _ = optimize_lp(a_cur, t_d, seed=rng.integers(0, 99999))
            score = float(evaluate_sequence(a_cur.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cur.copy()
                print(f"    After LP: {score:.6f}", flush=True)

    print(f"Phase 2 best: {best_score:.6f}", flush=True)

    # === Phase 3: Intensive polishing ===
    print("=== Phase 3: Polishing ===", flush=True)
    if best_sequence is not None:
        # 3a: Coordinate descent (long)
        t_cd = min(150.0, (deadline - time.monotonic() - 50) * 0.35)
        if t_cd > 10:
            a_cd, _ = coordinate_descent(best_sequence, t_cd, seed=123, top_k=5)
            score = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cd.copy()
                print(f"  CoordDesc: {score:.6f}", flush=True)

        # 3b: High-beta Nesterov
        t_hb = min(120.0, (deadline - time.monotonic() - 40) * 0.30)
        if t_hb > 5:
            a_hb, _ = optimize_nesterov(best_sequence, t_hb, beta_init=200, beta_end=1000,
                                        lr_init=0.003, momentum=0.95, seed=456)
            score = float(evaluate_sequence(a_hb.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_hb.copy()
                print(f"  HighBeta: {score:.6f}", flush=True)

        # 3c: Mirror descent fine
        t_mf = min(80.0, (deadline - time.monotonic() - 30) * 0.25)
        if t_mf > 5:
            a_mf, _ = optimize_mirror(best_sequence, t_mf, eta_init=0.1, eta_end=0.001,
                                      seed=789)
            score = float(evaluate_sequence(a_mf.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_mf.copy()
                print(f"  MirrorFine: {score:.6f}", flush=True)

        # 3d: Final coordinate descent
        t_fd = max(5.0, (deadline - time.monotonic() - 10) * 0.5)
        if t_fd > 5:
            a_fd, _ = coordinate_descent(best_sequence, t_fd, seed=321, top_k=3)
            score = float(evaluate_sequence(a_fd.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_fd.copy()
                print(f"  FinalCD: {score:.6f}", flush=True)

        # 3e: One more round of subgradient + CD
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 20:
            a_sg, _ = optimize_subgrad(best_sequence, remaining * 0.4, seed=555)
            score = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score):
                a_cur = a_sg if score < best_score else best_sequence
                if score < best_score:
                    best_score = score
                    best_sequence = a_sg.copy()

            remaining = deadline - time.monotonic() - 5.0
            if remaining > 10:
                a_cd2, _ = coordinate_descent(a_cur, remaining, seed=777, top_k=3)
                score = float(evaluate_sequence(a_cd2.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_cd2.copy()
                    print(f"  FinalPolish: {score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
