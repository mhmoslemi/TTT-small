import numpy as np
import time

def fast_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (S * S)

def compute_gradient(a, top_k=5):
    """Compute (sub)gradient of the objective using top-k peaks."""
    n = len(a)
    S = a.sum()
    conv = fast_convolve(a)
    
    # Find top-k peaks
    top_k_actual = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
    peak_values = conv[peak_indices]
    
    # Weights (softmax)
    temp = max(peak_values.max() * 0.1, 1e-10)
    weights = np.exp((peak_values - peak_values.max()) / temp)
    weights /= weights.sum()
    
    # Compute gradient
    # For each peak at position k: grad_k = 2 * (a[k-p] for valid p)
    # Full gradient = 4n/S^3 * sum_i w_i * (S * conv_grad_k_i - peak_i * 1)
    
    grad = np.zeros(n)
    for i, k in enumerate(peak_indices):
        w = weights[i]
        if w < 1e-10:
            continue
        # conv_grad at position p: 2*a[k-p] if 0 <= k-p < n, else 0
        # (for k != 2p; for k = 2p it's 2*a[p])
        p_range = np.arange(n)
        k_minus_p = k - p_range
        valid = (k_minus_p >= 0) & (k_minus_p < n)
        conv_grad = np.zeros(n)
        conv_grad[valid] = 2.0 * a[k_minus_p[valid]]
        # Note: for k = 2p, this formula gives 2*a[p] which is correct
        # (the "double counting" is already handled because in the sum
        #  conv[k] = sum_i a[i]*a[k-i], the term a[p]*a[p] appears once
        #  when i=p, and its derivative is 2*a[p])
        
        grad += w * (S * conv_grad - peak_values[i] * 1.0)
    
    grad *= 4.0 * n / (S ** 3)
    return grad, conv.max()

def nesterov_optimize(a0, time_limit, top_k=10, momentum=0.9):
    """Optimize using Nesterov accelerated projected gradient."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
	
    # Initial step size
    lr = 1.0 / (n * best_obj)
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute gradient at y
        grad, M = compute_gradient(y, top_k=top_k)
        
        # Backtracking line search
        alpha = lr
        obj_y = 2.0 * n * M / (y.sum() ** 2)
        improved = False
        for _ in range(20):
            candidate = y - alpha * grad
            candidate = np.maximum(candidate, 0.0)
            if candidate.sum() < 1e-15:
                alpha *= 0.5
                continue
            obj_cand = compute_obj(candidate)
            if obj_cand < obj_y - 1e-14:
                v_new = (candidate - a) / (1.0 - momentum)
                a = candidate
                v = v_new
                improved = True
                if obj_cand < best_obj - 1e-12:
                    best_obj = obj_cand
                    best_a = a.copy()
                break
            alpha *= 0.5
        
        if not improved:
            # Stuck, reduce learning rate
            lr *= 0.5
            if lr < 1e-15:
                break
        
        # Decay learning rate slowly
        lr *= 0.999
        it += 1
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov it={it}, obj={best_obj:.6f}, lr={lr:.2e}, elapsed={elapsed:.1f}s")
        
        # Convergence check
        if it > 100 and np.abs(v).max() < 1e-10:
            break
    
    return best_a, best_obj

        return np.zeros(n)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_vals = conv[peak_idx]
    
    # Softmax weights with adaptive temperature
    temp = max(peak_vals.max() * 0.05, 1e-12)
    w = np.exp((peak_vals - peak_vals.max()) / temp)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(k_actual):
        if w[i] < 1e-10:
            continue
        k = int(peak_idx[i])
        # d(conv[k])/da[p] = 2*a[k-p] if 0<=k-p<n else 0
        p_range = np.arange(n)
        km = k - p_range
        valid = (km >= 0) & (km < n)
        cg = np.zeros(n)
        cg[valid] = 2.0 * a[km[valid]]
        grad += w[i] * (S * cg - peak_vals[i])
    
    grad *= 4.0 * n / (S ** 3)
    return grad


def nesterov_optimize(a0, time_limit, top_k=10, momentum=0.85, verbose=False):
    """Nesterov accelerated projected gradient with backtracking."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        y = a + momentum * v
        grad = compute_multi_peak_gradient(y, top_k=top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Backtracking line search
        improved = False
        alpha = lr
        for _ in range(15):
            cand = y - alpha * grad
            cand = np.maximum(cand, 0.0)
            Sc = cand.sum()
            if Sc < 1e-15:
                alpha *= 0.5
                continue
            cand = cand / Sc  # normalize
            obj_c = 2.0 * n * fast_convolve(cand).max()
            if obj_c < best_obj + 1e-12 and obj_c < 2.0 * n * fast_convolve(y).max():
                v = (cand - a) / (1.0 - momentum)
                a = cand
                v = np.minimum(np.maximum(v, -1.0), 1.0)  # clip momentum
                if obj_c < best_obj - 1e-13:
                    best_obj = obj_c
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                improved = True
                break
            alpha *= 0.4
        
        if not improved:
            no_improve += 1
            lr *= 0.7
        
        lr = max(lr * 0.9995, 1e-8)
        it += 1
        
        if no_improve > 500:
            # Perturb to escape
            rng = np.random.default_rng(it)
            a = np.maximum(a + rng.standard_normal(n) * 0.005, 0.0)
            a = a / a.sum()
            no_improve = 0
            momentum = min(momentum + 0.02, 0.95)
        
        if verbose and it % 300 == 0:
            print(f"      Nesterov it={it}, obj={best_obj:.6f}, lr={lr:.2e}, el={time.monotonic()-t0:.0f}s")
        
        if it > 50 and np.abs(v).max() < 1e-9:
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with exact 1D subproblem via FFT."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    sweep = 0
    gr = (np.sqrt(5) - 1) / 2.0
    
    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        any_improve = False
        
        for p in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            ap = a[p]
            if ap < 1e-12:
                continue
            
            # Try x values: 0, ap/2, ap, 2*ap, and golden section refinement
            # Use golden section on [0, 3*ap] 
            x_lo = 0.0
            x_hi = 3.0 * ap
            
            def eval_x(x):
                a[p] = x
                S = a.sum()
                if S < 1e-15:
                    return 1e10
                a_norm = a / S
                return 2.0 * n * fast_convolve(a_norm).max()
            
            # Reset a[p]
            a[p] = ap
            
            # Evaluate at a few points to bracket
            x_trial = np.array([0.0, 0.25*ap, 0.5*ap, 0.75*ap, ap, 1.5*ap, 2.0*ap, 3.0*ap])
            x_trial = np.clip(x_trial, 0, 1.0)
            
            best_x_local = ap
            best_val_local = 2.0 * n * conv.max()
            
            for xt in x_trial:
                if abs(xt - ap) < 1e-10:
                    continue
                a[p] = xt
                S = a.sum()
                if S < 1e-15:
                    a[p] = ap
                    continue
                a_norm = a / S
                obj_try = 2.0 * n * fast_convolve(a_norm).max()
                if obj_try < best_val_local:
                    best_val_local = obj_try
                    best_x_local = xt
            
            a[p] = ap  # reset
            
            if best_x_local != ap:
                # Refine with golden section around best_x_local
                center = best_x_local
                span = max(0.3 * ap, 0.01 / n)
                x_lo_r = max(0.0, center - span)
                x_hi_r = min(1.0, center + span)
                
                def g_r(x):
                    a[p] = x
                    S = a.sum()
                    if S < 1e-15:
                        a[p] = ap
                        return 1e10
                    a_norm = a / S
                    val = 2.0 * n * fast_convolve(a_norm).max()
                    a[p] = ap
                    return val
                
                c_r = x_hi_r - gr * (x_hi_r - x_lo_r)
                d_r = x_lo_r + gr * (x_hi_r - x_lo_r)
                fc = g_r(c_r)
                fd = g_r(d_r)
                
                for _ in range(15):
                    if fc < fd:
                        x_hi_r = d_r
                        d_r = c_r
                        fd = fc
                        c_r = x_hi_r - gr * (x_hi_r - x_lo_r)
                        fc = g_r(c_r)
                    else:
                        x_lo_r = c_r
                        c_r = d_r
                        fc = fd
                        d_r = x_lo_r + gr * (x_hi_r - x_lo_r)
                        fd = g_r(d_r)
                
                x_opt = (x_lo_r + x_hi_r) / 2.0
                a[p] = x_opt
                S = a.sum()
                if S > 1e-15:
                    a = a / S
                    conv = fast_convolve(a)
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj - 1e-13:
                        best_obj = cur_obj
                        best_a = a.copy()
                    any_improve = True
            else:
                a[p] = ap
        
        sweep += 1
        if verbose and sweep % 5 == 0:
            print(f"      CD sweep={sweep}, obj={best_obj:.6f}, el={time.monotonic()-t0:.0f}s")
        
        if not any_improve and sweep > 3:
            # Try small random restart
            if rng.random() < 0.3:
                idx = rng.integers(n)
                a[idx] = max(0, a[idx] + rng.standard_normal() * 0.01 / n)
                a = a / a.sum()
                conv = fast_convolve(a)
                best_obj = 2.0 * n * conv.max()
                best_a = a.copy()
            else:
                break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Linear ramp (increasing)
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Quadratic ramp
    q = np.arange(1, n + 1, dtype=np.float64) ** 2
    inits['quad_ramp'] = q / q.sum()
    
    # Square root ramp
    s = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    inits['sqrt_ramp'] = s / s.sum()
    
    # Edge concentrated (high at both ends)
    x = np.arange(n, dtype=np.float64)
    edge = np.minimum(x, n - 1 - x)
    e = (n - edge) ** 2
    inits['edge'] = e / e.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Asymmetric: high on right
    asym = np.exp(x / (0.3 * n))
    inits['asym_right'] = asym / asym.sum()
    
    # Asymmetric: high on left
    asym2 = np.exp(-x / (0.3 * n))
    inits['asym_left'] = asym2 / asym2.sum()
    
    # Sine wave
    sine = np.sin(np.pi * (x + 0.5) / n) ** 0.5
    sine = np.maximum(sine, 0.01)
    inits['sine'] = sine / sine.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Bimodal
    bm = np.exp(-((x - 0.25*n)/0.1*n)**2) + np.exp(-((x - 0.75*n)/0.1*n)**2)
    inits['bimodal'] = bm / bm.sum()
    
    # Cosine window
    cosw = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cosw = np.maximum(cosw, 0.01)
    inits['cosine'] = cosw / cosw.sum()

            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
