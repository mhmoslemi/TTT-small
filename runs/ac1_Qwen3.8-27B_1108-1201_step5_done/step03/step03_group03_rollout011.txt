    a = a / s

    best_a = a.copy()

    eta = eta_init
    tau = tau_init

        # Compute smooth gradient
        b = conv
        # Softmax weights over peaks
        log_move = b - b.max()
        weights = np.exp(log_move / tau)

            for j in range(j_start, j_end + 1):

        tau *= 0.999

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative weights (mirror descent)
with adaptive peak weighting, LP-based refinement, and coordinate descent polish.
Key improvements over previous approach:
- Multiplicative weights update (mirror descent on simplex) as primary method
- Adaptive temperature for peak weighting
- Log-space momentum for faster convergence
- LP refinement for final push
- Multiple restarts with different initializations

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, conv, tau, top_k=50):
    """Compute smoothed gradient of max(conv) w.r.t. a."""
    n = len(a)
    # Get top k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Softmax weights
    peak_vals = conv[peaks]
    shifted = peak_vals - peak_vals.max()
    weights = np.exp(shifted / tau)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(len(peaks))
        weights[np.argmax(peak_vals)] = 1.0
    else:
        weights /= w_sum
    
    # Compute gradient: grad_j = 2 * sum_k w_k * a_{k-j}
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        wi = weights[idx]
        if wi < 1e-12:
            continue
        wi2 = 2.0 * wi
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] += wi2 * a[k - j]
    
    return grad


def mirror_descent_optimize(a0, time_limit, eta_init=0.05, tau_init=0.005,
                           top_k=50, momentum=0.7, anneal_rate=0.9995):
    """Mirror descent (multiplicative weights) with log-space momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    log_a = np.log(a + 1e-30)
    v = np.zeros(n)  # momentum in log-space
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    eta = eta_init
    tau = tau_init
    
    while time.monotonic() - t0 < time_limit:
        grad = compute_gradient(a, conv, tau, top_k)
        
        # Momentum update in log-space
        v = momentum * v - eta * grad
        log_a_new = log_a + v
        
        # Convert back to simplex
        log_a_new -= log_a_new.max()  # stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        a = a_new
        log_a = np.log(a + 1e-30)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Annealing
        eta *= anneal_rate
        tau *= anneal_rate
        
        # Restart if stuck
        if no_improve > 100:
            v *= 0.3
            eta = max(eta, 0.005)
            tau = max(tau, 0.001)
            no_improve = 0
        
        if eta < 1e-7:
            break
    
    return best_a, best_obj


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(f_init, time_limit, pairs, blend_init=0.3, blend_min=0.005):
    """LP-based directional optimization."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    uniform = np.ones(n) / n
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_direction(a, M, pairs, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        max_g = g.max()
        if max_g > 0.5:
            g = 0.7 * g + 0.3 * uniform
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= 10:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break
    
    return best_a, best_obj


def coordinate_descent(a_in, max_rounds=50):
    """Coordinate descent: try swapping mass between pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Try small moves from high-mass to low-mass positions
        sorted_idx = np.argsort(a)[::-1]
        for i in sorted_idx[:20]:
            if a[i] < 1e-6:
                break
            delta = a[i] * 0.03
            if delta < 1e-9:
                continue
            a_test = a.copy()
            a_test[i] -= delta
            # Distribute to a few random low-mass positions
            low_idx = sorted_idx[-10:]
            for j in low_idx[:5]:
                a_test[j] += delta / 5.0
            a_test = np.maximum(a_test, 0.0)
            st = a_test.sum()
            if st < 1e-15:
                continue
            a_test /= st
            conv_test = fast_convolve(a_test)
            obj_test = 2.0 * n * conv_test.max()
            if obj_test < best_obj - 1e-13:
                a = a_test
                best_obj = obj_test
                conv = conv_test
                improved = True
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""

    inits['triangle'] = tri / tri.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (multiplicative weights)
with adaptive peak weighting, LP-based directional refinement, and efficient coordinate
descent polish. Key innovations:
- Log-space momentum for faster mirror descent convergence
- Adaptive flat-top peak weighting (avoids softmax temperature sensitivity)
- Interleaved LP/mirror-descent iterations for direction quality + speed
- Efficient coordinate descent with limited pair trials
- Multi-length search with parent warm-start

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, conv, mode='flattop', epsilon=0.1, top_k=60):
    """
    Compute smoothed gradient of max(conv) w.r.t. a.
    mode='flattop': equal weight to all peaks within epsilon*max of the max
    mode='softmax': exponential weighting
    """
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    peak_vals = conv[peaks]
    pmax = peak_vals.max()
    if pmax < 1e-15:
        return np.zeros(n)

    if mode == 'flattop':
        threshold = pmax * (1.0 - epsilon)
        mask = peak_vals >= threshold
        if mask.sum() == 0:
            mask[np.argmax(peak_vals)] = True
        weights = np.where(mask, 1.0 / mask.sum(), 0.0)
    else:
        tau = pmax * 0.05
        shifted = peak_vals - pmax
        weights = np.exp(shifted / tau)
        ws = weights.sum()
        if ws < 1e-30:
            weights = np.zeros(len(peaks))
            weights[np.argmax(peak_vals)] = 1.0
        else:
            weights /= ws

    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        wi = weights[idx]
        if wi < 1e-12:
            continue
        wi2 = 2.0 * wi
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] += wi2 * a[k - j]

    return grad


def mirror_descent_solve(a0, time_limit, eta_init=0.06, mode='flattop',
                         epsilon=0.15, top_k=50, momentum=0.75,
                         anneal_eta=0.999, anneal_tighten=0.9995):
    """
    Mirror descent (multiplicative weights) with log-space momentum.
    Optimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    log_a = np.log(a + 1e-30)
    v = np.zeros(n)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0
    it_count = 0

    t0 = time.monotonic()
    eta = eta_init
    eps = epsilon

    while time.monotonic() - t0 < time_limit:
        grad = compute_smooth_gradient(a, conv, mode=mode, epsilon=eps, top_k=top_k)

        # Log-space momentum update
        v = momentum * v - eta * grad
        log_a_new = log_a + v

        # Convert to simplex
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        a = a_new
        log_a = np.log(a + 1e-30)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        it_count += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Annealing
        eta *= anneal_eta
        eps = max(0.01, eps * anneal_tighten)

        # Recovery from stagnation
        if no_improve > 80:
            v *= 0.2
            eta = max(eta, 0.01)
            eps = max(eps, 0.05)
            no_improve = 0
            # Small random perturbation
            if it_count > 200:
                rng_local = np.random.default_rng(hash(int(a.sum() * 1e10)) % (2**31))
                noise = rng_local.standard_normal(n) * 0.003
                a = project_simplex(np.exp(log_a) + noise)
                log_a = np.log(a + 1e-30)
                conv = fast_convolve(a)

        if eta < 1e-7:
            break

    return best_a, best_obj


def build_lp_matrix(n):
    """Build the LP constraint matrix for auto-convolution bounds."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = 1.0  # Will be scaled by f[idx] each call
    return A_ub, (2 * n - 1, n)


def solve_lp_for_direction(f, M, n):
    """Solve LP: max 1^T g s.t. g>=0, (f*g)_k <= M for all k."""
    # Build constraint: A_ub[k,j] = f[k-j] if valid, 0 otherwise
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_direction_step(f, v_log, n, blend, uniform):
    """One LP-based directional step. Returns updated a or None."""
    a = np.exp(v_log)
    a = a / a.sum()
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return None, None, None

    g = solve_lp_for_direction(a, M, n)
    if g is None:
        return None, None, None

    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, None, None
    g = g / g_sum

    # Blend with uniform if too sparse
    if g.max() > 0.4:
        g = 0.6 * g + 0.4 * uniform

    a_new = (1.0 - blend) * a + blend * g
    a_new = project_simplex(a_new)
    return a_new, g, M


def lp_refine(a0, time_limit, n, blend_init=0.25, blend_halve_every=12):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    uniform = np.ones(n) / n

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        a_new, g, M = lp_direction_step(a, np.log(a + 1e-30), n, blend, uniform)
        if a_new is None:
            break

        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(0.003, blend * 0.5)
            stall = 0
            if blend < 0.005:
                break

    return best_a, best_obj


def coord_descent_fast(a_in, max_rounds=30, trials_per_round=200, rng=None):
    """Fast coordinate descent with limited trials per round."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    sorted_idx = np.argsort(a)[::-1]

    for rnd in range(max_rounds):
        improved = False
        for _ in range(trials_per_round):
            # Pick a source (high mass) and target (low mass)
            i = sorted_idx[rng.integers(0, min(30, n))]
            j = sorted_idx[rng.integers(max(0, n - 30), n)]
            if i == j or a[i] < 1e-7:
                continue
            delta = a[i] * rng.uniform(0.02, 0.08)
            if delta < 1e-10:
                continue

            a_test = a.copy()
            a_test[i] -= delta
            a_test[j] += delta
            if a_test[i] < 0:
                continue
            st = a_test.sum()
            a_test /= st
            conv_test = fast_convolve(a_test)
            obj_test = 2.0 * n * conv_test.max()
            if obj_test < best_obj - 1e-14:
                a = a_test
                best_obj = obj_test
                conv = conv_test
                improved = True

        if not improved:
            # Try structured moves: move mass from center to edges or vice versa
            any_improve = False
            mid = n // 2
            for direction in [1, -1]:
                for frac in [0.01, 0.02, 0.03]:
                    src = mid if direction > 0 else 0
                    dst = 0 if direction > 0 else mid
                    if a[src] < 1e-7:
                        continue
                    delta = a[src] * frac
                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[dst] += delta
                    a_test = np.maximum(a_test, 0)
                    a_test /= a_test.sum()
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-14:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        any_improve = True
            if not any_improve:
                break

    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    g2 = (np.exp(-0.5 * ((x - 0.15 * n) / (0.08 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.85 * n) / (0.08 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Plateau in middle
    plat = np.zeros(n)
    m = n // 3
    plat[n//4:3*n//4] = 1.0
    inits['plateau'] = plat / plat.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(p)), p)
                p_rs = p_res.sum()
                if p_rs > 1e-10:
                    inits['parent_res'] = p_res / p_rs
    return inits


def hybrid_optimize(a0, time_limit, n, rng):

    Interleave mirror descent and LP steps for faster convergence.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)

    a = a / s
    uniform = np.ones(n) / n

    log_a = np.log(a + 1e-30)
    v = np.zeros(n)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0

    eta = 0.05
    momentum = 0.75
    eps = 0.15
    lp_blend = 0.2
    md_count = 0
    lp_count = 0

    while time.monotonic() - t0 < time_limit:
        # Every 50 MD iterations, do one LP step
        if md_count > 0 and md_count % 50 == 0 and lp_blend > 0.005:
            a_new, g, M = lp_direction_step(a, log_a, n, lp_blend, uniform)
            if a_new is not None:
                a = a_new

                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj

                else:
                    no_improve += 1
                lp_blend *= 0.7
                lp_count += 1
            else:
                lp_blend *= 0.5
            md_count = 0
        else:
            # Mirror descent step
            grad = compute_smooth_gradient(a, conv, mode='flattop',
                                           epsilon=eps, top_k=50)
            v = momentum * v - eta * grad
            log_a_new = log_a + v
            log_a_new -= log_a_new.max()
            a = np.exp(log_a_new)

                                      trials_per_round=150, rng=rng)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")

        # Final MD with very small steps
        remaining = deadline - time.monotonic() - 20.0
        t_final = min(60.0, remaining * 0.5)
        if t_final > 5:
            a_fin, _ = mirror_descent_solve(best_sequence, t_final,
                                            eta_init=0.005, mode='flattop',
                                            epsilon=0.03, top_k=80,
                                            momentum=0.9, anneal_eta=0.9998)
            score_fin = float(evaluate_sequence(a_fin.tolist()))
            if np.isfinite(score_fin) and score_fin < best_score:
                best_score = score_fin
                best_sequence = a_fin.copy()
                print(f"  Final MD: {score_fin:.6f}")

        # Perturbation restarts
        n_b = len(best_sequence)
        for pert_size in [0.003, 0.008, 0.015]:
            if time.monotonic() > deadline - 15:
                break
            for _ in range(3):
                if time.monotonic() > deadline - 10:
                    break
                noise = rng.standard_normal(n_b) * pert_size
                a_p = project_simplex(best_sequence + noise)
                a_p2, _ = hybrid_optimize(a_p, 8.0, n_b, rng)
                score_p = float(evaluate_sequence(a_p2.tolist()))
                if np.isfinite(score_p) and score_p < best_score:
                    best_score = score_p
                    best_sequence = a_p2.copy()
                    print(f"  Perturb({pert_size}): {score_p:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
