    """Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M for all k."""

    return None

        j = rng.integers(0, n)
        if i == j:
            continue

        # Adaptive transfer size based on temperature
        delta = min(a[i], a[j] + 1e-10) * t * 2 * np.sqrt(n)
        if delta < 1e-10:
            t *= cool_rate
            continue

        a_test = a.copy()
        a_test[i] -= delta
        a_test[j] += delta
        a_test = np.maximum(a_test, 0.0)
        st = a_test.sum()
        if st < 1e-15:
            continue
        a_test /= st

        conv_test = fast_convolve(a_test)
        obj_test = 2.0 * n * conv_test.max()

        delta_e = obj_test - cur_obj
        if delta_e < 0 or rng.random() < np.exp(-delta_e / max(t, 1e-15)):
            a = a_test
            cur_obj = obj_test
            conv = conv_test
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()

        t *= cool_rate

    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=50, time_limit=120):
    """Coordinate descent: try moving mass between pairs of coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    t0 = time.monotonic()
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Try moving mass from high-peak-contributing positions to others
        peaks = np.argpartition(conv, -10)[-10:]

        for i in range(0, n, max(1, n // 100)):
            if a[i] < 1e-8:
                continue
            for j in range(0, n, max(1, n // 50)):
                if j == i:
                    continue
                delta = a[i] * 0.03
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[i] = max(0, a_test[i] - delta)
                a_test[j] += delta
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True

        if not improved:
            break

    return a


def mirror_descent_step(a, sg, eta):
    """Multiplicative weights update (mirror descent with entropy)."""
    log_update = -eta * sg / max(sg.max(), 1e-15)
    # Clip to avoid overflow
    log_update = np.clip(log_update, -50, 50)
    weights = np.exp(log_update)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    # Anti-triangular (U-shape)
    u = np.maximum(np.abs(x - 0.5 * (n - 1)) / (0.5 * n), 0.05)
    inits['u_shape'] = u / u.sum()

    # Periodic pattern
    per = np.ones(n)
    period = max(2, n // 8)
    per[::period] = 2.0
    inits['periodic'] = per / per.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(parent)), p)
                p_interp = np.maximum(p_interp, 0)
                ps2 = p_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = p_interp / ps2

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining accelerated subgradient, LP refinement,
    simulated annealing, and coordinate descent to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if np.isfinite(par_score):
                best_score = par_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Multi-start subgradient optimization
    print("=== Phase 1: Multi-start Subgradient ===")
    phase1_end = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.3)

    # Try multiple lengths
    lengths = [256, 320, 384, 512, 640, 768]
    # If parent exists, include its length
    if parent is not None:
        pl = len(parent)
        if pl not in lengths:
            lengths.insert(0, pl)

    # Limit to reasonable lengths for time
    lengths = [l for l in lengths if l <= 1024]

    results = []
    n_starts = len(lengths) * 3  # up to 3 inits per length
    t_per_start = max(15.0, 400.0 / n_starts)

    init_names_pool = ['parent', 'parent_interp', 'uniform', 'bimodal',
                       'u_shape', 'triangle', 'random', 'edge', 'gauss',
                       'ramp', 'periodic']

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_inits(n, parent, rng)
        prio = [nm for nm in init_names_pool if nm in inits]
        # Take top 3 for this length
        for name in prio[:3]:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per_start, phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a0 = inits[name]
            # Subgradient optimization
            a_opt, obj_opt = subgradient_optimize(
                a0, t_alloc, lr_init=0.02, momentum=0.92,
                top_k=min(50, n // 5), restart_stall=80, restart_noise=0.008)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1 n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"  Phase 1 best: {best_score:.6f}")

    # Phase 2: LP refinement on top candidates
    print("=== Phase 2: LP Refinement ===")
    phase2_end = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.3)

    top_list = []
    seen = set()
    for score, n, name, a_opt in results[:6]:
        key = (n,)
        if key not in seen:
            top_list.append((n, a_opt))
            seen.add(key)
        if len(top_list) >= 4:
            break
    if best_sequence is not None:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = phase2_end - time.monotonic()
    t_per = max(20.0, remaining / max(1, len(top_list)))

    for n, a0 in top_list:
        if time.monotonic() > phase2_end - 10:
            break
        t_alloc = min(t_per, phase2_end - time.monotonic() - 8)
        if t_alloc < 10:
            break

        # LP refinement
        a_opt, obj_opt = lp_optimize(a0, t_alloc * 0.6,
                                     blend_init=0.2, blend_min=0.003,
                                     blend_halve_every=12)
        # Follow with subgradient
        a_opt2, obj2 = subgradient_optimize(
            a_opt, t_alloc * 0.4, lr_init=0.005, momentum=0.93,
            top_k=min(60, n // 4), restart_stall=150, restart_noise=0.003)
        if obj2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj2

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P2 n={n}: {score:.6f}")

    # Phase 3: Simulated annealing + deep subgradient on best
    print("=== Phase 3: SA + Deep Optimization ===")
    if best_sequence is not None:
        phase3_end = time.monotonic() + min(400.0, (deadline - time.monotonic()) * 0.25)
        n_b = len(best_sequence)

        # SA on best
        t_sa = min(80.0, phase3_end - time.monotonic() - 50)
        if t_sa > 10:
            a_sa, obj_sa = simulated_annealing(
                best_sequence, t_sa, t_init=0.015, t_min=0.00005,
                cool_rate=0.9997, top_k=30)
            score_sa = float(evaluate_sequence(a_sa.tolist()))
            if np.isfinite(score_sa) and score_sa < best_score:
                best_score = score_sa
                best_sequence = a_sa.copy()
                print(f"  SA: {score_sa:.6f}")

        # Deep subgradient with very small steps
        remaining = phase3_end - time.monotonic() - 20
        if remaining > 20:
            a_deep, obj_deep = subgradient_optimize(
                best_sequence, remaining, lr_init=0.003, momentum=0.94,
                top_k=min(80, n_b // 3), restart_stall=200, restart_noise=0.002)
            score_deep = float(evaluate_sequence(a_deep.tolist()))
            if np.isfinite(score_deep) and score_deep < best_score:
                best_score = score_deep
                best_sequence = a_deep.copy()
                print(f"  Deep SG: {score_deep:.6f}")

    # Phase 4: Final polish with coordinate descent + micro-LP
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50
        n_b = len(best_sequence)

        # Coordinate descent
        t_cd = min(80.0, remaining * 0.3)
        if t_cd > 5:
            a_cd = coordinate_descent_polish(best_sequence, max_rounds=40,
                                             time_limit=t_cd)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")

        # Final micro-optimization cycles
        remaining = deadline - time.monotonic() - 30
        for cycle in range(5):
            if time.monotonic() > deadline - 25:
                break
            t_cyc = min(remaining / (5 - cycle), 40.0)
            if t_cyc < 8:
                break

            # Very fine subgradient
            a_fine, obj_fine = subgradient_optimize(
                best_sequence, t_cyc * 0.5, lr_init=0.001, momentum=0.95,
                top_k=min(100, n_b // 2), restart_stall=300, restart_noise=0.001)
            # Quick LP
            a_lp, obj_lp = lp_optimize(
                a_fine, t_cyc * 0.4, blend_init=0.05, blend_min=0.001,
                blend_halve_every=8)
            # Subgradient again
            a_final, obj_final = subgradient_optimize(
                a_lp, t_cyc * 0.1, lr_init=0.001, momentum=0.95,
                top_k=min(100, n_b // 2), restart_stall=100, restart_noise=0.001)

            for candidate in [a_fine, a_lp, a_final]:
                sc = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_sequence = candidate.copy()
                    print(f"  Cycle {cycle}: {sc:.6f}")

            remaining = deadline - time.monotonic() - 25

        # Random perturbations at the end
        if time.monotonic() < deadline - 15:
            for ps in [0.002, 0.005, 0.001]:
                if time.monotonic() > deadline - 12:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 10:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_p2, _ = subgradient_optimize(
                        a_p, min(10.0, deadline - time.monotonic() - 8),
                        lr_init=0.002, momentum=0.93,
                        top_k=min(50, n_b // 4), restart_stall=50, restart_noise=0.001)
                    sc = float(evaluate_sequence(a_p2.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_sequence = a_p2.copy()
                        print(f"  Perturb({ps}): {sc:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
