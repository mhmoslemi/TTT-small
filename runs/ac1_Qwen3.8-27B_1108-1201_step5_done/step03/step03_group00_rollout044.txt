"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization strategy:
1. Soft-max smoothed L-BFGS optimization on the simplex (via softmax parameterization)
2. LP-based Frank-Wolfe method with adaptive constraint relaxation
3. Coordinate descent refinement
4. Subgradient polishing with adaptive temperature
Key innovation: smooth the non-differentiable max with a temperature-controlled
log-sum-exp, then use quasi-Newton methods for fast local convergence.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_parametrize(x):
    """Convert x to f on the simplex via softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_conv_gradient(f, weights):
    """Compute gradient of weighted sum of convolution entries w.r.t. f."""
    n = len(f)
    conv_len = 2 * n - 1
    sg = np.zeros(n)
    for i in range(conv_len):
        w = weights[i]
        if w < 1e-15:
            continue
        k = i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w * f[k - idx]
    return sg


def smooth_objective_and_grad(x, beta):
    """Compute smooth max(conv(f,f)) and its gradient w.r.t. x via softmax parametrization."""
    f = softmax_parametrize(x)
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv * beta
    scaled_max = scaled.max()
    shifted = scaled - scaled_max
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    smooth_val = (scaled_max + log_sum) / beta
    w = exp_shifted / exp_shifted.sum()
    sg = compute_conv_gradient(f, w)
    dot_sgf = np.dot(sg, f)
    grad_x = f * (sg - dot_sgf)
    return smooth_val, grad_x


def lbfgs_smooth_optimize(f0, time_limit, beta_start=50.0, beta_end=5000.0, n_betas=6):
    """Optimize using L-BFGS on soft-max smoothed objective with increasing beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    # Convert f to x in the softmax parameterization: x_i = log(f_i) (approximately)
    # Use log(f_i + eps) to avoid -inf
    eps = 1e-15
    x0 = np.log(f + eps) - np.log(f + eps).max()

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break

        time_per_beta = (time_limit - (time.monotonic() - t0)) / (n_betas - bi)

        def obj_grad(x):
            val, grad = smooth_objective_and_grad(x, beta)
            return val, grad

        x_current = x0.copy()
        max_iters = max(50, int(time_per_beta / 0.5))

        for trial in range(3):
            if time.monotonic() - t0 > time_limit:
                break

            res = minimize(obj_grad, x_current, method='L-BFGS-B',
                           options={'maxiter': max_iters, 'maxfun': max_iters * 2,
                                    'gtol': 1e-8, 'iprint': 0})

            x0 = res.x
            f_trial = softmax_parametrize(x0)
            obj = fast_convolve(f_trial).max()

            if obj < best_obj:
                best_obj = obj
                best_f = f_trial.copy()

        # Update x0 for next beta
        x0 = np.log(best_f + eps) - np.log(best_f + eps).max()

        if time.monotonic() - t0 > time_limit:
            break

    return best_f, best_obj


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 7 + 13)
            scale = 0.003 + 0.005 * (no_improve // 100)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.93, top_k=None, temp=0.00005):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if top_k is None:
        top_k = max(10, n // 10)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.004

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, n_sweeps=20):
    """Coordinate descent: for each coordinate, optimize it while keeping others fixed."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for k in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Try changing f[k] while keeping sum(f)=1
            # We'll adjust by moving mass between f[k] and the rest proportionally
            # Simple approach: try a small grid of values for f[k]
            fk_current = f[k]
            # Try reducing and increasing f[k]
            deltas = [0.001, -0.001, 0.01, -0.01, 0.05, -0.05, 0.1, -0.1]
            best_fk = fk_current
            best_val = fast_convolve(f).max()

            # Compute current conv once
            conv_cur = fast_convolve(f)
            best_val = conv_cur.max()

            for d in deltas:
                fk_new = fk_current + d
                if fk_new < 0 or fk_new > 1.0:
                    continue
                # Redistribute: reduce all other elements proportionally
                # f_other_new = f_other * (1 - fk_new) / (1 - fk_current)
                if abs(1.0 - fk_current) < 1e-15:
                    continue
                scale = (1.0 - fk_new) / (1.0 - fk_current)
                f_trial = f.copy()
                f_trial[k] = fk_new
                mask = np.ones(n, dtype=bool)
                mask[k] = False
                f_trial[mask] = f[mask] * scale
                if f_trial.sum() < 1e-10:
                    continue
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_fk = fk_new
                    f = f_trial.copy()
                    improved = True

            if not improved:
                # Reset f to before this coordinate if no improvement
                # (we already updated f in the loop, so we need to track)
                pass

        # Check overall improvement
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                # Try smaller steps
                f = best_f.copy()
                if no_improve > 20:
                    break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two-peak
    two_pk = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
             np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(two_pk / two_pk.sum())

    if parent is not None:
        if len(parent) != n:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        else:
            px = parent.copy()
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick exploration of different lengths
    Phase 2: L-BFGS smooth optimization (key method)
    Phase 3: LP-based Frank-Wolfe
    Phase 4: Final polishing (SG + coordinate descent)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={float(evaluate_sequence(parent.tolist())):.8f}")
    except Exception:
        pass

    # === Phase 1: Broad exploration (fast) ===
    print("=== Phase 1: Broad Exploration ===")
    t_phase1_end = time.monotonic() + 120

    candidates = []  # list of (f_normalized, n)

    if parent is not None:
        candidates.append((parent.copy(), len(parent)))
        try_seq(parent, "parent")

    # Try a few other lengths
    test_lengths = []
    if parent is not None:
        np_ = len(parent)
        test_lengths = [np_ - 64, np_ + 64, 256, 512]
        test_lengths = [l for l in test_lengths if l >= 64 and l != np_]
    else:
        test_lengths = [128, 256, 384, 512]

    for n in test_lengths[:3]:
        if time.monotonic() > t_phase1_end:
            break
        inits = generate_inits(n, parent, rng)
        for i, f_init in enumerate(inits[:2]):
            if time.monotonic() > t_phase1_end:
                break
            score = try_seq(f_init, f"exp_n{n}_i{i}")
            if score < best_score + 0.05:
                candidates.append((f_init.copy(), n))

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: L-BFGS smooth optimization ===
    print("\n=== Phase 2: L-BFGS Smooth Optimization ===")
    t_phase2_end = time.monotonic() + 700

    # Sort candidates by score
    for c in candidates:
        if time.monotonic() > t_phase2_end - 60:
            break

        f_cand, n = c
        time_alloc = min(200, t_phase2_end - time.monotonic() - 30)
        if time_alloc < 30:
            break

        print(f"  L-BFGS on candidate n={n}, t={time_alloc:.0f}s")
        t_start = time.monotonic()
        f_opt, obj_opt = lbfgs_smooth_optimize(f_cand, time_limit=time_alloc,
                                               beta_start=30.0, beta_end=8000.0, n_betas=5)
        elapsed = time.monotonic() - t_start
        score = try_seq(f_opt, f"lbfgs_n{n}_{elapsed:.0f}s")

    # Also try L-BFGS on the current best with more time
    if best_seq is not None and time.monotonic() < t_phase2_end - 50:
        time_alloc = t_phase2_end - time.monotonic() - 20
        print(f"  L-BFGS refinement on best, t={time_alloc:.0f}s")
        f_opt, _ = lbfgs_smooth_optimize(best_seq, time_limit=time_alloc,
                                         beta_start=200.0, beta_end=20000.0, n_betas=4)
        try_seq(f_opt, "lbfgs_refine")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: LP-based Frank-Wolfe ===
    print("\n=== Phase 3: LP-based Frank-Wolfe ===")
    t_phase3_end = time.monotonic() + 500

    if best_seq is not None:
        for p in range(4):
            rem = t_phase3_end - time.monotonic() - 30
            if rem < 25:
                break
            time_alloc = min(rem / max(1, 4 - p), 180)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(len(best_seq)) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.8 * best_seq + 0.2 * best_seq[::-1]
                f_init = project_simplex(f_init)
            else:
                noise = rng.standard_normal(len(best_seq)) * 0.01
                f_init = project_simplex(best_seq + noise)

            print(f"  FW-LP perturbation {p}, t={time_alloc:.0f}s")
            f_opt, _ = fw_lp_optimize(f_init, time_limit=time_alloc)
            try_seq(f_opt, f"fwlp_p{p}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Final polishing ===
    print("\n=== Phase 4: Final Polishing ===")
    t_phase4_end = deadline

    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent
        rem = t_phase4_end - time.monotonic() - 60
        if rem > 30:
            print(f"  Coordinate descent, t={min(rem, 120):.0f}s")
            f_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 120))
            try_seq(f_opt, "coord")

        # Subgradient polish
        rem = t_phase4_end - time.monotonic() - 30
        if rem > 20:
            print(f"  SG polish, t={min(rem, 80):.0f}s")
            f_opt, _ = sg_polish(best_seq, time_limit=min(rem, 80),
                                 lr=0.002, mom=0.95, top_k=max(20, n_best // 8), temp=0.00003)
            try_seq(f_opt, "sg")

        # Final L-BFGS with very high beta
        rem = t_phase4_end - time.monotonic() - 15
        if rem > 15:
            print(f"  Final L-BFGS, t={min(rem, 60):.0f}s")
            f_opt, _ = lbfgs_smooth_optimize(best_seq, time_limit=min(rem, 60),
                                             beta_start=5000.0, beta_end=50000.0, n_betas=3)
            try_seq(f_opt, "lbfgs_final")

        # Tiny perturbations
        rem = t_phase4_end - time.monotonic() - 5
        for _ in range(5):
            if time.monotonic() > t_phase4_end - 3:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            f_opt, _ = sg_polish(f_pert, time_limit=min(10, t_phase4_end - time.monotonic() - 2),
                                 lr=0.001, mom=0.96, top_k=max(10, n_best // 4), temp=0.00001)
            try_seq(f_opt, "tiny_pert")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
