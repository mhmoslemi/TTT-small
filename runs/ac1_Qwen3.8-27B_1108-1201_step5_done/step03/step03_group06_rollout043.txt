def adam_subgradient_optimize(a0, time_limit, beta_init=100, lr_init=0.01, seed=42):
    """Optimize using Adam-style subgradient descent on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr_init
    beta = beta_init
    
    t0 = time.monotonic()
    no_improve = 0
    patience = 200
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Compute smooth gradient
        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        grad = compute_grad_fft(w, a, n)
        
        # Project gradient: remove mean to stay on simplex
        grad_d = grad - grad.mean()
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_d
        v = beta2 * v + (1 - beta2) * grad_d ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == patience:
                lr *= 0.5
                m *= 0.5
                v *= 0.5
                no_improve = 0
            elif no_improve > patience * 3:
                # Restart with perturbation
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init
                beta = beta_init
                no_improve = 0
        
        # Gradually increase beta for sharper max approximation
        beta = min(beta * 1.002, beta_init * 10)
        
        # Periodic print
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam iter {t}: obj={best_obj:.6f}, lr={lr:.5f}, beta={beta:.0f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

"""
Subgradient optimization with adaptive active set for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Adam optimizer, softmax-smoothed max, coarse-to-fine strategy,
and multi-length exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(w, a, n):
    """Compute gradient of weighted sum of convolution values.
    grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices.
    """
    fft_size = 1 << (2 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    b = a[::-1]
    B = np.fft.rfft(b, fft_size)
    
    # term1 = (w * b)[n-1 : n-1+n]
    term1_full = np.fft.irfft(W * B, fft_size)
    term1 = term1_full[n - 1:n - 1 + n]
    
    # term2 = (a * w)[0 : n]
    term2_full = np.fft.irfft(A * W, fft_size)
    term2 = term2_full[:n]
    
    return term1 + term2


def smooth_max_gradient(a, beta):
    """Compute smooth approximation gradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    grad = compute_grad_fft(w, a, n)
    return grad, conv, w


def adam_optimize(a0, time_limit, beta_init=100.0, lr_init=0.005,
                  seed=42, verbose=False):
    """Adam subgradient optimization on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr_init
    beta = beta_init
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        grad, conv, w = smooth_max_gradient(a, beta)
        
        # Subtract mean to project onto simplex tangent
        grad_d = grad - grad.mean()
        
        t += 1
        m = b1 * m + (1 - b1) * grad_d
        v = b2 * v + (1 - b2) * grad_d * grad_d
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a - step)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.7
            if no_improve > 500:
                # Perturb and restart
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init
                beta = beta_init * 0.5
                no_improve = 0
        
        # Slowly increase beta
        beta = min(beta * 1.001, beta_init * 20)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam iter {t}: obj={best_obj:.8f}, lr={lr:.6f}, beta={beta:.1f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def generate_init(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Ramps
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits.append(('ramp', r))
    inits.append(('ramp_rev', r[::-1].copy()))
    
    # Gaussian at center
    g = np.exp(-0.5 * ((x - 0.5 * (n-1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits.append(('gauss', g))
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits.append(('bimodal', bi))
    
    # Cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits.append(('cosine', cos_w))
    
    # Random
    inits.append(('rand', rng.dirichlet(np.ones(n) * 0.5)))
    inits.append(('rand_sparse', rng.dirichlet(np.ones(n) * 0.1)))
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            p = np.maximum(parent[:n], 0)
            s = p.sum()
            if s > 1e-10:
                inits.append(('parent_crop', p / s))
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 1e-10:
                inits.append(('parent_interp', p / s))
        if len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 1e-10:
                inits.append(('parent_ds', p / s))
    
    return inits

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def obj_normalized(a):
    """Objective assuming sum(a)=1: 2n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_lp_matrix(a):
    """Build A_ub for LP: (a*b)[k] <= M. Row k has a[k-j] at column j."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

def line_search(a, b, n, max_evals=40):
    """Golden section line search for best t: a_new = (1-t)*a + t*b."""
    best_obj = obj_normalized(a)
    best_t = 0.0
    best_a = a.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0

    for _ in range(max_evals):
        if hi - lo < 1e-7:
            break
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)

        a1 = (1.0 - t1) * a + t1 * b
        a1 = np.maximum(a1, 0.0)
        s1 = a1.sum()
        if s1 > 1e-15:
            a1 = a1 / s1
        obj1 = obj_normalized(a1)

        a2 = (1.0 - t2) * a + t2 * b
        a2 = np.maximum(a2, 0.0)
        s2 = a2.sum()
        if s2 > 1e-15:
            a2 = a2 / s2
        obj2 = obj_normalized(a2)

        if obj1 < obj2:
            hi = t2
        else:
            lo = t1

    t_opt = (lo + hi) / 2.0
    a_opt = (1.0 - t_opt) * a + t_opt * b
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s > 1e-15:
        a_opt = a_opt / s
    obj_opt = obj_normalized(a_opt)

    if obj_opt < best_obj:
        best_obj = obj_opt
        best_t = t_opt
        best_a = a_opt

    return best_a, best_obj, best_t


def compute_smooth_grad(a, beta):
    """Smooth gradient of max(conv(a,a)) via softmax weighting."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # grad[i] = sum_k w[k]*(a[k-i] + a[i-k])
    fft_size = 1 << (2 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    b = a[::-1]
    B = np.fft.rfft(b, fft_size)

    term1 = np.fft.irfft(W * B, fft_size)[n - 1:n - 1 + n]
    term2 = np.fft.irfft(A * W, fft_size)[:n]
    grad = term1 + term2
    return grad, conv, w


def adam_subgrad_optimize(a0, time_limit, beta_init=100.0, lr_init=0.005,
                          seed=42, verbose=False):
    """Adam-style subgradient descent on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = obj_normalized(a)
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr_init
    beta = beta_init

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv, w = compute_smooth_grad(a, beta)
        grad_d = grad - grad.mean()

        t += 1
        m = b1 * m + (1.0 - b1) * grad_d
        v = b2 * v + (1.0 - b2) * grad_d * grad_d
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a - step)

        obj_new = obj_normalized(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr *= 0.7
            if no_improve > 600:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init
                beta = max(50.0, beta_init * 0.5)
                no_improve = 0

        beta = min(beta * 1.001, beta_init * 30)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t}: obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"beta={beta:.0f}, t={elapsed:.1f}s")

    return best_a, best_obj
