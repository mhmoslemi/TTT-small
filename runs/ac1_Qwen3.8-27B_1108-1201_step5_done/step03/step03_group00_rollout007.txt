    n = len(v)
    u = np.sort(v)[::-1]

    rho = np.nonzero(cond)[0][-1]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peaks, weights):
    """Compute weighted subgradient of max_k conv(a,a)[k]."""
    n = len(a)

        k = int(peaks[i])

        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += 2.0 * w * a[k - idx]
    return sg

            if rem < 15:
                break
            t_alloc = min(per_init, rem - 5)
            if t_alloc < 8:
                break
            a_opt, _ = md_optimize(f_init, time_limit=t_alloc,
                                    lr_init=0.025, temp_init=0.002)
            evaluate_and_track(a_opt, best_state)

    print(f"  Phase 1 best: {best_state['score']:.6f}")

    # === Phase 2: Deep optimization (main effort) ===
    print(f"\n=== Phase 2: Deep Optimization ===")

    if best_state['seq'] is None:
        best_state['seq'] = parent.copy()
        best_state['score'] = float(evaluate_sequence(parent.tolist()))

    f_curr = best_state['seq'].copy()
    n_curr = len(f_curr)
    phase2_end = deadline - 180

    # Main loop: alternate MD and LP
    iteration = 0
    while time.monotonic() < phase2_end:
        iteration += 1
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break

        # Determine MD duration based on remaining time
        if remaining > 600:
            md_dur = 200
        elif remaining > 300:
            md_dur = 150
        elif remaining > 150:
            md_dur = 100
        else:
            md_dur = max(30, remaining - 40)

        # Adaptive learning rate: decrease as we progress
        progress = 1.0 - max(0, remaining - 30) / max(1, phase2_end - (time.monotonic() - (deadline - 180) - (phase2_end - deadline + 180)))
        lr_scale = max(0.3, 1.0 - 0.5 * iteration / 10.0)

        a_md, _ = md_optimize(f_curr, time_limit=md_dur,
                               lr_init=0.015 * lr_scale,
                               temp_init=0.001 * lr_scale,
                               top_k_frac=0.15,
                               restart_interval=max(100, 400 - iteration * 20))
        evaluate_and_track(a_md, best_state)

        if best_state['seq'] is not None:
            f_curr = best_state['seq'] / best_state['seq'].sum()
            n_curr = len(f_curr)

        # LP move
        rem = phase2_end - time.monotonic()
        if rem > 40:
            a_lp, _ = lp_move(f_curr, m_exp=1.03)
            evaluate_and_track(a_lp, best_state)
            if best_state['score'] < 1.51:
                f_curr = best_state['seq'] / best_state['seq'].sum()

        # Subgradient polish occasionally
        rem = phase2_end - time.monotonic()
        if rem > 50 and iteration % 3 == 0:
            a_sg, _ = subgrad_polish(f_curr, time_limit=min(40, rem - 10),
                                      lr=0.004, mom=0.94, top_k=max(16, n_curr // 8))
            evaluate_and_track(a_sg, best_state)
            if best_state['seq'] is not None:
                f_curr = best_state['seq'] / best_state['seq'].sum()

        # Periodic perturbation for exploration
        rem = phase2_end - time.monotonic()
        if rem > 60 and iteration % 4 == 0:
            noise_scale = 0.005 * (1.0 + 0.5 * (iteration / 8.0))
            f_pert = project_simplex(f_curr + rng.standard_normal(n_curr) * noise_scale)
            a_pert, _ = md_optimize(f_pert, time_limit=min(30, rem - 20),
                                     lr_init=0.01 * lr_scale, temp_init=0.0008)
            evaluate_and_track(a_pert, best_state)

        if iteration % 5 == 0:
            print(f"    Iter {iteration}: best={best_state['score']:.6f}, remaining={remaining:.0f}s")

    print(f"  Phase 2 best: {best_state['score']:.6f}")

    # === Phase 3: Final polish (180s) ===
    print(f"\n=== Phase 3: Final Polish ===")

    if best_state['seq'] is not None:
        f_curr = best_state['seq'] / best_state['seq'].sum()
        n_curr = len(f_curr)

        # Fine subgradient
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            a_sg, _ = subgrad_polish(f_curr, time_limit=min(80, rem),
                                      lr=0.002, mom=0.95, top_k=n_curr // 6)
            evaluate_and_track(a_sg, best_state)

        # Very fine MD
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a_md, _ = md_optimize(best_state['seq'], time_limit=min(50, rem),
                                   lr_init=0.003, temp_init=0.0001)
            evaluate_and_track(a_md, best_state)

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for i in range(4):
            rem = deadline - time.monotonic() - 5
            if rem < 8:
                break
            f_p = project_simplex(best_state['seq'] + rng.standard_normal(n_curr) * 0.001)
            a_p, _ = md_optimize(f_p, time_limit=min(15, rem - 2),
                                  lr_init=0.002, temp_init=0.00005)
            evaluate_and_track(a_p, best_state)

        # Final LP nudge
        rem = deadline - time.monotonic() - 5
        if rem > 15 and best_state['seq'] is not None:
            f_c = best_state['seq'] / best_state['seq'].sum()
            a_lp, _ = lp_move(f_c, m_exp=1.01)
            evaluate_and_track(a_lp, best_state)

    # Fallback
    if best_state['seq'] is None:
        best_state['seq'] = parent.copy()
        best_state['score'] = float(evaluate_sequence(parent.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_state['seq'].tolist()))
    if np.isfinite(final_score):
        best_state['score'] = final_score

    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
