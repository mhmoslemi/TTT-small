import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_subgradient(a, conv, top_k=5):
    """
    Compute weighted subgradient of max(conv) using softmax weighting
    over the top-k peaks.
    """
    n = len(a)
    # Weight peaks by softmax
    top_indices = np.argsort(conv)[::-1][:top_k]
    w = np.zeros(len(conv))
    for i in top_indices:
        w[i] = 1.0
    
    # For each peak at index k, the gradient of conv[k] w.r.t. a[i] is a[k-i]
    # So the weighted subgradient is: sum_k w[k] * 2 * a[k-i]
    # This is 2 * cross_correlation(a, w)
    
    # cross[k] = sum_i a[i] * w[k+i] (with appropriate bounds)
    # Actually: grad[i] = 2 * sum_k w[k] * a[k-i] (where k-i is valid)
    # This is 2 * (w * a)[i] where * is cross-correlation
    
    # Using FFT: cross_correlation(w, a)[i] = sum_k w[k] * a[k-i]
    # = ifft(fft(w) * conj(fft(a)))... 
    
    # Actually let me think more carefully.
    # conv[k] = sum_i a[i] * a[k-i]
    # d(conv[k])/d(a[i]) = a[k-i]
    # 
    # Weighted subgradient: sum_k w[k] * a[k-i]
    # For fixed i: sum_k w[k] * a[k-i]
    # Let j = k-i, so k = j+i: sum_j w[j+i] * a[j]
    # This is the cross-correlation of w and a at lag i.
    
    fft_size = 1 << (n + len(conv) - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    
    return 2.0 * cross[:n]

def peak_equalization(a, conv, n_peaks=10, strength=0.01):
    """
    Try to equalize the top peaks by moving mass from high-contribution
    elements to low-contribution ones.
    """
    n = len(a)
    a_new = a.copy()
    
    # Sort peaks
    sorted_peaks = np.argsort(conv)[::-1]
    top_peaks = sorted_peaks[:n_peaks]
    max_val = conv[sorted_peaks[0]]
    mean_val = conv.mean()
    
    # For each top peak, identify the most contributing pair
    for k in top_peaks:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        
        # Find the pair with maximum product
        if i_end < i_start:
            continue
        products = np.array([a_new[i] * a_new[k - i] for i in range(i_start, i_end + 1)])
        if len(products) == 0:
            continue
        
        max_idx = np.argmax(products)
        i_max = i_start + max_idx
        j_max = k - i_max
        
        # Find a position to move to (one with low contribution to all top peaks)
        # Try a few random positions
        for _ in range(5):
            j_new = np.random.randint(0, n)
            if j_new == i_max or j_new == j_max:
                continue
            
            # Check if moving helps
            amt = a_new[i_max] * strength
            if amt < 1e-10:
                break
            a_try = a_new.copy()
            a_try[i_max] -= amt
            a_try[j_new] += amt
            conv_try = fast_convolve(a_try)
            if conv_try.max() < conv.max() * (1 - 1e-6):
                a_new = a_try
                break
    
    return a_new

        a = a / s
    n = len(a)

    conv = fast_convolve(a)

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    perturb_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        
        # Annealing schedule
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.7)
        lr = lr_start * (1.0 - 0.8 * frac)
        lr = max(lr, 0.0005)
        
        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)
        
        # Smoothed gradient
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        
        w = conv_y * p
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)

        grad = 2.0 * cross[:n]

        cur_obj = conv.max()
        total_iters += 1

def lp_refine(a0, M, n):
    """
    Solve LP to find direction b that maximizes sum(b) subject to conv(a,b) <= M, b >= 0.
    """
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a0[:k_end - k_start]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Monotone increasing (should reduce center peak)
    mono = np.arange(1, n + 1, dtype=np.float64)
    mono /= mono.sum()
    inits['mono_inc'] = mono
    inits['mono_dec'] = mono[::-1].copy()
    
    # Quadratic ramp
    quad = x**2 + 1
    quad /= quad.sum()
    inits['quad_inc'] = quad
    inits['quad_dec'] = quad[::-1].copy()
    
    # Exponential ramp
    exp_ramp = np.exp(0.5 * x / n)
    exp_ramp /= exp_ramp.sum()
    inits['exp_inc'] = exp_ramp
    inits['exp_dec'] = exp_ramp[::-1].copy()
    
    # Bell shapes
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - n/2) / (sigma * n))**2)
        g /= g.sum()
        inits[f'gauss_{int(sigma*100)}'] = g
    
    # Off-center
    g_off = np.exp(-0.5 * ((x - 0.25 * n) / (0.2 * n))**2)
    g_off /= g_off.sum()
    inits['gauss_off'] = g_off
    
    # Championship shape (high at edges, low in middle) - anti-bell
    champ = 1.0 / (1.0 + ((x - n/2) / (n/4))**2)
    champ /= champ.sum()
    inits['champ'] = champ
    
    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)
    
    # Parent-based
    if parent is not None:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.maximum(parent, 0)
            # Downsample
            idx = np.linspace(0, len(parent)-1, n).astype(int)
            pp = pp[idx]
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        else:
            pp = np.maximum(parent, 0)
            # Upsample
            pp = np.interp(x, np.linspace(0, n-1, len(parent)), pp)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
    
    return inits

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
- Phase 1: Quick scan over lengths and initializations (emphasizing monotone/asymmetric starts)
- Phase 2: Deep optimization with smoothed gradient + peak equalization + LP direction
- Phase 3: Coordinate refinement and fine-tuning
Key innovation: exploits monotone initializations (rearrangement inequality) to reduce
center convolution, combined with adaptive peak-targeted updates.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, conv, beta):
    """
    Gradient of logsumexp(beta*conv)/beta w.r.t. a via FFT cross-correlation.
    conv = auto-convolution of a (length 2n-1).
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def peak_targeted_update(a, conv, rng, strength=0.02, n_peaks=6):
    """
    Directly target the top peaks: find the most contributing pair for each peak
    and try to redistribute mass to reduce it.
    """
    n = len(a)
    a_new = a.copy()
    improved = False

    top_peaks = np.argsort(conv)[::-1][:n_peaks]

    for k in top_peaks:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end < i_start:
            continue

        # Find the pair with maximum product contributing to this peak
        products = np.array([a_new[i] * a_new[k - i]
                            for i in range(i_start, i_end + 1)])
        if len(products) == 0 or products.max() < 1e-15:
            continue

        max_p = np.argmax(products)
        i_pk = i_start + max_p
        j_pk = k - i_pk

        # Try moving mass from the peak contributor to random other positions
        for _ in range(4):
            j_new = rng.integers(0, n)
            if j_new in (i_pk, j_pk):
                continue
            for frac in [0.03, 0.07, 0.12]:
                amt = a_new[i_pk] * frac
                if amt < 1e-12:
                    break
                a_try = a_new.copy()
                a_try[i_pk] -= amt
                a_try[j_new] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < conv.max() * (1 - 1e-8):
                    a_new = a_try
                    improved = True
                    break
            if improved:
                break

    return a_new, improved


def gradient_optimize(a0, time_limit, seed=0, beta_start=50.0, beta_end=400.0,
                      lr_start=0.015, momentum=0.92, peak_eq_every=150,
                      peak_eq_strength=0.015, perturb_threshold=400):

    Nesterov accelerated projected gradient descent with periodic peak equalization.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0

        frac = min(1.0, elapsed / max(time_limit, 1e-6))

        beta = beta_start + (beta_end - beta_start) * (frac ** 0.6)
        lr = max(0.0004, lr_start * (1.0 - 0.75 * frac))

        # Nesterov lookahead
        y = a + momentum * v
        np.clip(y, 0, None, out=y)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys

        conv_y = fast_convolve(y)
        grad = smoothed_grad(y, conv_y, beta)

        # Momentum update
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Peak targeted update
        if total_iters % peak_eq_every == 0:
            a_eq, eq_improved = peak_targeted_update(
                a, fast_convolve(a), rng,
                strength=peak_eq_strength * (1.0 + 0.5 * np.sin(total_iters * 0.03)))
            if eq_improved:

                if conv_eq.max() < best_obj:
                    best_obj = conv_eq.max()
                    best_a = a_eq.copy()
                a = a_eq
                v *= 0.4
                no_improve = 0

        # Perturbation for escape
        if no_improve >= perturb_threshold:
            scale = 0.004 * (1.0 + 0.3 * np.sin(total_iters * 0.01))
            a = project_simplex(a + rng.standard_normal(n) * scale)
            v *= 0.15
            no_improve = 0
            perturb_threshold = int(perturb_threshold * 1.3)
            lr_start = max(0.002, lr_start * 0.6)

    return best_a, best_obj


def lp_solve(a, M, n, time_limit=45):
    """
    LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_end = min(2 * n - 1, i + n)
        A_ub[i:k_end, i] = a[:k_end - i]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum


def lp_optimize(a0, time_limit, seed=0):
    """
    LP-based iterative: repeatedly find direction b that keeps conv(a,b) <= M,
    then move toward b.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit - 5:
        conv_cur = fast_convolve(a)
        M = conv_cur.max()

        b = lp_solve(a, M * 1.0005, n, time_limit=min(45, time_limit - 10))
        if b is None:
            noise = rng.standard_normal(n) * 0.008
            a = project_simplex(a + noise)
            stuck += 1
            if stuck > 15:
                break
            continue

        stuck = 0
        improved = False
        for t in [0.15, 0.25, 0.35, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            if conv_try.max() < best_obj - 1e-12:
                best_obj = conv_try.max()
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        if not improved:
            for t in [0.05, 0.08]:
                a_try = (1 - t) * a + t * b
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                if conv_try.max() < M:
                    a = a_try.copy()
                    if conv_try.max() < best_obj:
                        best_obj = conv_try.max()
                        best_a = a.copy()
                    break

    return best_a, best_obj


def coordinate_polish(a0, time_limit, seed=0):
    """
    Fine coordinate descent: repeatedly adjust single coordinate pairs
    to reduce the peak.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))

        # Target the peak's top contributor
        i_start = max(0, max_idx - n + 1)
        i_end = min(n - 1, max_idx)
        if i_end >= i_start:
            prods = np.array([a[i] * a[max_idx - i]
                            for i in range(i_start, i_end + 1)])
            if len(prods) > 0:
                ip = i_start + int(np.argmax(prods))
                jp = max_idx - ip
                for frac in [0.01, 0.02, 0.04, 0.06, 0.1]:
                    amt = a[ip] * frac
                    if amt < 1e-11:
                        break
                    for _ in range(3):
                        jn = rng.integers(0, n)
                        if jn in (ip, jp):
                            continue
                        a_try = a.copy()
                        a_try[ip] -= amt
                        a_try[jn] += amt
                        ct = fast_convolve(a_try)
                        if ct.max() < best_obj - 1e-13:
                            best_obj = ct.max()
                            best_a = a_try.copy()
                            a = a_try.copy()
                            break

        # Random pair adjustment
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j and a[i] > 1e-9:
            for frac in [0.01, 0.03]:
                amt = a[i] * frac
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                ct = fast_convolve(a_try)
                if ct.max() < best_obj - 1e-13:
                    best_obj = ct.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a, best_obj


def make_inits(n, parent, rng):
    """Generate diverse initializations emphasizing asymmetric/monotone shapes."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Monotone (exploits rearrangement inequality to reduce center peak)
    mono = np.arange(1, n + 1, dtype=np.float64)
    mono /= mono.sum()
    inits['mono_inc'] = mono
    inits['mono_dec'] = mono[::-1].copy()

    # Exponential ramps (sharper than linear)
    for rate in [0.5, 1.0, 2.0]:
        e = np.exp(rate * x / n)
        e /= e.sum()
        inits[f'exp_inc_{int(rate*10)}'] = e.copy()
        inits[f'exp_dec_{int(rate*10)}'] = e[::-1].copy()

    # Power law
    for pwr in [2, 3]:
        pl = (x + 1) ** pwr
        pl /= pl.sum()
        inits[f'pow{pwr}_inc'] = pl.copy()
        inits[f'pow{pwr}_dec'] = pl[::-1].copy()

    # Gaussian (symmetric - baseline)
    for sf in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - n / 2) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Off-center Gaussian
    for pos in [0.2, 0.3, 0.7]:
        g = np.exp(-0.5 * ((x - pos * n) / (0.2 * n)) ** 2)
        g /= g.sum()
        inits[f'goff_{int(pos*100)}'] = g

    # Random Dirichlet
    inits['rand_a'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_b'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_c'] = rng.dirichlet(np.ones(n) * 2.5)

    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.maximum(parent, 0)
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            pp = pp[idx]
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        else:
            pp = np.maximum(parent, 0)
            pp = np.interp(x, np.linspace(0, n - 1, len(pp)), pp)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 50.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
            print(f"Parent: n={len(p)}, score={parent_score:.6f}", flush=True)
    except Exception as e:
        print(f"No parent available: {e}", flush=True)

    # ========== Phase 1: Quick Scan ==========
    print("=== Phase 1: Quick Scan ===", flush=True)
    lengths = [192, 256, 320, 384, 448, 512, 640]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)

    phase1_budget = 280.0
    t_p1 = time.monotonic()
    scan_results = []

    for n in lengths:
        if time.monotonic() - t_p1 > phase1_budget:
            break
        if time.monotonic() > deadline - 350:
            break

        inits = make_inits(n, parent, rng)

        # Priority: parent, then monotone (key insight), then diverse
        priority = ['parent', 'parent_ds', 'parent_up',
                    'mono_inc', 'mono_dec', 'exp_inc_10', 'exp_dec_10',
                    'exp_inc_20', 'pow2_inc', 'pow3_inc',
                    'goff_30', 'goff_70', 'uniform', 'rand_a']
        names = [nm for nm in priority if nm in inits]
        names = names[:7]

        per_init = max(4.0, phase1_budget / (max(1, len(lengths)) * 7))

        for name in names:
            if time.monotonic() > deadline - 300:
                break
            if time.monotonic() - t_p1 > phase1_budget:
                break
            t_alloc = min(per_init, 35.0)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt, _ = gradient_optimize(
                a0, t_alloc, seed=int(rng.integers(0, 100000)),
                beta_start=40, beta_end=250,
                lr_start=0.014, momentum=0.91,
                peak_eq_every=100, peak_eq_strength=0.02)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    for r in scan_results[:5]:
        print(f"  Scan: n={r[1]}, {r[2]}: {r[0]:.6f}", flush=True)
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # ========== Phase 2: Deep Optimization ==========
    print("=== Phase 2: Deep Optimization ===", flush=True)

    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_n:
            top_configs.append((n, a_opt))
            seen_n.add(n)
        if len(top_configs) >= 5:
            break

    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add independent restarts
    best_n = len(best_sequence) if best_sequence is not None else 384
    top_configs.append((best_n, make_inits(best_n, parent, rng).get('mono_inc', np.ones(best_n) / best_n)))
    top_configs.append((best_n, make_inits(best_n, parent, rng).get('exp_inc_10', rng.dirichlet(np.ones(best_n) * 0.5))))

    remaining = deadline - time.monotonic() - 120.0
    n_configs = max(1, len(top_configs))
    per_config = max(25.0, remaining / n_configs)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(per_config, 280.0)
        if t_alloc < 15:
            break

        print(f"  Config {i}: n={n}", flush=True)

        # Sub-phase A: Strong gradient (45%)
        t_a = t_alloc * 0.45
        a_a, _ = gradient_optimize(
            a0, t_a, seed=int(rng.integers(0, 100000)),
            beta_start=35, beta_end=200,
            lr_start=0.012, momentum=0.93,
            peak_eq_every=120, peak_eq_strength=0.018)

        score_a = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    A(Grad): {score_a:.6f}", flush=True)
        a_work = a_a

        # Sub-phase B: LP refinement (30%)
        t_b = t_alloc * 0.30
        if t_b > 15:
            a_b, _ = lp_optimize(a_work, t_b, seed=int(rng.integers(0, 100000)))
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    B(LP): {score_b:.6f}", flush=True)
            a_work = a_b

        # Sub-phase C: Fine gradient (25%)
        t_c = t_alloc * 0.25
        if t_c > 8:
            a_c, _ = gradient_optimize(
                a_work, t_c, seed=int(rng.integers(0, 100000)),
                beta_start=150, beta_end=600,
                lr_start=0.006, momentum=0.95,
                peak_eq_every=80, peak_eq_strength=0.01)

            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    C(Fine): {score_c:.6f}", flush=True)
            a_work = a_c

    # ========== Phase 3: Polish ==========
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0

        # Coordinate refinement
        t_coord = remaining * 0.30
        if t_coord > 10:
            a_cr, _ = coordinate_polish(best_sequence, t_coord, seed=123)
            score_cr = float(evaluate_sequence(a_cr.tolist()))
            if np.isfinite(score_cr) and score_cr < best_score:
                best_score = score_cr
                best_sequence = a_cr.copy()
                print(f"  Coord: {best_score:.6f}", flush=True)

        # Very fine gradient
        remaining = deadline - time.monotonic() - 10.0
        t_fg = remaining * 0.30
        if t_fg > 8:
            a_fg, _ = gradient_optimize(
                best_sequence, t_fg, seed=456,
                beta_start=400, beta_end=1500,
                lr_start=0.003, momentum=0.97,
                peak_eq_every=60, peak_eq_strength=0.008)
            score_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  FineGrad: {best_score:.6f}", flush=True)

        # Final coordinate pass
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 8:
            a_fp, _ = coordinate_polish(best_sequence, remaining, seed=789)
            score_fp = float(evaluate_sequence(a_fp.tolist()))
            if np.isfinite(score_fp) and score_fp < best_score:
                best_score = score_fp
                best_sequence = a_fp.copy()
                print(f"  FinalCoord: {best_score:.6f}", flush=True)

        # One more LP if time allows
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 20:
            a_lp, _ = lp_optimize(best_sequence, remaining, seed=321)
            score_lp = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"  FinalLP: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_sequence)} ===", flush=True)
    return [float(x) for x in best_sequence]
