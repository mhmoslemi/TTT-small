"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(p, conv, tol=0.01):
    """Compute subgradient of max_k conv_k at p using active set."""
    n = len(p)
    cmax = conv.max()
    if cmax < 1e-15:
        return np.zeros(n)
    active = np.where(conv >= cmax * (1.0 - tol))[0]
    m = len(active)
    # sg[j] = 2 * sum_{k in active} p[k-j] (with p out of range = 0)
    sg = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        sg[j_start:j_end + 1] += p[k - j_end:k - j_start + 1][::-1]
    return (2.0 * n / m) * sg


def subgradient_optimize(p_init, time_limit, lr_init=0.02, momentum=0.85,
                         tol=0.02, polyak_decay=0.995):
    """Projected subgradient with momentum and adaptive step size."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)

    conv = fast_convolve(p)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()
    v = np.zeros(n)
    lr = lr_init
    best_conv_val = best_obj / (2.0 * n)

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(p)
        cmax = conv.max()
        cur_obj = 2.0 * n * cmax

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_p = p.copy()
            best_conv_val = cmax
            no_improve = 0
        else:
            no_improve += 1

        sg = compute_subgradient(p, conv, tol=tol)
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm

        # Polyak-like step size
        gap = cmax - (1.0 / n)  # lower bound on max conv is 1/n (rough)
        if gap > 1e-8:
            lr_polyak = min(lr, 2.0 * gap / (sg_norm * sg_norm + 1e-15))
            lr = max(lr * polyak_decay, 1e-5)
            step = min(lr, lr_polyak)
        else:
            step = lr

        v = momentum * v - step * 2.0 * n * sg
        p = project_simplex(p + v)

        if no_improve > 300:
            rng_local = np.random.default_rng(hash(int(cmax * 1e10)) % (2**31))
            p = project_simplex(p + rng_local.standard_normal(n) * 0.01)
            v *= 0.3
            no_improve = 0
            lr = max(0.002, lr * 0.5)

    return best_p, best_obj


def build_lp_matrix(n):
    """Build sparse representation of LP constraint matrix."""
    rows = []
    cols = []
    vals = []
    return rows, cols, vals


def solve_lp_direction(p, M, n, max_iter=500):
    """Solve LP: max sum(g) s.t. g>=0, (p*g)_k <= M for all k.
    Returns normalized g on simplex, or None."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A_ub[k, j_start:j_end + 1] = p[k - j_end:k - j_start + 1][::-1]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                  'time_limit': 10})
        if result.success and result.fun > -1e10:
            g = np.maximum(result.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_subgradient_hybrid(p_init, time_limit, lp_every=10, lr_init=0.015,

    it = 0
    no_improve = 0

def simulated_annealing(p_init, time_limit, t0_init=0.005, cooling=0.9997,
                        move_scale=0.03):
    """Simulated annealing with diverse move types on the simplex."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)

    conv = fast_convolve(p)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    T = t0_init
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Generate a move
        move_type = rng.integers(0, 4)
        p_new = p.copy()

        if move_type == 0:
            # Transfer mass between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = rng.uniform(0, move_scale) * p[i]
                p_new[i] -= delta
                p_new[j] += delta
        elif move_type == 1:
            # Redistribute within a random block
            block_size = rng.integers(3, 8)
            start = rng.integers(0, n - block_size)
            block = p_new[start:start + block_size]
            block_sum = block.sum()
            if block_sum > 1e-10:
                new_block = rng.dirichlet(np.ones(block_size) * 2.0) * block_sum
                p_new[start:start + block_size] = new_block
        elif move_type == 2:
            # Smooth with small kernel
            kernel = np.exp(-0.5 * (np.arange(-2, 3) / 1.0) ** 2)
            kernel /= kernel.sum()
            p_new = np.convolve(p, kernel, mode='same')
        else:
            # Perturb a random subset
            mask = rng.random(n) < 0.1
            if mask.any():
                subset = p_new[mask]
                subset_sum = subset.sum()
                if subset_sum > 1e-10:
                    new_subset = project_simplex(subset + rng.standard_normal(mask.sum()) * 0.05) * subset_sum
                    p_new[mask] = new_subset

        p_new = np.maximum(p_new, 0.0)
        ps = p_new.sum()
        if ps < 1e-12:
            continue
        p_new /= ps

        conv_new = fast_convolve(p_new)
        cur_new = 2.0 * n * conv_new.max()
        delta = cur_new - cur_obj

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            p = p_new
            cur_obj = cur_new
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_p = p.copy()

        T *= cooling

    return best_p, best_obj


def coordinate_descent_polish(p_init, time_limit, step_frac=0.1):

    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        return p
    p = p / s
    n = len(p)

    conv = fast_convolve(p)
    best_obj = 2.0 * n * conv.max()

    t0 = time.monotonic()
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        sg = compute_subgradient(p, conv, tol=0.01)

        # Find good sources (high sg) and sinks (low sg)
        src_candidates = np.argsort(sg)[-10:]  # high gradient = want to reduce
        sink_candidates = np.argsort(sg)[:10]  # low gradient = want to increase

        for i in src_candidates:
            if p[i] < 1e-8:
                continue
            for j in sink_candidates:
                if j == i:
                    continue
                for frac in [step_frac, step_frac * 0.5, step_frac * 0.25]:
                    delta = p[i] * frac
                    if delta < 1e-10:
                        continue
                    p_test = p.copy()
                    p_test[i] -= delta
                    p_test[j] += delta
                    p_test = np.maximum(p_test, 0.0)
                    st = p_test.sum()
                    if st < 1e-12:
                        continue
                    p_test /= st
                    conv_test = fast_convolve(p_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-12:
                        p = p_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        break

        if not improved:
            # Try random pairs
            rng_local = np.random.default_rng(hash(int(best_obj * 1e8)) % (2**31))
            for _ in range(50):
                i = rng_local.integers(0, n)
                j = rng_local.integers(0, n)
                if i == j or p[i] < 1e-8:
                    continue
                delta = p[i] * 0.05
                p_test = p.copy()
                p_test[i] -= delta
                p_test[j] += delta
                p_test = np.maximum(p_test, 0.0)
                st = p_test.sum()
                if st < 1e-12:
                    continue
                p_test /= st
                conv_test = fast_convolve(p_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-12:
                    p = p_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True

        if not improved:
            break

    return p


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse smooth initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    xn = (x + 0.5) / n  # normalized to [0,1]

    # Sine-squared window (zero at edges)
    s1 = np.sin(np.pi * xn) ** 2
    s1 = np.maximum(s1, 0)
    if s1.sum() > 1e-10:
        inits['sine2'] = s1 / s1.sum()

    # Sine to odd power
    for p_val in [3, 5, 7]:
        sp = np.sin(np.pi * xn) ** p_val
        sp = np.maximum(sp, 0)
        if sp.sum() > 1e-10:
            inits[f'sine{p_val}'] = sp / sp.sum()

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * xn))
    rc = np.maximum(rc - 0.01, 0)
    if rc.sum() > 1e-10:
        inits['raised_cos'] = rc / rc.sum()

    # Triangle (zero at edges)
    tri = np.maximum(1 - 2 * np.abs(xn - 0.5), 0)
    if tri.sum() > 1e-10:
        inits['triangle'] = tri / tri.sum()

    # Polynomial window
    for q in [2, 3, 4]:
        pm = (1 - (2 * xn - 1) ** 2) ** q
        pm = np.maximum(pm, 0)
        if pm.sum() > 1e-10:
            inits[f'poly{q}'] = pm / pm.sum()

    # Gaussian (centered)
    for sigma in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((xn - 0.5) / sigma) ** 2)
        if g.sum() > 1e-10:
            inits[f'gauss_{sigma}'] = g / g.sum()

    # Bimodal
    g2 = np.exp(-0.5 * ((xn - 0.3) / 0.1) ** 2) + np.exp(-0.5 * ((xn - 0.7) / 0.1) ** 2)
    if g2.sum() >1e-10:
        inits['bimodal'] = g2 / g2.sum()

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'random_{alpha}'] = r

    # Edge-concentrated (mass near edges but not at them)
    e = np.zeros(n)
    w = max(2, n // 10)
    for i in range(w):
        e[i] = np.sin(np.pi * (i + 1) / (w + 1))
        e[n - 1 - i] = np.sin(np.pi * (i + 1) / (w + 1))
    if e.sum() > 1e-10:
        inits['edge'] = e / e.sum()

    # Parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                p_interp = np.maximum(p_interp, 0)
                ps2 = p_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_ds'] = p_interp / ps2
            else:
                p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                p_interp = np.maximum(p_interp, 0)
                ps2 = p_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_up'] = p_interp / ps2

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid search: smooth-parametric inits + subgradient + LP +
    simulated annealing + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if np.isfinite(par_score):
                best_score = par_score
                best_sequence = p.copy()
    except Exception:
        pass

    # === Phase 1: Global exploration ===
    print("=== Phase 1: Global Exploration ===")
    n_main = 512
    lengths_to_try = [n_main]
    if time.monotonic() < deadline - 600:
        lengths_to_try.append(384)
    if time.monotonic() < deadline - 800:
        lengths_to_try.append(640)

    all_results = []
    phase1_end = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.3)

    for n in lengths_to_try:
        if time.monotonic() > phase1_end - 30:
            break
        inits = generate_initializations(n, parent, rng)
        # Priority ordering
        prio = ['parent', 'parent_ds', 'parent_up', 'sine2', 'sine3', 'sine5',
                'raised_cos', 'triangle', 'poly2', 'poly3', 'gauss_0.3',
                'bimodal', 'edge', 'random_1.0', 'random_0.5', 'random_2.0',
                'sine7', 'poly4', 'gauss_0.2', 'gauss_0.4']
        names = [nm for nm in prio if nm in inits]
        names = names[:8]  # Limit to 8 per length

        t_per_init = max(10.0, 350.0 / (len(lengths_to_try) * 8))

        for name in names:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per_init, phase1_end - time.monotonic() - 10)
            if t_alloc < 5:
                break

            p0 = inits[name]
            # Fast subgradient
            p_opt, obj_opt = subgradient_optimize(p0, t_alloc, lr_init=0.02,
                                                  momentum=0.85, tol=0.02)
            score = float(evaluate_sequence(p_opt.tolist()))
            if np.isfinite(score):
                all_results.append((score, n, name, p_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = p_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    all_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # === Phase 2: Deep optimization on top candidates ===
    print("=== Phase 2: Deep Optimization ===")
    top_candidates = []
    seen = set()
    for score, n, name, p_opt in all_results[:10]:
        key = (n, name)
        if key not in seen:
            top_candidates.append((n, name, p_opt))
            seen.add(key)
        if len(top_candidates) >= 6:
            break
    # Also include current best
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 400.0
    t_per_cand = max(30.0, remaining / max(1, len(top_candidates)))

    for idx, (n, name, p0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_cand, deadline - time.monotonic() - 300)
        if t_alloc < 15:
            break

        print(f"  Deep opt: n={n} {name} (t={t_alloc:.0f}s)")

        # LP + subgradient hybrid
        t_lp = t_alloc * 0.5
        t_sg = t_alloc * 0.3
        t_sa = t_alloc * 0.2

        p_opt, obj_opt = lp_subgradient_hybrid(p0, t_lp, lp_every=8,
                                              lr_init=0.012, momentum=0.88, tol=0.012)
        score = float(evaluate_sequence(p_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = p_opt.copy()
            print(f"    LP+Sg: {score:.6f}")

        # Subgradient continuation
        if time.monotonic() < deadline - 280:
            p_opt2, obj2 = subgradient_optimize(p_opt, t_sg, lr_init=0.008,
                                               momentum=0.9, tol=0.01)
            score2 = float(evaluate_sequence(p_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = p_opt2.copy()
                print(f"    Sg cont: {score2:.6f}")

        # Simulated annealing
        if time.monotonic() < deadline - 260:
            p_opt3, obj3 = simulated_annealing(p_opt, t_sa, t0_init=0.003,
                                              cooling=0.9998, move_scale=0.02)
            score3 = float(evaluate_sequence(p_opt3.tolist()))
            if np.isfinite(score3) and score3 < best_score:
                best_score = score3
                best_sequence = p_opt3.copy()
                print(f"    SA: {score3:.6f}")

    print(f"Phase 2 best: {best_score:.6f}")

    # === Phase 3: Intensive refinement on best ===
    print("=== Phase 3: Intensive Refinement ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 150.0

        # Cycle through methods
        for cycle in range(min(4, max(1, int(remaining // 80)))):
            if time.monotonic() > deadline - 120:
                break
            t_cyc = min(remaining / (4 - cycle), 80.0)
            if t_cyc < 10:
                break

            # LP + subgrad
            t1 = t_cyc * 0.4
            p_opt, obj_opt = lp_subgradient_hybrid(best_sequence, t1, lp_every=6,
                                                  lr_init=0.006, momentum=0.9, tol=0.008)
            score = float(evaluate_sequence(p_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = p_opt.copy()
                print(f"  Cycle {cycle} LP+Sg: {score:.6f}")

            # SA
            if time.monotonic() < deadline - 100:
                t2 = t_cyc * 0.25
                p_opt2, obj2 = simulated_annealing(p_opt, t2, t0_init=0.001,
                                                  cooling=0.9999, move_scale=0.01)
                score2 = float(evaluate_sequence(p_opt2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = p_opt2.copy()
                    print(f"  Cycle {cycle} SA: {score2:.6f}")

            # Subgradient fine
            if time.monotonic() < deadline - 90:
                t3 = t_cyc * 0.2
                p_opt3, obj3 = subgradient_optimize(best_sequence, t3, lr_init=0.004,
                                                   momentum=0.92, tol=0.005)
                score3 = float(evaluate_sequence(p_opt3.tolist()))
                if np.isfinite(score3) and score3 < best_score:
                    best_score = score3
                    best_sequence = p_opt3.copy()
                    print(f"  Cycle {cycle} Sg: {score3:.6f}")

    # === Phase 4: Final polish ===
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0

        # Coordinate descent
        t_cd = min(100.0, remaining * 0.4)
        if t_cd > 5:
            p_cd = coordinate_descent_polish(best_sequence, t_cd, step_frac=0.08)
            score_cd = float(evaluate_sequence(p_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = p_cd.copy()
                print(f"  CD: {score_cd:.6f}")

        # Final subgradient with very small steps
        remaining = deadline - time.monotonic() - 30.0
        t_final = min(60.0, remaining * 0.5)
        if t_final > 5:
            p_fin, obj_fin = subgradient_optimize(best_sequence, t_final,
                                                 lr_init=0.002, momentum=0.93, tol=0.003)
            score_fin = float(evaluate_sequence(p_fin.tolist()))
            if np.isfinite(score_fin) and score_fin < best_score:
                best_score = score_fin
                best_sequence = p_fin.copy()
                print(f"  Final Sg: {score_fin:.6f}")

        # One more CD
        if time.monotonic() < deadline - 20:
            p_cd2 = coordinate_descent_polish(best_sequence, 15.0, step_frac=0.04)
            score_cd2 = float(evaluate_sequence(p_cd2.tolist()))
            if np.isfinite(score_cd2) and score_cd2 < best_score:
                best_score = score_cd2
                best_sequence = p_cd2.copy()
                print(f"  Final CD: {score_cd2:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
