def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix A_ub for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp_full(a, M):
    """Solve LP with full constraints: max sum(b) s.t. b >= 0, (a*b)[k] <= M."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = build_lp_matrix(a)
    b_ub = np.full(num_k, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_topk(a, M, top_k, conv):
    """Solve LP with only top-k peak constraints."""
    n = len(a)
    num_k = 2 * n - 1
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    A_ub_full = build_lp_matrix(a)
    A_ub = A_ub_full[peaks]
    b_ub = np.full(len(peaks), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_subgradient(a, k_star):
    """Compute gradient of conv(a,a)[k_star] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        grad[j] = val
    return grad


def eval_obj_normalized(a):
    """Evaluate 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def generate_parametric(n, shape_type, params):
    """Generate a parametric sequence of length n."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1  # range [-1, 1]
    
    if shape_type == 'raised_cosine':
        # a[i] = 0.5 + 0.5*cos(pi*t) -- peaks at 0, 1 (edges), zero at center
        # We want peak at center: a[i] = 0.5 - 0.5*cos(pi*t)
        a = 0.5 - 0.5 * np.cos(np.pi * t)
    elif shape_type == 'bell':
        p = params.get('p', 2.0)
        a = np.maximum(0, 1 - np.abs(t) ** p)
    elif shape_type == 'gaussian':
        sigma = params.get('sigma', 0.5)
        a = np.exp(-t ** 2 / (2 * sigma ** 2))
    elif shape_type == 'parabolic':
        a = np.maximum(0, 1 - t ** 2)
    elif shape_type == 'super_bell':
        p = params.get('p', 3.0)
        a = np.maximum(0, 1 - np.abs(t) ** p) ** params.get('q', 1.0)
    elif shape_type == 'mixture':
        # Mixture of two bells
        c1, s1 = params.get('c1', 0.5), params.get('s1', 0.5)
        c2, s2 = params.get('c2', 0.5), params.get('s2', 0.4)
        a = c1 * np.exp(-t ** 2 / (2 * s1 ** 2)) + c2 * np.exp(-(t - 0.3) ** 2 / (2 * s2 ** 2))
    elif shape_type == 'cosine_window':
        # Hamming-like: 0.54 + 0.46*cos(pi*t)
        alpha = params.get('alpha', 0.54)
        a = alpha + (1 - alpha) * np.cos(np.pi * t)
        a = np.maximum(a, 0.01)
    elif shape_type == 'sinc_like':
        # |sin(pi*p*t)/(pi*p*t)|^2
        p = params.get('p', 1.5)
        z = np.pi * p * t
        z[0] = 1e-10
        a = (np.sin(z) / z) ** 2
    elif shape_type == 'flat_top':
        # Flat top with raised cosine skirts
        w = params.get('w', 0.5)  # width of flat region
        a = np.ones(n)
        skirt = max(1, int(w * n))
        for i in range(skirt):
            fade = 0.5 * (1 - np.cos(np.pi * i / skirt))
            a[i] = fade
            a[n - 1 - i] = fade
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def smooth_gradient_nesterov(a0, time_limit, beta_init=500.0, lr_init=0.03,
                              verbose=False, seed=0):
    """
    Smooth gradient optimization with Nesterov acceleration.
    Uses log-sum-exp approximation of max with adaptive beta.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    y = a.copy()  # Nesterov momentum point
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    no_improve = 0
    t_k = 1.0
    
    while time.monotonic() - t0 < time_limit:
        # Evaluate at Nesterov point
        y = project_simplex(y)  # Ensure feasibility
        conv_y = fast_convolve(y)
        
        # Compute smooth gradient at y
        scaled = beta * conv_y
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv_y))
            w[np.argmax(conv_y)] = 1.0
        else:
            w /= w_sum
        
        # Gradient computation using FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        w_padded = np.zeros(fft_size)
        w_padded[:len(w)] = w
        a_padded = np.zeros(fft_size)
        a_padded[:n] = y
        
        W = np.fft.fft(w_padded)
        Y = np.fft.fft(a_padded)
        
        # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
        cross = np.fft.ifft(Y * np.conj(W)).real[:n]
        conv_wy = np.fft.ifft(Y * W).real[:n]
        grad = cross + conv_wy
        
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t_k ** 2)) / 2
        theta = (t_k - 1) / t_new
        v = theta * v - lr * grad
        y = project_simplex(a + v)
        a_new = project_simplex(y - theta * v)
        t_k = t_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(best_a + noise)
                y = a.copy()
                v = np.zeros(n)
                lr = lr_init * 0.5
                t_k = 1.0
                no_improve = 0
        
        # Increase beta slowly for sharper max approximation
        if no_improve % 100 == 0:
            beta = min(5000.0, beta * 1.05)
        
        # Adaptive learning rate
        if no_improve == 0 and lr < lr_init:
            lr = min(lr_init, lr * 1.1)
    
    conv_final = fast_convolve(best_a)

def downsample_sequence(a, target_n):
    """Downsample a sequence to target length using interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    x_orig = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(x_new, x_orig, a)
    return np.maximum(a_interp, 0.0)

    # ========== Phase 1: Shape Search + Quick Scan ==========
    print("=== Phase 1: Parametric Shape Search + Quick Scan ===")
    t_p1 = time.monotonic()
    p1_time = 150.0
    
    scan_results = []
    
    # Try different lengths
    lengths_to_try = [128, 192, 256, 320, 384, 448]
    
    # If parent exists, prioritize its length
    if parent_norm is not None:
        p_len = len(parent_norm)
        if p_len not in lengths_to_try:
            lengths_to_try.insert(0, p_len)
        # Move parent's length to front
        lengths_to_try.remove(p_len)
        lengths_to_try.insert(0, p_len)
    
    # Deduplicate
    lengths_to_try = list(dict.fromkeys(lengths_to_try))
    
    time_per_length = p1_time / len(lengths_to_try)
    
    for n in lengths_to_try:
        if time.monotonic() > t_p1 + p1_time:
            break
        
        # Parametric search
        a_param, param_score = optimize_parametric(n, time_limit=min(30, time_per_length * 0.4), rng=rng)
        if a_param is not None:
            scan_results.append((param_score, n, 'param', a_param.copy()))
        
        # Try parent-based initialization
        if parent_norm is not None:
            if len(parent_norm) == n:
                a_parent = parent_norm.copy()
            else:
                a_parent = downsample_sequence(parent_norm, n) if len(parent_norm) > n else upsample_sequence(parent_norm, n)
            a_parent = project_simplex(a_parent)
            score_p, _ = eval_obj_normalized(a_parent)
            scan_results.append((score_p, n, 'parent', a_parent.copy()))
        
        # Try a few structured inits
        x = np.linspace(0, 1, n)
        t = 2 * x - 1
        
        inits = {}
        inits['uniform'] = np.ones(n) / n
        inits['raised_cos'] = project_simplex(0.5 - 0.5 * np.cos(np.pi * t))
        inits['bell2'] = project_simplex(np.maximum(0, 1 - t**2))
        inits['gauss'] = project_simplex(np.exp(-t**2 / 0.4))
        
        for name, a_init in inits.items():
            score_i, _ = eval_obj_normalized(a_init)
            scan_results.append((score_i, n, name, a_init.copy()))
        
        # Quick LP on the best initialization for this length
        best_for_n = min(
            [r for r in scan_results if r[1] == n],
            key=lambda x: x[0], default=None
        )
        if best_for_n is not None:
            t_alloc = min(40, time_per_length * 0.5)
            if time.monotonic() + t_alloc < t_p1 + p1_time:
                a_opt, obj_opt = lp_optimize_refined(
                    best_for_n[3], time_limit=t_alloc, seed=n, use_full_lp=(n <= 512))
                scan_results.append((obj_opt, n, f'lp_{best_for_n[2]}', a_opt.copy()))
        
        elapsed = time.monotonic() - t_p1
        print(f"  n={n}: best so far in scan = {min(r[0] for r in scan_results if r[1]==n):.6f} (t={elapsed:.1f}s)")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {scan_results[0][0]:.6f} (n={scan_results[0][1]}, {scan_results[0][2]})")
    
    # Update global best with best scan result
    if scan_results:
        best_scan_a = scan_results[0][3]
        update_best(best_scan_a, f"scan_{scan_results[0][2]}")
    
    # ========== Phase 2: Deep LP Optimization ==========
    print("=== Phase 2: Deep LP Optimization ===")
    t_p2 = time.monotonic()
    p2_time = 600.0
    
    # Select top candidates for deep optimization
    deep_candidates = []
    seen = set()
    for score, n, name, a in scan_results:
        key = (n, round(score, 4))
        if key not in seen and len(deep_candidates) < 6:
            deep_candidates.append((n, a.copy()))
            seen.add(key)
    
    # Ensure we have the absolute best
    if best_sequence is not None:
        deep_candidates.insert(0, (len(best_sequence), best_sequence.copy()))
    
    # Also add the parent if good
    if parent_norm is not None and (len(parent_norm), ) not in [(n, ) for n, _ in deep_candidates]:
        deep_candidates.append((len(parent_norm), parent_norm.copy()))
    
    time_per_deep = p2_time / max(1, len(deep_candidates))
    
    for idx, (n_cand, a_cand) in enumerate(deep_candidates):
        if time.monotonic() > t_p2 + p2_time:
            break
        
        t_alloc = min(time_per_deep, t_p2 + p2_time - time.monotonic() - 5)
        if t_alloc < 10:
            break
        
        print(f"  Deep LP candidate {idx+1}/{len(deep_candidates)}: n={n_cand}, alloc={t_alloc:.0f}s")
        
        # Run LP optimization with different configurations
        configs = [
            {'use_full_lp': n_cand <= 384, 'seed': idx * 100 + 1},
            {'use_full_lp': n_cand <= 512, 'seed': idx * 100 + 2},
        ]
        
        time_per_cfg = t_alloc / len(configs)
        for cfg_i, cfg in enumerate(configs):
            if time.monotonic() > t_p2 + p2_time:
                break
            t_cfg_alloc = min(time_per_cfg, t_p2 + p2_time - time.monotonic() - 5)
            if t_cfg_alloc < 5:
                break
            
            a_opt, obj_opt = lp_optimize_refined(
                a_cand, time_limit=t_cfg_alloc, **cfg)
            
            score = update_best(a_opt, f"lp_deep_{idx}_{cfg_i}")
            if a_opt is not None:
                a_cand = a_opt  # Chain: use result as next start
    
    print(f"Phase 2 done. Best: {best_score:.6f}")
    
    # ========== Phase 3: Smooth Gradient Refinement ==========
    print("=== Phase 3: Smooth Gradient Refinement (Nesterov) ===")
    t_p3 = time.monotonic()
    p3_time = 400.0
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        
        grad_configs = [
            {'beta_init': 300, 'lr_init': 0.03, 'seed': 1},
            {'beta_init': 600, 'lr_init': 0.02, 'seed': 2},
            {'beta_init': 1000, 'lr_init': 0.015, 'seed': 3},
            {'beta_init': 200, 'lr_init': 0.04, 'seed': 4},
            {'beta_init': 500, 'lr_init': 0.025, 'seed': 5},
        ]
        
        # Also try from slightly perturbed versions
        perturbed_starts = [best_sequence.copy()]
        for ps in [0.005, 0.01]:
            noise = rng.standard_normal(n_best) * ps
            perturbed_starts.append(project_simplex(best_sequence + noise))
        
        time_per_run = p3_time / (len(grad_configs) + 2)
        
        for i, cfg in enumerate(grad_configs):
            if time.monotonic() > t_p3 + p3_time:
                break
            t_alloc = min(time_per_run, t_p3 + p3_time - time.monotonic() - 5)
            if t_alloc < 10:
                break
            
            start = perturbed_starts[i % len(perturbed_starts)]
            a_opt, obj_opt = smooth_gradient_nesterov(
                start, time_limit=t_alloc, **cfg)
            update_best(a_opt, f"grad_{i}")
        
        # Extra gradient runs with different starting points
        for extra_i in range(3):
            if time.monotonic() > t_p3 + p3_time:
                break
            noise = rng.standard_normal(n_best) * 0.02
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(time_per_run * 0.7, t_p3 + p3_time - time.monotonic() - 5)
            if t_alloc < 8:
                break
            a_opt, obj_opt = smooth_gradient_nesterov(
                a_pert, time_limit=t_alloc, beta_init=800, lr_init=0.02,
                seed=100 + extra_i)
            update_best(a_opt, f"grad_pert_{extra_i}")
    
    print(f"Phase 3 done. Best: {best_score:.6f}")
    
    # ========== Phase 4: Coordinate Polish + Final Search ==========
    print("=== Phase 4: Coordinate Polish + Final Search ===")
    t_p4 = time.monotonic()
    p4_time = deadline - time.monotonic()
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        
        # Coordinate polish
        if p4_time > 60:
            t_alloc = min(p4_time * 0.3, 120.0)
            print(f"  Coordinate polish: {t_alloc:.0f}s")
            a_opt, obj_opt = coordinate_polish(
                best_sequence, time_limit=t_alloc, seed=42)
            update_best(a_opt, "coord")
        
        # More LP runs with very small steps (polishing)
        remaining = deadline - time.monotonic() - 30
        if remaining > 60:
            t_alloc = min(remaining * 0.4, 200.0)
            print(f"  Final LP polish: {t_alloc:.0f}s")
            a_opt, obj_opt = lp_optimize_refined(
                best_sequence, time_limit=t_alloc, seed=999,
                use_full_lp=(n_best <= 512))
            update_best(a_opt, "lp_polish")
        
        # Tiny random perturbations
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            print("  Random perturbation search...")
            n_tries = min(int(remaining / 2), 50)
            for i in range(n_tries):
                if time.monotonic() > deadline - 25:
                    break
                for scale in [0.001, 0.002, 0.005, 0.01]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_pert.copy()
                        print(f"  Perturb improvement [{i}] s={scale}: {best_score:.6f}")
                        break
        
        # Try length variation: downsample/upsample best and re-optimize
        remaining = deadline - time.monotonic() - 30
        if remaining > 60:
            for target_n in [int(n_best * 0.8), int(n_best * 1.25)]:
                if remaining < 30:
                    break
                if target_n < 64 or target_n > 1024:
                    continue
                if target_n == n_best:
                    continue
                
                a_resized = upsample_sequence(best_sequence, target_n) if target_n > n_best else downsample_sequence(best_sequence, target_n)
                a_resized = project_simplex(a_resized)
                
                t_alloc = min(30, remaining - 10)
                if t_alloc > 5:
                    a_opt, obj_opt = lp_optimize_refined(
                        a_resized, time_limit=t_alloc, seed=target_n,
                        use_full_lp=(target_n <= 512))
                    update_best(a_opt, f"resized_{target_n}")
                remaining = deadline - time.monotonic() - 30
    
    # Final verification
    if best_sequence is None:
        # Fallback
        n_fb = 256
        x = np.linspace(0, 1, n_fb)
        t = 2 * x - 1
        best_sequence = project_simplex(np.maximum(0, 1 - t**2))
    
    # Ensure all values are in valid range
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    
    # Final score check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        # Emergency fallback
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    best_score = final_score
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
