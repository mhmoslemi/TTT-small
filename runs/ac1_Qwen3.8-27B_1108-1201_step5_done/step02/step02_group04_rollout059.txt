"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Spectral flattening (Gerchberg-Saxton style) in the frequency domain
2. LSE (log-sum-exp) based gradient descent with adaptive temperature
3. LP-guided alternating optimization
4. Multi-length search (n=512, 1024)
The key innovation is the spectral flattening approach which directly targets
flat power spectrum, and the LSE gradient which provides a smooth approximation
to the max-convolution objective.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_autocorr(p):
    """Linear auto-convolution of p with itself using FFT."""
    n = len(p)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    result = np.fft.irfft(P * P, fft_size)
    return result[:size]

def compute_conv_gradient(p, weights):
    """Compute sum_k weights[k] * d(p*p)_k/dp_j for each j.
    Result: grad[j] = 2 * sum_k weights[k] * p[k-j] (0 if k-j out of range)
    """
    n = len(p)
    # conv of weights (length 2n-1) with reversed p (length n)
    # c_j = (weights * p_rev)_{n-1+j}
    fft_size = 1 << (2 * (2 * n - 1) - 2 - 1).bit_length()  # >= (2n-1) + n - 1 = 3n-2
    W = np.fft.rfft(weights, fft_size)
    Pr = np.fft.rfft(p[::-1], fft_size)
    conv = np.fft.irfft(W * Pr, fft_size)
    # c_j is at index n-1+j for j=0..n-1
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad

def spectral_flatten(p, time_limit, iterations=None):
    """GS-style spectral flattening. Returns (best_p, best_obj)."""
    n = len(p)
    p = np.maximum(p.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-12:
        p = np.ones(n) / n
    else:
        p = p / s
    
    conv = fast_autocorr(p)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Adaptive step size: start aggressive, become gentle
        frac = min(1.0, it / 500.0)
        t = 0.3 * (1 - frac) + 0.01 * frac
        
        P = np.fft.rfft(p, fft_size)
        alpha = np.abs(P)
        
        if len(alpha) > 1:
            mean_non_dc = alpha[1:].mean()
            new_alpha = alpha.copy()
            new_alpha[1:] = (1 - t) * alpha[1:] + t * mean_non_dc
            # Also slightly boost DC to maintain sum
            # new_alpha[0] stays the same
        else:
            new_alpha = alpha
        
        phase = np.angle(P)
        P_new = new_alpha * np.exp(1j * phase)
        p_new = np.fft.irfft(P_new, fft_size)[:n]
        
        p_new = np.maximum(p_new, 0.0)
        s_new = p_new.sum()
        if s_new < 1e-12:
            break
        p_new = p_new / s_new
        
        conv = fast_autocorr(p_new)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj:
            best_obj = obj
            best_p = p_new.copy()
        
        p = p_new
        it += 1
    
    return best_p, best_obj

    """LSE-based gradient descent on the probability simplex.
    Uses multiplicative updates (mirror descent) with adaptive temperature.
    Returns (best_p, best_obj).
    """
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-12:

def make_lp_indices(n):
    """Precompute sparse matrix structure for cross-convolution constraints."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)

def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    vals = np.tile(f, n)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def lp_optimize(f0, time_limit, timeout_lp=10):
    """LP-based alternating optimization. Returns (best_f, best_obj)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    
    conv = fast_autocorr(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = make_lp_indices(n)
    
    t0 = time.monotonic()
    t = 0.5
    patience = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        g = solve_lp(f, M, rows, cols, n, timeout=timeout_lp)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-14:
            break
        
        # Backtracking line search
        improved = False
        for t_try in np.linspace(t, 0.05 * t, 5):
            f_new = np.maximum((1.0 - t_try) * f + t_try * g, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new = f_new / s
            conv_new = fast_autocorr(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.8
            if patience > 30:
                break
        
        it += 1
        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}, obj={best_obj:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj

def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g
    
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    
    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr
    
    # Random binary-like
    rb = rng.random(n)
    rb[rb > 0.5] = 1.0
    rb[rb <= 0.5] = 0.0
    if rb.sum() > 0:
        inits['bin'] = rb / rb.sum()
    
    # Alternating
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alt'] = alt / alt.sum()
    
    # Multi-peak
    npeaks = 3
    centers = np.linspace(n * 0.1, n * 0.9, npeaks)
    mp = np.zeros(n)
    for c in centers:
        mp += np.exp(-0.5 * ((x - c) / (n * 0.08)) ** 2)
    inits['multi'] = mp / mp.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Spectral flattening (Gerchberg-Saxton style) targeting flat power spectrum
2. LSE (log-sum-exp) based gradient descent with FFT-computed gradients and adaptive temperature
3. LP-guided alternating optimization
4. Multi-length search (n=512, 1024) with diverse initializations
Key innovations: FFT-based gradient computation for LSE objective, spectral domain
flattening as a new optimization axis, and adaptive temperature scheduling.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_autocorr(p):
    """Linear auto-convolution of p with itself using FFT."""
    n = len(p)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    result = np.fft.irfft(P * P, fft_size)
    return result[:size]


def compute_conv_gradient(p, weights):
    """Compute grad[j] = 2 * sum_k weights[k] * p[k-j] (0 if k-j out of range).
    Uses FFT for O(n log n) computation."""
    n = len(p)
    N = 1 << (3 * n - 3).bit_length()
    W = np.fft.rfft(weights, N)
    B = np.fft.rfft(p, N)
    R = np.fft.irfft(W * np.conj(B), N)
    grad = 2.0 * R[:n]
    return grad


def spectral_flatten(p, time_limit):
    """GS-style spectral flattening: iteratively blend power spectrum toward flat,
    reconstruct time domain, enforce non-negativity and normalization.
    Returns (best_p, best_obj)."""
    n = len(p)
    p = np.maximum(p.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-12:
        p = np.ones(n) / n
    else:
        p = p / s

    conv = fast_autocorr(p)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()

    fft_size = 1 << (2 * n - 2).bit_length()

    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, it / 800.0)
        t = 0.25 * (1.0 - frac) + 0.005 * frac

        P = np.fft.rfft(p, fft_size)
        alpha = np.abs(P)

        if len(alpha) > 1:
            mean_non_dc = alpha[1:].mean()
            new_alpha = alpha.copy()
            new_alpha[1:] = (1.0 - t) * alpha[1:] + t * mean_non_dc
        else:
            new_alpha = alpha

        phase = np.angle(P)
        P_new = new_alpha * np.cos(phase) + 1j * new_alpha * np.sin(phase)
        p_new = np.fft.irfft(P_new, fft_size)[:n]

        p_new = np.maximum(p_new, 0.0)
        s_new = p_new.sum()
        if s_new < 1e-12:
            break
        p_new = p_new / s_new

        conv = fast_autocorr(p_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj:
            best_obj = obj
            best_p = p_new.copy()

        p = p_new
        it += 1

    return best_p, best_obj


def lse_gradient_descent(p0, time_limit, T0=5.0, T_min=0.05, eta0=0.15):
    """LSE-based mirror descent on the probability simplex.
    Minimizes smooth approximation T*log(sum exp(c_k/T)) of max_k c_k.
    Uses FFT-based gradient and multiplicative updates.
    Returns (best_p, best_obj)."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-12:
        p = np.ones(n) / n
    else:
        p = p / s

    conv = fast_autocorr(p)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()

    T = T0
    eta = eta0
    t0 = time.monotonic()
    patience = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv_shifted = conv - conv.max()
        w = np.exp(conv_shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w = w / w_sum

        grad = compute_conv_gradient(p, w)
        grad = grad - grad.mean()

        log_update = -eta * grad
        log_update = np.clip(log_update, -5.0, 5.0)
        p = p * np.exp(log_update)
        p = np.maximum(p, 1e-30)
        s = p.sum()
        if s < 1e-12:
            p = np.ones(n) / n
        else:
            p = p / s

        conv = fast_autocorr(p)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_p = p.copy()
            patience = 0
            eta = min(eta * 1.01, 0.5)
        else:
            patience += 1
            if patience >= 50:
                eta *= 0.6
                patience = 0
                if eta < 0.01:
                    T = max(T_min, T * 0.7)
                    eta = 0.08

        T = max(T_min, T * 0.9995)
        it += 1

    return best_p, best_obj


def make_lp_indices(n):
    """Precompute sparse structure for cross-conv constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)_k <= M for all k=0..2n-2."""
    vals = np.tile(f, n)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, timeout_lp=10):
    """LP-based alternating optimization with backtracking line search.
    Returns (best_f, best_obj)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    conv = fast_autocorr(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_indices(n)

    t0 = time.monotonic()
    t = 0.5
    patience = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        g = solve_lp(f, M, rows, cols, n, timeout=timeout_lp)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in np.linspace(t, 0.03 * t, 6):
            if t_try < 0.01:
                continue
            f_new = np.maximum((1.0 - t_try) * f + t_try * g, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new = f_new / s
            conv_new = fast_autocorr(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.75
            if patience > 40:
                break

        it += 1
        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}, obj={best_obj:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uni'] = np.ones(n) / n

    for sf in [0.1, 0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr

    rb = (rng.random(n) > 0.5).astype(np.float64)
    if rb.sum() > 1:
        inits['bin'] = rb / rb.sum()

    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alt'] = alt / alt.sum()

    npeaks = 3
    centers = np.linspace(n * 0.15, n * 0.85, npeaks)
    mp = np.zeros(n)
    for c in centers:
        mp += np.exp(-0.5 * ((x - c) / (n * 0.07)) ** 2)
    inits['multi'] = mp / mp.sum()

    return inits
