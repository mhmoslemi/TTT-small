def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad(f, beta):
    """Compute gradient of 2n * LSE(beta * conv(f,f)) / beta w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Soft-max weights
    shifted = conv - conv.max()
    exp_vals = np.exp(beta * shifted)
    p = exp_vals / exp_vals.sum()
    
    # Gradient: 2 * (p cross-correlate f)[j] for j = 0, ..., n-1
    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    cross = np.fft.irfft(np.conj(P) * F, N)
    
    return 2.0 * cross[:n]


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} using Fisher's algorithm."""
    n = len(v)
    if n == 0:
        return v
    v = np.array(v, dtype=np.float64)
    # If all non-negative and sum is right, return
    if (v >= 0).all():
        sv = v.sum()
        if abs(sv - s) < 1e-15:
            return v
        if sv > 1e-15:
            return v * (s / sv)
    # General projection
    # The projection onto {x >= 0, sum(x) = s} is:
    # x_i = max(v_i - tau, 0) where tau is chosen so sum = s
    # Use bisection on tau
    lo = v.min() - s  # if tau <= this, all x_i > 0
    hi = v.max()  # if tau >= this, all x_i = 0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.maximum(v - mid, 0.0)
        sx = x.sum()
        if abs(sx - s) < 1e-12:
            break
        if sx > s:
            lo = mid
        else:
            hi = mid
    
    return np.maximum(v - mid, 0.0)

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based smoothed gradient descent
with logsumexp approximation, Adam optimizer, beta annealing, spectral initialization,
and multi-coordinate local search. Explores multiple sequence lengths (256-4096)
with diverse initializations including random spectral methods.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_softmax_weights(conv, beta):
    """Compute softmax weights for the soft-max approximation of max."""
    shifted = conv - conv.max()
    exp_vals = np.exp(beta * shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.full(len(conv), 1.0 / len(conv))
    return exp_vals / total


def compute_gradient(f, beta):
    """Compute gradient of 2n * LSE(beta * conv(f,f)) / beta w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    p = compute_softmax_weights(conv, beta)
    
    # Gradient: 2 * (p cross-correlate f)[j] for j = 0, ..., n-1
    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    cross = np.fft.irfft(np.conj(P) * F, N)
    
    return 2.0 * cross[:n]


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} using bisection on the threshold."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    if (v >= -1e-15).all():
        sv = v.sum()
        if sv > 1e-15:
            v = np.maximum(v, 0.0)
            return v * (s / v.sum())
        return np.full(n, s / n)
    
    lo = v.min() - s
    hi = v.max()
    mid = 0.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.maximum(v - mid, 0.0)
        sx = x.sum()
        if abs(sx - s) < 1e-13 * max(1.0, s):
            break
        if sx > s:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - mid, 0.0)


def adam_gd(f0, time_limit, beta_start=5.0, beta_end=500.0,
            lr_start=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with projected gradient and beta annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f = f / s
    
    n = len(f)
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    t0 = time.monotonic()
    m = np.zeros_like(f)  # first moment
    v = np.zeros_like(f)  # second moment
    t = 0
    lr = lr_start
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient
        grad = compute_gradient(f, beta)
        
        # Normalize gradient magnitude for stable updates
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new, 1.0)
        
        obj_new = 2.0 * n * fast_convolve(f_new).max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
        
        # Adaptive learning rate
        if obj_new > best_obj * 1.01 and lr > 1e-5:
            lr *= 0.9
        elif obj_new < best_obj * 0.9995 and lr < 0.1:
            lr *= 1.01
        
        f = f_new
    
    return best_f, best_obj


def momentum_gd(f0, time_limit, beta_start=5.0, beta_end=300.0,
                lr_start=0.05, mu=0.85):
    """Heavy-ball momentum gradient descent with projection."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f = f / s
    
    n = len(f)
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    t0 = time.monotonic()
    vel = np.zeros_like(f)
    lr = lr_start
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad = compute_gradient(f, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        vel = mu * vel + (1.0 - mu) * grad
        f_new = f - lr * vel
        f_new = project_simplex(f_new, 1.0)
        
        obj_new = 2.0 * n * fast_convolve(f_new).max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
        
        if obj_new > best_obj * 1.01 and lr > 1e-5:
            lr *= 0.88
        elif obj_new < best_obj * 0.999 and lr < 0.15:
            lr *= 1.015
        
        f = f_new
    
    return best_f, best_obj

def lp_kick(f, time_limit=5.0):
    """Quick LP-based kick to escape local minima."""
    import scipy.optimize
    n = len(f)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-12:
        return f.copy()
    f = f / s
    
    conv = fast_convolve(f)
    M = conv.max()
    
    try:
        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            end = min(2 * n - 1, j + n)
            A[j:end, j] = f[:end - j]
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, 
                                      method='highs', options={'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-12:
                g = g / gs
                # Line search
                best_t = 0.0
                best_obj = 2.0 * n * fast_convolve(f).max()
                for t in np.linspace(0, 1, 20):
                    a_trial = (1 - t) * f + t * g
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = 2.0 * n * fast_convolve(a_trial).max()
                    if o < best_obj:
                        best_obj = o
                        best_t = t
                if best_t > 0:
                    a_new = (1 - best_t) * f + best_t * g
                    a_new = np.maximum(a_new, 0.0)
                    a_new /= a_new.sum()
                    return a_new
    except Exception:
        pass
    return f.copy()


def spectral_init(n, rng):
    """Generate non-negative sequence with approximately flat auto-correlation."""
    N = 1 << (2 * n).bit_length()
    freqs = np.arange(N // 2 + 1)
    power = rng.uniform(0.3, 1.0, len(freqs))
    power[0] = 0.0
    phases = rng.uniform(0, 2 * np.pi, len(freqs))
    F = np.sqrt(power) * np.exp(1j * phases)

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s

        # Adam refinement
        remaining = deadline - time.monotonic() - 60.0
        ad_time = min(t_alloc * 0.3, max(3.0, remaining * 0.25))
        a_adam, _ = adam_gd(a_opt, time_limit=ad_time,
                            beta_start=50.0, beta_end=800.0,
                            lr_start=0.01)
        
        # Local search
        remaining = deadline - time.monotonic() - 60.0
        ls_time = min(15.0, max(2.0, remaining * 0.15))
        a_ls, _ = local_search_informed(a_adam, time_limit=ls_time)
        
        for a_trial in [a_opt, a_adam, a_ls]:
            score = float(evaluate_sequence(a_trial.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_trial.copy()
                print(f"  P2: n={n}, {name}, score={score:.6f}")
    
    # Phase 3: Intensive polish on best (remaining time)
    print("Phase 3: Intensive polish...")
    if best_sequence is not None:
        polish_deadline = deadline - 20.0
        
        while time.monotonic() < polish_deadline:
            remaining = polish_deadline - time.monotonic()
            if remaining < 15:
                break
            
            cycle = 0
            while time.monotonic() < polish_deadline:
                remaining = polish_deadline - time.monotonic()
                if remaining < 10:
                    break
                
                cycle += 1
                
                # Alternate between optimization methods
                if cycle % 3 == 1:
                    # Adam with high beta
                    t_gd = min(remaining * 0.3, 45.0)
                    if t_gd > 3:
                        a_new, _ = adam_gd(best_sequence, time_limit=t_gd,
                                            beta_start=100.0, beta_end=2000.0,
                                            lr_start=0.005)
                        score = float(evaluate_sequence(a_new.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_new.copy()
                            print(f"  P3 Adam: score={best_score:.6f}")
                
                elif cycle % 3 == 2:
                    # LP kick
                    t_lp = min(remaining * 0.2, 20.0)
                    if t_lp > 2:
                        a_new = lp_kick(best_sequence, time_limit=t_lp)
                        # Follow up with short GD
                        a_new, _ = adam_gd(a_new, time_limit=min(10.0, t_lp),
                                            beta_start=100.0, beta_end=500.0,
                                            lr_start=0.005)
                        score = float(evaluate_sequence(a_new.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_new.copy()
                            print(f"  P3 LP+Adam: score={best_score:.6f}")
                
                else:
                    # Local search
                    t_ls = min(remaining * 0.2, 30.0)
                    if t_ls > 2:
                        a_new, _ = local_search_informed(best_sequence, time_limit=t_ls)
                        score = float(evaluate_sequence(a_new.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_new.copy()
                            print(f"  P3 LS: score={best_score:.6f}")
    
    # Final: try a few more sequence lengths if time permits
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(lengths) < 7:
        for n in [8192]:
            if remaining > 60:
                print(f"  Trying n={n} with parent init...")
                inits = make_initializations(n, rng)
                if 'parent' in inits:
                    t_try = min(50.0, remaining - 30.0)
                    if t_try > 10:
                        a_try, _ = adam_gd(inits['parent'], time_limit=t_try,
                                            beta_start=5.0, beta_end=100.0,
                                            lr_start=0.02)
                        score = float(evaluate_sequence(a_try.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_try.copy()
                            print(f"  n={n}: score={score:.6f}")
                        remaining = deadline - time.monotonic()
    
    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(512) / 512.0
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_sequence]
    final_score = float(evaluate_sequence(final_seq))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(final_seq)}")
    return final_seq
