"""
Multi-strategy optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Uses peak-directed gradient updates,
multi-start initialization, and simulated annealing polishing.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_pressure(a, peak_idx):
    """Compute the pressure vector: contribution of each a[j] to conv[peak_idx]."""
    n = len(a)
    p = np.zeros(n)
    for j in range(n):
        k = peak_idx - j
        if 0 <= k < n:
            p[j] = a[k]
    return p

```python
def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.where(sorted_v * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.clip(v - theta, 0, None)
```

Phase 1: Peak-directed iterative optimization with FFT-accelerated pressure computation.

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

import numpy as np

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def compute_pressure(a, weights):
    """
    Compute pressure vector: pressure[j] = sum_k weights[k] * a[k-j].
    This represents the weighted contribution of a[j] to convolution peaks.
    """
    n = len(a)
    # pressure[j] = conv(a, weights_rev)[2n-2-j]
    conv_full = fft_convolve(a, weights[::-1])
    return conv_full[n - 1:2 * n - 1][::-1]

        if time.monotonic() > deadline - 100:

        time_alloc_total = max(20.0, frac / total_frac * available_time)

        # Generate inits for this length
        inits = []

        if provided_seq is not None and n == n_prov:
            inits.append(provided_seq.copy())
        elif provided_seq is not None:
            # Interpolate provided to this length
            x_old = np.linspace(0, 1, len(provided_seq))
            x_new = np.linspace(0, 1, n)
            inits.append(np.clip(np.interp(x_new, x_old, provided_seq), 0.05, None))

        inits.append(make_chirp_init(n, 0.4 + 0.2 * rng.random(), rng))
        inits.append(make_chirp_init(n, 0.6 + 0.2 * rng.random(), rng))
        inits.append(make_multitone_init(n, rng))
        inits.append(make_random_init(n, rng, 0.3 + 0.2 * rng.random()))
        inits.append(make_ramp_init(n, rng))

        # Score all inits, keep top 3
        scored_inits = []
        for a in inits:
            s = local_score(a)
            scored_inits.append((s, a))
        scored_inits = sort_by_score(scored_inits)
        top_inits = [a for _, a in scored_inits[:3]]

        # Allocate time among top inits
        time_per_init = time_alloc_total / len(top_inits)

        for idx, a0 in enumerate(top_inits):
            if time.monotonic() > deadline - 100:
                break

            t_alloc = min(time_per_init, deadline - time.monotonic() - 80)
            if t_alloc < 10:
                break

            init_score = local_score(a0)
            print(f"\n  n={n}, init_{idx}: score={init_score:.6f}, alloc={t_alloc:.0f}s")

            a_best, score = peak_directed_optimize(a0, t_alloc, rng, verbose=False)
            print(f"  PD result: score={score:.6f} (init was {init_score:.6f})")

            if score < best_score:
                best_score = score
                best_seq = a_best.copy()
                print(f"  *** New best: {best_score:.6f} (n={n}, init={idx}) ***")

    # Phase 2: SA polish
    print(f"\n{'=' * 60}")
    print(f"Phase 2: SA Polish")
    print(f"{'=' * 60}")

    if best_seq is not None and time.monotonic() < deadline - 40:
        sa_time = min(deadline - time.monotonic() - 20, 400.0)
        if sa_time > 20:
            print(f"  SA polish: n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, sa_score = sa_polish(best_seq, sa_time, rng, verbose=True)
            print(f"  SA result: {sa_score:.6f} (was {best_score:.6f})")
            if sa_score < best_score:
                best_score = sa_score
                best_seq = a_sa.copy()
                print(f"  *** New best after SA: {best_score:.6f} ***")

    # Phase 3: Final peak-directed refinement
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Final Peak-Directed Refinement")
    print(f"{'=' * 60}")

    if best_seq is not None and time.monotonic() < deadline - 20:
        final_time = min(deadline - time.monotonic() - 5, 200.0)
        if final_time > 15:
            print(f"  Final PD: n={len(best_seq)}, time={final_time:.0f}s")
            # Use a fresh rng seed for diversity
            rng_final = np.random.default_rng(seed + 999)
            a_final, final_score = peak_directed_optimize(best_seq, final_time, rng_final, verbose=False)
            print(f"  Final PD result: {final_score:.6f} (was {best_score:.6f})")
            if final_score < best_score:
                best_score = final_score
                best_seq = a_final.copy()
                print(f"  *** Final best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
