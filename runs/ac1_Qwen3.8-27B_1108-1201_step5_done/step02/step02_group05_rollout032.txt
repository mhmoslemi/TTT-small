def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    c = fast_convolve(a)

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def power_sum_val(c, p):
    """Compute (sum c_k^p)^(1/p) for c >= 0."""
    c_safe = np.maximum(c, 1e-30)
    return np.sum(c_safe ** p) ** (1.0 / p)


def power_sum_grad(c, a, p):
    """
    Compute (sum c_k^p)^(1/p) and its gradient w.r.t. a,
    where c = conv(a,a).
    """
    n = len(a)
    c_safe = np.maximum(c, 1e-30)
    c_p = c_safe ** p
    S = np.sum(c_p)
    F = S ** (1.0 / p)
    
    if S < 1e-30:
        return F, np.zeros(n)
    
    factor = S ** (1.0 / p - 1.0)
    w = c_safe ** (p - 1)
    
    # cross_corr[m] = sum_k w[k] * a[k-m] for valid indices
    # Using np.correlate: cc[i] = sum_k w[k] * a[k - i + (n-1)]
    # So cross_corr[m] = cc[n-1+m]
    cc = np.correlate(w, a, mode='full')
    cross = cc[n-1:n-1+n]
    
    # Correction: for k=2m, dc_k/da_m = a[m] not 2*a[m]
    # So we need to subtract w[2m]*a[m] for each m
    diag = w[2 * np.arange(n)] * a
    
    grad = factor * (2.0 * cross - diag)
    return F, grad

def make_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n)
    
    # Gaussian bumps
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g
    
    # Two bumps
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['two_bumps'] = g2
    
    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    inits['hann'] = np.maximum(hann, 0.0)
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5) * n
    
    # Parent
    if parent is not None:

    # Flat with random perturbation

    base += rng.random(n) * 0.5
    inits['flat_noise'] = base
    
    # Cosine window

    return inits

        a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for T in T_values:
        if time.monotonic() - t0 >= time_limit:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        max_iter = max(20, min(400, int(remaining * 3)))
        
        bounds = [(0.0, None)] * n
        try:
            res = scipy.optimize.minimize(
                _obj_func, a,
                args=(T, penalty),
                jac=_grad_func,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                          'maxls': 50}
            )
            a_new = np.maximum(res.x, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-12:
                a_new = a_new / s_new
                obj_new = 2.0 * n * fast_convolve(a_new).max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            break
    
    return best_a, best_obj


def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for j in range(n):
        end = min(num_constraints, j + n)
        A[j:end, j] = f[:end - j]
    c_vec = -np.ones(n)
    b = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c_vec, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit=60.0):
    """Iterative LP-based optimization."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 80
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp(f, M)
        
        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
        else:
            # Try a few step sizes
            best_t_obj = best_obj
            best_t = 0.0
            for t in [0.3, 0.5, 0.7, 1.0]:
                a_trial = (1.0 - t) * f + t * g
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                obj_trial = 2.0 * n * fast_convolve(a_trial).max()
                if obj_trial < best_t_obj:
                    best_t_obj = obj_trial
                    best_t = t
            
            if best_t > 0 and best_t_obj < best_obj - 1e-10:
                f = (1.0 - best_t) * f + best_t * g
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs > 1e-12:
                    f /= fs
                best_obj = best_t_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= max_no_improve:
            break
    
    return best_f, best_obj


def local_search(a, time_limit=30.0):
    """Aggressive local search: redistribute mass to reduce peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        return a, float('inf')
    a /= s
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 80
    
    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        c = fast_convolve(a)
        # Look at top 3 peaks
        top_indices = np.argsort(c)[-3:]
        
        found = False
        for k_star in top_indices:
            if found or time.monotonic() - t0 > time_limit:
                break
            
            # Compute contributions
            contrib = np.zeros(n)
            for i in range(n):
                j = int(k_star) - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
            
            order = np.argsort(contrib)
            bottom = order[:min(20, n // 2)]
            top = order[-min(20, n // 2):]
            
            for di in top:
                if found:
                    break
                if a[di] < 1e-10:
                    continue
                for si in bottom:
                    if found or di == si:
                        continue
                    for frac in [0.05, 0.1, 0.2, 0.35, 0.5]:
                        amount = a[di] * frac
                        if amount < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[di] = max(0.0, a_trial[di] - amount)
                        a_trial[si] += amount
                        ts = a_trial.sum()
                        if ts < 1e-12:
                            continue
                        a_trial /= ts
                        c_trial = fast_convolve(a_trial)
                        obj_trial = 2.0 * n * c_trial.max()
                        if obj_trial < best_obj - 1e-12:
                            best_obj = obj_trial
                            a = a_trial.copy()
                            found = True
                            break
        
        # Also try two-way transfers (reduce two, increase one)
        if not found:
            c = fast_convolve(a)
            k_star = int(np.argmax(c))
            contrib = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
            top2 = np.argsort(contrib)[-5:]
            bottom5 = np.argsort(contrib)[:5]
            for di in top2:
                if a[di] < 1e-10:
                    continue
                for si in bottom5:
                    if di == si:
                        continue
                    for frac in [0.1, 0.2, 0.3]:
                        amount = a[di] * frac
                        if amount < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[di] = max(0.0, a_trial[di] - amount)
                        a_trial[si] += amount
                        ts = a_trial.sum()
                        if ts < 1e-12:
                            continue
                        a_trial /= ts
                        c_trial = fast_convolve(a_trial)
                        obj_trial = 2.0 * n * c_trial.max()
                        if obj_trial < best_obj - 1e-12:
                            best_obj = obj_trial
                            a = a_trial.copy()
                            found = True
                            break
                if found:
                    break
        
        if not found:
            stuck += 1
        else:
            stuck = 0
    
    return a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_inits(n, rng, parent_seq):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n)
    
    # Various bump profiles
    for center, width in [(0.5, 0.2), (0.5, 0.35), (0.5, 0.5), (0.3, 0.15), (0.7, 0.15)]:
        g = np.exp(-0.5 * ((x / n - center) / width) ** 2)
        inits[f'bump_{int(center*100)}_{int(width*100)}'] = g + 1e-6
    
    # Two separated bumps
    g2 = np.exp(-0.5 * ((x / n - 0.2) / 0.08) ** 2) + np.exp(-0.5 * ((x / n - 0.8) / 0.08) ** 2)
    inits['two_bumps'] = g2 + 1e-6
    
    # Cosine tapers (edges high)
    cos_taper = 0.5 * (1 - np.cos(np.pi * x / n))
    inits['cos_taper'] = cos_taper + 0.1
    
    # Inverse (center low, edges high)
    inv_g = np.exp(0.5 * ((x / n - 0.5) / 0.3) ** 2)
    inits['inv_gauss'] = inv_g
    
    # Random smooth
    r = rng.random(n)
    k = max(2, n // 8)
    kernel = np.ones(k) / k
    inits['random_smooth'] = np.convolve(r, kernel, mode='same') + 0.1
    
    # Pure random
    inits['random'] = rng.random(n) + 0.01
    
    # Parent
    if parent_seq is not None:
        inits['parent'] = parent_seq
    
    # Parent with noise
    if parent_seq is not None:
        noisy = parent_seq * (1.0 + 0.2 * rng.standard_normal(n))
        inits['parent_noise'] = np.maximum(noisy, 0.0) + 1e-6
    
    return {k: v / v.sum() for k, v in inits.items()}


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses soft-max smoothed optimization with L-BFGS-B,
    temperature annealing, LP direction finding, random restarts, and
    local search to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s))
    deadline = time.monotonic() + (total_budget - 25.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Try to get parent
    parent_full = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_full = p / p.sum()
    except Exception:
        pass
    
    lengths = [256, 512, 1024]
    
    # === Phase 1: Broad scan (30% of budget) ===
    phase1_frac = 0.30
    phase1_time = (total_budget - 25.0) * phase1_frac
    print(f"Phase 1: Broad scan ({phase1_time:.0f}s)")
    
    all_configs = []
    for n in lengths:
        p_n = get_parent(n)
        inits = make_inits(n, rng, p_n)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))
    
    # Prioritize: parent first, then n=512, then others
    def config_key(item):
        n, name, _ = item
        if name == 'parent':
            return (0, n)
        if n == 512 and name in ['uniform', 'random_smooth']:
            return (1, n)
        return (2, n)
    
    all_configs.sort(key=config_key)
    
    time_per_scan = max(5.0, phase1_time / len(all_configs))
    T_values_scan = [0.05, 0.02, 0.01, 0.005]
    scan_results = []
    
    for n, name, a0 in all_configs:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(time_per_scan, 45.0, remaining - 30.0)
        if t_alloc < 4.0:
            break
        
        a_opt, _ = run_lBFGS(a0, T_values_scan, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P1: n={n}, {name}, score={score:.6f}")
    
    print(f"  Phase 1 done. Best: {best_score:.6f}")
    
    # === Phase 2: Deep optimization (35% of budget) ===
    phase2_time = (total_budget - 25.0) * 0.35
    print(f"Phase 2: Deep optimization ({phase2_time:.0f}s)")
    
    scan_results.sort(key=lambda x: x[0])
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 6:
            top_configs.append((n, name, a_opt))
            seen.add(key)
    
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))
    
    T_values_deep = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    time_per_deep = max(20.0, phase2_time / max(1, len(top_configs)))
    
    for n, name, a0 in top_configs:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(time_per_deep, remaining - 60.0)
        if t_alloc < 10.0:
            break
        
        # L-BFGS-B deep
        a_opt, _ = run_lBFGS(a0, T_values_deep, time_limit=t_alloc * 0.45)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P2 LBFGS: n={n}, {name}, score={score:.6f}")
        
        # LP on the result
        remaining = deadline - time.monotonic() - 30
        if remaining > 15:
            a_lp, _ = lp_optimize(a_opt, time_limit=min(60.0, remaining * 0.4))
            score = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_lp.copy()
                print(f"  P2 LP: n={n}, {name}, score={score:.6f}")
        
        # Local search
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            a_ls, _ = local_search(best_sequence if best_score < 999 else a_opt,
                                    time_limit=min(30.0, remaining * 0.5))
            score = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_ls.copy()
                print(f"  P2 LS: n={n}, {name}, score={score:.6f}")
    
    # === Phase 3: Intensive polish + random restarts (35% of budget) ===
    print(f"Phase 3: Polish + restarts ({(total_budget-25.0)*0.35:.0f}s), best={best_score:.6f}")
    
    polish_count = 0
    while time.monotonic() < deadline - 15.0:
        remaining = deadline - time.monotonic() - 15.0
        if remaining < 15.0:
            break
        
        polish_count += 1
        
        # Alternate between: L-BFGS-B, LP, random restart, local search
        strategy = polish_count % 4
        
        if strategy == 0:
            # L-BFGS-B polish
            t_alloc = min(90.0, remaining * 0.4)
            if t_alloc < 8:
                break
            a_opt, _ = run_lBFGS(best_sequence, [0.01, 0.005, 0.002, 0.001],
                                  time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 LBFGS#{polish_count}: score={score:.6f}")
        
        elif strategy == 1:
            # LP polish
            t_alloc = min(60.0, remaining * 0.4)
            if t_alloc < 8:
                break
            a_lp, _ = lp_optimize(best_sequence, time_limit=t_alloc)
            score = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_lp.copy()
                print(f"  P3 LP#{polish_count}: score={score:.6f}")
        
        elif strategy == 2:
            # Random restart: perturb best and re-optimize
            t_alloc = min(60.0, remaining * 0.4)
            if t_alloc < 10:
                break
            n = len(best_sequence)
            noise_scale = rng.choice([0.05, 0.1, 0.15, 0.2])
            perturbed = best_sequence * (1.0 + noise_scale * rng.standard_normal(n))
            perturbed = np.maximum(perturbed, 0.0)
            ps = perturbed.sum()
            if ps > 1e-10:
                perturbed /= ps
                a_r, _ = run_lBFGS(perturbed, [0.05, 0.02, 0.01, 0.005],
                                    time_limit=t_alloc)
                score = float(evaluate_sequence(a_r.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_r.copy()
                    print(f"  P3 Restart#{polish_count}: score={score:.6f}")
        
        else:
            # Local search
            t_alloc = min(45.0, remaining * 0.4)
            if t_alloc < 5:
                break
            a_ls, _ = local_search(best_sequence, time_limit=t_alloc)
            score = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_ls.copy()
                print(f"  P3 LS#{polish_count}: score={score:.6f}")
    
    # Final check
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Make sure we return a valid sequence
    final_seq = [max(0.0, float(x)) for x in best_sequence]
    final_score = float(evaluate_sequence(final_seq))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(final_seq)}")
    return final_seq
