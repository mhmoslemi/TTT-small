import scipy.signal
import time

def fast_auto_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Compute f_tau(a) = 2n/tau * log(sum exp(tau * conv(a,a)))"""
    n = len(a)
    conv = fast_auto_convolve(a)
    # Numerical stability
    conv_shifted = conv - conv.max()
    log_sum = np.log(np.sum(np.exp(tau * conv_shifted))) + tau * conv.max()
    return 2.0 * n / tau * log_sum

    """Compute gradient of f_tau(a)."""
    n = len(a)
    conv = fast_auto_convolve(a)
    conv_shifted = conv - conv.max()
    p = np.exp(tau * conv_shifted)
    p /= p.sum()

    # This is correlate(p, a, 'valid')
    grad = 2.0 * n * scipy.signal.correlate(p, a, mode='valid')
    return grad

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    z_prev = a.copy()
    t = 1.0
    alpha = 1.0  # initial step size
    best_obj = smooth_objective(a, tau)
    best_a = a.copy()

        if time.monotonic() - t0 > time_limit:

        grad_y = smooth_gradient(y, tau)

        # Update

    n = len(a)

    M = conv.max()

    weights = np.maximum(0, 1 - (M - conv) / (eps * M + 1e-15))

    grad = scipy.signal.correlate(weights, a, mode='valid')

import numpy as np
import scipy.signal
import time


def fast_auto_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Handle case where v is already on the simplex
    if v.sum() <= 1 + 1e-10 and np.min(v) >= -1e-10:
        v = np.maximum(v, 0)
        s = v.sum()
        if s > 1e-15:
            return v / s
        return np.ones(n) / n

    cond = u * np.arange(1, n + 1) > (cssv - 1)

    w = np.maximum(v - theta, 0)

    if s < 1e-15:

    return w / s

import time

    fft_size = 1 << (size - 1).bit_length()

        return v

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n

    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:

    p = exp_shifted / exp_shifted.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth (log-sum-exp) approximation of the peak with FISTA optimization
2. Tau-continuation for increasing accuracy

import numpy as np
import scipy.signal
import time


def fast_auto_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def true_obj(a):
    """True objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * fast_auto_convolve(a).max() / (s * s)


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(cond)))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def compute_grad(a, tau):
    """Gradient of (2n/tau)*logsumexp(tau*conv(a,a))."""
    n = len(a)
    conv = fast_auto_convolve(a)
    conv_max = conv.max()
    shifted = tau * (conv - conv_max)
    e = np.exp(shifted)
    p = e / e.sum()
    grad = 2.0 * n * scipy.signal.correlate(p, a, mode='valid')
    return grad


def fista_smooth(a0, tau, time_limit=30.0):
    """FISTA with adaptive step size for smooth objective."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    y = a.copy()
    z_prev = a.copy()
    t = 1.0
    alpha = 1.0 / (4.0 * n)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_auto_convolve(a).max()
    
    t0 = time.monotonic()
    
    for k in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad_y = compute_grad(y, tau)
        z = project_simplex(y - alpha * grad_y)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = z + (t - 1.0) / t_new * (z - z_prev)
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys > 1e-15:
            y_new /= ys
        else:
            y_new = z.copy()
        
        obj_z = 2.0 * n * fast_auto_convolve(z).max()
        if obj_z < best_obj:
            best_obj = obj_z
            best_a = z.copy()
        
        if k > 0 and k % 5000 == 0:
            alpha *= 0.97
        
        z_prev = z.copy()
        y = y_new
        t = t_new
    
    return best_a, best_obj


def subgradient_refine(a0, time_limit=30.0):
    """Subgradient method with adaptive smoothing."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_auto_convolve(a).max()
    
    t0 = time.monotonic()
    alpha = 1.0 / (2.0 * n)
    eps = 0.1  # smoothing width
    
    a_prev = a.copy()
    beta = 0.5  # momentum
    
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_auto_convolve(a)
        M = conv.max()
        # Softmax-like weights on the peak
        scores = (M - conv) / (eps * M + 1e-15)
        weights = np.exp(-np.maximum(0, scores))
        total_w = weights.sum()
        if total_w < 1e-15:
            break
        weights /= total_w
        
        grad = 2.0 * n * scipy.signal.correlate(weights, a, mode='valid')
        
        a_new = a - alpha * grad + beta * (a - a_prev)
        a_new = project_simplex(a_new)
        
        obj_new = 2.0 * n * fast_auto_convolve(a_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            eps = max(0.001, eps * 0.999)
        else:
            alpha *= 0.95
            eps = min(0.5, eps * 1.1)
        
        a_prev = a
        a = a_new
    
    return best_a, best_obj


def coordinate_descent_refine(a0, time_limit=30.0):
    """Coordinate descent: try moving mass from one coordinate to another."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_auto_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 500:
        conv = fast_auto_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Find coordinates that contribute to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        # Pick a "dangerous" coordinate (high contribution) and a "safe" one
        danger_thresh = 0.5 * max_c
        safe_thresh = 0.1 * max_c
        
        danger = np.where(contrib > danger_thresh)[0]
        safe = np.where(contrib < safe_thresh)[0]
        
        if len(danger) == 0 or len(safe) == 0:
            no_improve += 1
            continue
        
        improved = False
        # Try a few random pairs
        for _ in range(50):
            di = danger[rng.integers(len(danger))]
            si = safe[rng.integers(len(safe))]
            if di == si:
                continue
            
            for frac in [0.1, 0.2, 0.3, 0.5]:
                amt = a[di] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[di] -= amt
                a_trial[si] += amt
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-15:
                    continue
                a_trial /= ts
                o = 2.0 * n * fast_auto_convolve(a_trial).max()
                if o < best_obj - 1e-12:
                    best_obj = o
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
    
    return best_a, best_obj

def make_initializations(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine window
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    inits['raised_cos'] = np.maximum(rc, 0) / np.maximum(rc, 0).sum()
    
    # Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    # (Same as raised cosine, so let's use a different one)
    # Tukey window
    r = 0.3  # ramp width
    tukey = np.ones(n)
    m = int(r * n / 2)
    if m > 0:
        for i in range(m):
            tukey[i] = 0.5 * (1 + np.cos(np.pi * (i / m - 1)))
            tukey[n - 1 - i] = tukey[i]
    inits['tukey'] = np.maximum(tukey, 0.01) / np.maximum(tukey, 0.01).sum()
    
    # Gaussian
    for sf in [0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random'] = r
    
    # Windowed random
    for wname in ['raised_cos']:
        w = inits[wname]
        r2 = rng.random(n)
        wr = w * r2
        inits[f'{wname}_rand'] = wr / wr.sum()
    
    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
    
    # Perturbed parent
    if parent is not None:
        for noise_level in [0.01, 0.05, 0.1]:
            p_noisy = parent + noise_level * rng.standard_normal(n)
            p_noisy = np.maximum(p_noisy, 0)
            ps = p_noisy.sum()
            if ps > 1e-10:
                inits[f'parent_noise_{noise_level}'] = p_noisy / ps
    
    return inits

import numpy as np
import scipy.signal
import time


def fast_auto_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def true_peak(a):
    """2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_auto_convolve(a).max()


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(cond)))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def compute_smooth_grad(a, tau):
    """Gradient of (2n/tau)*logsumexp(tau*conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = fast_auto_convolve(a)
    conv_max = conv.max()
    shifted = tau * (conv - conv_max)
    e = np.exp(shifted)
    p = e / e.sum()
    grad = 2.0 * n * scipy.signal.correlate(p, a, mode='valid')
    return grad


def fista_optimize(a0, tau, time_limit=30.0):
    """FISTA optimization of smooth approximation with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    y = a.copy()
    z_prev = a.copy()
    t = 1.0
    alpha = 0.5 / n

    best_a = a.copy()
    best_obj = true_peak(a)

    t0 = time.monotonic()

    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        grad_y = compute_smooth_grad(y, tau)
        z = project_simplex(y - alpha * grad_y)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = z + (t - 1.0) / t_new * (z - z_prev)
        y_new = project_simplex(y_new)

        obj_z = true_peak(z)
        if obj_z < best_obj:
            best_obj = obj_z
            best_a = z.copy()

        if k > 0 and k % 3000 == 0:
            alpha *= 0.95

        z_prev = z.copy()
        y = y_new
        t = t_new

    return best_a, best_obj


def subgradient_refine(a0, time_limit=30.0):
    """Subgradient method with momentum and adaptive smoothing."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_obj = true_peak(a)

    t0 = time.monotonic()
    alpha = 1.0 / (2.0 * n)
    eps = 0.05
    a_prev = a.copy()
    beta = 0.4

    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_auto_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break

        scores = (M - conv) / (eps * M + 1e-15)
        weights = np.exp(-np.maximum(0.0, scores))
        total_w = weights.sum()
        if total_w < 1e-30:
            break
        weights /= total_w

        grad = 2.0 * n * scipy.signal.correlate(weights, a, mode='valid')

        a_new = a - alpha * grad + beta * (a - a_prev)
        a_new = project_simplex(a_new)

        obj_new = true_peak(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            eps = max(0.002, eps * 0.998)
        else:
            alpha *= 0.93
            eps = min(0.3, eps * 1.2)
            beta = 0.0
            a_prev = a_new.copy()

        a_prev = a
        a = a_new

    return best_a, best_obj


def coordinate_refine(a0, time_limit=30.0, seed=12345):
    """Coordinate descent: move mass from peak-contributing to safe positions."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_obj = true_peak(a)

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 800:
        conv = fast_auto_convolve(a)
        k_star = int(np.argmax(conv))

        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        danger = np.where(contrib > 0.5 * max_c)[0]
        safe = np.where(contrib < 0.1 * max_c)[0]

        if len(danger) == 0 or len(safe) == 0:
            no_improve += 1
            continue

        improved = False
        for _ in range(40):
            di = danger[rng.integers(len(danger))]
            si = safe[rng.integers(len(safe))]
            if di == si:
                continue
            for frac in [0.05, 0.15, 0.3, 0.5]:
                amt = a[di] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[di] -= amt
                a_trial[si] += amt
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-15:
                    continue
                a_trial /= ts
                o = true_peak(a_trial)
                if o < best_obj - 1e-12:
                    best_obj = o
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1

    return best_a, best_obj

def make_initializations(n, rng):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    rc = np.maximum(rc, 1e-6)
    inits['raised_cos'] = rc / rc.sum()

    # Gaussian windows
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-10)
        inits['gauss_{}'.format(int(sf * 100))] = g / g.sum()

    # Sparse random (half zeros)
    sparse = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    sparse[idx] = 1.0
    inits['sparse_rand'] = sparse / sparse.sum()

    # Fully random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    # Windowed random
    w = inits['raised_cos']
    r2 = rng.random(n) + 0.1
    wr = w * r2
    inits['windowed_rand'] = wr / wr.sum()

    # Alternating pattern
    alt = np.zeros(n)
    alt[::2] = 1.0
    inits['alternating'] = alt / alt.sum()

    # Parent
    parent = get_parent_resized(n)
    if parent is not None:
        inits['parent'] = parent
        for nl in [0.02, 0.05, 0.1]:
            p_noisy = parent + nl * rng.standard_normal(n)
            p_noisy = np.maximum(p_noisy, 0)
            ps = p_noisy.sum()
            if ps > 1e-10:
                inits['parent_n{}'.format(int(nl * 100))] = p_noisy / ps

    return inits
