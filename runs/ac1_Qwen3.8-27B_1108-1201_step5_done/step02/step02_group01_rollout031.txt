"""
Optimized step function search using multiplicative subgradient descent
with top-K peak focusing, LP active-set refinement, and block coordinate
descent. Targets minimization of 2n*max(conv(a,a))/sum(a)^2 for non-negative
sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize_to_sum(a, S):
    """Normalize a to have sum S."""
    s = np.sum(a)
    if s < 1e-12:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def subgradient_at_peaks(a, peak_indices):
    """Compute subgradient of max over peak_indices of (a*a)[k]."""
    n = len(a)
    sg = np.zeros(n)
    for k in peak_indices:
        for j in range(n):
            val = 0.0
            i1 = k - j
            if 0 <= i1 < n:
                val += a[i1]
            i2 = j - k
            if 0 <= i2 < n:
                val += a[i2]
            sg[j] += val
    return sg


def subgradient_fast(a, k):
    """Fast subgradient at lag k using the convolution structure."""
    n = len(a)
    # (a*a)[k] = sum_{j} a[j] * a[k-j] where 0<=k-j<n
    # d/d a[j] = a[k-j] (if valid) + a[j-k] (if valid)
    sg = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(k, n - 1)
    for j in range(j_min, j_max + 1):
        sg[j] += a[k - j]
    j_min2 = max(0, k - n + 1)  # for a[j-k], need 0 <= j-k < n, i.e., k <= j < k+n
    j_max2 = min(n - 1, k + n - 1)
    for j in range(max(0, k), min(n, k + n)):
        sg[j] += a[j - k]
    return sg


def multiplicative_subgradient_step(a, lr, top_k=5, S=None):
    """One step of multiplicative subgradient descent focusing on top peaks."""
    if S is None:
        S = np.sum(a)
    n = len(a)
    conv = fft_convolve(a)
    
    # Find top K peaks
    # Avoid the very edges (lag 0 and lag 2n-2) as they're trivially small
    peak_indices = np.argsort(conv)[::-1][:top_k]
    
    # Compute weighted subgradient
    weights = np.zeros(2 * n - 1)
    for i, k in enumerate(peak_indices):
        weights[k] = 1.0 / (i + 1)  # Higher weight for higher peaks
    
    sg = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] > 0:
            sg += weights[k] * subgradient_fast(a, k)
    
    # Normalize subgradient
    sg_mean = np.mean(sg)
    sg_centered = sg - sg_mean
    
    # Multiplicative update
    a_new = a * np.exp(-lr * sg_centered)
    a_new = np.clip(a_new, 1e-10, None)
    a_new = normalize_to_sum(a_new, S)
    
    return a_new, peak_indices[0], np.max(conv)


def build_conv_matrix(a):
    """Build the convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build convolution matrix using broadcasting (faster for moderate n)."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    k_idx = np.arange(2 * n - 1)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]  # (1, n)
    diff = k_idx - j_idx  # (2n-1, n)
    A = np.where((diff >= 0) & (diff < n), a[diff.clip(0, n-1)], 0.0)
    return A


def lp_solve_full(a, M):
    """Solve the full LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return np.ones(n) / n


def lp_solve_active(a, M, active_lags):
    """Solve LP with only active constraints."""
    n = len(a)
    A = build_conv_matrix_fast(a)
    A_active = A[active_lags, :]
    b_active = np.full(len(active_lags), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_active, b_ub=b_active, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return lp_solve_full(a, M)

def block_coordinate_descent(a, time_limit, block_size=16, verbose=True):

        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))

        # (a*a)[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contrib = np.zeros(n)
        j_start = max(0, peak_idx - n + 1)
        j_end = min(peak_idx, n - 1)
        for j in range(j_start, j_end + 1):

"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Uses fast subgradient descent with top-peak focusing, LP active-set
refinement, and block coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize_to_sum(a, S):
    """Normalize a to have sum exactly S."""
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def peak_subgradient(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a. Result: 2*a[k-j] where valid."""
    n = len(a)
    sg = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        indices = np.arange(j_start, j_end + 1)
        sg[indices] = 2.0 * a[k - indices]
    return sg


def multi_peak_subgradient(a, peak_lags, weights=None):
    """Compute weighted sum of subgradients at multiple peak lags."""
    if weights is None:
        weights = np.ones(len(peak_lags)) / len(peak_lags)
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peak_lags, weights):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            sg[indices] += w * 2.0 * a[k - indices]
    return sg


def project_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S} using bisection."""
    # Find t such that sum(max(a - t, 0)) = S
    lo = np.min(a) - S
    hi = np.max(a)
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(a - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.maximum(a - t, 0)


def subgradient_descent(a, time_limit, verbose=True, top_k=3, init_lr=None):
    """Subgradient descent with momentum and adaptive learning rate."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)
    if init_lr is None:
        lr = 0.5 / n
    else:
        lr = init_lr
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a))
    best_score = 2.0 * n * best_conv_max / (S * S)
    
    momentum = np.zeros(n)
    beta = 0.85
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        top_peaks = np.argsort(conv)[::-1][:top_k]
        
        # Softmax weights for peaks
        peak_vals = conv[top_peaks]
        weights = np.exp(peak_vals / (0.1 * peak_vals[0] + 1e-10))
        weights /= weights.sum()
        
        sg = multi_peak_subgradient(a, top_peaks, weights)
        sg -= np.mean(sg)  # Center to maintain sum
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * sg
        
        # Update
        a_new = a - lr * momentum
        a_new = project_simplex(a_new, S)
        
        new_conv = fft_convolve(a_new)
        new_conv_max = np.max(new_conv)
        new_score = 2.0 * n * new_conv_max / (S * S)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            best_conv_max = new_conv_max
            a = a_new
            stall = 0
        else:
            stall += 1
            a = a_new  # Accept anyway for exploration
            if stall > 30:
                a = best_a.copy()
                lr *= 0.9
                momentum = np.zeros(n)
                stall = 0
            if stall > 200:
                lr = min(lr, 2.0) * 1.05  # Try larger step
                stall = 0
        
        iter_count += 1
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SD iter {iter_count} ({elapsed:.0f}s): score={best_score:.6f}, lr={lr:.4f}")
    
    return best_a


def multiplicative_subgradient(a, time_limit, verbose=True, top_k=5):
    """Multiplicative subgradient descent (maintains positivity naturally)."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a)) / (S * S)
    
    lr = 1.0 / n
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        top_peaks = np.argsort(conv)[::-1][:top_k]
        
        # Uniform weights on top peaks
        weights = np.ones(top_k) / top_k
        sg = multi_peak_subgradient(a, top_peaks, weights)
        sg -= np.mean(sg)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * sg)
        a_new = np.clip(a_new, 1e-10, None)
        a_new = normalize_to_sum(a_new, S)
        
        new_score = 2.0 * n * np.max(fft_convolve(a_new)) / (S * S)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                stall = 0
            if lr < 1.0 / (10 * n):
                a = best_a.copy()
                lr = 5.0 / n
                stall = 0
        
        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    M-SD iter {iter_count} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a


def build_conv_matrix(a):
    """Build convolution matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            A[k, indices] = a[k - indices]
    return A


def solve_lp(a, M, active_lags=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Optionally use subset of constraints."""
    n = len(a)
    A = build_conv_matrix(a)
    if active_lags is not None:
        A_use = A[active_lags, :]
        b_use = np.full(len(active_lags), M)
    else:
        A_use = A
        b_use = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_use, b_ub=b_use, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return np.ones(n) / n


def lp_refine(a, time_limit, verbose=True):
    """LP-based iterative refinement with active set."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a)) / (S * S)
    
    active_lags = None
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        M = np.max(conv)
        
        # Update active set: lags where conv is within 10% of peak
        threshold = M * 0.90
        near_peak = np.where(conv >= threshold)[0]
        
        if active_lags is None or iter_count % 10 == 0:
            active_lags = near_peak
        else:
            active_lags = np.unique(np.concatenate([active_lags, near_peak]))
        
        # Limit active set size
        if len(active_lags) > 300:
            active_lags = np.sort(near_peak[np.argsort(conv[near_peak])[::-1]][:300])
        
        # Solve LP
        g = solve_lp(a, M, active_lags)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g_norm = normalize_to_sum(g, S)
        
        # Adaptive blend with line search
        best_t = 0.0
        best_new_score = np.inf
        for t_try in [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]:
            candidate = (1 - t_try) * a + t_try * g_norm
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_to_sum(candidate, S)
            score = 2.0 * n * np.max(fft_convolve(candidate)) / (S * S)
            if score < best_new_score:
                best_new_score = score
                best_t = t_try
        
        if best_t == 0.0 or best_new_score >= best_score * 1.01:
            # Stuck - reset and break
            if iter_count > 30:
                break
        else:
            a = (1 - best_t) * a + best_t * g_norm
            a = np.clip(a, 0.0, None)
            a = normalize_to_sum(a, S)
            if best_new_score < best_score:
                best_score = best_new_score
                best_a = a.copy()
        
        iter_count += 1
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count} ({elapsed:.0f}s): score={best_score:.6f}, t={best_t}")
    
    return best_a


def block_descent(a, time_limit, block_size=32, verbose=True):
    """Block coordinate descent focusing on peak-contributing blocks."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_convolve(a)) / (S * S)
    rng = np.random.default_rng(777)
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))
        
        # Contribution to peak from each position
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            contrib[indices] = a[indices] * a[peak_idx - indices]
        
        # Select blocks by contribution
        n_blocks = n // block_size
        if n_blocks > 0:
            block_sums = np.array([np.sum(contrib[i*block_size:(i+1)*block_size]) 
                                   for i in range(n_blocks)])
            top_blocks = np.argsort(block_sums)[::-1][:min(5, n_blocks)]
        else:
            top_blocks = np.array([0])
        
        improved = False
        for bi in top_blocks:
            start = int(bi) * block_size
            end = min(start + block_size, n)
            width = end - start
            
            # Try several perturbations
            for trial in range(30):
                candidate = a.copy()
                if trial < 10:
                    # Reduce the block
                    factor = 1.0 - 0.02 * (trial + 1)
                    candidate[start:end] *= factor
                elif trial < 20:
                    # Random perturbation
                    candidate[start:end] *= np.exp(rng.normal(0, 0.03, width))
                else:
                    # Increase the block (redistribute)
                    candidate[start:end] *= 1.0 + 0.01 * (trial - 19)
                
                candidate = np.clip(candidate, 0.0, None)
                candidate = normalize_to_sum(candidate, S)
                
                new_score = 2.0 * n * np.max(fft_convolve(candidate)) / (S * S)
                if new_score < best_score:
                    best_score = new_score
                    best_a = candidate.copy()
                    a = candidate
                    improved = True
                    break
        
        # Also try random blocks
        if not improved and rng.random() < 0.5:
            start = int(rng.integers(0, max(1, n - block_size)))
            candidate = a.copy()
            candidate[start:start+block_size] *= np.exp(rng.normal(0, 0.02, min(block_size, n-start)))
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_to_sum(candidate, S)
            new_score = 2.0 * n * np.max(fft_convolve(candidate)) / (S * S)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate
        
        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD iter {iter_count} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a

        nonlocal best_score, best_seq
