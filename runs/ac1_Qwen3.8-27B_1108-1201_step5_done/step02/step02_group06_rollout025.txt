def gs_iteration(a, num_iters):
    """Gerchberg-Saxton style iteration for flat spectrum."""
    n = len(a)
    for _ in range(num_iters):
        # Frequency domain: compute power spectrum
        spectrum = np.fft.rfft(a, 2*n)
        power = np.abs(spectrum)**2
        
        # Modify power spectrum to be flatter
        # Target: make power as uniform as possible
        mean_power = np.mean(power)
        # Blend towards uniform
        power_new = 0.5 * power + 0.5 * mean_power
        
        # Recover time domain
        # Take inverse FFT with random phase
        phase = np.angle(spectrum)
        spectrum_new = np.sqrt(power_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(spectrum_new, 2*n)[:n]
        
        # Enforce non-negativity
        a = np.clip(a_new, 0.0, None)
    
    return a

    return a

"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Combines diverse initializations, fast gradient
screening, LP-based deep optimization, coordinate descent, and simulated
annealing. Key improvements over previous: better initializations (window
functions, structured sequences), coordinate descent for exact 1D coordinate
optimization, and smarter perturbation strategies based on autocorrelation
structure.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_A(f):
    """Build constraint matrix where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. f convolved direction <= M, g >= 0."""
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
    if result.success:
        return result.x
    return None


def coordinate_descent_step(a, j):
    """Find optimal a[j] given all other elements fixed. Returns new value."""
    n = len(a)
    v = a[j]
    S_other = np.sum(a) - v
    c = fft_convolve(a, a)
    max_unaffected = 0.0
    for k in range(2 * n - 1):
        if k < j or k >= j + n:
            max_unaffected = max(max_unaffected, c[k])

    def score_at(v_new):
        if v_new < 0:
            return np.inf
        S_new = S_other + v_new
        if S_new < 0.01:
            return np.inf
        best = max_unaffected
        for k in range(j, j + n):
            if k >= 2 * n - 1:
                break
            if k == 2 * j:
                c_k = c[k] - v * v + v_new * v_new
            else:
                idx = k - j
                if 0 <= idx < n:
                    c_k = c[k] + (v_new - v) * a[idx]
                else:
                    c_k = c[k]
            if c_k > best:
                best = c_k
        return 2.0 * n * best / (S_new * S_new)

    # Golden section search
    lo, hi = 0.0, max(v * 5.0, 2.0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = score_at(c1)
    f2 = score_at(c2)
    for _ in range(40):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = score_at(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = score_at(c2)
    v_opt = (lo + hi) / 2.0
    return v_opt


def coordinate_descent(a, num_sweeps=50, time_limit=60.0):
    """Coordinate descent optimization."""
    n = len(a)
    best = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    for sweep in range(num_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            v_new = coordinate_descent_step(a, j)
            a[j] = v_new
            improved = True
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
        if not improved:
            break
    return best, best_score


def lp_optimize(f_init, time_limit, t_init=0.3, lp_time=8.0):
    """LP-based iterative optimization."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    rng_local = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)

        g = solve_lp(f, M, time_limit=lp_time)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        g_rescaled = g * (sum_f / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / sum_new)

        score_new = local_score(f_new)

        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            if t < 0.7:
                t = min(0.7, t * 1.03)
        else:
            f = f_new
            stagnant += 1
            if stagnant > 15:
                t *= 0.7
                stagnant = 0
            if stagnant > 60:
                f = f + 0.03 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.2
                stagnant = 0

    return best_f, best_score


def gradient_polish(f_init, time_limit, verbose=False):
    """Adam gradient descent on log-parameterization with annealed max."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / np.sum(f_safe))

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    tau_start = 10.0 / n
    tau_end = 0.2 / n
    base_lr = 1.0
    restart_count = 0
    max_restarts = 5

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S

        conv = fft_convolve(a, a)
        conv_max = np.max(conv)
        shifted_conv = (conv - conv_max) / tau
        exp_conv = np.exp(shifted_conv)
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp

        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]

        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)

        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr = base_lr / np.sqrt(1.0 + 0.05 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_check = e_new / np.sum(e_new) * S
        score = local_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

        # Periodic restart to escape local minima
        if step_count % 500 == 0 and restart_count < max_restarts:
            theta_perturbed = best_theta + 0.1 * np.random.default_rng(step_count).standard_normal(n)
            t_shifted2 = theta_perturbed - np.max(theta_perturbed)
            a_test = np.exp(t_shifted2) / np.sum(np.exp(t_shifted2)) * S
            if local_score(a_test) < best_score * 1.02:
                theta = theta_perturbed
                momentum1 = np.zeros(n)
                momentum2 = np.zeros(n)
                step_count = 0
                restart_count += 1

    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

                candidate[:mid] *= np.exp(rng.normal(0.015, 0.01, size=mid))
                candidate[mid:] *= np.exp(rng.normal(-0.015, 0.01, size=n - mid))
            else:
                candidate[:mid] *= np.exp(rng.normal(-0.015, 0.01, size=mid))
                candidate[mid:] *= np.exp(rng.normal(0.015, 0.01, size=n - mid))
        elif r < 0.90:
            # Mirror perturbation: perturb a[j] and adjust a[n-1-j]
            j = int(rng.integers(0, n))
            delta = rng.normal(0.0, 0.02)
            candidate[j] *= np.exp(delta)
            if n - 1 - j != j:
                candidate[n - 1 - j] *= np.exp(-delta)
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score

        temperature *= cooling
        if temperature < 1e-6:
            temperature = 0.002
            cooling = 0.99985

    return best_f, best_score


def make_window_init(n, window_type, rng):
    """Generate window function initializations."""
    t = np.arange(n, dtype=np.float64)
    if window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    elif window_type == 'hamming':
        a = 0.54 + 0.46 * np.cos(2 * np.pi * t / (n - 1))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t / (n - 1)) + 0.08 * np.cos(4 * np.pi * t / (n - 1))
    elif window_type == 'raised_cosine':
        a = np.cos(np.pi * t / (n - 1))
        a = np.abs(a)
    elif window_type == 'parabolic':
        x = 2.0 * t / (n - 1) - 1.0
        a = 1.0 - x * x
    elif window_type == 'gaussian':
        x = 2.0 * t / (n - 1) - 1.0
        alpha = 2.0 + 3.0 * rng.random()
        a = np.exp(-alpha * x * x)
    elif window_type == 'taylor':
        # Taylor-like window
        omega_group = -15.0 * rng.random() - 10.0
        A = 10 ** (omega_group / 20.0)
        sig = np.arccosh(A) / np.pi
        T = (0.5 * (n - 1)) * sig * (1.0 / np.sqrt(A * A - 1.0))
        a = np.ones(n)
        for k in range(1, 9):
            Ck = (A / k) * np.prod([(A * A - k * k) / (2 * (n - 1) * k * k - (n - 1) * (n - 1))] for _ in [1]) if False else 0
            # Simplified Taylor window
            Ck = ((A / k) * np.prod([(A * A - j * j) / ((n - 1) ** 2 - j * j)] for j in range(1, k + 1))) if k <= 8 else 0
            if k <= 8:
                a += Ck * np.cos(2 * np.pi * k * t / (n - 1))
    elif window_type == 'chirp':
        alpha = 2.0 + 8.0 * rng.random()
        a = 1.0 + 0.3 * np.cos(np.pi * alpha * t * t / n)
    elif window_type == 'anti_periodic':
        a = np.ones(n)
        half = n // 2
        a[:half] = 1.0 + 0.3 * rng.standard_normal(half)
        a[half:] = 1.0 - 0.3 * rng.standard_normal(n - half)
    elif window_type == 'multitone':
        a = np.ones(n)
        for _ in range(5 + int(rng.integers(0, 5))):
            k = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    elif window_type == 'random':
        spread = 0.2 + 0.4 * rng.random()
        a = 1.0 + spread * rng.standard_normal(n)
    elif window_type == 'bimodal':
        # Two peaks
        center1 = int(n * rng.uniform(0.2, 0.4))
        center2 = int(n * rng.uniform(0.6, 0.8))
        width = n // 8
        a = np.exp(-((t - center1) ** 2) / (2 * width ** 2)) + \
            np.exp(-((t - center2) ** 2) / (2 * width ** 2))
        a += 0.05
    elif window_type == 'sinc_like':
        # Approximate sinc
        x = t - n / 2
        a = np.sinc(2.0 * x / n)
        a = np.abs(a)
        a += 0.1
    else:
        a = np.ones(n)

    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Fast screening of diverse initializations with gradient optimization
    Phase 2: Deep LP optimization on top candidates
    Phase 3: Gradient polish with restarts
    Phase 4: Coordinate descent + SA + final LP
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            if len(provided_seq) > 0:
                s_prov = local_score(provided_seq)
                print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
                if s_prov < best_score:
                    best_score = s_prov
                    best_seq = provided_seq.copy()
        except Exception:
            pass

    # Target length
    n_target = len(provided_seq) if provided_seq is not None else 256

    # === Phase 1: Fast screening of diverse initializations ===
    print("\n" + "=" * 60)
    print(f"Phase 1: Screening diverse initializations (n={n_target})")
    print("=" * 60)

    phase1_time = min(350.0, (deadline - time.monotonic()) * 0.22)
    phase1_deadline = time.monotonic() + phase1_time

    init_types = [
        'provided', 'hann', 'hamming', 'blackman', 'gaussian',
        'chirp', 'anti_periodic', 'multitone', 'bimodal', 'sinc_like',
        'parabolic', 'taylor', 'random', 'raised_cosine'
    ]

    candidates = []
    per_init_time = max(15.0, phase1_time / len(init_types) - 5.0)

    for init_name in init_types:
        if time.monotonic() > phase1_deadline:
            break

        if init_name == 'provided' and provided_seq is not None:
            a0 = provided_seq.astype(np.float64).copy()
        else:
            a0 = make_window_init(n_target, init_name if init_name != 'provided' else 'random', rng)

        # Quick normalization
        a0 = a0 / np.sum(a0) * np.sqrt(2 * n_target)

        # Quick gradient optimization
        t_start = time.monotonic()
        a_opt, score_opt = gradient_polish(a0, per_init_time, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"  {init_name:15s}: score={score_opt:.6f} (time={elapsed:.1f}s)")

        candidates.append((score_opt, a_opt, init_name))

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} ({init_name}) ***")

    # Sort by score, keep top 5
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c for c in candidates if c[0] < np.inf][:5]
    print(f"\n  Top candidates: {[f'{c[0]:.6f} ({c[2]})' for c in top_candidates]}")

    # === Phase 2: Deep LP optimization on top candidates ===
    print("\n" + "=" * 60)
    print(f"Phase 2: Deep LP optimization")
    print("=" * 60)

    phase2_time = deadline - time.monotonic() - 500.0  # Save time for phases 3-4
    phase2_time = max(300.0, min(phase2_time, (deadline - time.monotonic()) * 0.45))

    if top_candidates and phase2_time > 50:
        n_lp = min(len(top_candidates), 3)
        lp_time_each = (phase2_time - 10.0) / n_lp

        for i, (sc, a_cand, name) in enumerate(top_candidates[:n_lp]):
            if time.monotonic() > deadline - 500:
                break
            remaining = deadline - time.monotonic() - 500
            alloc = min(lp_time_each, remaining)
            if alloc < 30:
                break

            print(f"\n  LP on candidate {i+1} (score={sc:.6f}, {name}), alloc={alloc:.0f}s")
            a_lp, score_lp = lp_optimize(a_cand, alloc, t_init=0.3, lp_time=6.0)
            print(f"  LP result: {score_lp:.6f}")

            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"  *** New best: {best_score:.6f} ***")

            # Also try coordinate descent on the LP result
            if time.monotonic() < deadline - 520:
                a_cd, score_cd = coordinate_descent(a_lp, num_sweeps=20, time_limit=20)
                if score_cd < best_score:
                    best_score = score_cd
                    best_seq = a_cd.copy()
                    print(f"  CD result: {best_score:.6f} *** New best ***")

    # === Phase 3: Gradient polish on the best ===
    print("\n" + "=" * 60)
    print(f"Phase 3: Gradient polish (score={best_score:.6f})")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 200.0
        grad_time = max(50.0, min(300.0, remaining * 0.5))
        print(f"  Gradient polish, time={grad_time:.0f}s")
        a_grad, score_grad = gradient_polish(best_seq, grad_time, verbose=False)
        print(f"  Gradient result: {score_grad:.6f}")
        if score_grad < best_score:
            best_score = score_grad
            best_seq = a_grad.copy()
            print(f"  *** New best: {best_score:.6f} ***")

        # Coordinate descent after gradient
        if time.monotonic() < deadline - 220:
            cd_time = min(40.0, deadline - time.monotonic() - 220)
            if cd_time > 10:
                a_cd, score_cd = coordinate_descent(best_seq, num_sweeps=30, time_limit=cd_time)
                if score_cd < best_score:
                    best_score = score_cd
                    best_seq = a_cd.copy()
                    print(f"  CD result: {best_score:.6f} *** New best ***")

    # === Phase 4: SA + final LP ===
    print("\n" + "=" * 60)
    print(f"Phase 4: SA + Final refinement (score={best_score:.6f})")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 10:
        # SA polish
        remaining = deadline - time.monotonic()
        sa_time = min(120.0, remaining * 0.4)
        if sa_time > 20:
            print(f"  SA, time={sa_time:.0f}s")
            a_sa, score_sa = sa_polish(best_seq, sa_time, rng)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

        # Final coordinate descent
        if time.monotonic() < deadline - 30:
            cd_time = min(30.0, deadline - time.monotonic() - 30)
            if cd_time > 5:
                a_cd, score_cd = coordinate_descent(best_seq, num_sweeps=50, time_limit=cd_time)
                if score_cd < best_score:
                    best_score = score_cd
                    best_seq = a_cd.copy()
                    print(f"  Final CD: {best_score:.6f} *** New best ***")

        # Final LP
        if time.monotonic() < deadline - 15:
            lp_time = max(5.0, deadline - time.monotonic() - 15)
            if lp_time > 10:
                print(f"  Final LP, time={lp_time:.0f}s")
                a_final, score_final = lp_optimize(best_seq, lp_time, t_init=0.2, lp_time=5.0)
                if score_final < best_score:
                    best_score = score_final
                    best_seq = a_final.copy()
                    print(f"  Final LP: {best_score:.6f} *** New best ***")

        # Very final coordinate descent
        if time.monotonic() < deadline - 5:
            a_cd, score_cd = coordinate_descent(best_seq, num_sweeps=10, time_limit=5)
            if score_cd < best_score:
                best_score = score_cd
                best_seq = a_cd.copy()

    # Also try a different length if time permits
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_alt = n_target * 2 if n_target <= 384 else n_target + 128
        remaining_alt = deadline - time.monotonic() - 10
        if remaining_alt > 60 and n_alt <= 1024:
            print(f"\n  Trying alternative length n={n_alt}, time={remaining_alt:.0f}s")
            a_alt = make_window_init(n_alt, 'hann', rng)
            a_alt = a_alt / np.sum(a_alt) * np.sqrt(2 * n_alt)
            alloc = min(remaining_alt * 0.7, 120.0)
            a_alt_opt, score_alt = gradient_polish(a_alt, alloc, verbose=False)
            if score_alt < best_score:
                best_score = score_alt
                best_seq = a_alt_opt.copy()
                print(f"  Alt length result: {best_score:.6f} *** New best ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
