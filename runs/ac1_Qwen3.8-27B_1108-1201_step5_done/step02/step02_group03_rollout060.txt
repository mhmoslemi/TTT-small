import numpy as np
import time
from scipy.optimize import minimize

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (S * S))

def normalize_to_sum(a, S_target):
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (S_target / S)

def soft_max_gradient(a, tau):
    """
    Gradient of the log-sum-exp (soft-max) of the convolution.
    This is a smooth approximation to the max.
    """
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_j w[i+j] * a[j] for valid j
    # This is: 2 * (cross-correlation of w with a)[i]
    # Using convolution: convolve(w, a[::-1])[n-1 + i] = sum_j w[n-1+i+j] * a[::-1][j]
    # Hmm, let me just do it directly for clarity
    a_rev = a[::-1]
    # conv(w, a_rev)[i + n - 1] = sum_j w[i+n-1-j] * a_rev[j] = sum_j w[i+n-1-j] * a[n-1-j]
    # Let k = n-1-j, then j = n-1-k:
    # = sum_k w[i+k] * a[k] for valid k
    # That's what we want!
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad, w

def objective_and_grad(a, tau, S_target):
    """Objective and gradient for the soft-max objective with sum constraint."""
    n = len(a)
    # Normalize
    a_norm = normalize_to_sum(a, S_target)
    
    c = np.convolve(a_norm, a_norm)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        logsumexp = c_max + np.log(max(w_sum, 1e-300)) / tau
    else:
        logsumexp = c_max + np.log(w_sum) / tau
    w /= w_sum
    
    # The soft-max objective is: logsumexp(tau * c) / tau
    # Its derivative w.r.t. a is: (1/tau) * tau * w * (dc/da) = w * (dc/da)
    # dc[k]/da[i] = a[k-i] + a[i-k] (for valid indices)
    # So d(logsumexp)/da[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    
    # Using the convolution trick:
    a_rev = a_norm[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    raw_grad = 2.0 * conv_result[n-1:n-1+n]
    
    # Project out the sum constraint: subtract mean
    grad = raw_grad - raw_grad.mean()
    
    # The actual objective value (hard max)
    obj = 2.0 * n * c_max / (S_target * S_target)
    
    return a_norm, raw_grad, grad, obj

def optimize_sequence(a0, deadline, verbose=True):
    """
    Optimize a sequence using gradient descent with soft-max,
    adaptive learning rate, and periodic restarts.
    """
    n = len(a0)
    S_target = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.maximum(a0.astype(np.float64), 0), S_target)
    
    best_a = a.copy()
    best_score = compute_obj(a)
    
    if verbose:
        print(f"    Start: n={n}, score={best_score:.8f}")
    
    momentum = np.zeros(n)
    momentum_decay = 0.9
    tau = 1.0
    lr = 0.5
    
    no_improve = 0
    best_no_improve_threshold = 20
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        # Anneal tau (increase over time)
        tau = 1.0 + 3.0 * it / 500.0
        tau = min(tau, 100.0)
        
        # Adaptive learning rate
        lr = 0.5 / (1.0 + 0.01 * it)
        
        # Compute gradient
        a_norm, raw_grad, grad, obj = objective_and_grad(a, tau, S_target)
        
        # Momentum step
        momentum = momentum_decay * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return float('inf')
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target_sum):
    """Normalize a to have the given sum."""
    S = float(np.sum(a))
    if S < 1e-15:
        return a.copy()
    return a * (target_sum / S)

def soft_max_obj_grad(a, tau, target_sum):
    """
    Compute soft-max objective and its gradient (projected onto sum constraint).
    Returns: (a_normalized, gradient, hard_max_obj)
    """
    n = len(a)
    a = normalize(np.maximum(a, 0), target_sum)
    
    c = np.convolve(a, a)
    c_max = float(c.max())
    c_shifted = c - c_max
    
    # Compute weights
    exp_vals = np.exp(tau * c_shifted)
    w_sum = float(exp_vals.sum())
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_vals / w_sum
    
    # Gradient: grad[i] = 2 * sum_{valid j} w[i+j] * a[j]
    a_rev = a[::-1]
    conv_res = np.convolve(w, a_rev, mode='full')
    raw_grad = 2.0 * conv_res[n-1:n-1+n]
    
    # Project onto sum constraint (subtract mean)
    grad = raw_grad - raw_grad.mean()
    
    # Hard max objective
    obj = 2.0 * n * c_max / (target_sum * target_sum)
    
    return a, grad, obj

    """One gradient step with momentum and non-negativity projection."""
    momentum_buf[:] = momentum_decay * momentum_buf - lr * grad
    a_new = a + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, target_sum)

def upsample(a, new_n):
    """
    Upsample a sequence by interpolation.
    """
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    old_idx = np.linspace(0, old_n - 1, old_n)
    new_idx = np.linspace(0, old_n - 1, new_n)
    a_new = np.interp(new_idx, old_idx, a)
    return np.maximum(a_new, 0)

def downsample(a, new_n):
    """
    Downsample a sequence by averaging.
    """
    old_n = len(a)
    if new_n >= old_n:
        return a.copy()
    idx = np.linspace(0, old_n - 1, new_n).astype(int)
    return a[idx]

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses multi-trajectory soft-max gradient descent with adaptive tau annealing,
cosine LR schedule, structured restarts (opposite-gradient jumps, log-space
reparameterization), active-set tracking, and coordinate descent polishing.
Differently from prior SOTA: runs parallel conceptual trajectories with
diverse tau schedules, uses log-space parameterization for some runs,
and employs "anti-basin" restarts that jump to the opposite side of the
current local minimum.

import numpy as np
import time


def compute_obj(a):
    """Compute objective: 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return float('inf')
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target_sum):
    """Normalize to target sum, maintaining non-negativity."""
    a = np.maximum(a, 0.0)
    S = float(np.sum(a))
    if S < 1e-15:
        return a.copy()
    return a * (target_sum / S)


def soft_max_gradient(a, tau, target_sum):
    """
    Compute normalized a and gradient of soft-max objective.
    Returns (a_norm, grad_projected, hard_obj).
    """
    n = len(a)
    a = normalize(a, target_sum)

    c = np.convolve(a, a)
    c_max = float(c.max())
    c_shifted = c - c_max

    exp_vals = np.exp(tau * c_shifted)
    w_sum = float(exp_vals.sum())
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_vals / w_sum

    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    a_rev = a[::-1]
    conv_res = np.convolve(w, a_rev, mode='full')
    raw_grad = 2.0 * conv_res[n - 1:n - 1 + n]

    # Project onto sum=const hyperplane
    grad = raw_grad - raw_grad.mean()

    obj = 2.0 * n * c_max / (target_sum * target_sum)
    return a, grad, obj


def soft_max_gradient_weighted(a, tau, target_sum, n_active=5):
    """
    Gradient that focuses on the top-n_active peaks more sharply.
    Uses a hybrid: weights are exp(tau*(c-cmax)) but with a floor
    so that near-peaks also get some weight.
    """
    n = len(a)
    a = normalize(a, target_sum)

    c = np.convolve(a, a)
    c_max = float(c.max())
    c_shifted = c - c_max

    # Create weights with emphasis on top peaks
    sorted_idx = np.argsort(c)[::-1]
    w = np.zeros_like(c)
    for rank, idx in enumerate(sorted_idx[:n_active]):
        if rank == 0:
            w[idx] = 1.0
        else:
            # Decaying weight for near-peaks
            frac = c[idx] / c_max
            w[idx] = 0.1 * frac

    w_sum = w.sum()
    if w_sum < 1e-300:
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    conv_res = np.convolve(w, a_rev, mode='full')
    raw_grad = 2.0 * conv_res[n - 1:n - 1 + n]
    grad = raw_grad - raw_grad.mean()

    obj = 2.0 * n * c_max / (target_sum * target_sum)
    return a, grad, obj


def gradient_descent_trajectory(a0, deadline, tau_start, tau_end, lr_start,
                                lr_end, momentum_decay, rng, use_log_space=False):
    """
    Run a gradient descent trajectory with specified hyperparameters.
    Returns (best_a, best_score).
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)

    if use_log_space:
        # Work in log-space: a = exp(x), optimize x
        a_init = normalize(a0, target_sum)
        a_init = np.maximum(a_init, 1e-6)
        x = np.log(a_init)
        best_a = a_init.copy()
        best_score = compute_obj(a_init)
    else:
        a = normalize(a0, target_sum)
        best_a = a.copy()
        best_score = compute_obj(a)

    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    total_iters_estimate = 10000

    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, it / total_iters_estimate)

        # Cosine annealing for tau and lr
        cos_factor = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_start + (tau_end - tau_start) * (1 - cos_factor)
        lr = lr_start + (lr_end - lr_start) * (1 - cos_factor)
        lr = max(lr, 0.005)

        if use_log_space:
            a = np.exp(x)
            a = normalize(a, target_sum)
            a_norm, grad, _ = soft_max_gradient(a, tau, target_sum)
            # Transform gradient to x-space: grad_x = grad_a * a (chain rule)
            grad = grad * a
            # Project
            grad = grad - grad.mean()
            momentum = momentum_decay * momentum - lr * grad
            x = x + momentum
            # Clip x to avoid overflow
            x = np.clip(x, np.log(1e-6), np.log(1000.0))
        else:
            a = normalize(a, target_sum)
            a, grad, _ = soft_max_gradient(a, tau, target_sum)
            momentum = momentum_decay * momentum - lr * grad
            a = a + momentum
            a = np.maximum(a, 0.0)
            a = normalize(a, target_sum)

        # Check improvement
        if it % 100 == 0:
            if use_log_space:
                a_check = np.exp(x)
                a_check = normalize(a_check, target_sum)
            else:
                a_check = a
            score = compute_obj(a_check)
            if score < best_score - 1e-11:
                best_score = score
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Restart when stuck
        if no_improve > 15:
            if no_improve % 8 == 0:
                if use_log_space:
                    x = x + rng.normal(0, 0.3, n)
                    x = np.clip(x, np.log(1e-5), np.log(1000.0))
                else:
                    # Anti-gradient jump: go opposite to current gradient
                    scale = 0.5 * np.max(np.abs(a))
                    a = a - lr * grad * 10.0
                    a = np.maximum(a, 0.0)
                    a = normalize(a, target_sum)
                momentum = np.zeros(n)
                no_improve = 0

            if no_improve > 50:
                break

    if use_log_space:
        best_a = np.exp(x)
        best_a = normalize(best_a, target_sum)

    return best_a, best_score


def multi_tau_optimize(a0, deadline, verbose=True):
    """
    Run optimization with multiple tau schedules and structured restarts.
    """
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a0, target_sum)

    best_a = a.copy()
    best_score = compute_obj(a)
    rng = np.random.default_rng(777 + n)

    # Define different tau schedules (conceptual parallel trajectories)
    # Each tuple: (tau_start, tau_end, lr_start, lr_end, mom_decay, use_log)
    schedules = [
        (1.0, 50.0, 0.3, 0.01, 0.9, False),
        (3.0, 80.0, 0.2, 0.01, 0.92, False),
        (0.5, 30.0, 0.5, 0.02, 0.85, True),
        (5.0, 100.0, 0.15, 0.01, 0.95, False),
    ]

    time_per_traj = max(10.0, (deadline - time.monotonic()) / len(schedules))

    for si, (ts, te, lrs, lre, md, use_log) in enumerate(schedules):
        if time.monotonic() > deadline - 10:
            break
        traj_deadline = min(deadline - 5, time.monotonic() + time_per_traj)

        if verbose:
            a_start = a.copy() if si == 0 else best_a.copy()
        else:
            a_start = a.copy() if si == 0 else best_a.copy()

        # Slightly perturb for diversity (except first)
        if si > 0:
            a_start = a_start + rng.normal(0, 0.02 * a_start.mean(), n)
            a_start = normalize(a_start, target_sum)

        traj_a, traj_score = gradient_descent_trajectory(
            a_start, traj_deadline, ts, te, lrs, lre, md, rng, use_log
        )

        if traj_score < best_score:
            best_score = traj_score
            best_a = traj_a.copy()
            if verbose:
                print(f"      Traj {si}: score={traj_score:.8f} (improved!)")
        elif verbose:
            print(f"      Traj {si}: score={traj_score:.8f}")

        # Update working point for next trajectory
        a = best_a.copy()

    return best_a, best_score


def active_set_refine(a, deadline, n_active=3):
    """
    Refine using a gradient that only considers the top-n_active peaks.
    This can help "sharpen" the solution.
    """
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a, target_sum)

    best_score = compute_obj(a)

    lr = 0.1
    it = 0

    while time.monotonic() < deadline:
        it += 1
        lr = max(0.005, 0.1 * (0.9995 ** it))
        tau = 10.0 + 5.0 * (it / 500.0)
        tau = min(tau, 50.0)

        a, grad, _ = soft_max_gradient_weighted(a, tau, target_sum, n_active)
        momentum = 0.9 * momentum - lr * grad
        a = a + momentum
        a = normalize(a, target_sum)

        if it % 200 == 0:
            score = compute_obj(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
            elif score > best_score + 1e-6:
                # Revert if we went too far
                a = best_a.copy()

def coordinate_descent_polish(a, deadline, max_rounds=5):
    """
    Coordinate descent: try local changes to each coordinate.
    """
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a.copy(), target_sum)
    best_score = compute_obj(a)
    rng = np.random.default_rng(42)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Randomize coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-12:
                # Try turning it on
                candidates = [0.01 * a.mean(), 0.005 * a.mean()]
            else:
                # Try modifying it
                candidates = [0.0, orig * 0.5, orig * 1.5, orig * 0.7, orig * 1.3]
                # Compute a "smart" candidate based on peak contribution
                c = np.convolve(a, a)
                k_star = int(np.argmax(c))
                j = k_star - i
                if 0 <= j < n and j != i:
                    if a[j] > orig:
                        candidates.append(orig * 0.3)
                    else:
                        candidates.append(orig * 2.0)

            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a = normalize(a, target_sum)
                s = compute_obj(a)
                if s < best_s - 1e-13:
                    best_s = s
                    best_val = val
                a[i] = orig

            if best_val != orig:
                a[i] = best_val
                a = normalize(a, target_sum)
                best_score = best_s
                improved = True

        if not improved:
            break

    return a, best_score


def upsample_sequence(a, new_n):
    """Upsample via linear interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    old_idx = np.linspace(0, old_n - 1, old_n)
    new_idx = np.linspace(0, old_n - 1, new_n)
    return np.maximum(np.interp(new_idx, old_idx, a), 0.0)


def downsample_sequence(a, new_n):
    """Downsample by averaging blocks."""
    old_n = len(a)
    if new_n >= old_n:
        return a.copy()
    # Create block boundaries
    boundaries = np.round(np.linspace(0, old_n, new_n + 1)).astype(int)
    result = np.zeros(new_n)
    for i in range(new_n):
        lo, hi = boundaries[i], boundaries[i + 1]
        if hi > lo:
            result[i] = np.mean(a[lo:hi])
        else:
            result[i] = a[lo] if lo < old_n else 0.0
    return result


def generate_diverse_starts(n, rng):
    """Generate diverse starting sequences."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n) * 0.5)

    # Edge-enhanced (high at edges, low in middle)
    edge_prox = np.minimum(i, n - 1 - i).astype(float)
    starts.append(0.1 + 0.9 * np.exp(-edge_prox / (n * 0.1)))

    # Linear ramp
    starts.append(0.1 + 0.9 * np.linspace(0, 1, n))

    # Cosine harmonics
    for freq in [2, 3, 5, 7, 11]:
        starts.append(np.maximum(0.3 + 0.4 * np.cos(2 * np.pi * freq * i / n), 0.01))

    # Gaussian centered
    starts.append(0.05 + 0.95 * np.exp(-0.5 * ((i - n / 2) / (n * 0.15)) ** 2))

    # Two gaussians
    starts.append(0.05 + 0.5 * np.exp(-0.5 * ((i - n * 0.25) / (n * 0.08)) ** 2)
                  + 0.5 * np.exp(-0.5 * ((i - n * 0.75) / (n * 0.08)) ** 2))

    # Random
    starts.append(0.2 + 0.6 * rng.random(n))

    # Sparse (1/3 nonzero)
    a = np.zeros(n)
    mask = rng.random(n) < (1.0 / 3.0)
    a[mask] = 0.2 + 0.8 * rng.random(mask.sum())
    starts.append(a + 0.01)

    # Triangle
    starts.append(0.1 + 0.9 * (1.0 - 2.0 * np.abs(i - n / 2) / n))

    # Alternating high-low
    a = np.zeros(n)
    a[::2] = 0.8
    a[1::2] = 0.2
    starts.append(a)

    return starts

    best_seq = None
    best_score = float('inf')

    print(f"=== Optimization start, budget={budget_s}s, deadline in {deadline - t_start:.1f}s ===")

    # Load previous best if available
    prev_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            prev_seq = gs.copy()
            prev_score = compute_obj(prev_seq)
            print(f"Previous best: n={len(gs)}, score={prev_score:.8f}")
            best_seq = prev_seq.copy()
            best_score = prev_score
    except NameError:
        print("No previous sequence found, starting from scratch.")

    # Phase 1: Multi-trajectory optimization on previous best (or uniform)
    print("\n--- Phase 1: Multi-trajectory on primary candidate ---")
    primary_n = len(prev_seq) if prev_seq is not None else 384
    if prev_seq is not None:
        a_primary = prev_seq.copy()
    else:
        a_primary = np.ones(primary_n) * 0.5

    phase1_deadline = min(deadline - 300, t_start + 600)
    seq, score = multi_tau_optimize(a_primary, phase1_deadline, verbose=True)
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
        print(f"  Phase 1 result: {best_score:.8f}")
    else:
        print(f"  Phase 1 result: {score:.8f} (no improvement)")

    # Phase 2: Active-set refinement on the best
    print("\n--- Phase 2: Active-set refinement ---")
    if best_seq is not None:
        n_best = len(best_seq)
        phase2_deadline = min(deadline - 200, t_start + 900)
        seq, score = active_set_refine(best_seq, phase2_deadline, n_active=3)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Phase 2 result: {best_score:.8f} (improved)")
        else:
            print(f"  Phase 2 result: {score:.8f}")

    # Phase 3: Try different starting points and lengths
    print("\n--- Phase 3: Diverse starts at various lengths ---")
    lengths_to_try = [256, 384, 512, 640]

    if best_seq is not None:
        # Also try the best length
        if len(best_seq) not in lengths_to_try:
            lengths_to_try.append(len(best_seq))

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break

        # Generate a few diverse starts
        starts = generate_diverse_starts(n_try, rng)

        # Also add upsampled/downsampled version of best
        if best_seq is not None:
            if n_try > len(best_seq):
                starts.append(upsample_sequence(best_seq, n_try))
            elif n_try < len(best_seq):
                starts.append(downsample_sequence(best_seq, n_try))

        # Try each start with limited time
        time_remaining = deadline - time.monotonic() - 60
        time_per_start = max(20.0, time_remaining / (len(starts) * len(lengths_to_try)))
        time_per_start = min(time_per_start, 90.0)

        for si, a0 in enumerate(starts[:4]):  # Limit to 4 per length
            if time.monotonic() > deadline - 120:
                break

            a0 = normalize(a0.astype(np.float64), np.sqrt(2.0 * n_try))
            trig_deadline = time.monotonic() + time_per_start

            # Use a subset of schedules for speed
            target_sum = np.sqrt(2.0 * n_try)
            single_deadline = trig_deadline

            # Quick single-trajectory optimization
            a_work = a0.copy()
            n = n_try
            momentum = np.zeros(n)
            tau = 2.0
            lr = 0.3
            no_imp = 0

            while time.monotonic() < single_deadline:
                tau = min(50.0, 2.0 + 0.5 * lr * 100)
                lr = max(0.01, 0.3 * (0.999 ** int(tau * 10)))

                a_work, grad, _ = soft_max_gradient(a_work, tau, target_sum)
                momentum = 0.9 * momentum - lr * grad
                a_work = a_work + momentum
                a_work = normalize(a_work, target_sum)

                if int(tau * 10) % 100 == 0:
                    s = compute_obj(a_work)
                    if s < best_score - 1e-10:
                        best_score = s
                        best_seq = a_work.copy()
                        no_imp = 0
                        print(f"    [n={n_try}, start{si}] New best: {best_score:.8f}")
                    else:
                        no_imp += 1
                    if no_imp > 20:
                        break

            # Also try the multi-trajectory on the best per-length result
            if time.monotonic() < deadline - 120:
                s = compute_obj(a_work)
                n_best_this_len = n_try
                a_try = a_work.copy() if s < 2.0 else best_seq[:n_try] if best_seq is not None and len(best_seq) >= n_try else a0
                # Only if promising
                if s < best_score + 0.05:
                    mt_deadline = min(deadline - 60, time.monotonic() + 60)
                    seq2, score2 = multi_tau_optimize(a_try, mt_deadline, verbose=False)
                    if score2 < best_score:
                        best_score = score2
                        best_seq = seq2.copy()
                        print(f"    [n={n_try}, multi] New best: {best_score:.8f}")

    # Phase 4: Final focused optimization on the overall best
    print(f"\n--- Phase 4: Final focused optimization (current best={best_score:.8f}) ---")
    if best_seq is not None and time.monotonic() < deadline - 60:
        final_deadline = deadline - 20
        seq, score = multi_tau_optimize(best_seq, final_deadline, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Phase 4 result: {best_score:.8f}")
        else:
            print(f"  Phase 4 result: {score:.8f}")

    # Phase 5: Coordinate descent polish
    print(f"\n--- Phase 5: Coordinate descent polish ---")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        if n <= 512:
            cd_deadline = deadline - 10
            seq, score = coordinate_descent_polish(best_seq, cd_deadline, max_rounds=3)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  CD result: {best_score:.8f} (improved)")
            else:
                print(f"  CD result: {score:.8f}")
        else:
            # For larger n, only polish a subset of coordinates
            cd_deadline = deadline - 10
            seq, score = coordinate_descent_polish(best_seq, cd_deadline, max_rounds=1)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  CD result (partial): {best_score:.8f}")

    # Final validation
    if best_seq is not None:
        # Ensure all values are valid
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.8f} ===")

        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.8f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

        # Ensure sum is valid
        if np.sum(best_seq) < 0.01:
            best_seq = best_seq * 1000
            print("Rescaled for minimum sum constraint.")

        return [float(x) for x in best_seq]

    # Fallback
    print("FALLBACK: returning uniform sequence")
    return [1.0] * 256
