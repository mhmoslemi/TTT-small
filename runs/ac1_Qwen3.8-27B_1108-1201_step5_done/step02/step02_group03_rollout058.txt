def compute_obj(a):

    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):

def hard_max_grad(a):
    """Gradient using hard max (single peak)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def adam_step(a, grad, state, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. State contains m, v, t."""
    m, v, t = state
    t = t + 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    state_new = (m, v, t)
    return a_new, state_new


def project_nonneg_sum(a, target_sum):
    """Project a onto the simplex {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / S)


def lp_refine(f, M, max_n=768):
    """
    LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    """
    n = len(f)
    if n > max_n:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix A where A[m, j] = f[m-j] if 0 <= m-j < n
    # Use efficient construction
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_start = j
        m_end = min(j + n, 2 * n - 1)
        A[m_start:m_end, j] = f[:m_end - m_start]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def generate_starts(rng, lengths):
    """Generate diverse starting sequences."""
    starts = []
    
    # Try to use known good sequence
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)) and gs.sum() > 0:
            for L in lengths:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
                else:
                    # Pad or truncate
                    if len(gs) >= L // 2:
                        a = gs[:L//2].copy()
                        a = np.pad(a, (0, L - len(a)), mode='edge')
                        starts.append(a)
                    else:
                        starts.append(np.resize(gs, L))
            # Perturbed versions
            for L in lengths[:2]:
                if L <= len(gs):
                    for _ in range(2):
                        pert = rng.normal(0, 0.1, L)
                        starts.append(np.maximum(gs[:L] + pert, 0))
    except (NameError, Exception):
        pass
    
    # Random starts
    for n in lengths:
        starts.append(rng.random(n) * 0.5 + 0.25)
        
        # Sparse
        a = np.zeros(n)
        mask = rng.random(n) < 0.5
        a[mask] = rng.random(mask.sum()) + 0.1
        starts.append(a)
        
        # Structured: raised cosine
        i = np.arange(n, dtype=np.float64)
        starts.append(0.5 + 0.5 * np.cos(np.pi * (2 * i / (n-1) - 1)) + 0.01)
        
        # Two bumps
        starts.append(np.exp(-((i - n*0.25)/(n*0.1))**2) + np.exp(-((i - n*0.75)/(n*0.1))**2) + 0.01)
        
        # Linear ramp
        starts.append(np.linspace(0.1, 1.0, n))
        
        # Alternating pattern
        starts.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.3))
    
    return starts

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-space (softmax) parameterization with Adam optimizer, smooth-max annealing,
LP-based refinement, and coordinate descent polishing. Multi-start with diverse
initializations including perturbations of known-good sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_sum(a, target=1.0):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-15:
        return np.full(len(a), target / len(a))
    return a * (target / S)


def smooth_max_weights(c, tau):
    """Compute soft-max weights for the autocorrelation vector."""
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = w / w_sum
    return w


def gradient_smooth_xspace(a, tau):
    """
    Gradient of smooth-max autocorrelation w.r.t. log-space x where a = softmax(x).
    Returns grad_x of shape (n,).
    """
    n = len(a)
    c = np.convolve(a, a)
    w = smooth_max_weights(c, tau)
    # R[i] = sum_j w[i+j] * a[j]
    R = np.correlate(w, a, mode='valid')
    mu = np.dot(w, c)
    grad_x = 2.0 * a * (R - mu)
    return grad_x, c


def gradient_soft_aspace(a, tau):
    """
    Gradient of smooth-max autocorrelation w.r.t. a (original space).
    """
    n = len(a)
    c = np.convolve(a, a)
    w = smooth_max_weights(c, tau)
    # grad[i] = 2 * R[i] where R[i] = sum_j w[i+j]*a[j]
    R = np.correlate(w, a, mode='valid')
    return 2.0 * R, c


def gradient_hard_aspace(a):
    """Gradient using hard max (single peak lag)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[mask] = 2.0 * a[idx[mask]]
    return grad, c


def gradient_hard_xspace(a):
    """Hard max gradient in x-space."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    # dF/da[i] = 2*a[k*-i] for valid k*-i
    # dF/dx[k] = a[k] * (dF/da[k] - sum_i dF/da[i]*a[i])
    # sum_i dF/da[i]*a[i] = 2*c[k*]
    dF_da = np.zeros(n)
    dF_da[mask] = 2.0 * a[idx[mask]]
    sum_term = 2.0 * c[k_star]
    grad_x = a * (dF_da - sum_term)
    return grad_x, c


def adam_update(x, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. Returns new x and updated state."""
    m, v, t = state
    t = t + 1
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, (m, v, t)


def project_simplex(a, target=1.0):
    """Project onto {x >= 0, sum(x) = target} (simple clipping + rescale)."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.full(len(a), target / len(a))
    return a * (target / S)


def lp_refine(f, M, max_n=448, time_limit=30):
    """
    LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    if n > max_n:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_end = min(j + n, 2 * n - 1)
        A[j:m_end, j] = f[:m_end - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_logspace(a0, deadline):
    """
    Optimize in log-space (x-space) with smooth-max annealing using Adam.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 1e-10, None)
    a = a / a.sum()
    x = np.log(a)

    best_a = a.copy()
    best_score = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    tau_start = 0.5
    tau_end = max(5.0, 3.0 * n)
    lr_start = 0.08
    lr_end = 0.0005

    total_iters = 8000
    no_improve = 0
    eval_interval = max(1, total_iters // 80)

    it = 0
    while it < total_iters and time.monotonic() < deadline:
        it += 1
        progress = it / total_iters

        # Anneal tau (exponential schedule, mostly in the soft regime early)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)
        # Anneal learning rate
        lr = lr_start * (lr_end / lr_start) ** (progress ** 0.8)

        if tau >= 20.0:
            grad_x, c = gradient_hard_xspace(a)
        else:
            grad_x, c = gradient_smooth_xspace(a, tau)

        x, (m, v, t) = adam_update(x, grad_x, (m, v, t), lr)

        # Update a from x
        a = np.exp(x - x.max())
        a = a / a.sum()

        if it % eval_interval == 0:
            score = 2.0 * n * c.max()  # Since sum(a)=1, S^2=1
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 15:
                    # Reset momentum to escape
                    m[:] = 0.0
                    v[:] = 0.0
                    t = 0
                    no_improve = 0
                    if progress > 0.8:
                        break

    return best_a, best_score


def optimize_aspace(a0, deadline):
    """
    Optimize in a-space with hard-max gradient and Adam.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None)
    a = normalize_sum(a, 1.0)

    best_a = a.copy()
    best_score = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    lr_start = 0.03
    lr_end = 0.0002
    total_iters = 6000
    no_improve = 0
    eval_interval = max(1, total_iters // 60)

    it = 0
    while it < total_iters and time.monotonic() < deadline:
        it += 1
        progress = it / total_iters
        lr = lr_start * (lr_end / lr_start) ** (progress ** 0.7)

        grad, c = gradient_hard_aspace(a)
        # Subtract mean to approximately maintain sum
        grad = grad - grad.mean()

        a, (m, v, t) = adam_update(a, grad, (m, v, t), lr)
        a = project_simplex(a, 1.0)

        if it % eval_interval == 0:
            score = 2.0 * n * c.max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 12:
                    m[:] = 0.0
                    v[:] = 0.0
                    t = 0
                    no_improve = 0
                    if progress > 0.85:
                        break

    return best_a, best_score


def lp_polish(a, deadline, max_iters=40):
    """LP-based refinement loop."""
    n = len(a)
    if n > 448:
        return a, compute_obj(a)

    best_a = a.copy()
    best_score = compute_obj(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(best_a, best_a)
        M = float(c.max())

        g = lp_refine(best_a, M)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        g_norm = g / g_sum

        improved = False
        for t_blend in [0.4, 0.25, 0.15, 0.05, 0.6, 0.1]:
            a_new = (1.0 - t_blend) * best_a + t_blend * g_norm
            a_new = project_simplex(a_new, 1.0)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-13:
                best_a = a_new
                best_score = s_new
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 6:
                break
            # Small gradient nudge
            grad, _ = gradient_hard_aspace(best_a)
            grad = grad - grad.mean()
            a_new = best_a - 0.005 * grad
            a_new = project_simplex(a_new, 1.0)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
                no_improve = 0
        else:
            no_improve = 0

    return best_a, best_score


def coordinate_polish(a, deadline, max_rounds=15):
    """Coordinate descent: try reducing/increasing each element."""
    n = len(a)
    if n > 512:

    best_a = a.copy()
    best_score = compute_obj(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved_any = False
        c = np.convolve(best_a, best_a)
        k_star = int(np.argmax(c))

        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = best_a[i]
            if orig < 1e-8:
                continue

            candidates = [orig * 0.6, orig * 0.4, orig * 0.2, 0.0, orig * 1.4]
            best_val = orig
            best_s = best_score

            for val in candidates:
                best_a[i] = val
                best_a = normalize_sum(best_a, 1.0)
                s = compute_obj(best_a)
                if s < best_s:
                    best_s = s
                    best_val = val

            if best_val != orig:
                best_a[i] = best_val
                best_a = normalize_sum(best_a, 1.0)
                best_score = best_s
                improved_any = True
            else:
                best_a[i] = orig
                best_a = normalize_sum(best_a, 1.0)

        if not improved_any:
            break

    return best_a, best_score

                                  np.linspace(0, 1, len(gs_norm)), gs_norm)
                    starts.append(a)
            # Perturbed versions of known good
            for L in lengths[:3]:
                if L <= len(gs_norm):
                    for sigma in [0.05, 0.1, 0.2]:
                        a0 = np.maximum(gs_norm[:L] + rng.normal(0, sigma, L), 0)
                        if a0.sum() > 0:
                            starts.append(a0 / a0.sum())
    except (NameError, Exception):
        pass

    # Random and structured starts
    for n in lengths:
        # Uniform random
        starts.append(rng.random(n) + 0.1)

        # Sparse
        a = np.zeros(n)
        mask = rng.random(n) < 0.6
        a[mask] = rng.random(int(mask.sum())) + 0.1
        starts.append(a)

        # Raised cosine
        i = np.arange(n, dtype=np.float64)
        starts.append(0.5 + 0.5 * np.cos(np.pi * (2.0 * i / (n - 1) - 1.0)) + 0.01)

        # Two Gaussian bumps
        starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.12)) ** 2) +
                      np.exp(-0.5 * ((i - n * 0.75) / (n * 0.12)) ** 2) + 0.01)

        # Alternating
        starts.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.4) + 0.01)

        # Linear
        starts.append(np.linspace(0.2, 1.0, n))

        # Cosine chirp
        starts.append(np.abs(np.cos(2.0 * np.pi * i * 3.0 / n)) + 0.05)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization (log-space Adam, a-space Adam,
    LP refinement, coordinate descent) with multi-start strategy.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    lengths = [256, 320, 384, 448, 512]

    print(f"=== Optimization start, budget={budget_s:.0f}s ===")
    print(f"Target: 1.5027, lengths to try: {lengths}")

    starts = make_starts(rng, lengths)
    print(f"Generated {len(starts)} starting sequences")

    # Sort by length (shorter first for quick feedback)
    starts.sort(key=lambda x: len(x))

    # Deduplicate
    seen = set()
    unique_starts = []
    for s in starts:
        key = (len(s), round(float(s.sum()), 2), round(float(s[0]), 3))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)
    starts = unique_starts
    print(f"After dedup: {len(starts)} starts")

    # First pass: try each start with time allocation
    n_starts = len(starts)
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 80.0:
            break

        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        n = len(a0)
        # Allocate more time to first few (likely better) and to n=384
        if idx < 2:
            t_alloc = min(350.0, remaining * 0.2)
        elif n == 384:
            t_alloc = min(400.0, remaining * 0.18)
        elif idx < 10:
            t_alloc = min(200.0, remaining * 0.12)
        else:
            t_alloc = min(120.0, remaining * 0.08)

        t_alloc = max(35.0, t_alloc)
        if time.monotonic() + t_alloc > deadline - 40:
            t_alloc = deadline - time.monotonic() - 40
            if t_alloc < 25:
                break

        print(f"\n[Start {idx}/{n_starts}] n={n}, alloc={t_alloc:.0f}s, "
              f"remaining={remaining:.0f}s")

        phase_deadline = time.monotonic() + t_alloc
        seq, score = optimize_full(a0, phase_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={len(seq)}) ***")
        else:
            print(f"  score={score:.6f} (best={best_score:.6f})")

    # Second pass: intensive refinement of the best
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Second pass: refining best ({best_score:.6f}, n={len(best_seq)}) ===")
        bn = len(best_seq)

        extra_starts = []
        # Small perturbations
        for sigma in [0.02, 0.05, 0.1, 0.15]:
            a0 = np.maximum(best_seq + rng.normal(0, sigma, bn), 0)
            if a0.sum() > 0:
                extra_starts.append(a0 / a0.sum())
        # Resized versions
        for dn in [-32, 32, -64, 64]:
            try_n = bn + dn
            if 128 <= try_n <= 640:
                a0 = np.interp(np.linspace(0, 1, try_n),
                               np.linspace(0, 1, bn), best_seq)
                extra_starts.append(a0)

        remaining = deadline - time.monotonic() - 30
        per_extra = min(150.0, remaining / max(1, len(extra_starts)))

        for idx, a0 in enumerate(extra_starts):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(per_extra, deadline - time.monotonic() - 15)
            if t_alloc < 20:
                break

            phase_deadline = time.monotonic() + t_alloc
            seq, score = optimize_full(a0, phase_deadline)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [extra {idx}] *** NEW BEST: {best_score:.6f} ***")

        # Final intensive pass on the absolute best
        if time.monotonic() < deadline - 40:
            print(f"\n=== Final intensive pass ===")
            t_alloc = deadline - time.monotonic() - 15
            phase_deadline = time.monotonic() + t_alloc
            seq, score = optimize_full(best_seq, phase_deadline)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
            print(f"  Final best: {best_score:.6f}")

    # Validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== RESULT: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        # Ensure all values are valid
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return [float(x) for x in best_seq]

    print("Falling back to uniform sequence.")
    return [1.0] * 256
