One more consideration: the simplex projection. Given a vector $v$ (not necessarily on the simplex), project it onto $\{x : \sum x_i = 1, x_i \geq 0\}$.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent
with soft-max smoothing, simplex projection, and local refinement.
Uses FFT-based convolution and gradient computation for efficiency.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def grad_soft_max_conv(a, epsilon):
    """Compute gradient of soft_max(conv(a,a), epsilon) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    # Soft-max weights
    c_shifted = c - c.max()  # for numerical stability
    w = np.exp(c_shifted / epsilon)
    w /= w.sum()
    # Cross-correlation: (a *star* w)_j = sum_u a_u * w_{u+j}
    # Using FFT: IFFT(FFT(a) * conj(FFT(w)))
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    xcorr = np.fft.irfft(A * np.conj(W), fft_size)[:n]
    g = 2.0 * xcorr
    return g, c, w

def simplex_project(v, target=1.0):
    """Project v onto {x : sum(x) = target, x >= 0}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - target) / rho > 0
    rho_val = np.max(cond.nonzero()[0]) + 1
    tau = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - tau, 0.0)

def soft_max_value(c, epsilon):
    """Compute soft max of c."""
    c_shifted = c - c.max()
    return c.max() + epsilon * np.log(np.exp(c_shifted / epsilon).sum())

def pgd_optimize(a0, time_limit, epsilon=0.01, verbose=False):
    """Projected gradient descent with soft-max smoothing."""
    a = simplex_project(a0.astype(np.float64).copy(), 1.0)
    n = len(a)
    
    c = fast_convolve(a)
    current_val = soft_max_value(c, epsilon)
    best_a = a.copy()
    best_val = current_val
    
    eta = 0.5  # initial step size
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        g, c, w = grad_soft_max_conv(a, epsilon)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Backtracking line search
        found = False
        for _ in range(30):
            a_trial = simplex_project(a - eta * g, 1.0)
            c_trial = fast_convolve(a_trial)
            val_trial = soft_max_value(c_trial, epsilon)
            if val_trial < current_val - 1e-15:
                a = a_trial
                current_val = val_trial
                found = True
                break
            eta /= 2
        
        if not found:
            # Try smaller eta
            eta = max(eta, 1e-8)
            a_trial = simplex_project(a - eta * g, 1.0)
            c_trial = fast_convolve(a_trial)
            val_trial = soft_max_value(c_trial, epsilon)
            if val_trial < current_val:
                a = a_trial
                current_val = val_trial
            else:
                break  # Stuck
        
        # Increase eta slightly if we succeeded with first try
        if found:
            eta = min(eta * 1.1, 2.0)

            best_a = a.copy()

            print(f"    PGD iter {iter_count}: val={current_val:.8f}, best={best_val:.8f}, eta={eta:.6f}")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent
with soft-max smoothing and adaptive local search.
Uses FFT-based convolution and cross-correlation for efficient gradient computation.
Multiple sequence lengths and annealing of the smoothing parameter are employed.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def soft_max_val(c, epsilon):
    """Stable soft-max of array c."""
    c_shift = c - c.max()
    return c.max() + epsilon * np.log(np.exp(c_shift / epsilon).sum())


def compute_grad(a, epsilon):
    """Compute gradient of soft_max(conv(a,a), epsilon) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_shift = c - c.max()
    w = np.exp(c_shift / epsilon)
    w /= w.sum()
    # Cross-correlation: (a star w)_j = sum_u a_u * w_{u+j}
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    xcorr = np.fft.irfft(A * np.conj(W), fft_size)[:n]
    g = 2.0 * xcorr
    return g, c


def simplex_project(v, target=1.0):
    """Project v onto {x : sum(x)=target, x>=0} (Euclidean projection)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho_val = idx[-1] + 1
    tau = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - tau, 0.0)


def pgd_step(a, epsilon, eta):
    """One projected gradient descent step. Returns new a and new soft-max value."""
    g, c = compute_grad(a, epsilon)
    a_new = simplex_project(a - eta * g, 1.0)
    c_new = fast_convolve(a_new)
    val_new = soft_max_val(c_new, epsilon)
    return a_new, val_new, c


def pgd_optimize(a0, time_limit, epsilon=0.01, verbose=False):
    """Project gradient descent with backtracking line search."""
    a = simplex_project(np.maximum(a0.astype(np.float64), 0), 1.0)
    c = fast_convolve(a)
    current_val = soft_max_val(c, epsilon)
    best_a = a.copy()
    best_val = current_val
    
    eta = 0.1
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        g, c = compute_grad(a, epsilon)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break
        
        # Backtracking line search
        improved = False
        for trial in range(40):
            a_new = simplex_project(a - eta * g, 1.0)
            c_new = fast_convolve(a_new)
            val_new = soft_max_val(c_new, epsilon)
            if val_new < current_val - 1e-15:
                a = a_new
                current_val = val_new
                c = c_new
                improved = True
                break
            eta *= 0.5
        
        if improved:
            if current_val < best_val:
                best_val = current_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            eta = min(eta * 1.05, 1.0)
        else:
            no_improve += 1
            # Try a small random perturbation to escape
            if no_improve > 50:
                pert = np.random.randn(len(a)) * 0.01
                a_pert = simplex_project(a + pert, 1.0)

                    best_a = a.copy()
                    best_val = val_pert
                    no_improve = 0
                else:
                    break
        
        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD [{elapsed:.1f}s] iter={iter_count} val={current_val:.10f} best={best_val:.10f} eta={eta:.6f}")
    
    return best_a, best_val
