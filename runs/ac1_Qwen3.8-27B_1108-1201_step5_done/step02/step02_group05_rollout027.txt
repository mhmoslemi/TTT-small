def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = n + m - 1

    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)

def project_simplex(v):
    """Project v onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

2. FFT convolution of a with itself: O(n log n)

4. FFT convolution of a with w_rev: O(n log n)

Actually, let me reconsider the whole approach. The key insight I want to exploit is:

**The objective is `2n * max(c) / (sum a)^2` where `c = a*a`.**

For normalized `a` (sum=1), this is `2n * max(c)`.

The sum of `c` is always 1 (since sum(a)=1). So we want to minimize the max while keeping the sum at 1. This is like making `c` as "flat" as possible.

The "flat" target is `c[k] = 1/(2n-1)` for all k. The "distance" from flat can be measured by the variance: `Var(c) = mean(c^2) - mean(c)^2`. Minimizing the max is approximately equivalent to minimizing the variance (when the distribution is close to uniform).

An alternative approach: **minimize the variance of c** (which is a smooth function of a) as a surrogate for minimizing the max. This gives a smooth optimization problem.

`Var(c) = (1/(2n-1)) * sum_k c[k]^2 - (1/(2n-1))^2`

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization
with softmax parameterization, soft-max loss with beta annealing, momentum,
variance pre-conditioning, spectral shaping, and local refinement.
Explores multiple sequence lengths (256, 512, 1024) with diverse initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def softmax_to_a(x):
    """Convert unconstrained params x to simplex point a (sum=1, a>=0)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    return a


def a_to_log(a):
    """Convert simplex point to log-space."""
    return np.log(np.maximum(a, 1e-30))


def compute_grad_soft_max(a, beta):
    """
    Compute gradient of soft_max(beta, c) w.r.t. x (softmax params).
    Returns (grad_x, objective_estimate).
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()

    # Soft max weights
    shifted = beta * (c - max_c)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()

    # Gradient of soft_max w.r.t. a:
    # d(soft_max)/da[i] = beta * sum_k w[k] * a[k-i] (valid indices)
    # = beta * cross_corr(a, w)[i]
    # cross_corr(a, w)[i] = sum_j a[j] * w[i+j]
    # = convolve(a, w_rev)[2n-2-i] where w_rev = w[::-1]
    w_rev = w[::-1]
    conv_result = fft_convolve(a, w_rev)
    grad_a = conv_result[n - 1:2 * n - 1][::-1] * beta

    # Convert to x-gradient: dx_loss = a * (grad_a - dot(a, grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)

    obj_est = 2.0 * n * max_c  # since sum(a)=1
    return grad_x, obj_est, c, w


def compute_grad_variance(a):
    """
    Compute gradient of sum(c^2) w.r.t. x.
    sum(c[k]^2) is a smooth surrogate for max(c).
    """
    n = len(a)
    c = fast_convolve(a)

    # d(sum c^2)/da[i] = 2 * sum_k c[k] * a[k-i]
    # = 2 * cross_corr(a, c)[i]
    # = 2 * convolve(a, c_rev)[2n-2-i]
    c_rev = c[::-1]
    conv_result = fft_convolve(a, c_rev)
    grad_a = 2.0 * conv_result[n - 1:2 * n - 1][::-1]

    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)

    obj_est = 2.0 * n * c.max()
    return grad_x, obj_est, c


def gradient_optimize(a0, time_limit, seed=42, use_variance_start=True):
    """
    Main gradient optimizer with softmax parameterization.
    Phase A: Minimize variance of convolution (smooth).
    Phase B: Minimize soft-max of convolution (targets peak).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a0 = np.maximum(a0, 1e-15).copy()
    a0 /= a0.sum()

    x = a_to_log(a0)
    best_a = a0.copy()
    best_obj = obj_value(a0)
    start = time.monotonic()

    # Phase A: Variance minimization (20% of time)
    if use_variance_start:
        var_time = time_limit * 0.2
        lr = 0.02
        momentum = 0.9
        v = np.zeros(n)
        no_improve = 0

        while time.monotonic() - start < start + var_time and no_improve < 200:
            a = softmax_to_a(x)
            grad_x, obj_est, c = compute_grad_variance(a)

            grad_norm = np.linalg.norm(grad_x)
            if grad_norm > 1e-15:
                v = momentum * v + grad_x / (grad_norm + 1e-10)
                x = x - lr * v * grad_norm

            a_new = softmax_to_a(x)
            obj_new = obj_value(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 50:
                    lr *= 0.95
                    v *= 0.5

        # Update x from best
        x = a_to_log(best_a)

    # Phase B: Soft-max minimization with beta annealing (80% of time)
    remaining_time = (start + time_limit) - time.monotonic()
    beta = 5.0
    lr = 0.03
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0

    # Batch processing for efficiency
    batch_size = 50

    while time.monotonic() - start < start + time_limit:
        t_rem = (start + time_limit) - time.monotonic()
        if t_rem < 0.1:
            break

        for _ in range(batch_size):
            if time.monotonic() - start >= start + time_limit:
                break
            a = softmax_to_a(x)
            grad_x, obj_est, c, w = compute_grad_soft_max(a, beta)

            grad_norm = np.linalg.norm(grad_x)
            if grad_norm > 1e-18:
                v = momentum * v + grad_x
                effective_lr = lr / (1.0 + 0.1 * grad_norm)
                x = x - effective_lr * v

        a = softmax_to_a(x)
        obj_new = obj_value(a)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Annealing schedule
        if no_improve == 50:
            beta *= 1.08
            lr *= 0.97
            v *= 0.7
        elif no_improve == 150:
            beta *= 1.15
            lr *= 0.9
            v *= 0.3
        elif no_improve == 300:
            # Random restart from best
            x = a_to_log(best_a) + 0.05 * rng.standard_normal(n)
            v = np.zeros(n)
            beta = max(3.0, beta * 0.7)
            lr = max(0.01, lr * 1.5)
            no_improve = 0
        elif no_improve > 500:
            break

    return best_a, best_obj


def spectral_shaping(a, time_limit, alpha=0.7):
    """
    Try to flatten the spectrum of a to reduce convolution peak.
    Modify |A(f)|^alpha to compress dynamic range.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    start = time.monotonic()

    for al in [0.5, 0.6, 0.7, 0.8, 0.9]:
        if time.monotonic() - start > time_limit:
            break
        A = np.fft.rfft(a, 2 * n)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        new_mag = np.power(np.maximum(mag, 1e-30), al)
        # Renormalize to keep sum(a) = 1
        new_mag /= np.sum(new_mag) * np.mean(new_mag) / (np.mean(mag) ** al) if np.mean(mag) > 0 else 1
        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        obj = obj_value(a_new)
        if obj < best_obj - 1e-8:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def local_refine(a, time_limit):
    """Local refinement: move mass from high-peak contributors to low contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a /= a.sum()
    best_obj = obj_value(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    stuck = 0

    while time.monotonic() - start < time_limit and stuck < 300:
        c = fast_convolve(a)
        k_star = int(np.argmax(c))

        # Compute per-element contribution to peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        danger = np.where(contrib > 0.4 * max_c)[0]
        safe = np.where(contrib < 0.05 * max_c)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            # Try next worst peak
            c_masked = c.copy()
            c_masked[k_star] = -np.inf
            if c_masked.max() < 0:
                break
            k_star = int(np.argmax(c_masked))
            contrib = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
            max_c = contrib.max()
            if max_c < 1e-15:
                break
            danger = np.where(contrib > 0.4 * max_c)[0]
            safe = np.where(contrib < 0.05 * max_c)[0]
            if len(danger) == 0 or len(safe) == 0:
                stuck += 1
                continue

        found = False
        n_d = min(12, len(danger))
        n_s = min(12, len(safe))
        d_sample = rng.choice(danger, size=n_d, replace=False)
        s_sample = rng.choice(safe, size=n_s, replace=False)

        for di in d_sample:
            if found:
                break
            for si in s_sample:
                if found or di == si:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.25, 0.4]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_value(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def get_parent(n):
    """Get parent sequence adapted to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_inits(n, rng, seed_offset=0):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
        # Perturbed parent
        for scale in [0.05, 0.1, 0.2]:
            pert = parent * (1.0 + scale * rng.standard_normal(n))
            pert = np.maximum(pert, 0.0)
            ps = pert.sum()
            if ps > 1e-10:
                inits[f'pert_{scale}'] = pert / ps

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        gs = g.sum()
        if gs > 1e-10:
            inits[f'gauss_{sf}'] = g / gs

    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0.01)
        inits['hann'] = hann / hann.sum()

    # U-shape
    u = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    u = np.maximum(u, 0.02)
    inits['u_shape'] = u / u.sum()

    # Inverted U
    inv_u = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))
    inv_u = np.maximum(inv_u, 0.02)
    inits['inv_u'] = inv_u / inv_u.sum()

    # Random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    return inits

            return (0, n)

        score = float(evaluate_sequence(a_opt.tolist()))

            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P1: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization
    phase2_budget = (total_budget - 20.0) * 0.45
    print(f"Phase 2: Deep optimization (budget={phase2_budget:.0f}s)")
    phase2_end = time.monotonic() + phase2_budget

    scan_results.sort(key=lambda x: x[0])
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 8:
            break

    if best_sequence is not None:
        best_n = len(best_sequence)
        if not any(t[0] == best_n for t in top_configs):
            top_configs.append((best_n, 'best', best_sequence.copy()))

    time_per_deep = max(5.0, phase2_budget / max(1, len(top_configs)))

    for n, name, a0 in top_configs:
        remaining_p2 = phase2_end - time.monotonic()
        remaining_total = deadline - time.monotonic()
        if remaining_p2 < 5 or remaining_total < 60:
            break
        t_alloc = min(time_per_deep, remaining_p2 - 3, remaining_total - 80)
        if t_alloc < 3:
            break

        a_opt, obj_opt = gradient_optimize(a0, time_limit=t_alloc, seed=seed + 100 + n)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P2: n={n}, {name}, score={score:.6f}")

        # Spectral shaping
        remaining_p2 = phase2_end - time.monotonic()
        if remaining_p2 > 5:
            ss_time = min(5.0, remaining_p2 * 0.2)
            a_ss, obj_ss = spectral_shaping(a_opt, time_limit=ss_time)
            score_ss = float(evaluate_sequence(a_ss.tolist()))
            if np.isfinite(score_ss) and score_ss < best_score:
                best_score = score_ss
                best_sequence = a_ss.copy()
                print(f"  P2 SS: n={n}, score={score_ss:.6f}")

        # Local refinement
        remaining_p2 = phase2_end - time.monotonic()
        if remaining_p2 > 5:
            ls_time = min(10.0, remaining_p2 * 0.3)
            a_ls, obj_ls = local_refine(a_opt, time_limit=ls_time)
            score_ls = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score_ls) and score_ls < best_score:
                best_score = score_ls
                best_sequence = a_ls.copy()
                print(f"  P2 LS: n={n}, score={score_ls:.6f}")

    # Phase 3: Polish the best
    if best_sequence is not None:
        phase3_budget = (total_budget - 20.0) * 0.30
        print(f"Phase 3: Polish (budget={phase3_budget:.0f}s)")
        phase3_end = min(time.monotonic() + phase3_budget, deadline)
        polish_count = 0

        while time.monotonic() < phase3_end and polish_count < 10:
            remaining = phase3_end - time.monotonic()
            if remaining < 10:
                break
            polish_count += 1

            # Gradient optimization from current best
            t_alloc = min(remaining * 0.5, 120.0)
            a_opt, obj_opt = gradient_optimize(best_sequence, time_limit=t_alloc,
                                              seed=seed + 500 + polish_count)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 grad: score={score:.6f}")

            # Spectral shaping
            remaining = phase3_end - time.monotonic()
            if remaining > 5:
                ss_time = min(10.0, remaining * 0.2)
                a_ss, _ = spectral_shaping(best_sequence, time_limit=ss_time)
                score_ss = float(evaluate_sequence(a_ss.tolist()))
                if np.isfinite(score_ss) and score_ss < best_score:
                    best_score = score_ss
                    best_sequence = a_ss.copy()
                    print(f"  P3 spec: score={score_ss:.6f}")

            # Local refinement
            remaining = phase3_end - time.monotonic()
            if remaining > 5:
                ls_time = min(20.0, remaining * 0.3)
                a_ls, _ = local_refine(best_sequence, time_limit=ls_time)
                score_ls = float(evaluate_sequence(a_ls.tolist()))
                if np.isfinite(score_ls) and score_ls < best_score:
                    best_score = score_ls
                    best_sequence = a_ls.copy()
                    print(f"  P3 local: score={score_ls:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL: n={len(best_sequence)}, score={best_score:.6f}")
    return [float(x) for x in best_sequence]
