"""
Optimization of non-negative step functions to minimize normalized peak 
autocorrelation, using L-BFGS with log-sum-exp smoothing and beta annealing,
combined with multi-start strategy and subgradient polish.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)


def a_from_x(x, S):
    """Convert log-parameters to non-negative sequence with given sum."""
    x_centered = x - np.max(x)  # For numerical stability
    return S * np.exp(x_centered) / np.sum(np.exp(x_centered))


def smooth_obj_and_grad_x(x, beta, S):
    """
    Compute smooth_max(conv(a,a))/beta and its gradient w.r.t. x,
    where a = S * softmax(x).
    """
    a = a_from_x(x, S)
    n = len(a)
    conv = fft_convolve(a, a)
    
    # Log-sum-exp
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    log_sum = beta * max_c + np.log(np.sum(np.exp(shifted)))
    obj = log_sum / beta
    
    # Weights
    w = np.exp(shifted)
    w /= np.sum(w)
    
    # Gradient of smooth_max w.r.t. a:
    # grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # Compute via cross-correlation
    w_rev = w[::-1]  # length 2n-1
    conv_result = fft_convolve(a, w_rev)  # length 3n-2
    grad_a = 2.0 * conv_result[2*n-2 : n-2 : -1]
    
    # Gradient w.r.t. x:
    # grad_x[j] = a[j] * (grad_a[j] - dot(a, grad_a) / S)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot / S)
    
    return obj, grad_x


def lbfgs_optimize(x_init, S, beta_schedule, max_iter_per_beta, time_limit):
    """
    Run L-BFGS with annealed beta.
    """
    t0 = time.monotonic()
    x = x_init.copy()
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_func(x):
            return smooth_obj_and_grad_x(x, beta, S)[0]
        
        def grad_func(x):
            return smooth_obj_and_grad_x(x, beta, S)[1]
        
        try:
            result = minimize(
                obj_func, x, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_beta, 'gtol': 1e-10}
            )
            x = result.x
        except Exception:
            break
    
    return x


def interpolate_seq(a, n_new):
    """Linearly interpolate a sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def subgradient_polish(a, S, time_limit, lr_init=0.01):
    """
    Polish using subgradient of the peak autocorrelation.
    Work in the space where sum(a) = S.
    """
    n = len(a)
    t0 = time.monotonic()
    a = a.copy()
    lr = lr_init
    steps = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        # Subgradient of conv[peak_idx] w.r.t. a:
        # d conv[peak_idx] / da[i] = 2*a[peak_idx - i] if 0 <= peak_idx - i < n
        grad = np.zeros(n)
        valid = np.arange(n)
        for i in valid:
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Step in the direction that reduces the peak, projected onto sum(a)=S, a>=0
        # Use a directional step: decrease a in the direction of grad
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s > 1e-12:
            a_new = a_new * S / s
        else:
            a_new = a.copy()
        
        new_conv = fft_convolve(a_new, a_new)
        new_peak = np.max(new_conv)
        
        if new_peak <= peak_val:
            a = a_new
            lr *= 1.05
        else:
            lr *= 0.9
        
        steps += 1
    
    return a
