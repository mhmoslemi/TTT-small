"""
Smoothed gradient optimization with FFT-accelerated cross-correlation,
momentum, beta annealing, and LP refinement for minimizing peak
autocorrelation of non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def cross_corr_fft(a, p):
    """Compute cross_correlation: result[i] = sum_j a[j] * p[i+j].
    a has length n, p has length 2n-1.
    Returns array of length n."""
    n = len(a)
    p_flip = p[::-1]
    conv = fft_convolve(a, p_flip)
    # conv has length n + (2n-1) - 1 = 3n-2
    # cross_corr[i] = conv[2n-2-i] for i=0,...,n-1
    # So we take conv[n-1:2n-1] and reverse
    return conv[n-1:2*n-1][::-1]


def smooth_gradient(a, beta):
    """Compute the smoothed gradient of max(conv(a,a)) w.r.t. a.
    Returns gradient vector of same length as a."""
    n = len(a)
    c = fft_convolve(a, a)
    # Softmax weights
    c_max = c.max()
    log_p = beta * (c - c_max)
    log_p -= log_p.max()  # numerical stability
    p = np.exp(log_p)
    p_sum = p.sum()
    p /= p_sum
    # Gradient: 2 * cross_corr(a, p)
    grad = 2.0 * cross_corr_fft(a, p)
    return grad

    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

        if elapsed >= time_limit - 2.0:
            break

        if anneal_iters == 0:
            progress = min(1.0, elapsed / max(1.0, time_limit * anneal_frac))
        beta = beta_start * (beta_end / beta_start) ** progress
        # beta = beta_start + (beta_end - beta_start) * progress

"""
Smoothed gradient optimization with FFT-accelerated cross-correlation,
momentum, beta annealing, LP refinement, and local search for minimizing
peak autocorrelation of non-negative sequences.
Key improvements: efficient FFT-based gradient, adaptive annealing,
multi-start across lengths, and LP finishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def cross_corr_top_k(a, c, top_k, beta):
    """Compute weighted gradient using top-K peaks of c.
    grad[i] = 2 * sum_k w_k * a[k-i] (for valid range)
    where w_k = softmax(beta * c[k]) over top-K indices.
    """
    n = len(a)
    top_k = min(top_k, 2 * n - 1)
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    c_top = c[top_idx].astype(np.float64)
    w = np.exp(beta * (c_top - c_top.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(top_k) / top_k
    else:
        w = w / s

    grad = np.zeros(n)
    for ki, wk in zip(top_idx, w):
        ki = int(ki)
        if wk < 1e-18:
            continue
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            grad[j] += wk * a[ki - j]
    return 2.0 * grad


def cross_corr_fft(a, p):
    """Compute cross_correlation: result[i] = sum_j a[j]*p[i+j] via FFT.
    a: length n, p: length 2n-1. Returns length n."""
    n = len(a)
    p_flip = p[::-1]
    conv = fft_convolve(a, p_flip)
    return conv[n - 1:2 * n - 1][::-1]


def optimize_smoothed_grad(f_init, time_limit, beta_start=2.0, beta_end=500.0,
                           lr_init=0.3, momentum=0.9, top_k=16,
                           verbose=True, seed=42):
    """Smoothed gradient descent with momentum and beta annealing."""
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    a = np.maximum(a, 0.001)
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_fast(a)
    v = np.zeros(n)
    lr = lr_init
    beta = beta_start
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    eval_interval = max(10, n // 10)

    while time.monotonic() - t0 < time_limit - 3.0:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit * 0.6))
        beta = beta_start * (beta_end / beta_start) ** progress
        top_k_cur = min(64, top_k + int(progress * 48))

        c = fft_convolve(a, a)
        grad = cross_corr_top_k(a, c, top_k_cur, beta)
        grad -= np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad
        a = a - lr * v
        a = np.maximum(a, 0.0)
        S = float(np.sum(a))
        if S < 1e-10:
            a = np.ones(n) / n
            v = np.zeros(n)
        else:
            a = a / S

        it += 1
        if it % eval_interval == 0:
            s = score_fast(a)
            if s < best_score - 1e-9:
                best_score = s
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if verbose and it % (eval_interval * 20) == 0:
                print(f"        [t={elapsed:5.1f}s it={it:7d} beta={beta:7.1f} lr={lr:.4f}] "
                      f"cur={s:.6f} best={best_score:.6f} no_imp={no_improve}")

            if no_improve > 50:
                lr *= 0.85
                v *= 0.7
                no_improve = 0

            if no_improve > 200:
                rng = np.random.default_rng(seed + it * 31)
                a += 0.015 * rng.standard_normal(n)
                a = np.maximum(a, 0.005)
                a = a / a.sum()
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0

    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(f_init, time_limit, verbose=True):
    """LP-based refinement: find direction b maximizing sum(b) s.t. conv(f,b) <= M."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = np.maximum(f, 0.001)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0

    A = build_conv_matrix(f)
    b_ub_arr = None

    while time.monotonic() - t0 < time_limit - 5.0:
        c = fft_convolve(f, f)
        M = float(np.max(c))

        if b_ub_arr is None or len(b_ub_arr) != 2 * n - 1:
            b_ub_arr = np.full(2 * n - 1, M)
        else:
            b_ub_arr[:] = M

        obj = -np.ones(n)
        bounds = [(0, None)] * n
        try:
            res = linprog(obj, A_ub=A, b_ub=b_ub_arr, bounds=bounds,
                          method='highs',
                          options={'time_limit': 5.0, 'presolve': True})
            if not res.success:
                break
            b_dir = res.x
        except Exception:
            break

        Sb = float(np.sum(b_dir))
        if Sb < 1e-12:
            break
        S = float(np.sum(f))
        b_dir = b_dir * (S / Sb)

        best_t = 0.05
        best_s = best_score
        for t_try in [0.03, 0.05, 0.1, 0.2, 0.3]:
            f_cand = (1.0 - t_try) * f + t_try * b_dir
            f_cand = np.maximum(f_cand, 0.001)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        if best_t == 0.05 and best_s >= best_score:
            break

        f = (1.0 - best_t) * f + best_t * b_dir
        f = np.maximum(f, 0.001)
        A = build_conv_matrix(f)
        it += 1

        if best_s < best_score:
            best_score = best_s
            best_f = f.copy()
            if verbose:
                print(f"        LP it={it} score={best_score:.6f}")

    return best_f, best_score


def gradient_nesterov(f_init, time_limit, verbose=True):
    """Nesterov-accelerated gradient descent with top-peak subgradient."""
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    it = 0
    beta = 10.0

    while time.monotonic() - t0 < time_limit - 3.0:
        it += 1
        t_k = 2.0 / (it + 1.0)

        c = fft_convolve(y, y)
        top_k = min(10, 2 * n - 1)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[top_idx]
        w = np.exp(beta * (c_top - c_top.max()))
        w /= w.sum()

        grad = np.zeros(n)
        for ki, wk in zip(top_idx, w):
            ki = int(ki)
            if wk < 1e-15:
                continue
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                grad[j] += wk * y[ki - j]
        grad = 2.0 * grad
        grad -= np.mean(grad)

        a_new = y - t_k * grad
        a_new = np.maximum(a_new, 0.0)
        S = float(np.sum(a_new))
        if S < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / S

        y = a_new + ((t_k - 1.0) / (it + 2.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        y = y / y.sum()
        a = a_new

        if it % 30 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose and it % 300 == 0:
                print(f"        Nest it={it} t_k={t_k:.4f} score={s:.6f} best={best_score:.6f}")

        if it % 500 == 0:
            beta *= 1.3

    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def local_search_polish(f_init, time_limit, verbose=True):
    """Targeted local search: reduce energy at peak-contributing positions."""
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    a = np.maximum(a, 0.001)
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit - 3.0:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        contribs = np.zeros(n)
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):
            contribs[j] = a[j] * a[peak_idx - j]

        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Decrease top contributors, increase bottom
        top_frac = max(3, n // 10)
        top_mask = np.zeros(n, dtype=bool)
        top_idx_c = np.argsort(contribs)[-top_frac:]
        top_mask[top_idx_c] = True

        bottom_frac = max(3, n // 15)
        bottom_mask = np.zeros(n, dtype=bool)
        valid = ~top_mask
        valid_idx = np.where(valid)[0]
        if len(valid_idx) > 0:
            bot_candidates = np.argsort(contribs[valid_idx])[:bottom_frac]
            bottom_mask[valid_idx[bot_candidates]] = True

        scale_down = np.exp(rng.normal(-0.02, 0.015, size=n))
        scale_down[top_mask] = np.exp(rng.normal(-0.04, 0.02, size=top_mask.sum()))
        scale_up = np.exp(rng.normal(0.02, 0.015, size=n))
        scale_up[bottom_mask] = np.exp(rng.normal(0.05, 0.02, size=bottom_mask.sum()))

        a_new = a * scale_down
        a_new[bottom_mask] = a[bottom_mask] * (1.0 + 0.03 * rng.random(bottom_mask.sum()))
        a_new = np.maximum(a_new, 0.001)

        s_new = score_fast(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
        elif rng.random() < 0.3:
            a = a_new

        it += 1
        if verbose and it % 200 == 0:
            print(f"        LS it={it} score={score_fast(a):.6f} best={best_score:.6f}")

    return best_a, best_score


def make_hann(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_smooth(n, rng):
    """Generate a smooth random non-negative sequence."""
    raw = rng.standard_normal(n)
    # Simple lowpass: moving average
    k = max(3, n // 32)
    kernel = np.ones(k) / k
    smooth = np.convolve(raw, kernel, 'same')
    a = 0.5 + 0.5 * smooth / (np.max(np.abs(smooth)) + 1e-10)
    t = np.arange(n)
    w = 0.7 + 0.3 * 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    a = a * w
    return np.clip(a, 0.01, None)


def make_bell(n, rng, sharpness=1.0):
    """Gaussian bell with noise."""
    t = np.arange(n) - n / 2.0
    a = np.exp(-sharpness * (t / (n / 4.0)) ** 2)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    for idx, (n, init_name, weight) in enumerate(tasks):
