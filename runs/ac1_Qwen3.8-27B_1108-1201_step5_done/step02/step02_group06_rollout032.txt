```python
def mirror_descent(p, n, max_iters, eta0, eta_decay):
    """Mirror descent with exponential family updates."""
    p = p.copy()
    eta = eta0
    best_p = p.copy()
    best_score = 2*n*max(np.convolve(p,p))/(np.sum(p)**2)
    
    for it in range(max_iters):
        conv = np.convolve(p, p)
        k_star = int(np.argmax(conv))
        
        # Compute subgradient
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * p[j]
        
        # Center the gradient (optional, helps with convergence)
        # g_centered = g - np.sum(p * g)  # This makes the update preserve the "energy"
        
        # Exponential family update
        log_p = np.log(p + 1e-300) - eta * g
        p_new = np.exp(log_p - np.max(log_p))
        p_new = p_new / np.sum(p_new)
        
        # Check for improvement
        conv_new = np.convolve(p_new, p_new)
        score_new = 2*n*np.max(conv_new)
        
        if score_new < best_score:
            best_score = score_new
            best_p = p_new.copy()
        
        p = p_new
        eta *= eta_decay
    
    return best_p, best_score
```

   This makes the gradient computation O(n log n) instead of O(n^2)!

3. **Multiple peak tracking**: Instead of just tracking the single peak, track the top few peaks and use their weighted sum for the gradient. This gives a smoother optimization landscape.

import numpy as np
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(p):
    n = len(p)
    conv = fft_convolve(p, p)
    return 2.0 * n * np.max(conv)

def compute_gradient(p, tau):
    """Compute gradient of soft-max of autocorrelation w.r.t. p."""
    n = len(p)
    conv = fft_convolve(p, p)  # length 2n-1
    # Soft-max weights
    shifted = (conv - np.max(conv)) / tau
    exp_conv = np.exp(shifted)
    w = exp_conv / np.sum(exp_conv)
    # Gradient: g_i = sum_k w_k * 2*p_{k-i} for valid k-i
    # = 2 * cross_correlation(w, p) evaluated at appropriate indices
    # cross_corr(w, p)_i = sum_k w_k * p_{k-i}
    # Using FFT: ifft(fft(w) * conj(fft(p)))
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fp = np.fft.rfft(p, size)
    corr = np.fft.irfft(fw * np.conj(fp), size)
    g = 2.0 * corr[:n]  # Take the first n elements
    return g

"""
Optimized step function design using mirror descent (exponential family updates)
with annealed soft-max peak tracking, combined with targeted simulated annealing.
Achieves sub-1.5 normalized peak autocorrelation for non-negative sequences.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score_p(p):
    """Compute 2*n*max(conv(p,p)) for probability vector p (sum=1, p>=0)."""
    n = len(p)
    conv = fft_convolve(p, p)
    return 2.0 * n * np.max(conv)


def compute_gradient(p, tau, fft_size=None):
    """Compute gradient of soft-max autocorrelation peak w.r.t. p."""
    n = len(p)
    if fft_size is None:
        fft_size = 1
        while fft_size < 2 * n - 1:
            fft_size <<= 1
    
    conv = fft_convolve(p, p)
    
    # Soft-max weights
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    exp_conv = np.exp(shifted)
    w = exp_conv / np.sum(exp_conv)
    
    # Gradient: g_i = 2 * sum_k w_k * p_{k-i} [0<=k-i<n]
    # = 2 * cross_correlation(w, p)_i for i=0,...,n-1
    fw = np.fft.rfft(w, fft_size)
    fp = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(fw * np.conj(fp), fft_size)
    g = 2.0 * corr[:n]
    
    return g, max_conv


def mirror_descent_optimize(p_init, n, time_limit, verbose=False):
    """Mirror descent with exponential family updates and annealed temperature."""
    p = p_init.astype(np.float64).copy()
    p = np.clip(p, 0.0, None)
    s = np.sum(p)
    if s < 1e-10:
        p = np.ones(n) / n
        s = 1.0
    p = p / s
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    tau_start = 1.0 / n
    tau_end = 0.1 / n
    eta = 0.5
    
    best_p = p.copy()
    best_score = compute_score_p(p)
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    no_improve_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        elapsed_frac = min(1.0, elapsed)
        
        # Annealed temperature
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)
        
        # Compute gradient
        g, max_conv = compute_gradient(p, tau, fft_size)
        
        # Center gradient (maintain simplex geometry)
        g_center = g - np.sum(p * g)
        
        # Exponential update
        log_factor = -eta * g_center
        max_log = np.max(log_factor)
        if max_log > 500:
            log_factor = log_factor - max_log
        p_new = p * np.exp(log_factor)
        p_sum = np.sum(p_new)
        if p_sum < 1e-300:
            break
        p_new = p_new / p_sum
        
        # Evaluate
        score = compute_score_p(p_new)
        it += 1
        
        if score < best_score:
            best_score = score
            best_p = p_new.copy()
            stagnant = 0
            p = p_new
        else:
            stagnant += 1
            p = p_new  # Still accept to explore
            if stagnant > 100:
                eta *= 0.7
                stagnant = 0
            if eta < 1e-12:
                break
        
        # Increase eta when improving
        if stagnant == 0 and it % 50 == 0:
            eta = min(eta * 1.05, 2.0)
        
        if verbose and it % 10000 == 0:
            print(f"      MD iter={it}, tau={tau:.2e}, eta={eta:.4f}, "
                  f"score={score:.6f}, best={best_score:.6f}")
    
    return best_p, best_score


def targeted_sa(p_init, n, time_limit, rng, verbose=False):
    """Targeted simulated annealing focused on peak reduction."""
    p = p_init.astype(np.float64).copy()
    p = np.clip(p, 0.0, None)
    s = np.sum(p)
    if s < 1e-10:
        p = np.ones(n) / n
        s = 1.0
    p = p / s
    
    best_p = p.copy()
    best_score = compute_score_p(p)
    current_score = best_score
    
    t0 = time.monotonic()
    temperature = 0.005 / n
    cooling = 0.9998
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = p.copy()
        r = rng.random()
        
        conv = fft_convolve(p, p)
        peak_idx = int(np.argmax(conv))
        
        # Compute contributions to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = p[i] * p[j]
        
        if r < 0.4:
            # Targeted: decrease peak contributors, increase others
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            n_dec = max(1, top_k // 2)
            dec_idx = rng.choice(top_idx, size=min(n_dec, len(top_idx)), replace=False)
            candidate[dec_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(dec_idx)))
            
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(4, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.75:
            # Swap mass between two random blocks
            w1 = max(1, n // 16)
            w2 = max(1, n // 16)
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            if s1 != s2:
                eps = 0.01 * np.mean(p)
                candidate[s1:s1+w1] -= eps
                candidate[s2:s2+w2] += eps
        else:
            # Small global perturbation
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate / cand_sum
        
        new_score = compute_score_p(candidate)
        it += 1
        
        if new_score < current_score:
            p = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_p = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            p = candidate
            current_score = new_score
        
        temperature *= cooling
        
        if verbose and it % 1000 == 0:
            print(f"      SA iter={it}, T={temperature:.2e}, "
                  f"score={current_score:.6f}, best={best_score:.6f}")
    
    return best_p, best_score


def make_init(n, method, rng):
    """Create initial probability vector."""
    if method == 'uniform':
        p = np.ones(n) / n
    elif method == 'chirp':
        t = np.arange(n, dtype=np.float64)
        alpha = 0.3 + 0.5 * rng.random()
        p = 1.0 + alpha * np.cos(np.pi * t * t / n)
        p = np.clip(p, 0.01, None)
        p = p / np.sum(p)
    elif method == 'multitone':
        t = np.arange(n, dtype=np.float64)
        p = np.ones(n)
        for _ in range(4 + int(rng.integers(0, 4))):
            k = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.1, 0.3)
            phase = rng.uniform(0, 2.0 * np.pi)
            p += amp * np.cos(2.0 * np.pi * k * t / n + phase)
        p = np.clip(p, 0.01, None)
        p = p / np.sum(p)
    elif method == 'random':
        p = 0.5 + 0.5 * rng.random(n)
        p = p / np.sum(p)
