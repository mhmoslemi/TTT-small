def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Cosine window
    cos_w = np.cos(np.pi * (x + 0.5) / n) ** 2
    cos_w = np.abs(cos_w)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Triangular
    tri = 1.0 - np.abs(x - (n-1)/2) / ((n-1)/2)
    tri = np.maximum(tri, 0.1)
    tri /= tri.sum()
    inits['triangular'] = tri
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - n/2) / (n*0.2))**2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) >= n:
                pp = p[:n]
                ps2 = pp.sum()
                if ps2 > 0:
                    inits['parent_trunc'] = pp / ps2
            p_n = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            p_n = np.maximum(p_n, 0)
            ps3 = p_n.sum()
            if ps3 > 0:
                inits['parent_resample'] = p_n / ps3
    
    return inits

def peak_reduction(a, n_iterations=1000, verbose=False):
    """
    Local optimization: iteratively find the peak of auto-convolution and
    reduce it by redistributing mass.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    rng = np.random.default_rng(123)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for _ in range(n_iterations):
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        
        # Find contributing pairs
        lo = max(0, k_max - n + 1)
        hi = min(n, k_max + 1)
        
        if hi - lo <= 1:
            break
        
        # Find the pair with maximum product
        # Product at position k_max: sum_{j=lo}^{hi-1} a[j] * a[k_max-j]
        # We want to reduce this by moving mass from a high-contribution element
        # to a position that doesn't contribute to this peak
        
        # Simple strategy: find j in [lo, hi-1] with max a[j], 
        # and move a small amount to a position far from the peak
        j_best = lo + int(np.argmax(a[lo:hi] * a[k_max-hi+1:k_max-lo+1][::-1]))
        
        # Find a target position far from contributing to this peak
        # A position that doesn't pair with any element to form k_max
        # Position p contributes to k_max if p + q = k_max for some q in [0, n-1]
        # i.e., q = k_max - p must be in [0, n-1]
        # So p contributes if 0 <= k_max - p < n, i.e., k_max - n + 1 <= p <= k_max
        # Non-contributing positions: p < k_max - n + 1 or p > k_max
        
        # But we also need the target to not increase other peaks too much
        # Simple heuristic: move to a random position
        target = rng.integers(0, n)
        
        # Make sure target != j_best
        attempts = 0
        while target == j_best and attempts < 10:
            target = rng.integers(0, n)
            attempts += 1
        
        if target == j_best:
            continue
        
        # Transfer a small amount
        amount = a[j_best] * 0.01
        if amount < 1e-15:
            break
        
        a_new = a.copy()
        a_new[j_best] -= amount
        a_new[target] += amount
        
        if a_new[j_best] < 0:
            a_new[j_best] = 0
            a_new[target] += a[j_best]
            a_new[j_best] = 0
            continue
        
        # Check if this improves
        conv_new = fast_convolve(a_new)
        max_new = conv_new.max()
        max_old = conv.max()
        
        if max_new < max_old:
            a = a_new
            if 2.0 * n * max_new < best_obj:
                best_obj = 2.0 * n * max_new
                best_a = a.copy()
    
    return best_a

"""
LP-based alternating direction method for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: fixing one sequence in the auto-convolution makes the max-convolution
constraint linear, enabling efficient LP-based direction finding. Combined with
adaptive step sizing, multi-phase search, peak reduction polishing, and random restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_constraint_matrix(f, n):
    """Build (2n-1) x n matrix where row k gives cross-conv f*b at position k."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = np.arange(lo, hi)
        A[k, lo:hi] = f[k - idx]
    return A


def solve_lp_direction(f, M, n):
    """Solve max sum(b) s.t. cross_conv(f,b)_k <= M for all k, b >= 0."""
    c = -np.ones(n)
    A = build_constraint_matrix(f, n)
    b_vec = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_vec, method='highs',
                  bounds=[(0, None)] * n,
                  options={'presolve': True})
    
    if res.success and res.x is not None:
        return res.x
    return None

        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = float(conv.max())
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    t = t_init
    rng = np.random.default_rng(seed)
    iteration = 0
    consecutive_fail = 0
    
    while time.monotonic() - t0 < time_limit:
        tension = tension_start + (tension_end - tension_start) * min(1.0, iteration / 500.0)
        M_tight = M * tension
        
        direction = solve_lp_direction(f, M_tight, n)
        if direction is None:
            direction = solve_lp_direction(f, M, n)
            if direction is None:
                break
        
        d_sum = direction.sum()
        if d_sum < 1e-15:
            break
        direction = direction / d_sum
        
        f_new = (1 - t) * f + t * direction
        f_new = np.maximum(f_new, 0.0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            break
        f_new = f_new / f_new_sum
        
        conv_new = fast_convolve(f_new)
        M_new = float(conv_new.max())
        obj_new = 2.0 * n * M_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            t = min(0.8, t * 1.03)
            consecutive_fail = 0
        else:
            t *= 0.7
            consecutive_fail += 1
            if consecutive_fail > 30 or t < 0.005:
                # Reset and inject noise
                t = max(t_init * 0.5, 0.1)
                noise = rng.standard_normal(n) * noise_scale
                f = f + noise
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                else:
                    f = np.ones(n) / n
                conv = fast_convolve(f)
                M = float(conv.max())
                consecutive_fail = 0
        
        iteration += 1
        if iteration % noise_every == 0 and noise_scale > 0:
            noise = rng.standard_normal(n) * noise_scale * 0.3
            f = f + noise
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-15:
                f = f / fs
            conv = fast_convolve(f)
            M = float(conv.max())
    
    return best_f, best_obj


def peak_reduction(a, max_iterations=3000, budget_s=60.0, seed=7):
    """
    Local peak shaving: find the max auto-correlation index and reduce it
    by transferring mass from contributing positions to non-contributing ones.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    count = 0
    stall = 0
    
    while count < max_iterations and time.monotonic() - t0 < budget_s:
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        
        lo = max(0, k_max - n + 1)
        hi = min(n, k_max + 1)
        
        if hi - lo <= 1:
            break
        
        # Products of pairs contributing to position k_max
        j_idx = np.arange(lo, hi)
        products = a[j_idx] * a[k_max - j_idx]
        
        # Find the pair with max product
        j_best = j_idx[np.argmax(products)]
        
        # Find non-contributing target positions (positions p where k_max - p is out of [0, n-1])
        # p contributes to k_max iff 0 <= k_max - p < n iff k_max - n + 1 <= p <= k_max
        # Non-contributing: p < k_max - n + 1 or p > k_max
        all_pos = np.arange(n)
        non_contrib = all_pos[~((all_pos >= lo) & (all_pos < hi))]
        
        if len(non_contrib) == 0:
            break
        
        # Choose target: prefer positions far from the peak
        target = non_contrib[rng.integers(0, len(non_contrib))]
        
        if target == j_best:
            target = non_contrib[0]
            if target == j_best:
                continue
        
        # Transfer amount
        frac = rng.uniform(0.005, 0.03)
        amount = a[j_best] * frac
        
        a_new = a.copy()
        a_new[j_best] -= amount
        a_new[target] += amount
        
        if a_new[j_best] < 0:
            amount = a[j_best] * 0.5
            a_new = a.copy()
            a_new[j_best] -= amount
            a_new[target] += amount
        
        conv_new = fast_convolve(a_new)
        max_new = float(conv_new.max())
        
        if max_new < conv[k_max] - 1e-12:
            a = a_new
            obj = 2.0 * n * max_new
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        count += 1
        if stall > 500:
            break
    
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Cosine window
    cos_w = np.cos(np.pi * (x + 0.5) / n) ** 2
    cos_w = np.abs(cos_w)
    cos_w = np.maximum(cos_w, 1e-8)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Triangular
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri = np.maximum(tri, 0.05)
    tri /= tri.sum()
    inits['triangular'] = tri
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n / 2.0) / (n * 0.25)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Bimodal edges
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) >= n:
                pp = p[:n]
                ps2 = pp.sum()
                if ps2 > 0:
                    inits['parent_trunc'] = pp / ps2
            p_n = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(p)), p)
            p_n = np.maximum(p_n, 0)
            ps3 = p_n.sum()
            if ps3 > 0:
                inits['parent_resample'] = p_n / ps3
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search using LP-based alternating direction, peak reduction,
    and random restarts to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    # === Phase 1: Exploration over lengths and initializations ===
    print("=== Phase 1: Exploration ===")
    lengths = [128, 256, 512, 1024]
    phase1_deadline = time.monotonic() + 380.0
    
    results = []
    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = generate_inits(n, rng, parent)
        init_names = list(inits.keys())
        
        # Limit number of inits per length
        if len(init_names) > 5:
            priority = ['parent_resample', 'parent_trunc', 'uniform', 'random', 'cosine']
            init_names = [nm for nm in priority if nm in inits] + \
                         [nm for nm in init_names if nm not in priority]
            init_names = init_names[:5]
        
        time_per = max(8.0, 320.0 / (len(lengths) * len(init_names)))
        
        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(time_per, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            
            a0 = inits[name].copy()
            a_opt, obj_opt = run_alternating_lp(a0, time_limit=t_alloc, t_init=0.3,
                                                 seed=rng.integers(0, 2**31))
            score = float(evaluate_sequence(a_opt.tolist()))
            results.append((score, n, name, a_opt.copy()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  n={n}, {name}: {score:.6f}")
    
    results.sort(key=lambda x: x[0] if np.isfinite(x[0]) else 1e9)
    print(f"Phase 1 best: {best_score:.6f}")
    
    # === Phase 2: Deep optimization on top results ===
    print("=== Phase 2: Deep Optimization ===")
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Deep optimize top unique (length) results
    seen_lengths = set()
    top_to_optimize = []
    for score, n, name, a_opt in results:
        if n not in seen_lengths and np.isfinite(score):
            top_to_optimize.append((n, a_opt.copy()))
            seen_lengths.add(n)
        if len(top_to_optimize) >= 4:
            break
    
    # Always include best
    if len(best_sequence) != top_to_optimize[0][0] if top_to_optimize else True:
        top_to_optimize.append((len(best_sequence), best_sequence.copy()))
    
    phase2_deadline = time.monotonic() + 500.0
    for n, a0 in top_to_optimize:
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(100.0, phase2_deadline - time.monotonic() - 20)
        if t_alloc < 10:
            break
        
        a_opt, obj_opt = run_alternating_lp(a0, time_limit=t_alloc, t_init=0.2,
                                             noise_scale=0.001, seed=rng.integers(0, 2**31))
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep n={n}: {score:.6f}")
    
    # === Phase 3: Multi-restart at best length ===
    print("=== Phase 3: Multi-Restart ===")
    best_n = len(best_sequence)
    phase3_deadline = time.monotonic() + 500.0
    
    n_restarts = 8
    t_per = max(15, (phase3_deadline - time.monotonic() - 20) / max(1, n_restarts))
    
    for r in range(n_restarts):
        if time.monotonic() > phase3_deadline - 15:
            break
        
        # Different perturbation strategies
        if r == 0:
            a_pert = best_sequence + rng.standard_normal(best_n) * 0.03
        elif r == 1:
            a_pert = best_sequence * np.exp(rng.standard_normal(best_n) * 0.1)
        elif r == 2:
            # Flip a random segment
            a_pert = best_sequence.copy()
            start = rng.integers(0, best_n - 10)
            end = min(start + rng.integers(5, best_n // 3), best_n)
            a_pert[start:end] *= 0.5
        elif r == 3:
            # Smooth version + noise
            from scipy.ndimage import gaussian_filter1d
            a_smooth = gaussian_filter1d(best_sequence, sigma=5)
            a_pert = a_smooth + rng.standard_normal(best_n) * 0.01
        else:
            a_pert = best_sequence + rng.standard_normal(best_n) * (0.02 + 0.01 * r)
        
        a_pert = np.maximum(a_pert, 0.0)
        a_pert_sum = a_pert.sum()
        if a_pert_sum < 1e-15:
            continue
        a_pert = a_pert / a_pert_sum
        
        a_opt, obj_opt = run_alternating_lp(a_pert, time_limit=t_per, t_init=0.15,
                                             noise_scale=0.001, seed=rng.integers(0, 2**31))
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Restart {r}: {score:.6f}")
    
    # === Phase 4: Peak reduction polishing ===
    print("=== Phase 4: Peak Reduction ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 15:
            t_peak = min(60.0, remaining * 0.5)
            a_polished = peak_reduction(best_sequence, max_iterations=5000,
                                        budget_s=t_peak, seed=rng.integers(0, 2**31))
            score = float(evaluate_sequence(a_polished.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_polished.copy()
                print(f"  Peak: {score:.6f}")
    
    # === Phase 5: Final LP polish ===
    print("=== Phase 5: Final LP Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 20:
            a_opt, obj_opt = run_alternating_lp(best_sequence, time_limit=remaining * 0.6,
                                                t_init=0.08, noise_scale=0.0003,
                                                tension_start=0.99, tension_end=0.97,
                                                seed=rng.integers(0, 2**31))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f}")
            
            # One more peak reduction
            remaining2 = deadline - time.monotonic() - 5.0
            if remaining2 > 10:
                a_polished = peak_reduction(best_sequence, max_iterations=3000,
                                            budget_s=remaining2 * 0.8,
                                            seed=rng.integers(0, 2**31))
                score = float(evaluate_sequence(a_polished.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_polished.copy()
                    print(f"  Final Peak: {score:.6f}")
    
    # Final safety
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
