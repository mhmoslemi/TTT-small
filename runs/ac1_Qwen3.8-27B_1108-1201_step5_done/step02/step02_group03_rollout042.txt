```python
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp

def compute_obj(a):
    """Compute objective for a sequence."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def compute_obj_smooth(a, K):
    """Smooth version using logsumexp."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    # logsumexp of K*c / K gives smooth max
    smooth_max = float(logsumexp(K * c) / K)
    return float(2.0 * n * smooth_max / (S * S))

def gradient_smooth(a, K):
    """Gradient of smooth objective."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 0.01:
        return np.zeros(n), float('inf')
    
    # Soft weights
    c_shifted = K * (c - c.max())
    w = np.exp(c_shifted)
```

```python
def flat_front_sequence(n):
    """Compute the sequence where conv(a,a)[0] = ... = conv(a,a)[n-1] = C."""
    a = np.zeros(n)
    a[0] = 1.0  # Set scale
    C = a[0]**2  # = 1
    
    for k in range(1, n):
        # conv(a,a)[k] = sum_{i} a[i]*a[k-i] = C
        # The terms where both indices are < k are already determined
        # The new term is a[0]*a[k] (from i=0, k-i=k)
        # Wait, let me think more carefully.
        # conv(a,a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
        # For k < n: = sum_{i=0}^{k} a[i]*a[k-i]
        # = a[0]*a[k] + a[1]*a[k-1] + ... + a[k]*a[0]
        # = 2*a[0]*a[k] + 2*a[1]*a[k-1] + ... (if k is even, the middle term is a[k/2]^2)
        # We want this to equal C.
        # The only unknown is a[k] (all a[0] through a[k-1] are known).
        # conv(a,a)[k] = 2*a[0]*a[k] + sum_{i=1}^{k-1} a[i]*a[k-i] (with the middle term halved if k is even)
        # Actually: sum_{i=0}^{k} a[i]*a[k-i] = a[0]*a[k] + a[k]*a[0] + sum_{i=1}^{k-1} a[i]*a[k-i]
        # = 2*a[0]*a[k] + sum_{i=1}^{k-1} a[i]*a[k-i]
        # (when i and k-i are different, which is the case for i=1 to k-1 when k >= 2)
        # For k=1: sum = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]
        # So: 2*a[0]*a[k] + sum_{i=1}^{k-1} a[i]*a[k-i] = C
        # a[k] = (C - sum_{i=1}^{k-1} a[i]*a[k-i]) / (2*a[0])
        
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k-i]
        a[k] = (C - s) / (2.0 * a[0])
    
    return a
```

```python
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # a[j] appears in rows j through j+n-1
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = a
```

def flat_front(n):
    """Generate flat-front sequence: conv(a,a)[k]=1 for k=0,...,n-1."""
    a = np.zeros(n)
    a[0] = 1.0

        for i in range(1, k):
