Multi-phase optimization for minimizing peak autocorrelation score of non-negative sequences.
Combines LP-based direction finding, subgradient descent with LP-dual weighting,
Newton refinement on active sets, and multi-scale initialization.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A_sparse(f):
    """Build A where A@b = conv(f,b). Returns as list of rows for linprog."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, timeout=3.0):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0. Returns b or None."""
    n = len(f)
    A = build_A_sparse(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_dual(f, M, timeout=3.0):
    """Solve dual: min M*sum(y) s.t. A^T y >= 1, y >= 0. Returns y or None."""
    n = len(f)
    A = build_A_sparse(f)
    c_obj = np.full(2 * n - 1, M)
    A_eq = A.T  # (n, 2n-1)
    b_eq = np.ones(n)
    try:
        res = linprog(c_obj, A_ub=-A.T, b_ub=-np.ones(n),
                      bounds=[(0, None)] * (2 * n - 1),
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def line_search_1d(f, direction, score_f0, max_trials=20):
    """Parabolic line search for f + t*direction. Returns best t and score."""
    best_t = 0.0
    best_s = score_f0
    t = 0.1
    # First find a bracket
    s1 = score_fast(np.clip(f + t * direction, 0.0, 1000.0))
    if s1 >= best_s:
        t = 0.05
        s1 = score_fast(np.clip(f + t * direction, 0.0, 1000.0))
    if s1 >= best_s:
        t = 0.02
        s1 = score_fast(np.clip(f + t * direction, 0.0, 1000.0))
    if s1 >= best_s:
        return 0.0, best_s

    best_t = t
    best_s = s1
    t_prev = t * 0.5
    s_prev = score_fast(np.clip(f + t_prev * direction, 0.0, 1000.0))

    # Parabolic interpolation / golden section
    for _ in range(max_trials):
        # Try to narrow down
        if best_t == 0:
            break
        t_try = best_t * 1.5
        s_try = score_fast(np.clip(f + t_try * direction, 0.0, 1000.0))
        if s_try > best_s:
            # The minimum is between best_t and t_try
            t_half = (best_t + t_try) * 0.5
            s_half = score_fast(np.clip(f + t_half * direction, 0.0, 1000.0))
            if s_half < best_s:
                best_t = t_half
                best_s = s_half
            break
        else:
            best_t = t_try
            best_s = s_try

    return best_t, best_s


def weighted_subgradient(f, top_k=15, use_lp_dual=False):
    """Compute weighted subgradient using top-k peaks or LP dual weights."""
    n = len(f)
    c = fft_convolve(f, f)
    M = float(np.max(c))
    idx = np.argsort(c)[-top_k:]

    if use_lp_dual and M > 0:
        y = solve_lp_dual(f, M, timeout=1.0)
        if y is not None:
            y_pos = np.maximum(y, 0)
            y_sum = y_pos.sum()
            if y_sum > 1e-12:
                weights = y_pos / y_sum
            else:
                weights = np.zeros(len(c))
                for ki in idx:
                    weights[int(ki)] = 1.0 / top_k
        else:
            weights = np.zeros(2 * n - 1)
            for ki in idx:
                weights[int(ki)] = c[int(ki)] / max(np.sum(c[idx]), 1e-12)
    else:
        weights = np.zeros(2 * n - 1)
        c_top = c[idx]
        w_sum = c_top.sum()
        if w_sum > 1e-12:
            for ki in idx:
                weights[int(ki)] = c[int(ki)] / w_sum
        else:
            for ki in idx:
                weights[int(ki)] = 1.0 / top_k

    g = np.zeros(n)
    for ki in np.where(weights > 1e-10)[0]:
        w = weights[ki]
        if w < 1e-10:
            continue
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            g[j] += w * 2.0 * f[ki - j]

    return g, c, idx

def newton_active_set(f_init, time_limit, verbose=False):
    """Newton method on the active set (top-k autocorrelation peaks)."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    top_k = min(10, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        idx = np.argsort(c)[-top_k:]

        # Build Hessian of sum_{k in S} c_k
        H = np.zeros((n, n), dtype=np.float64)
        grad = np.zeros(n)
        for ki in idx:
            ki = int(ki)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                other = ki - j
                if j == other:
                    H[j, j] += 1.0
                    grad[j] += 2.0 * f[other]
                else:
                    H[j, other] += 1.0
                    H[other, j] += 1.0
                    grad[j] += f[other]
                    grad[other] += f[j]

        # Regularize and solve
        H_reg = H + 1e-6 * np.eye(n)
        try:
            delta = np.linalg.solve(H_reg, -grad)
        except np.linalg.LinAlgError:
            try:
                delta = np.linalg.lstsq(H_reg, -grad, rcond=None)[0]
            except Exception:
                break

        # Line search
        score_cur = score_fast(f)
        best_t = 0.0
        best_s = score_cur
        for t_try in [0.3, 0.1, 0.05, 0.5]:
            f_cand = f + t_try * delta
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        if best_t > 0:
            f = f + best_t * delta
            f = np.clip(f, 0.0, 1000.0)

            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        it += 1
        if verbose and it % 30 == 0:

def make_hann_init(n, rng):

    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    a = 0.2 + 0.8 * w
    a += 0.03 * rng.standard_normal(n)

def make_kaiser_init(n, rng, beta=4.0):

    if n > 1:
        from scipy.special import i0
        alpha = (t - (n - 1) / 2.0) / ((n - 1) / 2.0)
        w = i0(beta * np.sqrt(1 - alpha**2)) / i0(beta)
    else:
        w = np.ones(n)
    a = 0.1 + 0.9 * w
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def spectral_polish(f_init, time_limit, verbose=False):
    """Frequency-domain refinement: adjust phases to reduce autocorrelation peak."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break

        F = rfft(f, size)
        # Small phase perturbation
        phase = np.angle(F)
        mag = np.abs(F)
        dphase = rng.normal(0, 0.02, size=size // 2 + 1)
        dphase[0] = 0  # Keep DC
        new_F = mag * np.exp(1j * (phase + dphase))
        f_cand = irfft(new_F, size)[:n]
        f_cand = np.clip(f_cand, 0.01, 1000.0)

        s_cand = score_fast(f_cand)
        if s_cand < best_score:
            best_score = s_cand
            best_f = f_cand.copy()
            f = f_cand

        it += 1
        if verbose and it % 100 == 0:
            print(f"    [spectral {time.monotonic()-t0:6.1f}s] it={it} best={best_score:.6f}")

        if it > 5000:
            break

    return best_f, best_score


def upsample_seq(f, factor=2):
    """Upsample a sequence by inserting interpolated values."""
    n = len(f)
    new_n = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, f)


def downsample_seq(f, factor=2):
    """Downsample by averaging pairs."""
    n = len(f)
    new_n = n // factor
    if new_n == 0:
        return f[:1]
    f_padded = np.zeros(new_n * factor)
    f_padded[:n] = f
    return f_padded.reshape(new_n, factor).mean(axis=1)


def combined_optimize(f_init, time_limit, verbose=False):
    """Combined LP + gradient + Newton optimization in cycles."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Phase A: LP steps (expensive but effective)
        lp_time = min(60.0, remaining * 0.4)
        if lp_time > 5:
            f, s = lp_optimize(f, lp_time, verbose=False)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                stall = 0
            else:
                stall += 1

        if time.monotonic() - t0 >= time_limit - 10:
            break

        # Phase B: Gradient steps (cheap, many iterations)
        remaining = time_limit - (time.monotonic() - t0)
        grad_time = min(30.0, remaining * 0.4)
        if grad_time > 5:
            f, s = gradient_optimize(f, grad_time, verbose=False)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                stall = 0
            else:
                stall += 1

        if time.monotonic() - t0 >= time_limit - 10:
            break

        # Phase C: Newton on active set
        remaining = time_limit - (time.monotonic() - t0)
        newton_time = min(20.0, remaining * 0.3)
        if newton_time > 5:
            f, s = newton_active_set(f, newton_time, verbose=False)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                stall = 0
            else:
                stall += 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    [combined {elapsed:6.1f}s] stall={stall} best={best_score:.6f}")

        if stall >= 6:
            # Big restart with noise
            rng = np.random.default_rng(int(time.monotonic() * 10000))
            f = best_f * (1.0 + 0.1 * rng.standard_normal(n))
            f = np.clip(f, 0.01, 1000.0)
            stall = 0

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Main search: multi-start, multi-method optimization."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided)
        print(f"Provided seq: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()

    # Define candidate tasks: (length, init_type, optimizer)
    # Strategy: try the provided length first, then a few others
    # Use combined optimizer for the main runs

    tasks = []
    if provided is not None:
        tasks.append((len(provided), 'provided', 'combined'))
        # Try neighbors of provided length
        n_prov = len(provided)
        for n_try in [n_prov // 2, n_prov * 2]:
            if 64 <= n_try <= 4096:
                tasks.append((n_try, 'hann', 'combined'))

    # Additional lengths to try
    for n in [192, 256, 320, 384, 512, 640, 768]:
        if provided is not None and abs(n - len(provided)) < 32:
            continue
        tasks.append((n, 'hann', 'combined'))
        tasks.append((n, 'blackman', 'combined'))

    # Add some with kaiser
    for n in [256, 384, 512]:
        tasks.append((n, 'kaiser', 'combined'))

    print(f"\nTotal tasks: {len(tasks)}")
    print(f"Deadline in {deadline - t_start:.0f}s")

    # Sort by priority: provided first, then by length (prefer moderate)
    def task_priority(task):
        n, init, opt = task
        if init == 'provided':
            return 0
        # Prefer lengths close to 256-512
        return abs(n - 384) / 1000.0

    tasks.sort(key=task_priority)

    for idx, (n, init_name, opt_name) in enumerate(tasks):
        if time.monotonic() > deadline - 60:
            print(f"\nTime nearly up, stopping main loop.")
            break

        remaining = deadline - time.monotonic()
        # Allocate time based on priority
        if idx == 0:
            t_alloc = min(400.0, remaining * 0.35)
        elif idx < 5:
            t_alloc = min(200.0, remaining * 0.15)
        else:
            t_alloc = min(120.0, remaining * 0.1)
        t_alloc = max(t_alloc, 20.0)

        # Generate initialization
        if init_name == 'provided' and provided is not None:
            if len(provided) == n:
                f0 = provided.copy()
            else:
                x_old = np.linspace(0, 1, len(provided))
                x_new = np.linspace(0, 1, n)
                f0 = np.interp(x_new, x_old, provided)
        elif init_name == 'hann':
            f0 = make_hann_init(n, rng)
        elif init_name == 'blackman':
            f0 = make_blackman_init(n, rng)
        elif init_name == 'kaiser':
            f0 = make_kaiser_init(n, rng, beta=3.0 + rng.random() * 4.0)
        elif init_name == 'bartlett':
            f0 = make_bartlett_init(n, rng)
        else:
            f0 = make_hann_init(n, rng)

        print(f"\n  Task {idx}: n={n:4d} init={init_name:10s} opt={opt_name} alloc={t_alloc:.0f}s")

        if opt_name == 'combined':
            f_best, s_best = combined_optimize(f0, t_alloc, verbose=True)
        elif opt_name == 'lp':
            f_best, s_best = lp_optimize(f0, t_alloc, verbose=True)
        elif opt_name == 'gradient':
            f_best, s_best = gradient_optimize(f0, t_alloc, verbose=True)
        else:
            f_best, s_best = combined_optimize(f0, t_alloc, verbose=True)

        print(f"  => score={s_best:.6f} (global best={best_score:.6f})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW GLOBAL BEST: {best_score:.6f} ***")

    # Final polish phase on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 20:
        polish_time = min(120.0, deadline - time.monotonic() - 15.0)
        if polish_time > 10:
            print(f"\n=== Final polish ({polish_time:.0f}s) ===")
            f_pol, s_pol = combined_optimize(best_seq, polish_time * 0.6, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = f_pol.copy()
                print(f"  After combined polish: {best_score:.6f}")

            remaining_polish = deadline - time.monotonic() - 15.0
            if remaining_polish > 10:
                f_spe, s_spe = spectral_polish(best_seq, min(30.0, remaining_polish), verbose=True)
                if s_spe < best_score:
                    best_score = s_spe
                    best_seq = f_spe.copy()
                    print(f"  After spectral: {best_score:.6f}")

    # Multi-scale refinement: downsample, optimize, upsample back
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = len(best_seq)
        if n_best >= 128:
            half = downsample_seq(best_seq, 2)
            ms_time = min(60.0, deadline - time.monotonic() - 10.0)
            if ms_time > 10:
                print(f"\n=== Multi-scale refinement ===")
                f_ms, s_ms_half = lp_optimize(half, ms_time * 0.5, verbose=True)
                f_ms_full = upsample_seq(f_ms_half, 2)
                if len(f_ms_full) == n_best:
                    f_ms_full = combined_optimize(f_ms_full, ms_time * 0.4, verbose=False)
                    s_ms_full = score_fast(f_ms_full)
                    print(f"  Multi-scale score: {s_ms_full:.6f}")
                    if s_ms_full < best_score:
                        best_score = s_ms_full
                        best_seq = f_ms_full.copy()
                        print(f"  *** Multi-scale improvement: {best_score:.6f} ***")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        if final_score <= best_score + 1e-6:
            best_score = final_score
        print(f"\n{'='*55}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*55}")
        return final_list

    # Fallback
    fallback = [1.0] * 256
    print(f"FALLBACK: returning uniform 256, score={evaluate_sequence(fallback):.6f}")
    return fallback
