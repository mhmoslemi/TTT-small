d(2xy)/dy = 2x. k-j = 1-1 = 0, a[0] = x. So 2*a[0] = 2x. ✓

Great! So the gradient formula is simply:
`d(conv[k])/da[j] = 2 * a[k-j]` if `0 <= k-j < n`, else 0.

This means: `grad[j] = sum_k w[k] * 2 * a[k-j] * [0<=k-j<n] = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j]` (with bounds on k)
`= 2 * sum_{m=0}^{n-1} w[j+m] * a[m]` (where m = k-j, and we need j+m <= 2n-2)

This is a cross-correlation of `w` and `a`! Specifically:
`grad[j] = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} w[j+m] * a[m]`

This is the cross-correlation `(w ⋆ a)[j]` where `w` has length `2n-1` and `a` has length `n`.

Using FFT: cross-correlation can be computed as `ifft(fft(w) * conj(fft(a)))` with appropriate padding.

Actually, `sum_m w[j+m] * a[m]` is the cross-correlation of `w` and `a` at lag `j`. In FFT terms:
`corr[j] = ifft(fft(w_padded) * conj(fft(a_padded)))[j]`

Let me verify: the cross-correlation `R[j] = sum_m w[m+j] * a[m]` (non-centered).
In FFT: `R = ifft(FFT(w) * conj(FFT(a)))` gives `R[j] = sum_m w[(m+j) mod N] * a[m]` (circular cross-correlation).

With proper zero-padding to avoid circular aliasing, this gives the linear cross-correlation.

So the algorithm is:
1. Compute `conv = fft_convolve(a, a)` (length 2n-1)
2. Compute softmax weights: `w[k] = exp(conv[k]/tau) / sum(exp(conv[k']/tau))`
3. Compute gradient: `grad = 2 * cross_correlate(w, a)[:n]`
4. Take step: `a_new = a - lr * grad`
5. Project: `a_new = max(a_new, 0)`
6. Normalize: `a_new = a_new * sqrt(2n) / sum(a_new)`

This is O(n log n) per step! Much faster than the LP approach.

The key hyperparameter is `tau` (the softmax temperature). Small tau focuses on the peak, large tau smooths over all values. We should start with a relatively small tau and increase it.

"""
Efficient softmax-gradient optimization for minimizing normalized peak
autocorrelation of non-negative step functions. Uses FFT-based gradient
computation with adaptive temperature annealing, combined with LP polishing
for final refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def compute_gradient(a, tau):
    """
    Compute the gradient of the softmax-smoothed max convolution.
    
    f(a) = tau * log(sum_k exp(conv(a,a)[k] / tau))
    grad[j] = (1/tau) * sum_k softmax_k * d(conv[k])/da[j]
            = (1/tau) * sum_k softmax_k * 2 * a[k-j] * [0<=k-j<n]
            = (2/tau) * cross_correlate(softmax, a)[j]
    """
    n = len(a)
    conv = fft_convolve(a, a)  # length 2n-1
    
    # Softmax weights (numerically stable)
    shifted = conv / tau - np.max(conv) / tau
    w = np.exp(shifted)
    w = w / np.sum(w)
    
    # Cross-correlation: grad[j] = 2/tau * sum_m w[j+m] * a[m]
    # This is the non-centered cross-correlation of w and a
    # Using FFT: ifft(fft(w) * conj(fft(a)))
    size = 1
    while size < (2*n - 1) + n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    
    # corr[j] = sum_m w[(m+j) mod size] * a[m]
    # With proper padding, corr[j] = sum_m w[m+j] * a[m] for j = 0, ..., n-1
    grad = (2.0 / tau) * corr[:n]
    
    return grad, conv


def softmax_gradient_descent(a, time_limit, tau_schedule='linear', verbose=True):
    """
    Softmax gradient descent with adaptive temperature.
    """
    n = len(a)
    a = normalize_seq(a.copy())
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)
    
    # Adaptive learning rate and temperature
    tau = 0.5  # Start with moderate temperature
    lr = 1.0 / (n * tau)  # Initial learning rate
    
    # Annealing parameters
    min_tau = 0.01
    max_tau = 2.0
    tau_decay = 0.999
    
    step = 0
    stall_count = 0
    prev_score = best_score
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        grad, conv = compute_gradient(a, tau)
        
        # Clip gradient to avoid instability
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm * min(grad_norm, 10.0)
        
        # Take gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        a_new = normalize_seq(a_new)
        
        # Evaluate
        new_score = local_score(a_new)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Decrease learning rate slightly on improvement
            lr *= 0.999
        elif new_score < prev_score * 1.01:
            a = a_new
            stall_count += 1
            if stall_count % 50 == 0:
                lr *= 0.95
                tau = max(min_tau, tau * 0.98)
        else:
            # Rejected step
            lr *= 0.9
            tau = min(max_tau, tau * 1.05)
            stall_count += 1
        
        prev_score = local_score(a)
        
        # Adaptive temperature: decrease when making progress
        if stall_count == 0 and step > 100:
            tau = max(min_tau, tau * tau_decay)
        
        step += 1
        
        if verbose and step % 500 == 0:
            print(f"    SGD step {step}: score={best_score:.6f}, tau={tau:.4f}, lr={lr:.6f}")
    
    return best_a


def lp_refine_fast(a, time_limit, verbose=False):

    Faster LP-based iterative refinement.

    """

    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    # Build constraint matrix once
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    
    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        M = np.max(conv_a)
        
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g = normalize_seq(g, S)
        
        # Find best blend
        best_t = 0.0
        for t_try in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:

                break

def adam_optimize(a, time_limit, verbose=True):
    """
    Adam optimizer for the softmax-smoothed objective.
    More robust than plain gradient descent.
    """
    n = len(a)
    a = normalize_seq(a.copy())
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)
    
    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 1.0 / (n ** 0.5)
    tau = 0.3
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 1
    
    step = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        grad, conv = compute_gradient(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new)
        
        new_score = local_score(a_new)
        t += 1
        
        if new_score < best_score:

            best_a = a_new.copy()
            a = a_new

"""
Efficient multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Combines Nesterov-accelerated Adam with softmax
smoothing, spectral domain refinement, and LP polishing. Uses FFT-based gradient
computation (O(n log n) per step) for fast convergence.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def compute_softmax_gradient(a, tau):
    """
    Compute gradient of softmax-smoothed max of conv(a,a).
    
    f(a) = tau * log(sum_k exp(conv(a,a)[k] / tau))
    
    The gradient is: grad[j] = (2/tau) * sum_m w[j+m] * a[m]
    where w = softmax(conv(a,a)/tau).
    
    This is the cross-correlation of w and a, computable via FFT.
    """
    n = len(a)
    conv = fft_convolve(a, a)  # length 2n-1
    
    # Numerically stable softmax
    shifted = conv / tau - np.max(conv) / tau
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.full(len(conv), 1.0 / len(conv))
    else:
        w = w / w_sum
    
    # Cross-correlation: corr[j] = sum_m w[j+m] * a[m]
    # Using FFT: ifft(FFT(w) * conj(FFT(a)))
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    
    grad = (2.0 / tau) * corr[:n]
    return grad, conv

def lp_refine(a, time_limit, verbose=True):
    """
    LP-based iterative refinement. For each step:
    1. Compute M = max(conv(a,a))
    2. Solve: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0
    3. Blend a with g
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    # Build constraint matrix
    def build_A(a_cur):
        A_mat = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                A_mat[k, j] = a_cur[k - j]
        return A_mat
    
    A_mat = build_A(a)
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        if iteration % 20 == 0 and time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        c_vec = -np.ones(n)
        res = linprog(c_vec, A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g = normalize_seq(g, S)
        
        # Find best blend
        best_t = 0.0
        b_conv_max_best = np.inf
        for t_try in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            b = (1.0 - t_try) * a + t_try * g
            b = np.clip(b, 0.0, None)
            sb = np.sum(b)
            if sb < 1e-10:
                continue
            b = b * S / sb
            b_conv_max = np.max(fft_convolve(b, b))
            if b_conv_max < b_conv_max_best:
                b_conv_max_best = b_conv_max
                best_t = t_try
            if b_conv_max <= M + 1e-8:
                break
        
        if best_t == 0.0:
            break
        
        a = (1.0 - best_t) * a + best_t * g
        a = np.clip(a, 0.0, None)
        sa = np.sum(a)
        if sa < 1e-10:
            break
        a = a * S / sa
        
        # Rebuild A
        A_mat = build_A(a)
        
        new_conv_max = np.max(fft_convolve(a, a))
        if new_conv_max < best_conv_max - 1e-10:
            best_conv_max = new_conv_max
            best_a = a.copy()
        
        if verbose and iteration % 20 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: score={score:.6f}, t={best_t:.3f}")
        
        if best_t < 0.005:
            break
        
        iteration += 1
    
    return best_a

def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old + 1)[:-1]
    x_new = np.linspace(0, 1, n_new + 1)[:-1]
    return np.interp(x_new, x_old, a)

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.03, 0.12)
        a += amp * np.cos(freq * t + phase)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing normalized peak autocorrelation.
    
    Phase 1: Nesterov-Adam gradient optimization (fast, O(n log n) per step)
    Phase 2: Spectral domain refinement
    Phase 3: Extended Nesterov-Adam with lower temperature
    Phase 4: Targeted local search
    Phase 5: LP polishing (if n allows)
    Phase 6: Final local search polish
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 15.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def _update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"  *** New best: score={s:.6f}, n={len(a)} ***")
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        _update_best(provided_seq)
    
    # === PHASE 1: Nesterov-Adam optimization ===
    print("\n=== Phase 1: Nesterov-Adam (primary) ===")
    if provided_seq is not None:
        n = len(provided_seq)
        remaining = deadline - time.monotonic()
        t_limit = min(360.0, remaining * 0.3)
        print(f"  n={n}, time_limit={t_limit:.0f}s")
        result = nesterov_adam_optimize(provided_seq, t_limit, verbose=True, tau_init=0.25, tau_min=0.02)
        _update_best(result)
    
    # Try a different length (interpolated)
    if provided_seq is not None:
        for n_try in [512, 768, 896]:
            if time.monotonic() > deadline - 400:
                break
            remaining = deadline - time.monotonic()
            t_per = min(80.0, remaining * 0.12)
            if t_per < 20:
                break
            a0 = interpolate_sequence(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            result = nesterov_adam_optimize(a0, t_per, verbose=False, tau_init=0.3, tau_min=0.03)
            _update_best(result)
    
    # === PHASE 2: Spectral refinement ===
    print("\n=== Phase 2: Spectral refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_limit = min(120.0, remaining * 0.15)
            print(f"  time_limit={t_limit:.0f}s")
            result = spectral_refine(best_seq, t_limit, verbose=True)
            _update_best(result)
    
    # === PHASE 3: Extended Adam with low temperature ===
    print("\n=== Phase 3: Extended Nesterov-Adam (fine) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_limit = min(240.0, remaining * 0.3)
            print(f"  time_limit={t_limit:.0f}s, tau_init=0.08")
            result = nesterov_adam_optimize(best_seq, t_limit, verbose=True, tau_init=0.08, tau_min=0.01)
            _update_best(result)
    
    # === PHASE 4: Targeted local search ===
    print("\n=== Phase 4: Targeted local search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_limit = min(150.0, remaining * 0.25)
            print(f"  time_limit={t_limit:.0f}s")
            result = targeted_local_search(best_seq, t_limit, verbose=True)
            _update_best(result)
    
    # === PHASE 5: LP polishing ===
    print
    print("\n=== Phase 5: LP polishing ===")
    if best_seq is not None and len(best_seq) <= 700:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_limit = min(120.0, remaining * 0.3)
            print(f"  n={len(best_seq)}, time_limit={t_limit:.0f}s")
            result = lp_refine(best_seq, t_limit, verbose=True)
            _update_best(result)
    
    # === PHASE 6: Final short Adam + local search ===
    print("\n=== Phase 6: Final polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t1 = min(60.0, remaining * 0.4)
            result = nesterov_adam_optimize(best_seq, t1, verbose=False, tau_init=0.05, tau_min=0.01)
            _update_best(result)
        
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t2 = min(40.0, remaining * 0.5)
            result = targeted_local_search(best_seq, t2, verbose=False)
            _update_best(result)
    
    # === FINAL ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    
    return [1.0] * 256
