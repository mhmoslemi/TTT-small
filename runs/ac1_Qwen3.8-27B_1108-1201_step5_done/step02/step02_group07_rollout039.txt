"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Combines smooth quartic optimization (Adam), top-k subgradient descent, and LP polishing.
Emphasizes asymmetric initializations since symmetric sequences are provably limited to score >= 2.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def conv_gradient_quad(a):
    """Gradient of sum_k conv(a,a)[k]^2 with respect to a.
    Returns grad where grad[j] = 2 * sum_k c[k] * a[k-j] (valid indices)."""
    c = fft_convolve(a, a)
    # cross-correlation: R[j] = sum_k c[k] * a[k-j]
    # Use FFT: ifft(FFT(c) * conj(FFT(a)))
    size = 1
    while size < 2 * len(a) + len(a) - 1:  # need size >= (2n-1) + n - 1 = 3n-2
        size <<= 1
    C = rfft(c, size)
    A = rfft(a, size)
    R = irfft(C * np.conj(A), size)
    return 2.0 * R[:len(a)]


def project_simplex(v):
    """Project v onto the simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    # Find lambda such that sum(max(v - lambda, 0)) = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def adam_simplex(a, objective_grad_fn, n_iter=5000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with simplex projection."""
    n = len(a)
    a = project_simplex(a.copy())
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_fast(a)
    
    for t in range(1, n_iter + 1):
        g = objective_grad_fn(a)
        # Project gradient onto simplex tangent space
        g = g - np.mean(g)
        
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
        
        if t % 500 == 0:
            s_val = score_fast(a)
            if s_val < best_score:
                best_score = s_val
                best_a = a.copy()
    
    s_val = score_fast(a)
    if s_val < best_score:
        best_score = s_val
        best_a = a.copy()
    
    return best_a, best_score


def topk_subgradient_descent(a, n_iter=2000, lr=0.005, top_k=5, momentum_beta=0.9):
    """Subgradient descent on max(conv(a,a)) with top-k smoothing."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_score = score_fast(a)
    momentum = np.zeros(n)
    
    for t in range(1, n_iter + 1):
        c = fft_convolve(a, a)
        # Get top-k indices
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[-k:]
        
        # Compute subgradient (average over top-k)
        g = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += a[ki - j]
        g /= k
        g = g - np.mean(g)  # project to tangent space
        
        momentum = momentum_beta * momentum + (1 - momentum_beta) * g
        current_lr = lr / (1 + 0.01 * t)
        
        a = a - current_lr * momentum
        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
        
        if t % 200 == 0:
            s_val = score_fast(a)
            if s_val < best_score:
                best_score = s_val
                best_a = a.copy()
    
    s_val = score_fast(a)
    if s_val < best_score:
        best_score = s_val
        best_a = a.copy()
    
    return best_a, best_score

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses U-shaped initializations (exploiting the insight that asymmetric sequences
can achieve scores below 2), smooth quartic Adam optimization, top-k subgradient
descent, exact coordinate descent, and LP polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * float(np.max(c)) / (S * S))


def quad_gradient(a):
    """Gradient of sum_k conv(a,a)[k]^2."""
    c = fft_convolve(a, a)
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    C = rfft(c, size)
    A = rfft(a, size)
    R = irfft(C * np.conj(A), size)
    return 2.0 * R[:n]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_u_shape(n, alpha=0.5, rng=None):
    """U-shaped initialization: high at edges, low in middle."""
    t = np.arange(n, dtype=np.float64)
    left = (t + 1.0) ** (-alpha)
    right = (n - t) ** (-alpha)
    a = left + right
    if rng is not None:
        a += 0.05 * a * rng.standard_normal(n)
    a = np.clip(a, 1e-4, None)
    a = a / np.sum(a)
    return a


def make_ramp(n, decay=2.0, rng=None):
    """Exponential decay from left."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * t / n)
    if rng is not None:
        a += 0.05 * a * rng.standard_normal(n)
    a = np.clip(a, 1e-4, None)
    a = a / np.sum(a)
    return a


def make_power_law(n, alpha=0.7, rng=None):
    """Power law decay."""
    t = np.arange(n, dtype=np.float64) + 1.0
    a = t ** (-alpha)
    if rng is not None:
        a += 0.03 * a * rng.standard_normal(n)
    a = np.clip(a, 1e-4, None)
    a = a / np.sum(a)
    return a


def make_asym_gauss(n, center=0.3, width=0.2, rng=None):
    """Asymmetric Gaussian (off-center)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    if rng is not None:
        a += 0.05 * a * rng.standard_normal(n)
    a = np.clip(a, 1e-4, None)
    a = a / np.sum(a)
    return a


def make_hann_asym(n, gamma=2.0, rng=None):
    """Asymmetric Hann-like window."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 0.5 * (1.0 + np.cos(np.pi * t)) ** gamma
    a += 1e-6
    if rng is not None:
        a += 0.05 * a * rng.standard_normal(n)
    a = np.clip(a, 1e-4, None)
    a = a / np.sum(a)
    return a


def phase_adam_quad(a, n_iter=3000, lr=0.01, verbose=False):
    """Phase 1: Minimize sum of squares of conv values using Adam."""
    n = len(a)
    a = project_simplex(a.copy())
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = score_fast(a)
    
    for t in range(1, n_iter + 1):
        g = quad_gradient(a)
        g = g - np.mean(g)
        
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
        
        if t % 500 == 0:
            s_val = score_fast(a)
            if s_val < best_score:
                best_score = s_val
                best_a = a.copy()
            if verbose:
                print(f"      Adam t={t}: score={s_val:.6f}, best={best_score:.6f}")
    
    s_val = score_fast(a)
    if s_val < best_score:
        best_score = s_val
        best_a = a.copy()
    
    return best_a, best_score


def phase_subgrad(a, n_iter=2000, lr=0.01, top_k=5, beta_m=0.9, verbose=False):
    """Phase 2: Top-k subgradient descent on max(conv)."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_score = score_fast(a)
    momentum = np.zeros(n)
    
    for t in range(1, n_iter + 1):
        c = fft_convolve(a, a)
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[-k:]
        
        g = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += a[ki - j]
        g /= k
        g -= np.mean(g)
        
        momentum = beta_m * momentum + (1 - beta_m) * g
        current_lr = lr / (1.0 + 0.005 * t)
        
        a = a - current_lr * momentum
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
        
        if t % 300 == 0:
            s_val = score_fast(a)
            if s_val < best_score:
                best_score = s_val
                best_a = a.copy()
            if verbose:
                print(f"      Subgrad t={t}: score={s_val:.6f}, best={best_score:.6f}")
    
    s_val = score_fast(a)
    if s_val < best_score:
        best_score = s_val
        best_a = a.copy()
    
    return best_a, best_score

    k_hi = min(j + n - 1, 2 * n - 2)

    max_unaffected = 0.0
    if j > 0:
        max_unaffected = max(max_unaffected, float(np.max(c[:j])))
    if j + n <= 2 * n - 2:
        max_unaffected = max(max_unaffected, float(np.max(c[j+n:])))

    lower = 0.0
    upper = max(old * 2.0, 0.01)

    f_upper = eval_f(upper)
    if f_upper < f_
