import numpy as np
import scipy.optimize
import time

def fast_autoconv(a):
    """Compute auto-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = fast_autoconv(a)
    return 2.0 * n * conv.max() / (s * s)

def softmax_gradient(a, temperature=0.01):
    """Compute smoothed gradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_autoconv(a)
    # Softmax weights
    conv_scaled = conv / temperature
    conv_scaled = conv_scaled - conv_scaled.max()  # numerical stability
    weights = np.exp(conv_scaled)
    weights /= weights.sum()
    
    # Gradient of sum_k w_k * conv(a,a)[k] with respect to a[i]
    # d(conv(a,a)[k])/da[i] = a[k-i] if 0 <= k-i < n
    # So gradient[i] = sum_k w_k * a[k-i] (if valid)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        # For this k, the valid i range is max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        # grad[i] += w[k] * a[k-i] for i in [i_start, i_end)
        # k-i ranges from k-i_end+1 to k-i_start
        indices = k - np.arange(i_start, i_end)
        grad[i_start:i_end] += weights[k] * a[indices]
    
    return 2.0 * n * grad  # The 2n factor from the objective

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fista_optimize(a0, time_limit, temperature=0.01, verbose=False):
    """FISTA (accelerated projected gradient) for minimizing the objective on the simplex."""
    n = len(a0)
    a = proj_simplex(a0.astype(np.float64).copy())
    y = a.copy()
    t = 1.0
    
    # Compute Lipschitz constant estimate for the smoothed objective
    # The gradient is bounded by 2n * max(a) * (2n-1) roughly
    # Let's use a step size based on the norm
    conv = fast_autoconv(y)
    M = conv.max()
    eta = 1.0 / (2.0 * n * 2 * n)  # Conservative initial step size
    
    best_a = a.copy()
    best_obj = objective(a)
    
    t0 = time.monotonic()
    iteration = 0
    temperature_cur = temperature
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient at y
        grad = softmax_gradient(y, temperature_cur)
        
        # Nesterov momentum
        t_old = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t_old - 1.0) / (t + 1.0) if t > 1 else 0.0
        
        # Backtracking line search
        a_new = proj_simplex(y - eta * grad)
        f_new = fast_autoconv(a_new)
        obj_new = f_new.max()
        
        # Simple backtracking
        if obj_new > fast_autoconv(y).max() + eta * np.sum(grad * (a_new - y)) + 0.5 * eta * np.sum(grad**2) * np.sum((a_new-y)**2) + 1e-6:
            eta *= 0.5
            a_new = proj_simplex(y - eta * grad)
        
        # Update y for next iteration
        y = a_new + beta * (a_new - a)
        y = proj_simplex(y)  # Keep y in simplex
        a = a_new
        
        # Update best
        cur_obj = 2.0 * n * f_new.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

            temperature_cur = max(0.001, temperature_cur * 0.95)
        
        iteration += 1
        
        if verbose and iteration % 1000 == 0:
            print(f"    FISTA iter {iteration}, obj={best_obj:.6f}, eta={eta:.2e}")

import numpy as np
import scipy.optimize
import time


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]
