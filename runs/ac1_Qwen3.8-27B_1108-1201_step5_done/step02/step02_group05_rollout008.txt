def active_set_lp(a, eps_ratio=0.01):
    """Solve the linearized active-set LP."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    # Active set: positions where conv is close to max
    threshold = M - eps_ratio * M
    active = np.where(conv >= threshold)[0]
    
    # Build LP: minimize t
    # Variables: a_0, ..., a_{n-1}, t (total n+1 variables)
    # Constraints:
    # 1. For each k in active: grad_k @ a <= t (linearized)
    #    i.e., sum_j grad_k[j] * a_j - t <= 0
    # 2. sum(a) = 1
    # 3. a_j >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    
    for idx, k in enumerate(active):
        grad = np.zeros(n)
        idxs = k - np.arange(n)
        valid = (idxs >= 0) & (idxs < n)
        grad[valid] = 2.0 * a[idxs[valid]]
        # Also need the constant term: conv[k] - grad @ a
        # Linearized: conv[k] + grad @ (a - a_cur) <= t
        # => grad @ a <= t - conv[k] + grad @ a_cur

    # sum(a) = 1 as equality
    # scipy linprog doesn't handle equalities well mixed with inequalities easily
    # Use A_eq, b_eq
    A_eq = np.zeros((1, n + 1))

def proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale to work with sum=1, then rescale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_k(a, k):
    """Compute gradient of (a*a)_k w.r.t. a. Result: 2*P_k*a."""
    n = len(a)
    grad = np.zeros(n)
    idxs = k - np.arange(n)
    valid = (idxs >= 0) & (idxs < n)
    grad[valid] = 2.0 * a[idxs[valid]]
    return grad

def active_set_lp(a, eps_ratio=0.05, max_iter=200, time_limit=30.0):
    """
    Active-set linearization method.
    Iterate: find active set, solve LP, update.
    """
    n = len(a)
    a = proj_simplex(a.copy())
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Active set: positions where conv is close to max
        threshold = M * (1.0 - eps_ratio)
        active = np.where(conv >= threshold)[0]
        
        # Limit active set size for LP tractability
        if len(active) > 100:
            # Keep only the top 100
            active_vals = conv[active]
            top_idx = np.argsort(active_vals)[::-1][:100]
            active = active[top_idx]
        
        # Build LP
        # Variables: a_0,...,a_{n-1}, t  (n+1 vars)
        # Minimize: t
        # Constraints:
        #   For each k in active: 2*(P_k a_cur) @ a - t <= conv[k]
        #   sum(a) = 1
        #   a >= 0, t >= M*0.5
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        m_active = len(active)
        A_ub = np.zeros((m_active, n + 1))
        b_ub = np.zeros(m_active)
        
        for idx, k in enumerate(active):
            grad = compute_grad_k(a, k)
            A_ub[idx, :n] = grad
            A_ub[idx, n] = -1.0
            b_ub[idx] = conv[k]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0.5 * M, None)]
        
        try:
            res = scipy.optimize.linprog(
                c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs',
                options={'time_limit': max(1.0, time_limit - (time.monotonic() - t0))}
            )
        except Exception:
            break
        
        if not res.success:
            break
        
        a_lp = res.x[:n]
        a_lp = np.maximum(a_lp, 0.0)
        s = a_lp.sum()
        if s < 1e-12:
            break
        a_lp /= s
        
        # Check improvement
        obj_lp = obj_norm(a_lp)
        
        if obj_lp < best_obj - 1e-12:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp
        else:
            # Try interpolation
            for t in [0.1, 0.25, 0.5]:
                a_try = (1.0 - t) * a + t * a_lp
                a_try = proj_simplex(a_try)
                obj_try = obj_norm(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, adjust eps and continue
                eps_ratio = max(eps_ratio * 1.5, 0.3)
                if eps_ratio > 0.4:
                    break
    
    return best_a, best_obj


def gauss_seidel_refine(a, time_limit):
    """
    Coordinate descent / Gauss-Seidel style refinement.
    For each coordinate, find the optimal value given all others fixed.
    """
    n = len(a)
    a = proj_simplex(a.copy())
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            # Try a few values for a[i]
            # The auto-convolution at lag k involves a[i]*a[k-i]
            # Changing a[i] affects conv at lags where i and k-i are both valid
            # i.e., k = i + j for j in [0, n), and k = j - i for j in [0, n)
            # So conv[k] changes for k in [i, i+n-1] (from a[i]*a[k-i]) 
            # and k in [i-n+1, i] (from a[k+i-i]*a[i] = a[k]*a[i])
            # Wait, conv[k] = sum_j a[j]*a[k-j]
            # a[i] appears in conv[k] when j=i (term a[i]*a[k-i]) or k-j=i (term a[k-i]*a[i])
            # So a[i] appears in conv[k] for k where 0<=k-i<n (i.e., i<=k<=i+n-1) 
            # AND for k where 0<=k-i<n (same condition from the other side)
            # So a[i] appears in conv[k] for k in [max(0,i-(n-1)), min(2n-2, i+n-1)]
            # which is k in [max(0, i-n+1), min(2n-2, i+n-1)]
            
            # Simplification: just try a few perturbations
            old_val = a[i]
            if old_val < 1e-15:
                # Try adding a small amount
                for frac in [0.001, 0.005, 0.01]:
                    amount = frac / n  # small amount
                    a_try = a.copy()
                    # Take from the largest element (not i)
                    j = int(np.argmax(np.where(np.arange(n) != i, a, 0)))
                    if a_try[j] > amount:
                        a_try[i] += amount
                        a_try[j] -= amount
                        obj_try = obj_norm(a_try)
                        if obj_try < best_obj - 1e-12:
                            a = a_try
                            best_obj = obj_try
                            improved = True
                            break
                continue
            
            # Try reducing
            for frac in [0.1, 0.25, 0.5]:
                amount = old_val * frac
                a_try = a.copy()
                a_try[i] -= amount
                # Distribute to others proportionally
                others = np.ones(n)
                others[i] = 0
                others /= others.sum()
                a_try += amount * others
                a_try = np.maximum(a_try, 0)
                a_try /= a_try.sum()
                obj_try = obj_norm(a_try)
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    improved = True
                    break
        
        if not improved:
            break
    
    return a, best_obj


def local_search_pairs(a, time_limit):
    """
    Pair-based local search: move mass between pairs of elements.
    Focus on pairs that affect the peak convolution positions.
    """
    n = len(a)
    a = proj_simplex(a.copy())
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]
        
        # Find elements contributing most to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Danger: high contribution, Safe: low contribution
        sorted_contrib = np.sort(contrib)[::-1]
        if sorted_contrib[0] < 1e-15:
            break
        
        danger_thr = 0.5 * sorted_contrib[0]
        safe_thr = 0.1 * sorted_contrib[0]
        danger = np.where(contrib > danger_thr)[0]
        safe = np.where(contrib < safe_thr)[0]
        
        found = False
        for di in danger[:20]:
            if found:
                break
            for si in safe[:20]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[di] -= amount
                    a_try[si] += amount
                    a_try = np.maximum(a_try, 0.0)
                    ts = a_try.sum()
                    if ts < 1e-12:
                        continue
                    a_try /= ts
                    obj_try = obj_norm(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        a = a_try
                        found = True
                        break
        
        if found:
            no_improve = 0
        else:
            no_improve += 1
    
    return a, best_obj


def get_parent(n):
    """Get parent sequence clipped/resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            # Interpolate to length n
            x_old = np.linspace(0, 1, p.size)
            x_new = np.linspace(0, 1, n)
            pc = np.interp(x_new, x_old, p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_inits(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-10)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    hann = np.maximum(hann, 0.0)
    inits['hann'] = hann / hann.sum()
    
    # Triangle
    tri = 1.0 - np.abs(x - (n-1)/2.0) / ((n-1)/2.0 + 1e-10)
    tri = np.maximum(tri, 0.0)
    inits['triangle'] = tri / tri.sum()
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Edge-concentrated
    edges = np.zeros(n)
    m = max(1, n // 5)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()
    
    # Sinc-like
    if n > 1:
        sinc = np.sinc(x / n - 0.5)  # shifted sinc
        sinc = np.maximum(sinc, 0.0)
        if sinc.sum() > 1e-10:
            inits['sinc'] = sinc / sinc.sum()
    
    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
    
    # Parent with noise
    if parent is not None:
        noisy = parent + rng.normal(0, 0.01, n)
        noisy = np.maximum(noisy, 0)
        inits['parent_noisy'] = noisy / noisy.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set linearization (ASL)
combined with pair-based local search and diverse initializations across
multiple sequence lengths. The ASL method solves a linearized subproblem
at the peak convolution positions each iteration, providing a principled
descent direction. Complemented with the previous LP-based direction finding
and aggressive local search for final polishing.

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find largest rho (1-indexed) such that rho*u[rho-1] > cssv[rho-1] - target_sum
    rhs = cssv - target_sum
    candidates = np.where(np.arange(1, n + 1) * u > rhs)[0]
    if len(candidates) == 0:
        return np.full(n, target_sum / n)
    rho = candidates[-1] + 1
    theta = (cssv[rho - 1] - target_sum) / rho
    return np.maximum(v - theta, 0.0)


def compute_grad_k(a, k):
    """Compute gradient of (a*a)_k w.r.t. a: result is 2*P_k*a."""
    n = len(a)
    grad = np.zeros(n)
    idxs = k - np.arange(n)
    valid = (idxs >= 0) & (idxs < n)
    if np.any(valid):
        grad[valid] = 2.0 * a[idxs[valid]]
    return grad


def active_set_lp(a, eps_ratio=0.05, max_iter=300, time_limit=60.0):
    """
    Active-set linearization: at each iteration, linearize the QCP constraints
    at active (peak) positions and solve the resulting LP.
    """
    n = len(a)
    a = proj_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 50

    for _ in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break

        # Active set: positions within eps_ratio of max
        threshold = M * (1.0 - eps_ratio)
        active_mask = conv >= threshold
        active = np.where(active_mask)[0]

        # Limit active set for LP size
        if len(active) > 80:
            active_vals = conv[active]
            top_idx = np.argsort(active_vals)[::-1][:80]
            active = active[top_idx]

        # Build LP: minimize t
        # Variables: a_0,...,a_{n-1}, t  => n+1 vars
        # Constraints: 2*(P_k a_cur)@a - t <= conv[k]  for k in active
        # Equality: sum(a) = 1
        # Bounds: a >= 0, t >= 0.5*M

        c = np.zeros(n + 1)
        c[-1] = 1.0

        m_act = len(active)
        A_ub = np.zeros((m_act, n + 1))
        b_ub = np.zeros(m_act)

        for idx, k in enumerate(active):
            grad = compute_grad_k(a, k)
            A_ub[idx, :n] = grad
            A_ub[idx, n] = -1.0
            b_ub[idx] = conv[k]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0.0, None)] * n + [(0.5 * M, None)]

        remaining_time = max(1.0, time_limit - (time.monotonic() - t0))
        try:
            res = scipy.optimize.linprog(
                c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs',
                options={'time_limit': remaining_time, 'presolve': True}
            )
        except Exception:
            break

        if not res.success:
            break

        a_lp = res.x[:n].copy()
        a_lp = np.maximum(a_lp, 0.0)
        s = a_lp.sum()
        if s < 1e-12:
            break
        a_lp /= s

        obj_lp = obj_norm(a_lp)

        if obj_lp < best_obj - 1e-13:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp
            stuck = 0
        else:
            # Try interpolation to get a smaller step
            improved_here = False
            for frac in [0.3, 0.5, 0.7]:
                a_try = (1.0 - frac) * a + frac * a_lp
                a_try = proj_simplex(a_try)
                obj_try = obj_norm(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved_here = True
                    break
            if not improved_here:
                stuck += 1
                eps_ratio = min(eps_ratio * 1.3, 0.4)
                if stuck >= max_stuck:
                    break

    return best_a, best_obj


def lp_direction_method(f, M, n):
    """
    Previous LP-based method: find g >= 0 maximizing sum(g)
    subject to conv(f, g) <= M. Used as an alternative direction.
    """
    if M < 1e-15:
        return f.copy()
    # Build constraint matrix: conv(f,g)_k = sum_j f_{k-j} * g_j <= M
    # For efficiency, use sparse structure
    A_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row[j] = f[k - j]
        A_rows.append(row)
    A = np.array(A_rows)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                      method='highs', options={'time_limit': 30.0})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def local_search_pairs(a, time_limit, max_stuck=300):
    """
    Pair-based local search: move mass between elements to reduce peak convolution.
    Targets elements contributing most to the peak and redistributes to safe elements.
    """
    n = len(a)
    a = proj_simplex(a.astype(np.float64).copy())
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]

        # Compute contributions to peak
        contrib = np.zeros(n)
        idxs = k_star - np.arange(n)
        valid = (idxs >= 0) & (idxs < n)
        contrib[valid] = a[valid] * a[idxs[valid]]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        danger_thr = 0.4 * max_c
        safe_thr = 0.08 * max_c
        danger = np.where(contrib > danger_thr)[0]
        safe = np.where(contrib < safe_thr)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        d_sample = danger[:25]
        s_sample = safe[:25]

        for di in d_sample:
            if found:
                break
            ai = a[di]
            if ai < 1e-15:
                continue
            for si in s_sample:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6]:
                    amount = ai * frac
                    if amount < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[di] -= amount
                    a_try[si] += amount
                    a_try = np.maximum(a_try, 0.0)
                    ts = a_try.sum()
                    if ts < 1e-12:
                        continue
                    a_try /= ts
                    obj_try = obj_norm(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        a = a_try
                        found = True
                        break

        if found:
            stuck = 0
        else:
            stuck += 1

    return a, best_obj

            if ai < 1e-15:

            conv = fast_convolve(a)
            k_peak = int(np.argmax(conv))

            # Find safe target: minimize contribution to all peak lags

            if min_contrib_target is None:

                obj_try = obj_norm(a_try)
                if obj_try < best_obj - 1e-13:

                    a = a_try

                    improved_any = True

def make_inits(n, rng):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-10)
        inits[f'gauss_{sf}'] = g / g.sum()

    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0.0)
        inits['hann'] = hann / hann.sum()

    # Flat-top (raise cosine to power)
    if n > 1:
        flat = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1))) ** 2
        flat = np.maximum(flat, 0.0)
        inits['flat_top'] = flat / flat.sum()

    # Triangle
    tri = 1.0 - 2.0 * np.abs(x - (n - 1) / 2.0) / (n - 1 + 1e-10)
    tri = np.maximum(tri, 0.0)
    if tri.sum() > 1e-10:
        inits['triangle'] = tri / tri.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Edge concentrated
    edges = np.zeros(n)
    m = max(1, n // 6)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Two-peak
    two_peak = np.zeros(n)
    c1 = n // 4
    c2 = 3 * n // 4
    w = n // 8
    two_peak[max(0, c1 - w):min(n, c1 + w)] = 1.0
    two_peak[max(0, c2 - w):min(n, c2 + w)] = 1.0
    if two_peak.sum() > 0:
        inits['two_peak'] = two_peak / two_peak.sum()

    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent

    # Parent with noise
    if parent is not None:
        for noise_level in [0.01, 0.03]:
            noisy = parent + rng.normal(0, noise_level, n)
            noisy = np.maximum(noisy, 0)
            if noisy.sum() > 1e-10:
                inits[f'parent_noise_{noise_level}'] = noisy / noisy.sum()

    return inits

        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure all values are valid
    best_sequence = np.maximum(best_sequence, 0.0)
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, length={len(best_sequence)}")
    return [float(x) for x in best_sequence]
