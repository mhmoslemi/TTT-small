def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / max(len(a), 1)
    return a * S / s


def build_lp_matrix(a):
    """Build the convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M, top_k=None):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M for all k, g >= 0.
    Optionally use only top_k constraints for speed.
    """
    n = len(a)
    A = build_lp_matrix(a)
    if top_k is not None and top_k < 2 * n - 1:
        # Find the most binding constraints
        test_g = np.ones(n) / n
        conv_test = A @ test_g
        top_idx = np.argsort(-conv_test)[:top_k]
        A = A[top_idx, :]
        b = np.full(len(top_idx), M)
    else:
        b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=False):

def smoothed_gradient(a, eps):
    """
    Compute gradient of f_eps(a) = 2n * logsumexp(conv(a,a)/eps) / (sum(a))^2
    using efficient FFT-based convolution.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    S = np.sum(a)
    if S < 1e-10:
        return np.zeros(n)

    # Log-sum-exp for stability
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / eps
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = max_conv / eps + np.log(np.sum(np.exp(shifted)))
    LSE = log_sum  # This is LSE/eps actually... let me be careful

    # Actually: logsumexp(conv/eps) = max(conv)/eps + log(sum(exp((conv-max(conv))/eps)))
    LSE = max_conv / eps + np.log(np.sum(np.exp(np.clip((conv - max_conv) / eps, -500, 500))))

    # Softmax weights p[k] = exp(C[k]/eps) / sum(exp(C[j]/eps))
    p = np.exp(np.clip((conv - max_conv) / eps, -500, 500))
    p = p / np.sum(p)

    # Compute q[i] = sum_{m=0}^{n-1} p[m+i] * a[m]  (cross-correlation)
    # scipy.signal.correlate(p, a, 'full')[i] = sum_m p[m+i]*a[m] for i >= 0
    q = sig_correlate(p, a, mode='full')[:n]

    # Compute r[i] = sum_{m=0}^{i} p[i-m] * a[m]  (convolution)
    r = np.convolve(p, a)[:n]

    # dLSE/da[i] = (q[i] + r[i]) / eps
    dLSE = (q + r) / eps

    # grad[i] = 2n * (dLSE[i] * S - 2*LSE) / S^3
    grad = 2.0 * n * (dLSE * S - 2.0 * LSE) / (S ** 3)

    return grad

def make_varied_initialization(n, init_type, rng):
    """Create various types of initial sequences with emphasis on asymmetry."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'exponential':
        rate = rng.uniform(0.3, 1.5)
        a = np.exp(rate * t / n)
    elif init_type == 'anti_symmetric':
        # First half large, second half small (or vice versa)
        a = np.ones(n)
        a[:n//2] *= rng.uniform(1.5, 3.0)
        a[n//2:] *= rng.uniform(0.3, 0.7)
    elif init_type == 'chirp':
        alpha = rng.uniform(0.5, 2.0)
        a = 1.0 + 0.5 * np.cos(np.pi * alpha * t * t / n)
    elif init_type == 'sine':
        f = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.4 * rng.standard_normal(n)
    elif init_type == 'two_hump':
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        sigma = n * 0.1
        a = 0.3 + np.exp(-0.5 * ((t - c1) / sigma) ** 2) + np.exp(-0.5 * ((t - c2) / sigma) ** 2)
    elif init_type == 'sawtooth':
        period = n // rng.integers(2, 5)
        a = 1.0 + 0.5 * ((t % period) / period - 0.5)
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, None)
    return a

    for init_type in init_types:
        if time.monotonic() - t1_start > t1_limit:
            break
        a0 = make_varied_initialization(n_base, init_type, rng)
        t_per = min(8.0, (t1_start + t1_limit - time.monotonic()) * 0.3)
        if t_per < 2:
            break
        result = lp_refine(a0, t_per, verbose=False)
        if try_update(result):
            print(f"  *** init={init_type}: new best score={best_score:.6f} ***")

    print(f"  Phase 1 best: {best_score:.6f} (t={time.monotonic()-t1_start:.0f}s)")

    # Phase 2: Smoothed gradient optimization (500s)
    print("\n=== Phase 2: Smoothed Gradient Optimization ===")
    t2_limit = min(500.0, (deadline - time.monotonic()) * 0.35)
    if t2_limit > 30:
        print(f"  Time limit: {t2_limit:.0f}s")
        result = smoothed_optimize(best_seq.copy(), t2_limit, verbose=True)
        try_update(result)
        print(f"  After smoothed opt: score={best_score:.6f}")

    # Phase 3: LP refinement (300s)
    print("\n=== Phase 3: LP Refinement ===")
    t3_limit = min(300.0, (deadline - time.monotonic()) * 0.25)
    if t3_limit > 20:
        print(f"  Time limit: {t3_limit:.0f}s")
        result = lp_refine(best_seq.copy(), t3_limit, verbose=True)
        try_update(result)
        print(f"  After LP: score={best_score:.6f}")

    # Phase 4: Simulated annealing (400s)
    print("\n=== Phase 4: Simulated Annealing ===")
    t4_limit = min(400.0, (deadline - time.monotonic()) * 0.3)
    if t4_limit > 30:
        print(f"  Time limit: {t4_limit:.0f}s")
        result = sim_annealing(best_seq.copy(), t4_limit, verbose=True)
        try_update(result)
        print(f"  After SA: score={best_score:.6f}")

    # Phase 5: Another LP round (200s)
    print("\n=== Phase 5: LP Refinement Round 2 ===")
    t5_limit = min(200.0, (deadline - time.monotonic()) * 0.25)
    if t5_limit > 15:
        print(f"  Time limit: {t5_limit:.0f}s")
        result = lp_refine(best_seq.copy(), t5_limit, verbose=True)
        try_update(result)
        print(f"  After LP2: score={best_score:.6f}")

    # Phase 6: Coordinate descent polish (100s)
    print("\n=== Phase 6: Coordinate Descent Polish ===")
    t6_limit = min(100.0, (deadline - time.monotonic()) * 0.2)
    if t6_limit > 10:
        result = coordinate_descent(best_seq.copy(), t6_limit, verbose=False)
        try_update(result)
        print(f"  After CD: score={best_score:.6f}")

    # Phase 7: Final quick LP if time remains
    remaining = deadline - time.monotonic() - 15
    if remaining > 10:
        print(f"\n=== Phase 7: Final LP ({remaining:.0f}s) ===")
        result = lp_refine(best_seq.copy(), remaining, verbose=False)
        try_update(result)
        print(f"  Final LP: score={best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_score={best_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
