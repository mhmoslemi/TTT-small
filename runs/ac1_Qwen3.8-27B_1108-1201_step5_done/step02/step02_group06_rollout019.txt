Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Combines active-set LP optimization, annealed

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (sum_a ** 2)


def build_T(f, active_idx):
    """Build constraint rows for active indices. Each row k: A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    K = len(active_idx)
    A = np.zeros((K, n))
    for idx, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            cols = np.arange(lo, hi + 1)
            A[idx, cols] = f[k - cols]
    return A

    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k active lags, g >= 0."""

                     method='highs', options={'presolve': True, 'time_limit': 5})
    if result.success:
        return result.x
    return None

        g_rescaled = g * (sum_f / sum_g)

        # Update
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = f_new.sum()
        if sum_new < 1e-12:
            t *= 0.5
            stagnant += 1
            continue
        f_new = f_new * (sum_f / sum_new)

        score_new = local_score(f_new)
        iteration += 1
        history.append(score_new)

        # Oscillation detection
        if len(history) > 10:
            recent = history[-10:]
            osc = np.std(recent) / max(np.mean(recent), 1e-10)
            if osc > 0.001:
                t *= 0.8

        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            # Slightly increase step if consistently improving
            if t < 0.6:
                t = min(0.6, t * 1.03)
        else:
            f = f_new
            stagnant += 1
            if stagnant > 15:
                t *= 0.75
                stagnant = 0
            if stagnant > 60:
                # Perturb to escape
                f = f + 0.01 * rng_local.standard_normal(n) * (f > 0.01)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / max(f.sum(), 1e-10))
                t = 0.2
                stagnant = 0
                history = []

        if verbose and iteration % 100 == 0:
            print(f"    LP iter={iteration}, t={t:.3f}, score={local_score(f):.6f}, "
                  f"best={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def gradient_polish(f_init, time_limit, verbose=False):
    """Adam gradient descent on log-softmax with annealed temperature."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-8, None)
    theta = np.log(f_safe / f_safe.sum())

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    tau_start = 10.0 / n
    tau_end = 0.5 / n
    base_lr = 1.2

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** frac
        # Periodic reheating
        if step_count % 500 == 490:
            tau = tau_start * 0.5

        ts = theta - theta.max()
        e = np.exp(ts)
        a = e / e.sum() * S

        h = fft_convolve(a, a)
        # Smoothed max using top-K
        h_sorted = np.sort(h)[::-1]
        # Use softmax over top 10 for gradient
        top_k = min(10, len(h))
        top_idx = np.argsort(h)[-top_k:]
        h_top = h[top_idx] / tau
        h_top -= h_top.max()
        w = np.exp(h_top)
        w /= w.sum()

        # Gradient of smoothed max w.r.t. a
        # d/d a_j [sum_k w_k * h_k] where h_k = sum_i a_i * a_{k-i}
        # = sum_k w_k * (2 * a_{k-j}) for valid (k,j)
        grad_a = np.zeros(n)
        for ki, k in enumerate(top_idx):
            wv = w[np.where(top_idx == k)[0][0]] if top_k > 1 else 1.0 / top_k
            # Actually w is already normalized over top_idx
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad_a[lo:hi+1] += wv * 2.0 * a[k - hi:k - lo + 1][::-1]

        # Simplify: gradient of h[k] w.r.t. a[j] is 2*a[k-j] if 0<=k-j<n
        # So weighted sum: grad_a[j] = sum_k w_norm[k] * 2 * a[k-j] (if 0<=k-j<n)
        grad_a = np.zeros(n)
        w_norm = w / w.sum()
        for ki, k in enumerate(top_idx):
            wv = w_norm[ki]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                grad_a[js] += wv * 2.0 * a[k - js]

        # Convert to theta gradient: a = softmax(theta)*S
        # da/dtheta = a*(e_j - avg(e)) for the softmax... 
        # Actually: a_j = S * exp(theta_j) / sum(exp(theta))
        # da_j/dtheta_l = a_j * (delta_jl - exp(theta_l)/sum(exp(theta)))
        #                = a_j * (delta_jl - a_l/S)
        # So dF/dtheta_l = sum_j (dF/da_j) * a_j * (delta_jl - a_l/S)
        #                = a_l * (sum_j grad_a_j * a_j - a_l * sum_j grad_a_j * a_j / S)
        #                = a_l * S * (grad_a @ (a/S) - (a @ grad_a) / S)
        # Let me redo: dF/dtheta_l = sum_j grad_a_j * a_j * (delta_jl - p_l)
        # where p_l = a_l / S
        # = a_l * grad_a_l - p_l * sum_j grad_a_j * a_j
        # = a_l * grad_a_l - (a_l/S) * (grad_a @ a)

        inner = np.dot(grad_a, a) / S
        g_theta = a * (grad_a - inner)

        step_count += 1
        m1 = b1 * m1 + (1.0 - b1) * g_theta
        m2 = b2 * m2 + (1.0 - b2) * (g_theta ** 2)
        m1_hat = m1 / (1.0 - b1 ** step_count)
        m2_hat = m2 / (1.0 - b2 ** step_count)
        lr = base_lr / np.sqrt(1.0 + 0.005 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        ts = theta - theta.max()
        a_check = np.exp(ts) / np.exp(ts).sum() * S
        score = local_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

        if verbose and step_count % 500 == 0:
            print(f"    Grad step={step_count}, score={score:.6f}, "
                  f"best={best_score:.6f}, tau={tau:.5f}")

    ts = best_theta - best_theta.max()
    a_best = np.exp(ts) / np.exp(ts).sum() * S
    return a_best, local_score(a_best)


def targeted_reduction(f_init, time_limit, rng):
    """Targeted mutations: reduce elements contributing most to peaks."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = f.sum()
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        h = fft_convolve(f, f)
        peak_val = h.max()
        # Find peaks above 95th percentile
        thresh = np.percentile(h, 95)
        peak_positions = np.where(h > thresh)[0]

        candidate = f.copy()
        r = rng.random()

        if r < 0.5 and len(peak_positions) > 0:
            # Reduce top contributors to a random peak
            k = peak_positions[int(rng.integers(0, len(peak_positions)))]
            contribs = np.zeros(n)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                contribs[js] = f[js] * f[k - js]
            # Reduce top contributors
            n_reduce = max(2, n // 15)
            top_idx = np.argsort(contribs)[-n_reduce:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_idx.size))
            # Increase non-contributors
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(8, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.75:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.9:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        else:
            # Two-block
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))

        candidate = np.clip(candidate, 0.0, None)
        cs = candidate.sum()
        if cs < 1e-10:
            continue
        candidate = candidate * (sum_f / cs)

        new_score = local_score(candidate)
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            f = candidate
            current_score = new_score

        temp *= cooling

    return best_f, best_score


def make_provided_init(provided, n, rng):
    if len(provided) == n:
        return provided.astype(np.float64).copy()
    x_old = np.linspace(0, 1, len(provided))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, provided)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    alpha = 0.2 + 0.5 * rng.random()
    freq = 1.0 + 2.0 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * freq * t * t / n)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_tones = 4 + int(rng.integers(0, 5))
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 3)))
        amp = rng.uniform(0.04, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng):
    a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_block_init(n, rng):
    """Initialize with block structure."""
    n_blocks = int(rng.integers(4, 13))
    a = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        start = i * block_size
        end = min(start + block_size, n)
        a[start:end] = rng.uniform(0.3, 1.5)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP with active set -> gradient polish ->
    targeted reduction -> SA fine-tune. Multiple initializations and
    adaptive time allocation.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if provided_seq.sum() > 0.01:
            s_prov = local_score(provided_seq)
            print(f"Provided: n={len(provided_seq)}, score={s_prov:.6f}")
            if s_prov < best_score:
                best_score = s_prov
                best_seq = provided_seq.copy()

    n_default = len(provided_seq) if provided_seq is not None else 256

    # === Phase 1: LP optimization with multiple starts ===
    print("=" * 60)
    print("Phase 1: LP-based optimization (multi-start)")
    print("=" * 60)

    avail = deadline - time.monotonic()
    # Define initializations to try
    inits = []
    if provided_seq is not None:
        inits.append(('provided_256', n_default, lambda: make_provided_init(provided_seq, n_default, rng)))
    inits.append(('chirp_256', 256, lambda: make_chirp_init(256, rng)))
    inits.append(('multi_256', 256, lambda: make_multitone_init(256, rng)))
    inits.append(('random_256', 256, lambda: make_random_init(256, rng)))
    inits.append(('block_256', 256, lambda: make_block_init(256, rng)))
    inits.append(('chirp_512', 512, lambda: make_chirp_init(512, rng)))
    inits.append(('multi_512', 512, lambda: make_multitone_init(512, rng)))
    inits.append(('random_512', 512, lambda: make_random_init(512, rng)))

    # Allocate time: more to provided, less to others
    time_fracs = []
    for name, n, _ in inits:
        if name.startswith('provided'):
            time_fracs.append(0.25)
        elif n == 256:
            time_fracs.append(0.10)
        else:
            time_fracs.append(0.07)

    total_frac = sum(time_fracs)
    phase1_time = avail * 0.50

    for i, (name, n, gen_fn) in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break
        alloc = time_fracs[i] / total_frac * phase1_time
        alloc = min(alloc, deadline - time.monotonic() - 100)
        if alloc < 10:
            break

        a0 = gen_fn()
        print(f"\n  {name}: n={n}, alloc={alloc:.0f}s, init_score={local_score(a0):.6f}")
        t_start = time.monotonic()
        f_best, score = lp_optimize(a0, alloc, t_init=0.25, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"  {name}: score={score:.6f}, time={elapsed:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} ({name}) ***")

    # === Phase 2: Gradient polish on best ===
    print("\n" + "=" * 60)
    print(f"Phase 2: Gradient polish (best so far: {best_score:.6f})")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(remaining * 0.35, 400.0)
        if grad_time > 20:
            print(f"  n={len(best_seq)}, time={grad_time:.0f}s")
            a_grad, score_grad = gradient_polish(best_seq, grad_time, verbose=True)
            print(f"  Gradient result: {score_grad:.6f}")
            if score_grad < best_score:
                best_score = score_grad
                best_seq = a_grad.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: Targeted peak reduction ===
    print("\n" + "=" * 60)
    print(f"Phase 3: Targeted peak reduction (best so far: {best_score:.6f})")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        target_time = min(remaining * 0.30, 300.0)
        if target_time > 15:
            print(f"  n={len(best_seq)}, time={target_time:.0f}s")
            a_target, score_target = targeted_reduction(best_seq, target_time, rng)
            print(f"  Targeted result: {score_target:.6f}")
            if score_target < best_score:
                best_score = score_target
                best_seq = a_target.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Final LP refinement ===
    print("\n" + "=" * 60)
    print(f"Phase 4: Final LP refinement (best so far: {best_score:.6f})")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        final_time = min(remaining * 0.8, 200.0)
        if final_time > 15:
            print(f"  n={len(best_seq)}, time={final_time:.0f}s")
            a_final, score_final = lp_optimize(best_seq, final_time, t_init=0.15)
            print(f"  Final LP: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Final gradient sweep ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n  Final gradient sweep, time={remaining-30:.0f}s")
            a_final2, score_final2 = gradient_polish(best_seq, remaining - 30, verbose=False)
            if score_final2 < best_score:
                best_score = score_final2
                best_seq = a_final2.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
