    # (w * a[::-1])[j + n - 1]: 
    # a[::-1] has DFT = conj(A) (for real a, shifted)... 
    # Actually, DFT of a[::-1] is not simply related to DFT of a.
    # Let me use a different approach.
    # (w * a_flipped)[m] where a_flipped = a[::-1]
    # = sum_k w[k] * a_flipped[m-k] = sum_k w[k] * a[n-1-m+k]
    # We want m = j + n - 1, so a[n-1-(j+n-1)+k] = a[k-j]
    
    # (w * a_flipped) = ifft(W * fft(a_flipped))
    # fft(a_flipped)[k] = sum_j a[n-1-j] * exp(-2πijk/N)
    # Let m = n-1-j: = sum_m a[m] * exp(-2πi(n-1-m)k/N) = exp(-2πi(n-1)k/N) * sum_m a[m] * exp(2πimk/N)
    # = exp(-2πi(n-1)k/N) * conj(A[k]) (for real a)
    
    # So fft(a_flipped) = exp(-2πi(n-1)k/N) * conj(A[k])
    # And (w * a_flipped) = ifft(W * exp(-2πi(n-1)k/N) * conj(A))
    
    # Alternatively, just compute it directly with rfft:
    a_flipped = a[::-1]
    AF = np.fft.rfft(a_flipped, fft_size)
    conv_waf = np.fft.irfft(W * AF, fft_size)
    
    # grad[j] = conv_waf[j + n - 1] + conv_aw[j]
    grad = conv_waf[n - 1:n - 1 + n] + conv_aw
    
    # Clip tiny negative values from FFT numerical error
    grad = np.maximum(grad, 0)  # Actually the gradient should be non-negative for non-negative a and w
    
    return grad


def mirror_descent_step(f, grad, eta, tau):
    """One step of mirror descent with entropic smoothing.
    Minimizes the entropic approximation of max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Entropic weights
    max_c = conv.max()
    shifted = tau * (conv - max_c)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Concentrate on the max
        w = np.zeros_like(conv)
        idx = np.argmax(conv)
        w[idx] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    g = compute_grad(f, w)
    
    # Mirror descent update
    # f_new = f * exp(-eta * g)
    # But we need to handle the constraint sum(f) = 1, f >= 0
    
    f_new = f * np.exp(-eta * g)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-300:
        f_new = np.ones(n) / n
    else:
        f_new /= s
    
    return f_new, conv

            f = f_new
            best_obj = obj_new
            best_f = f.copy()

            # Increase tau slowly (sharpen the max approximation)
            tau = min(tau * tau_growth, tau_max)
            # Slightly increase eta if we're making good progress

def make_lp_structure(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=5):
    """Solve LP: max sum(g) s.t. g>=0, cross_conv(f,g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit):
    """LP-based alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-15:
        f = np.ones(n) / n
    else:
        f /= f_sum
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = make_lp_structure(n)
    
    t0 = time.monotonic()
    t = 0.5
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp(f, M, rows, cols, n, timeout=3)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            break
        
        improved = False
        for t_try in [t, 0.5 * t, 0.25 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-12:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.1)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.8
            if patience > 30:
                break
        
        if iter_count % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iter_count}, obj={best_obj:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g
    
    # Bimodal with varying gap
    for frac in [0.2, 0.3, 0.4]:
        w = max(1, int(n * frac))
        bi = np.zeros(n)
        bi[:w] = 1.0
        bi[-w:] = 1.0
        inits[f'bi{int(frac * 100)}'] = bi / bi.sum()
    
    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()
    
    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr
    
    # "Anti-palindromic" - high at ends, low in middle, with structure
    ap = np.maximum(1.0 - 2.0 * np.abs(x / n - 0.5), 0.0) ** 0.5
    # Add some oscillation
    ap += 0.1 * np.abs(np.sin(4 * np.pi * x / n))
    ap = np.maximum(ap, 0.01)
    inits['anti'] = ap / ap.sum()
    
    # "Fourier-flat" initialization: try to make the DFT flat
    # This gives approximately flat autocorrelation
    target_spectrum = np.ones(n)
    a_fourier = target_spectrum / n  # uniform spectrum
    a_init = np.fft.irfft(a_fourier, n)
    a_init = np.maximum(a_init, 0)
    if a_init.sum() > 1e-10:
        inits['spec'] = a_init / a_init.sum()
    
    return inits


def upsample(f, n_new):
    """Upsample a sequence to a new length using interpolation."""
    n_old = len(f)
    if n_new == n_old:
        return f.copy()
    if n_new > n_old:
        x_old = np.linspace(0, 1, n_old)
        x_new = np.linspace(0, 1, n_new)
        f_new = np.interp(x_new, x_old, f)
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = np.ones(n_new) / n_new
        return f_new
    else:
        # Downsample
        x_old = np.linspace(0, 1, n_old)
        x_new = np.linspace(0, 1, n_new)
        f_new = np.interp(x_new, x_old, f)
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = np.ones(n_new) / n_new
        return f_new
