"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP iteration with line search, soft-max gradient refinement,
and multi-start strategy. Focuses on lengths near 384.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    """Normalize to have sum = target."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def autocorr(a):
    return np.convolve(a, a)

def active_set_lp(f, M, top_k=50):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top_k largest lags, g >= 0.
    Returns g or None.
    """
    n = len(f)
    c = np.convolve(f, f)
    
    # Get the top_k lags by value
    if top_k < len(c):
        top_indices = np.argsort(c)[-top_k:]
    else:
        top_indices = np.arange(len(c))
    
    # Build constraint matrix for only these lags
    # A[m,j] = f[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((len(top_indices), n))
    for idx, m in enumerate(top_indices):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[idx, j] = f[m - j]
    
    b = np.full(len(top_indices), M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def full_lp(f, M):
    """Full LP with all 2n-1 constraints."""
    n = len(f)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, min(j + n, 2 * n - 1)):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def soft_max_gradient(a, tau=10.0):
    """Compute gradient of soft-max of autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad

def gradient_descent_step(a, lr, momentum_buf, momentum_decay, tau=10.0):
    """One step of gradient descent with momentum on soft-max objective."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a_curr = normalize(a, S_t)
    grad = soft_max_gradient(a_curr, tau)
    # Project out mean to maintain sum constraint
    grad -= grad.mean()
    momentum_buf[:] = momentum_decay * momentum_buf - lr * grad
    a_new = a_curr + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, momentum_buf

def hard_max_gradient(a):
    """Gradient of hard max autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad

def line_search_blend(f, g, S_t, best_score):
    """Find optimal blend parameter t for f_new = (1-t)f + t*g."""
    def obj_at_t(t):
        a_new = (1.0 - t) * f + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        return compute_obj(a_new)
    
    result = minimize_scalar(obj_at_t, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-4})
    return result.x, result.fun

    best_score = compute_obj(f)
    best_f = f.copy()

    for it in range(max_iters):

                break

        if g_sum < 1e-6:
            break

        # Line search
        t_opt, score_at_t = line_search_blend(f, g_norm, S_t, best_score)

            f_new = (1.0 - t_opt) * f + t_opt * g_norm
            f_new = np.maximum(f_new, 0.0)
            f_new = normalize(f_new, S_t)
            score = compute_obj(f_new)
            if score < best_score - 1e-10:

                best_f = f_new.copy()
                f = f_new

"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive active-set LP iteration with line search as the main engine,
interleaved with soft-max gradient descent for escaping local minima.
Focuses on n=384 (previous best length) with multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to have sum = target."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def active_set_lp(f, M, threshold=0.85):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for lags where f*f > threshold*M, g >= 0.
    Uses adaptive active set based on threshold.
    Returns (g, active_indices) or (None, None).
    """
    n = len(f)
    c = np.convolve(f, f)
    active_mask = c >= threshold * M
    active_indices = np.where(active_mask)[0]
    
    if len(active_indices) == 0:
        active_indices = np.array([np.argmax(c)])
    
    if len(active_indices) == len(c):
        # All constraints active - use full LP
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            for m in range(j, min(j + n, 2 * n - 1)):
                A[m, j] = f[m - j]
        b = np.full(2 * n - 1, M)
    else:
        A = np.zeros((len(active_indices), n))
        for idx, m in enumerate(active_indices):
            lo = max(0, int(m) - n + 1)
            hi = min(n - 1, int(m))
            for j in range(lo, hi + 1):
                A[idx, j] = f[m - j]
        b = np.full(len(active_indices), M)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x, active_indices
    except Exception:
        pass
    return None, None


def top_k_lp(f, M, k=40):
    """Solve LP with only top-k most active constraints."""
    n = len(f)
    c = np.convolve(f, f)
    top_indices = np.argsort(c)[-k:]
    
    A = np.zeros((k, n))
    for idx, m in enumerate(top_indices):
        lo = max(0, int(m) - n + 1)
        hi = min(n - 1, int(m))
        for j in range(lo, hi + 1):
            A[idx, j] = f[m - j]
    
    b = np.full(k, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def soft_max_gradient(a, tau=10.0):
    """Compute gradient of soft-max of autocorrelation w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def hard_max_gradient(a):
    """Gradient of hard max autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def gradient_step(a, lr, momentum_buf, decay, tau=10.0):
    """One gradient descent step with momentum on soft-max objective."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a_curr = normalize(a, S_t)
    grad = soft_max_gradient(a_curr, tau)
    grad -= grad.mean()
    momentum_buf[:] = decay * momentum_buf - lr * grad
    a_new = a_curr + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, momentum_buf

        if s < 1e-6:
            return 1e10

    try:
        result = minimize_scalar(obj_t, bounds=(0.0, 1.0), method='bounded',

        # Line search
        t_opt, score_at_t = line_search(f, g_norm, S_t)
        
        improved = False
        if t_opt > 0.005:
            f_new = (1.0 - t_opt) * f + t_opt * g_norm
            f_new = np.maximum(f_new, 0.0)
            f_new = normalize(f_new, S_t)
            score = compute_obj(f_new)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
                improved = True
                if it % 20 == 0:
                    print(f"      LP iter {it}: score={best_score:.6f}, t={t_opt:.3f}")
        
        if not improved:
            # Try a few fixed blend values
            for t_try in [0.5, 0.3, 0.2, 0.1]:
                f_new = (1.0 - t_try) * f + t_try * g_norm
                f_new = np.maximum(f_new, 0.0)
                f_new = normalize(f_new, S_t)
                score = compute_obj(f_new)
                if score < best_score - 1e-12:
                    best_score = score
                    best_f = f_new.copy()
                    f = f_new
                    no_improve = 0
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Try to escape with gradient kick
                grad = hard_max_gradient(f)
                grad -= grad.mean()
                f = f - 0.02 * grad
                f = np.maximum(f, 0.0)
                f = normalize(f, S_t)
                s = compute_obj(f)
                if s < best_score:
                    best_score = s
                    best_f = f.copy()
                elif no_improve > 25:
                    break
        
        # Periodically tighten the threshold to focus more
        if it % 50 == 0 and it > 0:
            threshold = max(0.7, threshold - 0.02)
    
    return best_f, best_score


def gradient_optimize(f, deadline, max_iters=5000):
    """Soft-max gradient descent with momentum."""
    n = len(f)
    S_t = np.sqrt(2.0 * n)
    f = normalize(f.copy(), S_t)
    best_score = compute_obj(f)
    best_f = f.copy()
    momentum_buf = np.zeros(n)
    lr = 0.05
    tau = 5.0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        f, momentum_buf = gradient_step(f, lr, momentum_buf, 0.9, tau=tau)
        
        if it % 200 == 0:
            s = compute_obj(f)
            if s < best_score - 1e-9:
                best_score = s
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 15:
                    break
        
        # Anneal learning rate and temperature
        lr = max(0.001, 0.05 * (0.9995 ** it))
        tau = min(50.0, 5.0 + 3.0 * (it / 300.0))
    
    return best_f, best_score


def optimize_from_start(a0, deadline, max_time_single=None):
    """Full optimization: LP + gradient alternation."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    a = normalize(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    if max_time_single is None:
        max_time_single = max(20.0, deadline - time.monotonic() - 30.0)
    single_deadline = time.monotonic() + max_time_single
    
    print(f"    n={n}, initial_score={score:.6f}")
    
    # Phase 1: LP optimization (main engine)
    t1 = time.monotonic()
    phase1_end = min(single_deadline - max(30, max_time_single * 0.3),
                     t1 + max_time_single * 0.5)
    best, best_score = lp_optimize(best, phase1_end, max_iters=400, threshold=0.85)
    print(f"    After LP (phase 1): score={best_score:.6f}")
    
    # Phase 2: Gradient refinement
    if time.monotonic() < single_deadline - max(20, max_time_single * 0.2):
        phase2_end = min(single_deadline - max(10, max_time_single * 0.15),
                         time.monotonic() + max_time_single * 0.2)
        best, best_score = gradient_optimize(best, phase2_end, max_iters=4000)
        print(f"    After gradient (phase 2): score={best_score:.6f}")
    
    # Phase 3: Second LP round (tighter threshold)
    if time.monotonic() < single_deadline - max(10, max_time_single * 0.1):
        phase3_end = single_deadline - max(10, max_time_single * 0.05)
        best, best_score = lp_optimize(best, phase3_end, max_iters=300, threshold=0.75, top_k=50)
        print(f"    After LP (phase 3): score={best_score:.6f}")
    
    # Phase 4: Final gradient polish
    if time.monotonic() < single_deadline - 5:
        best, best_score = gradient_optimize(best, single_deadline - 3, max_iters=2000)
        print(f"    After final gradient: score={best_score:.6f}")
    
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses adaptive active-set LP with line search as primary
    optimizer, alternating with soft-max gradient descent. Multi-start with
    structured initializations focused on n=384.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    print(f"Starting optimization, budget={budget_s}s, deadline in {deadline - t_start:.1f}s")
    
    # Build starting points
    starts = []
    
    # Priority 0: height_sequence_1
    has_gs = False
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)) and np.all(gs >= 0):
            starts.append(('gs', gs.copy()))
            has_gs = True
            print(f"Loaded height_sequence_1: n={len(gs)}, score={compute_obj(gs):.6f}")
    except (NameError, Exception):
        print("height_sequence_1 not available")
    
    # Structured starts for key lengths
    for n in [384, 320, 448]:
        i = np.arange(n, dtype=np.float64)
        # Cosine modulations at various frequencies
        for freq in [2, 3, 5, 7]:
            a = 0.35 + 0.25 * np.cos(2 * np.pi * i * freq / n)
            starts.append((f'cos{freq}_{n}', a))
        # Random
        a = rng.random(n) * 0.3 + 0.25
        starts.append((f'rand_{n}', a))
        # Slowly varying
        a = 0.3 + 0.2 * np.sin(np.pi * i / n)
        starts.append((f'sin_{n}', a))
    
    # For n=256 (faster optimization)
    n = 256
    i = np.arange(n, dtype=np.float64)
    starts.append(('cos3_256', 0.35 + 0.25 * np.cos(2 * np.pi * i * 3.0 / n)))
    starts.append(('rand_256', rng.random(n) * 0.3 + 0.25))
    
    print(f"Total starts: {len(starts)}")
    
    # Sort: height_sequence_1 first, then by length (384 preferred)
    def sort_key(item):
        name, seq = item
        if name == 'gs':
            return (0, len(seq))
        return (1, abs(len(seq) - 384), len(seq))
    
    starts.sort(key=sort_key)
    
    # Allocate time: give the most time to the most promising starts
    results = []
    
    for idx, (name, a0) in enumerate(starts):
        if time.monotonic() > deadline - 90:
            break
        
        remaining = deadline - time.monotonic() - 60
        if remaining < 40:
            break
        
        # Time allocation strategy
        n_start = len(a0)
        if name == 'gs':
            t_alloc = min(500.0, remaining * 0.35)
        elif n_start == 384 and idx < 4:
            t_alloc = min(300.0, remaining * 0.2)
        elif n_start == 384:
            t_alloc = min(180.0, remaining * 0.15)
        elif n_start == 256:
            t_alloc = min(120.0, remaining * 0.1)
        else:
            t_alloc = min(150.0, remaining * 0.12)
        
        t_alloc = max(25.0, t_alloc)
        if time.monotonic() + t_alloc > deadline - 60:
            t_alloc = deadline - time.monotonic() - 60
        if t_alloc < 25:
            break
        
        print(f"\n[Start {idx}] '{name}', n={n_start}, time={t_alloc:.0f}s, "
              f"remaining={remaining:.0f}s")
        
        seq, score = optimize_from_start(a0, deadline, max_time_single=t_alloc)
        results.append((score, n_start, name, seq))
        
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  >>> New best: score={best_score:.6f}, n={n_start}, '{name}'")
        else:
            print(f"  score={score:.6f} (current best: {best_score:.6f})")
    
    # --- Second pass: intensive refinement of the best ---
    if best_seq is not None and time.monotonic() < deadline - 120:
        print(f"\n{'='*50}")
        print(f"Second pass: intensive refinement of best (score={best_score:.6f})")
        print(f"{'='*50}")
        
        n = len(best_seq)
        remaining = deadline - time.monotonic() - 30
        t_alloc = min(remaining, 600.0)
        
        # Slight perturbation to explore neighborhood
        a0 = best_seq + 0.015 * rng.randn(n)
        a0 = np.maximum(a0, 0.0)
        
        seq, score = optimize_from_start(a0, deadline, max_time_single=t_alloc)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Second pass new best: score={best_score:.6f}")
        else:
            print(f"  Second pass: score={score:.6f} (no improvement)")
    
    # --- Third pass: pure LP polishing on the best with various parameters ---
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*50}")
        print(f"Third pass: LP polish (current best: {best_score:.6f})")
        print(f"{'='*50}")
        
        remaining = deadline - time.monotonic() - 15
        # Try different thresholds
        for thr in [0.80, 0.70, 0.60]:
            if time.monotonic() > deadline - 20:
                break
            t_sub = min(200.0, remaining / 3.0)
            if t_sub < 20:
                break
            seq, score = lp_optimize(best_seq, deadline - 20,
                                      max_iters=300, threshold=thr, top_k=45)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  LP (thr={thr}): new best score={best_score:.6f}")
            else:
                print(f"  LP (thr={thr}): score={score:.6f}")
    
    # --- Final: quick gradient polish ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        seq, score = gradient_optimize(best_seq, deadline - 5, max_iters=3000)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Final gradient: new best score={best_score:.6f}")
    
    # --- Final validation ---
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.minimum(best_seq, 1000.0)
        final_score = compute_obj(best_seq)
        print(f"\n{'='*50}")
        print(f"FINAL: n={len(best_seq)}, score={final_score:.6f}")
        print(f"{'='*50}")
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 384
