"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Uses log-sum-exp smoothed L-BFGS-B optimization with softmax
parameterization, projected subgradient methods, and targeted local search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize to given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)


def smooth_obj_and_grad_x(x, S, beta):
    """
    Compute smoothed objective and gradient w.r.t. x (softmax parameterization).
    a = S * softmax(x)
    F = (1/beta) log(sum_k exp(beta * conv(a,a)[k]))
    """
    n = len(x)
    a = S * softmax(x)
    
    conv_aa = fft_convolve(a, a)
    
    # Log-sum-exp
    m = np.max(conv_aa)
    log_sum_exp = m + np.log(np.sum(np.exp(beta * (conv_aa - m))))
    obj = log_sum_exp / beta
    
    # Probabilities
    p = np.exp(beta * (conv_aa - m))
    p /= np.sum(p)
    
    # Gradient w.r.t. a
    flip_a = a[::-1]
    conv1 = fft_convolve(flip_a, p)
    term1 = conv1[n-1:2*n-1]
    conv2 = fft_convolve(a, p)
    term2 = conv2[0:n]
    grad_a = term1 + term2
    
    # Chain rule through softmax: da[i]/dx[k] = a[i](delta[ik] - a[k]/S)
    # dF/dx[k] = a[k] * (grad_a[k] - (1/S) * sum_i a[i]*grad_a[i])
    mean_grad = np.dot(a, grad_a) / S
    grad_x = a * (grad_a - mean_grad)
    
    return obj, grad_x, a


def lbfgs_smooth(a0, S, beta, time_limit, max_iter=5000, verbose=False):
    """
    Run L-BFGS-B on the softmax-parameterized smoothed problem.
    """
    n = len(a0)
    t0 = time.monotonic()
    
    # Initialize x from a0
    a_norm = normalize_seq(a0, S)
    a_floor = np.maximum(a_norm, 1e-10)
    x0 = np.log(a_floor / S)  # log of the softmax values
    
    func_counter = [0]
    
    def fun_and_grad(x):
        func_counter[0] += 1
        obj, grad, _ = smooth_obj_and_grad_x(x, S, beta)
        return obj, grad
    
    result = minimize(fun_and_grad, x0, method='L-BFGS-B',
                      options={'maxiter': max_iter, 
                               'maxfun': 20000,
                               'ftol': 1e-14,
                               'gtol': 1e-10,
                               'maxls': 50})
    
    x_opt = result.x
    a_opt = S * softmax(x_opt)
    a_opt = normalize_seq(a_opt, S)
    
    elapsed = time.monotonic() - t0
    if verbose:
        peak = np.max(fft_convolve(a_opt, a_opt))
        print(f"    L-BFGS-B (beta={beta}): peak={peak:.6f}, fcnt={func_counter[0]}, "
              f"iters={result.niter}, time={elapsed:.1f}s, success={result.success}")
    
    return a_opt

def targeted_search(a0, S, time_limit, verbose=False):
    """
    Targeted local search: reduce peak by adjusting contributing elements.
    """
    n = len(a0)
    a = normalize_seq(a0.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(42)
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_aa = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv_aa))
        current_peak = conv_aa[peak_idx]
        
        # Find top contributors to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j1 = peak_idx - i
            j2 = i - peak_idx
            val = 0.0
            if 0 <= j1 < n:
                val += a[i] * a[j1]
            if 0 <= j2 < n:
                val += a[i] * a[j2]
            contribs[i] = val
        
        # Strategy: reduce top contributors, increase low contributors
        top_k = max(5, n // 10)
        top_idx = np.argsort(contribs)[-top_k:]
        bottom_k = max(5, n // 10)
        bottom_idx = np.argsort(contribs)[:bottom_k]
        
        for attempt in range(3):
            candidate = a.copy()
            scale_down = np.exp(rng.normal(-0.03, 0.015, size=len(top_idx)))
            candidate[top_idx] *= scale_down
            
            scale_up = np.exp(rng.normal(0.03, 0.015, size=len(bottom_idx)))
            candidate[bottom_idx] *= scale_up
            
            candidate = np.maximum(candidate, 1e-10)
            candidate = normalize_seq(candidate, S)
            
            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = candidate.copy()
                a = candidate
                break
        else:
            # If no improvement in 3 attempts, try random perturbation
            candidate = a.copy()
            mask = rng.random(n) < 0.3
            candidate[mask] *= np.exp(rng.normal(0, 0.02, size=mask.sum()))
            candidate = np.maximum(candidate, 1e-10)
            candidate = normalize_seq(candidate, S)
            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = candidate.copy()
                a = candidate
            elif new_peak < current_peak * 1.02 and rng.random() < 0.2:
                a = candidate  # Accept non-improving move (simulated annealing-like)
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Targeted it={it}: best_peak={best_peak:.6f}, t={elapsed:.0f}s")
    
    return best_a


def build_lp_matrix(a):
    """Build convolution constraint matrix."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_refine(a0, S, time_limit, verbose=False):
    """LP-based iterative refinement."""
    n = len(a0)
    a = normalize_seq(a0, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        A = build_lp_matrix(a)
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        c_norm = normalize_seq(g, S)
        
        # Find best blend
        best_t = 0.0
        best_conv = M
        for t_try in [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03]:
            b = (1 - t_try) * a + t_try * c_norm
            b = np.maximum(b, 0.0)
            b = normalize_seq(b, S)
            conv_max = np.max(fft_convolve(b, b))
            if conv_max < best_conv:
                best_conv = conv_max
                best_t = t_try
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * c_norm
        a = np.maximum(a, 0.0)
        a = normalize_seq(a, S)
        
        new_peak = np.max(fft_convolve(a, a))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a.copy()
        
        if verbose and iteration % 50 == 0:
            print(f"    LP iter={iteration}: peak={best_peak:.6f}, t={best_t:.3f}")
    
    return best_a


def make_initialization(n, init_type, rng):
    """Create initial sequences of various types."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 4))
        amp = 0.2 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, n // 4))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.15 + 0.1 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'triangular':
        center = n // 2
        a = 1.0 + 0.3 * np.abs(1 - 2 * np.abs(t - center) / n)
    else:
        a = np.ones(n)
    
    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

        if time.monotonic() > deadline - 500:

        S_n = np.sqrt(2.0 * n)

            new_sc = local_score(current)
            if update_best(current):

                result = lp_refine(best_seq, S_n, t_limit, verbose=True)
                update_best(result)
                print(f"  After LP: {best_score:.6f}")
    
    # === Phase 5: More L-BFGS-B at very high beta ===
    print("\n=== Phase 5: High-beta L-BFGS-B ===")
    
    if best_seq is not None:
        n = len(best_seq)
        S_n = np.sqrt(2.0 * n)
        for beta in [2000, 5000]:

            result = lbfgs_smooth(best_seq, S_n, beta=beta, time_limit=t_per, verbose=True)
            update_best(result)
    
    # === Phase 6: Targeted search ===
    print("\n=== Phase 6: Targeted Search ===")
    
    if best_seq is not None:
        n = len(best_seq)
        S_n = np.sqrt(2.0 * n)
        t_limit = max(30.0, deadline - time.monotonic() - 20)
        if t_limit > 30:
            print(f"  Targeted search, time_limit={t_limit:.0f}s")
            result = targeted_search(best_seq, S_n, t_limit, verbose=True)
            update_best(result)
            print(f"  After targeted: {best_score:.6f}")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    
    return [1.0] * 256
