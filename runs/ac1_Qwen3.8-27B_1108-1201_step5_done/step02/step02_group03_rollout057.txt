"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses a multi-phase approach: diverse initialization, gradient descent with adaptive
step sizes, LP-based refinement with warm starts, and coordinate descent polishing.
Key improvements over previous: BB step size, top-K constraint LP, log-parameterization
exploration, and more aggressive LP blending.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def soft_max(c, beta):
    """Compute soft-max of c with temperature parameter beta."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    return c_max + np.log(e.sum() / len(c)) / beta

def gradient_soft(a, beta):
    """Gradient of soft_max(conv(a,a), beta) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w /= w.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # Using convolution: grad = 2 * (w * a_reversed) at appropriate offset
    a_rev = a[::-1]
    conv_w_a = np.convolve(w, a_rev)
    grad = 2.0 * conv_w_a[n-1:n-1+n]
    return grad

def bb_step_size(a_prev, a_curr, f_prev, f_curr, g_prev, g_curr):
    """Barzilai-Borwein step size (s/y variant)."""
    s = a_curr - a_prev
    y = g_curr - g_prev
    sy = np.dot(s, y)
    if abs(sy) < 1e-12:
        return 0.1
    ss = np.dot(s, s)
    yy = np.dot(y, y)
    if ss < 1e-12:
        return 0.1
    # Alternate between s/sy and y/sy
    alpha_s = ss / sy if sy > 0 else 0.1
    alpha_y = sy / yy if yy > 0 else 0.1
    # Return the one closer to the previous step size
    return max(0.001, min(1.0, alpha_s))

def gradient_descent_phase(a, deadline, max_iter=20000):
    """Gradient descent with BB step size and momentum."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    beta = 10.0
    momentum = np.zeros(n)
    mom_decay = 0.9
    base_lr = 0.1
    a_prev = a.copy()
    g_prev = gradient_soft(normalize(a_prev, S_t), beta)
    g_prev -= g_prev.mean()  # Project to maintain sum
    
    it = 0
    no_improve = 0
    last_eval = time.monotonic()
    
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        
        # Anneal beta
        beta = 10.0 + 10.0 * (it / 500.0)
        beta = min(beta, 100.0)
        
        # Anneal learning rate
        lr = base_lr * (0.9995 ** it)
        lr = max(lr, 0.005)
        
        g = gradient_soft(a, beta)
        g -= g.mean()  # Project to sum-zero subspace
        
        momentum = mom_decay * momentum - lr * g
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        
        # Periodic evaluation
        if it % 50 == 0:
            s = compute_obj(a_new)
            if s < best_score - 1e-8:
                best_score = s
                best = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    # Reset
                    a = best.copy()
                    momentum = np.zeros(n)
                    no_improve = 0
                    base_lr *= 0.5
                    if base_lr < 0.001:
                        break
            
            a = a_new
            g_prev = g
            a_prev = a.copy()
    
    # Final evaluation
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score

def lp_refine_topk(a, M, K=50, method='highs'):
    """
    LP: max sum(g) s.t. top K convolution constraints with a, g >= 0.
    Only uses the K most binding constraints for speed.
    """
    n = len(a)
    if n > 1024:
        return None
    
    c = np.convolve(a, a)
    # Get top K lags
    top_k = np.argsort(c)[-K:][::-1]
    
    # Build constraint matrix for only these K rows
    A_rows = []
    for m in top_k:
        row = np.zeros(n)
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            row[j] = a[m - j]
        A_rows.append(row)
    
    A = np.array(A_rows)
    b = np.full(K, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_refine_full(a, M, method='highs'):
    """Full LP with all 2n-1 constraints."""
    n = len(a)
    if n > 768:
        return None
    
    c_obj = -np.ones(n)
    
    # Build the full constraint matrix efficiently
    # A[m, j] = a[m-j] if 0 <= m-j < n, else 0
    # For each j, a[j] appears in rows j through j+n-1
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = a[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_phase(a, deadline, max_iters=200):
    """Iterative LP-based refinement."""
    n = len(a)

def coordinate_descent_polish(a, deadline, max_rounds=5):

        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() > deadline:
                break

                orig_i, orig_j = a[i], a[j]
                total = orig_i + orig_j

                else:
                    a[i] = orig_i

def generate_initializations(rng, n):

    starts = []
    i = np.arange(n, dtype=np.float64)
    
    # Uniform
    starts.append(np.ones(n) * 0.5)
    
    # Random
    starts.append(rng.random(n) * 0.5 + 0.25)
    
    # Cosine windows
    for freq in [1, 2, 3, 5, 7]:
        starts.append(0.3 + 0.7 * np.abs(np.cos(2 * np.pi * freq * i / n)))
    
    # Concentrated at start
    starts.append(np.exp(-3.0 * i / n))
    
    # Concentrated at end
    starts.append(np.exp(-3.0 * (n - 1 - i) / n))
    
    # Two bumps
    starts.append(np.exp(-5.0 * ((i - n*0.25)/n)**2) + np.exp(-5.0 * ((i - n*0.75)/n)**2))
    
    # Anti-symmetric (concentrated at odd positions)
    a = np.zeros(n)
    a[::2] = 1.0
    starts.append(a + 0.1)
    
    # Chirp
    starts.append(0.3 + 0.7 * np.abs(np.sin(2 * np.pi * i**2 / (4*n))))
    
    # Gaussian bump in middle

    # Periodic with randomness
    starts.append(0.3 + 0.3 * np.cos(2 * np.pi * 3 * i / n) + rng.random(n) * 0.2)
    
    return starts

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses multi-phase approach: diverse initialization, Nesterov gradient descent with
soft-max annealing, aggressive LP-based refinement with warm-starting, primal-dual
gradient guidance, and coordinate descent polishing. Key improvements: BB/Nesterov
step sizes, full-replacement LP blending, top-K constraint LP for larger n, and
adaptive active set management.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def soft_max(c, beta):
    """Compute soft-max of c with temperature parameter beta."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    return c_max + np.log(e.sum() / len(c)) / beta

def gradient_soft(a, beta):
    """Gradient of soft_max(conv(a,a), beta) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_w_a = np.convolve(w, a_rev)
    grad = 2.0 * conv_w_a[n-1:n-1+n]
    return grad

def gradient_hard(a):
    """Gradient using only the hard max (single peak)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad

def nesterov_gradient_phase(a, deadline, max_iter=30000):
    """Nesterov-accelerated gradient descent with soft-max annealing."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    velocity = np.zeros(n)
    mom = 0.9
    base_lr = 0.5
    beta = 5.0
    it = 0
    no_improve = 0
    lr = base_lr
    
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        beta = 5.0 + 15.0 * min(1.0, it / 3000.0)
        lr = base_lr * (0.9998 ** it)
        lr = max(lr, 0.002)
        
        # Nesterov lookahead
        y = a + mom * velocity
        y = np.maximum(y, 0.0)
        y = normalize(y, S_t)
        
        g = gradient_soft(y, beta)
        g -= g.mean()
        
        velocity = mom * velocity - lr * g
        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        
        if it % 100 == 0:
            s = compute_obj(a_new)
            if s < best_score - 1e-9:
                best_score = s
                best = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 15:
                    a = best.copy()
                    velocity = np.zeros(n)
                    no_improve = 0
                    base_lr *= 0.6
                    if base_lr < 0.001:
                        break
            a = a_new
    
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    return best, best_score

def build_lp_matrix(a, rows=None):
    """Build LP constraint matrix A where A[m,j] = a[m-j] if valid."""
    n = len(a)
    if rows is None:
        rows = np.arange(2 * n - 1)
    A = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[idx, j] = a[m - j]
    return A

def solve_lp(a, M, K=None, time_limit=None):
    """Solve LP: max sum(g) s.t. A(a)@g <= M, g >= 0. Returns (g, dual_y)."""
    n = len(a)
    if n > 1024:
        return None, None
    
    if K is None or K >= 2 * n - 1:
        rows = np.arange(2 * n - 1)
    else:
        c = np.convolve(a, a)
        top_k = np.argsort(c)[-K:][::-1]
        rows = top_k
    
    A = build_lp_matrix(a, rows)
    b = np.full(len(rows), M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    opts = {'presolve': True}
    if time_limit:
        opts['time_limit'] = time_limit
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs', options=opts)
        if res.status == 0:
            dual = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return res.x, dual
    except Exception:
        pass
    return None, None

def lp_refinement_phase(a, deadline, max_iters=300):
    """Aggressive LP-based refinement with full replacement and gradient alternation."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)

                y = np.maximum(y, 0)
                if y.sum() > 1e-12:
                    y /= y.sum()
                    grad_dual = A_top.T @ y

                    grad_dual = gradient_soft(a, 20.0)
            else:
                grad_dual = gradient_soft(a, 20.0)

            s_nudge = compute_obj(a_nudge)
            if s_nudge < best_score:
                a = a_nudge
                best = a.copy()
                best_score = s_nudge
                no_improve = 0

                # Try random perturbation to escape

                if s_pert < best_score:
                    a = a_pert
                    best = a.copy()
                    best_score = s_pert
                    no_improve = 0
                else:
                    a = best.copy()
                    no_improve = 0
                    # Reduce K to make LP easier
                    K = max(30, K // 2)
                    if no_improve > 15:
                        break
        else:
            no_improve = 0
    
    return best, best_score

def coordinate_descent_polish(a, deadline, max_rounds=5):
    """Pairwise coordinate descent for final polishing."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    
    # Precompute step size for iterating
    step = max(1, n // 300)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        
        for i in range(0, n, step):
            if time.monotonic() > deadline:
                break
            j = (i + 1) % n
            if j == i:
                continue
            orig_i, orig_j = a[i], a[j]
            total = orig_i + orig_j
            if total < 1e-12:
                continue
            
            best_frac = 0.5
            best_s = best_score
            for frac in [0.0, 0.15, 0.3, 0.4, 0.5, 0.6, 0.7, 0.85, 1.0]:
                a[i] = frac * total
                a[j] = (1.0 - frac) * total
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_s - 1e-12:
                    best_s = s
                    best_frac = frac
                a[i] = orig_i
                a[j] = orig_j
            
            if best_frac != 0.5:
                a[i] = best_frac * total
                a[j] = (1.0 - best_frac) * total
                a = normalize(a, S_t)
                best_score = best_s
                improved = True
        
        if not improved:
            break
    
    return a, best_score

def generate_initializations(rng, n):
    """Generate diverse initializations for a given length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)
    
    # Uniform
    starts.append(np.ones(n) * 0.5)
    
    # Random
    starts.append(rng.random(n) * 0.5 + 0.25)
    
    # Cosine windows (various frequencies)
    for freq in [1, 2, 3, 5, 7, 11]:
        starts.append(0.3 + 0.7 * np.abs(np.cos(2 * np.pi * freq * i / n)))
    
    # Exponential decay (start-concentrated)
    starts.append(np.exp(-2.5 * i / n) + 0.05)
    
    # Exponential decay (end-concentrated)
    starts.append(np.exp(-2.5 * (n - 1 - i) / n) + 0.05)
    
    # Two symmetric bumps
    starts.append(np.exp(-4.0 * ((i - n * 0.25) / (n * 0.3))**2) + np.exp(-4.0 * ((i - n * 0.75) / (n * 0.3))**2) + 0.05)
    
    # Three bumps
    starts.append(np.exp(-5.0 * ((i - n * 0.2) / (n * 0.15))**2) + np.exp(-5.0 * ((i - n * 0.5) / (n * 0.15))**2) + np.exp(-5.0 * ((i - n * 0.8) / (n * 0.15))**2) + 0.03)
    
    # Sparse (every other)
    a_sparse = np.zeros(n)
    a_sparse[::2] = 1.0
    starts.append(a_sparse * 0.5 + 0.2)
    
    # Chirp
    starts.append(0.3 + 0.7 * np.abs(np.sin(2 * np.pi * i**2 / (6.0 * n))))
    
    # Gaussian in middle
    starts.append(np.exp(-0.5 * ((i - n / 2) / (n * 0.2))**2) + 0.05)
    
    # Modulated
    starts.append(0.3 + 0.2 * np.cos(2 * np.pi * 3 * i / n) + 0.2 * np.cos(2 * np.pi * 7 * i / n) + 0.1 + rng.random(n) * 0.15)
    
    return starts
