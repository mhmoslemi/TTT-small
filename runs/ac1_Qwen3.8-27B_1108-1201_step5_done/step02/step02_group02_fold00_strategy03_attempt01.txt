import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf
    n = len(sequence)
    total_sum = np.sum(sequence)
    if total_sum < 0.01:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    # Convert all elements to float for consistency
    sequence_float = [float(x) for x in sequence]
    # Clip negative numbers
    sequence_float = [max(0.0, x) for x in sequence_float]
    # Create a copy to clip from both ends
    sequence_clipped = [max(0.0, min(sequence_float[i], total_sum/2)) for i in range(n)]
    # Now compute the actual score
    conv = np.convolve(sequence_float, sequence_float, mode='full')
    score = 2.0 * n * np.max(conv) / (total_sum ** 2)
    if score > 1000000:  # Guard against astronomical values
        return np.inf
    return score

def resample_sequence(seq, new_n):
    """Resample the sequence to a new length using linear interpolation."""
    if not isinstance(seq, np.ndarray) and not isinstance(seq, list):

    if not isinstance(seq, np.ndarray):

    old_t = np.linspace(0, 1, len(seq))
    new_t = np.linspace(0, 1, new_n)
    return np.interp(new_t, old_t, seq)
