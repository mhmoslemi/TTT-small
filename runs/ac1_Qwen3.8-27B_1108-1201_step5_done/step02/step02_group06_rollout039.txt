"""
Multi-phase optimization for minimizing normalized peak self-convolution
of non-negative step functions. Combines Nesterov-momentum gradient descent
with softmax-smoothed peak, targeted simulated annealing, and diverse
multi-start initialization including chirp, multitone, and two-level sequences.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S ** 2)

def compute_score_and_grad(a, tau=0.5):
    """Compute score and gradient using softmax smoothing."""
    a = np.clip(a, 0.0, None)
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf, np.zeros(n)
    
    c = fft_convolve(a, a)  # length 2n-1
    # Softmax weights
    c_shifted = (c - np.max(c)) / tau
    w = np.exp(c_shifted)
    w_sum = np.sum(w)
    w = w / w_sum
    
    M_smooth = np.dot(w, c)
    score = 2.0 * n * M_smooth / (S ** 2)
    
    # Gradient of M_smooth w.r.t. a
    # dM/da[j] = sum_k w[k] * a[k-j] (for 0<=k-j<n) + a[j]*w[2j] (for 2j < 2n-1)
    # The first term is a cross-correlation
    # g[j] = sum_{m=0}^{n-1} w[j+m] * a[m] (where j+m < 2n-1)
    
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.fft(w, size)
    A = np.fft.fft(a, size)
    # Cross-correlation: correlate(w, a)[k] = sum_m w[m+k]*a[m]
    # Using FFT: ifft(fft(w) * conj(fft(a))) gives circular correlation
    # For linear, we need zero-padding
    corr = np.fft.ifft(W * np.conj(A)).real[:n]
    
    # Add diagonal correction
    # For j where 2j < 2n-1, add a[j]*w[2j]
    diag_idx = np.arange(n)
    w_2j_idx = 2 * diag_idx
    valid_diag = w_2j_idx < (2 * n - 1)
    corr[valid_diag] += a[valid_diag] * w[w_2j_idx[valid_diag]]
    
    # Full gradient of score
    grad = (2.0 * n / (S ** 2)) * corr - (4.0 * n * M_smooth / (S ** 3)) * 1.0
    
    return score, grad

def nesterov_optimize(a_init, time_limit, tau_init=None, lr_init=0.5, verbose=False):
    """Nesterov momentum gradient descent with adaptive parameters."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    
    if tau_init is None:
        tau_init = 2.0 / n
    
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    step = 0
    lr = lr_init
    
    # Anneal tau from tau_init to tau_init * 0.05
    tau_final = tau_init * 0.05
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_final / tau_init) ** elapsed_frac
        # Decay learning rate
        lr = lr_init / (1.0 + 0.005 * step)
        
        # Nesterov: look-ahead step
        a_look = a + v * lr
        a_look = np.clip(a_look, 0.0, None)
        
        score_l, grad = compute_score_and_grad(a_look, tau)
        
        if not np.isfinite(score_l):
            a_look = a.copy()
            score_l, grad = compute_score_and_grad(a_look, tau)
        
        # Update momentum
        v = 0.9 * v - lr * grad
        
        # Take step
        a_new = a + v
        a_new = np.clip(a_new, 0.0, None)
        
        # Normalize to keep sum constant
        S_new = np.sum(a_new)
        if S_new > 1e-10:
            a_new = a_new * (S / S_new)
        
        # Evaluate
        new_score = compute_score(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # Don't accept, reset momentum partially
            v = v * 0.5
        
        step += 1
        if verbose and step % 500 == 0:
            print(f"    Nesterov step={step}, score={current_score:.6f}, "
                  f"best={best_score:.6f}, tau={tau:.4f}, lr={lr:.4f}")
    
    return best_a, best_score

def sa_refine(a_init, time_limit, rng, n_iter_per_move=1):
    """Simulated annealing with targeted moves."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9999
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.3:
            # Peak-targeted: reduce values that contribute most to the peak
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            # Find which pairs contribute to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            # Compensate by increasing low-contributing values
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(5, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.5:
            # Block modulation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.7:
            # Two-point swap-like: increase one, decrease another
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            while i2 == i1:
                i2 = int(rng.integers(n))
            delta = rng.normal(0.03, 0.01)
            candidate[i1] *= np.exp(delta)
            candidate[i2] *= np.exp(-delta)
        elif r < 0.85:
            # Broad small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        else:
            # Scale a random subset
            k = int(rng.integers(1, max(2, n // 4)))
            idx = rng.choice(n, size=k, replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.02, size=k))
        
        candidate = np.clip(candidate, 0.0, None)
        cand_S = np.sum(candidate)
        if cand_S < 1e-10:
            continue
        candidate = candidate * (S / cand_S)
        
        new_score = compute_score(candidate)
        step += 1
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score
        
        temperature *= cooling
        if temperature < 1e-6:
            temperature = 1e-6
    
    return best_a, best_score

def lp_polish(f_init, time_limit, t_init=0.25):
    """LP-based polishing (kept from previous approach, but more conservative)."""
    from scipy.optimize import linprog
    
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    t = t_init
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        M = np.max(c)
        
        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = f
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 20})
        
        if not result.success:
            break
        
        g = result.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (S / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (S / sum_new)
        
        score_new = compute_score(f_new)
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new
        else:
            f = f_new
            t *= 0.8
        
        if t < 0.01:
            t = 0.1
    
    return best_f, best_score

def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.05, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

def make_multitone(n, rng, n_tones=5):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

def make_two_level(n, rng, lo=0.3, hi=1.5):
    """Two-level sequence: random binary mapped to [lo, hi]."""
    binary = rng.random(n) < 0.5
    a = np.where(binary, lo, hi)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

def make_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

def make_windowed(n, rng, shape='raised_cos'):
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    if shape == 'raised_cos':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif shape == 'sinc':
        a = np.sinc((x - 0.5) * 2)
    else:
        a = np.ones(n)
    a = np.clip(a, 0.05, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S ** 2)


def compute_score_and_grad(a, tau):
    """Compute smoothed score and its gradient."""
    a = np.clip(a, 0.0, None)
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf, np.zeros(n)

    c = fft_convolve(a, a)
    c_shifted = (c - np.max(c)) / tau
    w = np.exp(c_shifted)
    w = w / np.sum(w)

    M_smooth = np.dot(w, c)
    score = 2.0 * n * M_smooth / (S ** 2)

    # Gradient: dM/da[j] = 2 * sum_m w[j+m] * a[m]
    # This is 2 * cross_correlation(w, a) at lags 0..n-1
    corr = np.correlate(w, a, mode='full')[:n]
    grad_M = 2.0 * corr

    grad = (2.0 * n / (S ** 2)) * grad_M - (4.0 * n * M_smooth / (S ** 3)) * 1.0
    return score, grad


def nesterov_optimize(a_init, time_limit, tau_init=None, lr_init=0.8, momentum=0.9):
    """Nesterov momentum gradient descent with annealed temperature."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)

    if tau_init is None:
        tau_init = 3.0 / n
    tau_final = max(0.1 / n, tau_init * 0.03)

    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score

    v = np.zeros(n)
    t0 = time.monotonic()
    step = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_final / tau_init) ** elapsed_frac
        lr = lr_init / (1.0 + 0.003 * step)

        a_look = a + v * lr
        a_look = np.clip(a_look, 0.0, None)
        S_look = np.sum(a_look)
        if S_look < 1e-10:
            a_look = a.copy()

        score_l, grad = compute_score_and_grad(a_look, tau)

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, None)

        S_new = np.sum(a_new)
        if S_new < 1e-10:
            v = v * 0.5
            step += 1
            continue
        a_new = a_new * (S / S_new)

        new_score = compute_score(a_new)
        step += 1

        if np.isfinite(new_score) and new_score < current_score:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            v = v * 0.3

def log_domain_optimize(a_init, time_limit, rng):

    n = len(a_init)
    S = np.sum(a_init)
    f_safe = np.clip(a_init, 1e-10, None)
    theta = np.log(f_safe / S)

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 1.5

    best_a = a_init.copy()
    best_score = compute_score(a_init)
    t0 = time.monotonic()
    step = 0
    tau_start = 3.0 / n
    tau_end = 0.15 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        t_shift = theta - np.max(theta)
        e = np.exp(t_shift)
        a = e / np.sum(e) * S

        c = fft_convolve(a, a)
        c_shifted = (c - np.max(c)) / tau
        w = np.exp(c_shifted)
        w = w / np.sum(w)

        M_smooth = np.dot(w, c)

        # Gradient of M_smooth w.r.t. a
        corr = np.correlate(w, a, mode='full')[:n]
        grad_M = 2.0 * corr

        # grad_M w.r.t. theta: chain rule
        # a = softmax(theta) * S, so da/dtheta[j] = a[j] - mean(a)*1 (in log-param)
        # Actually: a[j] = exp(theta[j]) * S / sum(exp(theta))
        # da[j]/dtheta[k] = a[j] * (delta[jk] - exp(theta[k])/sum(exp))
        # = a[j] * (delta[jk] - a[k]/S)
        # So grad_theta[k] = sum_j grad_M[j] * a[j] * (delta[jk] - a[k]/S)
        # = grad_M[k] * a[k] - (sum_j grad_M[j]*a[j]) * a[k] / S

        inner = np.dot(grad_M, a) / S
        g_theta = a * (grad_M - inner)

        # Scale gradient for the full score
        # d(score)/dtheta = (2n/S^2) * S * g_theta - (4n*M/S^3) * S * (sum(a) - S*mean_frac)
        # Simplified: g_theta already accounts for the softmax structure
        # The score = 2n*M_smooth/S^2, and S is constant in log-param (by construction)
        # So d(score)/dtheta = (2n/S^2) * d(M_smooth)/dtheta
        # = (2n/S^2) * S * a * (grad_M - inner)  [since S is fixed by construction]
        # Wait, S is NOT fixed in log-param. Let me reconsider.
        # Actually a = softmax(theta) * S, and S = sum(a_init) is fixed.
        # So S IS fixed. Good.
        g_score = (2.0 * n / S) * g_theta

        # Adam update
        step += 1
        m1 = beta1 * m1 + (1 - beta1) * g_score
        m2 = beta2 * m2 + (1 - beta2) * (g_score ** 2)
        m1_hat = m1 / (1 - beta1 ** step)
        m2_hat = m2 / (1 - beta2 ** step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Evaluate
        t_shift = theta - np.max(theta)
        e_new = np.exp(t_shift)
        a_check = e_new / np.sum(e_new) * S
        score = compute_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_check.copy()

        if step % 1000 == 0 and elapsed_frac > 0.5:
            lr *= 0.95

    return best_a, best_score


def sa_refine(a_init, time_limit, rng):
    """Simulated annealing with targeted peak-reduction moves."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.99992
    step = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.30:
            # Peak-targeted reduction
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.045, 0.018, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(6, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.045, 0.018, size=n_inc))
        elif r < 0.45:
            # Secondary peak reduction
            c = fft_convolve(a, a)
            # Find top 3 peaks
            sorted_peaks = np.argsort(c)[::-1][:3]
            for pk in sorted_peaks:
                pk = int(pk)
                contribs = np.zeros(n)
                for i in range(min(n, pk + 1)):
                    j = pk - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]
                top_k = max(2, n // 25)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=top_k))
        elif r < 0.60:
            # Block modulation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.75:
            # Pair exchange
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2:
                delta = rng.normal(0.035, 0.012)
                candidate[i1] *= np.exp(delta)
                candidate[i2] *= np.exp(-delta)
        elif r < 0.85:
            # Broad small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.004, size=n))
        else:
            # Structured: multiply by a smooth random function
            k = int(rng.integers(1, max(2, n // 10)))
            t = np.arange(n) / n
            candidate *= 1.0 + 0.05 * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))

        candidate = np.clip(candidate, 0.01, None)
        cand_S = np.sum(candidate)
        if cand_S < 1e-10:
            continue
        candidate = candidate * (S / cand_S)

        new_score = compute_score(candidate)
        step += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 5e-7)

    return best_a, best_score


def lp_polish(f_init, time_limit, t_init=0.2):
    """LP-based polishing using scipy linprog."""
    from scipy.optimize import linprog

    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        M = np.max(c)

        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = f

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 15})

        if not result.success or result.x is None:
            break

        g = result.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        g_rescaled = g * (S / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (S / sum_new)

        score_new = compute_score(f_new)

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new
            stagnant = 0
        else:
            f = f_new
            stagnant += 1
            if stagnant > 5:
                t *= 0.7
                stagnant = 0
            if stagnant > 20:
                break

    return best_f, best_score


def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.05, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_multitone(n, rng, n_tones=5):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.08, 0.3)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_two_level(n, rng):
    binary = rng.random(n) < 0.5
    lo = rng.uniform(0.2, 0.5)
    hi = rng.uniform(1.2, 2.0)
    a = np.where(binary, lo, hi)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_quadratic_residue(n, rng):
    """Based on quadratic residue pattern (Barker-like)."""
    t = np.arange(n)
    a = np.ones(n)
    # Use a pseudo-random quadratic pattern
    p = 2 * n + 1  # odd number
    while p % 2 == 0:
        p += 2
    for i in range(n):
        if (i + 1) * (i + 1) % p < p // 2:
            a[i] *= 0.6
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

            best_seq = provided_seq.copy()

    # === Phase 1: Multi-start gradient optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: Multi-start Nesterov gradient optimization")
    print("=" * 60)

    n_prov = len(provided_seq) if provided_seq is not None else 256

    configs = [
        (n_prov, 'provided'),
        (n_prov, 'chirp'),
        (n_prov, 'multitone'),
        (n_prov, 'two_level'),
        (n_prov, 'random'),
        (n_prov, 'quad_res'),
        (512, 'chirp'),
        (512, 'multitone'),
        (384, 'chirp'),
        (768, 'random'),
    ]

    phase1_end = t_global + (deadline - t_global) * 0.50
    per_config_time = (phase1_end - time.monotonic()) / len(configs)

    candidates = []

    for n, init_name in configs:
        if time.monotonic() > phase1_end - 5:
            break

        t_allowed = min(per_config_time, phase1_end - time.monotonic() - 5)
        if t_allowed < 8:
            break

        if init_name == 'provided' and provided_seq is not None:
            x_old = np.linspace(0, 1, len(provided_seq))
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, provided_seq)
            a0 = np.clip(a0, 0.05, None)
        elif init_name == 'chirp':
            a0 = make_chirp(n, 0.3 + 0.5 * rng.random(), rng)
        elif init_name == 'multitone':
            a0 = make_multitone(n, rng, n_tones=int(rng.integers(4, 9)))
        elif init_name == 'two_level':
            a0 = make_two_level(n, rng)
        elif init_name == 'quad_res':
            a0 = make_quadratic_residue(n, rng)
        else:
            a0 = make_random(n, rng, spread=0.2 + 0.3 * rng.random())

        init_sc = compute_score(a0)
        print(f"  n={n}, init='{init_name}', alloc={t_allowed:.0f}s, init_score={init_sc:.4f}")

        t_start = time.monotonic()
        # Use Nesterov for most, log-domain for some
        if init_name in ('provided', 'chirp', 'multitone'):
            a_best, score = nesterov_optimize(a0, t_allowed, tau_init=3.0/n, lr_init=0.8)
        elif init_name == 'two_level':
            a_best, score = log_domain_optimize(a0, t_allowed, rng)
        else:
            a_best, score = nesterov_optimize(a0, t_allowed, tau_init=2.0/n, lr_init=1.0)
        elapsed = time.monotonic() - t_start
        print(f"    -> score={score:.6f}, time={elapsed:.1f}s")

        candidates.append((score, a_best, n, init_name))

        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}, {init_name}) ***")

    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5: {[(f'{s:.5f}', n, nm) for s, _, n, nm in candidates[:5]]}")

    # === Phase 2: SA refinement on top candidates ===
    print("\n" + "=" * 60)
    print("Phase 2: Simulated annealing refinement")
    print("=" * 60)

    phase2_end = t_global + (deadline - t_global) * 0.78
    n_sa = min(3, len(candidates))
    sa_per = (phase2_end - time.monotonic()) / max(1, n_sa)

    for i in range(n_sa):
        if time.monotonic() > phase2_end - 5:
            break

        score_i, a_i, n_i, name_i = candidates[i]
        t_allowed = min(sa_per, phase2_end - time.monotonic() - 5)
        if t_allowed < 8:
            break

        print(f"  SA on cand {i} (n={n_i}, score={score_i:.6f}), alloc={t_allowed:.0f}s")
        a_sa, score_sa = sa_refine(a_i, t_allowed, rng)
        print(f"    -> score={score_sa:.6f}")

        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 3: Log-domain gradient polish on best ===
    print("\n" + "=" * 60)
    print("Phase 3: Log-domain gradient polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 40:
        phase3_end = t_global + (deadline - t_global) * 0.90
        t_allowed = min(phase3_end - time.monotonic(), 120.0)
        if t_allowed > 10:
            print(f"  Log-domain polish (n={len(best_seq)}), alloc={t_allowed:.0f}s")
            a_log, score_log = log_domain_optimize(best_seq, t_allowed, rng)
            print(f"    -> score={score_log:.6f}")
            if score_log < best_score:
                best_score = score_log
                best_seq = a_log.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 4: Nesterov + LP polish ===
    print("\n" + "=" * 60)
    print("Phase 4: Final Nesterov + LP polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 20:
        # Quick Nesterov
        t_nest = min(40.0, (deadline - time.monotonic()) * 0.4)
        if t_nest > 8:
            a_nest, score_nest = nesterov_optimize(best_seq, t_nest, tau_init=1.0/len(best_seq), lr_init=0.5)
            if score_nest < best_score:
                best_score = score_nest
                best_seq = a_nest.copy()
            print(f"  Nesterov final: {best_score:.6f}")

        # LP if feasible
        n_best = len(best_seq)
        if n_best <= 384 and time.monotonic() < deadline - 25:
            t_lp = min(60.0, deadline - time.monotonic() - 15)
            if t_lp > 10:
                a_lp, score_lp = lp_polish(best_seq, t_lp)
                if score_lp < best_score:
                    best_score = score_lp
                    best_seq = a_lp.copy()
                print(f"  LP final: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
