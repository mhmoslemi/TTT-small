    w = (conv / max_c) ** p  # w[k] in [0, 1], w[k] = 1 at peaks
    # Update: a[i] *= 1 / sqrt(sum_k w[k] * a[k-i])
    denom = np.zeros(n)
    for i in range(n):
        # sum_k w[k] * a[k-i] for valid k-i
        lo = max(0, i - (n-1))  # k - i >= 0 => k >= i
        hi = min(2*n-1, i + (n-1))  # k - i < n => k < i + n
        # Actually: k-i ranges from 0 to n-1, so k ranges from i to i+n-1
        lo = i
        hi = min(2*n-1, i + n - 1)
        denom[i] = np.dot(w[lo:hi+1], a[0:hi-lo+1])
    a_new = a / np.sqrt(np.maximum(denom, 1e-15))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Iterative proportional fitting (IPF) to flatten the auto-convolution
2. LP-based direction finding for fine-tuning
3. Local search for final polish

import numpy as np
import scipy.optimize
import scipy.sparse
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def evaluate_safe(a):
    """Safe evaluation that clips values."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)

def ipf_flatten(a, p=4.0, iterations=1000, tol=1e-10):
    """Iterative proportional fitting to flatten auto-convolution."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    prev_obj = obj_value(a)
    best_a = a.copy()
    best_obj = prev_obj
    
    for it in range(iterations):
        conv = fast_convolve(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Weights emphasizing peaks
        w = (conv / max_c) ** p
        
        # Compute denom[i] = sum_j a[j] * w[i+j]
        denom = np.zeros(n)
        for i in range(n):
            denom[i] = np.dot(a, w[i:i+n])
        
        # Update
        denom_safe = np.maximum(denom, 1e-15)
        a_new = a / np.sqrt(denom_safe)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        
        obj_new = obj_value(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        if abs(prev_obj - obj_new) < tol:
            break
        prev_obj = obj_new
        a = a_new
    
    return best_a, best_obj

def solve_lp_direction(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g) <= M."""
    n = len(f)
    # Build sparse constraint matrix
    # (f*g)[k] = sum_i f[i]*g[k-i] for valid i
    # Row k has non-zeros in columns max(0,k-n+1) to min(n-1,k)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

def lp_refine(a0, time_limit, max_iter=200):
    """LP-based refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_lp_direction(a, M)
        
        # Line search
        best_t = 0.0
        best_obj_t = obj_value(a)
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_t = (1-t)*a + t*g
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t < 1e-15:
                continue
            a_t /= s_t
            obj_t = obj_value(a_t)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t
        
        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            a = (1-best_t)*a + best_t*g
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a /= s
            best_obj = best_obj_t
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_a, best_obj

def local_refine(a0, time_limit):
    """Local search: try transferring mass from peak-contributing to safe positions."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    best_obj = obj_value(a)
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit and stuck < 100:
        conv = fast_convolve(a)
        # Find top few peaks
        peak_idx = np.argsort(conv)[-5:].tolist()
        
        improved = False
        for k_star in peak_idx:
            if time.monotonic() - t0 > time_limit:
                break
            # Find which (i, j) pairs contribute to this peak
            # conv[k_star] = sum_i a[i]*a[k_star-i]
            contrib = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
            
            max_c = contrib.max()
            if max_c < 1e-15:
                continue
            
            # Sort by contribution
            order = np.argsort(contrib)[::-1]
            # Try moving mass from high-contrib to low-contrib
            for di in order[:10]:
                if improved:
                    break
                for si in order[-20:]:
                    if improved:
                        break
                    if si == di:
                        continue
                    if a[di] < 1e-10:
                        continue
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        amount = a[di] * frac
                        a_trial = a.copy()
                        a_trial[di] -= amount
                        a_trial[si] += amount
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial /= s_t
                        o = obj_value(a_trial)
                        if o < best_obj - 1e-10:
                            best_obj = o
                            a = a_trial
                            improved = True
                            break
        
        if not improved:
            stuck += 1
        else:
            stuck = 0
    
    return a, best_obj

def random_perturb(a, eps, rng):
    """Apply random multiplicative perturbation."""
    noise = rng.normal(0, eps, len(a))
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

Wait, I realize there might be an issue with my IPF approach. Let me think about it more carefully.

The IPF update is:
`a_new[i] = a[i] / sqrt(sum_j a[j] * w[i+j])`

where `w = (conv/max_conv)^p`.

Intuition: if the convolution has a high peak at position `k*`, then `w[k*]` is large. The denominator for `a[i]` includes `w[i+j]` for all `j`, so elements `a[i]` where `i+j` is near a peak get reduced.

This should work, but let me think about convergence. The map `a -> a / sqrt(denom)` is not obviously a contraction, so it might oscillate. Let me add a damping factor:

`a_new[i] = a[i] / (denom[i])^{alpha/2}` where `alpha` is in (0, 2] and serves as a learning rate.

Or better: `a_new[i] = a[i] * (denom[i])^{-alpha}` for `alpha` in (0, 1).

Let me also think about whether the IPF approach is actually making progress. The key question is: does reducing the elements that contribute to peaks actually reduce the overall max?

Suppose the peak is at `k*`. The peak value is `conv[k*] = sum_i a[i]*a[k*-i]`. If I reduce `a[i]` for some `i`, the peak decreases, but other convolution values might increase (because the total mass is preserved by normalization).

The IPF update tries to balance this: it reduces elements proportionally to their "exposure" to peaks. This is similar to the idea in multiplicative weights algorithms.

I think the approach is sound. Let me refine the implementation.

One more thought: I should also try the approach of **optimizing the peak position**. If the peak is at position `k*`, and I can choose where to put the mass, maybe I should try to move the peak to a position where fewer elements are involved.

For example, if the peak is at `k* = 0`, only `a[0]` contributes (since `conv[0] = a[0]^2`). So I just need to keep `a[0]` small. But then the mass has to go somewhere else, and it might create a peak at a different position.

The optimal strategy is to spread the mass so that no single position in the convolution gets too much. This is essentially a load-balancing problem.

def smooth_obj(a, beta=100):
    """Smooth approximation of the objective using log-sum-exp."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    # Log-sum-exp for numerical stability
    max_c = conv.max()
    if max_c < 1e-15:
        return 0
    lse = max_c + np.log(np.sum(np.exp((conv - max_c) * beta)) / beta)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative proportional fitting (IPF)
to flatten auto-convolution, combined with LP-based refinement and local search.
The IPF update leverages the optimality condition for minimizing sum_k conv[k]^p
to iteratively reduce peak contributions.
"""

import numpy as np
import scipy.optimize
import scipy.sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def ipf_update(a, p=2.0, damp=1.0):
    """Single IPF update step.
    
    a_new[i] ∝ a[i] / (sum_j a[j] * conv[i+j]^(p-1))
    
    with damping: a_new = a^(1-damp) * (a/update_factor)^damp
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return a.copy()
    
    # Compute denominator: denom[i] = sum_{j=0}^{n-1} a[j] * conv[i+j]^(p-1)
    if p == 1:
        # Special case: denom[i] = sum_j a[j] * 1 = sum(a) = 1 (if normalized)
        # This gives no update. Use p=2 minimum.
        conv_power = np.ones(2*n-1)
    else:
        conv_power = np.maximum(conv, 1e-30) ** (p - 1)
    
    denom = np.zeros(n)
    for i in range(n):
        denom[i] = np.dot(a, conv_power[i:i+n])
    
    denom_safe = np.maximum(denom, 1e-30)
    
    # Multiplicative update with damping
    ratio = a / denom_safe
    if damp < 1.0:
        a_new = a * (ratio ** damp)  # a * (a/denom)^damp = a^(1+damp) / denom^damp
    else:
        a_new = ratio
    
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new /= s
    return a_new


def ipf_optimize(a0, p_values=None, max_iter=500, tol=1e-12, damp=1.0):
    """Run IPF optimization with multiple p values."""
    if p_values is None:
        p_values = [2, 4, 8, 16, 32]
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = obj_value(a)
    
    for p in p_values:
        prev_obj = best_obj
        for it in range(max_iter):
            a_new = ipf_update(a, p=p, damp=damp)
            obj_new = obj_value(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
            if abs(prev_obj - obj_new) < tol:
                break
            prev_obj = obj_new
            a = a_new
    
    return best_a, best_obj
