import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_to(a, target):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def soft_max_grad(a, tau):
    """Compute gradient of soft-max of convolution w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_res = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_res[n - 1:n - 1 + n]
    return grad, float(c.max())


def hard_max_grad(a):
    """Compute gradient of hard-max of convolution w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, float(c.max())


def lp_solve(f, M, time_limit=30.0):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    if n > 2048:
        return None

    c_obj = -np.ones(n)

    # Build the convolution constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = f[:end - start]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_solve_sparse(f, M, time_limit=30.0):
    """
    LP solve using scipy sparse matrix for efficiency.
    """
    from scipy.sparse import csr_matrix
    n = len(f)
    if n > 2048:
        return None

    c_obj = -np.ones(n)

    # Build sparse A: it's a banded matrix
    # Row k, col j is nonzero when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def grad_descent_phase(a, S_t, deadline, tau_start=3.0, tau_end=30.0, lr_start=0.3):
    """Soft-max gradient descent with momentum and adaptive temperature."""
    n = len(a)
    a = a.copy()
    momentum = np.zeros(n)
    mu = 0.9
    best_local = compute_obj(a)
    no_improve = 0
    it = 0
    max_it = 100000

    while time.monotonic() < deadline and it < max_it:
        it += 1
        progress = it / max_it
        tau = tau_start + (tau_end - tau_start) * min(1.0, progress * 2.0)
        lr = lr_start * (0.9995 ** it)
        lr = max(lr, 0.003)

        grad, M = soft_max_grad(a, tau)

        # Project gradient to maintain sum constraint
        grad -= grad.mean()

        momentum = mu * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to(a_new, S_t)
        a = a_new

        if it % 500 == 0:
            s = compute_obj(a)
            if s < best_local - 1e-9:
                best_local = s
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 6:
                    # Reset momentum
                    momentum = np.zeros(n)
                    no_improve = 0
                    # Small random perturbation
                    a = np.maximum(a + rng_perturb(n, 0.01), 0)
                    a = normalize_to(a, S_t)

    return a, best_local


def lp_alternation(a, S_t, deadline, max_iters=200, initial_blend=0.4):
    """
    LP-based alternation: repeatedly solve LP and blend.
    Uses adaptive blending and line search.
    """
    n = len(a)
    a = a.copy()
    best_score = compute_obj(a)
    best_a = a.copy()
    no_improve = 0
    blend = initial_blend
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = np.convolve(a, a)
        M = float(c.max())

        g = lp_solve(a, M, time_limit=20.0)
        if g is None:
            g = lp_solve_sparse(a, M, time_limit=20.0)
        if g is None:
            break

        g_sum = float(g.sum())
        if g_sum < 1e-8:
            break

        g_norm = normalize_to(g, S_t)

        # Line search over blend parameter
        improved = False
        for t in [blend, blend * 1.5, blend * 0.5, 0.1, 0.3, 0.5, 0.7, 0.9]:
            t = min(max(t, 0.01), 0.99)
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best_score = s_new
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            # Reduce blend for finer tuning
            blend = max(0.05, blend * 0.9)
        else:
            no_improve += 1
            # Increase blend to try larger steps
            blend = min(0.95, blend * 1.3)
            if no_improve > 15:
                # Add perturbation to escape
                a = best_a.copy()
                pert = rng_gauss(n, 0.05)
                a = np.maximum(a + pert, 0.0)
                a = normalize_to(a, S_t)
                no_improve = 0
                blend = 0.3

    return best_a, best_score


def coord_descent_1d(a, S_t, deadline, max_rounds=5):
    """
    1D coordinate descent: for each coordinate, find optimal value
    using golden section search on the 1D objective.
    """
    n = len(a)
    a = a.copy()
    best_score = compute_obj(a)
    c = np.convolve(a, a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False

        for i in range(n):
            if time.monotonic() > deadline:
                break

            orig = a[i]
            if orig < 1e-10:
                # Try setting to a small value
                a[i] = S_t / n * 0.1
                a = normalize_to(a, S_t)
                s = compute_obj(a)
                if s < best_score - 1e-10:
                    best_score = s
                    improved = True
                else:
                    a[i] = orig
                    a = normalize_to(a, S_t)
                continue

            # Golden section search over a[i] in [0, orig * 3]
            lo = 0.0
            hi = orig * 3.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0

            # Evaluate at two interior points
            x1 = lo + (1.0 - gr) * (hi - lo)
            x2 = lo + gr * (hi - lo)

            a[i] = x1
            a = normalize_to(a, S_t)
            f1 = compute_obj(a)

            a[i] = x2
            a = normalize_to(a, S_t)
            f2 = compute_obj(a)

            for _ in range(20):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = lo + (1.0 - gr) * (hi - lo)
                    a[i] = x1
                    a = normalize_to(a, S_t)
                    f1 = compute_obj(a)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    a[i] = x2
                    a = normalize_to(a, S_t)
                    f2 = compute_obj(a)

            # Use the better of x1, x2
            if f1 < f2:
                opt_val = x1
                opt_score = f1
            else:
                opt_val = x2
                opt_score = f2

            # Also check original
            if best_score < min(opt_score, f1, f2):
                opt_val = orig
                opt_score = best_score

            a[i] = opt_val
            a = normalize_to(a, S_t)
            s_final = compute_obj(a)

            if s_final < best_score - 1e-10:
                best_score = s_final
                improved = True
            else:
                # Revert if not improved
                a[i] = orig
                a = normalize_to(a, S_t)

        if not improved:
            break

    return a, best_score


def peak_targeted_reduction(a, S_t, deadline, max_iters=500):
    """
    Directly reduce coordinates that contribute most to the current peak.
    """
    n = len(a)
    a = a.copy()
    best_score = compute_obj(a)
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        M = c[k_star]

        # Find the coordinate i that, if reduced, most reduces the peak
        # The contribution of a[i] to c[k_star] involves a[i] * a[k_star-i]
        # Reducing a[i] by delta reduces c[k_star] by 2*delta*a[k_star-i]
        # But it also affects other c[k] values.
        # Simple heuristic: reduce the i with largest a[i]*a[k_star-i]
        # where a[k_star-i] is valid.

        best_reduction = -1
        best_i = -1
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                prod = a[i] * a[j]
                if prod > best_reduction:
                    best_reduction = prod
                    best_i = i

        if best_i < 0 or best_reduction < 1e-12:
            break

        # Try reducing a[best_i] by various amounts
        orig = a[best_i]
        improved = False
        for factor in [0.8, 0.6, 0.4, 0.2, 0.9, 0.7]:
            a[best_i] = orig * factor
            a = normalize_to(a, S_t)
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try increasing the mirror coordinate
            j = k_star - best_i
            if 0 <= j < n and j != best_i:
                orig_j = a[j]
                for factor in [1.2, 1.5, 2.0]:
                    a = best_a.copy()
                    a[j] = orig_j * factor
                    a = normalize_to(a, S_t)
                    s = compute_obj(a)
                    if s < best_score - 1e-10:
                        best_score = s
                        best_a = a.copy()
                        improved = True
                        break

        if improved:
            a = best_a.copy()
            no_improve = 0
        else:
            a = best_a.copy()
            no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_score


def rng_gauss(n, sigma):
    """Generate Gaussian random vector (using global rng)."""
    return global_rng.standard_normal(n) * sigma


def rng_perturb(n, sigma):
    """Generate random perturbation."""
    return global_rng.standard_normal(n) * sigma


global_rng = np.random.default_rng(12345)


def full_pipeline(a0, deadline):
    """Run the full optimization pipeline from a starting point."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a.sum() < 1e-8:
        a = np.ones(n) * 0.5
    a = normalize_to(a, S_t)

    best_score = compute_obj(a)
    best_a = a.copy()
    print(f"    Start: score={best_score:.8f}, n={n}")

    remaining = deadline - time.monotonic()

    # Phase 1: Gradient descent (20% of time)
    t1 = time.monotonic() + max(5.0, remaining * 0.15)
    if time.monotonic() < t1:
        a, s1 = grad_descent_phase(a, S_t, t1, tau_start=2.0, tau_end=20.0)
        if s1 < best_score:
            best_score = s1
            best_a = a.copy()
        print(f"    After GD: score={best_score:.8f} ({time.monotonic()-t1:.1f}s over)")

    # Phase 2: LP alternation (35% of time)
    remaining = deadline - time.monotonic()
    t2 = time.monotonic() + max(10.0, remaining * 0.35)
    if time.monotonic() < t2:
        a2, s2 = lp_alternation(best_a, S_t, t2, max_iters=100, initial_blend=0.3)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
        print(f"    After LP: score={best_score:.8f}")

    # Phase 3: Peak targeted reduction (15% of time)
    remaining = deadline - time.monotonic()
    t3 = time.monotonic() + max(5.0, remaining * 0.2)
    if time.monotonic() < t3:
        a3, s3 = peak_targeted_reduction(best_a, S_t, t3, max_iters=300)
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()
        print(f"    After peak reduction: score={best_score:.8f}")

    # Phase 4: Another LP round (20% of time)
    remaining = deadline - time.monotonic()
    t4 = time.monotonic() + max(10.0, remaining * 0.5)
    if time.monotonic() < t4:
        a4, s4 = lp_alternation(best_a, S_t, t4, max_iters=150, initial_blend=0.15)
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()
        print(f"    After LP2: score={best_score:.8f}")

    # Phase 5: Coordinate descent polish (remaining)
    remaining = deadline - time.monotonic()
    if remaining > 10 and n <= 512:
        t5 = deadline - 2.0
        a5, s5 = coord_descent_1d(best_a, S_t, t5, max_rounds=3)
        if s5 < best_score:
            best_score = s5
            best_a = a5.copy()
        print(f"    After coord descent: score={best_score:.8f}")

    return best_a, best_score
