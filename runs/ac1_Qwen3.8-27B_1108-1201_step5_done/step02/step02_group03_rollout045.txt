def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def grad_hard_max(a):
    """Gradient of max autocorrelation (hard max). Returns (grad, k_star)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, k_star


def grad_soft_max(a, tau):
    """Gradient using soft-max weighting."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()

        w /= w_sum

    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]

def lp_solve(f, M, timeout=30.0):
    """
    Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0.
    Uses the banded structure for efficiency.
    Returns optimal g or None.
    """
    n = len(f)
    if n > 1024:
        return None
    
    # Build constraint matrix efficiently
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    # For row m, valid j range is [max(0, m-n+1), min(n-1, m)]
    c_obj = -np.ones(n)
    
    # Use sparse representation conceptually, but build dense for HiGHS
    # For n <= 512, this is at most 1023 x 512 = ~500K entries, manageable

        # A[m, j] = f[m-j] where m-j ranges from 0 to n-1
        for i in range(n):
            A[j + i, j] = f[i]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.status == 0:
            g = res.x
            g = np.maximum(g, 0)  # Ensure non-negative
            return g
    except Exception:

    return None


def lp_solve_dual(f, M, timeout=30.0):
    """
    Solve the dual LP: min M*sum(y) s.t. A^T @ y >= 1, y >= 0.
    A^T[j, m] = f[m-j] if 0 <= m-j < n.
    The dual objective gives us information about the tightness.
    """
    n = len(f)
    if n > 1024:
        return None
    
    # Dual: min M * sum_{m=0}^{2n-2} y_m
    # s.t. for each j: sum_{m: 0<=m-j<n} f[m-j] * y_m >= 1
    #      y_m >= 0
    
    # A^T has shape (n, 2n-1)
    # A^T[j, m] = f[m-j] if 0 <= m-j < n, else 0
    c_obj = M * np.ones(2 * n - 1)
    
    A_eq = np.zeros((n, 2 * n - 1))
    for j in range(n):
        for i in range(n):
            A_eq[j, j + i] = f[i]
    
    # Minimize c^T y s.t. A_eq @ y >= 1, y >= 0
    # Convert to standard form: -A_eq @ y <= -1
    res = linprog(c_obj, A_ub=-A_eq, b_ub=-np.ones(n),
                  bounds=[(0.0, None)] * (2 * n - 1), method='highs',
                  options={'presolve': True, 'time_limit': timeout})
    if res.status == 0:
        return res.x
    return None

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses aggressive LP-based refinement, subgradient descent with momentum,
line search on blend factors, and multi-start strategy at n=384 and n=512.
Key improvements: more frequent LP calls, proper line search, better time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def grad_hard_max(a):
    """Gradient of max autocorrelation (hard max)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def grad_soft_max(a, tau):
    """Gradient using soft-max weighting."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def build_lp_matrix(n):
    """Build the LP constraint matrix A where A[m,j] = f[m-j] if valid."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            A[j + i, j] = 1.0  # Placeholder - we'll multiply by f
    return A


def lp_solve(f, M, timeout=15.0):
    """
    Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    if n > 1024:
        return None

    # Build A: A[m, j] = f[m-j] if 0 <= m-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[i] goes into A[j+i, j] for i = 0..n-1
        A[j:j + n, j] = f

    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None


def line_search_blend(a, g, S_t, best_score, n_trials=15):
    """
    Find optimal blend factor t in (0,1) for a_new = (1-t)*a + t*g.
    Returns (best_a, best_score, best_t).
    """
    best_a = a.copy()
    best_s = best_score
    best_t = 0.0

    for t in np.linspace(0.05, 0.95, n_trials):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            continue
        a_new = normalize(a_new, S_t)
        s = compute_obj(a_new)
        if s < best_s - 1e-12:
            best_s = s
            best_a = a_new.copy()
            best_t = t

    return best_a, best_s, best_t

def gradient_polish(a, deadline, verbose=False):
    """Fast gradient-only polish (no LP, for later stages)."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    best = a.copy()
    momentum = np.zeros(n)
    mu = 0.85

    for it in range(50000):
        if time.monotonic() > deadline:
            break
        lr = max(0.005, 0.2 * (0.9995 ** it))
        if it < 2000:
            grad = grad_hard_max(a)
        else:
            tau = 10.0 + 40.0 * min(1.0, (it - 2000) / 5000.0)
            grad = grad_soft_max(a, tau)
        grad -= grad.mean()
        momentum = mu * momentum - lr * grad
        a = a + momentum
        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)

        if it % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
                best = a.copy()

    return best, best_score
