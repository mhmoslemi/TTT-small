import time

    if b is None:

    size = 1
    while size < n + m - 1:

"""
Multi-phase optimization for minimizing peak self-convolution of non-negative sequences.
Combines LP-based direction finding, peak-directed local search, and gradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    """Build matrix A where A@b = conv(f,b) for full linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 2, j + n - 1)
        for k in range(lo, hi + 1):
            A[k, j] = f[k - j]
    return A


def solve_lp(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(f_init, time_limit, t_step=0.04, verbose=True):
    """LP-iteration optimizer."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    A_cached = None
    f_for_A = None

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        # Only rebuild A if f changed significantly
        if A_cached is None or f_for_A is None or np.max(np.abs(f - f_for_A)) > 1e-6:
            A_cached = build_A(f)
            f_for_A = f.copy()

        b = solve_lp_with_A(A_cached, M, timeout=3.0)
        if b is None:
            break

        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        # Line search
        best_t = t_step
        best_s = np.inf
        for t_try in [t_step, 2*t_step, 0.5*t_step, 0.1*t_step]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{elapsed:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f} stall={stall}")

        if stall >= 60:
            # Escape with random perturbation
            rng_esc = np.random.default_rng(42 + it * 7919)
            f += 0.02 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0
            A_cached = None

    return best_f, best_score


def solve_lp_with_A(A, M, timeout=5.0):
    """Solve LP with pre-built A matrix."""
    n = A.shape[1]
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2
over non-negative sequences. Combines LP-based direction finding, smooth-max
projected gradient descent, and peak-directed local search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def normalize_sum(a, target):
    """Scale a so that sum(a) = target."""
    S = float(np.sum(a))
    if S < 1e-15:
        return a
    return a * (target / S)


def project_sum_nonneg(x, S):
    """Project x onto {a: sum(a) = S, a >= 0} using water-filling."""
    n = len(x)
    lo = float(x.min()) - abs(float(S)) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(x - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.maximum(x - lam, 0.0)


def build_A(f):
    """Build matrix A where A@b = conv(f, b) for full linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(2 * n - 2, j + n - 1)
        for k in range(lo, hi + 1):
            A[k, j] = f[k - j]
    return A


def solve_lp_with_A(A, M, timeout=4.0):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0."""
    n = A.shape[1]
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(f_init, time_limit, verbose=True):
    """LP-iteration optimizer with cached A matrix."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    A_mat = build_A(f)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break
        
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        b = solve_lp_with_A(A_mat, M, timeout=3.0)
        if b is None:
            break
        
        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)
        
        # Adaptive step size
        diff = np.max(np.abs(b - f))
        if diff < 1e-8:
            stall += 1
        else:
            # Try several step sizes
            best_t = 0.04
            best_s = np.inf
            for t_try in [0.08, 0.04, 0.02, 0.01, 0.15]:
                f_cand = (1.0 - t_try) * f + t_try * b
                f_cand = np.clip(f_cand, 0.0, 1000.0)
                s_cand = score_fast(f_cand)
                if s_cand < best_s:
                    best_s = s_cand
                    best_t = t_try
            
            f = (1.0 - best_t) * f + best_t * b
            f = np.clip(f, 0.0, 1000.0)
            A_mat = build_A(f)
            it += 1

                best_score = best_s
                best_f = f.copy()
                stall = 0
            else:
                stall += 1

                print(f"    [LP {time.monotonic()-t0:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f} stall={stall}")
            
            if stall >= 50:

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2
over non-negative sequences. Combines LP-based direction finding, top-k
projected gradient descent, peak-directed local search, and multi-start
with adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def normalize_sum(a, target):
    """Scale a so that sum(a) = target."""
    S = float(np.sum(a))
    if S < 1e-15:
        return a
    return a * (target / S)


def project_sum_nonneg(x, S):
    """Project x onto {a: sum(a)=S, a>=0} via binary search water-filling."""
    lo = float(np.min(x)) - abs(float(S)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(x - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.maximum(x - lam, 0.0)


def build_A(f):
    """Build matrix A where A@b = conv(f, b) for full linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(2 * n - 2, j + n - 1)
        for k in range(lo, hi + 1):
            A[k, j] = f[k - j]
    return A


def solve_lp_with_A(A, M, timeout=4.0):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0."""
    n = A.shape[1]
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_gradient_topk(f, c, top_k=5):
    """Compute gradient of smoothed max using top-k peaks."""
    n = len(f)
    top_idx = np.argsort(c)[-top_k:]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / top_k
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            g[j] += 2.0 * w[k] * f[k - j]
    return g


def compute_gradient_softmax(f, c, beta=20.0):
    """Compute gradient of logsumexp(beta*c)/beta."""
    n = len(f)
    c_shifted = c - np.max(c)
    exp_c = np.exp(beta * c_shifted)
    w = exp_c / np.sum(exp_c)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            g[j] += 2.0 * w[k] * f[k - j]
    return g


def optimize_lp(f_init, time_limit, verbose=True):
    """LP-iteration optimizer with adaptive step and perturbation."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    A_mat = build_A(f)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        b = solve_lp_with_A(A_mat, M, timeout=3.0)
        if b is None:
            break

        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        diff = float(np.max(np.abs(b - f)))
        if diff < 1e-9:
            stall += 1
        else:
            best_t = 0.04
            best_s = np.inf
            for t_try in [0.06, 0.04, 0.02, 0.01, 0.10, 0.15]:
                f_cand = (1.0 - t_try) * f + t_try * b
                f_cand = np.clip(f_cand, 0.0, 1000.0)
                s_cand = score_fast(f_cand)
                if s_cand < best_s:
                    best_s = s_cand
                    best_t = t_try

            f = (1.0 - best_t) * f + best_t * b
            f = np.clip(f, 0.0, 1000.0)
            A_mat = build_A(f)
            it += 1

            if best_s < best_score - 1e-10:
                best_score = best_s
                best_f = f.copy()
                stall = 0
            else:
                stall += 1

            if verbose and it % 30 == 0:
                print(f"    [LP {time.monotonic()-t0:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f} stall={stall}")

            if stall >= 40:
                rng_esc = np.random.default_rng(42 + it * 7919)
                f += 0.015 * rng_esc.standard_normal(n)
                f = np.clip(f, 0.01, 1000.0)
                f = normalize_sum(f, float(np.sum(best_f)))
                stall = 0
                A_mat = build_A(f)

    return best_f, best_score

    n = len(f_init)
    target_sum = np.sqrt(2.0 * n)
    f = normalize_sum(f_init.astype(np.float64).copy(), target_sum)

            break

        g = compute_gradient_topk(f, c, top_k)
        g = g - np.mean(g)

        momentum = beta_m * momentum + (1.0 - beta_m) * g
        f_new = f - lr * momentum

def peak_shave_search(f_init, time_limit, verbose=True):
    """Peak-directed local search: reduce top contributors, boost others."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    alpha = 0.03

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        k_peak = int(np.argmax(c))
        peak_val = c[k_peak]
        S = float(np.sum(f))

        # Compute contributions to peak
        contribs = np.zeros(n)
        for j in range(n):
            idx = k_peak - j
            if 0 <= idx < n:
                contribs[j] = f[j] * f[idx]

        # Also consider second peak
        c_sorted = np.sort(c)[::-1]
        second_peak = c_sorted[1] if len(c_sorted) > 1 else 0
        k_second = int(np.argmax(c - np.where(np.arange(2 * n - 1) == k_peak, np.inf, 0)))

        if second_peak > 0.8 * peak_val:
            for j in range(n):
                idx = k_second - j
                if 0 <= idx < n:
                    contribs[j] += 0.5 * f[j] * f[idx]

        # Determine which to reduce and which to boost
        sorted_idx = np.argsort(contribs)
        n_reduce = max(1, n // 6)
        reduce_set = sorted_idx[-n_reduce:]
        boost_set = sorted_idx[:n // 4]

        f_new = f.copy()
        for j in reduce_set:
            f_new[j] *= (1.0 - alpha * rng.uniform(0.5, 1.0))

        delta = S - float(np.sum(f_new))
        if delta > 0 and len(boost_set) > 0:
            f_new[boost_set] += delta / len(boost_set) * rng.uniform(0.8, 1.2, size=len(boost_set))

        f_new = np.clip(f_new, 0.01, 1000.0)
        new_score = score_fast(f_new)

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            alpha = max(0.005, alpha * 0.98)
        elif new_score < score_fast(f) + 0.001:
            f = f_new
            alpha = max(0.005, alpha * 0.99)
        else:
            alpha = min(0.06, alpha * 1.05)

    return best_f, best_score


def make_hann_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng):

    t = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    a = a * (0.4 + 0.6 * w)
    return np.clip(a, 0.01, None)

def interpolate_sequence(seq, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
