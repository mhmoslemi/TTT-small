    w = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            grad[j] += w[k] * a[k - j]
    return obj, grad

The gradient `grad[j] = sum_k w[k] * a[k-j]` for valid `k-j` is the cross-correlation of `w` (length 2n-1) with `a` (length n), evaluated at lags `0, 1, ..., n-1`.

Cross-correlation of `w` and `a` at lag `j`: `sum_k w[k+j] * a[k]`... hmm, the indexing is different.

Let me define it more carefully. We want:
`grad[j] = sum_{k: 0<=k-j<n} w[k] * a[k-j]`

Let `m = k-j`, so `k = m+j`. Then:
`grad[j] = sum_{m: 0<=m<n} w[m+j] * a[m]` (where `w[m+j]` is 0 if `m+j >= 2n-1`)

This is the correlation of `w` (padded to length 2n-1) with `a` at lag `j`:
`grad[j] = (w ⊛ a)[j]` where ⊛ denotes cross-correlation.

We can compute this via FFT:
`grad = ifft(fft(w_padded) * conj(fft(a_padded)))` where we pad both to sufficient length.

"""
Smoothed subgradient descent with Adam optimizer for minimizing
normalized peak autocorrelation of non-negative step functions.
Uses log-sum-exp smoothing, temperature annealing, and multi-phase strategy.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s ** 2))

def compute_grad(a, tau):
    """Compute gradient of smoothed objective tau*log(sum(exp(c[k]/tau)))."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    
    # grad[j] = sum_k w[k] * a[k-j] for valid k-j
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # For j in [j_start, j_end]: grad[j] += wk * a[k-j]
        # a[k-j] for j=j_start..j_end:
        #   k-j_start = k - max(0,k-n+1) = min(k, n-1)
        #   k-j_end = k - min(n-1,k) = max(0, k-n+1)
        # So a[k-j] goes from a[min(k,n-1)] down to a[max(0,k-n+1)]
        # = a[max(0,k-n+1) : min(k,n-1)+1][::-1]
        segment = a[max(0, k - n + 1):min(k, n - 1) + 1]
        grad[j_start:j_end + 1] += wk * segment[::-1]
    
    return c_max + tau * np.log(sum_exp), grad

def adam_smoothed_descent(f_init, time_alloc, tau_start, tau_end, lr=0.5, seed=42):
    """Adam optimizer on smoothed subgradient with temperature annealing."""
    rng = np.random.default_rng(seed)
    a = f_init.astype(np.float64).copy()
    n = len(a)
    s = np.sum(a)
    a = a / s  # Normalize to sum=1
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_step = 0
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_alloc:
        elapsed_frac = (time.monotonic() - t0) / time_alloc
        # Anneal temperature
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        obj, grad = compute_grad(a, tau)
        
        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        # Learning rate with decay
        lr_eff = lr / (1.0 + 0.001 * t_step)
        a_new = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project: non-negative and sum=1
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()  # Restore
            continue
        a_new = a_new / s_new
        
        # Check score
        score = local_score(a_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def coordinate_descent(a_init, time_alloc, rng, step_size=0.01):
    """Coordinate descent: try adjusting individual elements."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_alloc:
        # Pick a few random coordinates to try
        n_try = max(1, n // 8)
        idxs = rng.choice(n, size=n_try, replace=False)
        improved = False
        
        for idx in idxs:
            # Try a few step sizes
            for ds in [step_size, -step_size, step_size*0.5, -step_size*0.5]:
                a_new = a.copy()
                a_new[idx] = max(0, a_new[idx] + ds * s / n)
                s_new = np.sum(a_new)
                if s_new < 1e-10:
                    continue
                score = local_score(a_new)
                if score < best_score - 1e-10:
                    a = a_new
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            # Try random perturbation to escape
            if no_improve > 5:
                a = a * np.exp(rng.normal(0, 0.01, n))
                a = np.clip(a, 0, None)
                a = a / np.sum(a) * s
                no_improve = 0
                step_size *= 1.5
    
    return best_a, best_score

def peak_reduction(a_init, time_alloc, rng):
    """Targeted peak reduction: identify peak, reduce contributing elements."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_alloc:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        
        # Find top contributors to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Reduce top contributors, increase others
        top_k = max(3, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]
        other_idx = np.setdiff1d(np.arange(n), top_idx)
        
        a_new = a.copy()
        # Reduce top contributors
        a_new[top_idx] *= np.exp(rng.normal(-0.03, 0.01, top_k))
        # Increase others (pick a few)
        if len(other_idx) > 0:
            n_inc = min(5, len(other_idx))
            inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
            a_new[inc_idx] *= np.exp(rng.normal(0.03, 0.01, n_inc))
        
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new * (s / s_new)
        
        score = local_score(a_new)
        if score < best_score:
            a = a_new
            best_score = score
            best_a = a.copy()
        elif rng.random() < 0.1:  # Accept some bad moves
            a = a_new
    
    return best_a, best_score

        w = exp_c / sum_exp
        
        # grad_a[j] = Σ_k w[k] * a[k-j]
        grad_a = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            segment = a[max(0, k - n + 1):min(k, n - 1) + 1]
            grad_a[j_start:j_end + 1] += wk * segment[::-1]
        
        # Convert to logit gradient
        # dF/dθ[i] = a[i] * (grad_a[i] - dot(a, grad_a))
        inner = np.dot(a, grad_a)
        grad_theta = a * (grad_a - inner)
        
        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        lr_eff = lr / (1.0 + 0.01 * t_step)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check score
        score = local_score(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Uses logit-parameterized Adam with annealed
log-sum-exp smoothing, exact subgradient descent, peak-directed local search,
and spectral shaping moves. Combines diverse initializations with focused
exploitation to beat the 1.5027 target.

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s ** 2))


def compute_smooth_data(a, tau):
    """Compute smoothed objective value, weights, and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp

    # Gradient: grad[j] = sum_k w[k] * a[k-j] for valid k-j
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        segment = a[max(0, k - n + 1):min(k, n - 1) + 1]
        grad[j_start:j_end + 1] += wk * segment[::-1]

    return obj, w, grad


def compute_exact_subgrad(a):
    """Compute exact subgradient at the peak of conv(a,a)."""
    n = len(a)
    c = fft_convolve(a, a)
    peak_val = np.max(c)
    # Find all indices where c is close to peak (within 1% for stability)
    threshold = peak_val * 0.999
    active = np.where(c >= threshold)[0]
    # Weight active peaks by how close they are to the max
    weights = (c[active] - threshold) / (peak_val - threshold + 1e-15)
    weights = weights / np.sum(weights)

    grad = np.zeros(n)
    for idx in range(len(active)):
        k = active[idx]
        wk = weights[idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        segment = a[max(0, k - n + 1):min(k, n - 1) + 1]
        grad[j_start:j_end + 1] += wk * segment[::-1]

    return grad, peak_val


deflogit_grad(grad_a, a):
    """Convert gradient w.r.t. a to gradient w.r.t. logit parameters.
    If a = softmax(theta), then dF/dtheta[i] = a[i]*(dF/da[i] - sum(j) a[j]*dF/da[j])
    """
    inner = np.dot(a, grad_a)
    return a * (grad_a - inner)


def adam_logit_optimize(f_init, time_alloc, tau_start, tau_end, lr=1.0,
                        seed=42, use_exact_switch=None):
    """Adam optimizer in logit parameterization with annealed smoothing.
    
    use_exact_switch: if not None, a fraction (0-1) of elapsed time after
    which we switch from smoothed to exact subgradient.
    """
    rng = np.random.default_rng(seed)
    n = len(f_init)
    a0 = np.clip(f_init, 1e-10, None)
    theta = np.log(a0) - np.log(np.sum(a0))

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_step = 0
    best_score = local_score(a0)
    best_theta = theta.copy()
    t0 = time.monotonic()
    stagnant = 0

    while time.monotonic() - t0 < time_alloc:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_alloc)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)

        # Current a from theta
        theta_shifted = theta - np.max(theta)
        e = np.exp(theta_shifted)
        a = e / np.sum(e)

        # Compute gradient
        use_exact = (use_exact_switch is not None and elapsed_frac > use_exact_switch)
        if use_exact:
            grad_a, _ = compute_exact_subgrad(a)
        else:
            _, _, grad_a = compute_smooth_data(a, tau)

        # Convert to logit gradient
        grad_theta = logit_grad(grad_a, a)

        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)

        # Adaptive learning rate
        progress = elapsed_frac
        lr_eff = lr * (1.0 - 0.5 * progress) / (1.0 + 0.005 * t_step)
        theta = theta - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

        # Compute score
        theta_shifted = theta - np.max(theta)
        e_new = np.exp(theta_shifted)
        a_new = e_new / np.sum(e_new)
        score = local_score(a_new)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                # Perturbation to escape
                theta += 0.1 * rng.standard_normal(n)
                stagnant = 0

    theta_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(theta_shifted) / np.sum(np.exp(theta_shifted))
    return a_best, best_score


def peak_directed_search(a_init, time_alloc, rng):
    """Local search that targets the peak of the convolution for reduction."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9999

    while time.monotonic() - t0 < time_alloc:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        r = rng.random()
        if r < 0.5:
            # Reduce top contributors to peak, increase others
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]

            top_k = max(5, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            other_idx = np.setdiff1d(np.arange(n), top_idx)

            a_new = a.copy()
            a_new[top_idx] *= np.exp(rng.normal(-0.04, 0.015, top_k))
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                a_new[inc_idx] *= np.exp(rng.normal(0.04, 0.015, n_inc))

        elif r < 0.75:
            # Block modification: modify a contiguous segment
            width = max(2, int(rng.integers(1, n // 6)))
            start = int(rng.integers(0, n - width))
            a_new = a.copy()
            a_new[start:start + width] *= np.exp(rng.normal(0.0, 0.03, width))

        elif r < 0.9:
            # Spectral shaping move: modify in frequency domain
            A = np.fft.rfft(a, 2 * n)
            # Randomly boost or attenuate a few frequency bins
            n_bands = max(2, n // 32)
            bands = rng.choice(len(A), size=n_bands, replace=False)
            for b in bands:
                A[b] *= np.exp(rng.normal(0, 0.02))
            a_new = np.fft.irfft(A, 2 * n)[:n]
            a_new = np.clip(a_new, 0, None)

        else:
            # Pair swap: move mass from one position to another
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                a_new = a.copy()
                transfer = 0.02 * s / n
                a_new[i] = max(0, a_new[i] - transfer)
                a_new[j] = a_new[j] + transfer

        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new * (s / s_new)

        new_score = local_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        if new_score < local_score(a) or rng.random() < np.exp(-(new_score - local_score(a)) / max(temp, 1e-12)):
            a = a_new
        temp *= cooling

    return best_a, best_score


def coordinate_polish(a_init, time_alloc, rng):
    """Fine coordinate descent polish."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    step = 0.003 * s / n
    no_improve = 0

    while time.monotonic() - t0 < time_alloc:
        # Batch of random coordinates
        n_try = max(4, n // 16)
        idxs = rng.choice(n, size=n_try, replace=False)
        improved = False

        for idx in idxs:
            for ds in [step, -step, step * 0.5, -step * 0.5, step * 2, -step * 2]:
                a_new = a.copy()
                a_new[idx] = max(0, a_new[idx] + ds)
                s_new = np.sum(a_new)
                if s_new < 1e-10:
                    continue
                score = local_score(a_new)
                if score < best_score - 1e-12:
                    a = a_new
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                a = a * np.exp(rng.normal(0, 0.005, n))
                a = np.clip(a, 0, None)
                a = a / np.sum(a) * s
                no_improve = 0
                step *= 1.3
            elif no_improve > 3:
                step *= 0.7

    return best_a, best_score


def make_chirp_init(n, rng):
    """Generate a chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = rng.uniform(0.2, 0.6)
    freq = rng.uniform(0.5, 3.0)
    a = 1.0 + alpha * np.cos(np.pi * freq * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    """Generate a multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(rng.integers(3, 8)):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_asymmetric_init(n, rng):
    """Generate an asymmetric initialization (important for breaking symmetry)."""
    a = np.ones(n)
    # Add a linear trend (makes it asymmetric)
    a += rng.uniform(-0.3, 0.3) * np.linspace(-1, 1, n)
    # Add some asymmetry
    a[:n // 2] *= 1.0 + rng.uniform(-0.2, 0.2)
    a[n // 2:] *= 1.0 + rng.uniform(-0.2, 0.2)
    # Add random component

    return np.clip(a, 0.01, None)


def make_burst_init(n, rng):
    """Initialize with a few bursts (sparse-like)."""
    a = 0.1 * np.ones(n)
    n_bursts = rng.integers(3, 8)
    for _ in range(n_bursts):
        center = int(rng.integers(0, n))
        width = int(rng.integers(1, n // 8))
        amp = rng.uniform(0.5, 2.0)
        for i in range(n):
            dist = abs(i - center)
            if dist < width:
                a[i] += amp * np.exp(-dist / (width * 0.5))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def logit_grad(grad_a, a):
    """Convert gradient w.r.t. a to gradient w.r.t. logit parameters."""
    inner = np.dot(a, grad_a)

        print(f"No provided seq, starting with ones: score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Diverse exploration ===
    print("\n" + "=" * 60)
    print("Phase 1: Diverse exploration (logit Adam + smoothed subgrad)")
    print("=" * 60)

    inits = []
    # Start 1: provided sequence
    if provided_seq is not None:
        inits.append(('provided', best_seq.copy()))
    # Start 2: chirp
    inits.append(('chirp', make_chirp_init(n_best, rng)))
    # Start 3: multitone
    inits.append(('multitone', make_multitone_init(n_best, rng)))
    # Start 4: asymmetric
    inits.append(('asym', make_asymmetric_init(n_best, rng)))
    # Start 5: burst
    inits.append(('burst', make_burst_init(n_best, rng)))
    # Start 6: random
    inits.append(('random', np.clip(1.0 + 0.3 * rng.standard_normal(n_best), 0.01, None)))

    n_starts = len(inits)
    phase1_time = (deadline - time.monotonic()) * 0.35
    time_per_start = phase1_time / n_starts
    print(f"  {n_starts} starts, ~{time_per_start:.0f}s each")

    for i, (name, a0) in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(time_per_start, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break

        print(f"  [{i+1}/{n_starts}] init='{name}', alloc={t_alloc:.0f}s", end="")
        a_res, score = adam_logit_optimize(a0, t_alloc, tau_start=8.0, tau_end=0.3,
                                           lr=1.5, seed=seed + i)
        print(f" -> score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 2: Focused exploitation ===
    print("\n" + "=" * 60)
    print("Phase 2: Focused exploitation (lower tau, longer runs)")
    print("=" * 60)

    if time.monotonic() < deadline - 100:
        phase2_configs = [
            (3.0, 0.05, 1.0, None),
            (5.0, 0.1, 0.8, 0.5),
            (2.0, 0.03, 1.2, 0.6),
        ]
        phase2_time = min(400.0, (deadline - time.monotonic()) * 0.30)
        time_per_config = phase2_time / len(phase2_configs)

        for ci, (tau_s, tau_e, lr, exact_frac) in enumerate(phase2_configs):
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(time_per_config, deadline - time.monotonic() - 30)
            if t_alloc < 15:
                break

            print(f"  Config {ci}: tau=[{tau_s},{tau_e}], lr={lr}, exact_at={exact_frac}", end="")
            a_res, score = adam_logit_optimize(best_seq, t_alloc, tau_start=tau_s,
                                               tau_end=tau_e, lr=lr,
                                               seed=seed + 100 + ci,
                                               use_exact_switch=exact_frac)
            print(f" -> score={score:.6f}")

            if score < best_score:
                best_score = score
                best_seq = a_res.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: Exact subgradient refinement ===
    print("\n" + "=" * 60)
    print("Phase 3: Exact subgradient refinement")
    print("=" * 60)

    if time.monotonic() < deadline - 80:
        t_alloc = min(200.0, (deadline - time.monotonic()) * 0.25)
        print(f"  Allocating {t_alloc:.0f}s")
        a_res, score = adam_logit_optimize(best_seq, t_alloc, tau_start=0.5,
                                           tau_end=0.01, lr=0.5,
                                           seed=seed + 200, use_exact_switch=0.0)
        print(f"  Result: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Peak-directed local search ===
    print("\n" + "=" * 60)
    print("Phase 4: Peak-directed local search")
    print("=" * 60)

    if time.monotonic() < deadline - 50:
        t_alloc = min(250.0, (deadline - time.monotonic()) * 0.35)
        print(f"  Allocating {t_alloc:.0f}s")
        a_res, score = peak_directed_search(best_seq, t_alloc, rng)
        print(f"  Result: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a_res.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Coordinate polish ===
    print("\n" + "=" * 60)
    print("Phase 5: Coordinate descent polish")
    print("=" * 60)

    if time.monotonic() < deadline - 30:
        t_alloc = min(outer := 120.0, deadline - time.monotonic() - 25)
        if t_alloc > 10:
            print(f"  Allocating {t_alloc:.0f}s")
            a_res, score = coordinate_polish(best_seq, t_alloc, rng)
            print(f"  Result: score={score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = a_res.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 6: Final short SA + polish cycle ===
    if time.monotonic() < deadline - 20:
        t_alloc = min(60.0, deadline - time.monotonic() - 15)
        if t_alloc > 10:
            print(f"\n  Final SA polish, {t_alloc:.0f}s")
            a_res, score = peak_directed_search(best_seq, t_alloc, rng)
            if score < best_score:
                best_score = score
                best_seq = a_res.copy()
            # Tiny coordinate polish
            if time.monotonic() < deadline - 15:
                t2 = min(30.0, deadline - time.monotonic() - 12)
                if t2 > 5:
                    a_res2, score2 = coordinate_polish(best_seq, t2, rng)
                    if score2 < best_score:
                        best_score = score2
                        best_seq = a_res2.copy()

    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"\n{'=' * 60}")
    print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
    print(f"{'=' * 60}")

    return final_list
