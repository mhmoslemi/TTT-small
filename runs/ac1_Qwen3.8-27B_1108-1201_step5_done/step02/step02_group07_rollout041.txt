def autoconv(a):
    """Linear autoconvolution via FFT."""
    n = len(a)
    L = 1
    while L < 2*n - 1:
        L <<= 1
    A = rfft(a, L)
    C = irfft(A * A, L)
    return C[:2*n-1]

def score(a):
    """Score = 2*n*max(autoconv(a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    C = autoconv(a)
    return float(2.0 * n * np.max(C) / (S*S))

        a = a_new
        if verbose and it % 200 == 0:
            s = score(a)

import numpy as np
import time
from scipy.fft import rfft, irfft


def autoconv(a):
    """Linear autoconvolution via FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    A = rfft(a, L)
    C = irfft(A * A, L)
    return C[:2*n-1]


def score(a):
    """Score = 2*n*max(autoconv(a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    C = autoconv(a)
    return float(2.0 * n * np.max(C) / (S * S))


def gs_optimize(a0, time_limit=30.0, relax=0.5, eps=1e-8, seed=42):
    """Gerchberg-Saxton with relaxation for flat autoconvolution."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    if S > 0:
        a = a / S
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_score = score(a)

        A = rfft(a, L)
        mag = np.abs(A)
        target = np.sqrt(np.mean(mag**2))
        
        # Relaxed update
        mag_new = (1 - relax) * mag + relax * target
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = irfft(A_new, L)[:n]
        a_new = np.clip(a_new, eps, None)

    if S < 0.01:
        return np.inf
    C = autoconv(a)
    return float(2.0 * n * np.max(C) / (S * S))


def gs_optimize(a0, time_limit=30.0, relax=0.5, eps=1e-8, target='flat'):
    """Gerchberg-Saxton with relaxation for flat autoconvolution."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    if S > 1e-15:
        a = a / S
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score(a)
    it = 0

    while time.monotonic() - t0 < time_limit:
        A = rfft(a, L)
        mag = np.abs(A)
        
        if target == 'flat':
            target_mag = np.full_like(mag, np.sqrt(np.mean(mag**2)))
        elif target == 'high_boost':
            # Slightly boost high frequencies to spread time-domain energy
            k = np.arange(len(mag))
            k_norm = k / max(1, len(mag) - 1)
            base = np.sqrt(np.mean(mag**2))
            target_mag = base * (1.0 + 0.3 * k_norm)
        else:
            target_mag = np.full_like(mag, np.sqrt(np.mean(mag**2)))

        mag_new = (1.0 - relax) * mag + relax * target_mag
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)

        a_new = irfft(A_new, L)[:n]
        a_new = np.clip(a_new, eps, None)
        S = np.sum(a_new)
        if S > 1e-10:
            a_new = a_new / S
        a = a_new
        it += 1

        if it % 100 == 0:
            s = score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()

    s = score(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()

    return best_a, best_score


def subgradient_optimize(a0, time_limit=30.0, top_k=5, lr_init=0.1, seed=42):
    """Subgradient descent on peak autoconvolution with adaptive restart."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score(a)
    lr = lr_init
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        C = autoconv(a)
        top_idx = np.argsort(C)[-top_k:]

        g = np.zeros(n)
        for k in top_idx:
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            g[valid] += a[idx[valid]]
        g /= top_k

        a = a - lr * g
        a = np.clip(a, 1e-8, None)
        S = np.sum(a)
        a = a / S

        it += 1
        if it % 50 == 0:
            s = score(a)
            if s < best_score - 1e-9:
                best_score = s
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            lr = max(lr * 0.995, 0.003)

            if stall > 80:
                rng = np.random.default_rng(seed + it * 7919)
                a = a * 0.5 + 0.5 * rng.random(n)
                a = np.clip(a, 1e-6, None)
                S = np.sum(a)
                a = a / S
                stall = 0
                lr = lr_init * 0.5

    s = score(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()

    return best_a, best_score


def lp_optimize(a0, time_limit=30.0, t_step=0.05, seed=42):
    """LP-based direction finding (from prior SOTA, streamlined)."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score(a)
    it = 0

    # Pre-build convolution matrix structure info
    # A @ b = conv(a, b) where A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For efficiency, build once per call
    A_mat = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A_mat[j:j+n, j] = a

    while time.monotonic() - t0 < time_limit:
        C = autoconv(a)
        M = float(np.max(C))

        # Solve LP: max sum(b) s.t. A@b <= M, b >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 2.0, 'presolve': True})
            if not res.success:
                break
            b = res.x
        except Exception:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        # Line search
        best_t = t_step
        best_s = score(a)
        for t_try in [t_step, 0.5*t_step, 0.2*t_step]:
            a_cand = (1.0 - t_try) * a + t_try * b
            a_cand = np.clip(a_cand, 0.0, None)
            s_cand = score(a_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        a = (1.0 - best_t) * a + best_t * b
        a = np.clip(a, 1e-8, None)
        S = np.sum(a)
        a = a / S
        it += 1

        if best_s < best_score:
            best_score = best_s
            best_a = a.copy()

        # Rebuild A_mat with updated a
        A_mat = np.zeros((2*n-1, n), dtype=np.float64)
        for j in range(n):
            A_mat[j:j+n, j] = a

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{elapsed:5.1f}s] it={it:3d} score={best_score:.6f}")

    return best_a, best_score


def local_search(a0, time_limit=20.0, seed=42):
    """Peak-directed local search with simulated annealing."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score(a)
    cur_score = best_score
    temp = 0.003

    while time.monotonic() - t0 < time_limit:
        C = autoconv(a)
        k_peak = int(np.argmax(C))

        idx = k_peak - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[valid] = a[valid] * a[idx[valid]]

        a_new = a.copy()
        r = rng.random()

        if r < 0.4:
            # Decrease top contributors to peak, increase others
            top_k = max(5, n // 8)
            threshold = np.partition(contrib, -top_k)[-top_k]
            top_mask = contrib > threshold
            if top_mask.sum() > 0:
                a_new[top_mask] *= np.exp(rng.normal(-0.03, 0.02, size=top_mask.sum()))
            other = ~top_mask
            if other.sum() > 0:
                n_inc = max(1, int(0.3 * other.sum()))
                inc_idx = np.where(other)[0]
                chosen = rng.choice(inc_idx, size=min(n_inc, len(inc_idx)), replace=False)
                a_new[chosen] *= np.exp(rng.normal(0.03, 0.02, size=len(chosen)))
        elif r < 0.7:
            # Windowed perturbation at peak-adjacent region
            center = min(n-1, max(0, k_peak // 2))
            width = max(2, n // 16)
            start = max(0, center - width//2)
            end = min(n, start + width)
            a_new[start:end] *= np.exp(rng.normal(0, 0.04, size=end-start))
        else:
            # Global smooth perturbation
            a_new *= np.exp(rng.normal(0, 0.015, size=n))

        a_new = np.clip(a_new, 1e-8, None)
        S = np.sum(a_new)
        a_new = a_new / S

        s_new = score(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
            cur_score = s_new
        elif rng.random() < np.exp(-(s_new - cur_score) / max(temp, 1e-10)):
            a = a_new
            cur_score = s_new

        temp *= 0.9997

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search: GS exploration + LP refinement + subgradient + local search."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence if available
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(provided) > 0 and np.sum(provided) > 0.01:
            s = score(provided)
            print(f"Provided: n={len(provided)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = provided.copy()

    n_values = [128, 192, 256, 320, 384, 512, 640, 768, 1024]

    # Phase 1: GS-based exploration (fast, many starts)
    print("\n=== Phase 1: GS Exploration ===")
    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - 180:
            break

        inits = []
        rng = np.random.default_rng(seed + n)

        # Random init
        inits.append(rng.random(n) + 0.1)
        # Hann window
        t = np.linspace(0, 1, n)
        inits.append(np.clip(0.5 * (1 + np.cos(2*np.pi*t)), 0.01, None))
        # Edge bump (large at edges)
        t_edge = np.abs(np.arange(n) - (n-1)/2) / max(1, (n-1)/2)
        inits.append(0.3 + 0.7 * t_edge)
        # Chirp-like
        t_chirp = np.arange(n) / n
        inits.append(np.clip(0.5 + 0.3*np.cos(2*np.pi*t_chirp**2) + 0.2*np.cos(2*np.pi*t_chirp), 0.01, None))

        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - 180:
                break
            a0 = np.array(a0, dtype=np.float64)
            a0 = a0 / np.sum(a0)

            t_gs = min(15.0, (deadline - time.monotonic() - 180) * 0.15)
            if t_gs < 2:
                break

            target = 'flat' if ii % 2 == 0 else 'high_boost'
            a_gs, s_gs = gs_optimize(a0, time_limit=t_gs, target=target)
            candidates.append((s_gs, n, a_gs.copy()))

            if s_gs < best_score:
                best_score = s_gs
                best_seq = a_gs.copy()
                print(f"  n={n:4d} init={ii} gs={s_gs:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n:4d} init={ii} gs={s_gs:.6f}")

    # Phase 2: Subgradient polish on top candidates
    print(f"\n=== Phase 2: Subgradient Polish ===")
    candidates.sort(key=lambda x: x[0])
    top_n_cands = min(5, len(candidates))

    for i in range(top_n_cands):
        if time.monotonic() > deadline - 100:
            break
        s_cand, n_cand, a_cand = candidates[i]
        t_sg = min(25.0, (deadline - time.monotonic() - 100) * 0.3)
        if t_sg < 3:
            break
        a_sg, s_sg = subgradient_optimize(a_cand, time_limit=t_sg, top_k=5)
        print(f"  Cand {i} (n={n_cand}, gs={s_cand:.6f}) -> sg={s_sg:.6f}")
        if s_sg < best_score:
            best_score = s_sg
            best_seq = a_sg.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 3: LP refinement on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        t_lp = min(80.0, deadline - time.monotonic() - 40.0)
        print(f"\n=== Phase 3: LP Refinement ({t_lp:.0f}s) ===")
        a_lp, s_lp = lp_optimize(best_seq, time_limit=t_lp)
        if s_lp < best_score:
            best_score = s_lp
            best_seq = a_lp.copy()
            print(f"  After LP: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  LP didn't improve: {s_lp:.6f} vs {best_score:.6f}")

    # Phase 4: More GS with different params on best
    if best_seq is not None and time.monotonic() < deadline - 40:
        t_gs2 = min(30.0, deadline - time.monotonic() - 30.0)
        print(f"\n=== Phase 4: GS Refinement ({t_gs2:.0f}s) ===")
        for relax in [0.3, 0.7]:
            if time.monotonic() > deadline - 30:
                break
            a_g2, s_g2 = gs_optimize(best_seq, time_limit=t_gs2/2, relax=relax)
            print(f"  GS relax={relax}: {s_g2:.6f}")
            if s_g2 < best_score:
                best_score = s_g2
                best_seq = a_g2.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 5: Local search
    if best_seq is not None and time.monotonic() < deadline - 20:
        t_ls = min(40.0, deadline - time.monotonic() - 15.0)
        print(f"\n=== Phase 5: Local Search ({t_ls:.0f}s) ===")
        a_ls, s_ls = local_search(best_seq, time_limit=t_ls)
        if s_ls < best_score:
            best_score = s_ls
            best_seq = a_ls.copy()
            print(f"  After LS: {best_score:.6f} *** NEW BEST ***")

    # Phase 6: Final subgradient + local search
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_fin = min(20.0, deadline - time.monotonic() - 10.0)
        if t_fin > 5:
            print(f"\n=== Phase 6: Final Polish ({t_fin:.0f}s) ===")
            a_f1, s_f1 = subgradient_optimize(best_seq, time_limit=t_fin*0.5, top_k=3, lr_init=0.05)
            a_f2, s_f2 = local_search(a_f1, time_limit=t_fin*0.5)
            s_final = min(s_f1, s_f2)
            a_final = a_f1 if s_f1 < s_f2 else a_f2
            if s_final < best_score:
                best_score = s_final
                best_seq = a_final.copy()
                print(f"  Final: {best_score:.6f} *** NEW BEST ***")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    fallback = [1.0] * 256
    return fallback
