"""
Multi-strategy optimizer for minimizing normalized peak self-convolution.
Uses smooth (log-sum-exp) gradient descent with annealing, combined with
simulated annealing polish and multiple restarts at different lengths.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def logsumexp(x, axis=None):
    """Numerically stable log-sum-exp."""
    x_max = np.max(x, axis=axis, keepdims=True)
    return (x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))).squeeze()

def smooth_loss_and_grad(theta, tau):
    """
    Compute smooth loss and gradient w.r.t. theta.
    a = softmax(theta), loss = 2*n*tau*logsumexp(conv(a,a)/tau)
    """
    n = len(theta)
    # softmax
    t = theta - np.max(theta)
    e = np.exp(t)
    a = e / np.sum(e)  # sum(a) = 1
    
    conv = fft_convolve(a, a)
    loss = 2.0 * n * tau * logsumexp(conv / tau)
    
    # Gradient of logsumexp(conv/tau) w.r.t. conv
    w = np.exp(conv / tau - logsumexp(conv / tau))  # normalized weights, length 2n-1
    
    # Gradient of conv w.r.t. a: for each m, d conv[k]/d a[m] = a[k-m] + a[m] (with boundary)
    # Sum over k of w[k] * d conv[k]/d a[m]
    # = sum_k w[k]*a[k-m] + sum_k w[k]*a[m]*1[0<=k-m<n]
    # First term: (w * a)[m] but w has length 2n-1 and a has length n
    # Actually, let's compute it as:
    # g_a[m] = sum_k w[k] * (a[k-m] if 0<=k-m<n else 0) + a[m] * sum_k w[k] * 1[0<=k-m<n]
    
    # First part: for each m, sum over k such that 0 <= k-m < n, i.e., m <= k < m+n
    # g_a1[m] = sum_{k=m}^{min(m+n-1, 2n-2)} w[k] * a[k-m]
    # This is a cross-correlation of w with a
    
    # Second part: g_a2[m] = a[m] * sum_{k=m}^{min(m+n-1, 2n-2)} w[k]
    # = a[m] * (cumsum_w_full[m+n-1] - cumsum_w_full[m-1]) (with appropriate handling)
    
    # Let's compute using the full weight vector
    w_full = w  # length 2n-1
    cum_w = np.concatenate([[0], np.cumsum(w_full)])
    
    # g_a2[m] = a[m] * (cum_w[min(m+n, 2n-1)] - cum_w[m])
    m_idx = np.arange(n)
    upper = np.minimum(m_idx + n, 2 * n - 1)
    g_a2 = a * (cum_w[upper] - cum_w[m_idx])
    
    # g_a1[m] = sum_{k=m}^{m+n-1} w[k] * a[k-m] (capped at 2n-2)
    # This is like a cross-correlation. For efficiency, use FFT.
    # g_a1[m] = sum_{j=0}^{n-1} w[m+j] * a[j] (where m+j <= 2n-2)
    # = sum_{j=0}^{min(n-1, 2n-2-m)} w[m+j] * a[j]
    
    # This is the cross-correlation of w and a at shift m.
    # cross_corr[m] = sum_j a[j] * w[m+j]
    # = convolve(a, w_reversed)[m + n - 1] ... hmm let me think differently.
    
    # Actually, let's just compute it with a sliding window approach or FFT.
    # For n up to a few thousand, a direct computation might be slow. Let's use FFT.
    
    # g_a1[m] = sum_j a[j] * w[m+j] for j = 0..min(n-1, 2n-2-m)
    # This is the cross-correlation of a and w at lag m.
    # cross_corr(m) = sum_j a[j] * w[j+m]
    # = <a, shift_m(w)>
    
    # Using FFT: if we define A = FFT(a, size) and W = FFT(w, size), then
    # the cross-correlation is ifft(A * conj(W)) ... but we need to be careful with the indexing.
    
    # Actually, cross_corr[m] = sum_j a[j] * w[j+m]
    # Let's define w_ext as w zero-padded to length >= 2n-1 (it already is)
    # Then cross_corr = ifft(FFT(a, 2n-1) * FFT(w_conj_reversed, 2n-1)) ... 
    
    # Simpler: just use the definition directly with numpy for moderate n,
    # or use the fact that this is a "sliding dot product".
    
    # For efficiency, let's use FFT-based cross-correlation:
    # cross_corr[m] = sum_j a[j] * conj(w)[conj(m+j)] ... no.
    
    # Let me just do it the straightforward way for now and optimize if needed.
    # For n <= 2048, a loop with numpy should be fast enough.
    
    # Actually, the cleanest FFT approach:
    # cross_corr[m] = sum_{j=0}^{n-1} a[j] * w[m+j]
    # = (a reversed) convolved with w, at appropriate index
    # Let a_rev = a[::-1], length n
    # conv(a_rev, w) has length n + 2n-2 - 1 = 3n-3
    # conv(a_rev, w)[m + n - 1] = sum_j a_rev[j] * w[m+n-1-j] = sum_j a[n-1-j] * w[m+n-1-j]
    # Let i = n-1-j: = sum_i a[i] * w[m+i]
    # Yes! So cross_corr[m] = conv(a_rev, w)[m + n - 1]
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w_full)  # length n + (2n-1) - 1 = 3n-2
    g_a1 = cross[n-1 : 2*n-1]  # indices n-1, n, ..., 2n-2 → length n
    
    g_a = g_a1 + g_a2  # gradient of loss w.r.t. a
    
    # Chain rule: d loss / d theta[m] = a[m] * (g_a[m] - sum_j g_a[j] * a[j])
    inner = np.dot(g_a, a)
    g_theta = a * (g_a - inner)
    
    return loss, g_theta

def adam_optimize(theta_init, time_limit, tau_start, tau_end, base_lr=1.0, verbose=False):
    """Adam optimization with annealed tau."""
    n = len(theta_init)
    theta = theta_init.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0
    
    best_loss = np.inf
    best_theta = theta.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential annealing of tau
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        loss, grad = smooth_loss_and_grad(theta, tau)
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        # Learning rate decay
        lr = base_lr / (1 + 0.05 * step) ** 0.5
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best (using actual max, not smooth approximation)
        t_shifted = theta - np.max(theta)
        a = np.exp(t_shifted) / np.sum(np.exp(t_shifted))
        actual_loss = local_score(a)
        
        if actual_loss < best_loss:
            best_loss = actual_loss
            best_theta = theta.copy()
        
        if verbose and step % 100 == 0:
            print(f"      step={step}, tau={tau:.5f}, smooth_loss={loss:.6f}, "
                  f"actual={actual_loss:.6f}, best={best_loss:.6f}")
    
    return best_theta, best_loss

def sa_polish(f_init, time_limit, rng):
    """Simulated annealing for final polishing."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Batch of mutations for speed
        batch_size = 1
        for _ in range(batch_size):
            candidate = f.copy()
            r = rng.random()
            
            if r < 0.4:
                # Target the peak contributors
                conv = fft_convolve(f, f)
                peak_idx = int(np.argmax(conv))
                # Find which (i, j) pairs contribute most to the peak
                if 0 <= peak_idx < 2 * n - 1:
                    contribs = np.zeros(n)
                    for i in range(n):
                        j = peak_idx - i
                        if 0 <= j < n:
                            contribs[i] = f[i] * f[j]
                    top_k = max(5, n // 8)
                    top_idx = np.argsort(contribs)[-top_k:]
                    candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=top_k))
                    other_idx = np.setdiff1d(np.arange(n), top_idx)
                    if len(other_idx) > 10:
                        n_inc = min(8, len(other_idx))
                        inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                        candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
            elif r < 0.65:
                # Local window modulation
                width = max(2, int(rng.integers(2, max(3, n // 4))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
            elif r < 0.8:
                # Two small windows
                for _ in range(2):
                    width = max(1, int(rng.integers(1, max(2, n // 8))))
                    start = int(rng.integers(0, n - width))
                    candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
            elif r < 0.9:
                # Global small perturbation
                candidate *= np.exp(rng.normal(0.0, 0.004, size=n))
            else:
                # Swap two segments
                w1 = max(1, int(rng.integers(1, max(2, n // 10))))
                w2 = max(1, int(rng.integers(1, max(2, n // 10))))
                s1 = int(rng.integers(0, n - w1))
                s2 = int(rng.integers(0, n - w2))
                if s1 != s2 and abs(s1 - s2) > max(w1, w2):
                    candidate[s1:s1+w1], candidate[s2:s2+w2] = candidate[s2:s2+w2].copy(), candidate[s1:s1+w1].copy()
            
            candidate = np.clip(candidate, 0.0, None)
            cand_sum = np.sum(candidate)
            if cand_sum < 1e-10:
                continue
            candidate = candidate * (sum_f / cand_sum)
            
            new_score = local_score(candidate)
            iter_count += 1
            
            if new_score < current_score:
                f = candidate
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_f = candidate.copy()
            elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
                f = candidate
                current_score = new_score
        
        temperature *= cooling ** batch_size
    
    return best_f, best_score

def make_init_from_angles(n, modes, amps, phases, rng):
    """Create initialization from sum of cosines."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.5
    for k, amp, phase in zip(modes, amps, phases):
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

"""
Multi-phase optimizer minimizing normalized peak self-convolution of non-negative sequences.
Uses efficient FFT-based gradient descent with annealed log-sum-exp smoothing,
combined with simulated annealing polish and multi-length search.
Key innovation: exact analytic gradient via cross-correlation identity,
piecewise tau schedule, and peak-targeted SA mutations.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smooth_loss_and_grad(theta, tau):
    """
    Compute smooth loss and exact gradient w.r.t. theta.
    a = softmax(theta), loss = 2n * tau * logsumexp(conv(a,a)/tau).
    Uses cross-correlation identity for O(n log n) gradient.
    """
    n = len(theta)
    t = theta - np.max(theta)
    e = np.exp(t)
    a = e / np.sum(e)

    conv = fft_convolve(a, a)
    conv_scaled = conv / tau

    c_max = np.max(conv_scaled)
    exp_shifted = np.exp(conv_scaled - c_max)
    sum_exp = np.sum(exp_shifted)
    log_sum_exp = c_max + np.log(sum_exp)

    loss = 2.0 * n * tau * log_sum_exp

    # Weights w[k] = exp(conv[k]/tau) / sum(exp(conv/tau))
    w = exp_shifted / sum_exp  # length 2n-1, sums to 1

    # Gradient w.r.t. a: g_a[m] = 4n * sum_j a[j] * w[m+j] for valid indices
    # This equals 4n * conv(a[::-1], w)[m + n - 1]
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    g_a = 4.0 * n * cross[n - 1:2 * n - 1]

    # Softmax chain rule: d loss/d theta[m] = a[m] * (g_a[m] - dot(g_a, a))
    inner = np.dot(g_a, a)
    g_theta = a * (g_a - inner)

    return loss, g_theta


def adam_optimize(theta_init, time_limit, tau_start, tau_end, base_lr=0.8,
                  verbose=False, num_restarts=2):
    """Adam optimization with piecewise tau annealing and restarts."""
    n = len(theta_init)
    theta = theta_init.copy()
    best_theta = theta.copy()
    best_score = local_score(np.exp(theta - np.max(theta)))

    t0 = time.monotonic()
    restart_time = time_limit / num_restarts
    rng_restart = np.random.default_rng(999)

    for restart in range(num_restarts):
        if restart > 0:
            # Perturb current best
            theta = best_theta + 0.05 * rng_restart.standard_normal(n)
            m = np.zeros(n)
            v = np.zeros(n)
            step = 0

        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        step = 0
        phase_start = time.monotonic()
        phase_limit = restart_time

        while time.monotonic() - phase_start < phase_limit:
            elapsed_frac = min(1.0, (time.monotonic() - phase_start) / phase_limit)

            # Piecewise tau schedule: 3 phases
            if elapsed_frac < 0.4:
                frac = elapsed_frac / 0.4
                tau = tau_start * (tau_start * 0.1 / tau_start) ** frac
            elif elapsed_frac < 0.75:
                frac = (elapsed_frac - 0.4) / 0.35
                tau = tau_start * 0.1 * (tau_start * 0.01 / (tau_start * 0.1)) ** frac
            else:
                frac = (elapsed_frac - 0.75) / 0.25
                tau = tau_start * 0.01 * (tau_end / (tau_start * 0.01)) ** frac

            loss, grad = smooth_loss_and_grad(theta, tau)

            step += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)

            lr = base_lr / (1.0 + 0.02 * step) ** 0.5
            theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Check actual score periodically
            if step % 20 == 0:
                t_shift = theta - np.max(theta)
                a_check = np.exp(t_shift)
                a_check = a_check / np.sum(a_check)
                actual = local_score(a_check)
                if actual < best_score:
                    best_score = actual
                    best_theta = theta.copy()

    return best_theta, best_score


def sa_polish(f_init, time_limit, rng, verbose=False):
    """Simulated annealing with peak-targeted mutations."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.003
    cooling = 0.99985
    iter_count = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-targeted: reduce contributors to peak, boost others
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(4, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.01, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 5:
                n_inc = min(6, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.01, size=n_inc))
        elif r < 0.55:
            # Multi-peak targeting
            conv = fft_convolve(f, f)
            # Find top 3 peaks
            peak_threshold = np.max(conv) * 0.95
            peak_indices = np.where(conv >= peak_threshold)[0]
            if len(peak_indices) > 0:
                # Reduce all main contributors
                contribs = np.zeros(n)
                for pk in peak_indices[:3]:
                    for i in range(n):
                        j = pk - i
                        if 0 <= j < n:
                            contribs[i] += f[i] * f[j]
                top_k = max(5, n // 8)
                top_idx = np.argsort(contribs)[-top_k:]
                dec_amp = 0.015 + 0.01 * (1.0 - (time.monotonic() - t0) / time_limit)
                candidate[top_idx] *= np.exp(rng.normal(-dec_amp, 0.005, size=top_k))
        elif r < 0.70:
            # Local window modulation
            width = max(2, int(rng.integers(2, max(3, n // 5))))
            start = int(rng.integers(0, n - width))
            amp = 0.015 + 0.01 * (1.0 - (time.monotonic() - t0) / time_limit)
            candidate[start:start + width] *= np.exp(rng.normal(0.0, amp, size=width))
        elif r < 0.80:
            # Two small windows
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.012, size=width))
        elif r < 0.90:
            # Global small perturbation
            amp = 0.003 + 0.002 * (1.0 - (time.monotonic() - t0) / time_limit)
            candidate *= np.exp(rng.normal(0.0, amp, size=n))
        else:
            # Mirror-like adjustment: swap influence of symmetric positions
            idx1 = int(rng.integers(0, n))
            idx2 = n - 1 - idx1
            if idx1 != idx2:
                swap = rng.uniform(0.05, 0.15)
                tmp = candidate[idx1]
                candidate[idx1] = (1 - swap) * candidate[idx1] + swap * candidate[idx2]
                candidate[idx2] = (1 - swap) * candidate[idx2] + swap * tmp

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            stagnant = 0
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
                if verbose and iter_count % 500 == 0:
                    print(f"    SA iter={iter_count}, best={best_score:.6f}")
        else:
            stagnant += 1
            if rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
                f = candidate
                current_score = new_score

        # Adaptive: if stuck, increase temperature briefly
        if stagnant > 200:
            temperature = max(temperature * 1.5, 0.001)
            stagnant = 0

        temperature *= cooling

    return best_f, best_score


def make_cosine_init(n, rng, num_components=8):
    """Create initialization as sum of random cosines."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.8
    for i in range(num_components):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.03, 0.15) * (1.0 - i / num_components)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_chirp_init(n, rng):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + 0.2 * np.cos(alpha * np.pi * t / n + rng.uniform(0, np.pi))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_bipartite_init(n, rng):
    """Bipartite-like initialization (alternating pattern with noise)."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.1 * np.cos(np.pi * t)
    a += 0.05 * rng.standard_normal(n)
    # Add smooth modulation
    a *= 1.0 + 0.1 * np.cos(2 * np.pi * t / n * rng.integers(1, 5) + rng.uniform(0, 2 * np.pi))
    return np.clip(a, 0.05, None)


def get_tau_range(a):
    """Estimate good tau range from current sequence."""
    conv = fft_convolve(a, a)
    conv_max = np.max(conv)
    conv_min = np.min(conv)
    spread = conv_max - conv_min
    tau_start = max(spread / 5.0, conv_max * 0.1)
    tau_end = max(spread / 500.0, conv_max * 0.001)
    return tau_start, tau_end


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Gradient-based smooth optimization (annealed log-sum-exp + Adam)
    2. SA polish with peak-targeted mutations
    3. Multiple initializations and sequence lengths
    4. Final micro-polish
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if np.sum(provided_seq) > 0.01:
            s_prov = local_score(provided_seq)
            print(f"Provided: n={len(provided_seq)}, score={s_prov:.6f}")
            if s_prov < best_score:
                best_score = s_prov
                best_seq = provided_seq.copy()

    n_base = len(provided_seq) if provided_seq is not None else 256

    # Search configurations: (n, time_fraction, init_type)
    configs = [
        (n_base, 0.22, "provided"),
        (n_base, 0.18, "cosine1"),
        (n_base, 0.12, "cosine2"),
        (n_base, 0.10, "chirp"),
        (int(n_base * 1.5), 0.12, "interpolate"),
        (int(n_base * 0.67), 0.08, "interpolate_short"),
        (n_base, 0.10, "bipartite"),
        (n_base, 0.08, "best_refine"),
    ]

    available_time = deadline - time.monotonic()
    print(f"Available time: {available_time:.0f}s, base n={n_base}")

    for cfg_idx, (n, frac, init_type) in enumerate(configs):
        if time.monotonic() > deadline - 80:
            break

        time_alloc = max(20.0, frac * available_time * 0.9)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 20)
        if time_alloc < 25:
            break

        print(f"\n[{cfg_idx+1}/{len(configs)}] n={n}, init={init_type}, alloc={time_alloc:.0f}s")

        # Generate initialization
        if init_type == "provided" and provided_seq is not None and len(provided_seq) == n:
            a0 = provided_seq.copy()
        elif init_type == "cosine1":
            a0 = make_cosine_init(n, rng, num_components=10)
        elif init_type == "cosine2":
            a0 = make_cosine_init(n, rng, num_components=15)
        elif init_type == "chirp":
            a0 = make_chirp_init(n, rng)
        elif init_type == "interpolate":
            if provided_seq is not None:
                x_old = np.linspace(0, 1, len(provided_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, provided_seq)
            else:
                a0 = make_cosine_init(n, rng, 8)
            a0 = np.clip(a0, 0.05, None)
        elif init_type == "interpolate_short":
            if provided_seq is not None:
                x_old = np.linspace(0, 1, len(provided_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, provided_seq)
            else:
                a0 = make_cosine_init(n, rng, 8)
            a0 = np.clip(a0, 0.05, None)
        elif init_type == "bipartite":
            a0 = make_bipartite_init(n, rng)
        else:  # best_refine
            if best_seq is not None:
                if len(best_seq) == n:
                    a0 = best_seq + 0.02 * rng.standard_normal(n)
                else:
                    x_old = np.linspace(0, 1, len(best_seq))
                    x_new = np.linspace(0, 1, n)
                    a0 = np.interp(x_new, x_old, best_seq)
            else:
                a0 = make_cosine_init(n, rng, 8)
            a0 = np.clip(a0, 0.05, None)

        # Split time: 70% gradient, 30% SA
        grad_time = time_alloc * 0.65
        sa_time = time_alloc * 0.35

        # Determine number of restarts for gradient phase
        num_restarts = 2 if grad_time > 100 else 1

        # Gradient optimization
        theta0 = np.log(a0 / np.sum(a0))
        tau_start, tau_end = get_tau_range(a0)

        f_grad, score_grad = adam_optimize(theta0, grad_time, tau_start, tau_end,
                                           base_lr=0.7, verbose=False,
                                           num_restarts=num_restarts)
        a_result = np.exp(f_grad - np.max(f_grad))
        a_result = a_result / np.sum(a_result) * np.sum(a0)
        a_result = np.clip(a_result, 0.01, 1000.0)
        score_grad = local_score(a_result)

        print(f"  Gradient: score={score_grad:.6f}")

        if score_grad < best_score:
            best_score = score_grad
            best_seq = a_result.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}) ***")

        # SA polish on current best
        if sa_time > 10 and best_seq is not None:
            f_sa, score_sa = sa_polish(best_seq, sa_time, rng, verbose=False)
            print(f"  SA: score={score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = f_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Final refinement phase ===
    remaining = deadline - time.monotonic()
    if remaining > 40 and best_seq is not None:
        print(f"\n{'='*50}")
        print(f"Final refinement: {remaining:.0f}s remaining")
        print(f"Current best: {best_score:.6f}, n={len(best_seq)}")

        # Short gradient polish
        grad_time_final = min(remaining * 0.4, 120.0)
        if grad_time_final > 15:
            theta0 = np.log(np.clip(best_seq, 0.01, None) / np.sum(best_seq))
            tau_s, tau_e = get_tau_range(best_seq)
            f_g, s_g = adam_optimize(theta0, grad_time_final, tau_s, tau_e,
                                     base_lr=0.3, num_restarts=1)
            a_g = np.exp(f_g - np.max(f_g))
            a_g = a_g / np.sum(a_g) * np.sum(best_seq)
            a_g = np.clip(a_g, 0.01, 1000.0)
            s_g = local_score(a_g)
            print(f"  Final gradient: {s_g:.6f}")
            if s_g < best_score:
                best_score = s_g
                best_seq = a_g.copy()

        # Final SA
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            f_f, s_f = sa_polish(best_seq, remaining2 * 0.85, rng)
            print(f"  Final SA: {s_f:.6f}")
            if s_f < best_score:
                best_score = s_f
                best_seq = f_f.copy()

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if provided_seq is not None and np.sum(provided_seq) > 0.01:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
