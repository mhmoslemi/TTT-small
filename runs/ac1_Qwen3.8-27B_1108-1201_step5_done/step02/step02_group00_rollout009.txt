def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """
    Compute weighted subgradient of max(conv) at point a.
    Uses softmax weighting over top-k peaks with given temperature.
    """
    n = len(a)
    conv_len = len(conv)
    
    if top_k >= conv_len:
        peaks = np.arange(conv_len)
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    # Compute weights
    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i]
        # grad c[k] / da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n and j!=k-j)
        # Simplified: for each valid j, contribution is a[k-j] if k-j in [0,n)
        # plus a[j-k] if j-k in [0,n) and j != k-j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            contrib = 0.0
            # First term: a[k-j]
            if 0 <= k - j < n:
                contrib += a[k - j]
            # Second term: a[j-k] (only if different from first)
            if 0 <= j - k < n and j != k - j:
                contrib += a[j - k]
            sg[j] += wi * contrib
    
    return sg

def compute_subgradient_v2(a, conv, top_k, temperature):

    For linear auto-convolution c[k] = sum_j a[j]*a[k-j],

        # a[k-m] is valid when 0 <= k-m <= n-1, i.e., k-n+1 <= m <= k
        # a[m-k] is valid when 0 <= m-k <= n-1, i.e., k <= m <= k+n-1

        p1_start = max(0, k - n + 1)
        p1_end = min(n - 1, k)
        if p1_end >= p1_start:

        p2_start = max(0, k)

    return 2.0 * n * conv.max()

        w = w / w_sum
        
        # d(conv_k)/dx = d(conv_k)/da * da/dx
        # da/dx = diag(a) - a*a^T (Jacobian of softmax)
        # So d(conv_k)/dx = (diag(a) - a*a^T)^T @ (d(conv_k)/da)
        # = a * (d(conv_k)/da) - a * sum(a * (d(conv_k)/da))
        
        # Compute d(conv_k)/da for weighted sum
        grad_a = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k_int in active:
            k = int(k_int)
            wk = w[k_int]
            # d(conv[k])/da[m] = 2*a[k-m] (if valid, counting both terms)
            # Actually: d(conv[k])/da[m] = a[k-m] + a[m-k] (with a[x]=0 outside [0,n-1])
            # = 2*a[k-m] when both are valid and m != k-m, or a[m] when m = k-m
            p1_start = max(0, k - n + 1)
            p1_end = min(n - 1, k)
            m_range = np.arange(p1_start, p1_end + 1)
            grad_a[m_range] += wk * 2.0 * a_local[k - m_range]
            # Symmetric part
            p2_start = max(0, k)
            p2_end = min(n - 1, k + n - 1)
            if p2_end >= p2_start:
                m_range2 = np.arange(p2_start, p2_end + 1)
                mask = np.ones(len(m_range2), dtype=bool)
                if k % 2 == 0:
                    mask[m_range2 == k // 2] = False
                if mask.any():
                    grad_a[m_range2[mask]] += wk * 2.0 * a_local[m_range2[mask] - k]
        
        # Apply softmax Jacobian
        grad_x = a_local * grad_a - a_local * np.dot(a_local, grad_a)
        grad_x = 2.0 * n * grad_x
        
        if time.monotonic() - t0 > time_limit + 5:
            return obj_val, grad_x
        
        return obj_val, grad_x
    
    # Run L-BFGS-B with time limit via callback
    state = {'done': False}
    
    def callback(xk):
        if time.monotonic() - t0 > time_limit - 5:
            state['done'] = True
        return state['done']
    
    try:
        result = scipy_minimize(
            obj_and_grad, x0, method='L-BFGS-B',
            jac=True, callback=callback,
            options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        if result.x is not None:
            x_opt = result.x
            x_shifted = x_opt - x_opt.max()
            exp_x = np.exp(x_shifted)
            a_opt = exp_x / exp_x.sum()
            a_opt = np.maximum(a_opt, 0)
            s_opt = a_opt.sum()
            if s_opt > 1e-15:
                a_opt = a_opt / s_opt
            conv_opt = fast_convolve(a_opt)
            obj_opt = objective_from_conv(n, conv_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
    except Exception:
        pass
    
    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian centered
    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        s = g.sum()
        if s > 1e-10:
            inits[f'gauss_{sf}'] = g / s
    
    # Edge-concentrated
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    s = bi.sum()
    if s > 0:
        inits['edge'] = bi / s
    
    # Lopsided
    lop = np.exp(-x / (0.3 * n))
    s = lop.sum()
    if s > 1e-10:
        inits['lopsided'] = lop / s
    inits['lopsided_r'] = inits['lopsided'][::-1].copy()
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    s = cos_w.sum()
    if s > 1e-10:
        inits['cosine'] = cos_w / s
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining adaptive subgradient,
    mirror descent, and L-BFGS polishing to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: length={len(p)}")
    except Exception:
        pass
    
    # Phase 1: Broad exploration (25% of time)
    print("=== Phase 1: Broad Exploration ===")
    t0 = time.monotonic()
    phase1_time = (deadline - t0) * 0.25
    
    # Try various lengths, weighted toward larger
    lengths = [256, 384, 512, 768, 1024, 1536, 2048]
    
    # Time allocation: more time for larger n
    time_weights = [1, 1, 1.5, 1.5, 2, 2, 3]
    total_w = sum(time_weights)
    
    candidates = []  # (score, n, a)
    
    for idx, n in enumerate(lengths):
        t_alloc = phase1_time * time_weights[idx] / total_w
        t_alloc = min(t_alloc, deadline - time.monotonic() - 20)
        if t_alloc < 10:
            break
        
        inits = generate_diverse_inits(n, parent, rng)
        init_names = list(inits.keys())
        
        # Select top initializations
        priority = ['parent', 'parent_ds', 'parent_up', 'gauss_0.2', 'gauss_0.3',
                    'edge', 'random', 'uniform', 'lopsided', 'cosine']
        init_names = [nm for nm in priority if nm in inits]
        init_names = init_names[:4]  # Try at most 4 per length
        
        t_per_init = t_alloc / max(1, len(init_names))
        
        for name in init_names:
            t_single = min(t_per_init, deadline - time.monotonic() - 15)
            if t_single < 5:
                break
            
            a0 = inits[name]
            
            # Use adaptive subgradient
            a_opt, obj_opt = optimize_adaptive(
                a0, time_limit=t_single, lr_base=0.04,
                momentum_base=0.85, top_k_frac=0.08, temp_base=0.002)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                candidates.append((score, n, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1: n={n}, {name}, score={score:.6f}")
    
    print(f"Phase 1 done. Best: {best_score:.6f}")
    candidates.sort(key=lambda x: x[0])
    
    # Phase 2: Focused optimization (50% of time)
    print("=== Phase 2: Focused Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = (deadline - phase2_start) * 0.50
    
    # Take top candidates and optimize further
    top_candidates = candidates[:6] if len(candidates) >= 6 else candidates
    
    # Add the current best if not already there
    if best_sequence is not None:
        top_candidates.append((best_score, len(best_sequence), best_sequence.copy()))
    
    # Also add fresh starts at the best-performing length
    if best_sequence is not None:
        best_n = len(best_sequence)
        for _ in range(2):
            fresh = rng.dirichlet(np.ones(best_n) * 0.3)
            top_candidates.append((float('inf'), best_n, fresh))
    
    seen = set()
    unique_candidates = []
    for sc, n, a in top_candidates:
        key = (n, tuple(np.round(a[:10], 4)))
        if key not in seen:
            seen.add(key)
            unique_candidates.append((sc, n, a))
        if len(unique_candidates) >= 8:
            break
    
    t_per_candidate = phase2_time / max(1, len(unique_candidates))
    
    for i, (sc, n, a0) in enumerate(unique_candidates):
        t_alloc = min(t_per_candidate, deadline - time.monotonic() - 30)
        if t_alloc < 10:
            break
        
        # Try multiple methods
        t_sub = t_alloc * 0.4
        t_mir = t_alloc * 0.3
        t_lbfgs = t_alloc * 0.3
        
        # Subgradient
        if t_sub > 5:
            a_opt, obj_opt = optimize_adaptive(
                a0, time_limit=t_sub, lr_base=0.03,
                momentum_base=0.88, top_k_frac=0.06, temp_base=0.001)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2-sub: n={n}, score={score:.6f}")
            a0 = a_opt  # Chain: use result as next init
        
        # Mirror descent
        if t_mir > 5 and time.monotonic() < deadline - 25:
            a_opt, obj_opt = optimize_mirror_v2(
                a0, time_limit=t_mir, eta_base=0.4, t_base=60.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2-mir: n={n}, score={score:.6f}")
            a0 = a_opt
        
        # L-BFGS smooth
        if t_lbfgs > 5 and time.monotonic() < deadline - 25:
            a_opt, obj_opt = optimize_lbfgs_smooth(
                a0, time_limit=t_lbfgs, T=300.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2-lbfgs: n={n}, score={score:.6f}")
    
    print(f"Phase 2 done. Best: {best_score:.6f}")
    
    # Phase 3: Intensive polishing (25% of time)
    print("=== Phase 3: Intensive Polishing ===")
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    phase3_start = time.monotonic()
    phase3_time = deadline - phase3_start - 10
    
    n_best = len(best_sequence)
    current_best = best_sequence.copy()
    
    # Multiple polishing strategies
    strategies = [
        # (name, method, params)
        ('sub_fine', 'sub', {'lr_base': 0.01, 'momentum_base': 0.92, 'top_k_frac': 0.03, 'temp_base': 0.0003}),
        ('sub_mom', 'sub', {'lr_base': 0.005, 'momentum_base': 0.95, 'top_k_frac': 0.02, 'temp_base': 0.0001}),
        ('mir_fine', 'mir', {'eta_base': 0.15, 't_base': 100.0}),
        ('lbfgs_highT', 'lbfgs', {'T': 500.0}),
        ('lbfgs_medT', 'lbfgs', {'T': 200.0}),
        ('sub_broad', 'sub', {'lr_base': 0.02, 'momentum_base': 0.90, 'top_k_frac': 0.10, 'temp_base': 0.001}),
    ]
    
    t_per_strategy = phase3_time / (len(strategies) + 4)
    
    for name, method, params in strategies:
        t_alloc = min(t_per_strategy, deadline - time.monotonic() - 10)
        if t_alloc < 5:
            break
        
        if method == 'sub':
            a_opt, obj_opt = optimize_adaptive(
                current_best, time_limit=t_alloc, **params)
        elif method == 'mir':
            a_opt, obj_opt = optimize_mirror_v2(
                current_best, time_limit=t_alloc, **params)
        elif method == 'lbfgs':
            a_opt, obj_opt = optimize_lbfgs_smooth(
                current_best, time_limit=t_alloc, **params)
        else:
            continue
        
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            current_best = a_opt.copy()
            print(f"  P3-{name}: score={score:.6f}")
        else:
            # Keep the one with better internal objective
            if obj_opt < objective_from_conv(n_best, fast_convolve(current_best)):
                pass  # Don't update best, but could use as next init
    
    # Random perturbation restarts
    print("  Running perturbation restarts...")
    remaining = deadline - time.monotonic() - 10
    if remaining > 30:
        n_perturb = max(2, int(remaining / 30))
        for pi in range(n_perturb):
            t_alloc = min(remaining / (n_perturb - pi), deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            
            # Different perturbation scales
            scale = 0.005 + 0.01 * pi / max(n_perturb, 1)
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(current_best + noise)
            
            a_opt, obj_opt = optimize_adaptive(
                a_pert, time_limit=t_alloc, lr_base=0.01,
                momentum_base=0.93, top_k_frac=0.04, temp_base=0.0005)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                current_best = a_opt.copy()
                print(f"  P3-pert{pi}: score={score:.6f}")
            
            remaining = deadline - time.monotonic() - 10
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
