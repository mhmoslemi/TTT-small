"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Uses fast coordinate descent with incremental convolution updates, subgradient method,
and simulated annealing across multiple sequence lengths and initializations.
Key innovation: incremental O(n) convolution updates per coordinate enable thousands
of sweeps per second for large n, outperforming LP-based approaches significantly.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def make_init(n, kind, rng):
    """Generate initial sequence of length n."""
    t = np.arange(n, dtype=np.float64)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((t - center) / (n / 5.0)) ** 2)
    elif kind == 'random':
        a = 0.5 + rng.random(n)
    elif kind == 'bell':
        center = (n - 1) / 2.0
        a = 1.0 / (1.0 + ((t - center) / (n / 6.0)) ** 4)
    elif kind == 'plateau':
        w = n // 3
        a = np.zeros(n)
        a[w//2:n-w//2] = 1.0
        # Smooth edges
        for i in range(w//2):
            a[i] = a[w//2] * (i + 1) / (w//2)
            a[n-1-i] = a[n-1-w//2] * (i + 1) / (w//2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a += 0.005 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return a


def coord_descent(a, time_limit=100.0, verbose=False, seed=42):
    """
    Coordinate descent with incremental convolution updates.
    For each coordinate j, try several multiplicative factors and pick the best.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(np.sum(a))
    conv = fft_convolve(a, a).copy()
    best_score = 2.0 * n * np.max(conv) / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Factors to try for each coordinate
    factors = np.array([0.0, 0.1, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0])

    sweep = 0
    stall = 0

    while time.monotonic() - t0 < time_limit - 5.0:
        improved = False
        order = rng.permutation(n)

        for j in order:
            x = a[j]
            if x < 1e-12:
                continue

            # Compute contribution vector: d[k] = a[k-j] for valid k, 0 otherwise
            # d is non-zero for k in [j, min(j+n-1, 2n-2)]
            d = np.zeros(2 * n - 1)
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            if k_start <= k_end:
                d[k_start:k_end + 1] = a[0:k_end - k_start + 1]
            # Zero out k=2j (handled separately)
            if 0 <= 2 * j < 2 * n - 1:
                d[2 * j] = 0.0

            # Current score
            current_max = np.max(conv)
            current_s = 2.0 * n * current_max / (S * S)

            # Try different y values
            best_y = x
            best_s = current_s

            for f in factors:
                y = x * f
                if abs(y - x) < 1e-15:
                    continue
                S_new = S - x + y
                if S_new < 0.01:
                    continue

                # Compute new max efficiently
                # delta[k] = 2*(y-x)*d[k] for k != 2j
                # delta[2j] = y^2 - x^2
                delta = 2.0 * (y - x) * d
                if 0 <= 2 * j < 2 * n - 1:
                    delta[2 * j] = y * y - x * x
                new_max = np.max(conv + delta)
                s_new = 2.0 * n * new_max / (S_new * S_new)
                if s_new < best_s:
                    best_s = s_new
                    best_y = y

            if best_y != x:
                # Update conv and S
                delta = 2.0 * (best_y - x) * d
                if 0 <= 2 * j < 2 * n - 1:
                    delta[2 * j] = best_y * best_y - x * x
                conv += delta
                S += best_y - x
                a[j] = best_y
                improved = True

        sweep += 1

        # Check improvement
        current_score = 2.0 * n * np.max(conv) / (S * S)
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"        CD sweep {sweep}: score={current_score:.8f}, best={best_score:.8f}, stall={stall}")

        # Escape stalling
        if stall > 30:
            # Perturb a random subset
            n_pert = max(10, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.1, size=n_pert))
            a = np.maximum(a, 0.01)
            S = float(np.sum(a))
            conv = fft_convolve(a, a)
            stall = 0

    return best_a, best_score


def subgradient_polish(a, time_limit=60.0, verbose=False):
    """
    Subgradient descent on max(conv(a,a)) with S fixed, using momentum.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a * (np.sqrt(2.0 * n) / S)  # Normalize
    S = float(np.sum(a))
    best_score = score_fast(a)
    best_a = a.copy()
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    it = 0

    while time.monotonic() - t0 < time_limit - 2.0:
        conv = fft_convolve(a, a)
        max_val = np.max(conv)

        # Find all argmax positions
        top_k = min(10, len(conv))
        top_idx = np.argsort(conv)[-top_k:]
        # Only use those that are at the max (within tolerance)
        tol = 1e-8
        active_idx = [k for k in top_idx if conv[int(k)] >= max_val - tol]
        if not active_idx:
            active_idx = [int(np.argmax(conv))]

        # Compute subgradient: average of gradients at active peaks
        g = np.zeros(n)
        for k in active_idx:
            ki = int(k)
            # d(conv[k])/da[j] = a[k-j] + a[j-k] (with a[i]=0 for invalid)
            for j in range(n):
                val = 0.0
                if 0 <= ki - j < n:
                    val += a[ki - j]
                if 0 <= j - ki < n:
                    val += a[j - ki]
                g[j] = val
        g /= len(active_idx)

        # Center the gradient (since we're optimizing on the simplex-like constraint)
        g -= np.mean(g)

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        a = a - lr * momentum
        a = np.maximum(a, 0.01)

        # Renormalize (score is scale-invariant, but keep S reasonable)
        S = float(np.sum(a))
        if S < 0.01:
            a = a * 10.0
            S = float(np.sum(a))

        it += 1
        if it % 100 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose:
                print(f"        Subgrad it {it}: score={s:.8f}, best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish_fast(a, time_limit=60.0, verbose=False):
    """
    Faster subgradient method using vectorized gradient computation.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 0.01:
        return a, score_fast(a)
    a = a / S  # Normalize to sum=1
    best_score = score_fast(a)
    best_a = a.copy()
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.005
    it = 0

    while time.monotonic() - t0 < time_limit - 2.0:
        conv = fft_convolve(a, a)
        max_val = np.max(conv)

        # Find active peaks (within tolerance of max)
        tol = max_val * 0.01 + 1e-10
        active_mask = conv >= max_val - tol
        active_idx = np.where(active_mask)[0]

        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(conv))])

        # Compute gradient vectorized
        # d(conv[k])/da[j] = a[k-j] + a[j-k] (with bounds)
        g = np.zeros(n)
        for ki in active_idx:
            ki = int(ki)
            # Vectorized: for j in 0..n-1
            # a[ki-j] is valid when 0 <= ki-j < n, i.e., max(0, ki-n+1) <= j <= min(n-1, ki)
            # a[j-ki] is valid when 0 <= j-ki < n, i.e., max(0, ki) <= j <= min(n-1, ki+n-1)
            lo1 = max(0, ki - n + 1)
            hi1 = min(n - 1, ki)
            if lo1 <= hi1:
                g[lo1:hi1 + 1] += a[ki - hi1:ki - lo1 + 1][::-1]
            lo2 = max(0, ki)
            hi2 = min(n - 1, ki + n - 1)
            if lo2 <= hi2:
                g[lo2:hi2 + 1] += a[ki:ki + (hi2 - lo2 + 1)]

        g /= len(active_idx)
        g -= np.mean(g)

        momentum = beta * momentum + (1.0 - beta) * g
        a = a - lr * momentum
        a = np.maximum(a, 1e-4)
        a = a / np.sum(a)  # Renormalize

        it += 1
        if it % 200 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose:
                print(f"        SubGrad it {it}: score={s:.8f}, best={best_score:.8f}")

    s_final = score_fast(a)
    if s_final < best_score:
        best_score = s_final
        best_a = a.copy()

    return best_a, best_score


def simulated_annealing(a, time_limit=60.0, verbose=False, seed=123):
    """
    Simulated annealing with structured moves focused on peak reduction.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    temp = 0.005
    cooling = 0.9995
    current_score = best_score
    it = 0

    while time.monotonic() - t0 < time_limit - 2.0:
        candidate = a.copy()
        r = rng.random()

        if r < 0.5:
            # Peak-targeted move: reduce elements contributing most to peak
            conv = fft_convolve(a, a)
            peak_k = int(np.argmax(conv))
            # Find top contributors
            contribs = np.zeros(n)
            for j in range(n):
                val = 0.0
                if 0 <= peak_k - j < n:
                    val += a[peak_k - j]
                if 0 <= j - peak_k < n:
                    val += a[j - peak_k]
                contribs[j] = val
            # Reduce top contributors, increase others
            top_m = max(5, n // 10)
            top_idx = np.argsort(contribs)[-top_m:]
            candidate[top_idx] *= np.exp(rng.normal(-0.08, 0.05, size=top_m))
            rest = np.setdiff1d(np.arange(n), top_idx)
            inc_m = min(len(rest), top_m // 2)
            if inc_m > 0:
                inc_idx = rng.choice(rest, size=inc_m, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.08, 0.05, size=inc_m))
        elif r < 0.75:
            # Block move
            width = rng.integers(2, max(3, n // 10))
            start = rng.integers(0, max(1, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.06, size=width))
        elif r < 0.9:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0, 0.02, size=n))
        else:
            # Correlated move (smooth perturbation)
            noise = rng.normal(0, 0.03, size=n)
            # Smooth the noise
            kernel = np.ones(5) / 5.0
            noise = np.convolve(noise, kernel, mode='same')
            candidate *= np.exp(noise)

        candidate = np.maximum(candidate, 0.01)
        new_score = score_fast(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-10)):
            a = candidate
            current_score = new_score

        temp *= cooling
        it += 1
        if verbose and it % 100 == 0:
            print(f"        SA it {it}: current={current_score:.8f}, best={best_score:.8f}, T={temp:.6f}")

    return best_a, best_score


def refine_length(a, time_limit=60.0, verbose=False):
    """
    Try to improve a sequence by trying slightly different lengths
    (interpolation to new length + coordinate descent).
    """
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    n = len(a)

    for new_n in [n * 2, n // 2, n + 64, n - 64]:
        if time.monotonic() - t0 > time_limit - 10:
            break
        if new_n < 64 or new_n > 8192:
            continue

        # Interpolate
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
        a_new = np.maximum(a_new, 0.01)

        # Quick coordinate descent
        t_remaining = time_limit - (time.monotonic() - t0) - 5.0
        if t_remaining < 5.0:
            continue
        a_ref, s_ref = coord_descent(a_new, time_limit=t_remaining * 0.6, verbose=False, seed=99 + new_n % 100)

        if s_ref < best_score:
            best_score = s_ref
            best_a = a_ref.copy()
            if verbose:
                print(f"        Length {new_n}: score={s_ref:.8f} (improved!)")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: multi-length, multi-initialization coordinate descent
    with subgradient polish and simulated annealing.
    """
    t_start = time.monotonic()

    if 'height_sequence_1' in globals() and height_sequence_1 is not None:

        if len(provided) > 0:
            s = score_fast(provided)
            print(f"Provided sequence: n={len(provided)}, score={s:.8f}")
            if s < best_score:
                best_score = s
                best_seq = provided.copy()

    # Define optimization tasks: (n, initialization_type)
    tasks = []
    if provided is not None:
        tasks.append((len(provided), 'provided'))

    # Larger sequences have better theoretical bounds
    for n in [512, 1024, 2048, 4096]:
        for init in ['uniform', 'hann', 'gaussian']:
            tasks.append((n, init))

    print(f"\nTotal tasks: {len(tasks)}, deadline in {budget_s - 25:.0f}s")
    print(f"{'='*60}")

    # Time allocation: weight by sqrt(1/n) since CD is O(n^2) per sweep
    # But larger n has better potential, so weight by 1/sqrt(n)
    time_weights = []
    for n, _ in tasks:
        time_weights.append(1.0 / np.sqrt(n / 128.0))
    total_w = sum(time_weights)

    task_idx = 0
    for n, init_type in tasks:
        if time.monotonic() > deadline - 60.0:
            break

        # Generate initial sequence
        if init_type == 'provided' and provided is not None:

        else:

        # Allocate time

        t_alloc = max(15.0, remaining * time_weights[task_idx] / total_w)

            if s2 < s1:

        # Phase 3: Simulated annealing

        # Update global best

            best_score = s1
            best_seq = f1.copy()
            print(f"    *** NEW GLOBAL BEST: {best_score:.8f} (n={n}) ***")

        task_idx += 1

    # Final refinement phase: take the best and try harder
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        t_remaining = deadline - time.monotonic() - 20.0

            if s_ref < best_score:
                best_score = s_ref
                best_seq = f_ref.copy()
                print(f"  After refinement CD: {best_score:.8f}")

        # Subgradient
        t_sub2 = min(60.0, t_remaining * 0.25)
        if t_sub2 > 10.0 and time.monotonic() < deadline - 15.0:
            f_sub, s_sub = subgradient_polish_fast(best_seq, time_limit=t_sub2, verbose=True)
            if s_sub < best_score:
                best_score = s_sub
                best_seq = f_sub.copy()
                print(f"  After refinement SubGrad: {best_score:.8f}")

        # SA
        t_sa2 = max(0.0, t_remaining - (t_cd2 if t_cd2 < 120 else 120.0) - (t_sub2 if t_sub2 < 60 else 60.0) - 10.0)
        if t_sa2 > 10.0 and time.monotonic() < deadline - 10.0:
            f_sa, s_sa = simulated_annealing(best_seq, time_limit=t_sa2, verbose=True, seed=777)
            if s_sa < best_score:
                best_score = s_sa
                best_seq = f_sa.copy()

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with actual evaluate function

        print(f"\n{'='*60}")

        print(f"Total time: {time.monotonic()-t_start:.1f}s")
        print(f"{'='*60}")
        return final_list

    # Fallback
    print("WARNING: No valid sequence found, returning fallback.")
    return [1.0] * 256
