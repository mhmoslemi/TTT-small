```python
import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)_k <= M for all k"""
    n = len(a)
    # Constraint matrix: row k is the row of the convolution matrix
    # (a*b)_k = sum_j a_{k-j} * b_j for j in [max(0,k-n+1), min(n-1,k)]
    # We build A such that A @ b <= M * ones(2n-1)
    
    # Build convolution matrix C where C[k,j] = a_{k-j} if valid, else 0
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    # Maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    b_constraint = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_constraint, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
def adaptive_subgradient(a, conv, M, dual_weights, top_k=None):
    """Subgradient weighted by LP dual variables."""
    n = len(a)
    if top_k is not None:
        indices = np.argpartition(conv, -top_k)[-top_k:]
    else:
        indices = np.arange(len(conv))
    
    sg = np.zeros(n)
    for k in indices:
        w = dual_weights[k] if dual_weights is not None else 1.0
        if w < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w * 2.0 * a[k - idx]
    return sg
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
smooth max approximation with gradient descent, and multi-phase search.
Key innovation: Uses linear programming to find optimal redistribution directions,
combined with temperature-annealed smooth max for fine-grained optimization.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective_normalized(a):
    """Compute 2n*max_b for a sequence with sum=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(b) subject to b >= 0, conv(a,b)_k <= M for all k.
    Returns the optimal b (unnormalized).
    """
    n = len(a)
    # Build constraint matrix A where row k gives conv(a,b)_k as a function of b
    # conv(a,b)_k = sum_j a[k-j] * b[j] for valid j
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            A[k, j] = a[k-j]
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

def optimize_lp(a0, time_limit, max_iters=50000):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > max_iters:
            break
        
        # Solve LP
        b = solve_lp_direction(a, M)
        if b is None:
            # LP failed, try subgradient step
            break
        
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        b_norm = b / b_sum
        
        # Check if b is very close to a (stuck)
        diff = np.linalg.norm(b_norm - a)
        if diff < 1e-6:
            # Perturb and continue
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
                continue
            else:
                continue
        
        # Try full update (t=1)
        a_new = b_norm
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        
        if M_new < M - 1e-10:
            a = a_new
            M = M_new
            no_improve = 0
            obj = 2.0 * n * M
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        else:
            # Try partial updates
            for t in [0.5, 0.25, 0.1, 0.05]:
                a_try = (1-t)*a + t*b_norm
                conv_try = fast_convolve(a_try)
                if conv_try.max() < M - 1e-10:
                    a = a_try
                    M = conv_try.max()
                    no_improve = 0
                    obj = 2.0 * n * M
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                    break
            else:
                no_improve += 1
        
        # Periodic perturbation
        if no_improve > 200:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
    
    return best_a, best_obj / (2*n)  # Return normalized M


def optimize_smooth_max(a0, time_limit, beta_init=10.0, beta_max=500.0):
    """
    Gradient descent on smooth max approximation:
    f(a) = log(sum_k exp(beta * conv_k(a,a)))
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    beta = beta_init
    lr = 0.05
    momentum = 0.9
    v = np.zeros(n)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax weights
        shifted = beta * (conv - M)
        w = np.exp(shifted)
        w_sum = w.sum()
        w /= w_sum
        
        # Compute gradient of smooth max
        grad = np.zeros(n)
        active = np.where(w > 1e-4)[0]
        for k in active:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n-1, ki)
            idx = np.arange(j_start, j_end+1)
            grad[idx] += 2.0 * w[ki] * a[ki - idx]
        
        # Center gradient (to maintain sum constraint)
        grad = grad - grad.mean()
        
        # Update with momentum
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v * 0.1)
        
        a = a_new
        conv = fast_convolve(a)
        M = conv.max()
        obj = 2.0 * n * M
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Schedule
        beta = min(beta_max, beta * 1.005)
        lr *= 0.9999
        if lr < 0.001:
            lr = 0.01
            momentum = 0.8
            v *= 0.5
    
    return best_a, best_obj / (2*n)


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2*np.pi*x/(n-1)))
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Gaussian centered
    g = np.exp(-0.5*((x - n/2)/(n*0.25))**2)
    g /= g.sum()
    inits['gauss_center'] = g
    
    # Edge-concentrated
    edge = np.minimum(x, n-1-x)
    e = np.exp(-edge/(n*0.1))
    e /= e.sum()
    inits['edge'] = e
    
    # Sine wave
    sine = np.sin(np.pi*x/n)**2 + 0.01
    sine /= sine.sum()
    inits['sine'] = sine
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n)*0.5)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
    
    return inits
```

```python
def solve_lp_adaptive(a, M, top_k=50):
    """Solve LP with only top-k constraints, then verify."""
    n = len(a)
    conv_ab = lambda b: fast_convolve_cross(a, b)  # Need cross-convolution
    
    # Start with top-k constraints
    # First compute a*self conv to find which lags are largest
    # Actually we need conv(a, b) but we don't know b yet
    # Use the self-conv of a to identify which lags to constrain
    
    self_conv = fast_convolve(a)
    top_indices = np.argpartition(self_conv, -top_k)[-top_k:]
    
    # Solve LP with only these constraints
    A = np.zeros((len(top_indices), n))
    for ii, k in enumerate(top_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            A[ii, j] = a[k-j]
    
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    
    b = res.x
    
    # Verify all constraints
    # conv(a, b) at all lags
    ab_conv = np.fft.irfft(np.fft.rfft(a, 4*n) * np.fft.rfft(b, 4*n), 4*n)[:2*n-1]
    max_violation = ab_conv.max()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
smooth-max gradient descent with annealing, and multi-phase search.
Core idea: LP finds optimal redistribution directions that respect the
convolution bound; smooth-max provides a differentiable surrogate for
fine-grained local optimization. Multi-start with diverse initializations
and periodic perturbations escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix C where C[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        C[k, j_start:j_end + 1] = a[k - j_start:k - j_end + 1][::-1]
    return C


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)_k <= M for all k."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_partial(a, M, top_k=100):
    """Solve LP using only top-k constraints for speed."""
    n = len(a)
    self_conv = fast_convolve(a)
    top_idx = np.argpartition(self_conv, -top_k)[-top_k:]
    C_full = build_conv_matrix(a)
    C = C_full[top_idx, :]
    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        b_norm = b / b_sum
        diff = np.linalg.norm(b_norm - a)

        if diff < 1e-7:
            no_improve += 1
            if no_improve > perturb_every:
                noise = rng.standard_normal(n) * perturb_scale
                a = project_simplex(a + noise)
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
                perturb_scale = max(0.001, perturb_scale * 0.5)
            continue

        # Try full replacement
        conv_new = fast_convolve(b_norm)
        M_new = conv_new.max()

        if M_new < M - 1e-12:
            a = b_norm
            M = M_new
            no_improve = 0
            if M < best_M:
                best_M = M
                best_a = a.copy()
        else:
            # Partial updates
            accepted = False
            for t in [0.7, 0.5, 0.3, 0.1]:
                a_try = (1.0 - t) * a + t * b_norm
                conv_try = fast_convolve(a_try)
                if conv_try.max() < M - 1e-12:
                    a = a_try
                    M = conv_try.max()
                    no_improve = 0
                    accepted = True
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                    break
            if not accepted:
                no_improve += 1
                if no_improve > 200:
                    noise = rng.standard_normal(n) * perturb_scale
                    a = project_simplex(a + noise)
                    conv = fast_convolve(a)
                    M = conv.max()
                    no_improve = 0

    return best_a, best_M


def compute_smooth_grad(a, softmax_w):
    """Compute gradient of smooth max: grad[i] = 2*sum_j a[j]*softmax[i+j]."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(a, softmax_w[i:i + n])
    return grad


def optimize_smooth_max(a0, time_limit, beta_start=20.0, beta_end=500.0, lr_start=0.08):
    """Gradient descent on smooth max approximation with annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_M = M
    best_a = a.copy()

    t0 = time.monotonic()
    beta = beta_start
    lr = lr_start
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Softmax weights
        shifted = beta * (conv - M)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Gradient
        grad = compute_smooth_grad(a, w)
        # Center to maintain sum constraint
        grad = grad - grad.mean()

        # Momentum update
        v = momentum * v - lr * grad
        a = project_simplex(a + v * 0.05)

        conv = fast_convolve(a)
        M = conv.max()

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Schedule
        progress = (time.monotonic() - t0) / time_limit
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (1.0 - 0.8 * progress)
        momentum = 0.85 + 0.1 * progress

        if no_improve > 300:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            v *= 0.3
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0

    return best_a, best_M


def optimize_subgrad_topk(a0, time_limit, top_k=30, lr_init=0.02, momentum=0.9):
    """Subgradient descent using top-k convolution peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_M = M
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1

        peaks = np.argpartition(conv, -top_k)[-top_k:]

        sg = np.zeros(n)
        for k_int in peaks:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += 2.0 * a[k - idx]

        sg = sg - sg.mean()
        v = momentum * v - lr * sg
        a = project_simplex(a + v * 0.1)

        conv = fast_convolve(a)
        M = conv.max()

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.5
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            lr = max(0.005, lr_init * 0.5)

    return best_a, best_M


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Cosine window
    cos_w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Edge-concentrated
    edge = np.minimum(x, n - 1 - x)
    e = np.exp(-edge / (0.15 * n))
    e = e / e.sum()
    inits['edge'] = e

    # Sine squared
    sine = np.sin(np.pi * x / n) ** 2 + 0.05
    sine /= sine.sum()
    inits['sine'] = sine

    # Bimodal
    bi = np.zeros(n)
    half = n // 2
    bi[:half // 2] = 1.0
    bi[-(half // 2):] = 1.0
    mid = n // 4
    bi[mid:mid + n // 2] += 0.5
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Chirp-based (real part shifted to non-negative)
    chirp = np.cos(np.pi * x * x / n)
    chirp_shifted = np.abs(chirp) + 0.1
    chirp_shifted /= chirp_shifted.sum()
    inits['chirp'] = chirp_shifted

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0.0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_ds'] = pp / ps2
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_up'] = pp / ps2

    return inits


def combine_directions(a1, a2, t):
    """Blend two normalized sequences."""
    return (1.0 - t) * a1 + t * a2


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining LP direction finding,
    smooth-max gradient descent, and subgradient methods.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = max(10.0, float(budget_s))
    deadline = start_time + total_budget - 30.0

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: len={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Helper to track best
    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a_clean = np.maximum(a, 0.0)
        sc = a_clean.sum()
        if sc < 1e-10:
            return
        score = float(evaluate_sequence(a_clean.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_clean.copy()
            print(f"  ** NEW BEST: {score:.6f} (n={len(a_clean)})")
        return score

    # ============ Phase 1: Explore lengths and initializations (350s) ============
    print("=== Phase 1: Exploration ===")
    t0 = time.monotonic()
    phase1_end = t0 + 350.0

    lengths_to_try = [192, 256, 320, 384]
    results = []

    for n in lengths_to_try:
        if time.monotonic() > phase1_end - 20:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = [nm for nm in ['parent', 'parent_ds', 'parent_up',
                                    'cosine', 'sine', 'chirp', 'random',
                                    'gauss_center', 'edge'] if nm in inits]
        if len(init_names) > 5:
            init_names = init_names[:5]

        time_per = max(8.0, (phase1_end - time.monotonic() - 10) /
                       max(1, (len(lengths_to_try) - lengths_to_try.index(n)) * len(init_names)))

        for name in init_names:
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(time_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 3:
                continue

            a0 = inits[name]
            # Quick LP optimization
            a_opt, m_opt = optimize_lp_iter(a0, time_limit=t_alloc, use_partial=(n > 192))
            try_update(a_opt)
            norm_obj = 2.0 * n * m_opt
            results.append((norm_obj, n, name, a_opt.copy()))

    results.sort(key=lambda x: x[0])
    for r in results[:5]:
        print(f"  n={r[1]}, {r[2]}, obj={r[0]:.6f}")
    print(f"Phase 1 best: {best_score:.6f}")

    # ============ Phase 2: Deep LP optimization (1000s) ============
    print("=== Phase 2: Deep LP Optimization ===")
    t0 = time.monotonic()
    phase2_end = t0 + 1000.0

    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Build candidate list
    candidates = []
    candidates.append(best_sequence.copy())

    # Add top scan results
    for obj, n, name, a in results[:4]:
        candidates.append(a.copy())

    # Add parent at 256
    if parent is not None:
        for target_n in [256, 320]:
            p = np.maximum(parent, 0.0)
            if len(p) >= target_n:
                pp = p[:target_n]
            else:
                pp = np.interp(np.linspace(0, 1, target_n),
                               np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                candidates.append(pp / ps)

    # Remove near-duplicates
    unique = [candidates[0]]
    for c in candidates[1:]:
        is_dup = False
        for u in unique:
            if len(c) == len(u) and np.linalg.norm(c - u) < 0.005:
                is_dup = True
                break
        if not is_dup:
            unique.append(c)

    time_per_cand = (phase2_end - time.monotonic() - 20) / max(1, len(unique))
    print(f"  {len(unique)} candidates, ~{time_per_cand:.0f}s each")

    for i, a0 in enumerate(unique):
        if time.monotonic() > phase2_end - 15:
            break
        t_alloc = min(time_per_cand * 1.3, phase2_end - time.monotonic() - 10)
        if t_alloc < 8:
            continue

        n_cand = len(a0)
        print(f"  Candidate {i+1}/{len(unique)}: n={n_cand}, t={t_alloc:.0f}s")

        # Run LP optimization
        a_opt, m_opt = optimize_lp_iter(
            a0, time_limit=t_alloc * 0.7,
            use_partial=(n_cand > 192), top_k=150)
        try_update(a_opt)

        # Follow up with smooth max if time remains
        t_remain = phase2_end - time.monotonic() - 10
        if t_remain > 10:
            a_sm, m_sm = optimize_smooth_max(a_opt, time_limit=min(t_remain * 0.6, 60))
            try_update(a_sm)
            if m_sm < m_opt:
                a_opt = a_sm
                m_opt = m_sm

    # ============ Phase 3: Alternation + fine-tuning (250s) ============
    print("=== Phase 3: Alternation ===")
    t0 = time.monotonic()
    phase3_end = t0 + 250.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a_curr = best_sequence / best_sequence.sum()

        # Alternate LP and smooth-max
        alt_count = 0
        while time.monotonic() < phase3_end - 10:
            t_seg = min(40.0, (phase3_end - time.monotonic() - 5) / 2)
            if t_seg < 5:
                break
            alt_count += 1

            # LP step
            a_lp, m_lp = optimize_lp_iter(
                a_curr, time_limit=t_seg, use_partial=(n_best > 192), top_k=200)
            if m_lp < fast_convolve(a_curr).max():
                a_curr = a_lp
            try_update(a_curr)

            # Smooth max step
            t_sm = min(30.0, phase3_end - time.monotonic() - 10)
            if t_sm < 5:
                break
            a_sm, m_sm = optimize_smooth_max(
                a_curr, time_limit=t_sm,
                beta_start=50.0, beta_end=400.0, lr_start=0.03)
            try_update(a_sm)
            a_curr = a_sm

        # Random restarts from best
        n_restart = len(best_sequence)
        for _ in range(4):
            if time.monotonic() > phase3_end - 10:
                break
            t_rem = phase3_end - time.monotonic() - 5
            if t_rem < 5:
                break
            noise = rng.standard_normal(n_restart) * 0.008
            a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)
            a_opt, _ = optimize_lp_iter(a_pert, time_limit=min(t_rem / 2, 30),
                                        use_partial=(n_restart > 192))
            try_update(a_opt)

    # ============ Phase 4: Final polish (120s) ============
    print("=== Phase 4: Polish ===")
    t0 = time.monotonic()
    phase4_end = t0 + 120.0

    if best_sequence is not None:
        n_fin = len(best_sequence)
        a_fin = best_sequence / best_sequence.sum()

        # Very fine subgradient
        t_sub = min(40.0, phase4_end - time.monotonic() - 10)
        if t_sub > 5:
            a_sg, m_sg = optimize_subgrad_topk(
                a_fin, time_limit=t_sub, top_k=n_fin // 8, lr_init=0.005, momentum=0.95)
            try_update(a_sg)

        # Fine LP
        t_lp = min(60.0, phase4_end - time.monotonic() - 5)
        if t_lp > 5:
            a_fine, m_fine = optimize_lp_iter(
                a_fin, time_limit=t_lp, use_partial=False)
            try_update(a_fine)

        # Try slightly different lengths via interpolation
        for target_n in [n_fin - 16, n_fin + 16, n_fin + 32]:
            if target_n < 64 or target_n > 512:
                continue
            if time.monotonic() > phase4_end - 5:
                break
            a_interp = np.interp(np.linspace(0, 1, target_n),
                                 np.linspace(0, 1, n_fin), a_fin)
            a_interp = np.maximum(a_interp, 0)
            a_interp = a_interp / a_interp.sum()
            t_interp = min(20.0, phase4_end - time.monotonic() - 5)
            if t_interp > 3:
                a_io, _ = optimize_lp_iter(a_interp, time_limit=t_interp, use_partial=True)
                try_update(a_io)

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
