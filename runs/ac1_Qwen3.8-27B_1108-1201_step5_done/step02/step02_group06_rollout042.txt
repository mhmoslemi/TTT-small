**Algorithm:**

```
1. Initialize a (e.g., from provided sequence or random)
2. Set theta = log(a / sum(a)) (softmax parameterization)
3. For each iteration:

   b. Compute softmax weights w = softmax(conv/tau)
   c. Compute h[i] = 2 * cross_corr(w_padded, a)[i] using FFT
   d. Compute v = w * conv
   e. Compute Q[i] = 2 * cross_corr(v_padded, a)[i] using FFT
   f. Compute P = sum(w * conv)
   g. Compute dP_da[i] = h[i] + (Q[i] - P*h[i])/tau
   h. Compute dF_da[i] = 2n/sum(a)^2 * (dP_da[i] - 2*P*a[i]/sum(a))
   i. Convert to theta gradient: dF_dtheta[k] = dF_da[k]*a[k] - (a[k]/S)*sum_i(dF_da[i]*a[i])
   j. Take Adam step
   k. Update tau (anneal)
4. Return best a found
```

```python
"""
Fast gradient-based optimization for minimizing normalized peak autocorrelation.
Uses analytical FFT-based gradients with softmax parameterization, adaptive tau annealing,
multiple initialization strategies, and peak-aware local search.
"""

import numpy as np
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_gradient(a, tau):
    """Compute gradient of soft-peak objective using FFT/correlation."""
    n = len(a)
    conv = fft_convolve(a, a)
    # Softmax weights
    shifted = (conv - np.max(conv)) / tau
    w = np.exp(shifted)
    w /= np.sum(w)
    P = np.dot(w, conv)  # soft peak value
    
    # h[i] = 2 * sum_j a[j] * w[i+j] for i=0..n-1
    # This is np.correlate(w, a, mode='valid') * 2
    h = 2.0 * np.correlate(w, a, mode='valid')
    
    # v = w * conv (element-wise)
    v = w * conv
    # Q[i] = 2 * sum_j a[j] * v[i+j] for i=0..n-1
    Q = 2.0 * np.correlate(v, a, mode='valid')
    
    # dP/da[i] = h[i] + (Q[i] - P * h[i]) / tau
    dP_da = h + (Q - P * h) / tau
    
    # dF/da[i] where F = 2n * P / sum(a)^2
    sum_a = np.sum(a)
    dF_da = 2.0 * n / sum_a**2 * dP_da - 4.0 * n * P * a / sum_a**3
    
    return dF_da

def gradient_descent(f_init, time_limit, tau_init=None, tau_min=None, verbose=False):
    """Adam gradient descent with softmax parameterization and tau annealing."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / np.sum(f_safe))
    
    if tau_init is None:
        tau_init = 5.0 / n
    if tau_min is None:
        tau_min = 0.1 / n
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 0.5
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Anneal tau exponentially
        tau = tau_init * (tau_min / tau_init) ** elapsed_frac
        
        # Decode a from theta
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S
        
        # Compute gradient
        dF_da = compute_gradient(a, tau)
        
        # Convert to theta gradient (softmax parameterization)
        # dF/dtheta[k] = dF/da[k] * a[k] - (a[k]/S) * sum_i dF/da[i] * a[i]
        energy_term = np.dot(dF_da, a)
        dF_dtheta = dF_da * a - (a / S) * energy_term
        
        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * dF_dtheta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (dF_dtheta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        
        # Adaptive learning rate with decay
        lr = base_lr / np.sqrt(1.0 + 0.005 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Evaluate
        t_shifted = theta - np.max(theta)
        a_new = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
        score = local_score(a_new)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        if verbose and step_count % 500 == 0:
            print(f"    GD step={step_count}, score={score:.6f}, best={best_score:.6f}, tau={tau:.6f}")
    
    # Decode best
    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

def peak_aware_sa(f_init, time_limit, rng):
    """Simulated annealing with peak-aware mutations."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.005
    cooling = 0.9999
    iter_count = 0
    best_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        # Batch of iterations (for speed)
        batch_size = max(1, min(50, int((time_limit - (time.monotonic() - t0)) * 10)))
        
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            candidate = f.copy()
            r = rng.random()
            
            if r < 0.4:
                # Peak-directed mutation: reduce top contributors to peak
                conv = fft_convolve(f, f)
                peak_idx = int(np.argmax(conv))
                # Find which pairs contribute to the peak
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = f[i] * f[j]
                # Reduce top contributors
                n_reduce = max(2, n // 15)
                top_idx = np.argsort(contribs)[-n_reduce:]
                candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_idx))
                # Increase some others
                other_idx = np.setdiff1d(np.arange(n), top_idx)
                n_inc = min(4, len(other_idx))
                if n_inc > 0:
                    inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
            elif r < 0.6:
                # Multi-peak reduction: target top K peaks
                conv = fft_convolve(f, f)
                top_k_peaks = np.argsort(conv)[-3:]
                for pk in top_k_peaks:
                    pk = int(pk)
                    contribs = np.zeros(n)
                    for i in range(n):
                        j = pk - i
                        if 0 <= j < n:
                            contribs[i] = f[i] * f[j]
                    n_reduce = max(1, n // 30)
                    top_idx = np.argsort(contribs)[-n_reduce:]
                    candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=top_idx))
            elif r < 0.75:
                # Block mutation
                width = max(1, int(rng.integers(1, max(2, n // 8))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
            elif r < 0.85:
                # Global scale perturbation
                candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
            else:
                # Leap: larger random perturbation
                candidate = f * np.exp(rng.normal(0.0, 0.05, size=n))
                candidate = np.clip(candidate, 0.01, None)
            
            candidate = np.clip(candidate, 0.01, None)
            cand_sum = np.sum(candidate)
            candidate = candidate * (sum_f / cand_sum)
            
            new_score = local_score(candidate)
            iter_count += 1
            
            if new_score < current_score:
                f = candidate
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_f = candidate.copy()
                    best_iter = iter_count
            elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
                f = candidate
                current_score = new_score
            
            temperature *= cooling
        
        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, T={temperature:.6f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_score

def make_init_chirp(n, rng):

```python
"""
Fast gradient-based optimization for minimizing normalized peak autocorrelation.
Uses analytical FFT-based gradients (O(n^2) via correlation) with softmax parameterization,
adaptive tau annealing, multiple diverse initializations, and peak-aware simulated annealing
for fine-tuning. Key improvement over previous LP-based approach: orders of magnitude more
gradient steps are feasible due to much cheaper per-iteration cost.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_gradient(a, tau):
    """
    Compute analytical gradient of F(a) = 2n * softpeak(conv(a,a)) / sum(a)^2.
    Uses the identity: dc_k/da_i = 2*a[k-i] if 0<=k-i<n else 0.
    h[i] = 2*sum_j a[j]*w[i+j] via np.correlate.
    Q[i] = 2*sum_j a[j]*v[i+j] where v = w*conv.
    dP/da_i = h[i]*(1 - P/tau) + Q[i]/tau
    dF/da_i = 2n/sum^2 * dP/da_i - 4n*P*a_i/sum^3
    """
    n = len(a)
    conv = fft_convolve(a, a)
    # Softmax weights (shifted for numerical stability)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    w = np.exp(shifted)
    w /= np.sum(w)
    P = np.dot(w, conv)  # soft-peak value

    # h[i] = 2 * sum_j a[j] * w[i+j], i=0..n-1
    h = 2.0 * np.correlate(w, a, mode='valid')
    # v = w * conv element-wise
    v = w * conv
    # Q[i] = 2 * sum_j a[j] * v[i+j], i=0..n-1
    Q = 2.0 * np.correlate(v, a, mode='valid')

    # dP/da_i = h[i] + (Q[i] - P*h[i]) / tau
    dP_da = h + (Q - P * h) / tau

    # dF/da_i
    sum_a = np.sum(a)
    dF_da = 2.0 * n / (sum_a ** 2) * dP_da - 4.0 * n * P * a / (sum_a ** 3)

    return dF_da


def gradient_descent(f_init, time_limit, tau_init=None, tau_min=None, base_lr=0.5,
                     beta1=0.85, beta2=0.999, verbose=False):
    """Adam gradient descent with softmax parameterization and tau annealing."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / np.sum(f_safe))

    if tau_init is None:
        tau_init = 8.0 / n
    if tau_min is None:
        tau_min = 0.05 / n

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    stagnant = 0
    stagnant_threshold = 2000

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential tau annealing
        tau = tau_init * (tau_min / tau_init) ** (elapsed_frac ** 0.7)

        # Decode a from theta (softmax)
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S

        # Compute gradient
        dF_da = compute_gradient(a, tau)

        # Convert to theta gradient: dF/dtheta_k = a_k * (dF/da_k - mean_weighted)
        weighted_sum = np.dot(dF_da, a) / S
        dF_dtheta = a * (dF_da - weighted_sum)

        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * dF_dtheta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (dF_dtheta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)

        # Learning rate with warmup and decay
        warmup = min(1.0, step_count / 50.0)
        lr = base_lr * warmup / np.sqrt(1.0 + 0.003 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + 1e-8)

        # Evaluate
        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_new = e_new / np.sum(e_new) * S
        score = local_score(a_new)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            stagnant = 0
        else:
            stagnant += 1

        # If stuck, perturb theta to escape
        if stagnant > stagnant_threshold:
            theta += 0.02 * np.random.standard_normal(n)
            stagnant = 0
            stagnant_threshold = max(500, stagnant_threshold // 2)

        if verbose and step_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD step={step_count}, score={score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.5f}, lr={lr:.4f}, elapsed={elapsed:.0f}s")

    # Decode best
    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

        elif r < 0.90:
            # Global subtle perturbation
            candidate *= np.exp(rng.normal(0.0, 0.006, size=n))
        else:
            # Leap: larger random perturbation
            candidate = f * np.exp(rng.normal(0.0, 0.05, size=n))

        candidate = np.clip(candidate, 0.001, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score

        temperature *= cooling

        if iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            remaining = time_limit - elapsed
            print(f"    SA iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, "
                  f"T={temperature:.5f}, elapsed={elapsed:.0f}s, remaining={remaining:.0f}s")

    return best_f, best_score


def make_init_chirp(n, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = rng.uniform(0.2, 0.6)
    f = rng.uniform(1.0, n / 4.0)
    phi = rng.uniform(0, 2 * np.pi)
    a = 1.0 + alpha * np.cos(2 * np.pi * f * t / n + phi)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_multitone(n, rng):
    """Multi-tone cosine initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_harm = int(rng.integers(3, 8))
    for _ in range(n_harm):
        k = int(rng.integers(1, max(2, n // 3)))
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_random(n, rng, spread=0.3):
    """Random initialization around 1."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_periodic(n, rng):
    """Periodic pattern initialization with random period and asymmetry."""
    period = int(rng.integers(2, max(3, n // 8)))
    a = np.ones(n)
    for i in range(n):
        a[i] = 1.0 + rng.uniform(-0.3, 0.3) * np.cos(2 * np.pi * i / period)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
