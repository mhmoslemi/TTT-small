import numpy as np
import scipy.optimize as optimize

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_normalized(a):
    """Compute 2*n*max(conv) for normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def proj_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.max(np.where(1 - u * idx > 0)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, active_mask):
    """Compute subgradient of max_k conv(a,a)[k] at active positions."""
    n = len(a)
    sg = np.zeros(n)
    active_idx = np.where(active_mask)[0]
    w = np.ones(len(active_idx)) / len(active_idx)
    for idx_pos in range(len(active_idx)):
        k = int(active_idx[idx_pos])
        wv = w[idx_pos]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            j = np.arange(lo, hi + 1)
            sg[j] += wv * a[k - j]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if hi2 >= lo2:
            j2 = np.arange(lo2, hi2 + 1)
            sg[j2] += wv * a[j2 - k]
    return sg


def fista_optimize(a0, time_limit, epsilon=0.02, restart_every=30):
    """FISTA-style projected subgradient with momentum and restarting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = obj_normalized(a)
    t0 = time.monotonic()
    step = 0.5 / n
    k_mom = 1
    y = a.copy()
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(y)
        max_conv = conv.max()
        current_obj = 2.0 * n * max_conv

        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        if patience > 200:
            break

        thresh = max_conv * (1.0 - epsilon)
        active_mask = conv >= thresh
        if not active_mask.any():
            active_mask[int(np.argmax(conv))] = True

        sg = compute_subgradient(y, conv, active_mask)
        grad_norm = np.linalg.norm(sg)
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        step = min(1.0 / (n * max_conv + 1e-10), max(step, 1e-8))
        a_new = y - step * sg
        a_new = proj_simplex(a_new)

        # FISTA momentum with restarting
        a_check = a_new
        obj_check = obj_normalized(a_check)
        if obj_check > current_obj:
            a_new = y  # Restart (no momentum)
            k_mom = 1
        else:
            k_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * k_mom * k_mom))
            a_new = a_new + ((k_mom - 1.0) / k_new) * (a_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            k_mom = k_new

        y = a_new
        if iter_count % restart_every == 0:
            k_mom = 1

    return best_a, best_obj


def make_lp_constraint_matrix(n, active_rows):
    """Build sparse constraint matrix for cross-conv at given rows."""
    rows_list = []
    cols_list = []
    vals_list = []
    for i in range(n):
        for k in active_rows:
            k = int(k)
            j = k - i
            if 0 <= j < n:
                rows_list.append(k)
                cols_list.append(i)
                vals_list.append(1.0)
    return (np.array(rows_list, dtype=np.int32),
            np.array(cols_list, dtype=np.int32),
            np.array(vals_list, dtype=np.float64))


def solve_lp_direction(f, M, active_rows, n):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for k in active_rows."""
        if len(active_rows) == 0:
            return None
    rows = np.array(active_rows, dtype=np.int32)
    rows_list = []
    cols_list = []
    vals_list = []
    f_arr = f
    for i in range(n):
        for idx_k in range(len(rows)):
            k = int(rows[idx_k])
            j = k - i
            if 0 <= j < n:
                rows_list.append(idx_k)
                cols_list.append(i)
                vals_list.append(f_arr[j])
    if len(rows_list) == 0:
        return None
    A = sparse.csc_matrix((vals_list, (rows_list, cols_list)), shape=(len(rows), n))
    c = -np.ones(n)
    b = np.full(len(rows), M)
    bounds = [(0, None)] * n
    try:
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                               options={'disp': False, 'time_limit': 10})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-12:
                return g / gs
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, max_constraint_reduction=0.3):
    """LP-guided alternating with constraint generation and backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    max_conv = conv.max()
    best_obj = 2.0 * n * max_conv
    best_a = a.copy()
    t0 = time.monotonic()
    t_blend = 0.5
    patience = 0
    iter_count = 0
    active_rows = list(range(2 * n - 1))

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Reduce active constraints if too many
        if len(active_rows) > n * max_constraint_reduction:
            conv_check = fast_convolve(a)
            thresh = conv_check.max() * 0.95
            new_active = list(np.where(conv_check >= thresh)[0])
            if len(new_active) >= max(10, n // 8):
                active_rows = new_active

        g = solve_lp_direction(a, max_conv, active_rows, n)
        if g is None:
            # Fallback: try with all constraints
            active_rows = list(range(2 * n - 1))
            g = solve_lp_direction(a, max_conv, active_rows, n)
            if g is None:
                break

        if np.max(np.abs(g - a)) < 1e-12:
            break

        # Backtracking line search
        improved = False
        for t_try in [t_blend, 0.7 * t_blend, 0.5 * t_blend, 0.3 * t_blend, 0.1 * t_blend]:
            a_new = (1.0 - t_try) * a + t_try * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-12:
                continue
            a_new = a_new / sn
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                a = a_new
                max_conv = conv_new.max()
                best_obj = obj_new
                best_a = a.copy()
                patience = 0
                t_blend = min(1.0, t_blend * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t_blend *= 0.85
            if patience > 40:
                break

        if iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iter_count}, obj={best_obj:.8f}, patience={patience}, elapsed={elapsed:.1f}s")

def mirror_polish(a0, time_limit, t_param=50.0, eta0=0.3):
    """Multiplicative weights / mirror descent polishing on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a = a / a.sum()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Compute importance weights from convolution
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            break
        w = w / w_sum

        # Compute subgradient weighted by convolution importance
        sg = np.zeros(n)
        threshold = w.max() * 0.001
        active = np.where(w > threshold)[0]
        for k in active:
            k = int(k)
            wv = w[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                j = np.arange(lo, hi + 1)
                sg[j] += wv * a[k - j]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if hi2 >= lo2:
                j2 = np.arange(lo2, hi2 + 1)
                sg[j2] += wv * a[j2 - k]

        # Mirror descent update (multiplicative weights)
        sg_centered = sg - sg.mean()
        log_f = -eta * sg_centered
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / s

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.75
                if eta < 0.005:
                    t_param = min(t_param * 1.5, 800.0)
                    eta = 0.2
            if patience > 150:
                break

    return best_a, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    # Uniform
    inits['uni'] = np.ones(n) / n

    # Gaussian (narrow, medium, wide)
    for sf in [0.1, 0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    # U-shaped (high at edges)
    u = 0.5 + 1.5 * ((x - mid) / mid) ** 2
    u = np.maximum(u, 0.1)
    u /= u.sum()
    inits['u_shape'] = u

    # V-shaped (low at edges, high in middle) - opposite of U
    v = 1.0 - 0.7 * ((x - mid) / mid) ** 2
    v = np.maximum(v, 0.05)
    v /= v.sum()
    inits['v_shape'] = v

    # Bimodal (two peaks)
    w1 = max(1, n // 6)
    bi = np.exp(-0.5 * ((x - w1) / (w1 * 0.5)) ** 2) + np.exp(-0.5 * ((x - n + w1) / (w1 * 0.5)) ** 2)
    bi = np.maximum(bi, 0.01)
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine bump
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb = np.maximum(cb, 0.01)
    cb /= cb.sum()
    inits['cos'] = cb

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Random Dirichlet (concentrated)
    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd_sparse'] = rr

    # Random Dirichlet (diffuse)
    rr2 = rng.dirichlet(np.ones(n) * 5.0)
    inits['rnd_diffuse'] = rr2

    # Triangle
    tri = np.maximum(1.0 - np.abs(x - mid) / (mid + 0.1), 0.05)
    tri /= tri.sum()
    inits['tri'] = tri

    # Power law decay from center
    pw = np.exp(-0.3 * np.abs(x - mid) / mid)
    pw /= pw.sum()
    inits['pw'] = pw

    return inits


def combined_optimize(a0, time_limit):
    """Run FISTA then LP then mirror polish in sequence."""
    a = a0.copy()
    best_a = a0.copy()
    best_obj = obj_normalized(a0)

    # Phase A: FISTA (40% of time)
    t_fista = time_limit * 0.4
    a_f, obj_f = fista_optimize(a, time_limit=t_fista, epsilon=0.02)
    if obj_f < best_obj:
        best_obj = obj_f
        best_a = a_f
    a = a_f

    # Phase B: LP (35% of time)
    if time_limit - t_fista > 10:
        t_lp = time_limit * 0.35
        a_l, obj_l = lp_optimize(a, time_limit=t_lp)
        if obj_l < best_obj:
            best_obj = obj_l
            best_a = a_l
        a = a_l

    # Phase C: Mirror polish (25% of time)
    if time_limit - t_fista - time_limit * 0.35 > 5:
        t_mirror = time_limit * 0.25
        a_m, obj_m = mirror_polish(a, time_limit=t_mirror, t_param=40.0, eta0=0.25)
        if obj_m < best_obj:
            best_obj = obj_m
            best_a = a_m

    return best_a, best_obj

    and mirror descent polishing across multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    if parent is not None:
        ps = parent.sum()
        if ps > 0:
            pn = parent / ps
            sc = float(evaluate_sequence(pn.tolist()))
            if np.isfinite(sc):
                best_score = sc
                best_seq = pn.copy()
                print(f"Parent sequence: n={len(pn)}, score={sc:.6f}")

    # Define lengths to try
    lengths = [256, 384, 512, 640, 768, 1024]

    # Filter lengths: skip if parent length is very different (to save time)
    if parent is not None:
        pn = len(parent)
        # Prioritize lengths close to parent
        lengths = sorted(lengths, key=lambda x: abs(x - pn))

    all_results = []
    n_inits = 6  # Number of initializations to try per length

    total_budget = deadline - time.monotonic()
    n_trials = len(lengths) * n_inits
    time_per_trial = total_budget * 0.6 / max(1, n_trials)  # 60% for exploration

    print(f"Exploration phase: {len(lengths)} lengths x {n_inits} inits = {n_trials} trials")
    print(f"Time per trial: ~{time_per_trial:.1f}s")

    trial_idx = 0
    for n in lengths:
        if time.monotonic() > deadline - total_budget * 0.4:
            break

        inits = gen_inits(n, rng)
        # Add parent as initialization if available
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size),
                               np.maximum(parent, 0.0))
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()

        # Select top n_inits initializations (prioritizing diverse types)
        init_names = list(inits.keys())
        # Always include 'uni', 'g35', 'u_shape', 'bimodal', 'parent' if available
        priority = ['parent', 'uni', 'g35', 'u_shape', 'bimodal', 'rnd_sparse', 'cos', 'rnd_diffuse', 'v_shape', 'pw']
        selected_names = []
        for pn_name in priority:
            if pn_name in init_names and pn_name not in selected_names:
                selected_names.append(pn_name)
            if len(selected_names) >= n_inits:
                break
        for nm in init_names:
            if nm not in selected_names and len(selected_names) < n_inits:
                selected_names.append(nm)

        for name in selected_names:
            trial_idx += 1
            if time.monotonic() > deadline - total_budget * 0.4:
                break
            ta = min(time_per_trial, deadline - time.monotonic() - total_budget * 0.4)
            if ta < 5:
                break

            a0 = inits[name].astype(np.float64)
            a_opt, obj_opt = combined_optimize(a0, time_limit=ta)

            # Evaluate with the actual evaluator
            seq_list = [float(x) for x in a_opt]
            score = float(evaluate_sequence(seq_list))

            if np.isfinite(score):
                all_results.append((score, n, name, a_opt))
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  [{trial_idx:2d}] n={n:4d} {name:12s} score={score:.6f} (obj_norm={obj_opt:.6f})")

    print(f"Exploration done. Best score: {best_score:.6f}")
    print(f"Found {len(all_results)} valid results.")

    # Phase 2: Deep optimization on top results
    all_results.sort(key=lambda x: x[0])
    top_k = min(5, len(all_results))
    if top_k > 0:
        remaining = deadline - time.monotonic() - 30
        if remaining > 60:
            print(f"\nPhase 2: Deep optimization on top {top_k}, {remaining:.0f}s remaining")
            t_each = remaining / top_k

            for idx in range(top_k):
                if time.monotonic() > deadline - 25:
                    break
                sc, n, name, f = all_results[idx]
                ta = min(t_each, deadline - time.monotonic() - 25)
                if ta < 10:
                    break

                print(f"  Deep [{idx}] n={n} {name} current_score={sc:.6f}, time={ta:.0f}s")

                # Long LP run
                f_lp, obj_lp = lp_optimize(f, time_limit=ta * 0.5)
                sc_lp = float(evaluate_sequence(f_lp.tolist()))
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = f_lp.copy()
                    print(f"    LP: score={sc_lp:.6f}")
                f = f_lp

                # Mirror polish with multiple params
                for tp, eta in [(30.0, 0.3), (80.0, 0.15), (200.0, 0.08)]:
                    if time.monotonic() > deadline - 20:
                        break
                    tr = min(20.0, deadline - time.monotonic() - 20)
                    if tr < 3:
                        break
                    f_m, _ = mirror_polish(f, time_limit=tr, t_param=tp, eta0=eta)
                    sc_m = float(evaluate_sequence(f_m.tolist()))
                    if np.isfinite(sc_m) and sc_m < best_score:
                        best_score = sc_m
                        best_seq = f_m.copy()
                        print(f"    Mirror(tp={tp:.0f}): score={sc_m:.6f}")
                    f = f_m

                # Quick FISTA restart
                if time.monotonic() < deadline - 15:
                    tr = min(15.0, deadline - time.monotonic() - 15)
                    f_f, _ = fista_optimize(f, time_limit=tr, epsilon=0.01)
                    sc_f = float(evaluate_sequence(f_f.tolist()))
                    if np.isfinite(sc_f) and sc_f < best_score:
                        best_score = sc_f
                        best_seq = f_f.copy()
                        print(f"    FISTA: score={sc_f:.6f}")
                    f = f_f

    # Phase 3: Final aggressive polish on the overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nPhase 3: Final polish. Best so far: {best_score:.6f}")
        tr_total = deadline - time.monotonic() - 10
        n_polish = 3
        tr_each = tr_total / n_polish

        for i in range(n_polish):
            if time.monotonic() > deadline - 8:
                break
            tr = min(tr_each, deadline - time.monotonic() - 8)
            if tr < 3:
                break

            # Alternate between LP and mirror
            if i % 2 == 0:
                f_p, _ = lp_optimize(best_seq, time_limit=tr * 0.7)
                f_p2, _ = mirror_polish(f_p, time_limit=tr * 0.3, t_param=60.0, eta0=0.2)
                f_final = f_p2
            else:
                f_final, _ = mirror_polish(best_seq, time_limit=tr, t_param=100.0, eta0=0.1)

            sc_p = float(evaluate_sequence(f_final.tolist()))
            if np.isfinite(sc_p) and sc_p < best_score:
                best_score = sc_p
                best_seq = f_final.copy()
                print(f"  Polish[{i}]: score={sc_p:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback to uniform: score={best_score:.6f}")

    # Final validation
    final_list = [float(max(0.0, x)) for x in best_seq]
    final_score = float(evaluate_sequence(final_list))
    if not np.isfinite(final_score) or final_score > best_score:
        # Use the best_seq as-is
        final_list = [float(max(0.0, x)) for x in best_seq]
        final_score = float(evaluate_sequence(final_list))

    print(f"\nFINAL: n={len(final_list)}, score={final_score:.6f}")
    return final_list
