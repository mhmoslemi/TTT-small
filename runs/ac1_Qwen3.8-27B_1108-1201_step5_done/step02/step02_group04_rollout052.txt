    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_score_normalized(a):
    """Compute score assuming sum(a)=1. Returns 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_lp_indices(n):
    """Build row/col indices for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        rows_list.extend(range(i, i + n))
        cols_list.extend([i] * n)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_cross_lp(f, M, rows, cols, n, timeout=8):
    """Solve max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    vals = np.tile(f, n).astype(np.float64)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    A_ub = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-15:
            return g
    return None


def compute_subgradient(a, w, n):
    """Compute weighted subgradient phi[p] = sum_k w[k] * d(conv_k)/d(a_p).
    Uses cross-correlation structure for efficiency."""
    # conv_k(a) = sum_{i+j=k} a[i]*a[j]
    # d(conv_k)/d(a_p) = 2*a[k-p] if 0<=k-p<n and 2p!=k, plus a[p] if 2p=k
    # phi[p] = 2*sum_k w[k]*a[k-p]*[0<=k-p<n] - a[p]*w[2p]*[2p<2n-1]
    # First term: cross-correlation: cross[p] = sum_j a[j]*w[p+j]
    j_idx = np.arange(n, dtype=np.float64)
    p_idx = np.arange(n, dtype=np.int32)
    # M[p,j] = w[p+j], shape (n, n)
    M = w[p_idx[:, None] + j_idx[None, :].astype(np.int32)]
    cross = M.dot(a)
    phi = 2.0 * cross
    # Diagonal correction
    two_p = 2 * p_idx
    valid = two_p < (2 * n - 1)
    phi[valid] -= a[valid] * w[two_p[valid]]
    return phi


def mirror_descent_step(a, T, eta, n):
    """One mirror descent step with softmax weighting."""
    conv = fast_convolve(a)
    # Softmax weights with temperature T
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    # Compute subgradient
    phi = compute_subgradient(a, w, n)
    # Center the subgradient
    phi_c = phi - phi.mean()
    # Multiplicative update
    log_factor = -eta * phi_c
    log_factor -= log_factor.max()
    factor = np.exp(log_factor)
    a_new = a * factor
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    return a_new, conv, w


def mirror_optimize(a0, time_limit, T0=0.5, T_min=0.05, eta0=0.5, eta_min=0.01):
    """Mirror descent with adaptive temperature and learning rate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    T = T0
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        a_new, conv, w = mirror_descent_step(a, T, eta, n)
        obj = 2.0 * n * conv.max()
        n_iters += 1

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            patience = 0
            # Slightly increase eta on success
            eta = min(eta * 1.02, 2.0)
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.8
                T *= 0.92
                if eta < eta_min:
                    eta = eta_min
                    T = max(T * 0.7, T_min)
            if patience > 500:
                break

        a = a_new

        # Periodic report
        if n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter {n_iters}, obj={best_obj:.8f}, T={T:.4f}, eta={eta:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def lp_alternate(a0, rows, cols, time_limit, max_iters=200):
    """LP-based alternating optimization with backtracking line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    step_size = 0.5
    patience = 0
    iters = 0

    while time.monotonic() - t0 < time_limit and iters < max_iters:
        iters += 1
        g = solve_cross_lp(a, M, rows, cols, n, timeout=min(8, max(2, (time_limit - (time.monotonic() - t0)) * 0.3)))
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - a)) < 1e-12:
            break

        # Backtracking line search
        improved = False
        for alpha in [step_size, 0.7 * step_size, 0.5 * step_size, 0.3 * step_size,
                      0.1 * step_size, 0.05 * step_size, 0.01 * step_size]:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new /= s_new
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()

            if obj_new < best_obj - 1e-13:
                a = a_new
                conv = conv_new
                M = conv.max()
                best_obj = obj_new
                best_a = a.copy()
                patience = 0
                step_size = min(1.0, step_size * 1.1)
                improved = True
                break

        if not improved:
            patience += 1
            step_size *= 0.75
            if patience > 40:
                break

        if iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iters}, obj={best_obj:.8f}, step={step_size:.3f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n

    # Gaussian with varying widths
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    # Beta-like distributions
    for p, q in [(2, 1), (1, 2), (3, 1), (1, 3), (2, 2), (0.5, 0.5), (0.3, 0.3)]:
        t = x / (n - 1) if n > 1 else np.array([0.5])
        b = t ** (p - 1) * (1 - t) ** (q - 1)
        b = np.maximum(b, 1e-10)
        b /= b.sum()
        key = f'b{p}_{q}'
        if key not in inits:
            inits[key] = b

    # Cosine power
    for pw in [1, 2, 4, 8]:
        c = np.abs(np.cos(np.pi * x / n)) ** pw
        c = np.maximum(c, 1e-10)
        c /= c.sum()
        inits[f'cos{pw}'] = c

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Bi-peak
    w = max(1, n // 5)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    # Triangular
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['tri'] = tri / tri.sum()

    # Random Dirichlet
    for alpha in [0.1, 0.5, 2.0]:
        rr = rng.dirichlet(np.ones(n) * alpha)
        inits[f'rnd{alpha}'] = rr

    # Chirp-like (modulated)
    freq = rng.uniform(0.1, 0.5)
    chirp = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
    chirp = np.maximum(chirp, 0.01)
    chirp /= chirp.sum()
    inits['chirp'] = chirp

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: multi-phase optimization combining LP and mirror descent."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Loaded parent sequence: n={p.size}, sum={p.sum():.4f}")
            ps = p.sum()
            if ps > 0.01:
                pscore = float(evaluate_sequence(p.tolist()))
                if np.isfinite(pscore):
                    best_score = pscore
                    best_seq = p.copy()
                    print(f"Parent score: {pscore:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Define lengths to try
    lengths = [512, 768, 1024]
    if parent is not None:
        # Also try the parent's length
        pl = parent.size
        if pl not in lengths:
            lengths.insert(0, pl)
        # Also try nearby
        for dl in [pl - 128, pl + 128]:
            if 128 <= dl <= 2048 and dl not in lengths:
                lengths.append(dl)
        lengths = sorted(set(lengths))[:5]

    print(f"Lengths to try: {lengths}")

    all_results = []

    # === Phase 1: Quick LP scan ===
    print("\n=== Phase 1: Quick LP scan ===")
    p1_time_each = 25  # seconds per config

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break

        inits = gen_inits(n, rng)

        # Add parent if available
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()

        rows, cols = build_lp_indices(n)

        # Limit number of inits to keep time manageable
        max_inits = 8
        init_items = list(inits.items())
        if len(init_items) > max_inits:
            # Keep most diverse ones
            priority = ['parent', 'uniform', 'g20', 'b0.5_0.5', 'cos4', 'bi', 'rnd0.1', 'chirp']
            init_items = sorted(init_items, key=lambda x: -1 if x[0] in priority else 0)
            init_items = init_items[:max_inits]

        for name, a0 in init_items:
            if time.monotonic() > deadline - 180:
                break
            t_alloc = min(p1_time_each, max(5, deadline - time.monotonic() - 180))
            if t_alloc < 3:
                break

            f_opt, obj_opt = lp_alternate(a0, rows, cols, time_limit=t_alloc, max_iters=50)
            score = float(evaluate_sequence(f_opt.tolist()))

            if np.isfinite(score):
                all_results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"  n={n:4d} {name:10s} score={score:.6f} (LP phase1)")

    print(f"Phase 1 done. Best: {best_score:.6f}, {len(all_results)} results")

    # === Phase 2: Deep optimization ===
    print("\n=== Phase 2: Deep optimization ===")
    all_results.sort(key=lambda x: x[0])
    top_k = min(6, len(all_results))
    top = all_results[:top_k]

    remaining = deadline - time.monotonic() - 120
    if remaining > 60 and top:
        t_each = remaining / top_k
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_each, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break

            print(f"\n  Deep [{idx}] n={n} {name} score={sc:.6f}, alloc={t_alloc:.0f}s")

            # Extended LP
            if n in [ar[1] for ar in all_results]:
                rows, cols = build_lp_indices(n)
            else:
                rows, cols = build_lp_indices(n)

            f2, obj2 = lp_alternate(f, rows, cols, time_limit=t_alloc * 0.4, max_iters=100)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = f2.copy()
                print(f"    Deep LP: score={sc2:.6f}")

            # Mirror descent - multiple temperature settings
            remaining2 = deadline - time.monotonic() - 90
            if remaining2 > 20:
                t_md = remaining2 * 0.35
                for T0, eta0 in [(0.3, 0.3), (0.1, 0.5), (1.0, 0.1)]:
                    if time.monotonic() > deadline - 80:
                        break
                    tr = min(t_md / 3.0, deadline - time.monotonic() - 80)
                    if tr < 5:
                        break
                    f3, obj3 = mirror_optimize(f2, time_limit=tr, T0=T0, eta0=eta0)
                    sc3 = float(evaluate_sequence(f3.tolist()))
                    if np.isfinite(sc3) and sc3 < best_score:
                        best_score = sc3
                        best_seq = f3.copy()
                        print(f"    MD T={T0}: score={sc3:.6f}")
                        f2 = f3  # chain improvements

            # Final LP polish
            remaining3 = deadline - time.monotonic() - 80
            if remaining3 > 15:
                f4, obj4 = lp_alternate(f2, rows, cols, time_limit=min(30, remaining3 * 0.5), max_iters=50)
                sc4 = float(evaluate_sequence(f4.tolist()))
                if np.isfinite(sc4) and sc4 < best_score:
                    best_score = sc4
                    best_seq = f4.copy()
                    print(f"    Final LP: score={sc4:.6f}")

    # === Phase 3: Final polish on best ===
    print("\n=== Phase 3: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = len(best_seq)
        rows, cols = build_lp_indices(n_best)
        a_norm = best_seq.copy()
        sn = a_norm.sum()
        if sn > 1e-10:
            a_norm /= sn

        remaining = deadline - time.monotonic() - 10

        # Extended mirror descent with fine-tuning
        if remaining > 30:
            tr = min(remaining * 0.4, 120)
            f5, obj5 = mirror_optimize(a_norm, time_limit=tr, T0=0.05, eta0=0.1)
            sc5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(sc5) and sc5 < best_score:
                best_score = sc5
                best_seq = f5.copy()
                print(f"  Fine MD: score={sc5:.6f}")

        # LP polish
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            f6, obj6 = lp_alternate(a_norm, rows, cols, time_limit=min(remaining * 0.4, 60), max_iters=50)
            sc6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(sc6) and sc6 < best_score:
                best_score = sc6
                best_seq = f6.copy()
                print(f"  Final LP: score={sc6:.6f}")

        # One more round of mirror with different params
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            f7, obj7 = mirror_optimize(a_norm, time_limit=min(remaining * 0.5, 30), T0=0.02, eta0=0.05)
            sc7 = float(evaluate_sequence(f7.tolist()))
            if np.isfinite(sc7) and sc7 < best_score:
                best_score = sc7
                best_seq = f7.copy()
                print(f"  Ultra-fine MD: score={sc7:.6f}")

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final evaluation
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL RESULT: n={len(best_seq)}, score={best_score:.8f}")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
