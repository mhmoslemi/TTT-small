"""
Optimized peak autocorrelation minimization for non-negative sequences.
Uses vectorized subgradient descent with momentum, active-set LP refinement,
and multi-start strategy across sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def compute_subgradient(f, top_k=10):
    """Compute subgradient of max_k conv(f,f)[k]."""
    n = len(f)
    c = fft_convolve(f, f)
    M = c.max()
    top_idx = np.argsort(c)[-top_k:]
    
    g = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        g[lo:hi+1] += 2.0 * f[ki-hi:ki-lo+1][::-1]
    g /= float(top_k)
    # Center to preserve sum
    g -= np.mean(g)
    return g, M, c


def subgradient_descent(f_init, time_limit, top_k=10, verbose=True):
    """Subgradient descent with momentum on normalized sequence."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    target_sum = 1.0  # normalize to sum=1
    f = f / float(np.sum(f))
    
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.005
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        g, M, c = compute_subgradient(f, top_k)
        
        # Adaptive learning rate
        # Use Polyak-like step: lr = (M - target) / (||g||^2 + eps)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-12:
            lr_adaptive = min(0.05, 0.01 / g_norm)
        else:
            lr_adaptive = lr
        
        momentum = beta * momentum + (1 - beta) * g
        f = f - lr_adaptive * momentum
        f = np.clip(f, 0.0, None)
        
        # Renormalize
        S = float(np.sum(f))
        if S < 1e-10:
            f = np.ones(n) / n
            S = 1.0
        f = f / S
        
        it += 1
        if it % 50 == 0:
            s = score_fast(f)
            if s < best_score - 1e-10:
                best_score = s
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if verbose and it % 200 == 0:
                print(f"    [sgd {time.monotonic()-t0:.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
            
            # Adapt step size
            if no_improve >= 10:
                lr *= 0.8
                momentum *= 0.5
                no_improve = 0
        
        # Periodic perturbation to escape local minima
        if it % 500 == 0 and it > 0:
            rng = np.random.default_rng(42 + it)
            f += 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.0, None)
            S = float(np.sum(f))
            f = f / S
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score

Consider the problem as a generalized eigenvalue problem. We want to minimize max_k c[k] s.t. sum(a) = 1, a >= 0.

At the optimum (interior), the gradient of the objective equals the gradient of the constraint:
(1/|A|) sum_{k in A} grad c[k] = lambda * 1

This means: at the optimum, the "average gradient" from the active peaks is constant across all coordinates. This is a fixed-point condition!

We can use this for a fixed-point iteration:
1. Start with some a.
2. Compute active set A (top peaks).
3. Compute g = (1/|A|) sum_{k in A} grad c[k].
4. Update: a_new = a - alpha * (g - mean(g)) [to maintain sum constraint].
5. Project onto a >= 0.
6. Repeat.

This is essentially the subgradient method I already described, but the fixed-point interpretation gives insight into convergence.

At the optimum, g should be constant (all components equal). So we can use ||g - mean(g)|| as a convergence criterion.

"""
Peak autocorrelation minimization for non-negative sequences.
Uses vectorized subgradient descent with momentum, active-set LP refinement,
coordinate descent polish, and multi-start across sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, f"a".encode() if False else a)
    return float(2.0 * n * np.max(c) / (S * S))


def compute_subgrad(f, top_k):
    """Compute subgradient of max_k conv(f,f)[k] and the convolution."""
    n = len(f)
    c = fft_convolve(f, f)
    M = float(c.max())
    top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        g[lo:hi + 1] += 2.0 * f[ki - hi:ki - lo + 1][::-1]
    g /= float(top_k)
    g -= np.mean(g)
    return g, M, c


def normalize(f):
    """Normalize to sum=1, clip to non-negative."""
    f = np.clip(f, 0.0, None)
    S = float(np.sum(f))
    if S < 1e-12:
        return np.ones(len(f)) / len(f)
    return f / S


def sgd_optimize(f_init, time_limit, top_k=8, verbose=False):
    """Subgradient descent with momentum and adaptive step size."""
    n = len(f_init)
    f = normalize(f_init.astype(np.float64))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.003
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        g, M, c = compute_subgrad(f, top_k)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            step = np.min([lr, 0.02 / (g_norm + 1e-10)])
        else:
            step = lr

        momentum = beta * momentum + (1.0 - beta) * g
        f = f - step * momentum
        f = normalize(f)
        it += 1

        if it % 40 == 0:
            s = score_fast(f)
            if s < best_score - 1e-12:
                best_score = s
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
            if verbose and it % 400 == 0:
                print(f"    [sgd {time.monotonic()-t0:.1f}s] it={it} score={s:.6f} best={best_score:.6f} stall={stall}")
            if stall >= 15:
                lr *= 0.7
                momentum *= 0.3
                stall = 0
                if lr < 1e-6:
                    break

        if it % 600 == 0:
            rng = np.random.default_rng(7 + it * 31)
            f = f * np.exp(rng.normal(0, 0.015, n))
            f = normalize(f)

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def build_A(f, active_only=False, top_k=8):
    """Build matrix A where A@b = conv(f,b). Optionally only active rows."""
    n = len(f)
    if active_only:
        c = fft_convolve(f, f)
        M = c.max()
        idx = np.where(c >= M * 0.95)[0]
        rows = idx
    else:
        rows = range(2 * n - 1)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for j in range(n):
        for i, r in enumerate(rows):
            r = int(r)
            if 0 <= r - j < n:
                A[i, j] = f[r - j]
    return A, [int(r) for r in rows]


def solve_lp_active(f, M, top_k=8, timeout=4.0):
    """Solve LP with only active constraints. Returns b or None."""
    n = len(f)
    c = fft_convolve(f, f)
    M = float(c.max())
    thresh = M * 0.9
    rows_idx = np.where(c >= thresh)[0]
    if len(rows_idx) == 0:
        rows_idx = np.argsort(c)[-top_k:]

    A = np.zeros((len(rows_idx), n), dtype=np.float64)
    for j in range(n):
        for i, r in enumerate(rows_idx.astype(int)):
            if 0 <= r - j < n:
                A[i, j] = f[r - j]

    c_obj = -np.ones(n)
    b_ub = np.full(len(rows_idx), M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, top_k=8, verbose=False):
    """LP-based iterative optimization with active set."""
    n = len(f_init)
    f = normalize(f_init.astype(np.float64))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    step_sizes = [0.3, 0.2, 0.1, 0.05, 0.02]

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(c.max())

        b = solve_lp_active(f, M, top_k, timeout=4.0)
        if b is None:
            break
        b = np.clip(b, 0, None)
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (float(np.sum(f)) / Sb)

        best_t = 0.1
        best_s = np.inf
        for t_try in step_sizes:
            fc = (1.0 - t_try) * f + t_try * b
            fc = normalize(fc)
            s_cand = score_fast(fc)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = normalize((1.0 - best_t) * f + best_t * b)
        it += 1

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 10 == 0:
            print(f"    [lp {time.monotonic()-t0:.1f}s] it={it} score={best_s:.6f} best={best_score:.6f}")

        if stall >= 20:
            rng = np.random.default_rng(99 + it * 7919)
            f = f * np.exp(rng.normal(0, 0.02, n))
            f = normalize(f)
            stall = 0
            if it > 100:
                break

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def coordinate_descent(f_init, time_limit, verbose=False):
    """Coordinate-wise local search: adjust one element at a time."""
    n = len(f_init)
    f = normalize(f_init.astype(np.float64))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    alpha = 0.05

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break

        # Pick a random set of coordinates to try
        n_try = max(5, n // 8)
        coords = rng.choice(n, size=n_try, replace=False)
        improved = False

        for j in coords:
            if time.monotonic() - t0 >= time_limit - 2.0:
                break
            for scale in [1 + alpha, 1 - alpha, 1 + alpha/2, 1 - alpha/2]:
                fc = f.copy()
                fc[j] *= scale
                if fc[j] < 0:
                    fc[j] = 0
                fc = normalize(fc)
                s = score_fast(fc)
                if s < best_score - 1e-12:
                    best_score = s
                    best_f = fc.copy()
                    f = fc.copy()
                    improved = True
                    break

        if not improved:
            # Try a random perturbation
            f = f * np.exp(rng.normal(0, alpha * 0.5, n))
            f = normalize(f)
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()

        if verbose:
            pass

    return best_f, best_score


def make_init(n, init_type, rng):
    """Create initial sequence."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    elif init_type == 'ramp':
        a = np.linspace(0.3, 1.0, n)
    elif init_type == 'ramp_rev':
        a = np.linspace(1.0, 0.3, n)
    elif init_type == 'bimodal':
        center = n / 2.0
        a = np.exp(-((t - center) ** 2) / (2 * (n / 4) ** 2)) + 0.3
    elif init_type == 'random':
        a = 0.5 + rng.random(n)
    else:
        a = np.ones(n)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def interp_sequence(f, n_new):
    """Interpolate sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, f)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Subgradient descent (multi-start, multiple n)
    Phase 2: LP refinement on top candidates
    Phase 3: Coordinate descent polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(provided) > 0 and np.sum(provided) > 0.01:
            s = score_fast(provided)
            print(f"Provided: n={len(provided)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = normalize(provided).copy()

    print(f"\n=== Phase 1: Subgradient multi-start ===")
    print(f"Initial best: {best_score:.6f}")

    # Define tasks: (n, init_type)
    tasks = []
    if provided is not None:
        tasks.append((len(provided), 'provided'))
    for n in [128, 192, 256, 320, 384, 512]:
        tasks.append((n, 'uniform'))
        tasks.append((n, 'hann'))
        tasks.append((n, 'ramp'))
        tasks.append((n, 'random'))
    if provided is not None:
        for n in [384, 512]:
            if n != len(provided):
                tasks.append((n, 'provided_interp'))

    # Time allocation: smaller n gets more time
    total_weight = sum(1.0 / (n / 100.0) for n, _ in tasks)
    phase1_time = min(900.0, deadline - t_start - 500.0)

    candidates = []  # (score, n, f_normalized)

    for idx, (n, init_type) in enumerate(tasks):
        if time.monotonic() > deadline - 500:
            break

        t_alloc = max(15.0, phase1_time * (100.0 / n) / total_weight)
        t_alloc = min(t_alloc, 120.0)

        if init_type == 'provided' and provided is not None:
            f0 = provided.copy()
            if len(f0) != n:
                f0 = interp_sequence(f0, n)
        elif init_type == 'provided_interp' and provided is not None:

        f_best, s_best = sgd_optimize(f0, t_alloc, top_k=8, verbose=False)

        if s_best < best_score - 1e-8:

            best_seq = f_best.copy()
            print(f"  n={n:4d} init={init_type:15s} score={s_best:.6f} *** NEW BEST ***")

            print(f"  n={n:4d} init={init_type:15s} score={s_best:.6f}")

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nPhase 1 complete. Best: {best_score:.6f}")
    for i, (s, n, _) in enumerate(candidates[:5]):

    print(f"\n=== Phase 2: LP Refinement ===")

    # Take top candidates for LP refinement
    top_cands = candidates[:min(4, len(candidates))]
    # Remove duplicates with same n
    seen_n = set()
    top_cands_filtered = []
    for s, n, f in top_cands:
        if n not in seen_n:
            top_cands_filtered.append((s, n, f))
            seen_n.add(n)
    if len(top_cands_filtered) == 0:

    for i, (s, n, f) in enumerate(top_cands_filtered):
        if time.monotonic() > deadline - 100:

        t_alloc = min(150.0, deadline - time.monotonic() - 80.0)

        f_lp, s_lp = lp_optimize(f, t_alloc, top_k=8, verbose=True)

        if s_lp < best_score - 1e-10:
            best_score = s_lp
            best_seq = f_lp.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")
        # Update candidate list
