def fft_convolve(a):
    """Compute autocorrelation (full convolution) of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

    """Fast score computation."""

    S = np.sum(a)

    return 2.0 * n * np.max(conv) / (S * S)
