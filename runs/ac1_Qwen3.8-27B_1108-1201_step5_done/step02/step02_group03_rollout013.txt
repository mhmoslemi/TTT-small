import numpy as np

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def compute_conv(a):
    """Full linear convolution a * a."""
    return np.convolve(a, a)


def gradient_softmax(a, tau):

    Returns (gradient, conv_values).

    w = np.exp(tau * c_shifted)

    if w_sum < 1e-300:

def gradient_hard(a):
    """
    Subgradient of max(conv(a,a)) at the argmax.
    grad[m] = 2*a[k*-m] if 0 <= k*-m < n, else 0.

    k_star = int(np.argmax(c))
    grad = np.zeros(n)

    hi = min(n - 1, k_star)
    for m in range(lo, hi + 1):

    F = mag_target * np.exp(1j * phase)
    # Enforce Hermitian symmetry for real signal
    for k in range(1, N // 2 + 1):
        F[N - k] = np.conj(F[k])
    if N % 2 == 1:
        F[N // 2] = np.abs(F[N // 2])

    best_a = None
    best_score = float('inf')

    for it in range(n_iters):
        # Time domain
        a_full = np.fft.ifft(F).real
        # Enforce support: keep first n, zero rest
        a_padded = np.zeros(N)
        a_padded[:n] = np.maximum(a_full[:n], 0.0)

        # Check quality periodically
        if it % 20 == 0 or it == n_iters - 1:
            a_trial = a_padded[:n].copy()
            if a_trial.sum() > 1e-6:
                a_trial = a_trial * (sum_target / a_trial.sum())
                s = compute_obj(a_trial)
                if s < best_score:
                    best_score = s
                    best_a = a_trial.copy()

        # Frequency domain
        F_new = np.fft.fft(a_padded)
        # Reset magnitude to target
        angles = np.angle(F_new)
        mags = np.abs(F_new)
        # Avoid zeroing out: use a floor
        mags = np.maximum(mags, mag_target * 0.1)
        F = mag_target * np.exp(1j * angles)
        # Enforce Hermitian symmetry
        for k in range(1, N // 2 + 1):
            F[N - k] = np.conj(F[k])
        if N % 2 == 1:
            F[N // 2] = np.abs(F[N // 2])

    if best_a is None:
        a_full = np.fft.ifft(F).real
        a = np.maximum(a_full[:n], 0)
        if a.sum() > 1e-6:
            best_a = a * (sum_target / a.sum())
        else:
            best_a = np.ones(n) * (sum_target / n)

    return best_a


def spectral_init_multi(n, sum_target, rng, n_phase_trials=5, n_iters=150):
    """Try multiple random phase initializations, return the best."""
    best_a = None
    best_score = float('inf')
    for _ in range(n_phase_trials):
        a = spectral_init(n, sum_target, rng, n_iters=n_iters)
        s = compute_obj(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    return best_a, best_score


def optimized_gradient_descent(a, deadline, verbose=False):
    """
    Main gradient descent loop with soft-max annealing and momentum.
    Returns (best_sequence, best_score).
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    momentum = np.zeros(n)
    momentum_decay = 0.85
    lr = 0.3
    tau = 2.0
    no_improve = 0

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # Anneal tau: start soft, get sharper
        tau = min(2.0 + it * 0.05, 80.0)
        lr = max(0.3 * (0.999 ** it), 0.01)
        momentum_decay = min(0.85 + it * 0.0001, 0.95)

        grad, c = gradient_softmax(a, tau)

        # Project out component along 1 (to maintain sum)
        grad -= grad.mean()

        # Scale gradient for stable updates
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        grad = grad / grad_norm * np.linalg.norm(a)

        # Momentum update
        momentum = momentum_decay * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        a = a_new

        if it % 50 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-9:
                best_score = s
                best = a.copy()
                no_improve = 0
                if verbose:
                    print(f"      it={it}, score={best_score:.6f}, tau={tau:.1f}")
            else:
                no_improve += 1
                if no_improve > 20:
                    # Reset and try harder
                    momentum[:] = 0
                    no_improve = 0
                    # Small random perturbation
                    a = a + rng_perturb.normal(0, 0.05, n)
                    a = np.maximum(a, 0)
                    a = normalize(a, S_t)
                    no_improve = 0

                if no_improve > 60:
                    break

    return best, best_score


# Global rng for perturbations
rng_perturb = np.random.default_rng(12345)


def lp_refine(f, M, n_max=1024):
    """
    LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0.
    """
    n = len(f)
    if n > n_max:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix efficiently
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    # For each column j, nonzeros are in rows j..j+n-1
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_start = j
        m_end = min(j + n, 2 * n - 1)
        A[m_start:m_end, j] = f[:m_end - m_start]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-6:
                return g
    except Exception:
        pass
    return None


def lp_iteration(a, deadline, verbose=False):
    """
    LP-based iterative refinement.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)

    no_improve = 0
    it = 0
    while time.monotonic() < deadline and no_improve < 8:
        it += 1
        c = compute_conv(a)
        M = float(c.max())

        g = lp_refine(a, M * 1.0)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        g_norm = g * (S_t / g_sum)

        improved = False
        for t_blend in [0.6, 0.4, 0.25, 0.1, 0.05, 0.8]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                if verbose:
                    print(f"      LP it={it}, t={t_blend}, score={best_score:.6f}")
                break

        if not improved:
            no_improve += 1
            # Nudge with gradient to escape
            grad, _ = gradient_hard(a)
            grad -= grad.mean()
            a = a - 0.02 * grad
            a = np.maximum(a, 0)
            a = normalize(a, S_t)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
                no_improve = 0
        else:
            no_improve = 0

    return best, best_score


def coordinate_polish(a, deadline, max_rounds=5):
    """
    Simple coordinate descent: for each coordinate, try a few values.
    Only effective for n <= 512.
    """
    n = len(a)
    if n > 600:
        return a, compute_obj(a)

    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = compute_conv(a)
        k_star = int(np.argmax(c))

        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            # Candidate values based on structure
            j = k_star - i
            candidates = [0.0, orig * 0.5, orig * 1.5]
            if 0 <= j < n and j != i:
                candidates.append(orig * 0.2)
                candidates.append(orig * 0.8)

            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                a[i] = orig

            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s
                improved = True

        if not improved:
            break

    return a, best_score


def full_pipeline(a0, deadline, verbose=False):
    """
    Full optimization pipeline from one starting point.
    Phases: gradient descent -> LP -> coordinate polish.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    a = normalize(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    if verbose:
        print(f"    n={n}, init_score={score:.6f}")

    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best, best_score

    # Phase 1: Gradient descent (60% of time)
    phase1_deadline = time.monotonic() + remaining * 0.55
    a, s1 = optimized_gradient_descent(a, phase1_deadline, verbose)
    if s1 < best_score:
        best = a.copy()
        best_score = s1
    if verbose:
        print(f"    Phase1 (grad): {best_score:.6f}")

    # Phase 2: LP refinement (25% of time)
    remaining = deadline - time.monotonic()
    if remaining > 10 and n <= 1024:
        phase2_deadline = time.monotonic() + remaining * 0.4
        a, s2 = lp_iteration(best, phase2_deadline, verbose)
        if s2 < best_score:
            best = a.copy()
            best_score = s2
        if verbose:
            print(f"    Phase2 (LP): {best_score:.6f}")

    # Phase 3: Coordinate polish (15% of time)
    remaining = deadline - time.monotonic()
    if remaining > 5:
        phase3_deadline = time.monotonic() + remaining * 0.8
        a, s3 = coordinate_polish(best, phase3_deadline)
        if s3 < best_score:
            best = a.copy()
            best_score = s3
        if verbose:
            print(f"    Phase3 (coord): {best_score:.6f}")

    return best, best_score


def generate_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    starts = []
    S_t = np.sqrt(2.0 * n)
    i = np.arange(n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n) * (S_t / n))

    # Cosine tapers
    for freq in [1, 2, 3, 5]:
        starts.append(np.maximum(0.3 + 0.7 * np.cos(2 * np.pi * freq * i / n), 0.01))

    # Gaussian bumps
    for center_frac in [0.5, 0.3, 0.7]:
        for width_frac in [0.15, 0.25, 0.4]:
            bump = np.exp(-0.5 * ((i - n * center_frac) / (n * width_frac)) ** 2)
            starts.append(bump + 0.05)

    # Two bumps
    starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.1)) ** 2) +
                  np.exp(-0.5 * ((i - n * 0.75) / (n * 0.1)) ** 2) + 0.02)

    # Random
    for _ in range(3):
        starts.append(rng.random(n) * 0.5 + 0.25)

    # Sparse-ish
    a = np.zeros(n)
    a[::3] = rng.random(n // 3 + 1) * 0.5 + 0.1
    starts.append(a)

    # Linear ramps
    starts.append(np.linspace(0.1, 1.0, n))
    starts.append(np.abs(np.linspace(-1, 1, n)) + 0.1)

    # Sinusoidal absolute
    starts.append(np.abs(np.sin(np.pi * i / n)) + 0.05)
    starts.append(np.abs(np.sin(2 * np.pi * i * 3.0 / n)) + 0.05)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses spectral initialization, gradient descent,
    LP refinement, and coordinate descent from multiple diverse starting points.
    """
    global rng_perturb
    rng = np.random.default_rng(seed)
    rng_perturb = np.random.default_rng(seed + 999)

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Try multiple lengths
    lengths = [256, 384, 512]

    print(f"Starting optimization, budget={budget_s}s")

    # --- Stage 1: Spectral initialization for each length ---
    spectral_inits = {}
    for n in lengths:
        S_t = np.sqrt(2.0 * n)
        if time.monotonic() > deadline - 120:
            break
        print(f"\nGenerating spectral init for n={n}...")
        t_spec = time.monotonic()
        a_spec, s_spec = spectral_init_multi(n, S_t, rng, n_phase_trials=3, n_iters=100)
        print(f"  Spectral init: score={s_spec:.6f}, time={time.monotonic()-t_spec:.1f}s")
        spectral_inits[n] = a_spec

        if s_spec < best_score:
            best_score = s_spec
            best_seq = a_spec.copy()

    # --- Stage 2: Run full pipeline on best spectral inits + structured starts ---
    # Build list of (sequence, label) to try, prioritizing by initial score
    candidates = []

    for n in lengths:
        if n not in spectral_inits:
            continue
        a_spec = spectral_inits[n]
        candidates.append((a_spec, f"spectral_n{n}", compute_obj(a_spec)))

        # Also try structured starts for this n
        starts = generate_starts(rng, n)
        for idx, s in enumerate(starts[:6]):  # Limit to 6 structured starts per n
            s_norm = normalize(s)
            s_score = compute_obj(s_norm)
            candidates.append((s_norm, f"structured_{n}_{idx}", s_score))

    # Also try height_sequence_1 if available
    try:
        h1 = np.asarray(height_sequence_1, dtype=np.float64)
        if h1.size >= 8 and np.all(np.isfinite(h1)):
            for n in lengths:
                if n <= len(h1):
                    a_h1 = h1[:n].copy()
                    a_h1 = np.maximum(a_h1, 0)
                    if a_h1.sum() > 0:
                        a_h1 = normalize(a_h1)
                        s_h1 = compute_obj(a_h1)
                        candidates.append((a_h1, f"h1_n{n}", s_h1))
                else:
                    # Pad or truncate
                    if len(h1) > n:
                        a_h1 = h1[:n].copy()
                    else:
                        a_h1 = np.zeros(n)
                        a_h1[:len(h1)] = h1
                    a_h1 = np.maximum(a_h1, 0)
                    if a_h1.sum() > 0:
                        a_h1 = normalize(a_h1)
                        s_h1 = compute_obj(a_h1)
                        candidates.append((a_h1, f"h1_pad_n{n}", s_h1))
    except NameError:
        pass

    # Sort by initial score (best first)
    candidates.sort(key=lambda x: x[2])

    print(f"\nTotal candidates to try: {len(candidates)}")
    print(f"Top 5 by initial score:")
    for i, (a, label, s) in enumerate(candidates[:5]):
        print(f"  {i}: {label}, n={len(a)}, score={s:.6f}")

    # --- Stage 3: Optimize top candidates with time allocation ---
    # Give more time to better initial candidates
    n_candidates = len(candidates)
    time_elapsed = time.monotonic() - t_start
    time_remaining = deadline - t_start

    # Allocate time: top candidates get more time
    for idx, (a0, label, init_score) in enumerate(candidates):
        if time.monotonic() > deadline - 30:
            break

        remaining = deadline - time.monotonic()

        # Time allocation: better initial scores get more time
        if idx < 3:
            t_alloc = min(300.0, remaining * 0.3)
        elif idx < 8:
            t_alloc = min(180.0, remaining * 0.2)
        elif idx < 15:
            t_alloc = min(120.0, remaining * 0.15)
        else:
            t_alloc = min(60.0, remaining * 0.1)

        t_alloc = max(15.0, t_alloc)
        if remaining - t_alloc < 30:
            t_alloc = remaining - 30
            if t_alloc < 10:
                break

        print(f"\n  [{idx}] Optimizing {label} (n={len(a0)}, init={init_score:.6f}), t_alloc={t_alloc:.0f}s")
        seq, score = full_pipeline(a0, time.monotonic() + t_alloc, verbose=True)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [{idx}] NEW BEST: {best_score:.6f}")
        else:
            print(f"  [{idx}] score={score:.6f} (not better)")

        # If we've beaten the target significantly, do a focused polish
        if best_score < 1.51 and time.monotonic() < deadline - 60:
            # Quick additional polish on the best
            remaining = deadline - time.monotonic() - 10
            if remaining > 30:
                print(f"\n  Extra polish on best (score={best_score:.6f})...")
                seq2, score2 = full_pipeline(best_seq, time.monotonic() + remaining * 0.8, verbose=True)
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2.copy()
                    print(f"  Extra polish: {best_score:.6f}")

    # --- Stage 4: Final attempts with perturbation of best ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        print(f"\nFinal perturbation attempts on best (n={n_best}, score={best_score:.6f})...")
        remaining = deadline - time.monotonic() - 10

        for trial in range(5):
            if time.monotonic() > deadline - 15:
                break
            # Random perturbation
            sigma = 0.1 / (1 + trial * 0.5)
            perturbed = best_seq + rng.normal(0, sigma, n_best)
            perturbed = np.maximum(perturbed, 0)
            perturbed = normalize(perturbed)

            t_alloc = max(15.0, remaining / (5 - trial))
            if t_alloc > 120:
                t_alloc = 120

            seq, score = full_pipeline(perturbed, time.monotonic() + min(t_alloc, remaining - 15), verbose=False)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Perturb [{trial}]: NEW BEST {best_score:.6f}")
            else:
                print(f"  Perturb [{trial}]: {score:.6f}")

    # --- Final validation ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL ===")
        print(f"  n={len(best_seq)}")
        print(f"  Computed score: {final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"  evaluate_sequence score: {eval_score:.6f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
