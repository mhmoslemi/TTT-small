"""
Soft-max gradient optimization for minimizing peak autocorrelation of
non-negative step functions. Uses log-space parameterization (softmax)
to handle non-negativity and sum constraints, with adaptive temperature
annealing and Adam optimizer. Includes local search polish phase.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]

def compute_score(a):
    """Fast score computation (scale-invariant version, assumes sum(a)=1)."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def compute_score_full(a):
    """Full score computation matching evaluate_sequence formula."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def softmax_gradient_step(b, tau, a, c, w):
    """Compute gradient of soft-max objective w.r.t. b (log-space params)."""
    n = len(a)
    # G[i] = 2 * sum_{m=0}^{n-1} w[m+i] * a[m]
    # This is 2 * dot(w[i:i+n], a)
    G = np.zeros(n)
    for i in range(n):
        G[i] = 2.0 * np.dot(w[i:i+n], a)
    H = np.dot(a, G)
    grad_b = a * (G - H)
    return grad_b

def softmax_gradient_step_fast(b, tau, a, c, w):
    """Faster gradient using matrix operations."""
    n = len(a)
    # Build the matrix where M[i,m] = w[m+i]
    # Then G = 2 * M @ a
    # M is an n x n matrix, M[i,m] = w[m+i]
    # This is a Hankel-like matrix (actually a sub-block of the Toeplitz)
    # For efficiency, use the fact that M[i,:m] = w[i:i+n]
    # We can compute G[i] = 2 * np.dot(w[i:i+n], a) for all i
    
    # Actually, for n=512, the loop is fast enough. Let's just do it.
    # But let's try a vectorized approach:
    # G[i] = 2 * sum_m w[m+i]*a[m] = 2 * sum_m w[m]*a[m-i] (letting m' = m+i, a[m] = a[m'-i])
    # = 2 * cross_correlation(w, a)[i] ... hmm
    # 
    # Actually: G[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # Let's define v = w (length 2n-1) and u = a (length n)
    # G[i] = 2 * <v[i:i+n], u>
    # This is the cross-correlation of v and u at specific lags.
    # Specifically, it's the "valid" cross-correlation.
    
    # For n=512, let's just use the loop (it's fast with numpy dot)
    G = np.array([2.0 * np.dot(w[i:i+n], a) for i in range(n)])
    H = np.dot(a, G)
    return a * (G - H)

def softmax_optimize(b_init, time_limit, rng, verbose=True, tau_init=1.0, tau_min=0.01):
    """Optimize b using soft-max gradient descent with Adam."""
    t0 = time.monotonic()
    n = len(b_init)
    b = b_init.astype(np.float64).copy()
    
    # Adam optimizer
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    step = 0
    best_a = None
    best_score = np.inf
    tau = tau_init
    stall_count = 0
    
    # Compute initial
    a = 1.0 / np.exp(-b - np.max(-b))  # softmax(b)
    a = a / np.sum(a)
    c = fft_convolve(a, a)
    current_score = compute_score(a)  # scale-invariant
    if current_score < best_score:
        best_score = current_score
        best_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        a = 1.0 / np.exp(-b - np.max(-b))
        a = a / np.sum(a)
        c = fft_convolve(a, a)
        
        # Soft-max weights
        c_scaled = c / tau
        c_max = np.max(c_scaled)
        exp_c = np.exp(c_scaled - c_max)
        w = exp_c / np.sum(exp_c)
        
        # Gradient
        grad_b = softmax_gradient_step_fast(b, tau, a, c, w)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * grad_b ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Compute new score
        a_new = 1.0 / np.exp(-b - np.max(-b))
        a_new = a_new / np.sum(a_new)
        new_score = compute_score(a_new)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Anneal temperature
        if step % 100 == 0:
            tau = max(tau_min, tau * 0.95)
            lr = max(0.001, lr * 0.98)
        
        # If stalled, try larger steps
        if stall_count > 500:
            b = b + rng.normal(0, 0.01, n)
            stall_count = 0
            tau = max(tau, tau_init * 0.3)
        
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    step={step}, score={new_score:.6f}, best={best_score:.6f}, tau={tau:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score
