def solve_lp_relaxed(f, M, relax=1.0):
    """LP: maximize sum(g) s.t. conv(f,g)[k] <= M*relax for all k, g >= 0"""
    n = len(f)
    # Build constraint matrix: conv(f,g)[k] = sum_j f[max(0,k-n+1):k+1] * g[j]
    # More precisely: conv(f,g)[k] = sum_{j} f[k-j] * g[j] where 0<=k-j<n and 0<=j<n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M * relax)
    bounds = [(0.0, None)] * n
    try:

                                       options={'presolve': True, 'time_limit': 30})

            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def subgradient_step(f, lr, top_k=1):
    """Take a subgradient step to reduce the top-k peaks."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Find top-k peak positions
    top_pos = np.argsort(conv)[-top_k:]
    
    # Compute gradient of sum of top-k peaks
    grad = np.zeros(n)
    for k in top_pos:
        for j in range(n):
            idx1 = k - j
            if 0 <= idx1 < n:
                grad[j] += f[idx1]
            idx2 = j - k
            if 0 <= idx2 < n:
                grad[j] += f[idx2]
    
    # Normalize gradient
    gn = np.linalg.norm(grad)
    if gn < 1e-12:
        return f.copy()
    grad /= gn
    
    # Take step
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-12:
        return f.copy()
    return f_new / s

def line_search(f, direction, obj_fn, max_iter=30):
    """Golden section or bisection line search."""
    # Try to find best t in [0, 1]
    best_t = 0.0
    best_obj = obj_fn(f)
    
    # Try a few values
    for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 0.95]:
        a = f + t * direction
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        o = obj_fn(a)
        if o < best_obj:
            best_obj = o
            best_t = t
    
    return best_t, best_obj

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s * s)


def normalize(a):
    """Project onto simplex: a>=0, sum(a)=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    rho_star = np.max(rho[cond])
    theta = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the matrix A such that A @ g = conv(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def solve_lp_direction(f, M, time_limit_s=10.0):
    """LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
            options={'presolve': True, 'time_limit': time_limit_s}
        )
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        for j in range(n):
            idx1 = k - j
            if 0 <= idx1 < n:
                grad[j] += wk * f[idx1]
            idx2 = j - k
            if 0 <= idx2 < n:
                grad[j] += wk * f[idx2]
    return grad


def smooth_objective(f, tau):
    """Smoothed objective: 2n*tau*log(sum(exp(conv/tau)))/(sum(f))^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        return np.inf
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    log_sum = tau * (scaled.max() + np.log(np.exp(scaled).sum()))
    return 2.0 * n * log_sum / (s * s)


def subgradient_step(f, lr):
    """Subgradient step on the peak of the convolution."""
    n = len(f)
    conv = fast_convolve(f)
    k_star = int(np.argmax(conv))

    grad = np.zeros(n)
    for j in range(n):
        idx1 = k_star - j
        if 0 <= idx1 < n:
            grad[j] += f[idx1]
        idx2 = j - k_star
        if 0 <= idx2 < n:
            grad[j] += f[idx2]

    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return f.copy()
    grad /= gn

    f_new = f - lr * grad
    return project_simplex(f_new)


def line_search_obj(f, direction, max_iter=40):
    """Golden section line search for f + t*direction on [0,1]."""
    def eval_t(t):
        a = f + t * direction
        a = project_simplex(a)
        return compute_objective(a)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_t(c)
    fd = eval_t(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_t(d)
    t = (lo + hi) / 2.0
    return t, eval_t(t)


def line_search_smooth(f, direction, tau, max_iter=40):
    """Line search on smoothed objective."""
    def eval_t(t):
        a = f + t * direction
        a = project_simplex(a)
        return smooth_objective(a, tau)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_t(c)
    fd = eval_t(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_t(d)
    t = (lo + hi) / 2.0
    return t, eval_t(t)


def local_search(a, time_limit):
    """Coordinate descent: move mass from peak-contributing to safe positions."""
    n = len(a)
    a = normalize(a.copy())
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 500

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        thr_high = 0.5 * max_c
        thr_low = 0.1 * max_c
        danger = np.where(contrib > thr_high)[0]
        safe = np.where(contrib < thr_low)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        for di in danger[:20]:
            if found:
                break
            for si in safe[:20]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = compute_objective(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            x_orig = np.linspace(0, 1, p.size)
            x_new = np.linspace(0, 1, n)
            pc = np.interp(x_new, x_orig, p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_candidates(rng, parent):
    """Generate diverse initial candidates."""
    candidates = []

    if parent is not None:
        for n in [512, 1024, 2048]:
            p = get_parent(n)
            if p is not None:
                candidates.append((p, n, f'parent_{n}'))

    for n in [512, 1024]:
        for alpha in [0.05, 0.1, 0.5, 1.0]:
            r = rng.dirichlet(np.ones(n) * alpha)
            candidates.append((r, n, f'dirichlet_{n}_{alpha}'))

    # Cone/gaussian initializations
    for n in [512, 1024]:
        x = np.arange(n, dtype=np.float64)
        for sf in [0.15, 0.25, 0.4]:
            g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
            g = normalize(g)
            candidates.append((g, n, f'gauss_{n}_{sf}'))

    # Anti-edges (mass at edges)
    for n in [512, 1024]:
        edges = np.zeros(n)
        m = max(1, n // 3)
        edges[:m] = 1.0
        edges[-m:] = 1.0
        edges = normalize(edges)
        candidates.append((edges, n, f'edges_{n}'))

    return candidates


def optimize_sequence(f0, time_limit, use_lp=True, use_smooth=True, use_subgrad=True):
    """
    Multi-technique optimization of a single sequence.
    Alternates between LP, smooth gradient, and subgradient methods.
    """
    f = normalize(f0.copy().astype(np.float64))
    best_f = f.copy()
    best_obj = compute_objective(f)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 50
    step_count = 0

    # Temperature for smooth objective: start high, anneal
    tau_init = best_obj / (2.0 * len(f)) * 0.5
    tau = max(tau_init, 1e-4)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        conv = fast_convolve(f)
        M = conv.max()
        s = f.sum()
        obj = 2.0 * len(f) * M / (s * s)

        if obj < best_obj - 1e-10:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            break

        step_count += 1

        # Choose method based on step count (cycle through)
        method = step_count % 3

        if method == 0 and use_lp and remaining > 5.0:
            # LP-based direction
            lp_time = min(remaining * 0.3, 15.0)
            g = solve_lp_direction(f, M, time_limit_s=lp_time)
            if g is not None:
                direction = g - f
                t, obj_t = line_search_obj(f, direction)
                if obj_t < best_obj - 1e-10:
                    f = project_simplex(f + t * direction)
                    best_obj = obj_t
                    best_f = f.copy()
                    no_improve = 0
                # Anneal tau
                tau = max(tau * 0.95, 1e-5)

        elif method == 1 and use_smooth:
            # Smooth gradient descent
            grad = smooth_gradient(f, tau)
            gn = np.linalg.norm(grad)
            if gn > 1e-15:
                grad /= gn
                # Adaptive learning rate
                lr = 0.5 / (len(f) ** 0.5)
                direction = -grad
                t, obj_t = line_search_smooth(f, direction, tau)
                if obj_t < smooth_objective(f, tau) - 1e-8:
                    f = project_simplex(f + t * direction)
                    # Check actual objective
                    actual_obj = compute_objective(f)
                    if actual_obj < best_obj - 1e-10:
                        best_obj = actual_obj
                        best_f = f.copy()
                        no_improve = 0
                # Anneal tau
                tau = max(tau * 0.9, 1e-5)

        elif method == 2 and use_subgrad:
            # Subgradient step
            lr = 1.0 / (np.sqrt(step_count + 1) * np.sqrt(len(f)))
            f_new = subgradient_step(f, lr)
            obj_new = compute_objective(f_new)
            if obj_new < best_obj - 1e-10:
                f = f_new
                best_obj = obj_new
                best_f = f_new.copy()
                no_improve = 0

        # Periodic local search
        if step_count % 30 == 0 and remaining > 10.0:
            ls_time = min(remaining * 0.2, 15.0)
            a_ls, obj_ls = local_search(f, time_limit=ls_time)
            if obj_ls < best_obj - 1e-10:
                f = a_ls
                best_obj = obj_ls
                best_f = f.copy()
                no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization combining LP, smoothed gradient,
    subgradient, and local search across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget

    best_sequence = None
    best_score = float('inf')

    parent = get_parent(512)

    # Generate candidates
    candidates = make_candidates(rng, parent)
    print(f"Generated {len(candidates)} candidates")

    # Phase 1: Quick scan (25% of budget)
    phase1_time = total_budget * 0.25
    phase1_deadline = time.monotonic() + phase1_time
    print(f"Phase 1: Quick scan ({phase1_time:.0f}s budget)")

    # Sort: parent first, then shorter n
    def candidate_priority(item):
        a, n, name = item
        if 'parent' in name:
            return (0, n)
        return (1, n)

    candidates.sort(key=candidate_priority)

    scan_results = []
    for a0, n, name in candidates:
        if time.monotonic() > phase1_deadline:
            break
        remaining_p1 = phase1_deadline - time.monotonic()
        if remaining_p1 < 5.0:
            break
        t_alloc = min(remaining_p1 * 0.8, 30.0)
        if t_alloc < 3.0:
            break

        a_opt, obj_opt = optimize_sequence(a0, time_limit=t_alloc,
                                            use_lp=True, use_smooth=True, use_subgrad=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase1: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization (50% of budget)
    phase2_time = total_budget * 0.50
    phase2_deadline = time.monotonic() + phase2_time
    print(f"Phase 2: Deep optimization ({phase2_time:.0f}s budget)")

    scan_results.sort(key=lambda x: x[0])
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 6:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    if not top_configs and parent is not None:
        top_configs.append((512, 'parent', parent.copy()))

    remaining_p2 = phase2_deadline - time.monotonic()
    if remaining_p2 < 10:
        remaining_p2 = max(10, total_budget * 0.1)
    time_per_deep = max(10.0, remaining_p2 / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        remaining_dl = deadline - time.monotonic() - 30.0
        if remaining_dl < 20:
            break
        t_alloc = min(time_per_deep, remaining_dl - 10.0)
        if t_alloc < 5.0:
            break

        a_opt, obj_opt = optimize_sequence(a0, time_limit=t_alloc,
                                            use_lp=True, use_smooth=True, use_subgrad=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase2 LP+smooth: n={n}, {name}, score={score:.6f}")

        # Additional: try smooth-only optimization
        remaining_dl = deadline - time.monotonic() - 30.0
        if remaining_dl > 10:
            t_alloc2 = min(15.0, remaining_dl * 0.3)
            a_opt2, obj_opt2 = optimize_sequence(a0, time_limit=t_alloc2,
                                                  use_lp=False, use_smooth=True, use_subgrad=True)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()
                print(f"  Phase2 smooth: n={n}, {name}, score={score2:.6f}")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Intensive polish (25% of budget)
    phase3_deadline = deadline - 5.0
    print(f"Phase 3: Intensive polish")

    if best_sequence is not None:
        polish_iter = 0
        while time.monotonic() < phase3_deadline:
            remaining = phase3_deadline - time.monotonic()
            if remaining < 10:
                break
            polish_iter += 1

            # LP optimization
            lp_time = min(remaining * 0.4, 60.0)
            a_opt, obj_opt = optimize_sequence(best_sequence, time_limit=lp_time,
                                                use_lp=True, use_smooth=True, use_subgrad=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase3[{polish_iter}] LP: score={score:.6f}")

            # Local search
            remaining = phase3_deadline - time.monotonic()
            if remaining < 5:
                break
            ls_time = min(remaining * 0.3, 30.0)
            a_ls, obj_ls = local_search(best_sequence, time_limit=ls_time)
            score_ls = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score_ls) and score_ls < best_score:
                best_score = score_ls
                best_sequence = a_ls.copy()
                print(f"  Phase3[{polish_iter}] LS: score={score_ls:.6f}")

            # Random perturbation + re-optimize
            remaining = phase3_deadline - time.monotonic()
            if remaining < 15:
                break
            if polish_iter % 3 == 0:
                n = len(best_sequence)
                perturbation = rng.dirichlet(np.ones(n) * 10.0)
                mixed = 0.9 * best_sequence + 0.1 * perturbation
                mixed = normalize(mixed)
                t_pert = min(remaining * 0.3, 30.0)
                a_p, obj_p = optimize_sequence(mixed, time_limit=t_pert,
                                                use_lp=True, use_smooth=True, use_subgrad=True)
                score_p = float(evaluate_sequence(a_p.tolist()))
                if np.isfinite(score_p) and score_p < best_score:
                    best_score = score_p
                    best_sequence = a_p.copy()
                    print(f"  Phase3[{polish_iter}] Perturb: score={score_p:.6f}")

    # Final check
    if best_sequence is None:
        best_sequence = np.ones(128) / 128.0
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
