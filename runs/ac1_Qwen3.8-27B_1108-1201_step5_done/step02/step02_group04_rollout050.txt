import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

def fast_auto_conv(a):
    """Compute auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_val(a):
    """Objective assuming sum(a)=1. Returns 2n * max(auto_conv)."""
    n = len(a)
    return 2.0 * n * fast_auto_conv(a).max()


def build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = sum_j f[k-j]*g[j] = (f*g)[k]."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(A_mat, M, n, timeout=8):
    """Solve max sum(g) s.t. A@g <= M, g >= 0. Return normalized g or None."""
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = opt.linprog(c, A_ub=A_mat, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-12:
            return g / s
    return None


def subgradient_step(f, k_peak, eta, n):

    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()

    return f_new / s

def gen_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations of length n summing to 1."""

    # Gaussian centered
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g

    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb = np.maximum(cb, 0.01)
    inits['cos'] = cb / cb.sum()

    # Bipolar (high at ends)
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    # Triangular
    tri = np.maximum(1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / (n - 1), 0.05)
    inits['tri'] = tri / tri.sum()

    # Random Dirichlet
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd1'] = rr
    rr2 = rng.dirichlet(np.ones(n) * 0.1)
    inits['rnd2'] = rr2

    # Sine-modulated
    sm = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x / n)
    sm = np.maximum(sm, 0.01)
    inits['sine'] = sm / sm.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.7
    inits['alt'] = alt / alt.sum()

    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()
        else:

            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()

    return inits

        inits = gen_diverse_inits(n, rng, parent)
        n_configs = len(inits)
        time_per = max(10.0, (p1_deadline - time.monotonic()) / max(1, n_configs))

        for name, a0 in inits.items():
            if time.monotonic() > p1_deadline - 10:
                break
            ta = min(time_per, p1_deadline - time.monotonic() - 5)
            if ta < 8:
                break

            f_opt, obj_opt = lp_optimize(a0, time_limit=ta, n=n)
            score = float(evaluate_sequence(f_opt.tolist()))

            if np.isfinite(score):
                all_results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"  n={n:4d} {name:5s} score={score:.6f} (short LP {ta:.0f}s)")

    print(f"Phase 1 done. Best score: {best_score:.6f}, {len(all_results)} results")

    # Sort results and pick top candidates for deep optimization
    all_results.sort(key=lambda x: x[0])
    top_candidates = all_results[:6]

    # Phase 2: Deep optimization with annealing on top candidates
    print("Phase 2: Deep anneal-LP optimization")
    phase2_deadline = deadline - 120.0
    if top_candidates:
        n_top = len(top_candidates)
        t_each = (phase2_deadline - time.monotonic()) / n_top
        t_each = max(60.0, t_each)

        for idx, (sc, n, name, f) in enumerate(top_candidates):
            if time.monotonic() > phase2_deadline - 10:
                break
            ta = min(t_each, phase2_deadline - time.monotonic() - 5)
            if ta < 30:
                break

            print(f"  Deep opt [{idx}] n={n} {name} initial_score={sc:.6f} time={ta:.0f}s")
            f2, obj2 = anneal_lp(f, time_limit=ta, n=n, rng=rng, verbose=True)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2):
                all_results.append((sc2, n, f'deep{idx}', f2))
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = f2.copy()
                    print(f"    NEW BEST: n={n} score={sc2:.6f}")

    print(f"Phase 2 done. Best score: {best_score:.6f}")

    # Phase 3: Final polishing on the overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        print("Phase 3: Final polishing")
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 5

        # Multiple rounds of different polishing
        f_cur = best_seq.copy()
        for round in range(3):
            if time.monotonic() > deadline - 5:
                break
            tr = min(remaining / (3 - round), deadline - time.monotonic() - 3)
            if tr < 3:
                break

            # LP polish
            f_cur, obj_cur = lp_optimize(f_cur, time_limit=tr * 0.5, n=n_best)
            sc_cur = float(evaluate_sequence(f_cur.tolist()))
            if np.isfinite(sc_cur) and sc_cur < best_score:
                best_score = sc_cur
                best_seq = f_cur.copy()
                print(f"  Final LP round {round}: score={sc_cur:.6f}")

            # Subgradient polish
            if time.monotonic() < deadline - 3:
                tr2 = min(tr * 0.3, deadline - time.monotonic() - 3)
                f_cur, obj_cur = multi_subgradient_polish(f_cur, n_best, tr2, eta0=0.03)
                sc_cur = float(evaluate_sequence(f_cur.tolist()))
                if np.isfinite(sc_cur) and sc_cur < best_score:
                    best_score = sc_cur
                    best_seq = f_cur.copy()
                    print(f"  Final SG round {round}: score={sc_cur:.6f}")

    # Phase 4: Try a few random restarts if time remains
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print("Phase 4: Random restarts")
        n_best = len(best_seq)
        for attempt in range(5):
            if time.monotonic() > deadline - 30:
                break
            # Perturb the best and try to improve
            noise = 0.1 / np.sqrt(n_best)
            f_try = best_seq + rng.normal(0, noise, n_best)
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()

            ta = min(30.0, deadline - time.monotonic() - 15)
            if ta < 10:
                break
            f_r, obj_r = lp_optimize(f_try, time_limit=ta, n=n_best)
            sc_r = float(evaluate_sequence(f_r.tolist()))
            if np.isfinite(sc_r) and sc_r < best_score:
                best_score = sc_r
                best_seq = f_r.copy()
                print(f"  Restart {attempt}: NEW BEST score={sc_r:.6f}")
            else:
                print(f"  Restart {attempt}: score={sc_r:.6f} (no improvement)")

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
