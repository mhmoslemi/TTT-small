import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def conv_max(a):
    """Return max of conv(a,a)."""
    return np.max(fft_convolve(a, a))


def normalize_to(a, S):
    """Scale a to have sum S."""
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_banded_conv_matrix(a, n):
    """Build the (2n-1) x n convolution matrix efficiently using banded structure."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max >= j_min:
            A[k, j_min:j_max + 1] = a[k - j_min:k - j_max + 1][::-1]
    return A


def solve_lp(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    A = build_banded_conv_matrix(a, n)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_direction(a, n):
    """Find LP optimal direction given current peak M."""
    M = conv_max(a)

        return a.copy()
    g = solve_lp(a, M, n)
    g_sum = np.sum(g)
    if g_sum < 1e-12:
        return a.copy()
    return g / g_sum

def subgradient_step(a, lr, momentum_prev, beta=0.9):
    """One subgradient step on the peak of conv(a,a) with momentum."""
    n = len(a)
    conv = fft_convolve(a, a)
    peak_idx = int(np.argmax(conv))

    # Subgradient of conv[peak_idx] w.r.t. a
    grad = np.zeros(n)
    for j in range(n):
        idx1 = peak_idx - j
        idx2 = j - peak_idx
        val = 0.0
        if 0 <= idx1 < n:
            val += a[idx1]
        if 0 <= idx2 < n:
            val += a[idx2]
        grad[j] = val

    # Momentum
    v = momentum_prev * momentum_prev * beta - lr * grad
    a_new = a - v
    a_new = np.maximum(a_new, 0.0)
    S = np.sqrt(2.0 * n)
    a_new = normalize_to(a_new, S)
    return a_new, v, peak_idx

def simulated_annealing(a, time_limit, n=None, verbose=False):

    if n is None:

        end = start + block_size

        # Try to optimize this block

            # Perturb block

def interpolate_to_length(a, n_new):
    """Interpolate sequence to new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def make_initialization(n, init_type, rng):
    """Create initial sequences of various types."""
    t = np.arange(n, dtype=np.float64)

        phase = rng.uniform(0, 2 * np.pi)

            amp = rng.uniform(0.03, 0.15)

    elif init_type == 'random':

    elif init_type == 'triangular':

        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)

        a_ref, cm_ref = lp_refine(a0, t_limit, n=n_try, verbose=False)
        if update_best(a_ref):
            print(f"      n={n_try} gave improvement!")

    # Phase 3: Structured initializations + LP
    print("\n=== Phase 3: Structured inits + LP ===")
    init_configs = [
        (512, 'multi_sine'), (512, 'chirp'), (640, 'multi_sine'),
        (640, 'random'), (768, 'sine'), (768, 'chirp'),
        (896, 'multi_sine'), (1024, 'random'), (1024, 'chirp'),
    ]
    for n_try, init_type in init_configs:
        if time.monotonic() > deadline - 300:
            break
        a0 = make_initialization(n_try, init_type, rng)
        t_limit = min(30.0, (deadline - time.monotonic()) * 0.08)
        a_ref, cm_ref = lp_refine(a0, t_limit, n=n_try, verbose=False)
        if update_best(a_ref):
            print(f"      n={n_try}, init={init_type} gave improvement!")

    # Phase 4: Subgradient refinement on best
    print("\n=== Phase 4: Subgradient refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(180.0, (deadline - time.monotonic()) * 0.25)
        a_ref, cm_ref = subgradient_optimize(best_seq, t_limit, verbose=True)
        update_best(a_ref)

    # Phase 5: More LP on the best
    print("\n=== Phase 5: Additional LP refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 120
        if remaining > 60:
            t_limit = min(remaining, 180.0)
            a_ref, cm_ref = lp_refine(best_seq, t_limit, verbose=True)
            update_best(a_ref)

    # Phase 6: Simulated annealing
    print("\n=== Phase 6: Simulated annealing ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 30:
            t_limit = min(remaining, 120.0)
            a_ref, cm_ref = simulated_annealing(best_seq, t_limit, verbose=True)
            update_best(a_ref)

    # Phase 7: Final LP + subgradient polish
    print("\n=== Phase 7: Final polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            t_lp = remaining * 0.5
            a_ref, cm_ref = lp_refine(best_seq, t_lp, verbose=False)
            update_best(a_ref)
            if time.monotonic() < deadline - 10:
                a_ref, cm_ref = subgradient_optimize(best_seq, deadline - time.monotonic() - 10, verbose=False)
                update_best(a_ref)

    # Final: try constant sequences as a sanity check
    print("\n=== Sanity: constant sequences ===")
    for n_test in [256, 512, 640, 1024, 2048]:
        a_const = np.ones(n_test)
        update_best(a_const)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: eval_score={final_score:.8f}, n={len(final_list)}, local={best_score:.8f} ===")
        return final_list

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 512
