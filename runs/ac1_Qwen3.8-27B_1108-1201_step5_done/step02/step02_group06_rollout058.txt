- If we write `a^T Q a = sum_{i,j} Q[i,j] a[i] a[j]`, then we need `Q[i,j] + Q[j,i] = 1` when `j = k-i` (i.e., `i+j = k`), and 0 otherwise.
- So `Q[i,j] = 1/2` when `i+j = k` and `i != j`, and `Q[i,i] = 1` when `2i = k`.

For this to be a convex quadratic form, `Q_k` needs to be positive semi-definite. Is it?

For `k = 0`: `conv(0) = a[0]^2`, so `Q_0 = [[1, 0, ...], [0, 0, ...], ...]`. This is PSD. ✓
For `k = n-1` (the center): `conv(n-1) = sum_i a[i]*a[n-1-i]`. The matrix has `Q[i, n-1-i] = 1/2` (and `Q[i,i] = 1` if `i = n-1-i`). This matrix is NOT necessarily PSD!

Example: `n=2`, `k=1`: `conv(1) = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]`. The matrix is `[[0, 1], [1, 0]]` which has eigenvalues `+1` and `-1`. NOT PSD!

So the problem is NOT convex in general! The quadratic forms are not all convex. This makes it a non-convex minimize-max problem.

OK so CVXPY won't directly work (unless we reformulate). Let me go back to the gradient-based approach.

"""
Hybrid optimization for minimizing normalized peak autocorrelation of non-negative sequences.
Uses flat-spectrum error reduction, projected gradient descent with logsumexp smoothing,
and coordinate descent for final polishing. Tries multiple sequence lengths.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def flat_spectrum_iteration(a, max_iter=1000):
    """Error reduction: alternate between flattening |F(a)| and enforcing a >= 0."""
    n = len(a)
    target_sum = np.sum(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    for _ in range(max_iter):
        # Frequency domain: flatten magnitude
        F = np.fft.rfft(a, fft_size)
        mag = np.abs(F)
        phase = F / (mag + 1e-15)
        # Target: flat magnitude = sqrt(mean(mag^2))
        target_mag = np.sqrt(np.mean(mag ** 2))
        # Soft flatten: interpolate between current and target
        new_mag = 0.5 * mag + 0.5 * target_mag
        F_new = phase * new_mag
        a_new = np.fft.irfft(F_new, fft_size)[:n]
        
        # Time domain: enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new * (target_sum / s)
        a = a_new
    
    return a

def gradient_descent(a_init, max_iter, tau_init, tau_end, lr_init, lr_end):
    """Projected gradient descent with logsumexp smoothing."""
    n = len(a_init)
    target_sum = np.sum(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = local_score(a)
    
    # Precompute FFT size for cross-correlation
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1
    
    for it in range(max_iter):
        # Adaptive temperature
        frac = it / max(max_iter - 1, 1)
        tau = tau_init * (tau_end / tau_init) ** frac
        lr = lr_init * (lr_end / lr_init) ** frac
        
        # Compute convolution
        conv = fft_convolve(a, a)
        
        # Softmax weights (for smooth max)
        shifted = (conv - np.max(conv)) / tau
        exp_conv = np.exp(shifted)
        w = exp_conv / np.sum(exp_conv)
        
        # Gradient: grad[i] = 2 * sum_j a[j] * w[j+i]
        # Using FFT cross-correlation
        fa = np.fft.rfft(a, fft_size)
        fw = np.fft.rfft(w, fft_size)
        v = np.fft.irfft(np.conj(fa) * fw, fft_size)[:n]
        grad = 2.0 * v
        
        # Project gradient: subtract mean to maintain sum constraint
        grad = grad - np.mean(grad)
        
        # Update
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            a = a_init.copy()
            continue
        a = a * (target_sum / s)
        
        # Track best
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def coordinate_descent(a, max_iter=100, step_size=0.05):
    """Simple coordinate descent: try small changes to each coordinate."""
    n = len(a)
    target_sum = np.sum(a)
    best_score = local_score(a)
    rng = np.random.default_rng(42)
    
    for _ in range(max_iter):
        improved = False
        # Random order
        indices = rng.permutation(n)
        for i in indices:
            for delta in [step_size, -step_size, step_size*0.5, -step_size*0.5]:
                a_test = a.copy()
                a_test[i] = max(0.0, a_test[i] + delta * target_sum / n)
                s = np.sum(a_test)
                if s < 1e-10:
                    continue
                a_test = a_test * (target_sum / s)
                score = local_score(a_test)
                if score < best_score:
                    a = a_test
                    best_score = score
                    improved = True
                    break
        if not improved:
            # Anneal: decrease step size
            step_size *= 0.8
            if step_size < target_sum * 1e-6 / n:
                break
    
    return a, best_score

def peak_shave_gradient(a_init, max_iter, verbose=False):
    """
    Targeted gradient: focus reduction on coordinates contributing most to the peak.
    This is a heuristic that can escape plateaus better than uniform gradient.
    """
    n = len(a_init)
    target_sum = np.sum(a_init)
    a = a_init.copy()
    best_a = a.copy()

    while grad_fft_size < 3 * n:
        grad_fft_size <<= 1
    
    lr_base = 1.0 / n
    
    for it in range(max_iter):
        # Decay learning rate
        lr = lr_base * (1.0 - 0.5 * it / max(max_iter, 1))
        
        fa = np.fft.rfft(a, conv_fft_size)
        conv = np.fft.irfft(fa * fa, conv_fft_size)[:2 * n - 1]
        peak_val = np.max(conv)
        peak_idx = int(np.argmax(conv))
        
        # Compute contribution of each a[i] to the peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contrib = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contrib[j] = a[j] * a[k]
        
        # Also compute contributions to near-peak values
        near_peaks = []
        for k in range(max(0, peak_idx - 3), min(2*n-1, peak_idx + 4)):
            if k == peak_idx:
                continue
            if conv[k] > 0.8 * peak_val:
                near_peaks.append(k)
        
        for k in near_peaks:
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    contrib[j] += 0.3 * a[j] * a[m]
        
        # Normalize contribution weights
        total_contrib = np.sum(contrib)
        if total_contrib < 1e-15:
            break
        w_contrib = contrib / total_contrib
        
        # Full gradient (smoothed)
        tau = peak_val * 0.01
        shifted = (conv - peak_val) / tau
        exp_c = np.exp(shifted)

        grad = grad - np.mean(grad)

        grad_shaped = 0.7 * grad + 0.3 * (w_contrib - 1.0 / n) * peak_val

            if s_prov < best_score:
                best_score = s_prov
                best_seq = provided_seq.copy()
    
    # Search configurations: (n, time_fraction)
    # Allocate more time to promising lengths
    search_configs = [
        (256, 0.10),
        (320, 0.08),
        (384, 0.10),
        (512, 0.14),
        (640, 0.10),
        (768, 0.12),
        (1024, 0.14),
        (1280, 0.10),
        (1536, 0.12),
    ]
    
    total_frac = sum(c[1] for c in search_configs)
    
    for n, frac in search_configs:
        if time.monotonic() > deadline - 80:
            break
        
        time_alloc = frac / total_frac * (budget_s - 80)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 30)
        if time_alloc < 20:
            break
        
        print(f"\n  === n={n}, allocated {time_alloc:.0f}s ===")
        
        # --- Phase 1: Multiple flat-spectrum initializations ---
        n_flat_iters = int(min(time_alloc * 0.3, 60.0) / max(n * 0.00002, 0.001))
        n_flat_iters = max(50, min(n_flat_iters, 3000))
        
        best_flat_a = None
        best_flat_score = np.inf
        
        # Try several different initializations
        n_inits = min(4, max(2, int(time_alloc * 0.15 / 10)))
        
        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 60:
                break
            
            if init_idx == 0 and provided_seq is not None:
                # Use provided sequence (resampled to n)
                if len(provided_seq) == n:
                    a0 = provided_seq.copy()
                else:
                    x_old = np.linspace(0, 1, len(provided_seq))
                    x_new = np.linspace(0, 1, n)
                    a0 = np.interp(x_new, x_old, provided_seq)
                    a0 = np.maximum(a0, 0.01)
            elif init_idx == 1:
                # Smooth random
                noise = rng.standard_normal(n)
                # Low-pass filter
                F = np.fft.rfft(noise, 2 * n)
                F[n // 4:] = 0
                a0 = np.fft.irfft(F, 2 * n)[:n]
                a0 = np.abs(a0) + 0.1
            elif init_idx == 2:
                # Random + chirp
                t = np.arange(n, dtype=np.float64)
                a0 = 1.0 + 0.5 * np.cos(2 * np.pi * t * t / (3 * n))
                a0 += 0.1 * rng.standard_normal(n)
                a0 = np.maximum(a0, 0.05)
            else:
                # Pure random positive
                a0 = np.abs(rng.standard_normal(n)) + 0.1
            
            a0 = a0 / np.sum(a0)
            
            a_flat, sc_flat = flat_spectrum_init(a0, n_flat_iters, alpha=0.6, verbose=False)
            
            if sc_flat < best_flat_score:
                best_flat_score = sc_flat
                best_flat_a = a_flat
        
        if best_flat_a is not None:
            print(f"    Flat spectrum best: score={best_flat_score:.6f}")
            if best_flat_score < best_score:
                best_score = best_flat_score
                best_seq = best_flat_a.copy()
                print(f"    *** New global best: {best_score:.6f} (n={n}) ***")
        
        # --- Phase 2: Projected gradient descent ---
        remaining = deadline - time.monotonic() - 40
        pgd_time = remaining * 0.4
        if pgd_time > 10 and best_flat_a is not None:
            # Estimate iterations
            est_time_per_iter = n * 0.000003
            max_pgd_iter = int(pgd_time / est_time_per_iter)
            max_pgd_iter = max(200, min(max_pgd_iter, 500000))
            
            a_pgd, sc_pgd = projected_gradient_descend(best_flat_a, max_iter=max_pgd_iter,
                                                        verbose=True, check_interval=20)
            print(f"    PGD: score={sc_pgd:.6f}")
            
            if sc_pgd < best_score:
                best_score = sc_pgd
                best_seq = a_pgd.copy()
                print(f"    *** New global best: {best_score:.6f} (n={n}) ***")
            
            # --- Phase 3: Peak-shaving gradient ---
            remaining = deadline - time.monotonic() - 30
            ps_time = remaining * 0.3
            if ps_time > 5:
                est_iter_ps = int(ps_time / est_time_per_iter)
                max_ps_iter = max(100, min(est_iter_ps, 200000))
                
                a_ps, sc_ps = peak_shave_gradient(a_pgd, max_iter=max_ps_iter, verbose=True)
                print(f"    PeakShave: score={sc_ps:.6f}")
                
                if sc_ps < best_score:
                    best_score = sc_ps
                    best_seq = a_ps.copy()
                    print(f"    *** New global best: {best_score:.6f} (n={n}) ***")
            
            # --- Phase 4: Coordinate descent ---
            remaining = deadline - time.monotonic() - 20
            if remaining > 5:
                a_cd, sc_cd = coordinate_descent_polin(a_pgd, max_rounds=20, rng=rng)
                print(f"    CoordDesc: score={sc_cd:.6f}")
                
                if sc_cd < best_score:
                    best_score = sc_cd
                    best_seq = a_cd.copy()
                    print(f"    *** New global best: {best_score:.6f} (n={n}) ***")
    
    # === Final polish on the global best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 10
        n = len(best_seq)
        print(f"\n  === Final polish: n={n}, time={remaining:.0f}s ===")
        
        est_time_per_iter = n * 0.000003
        max_final_iter = int(remaining / est_time_per_iter)
        max_final_iter = max(100, min(max_final_iter, 300000))
        
        a_final, sc_final = projected_gradient_descend(best_seq, max_iter=max_final_iter,
                                                        verbose=True, check_interval=50)
        print(f"    Final PGD: score={sc_final:.6f}")
        
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"    *** Final best: {best_score:.6f} ***")
        
        # Quick coordinate descent
        if time.monotonic() < deadline - 5:
            a_cd2, sc_cd2 = coordinate_descent_polin(best_seq, max_rounds=15, rng=rng)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()
                print(f"    *** CoordDesc final best: {best_score:.6f} ***")
    
    # Final verification with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"Local score was: {best_score:.6f}")
        print(f"{'='*60}")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
