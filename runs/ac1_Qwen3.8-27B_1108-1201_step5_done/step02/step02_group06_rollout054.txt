```python
import numpy as np
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def peak_contributions(a, peak_k, top_m=10):
    """For each position j, compute its contribution to the peak at position peak_k."""
    n = len(a)
    c = fft_convolve(a, a)
    # c[peak_k] = sum_i a[i] * a[peak_k - i]
    # Contribution of a[j] to c[peak_k] is a[j] * a[peak_k - j] (if valid)
    contribs = np.zeros(n)
    for j in range(n):
        k_j = peak_k - j
        if 0 <= k_j < n:
            contribs[j] = a[j] * a[k_j]
    return contribs, c

def multiplicative_update(a, c, eta, top_peaks=5):
    """Apply multiplicative update to reduce peak autocorrelation."""
    n = len(a)
    # Find top peak positions
    peak_indices = np.argsort(c)[-top_peaks:]
    peak_indices = sorted(peak_indices)
    
    # Compute the "pressure" at each position from all top peaks
    pressure = np.zeros(n)

            k_j = pk - j
            if 0 <= k_j < n:

    a_new = a * np.exp(-eta * pressure / np.max(pressure))
    a_new = np.clip(a_new, 1e-10, None)
    
    # Renormalize
```

```python
        peaks = np.where(c >= peak_thresh)[0]
        
        if len(peaks) == 0:
            peaks = np.argsort(c)[-3:]
        
        # Compute pressure from peaks
        # pressure[j] = sum over peaks k of a[k-j] (if valid)
        pressure = np.zeros(n)
        for k in peaks:
            for j in range(n):
                kj = int(k) - j
                if 0 <= kj < n:
                    pressure[j] += a[kj]
        
        # Normalize pressure
        p_max = np.max(pressure)
        if p_max < 1e-15:
            break
        pressure = pressure / p_max
        
        # Multiplicative update
        a_new = a * np.exp(-eta * pressure)
        a_new = np.clip(a_new, 1e-15, None)
        s = np.sum(a_new)
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        # Slowly decrease eta
        eta *= 0.999
        
        # Occasionally add noise to escape
        if rng.random() < 0.01:
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = np.clip(a, 1e-15, None)
            a = a / np.sum(a)
    
    return best_a, best_score
```

```python
"""
Smooth-max descent with multiplicative peak reduction and SA polish
for minimizing normalized peak autocorrelation of non-negative step functions.
Uses FFT-based gradient of soft-max approximation to the peak, combined with
targeted multiplicative updates and simulated annealing for final refinement.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smooth_max_descent(a_init, time_limit, seed=42, verbose=False):
    """Minimize max(conv(a,a)) using soft-max smoothed gradient descent."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    sum_a = np.sum(a)
    if sum_a < 1e-15:
        return a, np.inf
    a = a / sum_a

    best_a = a.copy()
    c = fft_convolve(a, a)
    best_score = 2.0 * n * np.max(c)
    t0 = time.monotonic()

    c_max_init = np.max(c)
    T0 = c_max_init / 4.0
    T1 = c_max_init * 0.03
    base_lr = 0.4

    # Pre-compute FFT size
    N = 1
    while N < 3 * n:
        N <<= 1

    step = 0
    stuck = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        T = T0 * (T1 / T0) ** min(elapsed, 1.0)
        lr = base_lr / (1.0 + 0.03 * step)

        c = fft_convolve(a, a)
        c_max = np.max(c)

        # Softmax weights
        shifted = (c - c_max) / T
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            break
        w = w / w_sum

        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
        W = np.fft.fft(w, N)
        A = np.fft.fft(a, N)
        grad = np.fft.ifft(W * np.conj(A)).real[:n] * 2.0

        # Gradient step
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new = a_new / s

        c_new = fft_convolve(a_new, a_new)
        c_new_max = np.max(c_new)
        score = 2.0 * n * c_new_max

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stuck = 0
        else:
            stuck += 1

        a = a_new
        step += 1

        # Periodic perturbation to escape local minima
        if stuck > 100 and rng.random() < 0.05:
            a = a * np.exp(rng.normal(0, 0.08, n))
            a = np.clip(a, 0.0, None)
            s = np.sum(a)
            if s > 1e-15:
                a = a / s
            stuck = 0

        if verbose and step % 200 == 0:
            print(f"    SMD step={step}, score={score:.6f}, best={best_score:.6f}, T={T:.6f}")

    best_a = best_a * np.sum(a_init)
    return best_a, compute_score(best_a)


def multiplicative_peak_reduction(a_init, time_limit, seed=42, verbose=False):
    """Reduce peak autocorrelation using targeted multiplicative updates."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    sum_a = np.sum(a)
    a = a / sum_a

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    eta = 0.08
    n_peaks = 3

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        c_max = np.max(c)

        # Find top peak positions
        if n_peaks <= 2 * n - 1:
            peaks = np.argsort(c)[-n_peaks:]
        else:
            peaks = np.arange(2 * n - 1)

        # Compute pressure from peaks (vectorized per peak)
        pressure = np.zeros(n)
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                a_lo = k - j_end
                a_hi = k - j_start
                pressure[j_start:j_end + 1] += a[a_lo:a_hi + 1][::-1]

        p_max = np.max(pressure)
        if p_max < 1e-15:
            break
        pressure = pressure / p_max

        # Multiplicative update
        a_new = a * np.exp(-eta * pressure)
        a_new = np.clip(a_new, 1e-15, None)
        s = np.sum(a_new)
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        eta *= 0.9995

        # Occasional escape
        if rng.random() < 0.008:
            a = a * np.exp(rng.normal(0, 0.06, n))
            a = np.clip(a, 1e-15, None)
            a = a / np.sum(a)

        if verbose and (t0 and time.monotonic() - t0 > 50):
            pass

    best_a = best_a * np.sum(a_init)
    return best_a, compute_score(best_a)


def sa_polish(a_init, time_limit, rng, verbose=False):
    """Simulated annealing with informed moves targeting peak contributors."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    sum_a = np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    t0 = time.monotonic()

    temperature = 0.004
    cooling = 0.9998
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.30:
            # Targeted: reduce top contributors to peak
            c = fft_convolve(a, a)
            peak_k = int(np.argmax(c))
            contribs = np.zeros(n)
            j_start = max(0, peak_k - n + 1)
            j_end = min(n - 1, peak_k)
            for j in range(j_start, j_end + 1):
                contribs[j] = a[j] * a[peak_k - j]

            # Find also off-peak positions to increase
            top_k = max(2, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))

            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(4, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))

        elif r < 0.50:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))

        elif r < 0.65:
            # Dual block
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))

        elif r < 0.80:
            # Mirror perturbation: perturb a[j] and a[n-1-j] together
            for _ in range(3):
                j = int(rng.integers(0, n))
                delta = rng.normal(0, 0.02)
                candidate[j] *= np.exp(delta)
                candidate[n - 1 - j] *= np.exp(-delta)

        elif r < 0.90:
            # Convolution-informed: reduce elements that create high off-center peaks
            c = fft_convolve(a, a)
            c_max = np.max(c)
            # Find second-highest peak
            c_sorted = np.sort(c)[::-1]
            if len(c_sorted) > 1 and c_sorted[1] > 0.5 * c_max:
                # There's a significant secondary peak
                peak2 = int(np.argsort(c)[::-1][1])
                contribs2 = np.zeros(n)
                j_start = max(0, peak2 - n + 1)
                j_end = min(n - 1, peak2)
                for j in range(j_start, j_end + 1):
                    contribs2[j] = a[j] * a[peak2 - j]
                top_idx = np.argsort(contribs2)[-max(2, n // 20):]
                candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))

        else:
            # Global small noise
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-15:
            continue
        candidate = candidate * (sum_a / cand_sum)

        new_score = compute_score(candidate)
        iter_count += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

        temperature *= cooling

    return best_a, best_score


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_chirp_init(n, rng, alpha=None):
    if alpha is None:
        alpha = 0.3 + 0.4 * rng.random()
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.05, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_multitone_init(n, rng, n_tones=5):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.04, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

            )
            print(f"  MPR: score={score_mpr:.6f}", flush=True)
            if score_mpr < best_score:
                best_score = score_mpr
                best_seq = a_mpr.copy()
                print(f"  *** New best: {best_score:.6f} ***", flush=True)

    # === Phase 3: SA polish ===
    print("\n" + "=" * 60)
    print("Phase 3: Simulated annealing polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 50:
        remaining = deadline - time.monotonic()
        sa_time = min(350.0, remaining * 0.55)
        if sa_time > 20:
            print(f"  n={len(best_seq)}, alloc={sa_time:.0f}s", flush=True)
            a_sa, score_sa = sa_polish(best_seq, sa_time, rng, verbose=False)
            print(f"  SA: score={score_sa:.6f}", flush=True)
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***", flush=True)

    # === Phase 4: Final smooth-max descent on best ===
    print("\n" + "=" * 60)
    print("Phase 4: Final refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        final_time = min(200.0, remaining * 0.7)
        if final_time > 15:
            print(f"  Final SMD, n={len(best_seq)}, alloc={final_time:.0f}s", flush=True)
            a_fin, score_fin = smooth_max_descent(
                best_seq, final_time, seed=seed + 5000
            )
            print(f"  Final SMD: score={score_fin:.6f}", flush=True)
            if score_fin < best_score:
                best_score = score_fin
                best_seq = a_fin.copy()
                print(f"  *** New best: {best_score:.6f} ***", flush=True)

            # Quick SA after
            if time.monotonic() < deadline - 20:
                rem = deadline - time.monotonic() - 5
                if rem > 10:
                    a_sa2, score_sa2 = sa_polish(best_seq, rem, rng)
                    if score_sa2 < best_score:
                        best_score = score_sa2
                        best_seq = a_sa2.copy()
                        print(f"  Final SA: {best_score:.6f}", flush=True)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
```
