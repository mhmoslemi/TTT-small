import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    # Let m = k-i: sum_m w[m+i] * a[m]
    # This is the cross-correlation of w and a at lag i
    # = (w * flip(a))[n-1 + i] where flip(a)[j] = a[n-1-j]
    
    a_flip = a[::-1]
    corr = fft_convolve(w, a_flip)  # length (2n-1) + (n-1) = 3n-2... 
    # Wait, w has length 2n-1, a_flip has length n
    # convolve(w, a_flip) has length (2n-1) + n - 1 = 3n-2
    # We want corr[i] = sum_m w[m+i] * a[m] for i = 0, ..., n-1
    # = sum_k w[k] * a[k-i]
    # In the full convolution (w * a_flip)[p] = sum_j w[j] * a_flip[p-j] = sum_j w[j] * a[n-1-p+j]
    # We want sum_k w[k] * a[k-i], so set n-1-p+j = k-i, j = k: p = n-1+k-(k-i) = n-1+i
    # So corr[i] = (w * a_flip)[n-1+i]
    
    full_corr = fft_convolve(w, a_flip)
    grad = 2.0 * n * full_corr[n-1:n-1+n]
    
    # Also need to account for the denominator (sum a)^2
    # Full objective: 2n * log_sum_exp / tau / (sum a)^2
    # d/d(a_i) [log_sum_exp / (sum a)^2] = (grad_raw_i * (sum a) - 2 * log_sum_exp) / (sum a)^3
    # where grad_raw_i = sum_k w[k] * 2 * a[k-i] (the 2 comes from the derivative of c[k])
    
    # Actually let me redo this properly.
    # Objective: f(a) = 2n * (1/tau) * log(sum_k exp(tau * c[k])) / (sum_j a[j])^2
    # where c = a * a
    # 
    # df/da_i = 2n/tau * [ (sum_k exp(tau*c[k]) * tau * dc[k]/da_i) / sum_k exp(tau*c[k]) * (sum a)^2 
    #                  - 2 * (1/tau) * log(sum exp(tau*c)) * sum a ] / (sum a)^4
    #           = 2n * [ (sum_k w_k * dc[k]/da_i) / (sum a)^2 - 2 * (1/tau) * log_sum_exp * ... ]
    
    # This is getting complicated. Let me just use the simpler form:
    # Since the objective is scale-invariant, fix sum(a) = 1 and minimize 2n * max_k c[k].
    # Then the gradient is just 2n * sum_k w_k * dc[k]/da_i = 2n * sum_k w_k * 2*a[k-i]
    # = 4n * (w * a_flip)[n-1+i]
    
    # But we need to project back to sum(a) = 1 after the update.
    
    grad = 4.0 * n * full_corr[n-1:n-1+n]
    return grad, log_sum_exp

def adam_update(a, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (np.sum(a) / s)  # Maintain normalization
    return a_new, m, v

def smoothed_gradient_descent(a0, time_limit, rng, tau_start=1.0, tau_end=50.0):
    """Phase 1: Smoothed gradient descent with Adam."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a = a / np.sum(a)  # Normalize to sum=1
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    total_steps = 0
    while time.monotonic() - t0 < time_limit:
        # Anneal tau
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        grad, _ = smoothed_gradient(a, tau)
        
        t += 1
        lr = 0.005 / (1 + 0.001 * t)
        a, m, v = adam_update(a, grad, m, v, t, lr=lr)
        total_steps += 1
        
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if total_steps % 5000 == 0:
            print(f"    SGD: step={total_steps}, tau={tau:.2f}, score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

    For each coordinate i, find the optimal value x >= 0 that minimizes the objective.
    """

    a = a.copy()
    
    # Full convolution (for reference)
    c_full = fft_convolve(a, a)
    sum_a = np.sum(a)
    
    # Coordinate order (randomized for better mixing)
    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)
    
    for i in order:
        a_i = a[i]
        S_mi = sum_a - a_i  # Sum of all other elements
        
        # Compute c^(0): autocorrelation with a[i] = 0
        # c_full[k] = sum_j a[j]*a[k-j]
        # c0[k] = c_full[k] - a[i]*a[k-i] - a[k-i]*a[i] + correction
        # = c_full[k] - 2*a[i]*a[k-i] (for k-i in range)
        # But if k-i == i (k=2i), we have a[i]*a[i] counted once, so:
        # c_full[2i] includes a[i]^2, and removing it gives c0[2i] = c_full[2i] - a[i]^2
        # For k != 2i: c_full[k] includes a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i] (if k-i in range)
        #   Removing: c0[k] = c_full[k] - 2*a[i]*a[k-i]
        
        # More carefully:
        # c_full[k] = sum_j a[j] * a[k-j]
        # The terms involving a[i]:
        #   j=i: a[i]*a[k-i] (if 0<=k-i<n)
        #   j=k-i (i.e., k-j=i): a[k-i]*a[i] (if 0<=k-i<n)
        # If k-i == i (k=2i): same term counted once, value = a[i]^2
        # If k-i != i: two terms, total = 2*a[i]*a[k-i]
        
        k_indices = np.arange(2 * n - 1)
        k_minus_i = k_indices - i
        valid = (k_minus_i >= 0) & (k_minus_i < n)
        
        # Contribution of a[i] to c_full
        contrib = np.zeros(2 * n - 1)
        # For valid k: contribution is 2*a[i]*a[k-i]
        # But for k=2i (where k-i == i): contribution is a[i]^2 (not 2*a[i]^2)
        contrib[valid] = 2 * a_i * a[k_minus_i[valid]]
        if 2 * i < 2 * n - 1:
            # k=2i: we overcounted by a[i]^2 (should be a[i]^2 not 2*a[i]^2)
            contrib[2*i] -= a_i ** 2
        
        c0 = c_full - contrib
        
        # Affects mask: which k values are affected by changing a[i]
        # d(c[k])/d(a[i]) = 2*a[k-i] for valid k, and the quadratic term for k=2i
        a_shifted = np.zeros(2 * n - 1)
        a_shifted[valid] = a[k_minus_i[valid]]
        
        quad_mask = np.zeros(2 * n - 1)
        if 2 * i < 2 * n - 1:
            quad_mask[2*i] = 1.0
        
        # Generate candidates
        candidates = [0.0, a_i]  # Always include 0 and current
        
        # FOC candidates for linear constraints
        # x_k = S_mi - c0[k] / a[k-i] (for a[k-i] > 0 and k != 2i)
        linear_affected = valid & (quad_mask == 0) & (a_shifted > 1e-12)
        if np.any(linear_affected):
            x_lin = S_mi - c0[linear_affected] / a_shifted[linear_affected]
            x_lin = x_lin[x_lin > 1e-10]
            # Limit to reasonable range
            x_lin = x_lin[x_lin < 10 * max(sum_a, 1.0)]
            candidates.extend(x_lin.tolist())
        
        # FOC candidate for quadratic constraint (k=2i)
        if 2 * i < 2 * n - 1 and S_mi > 1e-12:
            x_quad = c0[2*i] / S_mi
            if x_quad > 1e-10:
                candidates.append(min(x_quad, 10 * max(sum_a, 1.0)))
        
        # Also add some geometric candidates around current value
        if a_i > 1e-10:
            for factor in [0.5, 0.8, 1.2, 1.5, 2.0]:
                candidates.append(a_i * factor)
        
        # Evaluate all candidates
        candidates = np.array(candidates)
        candidates = candidates[candidates >= 0]
        candidates = np.unique(candidates)
        
        if len(candidates) == 0:
            candidates = np.array([0.0, a_i])
        
        # Vectorized evaluation
        # c_at_x[k] = c0[k] + 2*x*a_shifted[k] + x^2*quad_mask[k]
        # For each candidate x, compute max_k c_at_x[k]
        
        # Shape: (num_candidates, 2n-1)
        c_at_x = c0[None, :] + 2 * candidates[:, None] * a_shifted[None, :] + (candidates**2)[:, None] * quad_mask[None, :]
        max_c = np.max(c_at_x, axis=1)

        valid_scores = scores[np.isfinite(scores) & (S_mi + candidates > 0.01)]
        if len(valid_scores) > 0:

            best_x = candidates[candidates >= 0][0]  # fallback
            # Find the index of the best valid score in the original array
