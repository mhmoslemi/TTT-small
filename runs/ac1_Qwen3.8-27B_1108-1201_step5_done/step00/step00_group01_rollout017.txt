def do_iap(a, n, N, rng, num_iters=200, deadline=None):
    """Iterative Alternating Projections to flatten autocorrelation."""
    a = a.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    for it in range(num_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Forward FFT
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        # Target flat magnitude
        target = np.sqrt(np.mean(mag**2))
        # Preserve phase, set flat magnitude
        phase = A / (mag + 1e-30)

        s = np.sum(a_new)

import numpy as np
import time

def flatten_spectral(a, n, N, rng, num_iters, deadline):
    """IAP to flatten the power spectrum (and hence autocorrelation)."""
    a = np.array(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a /= s
    
    for it in range(num_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        target_mag = np.sqrt(np.mean(mag * mag))
        
        # Preserve phase, set flat magnitude
        # Handle zeros in mag
        phase = np.zeros_like(A)
        nonzero = mag > 1e-15
        phase[nonzero] = A[nonzero] / mag[nonzero]
        
        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, N)
        a_new = np.real(a_new[:n])
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
    
    return a

def compute_autocorr(a, N):
    """Compute linear autocorrelation via FFT."""
    A = np.fft.rfft(a, N)
    acorr = np.fft.irfft(A * np.conj(A), N)
    return np.real(acorr)

def subgradient_step(a, n, N, peak_frac, lr):
    """One subgradient descent step to minimize max autocorrelation."""
    b = compute_autocorr(a, N)[:2*n-1]
    max_b = np.max(b)
    
    # Weight peaks
    threshold = max_b * (1 - peak_frac)
    w = (b >= threshold).astype(np.float64)
    w_sum = np.sum(w)
    if w_sum == 0:
        return a, max_b
    w /= w_sum  # normalize weights to sum to 1
    
    # Gradient: grad[j] = sum_k w[k] * a[k-j] (valid indices)
    # = cross-correlation of a and w, evaluated at j
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    grad_full = np.fft.irfft(W * np.conj(A), N)
    grad = np.real(grad_full[:n])
    
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a, max_b
    a_new /= s
    return a_new, max_b

def refine_gradient(a, n, N, budget_iters, deadline):
    """Refine using subgradient descent with adaptive step size."""
    a = a.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a /= s
    
    best_a = a.copy()
    best_max_b = np.inf
    lr = 0.05
    peak_frac = 0.2
    
    for it in range(budget_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_new, max_b = subgradient_step(a, n, N, peak_frac, lr)
        
        # Check if we made progress
        if max_b < best_max_b:
            best_max_b = max_b
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive step size
        if it % 50 == 0 and it > 0:
            lr = max(0.001, lr * 0.95)
            # If stuck, increase step size
            if max_b > best_max_b * 1.05:
                lr = min(0.5, lr * 2)
    
    return best_a
