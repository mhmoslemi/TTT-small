```python
def convolve_fft(f):
    n = len(f)
    # Need at least 2n-1 points for full convolution
    size = 1
    while size < 2*n - 1:
        size *= 2
    F = np.fft.rfft(f, size)
    G = np.fft.irfft(F * F, size)[:2*n-1]
    return G
```
