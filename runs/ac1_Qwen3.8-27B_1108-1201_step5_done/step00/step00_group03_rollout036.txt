import numpy as np
from scipy.optimize import linprog
from scipy import sparse

def lp_step(f, M):
    """Solve the LP to find g that maximizes sum(g) subject to f*g <= M, g >= 0."""
    n = len(f)
    
    # Build the banded constraint matrix
    # A[k, i] = f[i] if 0 <= k-i <= n-1, else 0
    # i.e., A[k, i] = f[i] if k-n+1 <= i <= k and 0 <= i <= n-1
    
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if result.success:
        return result.x
    else:
        return None

Iterative LP-based optimization for minimizing the peak self-convolution
of a non-negative sequence. Uses the state-of-the-art LP projection method
with adaptive blending and local perturbations to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time


def build_conv_matrix(f, n):
    """Build the banded matrix A where A[k,i] = f[i] if 0<=k-i<=n-1."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, n):
    """Solve max sum(g) s.t. f*g <= M, g >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                     method='highs', options={'presolve': True, 'time_limit': 60})
    if result.success:
        return np.clip(result.x, 0, None)
    return None

        return None

    if s < 1e-10:

    f_new = f_new * (target_sum / s)

"""
LP-based iterative optimization for minimizing the peak self-convolution
of a non-negative sequence. Combines LP projection with adaptive line search
and random perturbations to escape local minima.

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time


def build_conv_matrix(f, n):
    """Build sparse banded matrix A where A[k,i] = f[i] if 0<=k-i<=n-1."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_step(f, M, n, time_limit=30):
    """Solve max sum(g) s.t. f*g <= M, g >= 0. Returns g or None."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                     method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
    if result.success and result.x is not None:
        return np.clip(result.x, 0, None)
    return None


def compute_score_fast(f):
    """Fast score computation without calling evaluate_sequence."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(f, f)
    return 2.0 * n * np.max(conv) / (s * s)

import numpy as np
from scipy.optimize import linprog
from scipy import sparse

def subgradient_step(f, target_sum, alpha):
    n = len(f)
    c = np.convolve(f, f)
    k_star = np.argmax(c)
    # Subgradient of max(c) w.r.t. f[i] is f[k_star - i] if valid
    sg = np.zeros(n)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    for i in range(i_lo, i_hi + 1):
        sg[i] = f[k_star - i]
    # Gradient descent step
    f_new = f - alpha * sg
    # Project to non-negative
    f_new = np.maximum(f_new, 0)
    # Rescale to target sum
    s = np.sum(f_new)
    if s < 1e-10:
        return f
    f_new = f_new * (target_sum / s)
    return f_new

def build_conv_matrix(f, n):
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
