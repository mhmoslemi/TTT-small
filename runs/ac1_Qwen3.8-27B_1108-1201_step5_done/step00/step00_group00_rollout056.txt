import numpy as np
import time
from scipy.optimize import linprog

def compute_score(f):
    """Compute the score directly (for speed, avoiding function call overhead)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = np.max(conv)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def lp_iteration(f, M):
    """
    Given f and M = max(f*f), solve the LP:
    maximize sum(g)
    subject to g >= 0, (f*g)(k) <= M for all k
    
    Returns the optimal g.
    """
    n = len(f)
    # (f*g)(k) = sum_j f(j) * g(k-j) for valid j
    # For k in 0..2n-2:
    #   (f*g)(k) = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f(j) * g(k-j)
    #   Let me re-index: for fixed k, the coefficient of g(i) is f(k-i) if 0 <= k-i <= n-1
    #   So (f*g)(k) = sum_i f(k-i) * g(i) where i ranges over max(0,k-n+1) to min(k,n-1)
    #   Equivalently: coefficient of g(i) in constraint k is f(k-i) if 0 <= k-i <= n-1, else 0
    
    # A matrix of shape (2n-1, n): A[k, i] = f(k-i) if 0 <= k-i <= n-1, else 0
    
    # This is a "convolution matrix" (Toeplitz-like)
    
    # For efficiency, we can use the structure. But for scipy.linprog, we just need the matrix.
    # For n ~ 4800, the matrix is 9601 x 4800, which is ~46 million entries. That's a lot.
    # Let's use the sparse version.
    
    from scipy.sparse import lil_matrix
    
    A = lil_matrix((2*n-1, n))
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            A[k, i] = f[k - i]
    
    A = A.tocsr()
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2*n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None

def build_conv_matrix(f):
    """Build the convolution matrix A where A[k,i] = f[k-i] if 0<=k-i<=n-1 else 0."""
    n = len(f)
    A = lil_matrix((2*n - 1, n))
    for i in range(n):
        # A[k, i] = f[k-i] for k = i, i+1, ..., i+n-1
        rows = np.arange(i, i + n)
        A[rows, i] = f

**Alternative: Subgradient method**

Since we want to minimize M = max_k (f*f)(k) subject to f >= 0, sum(f) = S:

The subgradient of M at f is in the convex hull of the gradients of the active constraints. If k* is the unique maximizer, the subgradient is:

The projection onto {g >= 0, sum(g) = S} can be done via bisection: find lambda such that sum(max(f_j - lambda, 0)) = S.

def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Clip to non-negative
    v = np.clip(v, 0, None)
    v_sum = np.sum(v)
    if v_sum == 0:
        return np.full(n, target_sum / n)
    # Scale to target sum
    v = v * (target_sum / v_sum)
    return v

    """Fast score computation."""

    conv = np.convolve(f, f)
    max_b = np.max(conv)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

    A = lil_matrix((2 * n - 1, n))
    for i in range(n):
        A[i:i+n, i] = f
    return A.tocsr()


def lp_iteration(f, M):
    """Solve max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x
    return None


def subgradient_descent(f, S, max_time=100.0, verbose=False):
    """Projected subgradient descent to minimize max(f*f) with sum(f)=S."""
    n = len(f)
    f = f.copy()
    S = np.sum(f)
    t_start = time.monotonic()
    
    # Initial step size
    eta = S / n * 0.1
    best_f = f.copy()
    best_M = float(np.max(np.convolve(f, f)))
    
    for it in range(1000000):
        if time.monotonic() - t_start > max_time:
            break
        
        conv = np.convolve(f, f)
        k_star = int(np.argmax(conv))
        M = conv[k_star]
        
        if M < best_M:
            best_M = M
            best_f = f.copy()
        
        # Subgradient
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * f[k_star - idx]
        
        # Step
        f = f - eta * grad
        f = project_simplex(f, S)
        
        # Decrease step size
        eta *= 0.9999
        
        if verbose and it % 10000 == 0:
            print(f"    SGD iter {it}: M={M:.4f}, best_M={best_M:.4f}")
    
    return best_f, best_M


def block_local_search(f, max_time=100.0, verbose=False):
    """Block perturbation local search."""
    n = len(f)
    f = f.copy()
    current_score = eval_score(f)
    rng = np.random.default_rng(123)
    t_start = time.monotonic()
    count = 0
    no_improve = 0
    
    while time.monotonic() - t_start < max_time:
        count += 1
        # Random block
        width = int(rng.integers(max(1, n // 500), max(2, n // 20)))
        start = int(rng.integers(0, n - width))
        
        candidate = f.copy()
        # Multiplicative perturbation
        noise = np.exp(rng.normal(0, 0.03, width))
        candidate[start:start+width] *= noise
        candidate = np.clip(candidate, 0, 1000)
        
        score = eval_score(candidate)
        if score < current_score:
            f = candidate
            current_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt noise level based on no_improve
        # (handled implicitly by the exponential)
        
        if verbose and count % 1000 == 0:
            print(f"    LS iter {count}: score={current_score:.6f}")
    
    return f, current_score
