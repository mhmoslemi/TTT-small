    if v.sum() <= 0:
        return np.ones(n) / n
    # Sorting-based projection
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.where(cond)[0][-1] + 1
    theta = (cv[rho_val - 1] - 1.0) / rho_val
    proj = np.maximum(v - theta, 0.0)
    proj_sum = proj.sum()
    if proj_sum < 1e-15:
        return np.ones(n) / n
    return proj / proj_sum

def _fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]
