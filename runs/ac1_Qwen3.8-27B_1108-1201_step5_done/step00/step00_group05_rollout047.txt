        a = np.maximum(a, 0.05)
    elif method == 7:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t * 2) * np.cos(2 * np.pi * t * 3)
        a = np.maximum(a, 0.05)
    else:
        a = np.ones(n) * (1.0 + 0.2 * rng.standard_normal(n))
        a = np.maximum(a, 0.05)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def optimize_trial(a_init, trial_deadline, rng, verbose=True):
    """Run optimization on a single trial. Returns best sequence and score."""
    n = len(a_init)
    best = a_init.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if verbose:
        print(f"  Trial start: n={n}, score={best_score:.6f}")

    top_k = int(rng.integers(5, 25))
    lr = float(rng.uniform(0.005, 0.03))
    use_softmax = rng.random() < 0.3
    temperature = 0.01 * np.std(np.convolve(best, best, mode='full'))
    no_improve = 0
    max_no_improve = 40
    step_count = 0

    while time.monotonic() < trial_deadline:
        a = best.copy()

        # Adaptive number of GD steps
        n_gd = int(rng.integers(3, 12))
        for _ in range(n_gd):
            if use_softmax:
                a = gd_step_softmax(a, temperature, lr)
            else:
                a = gd_step(a, top_k, lr)

        score = float(evaluate_sequence(a.tolist()))

        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            no_improve = 0
            step_count += 1
            if step_count % 20 == 0 and verbose:
                print(f"    Improved: score={best_score:.6f} (top_k={top_k}, lr={lr:.5f})")
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            no_improve = 0
            strategy = int(rng.integers(0, 6))

            if strategy == 0:
                lr = max(1e-6, lr * float(rng.uniform(0.3, 0.7)))
            elif strategy == 1:
                top_k = int(rng.integers(3, 60))
                use_softmax = rng.random() < 0.3
                if use_softmax:
                    temperature = float(rng.uniform(0.005, 0.1)) * np.std(
                        np.convolve(best, best, mode='full'))
            elif strategy == 2:
                intensity = float(rng.uniform(0.2, 0.8))
                best = block_perturb(best, rng, intensity)
                best_score = float(evaluate_sequence(best.tolist()))
            elif strategy == 3:
                intensity = float(rng.uniform(1.0, 2.5))
                best = block_perturb(best, rng, intensity)
                best_score = float(evaluate_sequence(best.tolist()))
            elif strategy == 4:
                # Anti-symmetry perturbation: shift mass
                shift = int(rng.integers(1, max(2, n // 50)))
                best = np.roll(best, shift)
                s = np.sum(best)
                best = best / s if s > 0 else np.ones(n) / n
                best_score = float(evaluate_sequence(best.tolist()))
            else:
                # Soft reset: blend with fresh random
                fresh = make_init(n, rng, int(rng.integers(0, 8)))
                alpha = float(rng.uniform(0.3, 0.7))
                best = alpha * best + (1 - alpha) * fresh
                s = np.sum(best)
                best = best / s if s > 0 else np.ones(n) / n
                best_score = float(evaluate_sequence(best.tolist()))
                if verbose:
                    print(f"    Reset: score={best_score:.6f}")

        # Adaptive: increase exploration pressure over time
        if no_improve > max_no_improve // 2:
            lr = lr * 0.8

    if verbose:
        print(f"  Trial end: n={n}, score={best_score:.6f}")
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize 2n*max(convolve(a,a))/(sum(a))^2 using analytical gradient descent
    with top-k/softmax autocorrelation weighting, diverse initializations,
    and adaptive block perturbations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Get provided sequence
    try:
        provided = np.array(height_sequence_1, dtype=np.float64)
        if provided.ndim != 1 or provided.size == 0 or not np.isfinite(provided).all():
            raise ValueError("bad shape")
        provided = np.clip(provided, 0.0, 1000.0)
    except Exception:
        provided = None

    # Global best tracking
    global_best = None
    global_best_score = float('inf')

    # Build list of trials: (sequence, label)
    trials = []

    if provided is not None and len(provided) > 20:
        n_p = len(provided)
        s = np.sum(provided)
        a_norm = provided / s if s > 0 else np.ones(n_p) / n_p
        trials.append((a_norm, "provided"))
    else:
        trials.append((np.ones(512) / 512, "uniform_512"))

    # Add diverse trials with different lengths and shapes
    lengths = []
    if provided is not None:
        n_p = len(provided)
        lengths = [n_p, n_p // 2, n_p * 2 if n_p * 2 < 20000 else n_p]
    else:
        lengths = [512, 1024, 2048]

    for n_try in lengths:
        n_try = max(32, min(n_try, 16384))
        for method in [1, 2, 3, 4, 6]:
            a = make_init(n_try, rng, method)
            trials.append((a, f"n{n_try}_m{method}"))

    # If provided length is unique, also add some medium-length trials
    has_medium = any(200 < len(t[0]) < 4000 for t in trials)
    if not has_medium:
        for n_try in [512, 1024, 2048]:
            a = make_init(n_try, rng, int(rng.integers(0, 8)))
            trials.append((a, f"n{n_try}_rnd"))

    # Shuffle trials (but keep provided first)
    if len(trials) > 1:
        rest = trials[1:]
        rng.shuffle(rest)
        trials = [trials[0]] + rest

    print(f"Total trials: {len(trials)}, budget: {budget_s}s")
    total_time = deadline - start_time
    time_per_trial = max(15.0, total_time / len(trials))

    for i, (a_init, label) in enumerate(trials):
        if time.monotonic() >= deadline:
            break

        # Allocate time: first trial gets more, later ones share remaining
        remaining = deadline - time.monotonic()
        n_remaining_trials = len(trials) - i
        trial_time = min(time_per_trial, remaining / n_remaining_trials)
        # Give at least 10 seconds
        trial_time = max(10.0, trial_time)
        # Cap at 400s per trial
        trial_time = min(trial_time, 400.0)
        trial_deadline = time.monotonic() + trial_time

        if i == 0:
            # Give the provided sequence extra time
            trial_deadline = time.monotonic() + min(600.0, remaining * 0.4)

        print(f"\nTrial {i+1}/{len(trials)}: {label}, time={trial_time:.0f}s")

        best_seq, best_sc = optimize_trial(a_init, trial_deadline, rng, verbose=True)

        if best_sc < global_best_score:
            global_best = best_seq.copy()
            global_best_score = best_sc
            print(f"  *** NEW GLOBAL BEST: {global_best_score:.6f} ***")

    # Final: if we still have time, do a focused optimization on the global best
    if global_best is not None and time.monotonic() < deadline - 30:
        print(f"\nFinal polish on best (score={global_best_score:.6f})...")
        polish_deadline = deadline
        global_best, global_best_score = optimize_trial(
            global_best, polish_deadline, rng, verbose=True)

    if global_best is None:
        global_best = np.ones(512) / 512
        global_best_score = float(evaluate_sequence(global_best.tolist()))

    elapsed = time.monotonic() - start_time
    print(f"\nFinal answer: n={len(global_best)}, score={global_best_score:.6f}, time={elapsed:.0f}s")
    return [float(x) for x in global_best]
