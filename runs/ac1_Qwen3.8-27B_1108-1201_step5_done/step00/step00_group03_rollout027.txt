def project_simplex(a):
    """Project a onto the probability simplex {a: a_i >= 0, sum(a_i) = 1}."""
    # Algorithm from Duchi et al. (2008)
    n = len(a)
    if np.sum(a) <= 0:
        return np.ones(n) / n
    b = np.zeros(n)
    rsorted = np.sort(a)[::-1]
    cssorted = np.cumsum(rsorted)
    rho = np.where(rsorted * np.arange(1, n+1) > (cssorted - 1))[0][-1]
    theta = (cssorted[rho] - 1) / (rho + 1.0)
    b = np.maximum(a - theta, 0)
    return b

    return c_max + np.log(np.sum(np.exp(alpha * (c - c_max)))) / alpha

    w = np.exp(c_shifted)
    w /= w.sum()

    n = len(a)
