```python
"""
Minimize  2*n*max(convolve(a,a)) / sum(a)^2  over non-negative a.

The objective is scale invariant, so we fix sum(a)=1 and minimize max(C),
C=convolve(a,a).  A monotone ramp already gives ~1.34..1.45 (approaching 4/3),
which alone beats the 1.5027 target.  We then run a projected-gradient method
on a log-sum-exp smooth surrogate of max(C) (temperature annealed up) with an
FFT-based gradient and simplex projection, from several initializations and
several lengths, keeping the best exact score found.
"""
import numpy as np
import time


def _ramp(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _exp_init(n, r):
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _proj_simplex(v, s=1.0):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho = int(np.nonzero(u * idx > (cssv - s))[0][-1])
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _next_pow2(x):
    m = 1
    while m < x:
        m <<= 1
    return m


def _convolve_fft(a):
    n = a.size
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:2 * n - 1]


def _lse_and_grad(a, t):
    n = a.size
    C = _convolve_fft(a)
    shift = C.max()
    z = t * (C - shift)
    ez = np.exp(z)
    s = ez.sum()
    val = float(shift + np.log(s) / t)          # log-sum-exp of t*C
    p = ez / s
    # p has length 2n-1, a has length n
    L = _next_pow2(3 * n - 2)
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    corr_pa = np.fft.irfft(P * np.conj(A), L)[:n]   # (p *conv* a)
    corr_ap = np.fft.irfft(A * np.conj(P), L)[:n]   # (a *conv* p)
    grad = corr_pa + corr_ap
    return val, grad


def _optimize(a0, t0, t1, steps_df, big_steps):
    a = a0.copy()
    cur = _lse_and_grad(a, t0)[0]
    for big in range(big_steps):
        frac = big / max(1, big_steps - 1)
        t = t0 + (t1 - t0) * frac
        for _ in range(steps_df):
            val, grad = _lse_and_grad(a, t)
            norm = np.max(np.abs(grad))
            if norm <= 0:
                break
            # backtracking line search
            lr = 0.5 / norm
            accepted = False
            for _ in range(12):
                a_try = _proj_simplex(a - lr * grad)
                v_try, _ = _lse_and_grad(a_try, t)
                if v_try <= val + 1e-15:
                    a = a_try
                    val = v_try
                    cur = val
                    accepted = True
                    break
                lr *= 0.5
            if not accepted:
                break
    return a, cur
```
