    # Small perturbation to break degeneracy
    rng = np.random.default_rng(12345)
    a = a + 0.001 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    # FFT size (need >= 3n-2 for gradient conv)
    fsize = 1
    while fsize < 4 * n:
        fsize *= 2

    # Initial peak
    conv = _convolve_fft(a, fsize)
    M = conv.max()

    print(f"Start: n={n}, peak={M:.8e}, score={best_score:.6f}")

    iteration = 0
    stall_count = 0
    last_score_check = time.monotonic()
    use_lp = True
    lp_fail_streak = 0
    beta = 80.0

    while time.monotonic() < deadline:
        if use_lp:
            b = _lp_direction(a, M, timeout=60)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-12:
                    # Normalize direction to match sum
                    g = b / b_sum  # sum(g) = 1, same as sum(a)
                    # Bisection for best blending parameter t
                    lo, hi = 0.0, 1.0
                    for _ in range(22):
                        mid = 0.5 * (lo + hi)
                        a_cand = (1.0 - mid) * a + mid * g
                        p = _convolve_fft(a_cand, fsize).max()
                        if p <= M + 1e-14:
                            lo = mid
                        else:
                            hi = mid
                    t = lo
                    if t > 1e-8:
                        a = (1.0 - t) * a + t * g
                        conv = _convolve_fft(a, fsize)
                        M = conv.max()
                        lp_fail_streak = 0
                        stall_count = 0
                    else:
                        lp_fail_streak += 1
                else:
                    lp_fail_streak += 1
            else:
                lp_fail_streak += 1

            if lp_fail_streak >= 8:
                use_lp = False
                lp_fail_streak = 0
                print(f"Iter {iteration}: LP failing, switching to gradient")
        else:
            # Soft-max gradient step
            conv = _convolve_fft(a, fsize)
            M = conv.max()
            grad = _soft_gradient(a, beta, fsize)

            improved = False
            for lr in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
                a_cand = a - lr * grad
                a_cand = np.maximum(a_cand, 0.0)
                s = a_cand.sum()
                if s < 1e-12:
                    continue
                a_cand = a_cand / s
                p = _convolve_fft(a_cand, fsize).max()
                if p < M:
                    a = a_cand
                    M = p
                    improved = True
                    break

            if not improved:
                # Perturb to escape
                a += 0.003 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = _convolve_fft(a, fsize)
                M = conv.max()
                # Sometimes switch back to LP
                if rng.random() < 0.3:
                    use_lp = True
                    lp_fail_streak = 0

        iteration += 1

        # Periodic scoring (every ~5 seconds of wall time)
        if time.monotonic() - last_score_check > 5.0:
            last_score_check = time.monotonic()
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"Iter {iteration}: score={best_score:.6f}, peak={M:.6e}")

    # Final evaluation
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = a.copy()

    print(f"Final: score={best_score:.6f}, iterations={iteration}")
    return [float(x) for x in best_a]
