                last_print = now
        else:
            no_improve += 1

        # Softmax probabilities (smooth approximation of argmax)
        conv_shift = conv - conv.max()
        conv_shift = np.maximum(conv_shift, -40.0 / T)
        exp_vals = np.exp(T * conv_shift)
        p_sum = exp_vals.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / p_sum

        # Gradient via FFT cross-correlation
        grad = _compute_grad(a, p)
        grad_centered = grad - grad.mean()

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad_centered
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum > 1e-10:
            a = a_new / a_sum
        else:
            a = a.copy()

        # Parameter schedule
        T = min(80.0, T * 1.0005)
        lr = max(0.0003, lr * 0.9997)

        if no_improve > 400:
            # Restart with perturbation of best
            scale = 0.03 + 0.02 * (no_improve / 1000.0)
            a = best_local + scale * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            else:
                a = best_local.copy()
            momentum = np.zeros(n)
            no_improve = 0
            T = max(1.0, T * 0.5)
            lr = max(lr_init * 0.5, lr)

        iter_count += 1

    score = 2.0 * n * best_local_max
    return best_local, score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-start gradient-based optimization to minimize the step function
    evaluation function 2n*max(convolve(a,a))/(sum(a))^2.

    Uses softmax-smoothed gradient with momentum, multiple initializations,
    and multiple sequence lengths to find a low-scoring sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    total_time = deadline - time.monotonic()

    # Collect starting sequences with different structures
    starts = []

    # Given sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a0 = np.array(height_sequence_1, dtype=np.float64)
        if a0.ndim == 1 and len(a0) > 0 and np.isfinite(a0).all():
            a0 = np.clip(a0, 0, 1000)
            if a0.sum() > 0.01:
                starts.append(('given', a0, 0.35))

    n_base = 2535

    # U-shaped: high at edges, low in middle
    i = np.arange(n_base, dtype=np.float64)
    u_shape = ((i - n_base / 2.0) / (n_base / 2.0)) ** 2 + 0.05
    starts.append(('u_shape', u_shape, 0.20))

    # Random
    starts.append(('random', rng.random(n_base) + 0.05, 0.15))

    # Longer uniform
    n_long = 5000
    starts.append(('uniform_5k', np.ones(n_long), 0.10))

    # Bimodal: two peaks
    n_med = 3000
    i2 = np.arange(n_med, dtype=np.float64)
    bimodal = np.exp(-((i2 - n_med * 0.2) / (n_med * 0.08)) ** 2) + \
              np.exp(-((i2 - n_med * 0.8) / (n_med * 0.08)) ** 2) + 0.01
    starts.append(('bimodal', bimodal, 0.10))

    best_score = float('inf')
    best_a = None

    # Phase 1: Multi-start exploration
    print(f"Phase 1: Multi-start exploration ({total_time:.1f}s budget)")
    for name, start_a, frac in starts:
        if time.monotonic() >= deadline - 5.0:
            break
        alloc = frac * total_time
        start_deadline = time.monotonic() + alloc
        print(f"  Optimizing '{name}' (n={len(start_a)}, budget={alloc:.1f}s)")

        result_a, result_score = _optimize_sequence(start_a, start_deadline, rng)

        if result_a is not None and result_score < best_score:
            best_score = result_score
            best_a = result_a.copy()
            print(f"  -> New best: score={best_score:.6f} (n={len(best_a)}, from '{name}')")

    if best_a is None:
        best_a = np.ones(256)
        best_score = evaluate_sequence(best_a.tolist())

    # Phase 2: Fine-tuning with tighter parameters
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        print(f"Phase 2: Fine-tuning best sequence ({remaining:.1f}s)")
        tune_deadline = time.monotonic() + remaining * 0.7

        result_a, result_score = _optimize_sequence(
            best_a, tune_deadline, rng, T_init=5.0, lr_init=0.02)

        if result_a is not None and result_score < best_score:
            best_score = result_score
            best_a = result_a.copy()
            print(f"  -> Fine-tuned score: {best_score:.6f}")

    # Phase 3: Try to improve with a very low learning rate polish
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        print(f"Phase 3: Polish ({remaining:.1f}s)")
        polish_deadline = time.monotonic() + remaining * 0.5

        result_a, result_score = _optimize_sequence(
            best_a, polish_deadline, rng, T_init=20.0, lr_init=0.005)

        if result_a is not None and result_score < best_score:
            best_score = result_score
            best_a = result_a.copy()
            print(f"  -> Polished score: {best_score:.6f}")

    # Final verification
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Final verified score: {final_score:.6f} (n={len(best_a)})")
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_a = np.ones(256)
        best_score = evaluate_sequence(best_a.tolist())

    return [float(x) for x in best_a]
