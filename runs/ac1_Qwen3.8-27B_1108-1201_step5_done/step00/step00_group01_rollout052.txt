import time
from scipy.optimize import linprog
from math import comb

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """LP-based iterative minimization of max self-convolution."""
    n = len(height_sequence_1) if 'height_sequence_1' in dir() else 6121
    
    # Initialize with central binomial coefficients
    a = np.zeros(n)
    for k in range(n):
        a[k] = comb(2*k, k) / (4**k)
    a = a / a.sum()
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score}")
    
    deadline = time.monotonic() + budget_s - 10
    
    iteration = 0
    while time.monotonic() < deadline:
        t_start = time.monotonic()
        
        # Compute current max convolution
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()
        
        # Set up LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
        n_vars = n
        n_constraints = 2 * n - 1
        
        # Objective: maximize sum(b) => minimize -sum(b)
        c = -np.ones(n_vars)
        
        # Constraint matrix A_ub: (a*b)[k] <= M
        # A_ub[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
        A_ub = np.zeros((n_constraints, n_vars))
        for k in range(n_constraints):
            for j in range(max(0, k - n + 1), min(k + 1, n)):
                A_ub[k, j] = a[k - j]
        
        b_ub = np.full(n_constraints, M)
        
        # Solve LP
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        
        if not result.success:
            break
        
        b = result.x
        b_sum = b.sum()
        
        if b_sum <= 0:
            break
        
        # Rescale b to match sum(a)
        a_sum = a.sum()
        b_scaled = b * (a_sum / b_sum)
        
        # Check if b_scaled improves
        conv_bb = np.convolve(b_scaled, b_scaled)
        max_bb = conv_bb.max()
        
        if max_bb < M:
            # Try direct replacement
            t = 1.0
        else:
            # Use soft update
            t = 0.1
        
        a_new = (1 - t) * a + t * b_scaled
        a_new = np.maximum(a_new, 0)
        
        # Evaluate

        max_new = conv_new.max()
        
        # Score (scale-invariant, so just 2n * max / sum^2)
        score_new = 2 * n * max_new / (a_new.sum()**2)
        
        if score_new < best_score:
            best = a_new.copy()
            best_score = score_new
            print(f"Iter {iteration}: score = {best_score}, M = {M}, max_new = {max_new}")
        
        a = a_new

        grad /= len(active)
        
        # Exponentiated gradient update
        a = a * np.exp(-lr * grad)
        a = a / a.sum()
        
        # Evaluate
        conv_new = np.convolve(a, a)
        score = 2 * n * conv_new.max() / (a.sum()**2)
        if score < best_score:
            best = a.copy()
            best_score = score
            if it % 500 == 0:
                print(f"Iter {it}: score = {best_score:.6f}, max_conv = {max_conv:.6f}")
        
        # Adaptive learning rate
        if it % 1000 == 0:
            lr *= 0.8
    
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"Final score: {final_score:.6f}")
    return [float(x) for x in best]
