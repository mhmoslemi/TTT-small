"""
Minimize  2*n*max(conv(a,a)) / (sum a)^2  over non-negative sequences a.

Key construction: a[i] = 1 + A*sin(2*pi*h*(i+c)/n) (and raised-cosine variants).
Rationale: the all-ones sequence gives exactly 2.0 with a triangular
autocorrelation peaking at n.  A single full-period sinusoidal modulation that
is (anti-)symmetric about the centre cancels the linear terms at the centre lag,
cutting the centre peak from n down to n*(1-A^2/2) while keeping the edge peaks
O(1).  For A near 1 this pushes the objective well below 2 (toward the
asymptotic lower bound 2n/(2n-1) ~ 1).  We (A) sweep this parametric family at
several lengths using a fast FFT-based score, then (B) refine the best seeds with
a projected steep-descent step that uses the (wrong-direction-only-matters)
subgradient of the soft-max-smoothed max-conv term, plus (C) cheap multiplicative
block local search.  The exact evaluator is only used to confirm the final answer.
"""

import time
import numpy as np


# --------------------------------------------------------------------------- #
# Fast score (O(n log n)) that is exactly equal to evaluate_sequence for
# non-negative, finite, in-range inputs (no clamping / no tiny-sum triggered).
# --------------------------------------------------------------------------- #
def _next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def conv_aa(a):
    """Full linear autocorrelation of 1-D real array a (length 2n-1) via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    out = np.fft.irfft(A * A, size)
    return out[:2 * n - 1]


def fast_score(a):
    """2*n*max(conv)/(sum)^2 ; inf on degenerate input."""
    s = float(np.sum(a))
    if s < 0.01 or not np.isfinite(s):
        return float("inf")
    c = conv_aa(a)
    return float(2.0 * a.size * float(np.max(c)) / (s * s))


def _lin_conv(U, V):
    n = U.size + V.size - 1
    size = _next_pow2(n)
    return np.fft.irfft(np.fft.rfft(U, size) * np.fft.rfft(V, size), size)[:n]


def smooth_value(a, t):
    c = conv_aa(a)
    M = float(np.max(c))
    return M + float(np.log(np.sum(np.exp(t * (c - M))))) / t


def smooth_grad(a, t):
    """Gradient of (1/t) logsumexp(t*conv(a,a)) w.r.t. a (FFT, O(n log n))."""
    n = a.size
    c = conv_aa(a)
    M = float(np.max(c))
    p = np.exp(t * (c - M))
    p /= p.sum()
    q = a[::-1]
    g = np.zeros(n)
    g += _lin_conv(p, a)[:n]                 # (p*a)[j]
    g += _lin_conv(p, q)[n - 1:n - 1 + n]     # (p*q)[n-1+j]
    return g


def project(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0:
        return a
    return a / s


def subgrad_step(a, step):
    """One projected step using the hard-max subgradient (always valid)."""
    n = a.size
    c = conv_aa(a)
    k = int(np.argmax(c))
    g = np.zeros(n)
    j1s = max(0, k - n + 1)
    j1e = min(n, k + 1)
    if j1s < j1e:
        g[j1s:j1e] += a[k - np.arange(j1s, j1e)]
    j2s = max(0, k)
    j2e = min(n, k + n)
    if j2s < j2e:
        g[j2s:j2e] += a[np.arange(j2s, j2e) - k]
    a = project(a - step * g)
    return a


def selftest_smooth():
    n = 16
    rng = np.random.default_rng(0)
    a = rng.random(n) + 0.2
    t = 6.0
    eps = 1e-5
    gs = smooth_grad(a, t)
    fd = np.zeros(n)
    for j in range(0, n, 3):
        e = np.zeros(n); e[j] = eps
        fd[j] = (smooth_value(a + e, t) - smooth_value(a - e, t)) / (2 * eps)
    rel = np.max(np.abs(gs - fd)) / (np.max(np.abs(fd)) + 1e-12)
    print("[selftest] smooth-grad rel err =", rel)
    return rel < 1e-3


def gen_sine(n, A, h, c0):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + A * np.sin(2.0 * np.pi * h * (i + c0) / n)
    return np.clip(a, 0.0, 1000.0)


def gen_raised(n, A, h, c0):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * A * (1.0 - np.cos(2.0 * np.pi * h * (i + c0) / n))
    return np.clip(a, 0.0, 1000.0)


def parametric_search(deadline):
    best_a, best_s = None, float("inf")
    lengths = [1024, 2048, 4096, 6144, 8192]

def block_perturb(base, rng):
    n = base.size
    c = base.astype(np.float64).copy()
    w = int(rng.integers(max(1, n // 400), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, st + w)
    c[st:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return np.clip(c, 0.0, 1000.0)


def refine(a, till, use_smooth):
    """Polish a seed with steep-descent (sub- or smooth-grad) then local search."""
    n = a.size
    a = project(a.copy())
    local_best = a.copy()
    local_bv = fast_score(a)
    it = 0
    base_lr = 0.5
    while time.monotonic() < till and it < 200000:
        if use_smooth:
            t = min(40.0, 2.0 + 0.02 * it)
            g = smooth_grad(a, t)
            g = g - np.mean(g) * a      # remove gauge component (scale invariance)
        else:
            a = subgrad_step(a, base_lr / (1.0 + it * 0.01))
            it += 1
            continue
        for lr in (base_lr / (1.0 + it * 0.01)):
            cand = project(a - lr * g)
            s = fast_score(cand)
            if s < local_bv:
                a = cand
                local_bv = s
                local_best = a.copy()
        it += 1
    # local search
    rng = np.random.default_rng(12345)
    while time.monotonic() < till:
        cand = block_perturb(local_best, rng)
        s = fast_score(cand)
        if s < local_bv:
            local_bv = s
            local_best = cand.copy()
    return local_best, local_bv
