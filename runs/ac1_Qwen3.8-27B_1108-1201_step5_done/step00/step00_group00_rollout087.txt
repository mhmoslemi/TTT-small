def spectral_square_root(C, A):
    root_direct = np.sqrt(np.abs(C)) * np.exp(0.5j*np.angle(C))
    denom = A*A
    absdenom = np.abs(denom)
    # scale eps based on max

    out[0] = 1.0  # A DC sum should be 1

def spectral_root(C, A):
    absC=np.abs(C)
    root=np.sqrt(absC)
    phase=np.angle(C)
    out=root*np.exp(0.5j*phase)
    denom=A*A
    absdenom=np.abs(denom)
    scale_max = np.max(absdenom) if absdenom.size else 0.0
    eps = 1e-18 * max(1.0, float(scale_max))
    mask = absdenom > eps
    if np.any(mask):
       ratio = C[mask]/denom[mask]
       out[mask] = A[mask] * np.sqrt(np.abs(ratio)) * np.exp(0.5j*np.angle(ratio))
    out[0] = 1.0
    # Remove tiny imaginary at DC
    out[0] = complex(1.0,0.0)
    return out

def spectral_root(C, A):
    absC = np.abs(C)
    root = np.sqrt(absC)
    phase = np.angle(C)
    out = root * np.exp(0.5j * phase)
    denom = A * A
    absdenom = np.abs(denom)
    if absdenom.size:
        scale = float(np.max(absdenom))
    else:
        scale = 1.0
    eps = 1e-18 * max(1.0, scale)
    mask = absdenom > eps
    if np.any(mask):
        ratio = C[mask] / denom[mask]
        out[mask] = A[mask] * np.sqrt(np.abs(ratio)) * np.exp(0.5j * np.angle(ratio))
    out[0] = 1.0 + 0.0j
    return out

"""
Search for a nonnegative step-function sequence by:
1) alternating c-domain peak clipping with spectral square-root/support projections,
2) polishing with a multiplicative gradient method,
3) finishing with random local search.
The objective is 2*n*max(conv(a,a))/(sum(a)**2), and candidate sequences are kept normalized.
"""

import numpy as np
import time


def next_fast_length(m):
    m = max(int(m), 1)
    k = 1
    while k < m:
        k <<= 1
    return k


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    if s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolution_fft(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2 * a.size - 1]


def fast_score(a, N):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf
    c = convolution_fft(a, N)
    m = float(np.max(c))
    return float(2.0 * a.size * m / (s * s))


def spectral_root(C, A):
    absC = np.abs(C)
    root = np.sqrt(absC)
    phase = np.angle(C)
    out = root * np.exp(0.5j * phase)

    denom = A * A
    absdenom = np.abs(denom)
    scale = float(np.max(absdenom)) if absdenom.size else 1.0
    eps = 1e-18 * max(1.0, scale)
    mask = absdenom > eps
    if np.any(mask):
        ratio = C[mask] / denom[mask]
        out[mask] = A[mask] * np.sqrt(np.abs(ratio)) * np.exp(0.5j * np.angle(ratio))

    out[0] = 1.0 + 0.0j
    return out


def dynamic_spectral(a, n, N, clip_score, time_end, seed):
    a = normalize_positive(a)
    n = a.size
    best = a.copy()
    best_score = fast_score(a, N)
    cur_score = best_score
    rng = np.random.default_rng(seed)

    if n > 1 and np.allclose(a, a[0], rtol=1e-12, atol=1e-12):
        a = normalize_positive(a * (1.0 + 0.05 * rng.standard_normal(n)))
        cur_score = fast_score(a, N)
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

    clip_val = max(1e-12, float(clip_score) / (2.0 * float(n)))
    alpha = 0.10
    stall = 0
    it = 0

    while time.monotonic() < time_end:
        c = convolution_fft(a, N)
        c = np.maximum(c, 0.0)
        s = float(np.sum(c))
        if s <= 1e-15:
            break
        c = c / s
        m = float(np.max(c))

        M = clip_val
        if M > m:
            M = 0.98 * m

        c_clip = None
        sc = 0.0
        for _ in range(10):
            c_clip = np.minimum(c, M)
            sc = float(np.sum(c_clip))
            if sc > 1e-10:
                break
            M *= 1.25
            if M > 2.0 * m:
                break

        if sc <= 1e-10:
            c_d = c.copy()
        else:
            c_d = c_clip / sc

        A = np.fft.rfft(a, N)
        C = np.fft.rfft(c_d, N)
        A_root = spectral_root(C, A)

        b = np.fft.irfft(A_root, N)[:n]
        b = np.maximum(b, 0.0)
        sb = float(np.sum(b))
        if sb <= 1e-12:
            stall += 1
            if stall > 25:
                alpha = max(0.05, 0.7 * alpha)
                stall = 0
            it += 1
            continue

        b = b / sb
        alpha = min(1.0, alpha + 0.0015)
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = normalize_positive(a_new)

        scand = fast_score(a_new, N)
        if scand < best_score:
            best = a_new.copy()
            best_score = scand

        if it < 1000:
            tol = 0.20
        elif it < 5000:
            tol = 0.05
        else:
            tol = 0.02

        if scand <= cur_score + tol:
            a = a_new
            cur_score = scand
            stall = 0
        else:
            stall += 1
            if stall > 35:
                clip_val = min(m, clip_val * 1.10 + 1e-12 / float(n))
                a = normalize_positive(a * (1.0 + 0.03 * rng.standard_normal(n)))
                cur_score = fast_score(a, N)
                if cur_score < best_score:
                    best = a.copy()
                    best_score = cur_score
                stall = 0
                alpha = max(0.10, alpha * 0.70)

        it += 1

    return best


def gradient_refine(a, N, time_end, seed):
    a = normalize_positive(a)
    n = a.size
    best = a.copy()
    best_score = fast_score(a, N)
    cur_score = best_score
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() < time_end:
        c = convolution_fft(a, N)
        c = np.maximum(c, 0.0)
        s = float(np.sum(c))
        if s <= 1e-15:
            break
        c = c / s
        m = float(np.max(c))
        if m <= 1e-15:
            break

        mean_c = float(np.mean(c))
        denom = max(m - mean_c, 1e-3 / float(n))
        scale = 12.0 / denom
        weights = scale * (c - m)
        ex = np.exp(weights)
        p = ex / np.sum(ex)

        ar = a[::-1]
        P = np.fft.rfft(p, N)
        AR = np.fft.rfft(ar, N)
        conv = np.fft.irfft(P * AR, N)
        g = 2.0 * conv[n - 1:n + n]

        d = -g
        d = d - np.mean(d)
        norm = float(np.sqrt(np.mean(d * d)))
        if norm < 1e-12:
            d = rng.standard_normal(n)
            d = d - np.mean(d)
            norm = float(np.sqrt(np.mean(d * d)))
        if norm <= 1e-12:
            it += 1
            continue
        d = d / norm

        accepted = False
        for t in (0.25, 0.125, 0.0625, 0.03125):
            loga = np.log(np.maximum(a, 1e-300) + 1e-300) + t * d
            cand = np.exp(loga - np.max(loga))
            cand = normalize_positive(cand)
            sc = fast_score(cand, N)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
            tol = 0.05 if it < 1500 else 0.01
            if sc <= cur_score + tol:
                a = cand
                cur_score = sc
                accepted = True
                break

        if not accepted:
            cand = normalize_positive(a * (1.0 + 0.02 * rng.standard_normal(n)))
            sc = fast_score(cand, N)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
            if sc <= cur_score + 0.02:
                a = cand
                cur_score = sc

        it += 1

    return best


def random_local_search(a, N, time_end, seed):
    a = normalize_positive(a)
    n = a.size
    best = a.copy()
    best_score = fast_score(a, N)
    cur_score = best_score
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)

    while time.monotonic() < time_end:
        cand = a.copy()
        op = int(rng.integers(0, 5))

        if op == 0:
            w = int(rng.integers(max(1, n // 250), max(2, n // 12)))
            st = int(rng.integers(0, max(1, n - w + 1)))
            cand[st:st + w] *= np.exp(rng.normal(0.0, 0.25))
        elif op == 1:
            st = int(rng.integers(-n // 10, n // 10 + 1))
            cand = np.roll(cand, st)
        elif op == 2:
            m = int(rng.integers(1, 25))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            cand *= (1.0 + 0.20 * np.cos(np.pi * m * x + ph))
        elif op == 3:
            al = float(rng.uniform(0.20, 0.55))
            cand = (1.0 - al) * cand + al * cand[::-1]
        else:
            cand *= np.exp(0.02 * rng.standard_normal(n))

        cand = normalize_positive(cand)
        sc = fast_score(cand, N)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
        if sc <= cur_score + 0.015:
            a = cand
            cur_score = sc

    return best


def init_ones(n):
    return normalize_positive(np.ones(n, dtype=np.float64))


def init_power(n, power, reverse=False):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(x, float(power))
    if reverse:
        a = a[::-1]
    return normalize_positive(a)


def init_cos(n, alpha, beta, gam):
    x = np.linspace(0.0, 1.0, n)
    a = (
        1.0
        + alpha * np.cos(np.pi * x)
        + beta * np.cos(2.0 * np.pi * x)
        + gam * np.sin(np.pi * x)
    )
    return normalize_positive(a)


def init_random(n, seed, sigma):
    rng = np.random.default_rng(seed)
    a = np.exp(rng.normal(0.0, float(sigma), n))
    return normalize_positive(a)


def exact_update(current_seq, current_score, cand):
    if cand is None:
        return current_seq, current_score
    arr = np.asarray(cand, dtype=np.float64).ravel()
    if arr.size == 0:
        return current_seq, current_score
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if s < 0.01:
        if s > 1e-12:
            arr = arr / s
        else:
            return current_seq, current_score
    lst = [float(v) for v in arr]
    sc = float(evaluate_sequence(lst))
    if np.isfinite(sc) and sc < current_score:
        return lst, sc
    return current_seq, current_score
