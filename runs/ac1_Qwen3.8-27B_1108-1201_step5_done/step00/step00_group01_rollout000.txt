```python
import time

def spectral_flatten(a, n_iterations=100):
    """Use spectral method to flatten the linear convolution."""
    n = len(a)
    L = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    L_fft = 1
    while L_fft < L:
        L_fft *= 2
    
    target_mag = 1.0 / np.sqrt(L)  # Since sum(a) = 1, target sum of conv = 1, each entry = 1/L
    
    for it in range(n_iterations):
        # Zero-pad and compute FFT
        a_pad = np.zeros(L_fft)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        # Set magnitudes to target
        phases = np.angle(A)
        A_new = target_mag * np.exp(1j * phases)
        
        # Inverse FFT
        a_new = np.fft.ifft(A_new).real
        
        # Take first n elements, clip to non-negative
        a = np.clip(a_new[:n], 0, None)
        
        # Renormalize to sum = 1
        s = a.sum()
        if s > 1e-10:
            a /= s
        else:
            break
    
    return a
```

```python
def spectral_iterate(a, L, n_iter, relax=1.0, target='uniform'):
    """

    - L: FFT length (should be >= 2n-1)
    - n_iter: number of iterations
    - relax: relaxation parameter (0 < relax <= 1)
    - target: 'uniform' for flat magnitude target
    
    Returns: optimized sequence
    """

    a_work = a.copy().astype(np.float64)
    target_mag = 1.0 / np.sqrt(L)

        # Forward FFT
        a_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a_work
```
