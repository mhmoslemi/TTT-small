"""
Optimizes non-negative step function to minimize 2n*max(a*a)/(sum(a))^2
using multiplicative subgradient descent, LP-based direction finding,
antisymmetric perturbations, and random restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def self_convolve(a):
    """Compute a * a (linear convolution with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = rfft(a, fft_size)
    return irfft(fa * fa, fft_size)[:size]


def compute_objective(a):
    """Compute 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = self_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))


def normalize(a):
    """Normalize a to sum to 1, clipping to non-negative."""
    a = np.clip(a, 0.0, 1000.0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a


def build_conv_matrix(f):
    """Build (2n-1) x n sparse matrix A such that A @ g = f * g."""
    n = len(f)
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A.tocsr()


def lp_direction(f, M, active_frac=0.05):
    """
    Find g maximizing sum(g) subject to (f*g)_k <= M for active constraints, g >= 0.
    Returns normalized g or None if LP fails.
    """
    n = len(f)
    conv = self_convolve(f)
    # Find active set: top active_frac fraction of constraints
    n_active = max(10, int(len(conv) * active_frac))
    top_idx = np.argpartition(conv, -n_active)[-n_active:]
    
    A_full = build_conv_matrix(f)
    A = A_full[top_idx, :].toarray()
    b = np.full(len(top_idx), M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.success and res.x is not None:
        g = np.clip(res.x, 0.0, None)
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None


def mul_subgradient_step(a, lr, top_k=10, soft_tau=0.0):
    """
    One step of multiplicative subgradient descent on max(a*a).
    Returns (new_a, convolution).
    """
    n = len(a)
    c = self_convolve(a)
    
    if soft_tau > 0:
        c_scaled = c / soft_tau
        c_max = np.max(c_scaled)
        w = np.exp(c_scaled - c_max)
        w /= np.sum(w)
        if top_k < len(c) and top_k > 0:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            w_masked = np.zeros_like(w)
            w_masked[top_idx] = w[top_idx]
            w_masked /= np.sum(w_masked)
            w = w_masked
    else:
        k_star = np.argmax(c)
        w = np.zeros(len(c))
        w[k_star] = 1.0
        top_k = 1
    
    nonzero_k = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for k in nonzero_k:
        wk = w[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * wk * a[idx]
    
    a_new = a * np.exp(-lr * grad)
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = np.sum(a_new)
    if s > 0:
        a_new /= s
    return a_new, c


def antisymmetric_perturbation(n, rng):
    """Generate a random unit-norm antisymmetric vector of size n."""
    v = np.zeros(n)
    half = n // 2
    v[:half] = rng.normal(0, 1, half)
    v[half:] = -v[:half][::-1] if n % 2 == 0 else -v[:half-1][::-1]
    # Fix for odd n: center element stays 0
    if n % 2 == 1:
        v[half:] = -v[:half-1][::-1]
        v[half] = 0.0
    # Redo properly for even n
    if n % 2 == 0:
        v = np.zeros(n)
        v[:half] = rng.normal(0, 1, half)
        v[half:] = -v[:half][::-1]
    else:
        v = np.zeros(n)
        h = (n - 1) // 2
        v[:h] = rng.normal(0, 1, h)
        v[h + 1:] = -v[:h][::-1]
        v[h] = 0.0
    norm = np.linalg.norm(v)
    if norm > 0:
        v /= norm
    return v


def run_subgradient_loop(a, n_iters, lr_start, lr_end, top_k=10, soft_tau=0.0,
                         rng=None, perturb_every=200, perturb_eps=0.05):
    """Run a loop of multiplicative subgradient steps with occasional perturbations."""
    if rng is None:
        rng = np.random.default_rng()
    a = a.copy()
    best = a.copy()
    best_score = compute_objective(a)
    
    for i in range(n_iters):
        frac = i / max(1, n_iters)
        lr = lr_start * (lr_end / lr_start) ** frac
        a, c = mul_subgradient_step(a, lr, top_k=top_k, soft_tau=soft_tau)
        
        if perturb_every > 0 and i > 0 and i % perturb_every == 0:
            v = antisymmetric_perturbation(len(a), rng)
            a_new = normalize(a + perturb_eps * v)
            s_new = compute_objective(a_new)
            if s_new < best_score:
                a = a_new
                best_score = s_new
                best = a.copy()
        
        s = compute_objective(a)
        if s < best_score:
            best_score = s
            best = a.copy()
    
    return best, best_score


def try_sequence_length(a_base, target_len, rng):
    """Resize a_base to target_len via interpolation."""
    from scipy.interpolate import interp1d
    n_old = len(a_base)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, target_len)
    f_interp = interp1d(x_old, a_base, kind='cubic', fill_value=0, bounds_error=False)
    a_new = np.clip(f_interp(x_new), 0.0, None)
    s = np.sum(a_new)
    if s > 0:
        a_new /= s
    return a_new
