def _smooth_obj_grad(x, tau, n, fft_len):
    """

    log_a[0] = 0.0
    log_a[1:] = x
    a = np.exp(log_a)
    S = a.sum()
    
    # Autocorrelation via FFT
    A = np.fft.rfft(a, n=fft_len)
    r = np.fft.irfft(A * A, n=fft_len)

"""
Minimizes 2n*max(autocorr)/(sum)^2 for non-negative sequences using L-BFGS
with log-sum-exp smoothing, FFT-based autocorrelation and gradient, log-domain
parameterization (a[i]=exp(x[i])), and tau annealing across multiple initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    m = 1
    while m < n:
        m <<= 1
    return m


def _smooth_obj_grad(x, tau, n, fft_len):
    """Compute smooth objective and gradient. x has length n-1 (log a[1:]/a[0])."""
    log_a = np.empty(n)
    log_a[0] = 0.0
    log_a[1:] = x
    a = np.exp(log_a)
    S = a.sum()

    # Linear autocorrelation via FFT (convolution, NOT correlation)
    A = np.fft.rfft(a, n=fft_len)
    r = np.fft.irfft(A * A, n=fft_len)
    r = r[:2 * n - 1]

    # Log-sum-exp of r/tau
    rs = r / tau
    rs_max = rs.max()
    ex = np.exp(rs - rs_max)
    ex_sum = ex.sum()
    L = rs_max + np.log(ex_sum)
    w = ex / ex_sum

    # g[i] = sum_{j=0}^{n-1} w[i+j]*a[j] via FFT cross-correlation
    flip_a = a[::-1]
    Wf = np.fft.rfft(w, n=fft_len)
    Ff = np.fft.rfft(flip_a, n=fft_len)
    conv_wf = np.fft.irfft(Wf * Ff, n=fft_len)
    g = conv_wf[n - 1:2 * n - 1]

    # Objective: 2n * tau * L / S^2
    obj = 2.0 * n * tau * L / (S * S)

    # Gradient: df/dx[i] = 4*n*tau*a[i]/S^3 * (S*g[i] - L*a[i])
    grad_full = 4.0 * n * tau * a / (S ** 3) * (S * g - L * a)
    grad = grad_full[1:]

    return obj, grad


def _fun_scalar(x, tau, n, fft_len):
    return _smooth_obj_grad(x, tau, n, fft_len)[0]


def _jac_vec(x, tau, n, fft_len):
    return _smooth_obj_grad(x, tau, n, fft_len)[1]


def _optimize_from(a0, n, fft_len, deadline, verbose=False):
    """Run L-BFGS with tau annealing from initial sequence a0."""
    a = np.array(a0, dtype=np.float64).copy()
    a = np.clip(a, 1e-10, 1000.0)
    if a.sum() < 1e-10:
        a = np.ones(n)

    x = np.log(a[1:] / a[0]).copy()

    best_score_local = float('inf')
    best_a_local = a.copy()

    tau = 1.0
    round_num = 0
    while time.monotonic() < deadline:
        round_num += 1
        t0 = time.monotonic()
        result = minimize(
            _fun_scalar, x, args=(tau, n, fft_len),
            method='L-BFGS-B', jac=_jac_vec,
            options={'maxiter': 400, 'gtol': 1e-10, 'ftol': 1e-14}
        )
        x = result.x
        a = np.empty(n)
        a[0] = 1.0
        a[1:] = np.exp(x)
        a = np.clip(a, 0.0, 1000.0)

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score_local:
            best_score_local = score
            best_a_local = a.copy()
            if verbose:
                print(f"    Round {round_num}, tau={tau:.5f}: {score:.6f} ({time.monotonic()-t0:.1f}s)")

        tau = max(0.01, tau * 0.55)
        if tau <= 0.011 and round_num > 4:
            break

    return best_a_local
