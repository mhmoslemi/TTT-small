        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    else:
        w = np.ones(n) / n
    return w

def compute_grad_fft(a, tau, N):
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    p = exp_c / exp_c.sum()
    R = np.fft.ifft(np.conj(np.fft.fft(a, N)) * np.fft.fft(p, N)).real[:n]
    g = (2.0 / tau) * R
    return g

def build_lp_matrix(f, n):
    """Build sparse constraint matrix for the LP."""
    # Row k, col j: f[k-j] if 0 <= k-j <= n-1 and 0 <= j <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(f, M, n):
    """Solve: max sum(g) s.t. f*g <= M, g >= 0."""
    A = build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

One important improvement: instead of a fixed learning rate, use a **backtracking line search**. At each step:

2. Try step `alpha * g` with `alpha = lr`.
3. If the objective (logsumexp) increases, halve `alpha` and retry.
4. Accept the first step that decreases the objective.

This is more robust than a fixed learning rate.

Actually, let me think about what shape of `a` gives the flattest `c = a*a`.

If `a` is constant, `c` is triangular (peak at the center, linearly decreasing to 0 at the edges).
If `a` is concentrated in the middle (like a bump), `c` will have a peak at the center (where the bump overlaps with itself most).

To make `c` flat, we want the "auto-overlap" to be the same for all shifts. This is like wanting `a` to have a flat "autocorrelation" in some sense.

For continuous functions, the only function with a constant autocorrelation is a constant function (on a circle) or a Dirac delta (on the line). On a finite interval with boundary effects, it's impossible to have a perfectly flat autocorrelation.

The best we can do is to minimize the variance of `c`. This is equivalent to minimizing `sum(c[k]^2) - n/sum(c[k])^2` (the variance, scaled).

`sum(c[k]^2) = (1/2π) ∫|A(ω)|^4 dω` (by Parseval, with appropriate normalization).

So we want to minimize `∫|A(ω)|^4 dω` subject to `A(0) = 1` (since `sum(a) = 1`) and `a >= 0`.

Without the `a >= 0` constraint, the optimal would be to make `|A(ω)|^2` as flat as possible (a "box" in frequency), which corresponds to a sinc-like function in time. But sinc functions are negative in the tails, violating the non-negativity constraint.

With the non-negativity constraint, the optimum is different. Let me just rely on numerical optimization.
