import numpy as np
import time

def compute_max_convolution_delta(c, a, i, x, n):
    """Compute max_k (c[k] + delta[k]) when a[i] is changed to x.
    c is the current full convolution, a is the current sequence.
    Returns the new max value.
    """
    # delta[k] = 2*(x - a[i]) * a[k-i] for k where 0 <= k-i < n and k != 2*i
    # delta[2*i] = x^2 - a[i]^2
    
    ai = a[i]
    diff = x - ai
    
    # The affected k values are those where 0 <= k-i < n, i.e., k in [i, i+n-1]
    k_lo = i
    k_hi = i + n - 1  # inclusive
    
    if k_lo >= k_hi:  # n=1 edge case
        return c.max()
    
    # Compute the new values for affected k
    # For k in [k_lo, k_hi], k != 2*i: new_c[k] = c[k] + 2*diff*a[k-i]
    # For k = 2*i (if in range): new_c[k] = c[k] + diff^2  (since x^2 - ai^2 = (x-ai)(x+ai) = diff*(x+ai))
    
    # Unaffected k: new_c[k] = c[k]
    
    # The new max is max(max_unaffected, max_affected)
    max_unaffected = c.max()  # Upper bound, might be tightened
    
    # Actually, for efficiency, we can compute:
    # max_affected = max over k in [k_lo, k_hi] of (c[k] + 2*diff*a[k-i]) (with special case at k=2i)
    
    # The affected slice is c[k_lo:k_hi+1]
    affected = c[k_lo:k_hi+1].copy()
    
    # Add 2*diff*a[k-i] for all affected
    # a[k-i] for k in [k_lo, k_hi] is a[0:n] = a
    affected += 2 * diff * a  # Since k_lo = i, k_hi = i+n-1, so k-i goes from 0 to n-1
    
    # Fix the k=2i term (if in range)
    if 2 * i >= k_lo and 2 * i <= k_hi:
        idx = 2 * i - k_lo  # Index within the affected slice
        # The correct value is c[2*i] + x^2 - ai^2
        # We added 2*diff*a[0] = 2*diff*ai, but the correct is diff*(x+ai) = diff*x + diff*ai
        # So we need to subtract 2*diff*ai and add diff*(x+ai)
        # Net correction: diff*(x+ai) - 2*diff*ai = diff*(x - ai) = diff^2
        affected[idx] = c[2*i] + x*x - ai*ai
    
    max_affected = affected.max()
    
    # The overall max is max(max of unaffected part, max_affected)
    # Unaffected part is c except the slice [k_lo:k_hi+1]
    if k_lo == 0 and k_hi == 2*n-2:
        return max_affected
    else:
        # Need max of the unaffected part
        # This is c[:k_lo].max() and c[k_hi+1:].max()
        max_unaffected = 0
        if k_lo > 0:
            max_unaffected = max(max_unaffected, c[:k_lo].max())
        if k_hi + 1 < len(c):
            max_unaffected = max(max_unaffected, c[k_hi+1:].max())
        return max(max_unaffected, max_affected)

When changing a[i] from old to new:
- delta[k] = 2*(new - old) * a_old[k-i] for k in [i, i+n-1], k != 2i
- delta[2i] = new^2 - old^2

Where a_old[k-i] means the value of a at position (k-i) BEFORE the update. Since we're only changing a[i], and k-i ranges from 0 to n-1, the only case where a_old[k-i] = a[i] is when k-i = i, i.e., k = 2i. But we handle k=2i separately. For all other k in the affected range, a_old[k-i] = a_current[k-i] (the value before this step).

So the update is:
1. Save old = a[i]
2. Compute delta for all affected k using current a (before update)
3. Update c: c[affected] += delta
4. Update a[i] = new
5. Update sum_a: sum_a += new - old

        best_score = float(evaluate_sequence(best_sequence.tolist()))
