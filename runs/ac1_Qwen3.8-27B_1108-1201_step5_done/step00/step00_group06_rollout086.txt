def resample_base(base, n):

    if base.size==0: return None
    base=base[np.isfinite(base)]

import numpy as np
import time


"""
Find a long non-negative step-function height sequence with very small
peak full self-convolution relative to the square of its sum.

The main idea is that asymmetric shapes such as 1/(1 + lambda*x) produce
convolution coefficients that are much flatter than symmetric/uniform shapes.
This program scans a small family of such shapes at a fixed length, keeps the
best candidates by fast FFT evaluation, re-checks the top candidates exactly,
and then spends any remaining time on light multiplicative local search.
"""


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    a = a / s
    if not np.all(np.isfinite(a)):
        return np.inf

    L = 2 * n - 1
    if n <= 3072:
        c = np.convolve(a, a)
    else:
        M = next_pow2(L)
        fa = np.fft.rfft(a, M)
        c = np.fft.irfft(fa * fa, M)[:L]

    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    a = a / s
    c = np.convolve(a, a)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx)


def make_inv_linear(t, lam):
    a = 1.0 / (1.0 + float(lam) * t)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return a / s


def make_power(t, lam, r):
    base = 1.0 + float(lam) * t
    a = np.power(base, -float(r))
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return a / s


def make_exp(t, lam):
    a = np.exp(-float(lam) * t)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return a / s


def make_quadratic(t, lam, mu):
    denom = 1.0 + float(lam) * t + float(mu) * t * (1.0 - t)
    if np.min(denom) <= 0.0:
        return None
    a = 1.0 / denom
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return a / s


def resample_base(base, n):
    base = np.asarray(base, dtype=np.float64).ravel()
    if base.size == 0:
        return None
    base = base[np.isfinite(base)]
    base = np.clip(base, 0.0, None)
    if base.size == 0:
        return None
    s = float(base.sum())
    if s <= 0.0:
        return None
    base = base / s
    if base.size == 1:
        return np.full(n, 1.0 / n)
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, base)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return out / s


def consider_candidate(score, arr, top, best):
    if not np.isfinite(score):
        return
    if score < best[0]:
        best[0] = score
        best[1] = arr.copy()
    if top is None:
        return
    if len(top) < 10:
        top.append((score, arr.copy()))
    else:
        idx = 0
        mx = top[0][0]
        for i in range(1, len(top)):
            if top[i][0] > mx:
                mx = top[i][0]
                idx = i
        if score < mx:
            top[idx] = (score, arr.copy())

        if time.monotonic() > deadline:

    # If a previous construction is present, try to break its symmetry by

    try:

    except Exception:

    if base_seq is not None:

                    break
                err = 1.0 / (1.0 + float(lam) * t)
                a = base_n * err
                s = float(a.sum())
                if s <= 0.0:
                    continue
                a = a / s
                consider_candidate(fast_score(a), a, top, best)

    # Re-aminate a few discrete analytic points to be safe.
    for lam in (3.7, 3.8, 3.9, 4.0, 4.1, 4.3):
        if time.monotonic() > deadline:

        if a is not None:

    # Exact verification of the top fast-scorers and guaranteed points.
    initial_arr = best[1] if best[1] is not None else make_inv_linear(t, 4.0)
    exact_best = [np.inf, None]
    consider_candidate(exact_score(initial_arr), initial_arr, None, exact_best)

    for _, arr in top:
        if time.monotonic() > deadline - 12.0:

        consider_candidate(exact_score(arr), arr, None, exact_best)

    for lam in (3.8, 3.9, 4.0, 4.1):
        if time.monotonic() > deadline - 12.0:
            break
        a = make_inv_linear(t, lam)
        if a is not None:
            consider_candidate(exact_score(a), a, None, exact_best)

    if exact_best[1] is None:
        exact_best = [exact_score(make_inv_linear(t, 4.0)), make_inv_linear(t, 4.0)]

    base_arr = exact_best[1].copy()
    base_score = float(exact_best[0])

    # Optional light local multiplicative refinement.  It is conservative:
    # the final exact score must improve the already exact-verified baseline.
    if time.monotonic() < deadline - 60.0:
        rng = np.random.default_rng(seed)
        local_arr = base_arr.copy()
        local_cur = fast_score(local_arr)
        local_steps = 0
        while time.monotonic() < deadline - 30.0 and local_steps < 3000:

            st = int(rng.integers(0, n - width + 1))
