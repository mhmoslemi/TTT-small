def lp_update(f, M, tol=1e-10):
    """Solve the LP to find the best direction g."""
    n = len(f)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n-1, k)
        j_lo = lo
        j_hi = hi
        # A[k, j] = f[k-j] for j in [lo, hi]
        A[k, j_lo:j_hi+1] = f[k - np.arange(j_lo, j_hi+1)]
    
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    c_obj = -np.ones(n)  # minimize -sum(g)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.x  # g
    else:
        return None

def optimize_sequence(f0, beta=50.0, max_iter=10000, time_limit=100.0):
    f = f0.copy().astype(float)
    n = len(f)

            break

        c = np.convolve(f, f)
        
        # Compute softmax weights
        c_shifted = c - c.max()
        w = np.exp(beta * c_shifted)
        w /= w.sum()
        
        # Compute gradient
        # grad[j] = 2 * sum_m f[m] * w[m+j]
        # Using FFT
        N = 1 << (3*n - 2).bit_length()  # next power of 2 >= 3n-2
        F = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        h = np.fft.irfft(np.conj(F) * W, N)
        grad = 2.0 * h[:n]
        
        # Center the gradient (project onto sum(f)=1 constraint)
        grad = grad - grad.mean()
        
        # Update

        f_new = np.clip(f_new, 0, None)
        if f_new.sum() < 1e-10:
            break
        f_new = f_new / f_new.sum()
        
        # Evaluate (every few iterations)
        if it % 10 == 0:
            score = evaluate_sequence(f_new.tolist())
            if score < best_score:
                best_score = score
                best_f = f_new.copy()
        
        f = f_new
        step *= 0.999  # slowly decrease step size
    
    return best_f, best_score
