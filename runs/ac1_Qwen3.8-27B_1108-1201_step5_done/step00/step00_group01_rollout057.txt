from scipy.optimize import linprog
from scipy.sparse import diags

import time

def next_pow2(n):
    n = max(1, int(n))
    return 1 << (n - 1).bit_length()

def make_initial(n, alpha=2.0):
    """U-shaped initial sequence."""
    t = np.linspace(0, 1, n)
    f = 1.0 + alpha * (2.0*t - 1.0)**2
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Soft-max mirror descent for minimizing 2n*max(f*f)/(sum(f))^2.
    
    Uses U-shaped initialization, FFT-based autocorrelation, and annealed
    temperature/learning rate. Multiple restarts with different initializations.
    """

    t0 = time.monotonic()

    n = 10000
    N_conv = next_pow2(2*n - 1)
    N_corr = next_pow2(3*n - 1)
    
    best_f = None
    best_score = float('inf')
    
    # Different initializations to try
    alphas = [1.0, 2.0, 3.0, 5.0]
    
    for ai, alpha_init in enumerate(alphas):
        if time.monotonic() > deadline:
            break
        
        f = make_initial(n, alpha_init)
        score = float(evaluate_sequence(f.tolist()))
        print(f"Init {ai} (alpha={alpha_init}): score={score:.6f}")
        
        if score < best_score:
            best_f = f.copy()
            best_score = score
        
        local_best = score
        no_improve_count = 0
        max_iters = 50000
        
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            frac = it / max_iters
            
            # Annealed parameters
            tau = 0.003 + 0.1 * np.exp(-5.0 * frac)
            lr = 0.3 + 0.4 * np.exp(-3.0 * frac)
            
            # Compute autocorrelation via FFT
            F = np.fft.rfft(f, N_conv)
            g = np.fft.irfft(F * F, N_conv)[:2*n-1]
            
            # Softmax weights (numerically stable)
            g_max = g.max()
            exp_g = np.exp((g - g_max) / tau)
            w = exp_g / exp_g.sum()
            
            # Compute pressure: p[j] = sum_k w[k] * f[k-j] (cross-correlation)
            W = np.fft.rfft(w, N_corr)
            Fc = np.fft.rfft(f, N_corr)
            p_full = np.fft.irfft(W * np.conj(Fc), N_corr)
            p = p_full[:n]
            
            # Multiplicative update
            p_max = p.max()
            if p_max > 1e-30:
                factor = 1.0 - lr * (p / p_max)
                factor = np.maximum(factor, 0.0)
                f_new = f * factor
                f_sum = f_new.sum()
                if f_sum > 1e-30:
                    f = f_new / f_sum
                else:
                    f = f.copy()  # no change
            # else: no pressure, no change needed
            
            # Evaluate periodically
            if it % 50 == 0 or it < 50:
                score = float(evaluate_sequence(f.tolist()))
                if np.isfinite(score):
                    if score < local_best:
                        local_best = score
                        no_improve_count = 0
                        if score < best_score:
                            best_f = f.copy()
                            best_score = score
                            if best_score < 1.6 or it % 500 == 0:
                                elapsed = time.monotonic() - t0
                                print(f"  [a={alpha_init}] it={it} best={best_score:.6f} t={elapsed:.0f}s")
                    else:
                        no_improve_count += 1
                else:
                    break  # invalid
            
            # Perturbation to escape local minima
            if no_improve_count > 200 and it % 500 == 499:
                # Add small multiplicative noise
                noise = np.exp(rng.normal(0, 0.05, n))
                f = f * noise
                f = np.maximum(f, 0)
                f_sum = f.sum()
                if f_sum > 0:
                    f /= f_sum
                no_improve_count = 0
            
            if no_improve_count > 500:
                break
        
        elapsed = time.monotonic() - t0
        print(f"  [a={alpha_init}] finished: local={local_best:.6f} global={best_score:.6f} t={elapsed:.0f}s")
    
    # One final evaluation

    print(f"FINAL: score={final_score:.6f} n={n}")
    
    return [max(0.0, float(x)) for x in best_f]

    n = max(1, int(n))

    if s > 0:
