import numpy as np
from scipy.optimize import linprog
import time

def propose_candidate(budget_s=1800):
    # Get initial sequence
    try:
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0, 1000)
        if a.ndim != 1 or len(a) == 0 or not np.all(np.isfinite(a)):
            a = np.ones(256)
    except NameError:
        a = np.ones(256)
    
    n = len(a)
    
    # Normalize
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s  # Now sum = 1
    
    def compute_score(a):
        c = np.convolve(a, a)
        return 2 * n * c.max() / (a.sum()**2)
    
    best_a = a.copy()
    best_score = compute_score(a)
    print(f"Initial score: {best_score}")
    
    deadline = time.monotonic() + budget_s - 5
    
    t = 0.1  # Blending parameter
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = c.max()
        
        # Set up LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
        # Constraints: sum_i a[i]*b[k-i] <= M
        # In matrix form: A @ b <= M * ones
        # A is (2n-1) x n, where A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
        
        # Build constraint matrix A
        # A[k, j] = a[k-j] if 0 <= k-j <= n-1
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            for j in range(max(0, k-n+1), min(k+1, n)):
                A[k, j] = a[k-j]
        
        # This is O(n^2) which for n=4500 is about 20M operations - might be slow
        # Better to use a banded structure or FFT-based approach
        
        # Actually, for n=4500, building A takes O(n^2) = ~20M which is fine
        # But the LP itself might be slow with 4500 vars and 9000 constraints
        
        # Let's use linprog
        c_obj = -np.ones(n)  # Minimize -sum(b) = maximize sum(b)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, None)]*n, method='highs')
        
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-10:
                b = b / b_sum  # Normalize to sum = 1
                a_new = (1-t)*a + t*b
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a_new = a_new / a_new.sum()
                    score_new = compute_score(a_new)
                    if score_new < best_score:
                        best_score = score_new
                        best_a = a_new.copy()
                        print(f"Improved! Score: {best_score}")
                    a = a_new
        else:
            # LP failed, try a different approach
            break
