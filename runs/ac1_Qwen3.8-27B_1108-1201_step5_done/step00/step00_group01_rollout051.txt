from scipy.optimize import linprog
import time

def solve_lp(f, M):
    n = len(f)
    # Variables: g_0, ..., g_{n-1} >= 0
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # A_ub @ g <= b_ub
    # (f*g)_k = sum_{i} f_i * g_{k-i} (with valid index ranges)
    
    # For k = 0: f_0 * g_0 <= M
    # For k = 1: f_0 * g_1 + f_1 * g_0 <= M
    # ...
    # For k = n-1: f_0 * g_{n-1} + f_1 * g_{n-2} + ... + f_{n-1} * g_0 <= M
    # For k = n: f_1 * g_{n-1} + ... + f_n * g_0 ... wait, f only has n elements
    # For k = n: sum_{i=1}^{n-1} f_i * g_{n-i} (since g has indices 0..n-1)
    #   Actually (f*g)_n = sum_{i} f_i * g_{n-i}, where both i and n-i must be in [0, n-1]
    #   So i ranges from max(0, n-(n-1)) = 1 to min(n-1, n-0) = n-1
    #   (f*g)_n = sum_{i=1}^{n-1} f_i * g_{n-i}
    
    # Let me construct the constraint matrix A_ub (shape (2n-1, n))
    A_ub = np.zeros((2*n - 1, n))
    b_ub = np.full(2*n - 1, M)
    
    for k in range(2 * n - 1):
        # (f*g)_k = sum_{i} f_i * g_{k-i} where 0 <= i <= n-1 and 0 <= k-i <= n-1
        # So: max(0, k-(n-1)) <= i <= min(n-1, k)
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A_ub[k, k - i] = f[i]
    
    # Bounds: g_i >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return -result.fun  # sum of g

from scipy.optimize import linprog
import time

def build_convolution_matrix(f):
    """Build the sparse matrix A where (A @ g)_k = (f*g)_k."""
    n = len(f)
    num_rows = 2 * n - 1
    # A is num_rows x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # For row k, the nonzeros are at columns j where 0 <= k-j <= n-1
    #   => k-(n-1) <= j <= k, intersected with [0, n-1]

    cols = []
    vals = []

        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    return A

"""
LP-based iterative optimization for minimizing 2n*max(a*a)/(sum(a))^2.
Uses the SOTA trick: fix f, solve LP for g maximizing sum(g) s.t. (f*g)_k <= M.
Iteratively blend f toward g. Multiple initializations + local search refinement.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build sparse matrix A (shape (2n-1, n)) where (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    # Row k has nonzeros at columns j where max(0,k-n+1) <= j <= min(n-1, k)
    # value at (k,j) is f[k-j]
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int64)
    cols = np.empty(total_nnz, dtype=np.int64)
    data = np.empty(total_nnz, dtype=np.float64)

    idx = 0
    n_minus_1 = n - 1
    for k in range(2 * n - 1):
        j_lo = max(0, k - n_minus_1)
        j_hi = min(n_minus_1, k)
        length = j_hi - j_lo + 1
        rows[idx:idx + length] = k
        cols[idx:idx + length] = np.arange(j_lo, j_hi + 1, dtype=np.int64)
        data[idx:idx + length] = f[k - np.arange(j_lo, j_hi + 1)]
        idx += length

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def evaluate_local(a):
    """Fast local evaluation (same as evaluate_sequence but avoids list conversion overhead)."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def solve_lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g."""
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None

    if result.x is None or not np.isfinite(result.x).any():
        return None

    g = np.maximum(result.x, 0.0)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    return g / g_sum


def initializations_for_n(n, rng):
    """Generate multiple reasonable initializations for sequence of length n."""
    inits = []
    x = np.arange(n)

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Raised cosine (high at endpoints)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n)
    inits.append(a / a.sum())

    # 3. Sinusoidal (zero at endpoints, high in middle)
    a = np.sin(np.pi * x / n) ** 2
    if a.sum() > 0:
        inits.append(a / a.sum())

    # 4. Linear ramp up then down (triangular)
    a = np.minimum(x + 1, n - x)
    inits.append(a.astype(np.float64) / a.sum())

    # 5. Two peaks
    a = np.exp(-4.0 * ((x - n * 0.25) / n) ** 2) + np.exp(-4.0 * ((x - n * 0.75) / n) ** 2)
    if a.sum() > 0:
        inits.append(a / a.sum())

    # 6. 1/cos profile
    if n > 10:
        a = 1.0 / (1.0 + 0.5 * np.cos(2.0 * np.pi * x / n))
        inits.append(a / a.sum())

    # 7. Random smooth
    a = np.abs(np.sin(x / n * 3 * np.pi + rng.random() * 2 * np.pi)) + 0.1
    inits.append(a / a.sum())

    return inits

            a_score = evaluate_local(a)

            local_best_score = a_score

            n_iter = 0

                if time.monotonic() > phase_deadline:

        no_improve = 0
        max_no_improve = 2000

        while time.monotonic() < total_deadline - 5.0 and no_improve < max_no_improve:
            width = int(rng.integers(base_width // 2, base_width * 3 + 1))
            width = min(width, n)
            start = int(rng.integers(0, n - width + 1))
            end = start + width

            candidate = a.copy()
            candidate[start:end] *= np.exp(rng.normal(0.0, sigma, size=width))

            # Occasionally try global rescaling of a subset
            if rng.random() < 0.2:
                w2 = int(rng.integers(base_width, n // 2))
                s2 = int(rng.integers(0, n - w2 + 1))
                candidate[s2:s2 + w2] *= float(np.exp(rng.normal(0.0, sigma * 2)))

            candidate = np.maximum(candidate, 0.0)
            c_sum = candidate.sum()
            if c_sum > 0:
                candidate = candidate / c_sum
                c_score = evaluate_local(candidate)
                if np.isfinite(c_score) and c_score < a_score:
                    a = candidate
                    a_score = c_score
                    if a_score < best_score:
                        best_score = a_score
                        best_sequence = a.copy()
                    no_improve = 0
                    sigma = min(sigma * 0.95, 0.02)
                else:
                    no_improve += 1
                    if no_improve % 500 == 0:
                        sigma = min(sigma * 1.1, 0.15)
            else:
                no_improve += 1

        # Final verification with actual evaluator
        final_score = evaluate_sequence(best_sequence.tolist())
        if np.isfinite(final_score):
            best_score = final_score

    print(f"Final best score: {best_score:.6f}, n={len(best_sequence) if best_sequence is not None else 0}")

    # Safety: return as list of floats
    if best_sequence is None:
        # Fallback: uniform sequence
        return [1.0] * 100
    return [float(x) for x in best_sequence]
