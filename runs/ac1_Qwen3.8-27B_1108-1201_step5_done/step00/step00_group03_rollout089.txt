maximizes sum(g) subject to conv(f,g) <= M and g >= 0, providing a principled
descent direction for the composite objective.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _precompute_lp_structure(n):
    """Precompute the CSR structure for the convolution constraint matrix."""
    num_rows = 2 * n - 1
    nnz_per_row = np.array([min(k + 1, n, num_rows - k) for k in range(num_rows)])
    indptr = np.concatenate([[0], np.cumsum(nnz_per_row)]).astype(np.int32)
    total_nnz = int(indptr[-1])
    indices = np.empty(total_nnz, dtype=np.int32)
    val_idx = np.empty(total_nnz, dtype=np.int32)
    for k in range(num_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        sl = slice(indptr[k], indptr[k + 1])
        js = np.arange(j_lo, j_hi + 1, dtype=np.int32)
        indices[sl] = js
        val_idx[sl] = k - js
    return indptr, indices, val_idx


def _lp_step(f, indptr, indices, val_idx, n, M):
    """Solve one LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    data = f[val_idx]
    A = csr_matrix((data, indices, indptr), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 20})
    if res.status == 0 and res.x is not None and np.all(np.isfinite(res.x)):
        return res.x
    return None


def _lp_optimize(f, deadline, n, verbose=True):
    """Iteratively optimize f using LP-based direction finding."""
    S = float(f.sum())
    if S < 0.01:
        f = np.ones(n)
        S = float(n)

    M = float(np.max(np.convolve(f, f)))
    score = 2.0 * n * M / (S * S)
    best_f = f.copy()
    best_score = score

    indptr, indices, val_idx = _precompute_lp_structure(n)
    rng = np.random.default_rng(12345 + n)

    no_improve = 0
    iters = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline - 2.0:
        iters += 1
        S = float(best_f.sum())
        if S < 0.01:
            best_f = np.ones(n)
            S = float(n)
        M = float(np.max(np.convolve(best_f, best_f)))

        g_star = _lp_step(best_f, indptr, indices, val_idx, n, M)

        if g_star is None:
            no_improve += 1
            if no_improve % 15 == 0:
                best_f = np.clip(best_f * np.exp(rng.normal(0, 0.12, n)), 0, 1000)
            if no_improve > 80:
                break
            continue

        sum_g = float(g_star.sum())

        if sum_g > S * 1.0001:
            g = g_star * (S / sum_g)
            improved = False
            for t_alpha in [0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05]:
                f_new = (1.0 - t_alpha) * best_f + t_alpha * g
                f_new = np.clip(f_new, 0.0, 1000.0)
                S_new = float(f_new.sum())
                if S_new < 0.01:
                    continue
                M_new = float(np.max(np.convolve(f_new, f_new)))
                sc_new = 2.0 * n * M_new / (S_new * S_new)
                if sc_new < best_score - 1e-12:
                    best_f = f_new
                    best_score = sc_new
                    improved = True
                    break

            if improved:
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve == 25:
            # Light multiplicative perturbation
            best_f = np.clip(best_f * np.exp(rng.normal(0, 0.08, n)), 0, 1000)
            no_improve = 0
        elif no_improve == 60:
            # Stronger perturbation
            best_f = np.clip(best_f * np.exp(rng.normal(0, 0.3, n)), 0, 1000)
            no_improve = 0
        elif no_improve > 150:
            break

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      n={n} it={iters} score={best_score:.8f} elapsed={elapsed:.1f}s")

    return best_f, best_score


def _stochastic_polish(f, deadline, n, verbose=True):
    """Fine-grained stochastic perturbations to escape shallow local minima."""
    rng = np.random.default_rng(99999 + n)
    best_f = f.copy()
    S = float(best_f.sum())
    M = float(np.max(np.convolve(best_f, f)))
    best_score = 2.0 * n * M / (S * S)
    iters = 0

    while time.monotonic() < deadline - 1.0:
        iters += 1
        # Random block perturbation
        block_len = max(2, n // (50 + rng.integers(0, 50)))
        start = rng.integers(0, max(1, n - block_len))
        f_candidate = best_f.copy()
        f_candidate[start:start + block_len] *= np.exp(rng.normal(0, 0.05, block_len))
        f_candidate = np.clip(f_candidate, 0, 1000)

        S_c = float(f_candidate.sum())
        if S_c < 0.01:
            continue
        M_c = float(np.max(np.convolve(f_candidate, f_candidate)))
        sc_c = 2.0 * n * M_c / (S_c * S_c)

        if sc_c < best_score - 1e-12:
            best_f = f_candidate
            best_score = sc_c
            if verbose and iters % 500 == 0:
                print(f"        polish n={n} it={iters} score={best_score:.8f}")

    return best_f, best_score
