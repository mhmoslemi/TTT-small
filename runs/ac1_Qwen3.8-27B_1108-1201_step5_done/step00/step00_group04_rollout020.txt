import time

def mwd_step(a, alpha, eta, rng=None):
    """One step of multiplicative weight descent."""
    n = a.size
    c = fftconvolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    c_powers = c_safe ** (alpha - 1)
    
    # Compute h[i] = sum_j c_powers[i+j] * a[j]
    # h = (c_powers * a_flip)[n-1 : 2n-1]
    a_flip = a[::-1]
    corr = fftconvolve(c_powers, a_flip)
    h = corr[(n-1):(2*n-1)]
    
    grad = 2.0 * alpha * h
    gn = np.abs(grad).max()
    if gn < 1e-30:
        return a, c.max()
    grad /= gn
    
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a, c.max()
    a_new /= s
    
    return a_new, c.max()

A good schedule might be:
- Start with eta = 1.0 (aggressive)
- Decay as eta = 1.0 / (1 + 0.01 * iteration) or eta = eta_0 * (0.999)^iteration

Let me also add: **momentum**. Keep a running average of the gradient and use that for the update. This can help with oscillations.

def _mwd_step(a, alpha, eta):
    """One MWD step: returns (new_a, c_max)."""
    n = a.size
    c = fftconvolve(a, a)
    c_max = c.max()
    c_safe = np.maximum(c, 1e-30)
    c_powers = c_safe ** (alpha - 1)
    
    a_flip = a[::-1]
    corr = fftconvolve(c_powers, a_flip)
    h = corr[(n-1):(2*n-1)]
    
    grad = 2.0 * alpha * h
    gn = np.abs(grad).max()
    if gn < 1e-30:
        return a, c_max
    grad = grad / gn
    
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a, c_max
    a_new = a_new / s
    return a_new, c_max

def _run_mwd(a, alpha_start, alpha_end, eta_start, eta_end, time_limit, rng):
    """Run MWD optimization with annealing until time_limit."""
    n = a.size
    t0 = time.monotonic()
    elapsed = 0.0
    
    while elapsed < time_limit:
        # Progress from 0 to 1
        progress = min(1.0, elapsed / max(0.1, time_limit))
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress
        
        a, c_max = _mwd_step(a, alpha, eta)
        
        # Periodic perturbation

            a = a * noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
