Let P = (1-t)*f + t*g. Then P = f + t*(g-f).
P*P = (f + t*(g-f)) * (f + t*(g-f))
= f*f + 2t*(f*(g-f)) + t^2*((g-f)*(g-f))
= f*f + 2t*(f*g - f*f) + t^2*((g-f)*(g-f))

- f*f (compute once per iteration)
- f*g (compute once per iteration)
- (g-f)*(g-f) (compute once per iteration)

Looking at this problem, I need to minimize `2 * n * max(convolve(a, a)) / (sum(a))^2`. This is a non-convex optimization problem where I need to find a sequence whose autocorrelation peak is as small as possible relative to its total mass.

My approach:
1. **Frank-Wolfe with LP**: At each step, fix the current sequence `f`, solve an LP to find the best direction `g` (maximizing sum subject to `f*g` being bounded), then do a line search for the optimal step size.
2. **Subgradient refinement**: Use cheap O(n) subgradient steps after the LP phase for additional gains.
3. **Multiple n values and initializations** to avoid local minima.

"""
Frank-Wolfe + Subgradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based Frank-Wolfe iterations followed by subgradient polishing.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def build_constraint_matrix(f):
    """Build A where A[j,k] = f[j-k] if 0<=j-k<n else 0. Shape (2n-1, n)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        j_end = min(k + n, 2 * n - 1)
        A[k:j_end, k] = f[:j_end - k]
    return A


def fw_lp_step(f, M):
    """Solve the LP: max sum(g) s.t. (f*g)[j] <= M, g >= 0. Returns g or None."""
    n = len(f)
    A_ub = build_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0.0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return res.x.copy()


def line_search(f, g):
    """Find t in [0,1] minimizing max(conv((1-t)f+tg, (1-t)f+tg))."""
    ff = np.convolve(f, f)
    fg = np.convolve(f, g)
    d = g - f
    dd = np.convolve(d, d)

    def h(t):
        return np.max(ff + 2.0 * t * (fg - ff) + t * t * dd)

    res = minimize_scalar(h, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-8})
    return res.x


def subgradient_step(f, alpha):
    """One subgradient step: reduce max(conv(f,f)) by moving mass away from peak contributors."""
    n = len(f)
    c = np.convolve(f, f)
    j_star = int(np.argmax(c))

    g = np.zeros(n)
    k_lo = max(0, j_star - n + 1)
    k_hi = min(n - 1, j_star)
    idx = j_star - np.arange(k_lo, k_hi + 1)
    g[k_lo:k_hi + 1] = 2.0 * f[idx]

    f_new = np.maximum(f - alpha * g, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    return f_new / s


def fw_phase(f, max_iters=500, time_limit=600.0, verbose=True):
    """Run Frank-Wolfe iterations until convergence or time limit."""
    start = time.monotonic()
    n = len(f)
    M = np.max(np.convolve(f, f))
    best_M = M
    best_f = f.copy()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        t0 = time.monotonic()
        g = fw_lp_step(f, M)
        lp_time = time.monotonic() - t0

        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * (f.sum() / g_sum)

        t = line_search(f, g)
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        M_new = np.max(np.convolve(f_new, f_new))

        if M_new < M - 1e-14:
            f = f_new
            M = M_new
            stall_count = 0
            if M_new < best_M:
                best_M = M_new
                best_f = f_new.copy()
        else:
            stall_count += 1
            if stall_count > 10:
                break

        if verbose and it % 5 == 0:
            score = 2.0 * n * M
            print(f"    FW iter {it}: score={score:.6f}, M={M:.8f}, t={t:.4f}, "
                  f"lp_time={lp_time:.2f}s")

    return best_f, best_M


def subgradient_phase(f, time_limit=200.0, verbose=True):
    """Run subgradient steps with decreasing step size."""
    start = time.monotonic()
    n = len(f)
    best_f = f.copy()
    best_M = np.max(np.convolve(f, f))
    total_iters = 0

    for epoch in range(50):
        if time.monotonic() - start > time_limit:
            break
        alpha = 0.5 / (npc := (epoch + 1))
        for _ in range(50):
            if time.monotonic() - start > time_limit:

            f = subgradient_step(f, alpha)
            total_iters += 1
            M = np.max(np.convolve(f, f))
            if M < best_M:

                best_f = f.copy()

        if verbose and epoch % 10 == 0:
            score = 2.0 * n * best_M
            print(f"    SG epoch {epoch}: best_score={score:.6f}, iter={total_iters}")

    return best_f, best_M


def local_block_search(f, time_limit=100.0, seed=42):
    """Block perturbation local search."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    step_size = 0.05

    for it in range(100000):

        pos = int(rng.integers(0, n - width))
        f_new = f.copy()
        f_new[pos:pos + width] *= np.exp(rng.normal(0.0, step_size, width))
        f_new = np.maximum(f_new, 0.0)
