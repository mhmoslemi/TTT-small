```python
def gradient_step(f, alpha, n_iter=100, step_size=0.01):
    """Smooth gradient descent on sum(exp(alpha * (f*f)_k)) with simplex projection."""
    n = len(f)
    f = f / np.sum(f)
    
    for i in range(n_iter):
        g = np.convolve(f, f)
        w = alpha * np.exp(alpha * g)
        # Gradient: grad[j] = 2 * sum_i w[i+j] * f[i]
        # This is 2 * (w * f)[j] where * denotes "cross-correlation"
        # More precisely: grad[j] = 2 * sum_{i=0}^{n-1} w[i+j] * f[i]
        # w has length 2n-1, so i+j ranges from j to j+n-1
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = 2.0 * np.dot(w[j:j+n], f)
        # Project gradient onto the tangent space of the simplex
        # (subtract the mean to ensure sum(grad)=0, approximately)
        grad -= np.mean(grad)
        # Gradient step
        f_new = f - step_size * grad
        # Project onto simplex {x: sum(x)=1, x>=0}
        f_new = project_simplex(f_new)
        f = f_new
    
    return f
```

```python
def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

import time
import scipy.optimize as opt
from scipy import sparse

def compute_score(f):
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    s = np.sum(f)
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * np.max(conv) / (s * s)

def gaussian_init(n, sigma, center=None):
    if center is None:
        center = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))

def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient_descent(f, alpha, n_steps, step_size, deadline):
    """Smooth gradient descent on sum(exp(alpha * C[k])) with simplex constraint."""
    f = f / np.sum(f)
    n = len(f)

        if time.monotonic() > deadline:
            break
        C = np.convolve(f, f)
        # Weight by exponential of scaled convolution
        w = np.exp(alpha * (C - C.max()))  # shift for numerical stability
        w = w / np.sum(w)  # normalize weights
        # Gradient: grad[j] = 2 * sum_i w[i+j] * f[i]
        # Compute efficiently
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = 2.0 * np.dot(w[j:j+n], f)
        # Project gradient to tangent space of simplex
        grad -= np.mean(grad)
        # Step
        f = f - step_size * grad
        # Project to simplex
        f = project_simplex(f)
    return f

def build_A_and_solve(f, n):
    """Build constraint matrix and solve the LP."""
    n_rows = 2 * n - 1
    # Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0
    # Precompute the structure
    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    pos = 0
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows[pos:pos+count] = k
        cols[pos:pos+count] = range(j_start, j_end + 1)
        pos += count
    
    idx = rows - cols
    vals = f[idx]
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(n_rows, n))
    b_ub = np.max(np.convolve(f, f)) * np.ones(n_rows)
    
    res = opt.linprog(-np.ones(n), A_ub=A, b_ub=b_ub, 
                      bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None
