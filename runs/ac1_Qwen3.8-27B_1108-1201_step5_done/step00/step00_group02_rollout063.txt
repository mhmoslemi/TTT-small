"""
Minimizes J(a) = 2*n*max(conv(a))/sum(a)^2 using smoothed subgradient descent
with Adam optimizer, multiple restarts, and adaptive soft-max smoothing.
The key insight is that non-palindromic sequences can have convolution peaks
below the Cauchy-Schwarz bound for palindromic sequences.
"""

import numpy as np
import time


def compute_smooth_gradient(b, beta):
    """Compute the smoothed gradient of max(conv(b))/sum(b)^2 with respect to b."""
    n = len(b)
    conv = np.convolve(b, b)
    max_conv = np.max(conv)
    # Softmax weights (numerically stable)
    shifted = beta * (conv - max_conv)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.exp_vals.sum()
    
    # Compute cross-correlation: cross[j] = sum_k weights[k] * b[k-j]
    # This equals convolve(weights, b[::-1])[n-1 : 2n-1]
    b_rev = b[::-1]
    full_cross = np.convolve(weights, b_rev)
    cross = full_cross[n-1:2*n-1]  # length n
    
    # Gradient: grad[j] = 2*cross[j] - weights[2*j]*b[j]
    grad = 2.0 * cross - weights[0:2*n-1:2] * b
    
    # Full gradient of J = 2*n*max_conv/sum(b)^2
    # dJ/db = 2*n/S^2 * (S*grad_max - 2*max_conv*1) where grad_max is subgrad of max

    # But we're optimizing max_conv directly (since J = 2n * max_conv when sum=1)
    # So we just need to minimize max_conv, and the gradient is `grad`
    
    return grad, max_conv, weights


def adam_update(state, grad, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer update step. Returns updated parameter and state."""
    m = state['m']
    v = state['v']
    
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    
    state['m'] = m
    state['v'] = v
    return update

"""
Minimizes J(a) = 2*n*max(conv(a))/sum(a)^2 using smoothed subgradient descent
with Adam optimizer, multiple restarts at different lengths, and adaptive soft-max
smoothing. Key insight: non-palindromic sequences can achieve J < 2 by reducing
the convolution peak below the Cauchy-Schwarz bound that constrains palindromic
sequences. Uses FFT-based convolution for efficiency.
"""

import numpy as np
import time


def compute_convolution(b):
    """Compute full linear convolution of b with itself."""
    return np.convolve(b, b)


def compute_smooth_gradient(b, beta):
    """
    Compute smoothed gradient of max_k conv[b][k] w.r.t. b.
    Returns (gradient, max_conv, weights).
    """
    n = len(b)
    conv = np.convolve(b, b)
    max_conv = np.max(conv)
    # Numerically stable softmax
    shifted = beta * (conv - max_conv)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights = exp_vals / total

    # cross[j] = sum_k weights[k] * b[k-j] for valid indices
    # = convolve(weights, b_rev)[n-1 : 2n-1]
    b_rev = b[::-1]
    full_cross = np.convolve(weights, b_rev)
    cross = full_cross[n - 1:2 * n - 1]

    # Exact gradient with factor-of-2 correction at center
    grad = 2.0 * cross - weights[0:2 * n - 1:2] * b

    return grad, float(max_conv), weights


def project_to_simplex(b):
    """Project b onto probability simplex (sum=1, non-negative)."""
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-300:
        return np.ones_like(b) / len(b)
    return b / s


def refine_sequence(b, time_limit, initial_lr=0.05, target_beta=200.0, seed=0):
    """
    Refine sequence b (normalized to sum=1) using smoothed subgradient with Adam.
    Minimizes 2*n*max(conv(b)) which equals J(b) when sum(b)=1.
    """
    n = len(b)
    b = b.astype(np.float64).copy()
    b = project_to_simplex(b)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_obj = np.inf
    best_b = b.copy()
    beta = 5.0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    no_improve_count = 0
    no_improve_threshold = 500

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, max_conv, _ = compute_smooth_gradient(b, beta)

        # Exact objective (since sum(b)=1): J = 2*n*max_conv
        obj = 2.0 * n * max_conv
        if obj < best_obj:
            best_obj = obj
            best_b = b.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive learning rate with cosine-like decay
        progress = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        lr_t = initial_lr * (1.0 - 0.7 * progress)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        update = lr_t * m_hat / (np.sqrt(v_hat) + eps)

        b = b - update
        b = project_to_simplex(b)

        # Increase beta (less smoothing) over time
        beta = min(target_beta, beta * (1.0 + 0.002 * (1.0 + progress)))

        # If stuck, do a small random perturbation
        if no_improve_count > no_improve_threshold:

            b += np.random.default_rng(t).normal(0, perturbation_scale, n)
            b = project_to_simplex(b)
            no_improve_count = 0
            no_improve_threshold = int(no_improve_threshold * 1.5)

    return best_b, best_obj

    else:
        # Asymmetric window
        x = np.linspace(0, 1, n)
        b = np.exp(-3.0 * x) + 0.3 * np.exp(-3.0 * (1 - x)) + 0.1

    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n)
        s = n
    return b / s


def structured_initial(n, rng, asymmetric=True):
    """Generate structured initial sequences that break palindromic symmetry."""
    method = rng.integers(0, 6)
    x = np.arange(n, dtype=np.float64)
    if method == 0:
        # Linear ramp (maximally asymmetric)
        b = np.linspace(0.1, 1.0, n)
    elif method == 1:
        # Two asymmetric bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.05, 0.15)
        b = np.exp(-0.5 * ((x - c1) / w1) ** 2) + 0.7 * np.exp(-0.5 * ((x - c2) / w2) ** 2)
    elif method == 2:
        # Cosine with phase shift (breaks symmetry)
        phase = rng.uniform(0, 2 * np.pi)
        b = 0.5 + 0.5 * np.cos(2 * np.pi * x / n + phase)
    elif method == 3:
        # Product of asymmetric windows
        a1 = np.linspace(0.5, 1.5, n)
        a2 = np.linspace(1.5, 0.5, n)
        b = a1 * a2 * np.exp(-0.5 * ((x - n * 0.3) / (n * 0.2)) ** 2)
    elif method == 4:
        # Random smooth (spline-like)
        n_ctrl = max(5, n // 50)
        ctrl = rng.uniform(0.1, 1.0, n_ctrl)
        ctrl_x = np.linspace(0, n - 1, n_ctrl)
        b = np.interp(x, ctrl_x, ctrl)
    else:
        # Asymmetric Gaussian
        center = n * rng.uniform(0.2, 0.6)
        width = n * rng.uniform(0.1, 0.3)
        b = np.exp(-0.5 * ((x - center) / width) ** 2) + 0.05

    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n)
        s = n
    return b / s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2*n*max(conv(a))/sum(a)^2.
    Phase 1: Explore different lengths and initializations.
    Phase 2: Deep refinement of best candidate.
    Phase 3: Fine-tuning with perturbed restarts.
    Uses smoothed subgradient with Adam for efficient gradient-based descent.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 5.0

    best_score = np.inf
    best_sequence = None

    # Try to use parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = None
    except NameError:
        parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0)
        pscore = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(pscore):
            best_score = pscore
            best_sequence = parent.copy()
            print(f"Parent: score={pscore:.6f}, n={parent.size}")

    # Determine lengths to explore
    base_n = parent.size if parent is not None else 2000
    candidate_lengths = sorted(set([
        base_n,
        max(100, base_n // 4),
        max(100, base_n // 2),
        int(base_n * 1.5),
        int(base_n * 2),
        500, 1000, 2000, 4000, 8000, 12000, 20000
    ]))
    # Filter to reasonable range
    candidate_lengths = [n for n in candidate_lengths if 100 <= n <= 30000]

    # ===== Phase 1: Broad exploration (45% of budget) =====
    phase1_limit = 0.45 * (total_budget - 5.0)
    t_phase1_start = time.monotonic()
    print(f"\n=== Phase 1: Exploration (limit={phase1_limit:.0f}s) ===")

    for length in candidate_lengths:
        if time.monotonic() - t_phase1_start > phase1_limit:
            break

        n_trials = 3 if length <= 5000 else 2
        time_per_trial = min(
            (phase1_limit - (time.monotonic() - t_phase1_start)) / max(1, n_trials),
            20.0
        )
        if time_per_trial < 2.0:
            continue

        for trial in range(n_trials):
            if time.monotonic() - t_phase1_start > phase1_limit:
                break

            # Choose initialization
            if trial == 0 and length == base_n and parent is not None:
                b = np.maximum(parent, 0.0)
                b = b / b.sum()
            elif trial == 0:
                b = structured_initial(length, rng)
            elif trial == 1:
                b = random_initial(length, rng)
            else:
                b = structured_initial(length, rng)

            # Quick refinement
            result_b, result_obj = refine_sequence(
                b, time_per_trial, initial_lr=0.1, target_beta=50.0
            )

            # Evaluate exactly
            score = float(evaluate_sequence(result_b.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = result_b.copy()
                print(f"  n={length}, trial={trial}: J={score:.6f} (NEW BEST)")

    print(f"Phase 1 done. Best so far: J={best_score:.6f}")

    # ===== Phase 2: Deep refinement (40% of budget) =====
    phase2_limit = 0.40 * (total_budget - 5.0)
    if best_sequence is not None:
        print(f"\n=== Phase 2: Deep refinement (limit={phase2_limit:.0f}s) ===")
        b = best_sequence.astype(np.float64).copy()
        b = np.maximum(b, 0.0)
        b = b / b.sum()

        result_b, result_obj = refine_sequence(
            b, phase2_limit, initial_lr=0.03, target_beta=300.0
        )

        score = float(evaluate_sequence(result_b.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = result_b.copy()
            print(f"Phase 2: J={score:.6f} (NEW BEST)")
        else:
            print(f"Phase 2: J={score:.6f} (no improvement over {best_score:.6f})")

    # ===== Phase 3: Perturbed restarts (15% of budget) =====
    phase3_limit = 0.15 * (total_budget - 5.0)
    if best_sequence is not None and phase3_limit > 5.0:
        print(f"\n=== Phase 3: Perturbed restarts (limit={phase3_limit:.0f}s) ===")
        n_best = len(best_sequence)
        n_restarts = max(3, int(phase3_limit / 8.0))

        for r in range(n_restarts):
            remaining = phase3_limit - (time.monotonic() - t_phase1_start - phase1_limit - phase2_limit)
            if remaining < 3.0:
                break
            time_this = min(remaining / max(1, n_restarts - r), 12.0)

            # Perturb best sequence
            b = best_sequence.copy().astype(np.float64)
            noise = rng.normal(0, 0.3 / np.sqrt(n_best), n_best)
            b += noise
            # Also try reversing a random segment
            if rng.random() < 0.5:
                seg_len = min(n_best, rng.integers(n_best // 10, n_best // 3))
                start = rng.integers(0, n_best - seg_len)
                b[start:start + seg_len] = b[start:start + seg_len][::-1]

            b = project_to_simplex(b)
            result_b, result_obj = refine_sequence(
                b, time_this, initial_lr=0.01, target_beta=500.0
            )

            score = float(evaluate_sequence(result_b.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = result_b.copy()
                print(f"  Restart {r}: J={score:.6f} (NEW BEST)")

    # ===== Final verification =====
    if best_sequence is not None:
        final_check = float(evaluate_sequence(best_sequence.tolist()))
        print(f"\nFinal: J={final_check:.6f}, n={len(best_sequence)}")
        if np.isfinite(final_check):
            # Make sure we return the actual best
            if final_check < best_score:
                best_score = final_check

    if best_sequence is None:
        # Absolute fallback
        best_sequence = np.ones(256, dtype=np.float64)
        print("WARNING: Using fallback ones(256)")

    return [float(x) for x in best_sequence]
