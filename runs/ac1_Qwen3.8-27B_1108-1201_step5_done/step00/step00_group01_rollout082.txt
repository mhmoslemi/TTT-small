    """
    n = len(g)
    W = fft(weights, L)
    G_rev = fft(g[::-1], L)
    conv = np.real(ifft(G_rev * W))
    # Slice indices n-1 through 2n-2
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def _make_init_two_level(n):
    g = np.zeros(n)
    g[:n // 2] = 2.0 / n
    g[n // 2:] = 1.0 / n
    return g


def _make_init_ramp(n, increasing):
    if increasing:
        g = np.arange(1, n + 1, dtype=np.float64)
    else:
        g = np.arange(n, 0, -1, dtype=np.float64)
    return g / g.sum()


def _make_init_cosine(n, freq, alpha, rng):
    j = np.arange(n, dtype=np.float64)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    g = 1.0 + alpha * np.cos(2.0 * np.pi * freq * j / n + phase)
    g = np.clip(g, 0.01, None)
    return g / g.sum()


def _make_init_gaussian(n, sigma_frac):
    j = np.arange(n, dtype=np.float64)
    center = n / 2.0
    g = np.exp(-0.5 * ((j - center) / (n * sigma_frac)) ** 2)
    return g / g.sum()


def _make_init_mix(n, rng):
    j = np.arange(n, dtype=np.float64)
    g = np.ones(n, dtype=np.float64)
    for _ in range(4):
        freq = rng.uniform(0.3, 4.0)
        alpha = rng.uniform(0.1, 0.35)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        g += alpha * np.cos(2.0 * np.pi * freq * j / n + phase)
    g = np.clip(g, 0.01, None)
    return g / g.sum()


def _make_init_random(n, rng):
    g = rng.random(n) + 0.05
    return g / g.sum()


def _make_init_three_level(n, rng):
    """Three-block construction with random ratios."""
    m1 = n // 3
    m2 = n // 3
    m3 = n - m1 - m2
    a = rng.uniform(0.5, 3.0)
    b = rng.uniform(0.3, 2.0)
    c = rng.uniform(0.5, 3.0)
    g = np.zeros(n)
    g[:m1] = a
    g[m1:m1 + m2] = b
    g[m1 + m2:] = c
    return g / g.sum()


def _optimize(g0, deadline):
    """
    Run smoothed gradient descent with exponential tilting.
    Returns (best_g, best_obj) found during the run.
    """
    n = len(g0)
    g = g0.copy()
    L_grad = next_fast_len(3 * n - 2)
    L_conv = next_fast_len(2 * n - 1)

    eta = 1.0
    epsilon = 0.05
    best_g = g.copy()
    best_obj = evaluate_sequence(g.tolist())
    if not np.isfinite(best_obj):
        best_obj = 1e10
    it = 0

    while time.monotonic() < deadline:
        # Self-convolution
        G = fft(g, L_conv)
        conv = np.real(ifft(G * G))[:2 * n - 1]

        # Smoothed weights (log-sum-exp style)
        max_conv = np.max(conv)
        exp_w = np.exp((conv - max_conv) / epsilon)
        exp_sum = exp_w.sum()
        if exp_sum <= 0:
            break
        weights = exp_w / exp_sum

        # Gradient
        grad = _gradient_smoothed(g, weights, L_grad)
        # Center for numerical stability
        grad = grad - np.mean(grad)

        # Exponential tilting update
        scale = np.max(np.abs(grad))
        if scale > 1e-15:
            g = g * np.exp(-eta * grad / scale)
            g = np.clip(g, 1e-30, None)
            s = g.sum()
            if s > 1e-15:
                g = g / s
            else:
                g = g0.copy()
                break

        it += 1
        # Adaptive decay
        eta *= 0.9997
        epsilon = max(epsilon * 0.9999, 0.0005)

        # Periodic evaluation
        if it % 300 == 0:
            obj = evaluate_sequence(g.tolist())
            if np.isfinite(obj) and obj < best_obj:
                best_obj = obj
                best_g = g.copy()

    # Final evaluation
    obj = evaluate_sequence(g.tolist())
    if np.isfinite(obj) and obj < best_obj:
        best_obj = obj
        best_g = g.copy()

    return best_g, best_obj


def _polish(g0, deadline):
    """
    Refinement phase: smaller epsilon, smaller step size, more frequent evals.
    """
    n = len(g0)
    g = g0.copy()
    L_grad = next_fast_len(3 * n - 2)
    L_conv = next_fast_len(2 * n - 1)

    eta = 0.3
    epsilon = 0.001
    best_g = g.copy()
    best_obj = evaluate_sequence(g.tolist())
    if not np.isfinite(best_obj):
        best_obj = 1e10
    it = 0

    while time.monotonic() < deadline:
        G = fft(g, L_conv)
        conv = np.real(ifft(G * G))[:2 * n - 1]

        max_conv = np.max(conv)
        exp_w = np.exp((conv - max_conv) / epsilon)
        exp_sum = exp_w.sum()
        if exp_sum <= 0:
            break
        weights = exp_w / exp_sum

        grad = _gradient_smoothed(g, weights, L_grad)
        grad = grad - np.mean(grad)

        scale = np.max(np.abs(grad))
        if scale > 1e-15:
            g = g * np.exp(-eta * grad / scale)
            g = np.clip(g, 1e-30, None)
            s = g.sum()
            if s > 1e-15:
                g = g / s
            else:
                break

        it += 1
        eta *= 0.9999
        epsilon = max(epsilon * 0.99999, 0.0001)

        if it % 100 == 0:
            obj = evaluate_sequence(g.tolist())
            if np.isfinite(obj) and obj < best_obj:
                best_obj = obj
                best_g = g.copy()

    obj = evaluate_sequence(g.tolist())
    if np.isfinite(obj) and obj < best_obj:
        best_obj = obj
        best_g = g.copy()

    return best_g, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Searches for a non-negative sequence minimizing 2*n*max(conv)/sum^2
    using smoothed gradient descent with exponential tilting,
    multiple diverse initializations, and a polishing phase.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 3.0

    n = len(height_sequence_1)

    # Build diverse initializations
    inits = []
    inits.append(_make_init_two_level(n))
    inits.append(_make_init_ramp(n, increasing=False))
    inits.append(_make_init_ramp(n, increasing=True))
    inits.append(_make_init_cosine(n, freq=1.0, alpha=0.5, rng=rng))
    inits.append(_make_init_cosine(n, freq=2.0, alpha=0.4, rng=rng))
    inits.append(_make_init_cosine(n, freq=0.5, alpha=0.7, rng=rng))
    inits.append(_make_init_cosine(n, freq=3.0, alpha=0.3, rng=rng))
    inits.append(_make_init_gaussian(n, sigma_frac=0.25))
    inits.append(_make_init_gaussian(n, sigma_frac=0.4))
    inits.append(_make_init_mix(n, rng))
    inits.append(_make_init_three_level(n, rng))
    inits.append(_make_init_three_level(n, rng))
    inits.append(_make_init_random(n, rng))

    # Include the given sequence
    g_given = np.clip(np.array(height_sequence_1, dtype=np.float64), 0.0, None)
    if g_given.sum() > 0:
        g_given = g_given / g_given.sum()
    else:
        g_given = np.ones(n) / n
    inits.append(g_given)

    # Evaluate all initializations to find the best start
    best = inits[0].copy()
    best_obj = evaluate_sequence(best.tolist())
    if not np.isfinite(best_obj):
        best_obj = 1e10

    for g0 in inits:
        obj = evaluate_sequence(g0.tolist())
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best = g0.copy()

    print(f"Initial best: {best_obj:.6f} (n={n})")

    # Phase 1: Explore each initialization with main optimizer
    n_inits = len(inits)
    phase1_budget = 0.6 * (float(budget_s) - 8.0)
    time_per_init = phase1_budget / n_inits

    for ii in range(n_inits):
        if time.monotonic() > total_deadline:
            break
        t_start = time.monotonic()
        t_deadline = t_start + time_per_init
        t_deadline = min(t_deadline, total_deadline)

        g, obj = _optimize(inits[ii], t_deadline)
        if obj < best_obj:
            best_obj = obj
            best = g
            print(f"Init {ii}: {best_obj:.6f}")

    # Phase 2: Polish from the best solution
    phase2_deadline = total_deadline
    if time.monotonic() < phase2_deadline:
        g, obj = _polish(best, phase2_deadline)
        if obj < best_obj:
            best_obj = obj
            best = g
            print(f"Polish: {best_obj:.6f}")

    # Phase 3: Try a few random perturbations of the best
    if time.monotonic() < total_deadline:
        remain = total_deadline - time.monotonic()
        n_perturb = max(1, int(remain // 5.0))
        n_perturb = min(n_perturb, 20)
        t_each = remain / n_perturb

        for p in range(n_perturb):
            if time.monotonic() > total_deadline:
                break
            pert = best * np.exp(rng.normal(0.0, 0.05, size=n))
            pert = np.clip(pert, 0.0, None)
            s = pert.sum()
            if s > 0:
                pert = pert / s
            t_dead = min(time.monotonic() + t_each, total_deadline)
            g, obj = _polish(pert, t_dead)
            if obj < best_obj:
                best_obj = obj
                best = g
                print(f"Perturb {p}: {best_obj:.6f}")

    print(f"Final result: {best_obj:.6f}")
    return [float(x) for x in best]
