"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated subgradient
descent with soft-max temperature annealing, multiple asymmetric initializations,
and periodic perturbations to escape local minima.
"""

import numpy as np
import time
from numpy.fft import fft, ifft


def fast_convolve(a, b=None):
    """FFT-based linear convolution."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1 << (na + nb - 2).bit_length()
    if size < na + nb - 1:
        size <<= 1
    A = fft(a, size)
    B = fft(b, size)
    return ifft(A * B).real[:na + nb - 1]


def subgradient_soft(a, r, T):
    """Compute soft-max weighted subgradient of max(r)."""
    n = len(a)
    r_max = r.max()
    # Log-sum-exp for stability
    shifted = (r - r_max) / max(T, 1e-12)
    max_sh = shifted.max()
    w = np.exp(shifted - max_sh)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(r)
        w[int(np.argmax(r))] = 1.0
    else:
        w /= ws

    g = np.zeros(n)
    # Only process lags with non-negligible weight
    thresh = w.max() * 1e-7
    active = np.where(w > thresh)[0]
    for k in active:
        wk = 2.0 * w[k]
        if k < n:
            g[:k + 1] += wk * a[k::-1]
        else:
            g[k - n + 1:] += wk * a[n - 1:k - n - 1:-1]
    return g


def subgradient_hard(a, r, frac=0.999):
    """Compute hard subgradient from active set (lags within frac of max)."""
    n = len(a)
    r_max = r.max()
    active = np.where(r >= r_max * frac)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(r))])
    g = np.zeros(n)
    for k in active:
        if k < n:
            g[:k + 1] += 2.0 * a[k::-1]
        else:
            g[k - n + 1:] += 2.0 * a[n - 1:k - n - 1:-1]
    g /= len(active)
    return g

    Nesterov-accelerated subgradient descent with temperature annealing.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """

    s = a.sum()
    if s < 1e-15:
        return a
    a /= s

    r0 = fast_convolve(a)
    M0 = r0.max()
    T = 0.005 * M0  # temperature
    T_min = 0.0001 * M0
    lr = 0.35
    mu = 0.91
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead

    best_M = M0
    best_a = a.copy()
    no_improve = 0
    patience = 6000
    use_soft = True

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # Evaluate at lookahead point
        r = fast_convolve(y)
        M_cur = r.max()

        if M_cur < best_M:
            best_M = M_cur
            best_a = y.copy()
            no_improve = 0
            lr = min(0.8, lr * 1.002)
        else:
            no_improve += 1

        # Escape from local minima
        if no_improve > patience:
            scale = 0.015 * best_a.std() + 1e-6
            y = best_a + rng.normal(0, scale, n)

            y /= y.sum()
            a = y.copy()

def try_lp_direction(a, n, M):
    """
    LP-based direction finding: max t s.t. b[i]>=t, (a*b)[k]<=M, b>=0.
    Returns blended direction or None.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[n] = -1.0

    # Build constraint matrix
    rows_l = []
    cols_l = []
    vals_l = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows_l.append(k)
            cols_l.append(i)
            vals_l.append(a[k - i])

    # b[i] - t >= 0 => -b[i] + t <= 0
    for i in range(n):
        rows_l.append(2 * n - 1 + i)
        cols_l.append(i)
        vals_l.append(-1.0)
        rows_l.append(2 * n - 1 + i)
        cols_l.append(n)
        vals_l.append(1.0)

    rows_arr = np.array(rows_l, dtype=np.int32)
    cols_arr = np.array(cols_l, dtype=np.int32)
    vals_arr = np.array(vals_l, dtype=np.float64)
    A = csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(3 * n - 1, nvar))
    b_ub = np.concatenate([np.full(2 * n - 1, M), np.zeros(n)])
    bounds = [(0, None)] * nvar

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 2})
        if res.success and res.x is not None:
            b = np.maximum(res.x[:n], 0)
            sb = b.sum()
            if sb > 1e-10:
                b /= sb
                return b
    except Exception:
        pass
    return None
