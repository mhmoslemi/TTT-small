"""
Minimize score(f) = 2*n*max(f*f)/(sum f)^2 over non-negative step functions f.

Approach: the score is scale invariant, so we optimize the shape of f directly.
We use a fully differentiable smooth surrogate of the peak (log-sum-exp) of the
convolution, reparameterize f = softplus(z) so that f > 0 automatically (no
negative values, no projection needed), and run a quasi-Newton optimizer
(L-BFGS-B) from many structured and random starts at several sequence lengths,
with a temperature anneal from coarse to fine so the smoothed peak converges to
the true max.  The convolution and its gradient are computed in O(n log n) via
FFT.  The best point (by the exact evaluator) over the whole search is returned.

This is a different algorithmic strategy from the previous block-perturbation
hill climber: it is a global-ish multi-start quasi-Newton optimization of a
smooth objective rather than local multiplicative block scaling.
"""

import time
import numpy as np
from scipy.optimize import minimize


def _softplus(z):
    """Numerically stable softplus: f = log(1 + exp(z)) > 0."""
    z = np.clip(z, -50.0, 50.0)
    return np.logaddexp(0.0, z)


def _convolve_self(y):
    """Full linear convolution y * y (length 2n-1) via FFT."""
    n = y.size
    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    Y = np.fft.fft(y, N)
    c = np.fft.ifft(Y * Y).real[:need]
    return np.maximum(c, 0.0)


def _cross_soft(p, y):
    """
    rho[i] = sum_{j=0}^{n-1} p[i+j] * y[j] for i = 0..n-1,
    where p has length 2n-1 (the softmax weights of the convolution) and
    y has length n.  Computed with one FFT-based convolution.
    """
    n = y.size
    q = p[::-1]                      # length 2n-1
    need = 3 * n - 2
    N = 1
    while N < need:
        N <<= 1
    a = np.zeros(N)
    b = np.zeros(N)
    a[:n] = y
    a_n = n
    a = np.zeros(N)
    a[:a_n] = y
    bl = q.size
    b[:bl] = q
    conv = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real[:need]
    block = conv[n - 1:2 * n - 1]    # positions n-1 .. 2n-2
    return block[::-1]               # index i -> position 2n-2-i


def _obj_grad(z, t):
    """Smooth surrogate score and gradient w.r.t. z (where f = softplus(z))."""
    y = _softplus(z)
    y = np.maximum(y, 1e-300)
    n = y.size

    c = _convolve_self(y)
    cm = float(c.max())
    e = np.exp((c - cm) / t)
    es = float(e.sum())
    p = e / es
    L = cm + t * np.log(es)          # log-sum-exp (smooth max)

    S = float(y.sum())

    rho = _cross_soft(p, y)           # rho[i] = sum_j p[i+j] y[j]
    dL_dy = 2.0 * rho                 # dL/dy_i = sum_k p_k d c_k/ d y_i = 2 rho_i

    # F = 2 n L / S^2  ->  dF/dy_i = (2 n / S^3) * (S * dL_i - 2 L)
    dF_dy = (2.0 * n) / (S ** 3) * (S * dL_dy - 2.0 * L)

    sig = 1.0 / (1.0 + np.exp(-np.clip(z, -50.0, 50.0)))
    grad = dF_dy * sig

    F = 2.0 * n * L / (S * S)
    if not np.isfinite(F):
        F = 1e300
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return F, grad


def _init_exponential(n, beta):
    """Exponential-decay start: f(i) = exp(-beta * i/(n-1)) (asymmetric)."""
    if n <= 1:
        return np.array([1.0])
    x = np.arange(n, dtype=np.float64)
    y = np.exp(-beta * x / (n - 1))
    return np.log(np.maximum(y, 1e-30) - 1e-30)  # z such that softplus(z) ~ y approx


def _init_parent(n):
    """Resample / build a non-negative start from the provided parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, None)
    if parent.size != n:
        # linear resample to length n
        xp = np.linspace(0.0, 1.0, parent.size)
        xp_new = np.linspace(0.0, 1.0, n)
        parent = np.interp(xp_new, xp, parent)
    parent = np.maximum(parent, 1e-3)
    s = float(parent.sum())
    if s < 1e-12:
        parent = np.ones(n)
    parent = parent / s
    # z with softplus(z) ~ parent: softplus(z)=y -> z = log(expm1(y))
    z = np.log(np.expm1(np.clip(parent, 1e-8, 50.0)))
    return z


def _run_anneal(z0, temps, deadline):
    """Run L-BFGS-B across a temperature anneal; return best z and best exact score."""
    best_z = z0.copy()
    best_exact = float(evaluate_sequence(_softplus(z0).tolist()))
    for t in temps:
        if time.monotonic() > deadline:
            break
        res = minimize(
            _obj_grad,
            z0.copy(),
            args=(t,),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 400, "maxfun": 8000, "ftol": 1e-12, "gtol": 1e-8},
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            z0 = res.x
            cand = _softplus(z0)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_exact:
                best_exact = sc
                best_z = z0.copy()
    return best_z, best_exact, z0


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the non-negative coefficient sequence minimizing the score."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 25.0)

    # ---- guaranteed sub-2 fallback from an asymmetric exponential start ----
    fallback_n = 2048
    fallback_z = _init_exponential(fallback_n, 1.0)
    best = _softplus(fallback_z)
    best_score = float(evaluate_sequence(best.tolist()))
    print("initial fallback score = %.6f  (n=%d)" % (best_score, best.size))

    temps_coarse = [0.5, 0.1, 0.02]
    temps_fine = [0.1, 0.02, 0.004]

    # pool of (n, kind) jobs to try
    jobs = []
    for n in [1024, 2048, 4096, 8192]:
        jobs.append((n, "parent"))
        for beta in [0.8, 1.0, 1.2]:
            jobs.append((n, ("exp", beta)))
    # a few random starts
    for n in [2048, 4096]:
        for _ in range(2):
            jobs.append((n, "rand"))

    attempt = 0
    for (n, kind) in jobs:
        if time.monotonic() > deadline:
            break
        attempt += 1
        if kind == "parent":
            z0 = _init_parent(n)
        elif kind == "rand":
            z0 = rng.normal(0.0, 0.6, size=n)
        else:
            beta = kind[1]
            z0 = _init_exponential(n, beta)

        z0 = np.clip(z0, -30.0, 30.0)
        t0 = time.monotonic()
        if time.monotonic() > deadline:
            continue
        _, sc, z_end = _run_anneal(z0.copy(), temps_coarse, deadline)
        if time.monotonic() > deadline:
            break
        # refine with finer temperatures from the coarse solution
        z_end2, sc2, _ = _run_anneal(z_end.copy(), temps_fine, deadline)
        cand = _softplus(z_end2)
        exact = float(evaluate_sequence(cand.tolist()))
        dt = time.monotonic() - t0
        print("attempt %2d  n=%d  kind=%s  exact=%.6f  (%.1fs)"
              % (attempt, n, str(kind), exact, dt))
        if np.isfinite(exact) and exact < best_score:
            best = cand
            best_score = exact
            print("   -> new best %.6f" % best_score)

    # final safety: make sure we return a finite, non-empty, non-negative list
    best = np.asarray(best, dtype=np.float64)
    if not np.all(np.isfinite(best)) or best.size == 0:
        best = _softplus(fallback_z)
    best = np.clip(best, 0.0, None)
    if float(best.sum()) < 0.01:
        best = best + 1.0
    sc_final = float(evaluate_sequence(best.tolist()))
    print("FINAL score = %.6f  (returned length %d)" % (sc_final, best.size))
    return [float(x) for x in best]
