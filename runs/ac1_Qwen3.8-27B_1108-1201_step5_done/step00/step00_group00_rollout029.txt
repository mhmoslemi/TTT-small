"""
Mirror descent optimization to find non-negative sequences minimizing
2n * max(conv(a,a)) / sum(a)^2. Uses FFT-based convolution, projected
multiplicative updates (mirror descent on the simplex), smoothed gradients,
and multiple structured starting points with adaptive step sizes.
"""
import numpy as np
import time


def _fft_conv_sq(a):
    """Compute full linear convolution a*a using real FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    ca = np.fft.irfft(fa * fa, fsize)
    return ca[:size]


def _simplex_project(v, s=1.0):
    """Project vector v onto the probability simplex {x>=0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = u * np.arange(1, n + 1) > (cssv - s)
    nz = np.nonzero(idx)[0]
    if len(nz) == 0:
        return np.full(n, s / n)
    rho = nz[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _run_mirror_descent(a0, deadline, rng):
    """Run mirror descent to minimize peak convolution. Returns (best_a, best_score)."""
    a = _simplex_project(a0.copy(), 1.0)
    n = len(a)
    if n < 2:
        return a, np.inf

    eta = 4.0 / np.sqrt(n)
    eta_decay = 0.9997
    eta_min = 1e-10 / n
    best_local = np.inf
    best_local_a = a.copy()
    patience = 0
    patience_max = 3000
    top_frac = 0.05
    score = np.inf

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        c = _fft_conv_sq(a)
        max_c = np.max(c)
        score = 2.0 * n * max_c
        if score < best_local:
            best_local = score
            best_local_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > patience_max:
            break

        # Find near-max positions for smoothed gradient
        thresh = 0.90 * max_c
        top_pos = np.nonzero(c >= thresh)[0]
        if len(top_pos) == 0:
            top_pos = np.array([int(np.argmax(c))])
        max_top = max(1, min(30, int(n * top_frac)))
        if len(top_pos) > max_top:
            sel = rng.choice(len(top_pos), max_top, replace=False)
            top_pos = top_pos[sel]

        # Compute averaged subgradient
        grad = np.zeros(n, dtype=np.float64)
        for k in top_pos:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                # grad[i] += 2*a[k-i] for i in [lo, hi]
                # a[k-lo : k-hi+1] reversed gives a[k-hi], a[k-hi+1], ..., a[k-lo]
                grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= len(top_pos)

        # Mirror (multiplicative) update
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
            break
        a /= s

        # Adaptive step size
        eta = max(eta * eta_decay, eta_min)

        # Periodic perturbation to escape local minima
        if it > 500 and it % 2000 == 1000:
            noise = rng.normal(0.0, 0.02, n)
            a = _simplex_project(a * (1.0 + noise), 1.0)
            eta = max(eta * 2.0, 2.0 / np.sqrt(n))
            patience = 0

    # Final evaluation
    if n >= 2:
        c = _fft_conv_sq(a)
        final_score = 2.0 * n * np.max(c)
        if final_score < best_local:
            best_local = final_score
            best_local_a = a.copy()

    return best_local_a, best_local


def _block_local_search(a0, deadline, rng, score0):
    """Local search with block multiplicative perturbations."""
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_score = score0
    iterations = 0

    while time.monotonic() < deadline:
        # Random block perturbation
        width = int(rng.integers(max(1, n // 500), max(2, n // 50)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.15))
        candidate = a.copy()
        candidate[start:stop] *= factor
        # Optional: symmetrize
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)
        s = candidate.sum()
        if s < 0.01:
            continue

        c = _fft_conv_sq(candidate)
        score = 2.0 * n * np.max(c) / (s * s)
        iterations += 1

        if score < best_score:
            best_score = score
            best = candidate.copy()

        # Accept with some probability (simulated annealing style)
        if score < score0 or rng.random() < np.exp(-(score - score0) / 0.01):
            a = candidate

        if iterations % 5000 == 0:
            score0 = np.max([best_score, 1.0])

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Build list of starting points
    starters = []

    # From parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1e6)
        if p.size > 10 and p.sum() > 0.01 and np.isfinite(p).all():
            starters.append(("parent", p / p.sum()))
    except Exception:
        pass

    # Uniform sequences of various lengths
    for n in [1024, 2048, 4096, 8192, 12288]:
        starters.append((f"uni_{n}", np.ones(n, dtype=np.float64) / n))

    # Hann windows
    for n in [2048, 4096, 8192]:
        j = np.arange(n, dtype=np.float64)
        h = 0.5 * (1.0 - np.cos(2.0 * np.pi * (j + 0.5) / n))
        starters.append((f"hann_{n}", h / h.sum()))

    # Triangular windows
    for n in [2048, 4096]:
        j = np.arange(n, dtype=np.float64)
        tri = np.minimum(j + 1.0, n - j)
        starters.append((f"tri_{n}", tri / tri.sum()))

    # U-shaped (inverse triangular)
    for n in [2048, 4096]:
        j = np.arange(n, dtype=np.float64)
        u = np.maximum(np.minimum(j + 1.0, n - j) - n / 4.0, 0.01)
        starters.append((f"u_{n}", u / u.sum()))

    # Parabolic
    for n in [2048, 4096]:
        j = np.arange(n, dtype=np.float64)
        x = 2.0 * (j + 0.5) / n - 1.0
        par = 1.0 - x * x
        par = np.maximum(par, 0.01)
        starters.append((f"par_{n}", par / par.sum()))

    # Phase-allocate time among starters
    t_start = time.monotonic()
    idx = 0
    for name, s in starters:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        remaining_starters = len(starters) - idx
        alloc = remaining / remaining_starters
        # Leave some time for local search at the end
        if idx == len(starters) - 1:
            alloc = remaining * 0.6
        sub_deadline = time.monotonic() + max(2.0, alloc * 0.7)

        a, score = _run_mirror_descent(s, sub_deadline, rng)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{name}] mirror descent score={score:.6f} n={len(a)}")
        else:
            print(f"[{name}] mirror descent score={score:.6f} (no improvement, best={best_score:.6f})")
        idx += 1

    # Local search phase on the best sequence
    if best_seq is not None and time.monotonic() < deadline:
        ls_deadline = time.monotonic() + max(1.0, (deadline - time.monotonic()) * 0.8)
        ls_a, ls_score = _block_local_search(best_seq, ls_deadline, rng, best_score)
        if ls_score < best_score:
            best_score = ls_score
            best_seq = ls_a.copy()
            print(f"[local_search] improved to {best_score:.6f}")
        else:
            print(f"[local_search] no improvement (best={best_score:.6f})")

    # One more round of mirror descent from the best found
    if best_seq is not None and time.monotonic() < deadline - 5:
        md_deadline = deadline
        md_a, md_score = _run_mirror_descent(best_seq, md_deadline, rng)
        if md_score < best_score:
            best_score = md_score
            best_seq = md_a.copy()
            print(f"[final_mirror] improved to {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64) / 256.0
        best_score = 2.0

    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
