"""
Multi-scale dual multiplicative-weights search.

The objective is scale invariant, so all sequences are kept on the simplex.
For a fixed current sequence f, the direction-generating LP
    min M s.t. conv(f, g) <= M, sum(g) = 1, g >= 0
is solved approximately by multiplicative weights on its dual.  The primal
best-response points are mixed back into f by exact/FFT line search.
Several lengths and initial curves are explored.
"""

import time

import numpy as np
from operator import itemgetter

_LEN_CACHE = {}


def get_fft_len(target):
    """Next power-of-two FFT length at least target."""
    cached = _LEN_CACHE.get(target)
    if cached is not None:
        return cached
    L = 1
    while L < target:
        L <<= 1
    _LEN_CACHE[target] = L
    return L


def clip_normalize(f):
    """Clip to non-negative and rescale to sum one."""
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return np.ones(1, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(f.size, dtype=np.float64) / float(f.size)
    return f / s


def fast_score(f):
    """FFT approximation to the official score for sum(f) = 1."""
    f = clip_normalize(f)
    n = f.size
    if n == 0:
        return np.inf
    L = get_fft_len(2 * n - 1)
    F = np.fft.rfft(f, L)
    c = np.fft.irfft(F * F, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m)


def exact_score(f):
    """Exact official score, with safe conversion to a Python list."""
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return np.inf
    return float(evaluate_sequence([float(x) for x in f]))


def resample_to(f, n):
    """Resample a non-negative sequence to length n."""
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if f.size == n:
        return clip_normalize(f)
    xp = np.linspace(0.0, 1.0, f.size)
    xn = np.linspace(0.0, 1.0, n)
    g = np.interp(xn, xp, f)
    return clip_normalize(g)


def initial_candidates(n, rng, best_f):
    """Generate diverse non-negative starting sequences length n."""
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64)
    den = max(1, n - 1)
    c = []

    c.append(np.ones(n, dtype=np.float64) / float(n))
    c.append(0.1 + rng.random(n).astype(np.float64))

    for p in (0.5, 1.0, 1.5, 2.0, 4.0):
        c.append(1.0 / (i + 1.0) ** p + 0.05)

    for a in (0.5, 1.0, 2.0, 4.0):
        c.append(np.exp(-a * i / den))
        c.append(np.exp(-a * (n - 1.0 - i) / den))

    c.append(1.0 - np.abs(i / den - 0.5) * 2.0)

    base = np.ones(n, dtype=np.float64)
    for alpha, beta in ((0.5, 0.0), (1.0, 0.0), (2.0, 0.0),
                        (0.5, 0.5), (0.0, 1.0), (1.0, 2.0)):
        b = base.copy()
        b[0] += alpha
        b[-1] += beta
        c.append(b)

    mid = base.copy()
    mid[n // 2] += 2.0
    c.append(mid)

    if best_f is not None and best_f.size >= 2:
        c.append(resample_to(best_f, n))

    out = []
    for a in c:
        out.append(clip_normalize(a))
    return out


def scan_initials(n, rng, best_f, best_score):
    """Score a small pool of initial sequences for length n."""
    n = max(int(n), 2)
    cands = initial_candidates(n, rng, best_f)

    scored = []
    for a in cands:
        sc = fast_score(a)
        if np.isfinite(sc):
            scored.append((sc, a))

    local_f = cands[0] if cands else None
    local_score = np.inf

    if scored:
        scored.sort(key=itemgetter(0))
        for sc, a in scored[:3]:
            se = exact_score(a)
            if np.isfinite(se):
                if se < local_score:
                    local_score = se
                    local_f = a.copy()
                if se < best_score:
                    best_f = a.copy()
                    best_score = se

    return best_f, best_score, local_f


def mw_s(y, Fflip, L, n):
    """Row payoffs s_j = sum_k y_k f[k-j] using FFT correlation."""
    q = np.fft.irfft(np.fft.rfft(y, L) * Fflip, L)
    return q[n - 1:n - 1 + n]


def setup_mw(f):
    """Initial multiplicative-weights state for a fixed f."""
    n = f.size
    fmax = float(np.max(f)) if f.size else 1.0
    if fmax <= 1e-300:
        fmax = 1.0
    fm = np.maximum(f / fmax, 1e-9)
    L = get_fft_len(3 * n - 2)
    F = np.fft.rfft(fm[::-1], L)
    y = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=np.float64)
    return fm, F, y, L


def refresh_mw(f, y):
    """Refresh f-dependent MW data while keeping the existing y."""
    n = f.size
    fmax = float(np.max(f)) if f.size else 1.0
    if fmax <= 1e-300:
        fmax = 1.0
    fm = np.maximum(f / fmax, 1e-9)
    L = get_fft_len(3 * n - 2)
    F = np.fft.rfft(fm[::-1], L)
    return fm, F, L


def make_g_candidates(s, rng):
    """Primal best-response/several soft responses from row payoffs s."""
    n = s.size
    if n == 0:
        return []
    if not np.all(np.isfinite(s)):
        return [np.ones(n, dtype=np.float64) / float(n)]

    out = []
    mn = float(np.min(s))
    rng_s = float(np.max(s) - mn)
    denom = rng_s + 1e-300

    tol = max(1e-15, 0.02 * rng_s)
    act = s <= (mn + tol)
    need = min(5, n)
    if np.count_nonzero(act) < need:
        idx = np.argpartition(s, need - 1 if need > 1 else 0)[:need]
        act[idx] = True
    g = np.zeros(n, dtype=np.float64)
    g[act] = 1.0 / float(np.count_nonzero(act))
    out.append(g)

    g = np.zeros(n, dtype=np.float64)
    g[int(np.argmin(s))] = 1.0
    out.append(g)

    for beta in (3.0, 10.0, 30.0):
        g = np.exp(-beta * (s - mn) / denom)
        g /= g.sum()
        out.append(g)

    g = 1.0 / (s + 1e-300)
    g /= g.sum()
    out.append(g)

    return out


def try_candidate(h, best_score):
    """If h is clearly better by FFT score, validate exactly."""
    h = clip_normalize(h)
    sc = fast_score(h)
    if not np.isfinite(sc) or not np.isfinite(best_score):
        return None
    threshold = max(1e-9, 1e-7 * best_score)
    if sc < best_score - threshold:
        se = exact_score(h)
        if np.isfinite(se) and se < best_score:
            return h, se
    return None


def try_blend(f, g, best_score):
    """Line-search a mixture of f and a simplex direction g."""
    for t in (1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05):
        h = (1.0 - t) * f + t * g
        res = try_candidate(h, best_score)
        if res is not None:
            return res
    return None


def stochastic_move(cur, best, rng, strength=0.1):
    """Random feasible perturbation used to escape stagnation."""
    f = cur.copy()
    n = f.size
    if n == 0:
        return cur
    method = int(rng.integers(6))

    if method == 0:
        f += strength * rng.standard_normal(n)
    elif method == 1:
        f *= np.exp(strength * rng.standard_normal(n))
    elif method == 2:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        delta = 0.1 * f[i]
        f[i] -= delta
        f[j] += delta
    elif method == 3:
        delta = 0.05 * f.sum()
        idx = 0 if rng.random() < 0.5 else n - 1
        f[idx] += delta
    elif method == 4:
        width = max(1, n // 50)
        start = int(rng.integers(0, max(1, n - width + 1)))
        f[start:start + width] *= np.exp(0.2 * rng.standard_normal(width))
    else:
        # Occasionally inherit from the global/local best.
        f = best.copy()
        f += strength * rng.standard_normal(n)

    return clip_normalize(f)


def dual_optimize(f0, alloc_s, seed):
    """Iterative dual-WF + feasible blend optimizer for one length."""
    f = clip_normalize(f0)
    n = f.size
    if n < 2 or alloc_s <= 0.0:
        return f, exact_score(f)

    best = f.copy()
    best_score = exact_score(best)
    if not np.isfinite(best_score):
        best = clip_normalize(np.ones(n, dtype=np.float64) / float(n))
        best_score = exact_score(best)

    cur = f.copy()
    rng = np.random.default_rng(seed)
    fm, Fflip, y, L = setup_mw(cur)

    eta = 12.0
    topk = 7
    stuck = 0
    start = time.monotonic()
    stop = start + float(alloc_s)

    while time.monotonic() < stop:
        t0 = time.monotonic()
        s = None

        while time.monotonic() - t0 < 0.6:
            if time.monotonic() >= stop:
                break
            s = mw_s(y, Fflip, L, n)
            if not np.all(np.isfinite(s)):
                y.fill(1.0 / float(2 * n - 1))
                s = np.full(n, 1.0 / float(n))

            k = min(topk, n)
            if k == 1:
                j = int(np.argmin(s))
            else:
                idx = np.argpartition(s, k - 1)[:k]
                j = int(idx[int(rng.integers(k))])

            y[j:j + n] *= np.exp(eta * fm[j:j + n])
            ssum = y.sum()
            if not np.isfinite(ssum) or ssum <= 1e-300:
                y.fill(1.0 / float(2 * n - 1))
            else:
                y /= ssum

        s = mw_s(y, Fflip, L, n)
        if not np.all(np.isfinite(s)):
            s = np.full(n, 1.0 / float(n))

        improved = False
        for g in make_g_candidates(s, rng):
            res = try_blend(cur, g, best_score)
            if res is not None:
                cur, sc = res
                if sc < best_score:
                    best = cur.copy()
                    best_score = sc
                improved = True
                fm, Fflip, L = refresh_mw(cur, y)
                stuck = 0
                break

        if not improved:
            stuck += 1
            if stuck % 4 == 0:
                strength = 0.08 + 0.015 * min(12, stuck)
                cur = stochastic_move(cur, best, rng, strength)
                fm, Fflip, L = refresh_mw(cur, y)

            if stuck > 36:
                cur = stochastic_move(best, best, rng, 0.25)
                fm, Fflip, y, L = setup_mw(cur)
                stuck = 0

    se = exact_score(cur)
    if np.isfinite(se) and se < best_score:
        best = cur.copy()
        best_score = se

    return best, best_score
