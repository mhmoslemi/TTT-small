def _stable_softmax(z):
    z = np.asarray(z, dtype=np.float64)
    m = np.max(z)
    e = np.exp(z - m)
    s = e.sum()
    if not np.isfinite(s) or s == 0:
        out = np.zeros_like(e)
        out[np.argmax(z)] = 1.0
        return out
    return e/s

def _lin_conv(a,b):
   nout=len(a)+len(b)-1
   L=1 << (nout-1).bit_length()
   fa=np.fft.rfft(a,L); fb=np.fft.rfft(b,L)
   out=np.fft.irfft(fa*fb,L)
   return out[:nout]

def _autocorr(x): return _lin_conv(x,x)

```
def resample_probs(x0, n):
    x0=np.clip(x0,0,1000)
    s=x0.sum(); if s<=0: return np.full(n,1.0/n)
    x0=x0/s
    n0=len(x0)
    cum=np.concatenate(([0.0], np.cumsum(x0)))
    edges_fine = np.linspace(0.0,1.0,n+1)
    # map fine edges to position in coarse edges 0..n0
    pos = edges_fine * n0
    left = np.floor(pos).astype(int)

    x = np.clip(x,0,1)
    s=x.sum(); if s>0: x=x/s
    return x
```

```
def _stable_softmax(y):
    y = np.asarray(y, dtype=np.float64)
    m = np.max(y)
    if not np.isfinite(m):
       return np.full(y.shape, 1.0/y.size)
    e = np.exp(y - m)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
       out=np.zeros_like(e)
       idx=int(np.argmax(y if np.all(np.isfinite(y)) else np.zeros_like(y)))
       out[idx]=1.0
       return out
    return e/s
```

```
def _lin_conv(a,b):
    a=np.asarray(a,dtype=np.float64); b=...
    if a.size==0 or b.size==0: return np.empty(0)
    nout=a.size+b.size-1
    if a.size*b.size <= 300000:
       return np.convolve(a,b,mode='full')
    L=1 << (nout-1).bit_length()
    fa=np.fft.rfft(a,L)
    fb=np.fft.rfft(b,L)
    out=np.fft.irfft(fa*fb,L)
    return out[:nout]
```

```
def resample_probs(x0, n):
   n=int(n)
   x=_normalize(x0)
   if x.size == n: return x
   if n <=0: return np.empty(0)
   n0=x.size
   if n0 ==0: return np.full(n,1.0/n)
   cum=np.empty(n0+1); cum[0]=0; cum[1:]=np.cumsum(x)
   edges=np.linspace(0.0,1.0,n+1)
   pos=edges*n0
   left=np.floor(pos).astype(np.int64)
   left=np.clip(left,0,n0-1)
   nxt=left+1
   frac=pos-left
   F=cum[left]+frac*(cum[nxt]-cum[left])
   out=np.diff(F)
   out=np.where(np.isfinite(out), out, 0.0)
   out=np.clip(out,0.0,1.0)
   s=out.sum()
   if s<=1e-300: return np.full(n,1.0/n)
   return out/s
```

```python
import numpy as np
import time


_TARGET_SCORE = 1.5027


def _normalize(x):
    arr = np.asarray(x, dtype=np.float64).reshape(-1)
    if arr.size == 0:
        return arr
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-300:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _score_key(item):
    return item[0]


def _lin_conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    if a.size == 0 or b.size == 0:
        return np.empty(0, dtype=np.float64)
    nout = a.size + b.size - 1
    if a.size * b.size <= 300000:
        return np.convolve(a, b, mode="full")
    L = 1 << (nout - 1).bit_length()
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout]


def _autocorr_full(x):
    return _lin_conv(x, x)


def _stable_softmax(y):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return y
    if not np.all(np.isfinite(y)):
        y = np.where(np.isfinite(y), y, 0.0)
    m = np.max(y)
    if not np.isfinite(m):
        return np.full(y.shape, 1.0 / float(y.size))
    e = np.exp(y - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        out = np.zeros_like(e)
        out[int(np.argmax(y))] = 1.0
        return out
    return e / s


def _softmax_weights(r, lam):
    r = np.asarray(r, dtype=np.float64)
    m = float(np.max(r))
    z = float(lam) * (r - m)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(r)
        w[int(np.argmax(r))] = 1.0
        return w
    return e / s


def _weighted_shift_conv(x, w):
    n = len(x)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    wrev = w[::-1]
    conv = _lin_conv(x, wrev)
    return conv[n - 1:2 * n - 1][::-1]


def _add_smooth_noise(y, rng, n, scale):
    y = np.asarray(y, dtype=np.float64)
    if n <= 1:
        return y
    noise = rng.normal(size=n)
    spec = np.fft.rfft(noise)
    freqs = np.fft.rfftfreq(n)
    band = np.exp(-5.0 * freqs)
    noise = np.fft.irfft(spec * band, n)
    std = float(noise.std())
    if std > 1e-12:
        noise = noise / std
    return y + float(scale) * noise


def _score_x(x):
    x = _normalize(x)
    if x.size == 0:
        return np.inf
    r = _autocorr_full(x)
    if not np.all(np.isfinite(r)):
        return np.inf
    m = float(max(0.0, np.max(r)))
    return float(2.0 * x.size * m)


def local_optimize(x0, time_limit=1.0, lam_factor_start=1.0, lam_factor_end=150.0,
                   lr=0.08, patience=400, seed=0):
    x = _normalize(x0)
    n = x.size
    if n == 0:
        return x, np.inf

    deadline = time.monotonic() + max(0.0, float(time_limit))
    y = np.log(np.maximum(x, 1e-300))
    y -= np.max(y)

    best_y = y.copy()
    best_x = x.copy()
    r = _autocorr_full(x)
    best_score = float(2.0 * n * max(0.0, np.max(r)))

    rng = np.random.default_rng(seed)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    it = 0
    last_best = 0
    step_factor = 1.0

    lam = max(1.0, float(lam_factor_start) * float(n))
    lam_max = max(lam, float(lam_factor_end) * float(n))

    while time.monotonic() < deadline:
        it += 1

        if not np.all(np.isfinite(y)):
            y = best_y.copy()
        y -= np.max(y)

        x = _stable_softmax(y)
        r = _autocorr_full(x)
        if not np.all(np.isfinite(r)):
            y = best_y.copy()
            step_factor = max(0.2, step_factor * 0.5)
            continue

        rmax = float(max(0.0, np.max(r)))
        score = float(2.0 * n * rmax)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_x = x.copy()
            best_y = y.copy()
            last_best = it
            step_factor = min(1.4, step_factor * 1.04)

        if (not np.isfinite(score)) or score > 4.0 * best_score + 1e-3:
            y = best_y.copy()
            step_factor = max(0.2, step_factor * 0.5)
            continue

        no_imp = it - last_best
        if no_imp > patience:
            w = _softmax_weights(r, lam)
            ent = -float(np.dot(w, np.log(np.maximum(w, 1e-300))))
            if ent < 1e-3 and lam > max(10.0, float(n)):
                lam = max(float(n), lam * 0.7)
            elif lam < 0.95 * lam_max:
                lam = min(lam_max, lam * 1.25)
            else:
                lam = max(float(n), lam * 0.8)
            y = _add_smooth_noise(y, rng, n, 0.02)
            y -= np.max(y)
            last_best = it
            patience = max(60, int(0.8 * patience))
            continue

        w = _softmax_weights(r, lam)
        c = _weighted_shift_conv(x, w)
        gx = 2.0 * lam * c
        gx = np.where(np.isfinite(gx), gx, 0.0)
        dot = float(np.dot(x, gx))
        gy = x * (gx - dot)
        gy = np.where(np.isfinite(gy), gy, 0.0)

        if float(np.max(np.abs(gy))) <= 1e-15:
            y = _add_smooth_noise(y, rng, n, 0.01)
            y -= np.max(y)
            last_best = it
            continue

        t += 1
        b1 = 0.9
        b2 = 0.99
        m = b1 * m + (1.0 - b1) * gy
        v = b2 * v + (1.0 - b2) * (gy * gy)
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        y = y - float(lr) * step_factor * mhat / (np.sqrt(vhat) + 1e-8)
        y -= np.max(y)

    if time.monotonic() < deadline + 1.0:
        if not np.all(np.isfinite(y)):
            y = best_y.copy()
        y -= np.max(y)
        x = _stable_softmax(y)
        r = _autocorr_full(x)
        score = float(2.0 * n * max(0.0, np.max(r)))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_x = x.copy()

    return best_x, best_score


def resample_probs(x0, n):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    x = _normalize(x0)
    if x.size == 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if x.size == n:
        return x

    n0 = x.size
    cum = np.empty(n0 + 1, dtype=np.float64)
    cum[0] = 0.0
    cum[1:] = np.cumsum(x)

    edges = np.linspace(0.0, 1.0, n + 1)
    pos = edges * float(n0)
    left = np.floor(pos).astype(np.int64)
    left = np.clip(left, 0, n0 - 1)
    nxt = left + 1
    frac = pos - left
    F = cum[left] + frac * (cum[nxt] - cum[left])
    out = np.diff(F)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return out / s


def _perturb_linear(x, rng, amp=0.5):
    x = _normalize(x)
    n = x.size
    if n == 0:
        return x
    t = np.linspace(0.0, 1.0, n)
    mult = 1.0 + float(amp) * (t - 0.5) + 0.05 * rng.normal(size=n)
    mult = np.clip(mult, 1e-8, None)
    y = x * mult
    y = np.clip(y, 0.0, 1000.0)
    s = float(y.sum())
    if s <= 1e-300:
        return x
    return y / s


def make_fourier_start(n, rng, modes=8, amp=1.5):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    phase = np.zeros(n, dtype=np.float64)
    for m in range(1, int(modes) + 1):
        scale = float(amp) / max(1.0, np.sqrt(float(m)))
        a = float(rng.normal(0.0, scale))
        b = float(rng.normal(0.0, scale))
        ph = 2.0 * np.pi * float(m) * t
        phase += a * np.cos(ph) + b * np.sin(ph)
    phase += float(rng.normal(0.0, float(amp) * 0.5)) * (t - 0.5)
    x = np.exp(phase - np.max(phase))
    x = np.where(np.isfinite(x), x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return x / s


def make_power_start(n, a, b, eps=1e-3):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    x = (t + float(eps)) ** float(a) * (1.0 - t + float(eps)) ** float(b)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 1e-300, None)
    s = float(x.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return x / s


def make_two_level(n, split, ratio):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    x = np.ones(n, dtype=np.float64)
    cut = int(round(float(split) * float(n)))
    cut = min(max(cut, 0), n)
    if cut < n:
        x[cut:] *= float(ratio)
    s = float(x.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return x / s


def make_polyexp_start(n, rng):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    a = float(rng.normal(0.0, 1.5))
    b = float(rng.normal(0.0, 1.5))
    c = float(rng.normal(0.0, 1.0))
    phase = a * t + b * (t * t) + c * (t * t * t)
    x = np.exp(phase - np.max(phase))
    x = np.where(np.isfinite(x), x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return x / s


def make_known_starts(n, rng):
    starts = []
    a = 1.0 / np.sqrt(5.0)
    b = (1.0 - a) / 2.0
    seqs = [
        np.array([a, b, b], dtype=np.float64),
        np.array([b, b, a], dtype=np.float64),
        np.array([b, a, b], dtype=np.float64),
        np.array([a, b, a], dtype=np.float64),
    ]
    for seq in seqs:
        xr = resample_probs(seq, n)
        starts.append(_perturb_linear(xr, rng, 0.25))
    return starts


def make_random_starts(n, rng, count=12):
    n = int(n)
    starts = []

    fourier_count = max(4, int(count))
    for _ in range(fourier_count):
        modes = int(rng.integers(3, 9))
        amp = float(rng.uniform(0.7, 2.2))
        starts.append(make_fourier_start(n, rng, modes, amp))

    poly_count = max(2, int(count) // 3)
    for _ in range(poly_count):
        starts.append(make_polyexp_start(n, rng))

    pows = [
        (0.0, 0.0), (1.0, 0.0), (2.0, 0.0), (0.0, 1.0), (0.0, 2.0),
        (1.0, 1.0), (2.0, 1.0), (1.0, 2.0), (0.5, 2.0), (2.0, 0.5)
    ]
    add_pows = max(0, int(count) - len(starts) + 3)
    for a, b in pows[:add_pows]:
        starts.append(make_power_start(n, a, b))

    for _ in range(max(1, int(count) // 4)):
        split = float(rng.uniform(0.15, 0.85))
        ratio = float(rng.uniform(0.3, 4.0))
        starts.append(make_two_level(n, split, ratio))

    starts.append(_perturb_linear(np.full(n, 1.0 / float(n), dtype=np.float64), rng, 0.9))
    starts.append(make_fourier_start(n, rng, modes=int(rng.integers(4, 10)), amp=float(rng.uniform(1.2, 2.8))))
    return starts


def make_initial_starts(n, global_best, base_sequence, rng, count):
    n = int(n)
    starts = []

    starts.extend(make_known_starts(n, rng))

    if base_sequence is not None:
        rb = resample_probs(base_sequence, n)
        starts.append(_perturb_linear(rb, rng, 0.6))
        starts.append(_perturb_linear(rb, rng, 1.2))

    if global_best is not None:
        rg = resample_probs(global_best, n)
        starts.append(_perturb_linear(rg, rng, 0.5))
        starts.append(_perturb_linear(rg, rng, 1.0))

    starts.extend(make_random_starts(n, rng, count))
    if len(starts) > 44:
        starts = starts[:44]
    return starts


def run_phase(starts, n, deadline, max_runs=8):
    n = int(n)
    items = []
    for x in starts:
        if x is None:
            continue
        arr = np.asarray(x, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_probs(arr, n)
        arr = _normalize(arr)
        sc = _score_x(arr)
        if np.isfinite(sc):
            items.append((sc, arr))
    if not items:
        return None, np.inf

    items.sort(key=_score_key)
    items = items[:int(max_runs)]

    best_x = None
    best_score = np.inf
    rem = max(0.0, float(deadline) - time.monotonic())
    if rem <= 0.0:
        return None, np.inf

    for idx, (sc, x) in enumerate(items):
        rem = float(deadline) - time.monotonic()
        if rem < 0.2:
            break

        if idx == len(items) - 1:
            limit = rem - 0.05
        else:
            per = rem / float(len(items) - idx)
            limit = min(per, max(0.2, rem * 0.5))
        if limit < 0.05:
            limit = max(0.05, rem * 0.5)

        if n <= 128:
            bx, bs = local_optimize(x, limit, lam_factor_start=3.0, lam_factor_end=300.0,
                                    lr=0.13, patience=120, seed=1000 + idx * 7 + n)
        elif n <= 512:
            bx, bs = local_optimize(x, limit, lam_factor_start=2.0, lam_factor_end=220.0,
                                    lr=0.11, patience=250, seed=2000 + idx * 13 + n)
        elif n <= 2048:
            bx, bs = local_optimize(x, limit, lam_factor_start=1.0, lam_factor_end=170.0,
                                    lr=0.08, patience=450, seed=3000 + idx * 17 + n)
        else:
            bx, bs = local_optimize(x, limit, lam_factor_start=1.0, lam_factor_end=160.0,
                                    lr=0.065, patience=750, seed=4000 + idx * 19 + n)

        if np.isfinite(bs) and bx is not None and bs < best_score:
            best_score = bs
            best_x = bx

        print("phase n=%d run=%d score=%.8f best=%.8f" % (n, idx, bs, best_score), flush=True)

    return best_x, best_score


def propose_candidate(seed=12345, budget_s=1800, **kwargs):
    t0 = time.monotonic()
    total = max(60.0, float(budget_s))
    deadline = t0 + total - 20.0
    rng = np.random.default_rng(int(seed))

    base_sequence = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.clip(arr, 0.0, 1000.0)
            if float(arr.sum()) > 0.0:
                base_sequence = arr
    except Exception:
        base_sequence = None

    if total > 1100.0:
        final_n = 4498
    else:
        final_n = 2048
    min_return_n = 2048 if final_n >= 4096 else 1024

    global_best = None
    global_score = np.inf
    large_best = None
    large_score = np.inf
    top_list = []

    phases = [
        (64, 0.05, 18),
        (256, 0.10, 14),
        (1024, 0.30, 8),
    ]

    search_span = max(0.0, float(deadline) - float(t0))
    cum = 0.0
    for idx, (n, frac, max_runs) in enumerate(phases):
        cum += frac
        phase_deadline = t0 + search_span * cum
        if time.monotonic() >= phase_deadline:
            break

        starts = make_initial_starts(n, global_best, base_sequence, rng, max_runs)
        bx, bs = run_phase(starts, n, phase_deadline, max_runs)

        if np.isfinite(bs) and bx is not None:
            if bs < global_score:
                global_score = bs
                global_best = bx
            if bx.size >= min_return_n and bs < large_score:
                large_score = bs
                large_best = bx
            top_list.append((bs, bx, int(n)))
            print("phase n=%d phase_best=%.8f global=%.8f large=%.8f" % (
                n, bs, global_score, large_score), flush=True)

        if time.monotonic() >= deadline:
            break

    final_phase_deadline = deadline - 70.0
    if time.monotonic() < final_phase_deadline:
        final_starts = []
        for sc, x, n in sorted(top_list, key=_score_key):
            final_starts.append(resample_probs(x, final_n))
        if base_sequence is not None:
            final_starts.append(_perturb_linear(resample_probs(base_sequence, final_n), rng, 0.5))
        if global_best is not None:
            final_starts.append(_perturb_linear(resample_probs(global_best, final_n), rng, 0.5))
        if large_best is not None:
            final_starts.append(_perturb_linear(large_best, rng, 0.35))

        final_starts.extend(make_random_starts(final_n, rng, count=4))
        final_starts = final_starts[:9]

        bx, bs = run_phase(final_starts, final_n, final_phase_deadline, max_runs=max(6, len(final_starts)))
        if np.isfinite(bs) and bx is not None:
            if bs < global_score:
                global_score = bs
                global_best = bx
            if bx.size >= min_return_n and bs < large_score:
                large_score = bs
                large_best = bx
            top_list.append((bs, bx, final_n))
            print("final phase best=%.8f global=%.8f large=%.8f" % (
                bs, global_score, large_score), flush=True)

    candidates = []
    if global_best is not None:
        if global_best.size >= min_return_n:
            candidates.append(global_best)
        else:
            candidates.append(resample_probs(global_best, final_n))
    if large_best is not None:
        candidates.append(large_best)
    if base_sequence is not None and base_sequence.size >= min_return_n:
        candidates.append(_normalize(base_sequence))

    chosen = None
    chosen_score = np.inf
    for cand in candidates:
        cand = _normalize(cand)
        if cand.size < min_return_n:
            cand = resample_probs(cand, final_n)
        sc = _score_x(cand)
        if np.isfinite(sc) and sc < chosen_score:
            chosen_score = sc
            chosen = cand

    if chosen is None:
        chosen = np.full(final_n, 1.0 / float(final_n), dtype=np.float64)
        chosen_score = _score_x(chosen)

    rem = float(deadline) - time.monotonic() - 12.0
    if chosen is not None and rem > 45.0:
        nch = chosen.size
        if nch <= 512:
            params = dict(lam_factor_start=2.0, lam_factor_end=220.0, lr=0.10, patience=300)
        elif nch <= 2048:
            params = dict(lam_factor_start=1.0, lam_factor_end=180.0, lr=0.08, patience=500)
        else:
            params = dict(lam_factor_start=1.0, lam_factor_end=170.0, lr=0.05, patience=850)
        bx, bs = local_optimize(chosen, rem, seed=987654, **params)
        if np.isfinite(bs) and bx is not None and bs < chosen_score:
            chosen = bx
            chosen_score = bs

    best_list = [float(v) for v in chosen]
    if not best_list or not all(np.isfinite(v) for v in best_list):
        fb = np.full(final_n, 1.0 / float(final_n), dtype=np.float64)
        best_list = [float(v) for v in fb]

    try:
        es = evaluate_sequence(best_list)
        print("final evaluator score=%.8f length=%d" % (es, len(best_list)), flush=True)
        if not np.isfinite(es):
            fb = np.full(final_n, 1.0 / float(final_n), dtype=np.float64)
            best_list = [float(v) for v in fb]
    except Exception:
        pass

    return best_list
```
