def fast_convolve(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    result = np.fft.irfft(A * A, size)
    return result[:2*n-1]

def generate_parametric(n, harmonics, rng):
    """Generate a random parametric sequence."""
    j = np.arange(n)
    a = np.ones(n)
    for k in harmonics:
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2*np.pi)
        a += amp * np.cos(2*np.pi*k*j/n + phase)
    return np.clip(a, 0, 1000)

def pressure_step(a, step_size, k_star):
    """Apply pressure step toward k_star."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):

        i2 = j - k_star
        if 0 <= i1 < n:
            g[j] += a[i1]
        if 0 <= i2 < n:
            g[j] += a[i2]
    g_centered = g - g.mean()
    a_new = a - step_size * g_centered
