    n = len(a)
    a = a.copy()
    a = a / a.sum()  # normalize
    
    for it in range(num_iters):
        c = np.convolve(a, a)
        # Compute pressure P[i] = sum_k c[k] * a[k-i] (weighted by c[k])
        # Or simpler: just use the max lag
        k_star = np.argmax(c)
        # Gradient of c[k_star] w.r.t. a[i] is 2*a[k_star - i]
        # P[i] = a[k_star - i] for i where 0 <= k_star - i <= n-1
        P = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                P[i] = a[j]
        
        a = a * np.exp(-eta * P)
        a = np.clip(a, 0, None)
        a = a / a.sum()
    
    return a

from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def compute_score(a):
    """Compute the evaluation score for a sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_b = c.max()
    return 2 * n * max_b / s**2

def multiplicative_weight_update(a, eta, num_iters, verbose=False):
    """Apply multiplicative weight updates to flatten the convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s  # normalize to sum 1
    
    for it in range(num_iters):
        c = np.convolve(a, a)
        # Compute pressure P[i] = sum_j a[j] * c[i+j]
        # P = conv(a[::-1], c)[n-1:2n-1]
        full_conv = np.convolve(a[::-1], c)
        P = full_conv[n-1:2*n-1]
        
        P_max = P.max()
        if P_max > 1e-30:
            P = P / P_max
            a = a * np.exp(-eta * P)
            a = np.clip(a, 1e-15, None)
            a = a / a.sum()
        
        if verbose and it % 1000 == 0:
            c = np.convolve(a, a)
            score = 2 * n * c.max() / a.sum()**2
            print(f"  MW iter {it}: score = {score:.6f}, peak = {c.max():.6f}")
    
    return a

def lp_direction(a, M):
    """Solve LP to find direction with conv(a, g) <= M, maximizing sum(g)."""
    n = len(a)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                     bounds=[(0, None)]*n, method='highs')
    
    if result.status == 0:
        return result.x, -result.fun
    return None, 0.0

def lp_iteration(a, t=0.1, verbose=False):
    """One LP-based iteration."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    c = np.convolve(a, a)
    M = c.max()
    
    g, sum_g = lp_direction(a, M)
    
    if g is None or sum_g <= s + 1e-10:
        return a
    
    # Normalize g to have same sum as a
    g_norm = a.sum() / sum_g * g
    g_norm = np.clip(g_norm, 0, None)
    g_norm = g_norm / g_norm.sum()
    
    # Update
    a_new = (1 - t) * a + t * g_norm
    a_new = np.clip(a_new, 0, None)
    a_new = a_new / a_new.sum()
    
    if verbose:
        c_new = np.convolve(a_new, a_new)
        print(f"  LP iter: sum_g/sum_a = {sum_g/s:.4f}, new peak = {c_new.max():.6f} (was {M:.6f})")
    
    return a_new

Algorithm: Combined multiplicative weight (MW) updates and LP-based refinement
to find a non-negative sequence with minimal peak convolution value.

The MW phase rapidly reduces the convolution peak by iteratively redistributing
mass from high-pressure to low-pressure positions. The LP phase then refines
the solution using the linear programming approach from the state of the art.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a, b):
    """Compute full convolution using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)[:n_out]
    return np.maximum(result, 0)  # clip small negative values


def compute_score(a):
    """Compute the evaluation score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    max_b = c.max()
    return float(2 * n * max_b / s**2)

    if s < 1e-15:
        return a
