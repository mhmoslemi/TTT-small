from scipy.sparse import csr_matrix

def solve_lp_for_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    f = np.asarray(f, dtype=np.float64)
