```python
"""
LP-guided local search for step function sequences minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses periodic LP direction finding
combined with local coordinate search.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def compute_score(a):
    """Fast score computation without calling evaluate_sequence."""
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    n = len(a)
    return 2.0 * n * max_b / (sum_a ** 2)


def solve_lp_direction(f, M):
    """Solve LP to find direction g: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build sparse A matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # For column j, A[k, j] = f[k-j] for k = j, j+1, ..., j+n-1
        for k_offset in range(n):
            k = j + k_offset
            rows.append(k)
            cols.append(j)
            vals.append(f[k_offset])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    return None

        base = np.clip(base, 0, 1000)
        if base.size > 0 and np.isfinite(base).all():
            candidates.append(base)
    except NameError:

    # Try various structured initializations
    for n in [2000, 4000, 6000, 8000]:
        # Gaussian bump

        bump = 1.0 + 2.0 * np.exp(-50 * (x - 0.3)**2) + 2.0 * np.exp(-50 * (x - 0.7)**2)
        candidates.append(bump)
        
        # Two-level based on quadratic residues
        # Find prime near n
        p = n
        while True:
            p += 1
            # Simple primality test
            is_p = True
            if p < 2:
                is_p = False
            for d in range(2, int(np.sqrt(p)) + 1):
                if p % d == 0:
                    is_p = False
                    break
            if is_p and p % 4 == 3:

        # Quadratic residue pattern
        qr = np.zeros(p)
        for j in range(p):
            if j == 0:

            else:

        # Tile to length n

        rem = n % p
```
