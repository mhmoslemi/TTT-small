def simplex_projection(v):

def compute_conv(a):

def smooth_max_grad(a, beta):

    conv = compute_conv(a)
    shifted = conv - np.max(conv)
    if beta > 500:

"""
Minimize the evaluation score by optimizing sequence length and shape.
Key insight: the objective J = 2n*max(conv(a,a))/(sum(a))^2 is scale-invariant,
so we normalize sum=1. For small n, asymmetric shapes beat the uniform sequence
(which always gives J=2). We use projected gradient with log-sum-exp smoothing
and beta-annealing to find near-optimal shapes for each n, then pick the best.
"""

import numpy as np
import time


def simplex_projection(v):
    """Project v onto the probability simplex {a: a_i >= 0, sum(a_i) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def optimize_for_n(n, rng, total_steps=20000, beta_max=500.0):
    """Optimize a nonnegative unit-sum sequence of length n to minimize max(conv(a,a))."""
    best_a = np.ones(n) / n
    best_max_conv = float(np.max(np.convolve(best_a, best_a)))

    # Multiple initializations
    inits = [np.ones(n) / n]
    ramp = np.arange(1, n + 1, dtype=float)
    inits.append(ramp / ramp.sum())
    inits.append(ramp[::-1] / ramp.sum())
    for _ in range(4):
        r = rng.random(n)
        inits.append(r / r.sum())
    x = np.arange(n, dtype=float)
    for sigma_frac in [0.2, 0.3, 0.5]:
        gauss = np.exp(-0.5 * ((x - n * 0.75) / (n * sigma_frac)) ** 2)
        if gauss.sum() > 0:
            inits.append(gauss / gauss.sum())
    # Power ramp
    for p in [2.0, 0.5]:
        pramp = np.power(np.arange(1, n + 1, dtype=float), p)
        inits.append(pramp / pramp.sum())
        inits.append(pramp[::-1] / pramp.sum())

    beta_levels = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, beta_max]
    steps_per_level = max(100, total_steps // len(beta_levels))

    for a0 in inits:
        a = a0.copy()
        v = np.zeros(n)
        for beta in beta_levels:
            # Reset momentum at each beta level for stability
            v = np.zeros(n)
            for step in range(steps_per_level):
                conv = np.convolve(a, a)
                max_conv = conv.max()
                shifted = conv - max_conv
                exp_vals = np.exp(beta * shifted)
                p = exp_vals / exp_vals.sum()
                # Gradient: 2 * correlate(p, a, 'valid')
                grad = 2.0 * np.correlate(p, a, mode='valid')

                lr = 1.0 / (beta + 1.0)
                v = 0.9 * v + 0.1 * grad
                a = simplex_projection(a - lr * v)

                if max_conv < best_max_conv - 1e-12:
                    best_max_conv = max_conv
                    best_a = a.copy()

    return best_a, best_max_conv


def local_polish(a, n, rng, deadline):
    """Fine local search to polish the best candidate."""
    a = a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / a.sum()
    current_max = float(np.max(np.convolve(a, a)))

    step_count = 0
    while time.monotonic() < deadline:
        step_count += 1
        # Perturb a random subset
        width = max(1, int(rng.integers(1, max(2, n // 10))))
        start = int(rng.integers(0, n - width + 1))
        factors = np.exp(rng.normal(0.0, 0.03, size=width))
        a_new = a.copy()
        a_new[start:start + width] *= factors
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            continue
        a_new = a_new / a_new.sum()

        new_max = float(np.max(np.convolve(a_new, a_new)))
        if new_max < current_max:
            current_max = new_max
            a = a_new
        if step_count % 5000 == 0:
            print(f"  polish: step={step_count}, max_conv={current_max:.8f}, score={2*n*current_max:.6f}")

    return a
