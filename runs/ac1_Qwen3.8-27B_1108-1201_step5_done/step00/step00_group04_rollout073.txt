```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def full_conv(a):
    """Compute np.convolve(a, a, 'full') using FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]

def build_conv_matrix_pattern(n):
    """Build the sparsity pattern for the convolution constraint matrix.
    Returns indptr and indices for a CSR matrix of shape (2n-1, n)."""
    rows = 2 * n - 1
    cols = n
    # Count non-zeros per row
    nnz_per_row = np.zeros(rows, dtype=np.int64)
    for k in range(rows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        nnz_per_row[k] = j_end - j_start + 1
    
    indptr = np.zeros(rows + 1, dtype=np.int64)
    indptr[1:] = np.cumsum(nnz_per_row)
    total_nnz = int(indptr[-1])
    
    indices = np.zeros(total_nnz, dtype=np.int32)
    pos = 0
    for k in range(rows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            indices[pos] = j
            pos += 1
    
    return indptr, indices, nnz_per_row

def fill_conv_matrix_data(a, indptr, indices, nnz_per_row):
    """Fill in the data values for the convolution matrix."""
    n = len(a)
    rows = 2 * n - 1
    total_nnz = int(indptr[-1])
    data = np.zeros(total_nnz, dtype=np.float64)
    pos = 0
    for k in range(rows):
        j_start = max(0, k - n + 1)
        for jj in range(nnz_per_row[k]):
            j = j_start + jj
            data[pos] = a[k - j]
            pos += 1
    return data
```

1. Find k* = argmax(c).
2. The constraint c[k*] <= M is active.
3. To reduce c[k*] = sum_i a[i]*a[k*-i], I need to reduce the products a[i]*a[k*-i].
4. One way: reduce a[i] and a[k*-i] simultaneously (for the pairs that contribute most).
5. To maintain the sum constraint, increase a[j] for j that don't contribute much to c[k*].

This is essentially a "pair swap" operation: move mass from indices that contribute to c[k*] to indices that don't.

Let me formalize:
- The contribution of a[j] to c[k*] is a[k*-j] (when valid).
- Indices with large a[k*-j] are "bad" (they contribute a lot to c[k*]).
- Indices with small a[k*-j] are "good" (they contribute little).
- Move mass from "bad" to "good" indices.

This is a form of gradient descent where the gradient is P_{k*} * a (the shifted a).

Let me implement this:

```python
def gradient_step(a, k_star, alpha, S_target):
    n = len(a)
    # Gradient of c[k_star] w.r.t. a[j] = 2 * a[k_star - j] (when valid)
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    for j in range(j_start, j_end + 1):
        grad[j] = 2 * a[k_star - j]
    # Gradient descent step
    a_new = a - alpha * grad
    # Project: non-negative
    a_new = np.maximum(a_new, 0)
    # Renormalize
    a_new = a_new / a_new.sum() * S_target
    return a_new
```

This is a simple projected gradient step. It should reduce c[k*] (at least for small alpha) without increasing other c[k] too much.
