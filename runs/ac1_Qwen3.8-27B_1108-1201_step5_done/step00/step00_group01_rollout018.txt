def compute_subgradient(a, k=5):
    """Compute subgradient of max(convolve(a,a)) using top-k maximizers."""
    n = len(a)
    conv = np.convolve(a, a)
    # Find top-k indices
    top_k = np.argsort(conv)[-k:]
    weights = np.zeros(len(conv))
    for idx in top_k:
        weights[idx] = 1.0 / k
    # Subgradient: for each j, grad_j = 2 * sum_k weights[k] * a[k-j] (if valid)
    # This is 2 * convolve(weights, a) but shifted... 
    # Actually, d(c_k)/d(a[j]) = 2*a[k-j] if 0 <= k-j < n
    # So grad[j] = 2 * sum_k weights[k] * a[k-j] (summed over valid k-j)
    # This is 2 * convolve(weights, a) at position... hmm
    # Let's just compute it directly
    grad = np.zeros(n)
    for k_idx in top_k:
        # d(c_k)/d(a[j]) = 2*a[k-j] for j in range(n) where 0 <= k-j < n
        # i.e., j in [k-(n-1), k] intersect [0, n-1]
        lo = max(0, k_idx - (n - 1))
        hi = min(n - 1, k_idx)
        if lo <= hi:
            grad[lo:hi+1] += (2.0 / k) * a[k_idx - hi : k_idx - lo + 1][::-1]
    return grad, conv

def project_nonneg_sum(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # Bisection for the offset t
    # sum(max(v - t, 0)) = target_sum
    lo = v.min() - 1.0  # definitely makes all positive
    hi = v.max() + 1.0  # definitely makes all zero
    
    for _ in range(50):  # 50 bisections gives ~1e-15 precision
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2.0
    return np.maximum(v - t, 0.0)

def soft_max_objective_and_grad(a, temperature):

    conv = np.convolve(a, a)
    max_c = conv.max()
