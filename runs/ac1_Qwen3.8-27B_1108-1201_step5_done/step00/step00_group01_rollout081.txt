```
def spectral_step(a, cap, N, current_b):
    L = len(a)*2 - 1
    b_target = np.minimum(current_b, cap)
    sum_b = float(np.sum(current_b)) # S^2 (approx)
    if b_target.sum() <= 0: return a
    b_target *= sum_b / b_target.sum()
    Bt = np.fft.rfft(b_target, N)
    A = np.fft.rfft(a, N)
    # Newton
    denom = A
    # safe division
    ratio = np.zeros_like(Bt)
    mask = np.abs(denom) > 1e-14
    ratio[mask] = Bt[mask] / denom[mask]
    A_new = 0.5 * (A + ratio)
    A_new[0] = A_new[0].real
    if N % 2 == 0: A_new[-1] = A_new[-1].real
    a_new = np.fft.irfft(A_new, N)[:len(a)]
    a_new = np.clip(a_new.real, 0.0, None)
    s = a_new.sum()
    if not np.isfinite(s) or s < 1e-12:
        return None
    a_new /= s
    return a_new
```

```
def make_wiggle_start(n, rng):
    x=np.arange(n,dtype=float)
    h=np.zeros(n)
    nterms=int(rng.integers(3,7))
    for _ in range(nterms):
       if rng.random()<0.5:
           f=float(rng.integers(1,25))
           ph=rng.uniform(0.0,2.0*np.pi)
           h += np.sin(2.0*np.pi*f*x/n + ph)
       else:
           f=int(rng.integers(1,13))

    mh=float(np.max(np.abs(h)))
    if mh <= 1e-14:

    alpha=float(rng.uniform(0.5,1.0))
```

```python
"""
Spectral-flattening plus active-constraint LP polish.

The evaluator is homogeneous, so we work with sum(a)=1.  The main loop uses
FFT-based Newton steps on A^2 = B_target, where B_target is the current
self-convolution with its largest coefficients clipped toward a target peak.
This is fast and forces the convolution to become flatter.  A secondary phase
uses small sparse LPs from scipy to polish the current best sequence.  Several
non-uniform anti-palindromic starts are used to avoid the uniform saddle point.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

try:
    from scipy.sparse import csr_matrix
except Exception:
    csr_matrix = None


def normalize_sum1(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return x
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    s = float(x.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return x / s


def next_fft_length(length):
    n = 1
    while n < length:
        n <<= 1
    return n


def convolve_full(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    L = 2 * n - 1
    if N is None:
        if L < 64:
            return np.convolve(a, a)
        N = next_fft_length(L)
    else:
        if N < L:
            N = next_fft_length(L)

    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[:L]
    if b.size and np.min(b) < 0.0:
        b[b < 0.0] = 0.0
    return b


def fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    b = convolve_full(a)
    if b.size == 0:
        return np.inf
    m = float(np.max(b))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def fast_score_N(a, N):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    b = convolve_full(a, N)
    if b.size == 0:
        return np.inf
    m = float(np.max(b))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def make_b_target(b, cap):
    b = np.asarray(b, dtype=np.float64).ravel()
    if b.size == 0:
        return b.copy()
    total = float(b.sum())
    if not np.isfinite(total) or total <= 0.0:
        return np.full(b.size, 1.0 / b.size, dtype=np.float64)
    if not np.isfinite(cap) or cap <= 0.0:
        cap = total / float(b.size)

    target = np.minimum(b, cap)
    st = float(target.sum())
    if st <= 1e-15:
        target = np.full(b.size, total / b.size, dtype=np.float64)
    else:
        target = target * (total / st)
    return target


def spectral_from_btarget(a, b_target, N, newton_iter=1):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return None
    if N is None:
        N = next_fft_length(2 * n - 1)
    if len(b_target) != 2 * n - 1:
        b_target = np.asarray(b_target, dtype=np.float64).ravel()
        if b_target.size < 2 * n - 1:
            b_target = np.concatenate((b_target, np.zeros(2 * n - 1 - b_target.size)))
        else:
            b_target = b_target[:2 * n - 1]

    Bt = np.fft.rfft(b_target, N)
    A = np.fft.rfft(a, N)

    for _ in range(max(1, int(newton_iter))):
        ratio = np.zeros_like(Bt, dtype=np.complex128)
        mask = np.abs(A) > 1e-14
        ratio[mask] = Bt[mask] / A[mask]
        A = 0.5 * (A + ratio)
        A[0] = A[0].real
        if N % 2 == 0 and A.size > 1:
            A[-1] = A[-1].real

    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.clip(np.real(a_new), 0.0, 1000.0)
    a_new = np.where(np.isfinite(a_new), a_new, 0.0)
    s = float(a_new.sum())
    if not np.isfinite(s) or s < 1e-12:
        return None
    return a_new / s


def pool_add(pool, a, max_len=4):
    if pool is None:
        return
    if a is None:
        return
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return
    pool.append(a.copy())
    while len(pool) > max_len:
        pool.pop(0)


def spectral_search(a, deadline, rng, pool=None, target_score=1.49):
    a = normalize_sum1(a)
    n = a.size
    if n == 0:
        return a, np.inf

    L = 2 * n - 1
    N = next_fft_length(L)
    b = convolve_full(a, N)
    m = float(np.max(b)) if b.size else 0.0
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return a, np.inf
    best_score = float(2.0 * n * m / (s * s))
    best = a.copy()
    cur = a.copy()
    cur_score = best_score

    pool_add(pool, best)

    cap_factor = 0.95
    low_factor = 0.55
    target_cap = max(1e-15, target_score / (2.0 * n))
    last_improve = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        b = convolve_full(cur, N)
        if b.size == 0:
            break
        m = float(np.max(b))
        s = float(cur.sum())
        if not np.isfinite(m) or not np.isfinite(s) or s < 0.01:
            break
        cur_score = float(2.0 * n * m / (s * s))

        if np.isfinite(cur_score) and cur_score < best_score - 1e-12:
            best = cur.copy()
            best_score = cur_score
            last_improve = time.monotonic()
            pool_add(pool, best)

        if time.monotonic() - last_improve > 25.0:
            mix = 0.05 + 0.05 * float(rng.random())
            cand = normalize_sum1((1.0 - mix) * cur + mix * rng.random(n))
            sc = fast_score_N(cand, N)
            if np.isfinite(sc) and sc < cur_score:
                cur = cand
                cur_score = sc
            if np.isfinite(sc) and sc < best_score:
                best = cand.copy()
                best_score = sc
                pool_add(pool, best)
            cap_factor = max(low_factor, cap_factor * 0.95)
            last_improve = time.monotonic()

        if it % 17 == 9:
            cap = max(target_cap * 0.9, low_factor * m)
            newton_iter = 2
        else:
            cap = max(target_cap * 0.95, cap_factor * m)
            newton_iter = 1

        b_target = make_b_target(b, cap)
        if rng.random() < 0.18:
            noise = 1.0 + 0.012 * rng.standard_normal(b_target.size)
            b_target = np.maximum(b_target * noise, 1e-15)
            st = float(b_target.sum())
            if st > 1e-15:
                b_target = b_target / st

        cand = spectral_from_btarget(cur, b_target, N, newton_iter=newton_iter)
        if cand is None:
            cand = normalize_sum1(0.75 * cur + 0.25 * rng.random(n))
        sc = fast_score_N(cand, N)

        accepted = False
        if np.isfinite(sc) and sc < cur_score - 1e-11:
            cur = cand
            cur_score = sc
            accepted = True
            last_improve = time.monotonic()
        elif np.isfinite(sc) and rng.random() < 0.04 and sc < cur_score + 0.04 * max(1.0, cur_score):
            cur = cand
            cur_score = sc
            accepted = True

        if np.isfinite(sc) and sc < best_score - 1e-11:
            best = cand.copy()
            best_score = sc
            last_improve = time.monotonic()
            pool_add(pool, best)

        if accepted:
            stall = 0
        else:
            stall += 1
            if stall > 8:
                cap_factor = max(low_factor, cap_factor * 0.94)
                cap = max(target_cap * 0.85, (low_factor + 0.05 * rng.random()) * m)
                b_target = make_b_target(b, cap)
                cand2 = spectral_from_btarget(cur, b_target, N, newton_iter=2)
                if cand2 is not None:
                    sc2 = fast_score_N(cand2, N)
                    if np.isfinite(sc2) and sc2 < best_score:
                        best = cand2.copy()
                        best_score = sc2
                        last_improve = time.monotonic()
                        pool_add(pool, best)
                    if np.isfinite(sc2) and sc2 < cur_score:
                        cur = cand2
                        cur_score = sc2
                stall = 0

        it += 1

    best = normalize_sum1(best)
    best_score = fast_score(best)
    pool_add(pool, best)
    return best, best_score


def anti_binary_start(n, rng):
    h = np.zeros(n, dtype=np.float64)
    m = n // 2
    if m > 0:
        first = rng.choice(np.array([-1.0, 1.0], dtype=np.float64), size=m)
        h[:m] = first
        h[n - m:] = -first[::-1]
    if n % 2 == 1:
        h[m] = 0.0
    a = (1.0 + h) / float(n)
    return normalize_sum1(a)


def make_wiggle_start(n, rng):
    x = np.arange(n, dtype=np.float64)
    h = np.zeros(n, dtype=np.float64)
    terms = int(rng.integers(3, 7))
    for _ in range(terms):
        if rng.random() < 0.5:
            f = float(rng.integers(1, 25))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            h += np.sin(2.0 * np.pi * f * x / float(n) + ph)
        else:
            f = int(rng.integers(1, 13))
            f = 2 * f - 1
            ph = float(rng.uniform(0.0, np.pi))
            h += np.sin(np.pi * f * (x + 0.5) / float(n) + ph)

    h = 0.5 * (h - h[::-1])
    h -= h.mean()
    mh = float(np.max(np.abs(h)))
    if mh <= 1e-14:
        return np.ones(n, dtype=np.float64) / float(n)

    h /= mh
    alpha = float(rng.uniform(0.5, 1.0))
    a = (1.0 + alpha * h) / float(n)
    return normalize_sum1(a)


def make_anti_sine_start(n, rng):
    x = np.arange(n, dtype=np.float64)
    h = np.zeros(n, dtype=np.float64)
    terms = int(rng.integers(1, 4))
    for _ in range(terms):
        f = int(rng.integers(1, 13))
        f = 2 * f - 1
        ph = float(rng.uniform(0.0, np.pi))
        amp = float(rng.uniform(0.5, 1.0))
        h += amp * np.sin(np.pi * f * (x + 0.5) / float(n) + ph)

    h = 0.5 * (h - h[::-1])
    h -= h.mean()
    mh = float(np.max(np.abs(h)))
    if mh <= 1e-14:
        return np.ones(n, dtype=np.float64) / float(n)

    h /= mh
    alpha = float(rng.uniform(0.7, 1.0))
    a = (1.0 + alpha * h) / float(n)
    return normalize_sum1(a)


def tile_sequence(base, L):
    base = np.asarray(base, dtype=np.float64).ravel()
    if L <= 0:
        return np.empty(0, dtype=np.float64)
    m = base.size
    if m == 0 or float(base.sum()) < 0.01:
        return np.ones(L, dtype=np.float64) / float(L)
    reps = (L + m - 1) // m
    a = np.tile(base, reps)[:L]
    return normalize_sum1(a)


def choose_lp_K(n, max_nonzeros):
    if n <= 0:
        return 0
    limit = max(1, int(max_nonzeros) // max(1, n))
    K = int(min(2 * n - 1, limit))
    if K < 1:
        K = 1
    return K


def select_active_indices(b, K, rng=None, random_fraction=0.0):
    L = b.size
    if L <= 0 or K <= 0:
        return np.empty(0, dtype=np.int64)
    if K >= L:
        return np.arange(L, dtype=np.int64)
    top = np.argpartition(b, L - K)[-K:]
    if rng is not None and random_fraction > 0.0:
        nr = int(K * random_fraction)
        if nr > 0:
            rand = rng.integers(0, L, size=nr).astype(np.int64)
            top = np.concatenate((top, rand))
    return top.astype(np.int64)


def build_cross_csr(a, ks, M, sum_bound, tol=1e-12):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    K = len(ks)
    indptr = np.empty(K + 2, dtype=np.int32)
    indptr[0] = 0
    dense = bool(np.all(a > tol))

    if dense:
        for i, k in enumerate(ks):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            cnt = (hi - lo + 1) if hi >= lo else 0
            indptr[i + 1] = indptr[i] + cnt
    else:
        for i, k in enumerate(ks):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            cnt = 0
            if hi >= lo:
                js = np.arange(lo, hi + 1, dtype=np.int32)
                vals = a[k - js]
                cnt = int(np.count_nonzero(vals > tol))
            indptr[i + 1] = indptr[i] + cnt

    indptr[K + 1] = indptr[K] + n
    total = int(indptr[K + 1])
    data = np.empty(total, dtype=np.float64)
    cols = np.empty(total, dtype=np.int32)
    pos = 0

    if dense:
        for i, k in enumerate(ks):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if hi >= lo:
                cnt = hi - lo + 1
                js = np.arange(lo, hi + 1, dtype=np.int32)
                cols[pos:pos + cnt] = js
                data[pos:pos + cnt] = a[k - js]
                pos += cnt
    else:
        for i, k in enumerate(ks):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if hi >= lo:
                js = np.arange(lo, hi + 1, dtype=np.int32)
                vals = a[k - js]
                mask = vals > tol
                if np.any(mask):
                    idx = np.nonzero(mask)[0]
                    cnt = idx.size
                    data[pos:pos + cnt] = vals[idx]
                    cols[pos:pos + cnt] = js[idx]
                    pos += cnt

    data[pos:pos + n] = 1.0
    cols[pos:pos + n] = np.arange(n, dtype=np.int32)

    A = csr_matrix((data, cols, indptr), shape=(K + 1, n))
    b = np.empty(K + 1, dtype=np.float64)
    b[:K] = M
    b[K] = sum_bound
    return A, b


def random_perturb(a, rng, strength=0.05):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a
    noise = rng.standard_normal(n)
    if n > 10:
        kernel = np.ones(3, dtype=np.float64) / 3.0
        noise = np.convolve(noise, kernel, mode="same")
    cand = a + strength * noise
    return normalize_sum1(cand)


def solve_lp_direction(a, ks, M, rng, tie_eps=1e-6, sum_bound=3.0, time_limit=20.0):
    if linprog is None or csr_matrix is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0 or ks is None or len(ks) == 0:
        return None

    try:
        A, b = build_cross_csr(a, ks, M, sum_bound)
    except Exception:
        return None

    c = -np.ones(n, dtype=np.float64)
    if tie_eps > 0.0:
        tie = rng.uniform(-1.0, 1.0, size=n)
        c = c - tie_eps * tie

    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(time_limit)}
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method="highs",
                options={"presolve": True}
            )
        except Exception:
            return None

    if res is None or getattr(res, "x", None) is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    if g.size != n:
        return None
    g = np.clip(g, 0.0, None)
    g = np.where(np.isfinite(g), g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return g


def lp_search(a, deadline, rng, pool=None, max_nonzeros=4_000_000, max_nonzeros_large=12_000_000):
    if linprog is None or csr_matrix is None:
        a = normalize_sum1(a)
        return a, fast_score(a)

    a = normalize_sum1(a)
    if a.size == 0:
        return a, np.inf

    best = a.copy()
    best_score = fast_score(best)
    pool_add(pool, best)
    n = best.size

    base_K = choose_lp_K(n, max_nonzeros)
    max_K = max(base_K, choose_lp_K(n, max_nonzeros_large))
    K = base_K
    t_list = (1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.06, 0.03)
    stall = 0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining <= 8.0:
            break
        lp_time = min(30.0, max(6.0, remaining * 0.6))
        if time.monotonic() + lp_time > deadline:
            lp_time = max(4.0, remaining - 1.0)
            if lp_time <= 0:
                break

        b = convolve_full(best)
        if b.size == 0:
            break
        M = float(np.max(b))
        if not np.isfinite(M):
            break

        found = False
        for _ in range(2):
            if time.monotonic() + lp_time > deadline:
                break
            ks = select_active_indices(b, K, rng, 0.1)
            g = solve_lp_direction(best, ks, M, rng, 1e-6, 3.0, lp_time)
            if g is None:
                continue

            s_best = float(best.sum())
            s_g = float(g.sum())
            if s_g <= 1e-12:
                continue
            g = (g / s_g) * s_best

            local_best = best
            local_score = best_score
            for t in t_list:
                if t >= 1.0:
                    new = g.copy()
                else:
                    new = (1.0 - t) * best + t * g
                new = np.clip(new, 0.0, 1000.0)
                snew = float(new.sum())
                if not np.isfinite(snew) or snew < 0.01:
                    continue
                new = new / snew
                sc = fast_score(new)
                if np.isfinite(sc) and sc < local_score:
                    local_score = sc
                    local_best = new

            if np.isfinite(local_score) and local_score < best_score - 1e-10:
                best = local_best
                best_score = local_score
                found = True
                pool_add(pool, best)
                print("LP improvement: %.9f" % best_score)
                break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 5:
                cand = random_perturb(best, rng, 0.05)
                sc = fast_score(cand)
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc
                    pool_add(pool, best)
                elif np.isfinite(sc) and rng.random() < 0.25:
                    best = cand
                    best_score = sc
                if stall > 10:
                    K = min(max_K, max(1, int(K * 1.4) + 1))
                stall = 0

    best = normalize_sum1(best)
    final_score = fast_score(best)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    pool_add(pool, best)
    return best, best_score


def optimize_small_track(parent_len, deadline, rng):
    if parent_len < 256:
        m = 64
    else:
        m = 256

    track_deadline = min(deadline, time.monotonic() + 120.0)
    best = None
    best_score = np.inf

    starts = [anti_binary_start(m, rng), make_wiggle_start(m, rng)]
    spectral_deadline = track_deadline - 30.0
    for s in starts:
        if time.monotonic() >= spectral_deadline:
            break
        sec = min(45.0, spectral_deadline - time.monotonic())
        if sec < 10.0:
            break
        b, sc = spectral_search(s, time.monotonic() + sec, rng, None, 1.49)
        if np.isfinite(sc) and sc < best_score:
            best = b
            best_score = sc

    if best is not None and time.monotonic() < track_deadline:
        b2, sc2 = lp_search(best, track_deadline, rng, None)
        if np.isfinite(sc2) and sc2 < best_score:
            best = b2
            best_score = sc2

    return best, best_score


def select_final_candidate(pool, default):
    cands = []
    if default is not None:
        cands.append(default)
    if pool:
        cands.extend(pool)

    unique = []
    for a in cands:
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            continue
        duplicate = False
        for u in unique:
            if u.size == a.size:
                duplicate = True
                break
        if not duplicate:
            unique.append(a)

    if not unique:
        default = np.ones(64, dtype=np.float64) / 64.0
        return normalize_sum1(default)

    scored = []
    for idx, a in enumerate(unique):
        fs = fast_score(a)
        if not np.isfinite(fs):
            fs = 1e100
        scored.append((fs, idx, a))
    scored.sort()

    max_n = max(int(a.size) for a in unique)
    limit = 5 if max_n <= 8000 else 2

    best_a = default if default is not None else unique[0]
    best_exact = np.inf

    for fs, idx, a in scored[:limit]:
        try:
            sc = float(evaluate_sequence([float(v) for v in a]))
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_a = a

    if best_a is None or not np.all(np.isfinite(best_a)):
        best_a = unique[0]

    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(best_a.sum())
    if not np.isfinite(s) or s < 0.01:
        best_a = np.ones(64, dtype=np.float64) / 64.0
    else:
        best_a = best_a / s
    return best_a


def propose_candidate():
    start_time = time.monotonic()
    deadline = start_time + 1790.0
    rng = np.random.default_rng(20240520)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size == 0 or parent.sum() < 0.01:
            parent = np.ones(6121, dtype=np.float64)
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    L = int(parent.size)
    if L <= 32:
        L = 6121
        parent = np.ones(L, dtype=np.float64)
    elif L > 20000:
        parent = parent[:20000]
        L = 20000

    parent = normalize_sum1(parent)
    global_best = parent.copy()
    global_score = fast_score(global_best)
    pool = []
    pool_add(pool, global_best)
    print("initial fast score: %.9f len=%d" % (global_score, L))

    starts_extra = []
    if L > 300:
        small_deadline = min(start_time + 150.0, deadline - 700.0)
        if time.monotonic() < small_deadline:
            small_best, small_score = optimize_small_track(L, small_deadline, rng)
            if small_best is not None and np.isfinite(small_score):
                if small_score < global_score:
                    global_best = small_best
                    global_score = small_score
                    pool_add(pool, global_best)
                else:
                    starts_extra.append(tile_sequence(small_best, L))

    starts = []
    starts.append(global_best.copy())
    starts.append(normalize_sum1(global_best + 0.05 * rng.random(L)))
    starts.append(anti_binary_start(L, rng))
    starts.append(make_wiggle_start(L, rng))
    starts.append(make_wiggle_start(L, rng))
    starts.append(make_anti_sine_start(L, rng))
    starts.extend(starts_extra)
    starts = starts[:8]

    if linprog is not None and L <= 20000:
        lp_reserve = 400.0
    else:
        lp_reserve = 80.0
    final_reserve = 30.0

    i = 0
    while i < len(starts) and time.monotonic() < deadline - lp_reserve - final_reserve:
        remaining = deadline - time.monotonic() - lp_reserve - final_reserve
        if remaining <= 0:
            break
        sec = min(300.0, max(60.0, remaining / max(1, len(starts) - i)))
        if sec < 20.0:
            break
        d_i = time.monotonic() + sec
        a = starts[i]
        b, sc = spectral_search(a, d_i, rng, pool, 1.49)
        if np.isfinite(sc) and sc < global_score:
            global_best = b
            global_score = sc
            pool_add(pool, global_best)
        i += 1

    polish_deadline = deadline - lp_reserve - final_reserve
    if time.monotonic() < polish_deadline - 30.0:
        b, sc = spectral_search(global_best, polish_deadline, rng, pool, 1.49)
        if np.isfinite(sc) and sc < global_score:
            global_best = b
            global_score = sc
            pool_add(pool, global_best)

    if linprog is not None and L <= 20000:
        lp_deadline = deadline - final_reserve
        if time.monotonic() < lp_deadline - 20.0:
            b, sc = lp_search(global_best, lp_deadline, rng, pool)
            if np.isfinite(sc) and sc < global_score:
                global_best = b
                global_score = sc
                pool_add(pool, global_best)

    forced_full = tile_sequence(global_best, L) if global_best.size != L else global_best.copy()
    pool_add(pool, forced_full)

    final = select_final_candidate(pool, global_best)
    return [float(x) for x in final]
```
