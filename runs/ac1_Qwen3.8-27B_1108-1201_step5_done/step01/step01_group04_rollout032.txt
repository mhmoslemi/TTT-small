import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = 1.0
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def softmax_grad_step(a, lr, tau, beta1=0.9, beta2=0.999, m=None, v=None, t=1):
    """Adam step with softmax gradient."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    # Gradient: grad[j] = 2 * sum_k a[k] * w[j+k] (for valid k)
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(0, j)  # j+k >= 0 always since j,k >= 0
        k_hi = min(n-1, 2*n-2-j)
        if k_hi >= k_lo:
            grad[j] = 2.0 * np.dot(a[k_lo:k_hi+1], w[j+k_lo:j+k_hi+1])
    
    # Project gradient: remove mean component (for sum constraint)
    grad -= grad.mean()
    
    # Adam update
    if m is None:
        m = np.zeros(n)
        v = np.zeros(n)
    
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, 1.0)
    
    return a_new, m, v, t + 1

def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 1024:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        lo = max(0, m-n+1)
        hi = min(n-1, m)
        for j in range(lo, hi+1):
            A[m, j] = f[m-j]
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except:
        pass
    return None

def optimize_adam_lp(a0, deadline, n_grad_iters=5000, n_lp_iters=200):
    """Optimize using Adam + LP refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a, 1.0)
    
    best = a.copy()
    best_score = compute_obj(a)
    
    # Phase 1: Adam
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 1
    tau = 50.0
    lr = 0.05
    
    for it in range(n_grad_iters):
        if time.monotonic() > deadline - 60:
            break
        # Anneal tau and lr
        progress = it / n_grad_iters
        tau = 30 + 40 * progress
        lr = 0.05 * (1 + np.cos(np.pi * progress)) * 0.5 + 0.005
        
        a, m, v, t_adam = softmax_grad_step(a, lr, tau, m=m, v=v, t=t_adam)
        
        if (it+1) % 200 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Phase 2: LP refinement
    if n <= 1024:
        c = np.convolve(a, a)
        M = c.max()
        
        for it in range(n_lp_iters):
            if time.monotonic() > deadline - 10:
                break
            g = lp_solve(a, M)
            if g is None:
                break
            improved = False
            for t_blend in [0.9, 0.7, 0.5, 0.3, 0.1]:
                a_new = (1-t_blend)*a + t_blend*g
                a_new = np.maximum(a_new, 0)
                a_new = normalize(a_new, 1.0)
                s_new = compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break
            if not improved:
                # Perturb
                a = a + 0.01 * np.random.randn(n)
                a = np.maximum(a, 0)
                a = normalize(a, 1.0)
            c = np.convolve(a, a)
            M = c.max()
    
    return best, best_score

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses SLSQP with smooth max (log-sum-exp) objective, LP-based refinement,
projected gradient with sharp softmax, and multi-start exploration.
Combines tau-annealing SLSQP, alternating LP blending, and fine-grained
projected gradient for progressive refinement.

import numpy as np
import time
from scipy.optimize import linprog, minimize


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def smooth_max_val_grad(a, tau):
    """Compute smooth max approximation and gradient."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = float(c.max())
    exp_c = np.exp((c - c_max) / tau)
    sum_exp = float(exp_c.sum())
    w = exp_c / sum_exp
    grad = 2.0 * np.correlate(w, a, mode='valid')
    val = c_max + tau * np.log(sum_exp)
    return val, grad


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve(f, M, n_max=1024):
    """LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    if n > n_max:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        A[m, lo:hi + 1] = f[m - hi:m - lo + 1][::-1]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s

    i = np.arange(n, dtype=np.float64)
    # Cosine-modulated
    start_strategies.append(("cos3", 0.3 + 0.7 * np.cos(2 * np.pi * i * 3.0 / n) ** 2))
    start_strategies.append(("cos1", 0.3 + 0.7 * np.cos(2 * np.pi * i / n) ** 2))
    # Edge-enhanced
    start_strategies.append(("edge", np.abs(np.cos(np.pi * i / n)) ** 0.5 + 0.1))
    # Two-bump
    start_strategies.append(("twpump", np.exp(-0.5 * ((i - n * 0.2) / (n * 0.15)) ** 2) +
                              np.exp(-0.5 * ((i - n * 0.8) / (n * 0.15)) ** 2) + 0.05))
    # Random
    start_strategies.append(("rand1", rng.random(n) * 0.5 + 0.2))
    start_strategies.append(("rand2", rng.random(n) * 0.3 + 0.1))
    # Geometric decay + mirror
    start_strategies.append(("geom", np.concatenate([np.exp(-i[:n//2] / (n*0.1)),
                                                     np.exp(-(n-1-i[n//2:]) / (n*0.1))]) + 0.05))

    # For n=1024 (longer sequence, lower theoretical bound)
    n2 = 1024
    i2 = np.arange(n2, dtype=np.float64)
    start_strategies.append(("cos3_1024", 0.3 + 0.7 * np.cos(2 * np.pi * i2 * 3.0 / n2) ** 2))
    start_strategies.append(("rand_1024", rng.random(n2) * 0.4 + 0.15))

    # Try starting from the global first (most likely to be near-optimal)
    # Then try others
    print(f"Total budget: {budget_s}s, strategies: {len(start_strategies)}")
    print(f"Starting optimization...")

    # Prioritize: global starts first, then structured, then random
    # Sort by priority: global > structured > random
    def priority(s):
        name = s[0]
        if "global" in name:
            return 0
        if "rand" in name:
            return 2
        return 1

    start_strategies.sort(key=lambda x: (priority(x), len(x[1])))

    for label, a0 in start_strategies:
        if time.monotonic() > deadline - 60.0:
            break
        try_sequence(a0, label)

    # --- Final refinement of the best sequence ---
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        print(f"\n=== Final refinement (current best: {best_score:.6f}) ===")
        for pass_i in range(8):
            if time.monotonic() > deadline - 15.0:
                break
            print(f"  Refinement pass {pass_i+1}...", end=" ", flush=True)
            t0 = time.monotonic()
            seq, score = optimize_combined(best_seq, deadline - 10.0)
            elapsed = time.monotonic() - t0
            print(f"score={score:.6f} [{elapsed:.1f}s]")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  *** IMPROVED: {best_score:.6f} ***")
            else:
                # Try a perturbation before next pass
                best_seq = best_seq + 0.01 * np.random.randn(len(best_seq))
                best_seq = normalize(best_seq)

    # --- Final evaluation ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
            if eval_score < final_score:
                final_score = eval_score
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
