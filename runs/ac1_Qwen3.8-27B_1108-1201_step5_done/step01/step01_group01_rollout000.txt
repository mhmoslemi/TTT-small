def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1

    while fft_size < n:
        fft_size *= 2

    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:n]

def _autocorr(a):
    """Compute a * a (linear convolution) via FFT."""
    return _fft_convolve(a, a)

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:

    rho = rho_candidates[-1]

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except (NameError, UnboundLocalError):
        parent = None

    if parent is not None and parent.size > 0:
        parent = np.clip(parent, 0.0, None)
        n = parent.size
    else:
        n = 6121
        parent = np.ones(n) / n

    print(f"Starting optimization with n={n}")

    # Prepare starting points
    starts = []

    # Parent
    a_parent = parent.copy()
    s = np.sum(a_parent)
    if s > 1e-10:
        a_parent /= s
    else:
        a_parent = np.ones(n) / n
    starts.append(("parent", a_parent))

    # Diverse starts
    starts.append(("antisym_0.3", _make_antisymmetric(n, 0.3, rng)))
    starts.append(("antisym_0.5", _make_antisymmetric(n, 0.5, rng)))
    starts.append(("antisym_0.7", _make_antisymmetric(n, 0.7, rng)))
    starts.append(("tapered_0.5", _make_tapered(n, 0.5, rng)))
    starts.append(("tapered_0.8", _make_tapered(n, 0.8, rng)))
    starts.append(("bipolar", _make_bipolar_smooth(n, rng)))

    # Uniform
    starts.append(("uniform", np.ones(n) / n))

    # Random smooth
    a_rand = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    a_rand = np.maximum(a_rand, 0.0)
    a_rand /= np.sum(a_rand)
    starts.append(("smooth_rand", a_rand))

    global_best = a_parent.copy()
    global_best_score = _score_normalized(a_parent)
    print(f"Initial parent score: {global_best_score:.6f}")

    # Phase 1: Optimize from each start (time varies by start quality)
    n_starts = len(starts)
    # Parent gets most time, others get less
    phase1_deadline = min(t_start + budget_s * 0.5, deadline)
    time_per_start = (phase1_deadline - time.monotonic()) / max(1, n_starts - 1)
    parent_time = 0.4 * (phase1_deadline - time.monotonic())

    for i, (name, a_start) in enumerate(starts):
        if time.monotonic() >= phase1_deadline:
            break

        if i == 0:
            t_alloc = min(parent_time, phase1_deadline - time.monotonic())
        else:
            t_alloc = min(time_per_start * 1.2, phase1_deadline - time.monotonic())

        sub_deadline = time.monotonic() + t_alloc
        best_a, best_s = _adam_smooth_descend(
            a_start, sub_deadline, rng,
            beta_start=5.0, beta_end=100.0,
            lr_start=0.03, lr_end=0.001,
            n_peaks=30, reheat_every=200,
            name=f"P1-{name}"
        )

        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  Phase1 {name}: NEW BEST {global_best_score:.6f}")
        else:
            print(f"  Phase1 {name}: {best_s:.6f} (best={global_best_score:.6f})")

    # Phase 2: Refine the best with higher beta
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 2: Refining with {remaining:.0f}s, score={global_best_score:.6f}")
        phase2_deadline = min(t_start + budget_s * 0.75, deadline)

        best_a, best_s = _adam_smooth_descend(
            global_best, phase2_deadline, rng,
            beta_start=50.0, beta_end=500.0,
            lr_start=0.01, lr_end=0.0003,
            n_peaks=60, reheat_every=400,
            name="P2-refine"
        )
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  Phase2: NEW BEST {global_best_score:.6f}")

        # Also try from a slightly perturbed version
        if time.monotonic() < deadline - 30:
            a_pert = global_best + 0.02 * rng.standard_normal(n)
            a_pert = _project_simplex(a_pert, 1.0)
            sub_dl = min(time.monotonic() + 60, deadline)
            best_a2, best_s2 = _adam_smooth_descend(
                a_pert, sub_dl, rng,
                beta_start=100.0, beta_end=500.0,
                lr_start=0.005, lr_end=0.0001,
                n_peaks=80, reheat_every=200,
                name="P2-perturb"
            )
            if best_s2 < global_best_score:
                global_best_score = best_s2
                global_best = best_a2.copy()
                print(f"  Phase2 perturbed: NEW BEST {global_best_score:.6f}")

    # Phase 3: Random perturbation polishing
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\nPhase 3: Perturbation polish with {remaining:.0f}s")
        best_a, best_s = _random_perturb_polish(
            global_best, deadline, rng, n_trials=10000, scale=0.02
        )
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  Phase3: NEW BEST {global_best_score:.6f}")

    # Phase 4: Pairwise transfer polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\nPhase 4: Pairwise polish with {remaining:.0f}s")
        best_a, best_s = _pairwise_transfer_polish(
            global_best, min(time.monotonic() + remaining * 0.5, deadline),
            rng, n_trials=5000, eps=0.005
        )
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  Phase4: NEW BEST {global_best_score:.6f}")

    # Final: one more quick Adam pass
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\nFinal pass with {remaining:.0f}s")
        best_a, best_s = _adam_smooth_descend(
            global_best, min(time.monotonic() + remaining * 0.8, deadline),
            rng, beta_start=200.0, beta_end=1000.0,
            lr_start=0.003, lr_end=0.0001,
            n_peaks=100, reheat_every=500,
            name="final"
        )
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  Final: NEW BEST {global_best_score:.6f}")

    # Prepare output
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.6f}")
    print(f"Time used: {time.monotonic() - t_start:.1f}s")
    return result
