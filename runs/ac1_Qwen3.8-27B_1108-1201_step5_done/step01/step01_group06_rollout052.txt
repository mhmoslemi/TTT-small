            if old_ai <= 0:

                pass

            # d[k] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            k_lo = max(0, i - n + 1)
            k_hi = min(2*n - 2, i + n - 1)
            
            # Get the affected k values
            ks = np.arange(k_lo, k_hi + 1)
            
            # d[k] computation
            ki = ks - i  # k - i
            ik = i - ks  # i - k
            
            d = np.zeros(len(ks))
            # Where 0 <= ki < n: d += a[ki]
            valid_ki = (ki >= 0) & (ki < n)
            d[valid_ki] += a[ki[valid_ki]]
            # Where 0 <= ik < n: d += a[ik]
            valid_ik = (ik >= 0) & (ik < n)
            d[valid_ik] += a[ik[valid_ik]]

            base = c[ks] - old_ai * d
            
            # Binary search for optimal t >= 0
            # f(t) = max(base + t*d) is convex in t
            # f'(t) = d[argmax(base + t*d)] (when unique)
            
            # Check t=0
            k_star = np.argmax(base)
            if d[k_star] >= -1e-15:
                # Optimum at t=0
                new_ai = 0.0
            else:
                # Binary search
                lo_t = 0.0
                hi_t = old_ai * 2.0 + max(a.max() * 0.1, 0.01)
                
                for _ in range(50):

                    k_star = np.argmax(base + mid_t * d)
                    if d[k_star] > 0:
                        hi_t = mid_t
                    else:
                        lo_t = mid_t
                
                new_ai = (lo_t + hi_t) * 0.5
                
                # Verify it's better than t=0
                if np.max(base + new_ai * d) >= np.max(base) + 1e-12:
                    new_ai = 0.0

                # Update a and c
                delta = new_ai - old_ai
                a[i] = new_ai
                
                # Update c: c[k] += delta * d[k] for affected k's
                c[ks] += delta * d
        
        # Recompute c periodically to avoid numerical drift
        if cycle % 20 == 0:
            c = fft_convolve_self(a)
        
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        
        if cycle % 100 == 0:
            obj = 2 * n * cur_max / S**2
            print(f"Cycle {cycle}: max_conv={cur_max:.6f}, obj={obj:.6f}, time={time.monotonic()-start_time:.1f}s")
    
    # Verify with evaluate_sequence
    result = best_a.tolist()
    ver = evaluate_sequence(result)

        # Try clipping
        result = [max(0.0, min(1000.0, x)) for x in result]
        ver = evaluate_sequence(result)
    
    return result
