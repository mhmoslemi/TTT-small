"""
Hybrid FFT-subgradient + LP optimization for minimizing peak auto-convolution
of non-negative sequences. Uses fast gradient descent for bulk optimization
and LP polishing for final refinement.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def score_fn(a):
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * M / (s * s)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]


def compute_subgradient(w, a):
    n = len(a)
    m = len(w)
    N = 1 << (m + n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(W * np.conj(A), N)
    return result[:n]


def gradient_optimize(a, time_budget=60.0, verbose=True, seed=0):
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a, a).max()
    
    temp = 30.0
    lr = 0.05
    stall_count = 0
    max_stall = 3000
    stagnation = 0
    
    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        
        c = fft_convolve(a, a)
        M = c.max()
        score_val = 2.0 * n * M
        
        if score_val < best_score - 1e-13:
            best_score = score_val
            best_a = a.copy()
            stall_count = 0
            stagnation = 0
        else:
            stall_count += 1
            if score_val < best_score + 1e-10:
                stagnation += 1
        
        if verbose and (it % 2000 == 0):
            print(f"        iter {it:6d}: score={score_val:.6f}, best={best_score:.6f}")
        
        if stall_count > max_stall:
            if stagnation > max_stall // 2:
                break
            # Restart with perturbation
            a = best_a.copy()
            idx = rng.choice(n, size=max(1, n // 5), replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.3, size=len(idx)))
            a = normalize_nonneg(a)
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            lr = min(0.1, lr * 1.5)
            continue

        w = np.exp(temp * c_shift)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[c.argmax()] = 1.0
        else:
            w /= w_sum

        g = compute_subgradient(w, a)

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)

def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n, dtype=np.int32))
    row_idx = np.concatenate(row_parts)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

            best_a = a.copy()
            stagnant = 0
            if verbose:
                print(f"        LP iter {it:3d}: score={score:.6f}")
        else:

            break
        
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        
        timeout = max(2.0, min(30.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.5, max(0.1, 1.0 / (1.0 + 0.1 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = 2.0 * n * np.convolve(a_new, a_new).max()
        if new_score <= score + 1e-14:
            a = a_new
        else:
            a = (1.0 - 0.1) * a + 0.1 * g
            a = normalize_nonneg(a)
    
    final_score = 2.0 * n * np.convolve(a, a).max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"        LP done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
