```python
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    """Compute 2*n*max(conv)/sum^2 efficiently."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def mirror_descent_step(a, lr):
    """One mirror descent (exponential weights) step."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    k_star = np.argmax(conv)
    # Gradient of conv[k_star] w.r.t. a
    # grad[m] = 2*a[k_star-m] if 0<=k_star-m<n
    m_idx = np.arange(n)
    k_minus_m = k_star - m_idx
    valid = (k_minus_m >= 0) & (k_minus_m < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[k_minus_m[valid]]
    # Normalize gradient (scale-invariant)
    if grad.max() < 1e-15:
        return a
    grad = grad / M  # normalize by the peak value
    # Exponential update
    a_new = a * np.exp(-lr * grad)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def run_mirror_descent(a, n_steps, lr_init, lr_end, seed=0):
    """Run mirror descent for n_steps with linearly decaying learning rate."""
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_sc = fast_score(a)
    a = a.copy()
    for i in range(n_steps):
        lr = lr_init * (lr_end / lr_init) ** (i / max(1, n_steps - 1))
        a = mirror_descent_step(a, lr)
        if i % 100 == 0:
            sc = fast_score(a)
            if sc < best_sc:
                best_sc = sc
                best = a.copy()
    # Final check
    sc = fast_score(a)
    if sc < best_sc:
        best = a.copy()
        best_sc = sc
    return best, best_sc
```

                best = a.copy()
                no_improve = 0

    if np.isfinite(sc) and sc < best_sc:

        best_sc = sc
