import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time
import math

def fast_score(f):
    """Compute the score quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def build_conv_matrix(n, f):
    """Build sparse matrix A where (A @ g)[k] = (f*g)[k] (linear conv)."""
    # A has shape (2n-1, n), A[k, j] = f[k-j] if 0 <= k-j < n else 0
    rows, cols, data = [], [], []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A

def lp_step(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except:

def peak_directed_move(f, rng, peak_margin=0.1):
    """
    Generate a move that targets the current peak lag.
    Returns a modified copy of f (same sum).
    """
    n = len(f)
    conv = np.convolve(f, f)
    peak_val = conv.max()
    peak_lags = np.where(conv >= peak_val * (1 - peak_margin))[0]
    k_star = peak_lags[rng.integers(len(peak_lags))]
    
    # Find heavy contributors to lag k_star
    # conv[k_star] = sum_{i} f[i]*f[k_star - i] (valid indices)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    contributors = []
    for i in range(lo, hi + 1):
        j = k_star - i
        if i != j:
            contributors.append((i, j, f[i] * f[j] * 2))
        else:
            contributors.append((i, j, f[i] * f[j]))
    
    if not contributors:
        return f.copy()
    
    # Sort by contribution (descending)
    contributors.sort(key=lambda x: -x[2])
    
    # Pick a heavy contributor (i, j) and reduce the larger one
    idx = rng.integers(min(len(contributors), max(1, len(contributors)//4)))
    i, j, _ = contributors[idx]
    
    # Decide which to reduce
    if f[i] >= f[j]:
        reduce_idx, reference_idx = i, j
    else:
        reduce_idx, reference_idx = j, i
    
    # How much to reduce
    # We want to reduce f[reduce_idx] but not to zero
    max_reduce = f[reduce_idx] * 0.5
    reduce_amt = rng.uniform(0.1, 1.0) * max_reduce
    
    f_new = f.copy()
    f_new[reduce_idx] -= reduce_amt
    
    # Distribute the mass to a "safe" position
    # Safe = a position where f[p] is small and it won't increase the peak
    # Find positions where f is small
    safe_positions = np.where(f < np.median(f))[0]
    if len(safe_positions) == 0:
        safe_positions = np.where(f < np.percentile(f, 75))[0]
    p = safe_positions[rng.integers(len(safe_positions))]
    
    # But check that this doesn't create a new peak
    # For now, just add it and we'll check the score later
    f_new[p] += reduce_amt
    
    # Ensure non-negativity
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s * f.sum()  # restore original sum
    
    return f_new

def block_perturb(f, rng, block_size=None):
    """Random multiplicative block perturbation."""
    n = len(f)
    if block_size is None:
        block_size = int(rng.integers(1, max(2, n // 8)))
    start = int(rng.integers(0, max(1, n - block_size + 1)))
    stop = min(n, start + block_size)
    f_new = f.copy()
    f_new[start:stop] *= float(np.exp(rng.normal(0, 0.3)))
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new * (f.sum() / s)
    return f_new

def make_initial(n, rng, method='uniform'):
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = np.maximum(a, 0.01)
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'two_bumps':
        x = np.arange(n, dtype=float)
        a = np.exp(-((x - n*0.25)/max(1,n*0.1))**2/2) + np.exp(-((x - n*0.75)/max(1,n*0.1))**2/2) + 0.01
    else:
        a = np.ones(n)
    a = a / a.sum()
    return a

    """
    Simulated annealing with peak-directed moves, block perturbations, and LP boosts.
    """

    f = f0.copy()

    best = f.copy()

    T = peak_val_estimate = best_sc - 2.0 + 0.1  # initial temperature
    if T < 0.01:
        T = 0.05
    
    start_time = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - start_time < budget_s:
        it += 1

        r = rng.random()
        if r < peak_prob:
            f_new = peak_directed_move(f, rng)
        elif r < peak_prob + lp_prob:

            M = conv.max()
            g = lp_step(f, M, time_limit=3.0)
            if g is not None:

                f_new = (1 - t_mix) * f + t_mix * g
                f_new = np.maximum(f_new, 0)
                s = f_new.sum()
                if s > 1e-10:
                    f_new = f_new / s
                else:
                    f_new = f
            else:
                f_new = block_perturb(f, rng)
        else:
            f_new = block_perturb(f, rng)
        
        sc_new = fast_score(f_new)
        sc_cur = fast_score(f)
        
        if sc_new < sc_cur:
            f = f_new
            if sc_new < best_sc:
                best_sc = sc_new
                best = f.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Metropolis criterion
            delta = sc_new - sc_cur
            if delta < 0 or rng.random() < math.exp(-delta / T):
                f = f_new
            stall += 1

        if stall > 200:
            f = block_perturb(f, rng, block_size=max(5, len(f)//5))
            T = 0.05
            stall = 0
