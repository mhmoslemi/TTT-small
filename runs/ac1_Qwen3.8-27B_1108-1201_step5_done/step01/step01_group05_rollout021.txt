from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def conv_full(p):
    """Compute full linear self-convolution using FFT."""
    n = len(p)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))[:2*n-1]
    return c

def score(p):
    """Compute the score for a normalized non-negative sequence."""
    n = len(p)
    s = np.sum(p)
    if s <= 0:
        return np.inf
    c = conv_full(p)
    max_b = np.max(c)
    return 2.0 * n * max_b / (s * s)

def normalize(p):
    """Normalize to sum 1, clip non-negative."""
    p = np.maximum(p, 0)
    s = np.sum(p)
    if s <= 0:
        return np.ones(len(p)) / len(p)
    return p / s

    # Build constraint matrix: A has 2n-1 rows, n columns

    rows = []
    cols = []

            cols.append(j)

The score 2n*max(p*p)/(sum p)^2 is minimized by iteratively solving a linear program
