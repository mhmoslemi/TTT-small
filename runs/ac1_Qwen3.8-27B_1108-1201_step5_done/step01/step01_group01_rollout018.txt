Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2
over non-negative sequences. Uses LSE-smoothed gradients with FFT acceleration,
both additive (simplex projection) and multiplicative (mirror descent) updates,

import numpy as np

def _simplex_project(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _grad_lse(a, b, alpha, N):
    """
    Gradient of (1/alpha)*log(sum(exp(alpha*b))) w.r.t. a.
    b = convolve(a, a) (precomputed, length 2n-1)
    a: length n
    N: FFT size >= 3n-1
    """
    n = len(a)
    b_max = np.max(b)
    exp_shifted = np.exp(alpha * (b - b_max))
    w_sum = np.sum(exp_shifted)
    w = exp_shifted / w_sum
    
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    wa = np.fft.irfft(W * A, N)
    term2 = wa[:n]
    
    a_flipped = a[::-1]
    AF = np.fft.rfft(a_flipped, N)
    w_af = np.fft.irfft(W * AF, N)
    term1 = w_af[n - 1:n - 1 + n]
    
    return term1 + term2
