"""
Improved LP-based iterative optimization with subgradient polishing,
adaptive step sizing, and multi-length search to minimize the
autocorrelation peak score: 2*n*max(conv(f,f))/sum(f)^2.

Key improvements over previous SOTA:
- Uses the provided sequence as primary warm start
- Backtracking line search for LP step size
- Subgradient steps between LP calls for faster convergence
- Adaptive length selection focusing time on best candidates
- Final coordinate-descent polish phase
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute the evaluation score quickly (mirrors evaluate_sequence)."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    n = len(f)
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] for 0<=k-j<n else 0."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_direction(f, M, time_limit_s=15.0):
    """Solve max sum(g) s.t. A g <= M*1, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def subgradient_step(f, lr):
    """One subgradient descent step on the score function."""
    n = len(f)
    conv = np.convolve(f, f)
    S = f.sum()
    M = conv.max()
    k_star = np.argmax(conv)
    # Subgradient of max(conv) w.r.t. f[j]
    sg = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = k_star - j
        if 0 <= i1 < n:
            val += f[i1]
        i2 = j - k_star
        if 0 <= i2 < n:
            val += f[i2]
        sg[j] = val
    # Subgradient of score = 2n*M/S^2
    g_score = (2.0 * n / (S * S)) * sg - (4.0 * n * M / (S ** 3))
    f_new = f - lr * g_score
    f_new = np.maximum(f_new, 0.0)
    s_new = f_new.sum()
    if s_new < 1e-10:
        return f
    return f_new / f_new.sum()


def lp_step_with_line_search(f, t_init=0.4, max_backtracks=5):
    """LP step with backtracking line search to ensure score doesn't increase."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return f, fast_score(f)
    
    g = solve_lp_direction(f, M, time_limit_s=15.0)
    if g is None:
        return f, fast_score(f)
    
    if np.linalg.norm(g - f) < 1e-10:
        return f, fast_score(f)
    
    sc_current = fast_score(f)
    t = t_init
    for _ in range(max_backtracks):
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new
        sc_new = fast_score(f_new)
        if sc_new < sc_current:
            return f_new, sc_new
        t *= 0.5
    
    # If no improvement with line search, try a very small step
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s_new = f_new.sum()
    if s_new < 1e-10:
        return f, sc_current
    f_new = f_new / s_new
    return f_new, fast_score(f_new)


def random_perturb(f, rng, scale=0.3):
    """Multiplicative random perturbation, renormalized."""
    n = len(f)
    cand = f.copy()
    # Perturb a random block
    w = max(1, int(rng.integers(2, max(3, n // 10))))
    start = int(rng.integers(0, n - w))
    cand[start:start + w] *= np.exp(rng.normal(0.0, scale, w))
    # Also try global small perturbation
    cand *= np.exp(rng.normal(0.0, 0.05, n))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def coordinate_descent_polish(f, budget_s, rng):
    """Final polish using pairwise coordinate swaps."""
    n = len(f)
    best = f.copy()
    best_sc = fast_score(best)
    deadline = time.monotonic() + budget_s
    lr = 0.1 / max(n, 10)
    attempts = 0
    
    while time.monotonic() < deadline:
        attempts += 1
        # Try perturbing a random pair symmetrically
        i = int(rng.integers(0, n))
        s = int(rng.integers(0, n))
        if i == s:
            continue
        delta = rng.normal(0.0, best.mean() * 0.1)
        cand = best.copy()
        cand[i] = max(0.0, cand[i] + delta)
        cand[s] = max(0.0, cand[s] - delta)
        sc = fast_score(cand)
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
        else:
            # Try the other direction
            cand[i] = max(0.0, cand[i] - delta)
            cand[s] = max(0.0, cand[s] + delta)
            sc = fast_score(cand)
            if sc < best_sc - 1e-12:
                best = cand
                best_sc = sc
    
    return best, best_sc


def make_initial(n, rng, method='cosine'):
    """Generate initial sequence of length n."""
    if method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'symmetric':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 12)
        a = np.exp(-0.5 * ((x - n // 4) / s) ** 2) + np.exp(-0.5 * ((x - 3 * n // 4) / s) ** 2)
    elif method == 'random':
        a = rng.random(n) + 0.1
    elif method == 'alternating':
        a = np.ones(n) * 0.5
        a[::2] = 1.0
        a[1::2] = 0.3
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def run_lp_optimization(f, budget_s, verbose=True):
    """Main LP-based optimization loop with subgradient interleaving."""
    n = len(f)
    best = f.copy()
    best_sc = fast_score(f)
    t = 0.4
    deadline = time.monotonic() + budget_s
    stall_count = 0
    lp_count = 0
    sg_count = 0
    last_print = time.monotonic()
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Alternate between LP and subgradient steps
        # LP is more powerful but slower; SG is fast but weaker
        if it % 3 == 0:
            # Subgradient step (fast)
            lr = max(0.01, 0.5 / n) * (0.95 ** (it // 100))
            f_new = subgradient_step(f, lr)
            sc = fast_score(f_new)
            if sc < best_sc:
                best = f_new.copy()
                best_sc = sc
                stall_count = 0
            else:
                stall_count += 1
            sg_count += 1
        else:
            # LP step with line search
            f_new, sc = lp_step_with_line_search(f, t)
            if sc < best_sc:
                best = f_new.copy()
                best_sc = sc
                stall_count = 0
                # Slightly increase t on success
                t = min(0.7, t * 1.05)
            else:
                stall_count += 1
                t = max(0.1, t * 0.9)
            f = f_new
            lp_count += 1
        
        # Periodic restart from best with perturbation if stalled
        if stall_count > 30:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            f = random_perturb(best, rng, scale=0.5)
            stall_count = 0
            t = 0.4
        
        # Print progress
        if verbose and time.monotonic() - last_print > 30:
            print("  [iter %d] score=%.6f, lp=%d, sg=%d, t=%.3f, time_left=%.0fs" %
                  (it, best_sc, lp_count, sg_count, t,
                   deadline - time.monotonic()), flush=True)
            last_print = time.monotonic()
    
    return best, best_sc
