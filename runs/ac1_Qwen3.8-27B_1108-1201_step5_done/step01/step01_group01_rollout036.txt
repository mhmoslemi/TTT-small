def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0][-1]
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def fft_convolve(a, b):

    fb = np.fft.rfft(b, size)

import numpy as np
import time

def project_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(a) + len(b) - 1
    size = 1 << (n-1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

# Top-level helper functions

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} in O(n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # Gradient
        grad = _smoothed_gradient(y, p)
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        
        # Update
        y_new = a + momentum * (a - a_prev)
        a_new = _project_simplex(y_new - eta * grad, 1.0)
        
        a_prev = a
        a = a_new
        t_k = t_new
        iter_count += 1
        
        # Periodic score check
        if iter_count % check_interval == 0:
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = iter_count
        
        # Break if stuck
        if iter_count - last_improve > 100000:
            break
    
    # Final check
    cur_score = _score(a)
    if cur_score < best_score:
        best_score = cur_score
        best = a.copy()
    
    return best, best_score

def _make_two_level(n, e):
    """Anti-symmetric two-level sequence."""
    a = np.ones(n)
    m = n // 2
    a[:m] = 1.0 + e
    a[m:] = 1.0 - e
    a = np.maximum(a, 0.0)
    a /= np.sum(a)
    return a

def _make_smooth_anti(n, e):
    """Smooth anti-symmetric: 1 + e*cos(pi*(2i/n - 1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + e * np.cos(np.pi * (2.0 * i / n - 1.0))
    a = np.maximum(a, 0.0)
    a /= np.sum(a)
    return a

def _make_quadratic_anti(n, e):
    """Quadratic anti-symmetric: 1 + e*(2i/n - 1)^2 - e/3 (centered)."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / n - 1.0  # Range [-1, 1]
    a = 1.0 + e * (t*t - 1.0/3.0)
    a = np.maximum(a, 0.0)
    a /= np.sum(a)
    return a
