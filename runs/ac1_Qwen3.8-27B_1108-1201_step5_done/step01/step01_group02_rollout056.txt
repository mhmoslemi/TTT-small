```python
def peak_shave(a, n_steps=100):
    a = np.maximum(0, a).astype(np.float64)
    s = a.sum()
    a = a / s  # normalize to sum = 1
    n = len(a)
    
    for step in range(n_steps):
        b = np.convolve(a, a)
        max_b = b.max()
        k_max = np.argmax(b)
        
        # Find all positions where b is close to max
        threshold = max_b * 0.95
        peak_positions = np.where(b >= threshold)[0]
        
        # For each peak position k, find the a_i that contribute most
        # b_k = sum_i a_i * a_{k-i}
        # The gradient of b_k wrt a_i is 2*a_{k-i} (approximately)
        
        # Compute the "pressure" on each a_i
        pressure = np.zeros(n)
        for k in peak_positions:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += 2 * a[j] / len(peak_positions)
        
        # Reduce a_i proportional to pressure
        # a_i = a_i - step_size * pressure_i (but keep non-negative)
        step_size = 0.01
        a = np.maximum(0, a - step_size * pressure)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    
    return a
```

```python
"""
Peak-shaving with momentum + LP refinement + local search to minimize
2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses subgradient-based peak reduction, alternating LP, and multiple starts.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_parabolic(n, p=1.0):
    t = np.arange(n, dtype=np.float64)
    x = 2.0 * t / (n - 1) - 1.0
    return np.maximum(0.0, 1.0 - np.abs(x) ** (2.0 * p))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def compute_pressure(a, b, threshold_frac=0.90):
    """Compute subgradient pressure from near-peak positions."""
    n = len(a)
    max_b = b.max()
    if max_b < 1e-30:
        return np.zeros(n)
    threshold = max_b * threshold_frac
    peak_indices = np.where(b >= threshold)[0]
    if len(peak_indices) == 0:
        idx = np.argmax(b)
        peak_indices = np.array([idx])
    pressure = np.zeros(n)
    for k in peak_indices:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        a_idx = k - np.arange(j_start, j_end + 1)
        pressure[j_start:j_end + 1] += 2.0 * a[a_idx]
    pressure /= len(peak_indices)
    return pressure


def peak_shave(a, time_limit=60.0, verbose=False, momentum=0.5, thresh=0.90):
    """Iteratively reduce peaks of auto-convolution using subgradient descent with momentum."""
    start = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    step_size = 0.03
    v = np.zeros(n)
    stagnant = 0
    it = 0

    while time.monotonic() - start < time_limit:
        b = np.convolve(a, a)
        max_b = b.max()
        score = 2.0 * n * max_b  # sum(a)=1 so denominator is 1

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 50 == 0 or it < 5):
                print(f"      PS iter {it:4d}: score={score:.8f}")
        else:
            stagnant += 1

        if stagnant > 200:
            if verbose:
                print(f"      PS stagnant at iter {it}, best={best_score:.8f}")
            break

        pressure = compute_pressure(a, b, threshold_frac=thresh)

        # Momentum update
        v = momentum * v - step_size * pressure
        a_new = np.maximum(0, a + v)
        s = a_new.sum()
        if s < 1e-15:
            step_size *= 0.5
            v *= 0.5
            it += 1
            continue
        a_new = a_new / s

        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            # Try smaller step
            v = momentum * v - step_size * 0.3 * pressure
            a_new = np.maximum(0, a + v)
            s = a_new.sum()
            if s < 1e-15:
                step_size *= 0.5
            else:
                a_new = a_new / s
                if score_sequence(a_new) <= score + 1e-14:
                    a = a_new
            step_size = max(step_size * 0.95, 1e-12)

        it += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      PS done: best={best_score:.8f} after {it} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse convolution matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=20.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=120.0, max_iters=200, verbose=True):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"      LP start: score={best_score:.8f}, n={n}")

    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"      LP iter {it:4d}: score={score:.8f}")
        else:
            stagnant += 1
        if stagnant > 50:
            if verbose:
                print(f"      LP stagnant at iter {it}, best={best_score:.8f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"      LP no improvement (sum_g={sum_g:.10f}) at iter {it}")
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.2 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"      LP done: best={best_score:.8f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=False):

    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(max(1, n // 300), max(2, n // 5))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        noise_scale = 0.05 + 0.05 * (1.0 - (time.monotonic() - start_time) / time_limit_s)
        candidate[start:stop] *= np.exp(rng.normal(0.0, noise_scale, size=stop - start))
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 20 == 0:
                print(f"      LS accept {accept_count}: score={best_score:.8f}")

    if verbose:
        print(f"      LS done: best={best_score:.8f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

        t = min(time_alloc, remaining * 0.85)
        if t < 10:
            return
        print(f"\n=== {label} (n={len(a)}, init_score={init_score:.6f}, time={t:.0f}s) ===")
        refined, score = combined_refine(a, time_limit=t, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = refined.copy()
            print(f"  *** NEW BEST: score={score:.8f}, n={len(a)}, elapsed={time.monotonic()-total_start:.1f}s")
        else:
            print(f"  No improvement: {score:.8f} >= {best_score:.8f}")

    # Try starting from height_sequence_1 if available
    try:
        h1 = np.array(height_sequence_1, dtype=np.float64)
        if len(h1) > 0 and h1.sum() > 0.01:
            h1 = normalize_nonneg(h1)
            h1_score = score_sequence(h1)
            print(f"Starting from height_sequence_1: n={len(h1)}, score={h1_score:.8f}")
            # Allocate significant time to refine the best known
            try_candidate(h1, "height_seq_1 refine", budget_s * 0.35)
    except (NameError, Exception) as e:
        print(f"No height_sequence_1 available: {e}")

    # Try various lengths and initializations
    configs = []
    # Smaller n (faster LP)
    configs.append((500, 'raised_cosine', 0.12))
    configs.append((500, 'blackman', 0.08))
    configs.append((700, 'raised_cosine', 0.10))
    configs.append((800, 'gaussian', 0.10))
    configs.append((1000, 'raised_cosine', 0.12))
    configs.append((1000, 'parabolic', 0.10))
    configs.append((1500, 'raised_cosine', 0.12))
    configs.append((2000, 'raised_cosine', 0.15))
    configs.append((2000, 'gaussian', 0.10))
    configs.append((1000, 'blackman', 0.08))
    configs.append((1500, 'parabolic', 0.08))
    configs.append((800, 'random', 0.06))
    configs.append((1200, 'random', 0.06))

    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        time_alloc = min(time_frac * budget_s, remaining * 0.8)
        if time_alloc < 15:
            continue

        if init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.25)
        elif init_type == 'blackman':
            a = initial_blackman(n)
        elif init_type == 'parabolic':
            a = initial_parabolic(n, p=rng.uniform(0.8, 2.0))
        elif init_type == 'random':
            a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 6)))
        else:
            a = initial_raised_cosine(n)

        try_candidate(a, f"{init_type}_{n}", time_alloc)

    # Final aggressive polishing of best solution
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")
            rng2 = np.random.default_rng(999)
            polished, score = local_search_refine(best_seq, time_limit_s=min(remaining * 0.3, 120), rng=rng2, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = polished.copy()
                print(f"  *** After LS: score={score:.8f}")

            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                polished2, score2 = peak_shave(best_seq, time_limit_s=min(remaining2 * 0.5, 120), verbose=True, momentum=0.2, thresh=0.97)
                if score2 < best_score:
                    best_score = score2
                    best_seq = polished2.copy()
                    print(f"  *** After final PS: score={score2:.8f}")

            remaining3 = deadline - time.monotonic()
            if remaining3 > 15 and len(best_seq) <= 1000:
                polished3, score3 = lp_refine(best_seq, time_limit_s=min(remaining3 * 0.8, 90), verbose=True)
                if score3 < best_score:
                    best_score = score3
                    best_seq = polished3.copy()
                    print(f"  *** After final LP: score={score3:.8f}")

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    print(f"\nFinal result: score={best_score:.8f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.8f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
```
