Let me verify: when `k = 2i`, `k-i = i`, so `dh[k]/da[i] = 2*a[i]`. ✓

So the gradient is simply: `dh[k]/da[i] = 2*a[k-i]` (where `a[j] = 0` for `j < 0` or `j >= n`).

This means the subgradient of `max_k h[k]` (w.r.t. `a[i]`) is:
`g[i] = (2/|S|) * sum_{k in S} a[k-i]` (where `a[j] = 0` for out-of-range `j`)

This is just a "smeared" version of `a` at the positions of the peaks!

If `S = {k*}` (single peak at `k*`), then `g[i] = 2*a[k*-i]`, which is a shifted and reflected version of `a`.

If there are multiple peaks, `g[i]` is a sum of shifted/reflected copies of `a`.

This is very interpretable! The subgradient tells us: "at position `i`, the rate of change of the peak is proportional to the value of `a` at the position that would shift a peak to location `i`."

So to reduce the peak at position `k*`, we should decrease `a[i]` where `a[k*-i]` is large, i.e., where `a` has significant values at positions that "feed into" the peak.

```python
"""
Multi-phase optimization to minimize peak auto-convolution of non-negative sequences.
Combines fast subgradient descent with LP-based refinement and local search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_square(n):
    return np.ones(n)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def gradient_step(a, h, top_m=10, tau=0.01):
    """Compute weighted subgradient of max(conv(a,a)) using top-m peaks."""
    n = len(a)
    # Find top m positions
    if top_m >= len(h):
        top_m = len(h)
    top_indices = np.argpartition(h, -top_m)[-top_m:]
    top_vals = h[top_indices]
    
    # Softmax weights
    if tau > 0:
        exp_vals = np.exp((top_vals - top_vals.max()) / tau)
        weights = exp_vals / exp_vals.sum()
    else:
        weights = np.ones(top_m) / top_m
    
    # Compute gradient: g[i] = sum_k w[k] * 2*a[k-i] (with a[j]=0 for out of range)
    g = np.zeros(n)
    for ii, k in enumerate(top_indices):
        w = 2.0 * weights[ii]
        # g[i] += w * a[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            # g[i_start:i_end] += w * a[k-i_start : k-i_end+1 : -1]
            g[i_start:i_end] += w * a[k - i_start : k - i_end + 1 : -1]
    
    return g


def subgradient_optimize(a, time_limit_s=100.0, eta0=0.05, top_m=None, tau=0.01, verbose=True):
    """
    Subgradient descent on max(conv(a,a)) with simplex projection.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    if top_m is None:
        top_m = max(5, min(50, n // 20))
    
    best_a = a.copy()
    best_score = score_sequence(a)
    eta = eta0
    stagnant = 0
    
    if verbose:
        print(f"    SubGrad start: score={best_score:.6f}, n={n}, top_m={top_m}")
    
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        h = np.convolve(a, a)
        score = 2.0 * n * h.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 100 == 0 or it < 5):
                print(f"    SubGrad iter {it:5d}: score={score:.6f}, eta={eta:.4f}")
        else:
            stagnant += 1
            if stagnant > 200:
                if verbose:
                    print(f"    SubGrad stagnant at iter {it}, best={best_score:.6f}")
                break
            # Reduce step size
            eta = max(eta * 0.9, eta0 * 0.01)
        
        it += 1
        if it % 5 == 0:
            eta = min(eta * 1.01, eta0 * 2.0)  # Slowly increase if not stuck
        
        # Compute gradient
        g = gradient_step(a, h, top_m=top_m, tau=tau)
        
        # Project gradient to simplex tangent (zero sum)
        g = g - g.mean()
        
        # Gradient descent step
        a_new = a - eta * g
        a_new = normalize_nonneg(a_new)
        
        # Check if score improved
        h_new = np.convolve(a_new, a_new)
        score_new = 2.0 * n * h_new.max() / (a_new.sum() ** 2)
        
        if score_new < score:
            a = a_new
        else:
            a = a  # Don't update, just reduce eta
            eta = eta * 0.5
    
    final_score = score_sequence(best_a)
    if verbose:
        print(f"    SubGrad done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, max_iters=200, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(3.0, min(60.0, remaining * 0.3))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement at iter {it}")
            break
        g = g / sum_g
        alpha = min(0.5, max(0.03, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(max(1, n // 300), max(2, n // 10))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.05, size=stop - start))
        if rng.random() < 0.3:

            accept_count += 1

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
```

"""
Multi-phase optimization to minimize peak auto-convolution of non-negative sequences.
Combines FFT-accelerated subgradient descent, Newton continuation method, 
LP-based refinement, and local search for comprehensive exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def fast_convolve(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_hamming(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = t / (n - 1)
    return np.maximum(0.0, 0.54 - 0.46 * np.cos(2 * np.pi * x))


def initial_kaiser(n, beta=4.0):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = 2.0 * t / (n - 1) - 1.0
    from scipy.special import i0
    return np.maximum(0.0, i0(beta * np.sqrt(np.maximum(0, 1 - x**2))) / i0(beta))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)

    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:

    return a / s


def compute_subgradient(a, h, top_m=10, tau=0.01):
    """Compute weighted subgradient of max(conv(a,a)) using top-m peaks."""
    n = len(a)
    if top_m >= len(h):
        top_m = len(h)
    top_indices = np.argpartition(h, -top_m)[-top_m:]
    top_vals = h[top_indices]
    
    if tau > 1e-10:
        exp_vals = np.exp((top_vals - top_vals.max()) / tau)
        weights = exp_vals / exp_vals.sum()
    else:
        weights = np.ones(top_m) / top_m
    
    # Use FFT-based computation: g[i] = sum_k w[k] * 2*a[k-i]
    # Build excess vector from top peaks
    e = np.zeros(len(h))
    for ii, k in enumerate(top_indices):
        e[k] = 2.0 * weights[ii]
    
    # g[i] = sum_k e[k] * a[k-i] = (e * a_rev)[n-1+i]
    size = 1
    while size < len(h) + n:
        size *= 2
    E = np.fft.rfft(e, size)
    A_rev = np.zeros(size)
    A_rev[:n] = a[::-1]
    G = E * np.fft.rfft(A_rev, size)
    g_full = np.fft.irfft(G, size)
    g = g_full[n - 1: n - 1 + n]
    
    return g

def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, max_iters=150, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = fast_convolve(a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 25:
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(3.0, min(45.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.4, max(0.02, 0.8 / (1.0 + 0.15 * it)))
        a = (1.0 - alpha) * a + alpha * g
        a = normalize_nonneg(a)
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
