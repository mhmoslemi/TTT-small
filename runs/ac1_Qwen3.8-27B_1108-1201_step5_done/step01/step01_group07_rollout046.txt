    n = len(a)

    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = len(v)

    """Project v onto {x >= 0, sum(x) = s} in O(n) expected time."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.zeros(n)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(a0, time_limit, T_start=20.0, T_end=500.0, use_subgrad_end=True):
    """FISTA with simplex projection. Uses log-sum-exp then switches to subgradient.
    Returns (best_a_normalized, best_max_conv)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a_sum

    c = fast_convolve(a)
    best_max_c = c.max()
    best_a = a.copy()

    z = a.copy()
    t_k = 1.0
    step_base = 1.0
    t0 = time.monotonic()

    it = 0
    while time.monotonic() - t0 < time_limit:
        frac = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** frac

        if T > 100 and use_subgrad_end:
            grad, c_z, max_c_z = compute_subgradient_max(z)
            step = step_base
        else:
            grad, c_z, max_c_z = compute_grad_lse(z, T)
            step = step_base * T / max(1.0, 2.0 * max_c_z * T)
            step = min(step, T * 0.5)

        if step < 1e-12:
            step = T * 0.1

        y = z - step * grad
        a_new = project_simplex(y, 1.0)

        c_new = fast_convolve(a_new)
        max_new = c_new.max()

        if max_new < best_max_c:
            best_max_c = max_new
            best_a = a_new.copy()
            it = 0
        else:
            it += 1
            if it % 50 == 0:
                step_base *= 0.8
            if it > 500:
                step_base = np.clip(step_base * 1.2, 0.01, 5.0)
                it = 200

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new
        z = a_new + alpha * (a_new - a)
        z = np.maximum(z, 0.0)
        z_sum = z.sum()
        if z_sum > 1e-15:
            z /= z_sum
        else:
            z = a_new.copy()
        a = a_new
        t_k = t_new

    return best_a, best_max_c


def multiplicative_optimize(a0, time_limit, T_start=30.0, T_end=300.0, eta0=0.3):
    """Improved multiplicative mirror descent with FFT-based gradient.
    Returns (best_a_normalized, best_max_conv)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a_sum

    c = fast_convolve(a)
    best_max_c = c.max()
    best_a = a.copy()

    eta = eta0
    T = T_start
    patience = 0
    max_patience = 100
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        frac = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** frac

        if T > 150:
            sg, c, max_c = compute_subgradient_max(a)
        else:
            sg, c, max_c = compute_grad_lse(a, T)

        if max_c < 1e-15:
            break

        # Center the subgradient
        sg_mean = sg.mean()
        log_factor = -eta * (sg - sg_mean)
        log_max = log_factor.max()
        if log_max > 50:
            log_factor = log_factor - log_max
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
        else:
            a /= a_sum

        c_new = fast_convolve(a)
        max_new = c_new.max()

        if max_new < best_max_c - 1e-10:
            best_max_c = max_new
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            if eta < 0.005:
                eta = 0.2
                max_patience = 50
            T = min(T * 1.5, T_end)

    return best_a, best_max_c


def lp_refine(a, max_iters=50):
    """LP-based refinement. Given normalized a, solve LP to find a better direction.
    Returns refined a (normalized)."""
    n = len(a)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        return a
    a = a / a_sum

    c = fast_convolve(a)
    M = c.max()
    if M < 1e-15:
        return a

    # Build constraint matrix: A[k,j] = a[j] if 0 <= k-j <= n-1, else 0
    # A is (2n-1) x n
    # For efficiency, build sparse
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])

    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    # Maximize sum(b) subject to A*b <= M, b >= 0
    # linprog minimizes, so minimize -sum(b)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')

    if res.status != 0 or res.x is None:
        return a

    b = np.maximum(res.x, 0.0)
    b_sum = b.sum()
    if b_sum < 1e-15:
        return a

    b = b / b_sum

    # Move towards b
    t = 0.3
    a_new = (1 - t) * a + t * b
    a_new = np.maximum(a_new, 0.0)
    a_new_sum = a_new.sum()
    if a_new_sum < 1e-15:
        return a
    a_new = a_new / a_new_sum
    return a_new


def try_lp_refine_full(a, time_budget=30.0):
    """Iteratively apply LP refinement within a time budget."""
    n = len(a)
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    t0 = time.monotonic()
    t_frac = 0.3

    while time.monotonic() - t0 < time_budget:
        a_new = lp_refine(a, max_iters=1)
        c_new = fast_convolve(a_new)
        max_new = c_new.max()
        if max_new < best_max - 1e-10:
            best_max = max_new
            best_a = a_new.copy()
            a = a_new
            t_frac = 0.3
        else:
            t_frac *= 0.7
            if t_frac < 0.01:
                break
            a = (1 - t_frac) * best_a + t_frac * a_new
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a /= s
    return best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g

    # Edge-concentrated
    edge = np.zeros(n)
    edge[:max(1, n // 4)] = 1.0
    edge[-max(1, n // 4):] = 1.0
    edge /= edge.sum()
    inits['edge'] = edge

    # Cosine
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b

    # Random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    # Random sparse
    n_nonzero = max(2, n // 4)
    idx = rng.choice(n, size=n_nonzero, replace=False)
    sparse = np.zeros(n)
    sparse[idx] = 1.0
    sparse /= sparse.sum()
    inits['sparse'] = sparse

    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    return inits


def run_search(seed, budget_s):
    """Main search routine with hierarchical time management."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Try to access parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    # Phase 1: Quick scan
    print("Phase 1: Quick scan...")
    t0 = time.monotonic()

    lengths = [64, 128, 192, 256, 384, 512]
    scan_configs = []

    for n in lengths:
        inits = generate_initializations(n, rng, parent)
        for name, a0 in inits.items():
            scan_configs.append((n, name, a0))

    # Time per scan config (quick)
    n_configs = len(scan_configs)
    phase1_time = min(300.0, (deadline - time.monotonic()) * 0.3)
    time_per_scan = max(2.0, phase1_time / max(1, n_configs))

    scan_results = []
    for i, (n, name, a0) in enumerate(scan_configs):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(time_per_scan, deadline - time.monotonic() - 120)
        if t_alloc < 1.5:
            break

        # Use FISTA for quick scan
        a_opt, obj_opt = fista_optimize(a0, time_limit=t_alloc,
                                         T_start=15.0, T_end=100.0)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization on top configs
    if time.monotonic() < deadline - 60:
        print("Phase 2: Deep optimization...")
        scan_results.sort(key=lambda x: x[0])

        # Select top configs with diversity
        top_configs = []
        seen = set()
        for score, n, name, a_opt in scan_results:
            key = (n, name)
            if key not in seen:
                top_configs.append((n, name, a_opt))
                seen.add(key)
                if len(top_configs) >= 10:
                    break

        # Always include current best
        if best_sequence is not None:
            top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

        remaining = deadline - time.monotonic() - 180
        time_per_deep = max(10.0, remaining / max(1, len(top_configs)))

        for i, (n, name, a0) in enumerate(top_configs):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(time_per_deep, deadline - time.monotonic() - 150)
            if t_alloc < 5:
                break

            # Try both FISTA and multiplicative
            half = t_alloc * 0.5

            a1, obj1 = fista_optimize(a0, time_limit=half,
                                       T_start=20.0, T_end=400.0)
            score1 = float(evaluate_sequence(a1.tolist()))
            if np.isfinite(score1) and score1 < best_score:
                best_score = score1
                best_sequence = a1.copy()
                print(f"  Deep FISTA: n={n}, {name}, score={score1:.6f}")

            if time.monotonic() < deadline - 100:
                start2 = a1 if (np.isfinite(score1) and score1 < best_score) else a0
                a2, obj2 = multiplicative_optimize(start2,
                                                    time_limit=deadline - time.monotonic() - 150,
                                                    T_start=30.0, T_end=200.0,
                                                    eta0=0.3)
                score2 = float(evaluate_sequence(a2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a2.copy()
                    print(f"  Deep Multi: n={n}, {name}, score={score2:.6f}")

    # Phase 3: LP refinement
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("Phase 3: LP refinement...")
        lp_budget = min(120.0, deadline - time.monotonic() - 20)
        a_lp = try_lp_refine_full(best_sequence, time_budget=lp_budget)
        score_lp = float(evaluate_sequence(a_lp.tolist()))
        if np.isfinite(score_lp) and score_lp < best_score:
            best_score = score_lp
            best_sequence = a_lp.copy()
            print(f"  LP refined: score={score_lp:.6f}")

        # Try LP from a slightly perturbed version
        if time.monotonic() < deadline - 20:
            perturbed = best_sequence + 0.01 * rng.standard_normal(len(best_sequence))
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps > 1e-10:
                perturbed /= ps
            a_lp2 = try_lp_refine_full(perturbed,
                                       time_budget=min(60.0, deadline - time.monotonic() - 15))
            score_lp2 = float(evaluate_sequence(a_lp2.tolist()))
            if np.isfinite(score_lp2) and score_lp2 < best_score:
                best_score = score_lp2
                best_sequence = a_lp2.copy()
                print(f"  LP perturbed: score={score_lp2:.6f}")

    # Phase 4: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        remaining = deadline - 10 - time.monotonic()
        n = len(best_sequence)

        # Try FISTA with very high temperature (focus on actual max)
        t_alloc = remaining * 0.4
        a_polish, obj_polish = fista_optimize(best_sequence, time_limit=t_alloc,
                                               T_start=200.0, T_end=1000.0,
                                               use_subgrad_end=True)
        score_polish = float(evaluate_sequence(a_polish.tolist()))
        if np.isfinite(score_polish) and score_polish < best_score:
            best_score = score_polish
            best_sequence = a_polish.copy()
            print(f"  Polish FISTA: score={score_polish:.6f}")

        # Try multiplicative with high T
        if time.monotonic() < deadline - 10:
            t_alloc2 = deadline - 10 - time.monotonic()
            a_polish2, _ = multiplicative_optimize(best_sequence,
                                                    time_limit=t_alloc2,
                                                    T_start=100.0, T_end=500.0,
                                                    eta0=0.1)
            score_polish2 = float(evaluate_sequence(a_polish2.tolist()))
            if np.isfinite(score_polish2) and score_polish2 < best_score:
                best_score = score_polish2
                best_sequence = a_polish2.copy()
                print(f"  Polish Multi: score={score_polish2:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses FISTA + multiplicative weights + LP refinement
    to minimize 2n*max(conv)/(sum)^2. Multi-phase hierarchical search.
    """
    result = run_search(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
