def compute_score(p):

def project_simplex(v):

    v = np.asarray(v, dtype=np.float64).copy()

    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()

def make_uniform(n):

def make_cup(n, a):

    x = np.linspace(0, 1, n)

def make_gaussian(n, mu, sigma):
    x = np.linspace(0, 1, n)
    p = np.exp(-((x - mu) ** 2) / (2 * sigma ** 2))
    return safe_normalize(p)


def make_chebyshev(n, coeffs):
    """Generate shape from Chebyshev polynomial coefficients in log-space."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    # Compute Chebyshev polynomials
    T0 = np.ones(n)
    if len(coeffs) > 0:
        logp = coeffs[0] * T0
    else:
        logp = np.zeros(n)
    if len(coeffs) > 1:
        T1 = t.copy()
        logp += coeffs[1] * T1
    for d in range(2, len(coeffs)):
        T2 = 2 * t * T1 - T0
        logp += coeffs[d] * T2
        T0, T1 = T1, T2
    # Add endpoint taper
    eps = 1e-6
    logp += 1.5 * np.log(x + eps) + 1.5 * np.log(1 - x + eps)
    m = np.max(logp)
    p = np.exp(logp - m)
    return safe_normalize(p)


def make_random_smooth(n, rng, nfreqs=5):
    x = np.linspace(0, 1, n)
    p = np.ones(n)
    for k in range(nfreqs):
        freq = rng.integers(1, max(2, n // 50))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.5) / (k + 1)
        p += amp * np.sin(2 * np.pi * freq * x + phase)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)

            lr = min(max(base_lr, 1e-8), 5.0)

    print(f"Time used: {time.monotonic() - t_start:.1f}s / {budget_s}s", flush=True)

    return [float(x) for x in result]
