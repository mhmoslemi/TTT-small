"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based refinement as the primary optimizer, combined with soft-max
gradient descent and targeted peak reduction. Explores multiple starting
points and sequence lengths within the time budget.

import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_direction_fast(f, M, A_cache=None):
    """Optimized LP with pre-built matrix."""
    n = len(f)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_cache, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def build_conv_matrix(f):
    """Build the convolution constraint matrix."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    return A

def softmax_gradient_step(a, lr, tau=30.0):
    """One gradient step using soft-max weighting on peaks."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    c = np.convolve(a, a)
    
    # Soft-max weighting
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    # Gradient: grad[j] = 2 * sum_k a[k] * w[j+k] for valid k
    grad = np.zeros(n)
    for j in range(n):
        lo_m = j  # m = j + k where k ranges
        # c[m] where m = j + k, 0 <= k < n
        # So m ranges from j to j+n-1, but m < 2n-1
        hi_m = min(2 * n - 2, j + n - 1)
        k_lo = max(0, j)  # k = m - j, need 0 <= k < n
        k_hi = min(n - 1, hi_m - j)
        # Actually: grad[j] = sum over valid m of 2*a[m-j]*w[m] where 0<=m-j<n
        # m-j ranges from 0 to n-1, so m ranges from j to j+n-1
        k_vals = np.arange(0, n)
        m_vals = j + k_vals
        valid = m_vals < (2 * n - 1)
        grad[j] = 2.0 * np.dot(a[k_vals[valid]], w[m_vals[valid]])
    
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    return normalize(a, S_t)

def hard_max_gradient_step(a, lr):
    """Gradient step targeting only the single peak."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    return normalize(a, S_t)

def multi_peak_gradient_step(a, lr, num_peaks=5, tau=10.0):
    """Gradient step targeting the top few peaks."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    c = np.convolve(a, a)
    
    # Weight peaks by soft-max over top values
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(num_peaks, len(c_sorted))-1]
    c_above = np.maximum(c - threshold, 0)
    w = np.exp(tau * (c - c.max()))
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        m_vals = j + np.arange(n)
        valid = m_vals < (2 * n - 1)
        grad[j] = 2.0 * np.dot(a[np.arange(n)[valid]], w[m_vals[valid]])
    
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    return normalize(a, S_t)

def generate_m_sequence_start(n, rng):
    """Generate a sequence inspired by m-sequences (low autocorrelation)."""
    # Use a pseudo-random binary sequence as inspiration
    binary = rng.integers(0, 2, size=n).astype(float)
    # Smooth it
    kernel_size = max(1, n // 32)
    if kernel_size > 1:
        kernel = np.ones(kernel_size) / kernel_size
        binary = np.convolve(binary, kernel, mode='same')
    # Make non-negative and normalize
    binary = binary - binary.min() + 0.1
    return binary

def generate_gold_seq_start(n, rng):
    """Generate a sequence with Gold-code-like properties."""
    # Sum of two m-sequences (Gold sequence concept)
    seq1 = rng.integers(0, 2, size=n).astype(float)
    seq2 = np.roll(seq1, n // 3)
    gold = seq1 + seq2  # Values in {0, 1, 2}
    # Smooth
    kernel_size = max(1, n // 64)
    if kernel_size > 1:
        kernel = np.ones(kernel_size) / kernel_size
        gold = np.convolve(gold, kernel, mode='same')
    gold = gold - gold.min() + 0.1
    return gold

def generate_flat_spectrum_start(n, rng):
    """Try to generate a sequence with a relatively flat power spectrum."""
    # Start with random, then iteratively flatten the spectrum
    a = rng.random(n) + 0.5
    for _ in range(100):
        # Compute power spectrum
        F = np.fft.fft(a, n=4*n)
        power = np.abs(F[:2*n])**2
        # Target: flat spectrum
        target = np.ones(2*n) * power.mean()
        # Gradient: move toward flat spectrum
        # d(power)/d(a) involves FFT
        # Simplified: a_new = a - eta * IFFT(2*F*(power-target)/target)
        err = (power - target) / (target + 1e-10)
        F_err = np.zeros(4*n, dtype=complex)
        F_err[:2*n] = 2 * F[:2*n] * err
        a = a - 0.5 * np.real(np.fft.ifft(F_err))[:n]
        a = np.maximum(a, 0.1)
    return a

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Primary method: LP-based directional optimization (maximize sum(g) s.t. conv(f,g)<=M).
Secondary: soft-max gradient descent and directed peak reduction.
Explores multiple starting points and sequence lengths within the time budget.
Key insight: The LP step finds the optimal direction to reduce the convolution
peak while maintaining non-negativity, making it the most powerful single step.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def build_conv_matrix(f):
    """Build the convolution constraint matrix A where (A@g)[m] = (conv(f,g))[m]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    return A


def lp_solve_direction(f, M, A_mat):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    c_obj = -np.ones(n)
    b = np.full(A_mat.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def softmax_gradient_step(a, lr, tau=30.0):
    """Gradient step using soft-max weighting on convolution peaks."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    c = np.convolve(a, a)

    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for j in range(n):
        m_vals = j + np.arange(n)
        valid = m_vals < (2 * n - 1)
        grad[j] = 2.0 * np.dot(a[np.arange(n)[valid]], w[m_vals[valid]])

    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    return normalize(a, S_t)


def hard_max_gradient_step(a, lr):
    """Gradient step targeting only the single peak position."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    return normalize(a, S_t)


def directed_peak_reduction(a, strength=0.7):
    """Directly reduce elements contributing to the peak."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    c = np.convolve(a, a)
    m_star = int(c.argmax())

    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)

    contributions = np.zeros(hi - lo + 1)
    for idx, j in enumerate(range(lo, hi + 1)):
        contributions[idx] = a[j] * a[m_star - j]

    for idx, j in enumerate(range(lo, hi + 1)):
        partner = m_star - j
        if j == partner:
            a[j] *= strength
        else:
            if a[j] >= a[partner]:
                a[j] *= strength
            else:
                a[partner] *= strength

    a = np.maximum(a, 0.0)
    return normalize(a, S_t)

def gen_m_seq_start(n, rng):

def gen_gold_start(n, rng):

def gen_flat_spectrum_start(n, rng):
    """Iteratively flatten the power spectrum to reduce autocorrelation peak."""
    a = rng.random(n) + 0.5

        delta = 0.3 * np.real(np.fft.ifft(F_err))[:n]

        print(f"  Candidate {idx+1}/{len(candidates)}: n={len(a0)}, method={method}")
        try_candidate(a0, method)

    # Phase 2: Intensive refinement of best
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        print(f"\n=== Phase 2: Refining best (score={best_score:.6f}) ===")
        for pass_i in range(15):
            if time.monotonic() > deadline - 5.0:
                break
            # Alternate between two optimizers
            if pass_i % 2 == 0:
                seq, score = optimize_lp_refined(best_seq, deadline)
            else:
                seq, score = optimize_hybrid(best_seq, deadline)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine {pass_i}: score={best_score:.6f}")
            else:
                # Try a perturbation-based restart from best
                if time.monotonic() > deadline - 30.0:
                    break
                rng_refine = np.random.default_rng(pass_i * 104729 + 42)
                for attempt in range(5):
                    if time.monotonic() > deadline - 5.0:
                        break
                    factor = np.exp(rng_refine.standard_normal(len(best_seq)) * 0.05)
                    perturbed = np.maximum(best_seq * factor, 0.0)
                    perturbed = normalize(perturbed)
                    seq, score = optimize_hybrid(perturbed, deadline)
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
                        print(f"  Perturb {pass_i}-{attempt}: score={best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = compute_objective(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"Eval warning: {e}")
        return [float(x) for x in best_seq]

    print("Fallback: uniform sequence")
    return [1.0] * 512
