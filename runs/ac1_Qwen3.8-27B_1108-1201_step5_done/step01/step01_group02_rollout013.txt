The gradient is:

import numpy as np
from scipy.fft import fft, ifft
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs = np.arange(1, n + 1)
    cond = u * cs > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def lse_gradient(a, tau, L):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*conv_k(a)))."""
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax
    max_c = conv.max()
    exp_t = np.exp(tau * (conv - max_c))
    p = exp_t / exp_t.sum()
    # FFT
    A = fft(a, L)
    P = fft(p, L)
    h1 = np.real(ifft(np.conj(A) * P))[:n]
    h2 = np.real(ifft(A * P))[:n]
    grad = h1 + h2
    return grad, p

def smooth_optimize(a_init, time_limit=300, tau_init=50, tau_max=5000,

    a = a_init.copy().astype(np.float64)

    a = a / a.sum()
    n = len(a)
    L = 1
    while L < 4 * n:
        L *= 2
    
    start = time.monotonic()

    best_score = score_sequence(a)
    
    tau = tau_init
    step = step_init
    stagnant = 0

        tau = min(tau_max, tau_init * (1.02 ** it))
        # Adaptive step: decrease if stuck
        if stagnant > 100:
            step *= 0.95
            stagnant = 0
        
        grad, p = lse_gradient(a, tau, L)
        a_new = project_simplex(a - step * grad)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:

```python
"""
Minimize peak auto-convolution of non-negative sequences using:
1. Mirror descent with log-sum-exp smoothing (fast spectral gradient)
2. Iterative spectral shaping (FFT-based flat spectrum pursuit)
3. Multi-start with diverse initializations and lengths
4. Local search polishing
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs = np.arange(1, n + 1)
    cond = u * cs > (cssv - 1)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_conv_gradient(a, p):
    """Compute sum_k p[k] * (∇conv_k(a)) using FFT.
    a: length n, p: length 2n-1 (softmax weights over convolution lags)
    Returns gradient of length n.
    """
    n = len(a)
    L = 1
    while L < 4 * n:
        L *= 2
    A = fft(a, L)
    P = fft(p, L)
    h1 = np.real(ifft(np.conj(A) * P))[:n]
    h2 = np.real(ifft(A * P))[:n]
    return h1 + h2


def mirror_descent(a_init, time_limit=200, tau_init=100, tau_max=10000,
                   eta_init=1.0, verbose=False):
    """Mirror descent with log-sum-exp smoothing."""
    a = normalize_nonneg(a_init.copy().astype(np.float64))
    n = len(a)
    L = 1
    while L < 4 * n:
        L *= 2
    
    start = time.monotonic()
    best_a = a.copy()
    best_score = score_sequence(a)
    tau = tau_init
    eta = eta_init
    stagnant = 0
    
    it = 0
    while time.monotonic() - start < time_limit:
        tau = min(tau_max, tau_init * (1.005 ** it))
        if stagnant > 200:
            eta *= 0.9
            if eta < 0.0001:
                break
            stagnant = 0
        
        conv = np.convolve(a, a)
        max_c = conv.max()
        # Numerically stable softmax
        exp_t = np.exp(tau * (conv - max_c))
        p_sum = exp_t.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[conv.argmax()] = 1.0
        else:
            p = exp_t / p_sum
        
        grad = compute_conv_gradient(a, p)
        
        # Multiplicative update (mirror descent with entropy)
        log_w = -eta * grad
        log_w -= log_w.max()
        w = a * np.exp(log_w)
        s = w.sum()
        if s > 1e-30:
            a_new = w / s
        else:
            a_new = a.copy()
        
        score = score_sequence(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            stagnant = 0
            if verbose and it % 1000 == 0:
                print(f"        MD iter {it}: score={score:.6f}, tau={tau:.0f}, eta={eta:.4f}")
        else:
            stagnant += 1
        
        it += 1
    
    if verbose:
        print(f"        MD done: {it} iters, best={best_score:.6f}, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def spectral_shaping(a_init, time_limit=60, n_iters=1000, verbose=False):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    a = normalize_nonneg(a_init.copy().astype(np.float64))
    n = len(a)
    L = 1
    while L < 2 * n:
        L *= 2
    
    start = time.monotonic()
    best_a = a.copy()
    best_score = score_sequence(a)
    
    alpha = 0.5  # mixing parameter
    
    for it in range(n_iters):
        if time.monotonic() - start > time_limit:
            break
        
        A = fft(a, L)
        power = np.abs(A) ** 2
        target = power.mean()
        
        # Shape: scale each frequency component
        scale = np.sqrt(target / (power + 1e-15))
        scale = np.clip(scale, 0.1, 10.0)
        
        A_new = A * scale
        a_new = np.real(ifft(A_new, L))[:n]
        
        # Enforce non-negativity and renormalize
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a.copy()
        
        # Mix with current
        a_candidate = (1 - alpha) * a + alpha * a_new
        a_candidate = normalize_nonneg(a_candidate)
        
        score = score_sequence(a_candidate)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate.copy()
            if verbose and it % 100 == 0:
                print(f"        Spec iter {it}: score={score:.6f}")
    
    if verbose:
        print(f"        Spec done: {it+1} iters, best={best_score:.6f}, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def local_search(a_init, time_limit=60, rng=None, verbose=False):
    """Local search with random segment perturbations."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = normalize_nonneg(a_init.copy().astype(np.float64))
    n = len(a)
    
    start = time.monotonic()
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0
    
    while time.monotonic() - start < time_limit:
        # Choose perturbation type
        r = rng.random()
        if r < 0.5:
            # Segment perturbation
            width = max(2, int(rng.integers(n // 100, max(3, n // 5))))
            start_i = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start_i:start_i + width] *= np.exp(rng.normal(0, 0.2, size=width))
        elif r < 0.7:
            # Symmetrization
            candidate = 0.5 * (a + a[::-1])
            candidate *= np.exp(rng.normal(0, 0.1, size=n))
        elif r < 0.85:
            # Single element perturbation
            candidate = a.copy()
            idx = int(rng.integers(0, n))
            candidate[idx] *= np.exp(rng.normal(0, 0.5))
            # Also perturb symmetric position
            candidate[n - 1 - idx] *= np.exp(rng.normal(0, 0.5))
        else:
            # Gaussian smooth perturbation
            candidate = a * np.exp(rng.normal(0, 0.05, size=n))
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score - 1e-12:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 50 == 0:
                print(f"        LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"        LS done: best={best_score:.6f}, accepts={accept_count}, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_centered_cosine(n):
    """Centered raised cosine (symmetric, peaked in middle)."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.maximum(0, 0.5 * (1.0 + np.cos(np.pi * (t - center) / (n / 2.0))))


def initial_gaussian(n, sigma_frac=0.2):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_de_ima(n, rng):
    """Deima-like sequence: high at edges, low in middle."""
    t = np.arange(n, dtype=np.float64)
    # Two Gaussians at edges
    center1 = n * 0.1
    center2 = n * 0.9
    sigma = n * 0.15
    a = np.exp(-0.5 * ((t - center1) / sigma) ** 2) + np.exp(-0.5 * ((t - center2) / sigma) ** 2)
    return a
