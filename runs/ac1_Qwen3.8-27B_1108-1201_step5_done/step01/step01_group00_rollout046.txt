def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = np.sum(a)

        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def smooth_max_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and gradient w.r.t. a.
    Returns (value, grad, conv, weights).
    """
    n = len(a)
    c = fft_convolve(a, a)
    cmax = np.max(c)
    shifted = (c - cmax) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = cmax + tau * np.log(sum_exp)
    a_rev = a[::-1]
    corr = fft_convolve(w, a_rev)
    grad = 2.0 * corr[n - 1: 2 * n - 1]
    return value, grad, c, w

    """
    Multi-phase search:
    1. Nesterov gradient optimization from multiple starts and lengths.
    2. Extended multi-strategy local search on best candidates.
    3. Coordinate descent polish on the final best.
    """
    global rng_global
    rng_global = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if len(start_seq) > 0 and np.sum(start_seq) > 0:
                s = compute_score(start_seq)
                print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
                if s < best_score:
                    best_score = s
                    best_seq = start_seq.copy()
        except Exception as e:
            print(f"Failed to load provided sequence: {e}")

    # Phase 1: Gradient optimization
    print("\n=== Phase 1: Nesterov gradient optimization ===")
    configs = []

    # Use provided sequence length as priority
    if start_seq is not None:
        n_prov = len(start_seq)
        configs.append((n_prov, 'provided', 0.30))  # 30% of gradient time

    # Other lengths to try
    configs.append((256, 'chirp', 0.12))
    configs.append((256, 'random', 0.08))
    configs.append((384, 'chirp', 0.10))
    configs.append((384, 'multitone', 0.08))
    configs.append((512, 'chirp', 0.10))
    configs.append((512, 'random', 0.08))
    configs.append((768, 'chirp', 0.08))
    configs.append((1024, 'causal', 0.06))

    grad_time_total = time_budget * 0.45  # 45% for gradient
    candidates = []  # (sequence, score, n)

    for n, init_type, time_frac in configs:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = grad_time_total * time_frac
        if t_alloc < 5:
            continue

        # Generate initialization
        if init_type == 'provided' and start_seq is not None:
            if len(start_seq) == n:
                a0 = start_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, start_seq)
        elif init_type == 'chirp':
            a0 = make_init_chirp(n, rng_global)
        elif init_type == 'multitone':
            a0 = make_init_multitone(n, rng_global)
        elif init_type == 'causal':
            a0 = make_init_causal(n, rng_global)
        else:
            a0 = make_init_random(n, rng_global)

        a0 = np.maximum(a0, 0.01)
        a0 = a0 / np.sum(a0)

        t0 = time.monotonic()
        a_best, sc_best = nesterov_refine(a0, t_alloc, verbose=(n <= 512))
        elapsed = time.monotonic() - t0

        actual_score = compute_score(a_best)
        print(f"  n={n}, init='{init_type}', score={actual_score:.6f}, time={elapsed:.1f}s")

        candidates.append((a_best, actual_score, n))

        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Sort candidates by score
    candidates.sort(key=lambda x: x[1])
    print(f"\n  Top candidates: " + ", ".join(f"{c[1]:.4f}(n={c[2]})" for c in candidates[:5]))

    # Phase 2: Local search on top candidates
    print(f"\n=== Phase 2: Multi-strategy local search ===")
    ls_time_total = time_budget * 0.35
    n_top = min(4, len(candidates))
    ls_per = ls_time_total / n_top

    for i in range(n_top):
        if time.monotonic() > deadline - 30:
            break
        a_cand, sc_cand, n_cand = candidates[i]
        print(f"  polishing candidate {i}: n={n_cand}, score={sc_cand:.6f}, time={ls_per:.0f}s")

        t0 = time.monotonic()
        a_ls, sc_ls = local_search_multi(a_cand, ls_per, rng_global)
        elapsed = time.monotonic() - t0
        print(f"    local search result: {sc_ls:.6f} (t={elapsed:.0f}s)")

        if sc_ls < best_score:
            best_score = sc_ls
            best_seq = a_ls.copy()
            print(f"    *** New best: {best_score:.6f} ***")

        # Also try on the current best if it's different
        if i == 0 and best_seq is not None and sc_ls > compute_score(best_seq):
            pass  # best_seq already better

    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Coordinate descent polish ===")
    if best_seq is not None:
        cd_time = min(120.0, max(10.0, deadline - time.monotonic()))
        print(f"  CD polish: current best={best_score:.6f}, time={cd_time:.0f}s")
        t0 = time.monotonic()
        a_cd, sc_cd = coordinate_descent_polish(best_seq, cd_time, rng_global)
        elapsed = time.monotonic() - t0
        print(f"  CD result: {sc_cd:.6f} (t={elapsed:.0f}s)")
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 4: Final short local search if time remains
    if time.monotonic() < deadline - 10 and best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            print(f"\n=== Phase 4: Final local search ({remaining:.0f}s) ===")
            a_final, sc_final = local_search_multi(best_seq, remaining, rng_global)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Return best sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
