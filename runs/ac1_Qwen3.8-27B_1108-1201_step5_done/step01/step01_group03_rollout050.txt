Key improvements over previous approach:
1. Adaptive step-size with backtracking line search

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute score = 2*n*max(conv)/(sum)^2, or inf if invalid."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse (2n-1 x n) matrix A where A[k,j] = f[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sp.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def build_conv_matrix_fast(n, f):
    """Faster construction using vectorized operations."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    valid = (row_idx >= 0) & (row_idx < 2 * n - 1)
    i_idx = i_idx[valid]
    j_idx = j_idx[valid]
    row_idx = row_idx[valid]
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit=5.0, top_k=None):
    """Solve max sum(g) s.t. (f*g)[k] <= M (top-k only if specified), g>=0."""
    n = len(f)
    if top_k is not None and top_k < 2 * n - 1:
        conv = np.convolve(f, f)
        top_idx = np.argsort(conv)[::-1][:top_k]
        # Build reduced constraint matrix
        A_full = build_conv_matrix_fast(n, f)
        A = A_full[top_idx, :]
        b = np.full(len(top_idx), M)
    else:
        A = build_conv_matrix_fast(n, f)
        b = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_one_step(f, t, time_limit=5.0, top_k=None):
    """One LP-driven update step. Returns (new_f, improved)."""
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return f, False
    g = solve_lp(f, M, time_limit, top_k)
    if g is None:
        return f, False
    gs = g.sum()
    if gs < 1e-12:
        return f, False
    g = g / gs
    if np.linalg.norm(g - f) < 1e-10:
        return f, False
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-12:
        return f, False
    f_new = f_new / s
    return f_new, True


def adaptive_lp_iterate(f, budget, t_init=0.3, t_min=0.05, t_decay=0.997,
                        lp_time=4.0, top_k=None, perturb_every=15):
    """
    Iterate LP with adaptive step size and periodic perturbations.
    Returns best f and score found.
    """
    f = f.copy()
    t = t_init
    best = f.copy()
    best_sc = fast_score(f)
    stall = 0
    n = len(f)
    
    for it in range(100000):
        if time.monotonic() > budget:
            break
        
        if it % perturb_every == 0 and stall > 5:
            # Inject perturbation
            f = block_perturb(f, np.random.default_rng(it * 31 + 7),
                              low=1, high=max(3, n // 16))
            stall = 0
        
        f_new, moved = lp_one_step(f, t, lp_time, top_k)
        if not moved:
            f_new = block_perturb(f, np.random.default_rng(it * 13 + 3),
                                  low=1, high=max(3, n // 16))
        
        f = f_new
        sc = fast_score(f)
        
        if sc < best_sc - 1e-10:
            if sc < best_sc:
                best_sc = sc
                best = f.copy()
                stall = 0
            t = min(0.5, t * 1.05)  # increase step if improving
        else:
            stall += 1
            t = max(t_min, t * t_decay)
        
        if stall > 40:
            # Larger perturbation to escape
            f = block_perturb(f, np.random.default_rng(9999 + it),
                              low=2, high=max(5, n // 8))
            stall = 0
    
    return best, best_sc


def make_initial(n, rng, method='uniform'):
    """Generate initial non-negative sequence normalized to sum=1."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'exponential':
        a = np.exp(-np.linspace(0, 2, n))
    elif method == 'power':
        x = np.linspace(0, 1, n)
        a = x ** 0.5
    elif method == 'two_humps':
        x = np.arange(n, dtype=float)
        s = max(1, n // 12)
        a = np.exp(-0.5 * ((x - 0.3 * n) / s) ** 2) + \
            np.exp(-0.5 * ((x - 0.7 * n) / s) ** 2)
    elif method == 'trapezoid':
        x = np.linspace(0, 1, n)
        a = np.clip(2 * x - 0.2, 0, 1) * np.clip(1.2 - 2 * x, 0, 1)
    elif method == 'random':
        a = rng.random(n)
    elif method == 'random_perturb_uniform':
        a = 1.0 + rng.normal(0, 0.3, n)
    elif method == 'prime_pattern':
        # Inspired by prime-generating structures
        a = np.ones(n)
        a[::2] = 0.5
        a[::4] = 0.8
        a[::7] = 0.3
    elif method == 'chebyshev':
        # Chebyshev-like distribution (concentrated at edges)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1 + x**2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def block_perturb(f, rng, low=1, high=20):
    """Random block perturbation: multiply a random block by a random factor."""
    n = len(f)
    cand = f.copy()
    w = int(rng.integers(low, max(low + 1, min(high, n))))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    factor = float(np.exp(rng.normal(0, 0.3)))
    cand[start:stop] *= factor
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-12:
        return f
    return cand / s


def coordinate_descent_polish(f, budget, n_trials_per_iter=200):
    """
    Coordinate descent: try moving mass between pairs of coordinates
    to reduce the peak of the convolution.
    """
    f = f.copy()
    n = len(f)
    best = f.copy()
    best_sc = fast_score(f)
    rng = np.random.default_rng(12345)
    
    # Precompute the peak lag
    conv = np.convolve(f, f)
    peak_lag = int(np.argmax(conv))
    
    for it in range(100000):
        if time.monotonic() > budget:
            break
        
        # Recompute peak lag periodically
        if it % 50 == 0:
            conv = np.convolve(f, f)
            peak_lag = int(np.argmax(conv))
        
        # Try random coordinate moves
        improved = False
        for _ in range(n_trials_per_iter):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or f[i] < 1e-8:
                continue
            
            # Try moving epsilon mass from i to j
            # Optimal epsilon found by 1D search
            best_eps = 0.0
            best_local_sc = fast_score(f)
            
            # Try a few epsilon values
            base = f[i] * 0.1
            for frac in [0.01, 0.03, 0.08, 0.15, 0.3, 0.5]:
                eps = base * frac
                if eps > f[i]:
                    continue
                f_try = f.copy()
                f_try[i] -= eps
                f_try[j] += eps
                sc = fast_score(f_try)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_eps = eps
            
            if best_eps > 0:
                f[i] -= best_eps
                f[j] += best_eps
                improved = True
                break  # restart trial loop after a move
        
        if time.monotonic() > budget:
            break
        
        sc = fast_score(f)
        if sc < best_sc:
            best_sc = sc
            best = f.copy()
    
    return best, best_sc


def peak_targeted_optimize(f, budget):
    """
    Target the specific peak lag: try to reduce the contribution
    of the peak lag by adjusting specific coordinates.
    """
    f = f.copy()
    n = len(f)
    best = f.copy()
    best_sc = fast_score(f)
    rng = np.random.default_rng(777)
    
    for it in range(10000):
        if time.monotonic() > budget:
            break
        
        conv = np.convolve(f, f)
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]
        
        # Find which pairs (j, peak_lag-j) contribute most to the peak
        j_min = max(0, peak_lag - n + 1)
        j_max = min(peak_lag, n - 1)
        contributions = []
        for j in range(j_min, j_max + 1):
            c = f[j] * f[peak_lag - j]
            contributions.append((c, j, peak_lag - j))
        contributions.sort(reverse=True)
        
        # Try to reduce the top contributor
        for c, j1, j2 in contributions[:5]:
            if c < 1e-10:
                break
            # Reduce f[j1] or f[j2] and redistribute
            # Option 1: reduce f[j1]
            for frac in [0.05, 0.1, 0.2, 0.35]:
                eps = f[j1] * frac
                if eps < 1e-12:
                    continue
                f_try = f.copy()
                f_try[j1] -= eps
                # Redistribute to a random other position
                k = int(rng.integers(0, n))
                f_try[k] += eps
                f_try = np.maximum(f_try, 0)
                s = f_try.sum()
                f_try = f_try / s
                sc = fast_score(f_try)
                if sc < best_sc:
                    best_sc = sc
                    best = f_try.copy()
                    f = f_try
                    break
            
            if time.monotonic() > budget:
                break
    
    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Tries multiple lengths and initializations,
    runs LP-based optimization with adaptive steps, then polishes
    with coordinate descent and peak-targeted optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    # Try to load previous best as a starting point
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = None
    
    results = []  # List of (score, sequence)
    
    def try_add(f, sc):
        if sc is not None and np.isfinite(sc) and f is not None:
            results.append((sc, f.copy()))
    
    # Evaluate parent if available
    if parent is not None and parent.size > 0 and parent.sum() > 0.01:
        p_sc = fast_score(parent)
        try_add(parent, p_sc)
        print("Parent score: %.6f (n=%d)" % (p_sc, parent.size), flush=True)
    
    # Phase 1: Multi-start LP optimization
    print("\n=== Phase 1: Multi-start LP optimization ===", flush=True)
    
    lengths = [256, 512, 1024, 2048]
    methods = ['uniform', 'cosine', 'sin2', 'exponential', 'power',
               'two_humps', 'trapezoid', 'random', 'random_perturb_uniform',
               'prime_pattern', 'chebyshev']
    
    # Time budget for phase 1
    phase1_budget = min(600.0, 0.35 * float(budget_s))
    n_combos = len(lengths) * len(methods)
    per_combo = phase1_budget / n_combos
    per_combo = max(8.0, min(per_combo, 45.0))
    
    combo_idx = 0
    for n in lengths:
        for m in methods:
            if time.monotonic() > deadline:
                break
            combo_idx += 1
            local_budget = min(deadline, time.monotonic() + per_combo)
            
            start = make_initial(n, rng, m)
            
            # Quick LP iterations
            top_k = None if n <= 512 else min(2 * n - 1, max(50, n))
            c_best, c_sc = adaptive_lp_iterate(
                start, local_budget,
                t_init=0.25, t_min=0.05, t_decay=0.997,
                lp_time=min(3.0, per_combo * 0.4),
                top_k=top_k, perturb_every=10
            )
            try_add(c_best, c_sc)
            
            if combo_idx % 4 == 0:
                best_so_far = min(s for s, _ in results) if results else float('inf')
                print("  [%d/%d] best=%.6f" % (combo_idx, n_combos, best_so_far), flush=True)
    
    results.sort(key=lambda x: x[0])
    if results:
        print("Phase 1 complete. Best: %.6f" % results[0][0], flush=True)
    
    # Phase 2: Deep optimization on top candidates
    print("\n=== Phase 2: Deep LP optimization ===", flush=True)
    
    # Take top 8 candidates
    top_candidates = results[:8]
    remaining_budget = deadline - time.monotonic()
    n_top = len(top_candidates)
    if n_top == 0:
        # Fallback: create a candidate
        top_candidates = [(fast_score(np.ones(512) / 512), np.ones(512) / 512)]
        n_top = 1
    
    per_candidate = max(20.0, remaining_budget * 0.55 / n_top)
    
    best_global = top_candidates[0]
    
    for idx, (sc, f0) in enumerate(top_candidates):
        if time.monotonic() > deadline:
            break
        
        local_budget = min(deadline, time.monotonic() + per_candidate)
        n = len(f0)
        top_k = None if n <= 512 else min(2 * n - 1, max(100, n))
        
        print("  Deep opt [%d/%d] n=%d start_score=%.6f budget=%.0fs" %
              (idx + 1, n_top, n, sc, per_candidate), flush=True)
        
        c_best, c_sc = adaptive_lp_iterate(
            f0, local_budget,
            t_init=0.2, t_min=0.03, t_decay=0.998,
            lp_time=min(5.0, per_candidate * 0.1),
            top_k=top_k, perturb_every=20
        )
        
        if c_sc < best_global[0]:
            best_global = (c_sc, c_best)
            print("    New global best: %.6f" % c_sc, flush=True)
        else:
            print("    Best from this run: %.6f" % c_sc, flush=True)
    
    print("After Phase 2: best=%.6f" % best_global[0], flush=True)
    
    # Phase 3: Peak-targeted optimization + coordinate descent
    print("\n=== Phase 3: Polish ===", flush=True)
    
    remaining = deadline - time.monotonic()
    f_polish = best_global[1].copy()
    sc_polish = best_global[0]
    
    # Peak-targeted phase
    peak_budget = min(remaining * 0.4, 200.0)
    if peak_budget > 10:
        f_pt, sc_pt = peak_targeted_optimize(f_polish, time.monotonic() + peak_budget)
        if sc_pt < sc_polish:
            sc_polish = sc_pt
            f_polish = f_pt
            print("  Peak-targeted improved to: %.6f" % sc_polish, flush=True)
    
    # Coordinate descent phase
    cd_budget = min(deadline - time.monotonic(), remaining * 0.5)
    if cd_budget > 10:
        f_cd, sc_cd = coordinate_descent_polish(f_polish, time.monotonic() + cd_budget)
        if sc_cd < sc_polish:
            sc_polish = sc_cd
            f_polish = f_cd
            print("  Coord descent improved to: %.6f" % sc_polish, flush=True)
    
    # Phase 4: Final short LP on the polished result
    final_lp_budget = max(5.0, (deadline - time.monotonic()) * 0.5)
    if final_lp_budget > 5:
        n = len(f_polish)
        top_k = None if n <= 512 else min(2 * n - 1, max(100, n))
        f_final, sc_final = adaptive_lp_iterate(
            f_polish, time.monotonic() + final_lp_budget,
            t_init=0.1, t_min=0.02, t_decay=0.999,
            lp_time=min(3.0, final_lp_budget * 0.1),
            top_k=top_k, perturb_every=30
        )
        if sc_final < sc_polish:
            sc_polish = sc_final
            f_polish = f_final
            print("  Final LP improved to: %.6f" % sc_polish, flush=True)
    
    # Ensure valid output
    f_polish = np.clip(f_polish, 0.0, 1000.0)
    if f_polish.size == 0 or f_polish.sum() < 0.01:
        f_polish = np.ones(512)
    
    # Final verification
    final_score = fast_score(f_polish)
    print("\n=== FINAL ===", flush=True)
    print("Score: %.6f (n=%d)" % (final_score, f_polish.size), flush=True)
    
    return [float(x) for x in f_polish]
