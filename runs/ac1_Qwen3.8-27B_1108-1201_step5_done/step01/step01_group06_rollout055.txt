def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def solve_lp(f, active_lags, M):
    n = len(f)
    # Build constraint matrix A: (len(active_lags) x n)
    # A[k, i] = f[lag_k - i] if 0 <= lag_k - i < n, else 0
    A = np.zeros((len(active_lags), n))
    for row, lag in enumerate(active_lags):
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        A[row, lo:hi+1] = f[lag - hi:lag - lo + 1][::-1]

    return None

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding.

Algorithm:
1. Normalize sequence to sum = sqrt(2n), so objective = max of self-convolution.
2. Iteratively find a direction g by solving an LP that maximizes sum(g)
   subject to (f*g)_k <= M for the top-k most active convolution lags.
3. Blend f towards g with a line-searched step size.
4. Multiple starting points + final polishing phase.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def convolve_self_fft(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_convolve_fft(f, g):
    """Cross-convolution f*g via FFT (same length n)."""
    n = len(f)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    c = np.real(ifft(F * G))
    return c[:size]


def build_lp_matrix(f, active_lags, n):
    """Build the constraint matrix for the direction LP.
    A[row, i] = f[lag-row - i] if 0 <= lag-row - i < n else 0.
    """
    k = len(active_lags)
    A = np.zeros((k, n), dtype=np.float64)
    for row in range(k):
        lag = int(active_lags[row])
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            A[row, lo:hi + 1] = f[lag - lo: lag - hi - 1: -1] if (lag - lo) > (lag - hi) else f[lag - lo: lag - hi + 1][::-1]
    return A


def solve_direction(f, M, active_lags):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for active lags, g >= 0.
    
    Returns the optimal g, or None if infeasible/fails.
    """
    n = len(f)
    A = build_lp_matrix(f, active_lags, n)
    b = np.full(len(active_lags), M, dtype=np.float64)
    
    try:
        res = linprog(-np.ones(n, dtype=np.float64),
                      A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception as e:
        print(f"  LP error: {e}")
    return None


def normalize_seq(a, n):
    """Clamp to [0, 1000] and normalize to sum = sqrt(2n)."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s < 1e-15:
        return np.full(n, S / n)
    return a * (S / s)


def objective_from_maxc(maxc, n):
    """Compute the evaluation score from the max convolution value (normalized)."""
    S = np.sqrt(2.0 * float(n))
    return 2.0 * float(n) * maxc / (S * S)  # = maxc since S^2 = 2n


def lp_optimize(a_init, deadline, rng, min_time=5.0):
    """Run the LP-based optimization until deadline. Returns (best_maxc, best_a)."""
    a = normalize_seq(a_init.astype(np.float64), len(a_init))
    n = len(a)
    S = np.sqrt(2.0 * n)
    
    best_maxc = convolve_self_fft(a).max()
    best_a = a.copy()
    
    k = 80  # number of active constraints
    blend = 0.25
    
    it = 0
    no_improve = 0
    start_time = time.monotonic()
    
    while time.monotonic() < deadline - min_time:
        it += 1
        c = convolve_self_fft(a)
        cur_max = c.max()
        
        if cur_max < best_maxc - 1e-10:
            best_maxc = cur_max
            best_a = a.copy()
            no_improve = 0
            if it % 50 == 0:
                score = 2.0 * n * best_maxc / (S * S)
                print(f"    it={it} maxc={best_maxc:.8f} score={score:.6f}")
        else:
            no_improve += 1
        
        # Adapt k: increase over time
        k = min(200, 50 + it // 10)
        
        # Find active lags
        top_k = np.argsort(c)[::-1][:k]
        
        # Solve LP
        g = solve_direction(a, cur_max, top_k)
        
        if g is not None and g.sum() > 1e-10:
            # Rescale g to have sum = S
            g = g * (S / g.sum())
            
            # Line search on blend parameter
            best_t = 0.1
            best_t_max = cur_max
            for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try * (S / s)
                try_max = convolve_self_fft(a_try).max()
                if try_max < best_t_max:
                    best_t_max = try_max
                    best_t = t
            
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.full(n, S / n)
            else:
                a = a * (S / s)
        else:
            # LP failed, try a small random perturbation
            a = a + rng.normal(0, 0.001 * S / n, n)
            a = normalize_seq(a, n)
        
        # Every 200 iterations, try a larger perturbation to escape local minima
        if it % 200 == 0 and rng.random() < 0.3:
            a = a + rng.normal(0, 0.01 * S / n, n)
            a = normalize_seq(a, n)
        
        # If stuck for too long, increase perturbation
        if no_improve > 500:
            a = a + rng.normal(0, 0.05 * S / n, n)
            a = normalize_seq(a, n)
            no_improve = 0
        
        if it % 100 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    it={it} elapsed={elapsed:.1f}s cur_max={cur_max:.6f} best={best_maxc:.6f} k={k}")
    
    return best_maxc, best_a


def gradient_polish(a, deadline, rng):
    """Polish with Adam gradient descent."""
    a = normalize_seq(a.astype(np.float64), len(a))
    n = len(a)
    S = np.sqrt(2.0 * n)
    
    best_maxc = convolve_self_fft(a).max()
    best_a = a.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1 * S / n
    
    while time.monotonic() < deadline - 5:
        t += 1
        c = convolve_self_fft(a)
        cur_max = c.max()
        
        if cur_max < best_maxc - 1e-10:
            best_maxc = cur_max
            best_a = a.copy()
            if t % 100 == 0:
                print(f"    polish it={t} maxc={best_maxc:.8f}")
        
        # Subgradient: weighted average of gradients at top lags
        top_k = np.argsort(c)[::-1][:20]
        top_vals = c[top_k]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights = top_vals / total
        
        grad = np.zeros(n)
        for w_val, lag in zip(weights, top_k):
            lag = int(lag)
            lo = max(0, lag - n + 1)
            hi = min(n - 1, lag)
            if lo <= hi:
                # d(dl_k)/da[i] = 2*a[lag-i] for valid i
                # For i in [lo, hi]: grad[i] += w * 2 * a[lag - i]
                grad[lo:hi + 1] += w_val * 2.0 * a[lag - lo: lag - hi - 1: -1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)
        
        lr = 0.1 * S / n * (0.5 ** (t / 500.0))
    
    return best_maxc, best_a

The cross-convolution (f*g)_k = sum_i f_i * g_{k-i} for k = 0, ..., 2n-2.

But we need: for each active lag k, (f*g)_k = sum_i f_i * g_{k-i} <= M.

The variable is g (vector of length n). For a fixed lag k:
(f*g)_k = sum_{i: 0<=k-i<n} f_i * g_{k-i}

Let j = k-i, so i = k-j. The sum becomes:
sum_{j: 0<=k-j<n, 0<=j<n} f_{k-j} * g_j
= sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} f_{k-j} * g_j

So the coefficient of g_j in constraint k is f_{k-j} (for valid j).

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding with
cutting-plane history of active lags, combined with gradient polishing.

Key ideas:
1. Normalize to sum=sqrt(2n) so objective equals max of self-convolution.
2. At each step, solve an LP: max sum(g) s.t. (f*g)_k <= M for historically
   active lags, g >= 0. This gives a direction that "spreads out" the convolution.
3. Maintain a growing set of active lags (cutting-plane) to prevent seesaw.
4. Polish with Adam subgradient descent at the end.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def convolve_self_fft(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def build_lp_matrix(f, active_lags, n):
    """Build constraint matrix A where A[row, j] = f[lag-row - j] if valid."""
    k = len(active_lags)
    A = np.zeros((k, n), dtype=np.float64)
    for row in range(k):
        lag = int(active_lags[row])
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            A[row, js] = f[lag - js]
    return A


def solve_direction_lp(f, M, active_lags, time_limit=10.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for active lags, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f, active_lags, n)
    b = np.full(len(active_lags), M, dtype=np.float64)
    try:
        res = linprog(-np.ones(n, dtype=np.float64),
                      A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def normalize_seq(a, n):
    """Clamp to [0,1000] and normalize to sum = sqrt(2n)."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s < 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def subgradient_topk(a, c, top_k_indices):
    """Compute weighted subgradient at top-k lags."""
    n = len(a)
    top_vals = c[top_k_indices]
    total = top_vals.sum()
    if total < 1e-15:
        return np.zeros(n)
    weights = top_vals / total
    grad = np.zeros(n, dtype=np.float64)
    for w_val, lag in zip(weights, top_k_indices):
        lag = int(lag)
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            grad[js] += w_val * 2.0 * a[lag - js]
    return grad


def lp_optimize(a_init, deadline, rng, k_initial=60, k_max=300, max_history=400):
    """Main LP-based optimization loop."""
    a = normalize_seq(a_init.astype(np.float64), len(a_init))
    n = len(a)
    S = np.sqrt(2.0 * n)

    c = convolve_self_fft(a)
    best_maxc = c.max()
    best_a = a.copy()

    active_history = set()
    it = 0
    no_improve = 0
    start_time = time.monotonic()
    last_print = 0.0

    while time.monotonic() < deadline - 5.0:
        it += 1
        c = convolve_self_fft(a)
        cur_max = c.max()

        if cur_max < best_maxc - 1e-9:
            best_maxc = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Print progress
        now = time.monotonic()
        if now - last_print > 5.0 or it <= 5:
            score = 2.0 * n * best_maxc / (S * S)
            print(f"    it={it} t={now-start_time:.1f}s maxc={cur_max:.8f} best={best_maxc:.8f} score={score:.6f} |hist|={len(active_history)}")
            last_print = now

        # Update active history with current top lags
        n_top = min(80, 2n - 1)
        top_idx = np.argsort(c)[::-1][:n_top]
        for lag in top_idx:
            active_history.add(int(lag))

        # Select constraints: use most active from history
        if len(active_history) > max_history:
            hist_list = list(active_history)
            hist_list.sort(key=lambda l: c[l], reverse=True)
            active_lags = np.array(hist_list[:max_history], dtype=np.int64)
        else:
            active_lags = np.array(sorted(active_history), dtype=np.int64)

        # Adaptive k: start small, grow over time
        k_use = min(len(active_lags), k_initial + it // 5)
        if k_use > k_max:
            k_use = k_max
        # Use the most active lags from our selection
        active_lags_used = active_lags[:k_use]

        # Solve LP
        g = solve_direction_lp(a, cur_max, active_lags_used, time_limit=8.0)

        if g is not None and g.sum() > 1e-10:
            g = g * (S / g.sum())
            # Line search on blend parameter
            best_t = 0.15
            best_try_val = cur_max
            for t_candidate in [0.4, 0.25, 0.15, 0.08, 0.04]:
                a_try = (1.0 - t_candidate) * a + t_candidate * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * (S / s_try)
                try_max = convolve_self_fft(a_try).max()
                if try_max < best_try_val:
                    best_try_val = try_max
                    best_t = t_candidate

            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.full(n, S / n)
            else:
                a = a * (S / s)
        else:
            # LP failed; small perturbation
            a = a + rng.normal(0, 0.003 * S / n, n)
            a = normalize_seq(a, n)

        # Periodic larger perturbation to escape local minima
        if no_improve > 300 and it % 100 == 0:
            a = a + rng.normal(0, 0.02 * S / n, n)
            a = normalize_seq(a, n)
            no_improve = 0

    return best_maxc, best_a


def gradient_polish(a_init, deadline, rng):
    """Polish using Adam with subgradient."""
    a = normalize_seq(a_init.astype(np.float64), len(a_init))
    n = len(a)
    S = np.sqrt(2.0 * n)

    c = convolve_self_fft(a)
    best_maxc = c.max()
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 0.3 * S / n

    start_time = time.monotonic()
    last_print = 0.0

    while time.monotonic() < deadline - 5.0:
        t += 1
        c = convolve_self_fft(a)
        cur_max = c.max()

        if cur_max < best_maxc - 1e-9:
            best_maxc = cur_max
            best_a = a.copy()

        now = time.monotonic()
        if now - last_print > 5.0 or t <= 5:
            score = 2.0 * n * best_maxc / (S * S)
            print(f"    grad it={t} t={now-start_time:.1f}s maxc={cur_max:.8f} best={best_maxc:.8f} score={score:.6f}")
            last_print = now

        # Subgradient at top lags
        top_k = np.argsort(c)[::-1][:30]
        grad = subgradient_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = base_lr * (0.5 ** (float(t) / 1000.0))

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

    return best_maxc, best_a
