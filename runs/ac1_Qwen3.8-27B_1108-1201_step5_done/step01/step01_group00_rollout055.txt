LP-guided iterative optimization for minimizing the peak of the linear
autocorrelation of non-negative step functions. Uses linear programming
to find descent directions that specifically target the convolution peak,
combined with line search, multiple initializations, and local search polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
import scipy.fft as sfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return sfft.irfft(sfft.rfft(a, size) * sfft.rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Compute the score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_lp_matrices(a, k_star, alpha):
    """Build matrices for the LP direction-finding problem."""
    n = len(a)
    num_lags = 2 * n - 1
    
    # Objective: minimize (a*d)[k_star]
    c_obj = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            c_obj[i] = a[j]

    A_rows = []
    b_vals = []
    for k in range(num_lags):
        if k == k_star:

        for i in range(n):
            j = k - i
            if 0 <= j < n:

    b_ub = np.array(b_vals)

    return None

                    stall_count = 0
                else:
                    stall_count += 1
                    if stall_count > 50:
                        break
            else:
                break
            continue
        
        # Line search
        best_t = 0.0
        best_new_peak = peak
        t_candidates = [0.1, 0.25, 0.5, 0.75, 1.0]
        # Add a few more for finer search
        if best_t == 0:
            t_candidates = [0.05] + t_candidates
        
        for t in t_candidates:
            a_trial = a + t * d
            if np.any(a_trial < 0):
                continue
            a_trial = a_trial / np.sum(a_trial)
            c_trial = fft_convolve(a_trial, a_trial)
            new_peak = np.max(c_trial)
            if new_peak < best_new_peak:
                best_new_peak = new_peak
                best_t = t
        
        if best_t == 0:
            stall_count += 1
            if stall_count > 100:
                break
            continue
        
        a = a + best_t * d
        a = np.clip(a, 0, None)
        a = a / np.sum(a)
        
        score = 2.0 * n * best_new_peak  # Since sum(a) = 1
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        # else: stall_count += 1 (not incrementing here to be more patient)
        
        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: peak={peak:.10f}, new_peak={best_new_peak:.10f}, score={score:.8f}, best={best_score:.8f}")
        
        # Adaptive alpha: if progress is slow, increase alpha
        if it % 500 == 499:
            alpha_base *= 1.5
    
    return best_a, best_score


def gradient_optimize(a_init, time_limit, seed=42):
    """Gradient-based optimization using softmax parameterization."""
    n = len(a_init)
    a_init = np.clip(a_init, 0.01, None)
    theta = np.log(a_init / np.sum(a_init))
    rng = np.random.default_rng(seed)
    
    a = theta.copy()
    momentum = np.zeros(n)
    # velocity
    v = np.zeros(n)
    m = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    lr = 1.0
    tau = 50.0 / n  # soft-max temperature
    
    best_a = a_init.copy()
    best_score = local_score(a_init)
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Convert theta to a
        x_max = np.max(theta)
        a = np.exp(theta - x_max)
        a_sum = np.sum(a)
        a = a / a_sum  # normalize to sum 1
        
        # Compute soft-max of convolution
        c = fft_convolve(a, a)
        c_max = np.max(c)
        shifted = c / tau  # for softmax
        shifted = shifted - np.max(shifted)
        w = np.exp(shifted)
        w = w / np.sum(w)
        
        # Gradient of soft_max(c, tau) w.r.t. a: 2 * conv(a, w) (approximately)
        # Actually: d/d a[j] of c[k] = a[k-j] + a[j] (where valid)
        # So grad_soft_max[j] = sum_k w[k] * d c[k] / d a[j]
        # = sum_k w[k] * (a[k-j] (if valid) + (a[j] if k-j is valid...))
        # More simply: grad = 2 * conv(a_reversed, w) adjusted
        # Let's use: grad[j] = 2 * sum_k w[k] * a[k-j] (valid k,j)
        # This is 2 * (w * a_reversed)[j + n - 1]... 
        # Actually let me just compute it directly for clarity
        
        # grad_a[j] = sum_k w[k] * (dc[k]/da[j])
        # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j] (if 0<=k-j<n, by the other term)
        # Wait, c[k] = sum_i a[i]*a[k-i]
        # dc[k]/da[j] = a[k-j] (term where i=j) + a[j] (term where k-i=j, i.e., i=k-j)
        # So dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] (if 0<=k-j<n)
        # Otherwise 0 (if k-j not in [0,n))
        
        # So grad_a[j] = sum_k w[k] * 2*a[k-j] (where 0<=k-j<n)
        # = 2 * sum_k w[k] * a[k-j]
        # = 2 * (w * a)[j] ... no, that's not right either.
        # (w*a)[j] = sum_i w[i]*a[j-i]
        # We want: sum_k w[k]*a[k-j] = sum_i w[i]*a[i-j] where i=k
        # Let m = i-j: sum_m w[m+j]*a[m] = (w_shifted * a) but that's cross-correlation
        
        # Simplest: compute via FFT
        # grad_a[j] = 2 * sum_k w[k]*a[k-j] (valid)
        # = 2 * sum_k w[k-j]*a[k] (substituting k->k+j)
        # Hmm, let me just use the correlation formula:
        # grad_a = 2 * ifft(fft(a) * conj(fft(w)))... no
        
        # Actually, the simplest correct formula:
        # c[k] = sum_j a[j]*a[k-j] (linear conv)
        # dc[k]/da[j] = a[k-j] + a[j] * (1 if k-2j is valid... no)
        # Let me be very careful.
        # c[k] = sum_i a[i]*a[k-i]
        # dc[k]/da[j]: the terms involving a[j] are:
        #   i=j: a[j]*a[k-j] (if 0<=k-j<n)
        #   k-i=j, i=k-j: a[k-j]*a[j] (if 0<=k-j<n)
        # So dc[k]/da[j] = a[k-j] + (a[k-j] if k != 2j, else a[j])
        # For k != 2j: dc[k]/da[j] = 2*a[k-j] (if 0<=k-j<n)
        # For k = 2j: dc[k]/da[j] = a[j] (just one term, since i=j=k-j)
        # 
        # For simplicity, let's use the approximation dc[k]/da[j] ≈ 2*a[k-j] (ignoring the k=2j case)
        # This is exact when n is large relative to the number of actual terms.
        
        # Using FFT: grad_a[j] ≈ 2 * sum_k w[k]*a[k-j]
        # Let's compute h[j] = sum_k w[k]*a[k-j] = (w * a_reversed)[j+n-1]... 
        # Actually: h[j] = sum_k w[k]*a[k-j]. Let m = k-j, then k = m+j.
        # h[j] = sum_m w[m+j]*a[m]
        # This is the cross-correlation of w and a at lag j.
        # = (w_rev * a)[j + (len(w)-1)] where w_rev = w[::-1]
        # = fft_convolve(w[::-1], a)[j + len(w) - 1]
        
        w_rev = w[::-1]
        corr = fft_convolve(w_rev, a)
        # corr has length len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
        # We want corr[j + len(w) - 1] = corr[j + 2n - 2] for j = 0, ..., n-1
        # Wait, that's not right. Let me think again.
        # h[j] = sum_m w[m+j]*a[m] for m such that 0<=m<n and 0<=m+j<2n-1
        # This is the correlation of w and a.
        # (w * a)[k] = sum_m w[m]*a[k-m]
        # h[j] = sum_m w[m+j]*a[m] = sum_m w[m']*a[m'-j] where m' = m+j
        #     = (w * a)[j]... no
        # Let me use: h[j] = sum_m a[m]*w[m+j]
        # (a * w)[k] = sum_m a[m]*w[k-m]
        # h[j] = sum_m a[m]*w[m+j] = (a * w)[?] 
        # Set k = m+j for each m? No, k is fixed.
        # (a*w)[j] = sum_m a[m]*w[j-m]
        # h[j] = sum_m a[m]*w[m+j]
        # These are different. h[j] = (a * w_rev)[j + len(w) - 1] where w_rev = w[::-1]
        # Because (a * w_rev)[k] = sum_m a[m]*w_rev[k-m] = sum_m a[m]*w[len(w_rev)-1-(k-m)]
        #     = sum_m a[m]*w[2n-2-(k-m)] = sum_m a[m]*w[2n-2-k+m]
        # Set k = j + 2n-2: sum_m a[m]*w[2n-2-(j+2n-2)+m] = sum_m a[m]*w[m-j]
        # Hmm, that gives w[m-j] not w[m+j].
        
        # OK let me just compute it directly for small n, or use a different approach.
        # For large n, the gradient can be approximated as:
        # grad_a ≈ 2 * (a * w) (circular convolution, since n is large)
        # This is because the "valid" range of the linear convolution covers most of the circular one.
        
        # Actually, the cleanest approach: use the full linear convolution formula.
        # grad_a[j] = 2 * sum_{k: 0<=k<2n-1, 0<=k-j<n} w[k] * a[k-j]
        # = 2 * sum_{k: max(j,0) <= k < min(2n-1, j+n)} w[k] * a[k-j]
        # = 2 * sum_{m: 0<=m<n} w[m+j] * a[m] (where m = k-j, valid when 0<=m+j<2n-1)
        
        # This is a valid cross-correlation. Let me compute it via FFT:
        # Define w_padded of length 3n-2 (to accommodate the full range)
        # Actually, let me just use np.correlate or compute via FFT.
        
        # For efficiency, let's use the approximation: treat it as circular convolution
        # grad_a[j] ≈ 2 * ifft(fft(a, 4n) * fft(w, 4n)) [first n elements]
        # This is an approximation that works well when the "wrap-around" effects are small.
        
        size = 4 * n
        fa = sfft.rfft(a, size)
        fw = sfft.rfft(w, size)
        grad_a = 2.0 * sfft.irfft(fa * fw, size)[:n]
        
        # Convert gradient from a-space to theta-space
        # a = softmax(theta) (approximately, since we normalize)
        # da/dtheta[j] = a[j] * (delta[j,k] - a[k])
        # grad_theta[k] = sum_j grad_a[j] * da[j]/dtheta[k]
        # = sum_j grad_a[j] * a[j] * (delta[j,k] - a[k])
        # = a[k] * (grad_a[k] - sum_j grad_a[j]*a[j])
        
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        
        # Adam step
        step_count += 1
        v = beta1 * v + (1 - beta1) * grad_theta
        m = beta2 * m + (1 - beta2) * (grad_theta ** 2)
        v_hat = v / (1 - beta1 ** step_count)
        m_hat = m / (1 - beta2 ** step_count)
        theta = theta - lr * v_hat / (np.sqrt(m_hat) + eps)
        
        # Check score
        a_new = np.exp(theta - np.max(theta))
        a_new = a_new / np.sum(a_new)
        score = 2.0 * n * np.max(fft_convolve(a_new, a_new))
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Anneal temperature
        tau *= 0.9995
        lr *= 0.9999
        
        if step_count % 1000 == 0:
            print(f"      Grad iter {step_count}: score={score:.8f}, best={best_score:.8f}, tau={tau:.2e}")
    
    return best_a, best_score
