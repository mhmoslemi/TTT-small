import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def compute_subgradient(w, a):
    """
    Compute sg[j] = sum_k w[k] * a[k-j] for j=0..n-1.
    This is the cross-correlation of w (length 2n-1) and a (length n).
    """
    n = len(a)
    a_flipped = a[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flipped, fft_size)
    conv = np.fft.irfft(W * AF, fft_size)[:size]
    return conv[n-1:n-1+n]

def solve_lp_direction(a, M):
    """
    Solve: maximize sum(b) subject to b >= 0, (a*b)[k] <= M for all k.
    Returns the optimal b (or None if infeasible/solver fails).
    """
    n = len(a)
    # Constraint matrix: each row k gives the coefficients of b in (a*b)[k]
    # (a*b)[k] = sum_j a[j] * b[k-j] (valid when 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k)
    # For row k, the coefficient of b[j] is a[k-j] if k-n+1 <= j <= k and 0 <= j <= n-1, else 0
    
    # This is a Toeplitz-like matrix. For n up to ~2000, building it explicitly is feasible.
    # But for very large n, it could be slow. Let's limit n for LP.
    
    if n > 3000:
        return None
    
    # Build constraint matrix A_ub (2n-1 rows, n columns)
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n and 0 <= j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A_ub[k, j_start:j_end+1] = a[k-j_end : k-j_start+1][::-1]
    
    b_ub = np.full(2*n-1, M)
    
    # Objective: maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def greedy_lp_direction(a, M, n_constraints=None):
    """
    Approximate LP: only use the top constraints (where a*a is largest).
    This is much faster for large n.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    if n_constraints is None:
        n_constraints = min(2*n-1, max(n, 2*n//2))
    
    # Select top constraints
    top_indices = np.argsort(conv)[::-1][:n_constraints]
    
    # Build constraint matrix for selected constraints only
    A_ub = np.zeros((len(top_indices), n))
    for i, k in enumerate(top_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A_ub[i, j_start:j_end+1] = a[k-j_end : k-j_start+1][::-1]
    
    b_ub = conv[top_indices].copy()
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def multiplicative_descent(a0, time_limit, t_param=40.0, eta0=0.3, use_momentum=True):
    """
    Fast multiplicative gradient descent with FFT-based subgradient.
    Returns (best_a, best_obj) where obj = 2*n*max(conv) for sum(a)=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    t = t_param
    patience = 0
    max_patience = 50
    t0 = time.monotonic()
    
    # Momentum variables
    v = np.zeros(n) if use_momentum else None
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Softmax weights
        scaled = t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Subgradient via FFT
        sg = compute_subgradient(w, a)
        
        # Center the subgradient (removes the constant component, 
        # which doesn't affect the simplex constraint)
        sg_centered = sg - sg.mean()
        
        # Multiplicative update
        if use_momentum:
            v = 0.9 * v + 0.1 * (-eta * sg_centered)
            log_factor = v
        else:
            log_factor = -eta * sg_centered
        
        # Normalize for numerical stability
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum
        
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        iteration += 1
        
        if patience >= max_patience:
            patience = 0
            eta *= 0.5
            if eta < 0.01:
                # Try a different temperature
                t *= 2.0
                eta = 0.2
                max_patience = 30
            t = min(t, 500.0)
    
    return best_a, best_obj
