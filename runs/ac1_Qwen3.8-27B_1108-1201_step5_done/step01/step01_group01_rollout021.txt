def _convolution(a):

def _subgrad_topm(a, b, top_m=20):
    """Compute subgradient of max_k b_k, using top-m peaks."""
    n = len(a)
    b_max = np.max(b)
    # Find top m indices
    if top_m >= len(b):
        idx = np.arange(len(b))
        wts = np.ones(len(idx)) / len(idx)
    else:
        idx = np.argpartition(b, -top_m)[-top_m:]
        # Weight by how close to max
        b_sel = b[idx]
        # Use softmax-like weights
        scale = b_max - b_sel
        wts = np.exp(-10 * scale / max(b_max, 1e-15))
        wts /= wts.sum()
    
    grad = np.zeros(n)

            continue
        w = wts[i]

        hi = min(n - 1, k)
        if lo <= hi:

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Hybrid FISTA + Mirror Descent optimizer for minimizing the normalized
autocorrelation peak of a non-negative sequence. Uses adaptive restart,
multiple diverse starting points, and coordinate descent polishing.
"""

import numpy as np
import time


def _score_normalized(a):
    """Compute 2*n*max_b for a sequence with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _subgrad_topm(a, b, top_m=30):
    """Compute weighted subgradient of max_k b_k using top-m peaks."""
    n = len(a)
    b_max = np.max(b)
    if top_m >= len(b):
        idx = np.arange(len(b))
        wts = np.ones(len(idx)) / len(idx)
    else:
        idx = np.argpartition(b, -top_m)[-top_m:]
        b_sel = b[idx]
        # Soft weighting: emphasize peaks closest to max
        margin = b_max - b_sel
        wts = np.exp(-20.0 * margin / max(b_max, 1e-15))
        wts /= wts.sum()

    grad = np.zeros(n)
    for i in range(len(idx)):
        w = wts[i]
        if w < 1e-10:
            continue
        k = idx[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            grad[j_range] += 2.0 * w * a[k - j_range]
    return grad


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fista_adaptive(a0, deadline, top_m=30, eta_init=None, use_restart=True):
    """
    FISTA with adaptive restart (O'Donoghue-Candes) to minimize max autocorrelation.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n

    if eta_init is None:
        eta_init = 0.5 / np.sqrt(n)

    best = a.copy()
    best_score = _score_normalized(best)

    t = 1.0
    x_old = a.copy()
    step = eta_init
    no_improve = 0
    patience = 300

    while time.monotonic() < deadline:
        # FISTA momentum with adaptive restart
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = a + momentum * (a - x_old)

        # Adaptive restart: if y makes things worse, reset momentum
        if use_restart:
            y_clipped = np.maximum(y, 0.0)
            s_y = np.sum(y_clipped)
            if s_y > 1e-10:
                y_norm = y_clipped / s_y
                b_y = np.convolve(y_norm, y_norm)
                if np.max(b_y) > _score_normalized(a) / (2.0 * n) * 1.05:
                    y = a.copy()
                    momentum = 0.0
                    t = 1.0
            else:
                y = a.copy()
                t = 1.0
                x_old = a.copy()
                continue
        else:
            y_clipped = np.maximum(y, 0.0)
            s_y = np.sum(y_clipped)
            if s_y > 1e-10:
                y_norm = y_clipped / s_y
            else:
                y_norm = a.copy()
                t_new = t

        # Compute subgradient
        b = np.convolve(y_norm, y_norm)
        grad = _subgrad_topm(y_norm, b, top_m)
        grad -= np.mean(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Update
        x_new = y_norm - step * grad / gnorm
        x_new = _project_simplex(x_new, 1.0)

        # Evaluate
        cur_score = _score_normalized(x_new)
        if cur_score < best_score - 1e-12:
            best = x_new.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            step *= 0.92
            no_improve = 0
            if step < 1e-14:
                break
            # Occasionally restart from best
            if no_improve == 0 and rng_fista_count[0] % 500 == 0:
                x_old = best.copy()
                a = best.copy()
                t = 1.0

        x_old = a.copy()
        a = x_new
        t = t_new
        rng_fista_count[0] += 1

    return best, best_score


def _mirror_descent(a0, deadline, top_m=30, eta_init=None):
    """Mirror descent (multiplicative weights) for the autocorrelation problem."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n

    if eta_init is None:
        eta_init = 0.5 / np.sqrt(n)

    best = a.copy()
    best_score = _score_normalized(best)
    step = eta_init
    no_improve = 0

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        grad = _subgrad_topm(a, b, top_m)
        grad -= np.mean(grad)

        # Multiplicative update: a_i <- a_i * exp(-eta * grad_i)
        log_scale = -step * grad
        # Clip to avoid overflow
        log_scale = np.clip(log_scale, -50, 50)
        a_new = a * np.exp(log_scale)
        s_new = np.sum(a_new)
        if s_new < 1e-10 or not np.isfinite(s_new):
            step *= 0.5
            if step < 1e-14:
                break
            continue
        a = a_new / s_new

        cur_score = _score_normalized(a)
        if cur_score < best_score - 1e-12:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            step *= 0.95
            no_improve = 0
            if step < 1e-14:
                break

    return best, best_score


def _fista_constant_step(a0, deadline, top_m=50, eta=0.1):
    """FISTA with constant (smaller) step for polishing."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s

    best = a.copy()
    best_score = _score_normalized(best)
    t = 1.0
    x_old = a.copy()
    step = eta
    no_improve = 0

    while time.monotonic() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t - 1.0) / t_new * (a - x_old)
        y = np.maximum(y, 0.0)
        s_y = np.sum(y)
        if s_y < 1e-10:
            y = a.copy()
            s_y = 1.0
        y = y / s_y

        b = np.convolve(y, y)
        grad = _subgrad_topm(y, b, top_m)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = y - step * grad / gnorm
        x_new = _project_simplex(x_new, 1.0)

        cur_score = _score_normalized(x_new)
        if cur_score < best_score - 1e-12:
            best = x_new.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                step *= 0.9
                no_improve = 0
                if step < 1e-14:
                    break

        x_old = a.copy()
        a = x_new
        t = t_new

    return best, best_score


def _make_anti_palindromic(n, eps=0.7, smoothness=3):
    """
    Create a sequence that is high in the first portion and low in the second,
    with smooth transition. This reduces the center convolution peak.
    """
    i = np.arange(n, dtype=np.float64)
    # Smooth step from 1+eps to 1-eps
    t = i / (n - 1)
    # Use a smoothstep-like function
    t_smooth = t * t * (3 - 2 * t)  # Hermite smoothstep
    a = (1 + eps) - 2 * eps * t_smooth
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def _make_gaussian_offset(n, offset=0.3, width=0.25):
    """Create a sequence with a Gaussian bump offset from center."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    center = 0.5 - offset
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0.001)
    s = np.sum(a)
    return a / s


def _make_two_level_smooth(n, eps=0.5, transition=0.1):
    """Two-level with smooth transition region."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    # Smooth transition from (1+eps) to (1-eps) around t=0.5
    transition_width = transition
    t_centered = (t - 0.5) / transition_width
    # Smooth step
    smooth = 0.5 * (1 + np.tanh(t_centered * 3))
    a = (1 + eps) - 2 * eps * smooth
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def _local_search(a0, deadline, rng, n_trials=2000):
    """
    Local search: try random perturbations of the current best and keep improvements.
    Uses structured perturbations (frequency-domain).
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    best = a.copy()
    best_score = _score_normalized(best)

    trial = 0
    while time.monotonic() < deadline and trial < n_trials:
        trial += 1
        # Generate a random low-frequency perturbation
        n_freqs = rng.integers(2, 8)
        freqs = rng.integers(1, max(2, n // 10), size=n_freqs)
        phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
        amps = rng.uniform(-1, 1, size=n_freqs)

        noise = np.zeros(n)
        for f, ph, amp in zip(freqs, phases, amps):
            noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + ph)
        noise /= (np.max(np.abs(noise)) + 1e-15)

        scale = rng.uniform(0.02, 0.15)
        a_try = a + scale * noise
        a_try = _project_simplex(a_try, 1.0)

        cur = _score_normalized(a_try)
        if cur < best_score - 1e-12:
            best = a_try.copy()
            best_score = cur
            a = a_try.copy()

    return best, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimization: FISTA with adaptive restart, mirror descent,
    diverse starting points, and local search to minimize the normalized
    autocorrelation peak of a non-negative sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Normalize parent
    a0 = parent.copy()
    S0 = np.sum(a0)
    if S0 > 1e-10:
        a0 = a0 / S0
    else:
        a0 = np.ones(n) / n

    initial_score = _score_normalized(a0)
    print(f"n={n}, Initial score: {initial_score:.8f}", flush=True)

    global_best = a0.copy()
    global_best_score = initial_score

    # === Phase 1: FISTA from parent (500s) ===
    print("Phase 1: FISTA from parent...", flush=True)
    t0 = time.monotonic()
    phase1_end = t0 + 500
    best_a, best_s = _fista_adaptive(a0, phase1_end, top_m=30)
    elapsed = time.monotonic() - t0
    if best_s < global_best_score:
        global_best = best_a.copy()
        global_best_score = best_s
    print(f"  Done in {elapsed:.0f}s: {global_best_score:.8f}", flush=True)

    # === Phase 2: Mirror descent (300s) ===
    print("Phase 2: Mirror descent...", flush=True)
    t0 = time.monotonic()
    phase2_end = t0 + 300
    best_a2, best_s2 = _mirror_descent(global_best, phase2_end, top_m=30)
    elapsed = time.monotonic() - t0
    if best_s2 < global_best_score:
        global_best = best_a2.copy()
        global_best_score = best_s2
    print(f"  Done in {elapsed:.0f}s: {global_best_score:.8f}", flush=True)

    # === Phase 3: FISTA with more peaks (300s) ===
    print("Phase 3: FISTA top-60...", flush=True)
    t0 = time.monotonic()
    phase3_end = t0 + 300
    best_a3, best_s3 = _fista_adaptive(global_best, phase3_end, top_m=60, eta_init=0.3 / np.sqrt(n))
    elapsed = time.monotonic() - t0
    if best_s3 < global_best_score:
        global_best = best_a3.copy()
        global_best_score = best_s3
    print(f"  Done in {elapsed:.0f}s: {global_best_score:.8f}", flush=True)

    # === Phase 4: Diverse starting points (300s) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"Phase 4: Diverse starts ({remaining:.0f}s)...", flush=True)
        starts = [
            _make_anti_palindromic(n, eps=0.7, smoothness=3),
            _make_anti_palindromic(n, eps=0.9, smoothness=3),
            _make_two_level_smooth(n, eps=0.6, transition=0.08),
            _make_gaussian_offset(n, offset=0.25, width=0.3),
        ]
        per_start = remaining / (len(starts) + 1)

        for i, start in enumerate(starts):
            if time.monotonic() >= deadline:
                break
            t_dead = min(time.monotonic() + per_start, deadline)
            sa, ss = _fista_adaptive(start, t_dead, top_m=30)
            if ss < global_best_score:
                global_best = sa.copy()
                global_best_score = ss
                print(f"  Start {i+1}: NEW BEST {ss:.8f}", flush=True)
            else:
                print(f"  Start {i+1}: {ss:.8f}", flush=True)

    # === Phase 5: Local search / perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"Phase 5: Local search ({remaining:.0f}s)...", flush=True)
        best_ls, best_ls_score = _local_search(
            global_best, min(time.monotonic() + remaining * 0.5, deadline), rng
        )
        if best_ls_score < global_best_score:
            global_best = best_ls.copy()
            global_best_score = best_ls_score
        print(f"  Local search: {global_best_score:.8f}", flush=True)

    # === Phase 6: Final polish with small steps ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"Phase 6: Final polish ({remaining:.0f}s)...", flush=True)
        eta_polish = 0.05 / np.sqrt(n)
        best_p, best_p_score = _fista_constant_step(
            global_best, deadline, top_m=100, eta=eta_polish
        )
        if best_p_score < global_best_score:
            global_best = best_p.copy()
            global_best_score = best_p_score
        print(f"  Polish: {global_best_score:.8f}", flush=True)

    # Final validation
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.8f}", flush=True)
    return result
