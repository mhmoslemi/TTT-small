def fast_score(f):
    """Compute the score without calling evaluate_sequence."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    n = len(f)
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))

def build_conv_matrix(n, f):
    """Sparse matrix A (2n-1 x n) where (A @ g)[k] = conv(f, g)[k]."""
    rows = []
    cols = []
    data = []
    for j in range(n):
        for k_offset in range(n):
            k = j + k_offset
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                data.append(f[k_offset])
    A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A

def solve_lp_direction(f, M, time_limit=5.0):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def lp_step_with_linesearch(f, M, lp_time=5.0):
    """One LP step with backtracking line search. Returns best (f_new, score)."""
    n = len(f)
    f = f / f.sum()  # normalize
    g = solve_lp_direction(f, M, lp_time)
    if g is None:
        return f, fast_score(f)
    
    best_f = f.copy()
    best_sc = fast_score(f)
    for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
        f_try = (1-t)*f + t*g
        f_try = np.maximum(f_try, 0.0)
        s = f_try.sum()
        if s < 1e-10:
            continue
        f_try = f_try / s
        sc = fast_score(f_try)
        if sc < best_sc:
            best_sc = sc
            best_f = f_try.copy()
    return best_f, best_sc

def gradient_step(f, alpha=0.5, method='sumsq'):
    """One gradient descent step on a smooth proxy objective."""
    n = len(f)
    f = f / f.sum()
    c = np.convolve(f, f)
    
    if method == 'sumsq':
        # J = sum(c_k^2), grad = 4 * conv(c, flipped_f)[0:n]
        flipped = f[::-1]
        grad_full = np.convolve(c, flipped)
        grad = grad_full[:n]
    elif method == 'logsumexp':
        # Smooth max approximation
        tau = 20.0  # temperature parameter
        c_max = c.max()
        weights = np.exp(tau * (c - c_max))
        weights /= weights.sum()
        # d/d f_i of sum_k exp(tau*c_k) / sum_k exp(tau*c_k)
        # = tau * (sum_k w_k * dc_k/df_i) 
        # dc_k/df_i = 2*f_{k-i} (as derived earlier)
        # So grad_i = tau * sum_k w_k * 2*f_{k-i}
        # = 2*tau * (w * flipped_f)[i] where w is the weight vector
        flipped = f[::-1]
        grad_full = np.convolve(weights, flipped)
        grad = 2.0 * tau * grad_full[:n]
    else:
        return f
    
    f_new = f - alpha * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s

def block_perturb(f, rng, max_width_ratio=0.2):
    """Random block perturbation."""
    n = len(f)
    cand = f.copy()
    w = max(2, int(n * max_width_ratio * rng.random() + 1))
    w = min(w, n)
    start = int(rng.integers(0, n - w + 1))
    factor = float(np.exp(rng.normal(0.0, 0.3)))
    cand[start:start+w] *= factor
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s

def make_initial(n, rng, method='uniform'):
    """Generate initial sequence."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)**2
    elif method == 'random':

    elif method == 'raised_cosine':

        a = 0.5 * (1 + np.cos(2*np.pi*x))

def run_lp_loop(f, deadline, method='lp', lp_time=5.0, alpha_init=0.3):

    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    alpha = alpha_init
    stall = 0
    
    while time.monotonic() < deadline:
        sc = fast_score(f)
        if not np.isfinite(sc):
            break

        elif method == 'grad_sumsq':
            f_new = gradient_step(f, alpha, 'sumsq')
            sc_new = fast_score(f_new)
        elif method == 'grad_lse':
            f_new = gradient_step(f, alpha, 'logsumexp')
            sc_new = fast_score(f_new)
        else:
            f_new = f
            sc_new = sc
        
        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best = f_new.copy()
            stall = 0
            # Slowly decrease learning rate

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LP-based direction finding with backtracking line search,
gradient descent on smooth proxy objectives (sum-of-squares and log-sum-exp),
and random perturbations for exploration. Starts from the known best solution.
Uses efficient gradient computation via convolution.
"""
import numpy as np
from scipy.sparse import coo_matrix
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute the score without calling evaluate_sequence."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    n = len(f)
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse matrix A (2n-1 x n) where (A @ g)[k] = conv(f, g)[k]."""
    i = np.tile(np.arange(n), n)
    j = np.repeat(np.arange(n), n)
    rows = j + i
    data = f[i]
    A = coo_matrix((data, (rows, j)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_direction(f, M, time_limit=5.0, delta=None):
    """
    Solve max sum(g) s.t. conv(f,g) <= M, g >= 0, and optionally ||g-f||_1 <= delta.
    Returns normalized g or None.
    """
    n = len(f)
    A = build_conv_matrix(n, f)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    if delta is not None:
        # Add constraint: sum|g_i - f_i| <= delta (linearize with auxiliary vars)
        # g_i - f_i <= u_i, f_i - g_i <= u_i, sum u_i <= delta
        # Simplified: just add -delta <= g_i - f_i <= delta for each i
        # This is: g_i - f_i <= delta AND f_i - g_i <= delta
        # Summing: sum(g_i) - sum(f_i) <= n*delta AND sum(f_i) - sum(g_i) <= n*delta
        # Since sum(f) = 1 (normalized): sum(g) <= 1 + n*delta AND sum(g) >= 1 - n*delta
        # But we're maximizing sum(g), so the lower bound is non-binding.
        # Add per-element: g_i <= f_i + delta, -g_i <= -(f_i - delta) i.e. g_i >= f_i - delta
        # But g_i >= 0 already, so: g_i >= max(0, f_i - delta) and g_i <= f_i + delta
        for i in range(n):
            lo = max(0.0, f[i] - delta)
            hi = f[i] + delta
            bounds[i] = (lo, hi)
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_step_with_linesearch(f, M, lp_time=5.0, use_delta=False):
    """One LP step with backtracking line search. Returns (best_f, best_score)."""
    n = len(f)
    f_norm = f / f.sum()
    delta = 0.3 / n if use_delta else None
    g = solve_lp_direction(f_norm, M, lp_time, delta=delta)
    if g is None:
        return f_norm, fast_score(f_norm)
    
    best_f = f_norm.copy()
    best_sc = fast_score(f_norm)
    for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01]:
        f_try = (1 - t) * f_norm + t * g
        f_try = np.maximum(f_try, 0.0)
        s = f_try.sum()
        if s < 1e-10:
            continue
        f_try = f_try / s
        sc = fast_score(f_try)
        if sc < best_sc:
            best_sc = sc
            best_f = f_try.copy()
    return best_f, best_sc


def compute_grad_sumsq(f):
    """Gradient of sum(conv(f,f)_k^2) with respect to f."""
    n = len(f)
    c = np.convolve(f, f)
    grad_full = np.convolve(c, f[::-1])
    return 4.0 * grad_full[n - 1: 2 * n - 1]


def compute_grad_lse(f, tau=30.0):
    """Gradient of (1/tau)*log(sum exp(tau*conv(f,f)_k)) with respect to f."""
    n = len(f)
    c = np.convolve(f, f)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    grad_full = np.convolve(w, f[::-1])
    return 2.0 * grad_full[n - 1: 2 * n - 1]


def gradient_step_sumsq(f, alpha):
    """One gradient descent step on sum-of-squares objective."""
    n = len(f)
    f = f / f.sum()
    grad = compute_grad_sumsq(f)
    f_new = f - alpha * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def gradient_step_lse(f, alpha, tau=30.0):
    """One gradient descent step on log-sum-exp objective."""
    n = len(f)
    f = f / f.sum()
    grad = compute_grad_lse(f, tau)
    f_new = f - alpha * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def block_perturb(f, rng, max_width_ratio=0.15, noise_scale=0.3):
    """Random block multiplicative perturbation."""
    n = len(f)
    cand = f.copy()
    w = max(2, int(n * max_width_ratio * (0.3 + 0.7 * rng.random())))
    w = min(w, n)
    start = int(rng.integers(0, max(1, n - w + 1)))
    factor = float(np.exp(rng.normal(0.0, noise_scale)))
    cand[start:start + w] *= factor
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def make_initial(n, rng, method='uniform'):
    """Generate initial normalized sequence."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0.001, np.pi - 0.001, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n) + 0.1
    elif method == 'two_peaks':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 8)
        a = np.exp(-0.5 * ((x - n // 4) / s) ** 2) + np.exp(-0.5 * ((x - 3 * n // 4) / s) ** 2)
    elif method == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif method == 'bump':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 4)
        a = np.exp(-0.5 * ((x - n / 2) / s) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def run_lp_loop(f, deadline, lp_time=5.0, use_delta=False, max_stall=50):
    """Run LP optimization loop until deadline. Returns (best_f, best_score)."""
    n = len(f)
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(f, f)
        M = conv.max()
        if M < 1e-15:
            break
        
        f_new, sc_new = lp_step_with_linesearch(f, M, lp_time, use_delta)
        
        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best = f_new.copy()
            stall = 0
        else:
            stall += 1
        
        f = f_new
        
        if stall > max_stall:
            rng_kick = np.random.default_rng(10000 * it + int(f[0] * 1e8))
            f = block_perturb(f, rng_kick, max_width_ratio=0.1, noise_scale=0.5)
            stall = 0
    
    return best, best_sc


def run_grad_loop(f, deadline, method='sumsq', alpha_init=0.05, max_stall=100):
    """Run gradient-based optimization loop until deadline."""
    n = len(f)
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    alpha = alpha_init
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        if method == 'sumsq':
            f_new = gradient_step_sumsq(f, alpha)
        else:
            f_new = gradient_step_lse(f, alpha, tau=30.0)
        
        sc_new = fast_score(f_new)
        
        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best = f_new.copy()
            stall = 0
            alpha = max(alpha * 0.995, 0.001)
        else:
            stall += 1
            alpha = max(alpha * 0.97, 0.001)
        
        f = f_new
        
        if stall > max_stall:
            rng_kick = np.random.default_rng(20000 * it + int(np.sum(f[:10]) * 1e8))
            f = block_perturb(f, rng_kick, max_width_ratio=0.1, noise_scale=0.5)

    return best, best_sc
