Actually the standard algorithm is simpler:
1. Sort b in decreasing order: b_(1) >= b_(2) >= ... >= b_(n).
2. Compute cumulative sums: S_m = b_(1) + ... + b_(m).

4. t = (S_m - 1) / m.
5. a[i] = max(b[i] - t, 0).

def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    # softmax
    c_shifted = c - max(c)
    exp_c = np.exp(c_shifted / tau)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    value = tau * (max(c) + np.log(sum_exp))
    # gradient
    N = 1
    while N < 3*n:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    v = np.fft.irfft(fw * np.conj(fa), N)[:n]
    grad = 2.0 * v
    return value, grad, c, w

def project_simplex(b):
    """Project b onto {x >= 0, sum(x) = 1}."""
    n = len(b)
    # Sort in decreasing order
    u = np.sort(b)[::-1]
    # Cumulative sum
    cssv = np.cumsum(u)
    # Find rho: largest m such that u[m] > (cssv[m] - 1) / (m+1)
    cond = u > (cssv - 1.0) / np.arange(1, n+1)
    rho = np.sum(cond)
    if rho == 0:
        # All b[i] <= 0, return uniform
        return np.ones(n) / n
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(b - theta, 0.0)

def optimize_sequence(a0, tau_start, tau_end, time_limit, lr_init=0.1, lr_decay=0.99):
    """Main optimization loop."""
    a = a0.copy()
    a = a / sum(a)  # normalize to sum = 1
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = lr_init
    tau = tau_start
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed_frac * 1.2)

            lr *= 0.5

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        lr *= lr_decay  # or some decay schedule
