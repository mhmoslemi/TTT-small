def init_sequence(rng, length):
    # Create a sequence with zeros and positive numbers
    a = rng.uniform(0, 1000, length)
    # Ensure there's at least one positive number by clipping any negatives
    a = np.clip(a, 0, 1000)
    return a

def crossover(parent1, parent2, rng):
    # One-point crossover for sequences of equal length
    point = rng.integers(0, 127)  # Random point between 0 and 126
    child = parent1[:point+1].copy()  # First part from parent1
    child = np.append(child, parent2[point+1:].copy())  # Second part from parent2
    return child

def mutation(a, rng, mutation_rate=0.2):
