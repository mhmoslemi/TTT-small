def spectral_flatten(a, n_iter, rng):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    n = len(a)
    for it in range(n_iter):
        # Compute FFT
        A = np.fft.fft(a, 2*n-1)
        power = np.abs(A)**2
        # Target: flat power spectrum
        target = np.mean(power)
        # Adjust in frequency domain
        # We want to reduce peaks in power spectrum
        # This means reducing |A| at high-power frequencies
        # A_new = A * exp(-alpha * (power - target) / (2*target))
        alpha = 0.1
        A_new = A * np.exp(-alpha * (power - target) / (2 * target))
        # Transform back
        a_new = np.fft.ifft(A_new, 2*n-1)[:n]
        a_new = np.real(a_new)
        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        # Renormalize
        if np.sum(a_new) > 0:
            a_new = a_new / np.sum(a_new) * np.sum(a)
        a = a_new
    return a

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Combines: (1) gradient-based soft-max optimization with adaptive learning rate,
(2) coordinate descent with 1D line search, (3) peak-reduction local search with
structured perturbations, and (4) multi-start over sequence lengths and initializations.
Uses FFT-based convolution for efficiency.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast computation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def grad_soft_max_conv(a, tau):
    """
    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # = 2 * conv(a[::-1], w)[j + n - 1]... actually:
    # g[j] = 2 * sum_m w[m] * a[m-j] for valid (m-j) in [0,n-1]
    # This is 2 * (w * a)[j] where (w*a) is cross-correlation
    # = 2 * conv(w, a[::-1])[n-1+j] ... let me just do it directly via FFT
    # g[j] = 2 * sum_k w[k] * a[k-j] (valid when 0<=k-j<n)
    # = 2 * (w cross-correlate a)[j]
    # Cross-correlation: (w ⋆ a)[j] = sum_k w[k] a[k+j]... 
    # Actually let me use: g[j] = 2 * sum_m w[m] * a[m-j]
    # Let r = m-j, then m = r+j: g[j] = 2 * sum_r w[r+j] * a[r]
    # This is 2 * (w_shifted * a)[r] summed... 
    
    # Simpler: use the fact that d(conv[k])/da[j] = 2*a[k-j] for valid k
    # So g[j] = sum_k w[k] * 2 * a[k-j] (for valid k-j)
    # = 2 * sum_k w[k] * a[k-j] where 0 <= k-j < n
    # Let l = k-j: g[j] = 2 * sum_l w[l+j] * a[l] where 0 <= l < n and 0 <= l+j < 2n-1
    # This is 2 * (w[j:] convolved with a) at position 0... 
    
    # Easiest: compute via FFT
    # g[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    # = 2 * (full_convolve(w, a))[j + n - 1] ... no
    # Let's think: define h[j] = sum_k w[k]*a[k-j] for 0<=k-j<n
    # This is the cross-correlation of w and a, evaluated at lag j
    # cross_corr(w, a)[j] = sum_k w[k]*a[k-j]
    # = conv(w, a[::-1]) at position (n-1+j)... 
    
    # Let me just compute it as: h = np.correlate(w, a, mode='full')
    # np.correlate(w, a, 'full')[k] = sum_n w[n] * a[n + k - (n_a - 1)]
    # For k from -(n_a-1) to (n_w-1)
    # We want sum_k w[k]*a[k-j] = cross_corr at lag -j... 
    
    # Let me use a different approach. Define:
    # h[j] = sum_{k=j}^{2n-2} w[k] * a[k-j] for j = 0,...,n-1
    # (where a[k-j] is only valid for 0 <= k-j < n, i.e., j <= k <= j+n-1)
    # So h[j] = sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} w[k] * a[k-j]
    # = sum_{l=0}^{n-1} w[l+j] * a[l]
    
    # This is: h = iff(fft(w) * conj(fft(a))) ... cross-correlation
    # Or simply: h[j] = dot(w[j:j+n], a) for j = 0,...,n-1
    # Can compute via FFT: this is the cross-correlation
    # h = np.correlate(w, a, mode='valid') ... no that gives different thing
    
    # Let me just do it with a sliding dot product using FFT:
    # h[j] = sum_{l=0}^{n-1} a[l] * w[j+l]
    # This is the convolution of a with w (reversed) at specific positions
    # conv(a, w_rev)[j + n - 1] where w_rev = w[::-1]
    # conv(a, w_rev)[k] = sum_i a[i]*w_rev[k-i] = sum_i a[i]*w_rev[k-i]
    # w_rev[m] = w[n-1-m]... no, w has length 2n-1
    # Let's just use: h = np.correlate(w, a, mode='full')  
    # np.correlate(c, a, 'full')[n] = sum_k c[k] * a[k+n-(n_a-1)]
    # This is getting confusing. Let me just do it directly for clarity in the gradient:
    
    # Actually the simplest correct formula:
    # g[j] = 2 * sum_k w[k] * a[k-j] for valid k (0<=k<2n-1, 0<=k-j<n)
    # = 2 * sum_{l=0}^{n-1} a[l] * w[l+j]
    # 
    # To compute all g[j] for j=0..n-1:
    # g[j] = 2 * dot(a, w[j:j+n])
    # This is a sliding window dot product, computable via FFT:
    # Define v = w (length 2n-1), and compute:
    # g[j] = 2 * sum_{l=0}^{n-1} a[l] * v[l+j]
    # This is the cross-correlation of a and v, at lag j
    # = conv(a, v[::-1])[n-1+j]... 
    
    # You know what, let me just use the direct formula with FFT:
    # g[j] = 2 * sum_l a[l] * w[l+j]
    # Let u[l] = a[l], and define the cross-correlation R[j] = sum_l u[l] * w[l+j]
    # R = np.correlate(w, u, mode='full') gives R[k] = sum_l w[l] * u[l+k-(n_u-1)]
    # Hmm, let me just use a simple approach:
    # R[j] = sum_{l=0}^{n-1} a[l] * w[l+j] for j = 0, ..., 2n-2-n = n-2... 
    # Actually j goes from 0 to (2n-2) - (n-1) = n-1. Good.
    
    # Using FFT: R[j] = IFFT(FFT(a) * conj(FFT(w))) at appropriate indices
    # The cross-correlation theorem: if R[j] = sum_l a[l]*w[l+j], 
    # then FFT(R) = conj(FFT(a)) * FFT(w) (with appropriate zero-padding)
    
    size = 1
    while size < 3 * n:
        size <<= 1
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    R = np.fft.irfft(np.conj(A) * W, size)
    # R[j] = sum_l a[l] * w[l+j] for j = 0, ..., size-1 (circular)
    # Due to zero-padding, R[j] for j=0..n-1 should be correct
    g = 2.0 * R[:n]
    
    return value, g, c


def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on threshold."""
    if np.sum(np.maximum(v, 0)) < S:
        # v is too small, just scale up the positive part or make uniform
        return np.maximum(v, 0) / max(np.sum(np.maximum(v, 0)), 1e-10) * S
    
    lo = np.min(v) - 1
    hi = np.max(v) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.maximum(v - mid, 0)
        s = np.sum(projected)
        if s > S:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.maximum(v - mid, 0)


def coordinate_descent_step(a, max_iter=5, rng=None):
    """
    Perform coordinate descent: optimize each a[j] to minimize the max convolution.
    Uses 1D bisection for each coordinate.
    """
    n = len(a)
    a = a.copy()
    
    # Precompute convolution
    conv = fft_convolve(a, a)
    current_max = np.max(conv)
    
    # Try a random subset of coordinates
    if rng is not None and n > 100:
        indices = rng.choice(n, size=min(max_iter * 10, n), replace=False)
    else:
        indices = np.arange(n)
    
    for j in indices:
        if len(indices) > max_iter * 5 and j != indices[0] and j % max(1, len(indices)//max_iter) != 0:
            continue
            
        # Compute the "other" convolution values (not involving a[j])
        # conv[k] = other[k] + 2*a[j]*a[k-j] for valid k
        # other[k] = conv[k] - 2*a[j]*a[k-j] (for valid k-j)
        old_val = a[j]
        
        # Determine valid range of k: k-j in [0, n-1], so k in [j, j+n-1]
        # But k also in [0, 2n-2]
        k_low = max(0, j)
        k_high = min(2*n-2, j+n-1)
        
        # The "other" part of conv for these k values:
        # For k in [k_low, k_high]: conv[k] = c[k] + 2*a[j]*a[k-j]
        # So c[k] = conv[k] - 2*old_val*a[k-j]
        # For k outside this range: conv[k] is unaffected by a[j]
        
        # To find optimal a[j]:
        # We want to minimize max_k (c[k] + 2*x*a[k-j]) subject to x >= 0
        # where c[k] is the "background" and the ramp is 2*a[k-j]*x
        
        a_j = a[j]
        
        # Compute background for affected positions
        affected = np.arange(k_low, k_high + 1)
        bg = conv[affected] - 2.0 * a_j * a[affected - j]
        
        # Unaffected max
        if k_low > 0:
            unaffected_prefix = conv[:k_low]
        else:
            unaffected_prefix = np.array([])
        if k_high < 2*n-2:
            unaffected_suffix = conv[k_high+1:]
        else:
            unaffected_suffix = np.array([])
        unaffected = np.concatenate([unaffected_prefix, unaffected_suffix])
        M_other = np.max(unaffected) if len(unaffected) > 0 else 0.0
        
        # Coefficients of x: coeff[k] = 2*a[k-j] for k in affected
        coeffs = 2.0 * a[affected - j]
        
        # Find optimal x: minimize max(M_other, max(bg + coeff*x))
        # At x=0: max_val = max(M_other, max(bg))
        # The ramps bg + coeff*x increase linearly
        # Optimal: find x where the steepest ramp that's below M_other reaches M_other
        # Or x=0 if all ramps are already above M_other
        
        # Simple approach: try a few candidate x values
        candidate_xs = [0.0, a_j * 0.5, a_j, a_j * 1.5]
        
        # Also try the "water-filling" solution:
        # For each affected k, the ramp hits M_other at x_k = (M_other - bg[k]) / coeff[k] (if coeff[k] > 0)
        valid_mask = coeffs > 1e-15
        if np.any(valid_mask):
            x_candidates = (M_other - bg[valid_mask]) / coeffs[valid_mask]
            x_candidates = x_candidates[x_candidates > 0]
            if len(x_candidates) > 0:
                # The optimal is the minimum x_candidate that brings all ramps down to M_other
                # Actually, we want the largest x that keeps all ramps <= M_other
                # That's min over k of (M_other - bg[k]) / coeff[k] for those where bg[k] < M_other
                below_mask = bg[valid_mask] < M_other
                if np.any(below_mask):
                    x_max_flat = np.min((M_other - bg[valid_mask][below_mask]) / coeffs[valid_mask][below_mask])
                    candidate_xs.append(max(0.0, x_max_flat))
        
        # Evaluate and pick best
        best_x = a_j
        best_val = np.inf
        for x in candidate_xs:
            x = max(0.0, x)
            # New conv max
            new_affected = bg + coeffs * x
            new_max = max(M_other, np.max(new_affected)) if len(new_affected) > 0 else M_other
            if new_max < best_val:
                best_val = new_max
                best_x = x
        
        # Fine-tune with a few more evaluations around best_x
        for mult in [0.8, 0.9, 1.0, 1.1, 1.2]:
            x = best_x * mult
            x = max(0.0, x)
            new_affected = bg + coeffs * x
            new_max = max(M_other, np.max(new_affected)) if len(new_affected) > 0 else M_other
            if new_max < best_val:
                best_val = new_max
                best_x = x
        
        a[j] = best_x
        # Recompute conv (approximate: only update affected positions)
        # For speed, just recompute fully every few steps
        if j % 16 == 0:
            conv = fft_convolve(a, a)
    
    conv = fft_convolve(a, a)
    return a, np.max(conv)


def gradient_descent_optimize(n, a_init, time_limit, rng, tau_start=None, tau_end=None):
    """
    Gradient descent with soft-max smoothing and Adam optimizer.
    a_init is the initial sequence (will be normalized internally).
    """
    S = np.sqrt(2.0 * n)
    a_init = np.clip(a_init, 0.01, None)
    theta = np.log(a_init / np.sum(a_init) + 1e-12)
    a = softmax_normalize(theta) * S
    
    if tau_start is None:
        tau_start = 50.0 / n
    if tau_end is None:
        tau_end = 0.5 / n
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_base = 3.0
    step = 0
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    # Adaptive learning rate
    prev_score = best_score
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        
        # Project gradient to maintain sum constraint
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        
        # Adaptive LR
        lr = lr_base / np.sqrt(step + 1)
        lr = lr * (1.0 + 0.5 * np.exp(-step / 500.0))  # higher early on
        
        theta_new = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Check new score
        a_new = softmax_normalize(theta_new) * S
        score_new = local_score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
        elif score_new < prev_score:
            patience = 0
        else:
            patience += 1
            if patience > 200:
                # Restart with smaller LR
                lr_base *= 0.7
                patience = 0
        
        if score_new < prev_score * 1.05:
            theta = theta_new
        else:
            # Reject step, try smaller step
            theta = theta - 0.3 * lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        prev_score = local_score(softmax_normalize(theta) * S)
        
        # Periodic coordinate descent
        if step % 500 == 0:
            a_cd, _ = coordinate_descent_step(a, max_iter=5, rng=rng)
            theta_cd = np.log(a_cd / np.sum(a_cd) + 1e-12)
            # Only accept if better
            score_cd = local_score(a_cd * S / np.sum(a_cd))
            if score_cd < best_score:
                best_score = score_cd
                best_a = (a_cd * S / np.sum(a_cd)).copy()
                theta = theta_cd
        
        if step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    step={step}, score={local_score(a):.6f}, best={best_score:.6f}, tau={tau:.2e}, t={elapsed:.1f}s")
    
    return best_a, best_score


def peak_reduction_search(a_init, time_limit, rng):
    """
    Local search that specifically targets peak reduction.
    Identifies positions contributing to the max convolution value
    and makes targeted adjustments.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    current = np.clip(a_init, 0.01, None).copy()
    current = current / np.sum(current) * S
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    temperature = 0.002
    cooling = 0.9998
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution and find peaks
        conv = fft_convolve(current, current)
        max_val = np.max(conv)
        
        # Find active set: positions where conv is close to max
        threshold = max_val * 0.95
        active_positions = np.where(conv >= threshold)[0]
        
        r = rng.random()
        candidate = current.copy()
        
        if r < 0.4:
            # Targeted: reduce contribution to active peaks
            for pos in active_positions[:min(5, len(active_positions))]:
                # Find which a[j] values contribute most to conv[pos]
                # conv[pos] = sum_m a[m]*a[pos-m]
                # Contributors: pairs (m, pos-m) where both in [0, n-1]
                lo = max(0, pos - (n-1))
                hi = min(n-1, pos)
                if hi < lo:
                    continue
                m_vals = np.arange(lo, hi+1)
                contribs = current[m_vals] * current[pos - m_vals]
                # Reduce the top contributors
                n_reduce = max(1, len(m_vals) // 4)
                top_idx = np.argsort(contribs)[-n_reduce:]
                for idx in top_idx:
                    m = m_vals[idx]
                    candidate[m] *= np.exp(rng.normal(-0.05, 0.02))
        elif r < 0.6:
            # Spread mass: reduce peaks, increase valleys
            conv_flat = conv / max_val
            low_mask = conv_flat < 0.3
            # Add mass at positions that don't contribute too much to peaks
            n_add = rng.integers(3, max(4, n // 10))
            if np.any(low_mask):
                # Find a[j] that contribute to low conv positions
                add_candidates = np.zeros(n)
                for pos in np.where(low_mask)[0][:10]:
                    lo = max(0, pos - (n-1))
                    hi = min(n-1, pos)
                    if hi >= lo:
                        m_vals = np.arange(lo, hi+1)
                        add_candidates[m_vals] += current[pos - m_vals]
                if np.sum(add_candidates) > 0:
                    add_idx = rng.choice(n, size=n_add, replace=False, p=add_candidates/add_candidates.sum())
                    candidate[add_idx] *= np.exp(rng.normal(0.04, 0.01))
            else:
                add_idx = rng.choice(n, size=n_add, replace=False)
                candidate[add_idx] *= np.exp(rng.normal(0.03, 0.01))
        elif r < 0.75:
            # Block perturbation around peak contributors
            peak_pos = int(np.argmax(conv))
            lo = max(0, peak_pos - (n-1))
            hi = min(n-1, peak_pos)
            center_j = (lo + hi) // 2
            width = max(1, n // 20)
            start = max(0, center_j - width)
            end = min(n, center_j + width)
            candidate[start:end] *= np.exp(rng.normal(0.0, 0.03, size=end-start))
        elif r < 0.9:
            # Symmetry exploitation: partially symmetrize
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.02, size=n))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))
        
        candidate = np.clip(candidate, 0.01, 1000.0)
        # Renormalize
        candidate = candidate / np.sum(candidate) * S
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            stall_count = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if best_score < 1.505:
                    print(f"    peak_reduction: new best = {best_score:.6f} at t={time.monotonic()-t0:.1f}s")
        else:
            stall_count += 1
            if rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-10)):
                current = candidate
                current_score = new_score
        
        temperature *= cooling
    
    return best, best_score


def make_init_chirp(n, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.3 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_multitone(n, rng, n_tones=5):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for i in range(n_tones):
        k = rng.integers(1, 20)
        amp = 0.1 + 0.2 / (i+1)
        phase = rng.uniform(0, 2*np.pi)
        a += amp * np.cos(2*np.pi*k*t/n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_blended(n, rng, base_seq=None):
    """Blend of different initialization strategies."""
    r = rng.random()
    if r < 0.3 and base_seq is not None:
        # Perturb base sequence
        if len(base_seq) == n:
            a = base_seq.copy()
        else:
            x_old = np.linspace(0, 1, len(base_seq))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, base_seq)
        a *= np.exp(rng.normal(0, 0.1, size=n))
    elif r < 0.6:
        a = make_init_chirp(n, rng)
    else:
        a = make_init_multitone(n, rng)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining gradient optimization, coordinate descent,
    and peak-reduction local search with multiple sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check for provided sequence
    base_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        base_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(base_seq)
        print(f"Provided sequence: n={len(base_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = base_seq.copy()
    
    # Define search configurations
    # (length, n_inits, time_fraction)
    configs = [
        (256, 3, 0.15),
        (320, 2, 0.10),
        (384, 2, 0.10),
        (448, 2, 0.10),
        (512, 3, 0.15),
        (640, 2, 0.10),
        (768, 2, 0.10),
        (1024, 1, 0.08),
    ]
    
    print(f"\n=== Phase 1: Gradient optimization ===")
    print(f"Time budget: {time_budget:.1f}s, deadline in {time_budget:.1f}s")
    
    for n, n_inits, time_frac in configs:
        if time.monotonic() > deadline - 60:
            break
        time_left = deadline - time.monotonic()
        time_per = max(5.0, time_left * time_frac / max(1, n_inits * (1 if n <= 512 else 2)))
        time_per = min(time_per, 120.0)
        
        for init_i in range(n_inits):
            if time.monotonic() > deadline - 45:
                break
            
            a0 = make_init_blended(n, rng, base_seq if base_seq is not None and n == len(base_seq) else None)
            
            t_start = time.monotonic()
            a_best, score = gradient_descent_optimize(n, a0, time_per, rng)
            elapsed = time.monotonic() - t_start
            print(f"  n={n}, init={init_i}, score={score:.6f}, time={elapsed:.1f}s")
            
            if score < best_score:
                best_score = score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}) ***")
    
    # Phase 2: Peak reduction local search on best candidate
    print(f"\n=== Phase 2: Peak reduction local search ===")
    if best_seq is not None:
        polish_time = min(180.0, max(30.0, deadline - time.monotonic()))
        print(f"  Starting with score={best_score:.6f}, time={polish_time:.1f}s")
        polished, polish_score = peak_reduction_search(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
    
    # Phase 3: Coordinate descent refinement
    print(f"\n=== Phase 3: Coordinate descent refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        cd_time = min(90.0, deadline - time.monotonic())
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a = best_seq / np.sum(best_seq) * S
        t0 = time.monotonic()
        while time.monotonic() - t0 < cd_time:
            a, conv_max = coordinate_descent_step(a, max_iter=min(20, n//10), rng=rng)
            a = a / np.sum(a) * S
            score = local_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"    CD: new best = {best_score:.6f}")
            if time.monotonic() - t0 > cd_time:
                break
    
    # Phase 4: Final peak reduction
    print(f"\n=== Phase 4: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        final_time = max(10.0, deadline - time.monotonic() - 5)
        polished, polish_score = peak_reduction_search(best_seq, final_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  Final polish: {best_score:.6f}")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if base_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in base_seq]
    return [1.0] * 256
