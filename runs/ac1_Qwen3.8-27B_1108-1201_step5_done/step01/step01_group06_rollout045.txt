"""
Minimize 2*n*max(self-convolution)/sum^2 via coordinate descent.

Key insight: The objective is scale-invariant, so we fix sum(a) = sqrt(2n).
For each coordinate j, the 1D subproblem (minimize max(c) over a[j]) is convex,
solved by golden section search. The self-convolution is updated incrementally.
Multiple initializations and random restarts help escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def run_coordinate_descent(a_init, n, time_budget, rng, verbose=True):
    """
    Coordinate descent to minimize max(self-convolution) with sum(a)=sqrt(2n), a>=0.
    Returns (best_a, best_maxc).
    """
    S = np.sqrt(2.0 * float(n))
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * (S / n)
    else:
        a = a * (S / s)
    
    c = fft_convolve_self(a)
    best_maxc = c.max()
    best_a = a.copy()
