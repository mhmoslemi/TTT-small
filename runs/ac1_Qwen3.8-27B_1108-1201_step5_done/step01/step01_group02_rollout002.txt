import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))

def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s

def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine(a, time_limit_s=600.0, max_iters=500, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 50 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 40:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(3.0, min(60.0, remaining * 0.3))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f}) at iter {it}")
            break
        g = g / sum_g
        alpha = min(0.6, max(0.05, 1.5 / (1.0 + 0.3 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.2
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def gradient_refine(a, time_limit_s=120.0, verbose=True, t_param=50.0):
    """Gradient descent on log-sum-exp smooth approximation."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    lr = 0.1 / t_param  # Initial learning rate
    stagnant = 0
    it = 0
    max_iters = 5000
    
    while time.monotonic() - start_time < time_limit_s and it < max_iters:
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 100 == 0 or it < 10):
                print(f"    GD iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 200:
            if verbose:
                print(f"    GD stagnant at iter {it}, best={best_score:.6f}")
            break
        
        # Compute smooth gradient
        max_conv = conv.max()
        shifted = t_param * (conv - max_conv)
        # Clip for numerical stability
        shifted = np.clip(shifted, -100, 100)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / exp_shifted.sum()
        
        # Gradient of sum_k p_k * conv[k] w.r.t. a[j]
        # = t * convolve(p, a, 'valid')
        grad = t_param * np.convolve(p, a, mode='valid')
        
        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        new_score = score_sequence(a_new)
        if new_score < score:
            a = a_new
            # Adaptive LR: increase slightly if improving
            lr = lr * 1.01
        else:
            # Decrease LR if not improving
            lr = lr * 0.5
            # Try smaller step
            a_new = a - lr * grad
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                if score_sequence(a_new) < score:
                    a = a_new
        
        it += 1
    
    if verbose:
        print(f"    GD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

"""
Multi-strategy optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2
over non-negative sequences. Combines LP-based refinement, multiplicative
gradient descent on log-sum-exp smooth approximation, and local search
with diverse initializations and warm-starting from parent sequences.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_flat_top(n):
    """Sequence with flat-top weighting that reduces peak autocorrelation."""
    t = np.arange(n, dtype=np.float64)
    # Use a super-Gaussian (flatter top)
    p = 6.0
    center = (n - 1) / 2.0
    x = (t - center) / (n / 2.0)
    return np.maximum(0.0, 1.0 - np.abs(x) ** p)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def solve_lp_step_tight(a, M, factor=0.99, timeout=30.0):
    """LP with tighter constraint to force more aggressive reduction."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M * factor)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, max_iters=500, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 100 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 50:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(3.0, min(60.0, remaining * 0.3))
        # Alternate between standard and tight LP
        if it % 3 == 0:
            g = solve_lp_step_tight(a, M, factor=0.995, timeout=timeout)
        else:
            g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement at iter {it}")
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.2 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def multiplicative_gd_refine(a, time_limit_s=120.0, verbose=True, t_start=20.0, t_end=200.0):
    """
    Multiplicative weight gradient descent on log-sum-exp smooth approximation.
    Uses annealing of the smoothness parameter t from t_start to t_end.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    lr = 0.5
    stagnant = 0
    it = 0
    max_iters = 10000

    while time.monotonic() - start_time < time_limit_s and it < max_iters:
        progress = min(1.0, it / (0.7 * max_iters))
        t_param = t_start + (t_end - t_start) * progress

        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 200 == 0 or it < 10):
                print(f"    MG iter {it:5d} (t={t_param:.0f}): score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 300:
            # Try restarting with smaller lr
            lr = max(0.01, lr * 0.3)
            stagnant = 0
            if lr < 0.02:
                if verbose:
                    print(f"    MG stagnant at iter {it}, best={best_score:.6f}")
                break

        # Compute softmax weights
        max_conv = conv.max()
        shifted = t_param * (conv - max_conv)
        shifted = np.clip(shifted, -500, 0)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / exp_shifted.sum()

        # Gradient: 2 * convolve(p, a, 'valid')
        grad = 2.0 * np.convolve(p, a, mode='valid')

        # Multiplicative update: a_j <- a_j * exp(-lr * grad_j)
        log_update = -lr * grad
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        new_score = score_sequence(a_new)
        if new_score < score:
            a = a_new
            lr = min(2.0, lr * 1.02)
        else:
            lr = lr * 0.7
            # Try with half step
            a_new = a * np.exp(0.5 * np.clip(-lr * grad, -5.0, 5.0))
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                if score_sequence(a_new) < score:
                    a = a_new

        it += 1

    if verbose:
        print(f"    MG done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def topk_gradient_refine(a, time_limit_s=120.0, verbose=True, k_frac=0.1):
    """
    Gradient descent focusing only on the top-K convolution lags.
    Provides a sparser gradient that targets the peak more directly.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    lr = 0.3
    stagnant = 0
    it = 0
    max_iters = 10000

    while time.monotonic() - start_time < time_limit_s and it < max_iters:
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 200 == 0 or it < 10):
                print(f"    TK iter {it:5d}: score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 200:
            lr = max(0.01, lr * 0.5)
            stagnant = 0
            if lr < 0.02:
                if verbose:
                    print(f"    TK stagnant at iter {it}, best={best_score:.6f}")
                break

        # Focus on top-K lags
        K = max(1, int(k_frac * (2 * n - 1)))
        sorted_idx = np.argsort(conv)[::-1]
        top_idx = sorted_idx[:K]

        # Uniform weights on top-K
        p = np.zeros_like(conv)
        p[top_idx] = 1.0 / K

        # Gradient
        grad = 2.0 * np.convolve(p, a, mode='valid')

        # Additive step with projection
        a_new = a - lr * grad
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        new_score = score_sequence(a_new)
        if new_score < score:
            a = a_new
            lr = min(1.0, lr * 1.01)
        else:
            lr = lr * 0.6
            a_new = a - lr * grad
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                if score_sequence(a_new) < score:
                    a = a_new

        it += 1

    if verbose:
        print(f"    TK done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Try different perturbation scales
        if rng.random() < 0.7:
            width = max(2, int(rng.integers(n // 200, n // 8)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate = a.copy()
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.08, size=stop - start))
        else:
            # Global scaling perturbation
            candidate = a * np.exp(rng.normal(0.0, 0.05, size=n))
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search minimizing 2*n*max(convolve(a,a))/sum(a)^2.
    Uses LP refinement, multiplicative gradient descent, top-K gradient,
    and local search with diverse initializations and warm-starting.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 15.0)

    best_score = float('inf')
    best_seq = None

    # Try to use parent sequence
    parent_np = None
    try:
        parent = height_sequence_1
        parent_np = np.array([max(0.0, min(1000.0, float(x))) for x in parent])
        parent_np = normalize_nonneg(parent_np)
        parent_score = score_sequence(parent_np)
        print(f"Parent: n={len(parent_np)}, score={parent_score:.6f}")
        if parent_score < best_score:
            best_score = parent_score
            best_seq = parent_np.copy()
    except Exception:
        pass

    # Define search configurations: (n, init_type, [phase_times])
    # Phases: LP, MG (multiplicative GD), TK (top-K GD), LS (local search)
    configs = []

    if parent_np is not None:
        pn = len(parent_np)
        configs.append((pn, 'parent', [300, 200, 100, 80]))

    configs.extend([
        (2000, 'raised_cosine', [200, 150, 100, 60]),
        (3000, 'raised_cosine', [180, 150, 100, 50]),
        (4000, 'raised_cosine', [150, 150, 80, 40]),
        (2000, 'gaussian', [150, 150, 80, 50]),
        (3000, 'gaussian', [120, 120, 80, 40]),
        (2000, 'blackman', [120, 120, 80, 40]),
        (1500, 'flat_top', [120, 100, 80, 40]),
        (2500, 'random', [100, 100, 60, 40]),
        (2000, 'random', [100, 100, 60, 40]),
    ])

    for n, init_type, phase_times in configs:
        remaining = deadline - time.monotonic()
        total_needed = sum(phase_times)
        if remaining < total_needed * 0.6:
            continue
        if remaining < 80:
            break

        scale = min(1.0, remaining * 0.9 / total_needed)
        phase_times = [t * scale for t in phase_times]

        print(f"\n=== n={n}, init={init_type}, phases={['%.0f' % t for t in phase_times]} ===")

        if init_type == 'parent' and parent_np is not None:
            a = parent_np.copy()
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.25)
        elif init_type == 'blackman':
            a = initial_blackman(n)
        elif init_type == 'flat_top':
            a = initial_flat_top(n)
        elif init_type == 'random':
            a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 6)))
        else:
            a = initial_raised_cosine(n)

        a = normalize_nonneg(a)
        init_score = score_sequence(a)
        print(f"    Init: {init_score:.6f}")

        # Phase 1: LP
        if phase_times[0] > 15:
            a, sc = lp_refine(a, time_limit_s=phase_times[0], verbose=True)
            print(f"    After LP: {sc:.6f}")

        # Phase 2: Multiplicative GD
        if phase_times[1] > 15:
            a, sc = multiplicative_gd_refine(a, time_limit_s=phase_times[1], verbose=True)
            print(f"    After MG: {sc:.6f}")

        # Phase 3: Top-K GD
        if phase_times[2] > 15:
            a, sc = topk_gradient_refine(a, time_limit_s=phase_times[2], verbose=True)
            print(f"    After TK: {sc:.6f}")

        # Phase 4: Local search
        if phase_times[3] > 10:
            a, sc = local_search_refine(a, time_limit_s=phase_times[3], rng=rng, verbose=True)
            print(f"    After LS: {sc:.6f}")

        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST: {sc:.6f}, n={n}, t={time.monotonic()-total_start:.0f}s")
        elif init_score < best_score:
            best_score = init_score
            best_seq = a.copy()

    # Final aggressive polishing of the best found
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n=== Final polish (remaining={remaining:.0f}s) ===")
            fractions = [0.25, 0.25, 0.2, 0.15, 0.15]
            times = [remaining * f for f in fractions]

            a, sc = lp_refine(best_seq, time_limit_s=times[0], verbose=True)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

            a, sc = multiplicative_gd_refine(best_seq, time_limit_s=times[1], verbose=True, t_start=50, t_end=300)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

            a, sc = topk_gradient_refine(best_seq, time_limit_s=times[2], verbose=True)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

            a, sc = local_search_refine(best_seq, time_limit_s=times[3], rng=rng, verbose=True)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

            a, sc = lp_refine(best_seq, time_limit_s=times[4], verbose=True)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

    if best_seq is None:
        best_seq = initial_raised_cosine(1000)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: {best_score:.6f}")

    # Ensure all values are valid for evaluate_sequence
    best_seq = np.maximum(0.0, best_seq)
    best_seq = np.minimum(1000.0, best_seq)

    print(f"\nFinal: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified: {verify_score:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
