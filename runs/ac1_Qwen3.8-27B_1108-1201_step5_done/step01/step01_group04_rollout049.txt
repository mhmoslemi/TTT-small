"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam optimizer with FFT-accelerated softmax gradient, diverse initializations,
multiple sequence lengths, and LP-based refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def compute_conv(a):
    """Compute full convolution of a with itself."""
    return np.convolve(a, a)

def compute_grad(a, tau=0.0):
    """Compute gradient of max(conv(a,a)) w.r.t. a using softmax.
    Uses FFT for efficiency."""
    n = len(a)
    c = np.convolve(a, a)
    if tau > 0:
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w /= w.sum()
    else:
        max_c = c.max()
        mask = (c >= max_c - 1e-12)
        w = mask.astype(float)
        w /= w.sum()
    
    # Cross-correlation: grad[j] = sum_q a[q] * w[j+q]
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    grad = np.fft.irfft(np.conj(A) * W, N)[:n]
    return grad

def normalize(a, target=None):
    """Normalize a to have specified sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (target / S)

def adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step with non-negativity projection and normalization."""
    n = len(a)
    target = np.sqrt(2.0 * n)
    
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, target)
    return a_new, m, v

def optimize_adam(a0, deadline, max_iters=5000):
    """Optimize a single sequence using Adam with adaptive softmax."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a)
    
    S_target = np.sqrt(2.0 * n)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Schedule: start with high tau (smooth), decrease to low tau (sharp)
    tau_start = 5.0
    tau_end = 50.0
    lr_start = 0.05
    lr_end = 0.001
    
    eval_interval = max(1, max_iters // 50)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        t += 1
        frac = it / max_iters
        tau = tau_start + (tau_end - tau_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac
        
        # Compute gradient
        grad = compute_grad(a, tau)
        
        # Remove mean component (for sum constraint)
        grad = grad - grad.mean()
        
        # Adam step
        a, m, v = adam_step(a, grad, m, v, t, lr)
        
        # Evaluate periodically
        if (it + 1) % eval_interval == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Final evaluation
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score

def lp_refine(a, M_target, deadline):
    """Solve LP to find g >= 0 maximizing sum(g) s.t. conv(a, g) <= M_target."""
    n = len(a)
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    # A[m, j] = a[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        A[m, lo:hi+1] = a[m - np.arange(lo, hi+1)]
    
    b = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': max(1.0, deadline - time.monotonic() - 5.0)})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_iteration(a, deadline, max_iters=30):
    """Iterative LP-based refinement."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        g = lp_refine(a, M, deadline)
        if g is None:
            break
        
        g = normalize(g, S_target)
        
        # Try different blend ratios
        improved = False
        for t_blend in [0.5, 0.3, 0.7, 0.2, 0.1]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_target)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_score

def make_chirp_init(n, freq=3.0, rng=None):
    """Create a chirp-like initialization."""
    i = np.arange(n, dtype=np.float64)
    # Quadratic chirp
    phase = 2 * np.pi * freq * (i / n) ** 2
    a = 0.5 + 0.5 * np.cos(phase)
    a = np.maximum(a, 0.01)
    return a

def make_modulated_init(n, base_freq=2.0, mod_freq=5.0, rng=None):
    """Create a modulated initialization."""

    base = np.cos(2 * np.pi * base_freq * i / n)
    mod = np.cos(2 * np.pi * mod_freq * i / n)
    a = 0.5 + 0.3 * base + 0.2 * mod
    a = np.maximum(a, 0.01)
    return a

def make_gaussian_init(n, center_frac=0.5, width_frac=0.3):

    width = width_frac * n
    a = np.exp(-0.5 * ((i - center) / width) ** 2)

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam optimizer with FFT-accelerated adaptive-softmax gradient, diverse
initializations across multiple sequence lengths, iterative LP refinement,
and restart-based local search to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_to(a, target):
    """Normalize a to have the specified sum."""
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a) / len(a) * target
    return a * (target / S)


def compute_grad(a, tau_norm=1.0):
    """Compute gradient of max(conv(a,a)) using normalized softmax.
    tau_norm controls sharpness relative to the signal range of c."""
    n = len(a)
    c = np.convolve(a, a)
    c_range = c.max() - c.min()
    if c_range < 1e-12:
        w = np.ones(len(c)) / len(c)
    else:
        tau = tau_norm / c_range
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum

    # Cross-correlation via FFT: grad[j] = sum_q a[q] * w[j+q]
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    grad = np.fft.irfft(np.conj(A) * W, N)[:n]
    return grad


def make_chirp_init(n, freq=3.0):
    """Quadratic chirp initialization."""
    i = np.arange(n, dtype=np.float64)
    phase = 2.0 * np.pi * freq * (i / n) ** 2
    a = 0.5 + 0.5 * np.cos(phase)
    return np.maximum(a, 0.01)


def make_modulated_init(n, f1=2.0, f2=5.0):
    """Modulated cosine initialization."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.3 * np.cos(2.0 * np.pi * f1 * i / n) + 0.2 * np.cos(2.0 * np.pi * f2 * i / n)
    return np.maximum(a, 0.01)


def make_gaussian_init(n, center_frac=0.5, width_frac=0.25):
    """Gaussian bump initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - center_frac * n) / (width_frac * n)) ** 2)
    return a


def make_random_init(n, rng):
    """Random non-negative initialization."""
    return rng.random(n) * 0.7 + 0.15


def optimize_adam(a0, deadline, max_iters=10000, tau_start=3.0, tau_end=40.0,
                  lr_start=0.05, lr_end=0.001, beta1=0.9, beta2=0.999):
    """Optimize using Adam with adaptive normalized softmax temperature."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    target = np.sqrt(2.0 * n)
    a = normalize_to(a, target)

    score = compute_obj(a)
    best = a.copy()
    best_score = score

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8
    eval_interval = max(1, max_iters // 60)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        t += 1
        frac = t / max_iters
        # Cosine schedule for tau and lr
        tau = tau_start + (tau_end - tau_start) * frac
        lr = lr_start + (lr_end - lr_start) * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, lr_end)

        grad = compute_grad(a, tau)
        # Remove mean for sum constraint
        grad = grad - grad.mean()

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = normalize_to(a, target)

        if (it + 1) % eval_interval == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def optimize_adam_restart(a0, deadline, n_restarts=3, max_iters_per=3000):
    """Adam with periodic restarts (perturb and retry if stuck)."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    target = np.sqrt(2.0 * n)
    a = normalize_to(a, target)

    best = a.copy()
    best_score = compute_obj(a)

    for restart in range(n_restarts):
        if time.monotonic() > deadline - 10.0:
            break

        seq, score = optimize_adam(a, deadline, max_iters=max_iters_per)
        if score < best_score:
            best_score = score
            best = seq.copy()
            a = seq.copy()
        else:
            # Perturb to escape
            n = len(a)
            noise = np.random.default_rng(best_score * 1000 + restart).random(n) * 0.1
            a = a + noise
            a = np.maximum(a, 0.0)
            a = normalize_to(a, target)

    return best, best_score


def solve_lp(f, M, time_limit=30.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 1024:
        return None

    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_refine_cycle(a, deadline, max_iters=25):
    """Iterative LP-based refinement (SOTA approach)."""
    n = len(a)
    target = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = np.convolve(a, a)
        M = float(c.max())
        remaining = deadline - time.monotonic() - 5.0
        if remaining < 3.0:
            break

        g = solve_lp(a, M, time_limit=min(remaining * 0.8, 60.0))
        if g is None:
            break

        g = normalize_to(np.maximum(g, 0.0), target)

        improved = False
        for t_blend in [0.5, 0.3, 0.7, 0.2, 0.1, 0.4, 0.6]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, target)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Small gradient nudge to escape
            grad = compute_grad(a, 10.0)
            grad = grad - grad.mean()
            a = a - 0.01 * grad
            a = np.maximum(a, 0.0)
            a = normalize_to(a, target)
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    return best, best_score
