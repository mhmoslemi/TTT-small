Minimize 2*n*max(conv(a,a))/sum(a)^2 using coordinate descent with exact 1D
sub-optimization at each coordinate. The key insight is that changing one
coordinate a[j] by delta affects only n convolution values (lags j..j+n-1),
with a known linear+quadratic structure, enabling fast exact minimization
over that coordinate.
"""
import numpy as np
import time


def _coord_descent(a, C, n, S, deadline, rng, verbose=True):

    no_improve = 0
