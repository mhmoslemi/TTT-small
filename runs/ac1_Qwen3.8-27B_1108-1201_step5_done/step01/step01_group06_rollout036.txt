Minimize 2*n*max_conv/sum^2 over non-negative sequences using log-sum-exp gradient
descent with FFT-accelerated full gradient computation and adaptive temperature.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_self_convolve(a):
    """Compute full self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

        if gnorm < 1e-15:

        grad = grad / gnorm  # Normalize gradient direction

        v_adam = beta2 * v_adam + (1 - beta2) * grad**2
        m_hat = m_adam / (1 - beta1**t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

import numpy as np

def fft_self_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_full_gradient(a, c, temp):
    """Compute full gradient of LSE(c/temp) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    # Softmax
    c_scaled = c / temp
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    softmax = exp_c / exp_c.sum()
    
    # Cross-correlation: cross[m] = sum_{j=0}^{n-1} a[j] * softmax[m+j]
    S_ext = np.zeros(fsize)
    S_ext[:size] = softmax
    cross = np.real(ifft(fft(S_ext, fsize) * np.conj(fft(a, fsize))))
    
    # Full gradient: (1/temp) * (cross[m] + softmax[2m]*a[m])
    correction = softmax[2 * np.arange(n)] * a
    grad = (cross[:n] + correction) / temp
    return grad


def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-10:
        return np.full(len(a), S / len(a))
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
1. Log-sum-exp gradient (FFT cross-correlation) with Adam optimizer and adaptive temperature
2. LP-based Frank-Wolfe with active-set linearization
3. Mirror descent with LSE gradient
Multiple initializations including the previous best (height_sequence_1).
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import lil_matrix
from scipy.optimize import linprog
from scipy.signal import fftconvolve


def fft_self_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_gradient(a, c, temp):
    """Full gradient of LSE(c/temp) w.r.t. a via FFT cross-correlation.
    dLSE/da[m] = (2/temp) * sum_j a[j] * softmax(c/temp)[m+j]
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    c_scaled = c / temp
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    softmax = exp_c / exp_c.sum()

    S_ext = np.zeros(fsize)
    S_ext[:size] = softmax
    cross = np.real(ifft(fft(S_ext, fsize) * np.conj(fft(a, fsize))))

    return 2.0 * cross[:n] / temp


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-10:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_lse_adam(a_init, n, S, max_time, rng, lr_init=0.1, temp_init=None, temp_end=None):
    """LSE gradient + Adam with adaptive temperature."""
    a = normalize_to_sum(a_init, S)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    if temp_init is None:
        temp_init = S * 0.3
    if temp_end is None:
        temp_end = S * 0.005

    c = fft_self_convolve(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    no_improve = 0

    while time.monotonic() - start < max_time:
        t += 1
        time_frac = min(1.0, (time.monotonic() - start) / max_time)
        temp = temp_init * (temp_end / temp_init) ** time_frac

        grad = compute_lse_gradient(a, c, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad ** 2
        m_hat = m_adam / (1 - beta1 ** t)
        v_hat = v_adam / (1 - beta2 ** t)

        lr = lr_init / (1.0 + 0.02 * t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, S)

        c = fft_self_convolve(a_new)
        cur = c.max()

        if cur < best_score:
            best_score = cur
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 800 and rng.random() < 0.03:
            noise = rng.normal(0, 0.03 * S / n, n)
            a = project_simplex(a + noise, S)
            no_improve = 0

    return best_a, best_score


def run_mirror_lse(a_init, n, S, max_time, rng, eta_init=0.3):
    """Mirror descent with LSE gradient."""
    a = normalize_to_sum(a_init, S)
    c = fft_self_convolve(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        t += 1
        time_frac = min(1.0, (time.monotonic() - start) / max_time)
        temp = S * 0.2 * (0.03 ** time_frac)

        grad = compute_lse_gradient(a, c, temp)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        eta = eta_init / (1.0 + 0.005 * t)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new * (S / a_new.sum())

        c = fft_self_convolve(a_new)
        cur = c.max()

        if cur < best_score:
            best_score = cur
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        if no_improve > 500 and rng.random() < 0.04:
            noise = rng.normal(0, 0.02 * S / n, n)
            a = project_simplex(a + noise, S)
            no_improve = 0

    return best_a, best_score


def run_lp_fw(a_init, n, S, max_time, rng, K=200, use_full=False):
    """LP-based Frank-Wolfe with active-set (or full) constraints."""
    a = normalize_to_sum(a_init, S)
    c = fft_self_convolve(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_self_convolve(a)
        M = c.max()
        if M < best_score:
            best_score = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            # Try a different active set or perturb
            if rng.random() < 0.1:
                noise = rng.normal(0, 0.02 * S / n, n)
                a = project_simplex(a + noise, S)
                no_improve = 0
            continue

        # Select active constraints
        if use_full and n <= 2048:
            active = np.arange(2 * n - 1)
        else:
            active = np.argsort(c)[::-1][:K]

        # Build sparse constraint matrix
        A_ub = lil_matrix((len(active), n))
        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_ub[idx, j] = a[k - j]
        A_ub = A_ub.tocsc()

        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break

        if not res.success or res.x is None:
            break
        g = res.x
        s_g = g.sum()
        if s_g <= 1e-10:
            break

        g = g * (S / s_g)
        gamma = 2.0 / (t + 2.0)
        a = (1.0 - gamma) * a + gamma * g

    return best_a, best_score
