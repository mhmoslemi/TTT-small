Combines FFT-based convolution, subgradient descent with adaptive step size,
LP-based refinement, and multi-start exploration across different lengths.

import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

    """Full linear convolution via FFT."""
    n_out = len(a) + len(b) - 1
    size = 1

    """Objective: 2*n*max(convolve(a,a))/(sum(a))^2."""

def gradient_fft(a, top_k=1, tau=None):
    """
    Efficient gradient computation using FFT.
    grad[i] = 2 * sum_k w[k] * a[k-i] where w[k] = 1 if k is argmax (or top-k).
    """
    n = len(a)
    c = fft_convolve(a, a)
    
    if tau is not None:
        # Soft-max weighting
        c_shift = c - c.max()
        w = np.exp(tau * c_shift)
        w /= w.sum()
    elif top_k == 1:
        w = np.zeros(len(c))
        w[int(c.argmax())] = 1.0
    else:
        idx = np.argsort(c)[-top_k:]
        w = np.zeros(len(c))
        w[idx] = 1.0 / top_k
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (where 0<=k-i<n)
    # This is 2 * (w * a_reversed) cross-correlated... 
    # Actually: sum_k w[k] * a[k-i] = (w conv a)[i]... no.
    # sum_k w[k] * a[k-i]: let j = k-i, then k = j+i, sum_j w[j+i] * a[j]
    # = sum_j a[j] * w[j+i] = (a conv w_reversed)[i]... 
    # Actually it's the cross-correlation of a and w, shifted.
    # Let me just use the direct formula: grad[i] = 2 * sum_{j: 0<=j<n, 0<=j+i<len(c)} a[j] * w[j+i]
    # = 2 * sum_j a[j] * w[j+i]
    # This is 2 * (a * w)[i] in the cross-correlation sense.
    # = 2 * ifft(fft(a) * conj(fft(w)))... hmm, let me think.
    
    # Cross-correlation: (a ⋆ w)[i] = sum_j a[j] * w[j+i]
    # In FFT: ifft(fft(a, N) * conj(fft(w, N))) gives circular cross-correlation.
    # For linear, we need N >= n + len(w) - 1.
    
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    fa = fft(a, N)
    fw = fft(w, N)
    corr = ifft(fa * np.conj(fw)).real[:n]
    grad = 2.0 * corr
    
    peak_val = float(c.max())
    return grad, peak_val, c


def lp_refine(f, M, n_max=2048):
    """
    LP: maximize sum(g) s.t. convolve(f,g) <= M (elementwise), g >= 0.
    Returns g or None.
    """
    n = len(f)
    if n > n_max:
        return None
    
    c_obj = -np.ones(n)
    
    # Constraint: for each m in [0, 2n-2], sum_j f[m-j] * g[j] <= M (where 0<=m-j<n)
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    # Use sparse representation for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            rows.append(m)
            cols.append(j)
            vals.append(f[m - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses momentum-based subgradient descent with adaptive peak selection,
LP-based refinement via HiGHS, and multi-start exploration across lengths.
Key improvements over prior approach: momentum, adaptive top-k peaks,
warm-started LP, asymmetric exploration, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_obj(a):
    """Compute objective: 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def grad_at_peak(a, c, top_k=1, tau=None):
    """
    Compute gradient of max(convolve(a,a)) w.r.t. a.
    If tau is given, use soft-max weighting over all lags.
    Otherwise use hard top-k.
    Returns (grad, peak_value).
    """
    n = len(a)

    if tau is not None:
        c_shift = c - c.max()
        w = np.exp(tau * c_shift)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c)
            w[int(c.argmax())] = 1.0
        else:
            w /= w_sum
    elif top_k == 1:
        w = np.zeros(len(c))
        w[int(c.argmax())] = 1.0
    else:
        idx = np.argsort(c)[-top_k:]
        w = np.zeros(len(c))
        w[idx] = 1.0 / top_k

    grad = np.zeros(n)
    for i in range(n):
        # grad[i] = 2 * sum_k w[k] * a[k-i] where 0 <= k-i < n
        k_lo = max(0, i)
        k_hi = min(len(c) - 1, i + n - 1)
        if k_lo <= k_hi:
            j_lo = k_lo - i
            j_hi = k_hi - i
            grad[i] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[j_lo:j_hi + 1])

    peak_val = float(c.max())
    return grad, peak_val


def lp_refine(f, M, n_max=1536):
    """
    LP: maximize sum(g) s.t. convolve(f,g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    if n > n_max:
        return None

    c_obj = -np.ones(n)

    rows = []
    cols = []
    vals = []
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            rows.append(m)
            cols.append(j)
            vals.append(f[m - j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def optimize_momentum(a0, deadline, max_iters=8000, use_lp=True, lp_n_max=1024):
    """
    Optimize using momentum-based subgradient descent, then LP refinement.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a)
    S_t = np.sqrt(2.0 * n)

    score = compute_obj(a)
    best = a.copy()
    best_score = score

    # Momentum parameters
    momentum = 0.5
    velocity = np.zeros(n)

    # Phase 1: Momentum subgradient descent
    lr_init = 0.2
    lr_min = 0.001
    stamina = 0
    stagnation_limit = 200

    for it in range(max_iters):
        if time.monotonic() > deadline - 60.0:
            break

        c = np.convolve(a, a)

        # Adaptive top_k: start focused, become broader
        frac = it / max(max_iters, 1)
        if frac < 0.3:
            top_k = 1
        elif frac < 0.6:
            top_k = 3
        else:
            top_k = 5

        grad, peak_val = grad_at_peak(a, c, top_k=top_k)

        # Project out mean (fixed-sum constraint)
        grad -= grad.mean()

        # Momentum update
        velocity = momentum * velocity + grad
        grad_mom = velocity

        # Learning rate schedule
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, lr_min)

        a_new = a - lr * grad_mom
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)

        s_new = compute_obj(a_new)

        if s_new < score:
            a = a_new
            score = s_new
            stamina = 0
            if score < best_score:
                best_score = score
                best = a.copy()
        else:
            stamina += 1
            # Reduce learning rate if stagnating
            if stamina > stagnation_limit:
                lr_init *= 0.8
                momentum = min(momentum + 0.05, 0.8)
                stamina = 0
            # Still move (smaller step)
            a_new = a - lr * 0.5 * grad_mom
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            a = a_new

    # Phase 2: LP-based refinement
    if use_lp and n <= lp_n_max and time.monotonic() < deadline - 30.0:

        M = float(c.max())
        lp_stagnation = 0

        for lp_it in range(200):
            if time.monotonic() > deadline - 15.0:
                break

            g = lp_refine(a, M, n_max=n)
            if g is None:
                break

            improved = False
            for t in [0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0.0)
                a_try = normalize(a_try, S_t)
                s_try = compute_obj(a_try)
                if s_try < best_score:
                    a = a_try
                    best = a.copy()
                    best_score = s_try
                    score = s_try
                    improved = True
                    lp_stagnation = 0
                    break

            if not improved:
                lp_stagnation += 1
                if lp_stagnation > 5:
                    # Escape: small gradient step
                    c = np.convolve(a, a)
                    grad, _ = grad_at_peak(a, c, top_k=1)
                    grad -= grad.mean()
                    a = a - 0.02 * grad
                    a = np.maximum(a, 0.0)
                    a = normalize(a, S_t)
                    score = compute_obj(a)
                    if score < best_score:
                        best_score = score
                        best = a.copy()
                    lp_stagnation = 0
                    if lp_stagnation > 20:
                        break

            c = np.convolve(a, a)
            M = float(c.max())

    # Phase 3: Fine-tuning
    if time.monotonic() < deadline - 10.0:
        for ft in range(3000):
            if time.monotonic() > deadline - 5.0:
                break
            c = np.convolve(a, a)
            grad, _ = grad_at_peak(a, c, top_k=1)
            grad -= grad.mean()
            a_new = a - 0.005 * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < score:
                a = a_new
                score = s_new
                if score < best_score:
                    best_score = score
                    best = a.copy()
            else:
                a_new = a - 0.002 * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                a = a_new

    return best, best_score


def optimize_soft(a0, deadline, max_iters=5000):

        a_new = np.maximum(a_new, 0.0)

def coordinate_descent(a, deadline, max_iters=5000):
    """
    Coordinate descent: try adjusting pairs of elements to reduce peak.
    Focus on elements that contribute to the peak convolution value.
    """
    n = len(a)
    S_t = a.sum()
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        k_star = int(c.argmax())
        peak_val = float(c[k_star])

        # Find elements contributing to the peak
        contributors = []
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i
            contributors.append((i, j, a[i] * a[j]))

        # Sort by contribution
        contributors.sort(key=lambda x: -x[2])

        # Try adjusting top contributors
        improved = False
        for i, j, contrib in contributors[:5]:
            if time.monotonic() > deadline:
                break

            # Try reducing a[i] and increasing a[j] (or vice versa)
            for delta in [0.01, 0.05, 0.1]:
                if a[i] < delta:
                    continue
                a_try = a.copy()
                if i != j:
                    a_try[i] -= delta
                    a_try[j] += delta
                    a_try = np.maximum(a_try, 0.0)
                    a_try = normalize(a_try, S_t)
                    s_try = compute_obj(a_try)
                    if s_try < best_score:
                        a = a_try
                        best = a.copy()
                        best_score = s_try
                        score = s_try
                        improved = True
                        break

        if not improved and it % 100 == 0:
            # Random perturbation to escape
            idx = np.random.randint(0, n)
            a_try = a.copy()
            a_try[idx] *= 0.9
            a_try = normalize(a_try, S_t)
            s_try = compute_obj(a_try)
            if s_try < best_score * 1.01:
                a = a_try
                score = s_try

    return best, best_score

    starts.append((i / n + 0.1, 'full', 'linear_inc'))
    # Two bumps (asymmetric placement)
    starts.append((np.exp(-((i - n * 0.3) / (n * 0.15)) ** 2) +
                   0.5 * np.exp(-((i - n * 0.7) / (n * 0.2)) ** 2) + 0.01,
                   'full', 'two_bump'))
    # Cosine (symmetric, for comparison)
    starts.append((0.5 * (1 + np.cos(2 * np.pi * i / n)) + 0.01, 'full', 'cos'))
    # Gaussian centered
    starts.append((np.exp(-0.5 * ((i - n / 2) / (n * 0.25)) ** 2) + 0.01, 'full', 'gauss'))

    # Longer sequences
    for n in [768, 1024, 2048]:
        starts.append((rng.random(n) + 0.05, 'full', f'rand_long_{n}'))

    # Sort: shorter first for faster results
    starts.sort(key=lambda x: len(x[0]))

    print(f"=== Starting optimization: {len(starts)} inits, budget={budget_s:.0f}s ===")
    print(f"Uniform n=512 baseline: {compute_obj(np.ones(512)):.4f}")

    # Run all starts
    for a0, method, label in starts:
        if time.monotonic() > deadline - 50.0:
            break
        track(a0, method, label)

    # --- Final refinement of best sequence ---
    if best_seq is not None and time.monotonic() < deadline - 20.0:
        print(f"\n=== Final refinement (score={best_score:.6f}, n={len(best_seq)}) ===")

        # Multiple passes with decreasing aggressiveness
        for pass_i in range(4):
            if time.monotonic() > deadline - 10.0:
                break
            n = len(best_seq)
            use_lp = n <= 768
            sub_deadline = deadline - 5.0
            seq, score = optimize_momentum(best_seq, sub_deadline,
                                           max_iters=2000,
                                           use_lp=use_lp,
                                           lp_n_max=768)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine pass {pass_i}: {best_score:.6f}")
            else:
                # Try coordinate descent as alternative
                if time.monotonic() < deadline - 10.0:
                    seq, score = coordinate_descent(best_seq, deadline - 5.0, max_iters=2000)
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
                        print(f"  Coord pass {pass_i}: {best_score:.6f}")
                    break

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            es = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {es:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    print("Fallback: uniform sequence n=256")
    return [1.0] * 256
