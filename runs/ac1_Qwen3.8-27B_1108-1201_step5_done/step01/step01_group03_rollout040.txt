import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def fast_score(f):
    """Compute 2*n*max(conv)/sum^2"""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] for valid indices."""
    rows = []
    cols = []
    data = []
    for j in range(n):
        for k in range(2*n-1):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                data.append(f[idx])
    A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A

def solve_lp_min_peak(f, time_limit=5.0):
    """Solve min t s.t. (f*g)[k] <= t for all k, sum(g) = 1, g >= 0.
    Returns (g, t) or (None, None)."""
    n = len(f)
    A = build_conv_matrix(n, f)
    # Variables: g[0],...,g[n-1], t
    # Objective: minimize t => c = [0,...,0, 1]
    # Constraints: A*g - t <= 0 => A_aug * [g; t] <= 0
    #              sum(g) = 1
    #              g >= 0, t free
    A_aug = np.hstack([A.toarray(), -np.ones((2*n-1, 1))])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    b = np.zeros(2*n-1)
    # Equality: sum(g) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_aug, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        t = res.x[n]
        gs = g.sum()
        if gs > 1e-10:
            g = g / gs
        return g, float(t)
    return None, None

def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] for 0<=k-j<n."""
    # For each j, the valid k values are j to j+n-1

def solve_lp_min_peak(f, time_limit=10.0):

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Use bisection on the threshold
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_subgradient(f):
    """Compute a subgradient of max_k (f*f)[k] at f.
    Returns the subgradient vector (before projection)."""
    conv = np.convolve(f, f)
    max_val = conv.max()
    # Find all indices where conv achieves max (within tolerance)
    tol = 1e-10 * max_val
    K = np.where(np.abs(conv - max_val) <= tol)[0]
    # Subgradient: sg[i] = (1/|K|) sum_{k in K} 2*f[k-i] * 1_{0<=k-i<n}
    n = len(f)
    sg = np.zeros(n)
    for k in K:
        # For this k, the gradient of (f*f)[k] w.r.t. f is:
        # [grad]_i = 2 * f[k-i] if 0 <= k-i < n, else 0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # i ranges from lo to hi, and k-i ranges from k-hi to k-lo
        # f[k-i] for i in [lo, hi] = f[k-lo : k-hi+1] reversed... 
        # Actually: for i = lo, k-i = k-lo; for i = hi, k-i = k-hi
        # So f[k-i] = f[k-hi : k-lo+1] but in reverse order
        # Let's just do it directly
        idx = k - np.arange(lo, hi + 1)  # this gives k-lo, k-lo-1, ..., k-hi
        sg[lo:hi+1] += 2.0 * f[idx]
    sg /= len(K)
    return sg


def subgradient_step(f, alpha, f_old_conv_max):
    """One subgradient step with projected update."""
    sg = compute_subgradient(f)
    # Center the subgradient (for the sum constraint)
    sg_centered = sg - sg.mean()
    v = f - alpha * sg_centered
    f_new = simplex_project(v)
    return f_new

        # Blend
        blend = 0.3
        new_block = (1 - blend) * block + blend * new_block
        
        f_new = f.copy()
        f_new[start:end] = new_block
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            continue

        new_sc = fast_score(f_new)
        delta = new_sc - cur_sc

            if cur_sc < best_sc:

                best = f.copy()
                print("   SA improvement: %.6f (it=%d, T=%.4f)" % (best_sc, it, T), flush=True)

    # = 2 * (p cross-correlated with f)[i] in a specific sense
    # Actually this is: (np.convolve(p, f[::-1]))[i + n - 1] * 2? Let me just do it directly
    # For efficiency, use the full convolution approach:
    # grad[i] = 2 * sum_{j=0}^{n-1} f[j] * p[i+j]
    # This is the cross-correlation: (f * p) at offset i
    # np.correlate(p, f, mode='valid') gives sum_j f[j]*p[i+j] for i=0..len(p)-n
    # len(p) = 2n-1, so valid gives n values. 
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    return grad, conv


def lse_descent(f0, budget_s, lam=50.0, alpha_init=0.005, alpha_decay=0.9995,
                eval_every=20, lam_growth=1.001, lam_max=500.0):
    """Log-sum-exp smooth descent on max(f*f) with simplex constraint."""
    f = f0.copy()
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    alpha = alpha_init
    lam_cur = lam
    stall = 0
    
    for it in range(500000):
        if time.monotonic() > budget_s:
            break
        
        grad, conv = smooth_gradient(f, lam_cur)
        # Center gradient for sum constraint
        grad_centered = grad - grad.mean()
        v = f - alpha * grad_centered
        f_new = simplex_project(v)
        f = f_new
        
        alpha = max(1e-6, alpha * alpha_decay)
        lam_cur = min(lam_max, lam_cur * lam_growth)
        
        if it % eval_every == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if stall > 50:
                # Restart with perturbation
                rng = np.random.default_rng(42 + it)
                f = f + rng.normal(0, 0.02, n)
                f = np.maximum(f, 0)
                f = f / f.sum()
                stall = 0
    
    return best, best_sc


def lp_search_with_line_search(f0, budget_s, lp_time=10.0, eval_every=3):
    """LP-based search with line search on step size."""
    f = f0.copy()
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    
    for it in range(50000):
        if time.monotonic() > budget_s:
            break
        g, t = solve_lp_min_peak(f, time_limit=lp_time)
        if g is None:
            break
        
        # Line search: try different step sizes
        best_alpha = 0
        best_f = f.copy()
        best_s = best_sc_local = fast_score(f)
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            f_trial = (1 - alpha) * f + alpha * g
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            sc = fast_score(f_trial)
            if sc < best_s:
                best_s = sc
                best_alpha = alpha
                best_f = f_trial.copy()
        
        if best_alpha > 0:
            f = best_f
            if best_s < best_sc:
                best_sc = best_s
                best = f.copy()
                print("    LP line search: %.6f (alpha=%.2f, it=%d)" % (best_sc, best_alpha, it), flush=True)
        else:
            # No improvement; try random perturbation
            rng = np.random.default_rng(1000 + it)
            w = int(rng.integers(1, max(2, n // 20) + 1))
            start = int(rng.integers(0, n - w + 1))
            f_pert = f.copy()
            f_pert[start:start+w] *= np.exp(rng.normal(0, 0.2, w))
            f_pert = np.maximum(f_pert, 0)
            s = f_pert.sum()
            if s > 1e-10:
                f = f_pert / s
    
    return best, best_sc


def simulated_annealing(f0, budget_s, T_init=0.03, T_min=0.00005, cooling=0.9997):
    """Simulated annealing with block redistribution moves."""
    f = f0.copy()
    f = f / f.sum()
    n = len(f)
    cur_sc = fast_score(f)
    best = f.copy()
    best_sc = cur_sc
    T = T_init
    rng = np.random.default_rng(99999)
    accept_count = 0
    
    for it in range(1000000):
        if time.monotonic() > budget_s:
            break
        
        # Random block move
        block_size = int(rng.integers(1, max(2, n // 8) + 1))
        start = int(rng.integers(0, n - block_size + 1))
        end = min(start + block_size, n)
        
        block = f[start:end]
        block_sum = block.sum()
        if block_sum < 1e-12:
            T *= cooling
            continue
        
        # Dirichlet perturbation
        concentration = 0.5
        weights = rng.dirichlet(concentration * np.ones(end - start))
        new_block = (1.0 - 0.4) * block + 0.4 * weights * block_sum
        new_block = np.maximum(new_block, 0)
        bs2 = new_block.sum()
        if bs2 < 1e-12:
            continue
        new_block = new_block / bs2 * block_sum
        
        f_new = f.copy()
        f_new[start:end] = new_block
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        
        new_sc = fast_score(f_new)
        delta = new_sc - cur_sc
        
        if delta < 0 or (T > T_min and rng.random() < np.exp(-max(0, -delta) / T) if delta > 0 else True):
            # Accept
            f = f_new
            cur_sc = new_sc
            accept_count += 1
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = f.copy()
                if accept_count % 50 == 1:
                    print("    SA: %.6f (it=%d, T=%.5f, accepts=%d)" % (best_sc, it, T, accept_count), flush=True)
        
        T *= cooling
    
    return best, best_sc


def make_initial(n, rng, method='uniform'):
    """Generate non-negative sum-normalized initial sequence."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'two_peaks':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 16)
        a = (np.exp(-0.5 * ((x - n / 4.0) / s) ** 2) +
             np.exp(-0.5 * ((x - 3 * n / 4.0) / s) ** 2))
    elif method == 'gabor':
        x = np.arange(n, dtype=np.float64)
        center = n / 2.0
        width = n / 4.0
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif method == 'plateau':
        a = np.ones(n)
        center = n // 2
        half = n // 5
        a[center - half:center + half] *= 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def combined_local_search(f0, budget_s):
    """Alternate between LP and LSE descent for local optimization."""
    f = f0.copy()
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    
    phase = 0
    stall = 0
    prev_sc = best_sc
    
    for it in range(200000):
        if time.monotonic() > budget_s:
            break
        
        if phase == 0:
            # LP step
            g, t = solve_lp_min_peak(f, time_limit=8.0)
            if g is not None:
                f_trial = 0.5 * f + 0.5 * g
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s > 1e-10:
                    f_trial = f_trial / s
                    f = f_trial
            phase = 1
        else:
            # LSE descent steps (do 50 at a time)
            lam = 30.0 + it * 0.01
            alpha = 0.003 * max(0.1, 1.0 - it / 50000.0)
            for _ in range(50):
                grad, conv = smooth_gradient(f, min(lam, 300.0))
                grad_c = grad - grad.mean()
                v = f - alpha * grad_c
                f = simplex_project(v)
            phase = 0
        
        sc = fast_score(f)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best = f.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 100:
            # Perturb to escape
            rng = np.random.default_rng(777 + it)
            f = f + rng.normal(0, 0.03, n)
            f = np.maximum(f, 0)
            f = f / f.sum()
            stall = 0
    
    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Search the best non-negative sequence minimizing the evaluation score."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    
    # Get parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = None
    
    if parent is not None and parent.ndim == 1 and parent.size > 0:
        if np.isfinite(parent).all() and parent.sum() > 0.01:
            best = parent.copy()
            bs = float(evaluate_sequence(best.tolist()))
            print("Parent: score=%.6f n=%d" % (bs, best.size), flush=True)
        else:
            best = np.ones(512)
            bs = 2.0
    else:
        best = np.ones(512)
        bs = 2.0
        print("No parent; using uniform n=512", flush=True)
    
    # Phase 1: Multi-start LP + line search
    print("\n=== Phase 1: LP multi-start ===", flush=True)
    configs = [
        (512, 'uniform'),
        (512, 'cosine'),
        (512, 'sin2'),
        (512, 'gabor'),
        (1024, 'uniform'),
        (1024, 'cosine'),
        (1024, 'two_peaks'),
        (2048, 'uniform'),
        (2048, 'cosine'),
    ]
    # Add parent length if reasonable
    if parent is not None and 64 <= parent.size <= 4096:
        pl = int(parent.size)
        if pl not in [c[0] for c in configs]:
            configs.append((pl, 'uniform'))
    
    phase1_time = 0.55 * float(budget_s)
    time_per = min(90.0, phase1_time / len(configs))
    
    for (n, method) in configs:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        if method == 'uniform' and n == (parent.size if parent is not None else -1) and parent is not None:
            f0 = np.maximum(parent, 0).copy()
            f0 = f0 / f0.sum()
        else:
            f0 = make_initial(n, rng, method)
        
        local_deadline = min(deadline, time.monotonic() + time_per)
        print("  LP n=%d %s (%.0fs)" % (n, method, time_per), flush=True)
        c_best, c_sc = lp_search_with_line_search(f0, local_deadline)
        if np.isfinite(c_sc) and c_sc < bs:
            sc2 = float(evaluate_sequence(c_best.tolist()))
            if np.isfinite(sc2) and sc2 < bs:
                bs = sc2
                best = c_best.copy()
                print("  -> NEW BEST %.6f (n=%d)" % (bs, best.size), flush=True)
    
    # Phase 2: Combined local search on best
    print("\n=== Phase 2: Combined LP+LSE on best (n=%d) ===" % best.size, flush=True)
    f2 = best.copy()
    f2 = f2 / f2.sum()
    phase2_deadline = deadline - 0.30 * float(budget_s)
    c_best, c_sc = combined_local_search(f2, phase2_deadline)
    if np.isfinite(c_sc) and c_sc < bs:
        sc2 = float(evaluate_sequence(c_best.tolist()))
        if np.isfinite(sc2) and sc2 < bs:
            bs = sc2
            best = c_best.copy()
            print("  -> NEW BEST %.6f" % bs, flush=True)
    
    # Phase 3: LSE descent with growing lambda
    print("\n=== Phase 3: LSE descent ===", flush=True)
    f3 = best.copy()
    f3 = f3 / f3.sum()
    phase3_deadline = deadline - 0.15 * float(budget_s)
    c_best, c_sc = lse_descent(f3, phase3_deadline, lam=20.0, alpha_init=0.003,
                               alpha_decay=0.9997, lam_growth=1.002, lam_max=800.0)
    if np.isfinite(c_sc) and c_sc < bs:
        sc2 = float(evaluate_sequence(c_best.tolist()))
        if np.isfinite(sc2) and sc2 < bs:
            bs = sc2
            best = c_best.copy()
            print("  -> NEW BEST %.6f" % bs, flush=True)
    
    # Phase 4: Simulated annealing
    print("\n=== Phase 4: Simulated Annealing ===", flush=True)
    f4 = best.copy()
    f4 = f4 / f4.sum()
    c_best, c_sc = simulated_annealing(f4, deadline)
    if np.isfinite(c_sc) and c_sc < bs:
        sc2 = float(evaluate_sequence(c_best.tolist()))
        if np.isfinite(sc2) and sc2 < bs:
            bs = sc2
            best = c_best.copy()
            print("  -> NEW BEST %.6f" % bs, flush=True)
    
    # Final validation
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(512)
    
    final_sc = float(evaluate_sequence(best.tolist()))
    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=%.6f n=%d elapsed=%.0fs" % (final_sc, best.size, elapsed), flush=True)
    return [float(x) for x in best]
