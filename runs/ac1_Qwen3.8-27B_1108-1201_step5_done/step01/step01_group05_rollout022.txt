"""
LP-based step function optimization for minimizing max autocorrelation.

Uses iterative LP best-response (min t s.t. (p*g)_k <= t, sum(g)=1, g>=0)
with line-search blending, multiple restarts, and gradient polishing.
Targets n ~ 500-800 where LP is tractable and autocorrelation can be very flat.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(max(1, p.size), 1.0 / max(1, p.size), dtype=np.float64)
    return p / s

def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    c = np.convolve(p, p)
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return np.inf
    return float(2.0 * n * cmax / (s * s))

def precompute_conv_structure(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        cnt = j_end - j_start + 1
        rows_list.append(np.full(cnt, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))
    rows = np.concatenate(rows_list) if rows_list else np.array([], dtype=np.int32)
    cols = np.concatenate(cols_list) if cols_list else np.array([], dtype=np.int32)
    return rows, cols

def build_conv_data(p, rows, cols):
    return p[rows - cols]

def solve_lp_best_response(p, rows, cols):
    n = len(p)
    data = build_conv_data(p, rows, cols)
    A_conv = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub_dense = np.zeros((2 * n - 1, n + 1))
    A_ub_dense[:, :n] = np.asarray(A_conv.todense())
    A_ub_dense[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub_dense, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return p.copy(), np.inf
    
    if res.status != 0 or res.x is None:
        return p.copy(), np.inf
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    t = float(res.x[n])
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return p.copy(), t
    g = g / s
    return g, t

def line_search_blend(p, g, n_evals=12):
    best_alpha = 0.0
    best_score = internal_score(p)
    alphas = np.linspace(0.05, 1.0, n_evals)
    for alpha in alphas:
        q = safe_normalize((1.0 - alpha) * p + alpha * g)
        sc = internal_score(q)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_alpha = alpha
    if best_alpha > 0:
        # Refine around best
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for alpha in np.linspace(lo, hi, 8):
            q = safe_normalize((1.0 - alpha) * p + alpha * g)
            sc = internal_score(q)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_alpha = alpha
    return best_alpha, best_score

def lp_optimize(p_init, time_limit, rng, max_restarts=8, verbose=True):
    p = safe_normalize(np.asarray(p_init, dtype=np.float64))
    n = int(p.size)
    rows, cols = precompute_conv_structure(n)
    
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    
    end = time.monotonic() + max(0.0, time_limit)
    total_iters = 0
    
    for restart in range(max_restarts):
        if time.monotonic() > end - 5.0:
            break
        
        iters_since_improve = 0
        p_score = current_score
        
        while time.monotonic() < end - 5.0:
            total_iters += 1
            iters_since_improve += 1
            
            if iters_since_improve > 200:
                break
            
            g_star, t_star = solve_lp_best_response(p, rows, cols)
            
            if not np.all(np.isfinite(g_star)):
                break
            
            alpha, new_score = line_search_blend(p, g_star)
            
            if alpha > 0 and np.isfinite(new_score) and new_score < p_score - 1e-12:
                p = safe_normalize((1.0 - alpha) * p + alpha * g_star)
                p_score = new_score
                iters_since_improve = 0
                
                if new_score < best_score:
                    best_score = new_score
                    best_p = p.copy()
                    if verbose and total_iters % 20 == 1:
                        print("    LP iter %d: score=%.10f (alpha=%.3f)" % (total_iters, best_score, alpha), flush=True)
            else:
                break
        
        if time.monotonic() > end - 10.0:
            break
        
        # Perturb for next restart
        if restart < max_restarts - 1:
            sigma = 0.005 + 0.003 * restart
            p = safe_normalize(best_p + sigma * rng.standard_normal(n))
            p_score = internal_score(p)
            current_score = p_score
    
    return best_p, best_score

def gradient_polish(p_init, time_limit, rng):
    p = safe_normalize(np.asarray(p_init, dtype=np.float64))
    n = int(p.size)
    best_p = p.copy()
    best_score = internal_score(best_p)
    curr_score = best_score
    
    end = time.monotonic() + max(0.0, time_limit)
    T = max(0.005 * n / (2 * n - 1), 1e-8)
    momentum = np.zeros(n)
    lr_base = 0.5
    mu = 0.9
    iters = 0
    stuck = 0
    
    while time.monotonic() < end - 2.0:
        iters += 1
        c = np.convolve(p, p)
        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax <= 0:
            break
        
        T = max(T * 0.9995, 1e-10)
        diff = (c - cmax) / T
        e = np.exp(diff - diff.max())
        w = e / e.sum()
        
        # Gradient of smoothed max: grad_j = 2 * sum_k w_k * p[k-j] (for valid k-j)
        # = 2 * (w_reversed * p)[j] restricted to valid range
        # Using FFT for speed
        from numpy.fft import fft, ifft
        L = 1
        while L < 3 * n - 2:
            L <<= 1
        Fp = fft(p, L)
        Fw = fft(w[::-1], L)
        h = np.real(ifft(Fp * Fw))
        
        # h[j] = sum_k Fp_k * Fw_k at position j (circular)
        # We need: grad_j = 2 * sum_{k: 0<=k-j<=n-1} w_k * p[k-j]
        # = 2 * sum_{m=0}^{n-1} w_{j+m} * p[m] (where j+m < 2n-1)
        # This is the cross-correlation of w and p
        # Using the FFT result: the linear cross-correlation is in h
        if n == 1:
            G = np.array([2.0 * float(np.sum(w)) * p[0]])
        else:
            # The full linear convolution of p and w has length 2n-1
            # We need the "anti-convolution": sum_k w_k * p[k-j]
            # This equals the cross-correlation
            # From h (which is ifft(fft(p)*fft(w[::-1]))), we get circular correlation
            # The linear part is:
            # For j in 0..n-1: sum_{k=j}^{j+n-1} w_k * p[k-j] (with w_k=0 if k>=2n-1)
            G = np.zeros(n)
            for j in range(n):
                k_min = j
                k_max = min(2 * n - 2, j + n - 1)
                if k_min <= k_max:
                    G[j] = 2.0 * float(np.dot(w[k_min:k_max+1], p[0:k_max-k_min+1]))
        
        if not np.all(np.isfinite(G)):
            G = np.zeros(n)
        
        gnorm = float(np.linalg.norm(G))
        if gnorm < 1e-15:
            break
        
        G = G / gnorm
        momentum = mu * momentum - lr_base * G
        v = p + momentum
        v = np.maximum(v, 0)
        vs = float(v.sum())
        if vs < 1e-15:
            v = p
            vs = 1.0
        v = v / vs
        
        sc = internal_score(v)
        if np.isfinite(sc) and sc < curr_score:
            curr_score = sc
            if sc < best_score:
                best_score = sc
                best_p = v.copy()
                stuck = 0
            p = v
        else:
            stuck += 1
            lr_base *= 0.9
            if stuck > 500:
                p = safe_normalize(best_p + 0.002 * rng.standard_normal(n))
                curr_score = internal_score(p)
                momentum *= 0.5
                lr_base = 0.3
                stuck = 0
        
        if best_score < 1.4990:
            break
    
    return best_p, best_score

def resize_sequence(p, n_target):
    p = np.asarray(p, dtype=np.float64)
    n_src = p.size
    if n_src == n_target:
        return p.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    p_resized = np.interp(x_tgt, x_src, p)
    return safe_normalize(p_resized)

def make_parametric(n, kind, params, rng):
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        p = np.ones(n)
    elif kind == 'gauss':
        mu, sig = params
        p = np.exp(-0.5 * ((x - mu) / sig) ** 2)
    elif kind == 'cosine':
        a, f, phi = params
        p = 1.0 + a * np.cos(2 * np.pi * f * x + phi)
    elif kind == 'sinc_like':
        a = params[0]
        p = np.abs(np.sin(np.pi * x) / (np.pi * x + 1e-10)) ** a + 0.01
    elif kind == 'beta':
        alpha, beta = params
        p = (x + 1e-8) ** alpha * (1 - x + 1e-8) ** beta
    elif kind == 'multi_cos':
        p = 1.0
        for freq, amp, ph in params:
            p += amp * np.cos(2 * np.pi * freq * x + ph)
    elif kind == 'random_smooth':
        p = 1.0
        for _ in range(3):
            f = rng.uniform(1, n / 20)
            a = rng.uniform(0.1, 0.5)
            ph = rng.uniform(0, 2 * np.pi)
            p += a * np.sin(2 * np.pi * f * x + ph)
    elif kind == 'chebyshev':
        z = 2 * x - 1
        p = np.ones(n)
        T_prev = np.ones(n)
        T_curr = z.copy()
        for d in range(1, min(8, n) + 1):
            coeff = params[min(d, len(params)-1)] if d < len(params) else 0.0
            p += coeff * T_curr
            T_new = 2 * z * T_curr - T_prev
            T_prev = T_curr
            T_curr = T_new
    else:
        p = np.ones(n)
    
    p = np.maximum(p, 0.001)
    return safe_normalize(p)
