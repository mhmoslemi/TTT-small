"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Combines smoothed gradient descent (log-sum-exp with temperature annealing),
LP-based directional search, and adaptive local random search.
Uses FFT-accelerated gradient computation and multi-length/multi-init exploration.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    """Compute the evaluation score directly."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    """Normalize to non-negative with sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x))


def initial_tukey(n, r=0.5):
    """Tukey (tapered cosine) window."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    taper = int(r * (n - 1) // 2)
    if taper > 0:
        for i in range(taper):
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * i / (r * (n - 1)) - 1.0)))
            a[n - 1 - i] = a[i]
    return a


def random_bump_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 10), max(3, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def compute_smoothed_gradient(a, tau):
    """
    Compute gradient of smoothed objective (1/tau)*log(sum exp(tau*c_k)).
    Returns (gradient, c_max).
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    # Numerically stable log-sum-exp
    c_shifted = c - c_max
    exp_c = np.exp(tau * c_shifted)
    w = exp_c / exp_c.sum()
    # Gradient: grad[j] = sum_k w[k] * a[k-j]
    # Use convolution: grad = convolve(w, a[::-1])[n-1:2n-1]
    a_rev = a[::-1]
    conv_res = np.convolve(w, a_rev)
    grad = conv_res[n - 1: 2 * n - 1]
    return grad, c_max, c


def smoothed_gradient_descent(a, time_limit_s, verbose=False, max_iters=500000):
    """
    Minimize max autocorrelation using log-sum-exp smoothed gradient descent
    with multiplicative updates and adaptive step size.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    # Temperature schedule
    temperature = max(1.0, n * 0.01)  # Start with smooth objective
    temperature_min = 0.01
    temp_decay = 0.998

    # Step size parameters
    base_eta = 0.3
    no_improve_count = 0
    stagnation_threshold = 500

    it = 0
    while time.monotonic() - start_time < time_limit_s and it < max_iters:
        tau = 1.0 / max(temperature, 1e-10)
        grad, c_max, c = compute_smoothed_gradient(a, tau)

        # Adaptive step size
        grad_norm_sq = np.dot(grad, a * grad)  # Weighted norm
        if grad_norm_sq < 1e-20:
            eta = 0.0
        else:
            eta = base_eta * c_max / (grad_norm_sq + 1e-15)
            eta = min(eta, 1.0)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = normalize_nonneg(a_new)

        score = score_sequence(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve_count = 0
        else:
            # Try smaller step
            a_new2 = a * np.exp(-eta * 0.05 * grad)
            a_new2 = normalize_nonneg(a_new2)
            score2 = score_sequence(a_new2)
            if score2 < best_score - 1e-12:
                best_score = score2
                best_a = a_new2.copy()
                a = a_new2.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
                # Reduce base_eta on stagnation
                if no_improve_count % 100 == 0:
                    base_eta *= 0.8
                    # Perturb to escape local minimum
                    if no_improve_count % 500 == 0 and no_improve_count > stagnation_threshold:
                        perturb = 1.0 + 0.05 * np.random.randn(n)
                        a_pert = a * perturb
                        a_pert = normalize_nonneg(a_pert)
                        score_pert = score_sequence(a_pert)
                        if score_pert < best_score:
                            best_score = score_pert
                            best_a = a_pert.copy()
                        a = a_pert.copy()
                        base_eta = 0.3
                        no_improve_count = 0

        temperature *= temp_decay
        temperature = max(temperature, temperature_min)
        it += 1

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"      SGD iter {it}: score={best_score:.6f}, temp={temperature:.4f}, eta={base_eta:.4f}, elapsed={elapsed:.1f}s")

    if verbose:
        print(f"      SGD done: best={best_score:.6f}, {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix for cross-convolution with a."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve LP: maximize sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s, verbose=False):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"      LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    it = 0
    while time.monotonic() - start_time < time_limit_s and it < 1000:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"      LP iter {it}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 60:
            if verbose:
                print(f"      LP stagnant at iter {it}, best={best_score:.6f}")
            break

        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 3.0:
            break
        timeout = max(2.0, min(45.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"      LP no improvement (sum_g={sum_g:.10f})")
            break
        g = g / sum_g

        # Adaptive step size
        alpha = min(0.5, max(0.03, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-12:
            a = a_new
        else:
            alpha2 = alpha * 0.1
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
        it += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      LP done: best={best_score:.6f}, {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit_s, rng, verbose=False):
    """Random local search with adaptive perturbation width."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"      LS start: score={best_score:.6f}")

    width_scale = 1.0
    accept_count = 0
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        it += 1
        width = max(2, int(rng.integers(max(2, n // 300), max(3, n // 5))) * width_scale)
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        noise_scale = 0.08 * width_scale
        candidate[start:stop] *= np.exp(rng.normal(0.0, noise_scale, size=stop - start))
        # Occasionally try global perturbation
        if rng.random() < 0.05:
            candidate *= np.exp(rng.normal(0.0, 0.02, size=n))
        # Occasionally try symmetry
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score - 1e-12:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            width_scale = min(width_scale * 1.1, 3.0)
            if verbose and accept_count % 30 == 0:
                print(f"      LS accept {accept_count}: score={best_score:.6f}")
        else:
            width_scale = max(width_scale * 0.95, 0.1)

    if verbose:
        print(f"      LS done: best={best_score:.6f}, {accept_count} accepts, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def alternating_refinement(a, time_limit_s, rng, verbose=False):
    """Alternate between smoothed gradient descent and LP refinement."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"      Alt start: score={best_score:.6f}, n={len(a)}")

    phase_duration = min(120.0, time_limit_s * 0.25)
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 10.0:
            break

        # Phase A: Smoothed gradient descent
        dur_a = min(phase_duration, remaining * 0.5)
        a_new, s_a = smoothed_gradient_descent(a, time_limit_s=dur_a, verbose=False)
        if s_a < best_score:
            best_score = s_a
            best_a = a_new.copy()
            a = a_new.copy()

        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 10.0:
            break

        # Phase B: LP refinement
        dur_b = min(phase_duration * 0.8, remaining * 0.5)
        a_new, s_b = lp_refine(a, time_limit_s=dur_b, verbose=False)
        if s_b < best_score:
            best_score = s_b
            best_a = a_new.copy()
            a = a_new.copy()

        it += 1
        if verbose and it % 2 == 0:
            print(f"      Alt round {it}: best={best_score:.6f}, elapsed={time.monotonic()-start_time:.1f}s")

    if verbose:
        print(f"      Alt done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses a hybrid of smoothed gradient descent, LP-based refinement,
    and adaptive local search with multi-length and multi-init exploration.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(60.0, budget_s - 30.0)

    best_score = float('inf')
    best_seq = None

    # Try to use previous best as a starting point
    start_candidates = []
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)
        if len(prev) > 0:
            prev_norm = normalize_nonneg(prev)
            s = score_sequence(prev_norm)
            print(f"Previous best: score={s:.6f}, n={len(prev)}")
            if s < best_score:
                best_score = s
                best_seq = prev_norm.copy()
            start_candidates.append(('prev', prev_norm))
    except Exception:
        pass

    # Generate diverse initial candidates
    lengths = [500, 750, 1000, 1500, 2000]
    inits_for_length = {}

    for n in lengths:
        candidates = []
        rc = normalize_nonneg(initial_raised_cosine(n))
        candidates.append(('rc', rc))
        gauss = normalize_nonneg(initial_gaussian(n, sigma_frac=0.25))
        candidates.append(('gauss25', gauss))
        gauss2 = normalize_nonneg(initial_gaussian(n, sigma_frac=0.4))
        candidates.append(('gauss40', gauss2))
        black = normalize_nonneg(initial_blackman(n))
        candidates.append(('black', black))
        tukey = normalize_nonneg(initial_tukey(n, r=0.4))
        candidates.append(('tukey', tukey))
        # Random bumps
        for bi in range(2):
            rb = random_bump_init(n, rng, n_bumps=rng.integers(2, 6))
            candidates.append((f'rand{bi}', rb))
        inits_for_length[n] = candidates

    # Evaluate all candidates
    print("\n=== Evaluating initial candidates ===")
    all_evaluated = []
    for n in lengths:
        for name, a in inits_for_length[n]:
            s = score_sequence(a)
            all_evaluated.append((s, n, name, a))
            if s < best_score:
                best_score = s
                best_seq = a.copy()
    all_evaluated.sort(key=lambda x: x[0])
    for i, (s, n, name, _) in enumerate(all_evaluated[:10]):
        print(f"  Init {i}: n={n}, type={name}, score={s:.6f}")

    # Select top candidates for refinement (deduplicate by length)
    selected = []
    lengths_seen = set()
    for s, n, name, a in all_evaluated:
        if n not in lengths_seen or len(selected) < 3:
            selected.append((n, a))
            lengths_seen.add(n)
        if len(selected) >= 5:
            break

    # If we have a previous best, make sure it's in the selection
    if 'prev' in [start_candidates[0][0]] if start_candidates else False:
        if len(selected) < 6:
            selected.append((len(start_candidates[0][1]), start_candidates[0][1]))

    print(f"\nSelected {len(selected)} candidates for refinement")

    # Phase 2: Smoothed gradient descent on top candidates
    print("\n=== Phase 2: Smoothed Gradient Descent ===")
    phase2_budget = min(budget_s * 0.35, deadline - time.monotonic() - 60)
    per_candidate = max(20.0, phase2_budget / len(selected))

    for idx, (n, a) in enumerate(selected):
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        t_limit = min(per_candidate, remaining * 0.7)
        print(f"  Candidate {idx} (n={n}): SGD for {t_limit:.0f}s")
        a_refined, s_refined = smoothed_gradient_descent(a, time_limit_s=t_limit, verbose=False)
        if s_refined < best_score:
            best_score = s_refined
            best_seq = a_refined.copy()
            # Store for further refinement
            if idx < len(selected):
                selected[idx] = (n, a_refined)
            print(f"    *** NEW BEST: score={s_refined:.6f}, n={n}")

    # Phase 3: Alternating refinement (SGD + LP)
    print("\n=== Phase 3: Alternating Refinement ===")
    phase3_budget = min(budget_s * 0.30, deadline - time.monotonic() - 60)
    if best_seq is not None and phase3_budget > 30:
        n_best = len(best_seq)
        print(f"  Refining best (n={n_best}) for {phase3_budget:.0f}s")
        a_alt, s_alt = alternating_refinement(best_seq, time_limit_s=phase3_budget, rng=rng, verbose=True)
        if s_alt < best_score:
            best_score = s_alt
            best_seq = a_alt.copy()
            print(f"    *** NEW BEST: score={s_alt:.6f}")

    # Try LP on other top candidates
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for idx, (n, a) in enumerate(selected[1:3]):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            t_limit = min(60.0, remaining * 0.4)
            print(f"  LP on candidate {idx+1} (n={n}) for {t_limit:.0f}s")
            a_lp, s_lp = lp_refine(a, time_limit_s=t_limit, verbose=False)
            if s_lp < best_score:
                best_score = s_lp
                best_seq = a_lp.copy()
                print(f"    *** NEW BEST: score={s_lp:.6f}")

    # Phase 4: Local search polishing
    print("\n=== Phase 4: Local Search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            ls_budget = min(remaining * 0.5, 120.0)
            print(f"  Local search for {ls_budget:.0f}s")
            a_ls, s_ls = local_search(best_seq, time_limit_s=ls_budget, rng=rng, verbose=True)
            if s_ls < best_score:
                best_score = s_ls
                best_seq = a_ls.copy()
                print(f"    *** After LS: score={s_ls:.6f}")

            # Final brief LP polish
            remaining = deadline - time.monotonic()
            if remaining > 15:
                final_budget = min(remaining * 0.6, 60.0)
                a_fin, s_fin = lp_refine(best_seq, time_limit_s=final_budget, verbose=False)
                if s_fin < best_score:
                    best_score = s_fin
                    best_seq = a_fin.copy()
                    print(f"    *** Final LP: score={s_fin:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = normalize_nonneg(initial_raised_cosine(500))
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Verify
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"\n=== FINAL RESULT ===")
    print(f"Score (internal): {best_score:.6f}")
    print(f"Score (verified): {verify_score:.6f}")
    print(f"Sequence length: {len(best_seq)}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
