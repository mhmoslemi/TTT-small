"""
Multi-phase optimizer for minimum peak autocorrelation non-negative sequences.
Combines fast LP refinement, smooth softmax gradient descent, and exact
coordinate descent to beat 1.5027.

import numpy as np

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if not np.isfinite(v).all() or v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    p = np.arange(1, n + 1)
    cond = u * p > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(cond))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n, f):
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    return sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()


def lp_step(f, M, A_cache, time_lim=0.35):

    try:
        res = linprog(c, A_ub=A_cache, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_lim, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
