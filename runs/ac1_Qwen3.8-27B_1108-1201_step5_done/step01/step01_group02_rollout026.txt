        if g is None:
            break
        sg = g.sum()
        if sg <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP converged (sum_g={sg:.6f}) at iter {it}")
            break
        g /= sg
        # Line search
        best_alpha = 0.1
        best_sc = sc
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = (1-alpha)*a + alpha*g
            a_try = np.maximum(0, a_try)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try /= st
            sc_try = score_seq(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_alpha = alpha
        a = (1-best_alpha)*a + best_alpha*g
        a = np.maximum(0, a)
        a /= a.sum()
    return best_a, best_score

def interp_upscale(a, factor):
    n = len(a)
    nn = n * factor
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n-1, nn)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new

def local_search(a, time_limit=60, seed=0):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    a = np.maximum(0, a.copy())
    a /= a.sum()
    best_a = a.copy()
    best_score = score_seq(a)
    n = len(a)
    while time.monotonic() - t0 < time_limit:
        w = max(2, int(rng.integers(n//300, n//5)))
        p = int(rng.integers(0, max(1, n-w)))
        cand = a.copy()
        cand[p:p+w] *= np.exp(rng.normal(0, 0.03, size=w))
        cand = np.maximum(0, cand)
        s = cand.sum()
        if s < 1e-10:
            continue
        cand /= s
        sc = score_seq(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand.copy()
    return best_a, best_score
