import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def lp_direction(f, M, n):
    """Solve the LP to find the best direction g_0.
    
    Maximize sum(b_j) subject to:
    - b_j >= 0
    - (f * b)_k <= M for all k
    
    Returns b_0 (the solution).
    """

    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return a uniform direction
        return np.ones(n) / n

    f = f * (S / np.sum(f))

    best_score = current_score

    t = t_init

import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix for the LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve the LP to find the best descent direction.
    
    Maximize sum(b_j) subject to:
    - b_j >= 0
    - (f * b)_k <= M for all k = 0, ..., 2n-2
    
    Returns the solution vector b.
    """
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: uniform direction
        return np.ones(n) / n


def lp_iterate(f0, time_limit, rng, verbose=True):
    """Iteratively apply LP-based updates to minimize the peak autocorrelation."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        f = np.ones(n)
        sum_f = n
    f = f * (S / sum_f)
    
    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t = 0.05  # initial step size
    stall_count = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))
        b_0 = solve_lp_direction(f, M)
        iter_count += 1
        
        sum_b = np.sum(b_0)
        if sum_b < 1e-10:
            break
        g = (S / sum_b) * b_0
        
        # Try the update
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        
        new_score = local_score(f_new)
        
        if new_score < current_score:
            f = f_new
            current_score = new_score
            stall_count = 0
            # Slightly increase step size
            t = min(0.2, t * 1.01)
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
        else:
            stall_count += 1
            t *= 0.7
            if stall_count > 50:
                # Perturb to escape
                f = f * np.exp(rng.normal(0, 0.02, n))
                f = np.clip(f, 0, None)
                f = f * (S / np.sum(f))
                current_score = local_score(f)
                stall_count = 0
                t = 0.05
        
        if verbose and iter_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, t={t:.4f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score, current_score


def polish_local_search(a_init, time_limit, rng):
    """Simulated annealing local search for final polish."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.3:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            # Find top contributors
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.7:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Pair swap (move mass from peak contributors to others)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * current[i]
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = max(0, current[j] + delta)
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 6):
        amp = 0.3 / k
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

"""
LP-based iterative optimization for minimizing the peak autocorrelation of
non-negative step functions. Uses the linear programming approach where at each
step, we find the best non-negative direction whose cross-correlation with the
current sequence is bounded by the current peak. Combined with adaptive step
size, subgradient sweeps, multiple starting points, and local search polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence (fast local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix A where A[k,i] = f[k-i] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve the LP to find the best descent direction.
    Maximize sum(b_j) subject to b>=0 and (f*b)_k <= M for all k.
    """
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(n) / n


def subgradient_step(f, lr):
    """One subgradient descent step on the peak autocorrelation."""
    n = len(f)
    c = fft_convolve(f, f)
    max_c = np.max(c)
    # Find all peak positions (within tolerance)
    peak_mask = c >= max_c * (1.0 - 1e-6)
    peak_indices = np.where(peak_mask)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in peak_indices:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                if idx == j:
                    grad[j] += 2.0 * f[j]
                else:
                    grad[j] += f[idx] + f[j]
    grad /= len(peak_indices)
    
    # Update
    f_new = f - lr * grad
    f_new = np.clip(f_new, 0.0, None)
    s = np.sum(f_new)
    if s < 0.01:
        return f.copy()
    return f_new * (np.sum(f) / s)


def lp_iterate(f0, time_limit, rng, verbose=True):
    """Iteratively apply LP-based updates with line search."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        f = np.ones(n) * (S / n)
    else:
        f = f * (S / sum_f)
    
    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t = 0.08
    stall_count = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))
        b_0 = solve_lp_direction(f, M)
        iter_count += 1
        
        sum_b = np.sum(b_0)
        if sum_b < 1e-10:
            break
        g = (S / sum_b) * b_0
        
        # Line search: try decreasing t values
        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1]:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0, None)
            new_score = local_score(f_new)
            if new_score < current_score:
                f = f_new
                current_score = new_score
                t = trial_t * 1.05
                t = min(t, 0.25)
                improved = True
                stall_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break
        
        if not improved:
            stall_count += 1
            t *= 0.7
            if stall_count > 80:
                # Perturb to escape local minimum
                f = f * np.exp(rng.normal(0, 0.03, n))
                f = np.clip(f, 0, None)
                sf = np.sum(f)
                if sf > 0.01:
                    f = f * (S / sf)
                    current_score = local_score(f)
                stall_count = 0
                t = 0.05
            # Every 200 stalled iterations, try subgradient
            if stall_count > 0 and stall_count % 20 == 0:
                sg_f = f.copy()
                for _ in range(50):
                    lr = 0.01 / (1 + _ * 0.01)
                    sg_f = subgradient_step(sg_f, lr)
                sg_score = local_score(sg_f)
                if sg_score < current_score:
                    f = sg_f
                    current_score = sg_score
                    if current_score < best_score:
                        best_score = current_score
                        best_f = f.copy()
                    stall_count = 0
        
        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, t={t:.4f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score


def polish_local_search(a_init, time_limit, rng):
    """Simulated annealing local search for final polish."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.9997
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.30:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(4, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(12, len(other) // 2)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.01, size=n_inc))
        elif r < 0.50:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.65:
            # Multi-block
            for _ in range(5):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.80:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.15, 0.15) * current[i]
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = max(0, current[j] + delta)
        elif r < 0.90:
            # Trilinear perturbation
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len(set([i, j, k])) == 3:
                delta = rng.uniform(-0.1, 0.1) * current[i]
                candidate[i] = max(0, current[i] - 2 * delta)
                candidate[j] = max(0, current[j] + delta)
                candidate[k] = max(0, current[k] + delta)
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.006, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 7):
        amp = 0.25 / k
        freq = rng.uniform(0.3, 2.5) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
