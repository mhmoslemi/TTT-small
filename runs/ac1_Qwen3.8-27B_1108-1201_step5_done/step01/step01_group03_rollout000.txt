def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2"""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)

def normalize(f):
    """Normalize to sum=1, clip to non-negative."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def build_conv_matrix(n, f):
    """Build sparse matrix A where (A @ g)[k] = (f*g)[k]"""
    # (f*g)[k] = sum_j f[k-j] * g[j] for j=0..n-1
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    data = []
    for j in range(n):
        # For this j, A[k,j] = f[k-j] for k = j, j+1, ..., j+n-1
        k_start = j
        k_end = j + n  # exclusive

            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                data.append(f[idx])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()

def subgradient_step(f, alpha):
    """One sub-gradient step on max(f*f) with simplex projection."""
    n = len(f)
    conv = np.convolve(f, f)
    k_star = np.argmax(conv)
    # Gradient of (f*f)[k_star] w.r.t. f[i] = 2*f[k_star-i] (if 0<=k_star-i<n)
    h = np.zeros(n)

        if 0 <= idx < n:
            h[i] = 2.0 * f[idx]
    # Descent direction: f - alpha*h
    v = f - alpha * h
    # Project onto simplex {x >= 0, sum(x) = 1}
    f_new = project_simplex(v)
    return f_new

def project_simplex(v):

    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    tau = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - tau, 0)

Multi-phase optimization for minimum peak autocorrelation non-negative step functions.
Uses LP-based successive linearization, sub-gradient descent, frequency-domain
initialization, and adaptive local search to minimize score = 2*n*max(f*f)/(sum(f))^2.

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(n, f):
    j_idx = np.repeat(np.arange(n), n)
    i_idx = np.tile(np.arange(n), n)
    k_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (k_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit_s=5.0):
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)

    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    h = np.zeros(n)
    valid_start = max(0, k_star - (n - 1))
    valid_end = min(n, k_star + 1)
    for i in range(valid_start, valid_end):
        idx = k_star - i
        if 0 <= idx < n:
            h[i] = 2.0 * f[idx]
    v = f - alpha * h
    return project_simplex(v)

    inits.append(normalize(np.sin(x) ** 2 + 0.01))
    # Exponential decay
    a = np.exp(-np.arange(n, dtype=float) / (n / 3))
    inits.append(normalize(a))
    # Asymmetric: more mass on one side
    a = np.exp(-np.abs(np.arange(n, dtype=float) - n * 0.3) / (n / 4))
    inits.append(normalize(a))
    # Two peaks
    x = np.arange(n, dtype=float)
    s = max(1, n // 10)
    a = np.exp(-0.5 * ((x - n * 0.25) / s) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / s) ** 2)
    inits.append(normalize(a))
    # Random
    inits.append(normalize(rng.random(n)))
    return inits


def freq_init(n, rng, n_iters=200):
    """Gerchberg-Saxton style frequency domain initialization."""
    f = normalize(1.0 + 0.3 * rng.random(n))
    N = 2 * n - 1
    for it in range(n_iters):
        f_pad = np.zeros(N)
        f_pad[:n] = f
        F = np.fft.fft(f_pad)
        psd = np.abs(F) ** 2
        # Target: as flat as possible, with DC fixed
        mean_psd = psd.mean()
        target = np.full(N, mean_psd)
        target[0] = psd[0]  # Keep DC (sum constraint)
        # Soft approach to target
        alpha = 0.05 + 0.05 * (it / n_iters)
        new_psd = (1 - alpha) * psd + alpha * target
        new_psd = np.maximum(new_psd, 1e-20)
        phase = np.angle(F)
        new_F = np.sqrt(new_psd) * np.exp(1j * phase)
        f_new = np.fft.ifft(new_F).real
        f_new = np.maximum(f_new, 0)
        s = f_new[:n].sum()
        if s < 1e-10:
            break
        f = f_new[:n] / s
    return f


def lp_phase(f, budget_s, t_init=0.2, lp_time=3.0):
    """LP-based iterative refinement with adaptive step size."""
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    t = t_init
    n = len(f)
    stall = 0
    it = 0

    while time.monotonic() < budget_s:
        it += 1
        conv = np.convolve(f, f)
        M = float(conv.max())
        if M < 1e-15:
            break

        g = solve_lp(f, M, lp_time)
        if g is None:
            # Fallback: sub-gradient step
            alpha = t * f.sum() / max(np.sum(np.abs(f)), 1e-10)
            f_new = subgradient_step(f, alpha * 0.1)
        else:
            gs = g.sum()
            if gs < 1e-10:
                f_new = subgradient_step(f, 0.01)
            elif np.linalg.norm(g / gs - f) < 1e-10:
                # LP gives same point, try perturbation
                rng = np.random.default_rng(it)
                f_new = f * np.exp(rng.normal(0, 0.1, n))
                f_new = normalize(f_new)
            else:
                g_norm = g / gs
                # Adaptive step: try multiple t values
                best_f_new = f
                best_sc_try = best_sc
                for t_try in [t, t * 0.5, t * 2.0, 0.05, 0.1, 0.3]:
                    t_try = min(1.0, max(0.01, t_try))
                    cand = (1 - t_try) * f + t_try * g_norm
                    cand = normalize(cand)
                    sc = fast_score(cand)
                    if sc < best_sc_try:
                        best_sc_try = sc
                        best_f_new = cand
                f_new = best_f_new

        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = f_new.copy()
            stall = 0
            t = min(0.4, t * 1.05)
        else:
            stall += 1
            t = max(0.02, t * 0.95)
            if stall > 30:
                # Escape: block perturbation
                rng = np.random.default_rng(10000 + it)
                f_new = f.copy()
                w = max(1, n // 20)
                start = int(rng.integers(0, n - w + 1))
                f_new[start:start+w] *= np.exp(rng.normal(0, 0.3, w))
                f_new = normalize(f_new)
                sc2 = fast_score(f_new)
                if np.isfinite(sc2):
                    f = f_new
                    if sc2 < best_sc:
                        best_sc = sc2
                        best = f_new.copy()
                stall = 0
                continue

        f = f_new

    return best, best_sc


def block_polish(f, budget_s, rng):
    """Fine-grained block perturbation polish."""
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    it = 0

    while time.monotonic() < budget_s:
        it += 1
        # Random block multiplicative perturbation
        w = int(rng.integers(1, max(2, n // 15)))
        start = int(rng.integers(0, n - w + 1))
        scale = np.exp(rng.normal(0, 0.15, w))
        cand = f.copy()
        cand[start:start+w] *= scale
        cand = normalize(cand)
        sc = fast_score(cand)

        # Also try single-element adjustment
        if it % 2 == 0:
            idx = int(rng.integers(0, n))
            cand2 = f.copy()
            delta = cand2[idx] * rng.uniform(-0.3, 0.3)
            cand2[idx] = max(0, cand2[idx] + delta)
            cand2 = normalize(cand2)
            sc2 = fast_score(cand2)
            if sc2 < sc:
                cand = cand2
                sc = sc2

        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = cand.copy()
            f = cand.copy()

    return best, best_sc


def stretch_and_optimize(f, n_new, budget_s, rng):
    """Interpolate sequence to new length and optimize."""
    n_old = len(f)
    if n_new == n_old:
        return f, fast_score(f)
    x_old = np.linspace(0, 1, n_old)
    y = f / f.sum()  # normalize
    x_new = np.linspace(0, 1, n_new)
    f_new = np.interp(x_new, x_old, y)
    f_new = normalize(f_new)
    return lp_phase(f_new, budget_s)


def propose_candidate(budget_s=1800):
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(42)

    # Get baseline from previous best if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(256)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256)

    best = normalize(parent.copy())
    best_sc = fast_score(best)
    print("Initial best: score=%.6f, n=%d" % (best_sc, best.size), flush=True)

    def try_update(cand, sc):
        nonlocal best, best_sc
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = np.asarray(cand, dtype=np.float64).copy()
            print("  NEW BEST: %.6f (n=%d)" % (best_sc, best.size), flush=True)

    # === Phase 1: Frequency domain initialization for multiple lengths ===
    print("Phase 1: Frequency-domain initialization", flush=True)
    for n in [512, 1024]:
        if time.monotonic() > deadline - 600:
            break
        t0 = time.monotonic()
        f_init = freq_init(n, rng, n_iters=100)
        sc = fast_score(f_init)
        print("  freq_init n=%d: score=%.6f (took %.1fs)" % (n, sc, time.monotonic() - t0), flush=True)
        try_update(f_init, sc)

    # === Phase 2: LP-based optimization from multiple starting points ===
    print("Phase 2: LP-based optimization", flush=True)

    # Strategy: run LP from several different starts
    starts = []
    # Start from previous best
    starts.append((best.copy(), "prev_best"))
    # Frequency inits
    for n in [512, 1024]:
        f_f = freq_init(n, rng, n_iters=80)
        starts.append((f_f, "freq_n%d" % n))
    # Structured inits
    for n in [512, 1024]:
        inits = make_initializations(n, rng)
        for i, ini in enumerate(inits[:3]):
            starts.append((ini, "struct_n%d_%d" % (n, i)))
    # Previous best stretched
    n_stretch = best.size * 2
    if n_stretch <= 2048:
        f_st, _ = stretch_and_optimize(best, n_stretch, 0, rng)
        starts.append((f_st, "stretched"))

    # Allocate time to each start
    time_per_start = max(30.0, (deadline - time.monotonic()) / (len(starts) + 1))
    time_per_start = min(time_per_start, 300.0)

    for (f_start, name) in starts:
        if time.monotonic() > deadline - 120:
            break
        n = len(f_start)
        lp_time = min(5.0, time_per_start / 50.0)
        if n > 1500:
            lp_time = min(3.0, time_per_start / 80.0)
        local_deadline = min(deadline - 120, time.monotonic() + time_per_start)
        print("  LP n=%d start=%s budget=%.0fs" % (n, name, time_per_start), flush=True)
        t0 = time.monotonic()
        c_best, c_sc = lp_phase(f_start, local_deadline, t_init=0.2, lp_time=lp_time)
        print("  -> %.6f (%.1fs)" % (c_sc, time.monotonic() - t0), flush=True)
        # Verify with authoritative evaluator
        c_conf = float(evaluate_sequence(c_best.tolist()))
        if np.isfinite(c_conf) and c_conf < c_sc:
            c_sc = c_conf
        try_update(c_best, c_sc)

    # === Phase 3: Local polish on current best ===
    print("Phase 3: Local polish (score=%.6f)" % best_sc, flush=True)
    polish_deadline = deadline
    f_pol, sc_pol = block_polish(best, polish_deadline, rng)
    sc_pol_conf = float(evaluate_sequence(f_pol.tolist()))
    if np.isfinite(sc_pol_conf) and sc_pol_conf < sc_pol:
        sc_pol = sc_pol_conf
    try_update(f_pol, sc_pol)

    # === Phase 4: One more round of LP from best, with different params ===
    if time.monotonic() < deadline - 30:
        print("Phase 4: Final LP refinement", flush=True)
       _lp_deadline = deadline
        f_fin, sc_fin = lp_phase(best, _lp_deadline, t_init=0.1, lp_time=3.0)
        sc_fin_conf = float(evaluate_sequence(f_fin.tolist()))
        if np.isfinite(sc_fin_conf) and sc_fin_conf < sc_fin:
            sc_fin = sc_fin_conf
        try_update(f_fin, sc_fin)

    # Final validation
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(256)
    final_sc = float(evaluate_sequence(best.tolist()))
    print("FINAL: score=%.6f, n=%d" % (final_sc, best.size), flush=True)
    return [float(x) for x in best]
