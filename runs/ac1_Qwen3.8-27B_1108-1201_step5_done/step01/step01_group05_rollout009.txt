def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0/n)
    return out / s

def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(len(p), 1.0/len(p))
    return p / s

def make_shape_1d(n, params):

    n = int(n)

        p *= np.exp(params[0] * x)  # linear taper

        p *= np.exp(params[1] * (1-x))  # linear taper from right

        p *= np.exp(params[3] * (x - 0.5))  # asymmetry
    if len(params) > 4:
        p *= (1.0 + params[4] * np.sin(2*np.pi*x))  # oscillation
    p = np.maximum(p, 1e-6)
    return p / p.sum()

def kkt_best_response(p, k_star, n_active=None):
    """
    Solve the KKT system for minimizing conv[p,p]_{k_star}
    subject to sum(p) = 1, p >= 0.
    
    The gradient of conv[p,p]_{k} w.r.t. p[j] is:
    p[j-k] + p[j+k] (for valid indices, 0 otherwise)
    
    KKT: p[j-k] + p[j+k] + lambda = 0 for all j with p[j] > 0
    and p[j-k] + p[j+k] + lambda >= 0 for all j with p[j] = 0
    
    For a given active set S, we solve:
    A[S,S] x[S] = -lambda * ones(S)
    sum(x[S]) = 1
    
    where A[i,j] = 1 if |i-j| == k_star else 0 (for the relevant indices)
    """
    n = len(p)
    p = np.asarray(p, dtype=np.float64)
    
    # Determine active set (indices where p > 0)
    if n_active is None:
        active = np.where(p > 1e-10)[0]
    else:
        active = np.where(p > n_active)[0]
    
    if len(active) == 0:
        return p, 0.0
    
    # For each active index j, the constraint is:
    # p[j-k] + p[j+k] + lambda = 0
    # where p[j-k] = 0 if j-k < 0, p[j+k] = 0 if j+k >= n
    
    # This gives a system in the active variables and lambda.
    # Let x = p[active], and A is the matrix where A[i,j] = 1 if active[j] - active[i] == k_star or active[j] - active[i] == -k_star
    
    m = len(active)
    A = np.zeros((m, m))
    for i in range(m):
        for j in range(m):
            if abs(active[j] - active[i]) == k_star:
                A[i, j] = 1.0
    
    # System: A @ x + lambda * ones = 0
    #         sum(x) = 1
    
    # Augmented system: [A | 1; 1^T | 0] [x; lambda] = [0; 1]
    if m == 1:
        # Single active variable: p[active[0]] = 1
        q = np.zeros(n)
        q[active[0]] = 1.0
        return q, 0.0
    
    M = np.zeros((m+1, m+1))
    M[:m, :m] = A
    M[:m, m] = 1.0
    M[m, :m] = 1.0
    b = np.zeros(m+1)
    b[m] = 1.0
    
    try:
        sol = np.linalg.solve(M, b)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(M, b, rcond=None)[0]
        except:
            return p, 0.0
    
    x = sol[:m]
    lam = sol[m]
    
    # Check feasibility
    q = np.zeros(n)
    q[active] = x
    
    if np.any(x < -1e-10):
        return p, 0.0  # Infeasible
    
    q = np.maximum(q, 0.0)
    s = q.sum()
    if s <= 1e-12:
        return p, 0.0
    q = q / s
    
    # Verify KKT conditions for inactive variables
    # For j not in active: p[j-k] + p[j+k] + lambda >= 0
    for j in range(n):
        if j not in set(active.tolist()):
            val = 0.0
            if j - k_star >= 0:
                val += q[j - k_star]
            if j + k_star < n:
                val += q[j + k_star]
            if val + lam < -1e-8:
                return p, 0.0  # KKT violated
    
    return q, float(score(q))

def optimize_local(p0, rng, time_limit, label=""):
    """Local optimization with adaptive temperature gradient descent."""
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = len(p)
    
    best_p = p.copy()
    best_score = score(best_p)
    current_score = best_score
    prev_p = p.copy()
    
    end = time.monotonic() + max(0.0, time_limit)
    T = 0.05 * best_score  # Initial temperature (relative to score)
    iter_no = 0
    since_improve = 0
    
    while time.monotonic() < end:
        iter_no += 1
        T = max(0.001 * best_score, T * 0.995)  # Annealing
        
        surr, grad, c, cmax, T_used = surrogate_and_grad(p, T)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
        
        # Adaptive step size
        gabs = float(np.max(np.abs(grad))) if n > 0 else 0.0
        p_scale = max(1.0/n, 0.1 * float(np.max(p)))
        
        if gabs > 1e-300:
            base_lr = p_scale / gabs
            lrs = (0.3 * base_lr, 0.1 * base_lr, 0.03 * base_lr, 0.01 * base_lr, 0.003 * base_lr)
        else:
            lrs = (0.1, 0.05, 0.01)
        
        accepted = False
        for lr in lrs:
            lr = min(float(lr), 5.0)
            v = p - lr * grad
            pt = project_simplex(v)
            
            c_t = conv_self(pt)
            cmax_t = float(np.max(c_t))
            s_t = float(pt.sum())
            score_t = float(2.0 * n * cmax_t / (s_t * s_t)) if s_t > 1e-12 else np.inf
            
            # Accept if surrogate improves or actual score improves
            surr_t = cmax_t + T_used * float(np.log(float(np.exp((c_t - cmax_t)/T_used).sum()))) if T_used > 0 else cmax_t
            
            if np.isfinite(score_t):
                if score_t < best_score:
                    best_score = score_t
                    best_p = pt.copy()
                    since_improve = 0
                if surr_t < surr - 1e-10 * max(1.0, abs(surr)) or score_t < current_score - 1e-10:
                    current_score = score_t
                    prev_p = p.copy()
                    p = pt
                    accepted = True
                    break
        
        if not accepted:
            since_improve += 1
            T = max(0.001 * best_score, T * 1.5)  # Increase temperature to escape
            
            if since_improve > 100:
                # Random perturbation
                sigma = 0.1 / n if since_improve < 300 else 0.5 / n
                pt = project_simplex(p + rng.normal(0, sigma, n))
                sc = score(pt)
                if np.isfinite(sc) and sc < current_score:
                    p = pt
                    current_score = sc
                    if sc < best_score:
                        best_score = sc
                        best_p = p.copy()
                    since_improve = 0
                    accepted = True
            
            if not accepted and since_improve > 500:
                # Revert towards best
                p = project_simplex(0.5 * best_p + 0.5 * p)
                current_score = score(p)
                since_improve = 0
                T = 0.05 * best_score
        
        # Periodic KKT best response
        if iter_no % 500 == 0 and time.monotonic() < end - 5:
            c = conv_self(p)
            k_star = int(np.argmax(c))
            if 0 < k_star < n:  # Only for non-trivial lags
                q, sc_q = kkt_best_response(p, k_star)
                if sc_q < best_score:
                    best_score = sc_q
                    best_p = q.copy()
                    p = q.copy()
                    current_score = sc_q
                    since_improve = 0
        
        if iter_no % 2000 == 0:
            print(f"{label} iter={iter_no} best={best_score:.10f} T={T:.6f}", flush=True)
    
    return best_p, best_score

        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = parent_arr
    except:
        parent = None

    if n < 100:
