import time


def _fft_convolve(a):
    """Compute full convolution using FFT for speed."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(Fa * Fa, fsize)[:size]

    """Compute 2*n*max(convolve(a,a)) for a sequence with sum=1."""
    n = len(a)
    b = _fft_convolve(a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:

    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_at_peaks(a, b, M, alpha):
    """Compute weighted subgradient at top M peaks of autocorrelation."""
    n = len(a)
    b_max = np.max(b)
    
    if M >= len(b):
        k_idx = np.arange(len(b))
        wts = np.ones(len(b)) / len(b)

        k_idx = np.argpartition(b, -M)[-M:]
        b_sel = b[k_idx]
        scale = b_max - b_sel
        max_scale = np.max(scale)
        if max_scale < 1e-15:
            wts = np.ones(len(k_idx)) / len(k_idx)
        else:
            wts = np.exp(-alpha * scale / max_scale)
            wts /= np.sum(wts)
    
    grad = np.zeros(n)
    for ki, k in enumerate(k_idx):
        w = wts[ki]
        if w < 1e-10:

    return grad

    FISTA (accelerated projected gradient) to minimize max_k b_k
    subject to sum(a) = 1, a >= 0.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_score = _compute_score(best)
    
    # Initial step size
    b0 = _fft_convolve(a)
    # Estimate Lipschitz constant
    grad0 = _grad_at_peaks(a, b0, M, alpha)
    L = 2.0 * n / (np.sum(a**2) + 1e-15)  # Rough upper bound
    step = 1.0 / L
    step = min(step, 0.1)  # Cap initial step
    
    t_k = 1.0
    y = a.copy()
    
    while time.monotonic() < deadline:
        # Compute gradient at y
        b_y = _fft_convolve(y)
        grad = _grad_at_peaks(y, b_y, M, alpha)
        grad -= np.mean(grad)  # Center for sum constraint
        
        # Gradient step
        z = y - step * grad
        x_new = _project_simplex(z, 1.0)
        
        # FISTA update

        # Evaluate

        if cur_score < best_score - 1e-13:

def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_smooth_anti(n, eps):
    """Smooth anti-symmetric profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_four_level(n, eps):
    """Four-level anti-symmetric."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3.0
    a[2 * q:3 * q] = 1.0 - eps / 3.0
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)

    """Triangular (inverted V) profile."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 - sharpness * np.abs(i - mid) / mid

"""
FFT-accelerated multi-start subgradient optimization for minimizing
the normalized autocorrelation peak of a non-negative sequence.
"""

import numpy as np
import time

def _fft_convolve(a):
    """Full convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(Fa * Fa, fsize)[:size]

def _score(a):
    """Score = 2n * max(autocorr) assuming sum(a)=1."""
    n = len(a)
    b = _fft_convolve(a)
    return 2.0 * n * np.max(b)

def _project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, 1^T x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0])
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgrad(a, b, M, alpha):
    """Weighted subgradient at top-M peaks."""
    n = len(a)
    bmax = b.max()
    k_idx = np.argpartition(b, -M)[-M:]
    b_sel = b[k_idx]
    spread = bmax - b_sel
    smax = spread.max()
    if smax < 1e-15:
        w = np.ones(M) / M
    else:
        w = np.exp(-alpha * spread / smax)
        w /= w.sum()
    g = np.zeros(n)
    for m in range(M):
        k = k_idx[m]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[m] * 2.0 * a[k - idx]
    return g

def _optimize_fista(a, deadline, rng, M=15, alpha=5.0):
    """FISTA with backtracking on the simplex."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = a.sum()
    a /= s if s > 1e-10 else n
    best, best_sc = a.copy(), _score(a)
    step = 0.05  # initial step
    t = 1.0
    y = a.copy()
    
    while time.monotonic() < deadline:
        b = _fft_convolve(y)
        g = _subgrad(y, b, M, alpha)
        g -= g.mean()
        
        # Backtracking line search
        t_new = y - step * g
        x = _project_simplex(t_new, 1.0)
        f_x = _score(x)
        f_y = _score(y)
        
        # Simple: accept if improved, else reduce step
        if f_x > f_y + 0.5 * step * np.dot(g, x - y):
            step *= 0.5
            if step < 1e-10:
                break
            continue
        
        # FISTA update
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_next) * (x - a)
        y = np.maximum(y, 0.0)
        y /= y.sum() + 1e-15
        t = t_next
        a = x
        
        if f_x < best_sc - 1e-12:
            best_sc = f_x
            best = x.copy()
    
    return best, best_sc

    f = 1 << (sz - 1).bit_length()
    A = np.fft.rfft(a, f)
    return np.fft.irfft(A * A, f)[:sz]


def _score(a):
    return 2.0 * len(a) * np.max(_fft_conv(a))


def _proj_simplex(v, t=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    c = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > c - t)[0])
    return np.maximum(v - (c[rho] - t) / (rho + 1.0), 0.0)


def _grad_peaks(a, b, M, temp):
    n = len(a)
    k = np.argpartition(b, -M)[-M:]
    bk = b[k]
    bmax = b.max()
    d = (bmax - bk) / (bmax + 1e-15)
    w = np.exp(-d / temp)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(M):
        km = k[m]
        lo, hi = max(0, km - n + 1), min(n - 1, km)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx] += w[m] * 2.0 * a[km - idx]
    return g


def _run(a, deadline, rng, M=15, temp=0.05, step0=0.03):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_s = _score(best)
    step = step0
    no_imp = 0
    t_acc = 1.0
    y = a.copy()
    
    while time.monotonic() < deadline:
        b = _fft_conv(y)
        g = _grad_peaks(y, b, M, temp)
        g -= g.mean()
        
        # FISTA-like update
        z = y - step * g
        x = _proj_simplex(z, 1.0)
        
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_acc**2))
        y = x + ((t_acc - 1.0) / t_next) * (x - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = best.copy()
            break
        y /= sy
        t_acc = t_next
        a = x
        
        sc = _score(x)
        if sc < best_s - 1e-13:
            best_s = sc
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive step: reduce if no improvement
        if no_imp == 50:
            step *= 0.7
            t_acc = 1.0  # Reset momentum
            no_imp = 0
            if step < 1e-12:
                break
        elif no_imp == 150:
            step *= 0.5
            t_acc = 1.0
            no_imp = 0
            if step < 1e-12:

def _perturb(a, rng, sc=0.1):
    n = len(a)
    nf = min(10, max(2, n // 10))
    freqs = rng.integers(1, max(2, n // 6), size=nf)
    noise = np.zeros(n)
    for f in freqs:
        ph = rng.uniform(0, 2 * np.pi)
        am = rng.uniform(-1, 1)
        noise += am * np.cos(2 * np.pi * f * np.arange(n) / n + ph)
    mn = np.max(np.abs(noise))
    if mn > 1e-15:
        noise /= mn
    a2 = np.maximum(a + sc * noise, 0.0)
    s = a2.sum()
    return a2 / s if s > 1e-10 else a.copy()

def _fft_conv(a):

def _proj_simplex(v, target=1.0):
