Smoothed gradient descent with Bregman updates to minimize peak auto-convolution.
Combines adaptive softmax smoothing, FFT-based gradients, and LP refinement.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

    return np.ones(n, dtype=np.float64) / n

    return a / a.sum()


def initial_gaussian(n, sigma_frac=0.25, rng=None):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return a / a.sum()


def initial_two_bump(n, rng, gap_frac=0.4):
    """Two Gaussian bumps separated by gap_frac * n."""
    gap = int(gap_frac * n)
    center1 = gap // 4
    center2 = n - 1 - gap // 4
    sigma = n * 0.15
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - center1) / sigma) ** 2) + np.exp(-0.5 * ((t - center2) / sigma) ** 2)
    return a / a.sum()


def initial_multi_bump(n, rng, n_bumps=3):
    """Multiple Gaussian bumps at random positions."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    positions = sorted(rng.integers(0, n, size=n_bumps))
    for pos in positions:
        sigma = n * rng.uniform(0.1, 0.25)
        a += np.exp(-0.5 * ((t - pos) / sigma) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_uniform(n)
    return a / s


def compute_gradient(a, beta):
    """Compute the smoothed gradient of max(a*a) using softmax."""
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax weights
    shifted = beta * (conv - conv.max())
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Gradient: grad[j] = sum_k weights[k] * a[k-j] (if 0 <= k-j < n)
    # = cross_correlation(weights, a)[j]
    # Use FFT
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)
    grad = R[:n]
    return grad, conv


def smooth_grad_descent(a, time_limit_s=100.0, beta_start=5.0, beta_end=2000.0, verbose=True):
    """Bregman gradient descent with adaptive beta."""
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)
    
    # Precompute FFT size
    N = 1
    while N < 3 * n:
        N *= 2
    
    stagnant = 0
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        elapsed = time.monotonic() - start_time
        remaining = time_limit_s - elapsed
        if remaining < 0.1:
            break
        
        # Adaptive beta
        progress = min(1.0, elapsed / (time_limit_s * 0.7))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad, conv = compute_gradient(a, beta)
        
        # Bregman (multiplicative) update
        # a_new[j] = a[j] * exp(-alpha * grad[j])
        # Try different alphas
        best_new_score = best_score
        best_alpha = None
        best_a_new = None
        
        # Scale gradient for step size
        # grad is O(1) in magnitude, a is O(1/n)
        # We want alpha * grad to be O(1) for meaningful updates
        grad_scale = grad.max()
        if grad_scale < 1e-10:
            break
        
        # Try alphas
        for alpha_mult in [1.0, 0.5, 0.1, 0.05, 0.01]:
            alpha = alpha_mult / grad_scale
            log_update = -alpha * grad
            log_update -= log_update.max()  # stability
            update = np.exp(log_update)
            a_new = a * update
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            s_new = score_sequence(a_new)
            if s_new < best_new_score:
                best_new_score = s_new
                best_alpha = alpha
                best_a_new = a_new.copy()
        
        if best_a_new is not None:
            a = best_a_new
            if best_score > best_new_score + 1e-14:
                stagnant = 0
            else:
                stagnant += 1
            best_score = min(best_score, best_new_score)
            best_a = a.copy()
        else:
            stagnant += 1
        
        if stagnant > 100:
            if verbose:
                print(f"    SGD stagnant at iter {it}, best={best_score:.6f}")
            break
        
        it += 1
        if verbose and (it % 200 == 0 or it < 3):
            print(f"    SGD iter {it}: score={score_sequence(a):.6f}, best={best_score:.6f}, beta={beta:.1f}")
    
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

    start_time = time.monotonic()

            break

                break

            a_new = (1 - alpha) * a + alpha * g

                best_alpha = alpha
