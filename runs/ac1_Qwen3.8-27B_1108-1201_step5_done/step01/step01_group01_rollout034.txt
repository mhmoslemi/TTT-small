def _compute_score_normalized(a):
    """Compute 2*n*max(conv(a,a)) for a normalized sequence (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _mirror_descent_step(a, b, tau, eta):
    """
    One mirror descent step.
    a: current sequence (sum=1, non-negative)
    b: convolve(a, a)
    tau: softmax temperature
    eta: step size
    Returns updated a.
    """
    n = len(a)
    # Softmax weights (shifted for numerical stability)
    b_max = np.max(b)
    exp_b = np.exp(np.clip(tau * (b - b_max), -700, 700))
    w = exp_b / np.sum(exp_b)
    
    # Gradient via correlation
    grad = 2.0 * np.correlate(w, a, mode='full')[:n]
    
    # Mirror (multiplicative) update
    log_update = np.clip(-eta * grad, -50, 100)
    a_new = a * np.exp(log_update)
    s = np.sum(a_new)
    if s < 1e-30:
        return a
    return a_new / s

def _interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""

    old_x = np.linspace(0, 1, old_n)

    a_new = np.maximum(a_new, 0.0)
    S = np.sum(a_new)

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
