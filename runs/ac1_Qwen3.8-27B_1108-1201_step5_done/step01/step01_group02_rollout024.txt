"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP-based refinement, gradient descent, and random search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def get_conv_and_peak(a):
    conv = np.convolve(a, a)
    peak_idx = np.argmax(conv)
    return conv, peak_idx, conv[peak_idx]


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_chirp(n, freq=1.0):
    t = np.arange(n, dtype=np.float64)
    return 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * t * t / n)


def initial_blackman(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def random_smooth_init(n, rng, n_bumps=None):
    if n_bumps is None:
        n_bumps = int(rng.integers(2, 7))
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 15), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    # For each column j (element a[j]), it contributes to rows j, j+1, ..., j+n-1
    # in the convolution output (row k gets contribution a[j]*a[k-j])
    # Actually, the LP is: conv(a, g)[k] = sum_j a[j]*g[k-j] <= M
    # So the matrix A has A[k,j] = a[j] if 0 <= k-j < n, else 0
    # This is a banded matrix
    row_idx = []
    col_idx = []
    data = []
    for j in range(n):
        # a[j] * g[k-j] contributes to row k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        for k in range(j, j + n):
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[j])
    return csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def gradient_descent_polish(a, time_limit_s=120.0, verbose=False):
    """Gradient descent on the peak auto-convolution value."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    eta = 0.3  # initial step size
    stagnant = 0
    it = 0
    
    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        peak_val = conv.max()
        # Find all positions within 1% of peak
        threshold = peak_val * 0.99
        peak_positions = np.where(conv >= threshold)[0]
        
        # Compute gradient (sum of gradients at all near-peak positions)
        grad = np.zeros(n)
        for k in peak_positions:
            for m in range(n):
                dist = abs(m - k)
                if dist < n:
                    if m == k:
                        grad[m] += 2.0 * a[0]
                    else:
                        grad[m] += a[dist]
        
        # Actually this O(n^2) per iteration is too slow for large n.
        # Let me use the structure: grad[m] = sum_k a[|m-k|] for k in peak_positions
        # If there's only one peak at k*, grad[m] = a[|m-k*|] (with 2*a[0] at m=k*)
        # For multiple peaks, it's the sum.
        
        # Center the gradient
        grad_mean = grad.mean()
        grad = grad - grad_mean
        
        # Step
        a_new = a - eta * grad
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = score_sequence(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            it += 1
            if verbose and it % 100 == 0:
                print(f"      GD iter {it}: score={score:.6f}, eta={eta:.4f}")
        else:
            stagnant += 1
            eta *= 0.8  # reduce step size
            if stagnant > 30:
                break
    
    return best_a, best_score


def gradient_descent_fast(a, time_limit_s=120.0, verbose=False):
    """Faster gradient descent using the structure of the gradient."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    eta = 0.2
    stagnant = 0
    it = 0
    best_eta = eta
    
    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        peak_val = conv.max()
        peak_idx = np.argmax(conv)
        
        # Gradient: grad[m] = a[|m - peak_idx|] for m != peak_idx, 2*a[0] for m = peak_idx
        # This is a "V-shaped" function centered at peak_idx
        dist = np.abs(np.arange(n) - peak_idx)
        grad = a[dist]  # a[|m - peak_idx|] for each m
        grad[peak_idx] = 2.0 * a[0]  # special case
        
        # Center
        grad = grad - grad.mean()
        
        # Step
        a_new = a - eta * grad
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = score_sequence(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            best_eta = eta
            eta = min(eta * 1.05, 0.5)  # slightly increase
            it += 1
            if verbose and it % 50 == 0:
                print(f"      GD-fast iter {it}: score={score:.6f}, eta={eta:.4f}")
        else:
            stagnant += 1
            eta = max(eta * 0.7, 0.001)
            if stagnant > 50:
                # Try jumping with a larger eta
                eta = best_eta * 2
                stagnant = 0
                if best_eta * 2 > 1.0:
                    break
    
    return best_a, best_score


def lp_refine(a, time_limit_s=300.0, verbose=False):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    stagnant = 0
    it = 0
    max_iters = 200
    
    while time.monotonic() - start_time < time_limit_s and it < max_iters:
        elapsed = time.monotonic() - start_time
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and it % 10 == 0:
                print(f"      LP iter {it}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 20:
            break
        
        timeout = max(2.0, min(30.0, remaining * 0.2))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= a.sum() * (1.0 + 1e-10):
            break
        
        g = g / g.sum()
        # Adaptive mixing
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-10:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
        it += 1
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"      LP done: best={best_score:.6f}, {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def random_search_polish(a, time_limit_s=60.0, rng=None, verbose=False):
    """Random smooth perturbation search."""
    if rng is None:
        rng = np.random.default_rng(123)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0
    
    while time.monotonic() - start_time < time_limit_s:
        # Random perturbation: multiply a window by a random factor
        width = max(2, int(rng.integers(n // 100, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        
        candidate = a.copy()
        noise_scale = rng.uniform(0.05, 0.3)
        candidate[start:stop] *= np.exp(rng.normal(0.0, noise_scale, size=stop - start))
        
        # Occasionally add a global smooth perturbation
        if rng.random() < 0.2:
            t = np.arange(n, dtype=np.float64)
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            candidate *= (1.0 + 0.1 * np.cos(2.0 * np.pi * freq * t / n + phase))
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 20 == 0:
                print(f"      RS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"      RS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def sqp_lp_step(a, timeout=30.0):
    n = len(a)
    conv_a = np.convolve(a, a)
    M = conv_a.max()
    
    # Variables: da[0],...,da[n-1], t => n+1 variables
    # x = [da[0],...,da[n-1], t]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    # conv(a,da)[k] = sum_j a[j]*da[k-j] (valid j)
    # So 2*sum_j a[j]*da[k-j] - t <= -conv_a[k]
    # In matrix form: row k has A[k, j] = 2*a[j] (if 0<=k-j<n) and A[k, n] = -1
    
    n_rows = 2 * n - 1
    A_ub = np.zeros((n_rows, n + 1))
    b_ub = -conv_a  # -conv_a[k]
    
    for j in range(n):
        # 2*a[j] * da[k-j] contributes to row k where 0<=k-j<n => j<=k<=j+n-1
        for k in range(j, j + n):
            A_ub[k, j] += 2.0 * a[j]
    A_ub[:, -1] = -1.0
    b_ub = -conv_a
    
    # Constraint 2: sum(da) = 0 => -sum(da) <= 0 AND sum(da) <= 0
    # Add as two inequality constraints
    A_ub2 = np.zeros((2, n + 1))
    A_ub2[0, :n] = 1.0  # sum(da) <= 0
    A_ub2[1, :n] = -1.0  # -sum(da) <= 0
    b_ub2 = np.zeros(2)
    
    # Constraint 3: da[m] >= -a[m] => -da[m] <= a[m]
    A_ub3 = np.zeros((n, n + 1))
    for m in range(n):
        A_ub3[m, m] = -1.0
    b_ub3 = a  # -da[m] <= a[m]
    
    # Combine
    A_all = np.vstack([A_ub, A_ub2, A_ub3])
    b_all = np.concatenate([b_ub, b_ub2, b_ub3])
    
    bounds = [(None, None)] * (n + 1)
    bounds[n] = (None, None)  # t is free
    
    try:
