import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def true_objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

def compute_grad_smoothed(a, tau):
    """
    Compute smoothed objective value and gradient dF/da.
    a should sum to 1 (normalized).
    """
    n = len(a)
    c = fast_convolve(a)
    
    # Softmax weights
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    w_sum = exp_c.sum()
    w = exp_c / w_sum
    smoothed_max = c.max() + tau * np.log(w_sum)
    
    # h_j = sum_k w_k * a_{k-j} (cross-correlation)
    w_rev = w[::-1]
    conv_len = n + (2*n - 1) - 1
    fsize = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fw = np.fft.rfft(w_rev, fsize)
    conv_full = np.fft.irfft(fa * fw, fsize)[:conv_len]
    h = conv_full[2*n-2::-1][:n]
    
    S = a.sum()
    value = 2.0 * n * smoothed_max / (S * S)
    grad = (4.0 * n / (tau * S * S)) * h - (4.0 * n * smoothed_max / (S * S * S)) * a
    
    return value, grad

def softmax_func(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def adam_step(z, m, v, t, grad_z, lr, beta1, beta2, eps):
    m = beta1 * m + (1 - beta1) * grad_z
    v = beta2 * v + (1 - beta2) * grad_z * grad_z
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v

def lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0.
    Returns b or None.
    """
    n = len(a)
    # Build A matrix: A[k,j] = a[k-j] if 0 <= k-j <= n-1
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    c = -np.ones(n)
    b_ub = M * np.ones(2*n-1)

                  options={'presolve': True})

        if b.sum() > 1e-10:

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam on softmax parameterization with exact gradient of smoothed objective,
LP-based direction finding for periodic restarts, adaptive tau annealing,
and multiple diverse initializations across several sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def true_obj_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def compute_grad_smoothed(a, tau):
    """
    Compute smoothed objective value and gradient dF/da for F = 2n*Smax(c)/S^2.
    a should be non-negative. S = sum(a).
    """
    n = len(a)
    c = fast_convolve(a)

    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    w_sum = exp_c.sum()
    w = exp_c / w_sum
    smoothed_max = c.max() + tau * np.log(w_sum)

    # h_j = sum_k w_k * a_{k-j} via FFT cross-correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1
    fsize = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fw = np.fft.rfft(w_rev, fsize)
    conv_full = np.fft.irfft(fa * fw, fsize)[:conv_len]
    h = conv_full[2 * n - 2::-1][:n]

    S = a.sum()
    value = 2.0 * n * smoothed_max / (S * S)
    grad = (4.0 * n / (tau * S * S)) * h - (4.0 * n * smoothed_max / (S * S * S)) * a

    return value, grad


def softmax_func(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def optimize_adam(a0, time_limit, tau_start, tau_end, lr, seed, perturb_interval=3000, perturb_scale=0.02):
    """
    Adam optimization on softmax parameterization.
    a0: initial non-negative sequence (normalized to sum 1 internally).
    Returns (best_a, best_true_obj_normalized).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(a + 1e-300)
    z = z - z.mean()

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_start

    best_a = a.copy()
    best_obj = true_obj_norm(a)

    start_time = time.monotonic()
    total_time = time_limit
    log_ratio = np.log(tau_end / tau_start)

    while True:
        elapsed = time.monotonic() - start_time
        if elapsed >= total_time:
            break

        progress = elapsed / total_time
        tau = tau_start * np.exp(log_ratio * progress)

        a = softmax_func(z)
        val, grad_a = compute_grad_smoothed(a, tau)

        mean_g = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_g)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z * grad_z
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 20 == 0:
            a_check = softmax_func(z)
            obj_check = true_obj_norm(a_check)
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a_check.copy()

        if t % perturb_interval == 0:
            rng = np.random.default_rng(seed + t)
            z = z + perturb_scale * rng.standard_normal(n)

    a_final = softmax_func(z)
    obj_final = true_obj_norm(a_final)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a_final.copy()

    return best_a, best_obj


def solve_lp_direction(a, M):
    """
    Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0.
    Returns normalized b (sum=1) or None.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            b = res.x
            bsum = b.sum()
            if bsum > 1e-10:
                return b / bsum
    except Exception:
        pass
    return None

        bm = np.maximum(bm, 1e-10)
