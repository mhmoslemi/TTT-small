```python
    best_score = score_sequence(a)

        # Compute softmax weights
        t = t_schedule(it)
        t_conv = t * conv
        t_conv -= t_conv.max()
        p = np.exp(t_conv)
        p /= p.sum()

        R = np.correlate(p, a, mode='full')
        grad = R[n-1:n-1+n] + a
        
        # Multiplicative weight update

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        score = score_sequence(a)
        if score < best_score:

            best_a = a.copy()
```

```python
    log_sum = max_tc + np.log(exp_shifted.sum())
    obj = log_sum / t
    
    # Softmax weights
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient with respect to a
    R = np.correlate(p, a, mode='full')
    grad_a = R[n-1:n-1+n] + a
    
    # Gradient with respect to log_x (chain rule through softmax)
    # da_j/dlog_x_k = a_j * (delta_jk - a_k)
    # dL/dlog_x_k = sum_j dL/da_j * da_j/dlog_x_k
    # = sum_j grad_a_j * a_j * (delta_jk - a_k)
    # = a_k * grad_a_k - a_k * sum_j grad_a_j * a_j
    mean_ga = np.dot(grad_a, a)
    grad_log_x = a * (grad_a - mean_ga)
    
    return obj, grad_log_x

def optimize_smooth_max(a_init, t_start, t_end, n_iters_per_t, n_t_steps, time_limit):
    """
    Animate t from t_start to t_end, using L-BFGS-B at each t.
    """
    start_time = time.monotonic()
    a = a_init / a_init.sum()
    log_x = np.log(a + 1e-30)  # Initialize log_x from a
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    t_schedule = np.geomspace(t_start, t_end, n_t_steps)
    
    for t_idx, t in enumerate(t_schedule):
        if time.monotonic() - start_time > time_limit:
            break
        
        # Use L-BFGS-B
        res = minimize(
            smooth_max_obj_and_grad,
            log_x,
            args=(t,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': n_iters_per_t, 'maxfun': n_iters_per_t * 2}
        )
        
        if res.x is not None:
            new_a = np.exp(res.x - res.x.max())
            new_a = new_a / new_a.sum()
            
            # Update best
            curr_score = score_sequence(new_a)
            if curr_score < best_score:
                best_score = curr_score
                best_a = new_a.copy()
            
            log_x = res.x
            a = new_a
    
    return best_a, best_score
```

Hmm, but L-BFGS-B might be slow for n=500 or larger since each function evaluation requires O(n^2) convolution. Let me think about how many iterations are feasible.

For n=500: convolution is O(n^2) = 250K operations, which is fast (microseconds in numpy).
For n=2000: O(n^2) = 4M operations, still fast.

L-BFGS-B does a few dozen to hundred iterations, each with a function eval and gradient eval. So maybe 100-200 evaluations total per t step. For 10 t steps, that's 1000-2000 evaluations. Each evaluation is ~1ms for n=500. Total: 1-2 seconds. That's very fast!

So I can afford to do many t steps and many restarts.

Let me also think about the t schedule:
- t_start: small, like n/10 = 50 for n=500 (very smooth, like an average)
- t_end: large, like 10*n = 5000 (close to true max)

Actually, the issue with large t is numerical: `exp(t * conv)` can overflow. But since I'm doing the shift by max, the largest exponent is 0, so it should be fine. The smallest could be very negative, but that's also fine (just underflows to 0).

The issue is that with very large t, the optimization landscape becomes very "peaked" (only the maximum lags matter), which can make gradient methods struggle. So a moderate t_end like 5*n might be better, followed by a "refinement" phase where we directly minimize the true max using subgradient methods.

Let me also consider: after the smooth max optimization, I can apply the LP-based refinement from the previous solution, which should work well as a polishing step.

Let me also add a **direct subgradient method** for the true max:
- At each step, find the argmax of the convolution
- Compute the subgradient at that point
- Take a step in the negative subgradient direction
- Project onto the simplex {a >= 0, sum(a) = 1}

The projection onto the simplex can be done efficiently:
1. Sort the values in descending order
2. Find the threshold tau such that sum(max(x_i - tau, 0)) = 1
3. Project: x_i <- max(x_i - tau, 0)

```python
"""
Hybrid optimization for peak auto-convolution minimization:
1. Smooth max (log-sum-exp) with L-BFGS-B gradient descent
2. Direct subgradient method on true max with simplex projection
3. LP-based refinement for final polishing
4. Multi-start with various n and initializations
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))

def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s

def uniforms_modulated(n, freq, amplitude):
    """Uniform sequence modulated by a sinusoid."""
    t = np.arange(n, dtype=np.float64)
    return np.maximum(0, 1.0 + amplitude * np.cos(2 * np.pi * freq * t / n))

def smooth_max_obj_and_grad(log_x, t):
    n = len(log_x)
    x_max = log_x.max()
    a = np.exp(log_x - x_max)
    a_sum = a.sum()
    a = a / a_sum
    
    conv = np.convolve(a, a)
    
    t_conv = t * conv
    max_tc = t_conv.max()
    exp_shifted = np.exp(t_conv - max_tc)
    log_sum = max_tc + np.log(exp_shifted.sum())
    obj = log_sum / t
    
    p = exp_shifted / exp_shifted.sum()
    
    R = np.correlate(p, a, mode='full')
    grad_a = R[n-1:n-1+n] + a
    
    mean_ga = np.dot(grad_a, a)
    grad_log_x = a * (grad_a - mean_ga)
    
    return obj, grad_log_x

def optimize_smooth_max_lagar(a_init, t_values, max_iter_per_t, time_limit, verbose=False):
    start_time = time.monotonic()
    a = a_init / a_init.sum()
    log_x = np.log(a + 1e-30)
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    for t in t_values:
        if time.monotonic() - start_time > time_limit:
            break
        
        res = minimize(
            smooth_max_obj_and_grad,
            log_x,
            args=(t,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_t, 'gtol': 1e-8}
        )
        
        if res.x is not None:
            new_a = np.exp(res.x - res.x.max())
            new_a = new_a / new_a.sum()
            
            curr_score = score_sequence(new_a)
            if curr_score < best_score:
                best_score = curr_score
                best_a = new_a.copy()
            
            log_x = res.x
            a = new_a
            if verbose:
                print(f"      t={t:.1f}, score={curr_score:.6f}")
    
    return best_a, best_score

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        # Fallback: uniform
        return np.ones(n) / n
    rho_hat = np.where(cond)[0][0] + 1
    theta = (cssv[rho_hat] - 1) / rho_hat
    return np.maximum(v - theta, 0)

def subgradient_max_optimize(a_init, n_iters, eta0, time_limit, verbose=False):
    """
    Direct subgradient method to minimize max(conv(a,a)) 
    subject to a >= 0, sum(a) = 1.
    """
    start_time = time.monotonic()
    a = a_init / a_init.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = score_sequence(a)
    best_max = np.convolve(a, a).max()
    
    for it in range(n_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv = np.convolve(a, a)
        max_idx = conv.argmax()
        max_val = conv[max_idx]
        
        if max_val < best_max:
            best_max = max_val
            best_a = a.copy()
            best_score = score_sequence(a)
        
        # Subgradient of max at a:
        # d(conv)_k / da_j = a_{k-j} + a_j (for valid indices)
        # So subgradient = a_{max_idx - j} + a_j for j = 0, ..., n-1
        # (with a_k = 0 for k outside [0, n-1])
        
        subgrad = np.zeros(n)
        for j in range(n):
            k_j = max_idx - j
            val = a[j]
            if 0 <= k_j < n:
                val += a[k_j]
            subgrad[j] = val
        
        # Step: a_new = a - eta * subgrad, then project to simplex
        eta = eta0 / (1.0 + it * 0.01)
        a_new = a - eta * subgrad
        a = project_simplex(a_new)
        
        if verbose and it % 100 == 0:
            print(f"      Subgrad iter {it}: max={max_val:.8f}, score={score_sequence(a):.6f}")
    
    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine(a, time_limit_s=120.0, max_iters=200, verbose=False):
    start_time = time.monotonic()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 30:
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(3.0, min(30.0, remaining * 0.2))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.5 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(0, a_new)
        a = a_new / a_new.sum()
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters")
    return best_a, best_score

def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(max(2, n // 200), max(3, n // 8))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.05, size=stop - start))
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.maximum(0, candidate)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 20 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}")
    return best_a, best_score
```

5. Update: a = (1-t)*a + t*y for some step size t

```python
"""
Hybrid optimization for peak auto-convolution minimization:
- Smooth max (log-sum-exp) with L-BFGS-B gradient descent via softmax parameterization
- Direct subgradient method on true max with simplex projection
- LP-based refinement for final polishing
- Multi-start with various n and initializations including frequency-modulated uniforms
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def uniforms_modulated(n, freq, amplitude):
    t = np.arange(n, dtype=np.float64)
    return np.maximum(0, 1.0 + amplitude * np.cos(2 * np.pi * freq * t / n))


def uniforms_dual_mod(n, f1, f2, a1, a2):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + a1 * np.cos(2 * np.pi * f1 * t / n) + a2 * np.cos(2 * np.pi * f2 * t / n)
    return np.maximum(0, a)


def smooth_max_obj_and_grad(log_x, t):
    n = len(log_x)
    x_max = log_x.max()
    a = np.exp(log_x - x_max)
    a_sum = a.sum()
    a = a / a_sum

    conv = np.convolve(a, a)

    t_conv = t * conv
    max_tc = t_conv.max()
    exp_shifted = np.exp(t_conv - max_tc)
    log_sum = max_tc + np.log(exp_shifted.sum())
    obj = log_sum / t

    p = exp_shifted / exp_shifted.sum()

    R = np.correlate(p, a, mode='full')
    grad_a = R[n - 1:n - 1 + n] + a

    mean_ga = np.dot(grad_a, a)
    grad_log_x = a * (grad_a - mean_ga)

    return obj, grad_log_x


def optimize_smooth_max_lbfgs(a_init, t_values, max_iter_per_t, time_limit, verbose=False):
    start_time = time.monotonic()
    a = a_init / a_init.sum()
    log_x = np.log(a + 1e-30)

    best_a = a.copy()
    best_score = score_sequence(a)

    for t in t_values:
        if time.monotonic() - start_time > time_limit:
            break

        res = minimize(
            smooth_max_obj_and_grad,
            log_x,
            args=(t,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_t, 'gtol': 1e-10, 'ftol': 1e-12}
        )

        if res.x is not None:
            new_a = np.exp(res.x - res.x.max())
            new_a = new_a / new_a.sum()

            curr_score = score_sequence(new_a)
            if curr_score < best_score:
                best_score = curr_score
                best_a = new_a.copy()

            log_x = res.x
            a = new_a
            if verbose:
                print(f"      t={t:.1f}, score={curr_score:.6f}")

    return best_a, best_score


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.where(cond)[0][0] + 1
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def subgradient_max_optimize(a_init, n_iters, eta0, time_limit, verbose=False):
    start_time = time.monotonic()
    a = a_init / a_init.sum()
    n = len(a)

    best_a = a.copy()
    best_score = score_sequence(a)
    best_max = np.convolve(a, a).max()

    for it in range(n_iters):
        if time.monotonic() - start_time > time_limit:
            break

        conv = np.convolve(a, a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        if max_val < best_max - 1e-15:
            best_max = max_val
            best_a = a.copy()
            best_score = score_sequence(a)

        # Vectorized subgradient computation
        k_vals = max_idx - np.arange(n)
        valid = (k_vals >= 0) & (k_vals < n)
        subgrad = a.copy()
        subgrad[valid] += a[k_vals[valid]]

        eta = eta0 / (1.0 + it * 0.05)
        a_new = a - eta * subgrad
        a = project_simplex(a_new)

        if verbose and it % 200 == 0:
            print(f"      Subgrad iter {it}: max={max_val:.8f}, score={score_sequence(a):.6f}")

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=20.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=120.0, max_iters=200, verbose=False):
    start_time = time.monotonic()
    a = np.maximum(0, a).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, score_sequence(a)
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 30:
            break

        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(25.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.5 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it + 1} iters")
    return best_a, best_score


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = np.maximum(0, a).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = initial_raised_cosine(len(a))
        s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(3, int(rng.integers(max(3, n // 300), max(5, n // 6))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        noise_scale = 0.03 + 0.04 * (1.0 - (time.monotonic() - start_time) / max(1, time_limit_s))
        candidate[start:stop] *= np.exp(rng.normal(0.0, noise_scale, size=stop - start))
        if rng.random() < 0.25:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.maximum(0, candidate)
        sc = candidate.sum()
        if sc < 1e-15:
            continue
        candidate = candidate / sc
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}")
    return best_a, best_score


def frank_wolfe_max(a_init, n_iters, time_limit, verbose=False):
    """Frank-Wolfe (conditional gradient) on the true max of auto-convolution."""
    start_time = time.monotonic()
    a = a_init / a_init.sum()
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    for it in range(n_iters):
        if time.monotonic() - start_time > time_limit:
            break

        conv = np.convolve(a, a)
        max_idx = int(conv.argmax())

        k_vals = max_idx - np.arange(n)
        valid = (k_vals >= 0) & (k_vals < n)
        subgrad = a.copy()
        subgrad[valid] += a[k_vals[valid]]

        # Linear minimization over simplex: min <subgrad, y> s.t. y in simplex
        # Solution: y = e_{argmin subgrad}
        min_idx = int(np.argmin(subgrad))

        # FW step size: t_k = 2/(k+2)
        t_step = 2.0 / (it + 2.0)
        y = np.zeros(n)
        y[min_idx] = 1.0
        a = (1.0 - t_step) * a + t_step * y
        a = np.maximum(0, a)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        if it % 100 == 0:
            curr_score = score_sequence(a)
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()
            if verbose:
                print(f"      FW iter {it}: score={curr_score:.6f}")

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimizer: smooth-max L-BFGS + subgradient + Frank-Wolfe + LP + local search.
    Multi-start over various lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 25.0)

    best_score = float('inf')
    best_seq = None

    # Try to use height_sequence_1 as a starting point
    has_prior = False
    try:
        hs1 = height_sequence_1
        if hs1 is not None and len(hs1) > 0:
            has_prior = True
            prior = np.array([float(max(0, x)) for x in hs1])
            ps = prior.sum()
            if ps > 1e-10:
                prior = prior / ps
                prior_score = score_sequence(prior)
                print(f"Prior sequence: n={len(prior)}, score={prior_score:.6f}")
                if prior_score < best_score:
                    best_score = prior_score
                    best_seq = prior.copy()
    except (NameError, AttributeError, TypeError):
        pass

    # Define search tasks: (n, init_type, time_frac)
    tasks = []

    # Prior-based refinement (if available)
    if has_prior:
        n_prior = len(prior)
        tasks.append((n_prior, 'prior', 0.12))

    # Core search tasks
    core_tasks = [
        (500, 'raised_cosine', 0.10),
        (500, 'uniform_mod1', 0.08),
        (500, 'uniform_mod2', 0.08),
        (500, 'gaussian', 0.07),
        (500, 'random', 0.06),
        (1000, 'raised_cosine', 0.10),
        (1000, 'uniform_mod1', 0.08),
        (1000, 'gaussian', 0.07),
        (2000, 'raised_cosine', 0.08),
        (2000, 'uniform_mod1', 0.06),
        (1500, 'raised_cosine', 0.06),
        (1500, 'uniform_mod2', 0.05),
    ]
    tasks.extend(core_tasks)

    # Sort by time fraction descending to prioritize longer budgets
    # Actually, let's just iterate in order

    for n, init_type, time_frac in tasks:
        remaining = deadline - time.monotonic()
        if remaining < 20.0:
            break

        t_budget = min(time_frac * budget_s, remaining * 0.9)
        if t_budget < 10.0:
            continue

        print(f"\n=== n={n}, init={init_type}, budget={t_budget:.0f}s ===")

        # Generate initial sequence
        if init_type == 'prior':
            a = prior if n == len(prior) else initial_raised_cosine(n)
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=float(rng.uniform(0.15, 0.45)))
        elif init_type == 'random':
            a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 7)))
        elif init_type == 'uniform_mod1':
            freq = int(rng.integers(1, max(2, n // 6)))
            amp = float(rng.uniform(0.15, 0.6))
            a = uniforms_modulated(n, freq, amp)
        elif init_type == 'uniform_mod2':
            f1 = int(rng.integers(1, max(2, n // 15)))
            f2 = int(rng.integers(max(2, n // 15 + 1), max(3, n // 4)))
            a1 = float(rng.uniform(0.2, 0.45))
            a2 = float(rng.uniform(0.1, 0.35))
            a = uniforms_dual_mod(n, f1, f2, a1, a2)
        else:
            a = initial_raised_cosine(n)

        a = np.maximum(0, a).astype(np.float64)
        s = a.sum()
        if s < 1e-10:
            a = initial_raised_cosine(n)
            s = a.sum()
        a = a / s

        init_score = score_sequence(a)
        print(f"    Init score: {init_score:.6f}")

        # Phase 1: Smooth max optimization (most effective)
        t1_frac = 0.45
        t1 = t_budget * t1_frac
        n_t_steps = max(5, int(np.log2(n) + 3))
        t_values = np.geomspace(n * 0.05, n * 6.0, num=n_t_steps)
        a1, s1 = optimize_smooth_max_lbfgs(a, t_values, max_iter_per_t=40, time_limit=t1, verbose=False)

        # Phase 2: Subgradient refinement
        t2_frac = 0.20
        t2 = t_budget * t2_frac
        a2, s2 = subgradient_max_optimize(a1, n_iters=800, eta0=0.05 * n, time_limit=t2, verbose=False)

        # Phase 3: Frank-Wolfe refinement
        t3_frac = 0.15
        t3 = t_budget * t3_frac
        a3, s3 = frank_wolfe_max(a2, n_iters=500, time_limit=t3, verbose=False)

        # Phase 4: LP refinement
        t4 = t_budget - t1 - t2 - t3
        if t4 < 5.0:
            t4 = 5.0
        a4, s4 = lp_refine(a3, time_limit_s=t4, verbose=False)

        # Track best
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()
            print(f"    *** NEW BEST: score={best_score:.6f} (n={n}, {init_type})")

    # Final polishing phase
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")

        if remaining > 40.0:
            # Local search
            t_ls = min(remaining * 0.25, 150.0)
            polished, s = local_search_refine(best_seq, time_limit_s=t_ls, rng=rng, verbose=True)
            if s < best_score:
                best_score = s
                best_seq = polished
                print(f"  After LS: {best_score:.6f}")

        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            # LP polish
            t_lp = min(remaining * 0.3, 120.0)
            polished2, s2 = lp_refine(best_seq, time_limit_s=t_lp, verbose=True)
            if s2 < best_score:
                best_score = s2
                best_seq = polished2
                print(f"  After LP: {best_score:.6f}")

        remaining = deadline - time.monotonic()
        if remaining > 20.0:
            # Another smooth max pass on the polished result
            n_cur = len(best_seq)
            t_values2 = np.geomspace(n_cur * 0.3, n_cur * 10.0, num=6)
            polished3, s3 = optimize_smooth_max_lbfgs(best_seq, t_values2,
                                                       max_iter_per_t=30,
                                                       time_limit=remaining * 0.4, verbose=True)
            if s3 < best_score:
                best_score = s3
                best_seq = polished3
                print(f"  After smooth max: {best_score:.6f}")

        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            # Final subgradient + LP
            polished4, s4 = subgradient_max_optimize(best_seq, n_iters=300, eta0=0.02 * len(best_seq),
                                                     time_limit=remaining * 0.3, verbose=False)
            if s4 < best_score:
                best_score = s4
                best_seq = polished4
            remaining = deadline - time.monotonic()
            if remaining > 10.0:
                polished5, s5 = lp_refine(best_seq, time_limit_s=remaining * 0.8, verbose=True)
                if s5 < best_score:
                    best_score = s5
                    best_seq = polished5
                    print(f"  After final LP: {best_score:.6f}")

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
```
