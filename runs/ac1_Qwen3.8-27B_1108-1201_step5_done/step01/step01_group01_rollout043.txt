import numpy as np
import time

def fft_convolve(a, b):
    """Compute convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L *= 2
    A = np.zeros(L)
    B = np.zeros(L)
    A[:len(a)] = a
    B[:len(b)] = b
    return np.fft.irfft(np.fft.rfft(A) * np.fft.rfft(B), n=L)[:n]

def softmax(v):
    v = v - np.max(v)
    e = np.exp(v)
    return e / np.sum(e)

def compute_b(a):
    """Compute a * a (full convolution)."""
    return fft_convolve(a, a)

def objective_and_grad(v, beta):
    """Compute the smooth objective and its gradient w.r.t. v."""
    n = len(v)
    a = softmax(v)
    b = compute_b(a)
    
    # Smooth objective: (1/beta) * log(sum(exp(beta * b)))
    log_weights = beta * b
    log_weights_shifted = log_weights - np.max(log_weights)
    log_sum = np.max(log_weights) + np.log(np.sum(np.exp(log_weights_shifted)))
    obj = log_sum / beta
    
    # Gradient of obj w.r.t. v
    # p = softmax(beta * b) (normalized)
    p = np.exp(log_weights_shifted) / np.sum(np.exp(log_weights_shifted))
    
    # df/da_i = sum_k p[k] * (db_k/da_i) = sum_k p[k] * a[k-i] (for valid k-i)
    # = cross-correlation of p with a (as derived earlier)
    # df_da[i] = sum_{m=0}^{n-1} a[m] * p[i+m]
    # = (a_reversed * p)[n-1 : n-1+n]
    
    a_rev = a[::-1]
    df_da_full = fft_convolve(a_rev, p)
    df_da = df_da_full[n-1 : n-1+n]
    
    # Chain rule through softmax:
    # a = softmax(v), da_i/dv_j = a_i * (delta_ij - a_j)
    # df/dv_i = sum_j df_da_j * da_j/dv_i = a_i * (df_da_i - sum_j df_da_j * a_j)
    s = np.dot(df_da, a)
    grad_v = a * (df_da - s)

def optimize_starting_point(v_init, deadline, beta_start=1.0, beta_end=100.0, n_steps=None):
    """Optimize from v_init using Adam with annealed beta."""
    n = len(v_init)
    v = v_init.copy()
    m = np.zeros(n)  # First moment
    u = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    t = 0
    best_score = np.inf
    best_v = v.copy()
    
    total_time = deadline - time.monotonic()
    if n_steps is None:
        n_steps = int(total_time * 50)  # ~50 iterations per second target
    
    for t in range(1, n_steps + 1):
        if time.monotonic() >= deadline:
            break
        
        # Anneal beta
        progress = t / n_steps
        beta = beta_start * (beta_end / beta_start) ** progress
        
        obj, grad = objective_and_grad(v, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        u = beta2 * u + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t) 
        u_hat = u / (1 - beta2 ** t)
        v -= lr * m_hat / (np.sqrt(u_hat) + eps)
        
        # Evaluate true objective periodically
        if t % 100 == 0:
            a = softmax(v)
            b = compute_b(a)
            score = 2.0 * n * np.max(b) / (np.sum(a) ** 2)
            if score < best_score:
                best_score = score
                best_v = v.copy()
    
    return best_v, best_score

    best_val = 2.0 * n * np.max(_compute_b(a))
    v_momentum = np.zeros(n, dtype=np.float64)
    eta = eta_init
    no_improve = 0
    step = 0
    total_steps_est = max(1, int((deadline - time.monotonic()) * 2000))

    while time.monotonic() < deadline:
        step += 1
        progress = min(1.0, step / total_steps_est)
        sharpness = beta_start * (beta_end / beta_start) ** progress

        b = _compute_b(a)
        b_max = b.max()
        cur_val = 2.0 * n * b_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            eta *= 0.95
            no_improve = 0
            v_momentum *= 0.5
            if eta < 1e-14:
                break

        M = min(50, max(10, int(10 + 40 * (1 - progress))))
        grad = _subgradient(a, b, M, sharpness)
        grad -= grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        v_momentum = momentum * v_momentum - eta * grad / gnorm
        a_new = a + v_momentum
        a_new = _project_simplex(a_new)

        if not np.all(np.isfinite(a_new)):
            break
        a = a_new

    return best_a, best_val


def _local_search(a_init, budget, rng, n_trials=200):
    """
    Stochastic local search: try random smooth perturbations, keep improvements.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_score = _true_score(a)
    deadline = time.monotonic() + budget

    for trial in range(n_trials):
        if time.monotonic() >= deadline:
            break
        # Generate a smooth random direction
        freqs = rng.integers(1, max(2, n // 5), size=min(4, n // 3))
        direction = np.zeros(n)
        for f in freqs:
            phase = rng.uniform(0, 2 * np.pi)
            dir_type = rng.random()
            if dir_type < 0.5:
                direction += np.sin(2 * np.pi * f * np.arange(n) / n + phase)
            else:
                direction += np.cos(2 * np.pi * f * np.arange(n) / n + phase)
        direction -= direction.mean()
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-10:
            continue
        direction /= dnorm

        # Try several step sizes
        for scale in [0.05, 0.02, 0.01]:
            a_trial = a + scale * direction
            a_trial = _project_simplex(a_trial)
            s = _true_score(a_trial)
            if s < best_score:
                best_score = s
                best_a = a_trial.copy()
                a = a_trial.copy()
                break

    return best_a, best_score


def _make_anti_sinusoidal(n, eps, freq=1, rng=None):
    """Anti-palindromic sinusoidal: proportional to 1 + eps*sin(2*pi*freq*i/(n-1) + phase)."""
    i = np.arange(n, dtype=np.float64)
    if rng is not None:
        phase = rng.uniform(0, 2 * np.pi)
    else:
        phase = 0.0
    a = 1.0 + eps * np.sin(2.0 * np.pi * freq * i / (n - 1) + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_sigmoid(n, alpha, rng=None):
    """Sigmoid: exactly anti-palindromic (a[i] + a[n-1-i] = 1 before normalization)."""
    i = np.arange(n, dtype=np.float64)
    x = alpha * (i - (n - 1) / 2.0)
    a = 1.0 / (1.0 + np.exp(-np.clip(x, -500, 500)))
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def _make_two_level_smooth(n, eps, transition_width=0.2):
    """Two-level with smooth transition: high on left, low on right."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    # Smooth step from 1 to 0 centered at 0.5
    t = (i - 0.5) / transition_width
    smooth_step = 1.0 / (1.0 + np.exp(4 * t))
    a = (1 - eps) + eps * smooth_step
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def _make_fourier_mix(n, rng, n_terms=3):
    """Random Fourier mixture."""
    a = np.ones(n, dtype=np.float64)
    for _ in range(n_terms):
        freq = rng.integers(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-0.4, 0.4)
        a += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0.05)
    s = a.sum()
    return a / s


def _refine_best(a_init, deadline, rng):
    """
    Multi-strategy refinement of the best sequence found so far.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _true_score(a_init)
    remaining = deadline - time.monotonic()

    if remaining < 30:
        return best_a, best_score

    # Phase A: Polished subgradient with focused peaks (small M, high sharpness)
    t_alloc = 0.3 * remaining
    d1 = min(time.monotonic() + t_alloc, deadline)
    a1, s1 = _optimize(best_a, d1, M=3, sharpness=30.0, eta_init=0.05,
                       momentum=0.95, patience=200, cool_rate=0.97)
    if s1 < best_score:
        best_a, best_score = a1, s1
        print(f"  Refine A: improved to {best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase B: Smoothed annealing
    t_alloc = 0.3 * remaining
    d2 = min(time.monotonic() + t_alloc, deadline)
    a2, s2 = _optimize_smooth(best_a, d2, beta_start=2.0, beta_end=100.0,
                              eta_init=0.05, momentum=0.9)
    if s2 < best_score:
        best_a, best_score = a2, s2
        print(f"  Refine B: improved to {best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase C: Local search
    a3, s3 = _local_search(best_a, remaining * 0.5, rng, n_trials=500)
    if s3 < best_score:
        best_a, best_score = a3, s3
        print(f"  Refine C: improved to {best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Phase D: Another polish
        d4 = time.monotonic() + remaining
        a4, s4 = _optimize(best_a, d4, M=5, sharpness=20.0, eta_init=0.03,
                           momentum=0.95, patience=300, cool_rate=0.98)
        if s4 < best_score:
            best_a, best_score = a4, s4
            print(f"  Refine D: improved to {best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
    
    Phase 1: Global search over diverse anti-palindromic starting points.
    Phase 2: Refinement of the best candidate using multiple strategies.
    Phase 3: Diversification from the best (perturb + re-optimize).
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    print(f"Sequence length: {n}")

    # Global best tracking
    global_best = None
    global_best_score = np.inf

    def _update_best(a, s):
        nonlocal global_best, global_best_score
        if s < global_best_score:
            global_best = a.copy()
            global_best_score = s

    # ================================================================
    # PHASE 1: Global search over diverse starting points
    # ================================================================
    print("=" * 60)
    print("PHASE 1: Global search")
    print("=" * 60)

    starts = []

    # Start 0: Given parent sequence
    a0 = parent.copy()
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 > 1e-10:
        a0 = a0 / s0
    else:
        a0 = np.ones(n) / n
    starts.append(("parent", a0))

    # Start 1-2: Sigmoid profiles (exactly anti-palindromic)
    for alpha in [2.0, 4.0]:
        starts.append((f"sigmoid_{alpha}", _make_sigmoid(n, alpha)))

    # Start 3-4: Smooth two-level
    for tw in [0.1, 0.3]:
        starts.append((f"twolvl_{tw}", _make_two_level_smooth(n, 0.8, tw)))

    # Start 5-6: Anti-sinusoidal
    for freq in [1, 2]:
        starts.append((f"anti_sin_{freq}", _make_anti_sinusoidal(n, 0.7, freq)))

    # Start 7-8: Random Fourier mixes
    starts.append(("rand_ft_0", _make_fourier_mix(n, rng, n_terms=3)))
    starts.append(("rand_ft_1", _make_fourier_mix(n, rng, n_terms=5)))

    # Allocate time: parent gets 25%, rest split equally
    n_starts = len(starts)
    total_time = max(1.0, total_deadline - time.monotonic())
    time_parent = 0.25 * total_time
    time_each_other = 0.75 * total_time / (n_starts - 1)

    for i, (name, a_start) in enumerate(starts):
        if time.monotonic() >= total_deadline:
            break
        if i == 0:
            t_alloc = time_parent
        else:
            t_alloc = time_each_other
        start_deadline = min(time.monotonic() + t_alloc, total_deadline)

        if t_alloc < 2.0:
            # Too little time, just evaluate
            s = _true_score(a_start)
            _update_best(a_start, s)
            print(f"  [{name}] (eval only): score={s:.8f}")
            continue

        # Use smooth annealing for the optimization
        best_a, best_s = _optimize_smooth(
            a_start, start_deadline,
            beta_start=3.0, beta_end=150.0,
            eta_init=0.15, momentum=0.9
        )
        _update_best(best_a, best_s)
        print(f"  [{name}] score={best_s:.8f} (best={global_best_score:.8f})")

    # ================================================================
    # PHASE 2: Refinement of the best
    # ================================================================
    print("=" * 60)
    print("PHASE 2: Refinement")
    print("=" * 60)
    print(f"  Current best: {global_best_score:.8f}")

    if global_best is not None:
        global_best, global_best_score = _refine_best(
            global_best, total_deadline, rng
        )
        print(f"  After refinement: {global_best_score:.8f}")

    # ================================================================
    # PHASE 3: Diversification
    # ================================================================
    remaining = total_deadline - time.monotonic()
    print("=" * 60)
    print(f"PHASE 3: Diversification ({remaining:.1f}s remaining)")
    print("=" * 60)

    if global_best is not None and remaining > 30:
        n_div = max(1, int(remaining / 80))
        per_div = remaining / (n_div + 0.5)

        for d in range(n_div):
            if time.monotonic() >= total

_total_deadline:
                break

            # Create diverse perturbation
            pert_scale = rng.uniform(0.05, 0.15)
            a_pert = global_best + pert_scale * _make_fourier_mix(n, rng, n_terms=2)
            a_pert = np.maximum(a_pert, 0.0)
            s = a_pert.sum()
            if s > 1e-10:
                a_pert /= s
            else:
                a_pert = global_best.copy()

            div_deadline = min(time.monotonic() + per_div, total_deadline)
            if div_deadline - time.monotonic() < 5.0:
                break

            best_d, score_d = _optimize_smooth(
                a_pert, div_deadline,
                beta_start=5.0, beta_end=100.0,
                eta_init=0.1, momentum=0.9
            )
            _update_best(best_d, score_d)
            print(f"  Div {d+1}/{n_div}: score={score_d:.8f} (best={global_best_score:.8f})")

    # ================================================================
    # FINAL POLISH
    # ================================================================
    remaining = total_deadline - time.monotonic()
    if global_best is not None and remaining > 10:
        print(f"Final polish with {remaining:.1f}s")
        a_final, s_final = _optimize(
            global_best, total_deadline,
            M=3, sharpness=50.0, eta_init=0.02,
            momentum=0.95, patience=500, cool_rate=0.98
        )
        _update_best(a_final, s_final)

    # Final local search
    remaining = total_deadline - time.monotonic()
    if global_best is not None and remaining > 10:
        a_ls, s_ls = _local_search(global_best, remaining, rng, n_trials=300)
        _update_best(a_ls, s_ls)

    # ================================================================
    # Return best
    # ================================================================
    if global_best is None:
        global_best = a0
        global_best_score = _true_score(global_best)

    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.8f}")
    print(f"  internal: {global_best_score:.8f}, evaluated: {final_score:.8f}")
    return result
