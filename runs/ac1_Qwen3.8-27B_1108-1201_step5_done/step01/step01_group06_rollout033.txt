```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences of length 8192.

Approach:
1. Multi-start optimization with diverse initializations.
2. Adam optimizer with projected gradient, adaptive top-k subgradient.
3. Periodic perturbations to escape local minima.
4. Final local search (pair-based coordinate descent).
5. Time-budgeted with verification via evaluate_sequence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_indices):
    """Subgradient of max(c) using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def run_optimization(a_init, n, S, max_time, rng, lr_init, top_k, kick_every, kick_scale_init):
    """Adam optimizer with simplex projection and periodic kicks."""
    a = safe_normalize(a_init, n)
    m_v = np.zeros(n, dtype=np.float64)
    v_v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    
    # Initial evaluation
    c = fft_convolve_self(a)
    best_score = c.max()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Compute subgradient with top-k lags
        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, n, top_indices, c)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm
        
        # Adam update
        m_v = beta1 * m_v + (1.0 - beta1) * grad
        v_v = beta2 * v_v + (1.0 - beta2) * (grad ** 2)
        m_hat = m_v / (1.0 - beta1 ** t)
        v_hat = v_v / (1.0 - beta2 ** t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Learning rate decay
        lr = lr_init * max(0.01, 1.0 / (1.0 + 0.002 * t))
        
        # Periodic kick to escape local minima
        if t % kick_every == 0:
            kick_scale = kick_scale_init * max(0.1, 1.0 / (1.0 + 0.01 * t))
            noise = rng.normal(0.0, kick_scale * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
        
        # Safety: ensure a is valid
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = np.full(n, S / n, dtype=np.float64)
        elif abs(s - S) / S > 0.1:
            a = a * (S / s)
    
    return best_local, best_score


def run_multiplicative(a_init, n, S, max_time, rng, eta_init, top_k):
    """Multiplicative weights / mirror descent."""
    a = safe_normalize(a_init, n)
    best_score = np.inf
    best_local = a.copy()
    eta = eta_init
    t = 0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        top_indices = np.argsort(c)[::-1][:top_k]
        grad = compute_subgradient(a, n, top_indices, c)
        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)
        
        eta = eta_init / (1.0 + 0.003 * t)
    
    return best_local, best_score


def run_local_search(a_init, n, S, max_time, rng, top_k=8):
    """Local search: try moving mass between pairs summing to peak lag."""
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Find peak lag
        peak_lag = int(np.argmax(c))
        
        # Find the pairs (i, j) with i + j = peak_lag that contribute most
        # c[peak_lag] = sum_i a[i] * a[peak_lag - i]
        lo = max(0, peak_lag - n + 1)
        hi = min(n - 1, peak_lag)
        
        if lo > hi:
            break
        
        # Get contributions from each pair
        contributions = a[lo:hi+1] * a[peak_lag - hi:peak_lag - lo + 1][::-1]
        
        # Try to reduce the largest contribution
        max_idx = int(np.argmax(contributions))
        i = lo + max_idx
        j = peak_lag - hi + (hi - lo - max_idx)  # = peak_lag - i
        j = peak_lag - i
        
        if j < 0 or j >= n or i == j:
            # Try a different approach: move mass to a random position
            if a[i] > 1e-10:
                delta = rng.uniform(0.0, 0.05 * a[i])
                a[i] -= delta
                a[rng.integers(n)] += delta
                a = np.maximum(a, 0.0)
                s = a.sum()
                a = a * (S / s)
            continue
        
        # Try moving a small amount from the larger of a[i], a[j] to reduce the product
        if a[i] >= a[j]:
            # Reduce a[i]
            delta = rng.uniform(0.0, 0.02 * a[i])
            a[i] -= delta
            # Put it at a random position far from the peak
            new_pos = rng.integers(n)
            a[new_pos] += delta
        else:
            delta = rng.uniform(0.0, 0.02 * a[j])
            a[j] -= delta
            new_pos = rng.integers(n)
            a[new_pos] += delta

        s = a.sum()
        if s > 0:
            a = a * (S / s)
        
        # Accept if better
        c_new = fft_convolve_self(a)
        if c_new.max() > best_score * 1.05:
            # Revert
            a = best_local.copy()

def generate_initializations(rng, n, given_seq=None):

        for _ in range(3):
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_indices):
    """Weighted subgradient of max(c) using top-k active lags.
    
    dc[k]/da[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)
```

```python
Minimize 2*n*max_conv / sum^2 over non-negative sequences of length 8192.

Strategy:
1. Multi-start with diverse initializations (given sequence, structured, random).
2. Adam optimizer with projected gradient onto the non-negative simplex,
   using a larger top-k subgradient for smoother descent.
3. Simulated annealing style informed kicks to escape local minima.
4. Multiplicative weights (mirror descent) as complementary optimizer.
5. Greedy local search polishing at the peak lag.
6. Time-budgeted with periodic verification via evaluate_sequence.

import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_indices):
    """Weighted subgradient of max(c) using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad

    best_score = c.max()

    kick_period = 2000
    T_scale = 0.04 * S / np.sqrt(n)

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()

            best_score = cur
            best_local = a.copy()

        top_indices = np.argsort(c)[::-1][:top_k]

def generate_initializations(rng, n, given_seq=None):
    """Generate diverse initializations."""
    inits = []

        except Exception:
            pass

    inits.append(("uniform", np.ones(n, dtype=np.float64)))

    center = (n - 1) / 2.0
    for sigma_mult in [2.0, 2.5, 3.0]:
        sigma = n / sigma_mult
        gauss = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits.append(("gauss_%d" % int(sigma_mult * 10), gauss))

    cos_win = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
    inits.append(("cosine", np.maximum(cos_win, 0.01)))

    for freq in [4, 8, 16, 32]:

        for _ in range(5):
            raw = (raw + np.roll(raw, 1) + np.roll(raw, -1)) / 3.0
        inits.append(("smooth_r", np.maximum(raw, 0.01)))

    # Two-bump
    n_half = n // 2
    for width in [n // 8, n // 6]:

        two_bump[:n_half] = np.exp(-0.5 * ((np.arange(n_half) - n_half // 2) / width) ** 2)
```
