def score_sequence(a):

    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')

def obj_and_grad(a, beta, n):
    """Compute smooth max objective and gradient.
    f(a) = 2n * logsumexp(beta * conv(a,a)) / (beta * sum(a)^2)
    """
    a = np.maximum(0, a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros_like(a)
    
    # logsumexp with numerical stability
    c_shifted = c - c.max()
    log_sum_exp = c.max() + np.log(np.sum(np.exp(beta * c_shifted)))

    f = 2.0 * n * log_sum_exp / (beta * S * S)

    weights = np.exp(beta * c_shifted)
    weights /= weights.sum()
    
    # C^T @ weights: (C^T @ w)[i] = sum_j a[j] * w[i+j]

    Ct_w = np.correlate(w, a, mode='valid')

    grad = 2.0 * n * Ct_w / (S * S) - 4.0 * n * log_sum_exp / (beta * S * S * S)
    
    return f, grad

            # And we minimize 2n * that
            grad = 2.0 * n * Ct_w

def proj_simplex(v, s=1.0):
    """Euclidean projection onto {a >= 0, sum(a) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - s) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, s / n)
    rho_star = idx[-1] + 1
    tau = (cssv[rho_star - 1] - s) / rho_star
    return np.maximum(0.0, v - tau)

Let me think about this differently. The key is to make the auto-convolution as flat as possible. For a "rectangular" pulse of width m within a sequence of length n (with zeros elsewhere), the auto-convolution is a triangle of width 2m-1 with peak m. Score = `2n * m / m^2 = 2n/m`. To minimize this, maximize m, but m <= n. If m = n, score = 2 (uniform). If m = n-1, score = `2n/(n-1) ≈ 2 + 2/n`.

So simple rectangular shapes can't beat 2. We need a more complex shape.
