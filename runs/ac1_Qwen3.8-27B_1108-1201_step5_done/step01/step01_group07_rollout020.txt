        return np.ones(n) / n, 2.0 * n
    a /= a_sum

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for round_num in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            # Try different values for a[i]
            # The optimal a[i] should balance the peaks
            # Simple approach: try a few candidate values
            old_val = a[i]
            if old_val < 1e-15:
                continue  # Skip zero coordinates for now
            
            # Candidate values: current, slightly less, slightly more, zero
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 0.25]
            
            best_val = old_val
            best_local_obj = np.inf
            
            for val in candidates:
                a[i] = val
                new_sum = a.sum()
                if new_sum < 1e-12:
                    a[i] = old_val
                    continue
                a_normalized = a / new_sum
                new_conv = fast_convolve(a_normalized)
                new_obj = 2.0 * n * new_conv.max()
                if new_obj < best_local_obj:
                    best_local_obj = new_obj
                    best_val = val
            
            if best_val != old_val:
                a[i] = best_val
                a /= a.sum()
                conv = fast_convolve(a)
                local_obj = 2.0 * n * conv.max()
                if local_obj < best_obj - 1e-12:
                    best_obj = local_obj
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break

    return best_a, best_obj
