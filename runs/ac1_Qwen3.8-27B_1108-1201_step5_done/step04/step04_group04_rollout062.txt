"""
SLP + LP-FW hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_matrix(a):
    """Compute G where G[k,i] = db_k/da_i for b = conv(a,a)."""
    n = len(a)
    G = np.zeros((2*n-1, n))
    for i in range(n):
        G[i:i+n, i] = 2.0 * a
    for i in range(n):
        if 2*i < 2*n - 1:
            G[2*i, i] = a[i]
    return G


def slp_solve(a, max_time=2.0):
    """Solve one SLP subproblem. Returns (a_new, t_value)."""
    n = len(a)
    conv = fast_convolve(a)
    G = compute_grad_matrix(a)
    
    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = conv
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': max_time, 'dual_feasibility_tolerance': 1e-9})
    
    if res.success and res.x is not None:
        return np.maximum(res.x[:n], 0), float(res.x[n])
    return a.copy(), float(conv.max())


def fw_lp_solve(a, M, max_time=2.0):
    """Solve the FW LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        A_ub[j:j+n, j] = a
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': max_time})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def subgradient_step(a, lr, top_k=20, temp=1e-5):
    """One subgradient step focusing on top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Find top-k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Compute subgradient
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            # db_k/da_j = 2*a[k-j] (or a[j] if 2j=k)
            ki = k - j
            if 2*j == k:
                sg[j] += w[idx] * a[j]
            else:
                sg[j] += 2.0 * w[idx] * a[ki]
    
    a_new = project_simplex(a - lr * sg)
    return a_new, conv.max()

    n = len(a0)
    f = np.maximum(a0.astype(np.float64), 0.0)

            if no_improve > 30:

                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)

        # Damped step with backtracking

        best_val = conv.max()

            conv = fast_convolve(f)

"""
SLP (Successive Linear Programming) + LP-Frank-Wolfe + Subgradient hybrid optimizer
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: SLP directly minimizes a linearized surrogate of max(conv),
providing well-calibrated steps without line search in many iterations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_matrix(a):
    """Compute G where G[k,i] = 2*a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    G = np.zeros((2 * n - 1, n))
    two_a = 2.0 * a
    for i in range(n):
        G[i:i + n, i] = two_a
    return G


def slp_solve(a, max_time=2.0):
    """Solve one SLP subproblem: min t s.t. G@t' - t <= conv, sum(t')=1, t'>=0."""
    n = len(a)
    conv = fast_convolve(a)
    G = compute_grad_matrix(a)

    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = conv
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': max_time, 'dual_feasibility_tolerance': 1e-10})

    if res.success and res.x is not None:
        return np.maximum(res.x[:n], 0.0), float(res.x[n])
    return a.copy(), float(conv.max())


def fw_lp_solve(a, M, max_time=2.0):
    """Solve FW LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': max_time})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def subgradient_step(a, lr, top_k=20, temp=1e-5):
    """One subgradient step focusing on top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)

    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            ki = k - j
            sg[j] += 2.0 * w[idx] * a[ki]

    a_new = project_simplex(a - lr * sg)
    return a_new


def slp_optimize(a0, time_limit, verbose=True):
    """SLP optimizer with damping and periodic restarts."""
    n = len(a0)
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.5:
            break

        max_time = min(2.5, rem * 0.6)
        a_slp, t_val = slp_solve(f, max_time=max_time)

        if a_slp is None or np.max(np.abs(a_slp - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng((it * 7919 + n * 13) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                no_improve = 0
            continue

        # Try multiple damping levels
        best_alpha = 0.0
        best_val = conv.max()
        for a_test in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            ft = (1.0 - a_test) * f + a_test * a_slp
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = a_test

        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * a_slp
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            obj = conv.max()

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    SLP iter {it}: obj={best_obj:.10f} (t={elapsed:.1f}s)")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng((it * 4441 + 7) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj
