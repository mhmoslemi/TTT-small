    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)

        for j in range(js, je + 1):
            A_ub[i, j + 1] = a[pk - j]
    b_ub = np.zeros(k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-a[i], None) for i in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 5})
        if res.success:
            return res.x[1:]
    except:
        pass
    return None

def gradient_descent_optimize(f0, time_limit, temp_init=0.005, temp_final=0.0005,
                               lr_init=0.005, mom_init=0.9, use_lp=True, lp_every=50):
    """Main optimization loop."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)  # momentum
    best_f = f.copy()
    best_obj = fast_conv(f).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_temp = temp_init
    cur_lr = lr_init
    cur_mom = mom_init
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv, cur_max = compute_grad_and_conv(f, cur_temp)
        obj = cur_max
        
        # Momentum update
        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        
        # Check improvement
        new_conv = fast_conv(f_new)
        new_obj = new_conv.max()
        
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            f_new = project_simplex(f - 0.5 * cur_lr * grad)
            new_conv2 = fast_conv(f_new)
            if new_conv2.max() < obj:
                f = f_new
            else:
                f = f  # stay
                cur_lr *= 0.95
        
        f = f_new if new_obj <= obj else f
        
        # LP correction
        if use_lp and it % lp_every == 0 and no_improve > 10:
            d = solve_lp_correction(f, fast_conv(f), top_k=min(20, n//10))
            if d is not None:
                target = f + d
                target = np.maximum(target, 0)
                ts = target.sum()
                if ts > 1e-15:
                    target = target / ts
                    # Line search for LP direction
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        ft = (1-alpha)*f + alpha*target
                        ft = np.maximum(ft, 0)
                        fs = ft.sum()
                        if fs > 1e-15:
                            ft = ft / fs
                            fo = fast_conv(ft).max()
                            if fo < best_obj:
                                best_obj = fo
                                best_f = ft.copy()
                                f = ft.copy()
                                no_improve = 0
                                break
        
        # Temperature update
        progress = min(1.0, it / 5000)
        cur_temp = temp_init * (temp_final / temp_init) ** progress
        
        # Learning rate decay
        cur_lr *= 0.9997
        cur_mom = min(0.98, cur_mom + 0.0001)
        
        # Restart
        if no_improve > 200:
            scale = 0.001 + 0.003 * (no_improve / 500)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = lr_init * 0.5
            cur_temp = max(temp_final, cur_temp * 2)
            no_improve = 0
    
    return best_f, best_obj * 2 * n

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent with
adaptive temperature, momentum, periodic LP corrections, and multi-restart
strategy. Includes multi-scale exploration and length variation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_and_conv(a, temp):
    """Compute gradient of softmax-smoothed max of auto-convolution."""
    n = len(a)
    conv = fast_conv(a)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, conv, conv.max()


def solve_lp_correction(a, conv, top_k=20, time_limit_lp=3.0):
    """Solve LP to find direction reducing top-k peaks."""
    n = len(a)
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((k, n + 1))
    A_ub[:, 0] = -1.0
    for i, pk in enumerate(peaks):
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        idx = np.arange(js, je + 1)
        A_ub[i, idx + 1] = a[pk - idx]
    b_ub = np.zeros(k)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-a[i], None) for i in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit_lp})
        if res.success:
            return res.x[1:]
    except Exception:
        pass
    return None

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

            if verbose:
                print(f"    Restart at iter {it}, obj={best_obj:.10f}")

    return best_f, best_obj


def multi_scale_optimize(f0, time_limit):
    """Optimize at coarse scale, then interpolate back and refine."""
    n = len(f0)
    # Coarse: average groups of 4
    group_size = 4
    n_coarse = n // group_size
    if n_coarse < 32:
        return gd_optimize(f0, time_limit)

    coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        coarse[i] = f0[i * group_size:(i + 1) * group_size].mean()
    coarse = coarse / coarse.sum()

    t_coarse = time_limit * 0.3
    t_fine = time_limit * 0.7

    coarse_opt, _ = gd_optimize(coarse, time_limit=t_coarse,
                                temp_init=0.005, temp_final=0.001,
                                lr_init=0.01, mom_init=0.88, lp_every=30)

    # Interpolate back
    fine_init = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, n_coarse), coarse_opt)
    fine_init = np.maximum(fine_init, 0)
    fine_init = fine_init / fine_init.sum()

    return gd_optimize(fine_init, time_limit=t_fine,
                       temp_init=0.003, temp_final=0.0003,
                       lr_init=0.003, mom_init=0.92, lp_every=40)


def try_new_lengths(parent, best_seq, best_score, deadline, rng):
    """Try interpolating to different lengths and quick-optimizing."""
    results = []
    n_parent = len(parent)
    parent_norm = parent / parent.sum()

    lengths = []
    for dl in [32, -32, 64, -64, 96, 128, -96]:
        nt = n_parent + dl
        if 128 <= nt <= 2048:
            lengths.append(nt)

    for nt in lengths[:4]:
        if time.monotonic() > deadline - 60:
            break
        cand = np.interp(np.linspace(0, 1, nt),
                         np.linspace(0, 1, n_parent), parent_norm)
        cand = np.maximum(cand, 0)
        cs = cand.sum()
        if cs < 1e-10:
            continue
        cand = cand / cs
        # Quick optimization
        a_opt, _ = gd_optimize(cand, time_limit=30,
                               temp_init=0.004, temp_final=0.0008,
                               lr_init=0.004, mom_init=0.90, lp_every=40)
        score = 2.0 * nt * fast_conv(a_opt).max()
        if score < best_score:
            results.append((a_opt, score))
            print(f"  New length n={nt}: score={score:.8f}")
    return results


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Load parent, quick exploration
    2. Main gradient descent with multiple hyperparameter configs
    3. Multi-scale optimization
    4. Length exploration
    5. Final fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(384)
        try_seq(parent, "uniform")

    n_parent = len(parent)
    parent_norm = parent / parent.sum()

    # === Phase 1: Quick exploration ===
    print("\n=== Phase 1: Exploration ===")

    # Try smoothed versions
    try:
        from scipy.ndimage import gaussian_filter1d
        for sigma in [2, 5, 8]:
            sm = gaussian_filter1d(parent_norm, sigma, mode='wrap')
            sm = np.maximum(sm, 0)
            if sm.sum() > 1e-10:
                try_seq(sm / sm.sum(), f"smooth_{sigma}")
    except Exception:
        pass

    # Try amplitude modulation
    x = np.arange(n_parent, dtype=np.float64)
    for freq, amp in [(1, 0.05), (2, 0.03), (3, 0.02), (5, 0.02)]:
        mod = parent_norm * (1.0 + amp * np.cos(2 * np.pi * freq * x / n_parent))
        mod = np.maximum(mod, 0)
        if mod.sum() > 1e-10:
            try_seq(mod / mod.sum(), f"mod_f{freq}")

    # Try edge-enhanced
    edge = parent_norm * (1.0 + 0.1 * (np.abs(2.0 * x / n_parent - 1.0) ** 2))
    edge = np.maximum(edge, 0)
    if edge.sum() > 1e-10:
        try_seq(edge / edge.sum(), "edge")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Main gradient descent (multiple configs) ===
    print("\n=== Phase 2: Main GD ===")
    best_norm = best_seq / best_seq.sum()
    remaining = deadline - time.monotonic() - 300
    t_p2 = min(remaining * 0.55, 900)

    configs = [
        # (temp_init, temp_final, lr_init, mom_init, lp_every, lp_top_k)
        (0.005, 0.0005, 0.005, 0.90, 50, 20),
        (0.003, 0.0003, 0.003, 0.93, 100, 15),
        (0.008, 0.0008, 0.008, 0.87, 30, 25),
        (0.002, 0.0002, 0.002, 0.94, 75, 12),
    ]

    time_per = t_p2 / len(configs)
    for ci, (ti, tf, li, mi, le, lk) in enumerate(configs):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(time_per, deadline - time.monotonic() - 300)
        if t_alloc < 40:
            break
        print(f"  Config {ci}: temp=[{ti},{tf}], lr={li}, mom={mi}, "
              f"budget={t_alloc:.0f}s")
        a_opt, raw_obj = gd_optimize(best_norm, time_limit=t_alloc,
                                     temp_init=ti, temp_final=tf,
                                     lr_init=li, mom_init=mi,
                                     lp_every=le, lp_top_k=lk,
                                     verbose=True)
        try_seq(a_opt, f"gd_c{ci}")
        # Update best_norm for next config
        best_norm = best_seq / best_seq.sum()

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Multi-scale ===
    print("\n=== Phase 3: Multi-scale ===")
    remaining = deadline - time.monotonic() - 200
    t_p3 = min(remaining * 0.25, 300)
    if t_p3 > 60:
        print(f"  Multi-scale budget={t_p3:.0f}s")
        a_opt, _ = multi_scale_optimize(best_seq / best_seq.sum(), t_p3)
        try_seq(a_opt, "multiscale")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Length exploration ===
    print("\n=== Phase 4: Length Exploration ===")
    remaining = deadline - time.monotonic() - 150
    if remaining > 120:
        results = try_new_lengths(parent, best_seq, best_score,
                                  deadline - 60, rng)
        for a_r, s_r in results:
            try_seq(a_r, "newlen")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Fine-tuning ===
    print("\n=== Phase 5: Fine-tuning ===")
    remaining = deadline - time.monotonic() - 50
    t_p5 = min(remaining * 0.6, 200)
    if t_p5 > 30:
        best_norm = best_seq / best_seq.sum()
        # Very fine GD
        a_opt, _ = gd_optimize(best_norm, time_limit=t_p5 * 0.6,
                               temp_init=0.001, temp_final=0.0001,
                               lr_init=0.001, mom_init=0.95,
                               lp_every=20, lp_top_k=10)
        try_seq(a_opt, "gd_fine")

        # Local search
        rem2 = deadline - time.monotonic() - 30
        if rem2 > 20:
            n_b = len(best_seq)
            bn = best_seq / best_seq.sum()
            scale = 0.002
            t0 = time.monotonic()
            while time.monotonic() - t0 < rem2 * 0.5:
                pert = bn + rng.standard_normal(n_b) * scale
                pert = project_simplex(pert)
                try_seq(pert * best_seq.sum(), "local")
                scale *= 0.97
                if scale < 0.0003:
                    scale = 0.001

    # Final check
    if best_seq is not None:
        final_s = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_s):
            best_score = final_s
        print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.10f}")

    return [float(x) for x in best_seq]
