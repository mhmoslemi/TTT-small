import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def fw_iteration(f, M, build_matrix=True):
    """One FW iteration: solve LP and do line search."""
    n = len(f)
    A = build_lp_matrix(f) if build_matrix else None
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return f, M, None
    g = np.maximum(res.x, 0)
    s = g.sum()
    if s < 1e-15:
        return f, M, None
    g = g / s
    
    # Line search
    best_alpha = 0
    best_val = M
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 0:
        f_new = (1 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        return f_new, best_val, g
    return f, M, g

def optimize_fw(f0, time_limit, m_relax=1.01):
    f = f0.copy() / f0.sum()
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    relax = m_relax
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f_new, M_new, g = fw_iteration(f, M * relax)
        if g is None:
            relax = min(relax * 1.05, 3.0)
            if relax > 3:
                break
            continue
        relax = m_relax
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
        f = f_new
        M = M_new
        if it % 100 == 0:
            print(f"  FW it={it}, M={M:.10f}, best={best_M:.10f}, t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_M

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M, A_mat):
    """Solve LP: max sum(g) s.t. A(a)@g <= M, g >= 0.
    Returns (g, dual_y) or (None, None) if infeasible."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            g = np.maximum(res.x, 0.0)
            dual_y = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return g, dual_y
    except Exception:
        pass
    return None, None


def quadratic_line_search(f, g, f_conv, M):
    """Quadratic line search: find alpha in [0,1] minimizing max(conv of (1-a)f+ag)."""
    n = len(f)
    best_alpha = 0.0
    best_val = M

    # Try a grid first
    alphas = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8, 1.0]
    for alpha in alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha <= 0:
        return 0.0, M

    # Refine with bisection
    lo = max(0.0, best_alpha - 0.08)
    hi = min(1.0, best_alpha + 0.08)
    for _ in range(8):
        mid = (lo + hi) / 2.0
        ft = (1.0 - mid) * f + mid * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            break
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid
            lo = mid
        else:
            hi = mid

    return best_alpha, best_val

def lse_gradient_descent(f0, time_limit, beta_start=10.0, beta_end=5000.0):
    """Log-sum-exp smoothed gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient: 2 * cross_correlation(w, f) of length n
        n_conv = len(f)
        f_flipped = f[::-1].copy()
        total_len = (2 * n_conv - 1) + n_conv - 1
        fft_size = 1 << (total_len - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f_flipped, fft_size)
        conv_wf = np.fft.irfft(W * F, fft_size)
        grad = 2.0 * conv_wf[n_conv - 1:n_conv - 1 + n_conv]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.93
                momentum = max(0.86, momentum - 0.004)
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.4, beta_end)
                momentum = 0.90
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj

    print(f"\n=== Phase 4: Multi-n Exploration ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    candidate_ns = [384, 420, 448, 480, 512, 544, 576]
    if n_parent in candidate_ns:
        candidate_ns.remove(n_parent)

    # Sort by distance from parent n (prefer nearby)
    candidate_ns.sort(key=lambda x: abs(x - n_parent))

    trials_done = 0
    max_trials = 4
    for n_alt in candidate_ns[:max_trials]:
        t_remaining = deadline - time.monotonic()
        if t_remaining < 80:
            break
        t_alloc = min(120, (t_remaining - 60) / (max_trials - trials_done))
        if t_alloc < 30:
            break

        trials_done += 1
        print(f"  Trying n={n_alloc if False else n_alt}, budget={t_alloc:.0f}s")

        # Interpolate or resize
        if current_best is not None:
            x_old = np.linspace(0, 1, len(current_best))
            x_new = np.linspace(0, 1, n_alt)
            f_interp = np.interp(x_new, x_old, current_best)
        else:
            f_interp = np.ones(n_alt) / n_alt
        f_interp = np.maximum(f_interp, 0.0)
        f_interp = f_interp / f_interp.sum()

        # FW-LP
        t1 = t_alloc * 0.6
        t2 = t_alloc * 0.4
        a_opt, obj = fw_lp_optimize(f_interp, time_limit=t1, init_relax=1.003)
        try_sequence(a_opt, best_tracker)

        # Brief LSE
        current_alt = best_tracker[1] if (best_tracker[1] is not None and len(best_tracker[1]) == n_alt) else a_opt
        if t2 > 15:
            a_opt2, obj2 = lse_gradient_descent(current_alt, time_limit=t2,
                                                beta_start=20.0, beta_end=2000.0)
            try_sequence(a_opt2, best_tracker)

    # === Phase 5: Final refinement on best found ===
    print(f"\n=== Phase 5: Final Refinement ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    t_remaining = deadline - time.monotonic()
    if t_remaining > 60 and current_best is not None:
        # Brief FW-LP
        t_alloc = min(150, t_remaining * 0.4)
        if t_alloc > 20:
            print(f"  Final FW-LP: {t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc, init_relax=1.002)
            try_sequence(a_opt, best_tracker)

    # Sparse perturbation search
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()
    t_alloc = min(80, t_remaining * 0.3)
    if t_alloc > 10 and current_best is not None:
        print(f"  Sparse perturbation: {t_alloc:.0f}s")
        a_opt, obj = sparse_perturbation_search(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # Final peak equalization
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()
    t_alloc = min(60, t_remaining * 0.3)
    if t_alloc > 10 and current_best is not None:
        print(f"  Final peak equalize: {t_alloc:.0f}s")
        a_opt, obj = peak_equalize(current_best, time_limit=t_alloc, n_peaks=4)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
    else:
        final_seq = np.ones(448) / 448

    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"\nFINAL: score={final_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
