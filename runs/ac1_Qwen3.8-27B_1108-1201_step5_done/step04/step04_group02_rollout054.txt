"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with Adam optimizer
2. Efficient FFT-based gradient computation
3. Temperature annealing
4. Multi-length exploration
5. Peak-chasing subgradient refinement
6. Careful time management

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    k = np.nonzero(cond)[0][-1]
    theta = (cssv[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, conv, T, fft_size):
    """Compute the gradient direction of the log-sum-exp smoothed objective.
    Returns grad = 2 * cross_correlation(softmax(conv/T), a)"""
    n = len(a)
    c = conv / T
    c -= c.max()
    exp_c = np.exp(c)
    w = exp_c / exp_c.sum()
    
    # Cross-correlation: R[m] = sum_j a[j] * w[j+m]
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    R = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * R[:n]

def smoothed_descent(a0, time_limit, T_init=0.5, T_min=0.001, lr=0.01, 
                     beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimizer on the log-sum-exp smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()  # for gradient computation
    # Actually we need fft_size >= n + (2n-1) - 1 = 3n-2
    # Let's use 4n rounded to power of 2
    fft_size = 1 << (4 * n - 1).bit_length()
    
    m = np.zeros(n)  # Adam momentum (first moment)
    v = np.zeros(n)  # Adam velocity (second moment)
    t = 0
    T = T_init
    T_decay = (T_min / T_init) ** (1.0 / max(1, int(time_limit * 10)))  # decay per step
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break
        
        # Compute gradient
        grad = compute_gradient(a, conv, T, fft_size)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        
        # Temperature decay
        T = max(T * T_decay, T_min)
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.95
            no_improve = 0
        
        if verbose and t % 1000 == 0:
            print(f"    step {t}, T={T:.4f}, lr={lr:.6f}, obj={best_obj:.8f}")
    
    return best_a, best_obj

def peak_chase(a0, time_limit, top_k=4):
    """Refine using exact subgradient focused on top peaks."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    lr = 0.005
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break
        
        # Find top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            w = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Weight by how much above average
            avg = conv.mean()
            w = np.maximum(pv - avg, 0.0)
            wsum = w.sum()
            if wsum < 1e-30:
                w = np.ones(len(peaks)) / len(peaks)
            else:
                w = w / wsum
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        # Gradient step
        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
                no_improve = 0
                if lr < 1e-6:
                    break
        
        a = a_new
        conv = conv_new
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with Adam optimizer (O(n log n) per step via FFT)
2. Temperature annealing for smooth-to-sharp transition
3. Multi-length, multi-initialization exploration
4. Peak-chasing subgradient refinement
5. Adaptive time allocation
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    k = np.nonzero(cond)[0][-1]
    theta = (cssv[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, conv, T, grad_fft_size):
    """
    Compute gradient direction of T*log(sum_k exp(conv[k]/T)) w.r.t. a.
    Result: 2 * cross_correlation(softmax(conv/T), a)[:n]
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    c = conv / T
    c -= c.max()
    exp_c = np.exp(c)
    w = exp_c / exp_c.sum()

    A = np.fft.rfft(a, grad_fft_size)
    W = np.fft.rfft(w, grad_fft_size)
    R = np.fft.irfft(A * np.conj(W), grad_fft_size)
    return 2.0 * R[:n]


def smoothed_adam(a0, time_limit, T_init=0.3, T_min=0.001, lr=0.015,
                  beta1=0.9, beta2=0.999, eps=1e-8, symmetrize=False):
    """
    Adam optimizer on the log-sum-exp smoothed objective.
    Works on the simplex (a >= 0, sum(a) = 1).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    grad_fft_size = 1 << (4 * n - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    lr_cur = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.3:
            break

        grad = compute_smooth_grad(a, conv, T, grad_fft_size)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        if symmetrize:
            a_new = 0.5 * (a_new + a_new[::-1])
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        # Temperature decay (geometric)
        T = max(T * 0.998, T_min)

        # Adaptive LR
        if no_improve > 200:
            lr_cur *= 0.9
            no_improve = 0
            if lr_cur < 1e-6:
                lr_cur = lr * 0.01
                m = np.zeros(n)
                v = np.zeros(n)
                T = max(T_init * 0.1, T_min * 10)

    return best_a, best_obj


def peak_chase(a0, time_limit, top_k=3, lr_init=0.003):
    """
    Refine using exact subgradient focused on top peaks.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break

        # Find top peaks and compute weighted subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv[peaks]
            avg = conv.mean()
            w = np.maximum(pv - avg, 1e-10)
            w = w / w.sum()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            pv_max = pv.max()
            pv_min = pv.min()
            if pv_max - pv_min < 1e-12:
                w = np.ones(len(peaks)) / len(peaks)
            else:
                w = (pv - pv_min + 1e-10) / (pv_max - pv_min)
                w = w / w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Step
        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.03)
        else:
            no_improve += 1
            if no_improve > 80:
                lr *= 0.85
                no_improve = 0
                if lr < 1e-7:
                    # Random perturbation escape
                    rng = np.random.default_rng(it * 7 + 3)
                    a = project_simplex(a + rng.standard_normal(n) * 0.002)
                    conv = fast_convolve(a)
                    lr = lr_init * 0.3

        a = a_new
        conv = conv_new

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    # U-shaped (high at edges)
    u_shape = 1.0 + 2.0 * np.cos(np.pi * x / (n - 1 + 1e-10))
    u_shape = np.maximum(u_shape, 0.01)
    inits.append(u_shape / u_shape.sum())

    # Cosine window
    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 2 + 0.05
    inits.append(c / c.sum())

    # Exponential decay from left
    e1 = np.exp(-x / max(n / 5.0, 1.0))
    inits.append(e1 / e1.sum())

    # Exponential decay from right
    e2 = np.exp(-x[::-1] / max(n / 5.0, 1.0))
    inits.append(e2 / e2.sum())

    # Double exponential (high at both edges)
    de = np.exp(-x / max(n / 6.0, 1.0)) + np.exp(-x[::-1] / max(n / 6.0, 1.0))
    inits.append(de / de.sum())

    # Random Dirichlet (various concentrations)
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.full(n, alpha)))

    # Gaussian-like (high in middle)
    center = n / 2.0
    g = np.exp(-0.5 * ((x - center) / max(n / 6.0, 1.0)) ** 2)
    inits.append(g / g.sum())

    # Parent-based interpolation
    if parent is not None and len(parent) != n:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
