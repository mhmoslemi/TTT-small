"""
Hybrid optimizer for minimizing 2*n*max(auto_convolve(a))/(sum(a))^2.
Uses: (1) LP-based Frank-Wolfe with adaptive line search,
(2) Smoothed gradient descent (log-sum-exp) with analytical gradients,
(3) Coordinate descent via pair-swap with 1D minimax,
(4) Multi-length portfolio search with adaptive time allocation,
(5) Nesterov-accelerated smoothing and structured restarts.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def smoothed_gradient_fft(conv, a, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def coord_descent_pair_swap(f, time_limit, n_swaps_per_iter=50, verbose=False):
    """Coordinate descent using pair-swap moves with 1D minimax."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    # Precompute sensitivity: for pair (i,j), d_k = a[k-i] - a[k-j]
    # We'll sample random pairs and solve the 1D minimax.

    while time.monotonic() - t0 < time_limit:
        for _ in range(n_swaps_per_iter):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            # Pick random pair
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue

            # d_k = a[k-i] - a[k-j] for valid k
            # new_c_k = c_k + 2*delta*(a[k-i] - a[k-j])
            # We want to minimize max_k new_c_k
            # subject to: f[i]+delta >= 0, f[j]-delta >= 0
            # i.e., -f[i] <= delta <= f[j]

            delta_lo = -f[i]
            delta_hi = f[j]
            if delta_lo >= delta_hi:
                continue

            # Compute the effect on each convolution coefficient
            # For peak k: new_val = conv[k] + 2*delta*(a[k-i] - a[k-j])
            # We only need to check peaks that could become the max
            # Use top-K peaks for efficiency
            top_k = min(30, 2 * n - 1)
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peak_idx]

            # For each peak, compute slope d_k = a[k-i] - a[k-j] (with boundary handling)
            slopes = np.zeros(len(peak_idx))
            for idx in range(len(peak_idx)):
                k = int(peak_idx[idx])
                # a[k-i] valid if 0 <= k-i < n
                val_1 = f[k - i] if 0 <= k - i < n else 0.0
                # a[k-j] valid if 0 <= k-j < n
                val_2 = f[k - j] if 0 <= k - j < n else 0.0
                slopes[idx] = 2.0 * (val_1 - val_2)

            # Solve: min_delta max_m (peak_vals[m] + delta * slopes[m])
            # subject to delta_lo <= delta <= delta_hi
            # This is a 1D convex optimization (min of max of linear functions)

            # Binary search for optimal delta
            lo = delta_lo
            hi = delta_hi
            for _ in range(20):
                mid = (lo + hi) / 2.0
                vals = peak_vals + mid * slopes
                max_above = np.max(vals[vals > peak_vals.max()]) if np.any(vals > peak_vals.max()) else peak_vals.max()
                # Check if slopes push up or down
                pos_slopes = slopes > 1e-12
                neg_slopes = slopes < -1e-12
                if np.any(pos_slopes) and np.any(neg_slopes):
                    # The max at mid: if it's at a pos-slope element, decrease; if at neg-slope, increase
                    max_at_pos = np.max(peak_vals[pos_slopes] + mid * slopes[pos_slopes]) if np.any(pos_slopes) else -np.inf
                    max_at_neg = np.max(peak_vals[neg_slopes] + mid * slopes[neg_slopes]) if np.any(neg_slopes) else -np.inf
                    if max_at_pos > max_at_neg + 1e-15:
                        hi = mid
                    else:
                        lo = mid
                elif np.all(slopes >= 0):
                    lo = mid  # Everything increases with delta, so go left
                elif np.all(slopes <= 0):
                    hi = mid  # Everything decreases with delta, so go right
                else:
                    if max_at_pos > max_at_neg + 1e-15:
                        hi = mid
                    else:
                        lo = mid

            delta = (lo + hi) / 2.0
            delta = max(delta_lo, min(delta_hi, delta))

            if abs(delta) < 1e-12:
                continue

            f[i] += delta
            f[j] -= delta
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = np.ones(n) / n
            else:
                f = f / fs

            conv = fast_convolve(f)
            obj = conv.max()

            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                # Revert if no improvement
                f[i] -= delta
                f[j] += delta
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                conv = fast_convolve(f)

            if no_improve > 200:
                # Restart from best with perturbation
                scale = 0.001 * min(no_improve / 200.0, 3.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                best_obj = conv.max()
                no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    CD iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_dir(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue

        # Line search (coarse + bisection)
        best_alpha = 0.0
        best_val = M
        for alpha in [0.03, 0.07, 0.15, 0.25, 0.4, 0.55, 0.7, 0.85, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.06)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if no_improve > 25:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.003 + 0.003 * min(no_improve // 25, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def smoothed_grad_descent(f0, time_limit, T_init=0.003, T_min=0.00005, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

def sg_polish(f0, time_limit, lr=0.001, top_k_frac=0.05, temp=0.00003):
    """Subgradient polish with weighted top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(5, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 60 == 0:
            lr *= 0.995
            temp *= 1.01

        if no_improve > 150:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0006)
            lr = 0.0006 * (1.0 + 0.5 * min(no_improve // 150, 4))
            no_improve = 0

    return best_f, best_obj


def resample_to_length(a, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated (U-shape) - often good for low autocorrelation
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.7
    inits.append(edge_w / edge_w.sum())

    # Stronger edge
    edge_w2 = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 1.5
    inits.append(edge_w2 / edge_w2.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Cosine trough
    c = np.abs(np.sin(np.pi * x / n)) ** 0.5 + 0.02
    inits.append(c / c.sum())

    # Random mirror-symmetric
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.4)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] = half[half_n - 1] * 0.5
    inits.append(sym / sym.sum())

    # Random Dirichlet (smooth-ish)
    alpha = np.ones(n) * (1.0 + 0.5 * np.cos(2 * np.pi * x / n))
    alpha = np.maximum(alpha, 0.1)
    inits.append(rng.dirichlet(alpha))

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        px2 = np.maximum(px + rng.standard_normal(n) * 0.01, 0)
        ps2 = px2.sum()
        if ps2 > 1e-10:
            inits.append(px2 / ps2)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase portfolio search:
    Phase 1: Quick exploration across multiple lengths and inits.
    Phase 2: Deep optimization on top candidates (LP-FW + Smoothed GD + Coord Descent).
    Phase 3: Fine polishing.
    Phase 4: Final verification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score~{evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    # Determine base length
    base_n = len(parent) if parent is not None else 384

    # === Phase 1: Broad Exploration ===
    print("=== Phase 1: Broad Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 250.0

    # Try parent with LP-FW
    if parent is not None:
        t_alloc = min(80.0, p1_budget - 30.0)
        if t_alloc > 20:
            print(f"  LP-FW on parent (n={base_n}), t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
            try_seq(a_opt, "parent_fw")

    # Try different lengths
    lengths = sorted(set([
        base_n,
        max(128, base_n - 32),
        max(128, base_n - 16),
        base_n + 16,
        base_n + 32,
        base_n + 64,
        max(128, base_n - 64),
        base_n + 96,
        256,
        512,
    ]))
    lengths = [l for l in lengths if 128 <= l <= 800]
    # Remove duplicates while preserving order
    seen = set()
    unique_lengths = []
    for l in lengths:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)
    lengths = unique_lengths

    # Shuffle non-base lengths
    others = [l for l in lengths if l != base_n]
    rng.shuffle(others)
    try_order = [base_n] + others[:6]  # Try base + 6 others

    candidates = []  # (sequence, length, score)

    for n_try in try_order:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 25:
            break

        t_alloc = min(35.0, rem - 10.0)
        if t_alloc < 15:
            break

        inits = generate_inits(n_try, parent, rng)
        # Quick evaluate top inits
        best_init = None
        best_init_obj = float('inf')
        for f_init in inits:
            conv = fast_convolve(f_init)
            obj = conv.max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = f_init

        if best_init is None:
            continue

        print(f"  Exploring n={n_try} (init_obj={best_init_obj:.8f}), t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_init, time_limit=t_alloc)
        candidates.append((a_opt, n_try))
        try_seq(a_opt, f"explore_n{n_try}")
        # Also try smoothed gradient on the same init
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem > 20:
            T_init = max(0.001, 0.004 * fast_convolve(a_opt).max())
            a_opt2, _ = smoothed_grad_descent(a_opt, time_limit=min(15.0, rem - 5.0),
                                              T_init=T_init, T_min=0.0001)
            try_seq(a_opt2, f"explore_sgd_n{n_try}")
            if a_opt2.sum() > 1e-10:
                candidates.append((a_opt2 / a_opt2.sum(), n_try))

    print(f"  Phase 1 complete: best={best_score:.6f}, {len(candidates)} candidates")

    # === Phase 2: Deep Optimization ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 1100.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)

        # Cycle through methods
        methods_cycle = ['fw', 'sgd', 'cd', 'fw', 'sgd', 'cd', 'fw', 'sgd']
        cycle_i = 0
        no_progress = 0

        while time.monotonic() - t_p2 < p2_budget:
            rem = p2_budget - (time.monotonic() - t_p2)
            if rem < 15:
                break

            method = methods_cycle[cycle_i % len(methods_cycle)]
            cycle_i += 1

            t_alloc = min(60.0, rem / 2.0)
            if t_alloc < 10:
                break

            prev_best = best_score

            if method == 'fw':
                print(f"  [2.{cycle_i}] FW  (n={n_work}, t={t_alloc:.0f}s, best={best_score:.6f})")
                a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
                try_seq(a_opt, f"p2_fw_{cycle_i}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
                    n_work = len(f_work)

            elif method == 'sgd':
                T_start = max(0.0005, 0.003 * fast_convolve(f_work).max())
                print(f"  [2.{cycle_i}] SGD (n={n_work}, t={t_alloc:.0f}s, T={T_start:.5f})")
                a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_alloc,
                                                 T_init=T_start, T_min=0.00003)
                try_seq(a_opt, f"p2_sgd_{cycle_i}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
                    n_work = len(f_work)

            elif method == 'cd':
                print(f"  [2.{cycle_i}] CD  (n={n_work}, t={t_alloc:.0f}s)")
                a_opt, _ = coord_descent_pair_swap(f_work, time_limit=t_alloc)
                try_seq(a_opt, f"p2_cd_{cycle_i}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
                    n_work = len(f_work)

            # Check progress
            if best_score < prev_best - 1e-8:
                no_progress = 0
            else:
                no_progress += 1

            # If stuck, try a different length or restart
            if no_progress >= 3 and rem > 60:
                no_progress = 0
                # Try resampling to a slightly different length
                delta_n = rng.choice([-16, 16, -32, 32])
                new_n = n_work + delta_n
                if 128 <= new_n <= 800:
                    print(f"  [2.{cycle_i}] Resample to n={new_n}")
                    f_resampled = resample_to_length(f_work, new_n)
                    t_resample = min(40.0, rem / 2.0)
                    a_opt, _ = fw_lp_optimize(f_resampled, time_limit=t_resample)
                    try_seq(a_opt, f"p2_resample_{new_n}")
                    if best_seq is not None:
                        f_work = best_seq / best_seq.sum()
                        n_work = len(f_work)
                else:
                    # Perturb and restart
                    f_pert = project_simplex(f_work + rng.standard_normal(n_work) * 0.005)
                    f_work = f_pert

    print(f"  Phase 2 complete: best={best_score:.6f}")

    # === Phase 3: Fine Polish ===
    print(f"\n=== Phase 3: Fine Polish ===")
    t_p3 = time.monotonic()
    p3_budget = 350.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)

        # Subgradient polish rounds
        sg_configs = [
            {"lr": 0.0008, "top_k_frac": 0.04, "temp": 0.00002},
            {"lr": 0.0004, "top_k_frac": 0.03, "temp": 0.00001},
            {"lr": 0.0002, "top_k_frac": 0.02, "temp": 0.000005},
        ]
        for i, cfg in enumerate(sg_configs):
            rem = p3_budget - (time.monotonic() - t_p3)
            if rem < 20:
                break
            t_alloc = min(60.0, rem - 10.0)
            print(f"  SG polish {i} (t={t_alloc:.0f}s)")
            a_opt, _ = sg_polish(f_work, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"p3_sg{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
                n_work = len(f_work)

        # Fine smoothed gradient
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            T_init = max(0.0002, 0.001 * fast_convolve(f_work).max())
            t_alloc = min(70.0, rem - 10.0)
            print(f"  Fine SGD (t={t_alloc:.0f}s, T={T_init:.6f})")
            a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_alloc,
                                             T_init=T_init, T_min=0.00001)
            try_seq(a_opt, "p3_fine_sgd")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
                n_work = len(f_work)

        # Coordinate descent polish
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 25:
            t_alloc = min(50.0, rem - 10.0)
            print(f"  CD polish (t={t_alloc:.0f}s)")
            a_opt, _ = coord_descent_pair_swap(f_work, time_limit=t_alloc,
                                               n_swaps_per_iter=80)
            try_seq(a_opt, "p3_cd")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
                n_work = len(f_work)

        # Final LP-FW
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 25:
            t_alloc = min(50.0, rem - 5.0)
            print(f"  Final FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_final_fw")

    print(f"  Phase 3 complete: best={best_score:.6f}")

    # === Phase 4: Final Tricks ===
    print(f"\n=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)

        # Tiny perturbations + quick polish
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            noise = rng.standard_normal(n_work) * (0.0003 + 0.0002 * trial)
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = sg_polish(f_pert, time_limit=5, lr=0.0002,
                                  top_k_frac=0.02, temp=0.000005)
            try_seq(a_opt, f"p4_pert{trial}")

        # Quick CD on best
        if time.monotonic() < deadline - 15:
            a_opt, _ = coord_descent_pair_swap(f_work, time_limit=10,
                                               n_swaps_per_iter=100)
            try_seq(a_opt, "p4_cd")

        # One final FW
        if time.monotonic() < deadline - 10:
            a_opt, _ = fw_lp_optimize(f_work, time_limit=8)
            try_seq(a_opt, "p4_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  FALLBACK: {best_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    elif not np.isfinite(final_score):
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
