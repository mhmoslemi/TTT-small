```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (multiplicative weights) with exact subgradient
2. LP-based direction finding with Newton step sizing
3. Fourier-domain peak flattening
4. Simulated annealing for exploration
5. Multi-start with diverse initializations
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, conv, top_k=50, temp=0.001):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    # Use top-k peaks for efficiency
    if top_k < len(conv):
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]
    else:
        peak_idx = np.arange(len(conv))
        peak_vals = conv
    
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # All weights underflow, use hard max
        w = np.zeros_like(peak_vals)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for idx in range(len(peak_idx)):
        if w[idx] < 1e-15:
            continue
        k = int(peak_idx[idx])
        # c[k] = sum_j a[j] * a[k-j], dc[k]/da[j] = a[k-j] if 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        idxs = np.arange(j_min, j_max + 1)
        contrib = a[k - idxs]  # a[k - j] for j in range
        grad[idxs] += w[idx] * contrib
    
    return grad


def mirror_descent(a0, time_limit, lr_init=0.01, temp=0.001, top_k=50,
                   momentum=0.9, patience=300):
    """Mirror descent with multiplicative weights and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)

        grad = compute_smooth_grad(a, conv, top_k=top_k, temp=temp)

        v = momentum * v - lr * grad
        a_new = a * np.exp(v)
        s_new = a_new.sum()

            v = v * 0.5

            conv = conv_new

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return None, 0.0, False


def lp_optimize(a0, time_limit, verbose=False):

def fourier_optimize(a0, time_limit, n_iter=1000):
    """
    Fourier-domain optimization: flatten |FFT(a)|^2 to reduce autocorrelation peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Zero-pad and FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        mag[mag < 1e-30] = 1e-30
        
        # Flatten: reduce peaks in |A|
        # Target: uniform magnitude (like a Zadoff-Chu sequence)
        target_mag = np.sqrt(fft_size / 2)  # roughly uniform
        
        # Blend current and target
        blend = 0.01 + 0.02 * min(it / n_iter, 1.0)  # adaptive blend
        new_mag = (1 - blend) * mag + blend * target_mag
        
        # But preserve the DC component (related to sum)
        # Actually, let's just reduce the dynamic range
        mag_min = np.percentile(mag, 5)
        mag_max = np.percentile(mag, 99.5)
        if mag_max > mag_min:
            # Compress dynamic range
            compressed = mag_min + (mag_max - mag_min) * ((mag - mag_min) / (mag_max - mag_min)) ** 0.7
            # Blend
            new_mag2 = (1 - 0.1) * mag + 0.1 * compressed
            # Reconstruct with original phase
            phase = np.angle(A)
            A_new = new_mag2 * np.exp(1j * phase)
        else:
            A_new = A
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller blend
            pass
    
    return best_a, best_obj


def simulated_annealing(a0, time_limit, T_init=0.01, T_min=0.0001, cooling=0.999):
    """Simulated annealing with targeted peak reduction."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T_init
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit and T > T_min:
        # Propose: pick two positions and transfer mass
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            T *= cooling
            continue
        
        delta = rng.uniform(-0.02, 0.02)
        a_new = a.copy()
        a_new[i] = max(0.0, a[i] + delta)
        a_new[j] = max(0.0, a[j] - delta)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        delta_obj = obj_new - cur_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            a = a_new
            conv = conv_new
            cur_obj = obj_new
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        
        T *= cooling
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Various structured inits
    for freq in [1, 2, 3, 4, 5]:
        name = f'sine{freq}'
        arr = (np.sin(2 * np.pi * freq * x / n) ** 2 + 0.1)
        inits[name] = arr / arr.sum()
    
    # Periodic with period ~n/4
    p = max(2, n // 4)
    per = np.zeros(n)
    per[::p] = 1.0
    inits['periodic4'] = per / per.sum()
    
    p = max(2, n // 3)
    per = np.zeros(n)
    per[::p] = 1.0
    inits['periodic3'] = per / per.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p_arr = np.maximum(parent, 0).astype(np.float64)
        ps = p_arr.sum()
        if ps > 1e-10 and np.isfinite(p_arr).all():
            if len(p_arr) == n:
                inits['parent'] = p_arr / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p_arr)), p_arr / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Mirror descent (multiplicative weights) with subgradient
2. LP-based direction finding with linear line search
3. Block coordinate descent for local refinement
4. Peak-targeted coordinate updates
5. Multi-start with diverse initializations and hierarchical n-search

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, conv, top_k=50, temp=0.001):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    if top_k < len(conv):
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]
    else:
        peak_idx = np.arange(len(conv))
        peak_vals = conv.copy()

    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(peak_vals)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for idx in range(len(peak_idx)):
        if w[idx] < 1e-15:
            continue
        k = int(peak_idx[idx])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        idxs = np.arange(j_min, j_max + 1)
        contrib = a[k - idxs]
        grad[idxs] += w[idx] * contrib
    return grad


def mirror_descent(a0, time_limit, lr_init=0.008, temp=0.001, top_k=40,
                   momentum=0.88, patience=250):
    """Mirror descent with multiplicative weights and backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_peak = conv.max()
        grad = compute_smooth_grad(a, conv, top_k=top_k, temp=temp)

        v = momentum * v - lr * grad
        # Clamp to prevent overflow
        v = np.clip(v, -5.0, 5.0)
        a_new = a * np.exp(v)
        s_new = a_new.sum()
        if s_new < 1e-300 or not np.isfinite(s_new):
            v *= 0.5
            continue
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < 2.0 * n * cur_peak - 1e-12:
            a = a_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
            no_improve = 0
        else:
            for _ in range(8):
                v *= 0.5
                a_new = a * np.exp(v)
                s_new = a_new.sum()
                if s_new < 1e-300 or not np.isfinite(s_new):
                    break
                a_new = a_new / s_new
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < 2.0 * n * cur_peak - 1e-12:
                    a = a_new
                    no_improve = 0
                    break
            else:
                no_improve += 1
                lr = max(lr * 1.02, 0.0003)

        if no_improve >= patience:
            break
        temp *= 1.001

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return None, 0.0, False


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based optimization with linear line search and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    max_patience = 80
    perturb_count = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        g, g_sum, success = lp_direction(a, M * 1.0005)

        if not success or g_sum <= 1.0 + 1e-6:
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-14:
                no_improve += 1
            else:
                # Linear line search with 12 candidates
                cur_peak = conv.max()
                best_t = 0.0
                best_peak = cur_peak

                for t in np.concatenate([np.linspace(0.02, 1.0, 10), [0.5, 0.3]]):
                    a_try = np.maximum(a + t * d, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    peak_try = conv_try.max()
                    if peak_try < best_peak - 1e-10:
                        best_peak = peak_try
                        best_t = t
                        a = a_try
                        conv = conv_try
                        M = conv_try.max()
                        if 2.0 * n * M < best_obj:
                            best_obj = 2.0 * n * M
                            best_a = a.copy()

                if best_t > 0:
                    no_improve = 0
                else:
                    no_improve += 1

        if verbose and no_improve % 50 == 0:
            print(f"      LP iter: obj={best_obj:.8f}, no_imp={no_improve}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_patience:
            scale = 0.002 + 0.003 * min(perturb_count, 12)
            a = np.maximum(best_a + rng.standard_normal(n) * scale, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_patience = min(250, 80 + 25 * perturb_count)
            if perturb_count > 50:
                break

    return best_a, best_obj


def block_descent(a0, time_limit, block_size=64):
    """Block coordinate descent: optimize sub-vectors while keeping rest fixed."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    max_patience = 50
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Pick a random block
        bs = min(block_size, n)
        start = rng.integers(0, max(1, n - bs))
        end = start + bs

        sub = a[start:end].copy()
        rest_sum = s - sub.sum()
        c_other = conv.copy()

        improved_block = False
        for _ in range(30):
            # Gradient of peak wrt sub
            peak_k = int(np.argmax(c_other))
            j_min = max(start, peak_k - (n - 1))
            j_max = min(end - 1, peak_k)
            if j_min > j_max:
                break
            idxs = np.arange(j_min, j_max + 1)
            grad_sub = np.zeros(bs)
            grad_sub[idxs - start] = a[peak_k - idxs]

            # Mirror step on sub
            lr = 0.01
            sub_new = sub * np.exp(-lr * grad_sub)
            sub_new = np.maximum(sub_new, 0.0)
            sub_new_sum = sub_new.sum()
            if sub_new_sum < 1e-15:
                break
            a_try = a.copy()
            a_try[start:end] = sub_new
            s_try = a_try.sum()
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            if conv_try.max() < conv.max() - 1e-10:
                a = a_try
                conv = conv_try
                sub = a[start:end].copy()
                improved_block = True
                if 2.0 * n * conv.max() < best_obj:
                    best_obj = 2.0 * n * conv.max()
                    best_a = a.copy()
                no_improve = 0
                break
            else:
                lr *= 0.5
                if lr < 0.0001:
                    break

        if not improved_block:
            no_improve += 1

        if no_improve >= max_patience:
            break

    return best_a, best_obj


def peak_targeted_update(a0, time_limit, rng=None):
    """Directly target the peak lag by reducing contributing products."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    max_patience = 200

    while time.monotonic() - t0 < time_limit:
        peak_k = int(np.argmax(conv))
        # Find contributing pairs for this lag
        j_min = max(0, peak_k - n + 1)
        j_max = min(n - 1, peak_k)
        if j_min >= j_max:
            break

        # Find the pair with maximum product
        idxs = np.arange(j_min, j_max + 1)
        products = a[idxs] * a[peak_k - idxs]
        max_pair_idx = idxs[np.argmax(products)]

        # Try reducing one element of the max pair
        partner = peak_k - max_pair_idx
        for element_to_reduce in [max_pair_idx, partner]:
            for frac in [0.01, 0.005, 0.002, 0.001]:
                delta = a[element_to_reduce] * frac
                if delta < 1e-15:
                    continue
                a_try = a.copy()
                a_try[element_to_reduce] -= delta
                # Redistribute to a random low element
                target = rng.integers(0, n)
                while target == element_to_reduce:
                    target = rng.integers(0, n)
                a_try[target] += delta
                s_try = a_try.sum()
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                if conv_try.max() < conv.max() - 1e-10:
                    a = a_try
                    conv = conv_try
                    no_improve = 0
                    if 2.0 * n * conv.max() < best_obj:
                        best_obj = 2.0 * n * conv.max()
                        best_a = a.copy()
                    break
            if no_improve == 0:
                break

        if no_improve > 0:
            no_improve += 1

        if no_improve >= max_patience:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine-based with various frequencies
    for freq in [1, 2, 3, 4, 6, 8]:
        arr = (np.sin(2 * np.pi * freq * x / n) ** 2 + 0.05)
        inits[f'sine{freq}'] = arr / arr.sum()

    # Cosine variants
    for freq in [1, 2, 3]:
        arr = (np.cos(2 * np.pi * freq * x / n) ** 2 + 0.05)
        inits[f'cos{freq}'] = arr / arr.sum()

    # Periodic spikes
    for num_spikes in [3, 4, 5, 7]:
        sp = np.zeros(n)
        positions = np.linspace(0, n - 1, num_spikes, dtype=int)
        sp[positions] = 1.0
        # Spread slightly
        for pos in positions:
            if pos > 0:
                sp[pos - 1] += 0.5
            if pos < n - 1:
                sp[pos + 1] += 0.5
        inits[f'spikes{num_spikes}'] = sp / sp.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Three-modal
    t1, t2 = n // 3, 2 * n // 3
    tm = np.zeros(n)
    tm[:t1//2] = 1.0
    tm[t1:t1+t1//2] = 1.0
    tm[-t1//2:] = 1.0
    inits['trimodal'] = tm / tm.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p_arr = np.maximum(parent, 0).astype(np.float64)
        ps = p_arr.sum()
        if ps > 1e-10 and np.isfinite(p_arr).all():
            if len(p_arr) == n:
                inits['parent'] = p_arr / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p_arr)), p_arr / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits

        t_per_init = max(10.0, min(40.0, remaining_for_phase1 / max(1, (len(lengths) - 1) * 5)))

        for name in init_names:
            if time.monotonic() > deadline - 350:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 300)
            if t_alloc < 8:
                break

            t_init_start = time.monotonic()
            a_opt, obj = mirror_descent(inits[name], time_limit=t_alloc,
                                         lr_init=0.008, temp=0.001, top_k=40,
                                         momentum=0.88, patience=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                candidates.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  MD: n={n}, {name}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  MD: n={n}, {name}: {score:.6f}")
            total_inits += 1

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s, {total_inits} inits). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP deep optimization (40% of time)
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Collect top candidates by unique length
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in candidates:
        if n not in seen_lengths:
            top_configs.append((n, a_opt.copy()))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None and len(best_sequence) not in seen_lengths:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 250.0
    n_cands = max(1, len(top_configs))
    t_per_lp = max(30.0, remaining * 0.55 / n_cands)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 200:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break

        print(f"  LP candidate {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"    -> {score:.6f} *NEW BEST*")
            else:
                print(f"    -> {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Multi-method polish (20% of time)
    # =========================================================
    print("=== Phase 3: Multi-Method Polish ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        n_best = len(best_sequence)
        t_per_method = remaining * 0.25

        # Block descent
        t_bd = min(t_per_method, 60.0)
        if t_bd > 8:
            a_opt, obj = block_descent(best_sequence, time_limit=t_bd, block_size=48)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  BlockDesc: {score:.6f} *NEW BEST*")
            else:
                print(f"  BlockDesc: {score:.6f}")

        # Peak-targeted
        t_pt = min(t_per_method, 40.0)
        if t_pt > 5:
            a_opt, obj = peak_targeted_update(best_sequence, time_limit=t_pt, rng=rng)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  PeakTarget: {score:.6f} *NEW BEST*")
            else:
                print(f"  PeakTarget: {score:.6f}")

        # Mirror descent with different params
        t_md = min(t_per_method, 50.0)
        if t_md > 8:
            a_opt, obj = mirror_descent(best_sequence, time_limit=t_md,
                                         lr_init=0.004, temp=0.0005, top_k=80,
                                         momentum=0.93, patience=400)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  MD polish: {score:.6f} *NEW BEST*")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Final LP polish with restarts (remaining time)
    # =========================================================
    print("=== Phase 4: Final LP Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 30:
            n_runs = min(4, int(remaining / 25) + 1)
            for ri in range(n_runs):
                if time.monotonic() > deadline - 10:
                    break
                t_alloc = min(remaining / (n_runs - ri) * 0.85, deadline - time.monotonic() - 8)
                if t_alloc < 8:
                    break

                if ri == 0:
                    a0 = best_sequence.copy()
                elif ri == 1:
                    noise = rng.standard_normal(len(best_sequence)) * 0.001
                    a0 = np.maximum(best_sequence + noise, 0)
                    a0 = a0 / a0.sum()
                elif ri == 2:
                    # Spherical perturbation
                    a0 = best_sequence.copy()
                    a0 += rng.standard_normal(len(a0)) * 0.003
                    a0 = np.maximum(a0, 0)
                    a0 = a0 / a0.sum()
                else:
                    #混合 with uniform
                    a0 = 0.95 * best_sequence + 0.05 * np.ones(len(best_sequence)) / len(best_sequence)
                    a0 = a0 / a0.sum()

                a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LP {ri}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Final LP {ri}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
