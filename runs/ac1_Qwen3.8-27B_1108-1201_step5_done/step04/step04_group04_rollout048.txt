"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) LP-Frank-Wolfe for global directions, (2) novel peak-equalization
gradient descent with FFT-based gradients, (3) coordinate swap refinement,
(4) multi-length exploration, (5) adaptive subgradient polishing.
Key innovation: peak-equalization step that equalizes top-K convolution peaks
via a smooth least-squares sub-objective, providing a different optimization
landscape than pure max or log-sum-exp approaches.

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_influence_gradient(a, w):

def peak_equalization(a0, time_limit, K_frac=0.1, lr_init=0.005, lr_decay=0.9995, verbose=False):
    """
    Peak equalization gradient descent.
    Minimizes sum_{k in topK} (c[k] - mean(c[topK]))^2 subject to a in simplex.
    This drives the top peaks toward equality, a necessary condition for optimality.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    K = max(5, int(n * K_frac))
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    T = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - T < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Find top K peaks
        top_k_idx = np.argpartition(conv, -K)[-K:]
        top_k_vals = conv[top_k_idx]
        M_target = np.mean(top_k_vals)
        
        # Build weight vector w
        w = np.zeros(len(conv))
        w[top_k_idx] = top_k_vals - M_target
        
        # Compute gradient
        g = compute_gradient_fast(f, w)
        
        # Project gradient to tangent space of simplex
        g = g - np.mean(g)
        
        # Update
        f_new = project_simplex(f - lr * g)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"      PE it {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
            f = f_new  # still take the step for momentum
        
        # Decay learning rate
        lr *= lr_decay
        
        if no_improve > 200:
            # Restart with fresh learning rate
            lr = lr_init * 0.5
            no_improve = 0
        
        if no_improve > 1000:
            # Perturb and restart
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = lr_init * 0.3
            K = max(3, int(n * K_frac * 0.5))
            no_improve = 0
            if time.monotonic() - T > time_limit:
                break
    
    return best_f, best_obj


def coordinate_swap_refine(a0, time_limit, max_passes=50, verbose=False):
    """
    Refine by trying pairwise mass transfers between coordinates.
    For each pair (i, j), try moving a small amount of mass from i to j
    (or j to i) and keep the best improvement.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    T = time.monotonic()
    no_improve_pass = 0
    
    for pageIndex in range(max_passes):
        if time.monotonic() - T > time_limit:
            break
        
        improved = False
        # Sample random pairs (for efficiency with large n)
        if n <= 200:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng = np.random.default_rng(pageIndex + 42)
            rng.shuffle(pairs)
        else:
            # Sample random pairs
            rng = np.random.default_rng(pageIndex + 42)
            num_pairs = min(5000, n * (n - 1) // 2)
            i1 = rng.integers(0, n, size=num_pairs)
            i2 = rng.integers(0, n, size=num_pairs)
            mask = i1 != i2
            i1, i2 = i1[mask], i2[mask]
            pairs = list(zip(i1, i2))
        
        for (i, j) in pairs:
            if time.monotonic() - T > time_limit:
                break
            if f[i] < 1e-10:
                continue
            
            # Try moving mass from i to j (and j to i)
            for (src, dst) in [(i, j), (j, i)]:
                if f[src] < 1e-10:
                    continue
                # Try different step sizes
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    delta = f[src] * frac
                    if delta < 1e-12:
                        continue
                    f_trial = f.copy()
                    f_trial[src] -= delta
                    f_trial[dst] += delta
                    conv_t = fast_convolve(f_trial)
                    obj_t = conv_t.max()
                    if obj_t < best_obj - 1e-12:
                        best_obj = obj_t
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
        
        if verbose and pageIndex % 5 == 0:
            print(f"      CS pass {pageIndex}: obj={best_obj:.10f}")
        
        if not improved:
            no_improve_pass += 1
            if no_improve_pass >= 3:
                break
        else:
            no_improve_pass = 0
    
    return best_f, best_obj


def lp_fw_step(f, M_target, n_constraints=None):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M_target, g >= 0.
    Uses a subset of constraints for speed."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: A_ub[k, j] contribution from f
    # (f*g)[k] = sum_j f[|k-j|] * g[j] (with bounds)
    # Actually (f*g)[k] = sum_j f[j] * g[k-j] where both indices valid
    # For LP: we need A_ub[k, j] = f[j] if k-j is in [0, n-1], i.e., j in [k-n+1, k]
    # And also f[k-j] if j is in [0, k]... 
    # Wait: (f*g)[k] = sum_{j=0}^{min(k, n-1)} f[j] * g[k-j] where 0 <= k-j < n
    # = sum_{j=max(0,k-n+1)}^{min(k, n-1)} f[j] * g[k-j]
    # So A_ub[k, j] = f[k-j] if max(0,k-n+1) <= j <= min(k, n-1), else 0
    # Equivalently: A_ub[k, j] = f[|k-j|] if |k-j| < n ... no that's not right.
    # Let me redo: (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # So coefficient of g[m] in (f*g)[k] is f[k-m] (if 0<=k-m<n)
    # A_ub[k, m] = f[k-m] if 0 <= k-m < n, else 0
    
    if n_constraints is None or n_constraints >= 2 * n - 1:
        k_range = range(2 * n - 1)
    else:
        # Use only constraints near the current peaks
        conv = fast_convolve(f)
        top_k = np.argpartition(conv, -n_constraints)[-n_constraints:]
        k_range = list(top_k)
    
    num_k = len(k_range)
    A_ub = np.zeros((num_k, n))
    for idx, k in enumerate(k_range):
        # A_ub[idx, m] = f[k-m] if 0 <= k-m < n
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            A_ub[idx, m_lo:m_hi + 1] = f[k - m_hi:k - m_lo + 1][::-1]
    
    b_ub = np.full(num_k, M_target)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, n_constraints=200, verbose=False):
    """LP-based Frank-Wolfe with constraint subset selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s

    M = conv.max()
    best_obj = M

    T = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - T < time_limit:
        it += 1
        g = lp_fw_step(f, M * m_exp, n_constraints=min(n_constraints, 2 * n - 1))
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:

        if no_improve > 40:
            rng = np.random.default_rng((it * 7919) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def subgradient_polish(a0, time_limit, top_k_frac=0.05, temp_init=0.00005, lr_init=0.001):
    """Subgradient method with adaptive peak weighting."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    K = max(5, int(n * top_k_frac))
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    T = time.monotonic()
    temp = temp_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - T < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Adaptive temperature
        top_k_idx = np.argpartition(conv, -K)[-K:]
        top_k_vals = conv[top_k_idx]
        spread = top_k_vals.max() - top_k_vals.min()
        if spread > 1e-15:
            target_temp = spread * 0.1
        else:
            target_temp = top_k_vals.max() * 0.01
        temp = 0.95 * temp + 0.05 * max(target_temp, 1e-8)
        
        # Compute weights
        scaled = top_k_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(K)
            w[np.argmax(top_k_vals)] = 1.0
        else:
            w /= ws
        
        # Build full weight vector and compute subgradient
        w_full = np.zeros(len(conv))
        w_full[top_k_idx] = w
        
        sg = compute_gradient_fast(f, w_full)
        sg = sg - np.mean(sg)
        
        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 50 == 0:
            lr *= 0.998
        
        if no_improve > 300:
            rng = np.random.default_rng((it * 4441 + 13) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_f, best_obj

    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    
    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    
    # Arnold-like: 1/sqrt pattern
    a = np.ones(n) / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    inits.append(a / a.sum())
    
    # Linear ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits.append(r / r.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP-FW, peak equalization,
    coordinate swap refinement, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, eval={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass
    
    # === Phase 1: Exploration (300s) ===
    print("=== Phase 1: Exploration ===")
    t_start = time.monotonic()
    phase1_budget = 300.0
    
    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 368
    
    # Try base length with LP-FW
    rem = phase1_budget - (time.monotonic() - t_start)
    if rem > 60:
        print(f"  LP-FW on parent/base n={n_base}")
        a_opt, _ = lp_fw_optimize(parent if parent is not None else np.ones(n_base) / n_base,
                                   time_limit=min(120.0, rem - 30))
        try_seq(a_opt, "p1_base_fw")
    
    # Try neighboring lengths
    rem = phase1_budget - (time.monotonic() - t_start)
    other_lengths = []
    for delta in [-16, -8, 8, 16, 32, -32, 64, -64]:
        cand = n_base + delta
        if 128 <= cand <= 800 and cand != n_base:
            other_lengths.append(cand)
    
    rng.shuffle(other_lengths)
    for n_try in other_lengths[:3]:
        rem = phase1_budget - (time.monotonic() - t_start)
        if rem < 40:
            break
        inits = generate_inits(n_try, rng, parent)
        # Pick best init quickly
        best_init = None
        best_init_obj = float('inf')
        for fi in inits[:4]:
            obj = fast_convolve(fi).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = fi
        if best_init is not None:
            print(f"  LP-FW n={n_try}")
            a_opt, _ = lp_fw_optimize(best_init, time_limit=min(50.0, rem - 20))
            try_seq(a_opt, f"p1_n{n_try}")
    
    # Peak equalization on current best
    rem = phase1_budget - (time.monotonic() - t_start)
    if rem > 60 and best_seq is not None:
        f_work = best_seq / best_seq.sum()
        print(f"  Peak-eq on best (n={len(f_work)})")
        a_opt, _ = peak_equalization(f_work, time_limit=min(100.0, rem - 20), K_frac=0.1, lr_init=0.008)
        try_seq(a_opt, "p1_peakeq")
    
    print(f"  Phase 1 done: best={best_score:.6f}")
    
    # === Phase 2: Deep optimization (1000s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()
    phase2_budget = 1000.0
    
    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)
        
        # Alternating: peak-eq -> LP-FW -> subgrad -> coordinate swap
        num_rounds = 8
        t_per_round = phase2_budget / num_rounds
        
        for r in range(num_rounds):
            rem = phase2_budget - (time.monotonic() - t_start2)
            if rem < 30:
                break
            t_each = min(t_per_round / 2.0, rem / 2.0)
            
            # A: Peak equalization
            if rem > 40:
                print(f"  Round {r}: Peak-eq (t={t_each:.0f}s)")
                a_opt, _ = peak_equalization(f_work, time_limit=t_each, K_frac=0.08, lr_init=0.004)
                try_seq(a_opt, f"p2_pe_{r}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
            
            # B: LP-FW
            rem = phase2_budget - (time.monotonic() - t_start2)
            if rem > 40:
                t_lpfw = min(t_per_round / 2.5, rem / 2.0)
                print(f"  Round {r}: LP-FW (t={t_lpfw:.0f}s)")
                a_opt, _ = lp_fw_optimize(f_work, time_limit=t_lpfw, n_constraints=150)
                try_seq(a_opt, f"p2_fw_{r}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
            
            # C: Subgradient
            rem = phase2_budget - (time.monotonic() - t_start2)
            if rem > 30:
                t_sg = min(t_per_round / 3.0, rem / 2.0)
                print(f"  Round {r}: Subgrad (t={t_sg:.0f}s)")
                a_opt, _ = subgradient_polish(f_work, time_limit=t_sg, top_k_frac=0.06,
                                               temp_init=0.00004, lr_init=0.0008)
                try_seq(a_opt, f"p2_sg_{r}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
            
            # D: Coordinate swap (every 2 rounds)
            rem = phase2_budget - (time.monotonic() - t_start2)
            if rem > 30 and r % 2 == 0:
                t_cs = min(60.0, rem / 2.0)
                print(f"  Round {r}: Coord-swap (t={t_cs:.0f}s)")
                a_opt, _ = coordinate_swap_refine(f_work, time_limit=t_cs, max_passes=20)
                try_seq(a_opt, f"p2_cs_{r}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()
        
        # Try slightly different lengths
        rem = phase2_budget - (time.monotonic() - t_start2)
        if rem > 60 and best_seq is not None:
            n_best = len(best_seq)
            for delta in [16, -16, 32]:
                new_n = n_best + delta
                if new_n < 128 or new_n > 800:
                    continue
                px = np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n_best), f_work)
                px = project_simplex(px)
                t_alloc = min(40.0, rem / 3.0)
                a_opt, _ = peak_equalization(px, time_limit=t_alloc, K_frac=0.08, lr_init=0.005)
                try_seq(a_opt, f"p2_len{new_n}")
                rem = phase2_budget - (time.monotonic() - t_start2)
                if rem < 30:
                    break
    
    print(f"  Phase 2 done: best={best_score:.6f}")
    
    # === Phase 3: Fine polishing (400s) ===
    print("\n=== Phase 3: Fine Polishing ===")
    t_start3 = time.monotonic()
    phase3_budget = 400.0
    
    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        
        # Multi-scale peak equalization
        configs = [
            {"K_frac": 0.05, "lr_init": 0.002, "label": "pe_k5"},
            {"K_frac": 0.03, "lr_init": 0.001, "label": "pe_k3"},
            {"K_frac": 0.02, "lr_init": 0.0005, "label": "pe_k2"},
        ]
        
        for cfg in configs:
            rem = phase3_budget - (time.monotonic() - t_start3)
            if rem < 30:
                break
            t_alloc = min(80.0, rem - 20)
            print(f"  {cfg['label']} (t={t_alloc:.0f}s)")
            a_opt, _ = peak_equalization(f_work, time_limit=t_alloc, **{k: v for k, v in cfg.items() if k != 'label'})
            try_seq(a_opt, cfg['label'])
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
        
        # Coordinate swap intensive
        rem = phase3_budget - (time.monotonic() - t_start3)
        if rem > 60:
            print(f"  Coord-swap intensive (t={min(120.0, rem-20):.0f}s)")
            a_opt, _ = coordinate_swap_refine(f_work, time_limit=min(120.0, rem - 20), max_passes=50)
            try_seq(a_opt, "p3_cs_int")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
        
        # Final subgradient
        rem = phase3_budget - (time.monotonic() - t_start3)
        if rem > 30:
            print(f"  Final subgrad (t={min(60.0, rem-15):.0f}s)")
            a_opt, _ = subgradient_polish(f_work, time_limit=min(60.0, rem - 15),
                                           top_k_frac=0.03, temp_init=0.00002, lr_init=0.0004)
            try_seq(a_opt, "p3_final_sg")
        
        # Tiny LP-FW
        rem = phase3_budget - (time.monotonic() - t_start3)
        if rem > 30:
            a_opt, _ = lp_fw_optimize(f_work, time_limit=min(50.0, rem - 15), n_constraints=100)
            try_seq(a_opt, "p3_tiny_fw")
    
    print(f"  Phase 3 done: best={best_score:.6f}")
    
    # === Phase 4: Final attempts ===
    print("\n=== Phase 4: Final Attempts ===")
    
    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)
        
        # Tiny perturbations
        for trial in range(5):
            if time.monotonic() > deadline - 20:
                break
            noise_scale = 0.0003 * (trial + 1)
            f_pert = project_simplex(f_work + rng.standard_normal(n_work) * noise_scale)
            a_opt, _ = peak_equalization(f_pert, time_limit=15, K_frac=0.05, lr_init=0.001)
            try_seq(a_opt, f"p4_pert{trial}")
        
        # Mirror symmetry attempt
        mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
        mirrored = project_simplex(mirrored)
        if np.max(np.abs(mirrored - f_work)) > 1e-6:
            a_opt, _ = peak_equalization(mirrored, time_limit=20, K_frac=0.05, lr_init=0.001)
            try_seq(a_opt, "p4_mirror")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
