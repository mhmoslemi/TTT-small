```
def fast_convolve(a):
    n = len(a)
    if n == 0: return np.array([], dtype=float)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    # eliminate tiny negatives
    mn = C.min()
    if mn < 0:
        C = np.maximum(C, 0.0)
    return C
```

```
def random_start(n, rng, kind):
    if n <=0: return np.ones(1)
    if kind == 0:
        a = rng.dirichlet(np.full(n, 0.5, dtype=float))
        return normalize_seq(a)
    if kind == 1:
        f = rng.normal(size=n)
        F = np.fft.rfft(f)
        freqs = np.fft.rfftfreq(F.size) * F.size
        scale = max(1.0, n / 8.0)
        mask = np.exp(-(freqs / scale) ** 2)
        g = np.abs(np.fft.irfft(F * mask, n))
        if g.sum() < 1e-12:
            g = np.abs(f)
        return normalize_seq(g)
    # kind 2
    t = np.arange(n, dtype=float)
    a = np.ones(n)
    a += 0.5 * rng.normal(size=n)
    a += 0.4 * np.cos(2.0 * np.pi * t / float(n)) ** 2
    a += 0.3 * np.sin(4.0 * np.pi * t / float(n)) ** 2
    return normalize_seq(a)
```

```python
"""
Searches for non-negative sequences minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

The main optimizer uses an active-set trust-region linear program for the
nonsmooth max-convolution objective:
  - maintain a small active set of convolution coefficients near the maximum;
  - solve a small LP for a direction h that decreases all active linearized
    coefficients within a trust box;
  - perform an FFT-based line search to accept only genuine improvements.

Log-sum-exp projected gradient polishing and random restarts are used to
escape local traps.  The code starts from height_sequence_1 when available.
"""

import time
import numpy as np
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear convolution a*a via zero-padded FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    if C.size and C.min() < 0.0:
        C = np.maximum(C, 0.0)
    return C


def normalize_seq(a):
    """Normalize to a non-negative probability vector."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([max(0.0, float(v[0])), 0.0][:1], dtype=np.float64) if v[0] > 0 else np.zeros(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    rho_idx = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho_idx] - 1.0) / float(rho_idx + 1)
    return np.maximum(v - theta, 0.0)


def local_score(q):
    """Score for a normalized sequence: 2*n*max(convolve(q,q))."""
    q = np.asarray(q, dtype=np.float64)
    n = q.size
    if n == 0:
        return float("inf")
    return float(2.0 * n * fast_convolve(q).max())


def try_sequence(seq, tracker):
    """Evaluate with the official evaluator and update tracker."""
    a = np.asarray(seq, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-15:
        return float("inf")
    a = a / s
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float("inf")
    if np.isfinite(score) and score < tracker[0]:
        tracker[0] = score
        tracker[1] = a.copy()
        print(f"  NEW BEST official score = {score:.10f}, n = {a.size}")
    return score


def get_parent():
    """Return height_sequence_1 if it exists and is valid."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    if p.size == 0 or p.sum() <= 1e-15:
        return None
    return p


def build_slice_info(n):
    """Precompute slice indices for gradient matrices of convolution."""
    m = 2 * n - 1
    k = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k - n + 1)
    je = np.minimum(n - 1, k)
    start = k - js
    end = k - je
    return js, je, start, end


def select_active(C, M, expand=1.0):
    """Select convolution coefficients near the current maximum."""
    C = np.asarray(C, dtype=np.float64)
    m = C.size
    if m == 0:
        return np.array([], dtype=np.int64)
    ef = max(1.0, float(expand))
    tol = max(1e-9, 1e-7 * ef, 1e-5 * float(M) * ef)
    idx = np.where(C >= M - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(C))], dtype=np.int64)
    # If too many become active, keep the largest 800 for speed.
    if idx.size > 800:
        idx = idx[np.argsort(C[idx])[::-1][:800]]
    return np.sort(idx.astype(np.int64))


def lp_trust_step(q, C, M, r, slice_info, lp_time=3.0, expand=1.0):
    """
    Solve an active-set trust-region LP:
      min u
      s.t. grad(C_k, q) . h <= u    for active k
           sum h = 0
           -min(q_i, r) <= h_i <= min(1-q_i, r)
    Returns (h, u, n_active). h is None on failure.
    """
    q = np.asarray(q, dtype=np.float64)
    n = q.size
    if n == 0 or M <= 0.0:
        return None, 0.0, 0

    C = np.asarray(C, dtype=np.float64)
    rows = select_active(C, M, expand)
    if rows.size == 0:
        return None, 0.0, 0

    js_arr, je_arr, start_arr, end_arr = slice_info
    A = np.zeros((rows.size, n + 1), dtype=np.float64)

    js_r = js_arr[rows]
    je_r = je_arr[rows]
    st_r = start_arr[rows]
    en_r = end_arr[rows]

    for t in range(rows.size):
        js = int(js_r[t])
        je = int(je_r[t])
        if js <= je:
            st = int(st_r[t])
            en = int(en_r[t])
            A[t, js:je + 1] = 2.0 * q[en:st + 1][::-1]

    A[:, -1] = -1.0
    b = np.zeros(rows.size, dtype=np.float64)

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, -1] = 0.0
    b_eq = np.zeros(1, dtype=np.float64)

    r = min(1.0, max(1e-4, float(r)))
    if r >= 1.0:
        lo = -q.copy()
        hi = (1.0 - q).copy()
    else:
        lo = -np.minimum(q, r)
        hi = np.minimum(1.0 - q, r)

    bad = lo > hi
    if np.any(bad):
        mid = 0.5 * (lo[bad] + hi[bad])
        lo[bad] = mid
        hi[bad] = mid

    bounds = [(float(lo[i]), float(hi[i])) for i in range(n)]
    bounds.append((-10.0, 0.0))

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(lp_time)},
        )
    except Exception:
        return None, 0.0, int(rows.size)

    if res.success:
        h = np.asarray(res.x[:n], dtype=np.float64)
        u = float(res.x[-1])
        if u < -1e-10 and float(np.linalg.norm(h)) > 1e-12:
            # Clean up tiny bound violations but do not spend time on full projection.
            h = np.clip(h, lo, hi)
            if abs(float(h.sum())) > 1e-7:
                h = h - float(h.mean())
            return h, u, int(rows.size)

    return None, 0.0, int(rows.size)


def line_search(q, h, cur_max):
    """Grid + local-grid line search for q + alpha*h."""
    q = np.asarray(q, dtype=np.float64)
    h = np.asarray(h, dtype=np.float64)
    if h.size == 0 or float(np.linalg.norm(h)) <= 1e-14:
        return None, float(cur_max), 0.0

    best_val = float(cur_max)
    best_q = None
    best_alpha = 0.0

    coarse = [0.05, 0.10, 0.20, 0.30, 0.50, 0.70, 1.00]
    for al in coarse:
        qn = q + al * h
        if qn.min() < -1e-9:
            continue
        qn = np.maximum(qn, 0.0)
        s = qn.sum()
        if s <= 1e-15:
            continue
        qn = qn / s
        val = float(fast_convolve(qn).max())
        if val < best_val:
            best_val = val
            best_q = qn
            best_alpha = float(al)

    if best_q is not None:
        width = 0.12 if 0.08 < best_alpha < 0.92 else 0.08
        lo = max(0.0, best_alpha - width)
        hi = min(1.0, best_alpha + width)
        for al in np.linspace(lo, hi, 15):
            if al <= 0.0:
                continue
            qn = q + al * h
            if qn.min() < -1e-9:
                continue
            qn = np.maximum(qn, 0.0)
            s = qn.sum()
            if s <= 1e-15:
                continue
            qn = qn / s
            val = float(fast_convolve(qn).max())
            if val < best_val:
                best_val = val
                best_q = qn
                best_alpha = float(al)

    if best_q is None:
        return None, float(cur_max), 0.0
    return best_q, best_val, best_alpha


def lse_gradient(weights, f):
    """Gradient of sum_k weights[k]*conv(f,f)[k] via FFT."""
    f = np.asarray(f, dtype=np.float64)
    weights = np.asarray(weights, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    fft_size = 1 << (total - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def lse_optimize(q0, time_limit, tracker=None, beta_start=20.0, beta_end=3000.0,
                 lr_init=0.010, top_k=None, verbose=False):
    """
    Log-sum-exp smoothed projected gradient optimizer.
    Returns (best_q, best_local_score).
    """
    q = normalize_seq(q0)
    n = q.size
    if n == 0:
        return q, float("inf")
    if time_limit <= 0.0:
        return q, local_score(q)

    C = fast_convolve(q)
    M = float(C.max())
    cur_score = 2.0 * n * M
    best_q = q.copy()
    best_score = cur_score
    if tracker is not None:
        try_sequence(best_q, tracker)

    v = np.zeros(n, dtype=np.float64)
    lr = float(lr_init)
    mom = 0.90
    beta = max(1.0, float(beta_start))
    beta_end = max(beta * 1.05, float(beta_end))
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        C = fast_convolve(q)
        M = float(C.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_q = q.copy()
            if tracker is not None and (cur_score < tracker[0] - 1e-7 or it % 25 == 0):
                try_sequence(best_q, tracker)

        m = C.size
        use_top = (top_k is not None and int(top_k) > 0 and int(top_k) < m)
        if use_top:
            k = min(int(top_k), m)
            if k >= m:
                idx = np.arange(m)
            else:
                idx = np.argpartition(C, m - k)[m - k:]
            csub = C[idx]
            wsub = np.exp(beta * (csub - csub.max()))
            ssub = wsub.sum()
            w = np.zeros(m, dtype=np.float64)
            if ssub > 0.0:
                w[idx] = wsub / ssub
        else:
            w = np.exp(beta * (C - C.max()))
            ssum = w.sum()
            if ssum <= 0.0:
                w = np.ones(m, dtype=np.float64) / m
            else:
                w = w / ssum

        # Tiny uniform smoothing avoids overly spiky weights.
        w = (1.0 - 1e-7) * w + 1e-7 / m
        w /= w.sum()

        grad = lse_gradient(w, q)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-13:
            no_improve += 25
        else:
            grad = grad / gnorm
            v = mom * v - lr * grad
            q_new = project_simplex(q + v)
            direction = q_new - q
            dx = float(np.linalg.norm(direction))
            if dx > 1e-13:
                best_val = M
                best_alpha = 0.0
                for al in (1.0, 0.6, 0.3, 0.15):
                    qc = q + al * direction
                    qc = np.maximum(qc, 0.0)
                    s = qc.sum()
                    if s <= 1e-15:
                        continue
                    qc = qc / s
                    val = float(fast_convolve(qc).max())
                    if val < best_val:
                        best_val = val
                        best_alpha = al

                if best_alpha > 0.0 and best_val < M - 1e-12:
                    q = q + best_alpha * direction
                    q = np.maximum(q, 0.0)
                    q = q / q.sum()
                    v = best_alpha * v
                    no_improve = 0
                    lr = min(0.06, lr * (1.0 + 0.35 * best_alpha))
                    mom = min(0.96, mom + 0.001)
                else:
                    no_improve += 1
                    v = 0.5 * v
                    if no_improve > 8:
                        lr *= 0.85
                        mom = max(0.80, mom - 0.01)
                    if no_improve > 70:
                        q = best_q.copy()
                        v = np.zeros(n, dtype=np.float64)
                        lr = float(lr_init)
                        mom = 0.90
                        beta = min(beta * 1.25, beta_end)
                        no_improve = 0

        beta = min(beta * 1.004, beta_end)
        if no_improve > 260:
            break
        if verbose and it % 100 == 0:
            print(f"      LSE it={it} best={best_score:.10f} beta={beta:.2f} lr={lr:.6f} no={no_improve}")

    if tracker is not None:
        try_sequence(best_q, tracker)
    return best_q, best_score


def run_lp_phase(q0, time_limit, tracker=None, r_init=0.25, lp_time=3.0, verbose=False):
    """
    Main active-set LP phase.  Returns (best_q, best_local_score).
    """
    q = normalize_seq(q0)
    n = q.size
    if n == 0:
        return q, float("inf")
    if time_limit <= 0.0:
        return q, local_score(q)

    slice_info = build_slice_info(n)
    C = fast_convolve(q)
    M = float(C.max())
    cur_score = 2.0 * n * M
    best_q = q.copy()
    best_score = cur_score
    if tracker is not None:
        try_sequence(best_q, tracker)

    r = float(r_init)
    expand = 1.0
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        C = fast_convolve(q)
        M = float(C.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_q = q.copy()
            if tracker is not None and (cur_score < tracker[0] - 1e-7 or it % 25 == 0):
                try_sequence(best_q, tracker)

        h, u, n_active = lp_trust_step(q, C, M, r, slice_info,
                                        lp_time=lp_time, expand=expand)

        if h is not None:
            qn, val, alpha = line_search(q, h, M)
            if qn is not None and val < M - 1e-12:
                q = qn
                new_score = 2.0 * n * float(val)
                no_improve = 0
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_q = q.copy()
                    if tracker is not None and (new_score < tracker[0] - 1e-7 or it % 25 == 0):
                        try_sequence(best_q, tracker)

                # Adapt trust radius and active tolerance.
                if alpha < 0.12:
                    expand = min(50.0, max(1.0, expand * 1.6))
                    r = max(0.03, r * 0.80)
                elif alpha < 0.40:
                    expand = min(50.0, max(1.0, expand * 1.2))
                    r = max(0.03, r * 0.95)
                elif alpha < 0.75:
                    expand = max(1.0, expand * 0.8)
                    r = min(1.0, r * 1.10)
                else:
                    expand = max(1.0, expand * 0.6)
                    r = min(1.0, r * 1.25)
            else:
                no_improve += 1
                # If a direction was found but line search failed, a nearby
                # coefficient probably spiked; expand active tolerance.
                if n_active > 5:
                    expand = min(50.0, expand * 1.5)
                else:
                    expand = min(50.0, expand * 1.2)
                r = max(0.03, r * 0.75)
        else:
            no_improve += 1
            # If no negative direction exists with current active set, the
            # active set is likely too large; tighten it if possible.
            if n_active > 5:
                expand = max(1.0, expand * 0.5)
            else:
                expand = 1.0
            r = max(0.03, r * 0.80)

        # Occasionally smooth with LSE.
        if no_improve >= 8:
            elapsed = time.monotonic() - t0
            rem = time_limit - elapsed
            if rem > 12.0:
                tk = max(16, n // 16)
                q_lse, obj_lse = lse_optimize(
                    best_q,
                    min(15.0, rem),
                    tracker=tracker,
                    beta_start=30.0,
                    beta_end=800.0,
                    lr_init=0.008,
                    top_k=tk,
                    verbose=False,
                )
                if obj_lse < best_score - 1e-12:
                    q = q_lse.copy()
                    best_q = q.copy()
                    best_score = obj_lse
                    if tracker is not None:
                        try_sequence(best_q, tracker)
                    no_improve = 0
                    r = 0.25
                    expand = max(1.0, expand * 0.5)
                else:
                    no_improve += 10

        # Random restart from best if stuck.
        if no_improve >= 25:
            elapsed = time.monotonic() - t0
            rem = time_limit - elapsed
            if rem > 30.0:
                extra = max(0, no_improve - 25)
                sigma = min(0.02, 0.002 + 0.0008 * float(extra))
                seed = (int(time.monotonic() * 1000000) + it * 101) % (2**31 - 1)
                rng = np.random.default_rng(seed)
                q = project_simplex(best_q + rng.standard_normal(n) * sigma)
                C = fast_convolve(q)
                M = float(C.max())
                cur_score = 2.0 * n * M
                if cur_score > best_score + 0.10 or cur_score > 2.0 * best_score:
                    q = best_q.copy()
                no_improve = 0
                r = 0.20
                expand = 1.0

        if no_improve >= 90:
            break

        if verbose and it % 50 == 0:
            print(f"      LP it={it} best={best_score:.10f} cur={cur_score:.10f} "
                  f"r={r:.3f} expand={expand:.2f} no={no_improve}")

    if tracker is not None:
        try_sequence(best_q, tracker)
    return best_q, best_score


def random_start(n, rng, kind):
    """Generate a diverse non-negative probability vector."""
    if n <= 0:
        return np.ones(1, dtype=np.float64)

    if kind == 0:
        a = rng.dirichlet(np.full(n, 0.5, dtype=np.float64))
        return normalize_seq(a)

    if kind == 1:
        f = rng.normal(size=n)
        F = np.fft.rfft(f)
        freqs = np.fft.rfftfreq(F.size) * F.size
        scale = max(1.0, n / 8.0)
        mask = np.exp(-(freqs / scale) ** 2)
        a = np.abs(np.fft.irfft(F * mask, n))
        if a.sum() < 1e-12:
            a = np.abs(f)
        return normalize_seq(a)

    t = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    a += 0.5 * rng.normal(size=n)
    a += 0.4 * np.cos(2.0 * np.pi * t / float(n)) ** 2
    a += 0.3 * np.sin(4.0 * np.pi * t / float(n)) ** 2
    return normalize_seq(a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Entry point.  Returns the best non-negative sequence found.
    """
    start = time.monotonic()
    actual = max(1.0, float(budget_s))
    margin = min(10.0, actual * 0.10)
    deadline = start + actual - margin
    if deadline < start:
        deadline = start + 0.1

    tracker = [float("inf"), None]

    parent = get_parent()
    parent_loaded = False
    if parent is not None:
        parent = normalize_seq(parent)
        sc = try_sequence(parent, tracker)
        q0 = parent
        parent_loaded = True
        print(f"Loaded parent: n={parent.size}, official score={sc:.10f}")
    else:
        q0 = np.ones(448, dtype=np.float64) / 448.0
        sc = try_sequence(q0, tracker)
        print(f"No parent loaded; starting from uniform n=448, score={sc:.10f}")

    # Phase 1: strong LP polishing from the parent / initial point.
    rem = deadline - time.monotonic()
    alloc = min(700.0, rem - 500.0)
    if alloc > 60.0:
        print(f"\nPhase 1: active-set LP polish, alloc={alloc:.0f}s")
        run_lp_phase(q0, alloc, tracker=tracker, r_init=0.25, lp_time=4.0, verbose=True)

    # Phase 2: LSE smoothing.
    rem = deadline - time.monotonic()
    cur = tracker[1] if tracker[1] is not None else q0
    alloc = min(250.0, rem - 350.0)
    if alloc > 30.0:
        print(f"\nPhase 2: LSE polish, alloc={alloc:.0f}s")
        lse_optimize(cur, alloc, tracker=tracker,
                     beta_start=20.0, beta_end=3000.0,
                     lr_init=0.010, top_k=max(32, cur.size // 8),
                     verbose=False)

    # Phase 3: second LP refinement.
    rem = deadline - time.monotonic()
    cur = tracker[1] if tracker[1] is not None else q0
    alloc = min(350.0, rem - 250.0)
    if alloc > 40.0:
        print(f"\nPhase 3: LP refinement, alloc={alloc:.0f}s")
        run_lp_phase(cur, alloc, tracker=tracker, r_init=0.15, lp_time=5.0, verbose=True)

    # Phase 4: try a symmetrized version.
    rem = deadline - time.monotonic()
    if rem > 250.0:
        cur = tracker[1] if tracker[1] is not None else q0
        sym = normalize_seq(0.5 * (cur + cur[::-1]))
        sc = try_sequence(sym, tracker)
        if sc <= tracker[0] + 0.02:
            alloc = min(180.0, rem - 150.0)
            if alloc > 30.0:
                print(f"\nPhase 4a: symmetrized LP polish, alloc={alloc:.0f}s")
                run_lp_phase(sym, alloc, tracker=tracker, r_init=0.2, lp_time=3.0, verbose=True)

    # Phase 5: try nearby lengths.
    for nalt in (512, 640, 384):
        rem = deadline - time.monotonic()
        if rem < 150.0:
            break
        base = tracker[1] if tracker[1] is not None else q0
        if base.size == nalt:
            continue
        x_old = np.linspace(0.0, 1.0, base.size)
        x_new = np.linspace(0.0, 1.0, nalt)
        qalt = np.interp(x_new, x_old, base)
        qalt = normalize_seq(qalt)
        sc = try_sequence(qalt, tracker)

        alloc = min(180.0, (rem - 100.0) / 3.0)
        if alloc > 30.0 and sc < tracker[0] + 0.25:
            print(f"\nPhase 5: alternative length {nalt}, alloc={alloc:.0f}s")
            run_lp_phase(qalt, alloc, tracker=tracker, r_init=0.20, lp_time=3.0, verbose=True)

            rem = deadline - time.monotonic()
            if rem > 80.0:
                cur2 = tracker[1] if tracker[1] is not None else qalt
                if cur2.size == nalt:
                    lse_optimize(cur2, min(60.0, rem - 60.0), tracker=tracker,
                                 beta_start=30.0, beta_end=1500.0,
                                 lr_init=0.008, top_k=max(16, cur2.size // 16),
                                 verbose=False)

    # Phase 6: if the parent was absent or poor, try several random basins.
    rem = deadline - time.monotonic()
    if rem > 250.0 and (not parent_loaded or tracker[0] > 1.55):
        rng = np.random.default_rng(987654321)
        for n_rand in (448, 512, 640):
            rem = deadline - time.monotonic()
            if rem < 120.0:
                break
            for kind in range(3):
                rem = deadline - time.monotonic()
                if rem < 70.0:
                    break
                qrand = random_start(n_rand, rng, kind)
                sc = try_sequence(qrand, tracker)
                alloc = min(50.0, rem - 60.0)
                if alloc > 20.0 and sc < max(2.0, tracker[0] + 0.3):
                    print(f"\nPhase 6: random basin n={n_rand} kind={kind}, alloc={alloc:.0f}s")
                    run_lp_phase(qrand, alloc, tracker=tracker,
                                 r_init=0.30, lp_time=2.0, verbose=False)

    # Final polish.
    rem = deadline - time.monotonic()
    if rem > 80.0:
        cur = tracker[1] if tracker[1] is not None else q0
        alloc = min(120.0, rem - 20.0)
        print(f"\nFinal LP polish, alloc={alloc:.0f}s")
        run_lp_phase(cur, alloc, tracker=tracker, r_init=0.08, lp_time=4.0, verbose=False)

    final = tracker[1]
    if final is None:
        final = normalize_seq(q0)
    # One last official evaluation and synchronization.
    try_sequence(final, tracker)

    final = tracker[1] if tracker[1] is not None else final
    final = np.asarray(final, dtype=np.float64)
    final = np.maximum(final, 0.0)
    s = final.sum()
    if s <= 1e-15:
        final = np.ones(1)
    else:
        final = final / s

    print(f"FINAL returned score tracked = {tracker[0]:.10f}, n={final.size}")
    return [float(x) for x in final]
```
