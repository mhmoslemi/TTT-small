"""

2. LP-based Frank-Wolfe for global direction finding

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(weights, f):
    """Compute gradient of LSE w.r.t. f."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP oracle."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - j] for j in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M (elementwise), g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def nesterov_lse(f0, time_limit, beta_start=30.0, beta_end=5000.0, 
                  lr_scale=1.5, momentum=0.95):
    """Nesterov-accelerated LSE gradient descent with adaptive beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # FISTA variables
    y = f.copy()  # extrapolation point
    t_k = 1.0
    lr = lr_scale / np.sqrt(n)  # initial step size
    
    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, y)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break
        grad = grad / grad_norm
        
        # FISTA update
        t_old = t_k
        f_new = project_simplex(y - lr * grad)
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.9
            if no_improve > 100:
                # Restart with perturbation
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.001 + 0.002 * min(1.0, (no_improve - 100) / 100)
                f_new = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv_new = fast_convolve(f_new)
                obj_new = conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_f = f_new.copy()
                f_new = best_f.copy()
                no_improve = 0
                lr = lr_scale / np.sqrt(n)
                beta = min(beta * 1.3, beta_end)
        
        # FISTA extrapolation
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0
        momentum = (t_old - 1.0) / t_k
        y = f_new + momentum * (f_new - f)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = f_new.copy()
        else:
            y = y / y_sum
        
        f = f_new
        
        # Ramp up beta
        if beta < beta_end:
            beta *= 1.005
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLSE it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, t={t_k:.1f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj

def fw_lp_optimize(f0, time_limit, lp_timeout=15):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

        f = f / s
    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP with relaxed constraint
        g = solve_lp_direction(f, M * m_relax)
        if g is None:
            m_relax *= 1.15
            if m_relax > 5.0:
                m_relax = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            m_relax = min(m_relax * 1.03, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.array([0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0])
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Refine with bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 4.0)
        
        if no_improve > 30:
            rng = np.random.default_rng(it * 13 + 42)
            scale = 0.002 + 0.003 * min(1.0, (no_improve - 30) / 50)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0
    
    return best_f, best_obj

def peak_aware_perturb(f, peaks, n_perturb=20, scale=0.005):
    """Perturb f by reducing mass at positions that contribute to peaks."""
    n = len(f)
    f_new = f.copy()
    rng = np.random.default_rng(int(time.monotonic() * 1000000) % 999983)
    
    for _ in range(n_perturb):
        # Pick a random peak
        k = peaks[rng.integers(len(peaks))]
        # Find positions that contribute to this peak
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        contributing = np.arange(j_start, j_end + 1)
        # Reduce some of them
        idx = contributing[rng.integers(len(contributing))]
        f_new[idx] *= (1.0 - scale)
    
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new = f_new / s
    return f_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Nesterov-accelerated log-sum-exp gradient descent with adaptive annealing
2. Periodic LP-based Frank-Wolfe jumps for escaping local minima
3. Multi-length exploration via interpolation and cross-optimization
4. Final high-precision polishing with peak-aware restarts

Key innovations over prior state-of-art:
- FISTA-style acceleration for faster LSE convergence
- Adaptive beta annealing with "cooling" restarts (reduce beta to escape)
- Peak-aware perturbations that target specific convolution peaks
- More aggressive multi-length search including larger n values
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """
    Compute gradient of weighted sum of convolution w.r.t. f.
    grad[j] = 2 * sum_k weights[k] * f[k-j] (with valid index checking).
    Uses FFT for efficiency.
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix A where row k represents (f*g)[k] = sum_j f[k-j]*g[j]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - j] for j in js)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, timeout=12):
    """Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def nesterov_lse(f0, time_limit, beta_start=20.0, beta_end=5000.0,
                 lr_init_scale=2.0, cool_prob=0.02):
    """
    Nesterov-accelerated LSE gradient descent with adaptive beta annealing.
    Includes simulated-annealing style beta cooling to escape local minima.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # FISTA state
    y = f.copy()
    t_fista = 1.0
    lr = lr_init_scale / n
    beta = beta_start

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at extrapolated point y
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, y)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Take gradient step with projection
        x_new = project_simplex(y - lr * grad)

        # FISTA extrapolation
        t_old = t_fista
        t_fista = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0
        momentum = (t_old - 1.0) / t_fista
        # Cap momentum to avoid instability
        momentum = min(momentum, 0.95)
        y = x_new + momentum * (x_new - f)
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s < 1e-15:
            y = x_new.copy()
        else:
            y = y / y_s

        # Evaluate
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = x_new.copy()
            no_improve = 0
            # Slightly increase lr when making progress
            lr = min(lr * 1.003, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.93
                t_fista = max(1.0, t_fista * 0.9)
            if no_improve > 120:
                # Peak-aware restart
                peaks = np.argsort(conv_new)[-max(5, n // 50):][::-1]
                scale = 0.002 + 0.004 * min(1.0, (no_improve - 120) / 100)
                # Reduce mass at positions contributing to peaks
                f_new = best_f.copy()
                for _ in range(10):
                    k = peaks[rng.integers(len(peaks))]
                    j_start = max(0, k - n + 1)
                    j_end = min(n - 1, k)
                    idx = rng.integers(j_start, j_end + 1)
                    f_new[idx] *= (1.0 - scale)
                f_new = np.maximum(f_new, 0.0)
                f_new = project_simplex(f_new)
                conv_new = fast_convolve(f_new)
                obj_new = conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_f = f_new.copy()
                x_new = best_f.copy()
                no_improve = 0
                lr = lr_init_scale / n
                t_fista = 1.0
                y = x_new.copy()
                # Annealing: sometimes decrease beta

                    beta = max(beta_start, beta * 0.5)
                else:
                    beta = min(beta * 1.3, beta_end)
                continue

        f = x_new

        # Ramp up beta gradually
        if beta < beta_end:
            beta *= 1.004

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLSE it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, t_fista={t_fista:.1f}, elapsed={elapsed:.0f}s")

def fw_lp_optimize(f0, time_limit, max_lp_solves=25):
    """Frank-Wolfe with LP oracle for global direction finding."""

def fast_convolve(a):

    n = len(a)

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(np.asarray(v, dtype=np.float64))[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(np.asarray(v, dtype=np.float64) - theta, 0.0)
