    """Equalize the top peaks using a direct linear system approach."""
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(best_f)
        M = conv.max()
        
        # Find top peaks
        k = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        peak_vals = conv[peaks]
        
        # Only consider peaks close to max
        threshold = M - 0.01 * M  # within 1% of max
        active = peaks[peak_vals >= threshold]
        if len(active) < 2:
            break
        
        # For each active peak k, the constraint is:
        # sum_{j} a_j * a_{k-j} = M
        # Gradient w.r.t. a: 2 * a_{k-j} for valid j
        # 
        # We want to find a direction d such that:
        # sum_j d_j * a_{k_i - j} = 0 for all active peaks k_i
        # (i.e., the direction that keeps all active peaks constant)
        # And d should be in the null space of the sum constraint: sum_j d_j = 0
        # 
        # Then we look for a direction that decreases M while keeping peaks equal.
        
        # Actually, a simpler approach: use the subgradient of the max.
        # The subgradient at the current point is the convex combination of gradients
        # of the active peaks. We want to move in the direction that reduces all of them.
        
        # Subgradient approach with exact active set:
        # grad = sum_i w_i * 2 * a[active_i - j] (for valid j)
        # where w_i are weights (equal for simplicity)
        
        active_set = active
        sg = np.zeros(n)
        for k_idx in active_set:
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * f[k_idx - idx]
        sg /= len(active_set)
        
        # Project negative gradient onto tangent space of simplex
        # Tangent: sum(d_j) = 0
        # Projected: d = -sg + mean(sg)
        d = -sg + np.mean(sg)
        
        # Line search
        best_alpha = 0
        best_val = M
        for alpha in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            ft = f + alpha * d
            ft = np.maximum(ft, 0)
            s = ft.sum()

                continue
            ft = ft / s
            val = fast_convolve(ft).max()
            if val < best_val - 1e-12:
                best_val = val
                best_alpha = alpha

            f = f + best_alpha * d
            f = np.maximum(f, 0)
            f = f / f.sum()
            new_M = fast_convolve(f).max()
            if new_M < best_obj - 1e-14:
                best_obj = new_M
                best_f = f.copy()
        else:
            break

import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lst_obj_grad(a, t):
    """Compute log-sum-exp objective and its gradient."""
    conv = fast_convolve(a)
    mx = conv.max()
    exp_t = np.exp(t * (conv - mx))
    Z = exp_t.sum()
    obj = mx + np.log(Z) / t
    p = exp_t / Z  # softmax weights
    
    # Gradient: grad_j = 2 * sum_m a[m] * p[j+m] = 2 * correlate(p, a, 'valid')[j]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return obj, grad

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with LP oracle (via cvxpy for speed)
2. Log-sum-exp FISTA for smooth optimization
3. Active-set peak equalization for final precision
4. Multiple restarts with diverse initializations

import numpy as np
import time

# Try to import cvxpy
try:
    import cvxpy as cp
    HAS_CVXPY = True
except:
    HAS_CVXPY = False

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_cvxpy(f, M):
    """Solve LP using cvxpy: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint: for each k=0,...,2n-2, sum_j f[k-j]*g[j] <= M
    # Use a sparse matrix for the Toeplitz structure
    # T[k,j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Actually, let's use cp expressions directly
    # conv(f, g)[k] = sum_j f[max(0, k-j):min(n, k-j+n)] ... this is messy
    # Let me use the matrix form
    
    # Build T as a sparse matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    T = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    constraints = [T @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    # Try different solvers
    for solver in [cp.MOSEK, cp.GUROBI, cp.CBC, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return g.value
        except:
            continue
    
    return None

def solve_lp_scipy(f, M):
    """Fallback LP solver using scipy."""
    from scipy.optimize import linprog
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if use_cvxpy:
            g = solve_lp_cvxpy(f, M * m_exp)
        else:
            g = solve_lp_scipy(f, M * m_exp)
        
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 20:
                break

        # Line search for step size
        best_alpha = 0.0
        best_val = M

        for alpha in np.arange(0.05, 1.01, 0.05):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:

                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)

4. Adaptive restarts with diverse perturbations
Key innovations: log-sum-exp continuation for smooth landscape navigation,

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_toeplitz_row(f, k):
    """Get row k of the Toeplitz matrix T(f) where T(f)@g = conv(f,g)."""
    n = len(f)
    row = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        row[j] = f[k - j]
    return row


def solve_lp_oracle(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
    if res.success:
        return res.x
    return None


def solve_lp_restricted(f, M, active_peaks):
    """Solve LP with only active peak constraints (faster)."""
    n = len(f)

            A_ub[i, j] = f[k - j]
    b_ub = np.full(len(active_peaks), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        # Verify all constraints are satisfied
        conv_fg = fast_convolve(f)  # Actually need conv(f, g)
        # Compute conv(f, g) properly
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        G = np.fft.rfft(g, fft_size)
        conv_fg = np.fft.irfft(F * G, fft_size)[:2 * n - 1]
        if conv_fg.max() <= M * (1 + 1e-8):

def lst_objective_and_grad(a, t):
    """Compute log-sum-exp smoothed objective and gradient."""
    conv = fast_convolve(a)
    mx = conv.max()
    exp_t = np.exp(t * (conv - mx))
    Z = exp_t.sum()
    obj = mx + np.log(Z) / t
    p = exp_t / Z
    # grad_j = 2 * sum_m a[m] * p[j+m] = 2 * correlate(p, a, 'valid')[j]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return obj, grad

    t0 = time.monotonic()
    it = 0
    last_obj = lst_objective_and_grad(x, t)[0]

    while time.monotonic() - t0 < time_limit:
        it += 1

        beta = (t_fista - 1) / (t_fista + 2)
        z = x + beta * (x - y)

        obj_z, grad_z = lst_objective_and_grad(z, t)
        x_new = project_simplex(z - lr * grad_z)

        # Evaluate true objective
        conv_new = fast_convolve(x_new)
        exact_new = conv_new.max()

        if exact_new < best_obj - 1e-15:
            best_obj = exact_new
            best_x = x_new.copy()

        # Backtracking: check if step was too large
        obj_new = lst_objective_and_grad(x_new, t)[0]
        if obj_new > obj_z + 0.5 * lr * np.dot(grad_z, x_new - z):
            lr = max(lr * 0.5, lr_min)
        elif it % 50 == 0 and obj_new < last_obj:
            lr = min(lr * 1.1, lr_max)

        last_obj = obj_new
        y = x
        x = x_new
        t_fista += 1

        if verbose and it % 200 == 0:
            print(f"    FISTA it={it}, exact={exact_new:.10f}, best={best_obj:.10f}")

    return best_x, best_obj


def active_peak_polish(f0, time_limit, verbose=False):
    """
    Final polishing using exact active set identification and
    very small subgradient steps with adaptive step size.
    """
    n = len(f0)
    f = f0.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    lr = 0.001
    momentum = np.zeros(n)
    mom_coeff = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Identify active peaks with tight tolerance
        tol = max(1e-6, 0.001 * M)
        active = np.where(conv >= M - tol)[0]

        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient (average over active peaks)
        sg = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * f[k - idx]
        sg /= len(active)

        # Momentum update
        momentum = mom_coeff * momentum - lr * sg
        f_new = project_simplex(f + momentum)

        # Evaluate
        conv_new = fast_convolve(f_new)
        exact_new = conv_new.max()

        if exact_new < best_obj - 1e-16:
            best_obj = exact_new
            best_f = f_new.copy()
            f = f_new
            # Slightly increase lr if making progress
            lr = min(lr * 1.01, 0.005)
        else:
            f = f_new  # Still move, but track best separately
            lr = max(lr * 0.95, 1e-5)

        # Periodic restart from best
        if it % 500 == 0:
            f = best_f.copy()
            momentum = np.zeros(n)

        if verbose and it % 500 == 0:
            print(f"    Active polish it={it}, exact={exact_new:.12f}, best={best_obj:.12f}")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal value
    that minimizes the max convolution, using ternary search.
    """
    n = len(f0)
    f = f0.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Shuffle coordinates for unbiased updates
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Try perturbation in coordinate i
            # Move mass from i to a random other coordinate
            j = rng.integers(0, n)
            if j == i:
                continue

            base_val = best_obj
            found_improvement = False

            for delta in [-0.001, -0.0005, -0.0002, 0.0002, 0.0005, 0.001,
                          0.002, -0.002, 0.005, -0.005]:
                f_trial = f.copy()
                f_trial[i] -= delta
                f_trial[j] += delta
                if f_trial[i] < 0 or f_trial[j] < 0:
                    continue
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial = f_trial / st
                val = fast_convolve(f_trial).max()
                if val < base_val - 1e-14:
                    f = f_trial
                    base_val = val
                    found_improvement = True

            if found_improvement and base_val < best_obj - 1e-15:
                best_obj = base_val
                best_f = f.copy()

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Two-hump
    g2 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Blend parent with uniform
        blend = 0.5 * (px / ps) + 0.5 * np.ones(n) / n
        inits.append(blend / blend.sum())

    return inits


def safe_evaluate(a):
    """Wrapper around evaluate_sequence with safety checks."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
        return score if np.isfinite(score) else float('inf')
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick FW-LP exploration (~300s)
    Phase 2: L-BFGS with log-sum-exp continuation (~400s)
    Phase 3: Deep FW-LP refinement (~500s)
    Phase 4: Active-set polish with multiple restarts (~300s)
    Phase 5: Final micro-polish (~170s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = safe_evaluate(a_norm)
        if score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            pscore = safe_evaluate(parent)
            print(f"Parent: n={len(parent)}, score={pscore:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"No parent: {e}")
        parent = np.ones(384) / 384

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform_default")

    n_p = len(parent)

    # === Phase 1: Quick FW-LP Exploration (~300s) ===
    print("\n=== Phase 1: FW-LP Exploration ===")
    t1_budget = min(300, deadline - time.monotonic() - 1300)
    if t1_budget > 30:
        a_opt, _ = fw_lp_optimize(parent, time_limit=t1_budget, verbose=True)
        try_seq(a_opt, "fw_phase1")

        # Try alternate initializations briefly
        rem = deadline - time.monotonic() - 1250
        if rem > 20:
            inits = generate_inits(n_p, parent, rng)
            for ii in range(2, min(5, len(inits))):
                rem = deadline - time.monotonic() - 1200
                if rem < 15:
                    break
                a_opt2, _ = fw_lp_optimize(inits[ii], time_limit=min(rem, 50))
                try_seq(a_opt2, f"fw_init_{ii}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: L-BFGS with Continuation (~400s) ===
    print("\n=== Phase 2: L-BFGS Continuation ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 900
        if rem > 30:
            t2_budget = min(rem, 400)
            t_values = [10, 30, 100, 300]
            a_opt, _ = lbfgs_lst(best_seq, t_values, time_limit=t2_budget, verbose=True)
            try_seq(a_opt, "lbfgs_cont")

            # Also try FISTA from the L-BFGS result
            rem = deadline - time.monotonic() - 800
            if rem > 20:
                a_opt2, _ = fista_lst(a_opt, t=200, time_limit=min(rem, 100))
                try_seq(a_opt2, "fista_after_lbfgs")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep FW-LP (~500s) ===
    print("\n=== Phase 3: Deep FW-LP ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t3_budget = min(rem, 500)
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t3_budget, verbose=True)
            try_seq(a_opt, "fw_deep")

            # Perturbed restart
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                noise = rng.standard_normal(len(best_seq)) * 0.003
                f_pert = project_simplex(best_seq + noise)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=min(rem, 150))
                try_seq(a_opt2, "fw_perturbed")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Active-Set Polish (~300s) ===
    print("\n=== Phase 4: Active-Set Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple passes with different tolerances
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t4_budget = min(rem, 250)
            a_opt, _ = active_peak_polish(best_seq, time_limit=t4_budget, verbose=True)
            try_seq(a_opt, "active_polish")

            # Coordinate descent
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                a_opt2, _ = coordinate_descent_polish(a_opt, time_limit=min(rem, 80))
                try_seq(a_opt2, "coord_descent")

            # Another active polish from best
            rem = deadline - time.monotonic() - 100
            if rem > 15:
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt3, _ = active_peak_polish(f_pert, time_limit=min(rem, 80))
                try_seq(a_opt3, "active_polish_2")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final Micro-Polish (~170s) ===
    print("\n=== Phase 5: Final Micro-Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine FISTA
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            a_opt, _ = fista_lst(best_seq, t=500, time_limit=min(rem, 60))
            try_seq(a_opt, "fine_fista")

        # Very fine active polish
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt2, _ = active_peak_polish(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt2, "fine_active")

        # Tiny random perturbations + quick polish
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            for trial in range(5):
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt3, _ = active_peak_polish(f_pert, time_limit=min(rem, 10))
                try_seq(a_opt3, f"micro_{trial}")

        # One more L-BFGS with very high t
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt4, _ = lbfgs_lst(best_seq, [500, 2000], time_limit=min(rem, 20))
            try_seq(a_opt4, "final_lbfgs")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = safe_evaluate(best_seq)

    # Final verification
    fs = safe_evaluate(best_seq)
    if fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
