    k_start = i
    k_end = min(i + n - 1, 2*n - 2)
    for k in range(k_start, k_end + 1):
        if k == 2 * i:

        else:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent
with direct convolution updates, combined with Frank-Wolfe LP oracle
for global moves and multi-start exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M, time_limit=30):
    """LP oracle: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def coord_descent(a, time_limit=60.0, n_x=40, max_sweeps=100000, restart_interval=2000):
    """
    Fast coordinate descent with direct O(n) convolution updates.
    Works on normalized sequences (sum=1).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    C = fast_convolve(a).copy()
    full_len = 2 * n - 1
    M = float(C.max())
    t0 = time.monotonic()
    
    if M < 1e-15:
        return a, M
    
    best_a = a.copy()
    best_M = M
    
    # Precompute prefix/suffix max
    def compute_aux(C):
        pref = np.full(full_len + 1, -np.inf)
        for k in range(full_len):
            pref[k+1] = max(pref[k], C[k])
        suff = np.full(full_len + 1, -np.inf)
        for k in range(full_len - 1, -1, -1):
            suff[k] = max(suff[k+1], C[k])
        return pref, suff
    
    pref, suff = compute_aux(C)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep_improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break
            v = a[i]
            
            # Generate candidates
            if v > 1e-8:
                xs = np.linspace(max(0, v * 0.05), v * 8.0, n_x)
            else:
                xs = np.linspace(0, 10.0 / n, n_x)
            xs = np.append(xs, v)
            xs = np.unique(xs)
            
            # Vectorized evaluation
            diff = xs - v
            new_S = 1.0 + diff
            valid = new_S > 1e-10
            
            # Affected: shape (len(xs), n)
            aff = C[i:i+n][None, :] + 2.0 * diff[:, None] * a[None, :]
            aff[:, i] = C[2*i] - v*v + xs*xs
            
            m_aff = aff.max(axis=1)
            m_unch = max(pref[i], suff[i+n]) if i+n < full_len else pref[i]
            # Wait: suff[i+n] is max(C[i+n], ..., C[full_len-1]). If i+n >= full_len, it's -inf.
            if i + n >= full_len:
                m_unch = pref[i]
            else:
                m_unch = max(pref[i], suff[i+n])
            if m_unch == -np.inf:
                m_unch = 0.0
            
            m_tot = np.maximum(m_unch, m_aff)
            obj = m_tot / (new_S * new_S)
            obj[~valid] = np.inf
            
            best_t = int(obj.argmin())
            best_x = float(xs[best_t])
            
            if best_x != v and obj[best_t] < M - 1e-13:
                a[i] = best_x
                d = best_x - v
                # Update C
                C[i:i+n] += 2.0 * d * a
                C[2*i] -= d * d
                # Renormalize
                S = a.sum()
                a = a / S
                C = C / (S * S)
                M = float(C.max())
                pref, suff = compute_aux(C)
                sweep_improved = True
                no_improve = 0
            
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
        
        if not sweep_improved:
            no_improve += 1
            if no_improve > 5:
                # Try a random restart
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                C = fast_convolve(a)
                M = float(C.max())
                pref, suff = compute_aux(C)
                no_improve = 0
                
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
    
    return best_a, best_M


def fw_step(a, M, time_limit=20):
    """Single Frank-Wolfe step with LP oracle."""
    n = len(a)
    g = solve_lp_oracle(a, M, time_limit=time_limit)
    if g is None:
        return None, None
    gs = g.sum()
    if gs < 1e-15:
        return None, None
    g = g / gs
    
    # Line search
    best_alpha = 0.0
    best_val = M
    alphas = np.linspace(0, 1, 20)
    for alpha in alphas:
        ft = (1 - alpha) * a + alpha * g
        ft = np.maximum(ft, 0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha < 1e-6:
        return None, None
    
    ft = (1 - best_alpha) * a + best_alpha * g
    ft = np.maximum(ft, 0)
    ft = ft / ft.sum()
    return ft, best_val


def generate_inits(n, rng, parent=None):

    inits.append(r / r.sum())

    for _ in range(3):
        r = rng.dirichlet(np.ones(n) * 0.5)
        inits.append(r)
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            for t in [0.3, 0.5, 0.7]:
                blended = t * (px/ps) + (1-t) * (np.ones(n)/n)
                inits.append(blended / blended.sum())
    
    return inits
