from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(n):
    """Build matrix C such that C @ g = convolve(a, g) where a is fixed.
    Returns the matrix for a general a (to be filled in)."""
    # Actually, let's build it properly

    # So C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = np.arange(n)  # placeholder, will be overwritten
    return C


def build_conv_matrix_for_a(a):
    """Build matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C


def objective_value(a):

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def adaptive_lp_direction(a, M, tighten=1.0, relax=1.0):

def line_search(a, d, initial_obj, max_iter=20):
    """

    f1 = objective_value(a1)

def lp_optimize_adaptive(a0, time_limit, verbose=False):
    """
    LP-based optimization with adaptive constraint tightness.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M / (s * s)
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 100
    perturb_count = 0
    tighten = 1.0  # Start with M * 1.0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        g, success = adaptive_lp_direction(a, M, tighten=tighten)
        
        if not success or g is None:
            # Try relaxing
            tighten = min(tighten * 1.5, 2.0)
            no_improve += 1
            continue
        
        g_sum = g.sum()
        if g_sum <= s * (1 + 1e-8):
            # No improvement possible in sum
            # Try tightening
            tighten = max(tighten * 0.9, 0.7)
            no_improve += 1
            continue
        
        # Compute direction
        d = (g / g_sum) * s - a  # Move towards normalized g, keeping sum
        
        # Actually, let's think about this differently.
        # We want to increase the sum while controlling the peak.
        # The LP gives us g with sum(g) > sum(a) and conv(a,g) <= M*tighten.
        # If we move a towards g, the sum increases.
        # But we need to check that the actual peak doesn't increase too much.
        
        # Let's use line search
        a_new, f_new = line_search(a, g - a, best_obj)
        
        # Normalize for comparison
        s_new = a_new.sum()
        if s_new > 1e-15:
            obj_new = 2.0 * n * fast_convolve(a_new).max() / (s_new * s_new)
        else:
            obj_new = best_obj
        
        if obj_new < best_obj - 1e-12:
            a = a_new
            conv = fast_convolve(a)
            M = conv.max()
            s = a.sum()
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            # Reward: relax the constraint slightly
            tighten = min(tighten * 1.01, 1.1)
        else:
            no_improve += 1
            # Penalty: tighten the constraint
            tighten = max(tighten * 0.95, 0.5)
        
        if verbose and no_improve % 50 == 0:
            print(f"    LP adapt: obj={best_obj:.8f}, tighten={tighten:.3f}, t={time.monotonic()-t0:.1f}s")
        
        if no_improve >= max_no_improve:
            # Perturb
            scale = 0.003 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            s = a.sum()
            tighten = 1.0
            no_improve = 0
            max_no_improve = min(300, 100 + 30 * perturb_count)
            perturb_count += 1
            
            if perturb_count > 50:
                break
    
    return best_a, best_obj


def gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=30, temp_init=0.002):
    """Nesterov-accelerated gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        conv_y = fast_convolve(y)
        
        # Compute gradient with soft-max weighting
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * y[k - idxs]
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.999
        temp *= 1.005
        mom = min(0.97, mom + 0.0005)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine-based
    for freq in [1, 2, 3]:
        s = (np.sin(freq * np.pi * x / n) ** 2 + 0.01)
        inits.append(s / s.sum())
    
    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(c / c.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Bimodal
    m = max(1, n // 4)
    bi = np.zeros(n)
    bi[:m] = 1.0
    bi[-m:] = 1.0
    inits.append(bi / bi.sum())
    
    # Frequency domain
    inits.append(freq_domain_init(n, rng))
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.1))
    
    # Power law
    pw = (x + 1.0) ** 0.5
    inits.append(pw / pw.sum())
    
    # "Comb" pattern
    comb = np.ones(n)
    comb[n//3::2] = 0.3
    inits.append(comb / comb.sum())
    
    return inits


def peak_aware_refine(a0, time_limit):
    """
    Refine by focusing on the top peaks. At each step, identify the peak,
    then find the coordinate that most reduces it when decreased.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find current peak
        peak_idx = int(np.argmax(conv))
        M = conv[peak_idx]
        
        # Find which coordinates contribute most to this peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j]
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        contributions = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contributions[j] = a[j] * a[peak_idx - j]
        
        # Try decreasing the coordinate with highest contribution
        # (and increase another coordinate to maintain sum)
        top_contrib = np.argsort(contributions)[::-1][:5]
        candidates_decrease = [j for j in top_contrib if a[j] > 0.01 * a.mean()]
        
        if not candidates_decrease:
            break
        
        improved = False
        for j_dec in candidates_decrease:
            # Try decreasing a[j_dec] and spreading to others
            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = a[j_dec] * frac
                if delta < 1e-10:
                    continue
                
                a_try = a.copy()
                a_try[j_dec] -= delta
                # Spread to positions that don't contribute to the peak
                other_positions = [j for j in range(n) if j != j_dec and 
                                   (j < j_start or j > j_end or peak_idx - j < 0 or peak_idx - j >= n)]
                if not other_positions:
                    other_positions = [j for j in range(n) if j != j_dec]
                
                # Spread proportionally to current values
                weights = np.array([a_try[j] + 0.001 for j in other_positions])
                weights /= weights.sum()
                for idx, j in enumerate(other_positions):
                    a_try[j] += delta * weights[idx]
                
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
    
    return best_a, best_obj
