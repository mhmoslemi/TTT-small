def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
(1) Adaptive subgradient descent with momentum and multi-peak softmax weighting
(2) Efficient coordinate descent with O(n) per-coordinate 1D optimization
(3) Linearized LP Frank-Wolfe for large jumps
(4) Multi-restart with diverse initializations and symmetry exploration
Key innovation: fast coordinate descent exploiting the structure of autocorrelation
to do O(n) work per coordinate, enabling thousands of passes within time budget.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp_reduced(f, conv, top_k=25):
    """Solve reduced linearized LP using only top_k peak lags."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    peaks = np.sort(peaks)

    c = np.zeros(n + 1)
    c[0] = 1.0

    B_full = build_conv_matrix(f)
    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B_full[peaks, :]
    b_ub = -conv[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def subgradient_optimize(a0, time_limit, lr=0.008, momentum=0.92, top_k=30, temp=0.0008):
    """Subgradient descent with softmax-weighted multi-peak targeting."""
    f = np.maximum(a0.astype(np.float64), 0.0)

    n = len(f)

    v = np.zeros(n)

    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):

        else:

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:

        sg = np.zeros(n)

            k = int(peaks[i])

        f = project_simplex(f + v)
        conv = fast_convolve(f)

            best_f = f.copy()
            no_improve = 0

            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.001)

            no_improve = 0
