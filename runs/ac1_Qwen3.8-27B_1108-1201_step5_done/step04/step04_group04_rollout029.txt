import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    A = np.fft.rfft(a, fft_size)

    if not A_ub_rows:
        return None
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'pdlp': {'max_iterations': 200}})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_full(f, M):
    """Solve full LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, use_reduced=True, top_k=30, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    reduced = use_reduced

    while time.monotonic() - t0 < time_limit:
        it += 1

        if reduced:
            g = solve_lp_reduced(f, M * m_exp, top_k=top_k)
        else:
            g = solve_lp_full(f, M * m_exp)
        
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if no_improve > 25:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.003 + 0.002 * min(no_improve // 25, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def adam_smoothed_simplex(f0, time_limit, T_init=0.002, T_min=0.00005,
                          lr_init=0.008, verbose=False):
    """Adam-based gradient descent on simplex with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_step = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_step += 1

        # Compute smoothed gradient
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = w / ws

        # grad[j] = sum_k p[k] * (a[k-j] + a[j]) = r[j] + a[j]
        r = cross_corr_p_a(p, f)
        grad = r + f
        grad = grad - np.mean(grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr = max(lr * 0.85, 0.0001)

        T = max(T * 0.998, T_min)

        if no_improve > 200:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(f)
            lr = max(lr * 1.5, 0.003)
            T = max(T_init * 0.3, T_min)
            m = np.zeros(n)
            v = np.zeros(n)
            t_step = 0
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit, n_sweeps=50, verbose=False):
    """Coordinate descent: optimize one coordinate at a time via 1D search."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    coord_order = np.arange(n)
    improved = True
    total_iter = 0

    while time.monotonic() - t0 < time_limit and improved:
        improved = False
        rng.shuffle(coord_order)

        for j in coord_order:
            if time.monotonic() - t0 > time_limit:
                break
            total_iter += 1

            a_j = a[j]
            if a_j < 1e-12:
                continue

            # Try several multipliers for a[j]
            best_delta = 0.0
            best_local_obj = conv.max()
            S_old = a.sum()

            # Candidate deltas: scale a[j] by factors
            factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.1, 3.0]
            for fac in factors:
                delta = a_j * (fac - 1.0)
                if delta <= -a_j * 0.999:
                    continue
                # Update convolution
                # c[k] changes by: 2*delta*a[k-j] + delta^2*I(k=2j)
                new_conv = conv.copy()
                for k in range(2 * n - 1):
                    idx = k - j
                    if 0 <= idx < n:
                        new_conv[k] += 2.0 * delta * a[idx]
                    if k == 2 * j:
                        new_conv[k] += delta * delta
                # Evaluate: 2*n*max(new_conv)/(S_old+delta)^2
                new_S = S_old + delta
                if new_S < 1e-10:
                    continue
                obj = 2.0 * n * new_conv.max() / (new_S * new_S)
                if obj < best_local_obj - 1e-12:
                    best_local_obj = obj
                    best_delta = delta

            if best_delta != 0.0:
                a[j] += best_delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
            else:
                conv = fast_convolve(a)

        if verbose and total_iter % 1000 == 0:
            print(f"    CD iter {total_iter}: obj={best_obj:.10f}")

    return best_a, best_obj


def coord_descent_fast(f0, time_limit, verbose=False):
    """Faster coordinate descent using analytical 1D minimization."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    S = a.sum()
    best_obj = 2.0 * n * conv.max() / (S * S)
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        coord_order = np.arange(n)
        rng.shuffle(coord_order)
        any_improved = False

        for j in coord_order:
            if time.monotonic() - t0 > time_limit:
                break

            a_j = a[j]
            if a_j < 1e-10:
                continue

            # The objective as function of a_new = a_j * t (scale factor t):
            # c_new[k] = c[k] + (t-1)*a_j*(a[k-j]+a[j]) + (t-1)^2*a_j^2*I(k=2j)
            # S_new = S + (t-1)*a_j
            # obj(t) = 2*n*max(c_new) / S_new^2
            # For small range of t, the max is achieved at the same k
            # Find the current argmax
            k_max = np.argmax(conv)
            
            # Compute coefficients for the active constraint k_max
            # c_new[k_max] = c[k_max] + (t-1)*a_j*(a[k_max-j] + a[j]) + (t-1)^2*a_j^2*I(k_max=2j)
            ikj = k_max - j
            if 0 <= ikj < n:
                B = a_j * (a[ikj] + a[j])
            else:
                B = a_j * a[j]
            C = a_j * a_j if k_max == 2 * j else 0.0
            A = conv[k_max]
            
            S_j = a_j  # (t-1)*a_j contribution to sum
            # obj(t) = 2*n*(A + B*(t-1) + C*(t-1)^2) / (S + S_j*(t-1))^2
            # Let u = t-1, minimize: 2*n*(A + B*u + C*u^2) / (S + S_j*u)^2
            # subject to u > -1 (a_j > 0) and other constraints
            
            # Try a grid of u values
            best_u = 0.0
            best_u_obj = 2.0 * n * conv.max() / (S * S)
            
            # Grid centered around current best
            u_vals = np.concatenate([
                np.linspace(-0.8, 0.0, 20),
                np.linspace(0.01, 2.0, 30),
            ])
            
            for u in u_vals:
                if u <= -0.99:
                    continue
                # Quick check: just eval the active constraint
                c_active = A + B * u + C * u * u
                S_new = S + S_j * u
                if S_new < 1e-10:
                    continue
                obj_est = 2.0 * n * c_active / (S_new * S_new)
                if obj_est < best_u_obj - 1e-12:
                    best_u_obj = obj_est
                    best_u = u
            
            # Fine tune around best_u
            if best_u != 0.0:
                lo = best_u - 0.1
                hi = best_u + 0.1
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    c_active = A + B * mid + C * mid * mid
                    S_new = S + S_j * mid
                    if S_new < 1e-10:
                        break
                    obj_est = 2.0 * n * c_active / (S_new * S_new)
                    if obj_est < best_u_obj:
                        best_u_obj = obj_est
                        best_u = mid
                        lo = mid
                    else:
                        hi = mid
                
                # Full evaluation
                a[j] = a_j * (1.0 + best_u)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_convolve(a)
                S = a.sum()
                obj = 2.0 * n * conv.max() / (S * S)
                
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    any_improved = True
            else:
                conv = fast_convolve(a)
                S = a.sum()

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}")
        
        if not any_improved and sweep > 5:
            break

    return best_a, best_obj


def palindromic_project(a):
    """Project to palindromic subspace and re-normalize to simplex."""
    n = len(a)
    half = (a[:n//2] + a[n//2::-1]) / 2.0
    if n % 2 == 0:
        result = np.concatenate([half, half[::-1]])
    else:
        result = np.concatenate([half, [a[n//2]], half[::-1]])
    return project_simplex(result)


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # U-shape (edge-concentrated)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    two_hump = h1 + h2
    inits.append(two_hump / two_hump.sum())

    # Beta-like distributions
    for alpha in [0.5, 2.0, 5.0]:
        beta_dist = rng.beta(alpha, alpha, size=n)
        inits.append(beta_dist / beta_dist.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Palindromic version
        init_p = palindromic_project(px)
        inits.append(init_p)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return

        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        # Normalize to avoid clip issues
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Length Sweep (250s) ===
    print("=== Phase 1: Length Sweep ===")
    t_p1 = time.monotonic()
    p1_budget = 250.0

    base_n = len(parent) if parent is not None else 368

        lengths.insert(0, base_n)

            conv = fast_convolve(f_init)
            obj = conv.max()
            if obj < best_init_obj:

        if best_init is not None:

                                        use_reduced=True, top_k=20)

    print(f"\n  Phase 2 complete: best={best_score:.6f}")

    # === Phase 3: Fine Polish (350s) ===

                                      (0.0001, 0.0005), (0.00005, 0.0002)]):
            rem = p3_budget - (time.monotonic() - t_p3)
            if rem < 20:
                break
            t_alloc = min(60.0, rem - 10)
            print(f"  Fine Adam round {i} (T={T}, lr={lr}, t={t_alloc:.0f}s)")
            a_opt, _ = adam_smoothed_simplex(f_work, time_limit=t_alloc,
                                              T_init=T, T_min=T * 0.1,
                                              lr_init=lr)
            try_seq(a_opt, f"p3_adam_{i}")
            if best_seq is not None:
                f_work = best_seq.copy()

        # Coordinate descent polish
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            t_alloc = min(80.0, rem - 10)
            print(f"  Fine CD (t={t_alloc:.0f}s)")
            a_opt, _ = coord_descent_fast(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_cd")
            if best_seq is not None:
                f_work = best_seq.copy()

        # LP-FW final
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            t_alloc = min(60.0, rem - 10)
            print(f"  Final LP-FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc,
                                       use_reduced=True, top_k=40)
            try_seq(a_opt, "p3_fw")
            if best_seq is not None:
                f_work = best_seq.copy()

    # === Phase 4: Final Tricks (100s) ===
    print("\n=== Phase 4: Final Tricks ===")
    t_p4 = time.monotonic()
    p4_budget = 100.0

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq.copy()

        # Tiny perturbations + quick Adam
        for i in range(5):
            rem = p4_budget - (time.monotonic() - t_p4)
            if rem < 15:
                break
            noise = rng.standard_normal(n_best) * 0.0005
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = adam_smoothed_simplex(f_pert, time_limit=min(15, rem - 5),
                                              T_init=0.0002, T_min=0.00002,
                                              lr_init=0.001)
            try_seq(a_opt, f"p4_pert_{i}")
            if best_seq is not None:
                f_work = best_seq.copy()

        # Mirror and optimize
        rem = p4_budget - (time.monotonic() - t_p4)
        if rem > 20:
            mirrored = (f_work + f_work[::-1]) / 2.0
            mirrored = project_simplex(mirrored)
            if np.max(np.abs(mirrored - f_work)) > 1e-8:
                a_opt, _ = adam_smoothed_simplex(mirrored, time_limit=min(20, rem - 5),
                                                  T_init=0.0002, T_min=0.00002,
                                                  lr_init=0.001)
                try_seq(a_opt, "p4_mirror")

        # Try n+1 and n-1 with interpolation
        rem = p4_budget - (time.monotonic() - t_p4)
        for delta in [1, -1, 2, -2]:
            if rem < 10:
                break
            new_n = n_best + delta
            if new_n < 128 or new_n > 600:
                continue
            px = np.interp(np.linspace(0, 1, new_n),
                           np.linspace(0, 1, n_best), f_work)
            px = project_simplex(px)
            a_opt, _ = adam_smoothed_simplex(px, time_limit=min(15, rem - 5),
                                              T_init=0.0003, T_min=0.00003,
                                              lr_init=0.002)
            try_seq(a_opt, f"p4_n{new_n}")
            rem = p4_budget - (time.monotonic() - t_p4)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
