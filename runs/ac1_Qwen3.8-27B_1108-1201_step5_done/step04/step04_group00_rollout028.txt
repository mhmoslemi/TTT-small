The optimal value of this LP tells us the directional derivative of the objective at f in the direction of the simplex. If the optimal value is 1 (i.e., g = f/sum(f) is optimal), then we're at a stationary point of the FW method (though not necessarily of the original problem).

For the dual:
min M * sum(y) s.t. (A^T y)[j] ≥ 1 for all j, y ≥ 0

where (A^T y)[j] = sum_k y[k] * f[k-j] (with appropriate index bounds).

The optimal y gives us the "dual certificate" that tells us which constraints are binding.

def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, weights, n):
    """
    Compute the gradient of sum_k weights[k] * (f*f)[k] with respect to f.
    Uses FFT cross-correlation for efficiency.
    weights: array of length 2n-1, the weights for each convolution value.
    Returns: gradient array of length n.
    """
    # grad[i] = 2 * sum_k weights[k] * f[k-i] * 1[0 <= k-i < n]
    # This equals 2 * (weights ⋆ f)[i] for i=0..n-1
    # Cross-correlation via FFT:
    # (w ⋆ f)[m] = ifft(fft(w) * conj(fft(f)))[m]
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f, fft_size)
    corr = np.fft.irfft(W * np.conj(F), fft_size)
    return 2.0 * corr[:n]


def subgradient_optimize(f0, time_limit, top_k=50, lr=0.001, mom=0.9, 
                         temp=0.001, lp_every=200, verbose=False):
    """
    Momentum subgradient optimization for minimizing max(f*f) on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_lr = lr
    cur_mom = mom
    cur_temp = temp
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute convolution
        conv = fast_convolve(f)
        c_max = conv.max()
        
        # Check improvement
        if c_max < best_obj - 1e-14:
            best_obj = c_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute weights for active peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        # Softmax weights
        scaled = pv / cur_temp
        scaled -= scaled.max()
        exp_w = np.exp(scaled)
        w_sum = exp_w.sum()
        if w_sum < 1e-15:
            w_vec = np.zeros(len(peaks))
            w_vec[np.argmax(pv)] = 1.0
        else:
            w_vec = exp_w / w_sum
        
        # Build full weight vector
        weights = np.zeros(2 * n - 1)
        for i in range(len(peaks)):
            if w_vec[i] > 1e-15:
                weights[peaks[i]] = w_vec[i]
        
        # Compute gradient via FFT
        grad = compute_gradient(f, weights, n)
        
        # Update with momentum
        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        
        # Check if we made progress
        f = f_new
        
        # Adapt hyperparameters
        cur_lr *= 0.9998
        cur_temp *= 1.01
        
        # Adaptive top_k: increase as we converge
        cur_top_k = min(int(top_k * (1 + it / 5000)), len(conv))
        
        # Periodically solve LP for a correction
        if lp_every > 0 and it % lp_every == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem > 1:
                g = solve_lp_direction(f, c_max, n)
                if g is not None:
                    g_sum = g.sum()
                    if g_sum > 1e-15:
                        g = g / g_sum
                        # Blend with current f
                        alpha = 0.3
                        f_blend = (1 - alpha) * f + alpha * g
                        f_blend = np.maximum(f_blend, 0)
                        f_blend = f_blend / f_blend.sum()
                        conv_blend = fast_convolve(f_blend)
                        if conv_blend.max() < c_max:
                            f = f_blend
                            v = np.zeros(n)
        
        if verbose and it % 100 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, c_max={c_max:.10f}")
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses the dual formulation for potentially faster solving."""
    # Dual: min M * sum(y) s.t. sum_k y[k]*f[k-j] >= 1 for j=0..n-1, y >= 0
    # In standard form for linprog:
    # minimize: M * sum(y)
    # subject to: -(A^T) y <= -1, y >= 0
    # where (A^T)[j,k] = f[k-j] if 0<=k-j<n
    
    # Build the constraint matrix for the dual
    # (A^T)[j,k] = f[k-j] if 0 <= k-j < n, else 0
    # So row j has f[0], f[1], ..., f[n-1] starting at column j
    
    A_dual = np.zeros((n, 2 * n - 1))
    for j in range(n):
        A_dual[j, j:j+n] = f
    
    # Dual: min M * ones(2n-1) @ y
    # s.t. A_dual @ y >= ones(n)  =>  -A_dual @ y <= -ones(n)
    #      y >= 0
    
    c = M * np.ones(2 * n - 1)
    A_ub = -A_dual
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    
    if res.success:
        # Recover primal solution from dual
        # The primal g is obtained from complementary slackness
        # But it's easier to just solve the primal directly from the dual info
        # Actually, let's just return the dual solution and use it differently
        
        # For FW, we need the primal g. Let's solve the primal instead.
        pass
    
    # If dual is too slow, fall back to primal
    # Actually, let's solve the primal directly - it's often faster with HiGHS
    c_p = -np.ones(n)
    A_primal = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_primal[j:j+n, j] = f
    b_primal = np.full(2 * n - 1, M)
    bounds_p = [(0, None)] * n
    
    res_p = linprog(c_p, A_ub=A_primal, b_ub=b_primal, bounds=bounds_p, 
                    method='highs', options={'presolve': True, 'time_limit': 5})
    
    if res_p.success:
        return res_p.x
    
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
                # Try a perturbation
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Refine with bisection
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 13 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        n_improved_coords = 0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
                    n_improved_coords += 1
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_cosine_window(n):
    """Create a cosine window normalized to sum 1."""
    x = np.linspace(0, np.pi, n)
    w = np.cos(x) ** 2
    w = np.maximum(w, 0)
    ss = w.sum()
    if ss > 0:
        w /= ss
    return w


def make_gaussian_window(n, sigma_frac=0.3):
    """Create a Gaussian window normalized to sum 1."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    ss = w.sum()
    if ss > 1e-15:
        w /= ss
    return w


def make_raised_cosine(n, beta=0.5):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    w = 1 - beta * np.cos(2 * np.pi * x)
    w = np.maximum(w, 0)
    ss = w.sum()
    if ss > 1e-15:
        w /= ss
    return w
