    c_scaled = c / T

    w = np.exp(c_scaled)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using annealed smoothed mirror descent
with FFT-based gradient computation. Explores multiple sequence lengths for
better solutions. Uses log-sum-exp smoothing with geometric temperature decay,
multiplicative weight (mirror descent) updates, and final subgradient polish.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient_fft(a, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] using FFT.
    w is the weight vector of length 2n-1.
    """
    n = len(a)
    w_flip = w[::-1]
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w_flip, N)
    corr = np.fft.irfft(A * W, N)
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * corr[idx]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def annealed_mirror_descent(a0, time_limit, T_ratio_start=1.0, T_ratio_end=0.001,
                            eta0=0.5, n_decay_steps=None, bts_max_iter=20, verbose=True):
    """
    Annealed smoothed mirror descent on the simplex.
    
    Minimizes max_k conv(a,a)[k] using log-sum-exp smoothing with geometric
    temperature decay and multiplicative weight updates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    max_c = c.max()
    best_obj = max_c
    best_a = a.copy()
    
    T_start = max_c * T_ratio_start
    T_end = max_c * T_ratio_end
    if n_decay_steps is None:
        n_decay_steps = max(100, int(time_limit * 50))  # ~50 temperature changes per second
    
    # Geometric ratio for temperature decay
    if T_start > T_end and T_start > 0:
        T_ratio = (T_end / T_start) ** (1.0 / n_decay_steps)
    else:
        T_ratio = 1.0
    
    T = T_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        max_c = c.max()
        
        # Update temperature (geometric decay)
        T = max(T * T_ratio, T_end)
        
        # Compute softmax weights
        c_scaled = c / T
        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient via FFT
        grad = compute_gradient_fft(a, w)
        
        # Mirror descent with backtracking line search
        eta = eta0
        improved = False
        for _ in range(bts_max_iter):
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 1e-30)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-15:
                eta *= 0.5
                continue
            a_new = a_new / a_new_sum
            c_new = fast_convolve(a_new)
            max_c_new = c_new.max()
            if max_c_new < max_c - 1e-15:
                a = a_new
                max_c = max_c_new
                improved = True
                break
            eta *= 0.5
        
        if improved:
            if max_c < best_obj - 1e-14:
                best_obj = max_c
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 7919 + 13)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            c = fast_convolve(a)
            max_c = c.max()
            T = max_c * 0.1  # reset temperature
            no_improve = 0
            eta0 = max(eta0 * 0.8, 0.05)
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AMD it={it}, obj={max_c:.8f}, T={T:.6f}, eta={eta:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit, top_k=20, eta0=0.002, verbose=True):
    """
    Final polish using exact subgradient of max(conv).
    Uses multiplicative weight update with the subgradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    eta = eta0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        max_c = c.max()
        
        # Find top-k peaks
        if top_k >= 2 * n - 1:
            peaks = np.arange(2 * n - 1)
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]
        
        # Weight: softmax with very low temperature (focus on peaks)
        pv = c[peaks]
        T = (pv.max() - pv.min()) / max(1, len(peaks)) * 0.1
        if T < 1e-12:
            T = 1e-12
        scaled = pv / T
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()
        
        # Build full weight vector
        w = np.zeros(2 * n - 1)
        w[peaks] = w_peaks
        
        # Compute gradient
        grad = compute_gradient_fft(a, w)
        
        # Multiplicative weight update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        
        c_new = fast_convolve(a_new)
        max_c_new = c_new.max()
        
        if max_c_new < best_obj - 1e-15:
            best_obj = max_c_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new  # still update, but track best separately
            no_improve += 1
        
        eta *= 0.9999
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 104729 + 7)
            noise_scale = 0.001 + 0.002 * min(1.0, no_improve / 600.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            c = fast_convolve(a)
            no_improve = 0
            eta = eta0 * 0.8
        
        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, obj={max_c:.8f}, eta={eta:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return a_new


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Cosine window
    x = np.arange(n, dtype=np.float64)
    cos_w = np.cos(np.pi * x / (2 * n)) ** 2 + 0.01
    inits.append(cos_w / cos_w.sum())
    
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
