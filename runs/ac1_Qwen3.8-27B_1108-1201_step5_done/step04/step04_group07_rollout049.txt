def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def kkt_active_set_optimize(p0, max_iters=500, tol=1e-10, top_k=20):
    """
    Solve KKT conditions for min max_k conv(p,p)[k] s.t. sum(p)=1, p>=0.
    At optimum: for j in support, sum_{a in active} 2*p[a-j] = lambda,
    and sum(p_support) = 1.
    """
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)

    for it in range(max_iters):
        conv = fast_convolve(p)
        t_star = conv.max()

        # Active set: lags within relative tolerance of max
        thresh = t_star - max(tol * t_star, 1e-14)
        A = np.where(conv >= thresh)[0]

        # Support: positive components
        S = np.where(p > 1e-10)[0]

        if len(A) == 0 or len(S) == 0:
            return p, t_star

        # Build KKT matrix: M[i,j] = 2 * |{a in A : a = S[i] + S[j]}|
        m = len(S)
        # Precompute active set as a set for fast lookup
        active_set = set(A.tolist())
        S_list = S.tolist()

        M = np.zeros((m, m))
        for i in range(m):
            for j in range(m):
                if (S_list[i] + S_list[j]) in active_set:
                    M[i][j] = 2.0

        # System: M @ p_S = lambda * 1, sum(p_S) = 1
        # [M  -1] [p_S]   [0]
        # [1...1 0] [lam] = [1]
        A_sys = np.zeros((m + 1, m + 1))
        A_sys[:m, :m] = M
        A_sys[:m, m] = -1.0
        A_sys[m, :m] = 1.0
        b_sys = np.array([0.0] * m + [1.0])

        # Solve with regularization if singular
        try:
            sol = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:
            # Add small regularization
            A_reg = A_sys.copy()
            A_reg[:m, :m] += 1e-8 * np.eye(m)
            try:
                sol = np.linalg.solve(A_reg, b_sys)
            except np.linalg.LinAlgError:
                return p, t_star

        p_new_S = sol[:m]
        lam = sol[m]

        # Check validity
        if np.all(p_new_S >= -1e-10) and lam >= -1e-10:
            p_new = np.zeros(n)
            p_new[S] = np.maximum(p_new_S, 0.0)
            ps = p_new.sum()
            if ps > 1e-12:
                p_new = p_new / ps
                conv_new = fast_convolve(p_new)
                obj_new = conv_new.max()

                if obj_new <= t_star + 1e-12:
                    # Check convergence
                    if obj_new < t_star - 1e-12:
                        p = p_new
                        continue
                    elif np.max(np.abs(p_new - p)) < 1e-12:
                        return p_new, obj_new
                    else:
                        p = p_new
                        continue

        # If KKT solution fails, try reducing the active set
        # Use only top peaks
        if len(A) > top_k:
            top_indices = np.argpartition(conv, -top_k)[-top_k:]
            A = np.sort(top_indices)
            active_set = set(A.tolist())
            M = np.zeros((m, m))
            for i in range(m):
                for j in range(m):
                    if (S_list[i] + S_list[j]) in active_set:
                        M[i][j] = 2.0

            A_sys = np.zeros((m + 1, m + 1))
            A_sys[:m, :m] = M
            A_sys[:m, m] = -1.0
            A_sys[m, :m] = 1.0
            b_sys = np.array([0.0] * m + [1.0])

                    p_new[S] = np.maximum(p_new_S, 0.0)

                        p = p_new

def linearized_lp_direction(f, conv_f, top_k=40):
    """Solve reduced linearized LP for Frank-Wolfe direction."""
    n = len(f)
    size = 2 * n - 1

    # Use top_k constraints
    if top_k >= size:
        peaks = np.arange(size)
    else:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
        peaks = np.sort(peaks)

    # Build constraint rows: 2*B[g][peak] <= -conv_f[peak] + t
    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0

    n_peaks = len(peaks)
    A_ub = np.zeros((n_peaks, n + 1))
    A_ub[:, 0] = -1.0
    b_ub = np.zeros(n_peaks)

    for i, k in enumerate(peaks):
        b_ub[i] = -conv_f[k]
        # (B*g)[k] = sum_j f[k-j] * g[j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, 1 + j] = 2.0 * f[k - j]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        return res.x[1:]
    return None


def maxsum_lp_direction(f, M, top_k=40):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for top peaks, g >= 0."""
    n = len(f)
    size = 2 * n - 1

    if top_k >= size:
        peaks = np.arange(size)
    else:
        # Use all constraints for correctness (or top peaks)
        peaks = np.arange(size)

    c = -np.ones(n)
    A_ub = np.zeros((len(peaks), n))
    for i, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = f[k - j]
    b_ub = np.full(len(peaks), M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None

            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
            continue

        if use_linearized:
            target = f + g
            target = np.maximum(target, 0.0)
            ts = target.sum()
            if ts < 1e-15:
                no_improve += 1
                continue
            target = target / ts
        else:
            target = g

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * target
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine
        if best_alpha > 0:
            lo = max(0.02, best_alpha * 0.7)
            hi = min(1.0, best_alpha * 1.3)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * target
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs

            if best_val < best_obj - 1e-13:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            scale = 0.002 + 0.001 * (no_improve // 20)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0

    return best_f, best_obj


def subgrad_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=50, temp=0.0001):
    """Smoothed subgradient with softmax-weighted multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = (pv - pv.max()) / (cur_temp * max(pv.max(), 1e-10))
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.003

        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0005)
            no_improve = 0

    return best_f, best_obj


def binarize_and_optimize(p, n_threshold=0.5, time_limit=60):
    """Round p toward 0-1, then re-optimize."""
    n = len(p)
    # Sort and threshold
    sorted_p = np.sort(p)[::-1]
    # Find threshold where values drop off
    # Try several thresholds
    best_p = p.copy()
    best_obj = fast_convolve(p).max()

    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        k = int(n * frac)
        if k < 2:
            continue
        thresh = sorted_p[k - 1]
        b = np.where(p >= thresh * 0.8, 1.0, 0.0)
        bs = b.sum()
        if bs < 2:
            continue
        b = b / bs
        obj = fast_convolve(b).max()
        if obj < best_obj:
            best_obj = obj
            best_p = b.copy()

    # Polish the best binarized solution
    if time_limit > 5:
        a_opt, obj_opt = subgrad_polish(best_p, time_limit=min(time_limit, 40),
                                         lr=0.001, mom=0.95, top_k=30, temp=0.0001)
        if obj_opt < best_obj:
            best_p = a_opt
            best_obj = obj_opt

    return best_p, best_obj


def generate_structured_init(n, rng, seed_var=0):
    """Generate structured initializations inspired by number theory."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Periodic with random period
    for period in [3, 4, 5, 7, 8]:
        base = rng.dirichlet(np.ones(period))
        p = np.tile(base, n // period + 1)[:n]
        p = p / p.sum()
        inits.append(p)

    # Symmetric random
    half = n // 2
    h = rng.dirichlet(np.ones(half) * 0.5)
    p = np.concatenate([h, h[::-1]])[:n]
    p = p / p.sum()
    inits.append(p)

    # Cosine modulated
    x = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) + 0.1
        inits.append(c / c.sum())

    # Two-peak
    for center1, center2 in [(n // 4, 3 * n // 4), (n // 3, 2 * n // 3)]:
        g1 = np.exp(-0.5 * ((x - center1) / (n * 0.1)) ** 2)
        g2 = np.exp(-0.5 * ((x - center2) / (n * 0.1)) ** 2)
        p = g1 + g2
        inits.append(p / p.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer:
    1. Load parent, generate diverse inits
    2. Active-set KKT solver (new, fast convergence)
    3. LP-based Frank-Wolfe (robust)
    4. Subgradient polish
    5. Binarization + re-optimize
    6. Multi-length and multi-restart exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    n_best = len(parent) if parent is not None else 384

    if parent is not None:
        try_seq(parent, "parent")
    else:
        parent = np.ones(n_best) / n_best
        try_seq(parent, "uniform")

    # Try structured inits at the best length
    inits = generate_structured_init(n_best, rng)
    for i, ini in enumerate(inits[:6]):
        try_seq(ini, f"init_{i}")

    # Try a few nearby lengths
    for n_try in [n_best - 8, n_best + 8, n_best - 16, n_best + 16]:
        if n_try < 32:
            continue
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, len(parent)), parent)
        else:
            px = np.ones(n_try) / n_try
        px = np.maximum(px, 0)
        px = px / px.sum()
        try_seq(px, f"len{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: Active-Set KKT Solver (primary) ===
    print("\n=== Phase 2: Active-Set KKT Solver ===")
    n_cur = len(best_seq)
    remaining = deadline - time.monotonic() - 300
    t_kkt = min(remaining * 0.25, 300)

    if t_kkt > 10:
        print(f"  Running KKT solver (n={n_cur}), budget={t_kkt:.0f}s")
        a_opt, obj_opt = kkt_active_set_optimize(best_seq, max_iters=200, tol=1e-9)
        try_seq(a_opt, "kkt_1")

        # Run KKT from multiple starting points
        rem = deadline - time.monotonic() - 250
        for restart in range(3):
            if rem < 15:
                break
            noise = rng.standard_normal(n_cur) * 0.01 * (restart + 1)
            f_start = project_simplex(best_seq + noise)
            t_each = min(rem * 0.3, 60)
            a_opt, obj_opt = kkt_active_set_optimize(f_start, max_iters=150, tol=1e-9)
            try_seq(a_opt, f"kkt_r{restart}")
            rem = deadline - time.monotonic() - 200

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: LP-based Frank-Wolfe ===
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 250
    t_fw = min(remaining * 0.35, 500)

    if t_fw > 30:
        n_cur = len(best_seq)
        print(f"  Running linearized FW (n={n_cur}), budget={t_fw:.0f}s")
        a_opt, _ = fw_with_lp(best_seq, time_limit=min(t_fw * 0.5, 250),
                              use_linearized=True, top_k=30, verbose=True)
        try_seq(a_opt, "linfw")

        rem = deadline - time.monotonic() - 200
        if rem > 30:
            a_opt, _ = fw_with_lp(best_seq, time_limit=min(rem * 0.4, 200),
                                  use_linearized=False, top_k=30)
            try_seq(a_opt, "maxfw")

        # KKT + FW hybrid: run KKT then polish with FW
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            a_kkt, _ = kkt_active_set_optimize(best_seq, max_iters=100, tol=1e-9)
            a_opt, _ = fw_with_lp(a_kkt, time_limit=min(rem * 0.4, 100),
                                  use_linearized=True, top_k=25)
            try_seq(a_opt, "kkt+fw")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Subgradient Polish ===
    print("\n=== Phase 4: Subgradient Polish ===")
    remaining = deadline - time.monotonic() - 150
    t_sg = min(remaining * 0.25, 200)

    if t_sg > 15:
        n_cur = len(best_seq)
        print(f"  Running SG polish (n={n_cur}), budget={t_sg:.0f}s")
        a_opt, _ = subgrad_polish(best_seq, time_limit=min(t_sg * 0.5, 100),
                                   lr=0.002, mom=0.94, top_k=n_cur // 5, temp=0.00005)
        try_seq(a_opt, "sg_1")

        rem = deadline - time.monotonic() - 100
        if rem > 15:
            a_opt, _ = subgrad_polish(best_seq, time_limit=min(rem * 0.5, 80),
                                       lr=0.0008, mom=0.96, top_k=n_cur // 3, temp=0.00002)
            try_seq(a_opt, "sg_fine")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Binarization and Structural Search ===
    print("\n=== Phase 5: Binarization & Structure ===")
    remaining = deadline - time.monotonic() - 100
    t_bin = min(remaining * 0.3, 120)

    if t_bin > 10:
        n_cur = len(best_seq)
        a_bin, obj_bin = binarize_and_optimize(best_seq, time_limit=min(t_bin, 60))
        try_seq(a_bin, "binarized")

        # KKT on binarized
        rem = deadline - time.monotonic() - 80
        if rem > 10:
            a_opt, _ = kkt_active_set_optimize(a_bin, max_iters=100, tol=1e-9)
            try_seq(a_opt, "kkt_bin")

        # FW on binarized
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            a_opt, _ = fw_with_lp(a_bin, time_limit=min(rem * 0.5, 40),
                                  use_linearized=True, top_k=20)
            try_seq(a_opt, "fw_bin")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Multi-restart with diverse strategies ===
    print("\n=== Phase 6: Multi-Restart ===")
    remaining = deadline - time.monotonic() - 50
    t_restart = min(remaining * 0.6, 200)

    if t_restart > 20:
        n_cur = len(best_seq)
        n_restarts = min(6, max(2, int(t_restart // 30)))
        t_each = t_restart / n_restarts

        for r in range(n_restarts):
            rem = deadline - time.monotonic() - 40
            if rem < 10:
                break
            t_alloc = min(t_each, rem)

            # Different strategy each restart
            noise_scale = 0.005 * (r + 1)
            f_start = project_simplex(best_seq + rng.standard_normal(n_cur) * noise_scale)

            if r % 3 == 0:
                a_opt, _ = kkt_active_set_optimize(f_start, max_iters=100, tol=1e-9)
                if rem > 20:
                    a_opt2, _ = fw_with_lp(a_opt, time_limit=min(rem * 0.3, 30),
                                           use_linearized=True, top_k=20)
                    try_seq(a_opt2, f"restart_kkt{r}")
                try_seq(a_opt, f"restart_kkt{r}_raw")
            elif r % 3 == 1:
                a_opt, _ = fw_with_lp(f_start, time_limit=min(t_alloc * 0.7, 50),
                                      use_linearized=True, top_k=25)
                try_seq(a_opt, f"restart_fw{r}")
            else:
                a_opt, _ = subgrad_polish(f_start, time_limit=min(t_alloc * 0.6, 40),
                                           lr=0.001, mom=0.95, top_k=30, temp=0.00005)
                # Then KKT polish
                a_opt2, _ = kkt_active_set_optimize(a_opt, max_iters=80, tol=1e-9)
                try_seq(a_opt2, f"restart_sg{r}")

    print(f"  Phase 6 best: {best_score:.8f}")

    # === Phase 7: Final KKT polish on the best ===
    print("\n=== Phase 7: Final KKT Polish ===")
    rem = deadline - time.monotonic() - 20
    if rem > 5 and best_seq is not None:
        a_opt, _ = kkt_active_set_optimize(best_seq, max_iters=200, tol=1e-11)
        try_seq(a_opt, "final_kkt")

        # One more FW pass
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = fw_with_lp(best_seq, time_limit=min(rem * 0.6, 20),
                                  use_linearized=True, top_k=20)
            try_seq(a_opt, "final_fw")

    # === Final verification ===
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
