            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def gradient_step(a, conv, lr, top_k=30, temp=0.004):
    """One projected gradient step on the soft peak objective."""
    n = len(a)
    grad = soft_gradient(a, conv, top_k=top_k, temp=temp)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def lp_solve(a, M_target, C):
    """Solve LP: max sum(g) s.t. C@g <= M_target, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > 1e-10 and np.all(g >= -1e-12):
                return np.maximum(g, 0)
    except Exception:
        pass
    return None


def bisection_step(a, d, n, best_obj):
    """Find best step size t in [0,1] for direction d using bisection."""
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_try = best_obj
    
    for _ in range(12):
        mid = (lo + hi) / 2
        a_try = a + mid * d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            hi = mid
            continue
        a_try = a_try / s_try
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_try - 1e-13:
            best_try = obj_try
            best_t = mid
            lo = mid
        else:
            hi = mid
    
    return best_t, best_try


def optimize_combined(a0, time_limit, verbose=False):
    """
    Combined LP + gradient optimization with adaptive strategies.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    no_improve = 0

            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-13:

                best_a = a.copy()

            else:
                no_improve += 1
            # Decay learning rate slowly
            lr_grad *= 0.999
        else:

            g = lp_solve(a, M * (1 + 1e-6), C)

                no_improve += 1
            else:

                        M = conv.max()
                        cur_obj = 2.0 * n * M
                        if cur_obj < best_obj - 1e-13:
                            best_obj = cur_obj
                            best_a = a_new.copy()
                            a = a_new
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
        
        iteration += 1
        
        # Handle stagnation
        if no_improve >= 25:

            scale = 0.002 + 0.003 * min(perturb_count, 8)

            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()

                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            lr_grad = 0.02  # Reset learning rate

                break

    return best_a, best_obj


def coordinate_refine(a0, time_limit):
    """Fine-tune using pairwise coordinate swaps."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    
    deltas = [0.0005, 0.001, 0.002, 0.005, 0.01, -0.0005, -0.001, -0.002, -0.005, -0.01]
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            break
        for _ in range(200):
            if time.monotonic() - t0 > time_limit - 1:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            improved = False
            for delta in deltas:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a[i] += delta
                a[j] -= delta
                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
                else:
                    a[i] -= delta
                    a[j] += delta
            if not improved:
                no_improve += 1
    
    return best_a, best_obj

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = np.convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def soft_gradient(a, conv, top_k=30, temp=0.004):
    """Softmax-weighted gradient of the peak of the convolution."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
