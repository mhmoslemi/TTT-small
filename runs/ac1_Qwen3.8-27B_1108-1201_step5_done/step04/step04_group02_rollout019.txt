Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with log-sum-exp smoothing,
FFT-based gradient computation, LP-based global moves, and adaptive temperature.

from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    rho = np.nonzero(cond)[0][-1]

def compute_gradient(a, w, fft_size):
    """Compute gradient of smooth objective: 2 * cross_correlation(w, a)."""
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(W) * A, fft_size)
    return 2.0 * cross[:len(a)]


def softmax_weights(C, t):
    """Compute softmax weights over convolution values."""
    tC = t * C
    tC -= tC.max()
    w = np.exp(tC)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(C))
        w[np.argmax(C)] = 1.0
    else:
        w /= wsum
    return w


def solve_lp(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix (2n-1) x n
    # Constraint k: sum_j f[j] * g[k-j] <= M (for valid k-j)
    # A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # This is a banded matrix.
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = 0
        k_end = min(n, 2 * n - 1 - j)
        # For constraint k, the coefficient of g[j] is f[k-j] if 0<=k-j<n
        # Equivalently, g[j] appears in constraints k = j, j+1, ..., j+n-1
        # with coefficients f[0], f[1], ..., f[n-1]
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fista_smooth(a_init, time_limit, t_init, t_final, t_decay, 
                  lp_every=200, lp_time=5, n_restarts=5, seed=42):
    """
    FISTA with log-sum-exp smoothed objective.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max() * 2 * n  # score = 2*n*max(C) for sum(a)=1
    
    z = a.copy()
    z_prev = a.copy()
    beta = 1.0
    t = t_init
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Compute objective and gradient
        C = fast_convolve(z)
        w = softmax_weights(C, t)
        grad = compute_gradient(z, w, fft_size)
        
        # FISTA step
        if it > 1:
            y = z + (beta - 1) / (beta + 1) * (z - z_prev)
        else:
            y = z
        
        # Step size: for the smooth objective with Lipschitz constant L
        # L is bounded by max |Hessian of smooth obj| which is O(t) * something
        # Use a practical step size
        step = 1.0 / (t * 4.0 * np.sum(a * a))
        step = min(step, 1.0)
        
        z_prev = z.copy()
        z = project_simplex(y - step * grad)
        beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        
        # Evaluate actual objective
        C_new = fast_convolve(z)
        obj = C_new.max() * 2 * n
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = z.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Temperature annealing
        t = max(t_final, t * t_decay)
        
        # LP escape
        if lp_every > 0 and no_improve > lp_every and rem > 20:
            M = C_new.max() * 1.05
            g = solve_lp(z, M, time_limit=lp_time)
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    # Line search
                    best_alpha = 0.0
                    best_val = best_obj
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        z_try = project_simplex((1 - alpha) * best_a + alpha * g)
                        obj_try = fast_convolve(z_try).max() * 2 * n
                        if obj_try < best_val:
                            best_val = obj_try
                            best_alpha = alpha
                    if best_alpha > 0:
                        best_a = project_simplex((1 - best_alpha) * best_a + best_alpha * g)
                        z = best_a.copy()
                        z_prev = best_a.copy()
                        beta = 1.0
                        no_improve = 0
                        t = t_init * 0.5
                        M = fast_convolve(z).max()
            
            no_improve = 0  # Reset regardless
        
        # Restart if stuck
        if no_improve > 3 * lp_every:
            noise = rng.standard_normal(n) * 0.01
            z = project_simplex(best_a + noise)
            z_prev = z.copy()
            beta = 1.0
            t = t_init * 0.3
            no_improve = 0
    
    return best_a, best_obj


def multiplicative_weights(a_init, time_limit, t_init, t_final, t_decay, seed=42):
    """
    Multiplicative weights (mirror descent) for the smoothed objective.
    Update: a_new[j] = a[j] * exp(-eta * grad[j]), then normalize.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max() * 2 * n
    
    eta = 1.0
    t = t_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        C = fast_convolve(a)
        w = softmax_weights(C, t)
        grad = compute_gradient(a, w, fft_size)
        
        # Multiplicative update
        log_a = np.log(a + 1e-30)
        new_a = np.exp(log_a - eta * grad)
        new_a = np.maximum(new_a, 0.0)
        ns = new_a.sum()
        if ns < 1e-15:
            break
        new_a = new_a / ns
        
        # Check improvement
        C_new = fast_convolve(new_a)
        obj = C_new.max() * 2 * n
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = new_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Revert
            pass  # Keep old a
        
        a = new_a  # Always accept for exploration
        
        # Adaptive eta and t
        eta *= 0.999
        t = max(t_final, t * t_decay)
        
        if no_improve > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            eta = 1.0 / n
            t = t_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    if n > 1:
        c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
        inits.append(c / c.sum())
    
    # Sine
    sn = np.sin(np.pi * (x + 1) / (n + 1)) + 0.05
    inits.append(sn / sn.sum())
    
    # Two bump (symmetric)
    b1 = np.exp(-((x - n*0.3) / (n*0.1)) ** 2) + np.exp(-((x - n*0.7) / (n*0.1)) ** 2) + 0.01
    inits.append(b1 / b1.sum())
    
    # Three bump
    b2 = np.exp(-((x - n*0.2) / (n*0.08)) ** 2) + np.exp(-((x - n*0.5) / (n*0.08)) ** 2) + np.exp(-((x - n*0.8) / (n*0.08)) ** 2) + 0.01
    inits.append(b2 / b2.sum())
    
    # Inverse sine (peaked in middle)
    if n > 1:
        inv = 1.0 / (np.sin(np.pi * (x + 1) / (n + 1)) + 0.1)
        inits.append(inv / inv.sum())
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
