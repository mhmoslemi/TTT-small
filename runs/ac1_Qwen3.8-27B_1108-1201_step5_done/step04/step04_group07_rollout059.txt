import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    rho = np.nonzero(cond)[0][-1]

    # We want (p * a)[i] = sum_k p[k] * a[k-i] for i in 0..n-1

    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(P * A, fft_size)[:n]

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    mom = 0.9
    lr = 0.01
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0

        scaled = conv / t
        scaled -= scaled.max()
        p = np.exp(scaled)
        w = p / p.sum()

        W = np.fft.rfft(w, fft_size)

            no_improve = 0

            no_improve = 0
            lr = 0.01 * max(0.1, t / t_start)
        
        if verbose and no_improve % 500 == 0:
            print(f"    t={t:.6f}, obj={best_obj:.10f}, no_improve={no_improve}")
    
    return best_a, best_obj

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, t):
    """Compute gradient of soft_max(conv(a,a), t) for a with sum=1.
    Returns gradient vector of same length as a."""
    n = len(a)
    conv_aa = fast_convolve(a)
    scaled = conv_aa / t
    scaled -= scaled.max()
    p = np.exp(scaled)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv_aa)] = 1.0
    else:
        p /= ps
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    cross = np.fft.irfft(P * AR, fft_size)
    grad = 2.0 * cross[n - 1:2 * n - 1]
    return grad

def pairwise_coordinate_descent(a0, time_limit, rng, max_sweeps=None, verbose=False):
    """Pairwise swap coordinate descent with golden section search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    obj = fast_convolve(a).max()
    best_obj = obj
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0

    golden = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        # Random permutation of pairs
        indices = rng.permutation(n)
        for i in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            ii = int(indices[i])
            jj = int(indices[i + 1])
            if a[ii] < 1e-10:
                continue

            # Golden section search for delta in [0, a[ii]]
            lo = 0.0
            hi = a[ii]
            a1 = hi - golden * (hi - lo)
            a2 = lo + golden * (hi - lo)

            # Evaluate at a1, a2
            a1_val = a.copy()
            a1_val[ii] -= a1
            a1_val[jj] += a1
            a1_obj = fast_convolve(a1_val).max()

            a2_val = a.copy()
            a2_val[ii] -= a2
            a2_val[jj] += a2
            a2_obj = fast_convolve(a2_val).max()

            for _ in range(12):
                if a1_obj < a2_obj:
                    hi = a2
                    a2 = a1
                    a2_obj = a1_obj
                    a1 = hi - golden * (hi - lo)
                    a1_val = a.copy()
                    a1_val[ii] -= a1
                    a1_val[jj] += a1
                    a1_obj = fast_convolve(a1_val).max()
                else:
                    lo = a1
                    a1 = a2
                    a1_obj = a2_obj
                    a2 = lo + golden * (hi - lo)
                    a2_val = a.copy()
                    a2_val[ii] -= a2
                    a2_val[jj] += a2
                    a2_obj = fast_convolve(a2_val).max()

            best_delta = (lo + hi) / 2.0
            a_new = a.copy()
            a_new[ii] -= best_delta
            a_new[jj] += best_delta
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns > 1e-15:
                a_new = a_new / ns
            new_obj = fast_convolve(a_new).max()

            if new_obj < obj - 1e-15:
                a = a_new
                obj = new_obj
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            print(f"      CD sweep={sweep}, obj={best_obj:.10f}")

        if max_sweeps is not None and sweep >= max_sweeps:
            break

    return best_a, best_obj

            best_half = half_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        half = half_new
        t = max(t * 0.998, t_end)
        lr = max(lr * 0.9997, 0.0001)

        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 7) % 999991)
            half = project_simplex(best_half + rng.standard_normal(nh) * 0.003)
            v = np.zeros(nh)
            no_improve = 0
            lr = lr_start * 0.5

        if verbose and it % 500 == 0:
            print(f"      PalOpt it={it}, t={t:.6f}, obj={best_obj:.10f}")

    return best_half, best_obj


def lp_frank_wolfe(a0, time_limit, M_expand=1.0, verbose=False):
    """Max-sum LP Frank-Wolfe (simplified version for polishing)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Solve max-sum LP: max sum(g) s.t. (a*g)[k] <= M*M_expand, g >= 0
        B = np.zeros((2 * n - 1, n))
        for j in range(n):
            B[j:j + n, j] = a

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * M_expand)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=B, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or res.x is None:
            M_expand = min(M_expand * 1.2, 5.0)
            if M_expand > 4.0:
                no_improve += 5
                if no_improve > 30:
                    break
            continue

        g = res.x
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - a)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            M_expand = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            M_expand = min(M_expand * 1.1, 4.0)

        if no_improve > 50:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            M = fast_convolve(a).max()
            M_expand = 1.0
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"      LP-FW it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Palindromic U-shape (high at edges)
    for width in [0.2, 0.35, 0.5]:
        u = 1.0 / (np.abs(x - mid) / (width * n) + 0.3)
        u = (u + u[::-1]) / 2
        inits.append(u / u.sum())

    # Palindromic concave (high in middle)
    for width in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - mid) / (width * n)) ** 2)
        inits.append(g / g.sum())

    # Two peaks
    for sep in [0.2, 0.3, 0.4]:
        p1 = np.exp(-0.5 * ((x - mid + sep * n) / (0.1 * n)) ** 2)
        p2 = np.exp(-0.5 * ((x - mid - sep * n) / (0.1 * n)) ** 2)
        two = p1 + p2
        inits.append(two / two.sum())

    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.05
    inits.append(c / c.sum())

    # Random Dirichlet
    for alpha in [0.3, 0.7, 1.5, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))

    # Parent interpolated
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Palindromic version of parent
        pp = (px + px[::-1]) / 2
        ps2 = pp.sum()
        if ps2 > 1e-10:
            inits.append(pp / ps2)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Diverse initialization across multiple lengths
    Phase 2: Smooth temperature-annealed gradient optimization
    Phase 3: Palindromic-structured optimization
    Phase 4: Simulated annealing for global exploration
    Phase 5: Pairwise coordinate descent for local refinement
    Phase 6: LP Frank-Wolfe polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_phase1_start = time.monotonic()

    lengths_to_try = [256, 320, 384, 448, 512]
    if parent is not None:
        n_parent = len(parent)
        if n_parent not in lengths_to_try:
            lengths_to_try.insert(0, n_parent)

    candidates = []
    for n_try in lengths_to_try:
        inits = generate_initializations(n_try, parent, rng)
        for i, ini in enumerate(inits):
            candidates.append((f"n{n_try}_i{i}", ini))

    # Quick eval of all candidates
    for label, cand in candidates:
        if time.monotonic() > deadline - 300:
            break
        try_seq(cand, f"init_{label}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    n_best = len(best_seq)
    print(f"  Phase 1 best: {best_score:.8f} (n={n_best})")

    # === Phase 2: Smooth temperature-annealed optimization ===
    print("\n=== Phase 2: Smooth Annealed Optimization ===")
    remaining = deadline - time.monotonic() - 200
    t_p2 = min(remaining * 0.35, 500)

    if t_p2 > 30:
        # Run on best candidate
        sub_time = min(t_p2 * 0.4, 200)
        print(f"  Smooth anneal on n={n_best}, budget={sub_time:.0f}s")
        a_opt, _ = smooth_anneal_optimize(best_seq, time_limit=sub_time,
                                          t_start=0.005, t_end=0.00005,
                                          lr_start=0.015, mom=0.85)
        try_seq(a_opt, "smooth_main")

        # Run with different parameters
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            sub_time2 = min(rem * 0.4, 200)
            print(f"  Smooth anneal (fast) on n={n_best}, budget={sub_time2:.0f}s")
            a_opt, _ = smooth_anneal_optimize(best_seq, time_limit=sub_time2,
                                              t_start=0.002, t_end=0.0001,
                                              lr_start=0.008, mom=0.9)
            try_seq(a_opt, "smooth_fast")

        # Try a different length
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            n_alt = n_best + 64 if n_best < 512 else n_best - 64
            if n_alt >= 128:
                px = np.interp(np.linspace(0, 1, n_alt),
                               np.linspace(0, 1, n_best), best_seq)
                px = px / px.sum()
                sub_time3 = min(rem * 0.5, 150)
                a_opt, _ = smooth_anneal_optimize(px, time_limit=sub_time3,
                                                  t_start=0.004, t_end=0.00008,
                                                  lr_start=0.012, mom=0.87)
                try_seq(a_opt, f"smooth_n{n_alt}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Palindromic optimization ===
    print("\n=== Phase 3: Palindromic Optimization ===")
    remaining = deadline - time.monotonic() - 150
    t_p3 = min(remaining * 0.2, 250)

    if t_p3 > 20:
        # Make best_seq palindromic
        n_cur = len(best_seq)
        pal = (best_seq + best_seq[::-1]) / 2.0
        pal = pal / pal.sum()
        half = pal[: (n_cur + 1) // 2]
        half = half / half.sum()

        sub_time = min(t_p3 * 0.5, 120)
        print(f"  Palindromic opt, budget={sub_time:.0f}s")
        half_opt, _ = palindromic_optimize(half, time_limit=sub_time,
                                           t_start=0.004, t_end=0.00006,
                                           lr_start=0.012, mom=0.85)
        full_opt = build_palindromic(half_opt)
        full_opt = full_opt / full_opt.sum()
        try_seq(full_opt, "palindromic")

        # Try a different palindromic init
        rem = deadline - time.monotonic() - 100
        if rem > 20 and n_cur >= 128:
            half_rand = rng.dirichlet(np.ones((n_cur + 1) // 2) * 0.5)
            sub_time2 = min(rem * 0.6, 100)
            half_opt, _ = palindromic_optimize(half_rand, time_limit=sub_time2,
                                               t_start=0.006, t_end=0.0001,
                                               lr_start=0.015, mom=0.82)
            full_opt = build_palindromic(half_opt)
            full_opt = full_opt / full_opt.sum()
            try_seq(full_opt, "pal_rand")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Simulated Annealing ===
    print("\n=== Phase 4: Simulated Annealing ===")
    remaining = deadline - time.monotonic() - 100
    t_p4 = min(remaining * 0.2, 200)

    if t_p4 > 20:
        sub_time = min(t_p4 * 0.6, 120)
        print(f"  SA, budget={sub_time:.0f}s")
        a_opt, _ = simulated_annealing(best_seq, time_limit=sub_time, rng=rng,
                                       T_start=0.003, T_end=0.00001,
                                       noise_start=0.008)
        try_seq(a_opt, "sa1")

        # Second SA with different params
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            sub_time2 = min(rem * 0.5, 80)
            a_opt, _ = simulated_annealing(best_seq, time_limit=sub_time2, rng=rng,
                                           T_start=0.001, T_end=0.000005,
                                           noise_start=0.004)
            try_seq(a_opt, "sa2")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Pairwise Coordinate Descent ===
    print("\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 60
    t_p5 = min(remaining * 0.3, 150)

    if t_p5 > 10:
        sub_time = min(t_p5 * 0.7, 100)
        print(f"  CD, budget={sub_time:.0f}s")
        a_opt, _ = pairwise_coordinate_descent(best_seq, time_limit=sub_time, rng=rng)
        try_seq(a_opt, "cd1")

        # Another CD pass
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = pairwise_coordinate_descent(best_seq, time_limit=min(rem * 0.5, 50),
                                                   rng=rng)
            try_seq(a_opt, "cd2")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: LP Frank-Wolfe Polish ===
    print("\n=== Phase 6: LP Polish ===")
    remaining = deadline - time.monotonic() - 20
    t_p6 = min(remaining * 0.5, 80)

    if t_p6 > 10 and n_best <= 512:
        sub_time = min(t_p6 * 0.8, 60)
        print(f"  LP-FW, budget={sub_time:.0f}s")
        a_opt, _ = lp_frank_wolfe(best_seq, time_limit=sub_time)
        try_seq(a_opt, "lp_polish")

    print(f"  Phase 6 best: {best_score:.8f}")

    # === Final: Quick attempts on remaining time ===
    rem = deadline - time.monotonic() - 5
    if rem > 10 and best_seq is not None:
        # Final smooth polish at very low temperature
        a_opt, _ = smooth_anneal_optimize(best_seq, time_limit=min(rem * 0.6, 30),
                                          t_start=0.0005, t_end=0.00002,
                                          lr_start=0.003, mom=0.92)
        try_seq(a_opt, "final_smooth")

        # Final CD
        rem2 = deadline - time.monotonic() - 3
        if rem2 > 5:
            a_opt, _ = pairwise_coordinate_descent(best_seq, time_limit=min(rem2 * 0.7, 20),
                                                   rng=rng)
            try_seq(a_opt, "final_cd")

    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
