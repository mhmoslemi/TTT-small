def fast_convolve(a):

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(f)
    peaks = np.where(conv >= conv.max() * 0.999)[0]
    sg = np.zeros(n)
    for k in peaks:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # dc[k]/da[j] = a[k-j] for j in js..je
        # So sg[j] += a[k-j]
        idx = np.arange(js, je + 1)
        sg[idx] += f[k - idx]
    return 2.0 * sg

def pairwise_cd(f0, time_limit, rng=None):
    """Pairwise coordinate descent: for each pair (i,j), find optimal mass transfer."""
    if rng is None:
        rng = np.random.default_rng(99)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break
        epoch += 1
        
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 3:
                break
            # Try moving mass from j to a few targets
            targets = [(j + 1) % n, (j - 1) % n, (j + n // 2) % n,
                       (j + n // 3) % n, rng.integers(n)]
            
            for t in targets:
                t = int(t) % n
                if t == j or f[j] < 1e-10:
                    continue
                
                # Binary search for optimal fraction to move
                lo, hi = 0.0, 1.0
                best_frac = 0.0
                best_val = fast_convolve(f).max()
                
                for _ in range(12):
                    mid = (lo + hi) / 2.0
                    delta = f[j] * mid
                    f_new = f.copy()
                    f_new[j] -= delta
                    f_new[t] += delta
                    if f_new[j] < 0:
                        delta = f[j]
                        f_new = f.copy()
                        f_new[j] = 0.0
                        f_new[t] += delta
                    f_new = f_new / f_new.sum()
                    val = fast_convolve(f_new).max()
                    if val < best_val:
                        best_val = val
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid
                
                if best_frac > 0.01:
                    delta = f[j] * best_frac
                    f = f.copy()
                    f[j] -= delta
                    f[t] += delta
                    f = f / f.sum()
                    conv = fast_convolve(f)
                    if conv.max() < best_obj - 1e-15:
                        best_obj = conv.max()
                        best_f = f.copy()
                        improved = True
        
        if not improved:
            # Try random restart
            if epoch > 3:
                break
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
    
    return best_f, best_obj

    n = len(v)

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe,
spectral equalization, adaptive subgradient descent, pairwise coordinate
descent with binary search, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n_lp_time=15):
    """LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': n_lp_time, 'presolve': True})
        if res.success and res.fun is not None:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, rng, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_expand = 1.0

    while time.monotonic() - t0 < time_limit - 3:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Solve LP with current M
        g = solve_lp_direction(f, M * m_expand, n_lp_time=min(10, remaining // 4))

        if g is None:
            m_expand = min(m_expand * 1.15, 6.0)
            if m_expand > 5.0:
                m_expand = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fixed point check
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            m_expand = min(m_expand * 1.08, 4.0)
            if no_improve > 40:
                break
            continue

        # Line search: golden-section style
        best_alpha = 0.0
        best_val = M
        # Coarse scan
        alphas = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.65, 0.8, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection around best
        if 0.02 < best_alpha < 0.98:
            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.06)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            m_expand = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_expand = min(m_expand * 1.12, 5.0)

        # Adaptive restart
        if no_improve > 35:
            scale = 0.003 + 0.003 * min(no_improve // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_expand = 1.0
            no_improve = 0

    return best_f, best_obj


def spectral_refine(f0, time_limit, rng):
    """Iteratively dampen frequency components that contribute to convolution peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length() - 1
    if fft_size < 2 * n - 1:
        fft_size = 1 << (2 * n - 1).bit_length()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.15
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 3:
        if time.monotonic() - t0 > time_limit - 5:
            break

        A = np.fft.rfft(f, fft_size)
        B = A * A
        c = np.fft.irfft(B, fft_size)[:2 * n - 1]

        # Compute error: how far c is from being flat
        target = 1.0 / (2 * n - 1)
        error = np.maximum(c - target, 0)  # only push down peaks

        # Convert error to frequency domain
        E = np.fft.rfft(error, fft_size)

        # Correction: reduce A where |A|^2 contributes to peaks
        # d(A^2)/dA = 2A, so to reduce B by E: dA = -E/(2A)
        A_mag = np.abs(A)
        A_safe = A.copy()
        mask = A_mag > 1e-12
        A_safe[mask] = A[mask]
        A_safe[~mask] = 1e-12

        dA = -lr * E / (2.0 * A_safe)
        # Ensure dA has proper conjugate symmetry (since f is real)
        # rfft handles this, so we just need dA to be "real-symmetric"
        # For rfft, this means dA[0] and dA[-1] should be real
        dA[0] = dA[0].real
        if len(dA) > 1:
            dA[-1] = dA[-1].real

        A_new = A + dA
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        fn = f_new.sum()
        if fn < 1e-15:
            break
        f_new = f_new / fn

        conv_new = fast_convolve(f_new)
        val = conv_new.max()

        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.03, 0.6)
        else:
            no_improve += 1
            lr = max(lr * 0.85, 0.02)
            # Try smaller step
            f_try = f - 0.1 * lr * dA
            f_try = np.maximum(f_try, 0)
            ft = f_try.sum()
            if ft > 1e-15:
                f_try = f_try / ft
                val_try = fast_convolve(f_try).max()
                if val_try < best_obj - 1e-14:
                    best_obj = val_try
                    best_f = f_try.copy()
                    f = f_try
                    no_improve = 0

        if no_improve > 80:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            no_improve = 0
            lr = 0.15

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, rng, lr0=0.008, momentum=0.92):
    """Subgradient descent with proper subgradient and adaptive step on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    lr = lr0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 3:
        it += 1
        if time.monotonic() - t0 > time_limit - 5:
            break

        conv = fast_convolve(f)
        cmax = conv.max()

        # Find all (near-)peaks
        threshold = cmax * (1.0 - 1e-7)
        peak_idx = np.where(conv >= threshold)[0]

        # Compute subgradient: average gradient over active peaks
        sg = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += f[k - idx]
        sg *= 2.0 / len(peak_idx)

        # Project subgradient onto tangent space of simplex
        sg_tan = sg - sg.mean()

        # Momentum + step
        v = momentum * v - lr * sg_tan

        # Line search along direction v
        best_alpha = 0.0
        best_val = cmax
        for alpha_mult in [0.1, 0.5, 1.0, 2.0, 5.0]:
            alpha = lr * alpha_mult
            f_try = f + alpha * v
            f_try = project_simplex(f_try)
            val = fast_convolve(f_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = f + best_alpha * v
            f = project_simplex(f)
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        lr *= 0.9995

        if no_improve > 250:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.003 + 0.001 * (no_improve // 300)))
            v = np.zeros(n)
            lr = max(lr0 * 0.5, 0.002)
            no_improve = 0

    return best_f, best_obj


def pairwise_cd(f0, time_limit, rng):
    """Pairwise coordinate descent with binary search on mass transfer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit - 5:
        epoch += 1
        if time.monotonic() - t0 > time_limit - 8:
            break

        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 5:
                break
            if f[j] < 1e-12:
                continue

            # Try a few target positions
            targets = [
                (j + 1) % n,
                (j - 1) % n,
                (j + n // 2) % n,
                (j + n // 4) % n,
                (j + 3 * n // 4) % n,
                int(rng.integers(n)),
            ]

            for t in targets:
                t = int(t) % n
                if t == j:
                    continue

                # Binary search: what fraction of f[j] to move to t?
                lo, hi = 0.05, 0.95
                best_frac = 0.0
                cur_best = fast_convolve(f).max()

                for _ in range(14):
                    mid = (lo + hi) / 2.0
                    delta = f[j] * mid
                    f_new = f.copy()
                    f_new[j] -= delta
                    f_new[t] += delta
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        break
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < cur_best:
                        cur_best = val
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid

                if best_frac > 0.03 and cur_best < best_obj - 1e-15:
                    delta = f[j] * best_frac
                    f = f.copy()
                    f[j] -= delta
                    f[t] += delta
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    best_obj = cur_best
                    best_f = f.copy()
                    improved = True

        if not improved and epoch > 2:
            # Perturb and try again
            if time.monotonic() - t0 < time_limit - 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            else:
                break

    return best_f, best_obj


def try_sequence(a, label, state, verbose=True):
    """Evaluate a sequence and update state if better."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return float('inf')
    a = a / a.sum()
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    elif verbose:
        print(f"  [{label}]: score={score:.8f} (not better than {state['best_score']:.8f})")
    return score


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return new_seq / s

            a_opt, _ = pairwise_cd(state['best_seq'], time_limit=min(rem, 50), rng=rng)
            try_sequence(a_opt, "p4_cd2", state)

    # Fallback
    if state['best_seq'] is None:
        print("  Using fallback uniform sequence!")
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    # Final verification
    final_a = state['best_seq']
    final_score = float(evaluate_sequence(final_a.tolist()))
    if np.isfinite(final_score):
        state['best_score'] = final_score

    print(f"\nFINAL: score={state['best_score']:.8f}, n={len(state['best_seq'])}")
    return [float(x) for x in state['best_seq']]
