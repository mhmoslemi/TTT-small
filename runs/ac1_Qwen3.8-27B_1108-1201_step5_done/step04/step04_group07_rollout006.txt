"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using central binomial initialization
and LP-based Frank-Wolfe optimization with multi-shape exploration.
Key innovation: the central binomial sequence (coefficients of 1/sqrt(1-x))
provides a much better starting point than uniform, with flat autocorrelation
for the first n lags, enabling faster convergence to the optimum.
"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def central_binomial_seq(n):
    """Generate the central binomial sequence C(2k,k)/4^k for k=0,...,n-1, normalized."""
    a = np.ones(n)
    for k in range(n):
        a[k] = comb(2*k, k) / (4.0**k)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def symmetric_central_binomial(n):
    """Symmetric version: use central binomial for first half, mirror for second half."""
    half = (n + 1) // 2
    a = np.ones(n)
    for k in range(half):
        a[k] = comb(2*k, k) / (4.0**k)
    # Mirror
    for k in range(half, n):
        a[k] = a[n - 1 - k]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def geometric_seq(n, ratio=0.95):

    a = ratio ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s

def binomial_seq(n, alpha):
    """Beta-like sequence proportional to (k+1)^alpha * (n-k)^alpha."""
    k = np.arange(n, dtype=np.float64) + 0.5
    a = k**alpha * (n - k)**alpha
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def solve_linearized_lp(f, conv_f):
    """Solve: min t s.t. t >= conv_f[k] + 2*(f*g)[k] for all k, g >= -f, sum(g)=0."""
    n = len(f)
    # Build constraint matrix
    # c[k] = sum_j f[j] * g[k-j], so the gradient of c[k] w.r.t. g is:
    # d c[k] / d g[j] = f[k-j] if 0 <= k-j < n else 0
    
    # Variables: [t, g_0, ..., g_{n-1}]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    # A_ub * x <= b_ub
    # For each k: -t + 2 * sum_j f[k-j]*g[j] <= -conv_f[k]
    # i.e., -t + 2 * (f*g)[k] <= -conv_f[k]
    # where (f*g)[k] = sum_j f[k-j]*g[j] (with valid indices)
    
    # Build the matrix efficiently
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    
    # For each k, the row is: 2 * [f[k], f[k-1], ..., f[k-n+1]] (padded with 0s)
    # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j]
    # So coefficient of g[j] in row k is: 2*f[k-j] if 0<=k-j<n else 0
    
    for j in range(n):
        # g[j] appears in c[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j] = f[0], f[1], ..., f[n-1]
        k_start = j
        k_end = j + n - 1
        A_ub[k_start:k_end+1, j+1] = 2.0 * f
    
    b_ub = -conv_f
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None
