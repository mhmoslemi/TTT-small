"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous version:
1. LP-based Frank-Wolfe with iterate averaging (faster convergence)
2. Exact piecewise-linear line search exploiting convexity

4. Nesterov-accelerated smoothed gradient for fine polish

6. Warm-started LP with adaptive constraint management

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_max(f):
    """Compute max of auto-convolution of f (assumes sum(f)=1)."""
    return fast_convolve(f).max()


def solve_lp_dir(f, M, n_constraints=None):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns normalized solution or None.
    """
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        g = res.x
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None


def exact_line_search(f, g, n_evals=20):
    """Find alpha in [0,1] minimizing max_k conv((1-a)f + a*g).
    The function is piecewise linear and convex in alpha.
    Uses ternary search on the piecewise-linear convex function.
    """
    n = len(f)
    
    def phi(alpha):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            return float('inf')
        ft = ft / st
        return fast_convolve(ft).max()
    
    # The function is convex on [0,1], so ternary search works
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = phi(m1)
        f2 = phi(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    alpha = (lo + hi) / 2.0
    return alpha, phi(alpha)


def fw_with_averaging(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with iterate averaging.
    
    The key insight: the average of FW iterates can converge faster
    than the last iterate, especially for the max-of-linear objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    # Averaging
    f_avg = f.copy()
    avg_weight = 1.0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break
        
        # Solve LP
        g = solve_lp_dir(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                m_exp = 1.0
                M = conv_max(f)
            continue
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 150:
                break
            m_exp = min(m_exp * 1.01, 8.0)
            continue
        
        # Line search
        alpha, best_val = exact_line_search(f, g, n_evals=16)
        
        if alpha < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 10.0)
            continue
        
        # Update
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        f = f_new
        
        # Update average (use decaying weights for recency bias)
        # Standard FW averaging: uniform average of all iterates
        avg_weight += 1.0
        f_avg = f_avg + (f_new - f_avg) / avg_weight
        
        conv = fast_convolve(f)
        M = conv.max()
        m_exp = 1.0
        
        # Check both current and averaged
        obj_curr = conv.max()
        obj_avg = conv_max(f_avg)
        
        if obj_avg < best_obj - 1e-15:
            best_obj = obj_avg
            best_f = f_avg.copy()
            no_improve = 0
        elif obj_curr < best_obj - 1e-15:
            best_obj = obj_curr
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 200 == 0:
            print(f"    FW-avg iter {it}: best={best_obj:.10f}, curr={obj_curr:.10f}, avg={obj_avg:.10f}")
        
        # Periodic restart
        if no_improve > 100:
            rng = np.random.default_rng((it * 104729 + n * 7919) % (2**31))
            scale = 0.0005 + 0.001 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            f_avg = f.copy()
            avg_weight = 1.0
    
    # Final check
    obj_avg = conv_max(f_avg)
    if obj_avg < best_obj:
        best_obj = obj_avg
        best_f = f_avg.copy()
    
    return best_f, best_obj


def smoothed_grad_nesterov(f0, time_limit, T_init=0.002, T_min=0.00005, verbose=False):
    """Nesterov-accelerated gradient descent on smoothed max-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = conv_max(f)
    best_f = f.copy()
    
    # Nesterov momentum
    y = f.copy()
    momentum = 0.0
    
    T = T_init
    lr = 0.005
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(y)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = w / ws
        
        # Gradient: grad[i] = 2 * sum_j f[j] * p[j+i] (cross-correlation)
        fft_size = 1 << (3 * n).bit_length()
        Y = np.fft.rfft(y, fft_size)
        P = np.fft.rfft(p, fft_size)
        cross = np.fft.irfft(np.conj(Y) * P, fft_size)
        grad = 2.0 * cross[:n]
        
        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)
        
        # Update momentum (Nesterov)
        momentum_new = (it + 1 + momentum) / (it + 2)
        y_new = project_simplex(f - lr * grad)
        y = (1.0 - momentum_new) * y_new + momentum_new * (y_new - f)
        y = project_simplex(y)
        momentum = momentum_new
        
        # Update f
        f = y_new
        
        obj = conv_max(f)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr = max(lr * 0.9, 0.0001)
        
        # Anneal temperature
        T = max(T * 0.998, T_min)
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
            y = f.copy()
            momentum = 0.0
            lr = max(0.002, lr)
            T = max(T_init * 0.2, T_min)
            no_improve = 0
    
    return best_f, best_obj


def active_set_polish(f0, time_limit, tol=1e-10, verbose=False):
    """Polish using active-set method.
    
    Near the optimum, only a few convolution values are at the maximum.
    We identify them and solve the KKT conditions directly.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = conv_max(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Identify active set (values close to max)
        threshold = M - max(0.001 * M, 1e-8)
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Build the system of KKT conditions
        # For active constraints k: (f*f)[k] = M
        # For all i: 2 * sum_{k in active} lambda_k * d(df*f/dk)/df[i] = mu + nu[i]
        # sum(f) = 1, f >= 0
        # lambda_k >= 0, complementary slackness
        
        # Simplified approach: use the subgradient from active constraints
        # and do a projected step
        subgrad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                subgrad[j] += f[k - j]
                if k - j == j:
                    subgrad[j] += f[j]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(subgrad)
        if sg_norm < 1e-15:
            break
        
        # Adaptive step size
        lr = min(0.01 / sg_norm, 0.01)
        f_new = project_simplex(f - lr * subgrad)
        
        obj_new = conv_max(f_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller step
            lr *= 0.5
            f_new = project_simplex(f - lr * subgrad)
            obj_new = conv_max(f_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
            else:
                if it > 100:
                    break
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - mid) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())
    
    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.25 * (n-1)) / (0.15 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * (n-1)) / (0.15 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    
    # Symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] = 0.5 * (half[-1] + half[0]) if half_n > 0 else 1.0
    inits.append(sym / sym.sum())
    
    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

            if cycle % 2 == 1:
                rem = p2_budget - (time.monotonic() - t_p2)
                if rem > 40:
                    new_n = n_best + 16
                    px = np.interp(np.linspace(0, 1, new_n),
                                   np.linspace(0, 1, n_best), f_work)
                    px = project_simplex(px)
                    t_alloc2 = min(30.0, rem / 3.0)
                    a_opt2, _ = fw_with_averaging(px, time_limit=t_alloc2)
                    try_seq(a_opt2, f"p2_len{new_n}")
        
        # Also try one more length at the end
        rem = p2_budget - (time.monotonic() - t_p2)
        if rem > 40 and best_seq is not None:
            for delta in [32, -32, 64]:
                rem = p2_budget - (time.monotonic() - t_p2)
                if rem < 30:
                    break
                new_n = len(best_seq) + delta
                if new_n < 160 or new_n > 1000:
                    continue
                f_work2 = best_seq / best_seq.sum()
                px = np.interp(np.linspace(0, 1, new_n),
                               np.linspace(0, 1, len(f_work2)), f_work2)
                px = project_simplex(px)
                t_alloc = min(30.0, rem / 2.0)
                a_opt, _ = fw_with_averaging(px, time_limit=t_alloc)
                try_seq(a_opt, f"p2_final_len{new_n}")
    
    print(f"  Phase 2 done: best={best_score:.6f}")
    
    # === Phase 3: Polishing (500s) ===
    print(f"\n=== Phase 3: Polishing ===")
    t_p3 = time.monotonic()
    p3_budget = 500.0
    
    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()
        
        # 3a: Nesterov smoothed gradient (long run)
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 60:
            T_init = max(0.001, 0.003 * conv_max(f_work))
            t_alloc = min(150.0, rem - 30.0)
            print(f"  Nesterov smoothed GD (t={t_alloc:.0f}s, T={T_init:.5f})")
            a_opt, _ = smoothed_grad_nesterov(f_work, time_limit=t_alloc,
                                                T_init=T_init, T_min=0.00003)
            try_seq(a_opt, "p3_nesterov")
        
        # 3b: Active set polish
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30 and best_seq is not None:
            f_work = best_seq / best_seq.sum()
            t_alloc = min(80.0, rem - 20.0)
            print(f"  Active set polish (t={t_alloc:.0f}s)")
            a_opt, _ = active_set_polish(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_active")
        
        # 3c: More FW with tighter settings
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 40 and best_seq is not None:
            f_work = best_seq / best_seq.sum()
            t_alloc = min(120.0, rem - 20.0)
            print(f"  Final FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_with_averaging(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_fw_final")
        
        # 3d: Nesterov with very small T
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 40 and best_seq is not None:
            f_work = best_seq / best_seq.sum()
            t_alloc = min(80.0, rem - 15.0)
            T_init = max(0.0003, 0.0005 * conv_max(f_work))
            print(f"  Fine Nesterov (t={t_alloc:.0f}s, T={T_init:.6f})")
            a_opt, _ = smoothed_grad_nesterov(f_work, time_limit=t_alloc,
                                                T_init=T_init, T_min=0.00001)
            try_seq(a_opt, "p3_fine_nesterov")
    
    print(f"  Phase 3 done: best={best_score:.6f}")
    
    # === Phase 4: Final tricks (100s) ===
    print(f"\n=== Phase 4: Final Tricks ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()
        
        # Small perturbations + quick polish
        for i in range(5):
            if time.monotonic() > deadline - 10:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = fw_with_averaging(f_pert, time_limit=15)
            try_seq(a_opt, f"p4_pert{i}")
        
        # One final FW
        if time.monotonic() < deadline - 20 and best_seq is not None:
            f_work = best_seq / best_seq.sum()
            a_opt, _ = fw_with_averaging(f_work, time_limit=20)
            try_seq(a_opt, "p4_final")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
