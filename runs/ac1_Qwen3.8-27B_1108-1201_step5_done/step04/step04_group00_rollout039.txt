from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, n):
    """Maximize sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fs)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)

def coord_descent_polish(f0, time_limit):

    return best_f, best_M

def interpolate(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    s = fi.sum()
    if s > 1e-15:
        fi /= s
    return fi

"""
Multi-start LP-FW with subgradient polishing and coordinate descent.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses diverse initializations, adaptive LP constraint relaxation,
ternary line search, and multi-phase polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fs)
    return np.fft.irfft(A * A, fs)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cs - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cs[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        if np.all(np.isfinite(g)) and g.sum() > 1e-18:
            return g
    return None


def line_search(f, g, cur_M, n):
    """Ternary + grid line search for best alpha in [0,1]."""
    best_alpha = 0.0
    best_val = cur_M

    alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]
    for alpha in alphas:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = float(fast_convolve(ft).max())
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha == 0.0:
        return 0.0, cur_M

    lo = max(0.0, best_alpha - 0.12)
    hi = min(1.0, best_alpha + 0.12)
    for _ in range(10):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.maximum(f + m1 * (g - f), 0)
        f2 = np.maximum(f + m2 * (g - f), 0)
        v1 = cur_M
        v2 = cur_M
        if f1.sum() > 1e-15:
            v1 = float((fast_convolve(f1 / f1.sum())).max())
        if f2.sum() > 1e-15:
            v2 = float((fast_convolve(f2 / f2.sum())).max())
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        if v1 < best_val:
            best_val = v1
            best_alpha = m1
        if v2 < best_val:
            best_val = v2
            best_alpha = m2

    return best_alpha, best_val


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation and noise restarts."""
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    b = fast_convolve(f)
    M = float(b.max())
    best_f = f.copy()
    best_M = M

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.15
            if m_relax > 8.0:
                no_improve += 2
            if no_improve > 40:
                break
            continue

        gs = g.sum()
        if gs < 1e-15:
            m_relax *= 1.1
            if m_relax > 8.0:
                no_improve += 2
            if no_improve > 40:
                break
            continue

        g = g / gs
        m_relax = 1.0

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        alpha, val = line_search(f, g, M, n)

        if val < M - 1e-13:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = val
            if M < best_M - 1e-15:
                best_M = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: M={best_M:.10f}, obj={2*n*best_M:.8f}")
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 2.5)

        if no_improve > 50:
            rng = np.random.default_rng(it * 104729 + 31)
            scale = 0.001 + 0.004 * min(no_improve // 50, 6)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            M = float(fast_convolve(f).max())
            m_relax = 1.0
            no_improve = 0

    return best_f, best_M


def subgradient_polish(f0, time_limit, lr=0.004, mom=0.92, temp=5e-5, top_k=40):
    """Subgradient method with softmax-weighted peaks and adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    b = fast_convolve(f)
    best_M = float(b.max())
    best_f = f.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        k_act = min(top_k, len(b))
        peak_idx = np.argpartition(b, -k_act)[-k_act:]
        peak_vals = b[peak_idx]

        scaled = peak_vals / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_idx))
            w[np.argmax(peak_idx)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                js = np.arange(j_lo, j_hi + 1)
                sg[js] += 2.0 * w[i] * f[k - js]

        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        b_new = fast_convolve(f_new)
        cur_M = float(b_new.max())

        if cur_M < best_M - 1e-15:
            best_M = cur_M
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003
        if cur_lr < 5e-7:
            cur_lr = 5e-7

        if no_improve > 300:
            rng = np.random.default_rng((int(time.monotonic() * 10000) + it) % 2**31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 1e-5)
            no_improve = 0

    return best_f, best_M


def coord_descent_polish(f0, time_limit):
    """Coordinate descent: try multiplier changes per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_M = float(fast_convolve(f).max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = np.argsort(f)[::-1]

        for j in order[:min(n, 400)]:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-11:
                continue

            old_val = f[j]
            best_local = old_val
            best_local_M = best_M

            for mult in [0.5, 0.7, 0.85, 1.1, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val <= 0:
                    continue
                trial = f.copy()
                trial[j] = new_val
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                val = float(fast_convolve(trial).max())
                if val < best_local_M - 1e-15:
                    best_local_M = val
                    best_local = new_val

            if best_local != old_val:
                f[j] = best_local
                f = f / f.sum()
                best_M = best_local_M
                best_f = f.copy()
                improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_M


def interpolate_seq(f, new_n):
    """Linear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    s = fi.sum()
    if s > 1e-15:
        fi = fi / s
    return fi

            gv = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
            inits.append(gv / gv.sum())

        rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
        inits.append(rc / rc.sum())

        par = np.maximum(1.0 - (2.0 * x - 1.0) ** 2, 0)
        inits.append(par / par.sum())

        for ns in [0.02, 0.05, 0.08]:
            noisy = best_seq + rng.standard_normal(n_best) * ns
            inits.append(project_simplex(noisy))

        max_diverse = 4
        for di in range(min(max_diverse, len(inits))):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t_alloc = min(rem / (max_diverse - di), 70)
            if t_alloc < 8:
                break
            print(f"  Init {di}, t={t_alloc:.0f}s")
            f_opt, _ = lp_fw_optimize(inits[di], time_limit=t_alloc)
            eval_and_track(f_opt, f"div_{di}")

    # === Phase 4: Deep LP-FW on current best ===
    print(f"\n=== Phase 4: Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_alloc = min(rem * 0.28, 250)
            print(f"  t={t_alloc:.0f}s")
            f_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            eval_and_track(f_opt, "deep_lpfw")

    # === Phase 5: Subgradient polishing ===
    print(f"\n=== Phase 5: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.22, 140)
            if t_alloc > 10:
                print(f"  SG coarse, t={t_alloc:.0f}s")
                f_opt, _ = subgradient_polish(
                    best_seq, time_limit=t_alloc,
                    lr=0.004, mom=0.92, temp=5e-5,
                    top_k=max(20, n_best // 8)
                )
                eval_and_track(f_opt, "sg_coarse")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_alloc = min(rem * 0.18, 100)
                if t_alloc > 8:
                    print(f"  SG fine, t={t_alloc:.0f}s")
                    f_opt, _ = subgradient_polish(
                        best_seq, time_limit=t_alloc,
                        lr=0.001, mom=0.95, temp=2e-5,
                        top_k=max(15, n_best // 5)
                    )
                    eval_and_track(f_opt, "sg_fine")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_alloc = min(rem * 0.25, 100)
            if t_alloc > 8:
                print(f"  CD, t={t_alloc:.0f}s")
                f_opt, _ = coord_descent_polish(best_seq, time_limit=t_alloc)
                eval_and_track(f_opt, "cd")

    # === Phase 7: Final polish cycle ===
    print(f"\n=== Phase 7: Final (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 8:
            t1 = min(rem * 0.35, 50)
            if t1 > 5:
                f_opt, _ = lp_fw_optimize(best_seq, time_limit=t1)
                eval_and_track(f_opt, "final_lpfw")

        rem = deadline - time.monotonic()
        if rem > 8:
            t2 = min(rem * 0.35, 50)
            if t2 > 5:
                f_opt, _ = subgradient_polish(
                    best_seq, time_limit=t2,
                    lr=0.0005, mom=0.96, temp=1e-5,
                    top_k=max(10, len(best_seq) // 5)
                )
                eval_and_track(f_opt, "final_sg")

        rem = deadline - time.monotonic()
        if rem > 5:
            t3 = min(rem * 0.5, 30)
            if t3 > 3:
                f_opt, _ = coord_descent_polish(best_seq, time_limit=t3)
                eval_and_track(f_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
