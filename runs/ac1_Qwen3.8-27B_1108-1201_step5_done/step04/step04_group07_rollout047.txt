2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key innovations: adaptive softmax peak targeting, Nesterov momentum,
entropy mirror descent polishing, structured restarts, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, temp):
    """Compute weighted subgradient targeting top_k peaks with softmax."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / max(temp, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def subgradient_nesterov(f0, time_limit, lr=0.004, mom=0.93, top_k=50,
                         temp0=0.001, temp_growth=1.003, cooldown=200):
    """Nesterov-accelerated subgradient descent with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov momentum variable
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temp0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov: evaluate at y = f + mom*(f - f_prev), approximated by f + mom*v/norm
        # Simplified: use momentum directly
        conv = fast_convolve(y)
        sg = compute_subgradient(y, conv, top_k, cur_temp)

        # Update
        v = mom * v - lr * sg
        f_new = project_simplex(y + v)
        y = f_new  # For next iteration's lookahead

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        cur_temp *= temp_growth
        lr *= 0.9998

        if no_improve > cooldown:
            # Structured restart: try a few different perturbation types
            restart_type = it % 4
            if restart_type == 0:
                # Random noise
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            elif restart_type == 1:
                # Shift-like perturbation (circular shift in frequency)
                F = np.fft.rfft(f)
                phase = rng.uniform(-0.1, 0.1, size=len(F))
                F_shifted = F * np.exp(1j * phase)
                f_new = np.real(np.fft.irfft(F_shifted, 2 * n)[:n])
                f = project_simplex(np.maximum(f_new, 0))
            elif restart_type == 2:
                # Inversion perturbation
                f_inv = project_simplex(best_f[::-1].copy())
                f = project_simplex(0.7 * best_f + 0.3 * f_inv)
            else:
                # Block swap perturbation
                f_copy = best_f.copy()
                block = max(1, n // 8)
                i = rng.integers(0, n - 2 * block)
                f_copy[i:i+block], f_copy[i+block:i+2*block] = f_copy[i+block:i+2*block].copy(), f_copy[i:i+block].copy()
                f = project_simplex(f_copy)

            v = np.zeros(n)
            y = f.copy()
            lr = max(lr * 0.7, 0.001)
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr=0.002, top_k=30, temp=0.0005,
                            cooldown=150):
    """Entropy mirror descent (multiplicative weights) for polishing."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temp
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        sg = compute_subgradient(f, conv, top_k, cur_temp)

        # Mirror descent update: f_new[i] ∝ f[i] * exp(-lr * sg[i])
        log_f = np.log(f)
        log_f_new = log_f - lr * sg
        # Normalize
        log_f_new -= np.max(log_f_new)
        exp_f_new = np.exp(log_f_new)
        f_new = exp_f_new / exp_f_new.sum()

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        cur_temp *= 1.005
        lr *= 0.9997

        if no_improve > cooldown:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            lr = max(lr * 0.8, 0.0005)
            no_improve = 0

    return best_f, best_obj


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=None):
    """Solve linearized LP: min t s.t. t >= conv_f[k] + 2*(B*g)[k], g >= -f, sum(g)=0."""
    n = len(f)
    if top_k is None or top_k >= len(conv_f):
        peaks = np.arange(len(conv_f))
    else:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks = np.sort(peaks)

    B_full = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B_full[peaks, :]
    b_ub = -conv_f[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def lp_frank_wolfe(f0, time_limit, use_maxsum=False, reduced_k=30, verbose=False):
    """Frank-Wolfe with LP oracle (linearized or max-sum)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(31415)

    while time.monotonic() - t0 < time_limit:
        it += 1

        if use_maxsum:
            g = solve_maxsum_lp(f, M * m_exp)
            if g is None:
                m_exp *= 1.12
                if m_exp > 4.0:
                    m_exp = 1.0
                    no_improve += 2
                continue
            g = g / g.sum()
        else:
            conv_cur = fast_convolve(f)
            g = solve_linearized_lp(f, conv_cur, top_k=reduced_k)
            if g is None:
                no_improve += 3
                if no_improve > 50:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                    no_improve = 0
                continue
            target = f + g
            target = np.maximum(target, 0.0)
            ts = target.sum()
            if ts < 1e-15:
                no_improve += 1
                continue
            g = target / ts

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 2
            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        alphas = np.linspace(0.05, 1.0, 15)
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Refine
            lo = max(0.0, best_alpha - 0.07)
            hi = min(1.0, best_alpha + 0.07)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 60:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    cx = 0.5 * (n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian bumps
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - cx) / sigma) ** 2)
        inits.append(g / g.sum())

    # Two peaks
    for gap_frac in [0.15, 0.25, 0.35]:
        gap = gap_frac * n
        g1 = np.exp(-0.5 * ((x - cx + gap) / (0.1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - cx - gap) / (0.1 * n)) ** 2)
        g = g1 + g2
        inits.append(g / g.sum())

    # Cosine
    c1 = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # U-shape
    d = np.abs(x - cx) / n
    u = 1.0 / (d + 0.1)
    inits.append(u / u.sum())

    # Beta-like
    for ab in [(1.5, 1.5), (2, 3), (3, 2), (1, 2), (2, 1)]:
        b = (x + 1) ** ab[0] * (n - x) ** ab[1]
        b = np.maximum(b, 1e-10)
        inits.append(b / b.sum())

    # Frequency-based: design |A|^2 to be concentrated
    # a = IFFT(sqrt(max(|A|^2, 0))) projected to non-negative
    A = np.zeros(n // 2 + 1)
    A[0] = 1.0
    for k in range(1, min(5, n // 2 + 1)):
        A[k] = 1.0 / (1 + 0.5 * k)
    a_freq = np.abs(np.fft.irfft(A * A, n))
    a_freq = np.maximum(a_freq, 0)
    if a_freq.sum() > 1e-10:
        inits.append(a_freq / a_freq.sum())

    # Sinc-like window
    w = np.sinc(x / (n // 4) - 1)
    w = np.abs(w) + 0.01
    inits.append(w / w.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def local_search(f0, time_limit, rng, num_per_round=200):
    """Local search with adaptive-scale random perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    scale = 0.003

    while time.monotonic() - t0 < time_limit:
        for _ in range(num_per_round):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f_pert.copy()
                scale = min(scale * 1.05, 0.01)

        scale *= 0.93
        if scale < 1e-5:
            scale = 0.002
            f_jump = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            obj = fast_convolve(f_jump).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_jump.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid optimizer:
    Phase 1: Multi-length initialization
    Phase 2: Fast Nesterov subgradient (broad exploration)
    Phase 3: LP-based Frank-Wolfe (targeted refinement)
    Phase 4: Mirror descent + fine subgradient (polishing)
    Phase 5: Local search (final polish)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Multi-length initialization ===
    print("\n=== Phase 1: Initialization ===")

    lengths_to_try = []
    if parent is not None:
        n_parent = len(parent)
        lengths_to_try = [n_parent, n_parent - 16, n_parent + 16, n_parent - 32, n_parent + 32]
    else:
        lengths_to_try = [256, 320, 384, 448, 512]

    # Filter valid lengths
    lengths_to_try = [n for n in lengths_to_try if 64 <= n <= 1024]
    # Remove duplicates
    seen = set()
    unique_lengths = []
    for n in lengths_to_try:
        if n not in seen:
            seen.add(n)
            unique_lengths.append(n)
    lengths_to_try = unique_lengths

    phase1_budget = 60.0
    per_length_time = max(5.0, phase1_budget / len(lengths_to_try))

    candidates_by_length = {}

    for n in lengths_to_try:
        if time.monotonic() > deadline - 500:
            break
        inits = generate_structured_inits(n, parent, rng)
        best_init = None
        best_init_obj = float('inf')

        for i, ini in enumerate(inits):
            if time.monotonic() > deadline - 500:
                break
            obj = fast_convolve(ini).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = ini

        if best_init is not None:
            candidates_by_length[n] = best_init
            # Quick subgradient pass
            t_sub = min(per_length_time * 0.6, 20)
            a_opt, _ = subgradient_nesterov(best_init, time_limit=t_sub,
                                            lr=0.005, mom=0.92, top_k=max(20, n // 10),
                                            temp0=0.002, temp_growth=1.005, cooldown=100)
            try_seq(a_opt, f"init_n{n}")

    # Ensure we have a best
    if best_seq is None:
        n_def = 384
        best_seq = np.ones(n_def) / n_def
        try_seq(best_seq, "fallback_uniform")
        candidates_by_length[n_def] = best_seq.copy()

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: Fast Nesterov subgradient ===
    print("\n=== Phase 2: Nesterov Subgradient ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 400
    t_pw2 = min(remaining * 0.35, 500)

    if t_pw2 > 20:
        print(f"  Budget: {t_pw2:.0f}s, n={n_best}")

        # Run with different hyperparameters
        t_each = t_pw2 / 3
        configs = [
            dict(lr=0.004, mom=0.93, top_k=n_best // 8, temp0=0.001, temp_growth=1.003, cooldown=150),
            dict(lr=0.006, mom=0.90, top_k=n_best // 4, temp0=0.0005, temp_growth=1.004, cooldown=200),
            dict(lr=0.003, mom=0.95, top_k=n_best // 12, temp0=0.002, temp_growth=1.002, cooldown=120),
        ]

        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 300
            if rem < 15:
                break
            t_alloc = min(t_each, rem * 0.3)
            a_opt, _ = subgradient_nesterov(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"sg_cfg{ci}")

        # Also try on a different length if time permits
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            for n_alt in [n_best + 32, n_best - 32]:
                if n_alt < 64 or n_alt > 1024:
                    continue
                if n_alt in candidates_by_length:
                    a_alt = candidates_by_length[n_alt]
                else:
                    a_alt = np.interp(np.linspace(0, 1, n_alt),
                                      np.linspace(0, 1, n_best), best_seq)
                    a_alt = np.maximum(a_alt, 0)
                    a_alt = a_alt / a_alt.sum()
                a_opt, _ = subgradient_nesterov(a_alt, time_limit=min(rem * 0.4, 40),
                                                lr=0.004, mom=0.93, top_k=n_alt // 8,
                                                temp0=0.001, temp_growth=1.003, cooldown=100)
                try_seq(a_opt, f"sg_alt_n{n_alt}")
                rem = deadline - time.monotonic() - 200
                if rem < 20:
                    break

    print(f"  Phase 2 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 3: LP-based Frank-Wolfe ===
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 300
    t_pw3 = min(remaining * 0.4, 500)

    if t_pw3 > 30:
        print(f"  Budget: {t_pw3:.0f}s, n={n_best}")

        # Linearized LP FW (reduced constraints for speed)
        t_lin = t_pw3 * 0.5
        a_opt, _ = lp_frank_wolfe(best_seq, time_limit=t_lin, use_maxsum=False,
                                  reduced_k=25, verbose=True)
        try_seq(a_opt, "lp_lin")

        # Max-sum LP FW
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_ms = min(rem * 0.6, 250)
            a_opt, _ = lp_frank_wolfe(best_seq, time_limit=t_ms, use_maxsum=True, verbose=True)
            try_seq(a_opt, "lp_maxsum")

        # Perturbed restart + LP
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            noise = rng.standard_normal(n_best) * 0.005
            f_pert = project_simplex(best_seq + noise)
            t_rs = min(rem * 0.5, 150)
            a_opt, _ = lp_frank_wolfe(f_pert, time_limit=t_rs, use_maxsum=False,
                                      reduced_k=20, verbose=False)
            try_seq(a_opt, "lp_restart")

    print(f"  Phase 3 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 4: Mirror descent + fine subgradient ===
    print("\n=== Phase 4: Mirror Descent + Fine Polish ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 150
    t_pw4 = min(remaining * 0.5, 300)

    if t_pw4 > 20:
        print(f"  Budget: {t_pw4:.0f}s, n={n_best}")

        # Mirror descent
        t_md = t_pw4 * 0.4
        a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md,
                                           lr=0.003, top_k=n_best // 10, temp=0.0005)
        try_seq(a_opt, "mirror1")

        # Fine subgradient (small lr, many peaks)
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_sf = min(rem * 0.6, 150)
            a_opt, _ = subgradient_nesterov(best_seq, time_limit=t_sf,
                                            lr=0.001, mom=0.96, top_k=n_best // 3,
                                            temp0=0.0003, temp_growth=1.002, cooldown=300)
            try_seq(a_opt, "sg_fine")

        # Another mirror pass on current best
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_md2 = min(rem * 0.7, 100)
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md2,
                                               lr=0.001, top_k=n_best // 5, temp=0.0002)
            try_seq(a_opt, "mirror2")

    print(f"  Phase 4 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 5: Local search + final LP ===
    print("\n=== Phase 5: Local Search ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 40
    t_pw5 = min(remaining * 0.7, 200)

    if t_pw5 > 15:
        print(f"  Budget: {t_pw5:.0f}s, n={n_best}")

        # Local search
        t_ls = t_pw5 * 0.5
        a_opt, _ = local_search(best_seq, time_limit=t_ls, rng=rng)
        try_seq(a_opt, "local")

        # Final LP polish
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_final = min(rem * 0.8, 100)
            a_opt, _ = lp_frank_wolfe(best_seq, time_limit=t_final, use_maxsum=False,
                                      reduced_k=15, verbose=False)
            try_seq(a_opt, "final_lp")

        # One more local search on result
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = local_search(best_seq, time_limit=min(rem, 30), rng=rng)
            try_seq(a_opt, "final_local")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"FALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
