Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Structured initialization (windowed, periodic, parent-based)
2. Adaptive subgradient descent with multi-peak softmax weighting

4. Simulated annealing for escaping local minima

import numpy as np
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(f):
    """Compute the objective value for a normalized (sum=1) non-negative sequence."""
    n = len(f)
    conv = fast_convolve(f)
    return conv.max(), conv


def subgradient_step(f, conv, peaks, weights, lr):
    """Compute subgradient and apply one step."""
    n = len(f)
    sg = np.zeros(n)
    for k in peaks:
        w = weights[np.argmax(peaks == k)] if len(peaks) == 1 else 0.0
    # More efficient: iterate in order
    sg = np.zeros(n)
    for i in range(len(peaks)):
        w = weights[i]
        if w < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * f[k - idx]
    return sg


def subgradient_optimize(f0, time_limit, lr_init=0.002, momentum=0.92,
                         top_k=50, temp_init=0.0001, verbose=False):
    """Adaptive subgradient descent with softmax multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 500 == 0 and verbose:
            print(f"      SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}, temp={cur_temp:.6f}")

        conv = fast_convolve(f)
        ck = len(conv)
        if top_k >= ck:
            peaks = np.arange(ck)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        if not np.all(np.isfinite(f_new)):
            f_new = f.copy()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0
        else:
            f = f_new.copy()
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, sweeps=10, verbose=False):
    """Coordinate descent with near-exact 1D optimization per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    S = 1.0  # sum is always 1

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            ai = f[i]
            if ai < 1e-15:
                # Try setting to a small value
                D = S - ai
                # M_other
                k_min = i
                k_max = min(i + n - 1, 2 * n - 2)
                if k_min <= k_max:
                    other_mask = np.ones(2 * n - 1, dtype=bool)
                    other_mask[k_min:k_max + 1] = False
                    M_other = conv[other_mask].max() if other_mask.any() else 0.0
                else:
                    M_other = conv.max()

                # Try a few candidate x values
                best_x = 0.0
                best_val = float('inf')
                for x_try in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    if D + x_try < 1e-10:
                        continue
                    # Compute new affected conv values
                    J = min(n - 1, 2 * n - 2 - i)
                    new_affected = conv[i:i + J + 1] + (x_try - 0.0) * f[:J + 1]
                    M_affected = new_affected.max()
                    M_total = max(M_other, M_affected)
                    val = M_total / (D + x_try) ** 2
                    if val < best_val:
                        best_val = val
                        best_x = x_try

                if best_x > 0 and best_val < best_obj / (2 * n) - 1e-14:
                    f[i] = best_x
                    S = D + best_x
                    # Update conv
                    conv[i:i + J + 1] += best_x * f[:J + 1]
                    best_obj = conv.max()
                    if best_obj < best_obj - 1e-14:  # always true if improved
                        best_f = f.copy()
                        best_obj = best_obj
                    improved = True
                continue

            D = S - ai
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)

            other_mask = np.ones(2 * n - 1, dtype=bool)
            other_mask[k_min:k_max + 1] = False
            M_other = conv[other_mask].max() if other_mask.any() else 0.0

            J = min(n - 1, 2 * n - 2 - i)
            # C_j = conv[i+j], a_j = f[j]
            C = conv[i:i + J + 1].copy()
            a_sub = f[:J + 1]

            # M_affected(x) = max_j (C_j + (x - ai) * a_j)
            # = max_j ((C_j - ai*a_j) + x * a_j)
            intercepts = C - ai * a_sub  # C_j - ai * a_j
            slopes = a_sub.copy()

            # Find candidate x values
            candidates = [0.0, ai]

            # x where each line hits M_other
            for j in range(J + 1):
                if slopes[j] > 1e-15:
                    x_hit = (M_other - intercepts[j]) / slopes[j]
                    if x_hit > 0:
                        candidates.append(x_hit)

            # x that minimizes each line/(D+x)^2: derivative zero
            # d/dx [(intercept + slope*x)/(D+x)^2] = 0
            # slope*(D+x)^2 - (intercept + slope*x)*2*(D+x) = 0
            # (D+x)[slope*(D+x) - 2*(intercept + slope*x)] = 0
            # slope*D + slope*x - 2*intercept - 2*slope*x = 0
            # -slope*x = 2*intercept - slope*D
            # x = (slope*D - 2*intercept) / slope
            for j in range(J + 1):
                if slopes[j] > 1e-15:
                    x_opt = (slopes[j] * D - 2.0 * intercepts[j]) / slopes[j]
                    if x_opt > 0:
                        candidates.append(x_opt)

            # Evaluate f at candidates
            best_x = ai
            best_val = float('inf')
            for x in candidates:
                if x < 0:
                    x = 0.0
                # M_affected at x
                vals = intercepts + x * slopes
                M_aff = vals.max()
                M_tot = max(M_other, M_aff)
                denom = (D + x) ** 2
                if denom < 1e-20:
                    continue
                val = M_tot / denom
                if val < best_val:
                    best_val = val
                    best_x = x

            # Also try a small grid around current and around best candidate
            if best_x != ai:
                for x_try in [ai * 0.5, ai * 1.5, best_x * 0.9, best_x * 1.1]:
                    if x_try < 0:
                        continue
                    vals = intercepts + x_try * slopes
                    M_aff = vals.max()
                    M_tot = max(M_other, M_aff)
                    denom = (D + x_try) ** 2
                    if denom < 1e-20:
                        continue
                    val = M_tot / denom
                    if val < best_val:
                        best_val = val
                        best_x = x_try

            # Golden section search to refine
            if best_val < S * S * best_obj / (S * S) - 1e-15:  # i.e., best_val < best_obj
                lo = max(0.0, min(ai, best_x) * 0.8)
                hi = max(ai, best_x) * 1.2 + 0.01

                for _ in range(8):
                    m1 = lo + 0.382 * (hi - lo)
                    m2 = lo + 0.618 * (hi - lo)
                    v1 = (intercepts + m1 * slopes).max()
                    v2 = (intercepts + m2 * slopes).max()
                    f1 = max(M_other, v1) / (D + m1) ** 2
                    f2 = max(M_other, v2) / (D + m2) ** 2
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1

                x_refined = (lo + hi) / 2.0
                v_r = (intercepts + x_refined * slopes).max()
                M_tot_r = max(M_other, v_r)
                val_r = M_tot_r / (D + x_refined) ** 2
                if val_r < best_val:
                    best_val = val_r
                    best_x = x_refined

            # Apply update if improved
            target_obj = best_val
            current_obj = best_obj
            if target_obj < current_obj * (1.0 - 1e-10):
                delta = best_x - ai
                f[i] = best_x
                conv[i:i + J + 1] += delta * f[:J + 1]
                S = D + best_x
                new_max = conv.max()
                if new_max < best_obj:
                    best_obj = new_max
                    best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_obj


def simulated_annealing(f0, time_limit, rng, t_init=0.01, t_min=0.0001,
                        cooling=0.9995, verbose=False):
    """Simulated annealing with convolution peak as energy."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    cur_obj = conv.max()
    best_obj = cur_obj
    best_f = f.copy()

    t0 = time.monotonic()
    temp = t_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 2000 == 0 and verbose:
            print(f"      SA iter {it}: best={best_obj:.10f}, temp={temp:.6f}")

        # Random perturbation: pick k random coordinates and adjust
        k = rng.integers(1, max(2, n // 10))
        idx = rng.integers(0, n, size=k)
        noise = rng.standard_normal(k) * temp
        f_pert = f.copy()
        f_pert[idx] = np.maximum(f_pert[idx] + noise, 0.0)
        fs = f_pert.sum()
        if fs < 1e-15:
            continue
        f_pert = f_pert / fs

        conv_pert = fast_convolve(f_pert)
        obj_pert = conv_pert.max()

        delta = obj_pert - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            f = f_pert
            cur_obj = obj_pert
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_f = f.copy()

        temp = max(temp * cooling, t_min)
        if temp <= t_min * 1.01:
            # Restart with larger temperature
            temp = t_init * 0.5
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            cur_obj = fast_convolve(f).max()

    return best_f, best_obj


def generate_structured_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Windowed (raised cosine)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    w = np.maximum(w, 1e-6)
    inits.append(w / w.sum())

    # Gaussian centered
    for sigma_frac in [0.15, 0.25, 0.35, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine window
    c = 0.5 * (1 + np.cos(np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    c = np.maximum(c, 1e-6)
    inits.append(c / c.sum())

    # S-shaped (tanh)
    t = np.tanh(3.0 * (x / n - 0.5))
    t = 0.5 * (t + 1)
    t = np.maximum(t, 1e-4)
    inits.append(t / t.sum())

    # Periodic modulation
    for freq in [2, 3, 5]:
        p = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n)
        p = np.maximum(p, 1e-4)
        inits.append(p / p.sum())

    # Beta-like
    for alpha_b in [1.5, 2.0, 3.0]:
        b = (x + 1) ** alpha_b * (n - x) ** alpha_b
        b = np.maximum(b, 1e-4)
        inits.append(b / b.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
    elif parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    candidates = []

    if parent is not None:
        candidates.append(("parent", parent, len(parent)))
        n_p = len(parent)
    else:
        n_p = 384
        candidates.append(("uniform", np.ones(n_p) / n_p, n_p))

    # Generate structured inits at parent length
    if parent is not None:
        inits = generate_structured_inits(n_p, parent, rng)
        for i, ini in enumerate(inits):
            candidates.append((f"struct_{i}", ini, n_p))

    # Try a few other lengths
    for n_try in [320, 400, 448, 512]:
        if n_try == n_p:
            continue
        inits = generate_structured_inits(n_try, parent, rng)
        candidates.append((f"len{n_try}_win", inits[1], n_try))
        candidates.append((f"len{n_try}_gauss", inits[3], n_try))

    # Evaluate all candidates
    for label, cand, n in candidates:
        if time.monotonic() > deadline - 300:
            break
        s = cand.sum()
        if s < 1e-10:
            continue
        try_seq(cand / s, f"init_{label}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: Subgradient optimization ===
    print("\n=== Phase 2: Subgradient Optimization ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 250

    # Run subgradient on best sequence
    t_sg = min(remaining * 0.4, 600)
    if t_sg > 30:
        print(f"  SG run 1 (n={n_best}), budget={t_sg:.0f}s")
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg * 0.6,
                                        lr_init=0.003, momentum=0.93,
                                        top_k=max(20, n_best // 8),
                                        temp_init=0.00005, verbose=True)
        try_seq(a_opt, "sg1")

        # Finer SG
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_sg2 = min(rem * 0.5, 200)
            print(f"  SG run 2 (finer), budget={t_sg2:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg2,
                                            lr_init=0.001, momentum=0.95,
                                            top_k=max(30, n_best // 4),
                                            temp_init=0.00002, verbose=True)
            try_seq(a_opt, "sg2")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Coordinate Descent ===
    print("\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 150
    t_cd = min(remaining * 0.4, 400)

    if t_cd > 20:
        print(f"  CD run (n={n_best}), budget={t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        try_seq(a_opt, "cd")

        # Another CD round
        rem = deadline - time.monotonic() - 120
        if rem > 15 and best_seq is not None:
            t_cd2 = min(rem * 0.5, 150)
            print(f"  CD run 2, budget={t_cd2:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd2)
            try_seq(a_opt, "cd2")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Simulated Annealing ===
    print("\n=== Phase 4: Simulated Annealing ===")
    remaining = deadline - time.monotonic() - 100
    t_sa = min(remaining * 0.35, 250)

    if t_sa > 15:
        print(f"  SA run, budget={t_sa:.0f}s")
        a_opt, _ = simulated_annealing(best_seq, time_limit=t_sa * 0.7,
                                       rng=rng, t_init=0.008, t_min=0.00005,
                                       cooling=0.9997, verbose=True)
        try_seq(a_opt, "sa")

        # Second SA with different params
        rem = deadline - time.monotonic() - 80
        if rem > 10:
            a_opt, _ = simulated_annealing(best_seq, time_limit=min(rem * 0.6, 100),
                                           rng=rng, t_init=0.003, t_min=0.00002,
                                           cooling=0.9999, verbose=False)
            try_seq(a_opt, "sa2")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final Polish ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic() - 40

    if remaining > 15:
        # Local refinement
        t_lr = min(remaining * 0.4, 100)
        print(f"  Local refine, budget={t_lr:.0f}s")
        a_opt, _ = local_refine(best_seq, time_limit=t_lr, rng=rng)
        try_seq(a_opt, "local")

        # Quick coordinate descent
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            t_cd3 = min(rem * 0.5, 60)
            print(f"  Final CD, budget={t_cd3:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd3)
            try_seq(a_opt, "cd_final")

        # One more subgradient pass
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_sg3 = min(rem * 0.6, 40)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg3,
                                            lr_init=0.0005, momentum=0.96,
                                            top_k=max(20, n_best // 5),
                                            temp_init=0.00001, verbose=False)
            try_seq(a_opt, "sg_final")

    # === Try alternate lengths ===
    rem = deadline - time.monotonic() - 15
    if rem > 20 and best_seq is not None:
        print("\n=== Phase 6: Alternate Lengths ===")
        for n_try in [n_best - 16, n_best + 16, n_best + 32]:
            if n_try < 64 or rem < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, len(best_seq)), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps <= 1e-10:
                continue
            px = px / ps
            # Quick optimization
            t_quick = min(rem * 0.3, 30)
            a_opt, _ = subgradient_optimize(px, time_limit=t_quick * 0.6,
                                            lr_init=0.002, momentum=0.93,
                                            top_k=max(15, n_try // 10),
                                            temp_init=0.00003, verbose=False)
            try_seq(a_opt, f"len{n_try}")
            rem = deadline - time.monotonic() - 12
            if rem < 10:
                break

    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
