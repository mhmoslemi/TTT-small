"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Log-sum-exp smooth optimization with L-BFGS (primary, smooth objective)
2. Spectral flattening via alternating projections in frequency domain
3. Coordinate descent for fine polishing
4. Frank-Wolfe with reduced LP oracle for larger jumps
5. Multi-start exploration with different lengths and initializations
Key innovation: smooth log-sum-exp surrogate enables use of second-order methods,
combined with frequency-domain spectral flattening for global exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def logsumexp_objective_and_grad(x, epsilon, conv_cache=None):
    """
    Compute log-sum-exp objective and gradient for the normalized sequence.
    x is projected onto simplex internally.
    Objective: epsilon * log(sum_k exp(conv_k / epsilon))
    """
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(x)) / len(x)
    else:
        a = a / s

    conv = fast_convolve(a)
    n = len(a)

    scaled = conv / epsilon
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    obj = epsilon * (np.log(exp_vals.sum()) + scaled.max())

    # Weights
    w = exp_vals / exp_vals.sum()

    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
    # Using Toeplitz structure
    idx_i = np.arange(n)[:, None]
    idx_j = np.arange(n)[None, :]
    indices = idx_i + idx_j
    valid = indices < len(w)
    W = np.where(valid, w[indices], 0.0)
    grad = 2.0 * (W @ a)

    # Project gradient onto tangent space of simplex (remove mean component)
    # Since we're optimizing over the simplex, we need grad - mean(grad)
    grad = grad - grad.mean()

    return obj, grad


def logsumexp_lbfgs(a0, time_limit, epsilon_start=0.05, epsilon_end=0.002,
                    n_phases=5, verbose=False):
    """
    Multi-phase L-BFGS optimization using log-sum-exp surrogate.
    Each phase uses a smaller epsilon (sharper approximation to max).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_convolve(a).max()

    # Epsilon schedule (geometric)
    epsilons = np.geomspace(epsilon_start, epsilon_end, n_phases)

    for phase, eps in enumerate(epsilons):
        if time.monotonic() - t0 > time_limit * 0.95:
            break

        phase_time = time_limit / max(1, n_phases - phase) * 0.9

        def fun_and_grad(x):
            obj, grad = logsumexp_objective_and_grad(x, eps)
            return obj, grad

        # Use L-BFGS-B with bounds (all zero or positive)
        bounds = [(0.0, None)] * n

        result = minimize(fun_and_grad, a, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 500,
                                                   'ftol': 1e-12,
                                                   'gtol': 1e-8,
                                                   'maxfun': 2000})

        a_new = np.maximum(result.x, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        score_new = conv_new.max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            if verbose:
                print(f"    LBF phase {phase}: eps={eps:.5f}, score={best_score:.10f}")
        else:
            # Keep old a, try smaller step
            if verbose:
                print(f"    LBF phase {phase}: eps={eps:.5f}, no improvement")

        # Check time
        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.9:
            break

    return best_a, best_score


def spectral_flatten(a0, time_limit, n_iter=500, verbose=False):
    """
    Alternating projection: flatten |A|^2 in frequency domain,
    project back to non-negative simplex in time domain.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    no_improve = 0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute DFT of zero-padded signal
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Flatten: compress dynamic range
        # Use log compression: new_power = (log(1 + power) / max_log) ** alpha
        max_power = power.max()
        if max_power < 1e-30:
            break
        log_power = np.log1p(power)
        max_log = log_power.max()
        # Compress: new_power = max_power * (log_power / max_log) ** 2
        compressed = max_power * (log_power / max_log) ** 2
        # Blend: keep some of original
        new_power = 0.7 * compressed + 0.3 * power

        # Reconstruct with same phase, new magnitude
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * phase
        # Make it conjugate symmetric
        A_full = np.zeros(fft_size, dtype=complex)
        half = fft_size // 2 + 1
        A_full[:half] = A_new
        A_full[half:] = A_new[1:half][::-1].conjugate() if fft_size > 1 else 0

        a_new = np.fft.irfft(A_full)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        score = fast_convolve(a_new).max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    Spectral iter {it}: score={best_score:.10f}")
        else:
            no_improve += 1
            # Adaptively adjust compression
            if no_improve > 50:
                # Try less compression
                a = best_a.copy()
                no_improve = 0

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Cyclic coordinate descent on the log-sum-exp smoothed objective.
    Each coordinate update solves a 1D problem.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_convolve(a).max()

    epsilon = 0.01  # Smoothing parameter
    improvement_factor = 0.95

    for sweep in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Minimize over a_i, keeping others fixed (on simplex)
            # The simplex constraint: sum_j a_j = 1, a_j >= 0
            # If we change a_i, we need to redistribute: a_i + delta = new_a_i
            # and scale others by (1 - new_a_i) / (1 - a_i) if a_i < 1
            # Simpler: just optimize a_i directly with the constraint that
            # sum = 1 means we scale the rest.

            # Actually, simplest: fix all j != i proportionally
            # a_old_sum_others = 1 - a[i]
            # new_a[i] = t, others scaled by (1-t)/(1-a[i])

            a_i_old = a[i]
            if a_i_old < 1e-10 or a_i_old > 1.0 - 1e-10:
                continue

            # Try a few values of t
            best_t = a_i_old
            best_obj = np.inf

            # Golden section search in [0, 0.9999]
            lo, hi = 0.0, min(1.0 - 1e-6, 2.0 * a_i_old + 0.05)
            if lo >= a_i_old:
                lo = max(0.0, a_i_old - 0.05)
            if hi <= a_i_old:
                hi = min(1.0 - 1e-6, a_i_old + 0.05)

            for _ in range(20):
                gr = (np.sqrt(5) - 1) / 2
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)

                # Evaluate at m1
                a_test = a.copy()
                scale_others = (1.0 - m1) / (1.0 - a_i_old)
                a_test[:i] = a[:i] * scale_others
                a_test[i] = m1
                a_test[i+1:] = a[i+1:] * scale_others
                obj1 = fast_convolve(a_test).max()

                # Evaluate at m2
                a_test[i] = m2
                a_test[:i] = a[:i] * ((1.0 - m2) / (1.0 - a_i_old))
                a_test[i+1:] = a[i+1:] * ((1.0 - m2) / (1.0 - a_i_old))
                obj2 = fast_convolve(a_test).max()

                if obj1 < obj2:
                    hi = m2
                else:
                    lo = m1

            best_t = (lo + hi) / 2.0
            scale_others = (1.0 - best_t) / (1.0 - a_i_old)
            a_new = a.copy()
            a_new[:i] = a[:i] * scale_others
            a_new[i] = best_t
            a_new[i+1:] = a[i+1:] * scale_others

            score = fast_convolve(a_new).max()
            if score < best_score - 1e-12:
                a = a_new
                best_score = score
                best_a = a.copy()
                improved = True

        if not improved:
            # Reduce epsilon or try larger steps
            epsilon *= 0.9
            if sweep > 100:
                break

    return best_a, best_score


def reduced_lp_fw(a0, time_limit, top_k=20, verbose=False):
    """
    Frank-Wolfe with reduced LP oracle (only top_k peak constraints).
    More aggressive than the log-sum-exp approach.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    no_improve = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        if no_improve > 30:
            # Perturb and restart
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0

        conv = fast_convolve(a)
        current_score = conv.max()

        # Get top_k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Build reduced LP:
        # min t s.t. 2*(B_k @ g) <= M - conv[k] for k in peaks, sum(g)=0, -a <= g <= inf
        # where B_k[g] = (a*g)[k] (a is fixed)
        # Variables: [t, g_0, ..., g_{n-1}]
        c = np.zeros(n + 1)
        c[0] = 1.0

        A_ub = np.zeros((len(peaks), n + 1))
        A_ub[:, 0] = -1.0
        b_ub = np.zeros(len(peaks))

        for idx, k in enumerate(peaks):
            # (a*g)[k] = sum_j a[j] * g[k-j] (valid j)
            # = sum_j g[j] * a[k-j] (valid k-j)
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    A_ub[idx, j + 1] += 2.0 * a[kj]
            b_ub[idx] = -conv[k]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.zeros(1)

        bounds = [(None, None)] + [(-a[i], None) for i in range(n)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 3.0})

        if not res.success:
            no_improve += 2
            continue

        g = res.x[1:]
        target = a + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - a)) < 1e-10:
            no_improve += 1
            continue

        # Line search
        best_alpha = 0.0
        best_val = current_score
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - alpha) * a + alpha * target
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            val = fast_convolve(a_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            a = (1.0 - best_alpha) * a + best_alpha * target
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_val < best_score - 1e-12:
                best_score = best_val
                best_a = a.copy()
                no_improve = 0
                if verbose:
                    print(f"    LP-FW: score={best_score:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_a, best_score


def multi_start_explore(parent, time_limit, rng, verbose=False):
    """
    Try different initializations and lengths to find good starting points.
    """
    best_a = parent.copy()
    best_score = fast_convolve(parent / parent.sum()).max()

    n_parent = len(parent)

    # Try different lengths
    for n_try in [n_parent - 32, n_parent + 32, n_parent // 2, n_parent * 2]:
        if n_try < 64 or n_try > 2048 or time.monotonic() > time.monotonic() + time_limit:
            continue
        try:
            # Interpolate parent to new length
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_parent), parent / parent.sum())
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                score = fast_convolve(px).max()
                if score < best_score:
                    best_score = score
                    best_a = px.copy()
        except:
            continue

    # Try structured initializations
    for n_try in [n_parent, 256, 512]:
        x = np.arange(n_try, dtype=np.float64)
        inits = [
            np.ones(n_try) / n_try,
            np.abs(np.cos(np.pi * x / n_try)) + 0.1,
            np.exp(-0.5 * ((x - n_try / 2) / (n_try * 0.2)) ** 2),
        ]
        for ini in inits:
            ini = np.maximum(ini, 0)
            s = ini.sum()
            if s > 1e-10:
                ini = ini / s
                score = fast_convolve(ini).max()
                if score < best_score:
                    best_score = score
                    best_a = ini.copy()

    return best_a, best_score


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    x = np.arange(n, dtype=np.float64)
    # Various Gaussian widths
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine variants
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) + 0.05
        inits.append(c / c.sum())

    # Chebyshev-like
    t = 2.0 * x / (n - 1) - 1.0
    for degree in [2, 4, 6]:
        c = np.abs(np.cos(degree * np.arccos(np.clip(t, -1, 1)))) + 0.05
        inits.append(c / c.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    print(f"  Phase 3 best: {best_score:.10f}")

    # === Phase 4: Reduced LP Frank-Wolfe ===
    print("\n=== Phase 4: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 100
    t_alloc = min(remaining * 0.2, 250)

    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, _ = reduced_lp_fw(best_seq, time_limit=t_alloc * 0.5,
                                  top_k=25, verbose=True)
        try_seq(a_opt, "lpfw")

        rem = deadline - time.monotonic() - 60
        if rem > 30:
            a_opt, _ = reduced_lp_fw(best_seq, time_limit=min(rem * 0.6, 150),
                                      top_k=15, verbose=True)
            try_seq(a_opt, "lpfw_reduced")

    print(f"  Phase 4 best: {best_score:.10f}")

    # === Phase 5: Coordinate Descent Polishing ===
    print("\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 60
    t_alloc = min(remaining * 0.25, 180)

    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "coord_descent")

    print(f"  Phase 5 best: {best_score:.10f}")

    # === Phase 6: Final Combined Refinement ===
    print("\n=== Phase 6: Final Refinement ===")
    remaining = deadline - time.monotonic() - 30

    # Alternate between methods on the current best
    for round_num in range(6):
        if remaining < 20:
            break
        t_each = min(remaining / max(1, 6 - round_num), 40)
        if t_each < 10:
            break

        # Small perturbation
        noise_scale = 0.003 * (round_num + 1)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)

        # Quick L-BFGS
        a_opt, _ = logsumexp_lbfgs(f_pert, time_limit=t_each * 0.5,
                                    epsilon_start=0.03, epsilon_end=0.001,
                                    n_phases=3)
        try_seq(a_opt, f"final_lbfgs_{round_num}")

        # Quick spectral
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            a_opt, _ = spectral_flatten(best_seq, time_limit=min(rem * 0.4, 30),
                                         n_iter=200)
            try_seq(a_opt, f"final_spectral_{round_num}")

        remaining = deadline - time.monotonic() - 20

    # === Try different lengths ===
    rem = deadline - time.monotonic() - 20
    if rem > 20 and best_seq is not None:
        for n_try in [n_best - 32, n_best + 32, n_best + 64]:
            if n_try < 64 or rem < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                a_opt, _ = logsumexp_lbfgs(px, time_limit=min(rem * 0.3, 30),
                                            epsilon_start=0.04, epsilon_end=0.002,
                                            n_phases=3)
                try_seq(a_opt, f"final_len{n_try}")
                rem = deadline - time.monotonic() - 15

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
