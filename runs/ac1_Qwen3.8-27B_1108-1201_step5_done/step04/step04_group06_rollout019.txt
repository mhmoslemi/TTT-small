"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of spectral deflation
and LP-based optimization. The spectral deflation method directly targets peaks
in the auto-convolution by working in the frequency domain: deflate the peak,
take the spectral square root, invert, and project back to the feasible set.
This is complemented by LP-based direction finding and Nesterov gradient descent.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def spectral_deflation_step(a, conv, alpha, top_k=5):
    """
    One spectral deflation step: deflate top-k peaks, invert, project.
    Returns (a_new, conv_new, obj_new) or (None, None, None) on failure.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Identify top peaks to deflate
    k_peaks = min(top_k, size)
    peak_indices = np.argpartition(conv, -k_peaks)[-k_peaks:]
    # Only deflate peaks that are significant
    peak_indices = peak_indices[conv[peak_indices] > 0.1 * conv.max()]
    if len(peak_indices) == 0:
        peak_indices = np.array([int(np.argmax(conv))])

    # Deflate the convolution
    c_def = conv.copy()
    for k in peak_indices:
        c_def[k] = max(0.0, c_def[k] - alpha)

    # FFT, sqrt, IFFT
    C_def = np.fft.fft(c_def, fft_size)
    # Take real part and clip to non-negative (safe for small deflation)
    C_real = np.real(C_def)
    C_real = np.maximum(C_real, 0.0)
    A_def = np.sqrt(C_real)
    a_def = np.fft.ifft(A_def).real[:n]

    # Project to non-negative and normalize
    a_new = np.maximum(a_def, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return None, None, None
    a_new = a_new / s_new

    conv_new = fast_convolve(a_new)
    obj_new = 2.0 * n * conv_new.max()
    return a_new, conv_new, obj_new


def spectral_deflation_optimize(a0, time_limit, top_k=5, verbose=False):
    """
    Optimize using spectral deflation with adaptive alpha and line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    alpha = 0.005 * conv.max()
    no_improve = 0
    rng = np.random.default_rng(42)
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        # Adaptive alpha based on current peak
        peak_val = conv.max()
        alpha = min(0.02 * peak_val, max(peak_val / (50 + no_improve), 1e-10))

        improved = False
        # Line search: try several alpha values
        for attempt in range(8):
            a_new, conv_new, obj_new = spectral_deflation_step(
                a, conv, alpha, top_k=top_k)
            if a_new is not None and obj_new < best_obj - 1e-13:
                a = a_new
                conv = conv_new
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1

                if s > 1e-15:

                else:
                    a = best_a.copy()
                conv = fast_convolve(a)
                no_improve = 0

        iteration += 1
        if verbose and iteration % 100 == 0:
            print(f"    SpecDef iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                         'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step(a, conv, n):
    """One LP-based step with step sizing."""
    M = conv.max()
    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.0001, C)

    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False

    best_a = a.copy()
    best_obj = 2.0 * n * M
    best_t = 0.0

    candidates = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    for t in candidates:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t
            break

    if best_t > 0:
        return best_a, True
    return a.copy(), False

def compute_soft_gradient(a, conv, top_k=30, temp=0.003):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def nesterov_grad_optimize(a0, time_limit, lr_init=0.01, momentum=0.9,
                           top_k=30, temp_init=0.003):
    """Nesterov-accelerated gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = np.maximum(a + mom * v, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = np.maximum(a + v, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.002
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def generate_initialization(n, parent, rng):

    x = np.arange(n, dtype=np.float64)

    a = np.maximum(a, 1e-6)
    a = a / a.sum()

    # Add parent influence if available

        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p_norm = p / ps
            if len(p_norm) == n:
                a = 0.7 * p_norm + 0.3 * a
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(p_norm)), p_norm)
                p_interp = np.maximum(p_interp, 0)
                pis = p_interp.sum()
                if pis > 1e-10:
                    p_interp = p_interp / pis
                    a = 0.7 * p_interp + 0.3 * a

    return a


def extend_sequence(a_short, n_target):
    """Extend via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:

    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target),

    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:

    return a_ext

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Spectral deflation on parent length (new technique)
    # =========================================================
    print("=== Phase 1: Spectral Deflation ===")
    t_start = time.monotonic()
    n_parent = 768 if parent is None else len(parent)

    # Try the parent length first
    a0 = generate_initialization(n_parent, parent, rng)
    a_opt, obj = spectral_deflation_optimize(a0, time_limit=180, top_k=5, verbose=True)
    score = float(evaluate_sequence(a_opt.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_sequence = a_opt.copy()
        print(f"  SpecDef n={n_parent}: {score:.6f}")

    # Try a different top_k
    a0 = generate_initialization(n_parent, parent, rng)
    a_opt, obj = spectral_deflation_optimize(a0, time_limit=120, top_k=8, verbose=False)
    score = float(evaluate_sequence(a_opt.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_sequence = a_opt.copy()
        print(f"  SpecDef k=8 n={n_parent}: {score:.6f} *NEW BEST*")

    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP deep optimization
    # =========================================================
    print("=== Phase 2: LP Optimization ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 400.0

    if best_sequence is not None:
        t_lp = max(30.0, min(remaining * 0.6, 500.0))
        a_opt, obj = lp_optimize(best_sequence, time_limit=t_lp, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP: {score:.6f} *NEW BEST*")

    print(f"Phase 2 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Spectral deflation again (after LP refinement)
    # =========================================================
    print("=== Phase 3: Spectral Deflation Round 2 ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 300.0

    if best_sequence is not None and remaining > 50:
        t_sd = max(30.0, min(remaining * 0.5, 250.0))
        a_opt, obj = spectral_deflation_optimize(
            best_sequence, time_limit=t_sd, top_k=3, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  SpecDef2: {score:.6f} *NEW BEST*")

    print(f"Phase 3 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Try different lengths
    # =========================================================
    print("=== Phase 4: Multi-length exploration ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 200.0

    try_lengths = [512, 1024]
    if remaining > 100:
        for n_try in try_lengths:
            if time.monotonic() > deadline - 150:
                break
            t_alloc = max(20.0, min(remaining / 2, 120.0))
            if t_alloc < 15:
                break

            a0 = extend_sequence(best_sequence, n_try) if best_sequence is not None else np.ones(n_try) / n_try
            a_opt, obj = spectral_deflation_optimize(a0, time_limit=t_alloc, top_k=5, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  MultiLen n={n_try}: {score:.6f} *NEW BEST*")
            else:
                print(f"  MultiLen n={n_try}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Gradient fine-tuning
    # =========================================================
    print("=== Phase 5: Gradient Fine-tuning ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 100.0

    if best_sequence is not None and remaining > 50:
        configs = [
            {'lr_init': 0.008, 'momentum': 0.91, 'top_k': 40, 'temp_init': 0.002},
            {'lr_init': 0.005, 'momentum': 0.93, 'top_k': 60, 'temp_init': 0.001},
            {'lr_init': 0.012, 'momentum': 0.88, 'top_k': 25, 'temp_init': 0.004},
        ]
        n_cfgs = min(3, int(remaining / 30))
        for gi in range(n_cfgs):
            if time.monotonic() > deadline - 50:
                break
            t_sg = max(10.0, min(remaining / (n_cfgs - gi) * 0.8, 80.0))
            if t_sg < 8:
                break
            cfg = configs[gi % len(configs)]
            a_opt, obj = nesterov_grad_optimize(best_sequence, time_limit=t_sg, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad {gi}: {score:.6f} *NEW BEST*")

    print(f"Phase 5 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 6: Final polish (short LP + spectral)
    # =========================================================
    print("=== Phase 6: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            # Short LP
            t_lp2 = min(remaining * 0.5, 60.0)
            if t_lp2 > 15:
                a_opt, obj = lp_optimize(best_sequence, time_limit=t_lp2, verbose=False)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LP: {score:.6f} *NEW BEST*")

        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic() - 8
            if remaining > 20:
                t_sd2 = min(remaining * 0.8, 50.0)
                if t_sd2 > 10:
                    a_opt, obj = spectral_deflation_optimize(
                        best_sequence, time_limit=t_sd2, top_k=3, verbose=False)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final SpecDef: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
