Multi-phase optimizer combining FW-LP, LSE gradient descent, and diverse
multi-start exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_smooth(f, beta):
    """Log-sum-exp smooth approximation of max."""
    conv = fast_convolve(f)
    bc = beta * conv
    bc = bc - bc.max()
    return np.exp(bc).sum() / beta  # This is (1/beta)*log(sum(exp(beta*conv))) approximately
    # Actually let me just use the standard: (1/beta) * log(sum(exp(beta * conv)))

    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lse_gradient_descent(f0, time_limit, beta_start=20.0, beta_end=10000.0, lr_start=None):
    """LSE gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_start is None:
        lr_start = 3.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_start
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 6.0 / n)
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_start * 0.5, 1.0 / n)
                beta = min(beta * 2.0, beta_end)
                momentum = 0.90
                no_improve = 0

        if beta < beta_end:
            beta = min(beta * 1.005, beta_end)

        if it % 1000 == 0:
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def softmax_param_optimize(f0, time_limit, lr=0.5, momentum=0.9, max_iter=None):
    """Optimize in log-space: a_i = exp(x_i) / sum(exp(x_j))."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)
    x = np.log(f)  # log parameters
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    it = 0
    
    def f_from_x(x):
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        return exp_x / exp_x.sum()
    
    while time.monotonic() - t0 < time_limit:
        if max_iter and it >= max_iter:
            break
        it += 1
        
        f = f_from_x(x)
        conv = fast_convolve(f)
        bc = 50.0 * conv  # fixed moderate beta for this phase
        bc_max = bc.max()
        w = np.exp(bc - bc_max)
        w = w / w.sum()
        
        grad_a = compute_lse_grad(w, f)
        
        # Chain rule: d f_from_x / d x
        # f_i = exp(x_i) / S where S = sum(exp(x_j))
        # df_i/dx_j = f_i * (delta_ij - f_j)
        # d obj / d x_j = sum_i (d obj / d f_i) * f_i * (delta_ij - f_j)
        # = f_j * (sum_i grad_a_i * f_i - grad_a_j)
        # = f_j * (dot(grad_a, f) - grad_a_j)
        dot_gf = np.dot(grad_a, f)
        grad_x = f * (dot_gf - grad_a)
        
        v = momentum * v - lr * grad_x
        x = x + v
        # Re-center to avoid numerical issues
        x = x - x.mean()
        
        f_new = f_from_x(x)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.95
            if no_improve > 200:
                break
    
    return best_f, best_obj

"""
Multi-phase optimizer for step function minimization: 2*n*max(conv(a,a))/(sum(a))^2.
Combines: diverse multi-start, FW-LP with sparse Oracle, LSE gradient descent,
coordinate descent polishing, escape-and-reoptimize, and length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Gradient of LSE(beta * max conv) w.r.t. f."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_res = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_res[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix A where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_optimize(f0, time_limit, max_lp_time=10):
    """Frank-Wolfe with LP oracle for bilinear minimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining_time = time_limit - (time.monotonic() - t0)
        if remaining_time < max_lp_time + 3:
            break

        A = build_lp_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0.0, None)] * n
        lp_timeout = min(max_lp_time, remaining_time * 0.4)

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_timeout})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
            else:
                g = None
        except Exception:
            g = None

        if g is None or g.sum() < 1e-15:
            m_exp = min(m_exp * 1.12, 5.0)
            if m_exp >= 5.0:
                no_improve += 2
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            m_exp = min(m_exp * 1.02, 2.5)
            if no_improve > 35:
                break
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M_new = conv.max()
            M = M_new
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 6.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 1091 + 42)
            scale = 0.003 + 0.004 * min(1.0, (no_improve - 60) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if it % 500 == 0:
            print(f"    FW it={it}, obj={best_obj:.12f}, elapsed={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def lse_gradient_descent(f0, time_limit, beta_start=20.0, beta_end=10000.0):
    """Log-sum-exp smoothed gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    lr_start = 3.0 / n
    lr = lr_start
    beta = beta_start
    momentum = 0.92
    v = np.zeros(n)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 8.0 / n)
        else:
            no_improve += 1
            if no_improve % 12 == 0:
                lr *= 0.94
                momentum = max(0.85, momentum - 0.004)
            if no_improve > 120:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_start * 0.4, 0.5 / n)
                beta = min(beta * 1.8, beta_end)
                momentum = 0.90
                no_improve = 0

        if beta < beta_end:
            beta = min(beta * 1.004, beta_end)

        if it % 2000 == 0:
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, n_sweeps_max=500):
    """Cyclic coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    for sweep in range(n_sweeps_max):
        if time.monotonic() - t0 > time_limit:
            break
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[i]
            if old_val < 1e-15:
                continue

            # Try a range of new values for coordinate i
            # The effect on convolution: changing a[i] by delta changes
            # conv[k] by delta*(a[k-i] + a[i-k]) for valid k, plus delta^2 at k=i (roughly)
            # Use a simpler approach: try scaling
            for factor in [0.7, 0.85, 1.15, 1.3]:
                f_test = f.copy()
                f_test[i] = f[i] * factor
                f_test = np.maximum(f_test, 0)
                st = f_test.sum()
                if st < 1e-15:
                    continue
                f_test = f_test / st
                val = fast_convolve(f_test).max()
                if val < best_obj - 1e-15:
                    f = f_test
                    conv = fast_convolve(f)
                    best_obj = conv.max()
                    best_f = f.copy()
                    sweep_improved = True
                    break

        if not sweep_improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

        if sweep % 50 == 0:
            print(f"    CD sweep={sweep}, obj={best_obj:.12f}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def diverse_initializations(n, parent=None, seed=42):
    """Generate diverse initializations of length n."""
    rng = np.random.default_rng(seed)
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(x_new, x_old, parent), 0))

    inits.append(np.ones(n) / n)

    c = np.cos(np.pi * np.arange(n) / (2 * n))
    c = np.maximum(c, 0)
    if c.sum() > 1e-10:
        inits.append(c / c.sum())

    t = 1.0 - np.abs(2.0 * np.arange(n) / max(n - 1, 1) - 1.0)
    t = np.maximum(t, 0)
    if t.sum() > 1e-10:
        inits.append(t / t.sum())

    r = rng.random(n) + 0.01
    inits.append(r / r.sum())

    center = n / 2.0
    width = n / 5.0
    bump = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    bump = bump + 0.005 * rng.random(n)
    inits.append(bump / bump.sum())

    p = 1.0 / (np.arange(n) + 1) ** 0.3
    inits.append(p / p.sum())

    # Random with a few FW pre-iterations
    r2 = rng.random(n) + 0.01
    r2 = r2 / r2.sum()
    for _ in range(3):
        conv = fast_convolve(r2)
        M = conv.max()
        A = build_lp_matrix(r2)
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.05)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 3})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0)
                if g.sum() > 1e-15:
                    g = g / g.sum()
                    r2 = 0.4 * r2 + 0.6 * g
                    r2 = np.maximum(r2, 0)
                    r2 = r2 / r2.sum()
        except Exception:
            pass
    inits.append(r2)

    result = []
    for a in inits:
        a = np.maximum(a, 0)
        if a.sum() > 1e-10 and np.isfinite(a).all():
            a = a / a.sum()
            result.append(a)
    return result


def score_sequence(a):
    """Safely evaluate a sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        return float('inf')
    try:
        s = float(evaluate_sequence(a.tolist()))
        return s if np.isfinite(s) else float('inf')
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer:
    1. Multi-start with quick FW to find good initial basin
    2. Extended FW-LP for global structure
    3. LSE gradient polishing
    4. FW-LP refinement
    5. Coordinate descent micro-polish
    6. Escape + re-optimize (multiple attempts)
    7. Length exploration
    8. Final LSE with high beta
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    def remaining():
        return max(0.0, deadline - time.monotonic())

    best_score = float('inf')
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        s = score_sequence(a)
        if s < best_score:
            best_score = s
            best_seq = np.array(a, dtype=np.float64).copy()
            print(f"  *** NEW BEST: {s:.10f} (n={len(a)}) ***")
        return s

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = score_sequence(parent)
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            update_best(parent)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        update_best(parent)

    n_parent = len(parent)

    # === Phase 1: Multi-start Quick FW ===
    print(f"\n=== Phase 1: Multi-start (remaining: {remaining():.0f}s) ===")
    if remaining() > 250:
        t1_budget = min(180.0, remaining() - 900)
        if t1_budget > 30:
            inits = diverse_initializations(n_parent, parent, seed)
            per_init = max(5.0, t1_budget / max(len(inits), 1))
            t1_start = time.monotonic()

            for i, init in enumerate(inits):
                t_rem = t1_budget - (time.monotonic() - t1_start)
                if t_rem < 5:
                    break
                t_alloc = min(per_init, t_rem)
                print(f"  Init {i+1}/{len(inits)}: {t_alloc:.0f}s")
                a_opt, _ = fw_lp_optimize(init, time_limit=t_alloc, max_lp_time=5)
                update_best(a_opt)

    # === Phase 2: Extended FW-LP ===
    print(f"\n=== Phase 2: Extended FW-LP (best: {best_score:.10f}, rem: {remaining():.0f}s) ===")
    if remaining() > 350:
        t2_budget = min(280.0, remaining() - 400)
        if t2_budget > 30:
            current = best_seq if best_seq is not None else parent
            print(f"  Allocating {t2_budget:.0f}s, n={len(current)}")
            a_opt, _ = fw_lp_optimize(current, time_limit=t2_budget, max_lp_time=15)
            update_best(a_opt)

    # === Phase 3: LSE Polish ===
    print(f"\n=== Phase 3: LSE Polish (best: {best_score:.10f}, rem: {remaining():.0f}s) ===")
    if remaining() > 400:
        t3_budget = min(300.0, remaining() - 200)
        if t3_budget > 30:
            current = best_seq if best_seq is not None else parent
            print(f"  Allocating {t3_budget:.0f}s")
            a_opt, _ = lse_gradient_descent(current, time_limit=t3_budget,
                                             beta_start=15.0, beta_end=8000.0)
            update_best(a_opt)

    # === Phase 4: FW-LP Refinement ===
    print(f"\n=== Phase 4: FW-LP Refinement (best: {best_score:.10f}, rem: {remaining():.0f}s) ===")
    if remaining() > 250:
        t4_budget = min(180.0, remaining() - 100)
        if t4_budget > 20:
            current = best_seq if best_seq is not None else parent
            print(f"  Allocating {t4_budget:.0f}s")
            a_opt, _ = fw_lp_optimize(current, time_limit=t4_budget, max_lp_time=20)
            update_best(a_opt)

    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: CD Polish (best: {best_score:.10f}, rem: {remaining():.0f}s) ===")
    if remaining() > 100:
        t5_budget = min(60.0, remaining() - 50)
        if t5_budget > 10:
            current = best_seq if best_seq is not None else parent
            print(f"  Allocating {t5_budget:.0f}s")
            a_opt, _ = coordinate_descent_polish(current, time_limit=t5_budget, n_sweeps_max=200)
            update_best(a_opt)

    # === Phase 6: Escape + Re-optimize ===
    print(f"\n=== Phase 6: Escape (best: {best_score:.10f}, rem: {remaining():.0f}s) ===")
    if remaining() > 200:
        t6_budget = min(200.0, remaining() - 60)
        n_escapes = min(4, int(t6_budget / 50))
        t6_start = time.monotonic()

        for esc in range(n_escapes):
            t_rem = t6_budget - (time.monotonic() - t6_start)
            if t_rem < 35:
                break

            current = best_seq if best_seq is not None else parent
            scale = 0.015 + 0.015 * esc
            perturbed = project_simplex(current + rng.standard_normal(len(current)) * scale)

            t_fw = min(35, t_rem * 0.45)
            t_lse = min(35, t_rem * 0.45)

            a_tmp, _ = fw_lp_optimize(perturbed, time_limit=t_fw, max_lp_time=10)
            update_best(a_tmp)

            a_opt, _ = lse_gradient_descent(a_tmp, time_limit=t_lse,
                                             beta_start=30, beta_end=5000)
            update_best(a_opt)

    # === Phase 7: Length Exploration ===
    print(f"\n=== Phase 7: Length Exploration (best: {best_score:.10f}, rem: {remaining():.0f}s) ===")
    if remaining() > 150:
        t7_budget = min(120.0, remaining() - 40)
        alt_lengths = [384, 400, 480, 512, 576]
        alt_lengths = [l for l in alt_lengths if l != n_parent]

        # Pick top 2 most promising
        n_try = min(2, len(alt_lengths))
        t7_start = time.monotonic()

        for n_alt in alt_lengths[:n_try]:
            t_rem = t7_budget - (time.monotonic() - t7_start)
            if t_rem < 40:
                break

            # Interpolate best to new length
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_alt)
            f_interp = np.interp(x_new, x_old, best_seq)
            f_interp = np.maximum(f_interp, 0)
            if f_interp.sum() > 1e-10:
                f_interp = f_interp / f_interp.sum()

            t_fw = min(35, t_rem * 0.5)
            t_lse = min(30, t_rem * 0.4)

            print(f"  Trying n={n_alt}: {t_rem:.0f}s remaining")
            a_tmp, _ = fw_lp_optimize(f_interp, time_limit=t_fw, max_lp_time=10)
            update_best(a_tmp)

            if remaining() > 50:
                a_opt, _ = lse_gradient_descent(a_tmp, time_limit=t_lse,
                                                 beta_start=20, beta_end=4000)
                update_best(a_opt)

    # === Phase 8: Final LSE Polish ===
    print(f"\n=== Phase 8: Final Polish (best: {best_score:.10f}, rem: {remaining():.0f}s) ===")
    if remaining() > 20 and best_seq is not None:
        t8_budget = max(15.0, remaining() - 10)
        print(f"  Allocating {t8_budget:.0f}s, high-beta LSE")
        a_opt, _ = lse_gradient_descent(best_seq, time_limit=t8_budget,
                                         beta_start=200, beta_end=50000)
        update_best(a_opt)

        # One more CD sweep if time
        if remaining() > 15:
            a_opt2, _ = coordinate_descent_polish(best_seq, time_limit=min(15, remaining()-5),
                                                   n_sweeps_max=50)
            update_best(a_opt2)

    # Final output
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        final_score = score_sequence(final)
        print(f"\nFINAL: score={final_score:.10f}, n={len(final)}")
        return [float(x) for x in final]
    else:
        fallback = np.ones(448) / 448
        print(f"\nFALLBACK: n=448")
        return [float(x) for x in fallback]
