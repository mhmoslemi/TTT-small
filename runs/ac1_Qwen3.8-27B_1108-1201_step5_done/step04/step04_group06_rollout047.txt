**Iterative approach (类似 Gerchberg-Saxton / error reduction):**
1. Start with a non-negative a.
2. Compute c = conv(a, a).
3. Modify c to be flatter (e.g., clip the peaks).
4. Find a new non-negative a' whose autocorrelation is close to the modified c.
   - This is done by taking the "square root" in the frequency domain: A' = sqrt(|C|) * phase(A), where C is the DFT of modified c and A is the DFT of current a.
   - Then a' = IFFT(A'), clipped to non-negative.
5. Repeat.

This is the "Fienup" or "error reduction" algorithm used in phase retrieval. It might work well here!

Let me code this up:

```python
def fienup_step(a, clip_fraction=0.9):
    """One step of the Fienup error reduction algorithm."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Forward transform
    A = np.fft.rfft(a, fft_size)
    C = A * A  # This gives the circular autocorrelation of zero-padded a
    c = np.fft.irfft(C, fft_size)[:size]  # Linear autocorrelation
    
    # Modify c: clip the peaks
    threshold = np.quantile(c, clip_fraction)
    c_modified = np.minimum(c, threshold)
    
    # Inverse: find new a
    C_mod = np.fft.rfft(c_modified, fft_size)
    # Take square root, preserving phase of A
    A_mod = np.sqrt(np.abs(C_mod)) * np.exp(1j * np.angle(A))
    a_new = np.fft.irfft(A_mod, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    
    return a_new
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Super-ellipse shape parameterization for informed initializations
2. Annealed soft-max gradient descent on the probability simplex
3. LP-based exact peak reduction for final polishing
4. Multi-length exploration with interpolation

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, conv, top_k=50, temp=0.005):
    """
    Compute gradient of soft-max of conv peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        # dc_k/da_i = a_{k-i} for valid i (0 <= k-i < n)
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a_{k-i} for i in [i_start, i_end]
        # k-i ranges from k-i_start to k-i_end
        indices = np.arange(k - i_start, k - i_end + 1)
        grad[i_start:i_end + 1] += w[idx] * a[indices]

    return grad


def annealed_gradient_optimize(a0, time_limit, lr_init=0.05, temp_high=0.001, temp_low=0.0001,
                                top_k=50, momentum_init=0.0, momentum_end=0.9,
                                n_anneal_steps=500):
    """
    Annealed soft-max gradient descent on the probability simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)  # momentum
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_anneal_steps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Annealing schedule: geometric from temp_high to temp_low
        progress = min(1.0, step / n_anneal_steps)
        temp = temp_high * (temp_low / temp_high) ** progress
        lr = lr_init * (1.0 - 0.7 * progress)
        momentum = momentum_init + (momentum_end - momentum_init) * progress

        # Compute gradient
        grad = compute_gradient(a, conv, top_k=top_k, temp=temp)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v

        # Project onto simplex
        a_new = project_simplex(a_new, 1.0)

        # Check improvement
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If stuck, try a larger step in a random direction
            if no_improve > 50:
                rng = np.random.default_rng(step * 31 + 7)
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(best_a + noise, 1.0)
                conv = fast_convolve(a)
                v = np.zeros(n)
                no_improve = 0

    return best_a, best_obj


def lp_optimize(a0, time_limit, max_iters=200):
    """
    LP-based iterative optimization: fix a, find best g s.t. a*g <= M, g >= 0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    # Build convolution matrix
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a

    for it in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Solve LP: max sum(g) s.t. C@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-10))
        bounds = [(0, None)] * n

        try:
            result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                            method='highs', options={'presolve': True})
            if result.success and result.x is not None:
                g = result.x
                g_sum = g.sum()
                if g_sum > 0:
                    g = g / g_sum  # Normalize to simplex

                    # Try several step sizes
                    improved = False
                    for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
                        a_try = (1.0 - t) * a + t * g
                        a_try = np.maximum(a_try, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        conv_try = fast_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        if obj_try < best_obj - 1e-10:
                            a = a_try
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break

                    if improved:
                        no_improve = 0
                        # Rebuild C with new a
                        C = np.zeros((2 * n - 1, n))
                        for j in range(n):
                            C[j:j + n, j] = a
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def super_ellipse_init(n, p=1.0, shift=0.0):
    """
    Generate a super-ellipse shaped initialization.
    p=1 gives triangle, p=2 gives circular arc, p->inf gives uniform.
    shift: fraction to shift the peak (0=centered, 0.5=right edge)
    """
    x = np.arange(n, dtype=np.float64) / n
    # Center the shape
    c = 0.5 + shift
    # Super-ellipse: (1 - |2*(x-c)/1|^p) clipped to [0,1]
    dist = np.abs(x - c) / max(c, 1.0 - c) if min(c, 1.0 - c) > 1e-10 else np.abs(x - c)
    shape = np.power(np.maximum(1.0 - dist, 0.0), 1.0 / max(p, 0.1))
    # Ensure positivity
    shape = shape + 0.001
    shape = shape / shape.sum()
    return shape


def cosine_init(n, power=1.0):
    """Cosine-based smooth initialization."""
    x = np.arange(n, dtype=np.float64)
    shape = np.cos(np.pi * x / (n - 1 + 1e-10))
    shape = np.maximum(shape, 0.0) ** power
    shape = shape + 0.001
    shape = shape / shape.sum()
    return shape


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}

    inits['uniform'] = np.ones(n) / n
    inits['triangle'] = super_ellipse_init(n, p=1.0)
    inits['circle'] = super_ellipse_init(n, p=2.0)
    inits['super4'] = super_ellipse_init(n, p=4.0)
    inits['super8'] = super_ellipse_init(n, p=8.0)
    inits['cos1'] = cosine_init(n, power=1.0)
    inits['cos2'] = cosine_init(n, power=2.0)
    inits['cos05'] = cosine_init(n, power=0.5)

    # Asymmetric shapes
    inits['ramp'] = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] /= inits['ramp'].sum()

    inits['ramp2'] = np.arange(1, n + 1, dtype=np.float64) ** 2
    inits['ramp2'] /= inits['ramp2'].sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    return inits


def interpolate_sequence(a, n_target):
    """Interpolate a sequence to a different length."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a / s
    a_new = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n), a_norm)
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid annealed gradient + LP + multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}")
    except Exception:
        pass

    def try_sequence(a, label=""):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a_clean = np.maximum(a, 0.0)
        s = a_clean.sum()
        if s < 0.01:
            return
        score = float(evaluate_sequence(a_clean.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_clean.copy()
            print(f"  {label}: score={score:.6f} *NEW BEST* (n={len(a_clean)})")

    # =========================================================
    # Phase 1: Quick gradient scan across shapes and lengths
    # =========================================================
    print("=== Phase 1: Shape/Length Scan ===")
    t_start = time.monotonic()

    lengths = [384, 512, 768, 1024]
    shapes_to_try = ['triangle', 'circle', 'super4', 'cos1', 'cos2', 'parent', 'super8', 'ramp']

    phase1_limit = 300.0
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break

        inits = generate_inits(n, rng, parent)

        for shape_name in shapes_to_try:
            if time.monotonic() > deadline - 500:
                break
            if shape_name not in inits:
                continue

            t_per = max(5.0, min(30.0, (phase1_limit - (time.monotonic() - t_start)) / 8.0))
            t_alloc = min(t_per, deadline - time.monotonic() - 400)
            if t_alloc < 3:
                break

            a0 = inits[shape_name]
            a_opt, obj = annealed_gradient_optimize(
                a0, time_limit=t_alloc,
                lr_init=0.03, temp_high=0.002, temp_low=0.0002,
                top_k=40, momentum_init=0.0, momentum_end=0.85,
                n_anneal_steps=300
            )

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, shape_name, a_opt.copy()))
                try_sequence(a_opt, f"Scan n={n}/{shape_name}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep optimization on top candidates
    # =========================================================
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()

    # Get top unique candidates
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_candidates.append((n, a_opt))
            seen.add(key)
        if len(top_candidates) >= 5:
            break

    # Always include parent if available
    if parent is not None and len(parent) not in [c[0] for c in top_candidates]:
        top_candidates.append((len(parent), parent.copy()))

    remaining = deadline - time.monotonic() - 300.0
    n_cands = max(1, len(top_candidates))
    t_per_deep = max(20.0, remaining * 0.5 / n_cands)

    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 200)
        if t_alloc < 10:
            break

        print(f"  Deep opt {i}: n={n}, t={t_alloc:.0f}s")

        # Long annealed gradient
        a_opt, obj = annealed_gradient_optimize(
            a0, time_limit=t_alloc * 0.5,
            lr_init=0.02, temp_high=0.003, temp_low=0.0001,
            top_k=60, momentum_init=0.1, momentum_end=0.88,
            n_anneal_steps=2000
        )
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            try_sequence(a_opt, f"DeepGrad {i}")

        # LP polish on the result
        remaining = deadline - time.monotonic() - 150.0
        if remaining > 10:
            t_lp = min(remaining * 0.3, 60.0)
            a_lp, obj_lp = lp_optimize(a_opt, time_limit=t_lp, max_iters=100)
            try_sequence(a_lp, f"LP {i}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Extended length exploration
    # =========================================================
    print("=== Phase 3: Extended Length ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        # Try extending to larger lengths
        for target_n in [1280, 1536, 2048]:
            if time.monotonic() > deadline - 150:
                break
            a_ext = interpolate_sequence(best_sequence, target_n)

            t_alloc = min(60.0, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break

            a_opt, obj = annealed_gradient_optimize(
                a_ext, time_limit=t_alloc,
                lr_init=0.015, temp_high=0.002, temp_low=0.0001,
                top_k=50, momentum_init=0.0, momentum_end=0.85,
                n_anneal_steps=500
            )
            try_sequence(a_opt, f"Ext n={target_n}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Intensive LP + gradient on best
    # =========================================================
    print("=== Phase 4: Intensive Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        n_best = len(best_sequence)

        # Multiple LP runs with perturbations
        for pi in range(5):
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(remaining / (5 - pi) * 0.6, 120.0)
            if t_alloc < 8:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.003 * (pi + 1)
                noise = rng.standard_normal(n_best) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s0 = a0.sum()
                if s0 > 1e-10:
                    a0 = a0 / s0
                else:
                    a0 = best_sequence.copy()

            # Alternate between gradient and LP
            if pi % 2 == 0:
                a_opt, obj = lp_optimize(a0, time_limit=t_alloc, max_iters=80)
            else:
                a_opt, obj = annealed_gradient_optimize(
                    a0, time_limit=t_alloc,
                    lr_init=0.01, temp_high=0.001, temp_low=0.00005,
                    top_k=80, momentum_init=0.0, momentum_end=0.8,
                    n_anneal_steps=1000
                )
            try_sequence(a_opt, f"Polish {pi}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final micro-optimization
    # =========================================================
    print("=== Phase 5: Final ===")

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # Final LP micro-step
        if remaining > 5:
            t_final = min(remaining * 0.5, 30.0)
            a_opt, _ = lp_optimize(best_sequence, time_limit=t_final, max_iters=50)
            try_sequence(a_opt, "Final LP")

        # Final gradient micro-step
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 3:
            a_opt, _ = annealed_gradient_optimize(
                best_sequence, time_limit=remaining,
                lr_init=0.005, temp_high=0.0005, temp_low=0.00005,
                top_k=100, momentum_init=0.0, momentum_end=0.7,
                n_anneal_steps=200
            )
            try_sequence(a_opt, "Final Grad")

    # Fallback
    if best_sequence is None:
        best_sequence = super_ellipse_init(512, p=2.0)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
