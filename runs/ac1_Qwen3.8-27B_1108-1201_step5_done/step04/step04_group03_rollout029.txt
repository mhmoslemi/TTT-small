"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed gradient descent
with temperature annealing, FW-LP refinement, and fine-tuning.
Key innovation: uses the log-sum-exp as a smooth surrogate for the max,
enabling efficient gradient computation via weighted correlation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_value_and_grad(a, T):
    """Compute LSE surrogate value and gradient."""
    c = fast_convolve(a)
    max_c = c.max()
    shifted = (c - max_c) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 0)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp  # softmax weights
    lse_val = T * np.log(sum_exp) + max_c
    
    n = len(a)
    # grad[j] = 2 * dot(w[j:j+n], a)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(w[j:j+n], a)
    
    return lse_val, grad, w


def lse_gradient_descent(a0, time_limit, T_start=0.01, T_end=0.0005, 
                         lr_start=0.01, lr_end=0.001, momentum=0.9,
                         n_anneal_steps=500, verbose=True):
    """LSE-smoothed projected gradient descent with temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum velocity
    no_improve = 0
    
    total_steps = 2000  # target number of steps
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit))
        
        # Anneal temperature and learning rate
        T = T_start * (T_end / T_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute LSE value and gradient
        lse_val, grad, w = lse_value_and_grad(a, T)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Evaluate actual objective
        c_new = fast_convolve(a_new)
        obj = c_new.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and no_improve == 0 and (int((elapsed) * 2) != int((elapsed - 0.5) * 2)):
                score = 2 * n * obj
                if score < 2.0:
                    print(f"  LSE-GD: score={score:.8f}, T={T:.6f}, elapsed={elapsed:.0f}s")
        else:
            no_improve += 1
            a_new = a  # rollback
            a = a  # stay
        
        a = a_new if obj < best_obj else a
        
        # Restart if stuck
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            noise_scale = 0.001 + 0.002 * min(1.0, no_improve / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj


def solve_lp(f, M):
    """LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, verbose=True):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 50:
                # Perturb and retry
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.8, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 80:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def mult_weight_optimize(f0, time_limit, lr_init=0.03, top_k=15, temp_init=0.0003,
                         verbose=True):
    """Multiplicative weight optimization for fine-tuning."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[i] * f[k - idx]
            log_update[idx] -= cur_lr * grad_contrib
        
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        cur_temp *= 1.002
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            f = best_f * np.exp(rng.standard_normal(n) * 0.003)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.005)
            no_improve = 0
    
    return best_f, best_obj
