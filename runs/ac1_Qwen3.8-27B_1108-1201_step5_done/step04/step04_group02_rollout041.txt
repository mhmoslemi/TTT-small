```python
                max_sl = max(max_sl, sl)

            sl = 2 * delta + 2 * a_i

                max_sl = max(max_sl, sl)

            hi = mid
```

```python
def coordinate_step(f, conv, i):
    """Find optimal f[i] to minimize max of new convolution."""
    n = len(f)
    a_i = f[i]
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)
    
    # U = max of conv outside [k_lo, k_hi]
    U = 0.0
    if k_lo > 0:
        U = max(U, conv[:k_lo].max())
    if k_hi < 2 * n - 2:
        U = max(U, conv[k_hi+1:].max())
    
    # Linear pieces: for k in [k_lo, k_hi], k != 2i
    # value_k(delta) = conv[k] + 2*delta*f[k-i]
    # slope_k = 2*f[k-i]
    
    # Quadratic: value_q(delta) = conv[2i] + 2*a_i*delta + delta^2
    # slope_q(delta) = 2*delta + 2*a_i
    
    # Find delta that minimizes max(U, max_k value_k(delta), value_q(delta))
    # with constraint delta >= -a_i (so f[i] >= 0)
    
    def eval_at(delta):
        """Returns (value, subgrad_interval) at given delta."""
        best_val = U
        # Linear pieces
        # Vectorized: for k in [k_lo, k_hi], k != 2i
        ks = np.arange(k_lo, k_hi + 1)
        mask = ks != 2 * i
        ks_m = ks[mask]
        js = ks_m - i  # indices into f
        valid = (js >= 0) & (js < n)
        ks_v = ks_m[valid]
        js_v = js[valid]
        
        lin_vals = conv[ks_v] + 2 * delta * f[js_v]
        lin_slopes = 2 * f[js_v]
        
        max_lin = lin_vals.max() if len(lin_vals) > 0 else -np.inf
        tied = np.abs(lin_vals - max_lin) < 1e-12
        min_sl_lin = lin_slopes[tied].min() if tied.any() else -np.inf
        max_sl_lin = lin_slopes[tied].max() if tied.any() else np.inf
        
        # Quadratic
        q_val = conv[2*i] + 2 * a_i * delta + delta * delta
        q_sl = 2 * delta + 2 * a_i
        
        # Combine
        total_val = best_val
        total_min_sl = 0.0  # U has slope 0
        total_max_sl = 0.0
        
        if max_lin > total_val + 1e-14:
            total_val = max_lin
            total_min_sl = min_sl_lin
            total_max_sl = max_sl_lin
        elif abs(max_lin - total_val) <= 1e-14:
            total_min_sl = min(total_min_sl, min_sl_lin)
            total_max_sl = max(total_max_sl, max_sl_lin)
        
        if q_val > total_val + 1e-14:
            total_val = q_val
            total_min_sl = q_sl
            total_max_sl = q_sl
        elif abs(q_val - total_val) <= 1e-14:
            total_min_sl = min(total_min_sl, q_sl)
            total_max_sl = max(total_max_sl, q_sl)
        
        return total_val, total_min_sl, total_max_sl
    
    # Bisection on delta
    lo = -a_i  # boundary
    hi = a_i  # start here, will expand if needed
    
    # Check slope at lo
    _, _, max_sl_lo = eval_at(lo)
    if max_sl_lo >= 0:
        return 0.0  # optimal is at the boundary
    
    # Find hi where slope > 0
    while True:
        _, min_sl_hi, _ = eval_at(hi)
        if min_sl_hi > 0:
            break
        hi *= 2.0
        if hi > 100 * max(a_i, 0.01) + 1.0:
            break
    
    # Bisection
    for _ in range(50):
        mid = (lo + hi) / 2.0
        val, min_sl, max_sl = eval_at(mid)
        if min_sl > 1e-10:
            hi = mid
        elif max_sl < -1e-10:
            lo = mid
        else:
            break
    
    optimal_delta = (lo + hi) / 2.0
    new_ai = max(0.0, a_i + optimal_delta)
    return new_ai
```

For the smoothed objective F_tau(f) = log-sum-exp(conv(f) / tau) * tau (or similar), the gradient is well-defined everywhere. Nesterov's method:

y_t = (1 - alpha_t) x_t + alpha_t x_{t-1}  (extrapolation)
x_{t+1} = argmin_{x in C} <g(y_t), x> + (1/(2L)) ||x - y_t||^2

For the simplex constraint and Euclidean projection, this becomes:
x_{t+1} = project_simplex(y_t - (1/L) * g(y_t))

With alpha_t = 2/(t+2) and L being the Lipschitz constant of the gradient.

The gradient of the log-sum-exp approximation:
F_tau(f) = tau * log(sum_k exp(conv(k)/tau))
grad F_tau(f) = sum_k (exp(conv(k)/tau) / sum_j exp(conv(j)/tau)) * grad conv(k)
= sum_k w_k * 2 * f[k::-1]  (approximately, with the right indexing)

where w_k = exp(conv(k)/tau) / sum_j exp(conv(j)/tau)

The Lipschitz constant of grad F_tau is bounded by 2/tau (roughly).

So the step size is 1/L = tau/2.

With Nesterov: alpha_t = 2/(t+2), and the effective step size is tau/2 * alpha_t... actually no, in Nesterov's method the step size is 1/L (not scaled by alpha).

```python
def nesterov_smooth_descent(f0, time_limit, init_temp=0.01, temp_decay=0.9999, 
                           nesterov=True):
    n = len(f0)
    f = f0.copy()
    f = f / f.sum()
    conv = fast_convolve(f)
    
    best_obj = 2 * n * conv.max()  # with sum=1
    best_f = f.copy()
    
    if nesterov:
        f_prev = f.copy()
        x_prev = f.copy()
    
    t0 = time.monotonic()
    it = 0
    temp = init_temp
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        it += 1
        
        # Nesterov extrapolation
        if nesterov and t > 1:
            alpha = 2.0 / (t + 2)
            y = (1 - alpha) * f + alpha * x_prev
            y = project_simplex(y)
        else:
            y = f
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        scaled = conv_y / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # grad[i] = 2 * sum_k w[k] * y[k-i] (for valid k-i)
        grad = np.zeros(n)
        for i in range(n):
            ke = min(i + n, len(conv_y))
            grad[i] = 2.0 * np.dot(w[i:ke], y[:ke-i])
        
        # Step: f_new = project_simplex(y - (1/L) * grad)
        # L ≈ 2 * max(conv_y) / temp + ... ≈ 2 / temp (rough bound)
        L = 4.0 / temp  # conservative
        f_new = project_simplex(y - grad / L)
        
        if nesterov:
            x_prev = f
            f = f_new
        else:
            f = f_new
        
        conv = fast_convolve(f)
        obj = 2 * n * conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
        
        temp *= temp_decay
    
    return best_f, best_obj / (2*n)  # return max(conv) for normalized f
```

Hmm, the Lipschitz constant bound might be too conservative. Let me think...

The gradient of F_tau(f) where F_tau = tau * log(sum exp(conv_k/tau)):
grad F_tau = sum_k p_k * grad conv_k where p_k = exp(conv_k/tau)/Z

The Hessian of F_tau is:
Hessian = (1/tau) * (sum_k p_k * grad conv_k * grad conv_k^T) - (1/tau^2) * ... (variance term)

The spectral norm of grad conv_k is at most 2*||f||_1 = 2 (when sum(f)=1).
So the Hessian norm is bounded by (1/tau) * 4 = 4/tau.

So L = 4/tau is a valid (though possibly loose) bound. The step size 1/L = tau/4.

With Nesterov, the convergence rate is O(1/t^2) for the objective gap, so after T iterations, the gap is O(L/D^2 * 1/T^2) where D is the diameter of the domain.

For our problem, D ≈ sqrt(2/n) (diameter of the simplex in Euclidean norm). So after T iterations, the gap is O(4/tau * n/2 * 1/T^2) = O(2n/(tau*T^2)).

To get a gap of 1e-6 with n=384, tau=0.01: T ≈ sqrt(2*384/(0.01*1e-6)) ≈ sqrt(7.68e10) ≈ 277K iterations. Each iteration takes O(n log n) for the FFT + O(n^2) for the gradient = O(n^2) ≈ 147K operations. Total: 277K * 147K ≈ 4e10 operations. That's too slow!

```python
Hybrid optimizer: LP Frank-Wolfe + Coordinate Descent for minimizing
2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build the A_ub matrix for the LP oracle."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, A_cache=None, time_limit=5):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    if A_cache is None:
        A = build_lp_matrix(f)
    else:
        A = A_cache
    c = -np.ones(n)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
```

```python
"""
Hybrid optimizer: LP Frank-Wolfe + Exact Coordinate Descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive time allocation, multiple phase progression, and incrementally-updated
convolution for fast coordinate sweeps.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M, time_limit=5):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for k=0..2n-2, g >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coord_find_optimal(f, conv, i):
    """Find optimal new f[i] minimizing max(conv) using bisection on subgradient."""
    n = len(f)
    a_i = f[i]

    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)

    # U = max of unchanged conv entries
    U = 0.0
    if k_lo > 0:
        U = max(U, float(conv[:k_lo].max()))
    if k_hi < 2 * n - 2:
        U = max(U, float(conv[k_hi + 1:].max()))

    # Linear pieces
    ks = np.arange(k_lo, k_hi + 1)
    lin_mask = ks != (2 * i)
    ks_lin = ks[lin_mask]
    js_lin = ks_lin - i
    valid = (js_lin >= 0) & (js_lin < n)
    ks_lin = ks_lin[valid]
    js_lin = js_lin[valid]

    has_lin = len(ks_lin) > 0
    if has_lin:
        lin_base = conv[ks_lin].copy()
        lin_slopes = 2.0 * f[js_lin].copy()

    # Quadratic piece
    has_quad = (2 * i >= 0) and (2 * i <= 2 * n - 2) and (2 * i >= k_lo) and (2 * i <= k_hi)
    if has_quad:
        quad_base = float(conv[2 * i])

    def eval_d(delta):
        best = U
        min_sg = 0.0
        max_sg = 0.0

        if has_lin:
            vals = lin_base + delta * lin_slopes
            mv = float(vals.max())
            if mv > best + 1e-13:
                best = mv
                tied = np.abs(vals - mv) < 1e-11
                min_sg = float(lin_slopes[tied].min())
                max_sg = float(lin_slopes[tied].max())
            elif abs(mv - best) <= 1e-13:
                tied = np.abs(vals - best) < 1e-11
                if tied.any():
                    min_sg = min(min_sg, float(lin_slopes[tied].min()))
                    max_sg = max(max_sg, float(lin_slopes[tied].max()))

        if has_quad:
            qv = quad_base + 2.0 * a_i * delta + delta * delta
            qs = 2.0 * a_i + 2.0 * delta
            if qv > best + 1e-13:
                best = qv
                min_sg = qs
                max_sg = qs
            elif abs(qv - best) <= 1e-13:
                min_sg = min(min_sg, qs)
                max_sg = max(max_sg, qs)

        return best, min_sg, max_sg

    # Check lower boundary
    _, _, msl_lo = eval_d(-a_i)
    if msl_lo >= -1e-11:
        return 0.0

    # Find upper bound
    hi = max(a_i, 0.005)
    for _ in range(40):
        _, msl_hi, _ = eval_d(hi)
        if msl_hi > 1e-11:
            break
        hi *= 2.0
        if hi > 50.0:
            break

    lo = -a_i
    for _ in range(55):
        mid = (lo + hi) / 2.0
        _, msl, xsl = eval_d(mid)
        if msl > 1e-10:
            hi = mid
        elif xsl < -1e-10:
            lo = mid
        else:
            break

    return max(0.0, a_i + (lo + hi) / 2.0)


def coordinate_descent(f0, time_limit, verbose=True):
    """Cyclic coordinate descent with incremental conv update."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f).copy()
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        for i in range(n):
            new_ai = coord_find_optimal(f, conv, i)
            delta = new_ai - f[i]
            if abs(delta) < 1e-14:
                continue

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Update conv for linear terms
            js = np.arange(k_lo, k_hi + 1) - i
            valid = (js >= 0) & (js < n)
            ks_v = np.arange(k_lo, k_hi + 1)[valid]
            js_v = js[valid]
            mask_ne2i = ks_v != (2 * i)
            ks_up = ks_v[mask_ne2i]
            js_up = js_v[mask_ne2i]
            if len(ks_up) > 0:
                conv[ks_up] += 2.0 * delta * f[js_up]

            # Update quadratic term
            if 2 * i <= 2 * n - 2 and 2 * i >= k_lo and 2 * i <= k_hi:
                conv[2 * i] += 2.0 * f[i] * delta + delta * delta

            f[i] = new_ai

        # End of sweep: check and renormalize if needed
        s = float(f.sum())
        if abs(s - 1.0) > 1e-8:
            f = f / s
            conv = fast_convolve(f).copy()

        obj = float(conv.max())
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps > 50:
            # Perturb and restart
            rng = np.random.default_rng((sweep * 31 + 7) % (2**32))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f).copy()
            no_improve_sweeps = 0
            if verbose and sweep % 100 == 0:
                print(f"    CD restart sweep={sweep}, obj={conv.max():.8f}")

        if verbose and sweep % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}, obj={obj:.10f}, best={best_obj:.10f}, t={elapsed:.1f}s")

    if verbose:
        print(f"  CD done: sweeps={sweep}, best={best_obj:.10f}")
    return best_f, best_obj


def smooth_gradient_step(f0, time_limit, init_temp=0.005, verbose=False):
    """Mirror descent (multiplicative weights) with softmax-smoothed subgradient."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    temp = init_temp

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break
        it += 1

        # Softmax weights
        scaled = conv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Subgradient: grad[i] = 2 * sum_k w[k] * f[k-i]
        grad = np.zeros(n)
        for i in range(n):
            ke = min(i + n, len(conv))
            grad[i] = 2.0 * np.dot(w[i:ke], f[:ke - i])

        # Mirror descent step: f_new = f * exp(-eta * grad) / sum(...)
        eta = 1.0 / (2.0 * np.sqrt(it + 1) * (1.0 + np.max(conv) / temp))
        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()

        conv = fast_convolve(f_new)
        obj = float(conv.max())

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()

        f = f_new
        temp *= 1.002

    if verbose:
        print(f"  MD done: it={it}, best={best_obj:.10f}")
    return best_f, best_obj

            lengths.insert(0, pn)
    # Add a couple other lengths for diversity
    for alt in [256, 512]:
        if alt not in lengths:
            lengths.append(alt)

    # === Phase 1: LP Frank-Wolfe on parent ===
    print("=== Phase 1: LP Frank-Wolfe ===")
    fw_results = []

    if parent is not None:
        n_p = len(parent)
        t1 = min(350, deadline - time.monotonic() - 500)
        if t1 > 30:
            print(f"  FW on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = fw_lp_optimize(parent, time_limit=t1, lp_time=5, verbose=True)
            try_seq(a_opt, "parent_fw")
            fw_results.append((a_opt, obj))

    # FW on a couple other initializations for the best length
    n_main = len(parent) if parent is not None else 384
    if parent is not None:
        inits_main = generate_inits(n_main, parent, rng)
        for idx, f_init in enumerate(inits_main[:4]):
            rem = deadline - time.monotonic() - 400
            if rem < 30:
                break
            t_alloc = min(rem // max(1, 4 - idx), 80)
            if t_alloc < 20:
                break
            print(f"  FW init {idx} (n={n_main}), t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, lp_time=3, verbose=False)
            try_seq(a_opt, f"fw_i{idx}")
            fw_results.append((a_opt, obj))

    # Try a different length
    if len(lengths) > 1:
        n_alt = lengths[1] if lengths[1] != n_main else (lengths[0] if lengths[0] != n_main else None)
        if n_alt is not None:
            rem = deadline - time.monotonic() - 300
            if rem > 40:
                f_init = np.ones(n_alt) / n_alt
                print(f"  FW on alt length n={n_alt}, t={min(rem//2, 120):.0f}s")
                a_opt, obj = fw_lp_optimize(f_init, time_limit=min(rem // 2, 120), lp_time=3, verbose=False)
                try_seq(a_opt, f"fw_n{n_alt}")
                fw_results.append((a_opt, obj))

    # Pick best so far for coordinate descent
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 2: Coordinate Descent (main refinement) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    print(f"  Starting from best score: {best_score:.8f}, n={len(best_seq)}")

    n_cd = len(best_seq)
    t2 = min(500, deadline - time.monotonic() - 200)
    if t2 > 30:
        print(f"  CD on best (n={n_cd}), t={t2:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t2, verbose=True)
        try_seq(a_opt, "cd_main")

    # CD with a perturbed start
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        f_pert = project_simplex(best_seq + rng.standard_normal(n_cd) * 0.005)
        t_alloc = min(rem, 120)
        print(f"  CD perturbed, t={t_alloc:.0f}s")
        a_opt, obj = coordinate_descent(f_pert, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, "cd_pert")

    # === Phase 3: Mirror Descent refinement ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  MD on best (n={n_cd}), t={t_alloc:.0f}s")
        a_opt, obj = smooth_gradient_step(best_seq, time_limit=t_alloc, init_temp=0.003, verbose=True)
        try_seq(a_opt, "md")

    # === Phase 4: Final refinement loop ===
    print(f"\n=== Phase 4: Final Refinement ===")
    print(f"  Current best: {best_score:.8f}")

    # Alternate between FW (short) and CD (short) for final squeeze
    while time.monotonic() < deadline - 30:
        rem = deadline - time.monotonic() - 30
        if rem < 15:
            break

        # Short CD burst
        t_cd = min(rem // 3, 60)
        if t_cd > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, verbose=False)
            try_seq(a_opt, "final_cd")

        # Short FW burst
        rem = deadline - time.monotonic() - 30
        if rem < 15:
            break
        t_fw = min(rem // 3, 40)
        if t_fw > 10:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_fw, lp_time=3, verbose=False)
            try_seq(a_opt, "final_fw")

        # Short CD again
        rem = deadline - time.monotonic() - 30
        if rem < 10:
            break
        t_cd2 = min(rem, 60)
        if t_cd2 > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd2, verbose=False)
            try_seq(a_opt, "final_cd2")

    # === Phase 5: Micro perturbations ===
    print(f"\n=== Phase 5: Micro Perturbations ===")
    rem = deadline - time.monotonic() - 10
    if rem > 5 and best_seq is not None:
        for trial in range(6):
            rem = deadline - time.monotonic() - 8
            if rem < 3:
                break
            noise_scale = 0.002 / (trial + 1)
            f_p = project_simplex(best_seq + rng.standard_normal(n_cd) * noise_scale)
            t_alloc = max(3, min(rem - 3, 15))
            a_opt, _ = coordinate_descent(f_p, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"micro_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
