import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_exact_subgradient(a, conv, top_k=5, temp=0.01):
    """Compute subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    
    # Softmax weights
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(top_k):
        k = int(peaks[idx])
        # subgrad for b_k: g_i = 2*a_{k-i} for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += w[idx] * 2.0 * a[k - i]
    return grad

def mirror_descent_optimize(a0, time_limit, eta_init=0.1, top_k=5, temp=0.01):
    """Mirror descent on the probability simplex."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        grad = compute_exact_subgradient(a, conv, top_k=top_k, temp=temp)
        
        # Multiplicative update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad
        # Normalize
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Reduce eta
            eta *= 0.995
        
        a = a_new
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

            grad[idxs] += 2.0 * a[k - idxs]
            return grad
        else:
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
            grad = np.zeros(n)
            for idx in range(top_k):
                if w[idx] < 1e-10:
                    continue
                k = int(peaks[idx])
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                idxs = np.arange(i_start, i_end + 1)
                grad[idxs] += w[idx] * 2.0 * a[k - idxs]
            return grad


def mirror_descent(a0, time_limit, eta_init=0.5, top_k=3, temp_init=0.0,
                   momentum=0.0, patience=1000):
    """
    Mirror descent (multiplicative weights) on the probability simplex.
    Minimizes max_k (a*a)_k subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    log_a = np.log(a + 1e-30)
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    eta = eta_init
    temp = temp_init
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        conv = fast_convolve(a)
        M = conv.max()
        cur_obj = 2.0 * n * M
        
        grad = compute_subgradient(a, conv, top_k=top_k, temp=temp)
        
        # Momentum
        v = momentum * v + grad
        
        # Multiplicative update in log-space
        log_a_new = log_a - eta * v
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Adaptive eta: reduce if not improving
            if no_improve % 50 == 0:
                eta *= 0.98
            if no_improve % 200 == 0:
                temp = max(temp * 1.05, 0.001)
        
        a = a_new
        log_a = np.log(a + 1e-30)
        step += 1
        
        if no_improve >= patience:
            break
        
        # Periodically increase top_k and temp to explore
        if step % 500 == 0:
            top_k = min(20, 3 + step // 500)
            temp = min(0.1, 0.001 * step / 500)
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_step(a, max_iter=50):
    """
    One LP-based step: find direction g that maximizes sum(g)
    subject to conv(a,g) <= M and g >= 0.
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max() * 1.0001
    
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'time_limit': 5})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-6:
                # Try different step sizes
                best_a = a.copy()
                best_obj = 2.0 * n * conv.max()
                d = g - a
                for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s < 1e-15:
                        continue
                    a_try = a_try / s
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a, best_obj
    except Exception:
        pass
    return a.copy(), 2.0 * n * conv.max()


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Sine-based
    inits['sine1'] = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine1'] /= inits['sine1'].sum()
    
    inits['sine2'] = np.sin(2 * np.pi * x / n) ** 2 + 0.01
    inits['sine2'] /= inits['sine2'].sum()
    
    inits['sine3'] = np.sin(3 * np.pi * x / n) ** 2 + 0.01
    inits['sine3'] /= inits['sine3'].sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Triangular (peaked in middle)
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n-1)) / n
    tri = np.maximum(tri, 0.01)
    inits['triangular'] = tri / tri.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps
    
    return inits


def coordinate_descent_polish(a0, time_limit):
    """Quick coordinate descent polish."""
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()

        return a, float('inf')
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()

    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        # Find peak
        k_star = int(np.argmax(conv))
        
        # Try to reduce the peak by adjusting elements
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        
        # Get the contributing pairs
        pairs = []
        for i in range(i_start, i_end + 1):
            j = k_star - i
            pairs.append((i, j))
        
        # Sort by contribution (largest first)
        pairs.sort(key=lambda p: a[p[0]] * a[p[1]], reverse=True)
        
        improved = False
        for i, j in pairs[:20]:
            # Try moving mass from the larger to the smaller
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            
            for frac in [0.01, 0.05, 0.1, 0.2]:
                delta = frac * a[src]
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            # Try random perturbations
            idx = rng.integers(0, n)
            for delta in [0.001, -0.001, 0.005, -0.005]:
                a_try = a.copy()
                a_try[idx] += delta
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Fast mirror descent (multiplicative weights) on the probability simplex
   with exact subgradients, adaptive top-k peak weighting, and log-space momentum
2. Multi-initialization rapid scan followed by deep refinement
3. LP-based direction finding for escape from local minima
4. Targeted coordinate descent polish on peak-contributing pairs
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, conv, top_k=1, temp=0.0):
    """
    Compute (softmax-weighted) subgradient of max_k b_k w.r.t. a.
    For top_k=1: exact subgradient at the argmax lag.
    For top_k>1 with temp>0: smoothed approximation.
    """
    n = len(a)
    if top_k <= 1 or temp <= 0.0:
        k = int(np.argmax(conv))
        grad = np.zeros(n)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            idxs = np.arange(i_start, i_end + 1)
            grad[idxs] += 2.0 * a[k - idxs]
        return grad
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            k = int(peaks[np.argmax(peak_vals)])
            grad = np.zeros(n)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idxs = np.arange(i_start, i_end + 1)
            grad[idxs] += 2.0 * a[k - idxs]
            return grad
        w /= w_sum
        grad = np.zeros(n)
        for idx in range(top_k):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end < i_start:
                continue
            idxs = np.arange(i_start, i_end + 1)
            grad[idxs] += w[idx] * 2.0 * a[k - idxs]
        return grad


def mirror_descent(a0, time_limit, eta_init=0.3, top_k=1, temp_init=0.0,
                   momentum=0.0, patience=2000, anneal_rate=1.001):
    """
    Mirror descent (multiplicative weights) on the probability simplex.
    Minimizes max_k (a*a)_k subject to a >= 0, sum(a) = 1.
    Uses log-space parameterization for efficiency.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    log_a = np.log(a + 1e-30)
    v = np.zeros(n)  # momentum buffer

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    temp = temp_init
    no_improve = 0
    step = 0
    top_k_cur = top_k

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at current point
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        grad = compute_subgradient(a, conv, top_k=top_k_cur, temp=temp)

        # Momentum in log-space
        v = momentum * v + (1.0 - momentum) * grad

        # Multiplicative update: a_new_i proportional to a_i * exp(-eta * v_i)
        log_a_new = log_a - eta * v
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            break
        a_new /= a_new_sum

        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        log_a = np.log(a + 1e-30)
        step += 1

        # Adaptive parameter updates
        if no_improve > 100:
            eta *= 0.97
        if no_improve > 300:
            eta *= 0.95
            temp = max(temp * anneal_rate, 0.001)
            top_k_cur = min(15, top_k_cur + 1)

        # Periodic "exploration" bump
        if step % 1000 == 0 and no_improve < 500:
            top_k_cur = min(10, 1 + step // 500)
            temp = min(0.05, 0.001 * (1 + step // 1000))

        if no_improve >= patience:
            break

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_refine(a, time_limit=30.0):
    """
    LP-based refinement: repeatedly find direction g maximizing sum(g)
    subject to conv(a,g) <= M, g >= 0, then take a step toward g.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 30:
        M = conv.max() * 1.0001
        C = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        d = None
        try:
            result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'time_limit': 10})
            if result.success:
                g = result.x
                g_sum = g.sum()
                if g_sum > 1.0 + 1e-5:
                    d = g - a
        except Exception:
            pass

        if d is None or np.linalg.norm(d) < 1e-12:
            no_improve += 1
            break

        # Line search
        improved = False
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]:
            a_try = a + t * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-10:
                a = a_try
                conv = conv_try
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1

    return best_a, best_obj


def coordinate_polish(a0, time_limit=30.0):
    """
    Targeted coordinate descent: identify peak-contributing pairs and
    try to reduce their products by mass transfer.
    """
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 80:
        k_star = int(np.argmax(conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)

        # Get contributing pairs sorted by product
        pairs = []
        for i in range(i_start, i_end + 1):
            j = k_star - i
            pairs.append((i, j, a[i] * a[j]))
        pairs.sort(key=lambda x: x[2], reverse=True)

        found_improvement = False
        for i, j, _ in pairs[:30]:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            # Try to equalize a[i] and a[j] (reduces product for fixed sum)
            if abs(a[i] - a[j]) < 1e-10:
                continue
            src, dst = (i, j) if a[i] > a[j] else (j, i)
            for frac in [0.1, 0.2, 0.3, 0.5]:
                delta = frac * (a[src] - a[dst])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-11:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    found_improvement = True
                    break

        if not found_improvement:
            # Try random small perturbations
            improved_random = False
            for _ in range(5):
                idx = rng.integers(0, n)
                for delta in [0.002, 0.005, -0.002, -0.005, 0.01]:
                    a_try = a.copy()
                    a_try[idx] += delta
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-11:
                        a = a_try
                        conv = conv_try
                        best_obj = obj_try
                        best_a = a.copy()
                        no_improve = 0
                        improved_random = True
                        break
                if improved_random:
                    break
            if not improved_random:
                no_improve += 1

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for the optimizer."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Sine-based (suppresses autocorrelation at certain lags)
    for freq in [1, 2, 3]:
        key = 'sine{}'.format(freq)
        sig = np.sin(freq * np.pi * x / n) ** 2 + 0.01
        inits[key] = sig / sig.sum()

    # Bimodal (mass at edges reduces central autocorrelation)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Cosine window
    cos_w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Alternating (reduces even-lag autocorrelation)
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid mirror descent + LP + coordinate descent search.
    Strategy:
    - Phase 1: Rapid scan of many initializations with short mirror descent runs
    - Phase 2: Deep mirror descent refinement on top candidates
    - Phase 3: LP-based refinement for larger jumps
    - Phase 4: Coordinate descent + final micro-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print("Parent loaded: n={}".format(len(p)))
    except Exception:
        pass

    # Primary length
    if parent is not None:
        n_primary = len(parent)
    else:
        n_primary = 512

    # =========================================================
    # Phase 1: Rapid mirror descent scan
    # =========================================================
    print("=== Phase 1: Rapid MD Scan ===")
    t_start = time.monotonic()

    lengths_to_try = [n_primary]
    if n_primary != 512:
        lengths_to_try.append(512)
    if n_primary != 1024:
        lengths_to_try.append(1024)

    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.3)
    results = []

    for n in lengths_to_try:
        if time.monotonic() > deadline - 900:
            break

        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'uniform', 'sine1', 'sine2', 'cosine', 'alternating',
                    'bimodal', 'sine3', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:6]

        n_configs = 2
        t_per = max(8.0, phase1_budget / (len(lengths_to_try) * len(init_names) * n_configs))

        for name in init_names:
            if time.monotonic() > deadline - 850:
                break
            a0 = inits[name]
            for ci in range(n_configs):
                t_alloc = min(t_per, deadline - time.monotonic() - 800)
                if t_alloc < 5:
                    break
                if ci == 0:
                    cfg = {'eta_init': 0.5, 'top_k': 1, 'temp_init': 0.0,
                           'momentum': 0.0, 'patience': 2000}
                else:
                    cfg = {'eta_init': 1.0, 'top_k': 3, 'temp_init': 0.005,
                           'momentum': 0.3, 'patience': 1500}
                a_opt, obj = mirror_descent(a0, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    results.append((score, n, name, ci, a_opt.copy()))
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print("  MD n={}, {}: {:.6f}".format(n, name, score))

    results.sort(key=lambda x: x[0])
    print("Phase 1 done ({:.0f}s). Best: {:.6f}".format(time.monotonic() - t_start, best_score))

    # =========================================================
    # Phase 2: Deep mirror descent refinement
    # =========================================================
    print("=== Phase 2: Deep MD Refinement ===")
    t_start2 = time.monotonic()

    # Select top candidates (diverse by length+init)
    top_candidates = []
    seen_keys = set()
    for score, n, name, ci, a_opt in results:
        key = (n, name)
        if key not in seen_keys:
            top_candidates.append((n, a_opt))
            seen_keys.add(key)
        if len(top_candidates) >= 4:
            break

    if best_sequence is not None:
        bl = len(best_sequence)
        if bl not in [c[0] for c in top_candidates]:
            top_candidates.append((bl, best_sequence.copy()))

    remaining = deadline - time.monotonic() - 350
    n_cand = max(1, len(top_candidates))
    t_per_cand = max(30.0, remaining * 0.6 / n_cand)

    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 300:
            break
        t_cand = min(t_per_cand, deadline - time.monotonic() - 250)
        if t_cand < 15:
            break

        # Run 3 configs with slight perturbation
        for pi in range(3):
            if time.monotonic() > deadline - 250:
                break
            t_run = t_cand / 3.0
            if t_run < 8:
                break

            if pi == 0:
                a_start = a0.copy()
            elif pi == 1:
                noise = rng.standard_normal(n) * 0.003
                a_start = np.maximum(a0 + noise, 0)
                a_start = a_start / a_start.sum()
            else:
                noise = rng.standard_normal(n) * 0.008
                a_start = np.maximum(a0 + noise, 0)
                a_start = a_start / a_start.sum()

            cfgs = [
                {'eta_init': 0.3, 'top_k': 1, 'temp_init': 0.0, 'momentum': 0.0, 'patience': 5000},
                {'eta_init': 0.6, 'top_k': 2, 'temp_init': 0.002, 'momentum': 0.2, 'patience': 3000},
                {'eta_init': 1.5, 'top_k': 5, 'temp_init': 0.01, 'momentum': 0.5, 'patience': 2000},
            ]
            a_opt, obj = mirror_descent(a_start, time_limit=t_run, **cfgs[pi])
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  DeepMD {}-{} (n={}): {:.6f} *NEW BEST*".format(i, pi, n, score))

    print("Phase 2 done ({:.0f}s). Best: {:.6f}".format(time.monotonic() - t_start2, best_score))

    # =========================================================
    # Phase 3: LP refinement
    # =========================================================
    print("=== Phase 3: LP Refinement ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining_lp = deadline - time.monotonic() - 200
        if remaining_lp > 30:
            n_best = len(best_sequence)
            a = best_sequence.copy()
            a = np.maximum(a, 0)
            a = a / a.sum()

            for li in range(6):
                if time.monotonic() > deadline - 150:
                    break
                t_lp = min(remaining_lp / (6 - li) * 0.7, 45.0)
                if t_lp < 5:
                    break

                a_opt, obj = lp_refine(a, time_limit=t_lp)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    a = np.maximum(a_opt, 0)
                    a = a / a.sum()
                    print("  LP {}: {:.6f} *NEW BEST*".format(li, score))
                else:
                    # Perturb before next attempt
                    noise = rng.standard_normal(n_best) * 0.003
                    a = np.maximum(np.maximum(best_sequence, 0) / np.maximum(best_sequence, 0).sum() + noise, 0)
                    a = a / a.sum()

    print("Phase 3 done ({:.0f}s). Best: {:.6f}".format(time.monotonic() - t_start3, best_score))

    # =========================================================
    # Phase 4: Final polish (coordinate descent + micro MD)
    # =========================================================
    print("=== Phase 4: Final Polish ===")

    if best_sequence is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 20

        # Coordinate descent
        t_cd = min(remaining * 0.4, 40.0)
        if t_cd > 5:
            a_opt, obj = coordinate_polish(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  CoordDesc: {:.6f} *NEW BEST*".format(score))

        # Final micro mirror descent
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            a_opt, obj = mirror_descent(best_sequence, time_limit=remaining,
                                        eta_init=0.1, top_k=1, temp_init=0.0,
                                        momentum=0.0, patience=10000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  FinalMD: {:.6f} *NEW BEST*".format(score))

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score={:.6f}, n={}".format(best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
