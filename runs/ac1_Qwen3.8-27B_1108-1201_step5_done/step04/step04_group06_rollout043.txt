def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smoothed_gradient(a, conv, temp, top_k=50):
    """Compute smoothed gradient of max(conv) using top-k lags."""
    n = len(a)
    # Use top-k lags with softmax weighting
    k_sel = np.argpartition(conv, -top_k)[-top_k:]
    vals = conv[k_sel]
    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Build full weight vector
    w_full = np.zeros(2*n - 1)
    w_full[k_sel] = w
    
    # Gradient: g_j = sum_k w_full[k] * da*a[k]/da[j]
    # da*a[k]/da[j] = 2*a[k-j] (if k-j in [0,n-1] and 2j != k) or a[k-j] (if 2j == k)
    # Approximation: g_j ≈ 2 * (w_full correlated with a)[j]
    
    # Use FFT: g/2 = correlate(w_full, a_rev) where a_rev[j] = a[n-1-j]
    a_rev = a[::-1]
    fft_size = 1 << (2*n - 1 + n - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    R = np.fft.rfft(a_rev, fft_size)
    corr = np.fft.irfft(W * R, fft_size)
    # (w_full * a_rev)[k] = corr[k], and g/2 at position j is corr[j + n - 1]
    g = 2.0 * corr[n-1 : n-1+n]
    
    # Fix the diagonal (where 2j == k)
    for j in range(n):
        k = 2*j
        if k < 2*n - 1:
            g[j] -= w_full[k] * a[j]  # Remove one copy since we double-counted
    
    return g

            break

def lp_step(a, conv):
    """One LP-based step (simplified version)."""
    n = len(a)
    M = conv.max()
    
    # Build constraint matrix C such that (C @ g)[k] = conv(a, g)[k]
    # conv(a, g)[k] = sum_j a[j] * g[k-j]
    # C is (2n-1) x n
    C = np.zeros((2*n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    
    # Solve: max sum(g) s.t. C@g <= M, g >= 0
    c = -np.ones(n)
    b = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                        options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-8:
                # Line search
                best_a = a.copy()
                best_obj = M
                for t in [0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
                    a_try = np.maximum(a + t * (g - a), 0)
                    s = a_try.sum()
                    if s < 1e-15:

                    a_try /= s
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                return best_a, best_obj
    except:
        pass
    return a.copy(), M

def compute_gradient(a, w_full):
    """Compute gradient of weighted sum of convolution terms."""
    n = len(a)
    L = 1 << (3*n - 2).bit_length()
    w_pad = np.zeros(L)

    a_pad = np.zeros(L)
    a_pad[:n] = a
    corr = np.fft.ifft(np.fft.fft(w_pad) * np.conj(np.fft.fft(a_pad))).real

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric initialization (t^alpha * (1-t)^beta family)
2. Mirror descent (exponential weights) with FFT-accelerated gradients
3. Adaptive temperature annealing and step sizing
4. Periodic LP-based restarts
5. Coordinate descent polishing
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_gradient(a, conv, temp, top_k=50):
    """
    Compute smoothed gradient of max(conv(a,a)) using top-k peak weighting.
    Returns gradient vector g of length n.
    """
    n = len(a)
    # Select top-k peaks
    if top_k >= 2 * n - 1:
        k_sel = np.arange(2 * n - 1)
    else:
        k_sel = np.argpartition(conv, -top_k)[-top_k:]
    
    vals = conv[k_sel]
    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum
    
    # Build full weight vector
    w_full = np.zeros(2 * n - 1)
    w_full[k_sel] = w
    
    # Compute gradient via FFT: g[m] = 2 * sum_j w_full[m+j] * a[j]
    # This is 2 * (w_full cross-correlate a)[m]
    L = 1 << (3 * n - 2).bit_length()
    w_pad = np.zeros(L)
    w_pad[:2 * n - 1] = w_full
    a_pad = np.zeros(L)
    a_pad[:n] = a
    
    wf = np.fft.fft(w_pad)
    af = np.fft.fft(a_pad)
    corr = np.fft.ifft(wf * np.conj(af)).real
    
    g = 2.0 * corr[:n]
    return g


def mirror_descent_optimize(a0, time_limit, temp_init=0.001, eta_init=1.0, top_k=50):
    """
    Mirror descent on the simplex to minimize max(conv(a,a)).
    Uses exponential weights update and adaptive parameters.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    cur_peak = conv.max()
    best_peak = cur_peak
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = eta_init
    temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smoothed gradient
        g = compute_smoothed_gradient(a, conv, temp, top_k=top_k)
        
        # Mirror descent (exponential weights) update
        # a_new[j] = a[j] * exp(-eta * g[j]) / Z
        log_weights = np.log(a) - eta * g
        log_weights -= log_weights.max()  # Numerical stability
        exp_weights = np.exp(log_weights)
        Z = exp_weights.sum()
        if Z < 1e-300:
            # Underflow: fall back to uniform
            a = best_a.copy()
            conv = fast_convolve(a)
            break
        a_new = exp_weights / Z
        
        conv_new = fast_convolve(a_new)
        peak_new = conv_new.max()
        
        if peak_new < best_peak - 1e-14:
            best_peak = peak_new
            best_a = a_new.copy()
            no_improve = 0
            # Reward: increase eta slightly
            eta = min(eta * 1.02, 50.0)
        elif peak_new < cur_peak:
            no_improve += 1
            eta *= 0.95
        else:
            no_improve += 1
            eta *= 0.85
        
        a = a_new
        conv = conv_new
        cur_peak = peak_new
        
        # Anneal temperature (slowly increase for exploration, then decrease)
        fraction = (time.monotonic() - t0) / time_limit
        if fraction < 0.3:
            temp *= 1.001  # Warm up
        elif fraction < 0.7:
            temp *= 0.999  # Cool down
        else:
            temp *= 0.998  # Cool faster
        
        temp = max(temp, 0.0001)
        
        if no_improve > 1000:
            break
    
    return best_a, 2.0 * n * best_peak


def lp_step(a, conv):
    """One LP-based step. Returns (a_new, peak_new)."""
    n = len(a)
    M = conv.max()
    
    # Build constraint matrix: (C@g)[k] = conv(a, g)[k]
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'presolve': True,
                        'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-6:
                # Line search
                best_a = a.copy()
                best_peak = M
                for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
                    a_try = np.maximum(a + t * (g - a), 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try /= s_try
                    conv_try = fast_convolve(a_try)
                    pk = conv_try.max()
                    if pk < best_peak:
                        best_peak = pk
                        best_a = a_try.copy()
                return best_a, best_peak
    except Exception:
        pass
    return a.copy(), M


def coordinate_descent(a0, time_limit, rng=None):
    """Coordinate descent: random pairwise mass transfers."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    a = np.maximum(a0, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_peak = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(300000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.001, 0.003, 0.005, 0.01, -0.001, -0.003, -0.005, -0.01]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a[i] += delta
                a[j] -= delta
                conv_try = fast_convolve(a)
                pk = conv_try.max()
                if pk < best_peak - 1e-12:
                    best_peak = pk
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
                else:
                    a[i] -= delta
                    a[j] += delta
                    break
            else:
                no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, 2.0 * n * best_peak


def parametric_init(n, alpha, beta):
    """Create parametric initialization f(t) = t^alpha * (1-t)^beta."""
    t = np.linspace(0, 1, n, endpoint=False)
    # Avoid 0^0 issues
    t_safe = np.clip(t, 1e-10, 1 - 1e-10)
    a = t_safe**alpha * (1 - t_safe)**beta
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
