    # Cross-correlation: sum_p a[p] * w[p+i] = ifft(conj(A) * W)

    second = np.fft.irfft(A_fft * W_fft, N)[:n]

def fast_convolve(a):
    """Linear auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_gradient(a, R, t):
    """
    Compute gradient of F(a) = (1/t)*log(sum_k exp(t*R_k)) w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns gradient vector of length n.
    """
    n = len(a)
    tR = t * R
    tR -= tR.max()
    w = np.exp(tR)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(R))
        w[np.argmax(R)] = 1.0
    else:
        w /= w_sum

    N = 1 << (4 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)

    first = np.fft.irfft(np.conj(A_fft) * W_fft, N)[:n]
    second = np.fft.irfft(A_fft * W_fft, N)[:n]

    return first + second


def obj_value(a):
    """Compute objective value 2*n*max_b/(sum_a)^2."""
    n = len(a)
    s = a.sum()

    R = fast_convolve(a)
    return 2.0 * n * R.max() / (s * s)


def normalize(a):

    a = np.maximum(a, 0.0)

def build_conv_matrix(a):

def multiplicative_update_optimize(a0, time_limit, num_iters=500):
    """
    Multiplicative update heuristic for flattening autocorrelation.
    Similar to updates used in radar waveform design.
    """
    n = len(a0)
    a = normalize(a0.copy())
    R = fast_convolve(a)
    best_obj = 2.0 * n * R.max()
    best_a = a.copy()
    t0 = time.monotonic()

    for it in range(num_iters):
        if time.monotonic() - t0 > time_limit:
            break

        R = fast_convolve(a)
        cur_obj = 2.0 * n * R.max()
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()

        # Find the peak lag
        k_peak = int(np.argmax(R))
        R_peak = R[k_peak]

        # Compute "pressure" on each element from the peak
        # pressure[i] = a[k_peak - i] + a[i - k_peak] (with bounds)
        pressure = np.zeros(n)
        for i in range(n):
            j1 = k_peak - i
            j2 = i - k_peak
            if 0 <= j1 < n:
                pressure[i] += a[j1]
            if 0 <= j2 < n:
                pressure[i] += a[j2]

        # Multiplicative update: reduce elements with high pressure, increase those with low
        target = R_peak / n  # desired "flat" level
        ratio = target / np.maximum(pressure, 1e-15)
        ratio = np.clip(ratio, 0.3, 3.0)
        eta = 0.1
        a_new = a * ratio ** eta
        a_new = normalize(a_new)

        # Accept if improved (with small probability of acceptance for diversity early on)
        R_new = fast_convolve(a_new)
        obj_new = 2.0 * n * R_new.max()
        if obj_new < cur_obj or (it < 50 and rng_float() < 0.1):
            a = a_new
        else:
            # Try smaller step
            a_new2 = a * (ratio ** (eta / 3))
            a_new2 = normalize(a_new2)
            R_new2 = fast_convolve(a_new2)
            obj_new2 = 2.0 * n * R_new2.max()
            if obj_new2 < cur_obj:
                a = a_new2

    return best_a, best_obj

def simulated_annealing(a0, time_limit, T_init=0.01, T_min=0.0001, cool_rate=0.999):
    """
    Simulated annealing on the sequence space.
    Perturbations: random pair swaps and scaling.
    """
    n = len(a0)
    a = normalize(a0.copy())
    R = fast_convolve(a)
    best_obj = 2.0 * n * R.max()
    best_a = a.copy()

    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        # Generate perturbation
        pert_type = rng.integers(0, 3)
        if pert_type == 0:
            # Scale two elements
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            factor = rng.uniform(0.5, 2.0)
            a_try = a.copy()
            a_try[i] *= factor
            a_try[j] /= factor
        elif pert_type == 1:
            # Gaussian perturbation
            mask = rng.random(n) < 0.3
            noise = rng.standard_normal(n) * 0.02
            a_try = a.copy()
            a_try[mask] += noise[mask]
        else:
            # Swap mass
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            amt = a[i] * rng.uniform(0.1, 0.5)
            a_try = a.copy()
            a_try[i] -= amt
            a_try[j] += amt

        a_try = normalize(a_try)
        R_try = fast_convolve(a_try)
        obj_try = 2.0 * n * R_try.max()
        cur_obj = 2.0 * n * R.max()

        # Metropolis criterion
        delta = obj_try - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_try
            R = R_try
            if obj_try < best_obj - 1e-11:
                best_obj = obj_try
                best_a = a_try.copy()
                no_improve = 0
            T *= cool_rate
        else:
            no_improve += 1
            T *= cool_rate

        if no_improve > 500:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):

                    pp_smooth = gaussian_filter1d(pp, sigma=2)
                    pp_smooth = np.maximum(pp_smooth, 0)
                    pps2 = pp_smooth.sum()
                    if pps2 > 1e-10:
                        inits['parent_smooth'] = pp_smooth / pps2

    return inits


def extend_to_length(a, target_n):
    """Resample a sequence to a different length."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(target_n) / target_n
    a_norm = a / s
    if n == target_n:
        return a_norm.copy()
    a_ext = np.interp(np.linspace(0, 1, target_n), np.linspace(0, 1, n), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    else:
        a_ext = np.ones(target_n) / target_n
    return a_ext


def coordinate_polish(a0, time_limit):
    """
    Fine-grained local optimization: try small adjustments to individual elements.
    Focus on elements that contribute most to the peak autocorrelation.
    """
    n = len(a0)
    a = normalize(a0.copy())
    R = fast_convolve(a)
    best_obj = 2.0 * n * R.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        R = fast_convolve(a)
        cur_obj = 2.0 * n * R.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        # Identify important lags (top peaks)
        top_k = max(3, min(10, n // 50))
        peak_indices = np.argpartition(R, -top_k)[-top_k:]

        # For each peak, find which elements contribute most and try adjusting them
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit - 1:
                break
            # Elements contributing to lag k
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            contrib_idx = np.arange(j_start, j_end + 1)
            # Contribution of element j to lag k: a[j] * a[k-j]
            contrib = a[contrib_idx] * a[k - contrib_idx]
            # Try reducing the highest contributors
            top_contrib = np.argsort(contrib)[-min(5, len(contrib)):]
            for ci in top_contrib:
                j = contrib_idx[ci]
                for delta in [0.001, 0.003, 0.01, -0.003]:
                    if a[j] + delta < 0:
                        continue
                    a_try = a.copy()
                    a_try[j] += delta
                    # Compensate: reduce another element
                    others = np.delete(np.arange(n), j)
                    comp = rng.integers(0, len(others))
                    j2 = others[comp]
                    a_try[j2] -= delta
                    a_try = normalize(a_try)
                    R_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * R_try.max()
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = obj_try
                        best_a = a_try.copy()
                        no_improve = 0
                        break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-method search for optimal sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = p / p.sum()
            R_p = fast_convolve(p_norm)
            parent_score = 2.0 * len(p) * R_p.max()
            print(f"Parent: n={len(p)}, obj={parent_score:.6f}")
    except Exception:
        pass

    def record_result(a, label=""):
        nonlocal best_sequence, best_score
        if a is None:
            return
        score = float(evaluate_sequence([float(x) for x in a]))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  {label}: {score:.6f} *NEW BEST*")
        elif np.isfinite(score):
            print(f"  {label}: {score:.6f}")

    # =========================================================
    # Phase 1: Quick multi-length gradient scan (200s)
    # =========================================================
    print("=== Phase 1: Quick Gradient Scan ===")
    t_start = time.monotonic()
    phase1_time = min(200.0, (deadline - time.monotonic()) * 0.2)

    lengths_to_try = [512, 768, 1024]
    if parent is not None:
        lengths_to_try.insert(0, len(parent))
        lengths_to_try = sorted(set(lengths_to_try))

    scan_results = []
    t_per_length = phase1_time / len(lengths_to_try)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 300:
            break
        inits = generate_initializations(n, parent, rng)
        # Prioritize promising initializations
        priority = ['parent', 'parent_interp', 'parent_smooth', 'half_sine', 'sine1', 'gaussian', 'sym_exp', 'random', 'uniform']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]

        t_per_init = max(8.0, (t_per_length - 5) / len(init_names))

        for name in init_names:
            if time.monotonic() > deadline - 300:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 250)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj = gradient_optimize(a0, time_limit=t_alloc, t_init=0.02, eta_init=0.05)
            score_local = obj / (a_opt.sum() ** 2)  # approximate
            scan_results.append((score_local, n, name, a_opt.copy()))
            record_result(a_opt, f"Scan n={n} {name}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best so far: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep optimization on top candidates (400s)
    # =========================================================
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()
    phase2_time = min(400.0, (deadline - time.monotonic()) * 0.4)

    # Select top unique candidates
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n,)
        if key not in seen:
            top_candidates.append((n, a_opt))
            seen.add(key)
        if len(top_candidates) >= 4:
            break

    # Always include best
    if best_sequence is not None:
        if len(best_sequence) not in seen:
            top_candidates.append((len(best_sequence), best_sequence.copy()))
            seen.add(len(best_sequence))

    t_per_deep = max(20.0, phase2_time / max(1, len(top_candidates)))

    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 200:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 150)
        if t_alloc < 10:
            break

        print(f"  Deep opt {i}: n={n}, t={t_alloc:.0f}s")
        # Gradient with LP assist
        a_opt, obj = gradient_optimize(a0, time_limit=t_alloc * 0.6, t_init=0.015,
                                        eta_init=0.04, use_lp=True, lp_interval=30)
        record_result(a_opt, f"Deep-grad {i} n={n}")

        # Multiplicative update from best so far
        if time.monotonic() < deadline - 100:
            t_mu = min(30.0, deadline - time.monotonic() - 80)
            if t_mu > 5:
                a_mu, obj_mu = multiplicative_update_optimize(best_sequence if best_sequence is not None else a0,
                                                              time_limit=t_mu)
                record_result(a_mu, f"Multi-update {i}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Focused optimization with varied strategies (300s)
    # =========================================================
    print("=== Phase 3: Focused Optimization ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining3 = deadline - time.monotonic() - 100

        # Strategy A: Gradient with very low temperature (focus on single peak)
        t_a = min(60.0, remaining3 * 0.2)
        if t_a > 5:
            a_a, _ = gradient_optimize(best_sequence, time_limit=t_a, t_init=0.05, eta_init=0.02)
            record_result(a_a, "Focus-peak")

        # Strategy B: Simulated annealing
        t_b = min(50.0, remaining3 * 0.2)
        if t_b > 5:
            a_b, _ = simulated_annealing(best_sequence, time_limit=t_b, T_init=0.005, cool_rate=0.998)
            record_result(a_b, "Annealing")

        # Strategy C: Gradient with high temperature (broad smoothing)
        t_c = min(50.0, remaining3 * 0.15)
        if t_c > 5:
            a_c, _ = gradient_optimize(best_sequence, time_limit=t_c, t_init=0.005, eta_init=0.08)
            record_result(a_c, "Broad-smooth")

        # Strategy D: Try extending to larger n
        for n_ext in [n_best * 2]:
            if n_ext > 4096 or time.monotonic() > deadline - 100:
                break
            t_d = min(40.0, deadline - time.monotonic() - 80)
            if t_d > 5:
                a_ext = extend_to_length(best_sequence, n_ext)
                # Add slight perturbation
                noise = rng.standard_normal(n_ext) * 0.01
                a_ext = normalize(a_ext + noise)
                a_d, _ = gradient_optimize(a_ext, time_limit=t_d, t_init=0.02, eta_init=0.05)
                record_result(a_d, f"Extended n={n_ext}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Polish with coordinate descent and fine gradient
    # =========================================================
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining4 = deadline - time.monotonic() - 20

        # Coordinate polish
        t_cp = min(40.0, remaining4 * 0.3)
        if t_cp > 3:
            a_cp, _ = coordinate_polish(best_sequence, time_limit=t_cp)
            record_result(a_cp, "Coord-polish")

        # Fine gradient with multiple temperature settings
        remaining4 = deadline - time.monotonic() - 15
        if remaining4 > 10:
            for t_setting in [0.003, 0.01, 0.03]:
                if time.monotonic() > deadline - 10:
                    break
                t_fg = min(15.0, remaining4 * 0.3)
                if t_fg < 3:
                    break
                a_fg, _ = gradient_optimize(best_sequence, time_limit=t_fg,
                                            t_init=t_setting, eta_init=0.01)
                record_result(a_fg, f"Fine-grad t={t_setting}")

        # One more round of multiplicative updates
        remaining4 = deadline - time.monotonic() - 10
        if remaining4 > 5:
            a_mu2, _ = multiplicative_update_optimize(best_sequence, time_limit=min(15.0, remaining4))
            record_result(a_mu2, "Multi-update-final")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Ultimate final tweaks
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_final = min(10.0, deadline - time.monotonic() - 3)
        if t_final > 2:
            # Very small gradient steps
            a_final, _ = gradient_optimize(best_sequence, time_limit=t_final,
                                           t_init=0.02, eta_init=0.005)
            record_result(a_final, "Ultimate")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score
        best_sequence = np.array([float(max(0, x)) for x in best_sequence])

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
