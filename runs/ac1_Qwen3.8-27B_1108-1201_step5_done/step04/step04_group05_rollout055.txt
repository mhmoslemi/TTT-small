import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

"""
Multi-stage optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FW-LP as primary optimizer with multi-length exploration,
subgradient polishing with adaptive temperature, mirror descent steps,
and coordinate descent for final refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_step(f, sg, eta):
    """Exponential gradient (mirror) step on the simplex."""
    log_f = np.where(f > 1e-300, np.log(f), -30.0)
    new_log = log_f - eta * sg
    new_log -= new_log.max()
    f_new = np.exp(new_log)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_subgradient(f, conv, top_k=None, temp=0.001):
    """Compute softmax-weighted subgradient of max(conv) at f."""
    n = len(f)
    if top_k is None or top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()

            c = d
            fc = fd
            d = lo + gr * (hi - lo)

"""
Multi-stage optimization combining FW-LP with subgradient polishing.
Key improvements: adaptive LP frequency, all-peak subgradient, 
mirror descent steps, and aggressive final polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def subgradient_step(f, conv, lr, temp, use_all_peaks=False):
    """One subgradient step with softmax weighting."""
    n = len(f)
    if use_all_peaks:
        peaks = np.arange(len(conv))
        # Use all peaks but weight by softmax
    else:
        top_k = max(10, n // 3)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * 2.0 * f[k - idx]
    
    f_new = project_simplex(f - lr * sg)
    return f_new


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())
    
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)))
    inits.append(peak / peak.sum())
    
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def fw_lp_optimize(f0, time_limit, verbose=False, lp_every=1, sg_iters=3, 
                   init_temp=0.001, lr_init=0.005):

    Hybrid FW-LP + subgradient optimizer.
    Every lp_every subgradient iterations, solve LP for a new direction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    temp = init_temp
    lr = lr_init
    m_exp = 1.0
    rng_local = np.random.default_rng(999 + n)
    momentum = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Every lp_every iterations, solve LP
        if it % lp_every == 1:
            g = solve_lp(f, M * m_exp)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    diff = np.max(np.abs(g - f))
                    if diff > 1e-8:
                        # Line search for LP direction
                        best_alpha = 0.0
                        best_val = M
                        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
                        for alpha in alphas:

                            ft = np.maximum(ft, 0.0)

                        if best_alpha > 1e-10:
                            f = (1.0 - best_alpha) * f + best_alpha * g
                            f = np.maximum(f, 0.0)
                            fs = f.sum()
                            f = f / fs
                            conv = fast_convolve(f)
                            M = conv.max()
                            m_exp = 1.0
                            if M < best_obj - 1e-14:
                                best_obj = M
                                best_f = f.copy()
                                no_improve = 0
                            else:
                                no_improve += 1
                            continue
            m_exp = min(m_exp * 1.05, 3.0)

        # Subgradient step(s)
        for _ in range(sg_iters):
            top_k = max(10, n // 4)
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            scaled = pv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws
            
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-14:
                    continue
                k = int(peaks[i])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                sg[idx] += w[i] * 2.0 * f[k - idx]
            
            # Momentum
            momentum = 0.9 * momentum - lr * sg
            f_new = project_simplex(f + momentum)
            
            conv = fast_convolve(f_new)
            co = conv.max()
            
            if co < M:
                f = f_new
                M = co
            else:
                # Reject if it gets worse
                f = f  # keep old
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Decay
        temp = min(temp * 1.003, 0.05)
        lr *= 0.9995
        
        # Escape
        if no_improve > 60:
            noise = rng_local.standard_normal(n) * (0.002 + 0.001 * (no_improve // 30))
            f = project_simplex(best_f + noise)
            momentum = np.zeros(n)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            m_exp = 1.0
            temp = init_temp * 0.5
            lr = lr_init * 0.5

    return best_f, best_obj

        order = rng_local.permutation(n)

"""
Multi-stage optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses hybrid FW-LP + subgradient with momentum, multi-length exploration,
adaptive temperature scheduling, and aggressive multi-technique polishing.
Key innovations over prior approach: hybrid optimizer combining LP directions

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_sg(f, conv, top_k, temp):
    """Compute softmax-weighted subgradient."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * 2.0 * f[k - idx]
    return sg

                                continue

def sg_momentum_polish(f0, time_limit, lr_init=0.003, mom_init=0.92,

            temp = temp_init

def coord_descent_polish(f0, time_limit):
    """Random coordinate descent with multi-fraction search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng_local = np.random.default_rng(31337 + n)
    fractions = [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        improved = False
        for _ in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            j = int(rng_local.integers(n))
            src = int(rng_local.integers(n))
            if src == j or f[src] < 1e-14:
                continue

            best_delta = 0.0
            best_val = best_obj
            for frac in fractions:
                delta = f[src] * frac
                f_new = f.copy()
                f_new[j] += delta
                f_new[src] -= delta
                if f_new[src] < -1e-14:
                    continue
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                if sn < 1e-14:
                    continue
                f_new = f_new / sn
                val = fast_convolve(f_new).max()
                if val < best_val:
                    best_val = val
                    best_delta = delta

            if best_delta != 0.0:
                f = f.copy()
                f[j] += best_delta
                f[src] -= best_delta
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                best_obj = best_val
                best_f = f.copy()
                improved = True

        if not improved:
            # Try very small moves as last resort
            small_improved = False
            for _ in range(n // 3):
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                j = int(rng_local.integers(n))
                src = int(rng_local.integers(n))
                if src == j or f[src] < 1e-14:
                    continue
                for frac in [0.005, 0.01]:
                    delta = f[src] * frac
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[src] -= delta
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-14:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-15:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        small_improved = True
                        break
                if small_improved:
                    break
            if not small_improved:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    peak = (np.exp(-((x - n * 0.25) ** 2) / (2.0 * (n * 0.1) ** 2)) +
            np.exp(-((x - n * 0.75) ** 2) / (2.0 * (n * 0.1) ** 2)))
    inits.append(peak / peak.sum())

    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Bell-shaped with flat top
    bell = np.exp(-((x - n * 0.5) ** 2) / (2.0 * (n * 0.4) ** 2))
    inits.append(bell / bell.sum())

    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Perturbed parent
            px2 = np.maximum(px + rng.standard_normal(n) * 0.02, 0)
            px2 = px2 / px2.sum()
            inits.append(px2)

    return inits

    final_score = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(final_score):
        state['best_score'] = final_score

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
