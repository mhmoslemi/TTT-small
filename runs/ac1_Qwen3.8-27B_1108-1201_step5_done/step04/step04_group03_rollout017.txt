"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a robust multi-phase approach:
1. Multi-length initialization with diverse starting points
2. Frank-Wolfe with LP oracle for major improvements
3. Projected gradient with adaptive step size for refinement  
4. Coordinate descent for fine-tuning
5. Multi-restart local search to escape local minima
Key innovations: trying multiple sequence lengths, combining LP-with-FW
with lightweight gradient steps, and using a 2D coordinate descent
that directly optimizes pairs of coordinates.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n, f):
    """Build the (2n-1) x n matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = f[:k_end - j]
    return A


def solve_lp_conv(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_conv_matrix(n, f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success or res.status == 1:
            return np.maximum(res.x, 0)
    except:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_conv(f, M)
        if g is None:
            # Try with slightly relaxed M
            g = solve_lp_conv(f, M * 1.01)
            if g is None:
                no_improve += 5
                if no_improve > 30:
                    break
                continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919)
            scale = 0.003 + 0.005 * min(1.0, no_improve / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def proj_gradient_optimize(f0, time_limit, lr_init=0.01, n_top=10):
    """Projected gradient with top-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Get subgradient from top peaks
        if n_top >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -n_top)[-n_top:]
            pv = conv[peaks]
            weights = pv / pv.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if weights[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_idx = np.arange(js, je + 1)
            sg[j_idx] += 2.0 * weights[idx] * f[k - j_idx]

        f_new = project_simplex(f - lr * sg)
        if np.max(np.abs(f_new - f)) < 1e-12:
            lr *= 0.5
        else:
            f = f_new

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 100:
            lr *= 0.5
            if no_improve > 300:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999997)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
                lr = lr_init * 0.5

    return best_f, best_obj


def coord_descent(f0, time_limit, n_pairs=None):
    """Coordinate descent: optimize pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Precompute which pairs to try
    if n_pairs is None:
        n_pairs = min(n, 64)
    
    # Use a deterministic set of pairs based on symmetry
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        # Try a set of coordinate pairs
        # Strategy: pair elements that are "opposite" in the sequence
        num_pairs = min(n_pairs, n // 2)
        for _ in range(num_pairs):
            i, j = rng.choice(n, size=2, replace=False)
            if i == j:
                continue
            
            # Optimize t such that a[i] += t, a[j] -= t
            # t in [-a[i], a[j]]
            t_lo = -f[i]
            t_hi = f[j]
            
            if t_hi - t_lo < 1e-15:
                continue

            # Binary search for optimal t
            # The objective is max_k (conv[k] + t*(f[k-i] - f[k-j]) + t^2*[k==2i] - t^2*[k==2j])
            # This is a max of affine+quadratic functions, which is convex in t
            # So we can use ternary search on the convex function
            
            def obj_of_t(t):
                # Compute new conv and return max
                # For efficiency, compute the change
                # conv_new[k] = conv[k] + t*(f[k-i] - f[k-j]) + t^2*(delta_{k,2i} - delta_{k,2j})
                # But only where the indices are valid
                # For simplicity, just recompute
                f_new = f.copy()
                f_new[i] += t
                f_new[j] -= t
                if f_new[i] < -1e-15 or f_new[j] < -1e-15:
                    return 1e10
                c = fast_convolve(f_new)
                return c.max()
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            a, b = t_lo + 1e-12, t_hi - 1e-12
            if b - a < 1e-12:
                continue
            
            # First check endpoints and current
            c0 = obj_of_t(0.0)
            ca = obj_of_t(a)
            cb = obj_of_t(b)
            
            if ca < c0:
                c0 = ca
                best_t = a
            elif cb < c0:
                c0 = cb
                best_t = b
            else:
                best_t = 0.0
            
            # Golden section
            c1 = a + gr * (b - a)  # wait, golden section for minimization
            # Actually: for minimization on [a,b]:
            # c = a + (1-gr)*(b-a), d = a + gr*(b-a) where gr = (sqrt(5)-1)/2
            invgr = (np.sqrt(5) - 1) / 2  # = 0.618...
            
            c_pt = a + invgr * (b - a)
            d_pt = a + (1 - invgr) * (b - a)
            
            fc = obj_of_t(c_pt)
            fd = obj_of_t(d_pt)
            
            for _ in range(15):  # 15 iterations of golden section
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = a + invgr * (b - a)
                    fc = obj_of_t(c_pt)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + (1 - invgr) * (b - a)
                    fd = obj_of_t(d_pt)
            
            best_t = (a + b) / 2
            # Check this t
            ft = obj_of_t(best_t)
            if ft < c0 - 1e-14:
                best_t = (a + b) / 2
                f[i] += best_t
                f[j] -= best_t
                f = np.maximum(f, 0.0)
                improved = True
        
        if improved:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    x = np.arange(n, dtype=np.float64)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(cos_w / cos_w.sum())
    
    # U-shape (high at edges)
    u_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(u_w / u_w.sum())
    
    # Inverted U (high in middle)
    inv_u = (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(inv_u / inv_u.sum())
    
    # Beta-like (concentrated in middle)
    beta_w = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(beta_w / beta_w.sum())
    
    # Two bumps
    b1 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append((b1 + b2) / (b1 + b2).sum())
    
    # Dirichlet random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Parent interpolated
    if parent is not None and len(parent) > 1:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
