"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed subgradient descent
with Nesterov acceleration, adaptive temperature, and periodic LP-based escapes.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv(k))) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    # grad[p] = 2 * sum_j a[j] * w[j+p]
    # Computed via: reverse w, convolve with a, extract appropriate indices
    w_rev = w[::-1]
    # Need L >= n + (2n-1) - 1 = 3n - 2
    L = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w_rev, L)
    conv_result = np.fft.irfft(A * W, L)
    # c[p] = conv_result[2n-2-p] for p=0,...,n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv_result[n-1:2*n-1][::-1]
    return grad, conv

def lse_optimize(a_init, time_limit, beta_start=20.0, beta_end=5000.0, 
                 lr_start=0.05, momentum=0.9, use_nesterov=True):

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    beta = beta_start
    lr = lr_start
    no_improve = 0
    it = 0
    
    if use_nesterov:
        y = a.copy()  # lookahead point
        t = 1.0
    
    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        
        if use_nesterov:
            grad, conv = compute_lse_gradient(y, beta)
        else:
            grad, conv = compute_lse_gradient(a, beta)
        
        # Step
        a_new = y - lr * grad if use_nesterov else a - lr * grad
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

            t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
            gamma = (t - 1) / t_new
            y = a_new + gamma * (a_new - a)
            t = t_new
        else:
            y = a_new

            y = a.copy()
            t = 1.0
            conv = fast_convolve(a)
            beta = beta_start
            lr = lr_start
            no_improve = 0
    
    return best_a, best_obj

def solve_lp_escape(f, M, time_limit_lp=5):

    n = len(f)

        if res.success:
            return res.x
    except Exception:
        pass
    return None

    return None, None
