"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FISTA on log-sum-exp smoothing with adaptive temperature annealing
2. LP-based Frank-Wolfe with adaptive constraint scaling
3. Subgradient polishing with top-k peak weighting and momentum
4. Pairwise coordinate optimization for fine refinement
5. Final LP-FW squeeze
Key innovation: accelerated first-order methods (FISTA) with FFT-based
gradient computation for the smooth approximation, combined with LP-FW
for precise direction finding and careful temperature scheduling.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_fft(a, w, T):
    """Compute gradient of T*log(sum(exp(conv/T))) using FFT-based cross-correlation.
    w[k] = exp(conv[k]/T) normalized (softmax weights), length 2n-1.
    Returns grad of length n."""
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, N)
    A_rev = a[::-1]
    A = np.fft.rfft(A_rev, N)
    conv_result = np.fft.irfft(W * A, N)
    cc = conv_result[n - 1:2 * n - 1]
    return (2.0 / T) * cc


def smooth_objective_value(conv, T):
    """Compute T * log(sum(exp(conv/T)))."""
    z = conv / T
    z -= z.max()
    return T * (z.max() + np.log(np.sum(np.exp(z))))


def fista_smooth(a0, time_limit, T_init=None, T_min=None, verbose=False):
    """FISTA on log-sum-exp smoothed objective with temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    max_conv = conv.max()
    if T_init is None:
        T_init = max_conv * 5.0
    if T_min is None:
        T_min = max_conv * 0.02

    T = T_init
    best_obj = max_conv
    best_a = a.copy()

    y = a.copy()
    t = 1.0
    step = 1.0 / (4.0 / T)  # initial step size estimate
    no_improve = 0
    T_decrease_interval = 200
    T_decrease_rate = 0.92
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute convex at y
        conv_y = fast_convolve(y)
        MAX_CONV = conv_y.max() + 1e-15

        # Compute softmax weights
        z = conv_y / T
        z -= z.max()
        w = np.exp(z)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv_y))
            w[np.argmax(conv_y)] = 1.0
        else:
            w /= w_sum

        # Compute gradient
        grad = smooth_grad_fft(y, w, T)

        # Backtracking line search
        c = smooth_objective_value(conv_y, T)
        grad_norm_sq = np.sum(grad * grad)
        if grad_norm_sq < 1e-30:
            break
        L = 4.0 / T + 1.0  # initial Lipschitz estimate
        alpha = 1.0 / L

        for _ in range(30):
            x_new = project_simplex(y - alpha * grad)
            conv_x = fast_convolve(x_new)
            c_new = smooth_objective_value(conv_x, T)
            # Quadratic model: c + grad^T(x_new - y) + L/2 * ||x_new - y||^2
            model = c + np.dot(grad, x_new - y) + 0.5 * L * np.sum((x_new - y) ** 2)
            if c_new <= model + 1e-15:
                break
            L *= 2.0
            alpha = 1.0 / L
        else:
            x_new = project_simplex(y - alpha * grad)

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        a = x_new
        t = t_new

        # Track best
        conv_a = fast_convolve(a)
        obj = conv_a.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Temperature annealing
        if it % T_decrease_interval == 0 and T > T_min:
            T *= T_decrease_rate
        if no_improve > 100 and T > T_min:
            T = max(T * 0.85, T_min)

        if no_improve > 500:
            break

        if verbose and it % 500 == 0:
            print(f"    FISTA iter {it}: obj={best_obj:.10f}, T={T:.4e}")

    return best_a, best_obj


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng_local = np.random.default_rng(12345 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        g = solve_lp_direction(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
                # Try subgradient direction instead
                conv_f = fast_convolve(f)
                peaks = np.where(conv_f >= conv_f.max() - 1e-12)[0]
                sg = np.zeros(n)
                for k in peaks:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    sg[js:je + 1] += 2.0 * f[k - np.arange(js, je + 1)]
                sg /= len(peaks)
                g = project_simplex(f - 0.1 * sg)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fixed point check
        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 3.0)
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = fast_convolve((1 - c_pt) * f + c_pt * g).max()
        fd = fast_convolve((1 - d_pt) * f + d_pt * g).max()

        for _ in range(20):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = fast_convolve((1 - c_pt) * f + c_pt * g).max()
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = fast_convolve((1 - d_pt) * f + d_pt * g).max()

        best_alpha = (lo + hi) / 2.0
        ft = (1.0 - best_alpha) * f + best_alpha * g
        best_val = fast_convolve(ft).max()

        if best_val < M - 1e-14:
            f = ft
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    FW-LP iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        # Restart
        if no_improve > 60:
            strategies = it % 4
            if strategies == 0:
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            elif strategies == 1:
                f = project_simplex((1 - 0.2) * best_f + 0.2 * best_f[::-1])
            elif strategies == 2:
                f = project_simplex(best_f * (1 + 0.05 * rng_local.standard_normal(n)))
            else:
                f = project_simplex(rng_local.dirichlet(np.ones(n) * 1.5))
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr_init=0.005, mom=0.93, top_k=50, temp_init=0.0001):
    """Subgradient descent with multi-peak softmax weighting and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    rng_local = np.random.default_rng(8888)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Softmax weights over peaks
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Momentum update
        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.003

        if no_improve > 400:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.001)
            cur_temp = temp_init * 5
            no_improve = 0

        if cur_lr < 1e-7:
            break

    return best_f, best_obj


def pairwise_optimize(f0, time_limit):
    """Pairwise coordinate optimization: optimize pairs of coordinates jointly."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng_local = np.random.default_rng(31337)

    # Precompute which convolution positions are near the peak
    peak_threshold = best_obj * 0.95

    while time.monotonic() - t0 < time_limit - 1:
        improved_this_round = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break

            # Choose partner: prefer positions that affect peak conv positions
            # The peak is affected by pairs (i, k-i) for peak positions k
            peak_positions = np.where(conv >= peak_threshold)[0]
            # For position j, the relevant peaks are k where k-j is in [0,n-1]
            relevant = [k for k in peak_positions if 0 <= k - j < n]

            if relevant:
                # Partner that maximizes interaction with peak
                best_partner = -1
                best_score = -1
                for k in relevant[:10]:
                    partner = k - j
                    if 0 <= partner < n and partner != j:
                        score = f[partner] * f[j]
                        if score > best_score:
                            best_score = score
                            best_partner = partner
                if best_partner >= 0:
                    partners = [best_partner]
                else:
                    partners = [rng_local.integers(n) % n]
            else:
                partners = [rng_local.integers(n) % n]

            for partner in partners[:3]:
                partner = int(partner) % n
                if partner == j:
                    continue

                total = f[j] + f[partner]
                if total < 1e-15:
                    continue

                # 1D optimization: vary f[j] = t, f[partner] = total - t
                # Objective: max_k conv_new[k] where conv changes only at positions involving j or partner
                # Use golden section search on t in [0, total]
                lo_t, hi_t = 0.0, total
                gr = (np.sqrt(5.0) - 1.0) / 2.0

                c_t = hi_t - gr * (hi_t - lo_t)
                d_t = lo_t + gr * (hi_t - lo_t)

                def eval_t(t_val):
                    f_trial = f.copy()
                    f_trial[j] = t_val
                    f_trial[partner] = total - t_val
                    if f_trial[j] < -1e-15 or f_trial[partner] < -1e-15:
                        return best_obj + 1.0
                    f_trial = np.maximum(f_trial, 0.0)
                    return fast_convolve(f_trial).max()

                fc = eval_t(c_t)
                fd = eval_t(d_t)

                for _ in range(12):
                    if fc < fd:
                        hi_t = d_t
                        d_t = c_t
                        fd = fc
                        c_t = hi_t - gr * (hi_t - lo_t)
                        fc = eval_t(c_t)
                    else:
                        lo_t = c_t
                        c_t = d_t
                        fc = fd
                        d_t = lo_t + gr * (hi_t - lo_t)
                        fd = eval_t(d_t)

                best_t = (lo_t + hi_t) / 2.0
                f_trial = f.copy()
                f_trial[j] = best_t
                f_trial[partner] = total - best_t
                f_trial = np.maximum(f_trial, 0.0)
                val = fast_convolve(f_trial).max()

                if val < best_obj - 1e-14:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    conv = fast_convolve(f)
                    peak_threshold = best_obj * 0.95
                    improved_this_round = True

        if not improved_this_round:
            # Try random perturbation to escape
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.001)
            conv = fast_convolve(f)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_f = f.copy()
            else:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.8))

    # Edge-concentrated (inverse distance to center)
    center = 0.5 * (n - 1)
    edge_w = 1.0 / (np.abs(x - center) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - center) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine taper
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Power law
    pl = (x + 1.0) ** (-0.3)
    inits.append(pl / pl.sum())

    # Triangle
    tri = np.minimum(x + 1, n - x)
    inits.append(tri / tri.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if better."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s * np.sqrt(2 * len(a))  # normalize for evaluation
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining FISTA, LP-FW,
    subgradient polishing, and pairwise optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent available: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_seq'] = parent.copy()
                state['best_score'] = s
    except Exception:
        pass

    # === Phase 1: Quick exploration ===
    print("=== Phase 1: Quick Exploration ===")
    phase1_end = time.monotonic() + (float(budget_s) * 0.15)

    lengths = []
    if parent is not None:
        n_p = len(parent)
        # Quick FISTA on parent
        t_alloc = min(80, phase1_end - time.monotonic() - 30)
        if t_alloc > 15:
            print(f"  FISTA on parent (n={n_p}), t={t_alloc:.0f}s")
            a_opt, obj = fista_smooth(parent, time_limit=t_alloc)
            try_sequence(a_opt, "parent_fista", state)
        lengths.append(n_p)

    # Add other lengths
    for n in [256, 320, 384, 448, 512, 640, 768]:
        if n not in lengths:
            lengths.append(n)
    lengths.sort(key=lambda x: abs(x - (len(parent) if parent is not None else 384)))

    for n in lengths[:5]:
        if time.monotonic() > phase1_end - 20:
            break
        inits = generate_inits(n, parent, rng)
        n_try = min(3, len(inits))
        t_per = max(10, (phase1_end - time.monotonic() - 15) / n_try)
        t_per = min(t_per, 40)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 8:
                break
            a_opt, obj = fista_smooth(inits[i], time_limit=t_alloc)
            try_sequence(a_opt, f"n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep optimization ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + (float(budget_s) * 0.50)

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}")

        # FISTA deep run
        rem = phase2_end - time.monotonic()
        t_fista = min(200, rem * 0.4)
        if t_fista > 30:
            print(f"  FISTA deep, t={t_fista:.0f}s")
            a_opt, obj = fista_smooth(state['best_seq'], time_limit=t_fista, verbose=True)
            try_sequence(a_opt, "fista_deep", state)

        # LP-FW run
        rem = phase2_end - time.monotonic()
        t_fw = min(200, rem * 0.4)
        if t_fw > 20:
            print(f"  LP-FW, t={t_fw:.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_fw, verbose=True)
            try_sequence(a_opt, "fw_deep", state)

        # Perturbation + re-optimize
        perturb_scales = [0.003, 0.008, 0.02, 0.05]
        for ps in perturb_scales:
            rem = phase2_end - time.monotonic() - 10
            if rem < 15:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            t_alloc = min(rem * 0.3, 80)
            if t_alloc < 10:
                break
            a_opt, obj = fista_smooth(f_init, time_limit=t_alloc)
            try_sequence(a_opt, f"pert_{ps}", state)

        # Alternate FISTA and FW
        rem = phase2_end - time.monotonic()
        if rem > 40:
            t_half = rem * 0.5
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(t_half, 100))
            try_sequence(a_opt, "alt_fw", state)
            rem = phase2_end - time.monotonic()
            if rem > 20:
                a_opt, obj = fista_smooth(state['best_seq'], time_limit=min(rem - 5, 80))
                try_sequence(a_opt, "alt_fista", state)

    # === Phase 3: Polishing ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline - 30

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f}, n={n_best}")

        # Subgradient polish
        rem = phase3_end - time.monotonic() - 30
        if rem > 20:
            t_sg = min(rem * 0.4, 150)
            a_opt, obj = sg_polish(state['best_seq'], time_limit=t_sg,
                                   lr_init=0.003, mom=0.94,
                                   top_k=max(20, n_best // 4), temp_init=0.00005)
            try_sequence(a_opt, "sg_polish", state)

        # Pairwise optimization
        rem = phase3_end - time.monotonic() - 15
        if rem > 15:
            t_pw = min(rem * 0.5, 100)
            a_opt, obj = pairwise_optimize(state['best_seq'], time_limit=t_pw)
            try_sequence(a_opt, "pairwise", state)

        # Fine subgradient
        rem = phase3_end - time.monotonic() - 10
        if rem > 15:
            t_sg2 = min(rem * 0.6, 60)
            a_opt, obj = sg_polish(state['best_seq'], time_limit=t_sg2,
                                   lr_init=0.001, mom=0.96,
                                   top_k=max(10, n_best // 6), temp_init=0.00002)
            try_sequence(a_opt, "sg_fine", state)

        # Final LP-FW squeeze
        rem = phase3_end - time.monotonic()
        if rem > 10:
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem - 2, 50))
            try_sequence(a_opt, "fw_final", state)

        # Another round of pairwise
        rem = phase3_end - time.monotonic()
        if rem > 10:
            a_opt, obj = pairwise_optimize(state['best_seq'], time_limit=min(rem - 2, 40))
            try_sequence(a_opt, "pairwise_final", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform n=384: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
