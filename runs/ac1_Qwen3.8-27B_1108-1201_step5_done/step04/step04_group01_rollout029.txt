def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    rho = np.nonzero(cond)[0][-1]

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(weights, f):
    """
    Compute gradient of LSE objective w.r.t. f.
    grad[i] = 2 * sum_j weights[i+j] * f[j]
    Computed via FFT: grad = 2 * (flip(f) * weights)[n-1 : n-1+n]
    """
    n = len(f)
    b = f[::-1]  # b[j] = f[n-1-j]
    # b has length n, weights has length 2n-1
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_s = 1 << (conv_len - 1).bit_length()
    B = np.fft.rfft(b, fft_s)
    W = np.fft.rfft(weights, fft_s)
    bw = np.fft.irfft(B * W, fft_s)
    return 2.0 * bw[n - 1: n - 1 + n]


def lse_objective(a, beta):
    """Compute LSE approximation of max(conv)."""
    conv = fast_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    return (bc_max + np.log(np.exp(bc - bc_max).sum())) / beta

def diverse_init(n, rng):
    """Generate a diverse random initialization on the simplex."""
    # Mix of uniform and structured
    choice = rng.random()
    if choice < 0.3:
        # Uniform
        return np.ones(n) / n
    elif choice < 0.5:
        # Exponential decay
        f = np.exp(-np.linspace(0, 3, n))
        return f / f.sum()
    elif choice < 0.7:
        # Two peaks
        f = np.zeros(n)
        p1 = rng.integers(0, n // 2)
        p2 = n // 2 + rng.integers(0, n // 2)
        width = n // 8
        f[max(0, p1-width):p1+width+1] = 1.0
        f[max(0, p2-width):p2+width+1] = 1.0
        f += rng.standard_normal(n) * 0.1
        f = np.maximum(f, 0)
        return f / f.sum()
    else:
        # Random Dirichlet
        alpha = rng.uniform(0.5, 2.0, n)
        f = rng.dirichlet(alpha)
        return f

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multi-phase LSE (log-sum-exp) smoothed gradient descent with momentum
- Adaptive beta annealing, learning rate scheduling, and Nesterov-style acceleration
- Direct peak-shaving phase using weighted subgradient on active peaks
- Periodic structured restarts (block shuffles, noise injection) to escape local minima
- FFT-based fast convolution and gradient computation throughout
- Multiple diverse starting points for robustness
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(weights, f):
    """
    Compute gradient of LSE objective w.r.t. f.
    grad[i] = 2 * sum_j weights[i+j] * f[j]
    Computed via FFT: 2 * (flip(f) * weights)[n-1 : n-1+n]
    """
    n = len(f)
    b = f[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_s = 1 << (conv_len - 1).bit_length()
    B = np.fft.rfft(b, fft_s)
    W = np.fft.rfft(weights, fft_s)
    bw = np.fft.irfft(B * W, fft_s)
    return 2.0 * bw[n - 1: n - 1 + n]


def lse_descent(f0, time_limit, beta_start=30.0, beta_end=5000.0,
                lr_init=None, momentum_init=0.85, seed=42, use_nesterov=True):
    """
    LSE-smoothed gradient descent with momentum, adaptive parameters,
    optional Nesterov acceleration, and periodic restarts.
    Returns (best_f, best_max_conv) for normalized f (sum=1).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 4.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov: evaluate gradient at lookahead point
        if use_nesterov:
            y = project_simplex(f + momentum * v)
            eval_point = y
        else:
            eval_point = f

        conv = fast_convolve(eval_point)
        cur_max = conv.max()

        if cur_max < best_obj - 1e-13:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0

        # Compute LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_bc / w_sum

        # Compute gradient
        grad = compute_grad(w, eval_point)

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        # Evaluate new point
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0

        if obj_new < cur_max * (1.0 + 1e-10):
            f = f_new
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.94
            if no_improve >= 15:
                momentum = max(0.78, momentum - 0.008)
            if no_improve > 120:

                scale = 0.001 + 0.003 * min(1.0, (no_improve - 120) / 100.0)
                perturbation = rng.standard_normal(n) * scale

                    idx = np.arange(n)
                    rng.shuffle(idx)
                    perturbation = (best_f[idx] - best_f) * 0.1 + perturbation
                f = project_simplex(best_f + perturbation)
                v = np.zeros(n)
                lr = max(lr_init * 0.4, 0.001)
                momentum = momentum_init
                no_improve = 0

        # Anneal beta (logarithmic schedule)
        if beta < beta_end:
            progress = min(1.0, it / max(1, int(time_limit * 30)))
            beta = beta_start * (beta_end / beta_start) ** progress
            beta = min(beta, beta_end)

    return best_f, best_obj


def peak_shave(f0, time_limit, top_k=None, seed=42, lr_init=None):
    """
    Direct subgradient peak shaving with adaptive peak selection.
    Focuses gradient on the highest convolution peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 1.5 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    if top_k is None:
        top_k = max(6, n // 14)

    t0 = time.monotonic()
    lr = lr_init
    momentum = 0.90
    v = np.zeros(n)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    temp = 0.0005
    cur_mom = momentum

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        cur_max = conv.max()

        if cur_max < best_obj - 1e-13:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0

        # Select peaks: use a threshold-based approach + top-k
        threshold = best_obj * 0.995
        mask = conv >= threshold
        peak_indices = np.where(mask)[0]

        if len(peak_indices) < top_k:
            # Expand to top_k
            remaining = np.setdiff1d(np.arange(len(conv)), peak_indices)
            if len(remaining) > 0:
                top_remaining = np.partition(conv[remaining], -(top_k - len(peak_indices)))[-(top_k - len(peak_indices)):]
                extra_mask = conv >= top_remaining.min() * (1 - 1e-9)
                extra_indices = np.where(extra_mask & ~mask)[0]
                peak_indices = np.concatenate([peak_indices, extra_indices])

        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Softmax weights on selected peaks
        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        weights = np.exp(scaled)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(len(peak_indices))
            weights[np.argmax(pv)] = 1.0
        else:
            weights /= w_sum

        # Compute weighted subgradient
        sg = np.zeros(n)
        for idx in range(len(peak_indices)):
            if weights[idx] < 1e-14:
                continue
            k = int(peak_indices[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                sg[js] += 2.0 * weights[idx] * f[k - js]

        # Momentum + projection
        cur_mom = 0.97 * cur_mom + 0.03 * momentum  # slowly adapt
        v = cur_mom * v - lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < cur_max - 1e-14:
            f = f_new
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr *= 0.93
            if no_improve > 300:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0004)
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.0002)
                cur_mom = momentum
                no_improve = 0

        # Slowly increase temperature
        temp *= 1.0008

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS: it={it}, max_conv={best_obj:.10f}, score_est={2*n*best_obj:.6f}, lr={lr:.6f}, no_imp={no_improve}, t={elapsed:.0f}s")

    return best_f, best_obj


def diverse_init(n, rng):
    """Generate a diverse random initialization on the simplex."""
    choice = rng.random()
    if choice < 0.25:
        return np.ones(n) / n
    elif choice < 0.45:
        f = np.exp(-np.linspace(0, 4, n))
        f += rng.standard_normal(n) * 0.05
        f = np.maximum(f, 0)
        return f / f.sum()
    elif choice < 0.65:
        f = np.zeros(n)
        n_peaks = rng.integers(2, 5)
        for _ in range(n_peaks):
            center = rng.integers(0, n)
            width = rng.integers(n // 20, n // 5)
            f[max(0, center - width):min(n, center + width + 1)] += 1.0
        f += rng.standard_normal(n) * 0.05
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return f / s
    else:
        alpha = rng.uniform(0.5, 3.0, n)
        return rng.dirichlet(alpha)
