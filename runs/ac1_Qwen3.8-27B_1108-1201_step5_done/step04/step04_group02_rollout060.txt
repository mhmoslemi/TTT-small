def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, conv, T):

"""
Hybrid optimizer combining mirror descent (multiplicative weights with adaptive
temperature annealing), momentum subgradient descent, and coordinate descent
polishing. Key innovations over previous approach:
1. Multiplicative weights update naturally respects simplex constraint
2. Gradient centering for better conditioning
3. Soft-to-hard temperature schedule for the softmax peak weighting
4. Focused deep optimization on the parent sequence
5. Multi-length exploration to find optimal sequence length
"""

import numpy as np
import time


def fast_convolve(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, conv, T):
    """Compute smoothed subgradient: g_j = 2 * sum_m a_m * softmax(conv)_j+m.
    Uses softmax weighting over all convolution peaks controlled by temperature T."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    g = np.empty(n)
    for j in range(n):
        g[j] = 2.0 * np.dot(a, w[j:j + n])
    return g


def compute_subgrad_exact(a, conv):
    """Exact subgradient using only the argmax position(s)."""
    n = len(a)
    max_val = conv.max()
    peak_mask = np.abs(conv - max_val) < 1e-12
    w = peak_mask.astype(np.float64)
    w /= w.sum()
    g = np.empty(n)
    for j in range(n):
        g[j] = 2.0 * np.dot(a, w[j:j + n])
    return g


def mirror_descent(a0, time_limit, T_initial=0.05, T_final=0.0005,
                   lr_initial=None, lr_decay=0.99995, exact_frac=0.7):
    """Mirror descent (multiplicative weights) with adaptive temperature.
    
    The multiplicative update a_j <- a_j * exp(-lr * g_j_centered) is the
    natural descent direction for the negative entropy barrier on the simplex.
    Temperature annealing transitions from smooth (many peaks) to sharp (few peaks).
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    if lr_initial is None:
        lr_initial = n * 0.008

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    T = T_initial
    lr = lr_initial
    no_improve = 0
    it = 0
    T_ratio = T_final / T_initial
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        # Use exact subgradient in later phase
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        if frac > exact_frac:
            g = compute_subgrad_exact(a, conv)
        else:
            g = compute_subgrad(a, conv, T)

        # Center gradient for balanced multiplicative update
        g_centered = g - g.mean()

        # Multiplicative update with clipping
        log_update = -lr * g_centered
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)

        obj = conv_new.max()

            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            # Backtracking: try smaller steps
            success = False
            for factor in [0.5, 0.2, 0.05]:
                log_upd = np.clip(-lr * factor * g_centered, -2.0, 2.0)
                a_trial = a * np.exp(log_upd)

                    no_improve = 0

        # Anneal temperature (exponential decay over time)
        T = T_initial * (T_ratio ** frac)
        T = max(T, T_final)

        # Slow decay of learning rate
        lr *= lr_decay

    return best_a, best_obj


def sg_momentum(a0, time_limit, lr=0.004, mom=0.91, T=0.001, T_growth=1.003):

    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    cur_T = T

        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        g = compute_subgrad(a, conv, cur_T)
        v = mom * v - cur_lr * g
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:

                v *= 0.5
            if no_improve > 500:
                # Reset with perturbation
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(a)
                cur_lr = lr * 0.5
                cur_T = max(T * 0.5, 1e-5)
                v = np.zeros(n)
                no_improve = 0

        cur_lr *= 0.99995

    return best_a, best_obj


def coord_polish(a0, time_limit):
    """Coordinate descent style polishing: try pairwise mass transfers."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(4242)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        improved_this_round = False
        # Try random pairwise transfers
        for _ in range(30):
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # Try moving a small fraction of mass from i to j (and reverse)
            for frac in [0.05, 0.02, 0.01]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                # Transfer from i to j
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                s = a_trial.sum()
                a_trial = a_trial / s
                c = fast_convolve(a_trial)
                if c.max() < best_obj - 1e-15:
                    a = a_trial
                    conv = c
                    best_obj = c.max()
                    best_a = a.copy()
                    improved_this_round = True
                    break
                # Transfer from j to i
                a_trial2 = a.copy()
                a_trial2[j] -= amt
                a_trial2[i] += amt
                a_trial2 = a_trial2 / a_trial2.sum()
                c2 = fast_convolve(a_trial2)
                if c2.max() < best_obj - 1e-15:
                    a = a_trial2
                    conv = c2
                    best_obj = c2.max()
                    best_a = a.copy()
                    improved_this_round = True
                    break

        if not improved_this_round and it % 20 == 0:
            # Broader exploration step
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(a)

    return best_a, best_obj


def fg_smooth_optimize(a0, time_limit, T_start=0.02, T_end=0.0005):
    """Faster Gradient method with LSE-smoothed objective.
    Uses the smooth approximation L_T(max) = T*log(sum(exp(x/T))) which has
    a well-defined gradient everywhere, enabling better step size selection."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    T = T_start
    lr = n * 0.005
    no_improve = 0
    it = 0
    T_ratio = T_end / T_start
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = max(T_start * (T_ratio ** frac), T_end)

        # LSE gradient = softmax(conv/T)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient of LSE w.r.t. a
        g = np.empty(n)
        for j in range(n):
            g[j] = 2.0 * np.dot(a, w[j:j + n])

        # Projected gradient step with backtracking
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            a_try = project_simplex(a - alpha * g)
            c_try = fast_convolve(a_try)
            if c_try.max() < best_val:
                best_val = c_try.max()
                best_alpha = alpha
                a = a_try
                conv = c_try

        if best_alpha > 0 and conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 300:
                a = a * np.exp(rng.standard_normal(n) * 0.015)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                conv = fast_convolve(a)
                lr = n * 0.003
                no_improve = 0

    return best_a, best_obj

    print(f"  After Phase 3: best = {state[1]:.8f}")

    # === Phase 4: Deep optimization on best ===
    print("\n=== Phase 4: Deep Optimization ===")
    if state[0] is not None:
        best_a = state[0]

        for rnd in range(8):
            rem = deadline - time.monotonic() - 60
            if rem < 25:
                break
            t_alloc = min(rem // (9 - rnd), 120)
            n_best = len(best_a)

            if rnd % 3 == 0:
                # Mirror descent with perturbation
                if rnd == 0:
                    f_init = best_a.copy()
                else:
                    f_init = best_a * np.exp(rng.standard_normal(n_best) * 0.008)
                    f_init = np.maximum(f_init, 1e-15)
                    f_init = f_init / f_init.sum()
                a_opt, _ = mirror_descent(f_init, time_limit=t_alloc,
                                          T_initial=0.03, T_final=0.0001,
                                          lr_initial=n_best * 0.008, lr_decay=0.9999)
                eval_and_update(a_opt, f"deep_md_{rnd}", state)

            elif rnd % 3 == 1:
                # SG momentum
                f_init = best_a.copy()
                if rnd > 2:
                    f_init = project_simplex(f_init + rng.standard_normal(n_best) * 0.004)
                a_opt, _ = sg_momentum(f_init, time_limit=t_alloc,
                                       lr=0.002, mom=0.93, T=0.0003)
                eval_and_update(a_opt, f"deep_sg_{rnd}", state)

            else:
                # LSE smooth
                f_init = best_a.copy()
                a_opt, _ = fg_smooth_optimize(f_init, time_limit=t_alloc,
                                              T_start=0.015, T_end=0.0003)
                eval_and_update(a_opt, f"deep_lse_{rnd}", state)

            if state[0] is not None and not np.array_equal(state[0], best_a):
                best_a = state[0].copy()

    # === Phase 5: Final polishing ===
    print("\n=== Phase 5: Final Polish ===")
    if state[0] is not None:
        n_best = len(state[0])

        # Coordinate polish
        rem = deadline - time.monotonic() - 30
        if rem > 25:
            a_opt, _ = coord_polish(state[0], time_limit=min(rem, 80))
            eval_and_update(a_opt, "coord", state)

        # Ultra-fine mirror descent
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            a_opt, _ = mirror_descent(state[0], time_limit=min(rem, 50),
                                      T_initial=0.005, T_final=0.00005,
                                      lr_initial=n_best * 0.004, lr_decay=0.99995)
            eval_and_update(a_opt, "final_md", state)

        # Ultra-fine SG
        rem = deadline - time.monotonic() - 8
        if rem > 15:
            a_opt, _ = sg_momentum(state[0], time_limit=min(rem, 35),
                                   lr=0.0008, mom=0.96, T=0.0001, T_growth=1.002)
            eval_and_update(a_opt, "final_sg", state)

        # Micro perturbations + quick polish
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            for _ in range(3):
                rem = deadline - time.monotonic() - 4
                if rem < 8:
                    break
                f_pert = state[0] * np.exp(rng.standard_normal(n_best) * 0.003)
                f_pert = np.maximum(f_pert, 1e-15)
                f_pert = f_pert / f_pert.sum()
                a_opt, _ = mirror_descent(f_pert, time_limit=min(rem, 15),
                                          T_initial=0.003, T_final=0.00003,
                                          lr_initial=n_best * 0.003, lr_decay=0.99995)
                eval_and_update(a_opt, "micro", state)

    # Fallback
    if state[0] is None:
        fb = np.ones(384) / 384
        state[0] = fb
        state[1] = float(evaluate_sequence(fb.tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[0].tolist()))
    if np.isfinite(fs):
        state[1] = fs
    else:
        state[0] = np.ones(384) / 384
        state[1] = float(evaluate_sequence(state[0].tolist()))

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={state[1]:.8f}, n={len(state[0])}, time={elapsed:.1f}s")
    return [float(x) for x in state[0]]
