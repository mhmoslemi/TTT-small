Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses coordinate descent, mirror descent, and Nesterov-accelerated LSE.

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines coordinate descent, mirror descent (multiplicative updates),
Nesterov-accelerated log-sum-exp, and peak-shaving heuristics.
Key innovation: coordinate descent with analytical peak-equalization candidates.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(weights, f):
    """Compute gradient of LSE objective: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def peak_shave(f0, time_limit=30.0):
    """Directly shave peaks by reducing contributing elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    S = f.sum()
    if S < 1e-15:
        f = np.ones(len(f)) / len(f)
        S = 1.0
    f = f / S
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()
        obj = 2.0 * n * M

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()

        # Find the peak
        k_star = int(np.argmax(conv))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = f[k_star - idx] * f[idx]

        # Find the top contributors
        n_contrib = len(contribs)
        top_n = min(n_contrib, max(4, n_contrib // 4))
        top_idx = np.argpartition(contribs, -top_n)[-top_n:]

        # Reduce top contributors
        reduce_amount = 0.005
        for ti in top_idx:
            real_idx = idx[ti]
            f[real_idx] = max(0.0, f[real_idx] * (1.0 - reduce_amount))
            f[k_star - real_idx] = max(0.0, f[k_star - real_idx] * (1.0 - reduce_amount * 0.5))

        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        # Every 100 iterations, do a small CD step
        if it % 100 == 0:
            f, _ = coordinate_descent(f, time_limit=2.0)

    return best_f, best_obj / (2.0 * n)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize_seq(p)
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.10f}")
            best_tracker[0] = parent_score
            best_tracker[1] = parent.copy()
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, score={best_tracker[0]:.10f}")

    # === Phase 1: Coordinate Descent ===
    print("\n=== Phase 1: Coordinate Descent ===")
    t1 = min(300, deadline - time.monotonic() - 1000)
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")
        a_opt, _ = coordinate_descent(parent, time_limit=t1)
        try_sequence(a_opt, best_tracker)
        print(f"  After CD: {best_tracker[0]:.10f}")

    # === Phase 2: Mirror Descent ===
    print("\n=== Phase 2: Mirror Descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t2 = min(300, deadline - time.monotonic() - 700)
    if t2 > 30:
        print(f"  Allocating {t2:.0f}s")
        a_opt, _ = mirror_descent(current_best, time_limit=t2, eta_init=0.003)
        try_sequence(a_opt, best_tracker)
        print(f"  After Mirror: {best_tracker[0]:.10f}")

    # === Phase 3: Nesterov LSE ===
    print("\n=== Phase 3: Nesterov LSE ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t3 = min(350, deadline - time.monotonic() - 400)
    if t3 > 30:
        print(f"  Allocating {t3:.0f}s")
        a_opt, _ = nesterov_lse(current_best, time_limit=t3,
                                beta_start=20.0, beta_end=3000.0)
        try_sequence(a_opt, best_tracker)
        print(f"  After NLSE: {best_tracker[0]:.10f}")

    # === Phase 4: Coordinate Descent Round 2 ===
    print("\n=== Phase 4: CD Round 2 ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t4 = min(200, deadline - time.monotonic() - 250)
    if t4 > 20:
        print(f"  Allocating {t4:.0f}s")
        a_opt, _ = coordinate_descent(current_best, time_limit=t4)
        try_sequence(a_opt, best_tracker)
        print(f"  After CD2: {best_tracker[0]:.10f}")

    # === Phase 5: Peak Shaving ===
    print("\n=== Phase 5: Peak Shaving ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t5 = min(100, deadline - time.monotonic() - 150)
    if t5 > 15:
        a_opt, _ = peak_shave(current_best, time_limit=t5)
        try_sequence(a_opt, best_tracker)
        print(f"  After Shave: {best_tracker[0]:.10f}")

    # === Phase 6: Final NLSE Polish ===
    print("\n=== Phase 6: Final NLSE Polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t6 = min(150, deadline - time.monotonic() - 60)
    if t6 > 20:
        a_opt, _ = nesterov_lse(current_best, time_limit=t6,
                                beta_start=80.0, beta_end=6000.0)
        try_sequence(a_opt, best_tracker)
        print(f"  After Final NLSE: {best_tracker[0]:.10f}")

    # === Phase 7: Final CD ===
    print("\n=== Phase 7: Final CD ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t7 = min(80, deadline - time.monotonic() - 30)
    if t7 > 10:
        a_opt, _ = coordinate_descent(current_best, time_limit=t7)
        try_sequence(a_opt, best_tracker)
        print(f"  After Final CD: {best_tracker[0]:.10f}")

    # === Phase 8: Alternative Lengths ===
    print("\n=== Phase 8: Alternative Lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    alt_lengths = [384, 400, 432, 464, 480, 512]
    alt_lengths = [x for x in alt_lengths if x != n_best]

    for n_alt in alt_lengths[:3]:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 60:
            break
        t_alloc = min(100, t_remaining // max(1, 3))
        if t_alloc < 30:
            break

        print(f"  Trying n={n_alt}, alloc={t_alloc:.0f}s")
        # Interpolate
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = normalize_seq(f_interp)

        # CD then NLSE
        t_cd = t_alloc * 0.4
        t_lse = t_alloc * 0.6
        a_opt, _ = coordinate_descent(f_interp, time_limit=t_cd)
        try_sequence(a_opt, best_tracker)

        if len(best_tracker[1]) == n_alt:
            a_opt2, _ = nesterov_lse(best_tracker[1], time_limit=t_lse,
                                     beta_start=30.0, beta_end=2000.0)
            try_sequence(a_opt2, best_tracker)

    # === Final ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        else:
            best_score = best_tracker[0]
            final_seq = best_tracker[1]
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.10f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
