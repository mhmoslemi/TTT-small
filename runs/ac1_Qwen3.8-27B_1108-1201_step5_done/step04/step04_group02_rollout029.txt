Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp optimization
with Adam optimizer and simplex projection. Uses multi-length exploration,
adaptive temperature annealing, and periodic restarts.

import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(a, conv, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a."""
    n = len(a)
    # Softmax weights
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    w /= wsum
    
    # Gradient: grad_i = 2 * sum_k w[k] * a[k-i] (valid range)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wk * a[k - idx]
    
    return grad

def smooth_objective(a, T):
    """Compute T*log(sum(exp(conv/T))) and its gradient."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    logsumexp = T * (np.log(np.exp(scaled).sum()) + scaled.max())
    # Actually: T * log(sum(exp(conv/T))) = T * (max(conv/T) + log(sum(exp((conv - max(conv))/T))))

    max_conv = conv.max()
    scaled2 = (conv - max_conv) / T
    logsumexp = max_conv + T * np.log(np.exp(scaled2).sum())
    grad = compute_grad(a, conv, T)
    return logsumexp, grad, conv

                  beta1=0.9, beta2=0.999, eps=1e-8, restart_every=500):

    a = project_simplex(a0.copy())

    v = np.zeros(n)
    t = 0

    best_max = conv.max()
    best_a = a.copy()
    
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute gradient
        logsumexp, grad, conv = smooth_objective(a, T)
        max_conv = conv.max()
        
        if max_conv < best_max - 1e-12:
            best_max = max_conv
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))
        
        # Check for NaN
        if not np.all(np.isfinite(a_new)):
            a_new = project_simplex(a)
        
        a = a_new

            a = project_simplex(best_a + rng.standard_normal(n) * noise_level)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            
            # Also try a larger perturbation occasionally
            if no_improve > restart_every * 3:
                a = project_simplex(rng.dirichlet(np.ones(n) * 0.5))
    
    return best_a, best_max

    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_grad_a(a, conv, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a."""
    n = len(a)
    max_c = conv.max()
    scaled = (conv - max_c) / T
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()

    ga = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        ga[idx] += 2.0 * wk * a[k - idx]
    return ga


def lbfgs_softmax_optimize(a0, T, max_iter=300, tol=1e-12):
    """Optimize log-sum-exp objective using L-BFGS with softmax parameterization."""
    a_safe = np.maximum(a0, 1e-10)
    a_safe = a_safe / a_safe.sum()
    x0 = np.log(a_safe) - np.log(a_safe.sum())

    def fun_g(x):
        a = softmax_fn(x)
        conv = fast_convolve(a)
        max_c = conv.max()
        scaled = (conv - max_c) / T
        obj = max_c + T * np.log(np.exp(scaled).sum())
        ga = compute_grad_a(a, conv, T)
        gx = a * (ga - np.dot(ga, a))
        return obj, gx

    try:
        res = minimize(fun_g, x0, jac=True, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol,
                                'maxls': 50})
        a_opt = softmax_fn(res.x)
        return a_opt
    except Exception:
        return a0


def adam_simplex_optimize(a0, time_limit, T_start=0.004, T_end=0.0004,
                          lr=0.008, beta1=0.9, beta2=0.999, eps=1e-8,
                          restart_patience=600):
    """Adam with simplex projection, annealing temperature, and restarts."""
    n = len(a0)
    a = project_simplex(a0.copy())
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        t += 1
        progress = min(1.0, (time.monotonic() - t0) / max(1.0, time_limit))
        T = T_start * (T_end / T_start) ** progress

        conv = fast_convolve(a)
        max_c = conv.max()

        if max_c < best_max - 1e-13:
            best_max = max_c
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient via log-sum-exp weights
        scaled = (conv - max_c) / T
        exp_s = np.exp(scaled)
        w = exp_s / exp_s.sum()

        ga = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            ga[idx] += 2.0 * wk * a[k - idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * ga
        v = beta2 * v + (1 - beta2) * ga ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))

        if not np.all(np.isfinite(a)):
            a = project_simplex(best_a)
            m = np.zeros(n)
            v = np.zeros(n)

        # Restart if stagnating
        if no_improve > restart_patience:
            rng = np.random.default_rng(t * 11 + 7)
            noise_scale = 0.003 * min(3.0, 1.0 + no_improve / restart_patience)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_max


def lbfgs_then_adam(a0, T_lbf, t_adam, T_ad_start, T_ad_end, lr_adam):
    """Two-stage: L-BFGS polish then Adam refinement."""
    a1 = lbfgs_softmax_optimize(a0, T_lbf, max_iter=200)
    conv1 = fast_convolve(a1)
    a2, max2 = adam_simplex_optimize(a1, t_adam, T_ad_start, T_ad_end, lr_adam)
    conv2 = fast_convolve(a2)
    if conv2.max() < conv1.max():
        return a2
    return a1


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations for length n."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())

    ramp2 = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp2 / ramp2.sum())

    # Quadratic U-shape
    q = ((x - n / 2) / (n / 4)) ** 2 + 0.1
    inits.append(q / q.sum())

    # Inverted U (peak in middle)
    iq = 1.0 - ((x - n / 2) / (n / 4)) ** 2
    iq = np.maximum(iq, 0.01)
    inits.append(iq / iq.sum())

    # Cosine
    c = np.cos(np.pi * x / (n - 1)) ** 2 + 0.05
    inits.append(c / c.sum())

    # Exponential decay
    e1 = np.exp(-3.0 * x / n)
    inits.append(e1 / e1.sum())

    # Two-sided exponential
    e2 = np.exp(-4.0 * np.abs(x - n / 2) / n)
    inits.append(e2 / e2.sum())

    # Sine
    s = np.sin(np.pi * (x + 0.5) / n) + 0.1
    inits.append(s / s.sum())

    # Power law
    pw = (x + 1.0) ** 0.3
    inits.append(pw / pw.sum())

    # Hybrid: large at both ends, small in middle
    h = np.minimum(x, n - 1 - x).astype(np.float64)
    h = (h + 1.0) ** (-0.5)
    inits.append(h / h.sum())

    # Random
    r = rng.dirichlet(np.ones(n) * 0.2)
    inits.append(r)

    # Another random with different concentration
    r2 = rng.dirichlet(np.ones(n) * 1.0)
    inits.append(r2)

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            inits.append(px)
            # Blend with various shapes
            inits.append(project_simplex(0.7 * px + 0.3 * (np.ones(n) / n)))
            inits.append(project_simplex(0.7 * px + 0.3 * (ramp / ramp.sum())))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses L-BFGS with log-sum-exp smooth objective
    (softmax parameterization) and Adam with simplex projection
    (annealed temperature) in a multi-phase strategy.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, sum={p.sum():.4f}")
            print(f"Parent score: {evaluate_sequence(p.tolist()):.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    # === Phase 1: L-BFGS on multiple initializations ===
    print("\n=== Phase 1: L-BFGS Exploration ===")
    t0 = time.monotonic()

    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(len(parent))
    for L in [384, 256, 448, 512, 320]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    phase1_candidates = []

    for n in lengths_to_try:
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break

        if parent is not None and n == len(parent):
            inits = [parent]
            # Add a few diverse ones
            extra_inits = generate_inits(n, parent, rng)
            inits.extend(extra_inits[:3])
        else:
            inits = generate_inits(n, parent, rng)

        n_inits = min(len(inits), 6)
        t_per_init = max(5, min(40, rem / max(1, n_inits * (len(lengths_to_try) - 1 + 1))))

        for i in range(n_inits):
            if time.monotonic() > deadline - 450:
                break
            f_init = inits[i]
            fnorm = f_init / f_init.sum()

            # L-BFGS with moderate T
            T_lb = 0.002
            try:
                a_opt = lbfgs_softmax_optimize(fnorm, T_lb, max_iter=150)
                try_seq(a_opt, f"p1_n{n}_i{i}")
                phase1_candidates.append((a_opt, n))
            except Exception as e:
                print(f"  L-BFGS failed: {e}")

            # Quick Adam follow-up
            rem2 = deadline - time.monotonic() - 400
            if rem2 > 10:
                try:
                    a_opt2, _ = adam_simplex_optimize(a_opt, time_limit=min(rem2, 15),
                                                       T_start=0.003, T_end=0.001,
                                                       lr=0.005, restart_patience=200)
                    try_seq(a_opt2, f"p1a_n{n}_i{i}")
                except Exception:
                    pass

    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s, best so far: {best_score:.8f}")

    # === Phase 2: Deep Adam optimization on best candidates ===
    print("\n=== Phase 2: Deep Adam Optimization ===")
    t1 = time.monotonic()

    # Sort candidates by their actual max(conv)
    if phase1_candidates:
        scored = []
        for a, n in phase1_candidates:
            conv = fast_convolve(a / a.sum())
            scored.append((conv.max(), a, n))
        scored.sort(key=lambda x: x[0])
        top_candidates = [x[1] for x in scored[:5]]
    else:
        top_candidates = []

    if best_seq is not None and len(best_seq) > 0:
        if best_seq not in top_candidates:
            top_candidates.insert(0, best_seq / best_seq.sum())

    # Also add parent if available
    if parent is not None and parent not in top_candidates:
        top_candidates.append(parent)

    # Deduplicate by length
    seen_lengths = set()
    unique_candidates = []
    for cand in top_candidates:
        L = len(cand)
        if L not in seen_lengths:
            unique_candidates.append(cand)
            seen_lengths.add(L)

    for ci, cand in enumerate(unique_candidates[:4]):
        rem = deadline - time.monotonic() - 250
        if rem < 20:
            break
        n = len(cand)
        t_alloc = min(rem // max(1, 4 - ci), 120)
        t_alloc = max(t_alloc, 15)

        print(f"  Deep Adam on candidate {ci} (n={n}), t={t_alloc:.0f}s")

        # Run Adam with decreasing T
        a_opt, max_val = adam_simplex_optimize(
            cand, time_limit=t_alloc,
            T_start=0.003, T_end=0.0003,
            lr=0.006, beta1=0.9, beta2=0.999,
            restart_patience=500)
        try_seq(a_opt, f"p2_deep_{ci}")

        # L-BFGS polish on the result
        rem2 = deadline - time.monotonic() - 200
        if rem2 > 10:
            try:
                a_polished = lbfgs_softmax_optimize(a_opt, T=0.0008, max_iter=100)
                try_seq(a_polished, f"p2_lb_{ci}")
            except Exception:
                pass

    print(f"  Phase 2 time: {time.monotonic()-t1:.1f}s, best so far: {best_score:.8f}")

    # === Phase 3: Alternating L-BFGS + Adam cycles ===
    print("\n=== Phase 3: Alternating Refinement ===")
    t2 = time.monotonic()

    if best_seq is not None:
        base = best_seq / best_seq.sum()
        n_best = len(base)

        for cycle in range(6):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break

            # Perturb base
            if cycle == 0:
                f_start = base.copy()
            elif cycle == 1:
                f_start = project_simplex(base + rng.standard_normal(n_best) * 0.002)
            elif cycle == 2:
                f_start = project_simplex(base + rng.standard_normal(n_best) * 0.005)
            elif cycle == 3:
                f_start = project_simplex(0.9 * base + 0.1 * (np.ones(n_best) / n_best))
            elif cycle == 4:
                f_start = project_simplex(base * (1 + rng.standard_normal(n_best) * 0.01))
            else:
                f_start = project_simplex(base + rng.standard_normal(n_best) * 0.001)

            # Adam phase
            t_adam = min(rem * 0.5, 60)
            T_s = 0.002 * (0.8 ** cycle)
            T_e = 0.0003 * (0.8 ** cycle)
            lr = 0.005 * (0.9 ** cycle)
            try:
                a_mid, _ = adam_simplex_optimize(f_start, time_limit=t_adam,
                                                   T_start=T_s, T_end=T_e,
                                                   lr=lr, restart_patience=400)
                mom = a_mid
            except Exception:
                a_mid = f_start
                mom = f_start

            # L-BFGS phase
            rem2 = deadline - time.monotonic() - 80
            if rem2 > 8:
                T_lb = 0.001 * (0.85 ** cycle)
                try:
                    a_final = lbfgs_softmax_optimize(mom, T=T_lb, max_iter=100)
                    try_seq(a_final, f"p3_c{cycle}")
                    if fast_convolve(a_final).max() < fast_convolve(mom).max():
                        mom = a_final
                except Exception:
                    pass

            try_seq(mom, f"p3m_c{cycle}")

    print(f"  Phase 3 time: {time.monotonic()-t2:.1f}s, best so far: {best_score:.8f}")

    # === Phase 4: Try other lengths on best structure ===
    print("\n=== Phase 4: Cross-length Transfer ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_src = len(best_seq)
        other_lengths = [L for L in [256, 320, 448, 512, 640] if L != n_src]

        for L in other_lengths[:2]:
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break

            # Interpolate parent to new length
            x_new = np.linspace(0, 1, L)
            x_old = np.linspace(0, 1, n_src)
            interp = np.interp(x_new, x_old, best_seq / best_seq.sum())
            interp = np.maximum(interp, 0)
            interp = interp / interp.sum()

            # Quick optimization
            try:
                a_opt = lbfgs_softmax_optimize(interp, T=0.002, max_iter=100)
                try_seq(a_opt, f"p4_interp_L{L}")

                rem2 = deadline - time.monotonic() - 60
                if rem2 > 10:
                    a_opt2, _ = adam_simplex_optimize(a_opt, time_limit=min(rem2, 20),
                                                       T_start=0.002, T_end=0.0005,
                                                       lr=0.004, restart_patience=200)
                    try_seq(a_opt2, f"p4_adam_L{L}")
            except Exception:
                pass

    print(f"  Phase 4 time: {time.monotonic()-t3:.1f}s, best so far: {best_score:.8f}")

    # === Phase 5: Final ultra-fine polish ===
    print("\n=== Phase 5: Final Polish ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine Adam
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            try:
                a_pol, _ = adam_simplex_optimize(
                    best_seq, time_limit=min(rem, 40),
                    T_start=0.0008, T_end=0.0001,
                    lr=0.001, beta1=0.95, beta2=0.999,
                    restart_patience=1000)
                try_seq(a_pol, "polish_fine")
            except Exception:
                pass

        # L-BFGS with very small T
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            try:
                a_pol2 = lbfgs_softmax_optimize(best_seq, T=0.0003, max_iter=50)
                try_seq(a_pol2, "polish_lbfgs")
            except Exception:
                pass

        # Micro perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0005)
                try:
                    a_p = lbfgs_softmax_optimize(pert, T=0.0005, max_iter=30)
                    try_seq(a_p, "micro")
                except Exception:
                    pass

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  FALLBACK used: {best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - (deadline - budget_s + 20):.1f}s")
    return [float(x) for x in best_seq]
