The problem is:

def anderson_acceleration(F, x0, m=5, max_iter=10000, tol=1e-10):
    """Anderson acceleration for fixed-point iteration x = F(x)."""
    n = len(x0)
    X = [x0.copy()]
    G = [F(x0)]  # G[i] = F(X[i]) - X[i] (residual)
    x = x0.copy()
    
    for k in range(max_iter):
        if k < m:
            x_new = F(x)
            X.append(x_new)
            G.append(x_new - x)
            x = x_new
        else:
            # Anderson acceleration
            history_X = np.array(X[-m:])  # m x n
            history_G = np.array(G[-m:])  # m x n
            
            # Minimize ||G_i - alpha_i * mean(G)||
            # Standard Anderson: solve min ||sum alpha_i G_i|| s.t. sum alpha_i = 1
            G_mean = history_G.mean(axis=0)
            G_diff = history_G - G_mean
            # Solve: (G_diff^T G_diff) alpha = G_diff^T G_mean
            A = G_diff.T @ G_diff
            b = G_diff.T @ G_mean
            try:
                alpha = np.linalg.solve(A + 1e-10*np.eye(m), b)
            except:
                alpha = np.ones(m) / m
            
            x_new = x - G_mean - G_diff @ alpha

            # Actually the standard formula is:
            # x_new = (1/m) * sum X_i - (G_diff @ alpha)  -- no this isn't right either

            best_f = a_trial.copy()
            no_improve = 0
            a = a_trial
            if verbose and it % 500 == 0:
                print(f"    DualFP iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            a = a_trial  # still move, even if not improving best
        
        # Anneal T
        T = max(T * 0.998, T_min)
        
        # Adapt damping
        if no_improve > 100:
            damping = min(damping * 1.05, 0.9)
        elif no_improve > 0:
            damping = max(damping * 0.98, 0.3)
        
        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(it * 7919)
            a = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            T = max(T_init * 0.5, T_min)
            damping = 0.5
            no_improve = 0
    
    return best_f, best_obj


def nest_subgrad(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient on the exact max."""
    f = f0.copy()
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    t = 0
    f_prev = f.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Nesterov extrapolation
        momentum = (t - 1) / (t + 2)
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0)
        s = y.sum()
        if s > 1e-15:
            y /= s
        
        conv_y = fast_convolve(y)
        k_star = int(conv_y.argmax())
        
        # Subgradient
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        js = np.arange(lo, hi + 1)
        g[js] = 2.0 * y[k_star - js]
        g -= g.mean()
        
        # Step size: O(1/sqrt(t))
        step = 2.0 / np.sqrt(t)
        f_new = project_simplex(y - step * g)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        f_prev = f
        f = f_new
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            if verbose and t % 1000 == 0:
                print(f"    NestSG iter {t}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def smoothed_grad_adam(f0, time_limit, T_init=0.002, T_min=0.0001,
                       lr_init=0.005, verbose=False):
    """Smoothed gradient descent with Adam optimizer."""
    f = f0.copy()
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    T = T_init
    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        p = w / ws
        
        # Gradient: g[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
        # = 2 * cross_correlation(p, f)
        fft_size = 1 << (2 * n).bit_length()
        F = np.fft.rfft(f, fft_size)
        P = np.fft.rfft(p, fft_size)
        cross = np.fft.irfft(np.conj(F) * P, fft_size)[:n]
        g = 2.0 * cross
        g -= g.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr_init * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            if verbose and t % 1000 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.10f}, T={T:.6f}, lr={lr_init:.6f}")
        
        f = f_new
        
        # Anneal
        T = max(T * 0.9985, T_min)
        lr_init = max(lr_init * 0.9999, 0.0001)
    
    return best_f, best_obj


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_dir(f, M * m_exp)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = project_simplex((1 - alpha) * f + alpha * g)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = project_simplex((1 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)
        
        if no_improve > 30:
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return

        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_seq = parent.copy()

    except Exception:
