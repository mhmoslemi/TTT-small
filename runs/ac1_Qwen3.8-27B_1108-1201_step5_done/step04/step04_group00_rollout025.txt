import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_hann_window(n):
    """Hann window normalized to sum=1."""
    j = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * j / (n - 1)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    return w


def make_raised_cosine(n):
    """Raised cosine (flat-top-like) window normalized to sum=1."""
    j = np.arange(n)
    t = np.pi * j / (n - 1)
    w = 1.0 - 0.5 * (1 - np.cos(2 * t))
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    return w


def make_gaussian_window(n, sigma_frac=0.3):
    """Gaussian window normalized to sum=1."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-(j - center) ** 2 / (2 * sigma ** 2))
    s = w.sum()
    if s > 1e-15:
        w /= s
    return w


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k=0..2n-2, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def peak_targeted_cd(f0, time_limit, verbose=False):
    """
    Coordinate descent that prioritizes coordinates contributing most to peaks.
    Uses a wider set of multipliers and re-evaluates peaks dynamically.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()

    multipliers = [0.2, 0.35, 0.5, 0.65, 0.8, 0.9, 1.1, 1.25, 1.4, 1.6, 2.0]

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        conv = fast_convolve(f)

        # Find top peaks and their contributing coordinates
        k_peaks = min(30, 2 * n - 1)
        peak_indices = np.argpartition(conv, -k_peaks)[-k_peaks:]

        # Compute coordinate importance based on peak contributions
        coord_score = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            pk = conv[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                other = k - j
                if 0 <= other < n:
                    coord_score[j] += pk * f[j] * f[other] / max(f.sum(), 1e-15)

        order = np.argsort(-coord_score)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()
            best_j_obj = cur_peak
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < -1e-15 or new_val > 0.99:
                    continue
                new_val = max(0.0, new_val)
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    ss = f.sum()
                    if ss > 1e-15:
                        f = f / ss

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try a few random coordinate perturbations before giving up
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            for _ in range(5):
                j = rng.integers(0, n)
                old_val = f[j]
                if old_val < 1e-14:
                    continue
                delta = rng.standard_normal() * 0.02 * old_val
                new_val = max(0.0, old_val + delta)
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break

            if not improved:
                break

    return best_f, best_obj


def softmax_sg_polish(f0, time_limit, lr=0.003, momentum=0.93,
                      top_k=80, init_temp=0.0001):
    """
    Subgradient polishing using softmax-weighted peak contributions.
    Provides a smooth approximation to the max objective gradient.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = init_temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = (pv - pv.max()) / cur_temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                other = k - j
                if 0 <= other < n:
                    sg[j] += 2.0 * w[i] * f[other]

        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with diverse initializations, LP-FW,
    peak-targeted CD, softmax SG polishing, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_hann_window(768)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initial (Hann n=768): {best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: LP-FW on parent ===
    print("\n=== Phase 1: LP-FW on parent ===")
    t1 = min(200, deadline - time.monotonic() - 700)
    if t1 > 30:
        a_opt, _ = lpfw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Alternative initializations at same length ===
    print(f"\n=== Phase 2: Alt inits (best={best_score:.6f}) ===")
    inits = {
        "hann": make_hann_window(n_parent),
        "raised_cos": make_raised_cosine(n_parent),
        "gauss": make_gaussian_window(n_parent, 0.25),
        "gauss2": make_gaussian_window(n_parent, 0.35),
    }

    # Also try mixing parent with windows
    inits["mix_hann"] = project_simplex(0.5 * best_seq + 0.5 * inits["hann"])
    inits["mix_rev"] = project_simplex(0.6 * best_seq + 0.4 * best_seq[::-1])

    for name, init in list(inits.items()):
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(rem * 0.3, 120)
        if t_alloc < 10:
            break
        print(f"  LP-FW [{name}] n={n_parent}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(init, time_limit=t_alloc)
        try_seq(a_opt, f"lpfw_{name}")

    # === Phase 3: Longer sequences ===
    print(f"\n=== Phase 3: Longer sequences (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    for target_n in [896, 1024, 1280]:
        rem = deadline - time.monotonic() - 250
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        # Also try Hann at this length
        f_hann = make_hann_window(target_n)
        # Mix
        f_mix = project_simplex(0.6 * f_interp + 0.4 * f_hann)

        t_alloc = min(rem * 0.4, 150)
        if t_alloc < 10:
            break

        # Try interpolated
        print(f"  LP-FW [interp] n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"lpfw_interp_n{target_n}")

        rem = deadline - time.monotonic() - 200
        if rem < 20:
            continue
        t_alloc2 = min(rem * 0.3, 120)
        if t_alloc2 < 10:
            continue

        # Try mixed init
        print(f"  LP-FW [mix] n={target_n}, t={t_alloc2:.0f}s")
        a_opt, _ = lpfw_optimize(f_mix, time_limit=t_alloc2)
        try_seq(a_opt, f"lpfw_mix_n{target_n}")

    # === Phase 4: Deep optimization on best ===
    print(f"\n=== Phase 4: Deep Opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150

        # LP-FW again with more time
        t_alloc = min(rem * 0.4, 200)
        if t_alloc > 20:
            print(f"  LP-FW deep n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_deep")

        # Restart LP-FW from perturbation
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            noise = rng.standard_normal(n_best) * 0.005
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.5, 150)
            print(f"  LP-FW perturb n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, "lpfw_perturb")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.6, 180)
            print(f"  Peak-CD t={t_cd:.0f}s")
            a_opt, _ = peak_targeted_cd(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "peak_cd")

            # CD from perturbed
            rem = deadline - time.monotonic() - 50
            if rem > 15:
                noise = rng.standard_normal(len(best_seq)) * 0.003
                f_pert = project_simplex(best_seq + noise)
                t_cd2 = min(rem * 0.7, 100)
                a_opt, _ = peak_targeted_cd(f_pert, time_limit=t_cd2)
                try_seq(a_opt, "peak_cd_pert")

    # === Phase 6: Softmax SG polish ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_sg = min(rem * 0.7, 120)
            top_k = max(30, n_best // 4)
            print(f"  SG polish t={t_sg:.0f}s, top_k={top_k}")
            a_opt, _ = softmax_sg_polish(best_seq, time_limit=t_sg,
                                         lr=0.002, momentum=0.94,
                                         top_k=top_k, init_temp=0.00005)
            try_seq(a_opt, "sg_polish")

            # More aggressive SG
            rem = deadline - time.monotonic() - 25
            if rem > 10:
                a_opt, _ = softmax_sg_polish(best_seq, time_limit=min(rem, 60),
                                             lr=0.001, momentum=0.95,
                                             top_k=max(20, n_best // 6),
                                             init_temp=0.00002)
                try_seq(a_opt, "sg_fine")

    # === Phase 7: Final interleaved polish ===
    print(f"\n=== Phase 7: Final (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Short LP-FW burst
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "final_lpfw")

        # Tiny CD
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_opt, _ = peak_targeted_cd(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "final_cd")

        # Tiny SG
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt, _ = softmax_sg_polish(best_seq, time_limit=min(rem, 30),
                                         lr=0.0008, momentum=0.95,
                                         top_k=max(15, n_best // 8),
                                         init_temp=0.00001)
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = make_hann_window(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
