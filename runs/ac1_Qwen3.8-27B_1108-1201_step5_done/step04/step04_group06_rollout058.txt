import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute the objective 2n*max_b/S^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_gradient(a, conv, temp, top_k=50):
    """
    Compute the gradient of the log-sum-exp smoothed max of conv.
    f_smooth(a) = (1/temp) * log(sum_k exp(temp * conv_k))
    grad = sum_k p_k * grad(conv_k) where p_k = softmax(temp * conv)
    """
    n = len(a)
    # Use top-k for efficiency
    k = min(top_k, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    vals = conv[top_idx]
    scaled = temp * vals
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(k)
        p[np.argmax(vals)] = 1.0
    else:
        p /= p_sum

    # Compute gradient: sum_k p_k * grad(conv_k)
    # grad(conv_k)[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    for idx in range(k):
        weight = p[idx]
        if weight < 1e-12:
            continue
        lag = int(top_idx[idx])
        # grad entry i = 2 * a[lag - i] if 0 <= lag - i < n
        i_start = max(0, lag - n + 1)
        i_end = min(n - 1, lag)
        idxs = np.arange(i_start, i_end + 1)
        grad[idxs] += weight * 2.0 * a[lag - idxs]
    return grad


def subgradient_active(a, conv, threshold=0.999):
    """
    Compute the exact subgradient using only active constraints (conv_k >= threshold * max).
    """
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n)
    active = np.where(conv >= threshold * max_val)[0]
    
    grad = np.zeros(n)
    for lag in active:
        i_start = max(0, lag - n + 1)
        i_end = min(n - 1, lag)
        idxs = np.arange(i_start, i_end + 1)
        grad[idxs] += 2.0 * a[lag - idxs]
    grad /= len(active)
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lp_direction(a, M=None):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns the direction g - a.
    """
    from scipy.optimize import linprog
    
    n = len(a)
    if M is None:
        conv = fast_convolve(a)
        M = conv.max()
    
    # Build constraint matrix: (a*g)[k] <= M for all k
    # (a*g)[k] = sum_j a[k-j] * g[j] (with bounds)
    # This is a Toeplitz-like matrix
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        C[start:end, j] = a[:end - start]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * 1.0001)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = np.maximum(result.x, 0)
            return g
    except Exception:
        pass
    return None


def lp_step(a, time_remaining):
    """One LP-based step with line search. Returns (a_new, improved)."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    g = lp_direction(a, M)
    if g is None:
        return a.copy(), False
    
    g_sum = g.sum()
    if g_sum < 1e-10:
        return a.copy(), False
    
    # Direction: move from a towards g (both normalized)
    g_norm = g / g_sum
    d = g_norm - a
    
    # Line search
    best_t = 0
    best_max = M
    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        max_try = conv_try.max()
        if max_try < best_max:
            best_max = max_try
            best_t = t
    
    if best_t > 0:
        a_new = a + best_t * d
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        return a_new, True
    return a.copy(), False

At that point, we can solve the equality-constrained problem: minimize (a*a)[k*] subject to (a*a)[k*] = (a*a)[k] for all active k, a >= 0, sum a = 1. This is a quadratic optimization problem that can be solved more precisely.

Let me implement this approach:

Phase 1: Smoothed gradient descent to get close to the optimum.
Phase 2: Identify the active set.
Phase 3: Solve the quadratic program on the active set.
Phase 4: Verify and polish.

For Phase 3, if the active set is S, we want to minimize (a*a)[k0] subject to:
- (a*a)[k] = (a*a)[k0] for all k in S (all active lags equal)
- a >= 0
- sum a = 1

This is a QP with equality constraints. We can solve it with scipy.optimize.minimize using SLSQP or trust-constr.

Actually, a simpler approach for Phase 3: use the fact that at the optimum, all active lags have the same value M. The KKT conditions give us:
- grad((a*a)[k0]) = lambda * grad(sum a) + sum_{k in S, k≠k0} mu_k * grad((a*a)[k] - (a*a)[k0]) + nu (for non-negativity)

This is complex. Let me just use a numerical optimizer for the final polish.

Let me write the complete solution now:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive log-sum-exp smoothed Nesterov gradient descent
2. Active-set subgradient polishing
3. QP refinement on the identified active constraint set
4. Multi-length exploration seeded from parent
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute 2n * max_b / S^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_max_gradient(a, conv, temp, top_k=60):
    """Gradient of log-sum-exp smoothed max of conv."""
    n = len(a)
    k = min(top_k, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    vals = conv[top_idx]
    scaled = temp * vals
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(k)
        p[np.argmax(vals)] = 1.0
    else:
        p /= p_sum

    grad = np.zeros(n)
    for idx in range(k):
        w = p[idx]
        if w < 1e-12:
            continue
        lag = int(top_idx[idx])
        i_start = max(0, lag - n + 1)
        i_end = min(n - 1, lag)
        idxs = np.arange(i_start, i_end + 1)
        grad[idxs] += w * 2.0 * a[lag - idxs]
    return grad


def exact_subgradient(a, conv, rel_threshold=0.9995):
    """Exact subgradient using active set (conv_k >= threshold * max)."""
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n)
    active = np.where(conv >= rel_threshold * max_val)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    grad = np.zeros(n)
    for lag in active:
        i_start = max(0, lag - n + 1)
        i_end = min(n - 1, lag)
        idxs = np.arange(i_start, i_end + 1)
        grad[idxs] += 2.0 * a[lag - idxs]
    grad /= len(active)
    return grad


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_optimize(a0, time_limit, temp_init=1.0, temp_decay=0.997,
                       lr_init=0.008, momentum_init=0.85, top_k=60):
    """Nesterov-accelerated gradient descent on smoothed objective."""
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    temp = temp_init
    lr = lr_init
    mom = momentum_init
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = project_simplex(y)
        
        conv_y = fast_convolve(y)
        grad = smooth_max_gradient(y, conv_y, temp, top_k)
        
        # Momentum update
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        cur_max = conv_new.max()
        a = a_new
        conv = conv_new
        
        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive scheduling
        temp *= temp_decay
        lr *= 0.9992
        mom = min(0.96, mom + 0.00005)
        
        iter_count += 1
        if no_improve >= 150:
            break
    
    return best_a, 2.0 * n * best_max


def subgradient_polish(a0, time_limit, lr_init=0.003):
    """Polish with exact subgradient of active set."""
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    threshold = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        grad = exact_subgradient(a, conv, threshold)
        
        a_new = project_simplex(a - lr * grad)
        conv_new = fast_convolve(a_new)
        cur_max = conv_new.max()
        
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            a = a_new  # Still move
            no_improve += 1
        
        lr *= 0.997
        if no_improve >= 30:
            threshold = max(0.99, threshold - 0.001)
            lr = lr_init * 0.5
            no_improve = 0
            if threshold < 0.95:
                break
    
    return best_a, 2.0 * n * best_max


def qp_refine(a0, time_limit=30.0):
    """
    Try to refine using scipy minimize on the active set.
    Minimize max_k (a*a)[k] using the smoothed objective with very low temperature.
    """
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    active = np.where(conv >= 0.999 * conv.max())[0]
    max_val = conv.max()
    
    def objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        c = fast_convolve(x)
        return c.max()
    
    def gradient(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros(len(x))
        x = x / s
        c = fast_convolve(x)
        return smooth_max_gradient(x, c, temp=500.0, top_k=len(active) + 10)
    
    x0 = a.copy()
    try:
        result = minimize(objective, x0, method='L-BFGS-B',
                          options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12})
        if result.success and result.fun < max_val:
            a_new = normalize_nonneg(result.x)
            return a_new, 2.0 * n * fast_convolve(a_new).max()
    except Exception:
        pass
    return a.copy(), 2.0 * n * max_val


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives the convolution a * g."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_polish(a0, time_limit=30.0):
    """LP-based polishing step."""
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_max = M
    
    t0 = time.monotonic()
    C = build_conv_matrix(a)
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.00001)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
            if not result.success:
                break
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum
        except Exception:
            break
        
        d = g_norm - a
        # Line search
        improved = False
        for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_try = a + t * d
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            conv_try = fast_convolve(a_try)
            if conv_try.max() < best_max - 1e-13:
                best_max = conv_try.max()
                best_a = a_try.copy()
                a = a_try
                M = best_max
                C = build_conv_matrix(a)
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, 2.0 * n * best_max
