"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-n exploration with FW-LP for diverse starting points
2. FISTA (Nesterov accelerated projected gradient) with log-sum-exp smoothing continuation
3. Active-set SLSQP refinement for fast convergence near optimum
4. Fine polish with adaptive subgradient + momentum
Key innovations: continuation method on smoothing parameter for fast convergence,
active-set identification for SLSQP, and structured asymmetric perturbations.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_and_grad(a, tau, conv=None):
    """
    Compute smoothed max of auto-convolution and its gradient.
    Uses log-sum-exp for numerical stability.
    """
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_terms = np.exp(shifted)
    sum_exp = np.sum(exp_terms)
    F = max_c + tau * np.log(sum_exp)
    w = exp_terms / sum_exp

    # Gradient: sum_k w[k] * 2 * a[k-j] for valid j
    grad = np.zeros(n)
    thresh = 1e-12
    for k in range(len(conv)):
        if w[k] < thresh:
            continue
        wk = 2.0 * w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += wk * a[k - idx]
    return F, grad


def fista_optimize(a0, time_limit, verbose=True):
    """
    FISTA (Nesterov accelerated projected gradient) on smoothed max of auto-conv.
    Uses continuation on tau: start large, decrease gradually.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Continuation schedule: tau values to try
    initial_conv = fast_convolve(a)
    scale = initial_conv.max()
    tau_values = [scale * f for f in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]]

    for tau in tau_values:
        if time.monotonic() - t0 > time_limit - 30:
            break
        # Run FISTA for this tau
        a_curr = a.copy()
        a_prev = a.copy()
        t_mom = 1.0
        step_size = 0.4  # Conservative step (Lipschitz ~2, so 1/2 would be safe, use 0.4)
        iters = 0
        max_iters = 300
        no_improve = 0

        while iters < max_iters and time.monotonic() - t0 < time_limit - 30:
            iters += 1
            # Nesterov momentum
            beta = (t_mom - 1.0) / (t_mom + 1.5)
            y = a_curr + beta * (a_curr - a_prev)
            y = project_simplex(y)

            # Compute objective and gradient at y
            conv_y = fast_convolve(y)
            F_y, grad_y = smoothed_max_and_grad(y, tau, conv_y)

            # Backtracking line search
            alpha = step_size
            F_curr = fast_convolve(a_curr).max()
            for _ in range(20):
                a_trial = project_simplex(y - alpha * grad_y)
                F_trial, _ = smoothed_max_and_grad(a_trial, tau)
                # Descent condition (relaxed)
                if F_trial <= F_y - 0.5 * alpha * np.sum(grad_y * (a_trial - y)):
                    break
                alpha *= 0.5

            a_prev = a_curr.copy()
            a_curr = a_trial
            t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0

            # Check improvement
            obj = fast_convolve(a_curr).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_curr.copy()
                no_improve = 0
                # Adapt step size upward if converging well
                step_size = min(step_size * 1.02, 1.0)
            else:
                no_improve += 1
                if no_improve > 20:
                    step_size *= 0.9

        a = a_curr
        if verbose:
            print(f"    FISTA tau={tau:.4f}: obj={best_obj:.10f} ({iters} iters)")

    return best_a, best_obj


def lbfqgs_reduced(a0, tau, time_limit, verbose=True):
    """
    L-BFGS-B optimization with one variable fixed (sum constraint handled implicitly).
    Optimizes over a[0:n-1] with a[n-1] = 1 - sum(a[0:n-1]).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_full = a.copy()

    def objective(x):
        """x has length n-1, a[n-1] = 1 - sum(x)."""
        if time.monotonic() - t0 > time_limit - 5:
            return best_obj
        full = np.zeros(n)
        full[:n-1] = x
        full[n-1] = max(0.0, 1.0 - np.sum(x))
        s = full.sum()
        if s < 1e-15:
            return 1e10
        full = full / s
        conv = fast_convolve(full)
        val, _ = smoothed_max_and_grad(full, tau, conv)
        return val

    def gradient(x):
        full = np.zeros(n)
        full[:n-1] = x
        full[n-1] = max(0.0, 1.0 - np.sum(x))
        s = full.sum()
        if s < 1e-15:
            return np.ones(n-1)
        full = full / s
        conv = fast_convolve(full)
        _, grad_full = smoothed_max_and_grad(full, tau, conv)
        # Chain rule: dF/da[j] - dF/da[n-1] for j < n-1
        return grad_full[:n-1] - grad_full[n-1]

    x0 = a[:n-1].copy()
    bounds = [(0.0, 1.0)] * (n - 1)

    result = minimize(objective, x0, jac=gradient, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})

    if result.success or result.fun < best_obj:
        full = np.zeros(n)
        full[:n-1] = np.maximum(result.x, 0)
        full[n-1] = max(0.0, 1.0 - np.sum(np.maximum(result.x, 0)))
        full = full / full.sum()
        obj = fast_convolve(full).max()
        if obj < best_obj:
            best_obj = obj
            best_full = full

    return best_full, best_obj


def active_set_slsqp(a0, time_limit, verbose=True):
    """
    Identify active set (top peaks) and use SLSQP to optimize with those as constraints.
    Iteratively adds new active constraints.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    for outer_iter in range(5):
        if time.monotonic() - t0 > time_limit - 20:
            break

        conv = fast_convolve(a)
        M = conv.max()
        # Active set: positions where conv is within 1% of max
        active_mask = conv > M * 0.995
        active_positions = np.where(active_mask)[0]

        if len(active_positions) == 0:
            break
        if len(active_positions) > 20:
            # Keep only top 20
            top_idx = np.argpartition(conv, -20)[-20:]
            active_positions = top_idx

        # Define objective: sum of active conv values (proxy for max)
        # Actually, use the max of active ones
        def obj(x):
            if time.monotonic() - t0 > time_limit - 5:
                return best_obj
            full = np.maximum(x, 0)
            s = full.sum()
            if s < 1e-15:
                return 1e10
            full = full / s
            c = fast_convolve(full)
            return np.max(c[active_positions])

        def obj_grad(x):
            full = np.maximum(x, 0)
            s = full.sum()
            if s < 1e-15:
                return np.ones(n)
            full = full / s
            c = fast_convolve(full)
            # Gradient of max over active positions
            # Use the position with the max value
            k_star = active_positions[np.argmax(c[active_positions])]
            # d(conv[k_star])/d(a[j]) = 2*a[k_star-j]
            grad = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            grad[idx] = 2.0 * full[k_star - idx]
            # Adjust for normalization (sum constraint)
            mean_grad = grad.mean()
            grad -= mean_grad
            return grad

        # Use SLSQP with sum constraint
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0,
                      'jac': lambda x: np.ones(n)}
        bounds_slsqp = [(0.0, None)] * n

        result = minimize(obj, a, jac=obj_grad, method='SLSQP',
                          bounds=bounds_slsqp, constraints=constraint,
                          options={'maxiter': 200, 'ftol': 1e-12})

        if result.success:
            a_new = np.maximum(result.x, 0)
            a_new = a_new / a_new.sum()
            new_obj = fast_convolve(a_new).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"    SLSQP iter {outer_iter}: obj={best_obj:.10f} (active={len(active_positions)})")
            else:
                break
        else:
            # Try L-BFGS-B as fallback
            break

    return best_a, best_obj


def solis_wettstein(a0, time_limit, top_k=10, verbose=True):
    """
    Pattern search (Solis-Wettstein) with adaptive step size.
    Explores directions based on recent gradient history.
    Very effective for piecewise-smooth objectives like max of quadratics.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    pattern = np.zeros(n)
    delta = 1.0 / n  # Initial step size
    improving_streak = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 10:
        # Compute gradient direction
        conv = fast_convolve(a)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        w = pv / pv.sum()

        grad = np.zeros(n)
        for i in range(len(peaks)):
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]

        # Project gradient to tangent space of simplex
        grad -= grad.mean()

        # Update pattern with momentum
        pattern = 0.7 * pattern + 0.3 * grad
        pat_norm = np.linalg.norm(pattern)
        if pat_norm > 1e-15:
            pattern /= pat_norm

        # Try step in pattern direction
        a_trial = a + delta * pattern
        a_trial = project_simplex(a_trial)
        obj_trial = fast_convolve(a_trial).max()

        if obj_trial < best_obj - 1e-15:
            a = a_trial
            improving_streak += 1
            no_improve = 0
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a.copy()
            # Increase step
            delta = min(delta * 1.2, 1.0)
        else:
            # Try opposite direction
            a_trial = a - delta * pattern
            a_trial = project_simplex(a_trial)
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_obj - 1e-15:
                a = a_trial
                pattern = -pattern
                best_obj = min(best_obj, obj_trial)
                best_a = a.copy()
                no_improve = 0
                delta = min(delta * 1.1, 1.0)
            else:
                # Try random direction
                rng = np.random.default_rng(hash((int(time.monotonic()*1000), no_improve)) % (2**32))
                rand_dir = rng.standard_normal(n)
                rand_dir -= rand_dir.mean()
                a_trial = project_simplex(a + delta * rand_dir * 0.5)
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_obj - 1e-15:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    delta *= 0.8

        # Restart if stuck
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**32))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            pattern = np.zeros(n)
            delta = 1.0 / n
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                no_improve += 5
                if no_improve > 50:
                    rng = np.random.default_rng(it * 31 + 17)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    conv = fast_convolve(f)
                    M = conv.max()
                    no_improve = 0
                continue
            g = res.x
        except Exception:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + np.random.default_rng(it).standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        c
