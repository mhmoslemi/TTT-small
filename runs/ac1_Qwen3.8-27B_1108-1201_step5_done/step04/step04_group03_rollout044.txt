"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed mirror descent with adaptive temperature (main workhorse)
2. Refined Frank-Wolfe with LP oracle
3. Coordinate descent polish on the smoothed objective
4. Basin hopping with large perturbations
5. Multi-length exploration (384, 512, 768)
Key innovation: clean gradient formula grad = 2*correlate(a, softmax(t*conv(a,a)))
with adaptive temperature annealing and multiplicative weight updates.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad(a, t):
    """Compute gradient of LSE-smoothed max convolution. Returns (smoothed_max, grad)."""
    conv_a = fast_convolve(a)
    max_c = conv_a.max()
    shifted = t * (conv_a - max_c)
    exp_shifted = np.exp(shifted)
    log_sum = max_c + (1.0 / t) * np.log(exp_shifted.sum())
    w = exp_shifted / exp_shifted.sum()
    # grad_i = 2 * sum_j a[j] * w[j+i] = 2 * correlate(a, w, 'full')[i]
    grad = 2.0 * np.correlate(a, w, 'full')[:len(a)]
    return log_sum, grad


def lse_mirror_descent(f0, time_limit, t_init=50.0, t_final=5000.0,
                       lr_init=0.1, lr_decay=0.9998, t_incr=1.003,
                       restart_patience=200, verbose=False):
    """
    Mirror descent (multiplicative weights) with LSE-smoothed objective.
    Adaptively increases temperature for sharper max approximation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t = t_init
    lr = lr_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_a = fast_convolve(f)
        M = conv_a.max()

        # Compute LSE gradient
        _, grad = lse_grad(f, t)

        # Multiplicative weight update (mirror descent)
        # f_i <- f_i * exp(-lr * grad_i) / sum
        log_up = -lr * grad
        # Clamp for numerical stability
        log_up = np.clip(log_up, -5.0, 5.0)
        f_new = f * np.exp(log_up)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            f = f_new
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            # Try smaller step
            lr_half = lr * 0.5
            log_up2 = -lr_half * grad
            log_up2 = np.clip(log_up2, -5.0, 5.0)
            f_try = f * np.exp(log_up2)
            f_try = np.maximum(f_try, 1e-30)
            f_try = f_try / f_try.sum()
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_obj - 1e-15:
                f = f_try
                best_obj = obj_try
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                lr = lr_half

        lr = min(lr * 1.005, lr_init)  # slowly recover learning rate
        t *= t_incr

        # Restart if stuck
        if no_improve > restart_patience:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.003 + 0.004 * min(1.0, no_improve / (2 * restart_patience))
            f = best_f * np.exp(rng.standard_normal(n) * noise_scale)
            f = np.maximum(f, 1e-10)
            f = f / f.sum()
            t = max(t_init, t * 0.5)
            lr = lr_init * 0.8
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-MD: it={it}, obj={best_obj:.10f}, t={t:.1f}, lr={lr:.5f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def lse_projected_sg(f0, time_limit, t_init=100.0, t_final=10000.0,
                     lr_init=0.005, mom=0.9, t_incr=1.004,
                     restart_patience=300, verbose=False):
    """
    Projected subgradient descent with momentum on LSE-smoothed objective.
    Uses Euclidean projection onto simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    t = t_init
    lr = lr_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        _, grad = lse_grad(f, t)

        # Project gradient onto tangent space of simplex
        grad_flat = grad - np.mean(grad)

        v = mom * v - lr * grad_flat
        f = project_simplex(f + v)

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        t *= t_incr

        if no_improve > restart_patience:
            rng = np.random.default_rng(it * 104729 + 37)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = max(lr_init * 0.7, 0.0005)
            t = max(t_init, t * 0.7)
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-SG: it={it}, obj={best_obj:.10f}, t={t:.1f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, t_lse=500.0, verbose=False):
    """
    Cyclic coordinate descent on the LSE-smoothed objective.
    For each coordinate, do a 1D golden-section search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    def eval_coord(f_arr):
        return fast_convolve(f_arr).max()

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Try to optimize coordinate i
            # We want to change f[i] while keeping sum(f) = 1
            # Parameterize: f[i] = f[i] + delta, adjust others proportionally
            # Simpler: just try f[i] * (1+delta) and renormalize
            best_delta = 0.0
            best_val = best_obj

            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                f_try = f.copy()
                f_try[i] = max(0.0, f_try[i] * (1.0 + delta))
                f_try = f_try / f_try.sum()
                val = eval_coord(f_try)
                if val < best_val:
                    best_val = val
                    best_delta = delta

            if best_delta != 0.0:
                f[i] = max(0.0, f[i] * (1.0 + best_delta))
                f = f / f.sum()
                best_obj = best_val
                best_f = f.copy()
                improved = True

        if not improved:
            # Try smaller steps
            for i in range(0, n, 3):
                if time.monotonic() - t0 > time_limit:
                    break
                best_delta = 0.0
                best_val = best_obj
                for delta in [0.003, -0.003, 0.007, -0.007, 0.001, -0.001]:
                    f_try = f.copy()
                    f_try[i] = max(0.0, f_try[i] * (1.0 + delta))
                    f_try = f_try / f_try.sum()
                    val = eval_coord(f_try)
                    if val < best_val:
                        best_val = val
                        best_delta = delta
                if best_delta != 0.0:
                    f[i] = max(0.0, f[i] * (1.0 + best_delta))
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True

        if verbose and it % 10 == 0:
            print(f"    CD: epoch={it}, obj={best_obj:.10f}")

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle (refined version)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            no_improve += 1
            if no_improve > 20:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 41 + 23)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 61 + 29)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-LP: it={it}, obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def basin_hopping(f0, time_limit, n_perturbations=10, perturb_scale=0.05, verbose=False):
    """
    Basin hopping: make large perturbations and re-optimize each one briefly.
    Returns the best solution found.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    elapsed_budget = time_limit
    time_per_perturb = elapsed_budget / max(1, n_perturbations)

    for p in range(n_perturbations):
        if time.monotonic() - t0 > time_limit:
            break

        rng = np.random.default_rng(p * 7919 + 42)
        # Large multiplicative perturbation
        noise = rng.standard_normal(n) * perturb_scale * (1.0 + 0.5 * p / n_perturbations)
        f_pert = best_f * np.exp(noise)
        f_pert = np.maximum(f_pert, 1e-10)
        f_pert = f_pert / f_pert.sum()

        # Quick LSE mirror descent on the perturbed point
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_alloc = min(time_per_perturb, remaining)

        a_opt, obj = lse_mirror_descent(
            f_pert, time_limit=t_alloc,
            t_init=50.0, t_final=5000.0,
            lr_init=0.08, lr_decay=0.9998, t_incr=1.004,
            restart_patience=100
        )

        if obj < best_obj:
            best_obj = obj
            best_f = a_opt.copy()
            if verbose:
                print(f"    Basin hop {p}: improved to {best_obj:.10f}")

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)

    # Dirichlet with concentration 0.5 (more spread)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Concentrated in middle
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c1 / c1.sum())

    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Random
    r = rng.dirichlet(np.ones(n) * 1.0)
    inits.append(r)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    # === Phase 1: LSE Mirror Descent on parent (~400s) ===
    print("\n=== Phase 1: LSE Mirror Descent ===")
    rem = deadline - time.monotonic() - 1300
    if rem > 30:
        t_alloc = min(rem, 400)
        print(f"  LSE-MD on parent (n={n_p}), t={t_alloc:.0f}s")
        a_opt, _ = lse_mirror_descent(parent, time_limit=t_alloc, t_init=30.0, lr_init=0.1)
        try_seq(a_opt, "lse_md_1")

    # === Phase 2: LSE Projected SG (~200s) ===
    print("\n=== Phase 2: LSE Projected SG ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 1000
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  LSE-SG, t={t_alloc:.0f}s")
            a_opt, _ = lse_projected_sg(best_seq, time_limit=t_alloc, t_init=50.0, lr_init=0.008)
            try_seq(a_opt, "lse_sg_1")

    # === Phase 3: FW-LP refinement (~300s) ===
    print("\n=== Phase 3: FW-LP ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem, 300)
            print(f"  FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_lp_1")

    # === Phase 4: Basin Hopping (~250s) ===
    print("\n=== Phase 4: Basin Hopping ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 50:
            t_alloc = min(rem, 250)
            print(f"  Basin hopping, t={t_alloc:.0f}s")
            a_opt, _ = basin_hopping(best_seq, time_limit=t_alloc, n_perturbations=8, perturb_scale=0.04)
            try_seq(a_opt, "basin_hop")

    # === Phase 5: Coordinate Descent Polish (~150s) ===
    print("\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem, 150)
            print(f"  CD, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd_1")

    # === Phase 6: Fine LSE polish (~100s) ===
    print("\n=== Phase 6: Fine LSE Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  Fine LSE-MD, t={t_alloc:.0f}s")
            a_opt, _ = lse_mirror_descent(best_seq, time_limit=t_alloc, t_init=200.0, lr_init=0.02)
            try_seq(a_opt, "fine_lse")

    # === Phase 7: Try different lengths (~remaining) ===
    print("\n=== Phase 7: Multi-length exploration ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        for n_try in [512, 256, 768]:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            print(f"  Trying n={n_try}, t={rem:.0f}s")
            inits = generate_inits(n_try, best_seq, rng)
            # Use the rescaled parent as best init
            best_local = None
            best_local_score = float('inf')
            for ii, ini in enumerate(inits[:4]):  # try first 4 inits
                rem = deadline - time.monotonic() - 25
                if rem < 10:
                    break
                t_alloc = min(rem, 40)
                a_opt, obj = lse_mirror_descent(ini, time_limit=t_alloc, t_init=30.0, lr_init=0.08)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_local_score:
                    best_local_score = score
                    best_local = a_opt
            if best_local is not None:
                try_seq(best_local, f"n{n_try}")

    # === Phase 8: Final micro-polish ===
    print("\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            # Very fine LSE
            a_opt, _ = lse_mirror_descent(best_seq, time_limit=min(rem, 30),
                                          t_init=1000.0, lr_init=0.005)
            try_seq(a_opt, "final_lse")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            # Tiny perturbations + quick optimize
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = lse_mirror_descent(f_pert, time_limit=min(rem, 8),
                                              t_init=500.0, lr_init=0.003)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
