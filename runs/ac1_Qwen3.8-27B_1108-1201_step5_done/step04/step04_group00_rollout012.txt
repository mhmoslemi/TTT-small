import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Compute subgradient of max(conv(a,a)) w.r.t. a.

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent with paired update and 1D search.
    For each coordinate j, we optimize a[j] while scaling all others to maintain sum=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    pass_count = 0
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False
        cur_a = a.copy()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            v = cur_a[j]  # current value of a[j]
            if v < 1e-15:
                # Try to turn it on
                candidates = [0.001, 0.002, 0.005, 0.01]
            else:
                # Try a range of multipliers around current value
                candidates = [v * m for m in [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0]]
                # Also try some absolute values
                candidates += [0.001, 0.005, 0.01, 0.05]
            
            best_v = v
            best_peak = fast_convolve(cur_a).max()  # Current peak
            
            for w in candidates:
                w = max(0.0, min(w, 1.0 - 1e-15))
                if abs(w - v) < 1e-12:
                    continue
                if 1 - v < 1e-15:
                    scale = 0.0
                else:
                    scale = (1.0 - w) / (1.0 - v)
                
                trial = cur_a.copy()
                trial[j] = w
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                
                # Normalize (should already be ~1, but let's be safe)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                
                peak = fast_convolve(trial).max()
                if peak < best_peak - 1e-14:
                    best_peak = peak
                    best_v = w
                    best_trial = trial
            
            if best_v != v:
                cur_a = best_trial
                if best_peak < best_obj - 1e-12:
                    best_obj = best_peak
                    best_a = cur_a.copy()
                    pass_improved = True
        
        a = cur_a
        
        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                if verbose:
                    print(f"  CD converged at pass {pass_count}")
                break
        else:
            no_improve_passes = 0
        
        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD pass {pass_count}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj
