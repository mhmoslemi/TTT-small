Let me also think about whether there's a theoretical insight I'm missing.

The objective is: 2n * max_k (a*a)(k) / (sum a)^2

By normalization (sum a = 1): 2n * max_k (a*a)(k)

We want to minimize the peak of the auto-convolution of a probability vector.

This is related to the problem of finding a probability distribution on Z_n (or {0,...,n-1}) whose convolution with itself is as flat as possible.

For the circular convolution (on Z_n), the Fourier transform gives:
max_k (a∘a)(k) >= (sum |a_hat|^4) / n (by Parseval-like arguments)

For the linear convolution, the problem is different.

I recall that the minimum peak for the linear auto-convolution of a non-negative sequence of length n with unit sum is achieved by a sequence that is approximately proportional to the square root of a "flat" spectral density. In the frequency domain, we want the Fourier transform of a to have |a_hat(θ)|^2 as flat as possible (which makes the convolution flat).

But for non-negative a, the Fourier transform has constraints (it must be the Fourier transform of a non-negative sequence).

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using coordinate descent with
efficient convolution updates, LP-based Frank-Wolfe for escape,
and multi-scale exploration.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_oracle(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] <= M for k=0,...,2n-2
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # This is a Toeplitz-like matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in (f*g)[k] for k = j to j+n-1 with coefficient f[k-j]
        lo = j
        hi = min(2 * n - 1, j + n)
        A_ub[lo:hi, j] = f[:hi-lo]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def coord_descent(a_init, time_limit, max_sweeps=200, seed=42):
    """
    Coordinate descent for minimizing 2n*max(conv)/sum^2.
    Normalizes to sum=1 internally.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

            best_t = old_val
            best_mc = conv.max()  # current max conv
            
            for t in t_grid:
                if abs(t - old_val) < 1e-14:
                    continue
                d = t - old_val
                # Linear values for affected indices:
                vals = conv_ks + 2.0 * d * a_ks
                # Override at k = 2j:
                if 2 * j >= lo_dep and 2 * j <= hi_dep:
                    idx = 2 * j - lo_dep
                    vals[idx] = conv[2 * j] + t * t - old_val * old_val
                mc = max(vals.max(), max_other)
                if mc < best_mc:
                    best_mc = mc
                    best_t = t

                        break

                    # Evaluate at m1
                    d1 = m1 - old_val
                    vals1 = conv_ks + 2.0 * d1 * a_ks
                    if 2 * j >= lo_dep and 2 * j <= hi_dep:
                        idx = 2 * j - lo_dep
                        vals1[idx] = conv[2 * j] + m1 * m1 - old_val * old_val
                    mc1 = max(vals1.max(), max_other)
                    
                    # Evaluate at m2
                    d2 = m2 - old_val
                    vals2 = conv_ks + 2.0 * d2 * a_ks
                    if 2 * j >= lo_dep and 2 * j <= hi_dep:

                    mc2 = max(vals2.max(), max_other)
                    
                    if mc1 < mc2:
                        hi = m2
                    else:
                        lo = m1
                
                best_t = (lo + hi) / 2.0
                # Recompute best_mc
                d = best_t - old_val
                vals = conv_ks + 2.0 * d * a_ks

            # Apply update

                        conv[k] += best_t * best_t - old_val * old_val
                    else:
                        conv[k] += 2.0 * d * a[k - j]
                a[j] = best_t
                sweep_improved = True
                global_no_improve = 0
                
                cur_mc = conv.max()
                cur_obj = 2.0 * n * cur_mc
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()

                global_no_improve += 1

            a = a / a.sum()
            conv = fast_convolve(a)
            global_no_improve = 0
        
        if not sweep_improved and sweep > 5:
            break
    
    return best_a, best_obj

def fw_lp_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

        f = f / s

    best_f = f.copy()
    
    t0 = time.monotonic()

    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        g = solve_lp_oracle(f, M * m_exp, time_limit=min(15, rem * 0.2))
        if g is None:
            m_exp *= 1.2
            if m_exp > 8.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.1, 1.0, 10)
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
