```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth penalty (soft-max) 
with multiplicative weight updates and temperature annealing.
Key innovation: FFT-based gradient computation and adaptive temperature schedule.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_cov(w, f, L):
    """Compute grad[j] = 2 * sum_m w[j+m] * f[m] using FFT cross-correlation."""
    n = len(f)
    W = np.fft.rfft(w, L)
    F = np.fft.rfft(f, L)
    corr = np.fft.irfft(W * np.conj(F), L)
    return 2.0 * corr[:n]

def smooth_mw_optimize(f0, time_limit, t_start=5.0, t_end=200.0, 
                       lr_init=0.3, lr_decay=0.9999, floor=1e-12):
    """
    Multiplicative weight optimization with soft-max smoothing and temperature annealing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    L = 1 << (3 * n - 1).bit_length()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = (time.monotonic() - t0) / time_limit
        
        # Temperature schedule: exponential increase
        t = t_start * (t_end / t_start) ** elapsed
        
        conv = fast_convolve(f)
        
        # Softmax weights (numerically stable)
        scaled = t * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        grad = compute_grad_cov(w, f, L)
        
        # Adaptive step size
        # Scale gradient so that the update is moderate
        grad_scale = grad.max()
        if grad_scale > 0:
            grad = grad / grad_scale
        
        f_new = f * np.exp(-cur_lr * grad)
        f_new = np.maximum(f_new, floor)
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            cur_lr *= 0.95
            if no_improve > 50:
                f = best_f.copy()
                cur_lr = lr_init * 0.5
                no_improve = 0
            if no_improve > 200:

                rng = np.random.default_rng(it)
                f = best_f * np.exp(rng.standard_normal(n) * 0.01)
                f = f / f.sum()
                cur_lr = lr_init
                no_improve = 0

def coordinate_refine(f0, time_limit):
    """
    Coordinate descent: try small perturbations of pairs of coordinates
    to reduce the maximum peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    best_obj = conv.max()

        it += 1

        max_idx = np.argmax(conv)
        max_val = conv[max_idx]

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        for delta in [0.001, 0.005, 0.01, 0.02, 0.05]:

            f_trial[j] += delta
            if f_trial[i] < 0:
                f_trial[j] += f_trial[i]
                f_trial[i] = 0
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            
            conv_trial = fast_convolve(f_trial)
            if conv_trial.max() < best_obj - 1e-14:

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)

    two_p = p1 + p2
    inits.append(('two_peaks', two_p / two_p.sum()))
    
    # Random binary (Barker-like)
    b = rng.integers(0, 2, size=n).astype(np.float64)

    inits.append(('binary', b / b.sum()))

    elif parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth penalty (soft-max)
with FFT-based gradient, multiplicative weight updates, temperature annealing,
and peak-targeted refinement. Key innovations: fast FFT cross-correlation gradient,
adaptive temperature schedule, and multi-phase exploration+exploitation strategy.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_cov(w, f, fft_size):
    """Compute grad[j] = 2 * sum_m w[j+m] * f[m] using FFT cross-correlation.
    w has length 2n-1, f has length n, result has length n.
    """
    n = len(f)
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    corr = np.fft.irfft(W * np.conj(F), fft_size)
    return 2.0 * corr[:n]


def smooth_mw_optimize(f0, time_limit, t_start=5.0, t_end=200.0,
                       lr_init=0.3, lr_decay=0.99995, floor=1e-15,
                       momentum=0.0, verbose=False):
    """
    Multiplicative weight optimization with soft-max smoothing and
    temperature annealing. Uses FFT-based gradient for speed.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (3 * n - 1).bit_length()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    it = 0
    v_mom = np.zeros(n) if momentum > 0 else None

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = (time.monotonic() - t0) / time_limit

        # Temperature schedule: exponential increase
        t = t_start * (t_end / t_start) ** elapsed

        conv = fast_convolve(f)

        # Softmax weights (numerically stable)
        scaled = t * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Gradient via FFT cross-correlation
        grad = compute_grad_cov(w, f, fft_size)

        # Normalize gradient for stable updates
        grad_max = np.abs(grad).max()
        if grad_max > 1e-30:
            grad = grad / grad_max

        if momentum > 0 and v_mom is not None:
            v_mom = momentum * v_mom - cur_lr * grad
            effective_grad = -v_mom
        else:

        f_new = np.maximum(f_new, floor)

def peak_targeted_optimize(f0, time_limit, lr_init=0.06, peak_band=0.02,

            w_peaks /= w_peaks.sum()
        else:
            peak_mask = np.zeros(len(conv), dtype=bool)
            peak_mask[np.argmax(conv)] = True

        w = np.zeros(len(conv))
        w[peak_mask] = w_peaks

        grad_max = np.abs(grad).max()
        if grad_max > 1e-30:
            grad = grad / grad_max

        f_new = f * np.exp(-cur_lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        f_new = f * np.exp(-cur_lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            cur_lr *= 0.98
            if no_improve > 50:
                f = best_f.copy()
                cur_lr = max(lr_init * 0.5, 0.005)
                no_improve = 0
            if no_improve > 200:
                rng = np.random.default_rng(it * 31 + 47)
                f = best_f * np.exp(rng.standard_normal(n) * 0.008)
                f = f / f.sum()
                cur_lr = lr_init
                cur_temp = temp_init * 1.5
                no_improve = 0

        cur_lr *= 0.9999
        cur_temp *= temp_decay

    return best_f, best_obj


def local_pair_optimize(f0, time_limit, n_trials_per_sec=2000):
    """
    Local optimization via random pair exchanges.
    For each pair (i,j), try transferring mass from i to j (and vice versa)
    and accept if it reduces the max peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    rng = np.random.default_rng(999)
    t0 = time.monotonic()
    accepted = 0
    total = 0

    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.001, 0.008])

    while time.monotonic() - t0 < time_limit:
        # Batch of random pairs
        batch = min(20, int(n_trials_per_sec * 0.01))
        for _ in range(batch):
            if time.monotonic() - t0 >= time_limit:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            d = deltas[rng.integers(0, len(deltas))]
            total += 1

            # Try moving from i to j
            if f[i] > d:
                f_trial = f.copy()
                f_trial[i] -= d
                f_trial[j] += d
                # No need to renormalize (sum is preserved)
                if f_trial[i] >= 0:
                    conv_t = fast_convolve(f_trial)
                    if conv_t.max() < best_obj - 1e-15:
                        best_obj = conv_t.max()
                        best_f = f_trial.copy()
                        f = f_trial
                        accepted += 1

            # Try moving from j to i
            if f[j] > d:
                f_trial = f.copy()
                f_trial[j] -= d
                f_trial[i] += d
                if f_trial[j] >= 0:
                    conv_t = fast_convolve(f_trial)
                    if conv_t.max() < best_obj - 1e-15:
                        best_obj = conv_t.max()
                        best_f = f_trial.copy()
                        f = f_trial
                        accepted += 1

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for optimization."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n) / n))
    inits.append(('dirichlet', rng.dirichlet(np.ones(n) * 0.3)))
    inits.append(('dirichlet_sym', rng.dirichlet(np.ones(n) * 0.5) * 1.0))

    # Edge concentration (various powers)
    for pw in [0.5, 1.0, 1.5]:
        edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** pw
        inits.append((f'edge_{pw}', edge_w / edge_w.sum()))

    # Gaussian center (various widths)
    for sigma in [0.1, 0.15, 0.25]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sigma * n)) ** 2)
        inits.append((f'gauss_{sigma}', g / g.sum()))

    # Cosine patterns
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) ** 2 + 0.05
        inits.append((f'cos_{freq}', c / c.sum()))

    # Two-peak
    p1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.08 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.08 * n)) ** 2)
    inits.append(('two_peak', (p1 + p2) / (p1 + p2).sum()))

    # Random smooth
    rs = rng.standard_normal(n)
    # Smooth it
    from scipy.ndimage import gaussian_filter1d
    rs_smooth = gaussian_filter1d(rs, sigma=5)
    rs_smooth = np.maximum(rs_smooth, 0.01)
    inits.append(('random_smooth', rs_smooth / rs_smooth.sum()))

    # Random binary (Barker-like)
    b = rng.integers(0, 2, size=n).astype(np.float64)
    b[b == 0] = 0.15
    inits.append(('binary', b / b.sum()))

    if parent is not None:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            px = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.append(('parent_interp', px / ps))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: Multi-phase optimization combining smooth penalty
    (soft-max with temperature annealing), peak-targeted refinement,
    and local pair search. Uses FFT-based gradients for speed.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"No parent: {e}")

    n = len(parent) if parent is not None else 384

    print(f"\nStarting optimization with n={n}, budget={budget_s}s")
    print(f"Initial best: {best_score:.8f}")

    # === Phase 1: Diverse Exploration (~350s) ===
    print("\n=== Phase 1: Diverse Exploration ===")
    t1_end = time.monotonic() + min(350, max(60, deadline - time.monotonic() - 1100))

    inits = generate_inits(n, parent, rng)
    # Prioritize parent and structured inits
    inits.sort(key=lambda x: 0 if x[0] in ('parent', 'parent_interp', 'uniform')
               else 1 if x[0].startswith(('edge', 'gauss', 'cos')) else 2)

    t_per_init = (t1_end - time.monotonic()) / max(1, len(inits))
    t_per_init = max(t_per_init, 15.0)

    for name, init in inits:
        remaining = t1_end - time.monotonic()
        if remaining < 10:
            break
        t_alloc = min(t_per_init, remaining)

        # Quick smooth MW
        f_opt, _ = smooth_mw_optimize(init, time_limit=t_alloc,
                                       t_start=5.0, t_end=150.0,
                                       lr_init=0.3, lr_decay=0.9999)
        try_seq(f_opt, f"p1_smooth_{name}")

    print(f"  Phase 1 best: {best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = evaluate_sequence(best_seq.tolist())

    # === Phase 2: Deep Optimization (~600s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t2_end = time.monotonic() + min(600, max(100, deadline - time.monotonic() - 600))

    # Deep smooth MW with high temperature range
    t_alloc = min(350, t2_end - time.monotonic())
    if t_alloc > 30:
        print(f"  Deep smooth MW (t: 20->800), time={t_alloc:.0f}s")
        f_opt, _ = smooth_mw_optimize(best_seq, time_limit=t_alloc,
                                       t_start=20.0, t_end=800.0,
                                       lr_init=0.25, lr_decay=0.99995)
        try_seq(f_opt, "p2_deep_smooth")

    # Mixed peak optimization
    t_alloc = min(200, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Mixed peak (top-k=30), time={t_alloc:.0f}s")
        f_opt, _ = mixed_peak_optimize(best_seq, time_limit=t_alloc,
                                        lr_init=0.05, top_k=30,
                                        temp_init=0.02, temp_decay=1.005)
        try_seq(f_opt, "p2_mixed_peak")

    # Peak targeted
    t_alloc = min(200, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Peak targeted, time={t_alloc:.0f}s")
        f_opt, _ = peak_targeted_optimize(best_seq, time_limit=t_alloc,
                                           lr_init=0.08, peak_band=0.03)
        try_seq(f_opt, "p2_peak")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Fine-tuning (~450s) ===
    print(f"\n=== Phase 3: Fine-tuning ===")
    t3_end = time.monotonic() + min(450, max(60, deadline - time.monotonic() - 150))

    # Very high temperature (near-hard max)
    t_alloc = min(200, t3_end - time.monotonic())
    if t_alloc > 20:
        print(f"  High-t smooth (t: 100->2000), time={t_alloc:.0f}s")
        f_opt, _ = smooth_mw_optimize(best_seq, time_limit=t_alloc,
                                       t_start=100.0, t_end=2000.0,
                                       lr_init=0.12, lr_decay=0.99995)
        try_seq(f_opt, "p3_high_t")

    # Peak targeted with narrow band
    t_alloc = min(150, t3_end - time.monotonic())
    if t_alloc > 15:
        print(f"  Narrow peak target, time={t_alloc:.0f}s")
        f_opt, _ = peak_targeted_optimize(best_seq, time_limit=t_alloc,
                                           lr_init=0.03, peak_band=0.01)
        try_seq(f_opt, "p3_narrow_peak")

    # Local pair optimization
    t_alloc = min(150, t3_end - time.monotonic())
    if t_alloc > 10:
        print(f"  Local pair search, time={t_alloc:.0f}s")
        f_opt, _ = local_pair_optimize(best_seq, time_limit=t_alloc)
        try_seq(f_opt, "p3_local")

    # Another mixed peak pass with different params
    t_alloc = min(150, t3_end - time.monotonic())
    if t_alloc > 15:
        print(f"  Mixed peak (top-k=10, cold), time={t_alloc:.0f}s")
        f_opt, _ = mixed_peak_optimize(best_seq, time_limit=t_alloc,
                                        lr_init=0.03, top_k=10,
                                        temp_init=0.005, temp_decay=1.01)
        try_seq(f_opt, "p3_cold_mixed")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Final Polish and Restarts (remaining time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    restart_count = 0
    while time.monotonic() < deadline - 30:
        restart_count += 1

        # Try a small random perturbation + quick optimization
        pert_scale = 0.003 + 0.005 * min(1.0, (best_score - 1.5) * 10)
        f_pert = best_seq * np.exp(rng.standard_normal(n) * pert_scale)
        f_pert = np.maximum(f_pert, 1e-15)
        f_pert = f_pert / f_pert.sum()

        t_alloc = min(80, deadline - time.monotonic() - 20)
        if t_alloc < 10:
            break

        # Quick high-t smooth
        f_opt, _ = smooth_mw_optimize(f_pert, time_limit=min(t_alloc * 0.6, 50),
                                       t_start=50.0, t_end=500.0, lr_init=0.15)
        try_seq(f_opt, f"r{restart_count}_smooth")

        # Quick peak target
        t_alloc2 = min(t_alloc * 0.4, deadline - time.monotonic() - 15)
        if t_alloc2 > 5:
            f_opt2, _ = peak_targeted_optimize(f_opt, time_limit=t_alloc2,
                                                lr_init=0.04, peak_band=0.015)
            try_seq(f_opt2, f"r{restart_count}_peak")

        # Local pair on current best
        t_alloc3 = min(30, deadline - time.monotonic() - 10)
        if t_alloc3 > 5:
            f_opt3, _ = local_pair_optimize(best_seq, time_limit=t_alloc3)
            try_seq(f_opt3, f"r{restart_count}_local")

    print(f"  Total restarts: {restart_count}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
```
