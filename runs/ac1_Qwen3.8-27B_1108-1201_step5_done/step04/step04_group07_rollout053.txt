def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten(f0, time_limit, alpha=0.1, iterations=None, verbose=False):
    """Iterative spectral flattening to make auto-convolution flat."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_alpha = alpha
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute FFT of current f
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        C = A * np.conj(A)  # Power spectrum = |A|^2
        
        # Target: flatten the power spectrum
        # C_flat = mean(C) (DC component only)
        C_flat = np.full_like(C, C.mean())
        
        # Blend
        C_new = (1.0 - cur_alpha) * C + cur_alpha * C_flat
        
        # Ensure non-negative
        C_new = np.maximum(C_new, 0)
        
        # Take square root (keep phase of A)
        phase = np.angle(A)
        mag = np.sqrt(C_new)
        A_new = mag * np.exp(1j * phase)
        
        # Inverse FFT
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-15:
            break
        f_new = f_new / fs
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            if verbose and it % 50 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.10f}, alpha={cur_alpha:.4f}")
        else:
            # Reduce alpha and try
            cur_alpha *= 0.95
            if cur_alpha < 0.005:
                cur_alpha = alpha * 0.5
        
        # Occasionally increase alpha to escape local minima
        if it % 200 == 0:
            cur_alpha = min(alpha * 2, 0.5)
    
    return best_f, best_obj

def spectral_flatten_v2(f0, time_limit, verbose=False):
    """
    Improved spectral flattening: instead of flattening the power spectrum,
    directly flatten the time-domain convolution using a soft max approach.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    momentum = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        max_c = conv.max()
        
        # Compute gradient of max(c) wrt f
        # max(c) is the max of linear functions in f (each c[m] is bilinear in f)
        # Subgradient of c[m] wrt f: 2*f reversed and shifted
        peak_idx = np.argmax(conv)
        
        # Gradient of c[peak_idx] wrt f
        # c[m] = sum_i f[i]*f[m-i], dc[m]/df[j] = f[m-j] + f[j-m] (for valid indices)
        grad = np.zeros(n)
        m = peak_idx
        for j in range(n):
            # ci = m - j (first term)
            ci = m - j
            if 0 <= ci < n:
                grad[j] += f[ci]
            # cj = j - m (second term, only if j > m)
            cj = j - m
            if 0 <= cj < n and cj != ci:  # avoid double counting
                grad[j] += f[cj]
        
        # Actually, for c[m] = sum_i f[i]*f[m-i]:
        # dc[m]/df[j] = f[m-j] + f[j-m] (where valid)
        # But since c[m] = c[2n-2-m] (symmetry), let's just compute directly
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            i1 = m - j  # f[i1] term
            if 0 <= i1 < n:
                val += f[i1]
            i2 = j - m  # f[i2] term (this is the same as the i1 term for the symmetric part)
            # Actually c[m] = sum over i where both i and m-i are in [0,n-1]
            # dc[m]/df[j]: j appears as i=j (contribution f[m-j]) and as i=m-j (contribution f[j])
            # But these are the same term when j = m-j, i.e., j = m/2
            # So: dc[m]/df[j] = f[m-j] + f[j-m] if both valid, or just one if only one valid
            # Wait, c[m] = sum_i f[i]*f[m-i]. j appears when i=j (gives f[m-j]) or when m-i=j i.e. i=m-j (gives f[j])
            # These are the same term if j = m-j, i.e., j = m/2
            # So dc[m]/df[j] = f[m-j] + (f[j] if j != m-j else 0) ... no
            # c[m] = sum_i f[i]*f[m-i]
            # = ... + f[j]*f[m-j] + ...
            # dc[m]/df[j] = f[m-j] + f[j-m] (the j in the first factor and the j in the second factor)
            # But f[j-m] is only valid if j-m >= 0 and j-m < n
            # And f[m-j] is only valid if m-j >= 0 and m-j < n
            
            # Actually it's simpler: c[m] = (f*f)[m] where * is linear convolution
            # The derivative of (f*f)[m] wrt f[j] is f[m-j] + f[j-m] (with f[k]=0 for k out of range)
            # But since f is defined only for 0..n-1, f[m-j] is valid if 0<=m-j<n, and f[j-m] is valid if 0<=j-m<n
            # Since m and j are both in [0, 2n-2] and [0, n-1] respectively:
            # m-j is in [-(n-1), 2n-2], valid if 0<=m-j<n
            # j-m is in [-(2n-2), n-1], valid if 0<=j-m<n
            
            # For non-negative f, both terms are non-negative
            if 0 <= m - j < n:
                val += f[m - j]
            if 0 <= j - m < n:
                val += f[j - m]
            grad[j] = val
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Momentum update
        momentum = 0.9 * momentum - lr * grad
        f_new = project_simplex(f + momentum)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            lr *= 0.95
            momentum *= 0.5
        
        if verbose and it % 100 == 0:
            print(f"    SpecFlatten2 iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj

def solve_linearized_lp(f, conv_f):
    """Solve: min t s.t. t >= conv_f[k] + 2*(B*g)[k] for all k, g >= -f, sum(g)=0."""
    n = len(f)
    
    # Variables: [t, g_0, ..., g_{n-1}]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    # Build B matrix where (B*g)[k] = sum_j B[k,j]*g[j] = (f*g)[k]
    # (f*g)[k] = sum_j f[k-j]*g[j] for valid k-j
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    for j in range(n):
        # (f*g)[k] gets contribution f[k-j]*g[j]
        # k-j is valid when 0 <= k-j < n, i.e., j <= k < j+n
        rows = slice(j, j + n)
        A_ub[rows, 1 + j] = 2.0 * f  # f[k-j] for k = j..j+n-1 gives f[0]..f[n-1]
    
    b_ub = -conv_f
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None

def reduced_lp(f, conv_f, top_k=25):
    """Reduced LP using only top_k active constraints."""
    n = len(f)
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, 0] = -1.0
    for j in range(n):
        for ii, k in enumerate(peaks):
            if j <= k < j + n:
                A_ub[ii, 1 + j] += 2.0 * f[k - j]
    
    b_ub = -conv_f[peaks]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None

def fw_optimize(f0, time_limit, use_reduced=True, reduced_k=25, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        if use_reduced:
            g = reduced_lp(f, conv, top_k=reduced_k)
        else:
            g = solve_linearized_lp(f, conv)
        
        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_new = (1.0 - alpha) * f + alpha * target
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 30:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0
    
    return best_f, best_obj

def subgrad_optimize(f0, time_limit, lr=0.002, mom=0.93, top_k=40, verbose=False):
    """Subgradient descent targeting top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    cur_lr = lr
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Get top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Softmax weights
            scaled = pv / 0.001
            scaled -= scaled.max()
            w = np.exp(scaled)
            weights = w / w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for ii, k in enumerate(peaks):
            if weights[ii] < 1e-12:
                continue
            w = weights[ii]
            k = int(k)
            for j in range(n):
                # dc[k]/df[j] = f[k-j] + f[j-k] (for valid indices)
                val = 0.0
                if 0 <= k - j < n:
                    val += f[k - j]
                if 0 <= j - k < n:
                    val += f[j - k]
                sg[j] += 2.0 * w * val
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9995
        
        if verbose and it % 200 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}")
        
        if no_improve > 500:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    
    return best_f, best_obj

def local_search(f0, time_limit, rng, verbose=False):
    """Local search with decreasing perturbation scale."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    scale = 0.003
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Try multiple perturbations
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()
        
        scale *= 0.92
        if scale < 1e-7:
            scale = 0.002
            # Big jump
            f_jump = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            obj = fast_convolve(f_jump).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_jump.copy()
    
    return best_f, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    x = np.arange(n, dtype=np.float64)
    
    # Lopsided (geometric-inspired)
    for r in [0.98, 0.95, 0.9]:
        g = r ** x
        inits.append(g / g.sum())
    
    # One-sided concentration
    for frac in [0.3, 0.5, 0.7]:
        g = np.exp(-x / (frac * n))
        inits.append(g / g.sum())
    
    # Gaussian various widths
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Cosine taper
    c1 = 0.5 + 0.5 * np.cos(np.pi * x / n)
    inits.append(c1 / c1.sum())
    
    # Linear ramp
    r = np.linspace(0.1, 1.0, n)
    inits.append(r / r.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Diverse initialization (lopsided sequences)
    2. Spectral flattening (new frequency-domain method)
    3. Frank-Wolfe LP (targeted peak reduction)
    4. Subgradient polishing (multi-peak aware)
    5. Local search (fine-tuning)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
