import numpy as np
import time
from scipy.optimize import minimize_scalar

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(weights, f):
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    # Nesterov parameters
    t = 1.0
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    
    lr = 2.0 / (n + 1)  # initial learning rate
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y (Nesterov lookahead point)
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, y)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 10
            continue
        grad = grad / grad_norm
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        theta = (t - 1) / t_new

        y_new = project_simplex(y_new)
        
        # Gradient step at y

        obj_cur = conv_cur.max()
        
        alpha = 1.0
        ft = f_new

            if obj_t < obj_cur + 1e-14:

            alpha *= 0.5
            if alpha < 1e-8:
                break
        
        f = ft
        y = f + theta * (f - f_new)  # Nesterov extrapolation
        y = project_simplex(y)
        t = t_new
        
        conv = fast_convolve(f)
        obj = conv.max()

            best_f = f.copy()
            no_improve = 0

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_from_weights(weights, f):
    """Compute gradient of sum_k weights[k]*(f*f)[k] w.r.t. f.
    grad[i] = 2 * sum_{m} weights[i+m] * f[m]
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def make_weights_from_conv(conv, beta):
    """Create softmax weights from convolution values."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_bc / total


def nesterov_lse(f0, time_limit, beta_start=5.0, beta_multiplier=1.008,
                  beta_max=20000.0, lr_init=None, mom_init=0.85):
    """Nesterov-accelerated LSE with backtracking and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 1.5 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    momentum = mom_init
    v = np.zeros(n)
    f_prev = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        w = make_weights_from_conv(conv, beta)
        grad = compute_grad_from_weights(w, f)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 50
            break

        # Heavy-ball / Nesterov momentum
        v = momentum * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-10:
            v = v / v_norm

        # Backtracking line search
        obj_cur = conv.max()
        alpha = lr
        best_alpha = 0.0
        best_step_obj = obj_cur

        for _ in range(15):
            f_trial = project_simplex(f - alpha * v)
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            if obj_trial < best_step_obj:
                best_step_obj = obj_trial
                best_alpha = alpha
            alpha *= 0.6

        if best_alpha > 0:
            f = project_simplex(f - best_alpha * v)
        else:
            no_improve += 5
            lr *= 0.8
            if lr < 0.1 / n:
                # Restart
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init, 1.0 / n)
                momentum = max(0.8, momentum - 0.05)
                no_improve = 0
            continue

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            # Slightly increase lr on success
            lr = min(lr * 1.01, 3.0 / n)
            best_lr = lr
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.93
            if no_improve > 50:
                momentum = max(0.7, momentum - 0.02)
            if no_improve > 150:
                # Strong restart with perturbation
                rng = np.random.default_rng(hash((it, n)) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * (0.003 + 0.002 * min(1, (no_improve-150)/100)))
                v = np.zeros(n)
                lr = best_lr
                momentum = mom_init
                beta = min(beta * 1.2, beta_max)
                no_improve = 0

        # Ramp up beta (slower for smoother landscape)
        beta = min(beta * beta_multiplier, beta_max)

        if it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLSE it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.5f}, mom={momentum:.3f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def active_subgrad_descent(f0, time_limit, top_k=16, lr_init=0.002):
    """Coordinate descent using active subgradients (top-k peaks)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Get top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Weight peaks by their excess over the median
        pv = conv[peaks]
        pv_scaled = (pv - pv.min()) / (pv.max() - pv.min() + 1e-15)
        w = pv_scaled ** 2
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        # Compute subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w[i] * f[k - idx]
        grad *= 2.0

        # Gradient step with momentum
        v = 0.9 * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = project_simplex(f - 0.5 * v)  # undo half step
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.92
            if no_improve > 200:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.0003)
                no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, max_rounds=500):
    """Exact coordinate descent: for each coordinate, find optimal 1D value."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit and no_improve_rounds < 30:
        # One full round of coordinate updates
        round_improved = False
        obj_before = fast_convolve(f).max()

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            # Find affected convolution positions
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            # Current convolution values at affected positions
            # We want to find x (new value of f[i]) that minimizes
            # max of affected convolution values, subject to x >= 0
            # and we'll renormalize afterwards.

            # The change in conv[k] when f[i] changes from old to x:
            # For k != 2*i: delta_conv[k] = 2*(x - f[i]) * f[k-i] (if 0 <= k-i < n)
            # For k == 2*i: delta_conv[2*i] = x^2 - f[i]^2

            old_val = f[i]
            if old_val < 1e-15:
                continue

            # Compute the "slope" and "base" for each affected position
            # conv_new[k] = conv[k] + 2*(x - old_val)*f[k-i] for k != 2i
            # conv_new[2i] = conv[2i] + x^2 - old_val^2

            # We want to minimize max_k conv_new[k]
            # Try a few candidate x values
            # The optimal x is where the maximum is minimized
            # Since all slopes are positive (f[k-i] >= 0), reducing x always helps
            # But we need x >= 0 and we want to maintain the sum

            # Simple heuristic: try reducing f[i] by a fraction
            # and see if max improves
            for frac in [0.5, 0.7, 0.9]:
                new_val = old_val * frac
                f[i] = new_val
                # Renormalize
                s = f.sum()
                if s < 1e-15:
                    f = best_f.copy()
                    break
                f = f / s
                conv_new = fast_convolve(f)
                obj_new = conv_new.max()
                if obj_new < obj_before:
                    round_improved = True
                    obj_before = obj_new
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_f = f.copy()
                    break
                else:
                    f[i] = old_val
                    f = f / f.sum()

        if not round_improved:
            no_improve_rounds += 1
        else:
            no_improve_rounds = 0

    return best_f, best_obj


def diverse_init(n, rng, parent=None):
    """Create diverse initializations of length n."""
    inits = []

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Gaussian
    g = np.abs(rng.standard_normal(n))
    inits.append(g / g.sum())

    # 3. Raised cosine
    x = np.arange(n) / (n - 1)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # 4. Gaussian bump
    x = np.arange(n) / (n - 1) - 0.5
    gb = np.exp(-8 * x * x)
    inits.append(gb / gb.sum())

    # 5. Interpolated parent
    if parent is not None:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        f_interp = np.interp(x_new, x_old, parent)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            inits.append(f_interp / f_interp.sum())

    # 6. Bleached noise (flattened)
    noise = rng.standard_normal(n)
    noise = noise - noise.mean()
    noise = np.abs(noise) ** 0.3
    inits.append(noise / noise.sum())

    # 7. Two-peak structure
    peak1 = np.exp(-((np.arange(n) / n - 0.3) ** 2) / 0.01)
    peak2 = np.exp(-((np.arange(n) / n - 0.7) ** 2) / 0.01)
    two = peak1 + peak2
    inits.append(two / two.sum())

    return inits

    print("\n=== Phase 1: Quick LSE multi-start ===")
    t_per_n = max(20, (deadline - time.monotonic() - 200) / max(1, len(candidate_n)))
    per_init_time = max(5, t_per_n / 4)

    for n_try in candidate_n:
        t_remaining = deadline - time.monotonic() - 100
        if t_remaining < 30:
            break

        inits = diverse_init(n_try, rng, parent)
        best_for_n = None
        best_obj_for_n = float('inf')

        for init in inits[:4]:
            t_rem = deadline - time.monotonic() - 80
            if t_rem < per_init_time:
                break
            t_alloc = min(per_init_time, t_rem)
            a_opt, obj = nesterov_lse(init, time_limit=t_alloc,
                                       beta_start=5.0, beta_multiplier=1.01,
                                       beta_max=5000.0, lr_init=2.0/n_try)
            if obj < best_obj_for_n:
                best_obj_for_n = obj
                best_for_n = a_opt
            try_sequence(a_opt, best_tracker)

        if best_for_n is not None:
            print(f"  n={n_try}: best obj={best_obj_for_n:.10f}")

    # Phase 2: Deep Nesterov LSE on current best
    print("\n=== Phase 2: Deep Nesterov LSE ===")
    current_best = best_tracker[1]
    n_best = len(current_best)
    t_alloc = min(350, deadline - time.monotonic() - 200)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s, n={n_best}")
        a_opt, obj = nesterov_lse(current_best, time_limit=t_alloc,
                                   beta_start=20.0, beta_multiplier=1.006,
                                   beta_max=15000.0, lr_init=1.0/n_best,
                                   mom_init=0.88)
        try_sequence(a_opt, best_tracker)

    # Phase 3: Active subgradient polish
    print("\n=== Phase 3: Active Subgradient ===")
    current_best = best_tracker[1]
    n_best = len(current_best)
    t_alloc = min(150, deadline - time.monotonic() - 100)
    if t_alloc > 15:
        print(f"  Time: {t_alloc:.0f}s, n={n_best}")
        a_opt, obj = active_subgrad_descent(current_best, time_limit=t_alloc,
                                             top_k=max(12, n_best // 24),
                                             lr_init=0.001)
        try_sequence(a_opt, best_tracker)

    # Phase 4: Second deep LSE with different params
    print("\n=== Phase 4: Second LSE (high beta) ===")
    current_best = best_tracker[1]
    n_best = len(current_best)
    t_alloc = min(200, deadline - time.monotonic() - 60)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s, n={n_best}")
        a_opt, obj = nesterov_lse(current_best, time_limit=t_alloc,
                                   beta_start=100.0, beta_multiplier=1.004,
                                   beta_max=30000.0, lr_init=0.5/n_best,
                                   mom_init=0.90)
        try_sequence(a_opt, best_tracker)

    # Phase 5: Coordinate descent polish
    print("\n=== Phase 5: Coordinate Descent ===")
    current_best = best_tracker[1]
    t_alloc = min(100, deadline - time.monotonic() - 30)
    if t_alloc > 10:
        print(f"  Time: {t_alloc:.0f}s, n={len(current_best)}")
        a_opt, obj = coordinate_descent_polish(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # Phase 6: Final LSE squeeze
    print("\n=== Phase 6: Final Squeeze ===")
    current_best = best_tracker[1]
    n_best = len(current_best)
    t_alloc = max(10, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  Time: {t_alloc:.0f}s, n={n_best}")
        a_opt, obj = nesterov_lse(current_best, time_limit=t_alloc,
                                   beta_start=500.0, beta_multiplier=1.003,
                                   beta_max=50000.0, lr_init=0.3/n_best,
                                   mom_init=0.92)
        try_sequence(a_opt, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(448) / 448
    print(f"\nFALLBACK: score={evaluate_sequence(fallback.tolist()):.8f}")
    return [float(x) for x in fallback]
