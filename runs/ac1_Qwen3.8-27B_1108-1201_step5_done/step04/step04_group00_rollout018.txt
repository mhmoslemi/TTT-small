def coord_descent(f0, time_limit):
    """Coordinate descent: for each coordinate, try multiple multipliers."""
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    multipliers = [0.3, 0.5, 0.7, 1.4, 1.7, 2.5]
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 4:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.93, top_k=80, temp=5e-5):
    """Subgradient polish using softmax-weighted peak contributions."""
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, 2 * n - 1)
        if k_actual < 2:
            k_actual = 2
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 7919) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length via linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse initializations, LP-FW with adaptive
    constraints, coordinate descent polishing, subgradient refinement,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if len(a) == 0:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default start: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 0: Quick diverse starts (30s) ===
    print("\n=== Phase 0: Diverse Initialization ===")
    t0 = time.monotonic()
    phase0_budget = 35.0

    # Try sinusoidal and Chebyshev-like shapes at parent length
    init_candidates = [parent.copy()]

    # Sinusoidal: a[i] = 1 + 0.5*cos(2*pi*i/n)
    n = n_parent
    x = np.linspace(0, 1, n)
    sinusoidal = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    sinusoidal = np.maximum(sinusoidal, 0)
    init_candidates.append(project_simplex(sinusoidal))

    # Raised cosine
    raised = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    init_candidates.append(project_simplex(np.maximum(raised, 1e-6)))

    # Two-bump
    two_bump = np.exp(-20 * (x - 0.25)**2) + np.exp(-20 * (x - 0.75)**2)
    init_candidates.append(project_simplex(np.maximum(two_bump, 1e-6)))

    # Flat-dot (high center)
    flat_dot = np.ones(n)
    center = n // 2
    width = max(1, n // 8)
    flat_dot[center - width:center + width + 1] = 3.0
    init_candidates.append(project_simplex(flat_dot))

    for i, ic in enumerate(init_candidates):
        if time.monotonic() - t0 > phase0_budget:
            break
        try_seq(ic, f"init_{i}")

    # Quick LP-FW on best (short)
    if best_seq is not None and time.monotonic() - t0 < phase0_budget:
        t_alloc = phase0_budget - (time.monotonic() - t0)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(t_alloc, 25))
        try_seq(a_opt, "quick_lpfw")

    # === Phase 1: Main LP-FW (long) ===
    print(f"\n=== Phase 1: Main LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 60:
            n_best = len(best_seq)
            t1 = min(rem, 900)
            print(f"  Running LP-FW n={n_best}, t={t1:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t1, verbose=True, momentum=0.4)
            try_seq(a_opt, "lpfw_main")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-Length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        target_lengths = [512, 640, 896, 1024]
        # Add parent length if not already
        if n_parent not in target_lengths:
            target_lengths.append(n_parent)
        target_lengths.sort()

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            if target_n == len(base):
                f_interp = base.copy()
            else:
                f_interp = interpolate_seq(base, target_n)
            t_alloc = min(rem * 0.5, 150)
            if t_alloc < 20:
                t_alloc = max(rem * 0.3, 15)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=(target_n == n_parent))
            try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_cd = min(rem * 0.6, 200)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 4: Subgradient Polish ===
    print(f"\n=== Phase 4: SG Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_sg = min(rem * 0.7, 120)
            top_k = max(20, n_best // 4)
            print(f"  SG t={t_sg:.0f}s, top_k={top_k}")
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg, lr=0.002,
                                 mom=0.94, top_k=top_k, temp=3e-5)
            try_seq(a_opt, "sg_main")

        # Second SG with different params
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 60), lr=0.001,
                                 mom=0.95, top_k=max(10, n_best // 8), temp=1e-5)
            try_seq(a_opt, "sg_fine")

    # === Phase 5: Deep iteration on best ===
    print(f"\n=== Phase 5: Deep Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # LP-FW with different momentum
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 80), momentum=0.6)
            try_seq(a_opt, "lpfw_deep1")

        # Perturbation + SG
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            for trial in range(3):
                rem = deadline - time.monotonic() - 10
                if rem < 10:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem, 25),
                                     lr=0.0015, mom=0.94,
                                     top_k=max(10, n_best // 6), temp=2e-5)
                try_seq(a_opt, f"perturb_{trial}")

        # Final mini LP-FW
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 30), momentum=0.3)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
