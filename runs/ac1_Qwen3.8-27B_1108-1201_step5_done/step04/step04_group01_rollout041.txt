import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, lp_timeout=15):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        A_cache = build_lp_matrix(f)
        g = solve_lp_for_direction(f, M * m_exp, A_cache)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()

            else:
                no_improve += 1
        else:

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0):
    """Log-sum-exp smoothed gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.0 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.5, beta_end)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

    return best_f, best_obj


def coordinate_descent(a0, time_limit, n_candidates=12):
    """Fast coordinate descent with vectorized updates."""
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    n = len(a)
    c = fast_convolve(a)
    S = a.sum()
    c_max = c.max()
    best_obj = 2.0 * n * c_max / (S * S)
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        c_max_global = c.max()

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_v = a[i]
            # Affected range of c: k in [i, i+n-1] intersected with [0, 2n-2]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            length = k_hi - k_lo + 1

            # The update to c is: c[k] += 2*delta*a[k-i] (for k != 2i)
            # At k=2i: c[2i] += 2*delta*a[i] + delta^2
            # a[k-i] for k in [k_lo, k_hi] is a[max(0,k_lo-i) : max(0,k_lo-i)+length]
            # Since k_lo >= i (because k_lo = max(0,i) = i for i>=0), a[k-i] = a[0:length]
            # Wait: k_lo = max(0, i) = i (since i >= 0), so k-i ranges from 0 to length-1
            # So a[k-i] = a[0:length]

            sens = a[0:length]  # sensitivity vector: a[k-i] for k in [k_lo, k_hi]

            # Precompute the unaffected max
            # c outside [k_lo, k_hi]
            if k_lo > 0 and k_hi < 2 * n - 2:
                unaff_max = max(c[:k_lo].max(), c[k_hi + 1:].max())
            elif k_lo > 0:
                unaff_max = c[:k_lo].max()
            elif k_hi < 2 * n - 2:
                unaff_max = c[k_hi + 1:].max()
            else:
                unaff_max = -np.inf

            # Current c values in affected range
            c_aff = c[k_lo:k_hi + 1].copy()

            # Candidate values for a[i]
            candidates = [0.0]
            if old_v > 1e-15:
                candidates.extend([old_v * f for f in [0.1, 0.3, 0.5, 0.7, 1.1, 1.3, 1.5, 2.0, 3.0]])
            # Add some values based on the "target"
            # The ideal would balance the active constraints
            if old_v > 1e-15:
                candidates.extend([old_v * 0.85, old_v * 0.92, old_v * 1.05, old_v * 1.12])
            candidates = sorted(set(max(0.0, v) for v in candidates))

            best_local = best_obj
            best_candidate = old_v

            for v in candidates:
                if abs(v - old_v) < 1e-15:
                    continue
                delta = v - old_v
                S_new = S + delta
                if S_new < 1e-10:
                    continue

                # New c values in affected range
                # c_new[k] = c_aff[idx] + 2*delta*sens[idx] for k != 2i
                # At k=2i (if in range): c_new = c_aff[idx] + 2*delta*a[i] + delta^2
                c_new_aff = c_aff + 2.0 * delta * sens
                # Fix the k=2i case
                k_2i = 2 * i
                if k_lo <= k_2i <= k_hi:
                    idx_2i = k_2i - k_lo
                    c_new_aff[idx_2i] = c_aff[idx_2i] + 2.0 * delta * a[i] + delta * delta

                c_new_max = max(c_new_aff.max(), unaff_max) if np.isfinite(unaff_max) else c_new_aff.max()
                local_obj = 2.0 * n * c_new_max / (S_new * S_new)

                if local_obj < best_local:
                    best_local = local_obj
                    best_candidate = v

            if best_candidate != old_v:
                delta = best_candidate - old_v
                # Update c
                c[k_lo:k_hi + 1] = c_aff + 2.0 * delta * sens
                k_2i = 2 * i
                if k_lo <= k_2i <= k_hi:
                    idx_2i = k_2i - k_lo
                    c[k_2i] = c_aff[idx_2i] + 2.0 * delta * a[i] + delta * delta
                a[i] = best_candidate
                S += delta
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 5:
                break
        else:
            no_improve_sweeps = 0
            c_max_new = c.max()
            obj_new = 2.0 * n * c_max_new / (S * S)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a.copy()

    return best_a, best_obj

def make_raised_cosine(n):
    """Raised cosine window."""
    t = np.linspace(0, 1, n)
    return 0.5 * (1 + np.cos(2 * np.pi * t))


def make_gaussian_window(n, sigma_frac=0.3):
    """Gaussian window."""
    t = np.linspace(0, 1, n)
    center = 0.5
    return np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)


def make_tukey_window(n, r=0.3):
    """Tukey (tapered cosine) window."""
    t = np.linspace(0, 1, n)
    w = np.ones(n)
    n_r = int(r * n / 2)
    if n_r > 0:
        w[:n_r] = 0.5 * (1 + np.cos(np.pi * (2 * np.arange(n_r) + 1) / (r * n + 1)))
        w[-n_r:] = w[:n_r][::-1]
    return w

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 12})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

                # Try random perturbation to escape
                if sweep % 4 == 0:
                    rng = np.random.default_rng(sweep * 1000 + 42)
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                    c = fast_convolve(a)
                    S = a.sum()
                    obj = 2.0 * n * c.max() / (S * S)
                    no_improve_sweeps = 0
                elif no_improve_sweeps > 20:
                    break
        else:
            no_improve_sweeps = 0
            obj_new = 2.0 * n * c.max() / (S * S)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a.copy()
            obj = obj_new

    return best_a, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def make_raised_cosine(n):
    """Raised cosine window."""
    t = np.linspace(0, 1, n)
    return 0.5 * (1 + np.cos(2 * np.pi * t))


def make_gaussian_window(n, sigma_frac=0.25):
    """Gaussian window."""
    t = np.linspace(0, 1, n)
    center = 0.5
    return np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)


def make_tukey_window(n, r=0.3):
    """Tukey window."""
    t = np.linspace(0, 1, n)
    w = np.ones(n)
    n_r = max(1, int(r * n / 2))
    if n_r < n // 2:
        w[:n_r] = 0.5 * (1 + np.cos(np.pi * (2 * np.arange(n_r) + 1 - n_r) / (r * n + 1)))
        w[-n_r:] = w[:n_r][::-1]
    return w


def make_bell_window(n, p=2.0):
    """Bell-shaped window a[t] = (1-t^2)^p for t in [-1,1]."""
    t = np.linspace(-1, 1, n)
    return np.maximum(1.0 - t ** p, 0.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry. Multi-phase:
    1. Diverse structural initialization
    2. FW-LP global optimization
    3. LSE gradient polishing
    4. Fast coordinate descent
    5. Alternating final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t_phase1_start = time.monotonic()
    candidates = {}

    lengths = [320, 384, 448, 512, 576, 640, 720]
    for n in lengths:
        candidates[f'cos_{n}'] = make_raised_cosine(n)
        candidates[f'gauss_{n}'] = make_gaussian_window(n, 0.25)
        candidates[f'tukey_{n}'] = make_tukey_window(n, 0.3)
        candidates[f'bell_{n}'] = make_bell_window(n, 2.0)
        candidates[f'uni_{n}'] = np.ones(n)

    # Random with structure
    for n in [384, 448, 512, 576]:
        for trial in range(2):
            base = make_raised_cosine(n)
            noise = np.abs(rng.standard_normal(n)) * 0.3
            candidates[f'rstruct_{n}_{trial}'] = base + noise

    # Interpolate parent
    for n in [384, 512, 576, 640]:
        if n != n_parent and n <= 800:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, parent)
            interp = np.maximum(interp, 0)
            if interp.sum() > 1e-10:
                candidates[f'interp_{n}'] = interp

    # Score all candidates
    scored = []
    for name, a in candidates.items():
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            continue
        a_norm = a / s
        c = fast_convolve(a_norm)
        obj_val = 2.0 * len(a_norm) * c.max()
        scored.append((obj_val, name, a_norm))

    scored.sort(key=lambda x: x[0])
    print(f"  {len(scored)} candidates generated. Top 5:")
    for obj_val, name, _ in scored[:5]:
        print(f"    {name}: obj={obj_val:.6f}")

    # Select diverse top candidates (different lengths)
    top_cands = []
    seen_lens = set()
    for obj_val, name, a in scored:
        if len(a) not in seen_lens:
            top_cands.append(a)
            seen_lens.add(len(a))
        if len(top_cands) >= 4:
            break
    if parent not in top_cands:
        top_cands.append(parent)

    t_phase1 = time.monotonic() - t_phase1_start
    print(f"  Phase 1 done in {t_phase1:.1f}s")

    # === Phase 2: FW-LP on top candidates ===
    print("\n=== Phase 2: FW-LP ===")
    t2_start = time.monotonic()
    # Allocate time: leave 800s for phases 3-5
    t2_total = min(450, max(60, deadline - t2_start - 850))
    n_fw = min(3, len(top_cands))
    t_per = t2_total / max(1, n_fw)

    for i in range(n_fw):
        t_rem = deadline - time.monotonic() - 800
        t_alloc = min(t_per, t_rem)
        if t_alloc < 25:
            break
        cand = np.maximum(top_cands[i], 0)
        cand = cand / cand.sum()
        print(f"  FW-LP #{i+1} (n={len(cand)}, {t_alloc:.0f}s)")
        a_opt, _ = fw_lp_optimize(cand, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: LSE Polish ===
    print(f"\n=== Phase 3: LSE Polishing ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t3_start = time.monotonic()
    t3_total = min(300, max(50, deadline - t3_start - 500))
    print(f"  {t3_total:.0f}s on n={len(current_best)}")
    a_opt, _ = lse_optimize(current_best, time_limit=t3_total,
                            beta_start=10.0, beta_end=10000.0)
    try_sequence(a_opt, best_tracker)

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t4_start = time.monotonic()
    t4_total = min(350, max(50, deadline - t4_start - 250))
    print(f"  {t4_total:.0f}s on n={len(current_best)}")
    a_opt, _ = coordinate_descent(current_best, time_limit=t4_total)
    try_sequence(a_opt, best_tracker)

    # === Phase 5: Alternating LSE + CD ===
    print(f"\n=== Phase 5: Final Refinement ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t5_start = time.monotonic()
    t5_total = max(30, deadline - t5_start - 40)

    iteration = 0
    while time.monotonic() < t5_start + t5_total - 20:
        iteration += 1
        t_sub = min(80, deadline - time.monotonic() - 30)
        if t_sub < 30:
            break

        # LSE with high beta
        a_opt, _ = lse_optimize(current_best, time_limit=t_sub * 0.35,
                                beta_start=100.0 + 50 * iteration,
                                beta_end=20000.0)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best

        t_sub2 = min(80, deadline - time.monotonic() - 30)
        if t_sub2 < 20:
            break
        a_opt, _ = coordinate_descent(current_best, time_limit=t_sub2)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # === Final CD pass ===
    if time.monotonic() < deadline - 15:
        print(f"\n=== Final CD ===")
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        t_final = max(10, min(80, deadline - time.monotonic() - 10))
        a_opt, _ = coordinate_descent(current_best, time_limit=t_final)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL SCORE: {final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
        # If evaluation failed, try clipping
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL (clipped): {final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(448) / 448
    score = float(evaluate_sequence(fallback.tolist()))
    print(f"\nFALLBACK: {score:.8f}")
    return [float(x) for x in fallback]
