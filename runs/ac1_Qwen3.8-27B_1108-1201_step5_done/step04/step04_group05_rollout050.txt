def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, peak_weights):
    """Compute weighted subgradient of max(conv(a,a))."""
    n = len(a)

            continue

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * a[k - idx]
    return sg

Wait, I realize there's an issue with the greedy oracle. The direction g should minimize `<g, grad_f>` where grad_f is the subgradient at f. The subgradient of max_k conv(f,f)[k] with respect to f[j] is 2*f[k*-j] (for the peak k*). So we want to maximize `<g, -grad_f>` = minimize `<g, grad_f>`.

Actually, in Frank-Wolfe, we minimize `<g, grad_f>` over the feasible set. The feasible set here is {g >= 0, sum(g) = 1, conv(f, g) <= M} (approximately).

Without the conv constraint, the minimizer of `<g, grad_f>` over the simplex is: g[j] = 1 where grad_f[j] is minimum.

With the conv constraint, it's more complex. But as an approximation, we can:
1. Find positions where grad_f is small (i.e., f[k*-j] is small)
2. Among those, prefer positions where conv(f, g) is least likely to exceed M

A simpler approach: just set g[j] proportional to max(0, C - f[k*-j]) where C is some threshold. This favors positions where f[k*-j] is small.

Actually, I think the simplest effective approach is:
- g[j] = 1 for positions j where f[k*-j] is in the bottom half of values
- Normalize g

This is simple, fast, and should give good directions.

Let me also think about a **spectral** approach. The key insight is:

conv(a, a)[k] = sum_i a[i] * a[k-i]

If we think of a as a signal, then conv(a,a) is the aperiodic autocorrelation. We want this to be as flat as possible.

In the frequency domain: FFT(conv(a,a)) = |FFT(a)|^2

So we want |FFT(a)|^2 to have a flat inverse FFT, which means |FFT(a)| should be approximately constant (white spectrum).

This gives another good initialization: generate a random phase, set the magnitude to be constant, take IFFT, and take the absolute value (or project to non-negative with fixed sum).

"""
Hybrid Greedy Frank-Wolfe + LP optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses a fast greedy oracle (1000x faster than LP) as the main driver, with periodic
LP-based Frank-Wolfe corrections, spectral initialization, Nesterov-accelerated

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def greedy_oracle(f, n_peaks=5, quantile=0.4):
    """
    Fast greedy Frank-Wolfe oracle.
    Finds a direction g that should reduce the peaks of conv(f, g).
    Strategy: prefer positions where f[peak - j] is small for all top peaks.
    """
    n = len(f)
    conv = fast_convolve(f)
    
    # Find top peaks
    k_p = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k_p)[-k_p:]
    
    # Compute "danger" profile: for each position j, how much would it
    # contribute to the peaks if we put mass there?
    danger = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        vals = f[k - idx]
        danger[js:je + 1] = np.maximum(danger[js:je + 1], vals)
    
    # Set g[j] = 1 where danger is low (below quantile threshold)
    threshold = np.quantile(danger, quantile)
    g = (danger <= threshold).astype(np.float64)
    gs = g.sum()
    
    if gs < max(4, n * 0.05):
        # Fallback: use soft assignment
        g = 1.0 / (1.0 + danger * 10.0)
        gs = g.sum()
    
    if gs < 1e-10:
        g = np.ones(n)
        gs = n
    else:
        g /= gs
    
    return g


def greedy_fw_optimize(f0, time_limit, verbose=False, lp_every=0, lp_start_it=0):
    """
    Frank-Wolfe with greedy oracle and optional periodic LP corrections.
    lp_every: if > 0, use LP oracle every lp_every iterations.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng_local = np.random.default_rng(9999 + n)
    n_peaks = 3 + min(5, n // 100)

    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        use_lp = (lp_every > 0 and it > lp_start_it and it % lp_every == 0)
        
        if use_lp:
            g = solve_lp(f, M * m_exp)
            if g is None:
                m_exp *= 1.15
                if m_exp > 5.0:
                    m_exp = 1.0
                g = greedy_oracle(f, n_peaks, quantile)
            else:
                gs = g.sum()
                if gs < 1e-15:
                    g = greedy_oracle(f, n_peaks, quantile)
                else:
                    g = g / gs
                    m_exp = 1.0

            continue

                    lo = mid

                    hi = mid

            f = np.maximum(f, 0.0)

                f = f / fs
            conv = fast_convolve(f)

                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.01)

                if fs2 > 1e-10:

                else:

def nesterov_subgradient(f0, time_limit, lr_init=0.004, momentum_init=0.85):
    """Nesterov-accelerated projected subgradient with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)  # Velocity
    t_val = 2.0  # Nesterov t parameter
    
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    temp = 0.0008
    top_k = max(8, n // 5)
    rng_local = np.random.default_rng(4321 + n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        
        # Multi-scale peak weighting with temperature
        if top_k >= len(conv_y):
            peak_idx = np.arange(len(conv_y))
        else:
            peak_idx = np.argpartition(conv_y, -top_k)[-top_k:]
        
        pv = conv_y[peak_idx]
        scaled = (pv - pv.min()) / max(temp, 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_idx))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-13:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_new
        
        x_new = project_simplex(y - lr * sg)
        y = x_new + beta * (x_new - f)
        
        # Evaluate
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = x_new
        t_val = t_new
        
        # Adaptive step size
        if no_improve > 150:
            lr *= 0.92
            temp *= 0.97
            no_improve = 0
        
        # Periodic restart
        if no_improve > 400:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.005)
            y = f.copy()
            v = np.zeros(n)
            t_val = 2.0
            lr = max(lr_init * 0.5, 0.0005)
            no_improve = 0
        
        # Slowly increase temperature (explore more peaks)
        temp *= 1.0005
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_pairs=8):
    """Coordinate descent: try 2-point moves targeting the current peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng_local = np.random.default_rng(7890 + n)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Find the current peak
        k_star = int(np.argmax(conv))
        
        # Find source candidates: positions that contribute most to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        products = f[js:je + 1] * f[k_star - (js:je + 1)]
        
        # Get top contributors
        top_count = min(n_pairs, len(products))
        top_local = np.argsort(products)[::-1][:top_count]
        sources = js + top_local
        
        improved = False
        for src in sources:
            src = int(src)
            if f[src] < 1e-10:
                continue
            
            # Try moving mass to different targets
            # Good targets: positions far from the peak
            # k_star - target should be in [0, 2n-2]
            # We want positions where f[k_star - target] is small
            # i.e., where the "partner" value is small
            
            # Try a few random targets
            for _ in range(5):
                tgt = int(rng_local.integers(n))
                if tgt == src:
                    continue
                
                for frac in [0.1, 0.3, 0.5, 0.8]:
                    delta = f[src] * frac
                    if delta < 1e-12:
                        continue
                    f_new = f.copy()
                    f_new[src] -= delta
                    f_new[tgt] += delta
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-14:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 5000:
                # Restart with perturbation
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.008)
                conv = fast_convolve(f)
                no_improve = 0
                if conv.max() > best_obj * 1.001:
                    f = best_f.copy()
                    conv = fast_convolve(f)
                    no_improve = 2000
    
    return best_f, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Spectral (flat power spectrum)
    inits.append(spectral_init(n, rng))
    
    # Another spectral
    inits.append(spectral_init(n, rng))
    
    # Raised cosine window
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x + 0.5) / n))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            s_parent = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s_parent):
                print(f"  Parent score: {s_parent:.6f}")
                state['best_score'] = s_parent
                state['best_seq'] = parent.copy()
            else:
                print(f"  Parent invalid: {s_parent}")
    except Exception as e:
        print(f"  No parent or error: {e}")
    
    # === Phase 1: Multi-length exploration ===
    print(f"\n=== Phase 1: Exploration ({(float(budget_s) * 0.20):.0f}s budget) ===")
    phase1_end = time.monotonic() + float(budget_s) * 0.20
    
    # Lengths to explore
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for L in [320, 352, 368, 384, 400, 416, 432, 448]:
        if L not in lengths:
            lengths.append(L)
    
    # Sort: parent length first, then by proximity
    if parent is not None:
        lengths.sort(key=lambda x: abs(x - len(parent)))
    
    n_lengths = min(5, len(lengths))
    t_per_length = (phase1_end - time.monotonic() - 10) / n_lengths
    
    for li in range(n_lengths):
        n = lengths[li]
        t_start = time.monotonic()
        t_alloc = min(t_per_length - 5, phase1_end - time.monotonic() - 10)
        if t_alloc < 15:
            break
        
        print(f"  Exploring n={n} (t_alloc={t_alloc:.0f}s)")
        inits = generate_inits(n, parent, rng)
        n_try = min(3, len(inits))
        t_per_init = t_alloc / n_try
        
        for i in range(n_try):
            t_init_alloc = min(t_per_init - 2, phase1_end - time.monotonic() - 5)
            if t_init_alloc < 10:
                break
            f_init = inits[i]
            a_opt, obj = greedy_fw_optimize(f_init, time_limit=t_init_alloc)
            try_sequence(a_opt, f"expl_n{n}_i{i}", state)
        
        el = time.monotonic() - t_start
        print(f"    n={n} done in {el:.1f}s")
    
    print(f"  Phase 1 best: {state['best_score']:.6f}")
    
    # === Phase 2: Deep hybrid optimization ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + float(budget_s) * 0.45
    
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        remaining_p2 = phase2_end - time.monotonic() - 10
        print(f"  Optimizing n={n_best}, budget={remaining_p2:.0f}s")
        
        # Main deep run: greedy FW with periodic LP
        t_main = min(remaining_p2 * 0.45, 300)
        if t_main > 30:
            print(f"  Main hybrid run (t={t_main:.0f}s, LP every 300 iters)")
            a_opt, obj = greedy_fw_optimize(state['best_seq'], time_limit=t_main,
                                            verbose=True, lp_every=300, lp_start_it=50)
            try_sequence(a_opt, "deep_hybrid", state)
        
        # Second run from perturbed start
        rem = phase2_end - time.monotonic() - 20
        if rem > 50:
            t2 = min(rem * 0.35, 150)
            if t2 > 20:
                noise = rng.standard_normal(n_best) * 0.02
                f_start = project_simplex(state['best_seq'] + noise)
                print(f"  Perturbed run (t={t2:.0f}s)")
                a_opt, obj = greedy_fw_optimize(f_start, time_limit=t2,
                                                verbose=False, lp_every=400, lp_start_it=100)
                try_sequence(a_opt, "deep_pert", state)
        
        # Mirror blend run
        rem = phase2_end - time.monotonic() - 15
        if rem > 30:
            t3 = min(rem * 0.4, 100)
            if t3 > 20:
                f_mirror = project_simplex(0.8 * state['best_seq'] + 0.2 * state['best_seq'][::-1])
                print(f"  Mirror run (t={t3:.0f}s)")
                a_opt, obj = greedy_fw_optimize(f_mirror, time_limit=t3, lp_every=500)
                try_sequence(a_opt, "deep_mirror", state)
        
        # Dedicated LP-FW run (fewer but higher quality steps)
        rem = phase2_end - time.monotonic() - 10
        if rem > 40:
            t4 = min(rem, 80)
            if t4 > 20:
                print(f"  LP-heavy run (t={t4:.0f}s, LP every 50 iters)")
                a_opt, obj = greedy_fw_optimize(state['best_seq'], time_limit=t4,
                                                lp_every=50, lp_start_it=0)
                try_sequence(a_opt, "deep_lp", state)
    
    print(f"  Phase 2 best: {state['best_score']:.6f}")
    
    # === Phase 3: Nesterov subgradient polish ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    phase3_end = deadline - 60
    
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        
        # Long Nesterov run
        rem = phase3_end - time.monotonic() - 40
        if rem > 30:
            t_sg = min(rem * 0.5, 200)
            print(f"  Nesterov SG (t={t_sg:.0f}s)")
            a_opt, obj = nesterov_subgradient(state['best_seq'], time_limit=t_sg,
                                              lr_init=0.003, momentum_init=0.88)
            try_sequence(a_opt, "sg_nested", state)
        
        # Second SG from perturbed
        rem = phase3_end - time.monotonic() - 20
        if rem > 30:
            t_sg2 = min(rem * 0.4, 100)
            if t_sg2 > 15:
                noise = rng.standard_normal(n_best) * 0.005
                f_start = project_simplex(state['best_seq'] + noise)
                print(f"  Nesterov SG perturb (t={t_sg2:.0f}s)")
                a_opt, obj = nesterov_subgradient(f_start, time_limit=t_sg2,
                                                  lr_init=0.002, momentum_init=0.90)
                try_sequence(a_opt, "sg_nested_pert", state)
    
    print(f"  Phase 3 best: {state['best_score']:.6f}")
    
    # === Phase 4: Coordinate descent + final squeeze ===
    print(f"\n=== Phase 4: Final Polish ===")
    
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        
        # Coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_cd = min(rem * 0.5, 100)
            print(f"  Coordinate descent (t={t_cd:.0f}s)")
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=t_cd, n_pairs=6)
            try_sequence(a_opt, "cd_final", state)
        
        # Quick greedy FW squeeze
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            print(f"  Final greedy FW squeeze (t={rem:.0f}s)")
            a_opt, obj = greedy_fw_optimize(state['best_seq'], time_limit=rem)
            try_sequence(a_opt, "final_squeeze", state)
    
    # Fallback
    if state['best_seq'] is None:
        print("  WARNING: No valid sequence found, using uniform fallback")
        n_fb = 384
        state['best_seq'] = np.ones(n_fb) / n_fb
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(final_score):
        state['best_score'] = final_score
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    print(f"{'='*50}")
    
    return [float(x) for x in state['best_seq']]
