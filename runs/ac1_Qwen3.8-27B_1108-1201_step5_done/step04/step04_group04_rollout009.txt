"""

Combines: (1) LP-based Frank-Wolfe, (2) Entropic mirror descent (multiplicative weights),

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(out, 0.0)  # Clamp tiny negative from FFT


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = res.x
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None


def smoothed_obj_and_grad(f, T):
    """Compute log-sum-exp smoothed max of convolution and its gradient."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    obj = T * (np.log(ws) + conv.max() / T)  # = T*log(sum(exp(conv/T)))
    # Gradient: 2 * cross_correlation(f, p)
    fft_size = 1 << (3 * n).bit_length()
    F = np.fft.rfft(f, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(F) * P, fft_size)
    grad = 2.0 * cross[:n]
    return obj, grad, conv

def adam_softmax_optimize(f0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, verbose=False):
    """Adam optimizer on softmax parameterization."""
    f = np.maximum(f0.astype(np.float64), 1e-30)
    f = f / f.sum()
    n = len(f)
    # Initialize in log-space
    log_f = np.log(f + 1e-30)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        obj, grad, conv = smoothed_obj_and_grad(f, T=0.005 * best_obj)
        grad_tangent = grad - np.mean(grad)

        m = beta1 * m + (1 - beta1) * grad_tangent
        v = beta2 * v + (1 - beta2) * grad_tangent ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        log_f = log_f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # Softmax normalization
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9

        if no_improve > 300:
            rng = np.random.default_rng((t * 31337 + 11) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            log_f = np.log(f + 1e-30)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.01
            no_improve = 0

    return best_f, best_obj

        conv = fast_convolve(f)
        # Find top peaks and weight them
        top_k = max(3, int(n * 0.05))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f_new = project_simplex(f - lr * sg)
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.9

        if it % 100 == 0:
            temp *= 1.02
            lr = max(lr * 1.005, 0.0002)

        if no_improve > 300:
            rng = np.random.default_rng((it * 991 + 3) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            lr = 0.0008
            no_improve = 0

    return best_f, best_obj


def mirror_symmetrize(f):
    """Average with mirror to enforce palindromic symmetry."""
    n = len(f)
    mirrored = 0.5 * f + 0.5 * f[::-1]
    return project_simplex(mirrored)


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # U-shape (edge-concentrated)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.5)
    inits.append(edge_w / edge_w.sum())

    # Stronger U-shape
    edge_w2 = 1.0 / (np.minimum(x, n - 1 - x) + 0.5) ** 1.5
    inits.append(edge_w2 / edge_w2.sum())

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n)) ** 2 + 0.01
    inits.append(cos_w / cos_w.sum())

    # Two bumps
    h1 = np.exp(-0.5 * ((x - 0.15 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.85 * n) / (0.08 * n)) ** 2)
    two_bump = h1 + h2 + 0.05
    inits.append(two_bump / two_bump.sum())

    # Mirror-symmetric Dirichlet
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.7)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] += 0.01
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: Multi-method hybrid search.
    Phase 1: Quick multi-length exploration with LP-FW.
    Phase 2: Deep optimization alternating FW, entropic MD, and Adam.
    Phase 3: Fine polishing with subgradient + very fine smoothed GD.
    Phase 4: Final tricks (mirror, perturbation, length tweaks).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # === Phase 1: Multi-length Exploration (200s) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 200.0

    base_n = len(parent) if parent is not None else 384
    lengths = sorted(set([
        base_n, base_n - 8, base_n - 16, base_n + 8, base_n + 16,
        base_n - 32, base_n + 32, base_n - 64, base_n + 64,
        256, 320, 384, 448, 512, 288, 416, 448
    ]))
    lengths = [l for l in lengths if 128 <= l <= 1024]

    # Start with parent length
    lengths_sorted = sorted(lengths, key=lambda l: abs(l - base_n))

    for n in lengths_sorted[:6]:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 25:
            break
        t_alloc = min(30.0, rem - 15.0)
        if t_alloc < 15:
            break

        inits = generate_inits(n, parent, rng)
        # Pick best init quickly
        best_init = None
        best_init_obj = float('inf')
        for f_init in inits:
            obj = fast_convolve(f_init).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = f_init

        if best_init is not None:
            print(f"  Exploring n={n}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_init, time_limit=t_alloc)
            try_seq(a_opt, f"explore_n{n}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep Optimization (800s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 800.0

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Alternate methods in cycles
        n_cycles = 5
        t_per_cycle = p2_budget / n_cycles

        for cycle in range(n_cycles):
            rem = p2_budget - (time.monotonic() - t_p2)
            if rem < 30:
                break

            # Method A: LP-FW
            t_a = min(t_per_cycle * 0.35, rem * 0.3)
            if t_a > 20:
                print(f"  Cycle {cycle}: LP-FW ({t_a:.0f}s)")
                a_opt, _ = fw_lp_optimize(f_work, time_limit=t_a)
                try_seq(a_opt, f"fw_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # Method B: Entropic Mirror Descent
            rem = p2_budget - (time.monotonic() - t_p2)
            t_b = min(t_per_cycle * 0.3, rem * 0.3)
            if t_b > 20:
                print(f"  Cycle {cycle}: Entropic MD ({t_b:.0f}s)")
                a_opt, _ = entropic_mirror_descent(f_work, time_limit=t_b,
                                                   eta_init=0.3 + 0.2 * cycle * 0.1)
                try_seq(a_opt, f"emd_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # Method C: Adam on softmax
            rem = p2_budget - (time.monotonic() - t_p2)
            t_c = min(t_per_cycle * 0.3, rem * 0.3)
            if t_c > 20:
                print(f"  Cycle {cycle}: Adam Softmax ({t_c:.0f}s)")
                lr_c = 0.005 * (1.0 + 0.3 * cycle)
                a_opt, _ = adam_softmax_optimize(f_work, time_limit=t_c, lr=lr_c)
                try_seq(a_opt, f"adam_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # Mirror symmetrize + quick polish
            rem = p2_budget - (time.monotonic() - t_p2)
            if rem > 20:
                mirrored = mirror_symmetrize(f_work)
                if np.max(np.abs(mirrored - f_work)) > 1e-6:
                    t_m = min(15.0, rem * 0.2)
                    a_opt, _ = fw_lp_optimize(mirrored, time_limit=t_m)
                    try_seq(a_opt, f"mirror_{cycle}")
                    if best_seq is not None:
                        f_work = best_seq / best_seq.sum()

    # Try a couple of different lengths with the best shape interpolated
    rem = p2_budget - (time.monotonic() - t_p2)
    if rem > 50 and best_seq is not None:
        n_best = len(best_seq)
        for delta in [16, -16, 32]:
            rem = p2_budget - (time.monotonic() - t_p2)
            if rem < 30:
                break
            new_n = n_best + delta
            if new_n < 128 or new_n > 1024:
                continue
            px = np.interp(np.linspace(0, 1, new_n),
                           np.linspace(0, 1, n_best),
                           best_seq / best_seq.sum())
            px = project_simplex(px)
            t_alloc = min(30.0, rem / 2.0)
            a_opt, _ = fw_lp_optimize(px, time_limit=t_alloc)
            try_seq(a_opt, f"p2_len{new_n}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Fine Polishing (500s) ===
    print(f"\n=== Phase 3: Fine Polishing ===")
    t_p3 = time.monotonic()
    p3_budget = 500.0

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Subgradient polish with decreasing parameters
        sg_configs = [
            {"lr": 0.0015, "temp": 0.0005},
            {"lr": 0.001, "temp": 0.0002},
            {"lr": 0.0005, "temp": 0.0001},
            {"lr": 0.0003, "temp": 0.00005},
            {"lr": 0.0002, "temp": 0.00002},
        ]

        for i, cfg in enumerate(sg_configs):
            rem = p3_budget - (time.monotonic() - t_p3)
            if rem < 25:
                break
            t_alloc = min(70.0, rem - 15.0)
            if t_alloc < 15:
                break
            print(f"  SG polish {i} (t={t_alloc:.0f}s, lr={cfg['lr']}, T={cfg['temp']})")
            a_opt, _ = fine_subgradient_polish(f_work, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"sg_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Very fine smoothed gradient (low temperature)
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 40:
            T_init = max(0.0002, 0.0005 * fast_convolve(f_work).max())
            t_alloc = min(80.0, rem - 15.0)
            print(f"  Fine smoothed GD (t={t_alloc:.0f}s, T={T_init:.6f})")
            a_opt, _ = adam_softmax_optimize(f_work, time_limit=t_alloc, lr=0.003)
            try_seq(a_opt, "fine_adam")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Final LP-FW push
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            t_alloc = min(60.0, rem - 10.0)
            print(f"  Final LP-FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Final Tricks ===
    print(f"\n=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Mirror + quick polish
        mirrored = mirror_symmetrize(f_work)
        if np.max(np.abs(mirrored - f_work)) > 1e-7:
            a_opt, _ = fine_subgradient_polish(mirrored, time_limit=15, lr=0.0005, temp=0.00005)
            try_seq(a_opt, "p4_mirror")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Tiny perturbations
        for i in range(5):
            if time.monotonic() > deadline - 20:
                break
            noise = rng.standard_normal(n_best) * 0.0003 * (i + 1)
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = fine_subgradient_polish(f_pert, time_limit=8, lr=0.0004, temp=0.00003)
            try_seq(a_opt, f"p4_perturb_{i}")

        # One last micro LP-FW
        if time.monotonic() < deadline - 35:
            a_opt, _ = fw_lp_optimize(f_work, time_limit=25)
            try_seq(a_opt, "p4_final_fw")

        # Try slightly different lengths one more time
        if time.monotonic() < deadline - 30:
            for delta in [8, -8]:
                new_n = n_best + delta
                if new_n < 128 or new_n > 1024:
                    continue
                px = np.interp(np.linspace(0, 1, new_n),
                               np.linspace(0, 1, n_best), f_work)
                px = project_simplex(px)
                a_opt, _ = fw_lp_optimize(px, time_limit=15)
                try_seq(a_opt, f"p4_len{new_n}")
                if time.monotonic() > deadline - 15:
                    break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
