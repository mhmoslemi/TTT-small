"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based Frank-Wolfe, coordinate descent with adaptive 1D search,
smoothed subgradient descent, and multi-start exploration.
Key improvements over previous: adaptive multi-peak LP constraints,
efficient coordinate descent exploiting 1D structure, and n-value diversity.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = sum_j f[k-j]*g[j] (cross-convolution)."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B

    A_ub[:, 1:] = -2.0 * B_full[peaks_sorted, :]
    b_ub = -conv_f[peaks_sorted]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[1:]
    return None


def lp_direction_weighted(f, conv_f, top_k=60, temperature=0.001):
    """
    Weighted LP direction: constraints weighted by softmax of peak values.
    Higher peaks get tighter constraints.
    """
    n = len(f)
    num_constr = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv_f, -num_constr)[-num_constr:]
    peak_vals = conv_f[peaks]

    # Softmax weights
    scaled = peak_vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    B_full = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((num_constr, n + 1))
    A_ub[:, 0] = -1.0
    for i, pk in enumerate(peaks):
        A_ub[i, 1:] = -2.0 * B_full[pk, :] * w[i]
    b_ub = -peak_vals * w

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[1:]
    return None


def golden_line_search(f, target, current_obj, max_iter=30):
    """Golden section line search for alpha in (0, 1]."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0

    a1 = hi - phi * (hi - lo)
    a2 = lo + phi * (hi - lo)

    def objective(alpha):
        if alpha < 1e-10:
            return current_obj
        f_new = (1.0 - alpha) * f + alpha * target
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            return float('inf')
        f_new = f_new / s
        return fast_convolve(f_new).max()

    f1 = objective(a1)
    f2 = objective(a2)

    best_alpha = 0.0
    best_val = current_obj
    if f1 < best_val:
        best_val = f1
        best_alpha = a1
    if f2 < best_val:
        best_val = f2
        best_alpha = a2

    for _ in range(max_iter):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - phi * (hi - lo)
            f1 = objective(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + phi * (hi - lo)
            f2 = objective(a2)

        if f1 < best_val:
            best_val = f1
            best_alpha = a1
        if f2 < best_val:
            best_val = f2
            best_alpha = a2

        if hi - lo < 1e-6:
            break

    return best_alpha, best_val


def fw_optimize(f0, time_limit, top_k=40, use_weighted=False, verbose=False):
    """Frank-Wolfe with LP oracle and golden section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    temp = 0.002

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Adaptive top_k: more peaks when stuck
        current_top_k = top_k if no_improve < 30 else min(top_k + 20, 2 * n - 1)
        current_temp = temp * (1.0 + no_improve * 0.05)

        if use_weighted:
            g = lp_direction_weighted(f, conv, top_k=current_top_k, temperature=current_temp)
        else:
            g = lp_direction(f, conv, top_k=current_top_k)

        if g is None:
            no_improve += 3
            if no_improve > 80:
                scale = 0.003 + 0.001 * (no_improve // 20)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                no_improve = 0
            continue

        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        # Check if target is too similar to f
        if np.max(np.abs(target - f)) < 5e-9:
            no_improve += 1
            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                no_improve = 0
            continue

        alpha, obj_val = golden_line_search(f, target, best_obj)

        if alpha > 1e-8:
            f = (1.0 - alpha) * f + alpha * target
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            co = conv.max()

            if co < best_obj - 1e-15:
                best_obj = co
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, alpha={alpha:.4f}, t={elapsed:.0f}s")
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def coordinate_descent(f0, time_limit, rng, verbose=False):
    """
    Coordinate descent with adaptive 1D search.
    For each coordinate, try a grid of multiplicative and additive changes.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    pass_count = 0
    no_improve_passes = 0
    order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        np.random.shuffle(order)
        pass_count += 1
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-10:
                # Try small positive values
                candidates = np.array([0.001, 0.005, 0.01, 0.02]) / n
            else:
                # Multiplicative and additive candidates
                mult_factors = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]
                candidates = old_val * np.array(mult_factors)
                # Also try some absolute offsets
                spread = f.max() * 0.1
                candidates = np.concatenate([candidates, [old_val - spread * [0.3, 0.6],
                                                          old_val + spread * [0.3, 0.6]]])
                candidates = candidates[candidates >= 0]

            best_local = old_val
            best_local_obj = best_obj

            for cv in candidates:
                f[j] = cv
                fs = f.sum()
                if fs < 1e-15:
                    f[j] = old_val
                    continue
                f = f / fs
                co = fast_convolve(f).max()
                if co < best_local_obj - 1e-14:
                    best_local_obj = co
                    best_local = cv

            f[j] = best_local if best_local != old_val else old_val
            if best_local != old_val:
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                improved = True

                co = fast_convolve(f).max()
                if co < best_obj - 1e-14:
                    best_obj = co
                    best_f = f.copy()

        if improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1

        if verbose and pass_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}, t={elapsed:.0f}s")

        if no_improve_passes > 15:
            # Perturb and restart
            scale = 0.002 + 0.001 * no_improve_passes
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_f = f.copy()
            no_improve_passes = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.005, momentum=0.9, top_k=30, temp=0.0005):
    """Momentum subgradient descent on the max-conv objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1

        num_peaks = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -num_peaks)[-num_peaks:]
        pv = conv[peaks]

        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    sg[j] += w[i] * (f[kj] + f[j])

        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        temp *= 1.003

        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.001)
            no_improve = 0

    return best_f, best_obj


def try_different_n(best_f, n_new, time_budget):
    """Interpolate to new length and run quick optimization."""
    n_old = len(best_f)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    f_new = np.interp(x_new, x_old, best_f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return None
    f_new = f_new / s

    t0 = time.monotonic()
    f_opt, _ = fw_optimize(f_new, time_limit=time_budget * 0.6, top_k=30)
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        f_opt, _ = coordinate_descent(f_opt, time_limit=rem,
                                       rng=np.random.default_rng(99))
    return f_opt


def generate_structured_init(n, rng):
    """Generate a structured initialization with low autocorrelation bias."""
    x = np.linspace(0, 1, n)
    inits = []

    # Cosine-based (low autocorrelation at center)
    c = np.cos(np.pi * x) ** 2 + 0.1
    inits.append(c / c.sum())

    # Walsh-like (alternating)
    w = np.ones(n)
    for i in range(n):
        w[i] = 1.0 + 0.5 * np.cos(2 * np.pi * i / n) + 0.3 * np.cos(4 * np.pi * i / n)
    w = np.maximum(w, 0.01)
    inits.append(w / w.sum())

    # Gaussian mixtures (2-3 bumps)
    for _ in range(3):
        centers = sorted(rng.uniform(0.1, 0.9, rng.integers(2, 5)).tolist())
        sigmas = [rng.uniform(0.05, 0.2) for _ in centers]
        amps = [rng.uniform(0.5, 1.0) for _ in centers]
        g = np.zeros(n)
        for c_, s_, a_ in zip(centers, sigmas, amps):
            g += a_ * np.exp(-0.5 * ((x - c_) / s_) ** 2)
        g = np.maximum(g, 1e-6)
        inits.append(g / g.sum())

    # Linear ramp
    r = np.linspace(0.1, 1.0, n)
    inits.append(r / r.sum())

    # U-shape
    u = 1.0 / (0.5 + (x - 0.5) ** 2 * n * 0.1)
    inits.append(u / u.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimizer: LP-Frank-Wolfe + Coordinate Descent + Subgradient.
    Multi-phase with n-value diversity and multi-start exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0).astype(np.float64)
        if a.sum() < 0.01:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    # === Phase 1: Initialization (100s) ===
    print("\n=== Phase 1: Initialization ===")
    t_start = time.monotonic()
    t_phase1_end = t_start + 100

    if parent is not None:
        n_main = len(parent)
        start_f = parent.copy()
    else:
        n_main = 384
        start_f = np.ones(n_main) / n_main
        try_seq(start_f, "uniform_init")

    # Try a few structured inits at n_main
    inits = generate_structured_init(n_main, rng)
    for i, ini in enumerate(inits[:4]):
        try_seq(ini, f"struct_{i}")

    # Also try n=512
    n_alt = 512
    f_alt = try_different_n(start_f, n_alt, 30)
    if f_alt is not None:
        try_seq(f_alt, "n512_init")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")

    # === Phase 2: LP Frank-Wolfe (600s) ===
    print("\n=== Phase 2: LP Frank-Wolfe ===")
    n_best = len(best_seq)
    t2_budget = min(600, deadline - time.monotonic() - 400)
    if t2_budget > 30:
        print(f"  Running FW (n={n_best}), budget={t2_budget:.0f}s")
        f_opt, _ = fw_optimize(best_seq, time_limit=t2_budget * 0.55, top_k=40, verbose=True)
        try_seq(f_opt, "fw_main")

        rem = deadline - time.monotonic() - 350
        if rem > 30:
            f_opt, _ = fw_optimize(best_seq, time_limit=min(rem * 0.6, 200),
                                    top_k=50, use_weighted=True)
            try_seq(f_opt, "fw_weighted")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Coordinate Descent (500s) ===
    print("\n=== Phase 3: Coordinate Descent ===")
    t3_budget = min(500, deadline - time.monotonic() - 200)
    if t3_budget > 20:
        print(f"  Running CD (n={len(best_seq)}), budget={t3_budget:.0f}s")
        f_opt, _ = coordinate_descent(best_seq, time_limit=t3_budget, rng=rng, verbose=True)
        try_seq(f_opt, "cd_main")

        rem = deadline - time.monotonic() - 150
        if rem > 20:
            f_opt, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.7, 200),
                                           rng=np.random.default_rng(seed + 1))
            try_seq(f_opt, "cd_pass2")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Subgradient + FW combo (300s) ===
    print("\n=== Phase 4: Subgradient + FW ===")
    t4_budget = min(300, deadline - time.monotonic() - 100)
    if t4_budget > 20:
        # Subgradient
        t_sg = min(t4_budget * 0.4, 100)
        f_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                         lr=0.004, momentum=0.92,
                                         top_k=len(best_seq) // 5, temp=0.0003)
        try_seq(f_opt, "sg")

        rem = deadline - time.monotonic() - 80
        if rem > 30:
            # Quick FW after SG
            f_opt, _ = fw_optimize(f_opt, time_limit=min(rem * 0.5, 100), top_k=35)
            try_seq(f_opt, "sg_fw")

            rem2 = deadline - time.monotonic() - 50
            if rem2 > 15:
                f_opt2, _ = coordinate_descent(f_opt, time_limit=rem2,
                                                rng=np.random.default_rng(seed + 2))
                try_seq(f_opt2, "sg_cd")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Multi-start exploration (200s) ===
    print("\n=== Phase 5: Multi-start ===")
    t5_budget = min(200, deadline - time.monotonic() - 40)
    if t5_budget > 30:
        n_trials = min(4, int(t5_budget // 40))
        t_each = t5_budget / max(1, n_trials)
        for trial in range(n_trials):
            if time.monotonic() > deadline - 30:
                break
            scale = 0.005 + 0.005 * trial
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
            t_sub = min(t_each * 0.5, 50)
            f_opt, _ = fw_optimize(f_pert, time_limit=t_sub, top_k=30)
            rem = t_each - (time.monotonic() - (deadline - t5_budget + trial * t_each))
            rem = max(rem, 5)
            rem = min(rem, t_each * 0.4)
            f_opt, _ = coordinate_descent(f_opt, time_limit=rem,
                                           rng=np.random.default_rng(seed + 100 + trial))
            try_seq(f_opt, f"multi_{trial}")

        # Try different n
        rem = deadline - time.monotonic() - 30
        if rem > 30 and best_seq is not None:
            for n_try in [len(best_seq) - 32, len(best_seq) + 32, 512]:
                if n_try < 64 or n_try > 1024 or rem < 20:
                    continue
                f_new = try_different_n(best_seq, n_try, rem * 0.4)
                if f_new is not None:
                    try_seq(f_new, f"n{n_try}")
                rem = deadline - time.monotonic() - 25

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Final polish (remaining time) ===
    print("\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic() - 5
    if rem > 10 and best_seq is not None:
        f_opt, _ = coordinate_descent(best_seq, time_limit=rem,
                                       rng=np.random.default_rng(seed + 999))
        try_seq(f_opt, "final_cd")

    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
