```python
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    return np.fft.irfft(A * A, fft_size)[:size]

    """Project v onto the simplex {x >= 0, sum(x) = s}."""

    if n == 0:
```

```python
def fast_convolve(a):
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v, s):
    """Project onto {x >= 0, sum(x) = s}."""
    v = v.copy()
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho such that u[rho] > (cssv[rho] - s) / (rho+1)
    rho_candidates = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0]

        return np.full(n, s/n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    s = a.sum()
    if s < 1e-15:

    return 2.0 * n * conv.max() / (s * s)

    n = len(a)
    peaks = np.argpartition(conv, -k)[-k:]

    w = 1.0 / k
    grad = np.zeros(n)
    for pk in peaks:

        # (P_pk a)[i] = a[pk - i] if 0 <= pk - i < n
        i_lo = max(0, pk - n + 1)
        i_hi = min(n - 1, pk)
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += w * a[pk - idx]
    return 2.0 * grad  # gradient of a^T P a is 2 P a
```

```python
def lp_direction(a):
    """Find direction g that minimizes max conv(a,g) subject to sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    
    # Build constraint matrix C: C[k, j] = a[k-j] if valid
    C = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] appears in conv[k] for k where a[k-j] is valid, i.e., 0 <= k-j < n
        # So k ranges from j to j+n-1
        C[j:j+n, j] = a
    
    # LP: min t s.t. C@g - t <= 0, sum(g) = 1, g >= 0
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: C@g - t <= 0
    A_ub = np.hstack([C, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    if result.success:
        return result.x[:n]
    return None
```

```python
import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute objective 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def conv_obj(a):
    """Return (conv, objective) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return conv, 2.0 * n * conv.max()


def project_simplex(v, s=1.0):
    """Project v onto simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_mask = u * idx > (cssv - s)
    rho_candidates = np.nonzero(rho_mask)[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_top_k(a, conv, k=5):
    """Exact subgradient using top-k peaks with proportional weighting."""
    n = len(a)
    if k >= len(conv):
        k = len(conv)
    peaks = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peaks]
    w_sum = peak_vals.sum()
    if w_sum < 1e-30:
        w = np.full(k, 1.0 / k)
    else:
        w = peak_vals / w_sum
    grad = np.zeros(n)
    for idx in range(k):
        if w[idx] < 1e-15:
            continue
        pk = int(peaks[idx])
        i_lo = max(0, pk - n + 1)
        i_hi = min(n - 1, pk)
        idxs = np.arange(i_lo, i_hi + 1)
        grad[idxs] += w[idx] * a[pk - idxs]
    return 2.0 * grad


def nesterov_optimize(a0, time_limit, k_peaks=5, lr_init=0.02, verbose=False):
    """
    Nesterov-accelerated projected subgradient with adaptive restarts.
    Works on sum=1 normalized sequences.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    best_a = a.copy()
    _, best_obj = conv_obj(a)

    t0 = time.monotonic()
    lr = lr_init
    momentum = 0.85
    no_improve = 0
    iteration = 0
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        batch = 50
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            # Nesterov lookahead
            y = project_simplex(a + momentum * v, 1.0)
            conv_y = fast_convolve(y)
            grad = subgradient_top_k(y, conv_y, k=k_peaks)

            v = momentum * v - lr * grad
            a_new = project_simplex(a + v, 1.0)
            _, obj_new = conv_obj(a_new)

            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
            iteration += 1

        # Adaptive updates
        if no_improve >= 40:
            lr *= 0.7
            momentum = min(0.97, momentum + 0.01)
        if no_improve >= 150:
            restart_count += 1
            perturb = min(0.005, 0.0005 * restart_count)
            rng = np.random.default_rng(42 + restart_count * 1000 + iteration)
            a = project_simplex(best_a + rng.standard_normal(n) * perturb, 1.0)
            v = np.zeros(n)
            lr = lr_init * max(0.1, 0.9 ** restart_count)
            momentum = 0.85
            no_improve = 0
            if restart_count > 50:
                break

        if verbose and iteration % 2000 == 0:
            print(f"    Nest iter={iteration}, obj={best_obj:.8f}, lr={lr:.5f}, "
                  f"mom={momentum:.3f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def lp_frank_wolfe(a, time_limit, max_iter=None, verbose=False):
    """
    Frank-Wolfe: find g on simplex minimizing max conv(a,g), then step.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Pre-build constraint matrix: C[k,j] = a[k-j] if 0<=k-j<n
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a

    best_a = a.copy()
    _, best_obj = conv_obj(a)
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if max_iter is not None and iteration >= max_iter:
            break

        # Solve LP: min t s.t. C@g <= t, sum(g)=1, g>=0
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(-np.inf, np.inf)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 2})
        except Exception:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        if not res.success:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        g = res.x[:n]
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            no_improve += 1
            continue
        g = g / gs

        # Step size: try geometric sequence
        a_curr = best_a.copy()
        for tau in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            a_try = (1.0 - tau) * a_curr + tau * g
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            _, obj_try = conv_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                no_improve = 0
                break
        else:
            no_improve += 1

        # Update current point for next iteration
        a = best_a
        # Rebuild C if a changed significantly
        if no_improve == 0:
            for j in range(n):
                C[j:j + n, j] = a

        iteration += 1
        if no_improve >= 50:
            # Restart with perturbation
            perturb = 0.003 * min(no_improve // 50, 5)
            rng = np.random.default_rng(iteration * 31 + 7)
            best_a = project_simplex(best_a + rng.standard_normal(n) * perturb, 1.0)
            a = best_a.copy()
            for j in range(n):
                C[j:j + n, j] = a
            no_improve = 0
            if perturb > 0.015:
                break

        if verbose and iteration % 20 == 0:
            print(f"    FW iter={iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: adjust pairs to reduce the current peak.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    peak_k = int(np.argmax(conv))
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    no_improve = 0
    iters_per_check = max(1, int(30000 / max(1, n)))

    while time.monotonic() - t0 < time_limit:
        # Focus on the current peak
        peak_k = int(np.argmax(conv))
        # Elements contributing to peak: a[i] * a[peak_k - i]
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)

        # Try moving mass from high contributors to low positions
        for _ in range(iters_per_check):
            if time.monotonic() - t0 > time_limit:
                break
            # Pick a contributor i with high a[i]*a[peak_k-i]
            contrib = a[i_lo:i_hi + 1] * a[peak_k - np.arange(i_lo, i_hi + 1)]
            if contrib.max() < 1e-15:
                break
            i_pick = i_lo + int(np.argmax(contrib))
            partner = peak_k - i_pick

            # Find a "safe" position j (far from peak) to move mass to
            dist = np.abs(np.arange(n) - i_pick)
            safe_mask = dist > n // 4
            if not safe_mask.any():
                safe_mask = np.ones(n, dtype=bool)
            safe_mask[i_pick] = False
            j_pick = safe_mask.nonzero()[0][rng.integers(0, safe_mask.sum())]

            # Try moving small delta from i_pick to j_pick (or vice versa)
            for delta in [0.001, 0.003, 0.005, 0.01, 0.02,
                          -0.001, -0.003, -0.005, -0.01]:
                if a[i_pick] + delta < 0 or a[j_pick] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i_pick] += delta
                a_try[j_pick] -= delta
                st = a_try.sum()
                if st < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-11:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
            else:
                no_improve += 1

        if no_improve > 300:
            no_improve = 0
            # Perturb
            scale = 0.002
            a = project_simplex(best_a + rng.standard_normal(n) * scale, 1.0)
            conv = fast_convolve(a)

        if verbose:
            print(f"    CD t={time.monotonic()-t0:.1f}s, obj={best_obj:.8f}")

    return best_a, best_obj


def generate_structural_inits(n, parent, rng):
    """Generate diverse structural initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Linear ramp (increasing)
    ramp = x + 1.0
    inits['ramp_inc'] = ramp / ramp.sum()

    # Linear ramp (decreasing)
    inits['ramp_dec'] = (n - x) / (n * (n + 1) / 2.0)

    # Parabolic (symmetric, high in middle)
    center = (n - 1) / 2.0
    para = 1.0 - ((x - center) / center) ** 2
    para = np.maximum(para, 0.01)
    inits['parabola'] = para / para.sum()

    # Exponential decay
    exps = np.exp(-x / (n / 4.0))
    inits['exp_decay'] = exps / exps.sum()

    # Sine squared
    s1 = np.sin(np.pi * x / n) ** 2
    inits['sine'] = s1 / s1.sum()

    # Bimodal (high at edges)
    edges = min(n // 4, 100)
    bi = np.zeros(n)
    bi[:edges] = 1.0
    bi[-edges:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Cosine window
    cosw = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cosw = np.maximum(cosw, 0.001)
    inits['cosine'] = cosw / cosw.sum()

    # Asymmetric: high left, low right with smooth transition
    asym = np.exp(-x / (n / 3.0)) + 0.3 * np.exp(-(n - 1 - x) / (n / 3.0))
    inits['asym'] = asym / asym.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Quick gradient scan across multiple initializations
    Phase 2: LP Frank-Wolfe deep optimization
    Phase 3: Coordinate descent polishing
    Phase 4: Multi-restart with structural inits
    Phase 5: Final LP + gradient polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent = parent / parent.sum()
            n_parent = len(parent)
            print(f"Parent loaded: n={n_parent}, score={obj_value(parent):.6f}")
    except Exception:
        n_parent = 768

    n = n_parent if parent is not None else 768

    def update_best(a, label=""):
        nonlocal best_seq, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  {label}: {score:.8f} *NEW BEST*")
            return True
        else:
            if np.isfinite(score):
                print(f"  {label}: {score:.8f}")
            return False

    # =========================================================
    # Phase 1: Quick Nesterov scan (multiple inits, short time each)
    # =========================================================
    print("=== Phase 1: Nesterov Scan ===")
    t_start = time.monotonic()
    phase1_budget = min(250.0, (deadline - time.monotonic()) * 0.2)

    inits = generate_structural_inits(n, parent, rng)
    # Prioritize parent
    priority = ['parent', 'parent_interp', 'asym', 'ramp_inc', 'ramp_dec',
                'parabola', 'sine', 'uniform', 'exp_decay', 'bimodal', 'cosine', 'random']
    init_names = [nm for nm in priority if nm in inits]
    init_names = init_names[:5]  # Limit to 5 best inits

    t_per = max(15.0, phase1_budget / max(1, len(init_names)))
    candidates = []

    for name in init_names:
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 500)
        if t_alloc < 5:
            break
        a0 = inits[name]
        a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc, k_peaks=5, lr_init=0.015)
        update_best(a_opt, f"Nest[{name}]")
        candidates.append((obj, name, a_opt.copy()))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP Frank-Wolfe on best candidate
    # =========================================================
    print("=== Phase 2: LP Frank-Wolfe ===")
    t_start2 = time.monotonic()
    remaining = deadline - time.monotonic() - 400.0
    phase2_budget = min(500.0, max(100.0, remaining * 0.5))

    if best_seq is not None:
        t_alloc = phase2_budget
        print(f"  LP on best (n={n}), budget={t_alloc:.0f}s")
        a_opt, obj = lp_frank_wolfe(best_seq, time_limit=t_alloc, verbose=True)
        update_best(a_opt, "LP-FW")

    # Also try LP on second-best if time permits
    if len(candidates) > 1 and time.monotonic() < deadline - 300:
        t_alloc2 = min(100.0, (deadline - time.monotonic()) - 250)
        if t_alloc2 > 20:
            print(f"  LP on 2nd best, budget={t_alloc2:.0f}s")
            a_opt2, obj2 = lp_frank_wolfe(candidates[1][2], time_limit=t_alloc2)
            update_best(a_opt2, "LP-FW-2nd")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Coordinate descent polishing
    # =========================================================
    print("=== Phase 3: Coordinate Polishing ===")
    t_start3 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 200.0
        t_cd = min(150.0, max(30.0, remaining * 0.3))
        a_opt, obj = coordinate_polish(best_seq, time_limit=t_cd, verbose=True)
        update_best(a_opt, "CoordDesc")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Multi-restart with perturbations
    # =========================================================
    print("=== Phase 4: Multi-Restart ===")
    t_start4 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150.0
        n_restarts = max(1, int(remaining / 40))
        n_restarts = min(n_restarts, 8)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(35.0, (deadline - time.monotonic()) - 100)
            if t_alloc < 10:
                break

            scale = 0.002 + 0.002 * ri
            noise = rng.standard_normal(n) * scale
            a0 = project_simplex(best_seq + noise, 1.0)

            # Alternate between gradient and LP
            if ri % 2 == 0:
                a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc, k_peaks=3, lr_init=0.008)
            else:
                a_opt, obj = lp_frank_wolfe(a0, time_limit=t_alloc)
            update_best(a_opt, f"Restart[{ri}]")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final polish (short LP + gradient)
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    t_start5 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            # Short gradient run
            t_g = min(30.0, remaining * 0.5)
            if t_g > 5:
                a_opt, obj = nesterov_optimize(best_seq, time_limit=t_g, k_peaks=3, lr_init=0.003)
                update_best(a_opt, "FinalGrad")

            # Short LP run
            remaining = deadline - time.monotonic() - 10.0
            if remaining > 5:
                a_opt, obj = lp_frank_wolfe(best_seq, time_limit=remaining, max_iter=50)
                update_best(a_opt, "FinalLP")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
```
