import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_objective_grad(f, beta):
    """Compute LSE-smoothed max of conv(f,f) and its gradient w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    smooth_val = bc_max / beta + np.log(exp_bc.sum()) / beta

    # Gradient: d(soft_max)/d(f[i]) = sum_k w[k] * d(conv[k])/d(f[i])
    # d(conv[k])/d(f[i]) = 2*f[k-i] - f[i]*I(k==2i)
    # Computed via cross-correlation of w with f (reversed)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_wf = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * conv_wf[n - 1:n - 1 + n]

    for i in range(n):

        if 0 <= k < 2 * n - 1:

def subgradient_polish(f0, time_limit, top_k=20, lr_init=0.003, momentum=0.95):
    """Subgradient descent using top-k weighted subgradients."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    v = np.zeros(n)
    cur_lr = lr_init
    no_improve = 0
    temp = 0.0001

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        # Top-k positions
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k_pos = int(peaks[idx])
            js = max(0, k_pos - n + 1)
            je = min(n - 1, k_pos)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += 2.0 * w[idx] * f[k_pos - idx_arr]

        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)
        obj = fast_convolve(f).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= 0.95
            if no_improve > 200:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0003)
                no_improve = 0

        temp *= 1.003

    return best_f, best_obj

def generate_analytical_shapes(n, seed=0):

        g = np.exp(-0.5 * ((np.arange(n) - center) / (sigma * n / 2)) ** 2)
        shapes.append(g / g.sum())

    # Tukey (tapered)
    for r in [0.3, 0.5, 0.7]:
        n_r = int(r * n)
        tukey = np.ones(n)
        taper = n_r
        if taper > 0:
            for i in range(taper):
                val = 0.5 * (1 + np.cos(np.pi * (i / taper - 0.5)))
                tukey[i] = val
                tukey[n - 1 - i] = val
        tukey = np.maximum(tukey, 0)
        if tukey.sum() > 0:
            shapes.append(tukey / tukey.sum())

    # Random smooth
    rng = np.random.default_rng(seed)
    r = rng.standard_normal(n)
    # Smooth it
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    r_smooth = np.convolve(r, kernel, 'same')
    r_smooth = np.maximum(r_smooth, 0)
    if r_smooth.sum() > 0:
        shapes.append(r_smooth / r_smooth.sum())

    return shapes

                                        top_k=max(10, n_best // 10),
                                        lr_init=0.002, momentum=0.95)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 200)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s for coord descent")
        a_opt, obj = coord_descent_polish(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: Second LSE round with higher beta ===
    print(f"\n=== Phase 4: High-Beta LSE Polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(250, deadline - time.monotonic() - 100)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s for high-beta LSE")
        a_opt, obj = lse_optimize(current_best, time_limit=t_alloc,
                                  beta_start=200.0, beta_end=15000.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Multi-restart with different shapes ===
    print(f"\n=== Phase 5: Multi-Restart ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    t_remaining = deadline - time.monotonic() - 30
    n_restarts = max(1, min(4, int(t_remaining // 60)))
    time_per_restart = min(60, (t_remaining - 10) / max(1, n_restarts))

    for r in range(n_restarts):
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 30:
            break

        # Generate a perturbed starting point
        rng_r = np.random.default_rng(seed + 100 + r)
        if r == 0:
            # Start from analytical shape
            shapes = generate_analytical_shapes(n_best, seed=seed + r)
            start = shapes[r % len(shapes)].copy()
        elif r == 1:
            # Interpolate between parent and uniform
            alpha = 0.3 + 0.2 * r
            start = (1 - alpha) * current_best + alpha * np.ones(n_best) / n_best
            start = np.maximum(start, 0)
            start = start / start.sum()
        else:
            # Random smooth perturbation
            noise = rng_r.standard_normal(n_best)
            kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
            noise_smooth = np.convolve(noise, kernel, 'same')
            start = np.maximum(current_best + noise_smooth * 0.01, 0)
            start = start / start.sum()

        t1 = time_per_restart * 0.7
        t2 = time_per_restart * 0.3
        a_opt, obj = lse_optimize(start, time_limit=t1,
                                  beta_start=30.0, beta_end=5000.0)
        try_sequence(a_opt, best_tracker)

        current_r = best_tracker[1] if best_tracker[1] is not None else a_opt
        if len(current_r) == n_best and t2 > 15:
            a_opt2, obj2 = subgradient_polish(current_r, time_limit=t2,
                                              top_k=max(8, n_best // 12),
                                              lr_init=0.0015, momentum=0.96)
            try_sequence(a_opt2, best_tracker)

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    alt_lengths = []
    for nl in [384, 512, 576, 448, 320]:
        if nl != n_cur:
            alt_lengths.append(nl)

    for n_alt in alt_lengths[:2]:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 60:
            break

        # Interpolate
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 0:
            f_interp = f_interp / f_interp.sum()

        t1 = min(40, t_remaining // 3)
        t2 = min(30, t_remaining // 3)

        if t1 > 15:
            a_opt, obj = lse_optimize(f_interp, time_limit=t1,
                                      beta_start=40.0, beta_end=5000.0)
            try_sequence(a_opt, best_tracker)

        current_alt = best_tracker[1] if best_tracker[1] is not None else f_interp
        if len(current_alt) == n_alt and t2 > 10:
            a_opt2, obj2 = subgradient_polish(current_alt, time_limit=t2,
                                              top_k=max(8, n_alt // 10),
                                              lr_init=0.002, momentum=0.95)
            try_sequence(a_opt2, best_tracker)

    # === Phase 7: Final coordinate descent on best ===
    print(f"\n=== Phase 7: Final Coord Descent ===")
    if best_tracker[1] is not None:
        current_best = best_tracker[1]
        t_remaining = deadline - time.monotonic() - 10
        if t_remaining > 20:
            a_opt, obj = coord_descent_polish(current_best, time_limit=min(t_remaining, 60))
            try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
