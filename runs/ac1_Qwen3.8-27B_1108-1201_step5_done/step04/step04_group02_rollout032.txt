"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Structured initializations (ramp, staircase, power-law)
2. Frank-Wolfe with LP oracle for global optimization
3. Subgradient descent with momentum for local polishing
4. Coordinate descent for final refinement
5. Multi-length exploration

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def fw_optimize(f0, time_limit, verbose=True):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Try to solve LP
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.5
            if m_exp > 16:
                m_exp = 1.0
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        # Coarse
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = project_simplex((1.0 - alpha) * f + alpha * g)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = project_simplex((1.0 - mid) * f + mid * g)
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose:
                    print(f"    FW it{it}: obj={M:.8f}, alpha={best_alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.2, 8.0)

        if no_improve > 40:
            # Perturb and restart
            rng = np.random.default_rng(hash((it, n)) % 2**31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj

def compute_subgradient(f, conv, top_k=128, temp=0.001):
    """Subgradient of max(conv) with softmax weighting."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg

def sg_polish(f0, time_limit, lr=0.001, mom=0.95, top_k=100, temp=0.0001):
    """Fine subgradient polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        sg = compute_subgradient(f, conv, top_k=top_k, temp=temp)
        v = mom * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp *= 1.001
        lr *= 0.9999

        if no_improve > 300:
            rng = np.random.default_rng(it + 500)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj

def coordinate_descent(f0, n_iter=5000, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    n = len(f)
    conv = fast_convolve(f)
    obj = conv.max()
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - t0 > 60:  # 60 sec limit per CD phase
            break

        # Pick a random coordinate
        idx = np.random.randint(n)
        old_val = f[idx]

        # The contribution of f[idx] to conv[k] is f[idx] * f[k-idx] + f[idx] * f[idx-k]
        # (with appropriate bounds). To minimize max conv while keeping sum constant,
        # we can try adjusting f[idx] and redistributing to another coordinate.

        # Simple approach: try a few values
        best_v = old_val
        best_o = obj

        for scale in [0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 0.0]:
            new_val = max(0, old_val * scale)
            # Redistribute the difference to a random other coordinate
            diff = old_val - new_val
            if abs(diff) < 1e-15:
                continue
            jdx = np.random.randint(n)
            if jdx == idx:
                continue
            f[jdx] = max(0, f[jdx] + diff)
            f[idx] = new_val

            # Renormalize
            s = f.sum()
            if s < 1e-15:
                f[idx] = old_val
                f[jdx] = f[jdx] - diff
                continue
            f = f / s

            new_conv = fast_convolve(f)
            new_obj = new_conv.max()

            if new_obj < best_o - 1e-15:
                best_o = new_obj
                best_v = new_val
                # Keep this change
            else:
                # Revert
                f[jdx] = f[jdx] - diff * (f.sum() / s)  # This is getting messy
                # Let me just revert properly
                break

        # Actually, let me simplify: just try reducing the largest element
        # and adding to a smaller one

    return f, obj

def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Ramp (best known structured init, gives ~4/3)

    # Reverse ramp

    # Quadratic ramp
    q = np.arange(1, n + 1, dtype=np.float64) ** 1.5
    inits.append(q / q.sum())

    # Cube root
    cr = np.arange(1, n + 1, dtype=np.float64) ** (1.0/3.0)
    inits.append(cr / cr.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (concentrated in middle)
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())

    # Exponential
    e = np.exp(x / (n / 4.0))
    inits.append(e / e.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Staircase
    step = max(1, n // 8)
    st = np.repeat(np.arange(1, n // step + 2)[:n // step], step)[:n]
    inits.append(st.astype(np.float64) / st.astype(np.float64).sum())

    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Structured initializations (ramp gives ~4/3, much better than uniform's 2)
2. Frank-Wolfe with LP oracle for global optimization
3. Subgradient descent with momentum for local polishing
4. Away-step FW variant and adaptive LP relaxation
5. Multi-length exploration with time-adaptive allocation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Build the (2n-1) x n constraint matrix for the LP oracle."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = np.arange(n + 1)[1:n + 1]  # placeholder, will scale
    return A


def solve_lp(f, M, time_limit=20):
    """Solve LP: max 1'g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'mip_rel_gap': 0.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=True, use_away=True):
    """Frank-Wolfe with LP oracle and optional away-step."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    lp_time = 20

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        # Solve LP oracle
        g = solve_lp(f, M * m_exp, time_limit=lp_time)
        if g is None:
            m_exp *= 1.5
            if m_exp > 32:
                m_exp = 1.0
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 2.0)
            continue

        # Standard FW line search (towards g)
        best_alpha = 0.0
        best_val = M
        best_type = 'towards'

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            ft = project_simplex((1.0 - alpha) * f + alpha * g)
            val = fast_convolve(ft).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha
                best_type = 'towards'

        # Fine search
        if best_alpha > 0 and best_type == 'towards':
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                if hi - lo < 1e-7:
                    break
                mid = (lo + hi) / 2.0
                ft = project_simplex((1.0 - mid) * f + mid * g)
                val = fast_convolve(ft).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        # Away-step FW
        if use_away:
            # Find the support element farthest from g
            support = np.where(f > 1e-8)[0]
            if len(support) > 1:
                # Pick the element in support that is most positive in f dot (g - f)
                # Actually, the away vertex is the one in support with max f_i / (something)
                # Simpler: pick the support element with largest f value
                v_idx = support[np.argmax(f[support])]
                v = np.zeros(n)
                v[v_idx] = 1.0

                for beta in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    ft = f - beta * v
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val - 1e-15:
                        best_val = val
                        best_alpha = beta
                        best_type = 'away'

        # Composite step: towards and away simultaneously
        if use_away and best_type == 'towards':
            support = np.where(f > 1e-8)[0]
            if len(support) > 1:
                v_idx = support[np.argmax(f[support])]
                v = np.zeros(n)
                v[v_idx] = 1.0
                for alpha in [0.1, 0.2, 0.3, 0.5]:
                    for beta in [0.05, 0.1, 0.2]:
                        ft = (1.0 - alpha - beta) * f + alpha * g + beta * v
                        ft = np.maximum(ft, 0.0)
                        st = ft.sum()
                        if st < 1e-15:
                            continue
                        ft = ft / st
                        val = fast_convolve(ft).max()
                        if val < best_val - 1e-15:
                            best_val = val
                            best_alpha = alpha
                            best_type = f'composite_{v_idx}'

        # Apply the best step
        if best_val < M - 1e-14:
            if best_type == 'towards':
                f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            elif best_type == 'away':
                v = np.zeros(n)
                support = np.where(f > 1e-8)[0]
                if len(support) > 0:
                    v_idx = support[np.argmax(f[support])]
                    v[v_idx] = 1.0
                f = f - best_alpha * v
                f = np.maximum(f, 0.0)
                f = f / f.sum()
            else:
                # Composite
                v = np.zeros(n)
                support = np.where(f > 1e-8)[0]
                if len(support) > 0:
                    v_idx = support[np.argmax(f[support])]
                    v[v_idx] = 1.0
                alpha_t = best_alpha
                beta_t = 0.1
                f = (1.0 - alpha_t - beta_t) * f + alpha_t * g + beta_t * v
                f = np.maximum(f, 0.0)
                f = f / f.sum()

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            lp_time = max(5, lp_time * 0.99)

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    FW it{it}: obj={M:.8f}, t={elapsed:.0f}s")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 8.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + n)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            lp_time = 20

    return best_f, best_obj


def compute_subgradient(f, conv, top_k=128, temp=0.001):
    """Subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg


def sg_polish(f0, time_limit, lr=0.001, mom=0.95, top_k=100, temp=0.0001,
              verbose=False):
    """Subgradient descent with momentum for fine polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= 1.002
        cur_lr *= 0.99995

        if no_improve > 400:
            rng = np.random.default_rng(it * 31 + n)
            noise_scale = max(0.001, 0.005 * (1.0 - it / 5000.0))
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            conv = fast_convolve(f)
            cur_lr = lr * 0.5
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}")

    return best_f, best_obj


def generate_initializations(n, rng):
    """Generate diverse structured initializations. Ramp is key (~4/3)."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Ramp (best structured init, asymptotic objective ~4/3)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())

    # Power-law ramps
    for p in [0.5, 1.5, 2.0, 0.7]:
        pw = np.arange(1, n + 1, dtype=np.float64) ** p
        inits.append(pw / pw.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    c = np.cos(np.pi * (x / max(1, n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())

    # Exponential
    e = np.exp(x / max(1, n // 4))
    inits.append(e / e.sum())

    # Staircase
    step = max(1, n // 10)
    n_steps = (n + step - 1) // step
    st = np.repeat(np.arange(1, n_steps + 1, dtype=np.float64), step)[:n]
    inits.append(st / st.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    return inits
