import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_exact(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)_k = sum_j f[j] * g[k-j] for valid j
    # A_ub[k, j] = f[j] if 0 <= k-j <= n-1
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

def smoothed_gradient_descent(f0, time_limit, T_init=0.002, T_min=0.00005, verbose=False):

    if s < 1e-15:

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.005
    momentum_buf = np.zeros(n)
    beta = 0.9  # momentum

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = w / ws

        # Gradient: grad[i] = 2 * sum_{k} p[k] * a[k-i] (cross-correlation)
        fft_size = 1 << (3 * n).bit_length()
        A = np.fft.rfft(f, fft_size)
        P = np.fft.rfft(p, fft_size)
        cross = np.fft.irfft(np.conj(A) * P, fft_size)
        grad = 2.0 * cross[:n]

        # Project gradient to tangent space
        grad = grad - np.mean(grad)

        # Momentum
        momentum_buf = beta * momentum_buf + grad
        f_new = project_simplex(f - lr * momentum_buf)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

            best_f = f_new.copy()

        else:
            no_improve += 1
            if no_improve % 50 == 0:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_exact(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0.
    Always feasible since g=f satisfies the constraints.
    """
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def subgrad_direction(f, conv, top_k=20, temp=0.001):
    """Compute a weighted subgradient direction focusing on top peaks."""
    n = len(f)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg

def smoothed_grad_descent(f0, time_limit, T_init=0.002, T_min=0.00003, verbose=False):
    """Smoothed gradient descent with momentum on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.004
    momentum_buf = np.zeros(n)
    beta = 0.85

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = w / ws

        # Cross-correlation for gradient
        fft_size = 1 << (3 * n).bit_length()
        A = np.fft.rfft(f, fft_size)
        P = np.fft.rfft(p, fft_size)
        cross = np.fft.irfft(np.conj(A) * P, fft_size)
        grad = 2.0 * cross[:n]
        grad = grad - np.mean(grad)

        momentum_buf = beta * momentum_buf + grad
        f_new = project_simplex(f - lr * momentum_buf)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            if no_improve % 80 == 0:
                lr = max(lr * 0.82, 0.0003)

        T = max(T * 0.9975, T_min)

        if no_improve > 300:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            lr = max(lr * 1.4, 0.003)
            momentum_buf = np.zeros(n)
            no_improve = 0

    return best_f, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated (U-shape)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Cosine bump
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append((c1 + 0.1) / (c1 + 0.1).sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.12 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.12 * n)) ** 2)
    inits.append((h1 + h2 + 0.03) / (h1 + h2 + 0.03).sum())

    # Mirror-symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.4)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_obj = 2 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, obj={p_obj:.6f}")
    except Exception:
        pass

    # === Phase 1: Exploration (200s) ===
    print("=== Phase 1: Exploration ===")
    t1 = time.monotonic()
    p1_budget = 200.0

    base_n = len(parent) if parent is not None else 384
    lengths = sorted(set([
        base_n, max(256, base_n - 32), base_n + 16,
        base_n + 32, base_n + 64, 400, 448, 480, 512
    ]))
    lengths = [l for l in lengths if 200 <= l <= 560]
    # Deduplicate preserving order
    seen = set()
    ul = []
    for l in lengths:
        if l not in seen:
            seen.add(l)
            ul.append(l)
    lengths = ul

    # Prioritize: parent length first, then nearby
    if base_n in lengths:
        lengths.remove(base_n)
    lengths = [base_n] + lengths

    for n in lengths:
        rem = p1_budget - (time.monotonic() - t1)
        if rem < 25:
            break
        t_alloc = min(45.0, rem - 10.0)
        if t_alloc < 15:
            break

        inits = generate_inits(n, parent, rng)
        best_init = None
        best_init_obj = float('inf')
        for fi in inits[:5]:
            obj = fast_convolve(fi).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = fi

        if best_init is not None:
            print(f"  n={n}, init_obj={best_init_obj:.8f}")
            a_opt, _ = fw_lp_optimize(best_init, time_limit=t_alloc)
            try_seq(a_opt, f"expl_n{n}")

    print(f"  Phase 1 done, best={best_score:.6f}")

    # === Phase 2: Deep LP-FW (1100s) ===
    print(f"\n=== Phase 2: Deep LP-FW ===")
    t2 = time.monotonic()
    p2_budget = 1100.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        n_cycles = 10
        for cycle in range(n_cycles):
            rem = p2_budget - (time.monotonic() - t2)
            if rem < 50:
                break

            t_fw = min(110.0, rem - 30)
            if t_fw < 25:
                break

            print(f"  Cycle {cycle}: FW (t={t_fw:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_fw, verbose=True)
            try_seq(a_opt, f"fw_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

            # SGD between FW cycles
            rem = p2_budget - (time.monotonic() - t2)
            if rem > 40:
                t_sgd = min(70.0, rem - 15)
                if t_sgd > 20:
                    conv_m = fast_convolve(f_work).max()
                    T_start = max(0.0008, 0.003 * conv_m)
                    print(f"  Cycle {cycle}: SGD (t={t_sgd:.0f}s, T={T_start:.5f})")
                    a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_sgd, T_init=T_start)
                    try_seq(a_opt, f"sgd_{cycle}")
                    if best_seq is not None:
                        f_work = best_seq / best_seq.sum()

    print(f"  Phase 2 done, best={best_score:.6f}")

    # === Phase 3: Fine Polish (350s) ===
    print(f"\n=== Phase 3: Fine Polish ===")
    t3 = time.monotonic()
    p3_budget = 350.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        configs = [
            {"lr": 0.0006, "top_k_frac": 0.04, "temp": 0.00002},
            {"lr": 0.0003, "top_k_frac": 0.03, "temp": 0.00001},
            {"lr": 0.00015, "top_k_frac": 0.02, "temp": 0.000005},
            {"lr": 0.00008, "top_k_frac": 0.02, "temp": 0.000002},
        ]

        for i, cfg in enumerate(configs):
            rem = p3_budget - (time.monotonic() - t3)
            if rem < 25:
                break
            t_alloc = min(70.0, rem - 10.0)
            if t_alloc < 15:
                break
            print(f"  SG {i} (t={t_alloc:.0f}s)")
            a_opt, _ = sg_polish(f_work, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"polish_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Fine smoothed gradient
        rem = p3_budget - (time.monotonic() - t3)
        if rem > 30:
            T_init = max(0.0002, 0.001 * fast_convolve(f_work).max())
            t_alloc = min(80.0, rem - 10.0)
            print(f"  Fine SGD (t={t_alloc:.0f}s, T={T_init:.6f})")
            a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_alloc,
                                              T_init=T_init, T_min=0.00001)
            try_seq(a_opt, "fine_sgd")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Final FW
        rem = p3_budget - (time.monotonic() - t3)
        if rem > 25:
            t_alloc = min(50.0, rem - 10.0)
            print(f"  Final FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

    print(f"  Phase 3 done, best={best_score:.6f}")

    # === Phase 4: Final tricks ===
    print(f"\n=== Phase 4: Final ===")

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(f_work)

        # Mirror symmetrization
        mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
        mirrored = project_simplex(mirrored)
        if np.max(np.abs(mirrored - f_work)) > 1e-5:
            if time.monotonic() < deadline - 20:
                a_opt, _ = sg_polish(mirrored, time_limit=12, lr=0.0002,
                                      top_k_frac=0.02, temp=0.00001)
                try_seq(a_opt, "mirror")

        # Tiny perturbations
        for i in range(5):
            if time.monotonic() > deadline - 10:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(f_work + noise)
            if time.monotonic() < deadline - 8:
                a_opt, _ = sg_polish(f_pert, time_limit=6, lr=0.00015,
                                      top_k_frac=0.02, temp=0.000005)
                try_seq(a_opt, f"pert_{i}")

        # Last LP-FW
        if time.monotonic() < deadline - 35:
            a_opt, _ = fw_lp_optimize(f_work, time_limit=30)
            try_seq(a_opt, "last_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
