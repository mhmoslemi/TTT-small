"""
Multi-phase optimizer using L-BFGS with softmax parameterization and analytical
gradients, combined with LP-Frank-Wolfe and subgradient polishing, to minimize
2*n*max(conv(a,a))/(sum(a))^2. Key innovation: second-order optimization via
L-BFGS on the softmax manifold with FFT-based analytical gradients.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(theta):
    """Compute softmax efficiently."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    return exp_theta / exp_theta.sum()


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbfgs_objective_and_grad(theta, T):
    """
    Compute smoothed max of softmax(theta)*softmax(theta) and its gradient.
    Uses analytical gradient: dL/dtheta_j = 2*p_j*(s_j - <w,c>)
    where s_j = sum_k w_k * p_{k-j}, c = p*p, w = softmax(c/T).
    """
    p = softmax(theta)
    n = len(p)
    conv = fast_convolve(p)
    
    # Softmax weights over convolution
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    # Objective: T * log(sum(exp(conv/T))) = conv.max() + T * log(sum(exp((conv-conv.max())/T)))
    obj = conv.max() + T * np.log(np.exp(scaled).sum())
    
    # Gradient computation
    # s_j = sum_k w_k * p_{k-j} = cross_correlation(w, p)[j]
    # cross_corr[j] = sum_m w[m+j] * p[m]
    # FFT: IFFT(conj(FFT(w)) * FFT(p)) gives cross-correlation
    fft_size = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(W) * P, fft_size)
    s = cross[:n]
    
    w_conv = np.dot(w, conv)
    grad = 2.0 * p * (s - w_conv)
    
    return obj, grad


def lbfgs_optimize(f0, T_start, T_end, time_limit, n_max_iter=20000, verbose=False):
    """
    L-BFGS optimization on softmax manifold with temperature annealing.
    Returns (best_p, best_obj).
    """
    n = len(f0)
    p = np.maximum(f0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    
    # Initialize theta from p
    theta = np.log(p + 1e-30)
    
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    
    t0 = time.monotonic()
    n_temps = 8
    temps = np.geomspace(T_start, max(T_end, T_start * 1e-3), n_temps)
    
    for T in temps:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        def fun_wrapped(theta_cur):
            val, _ = lbfgs_objective_and_grad(theta_cur, T)
            return val
        
        def jac_wrapped(theta_cur):
            _, g = lbfgs_objective_and_grad(theta_cur, T)
            return g
        
        try:
            res = minimize(fun_wrapped, theta, method='L-BFGS-B', jac=jac_wrapped,
                          options={'maxiter': n_max_iter, 'maxfun': 50000,
                                   'ftol': 1e-12, 'gtol': 1e-8, 'iprint': 0})
            
            if res.success or res.fun < best_obj:
                p_new = softmax(res.x)
                conv_new = fast_convolve(p_new)
                obj_new = conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_p = p_new.copy()
                    if verbose:
                        print(f"      L-BFGS T={T:.6f}: obj={best_obj:.10f}")
                theta = res.x
        except Exception:
            pass
        
        # Also try from current p (in case L-BFGS diverged)
        p_cur = softmax(theta)
        conv_cur = fast_convolve(p_cur)
        if conv_cur.max() < best_obj:
            best_obj = conv_cur.max()
            best_p = p_cur.copy()
    
    return best_p, best_obj


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0, sum(g)=1."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] <= M for k=0,...,2n-2
    # (f*g)[k] = sum_j f[k-j] * g[j] (with f out of range = 0)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A_ub[k, j] = f[k-j] if 0 <= k-j < n
        start = max(0, j - (n - 1))
        end = min(2 * n - 2, j + n - 1)
        for k in range(start, end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    
    # Add equality constraint: sum(g) = 1
    A_eq = np.ones((1, n))
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

def sg_polish(f0, time_limit, lr=0.001, top_k_frac=0.05, temp=0.00003):
    """Subgradient polish with weighted top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(5, int(n * top_k_frac))
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # d(conv[k])/d(f[j]) = 2*f[k-j] (with bounds)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 100 == 0:
            lr *= 0.995
            temp *= 1.005
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = 0.0008
            no_improve = 0
    
    return best_f, best_obj


def lbfgs_anneal_restart(parent, n, T_start, T_end, time_limit, rng, n_restarts=5):
    """
    Multiple L-BFGS restarts with different initializations and temperatures.
    Returns best (p, obj).
    """
    best_p = None
    best_obj = float('inf')
    
    for r in range(n_restarts):
        rem = time_limit - 0  # We'll track internally
        if best_p is None:
            p_init = parent.copy()
        else:
            # Perturb from best
            noise_scale = 0.002 * (1 + r * 0.5)
            p_init = project_simplex(best_p + rng.standard_normal(n) * noise_scale)

def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    
    x = np.arange(n, dtype=np.float64)
    
    # Edge-concentrated (U-shape)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.7
    inits.append(edge_w / edge_w.sum())
    
    # Cosine with offset
    c1 = np.abs(np.cos(np.pi * x / n)) ** 0.5 + 0.02
    inits.append(c1 / c1.sum())
    
    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.15 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.15 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Mirror-symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.3)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[n // 2] = half[half_n // 2]
    inits.append(sym / sym.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

            if rem < 30:
                break
            
            # L-BFGS step
            conv_w = fast_convolve(f_work)
            T0 = 0.005 * conv_w.max()
            T_end = 0.0003 * conv_w.max()
            t_lb = min(80.0, rem / 2.5)
            print(f"  Cycle {cycle}: L-BFGS (t={t_lb:.0f}s)")
            p_lb, _ = lbfgs_optimize(f_work, T0, T_end, t_lb)
            try_seq(p_lb, f"p2_lb_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
            
            # L-BFGS from perturbed best
            rem = phase2_budget - (time.monotonic() - t2_start)
            if rem < 30:
                break
            f_pert = project_simplex(f_work + rng.standard_normal(n_best) * 0.003)
            t_lb2 = min(50.0, rem / 2.0)
            p_lb2, _ = lbfgs_optimize(f_pert, T0, T_end, t_lb2)
            try_seq(p_lb2, f"p2_lb_pert_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
    
    print(f"  Phase 2 done. Best: {best_score:.6f}")
    
    # === Phase 3: Fine polishing (400s) ===
    print(f"\n=== Phase 3: Fine Polishing ===")
    t3_start = time.monotonic()
    phase3_budget = 400.0
    
    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()
        
        # Subgradient polish with decreasing parameters
        configs = [
            {"lr": 0.0008, "top_k_frac": 0.04, "temp": 0.00003},
            {"lr": 0.0004, "top_k_frac": 0.03, "temp": 0.00002},
            {"lr": 0.0002, "top_k_frac": 0.02, "temp": 0.00001},
        ]
        
        for i, cfg in enumerate(configs):
            rem = phase3_budget - (time.monotonic() - t3_start)
            if rem < 30:
                break
            t_alloc = min(80.0, rem - 20)
            print(f"  SG round {i} (t={t_alloc:.0f}s)")
            p_sg, _ = sg_polish(f_work, time_limit=t_alloc, **cfg)
            try_seq(p_sg, f"p3_sg_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
        
        # Very fine L-BFGS
        rem = phase3_budget - (time.monotonic() - t3_start)
        if rem > 40:
            conv_w = fast_convolve(f_work)
            T0 = 0.002 * conv_w.max()
            T_end = 0.0001 * conv_w.max()
            t_alloc = min(100.0, rem - 20)
            print(f"  Fine L-BFGS (t={t_alloc:.0f}s, T={T0:.6f}->{T_end:.8f})")
            p_fine, _ = lbfgs_optimize(f_work, T0, T_end, t_alloc)
            try_seq(p_fine, "p3_fine_lbfgs")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()
        
        # One more LP-FW
        rem = phase3_budget - (time.monotonic() - t3_start)
        if rem > 30:
            t_alloc = min(60.0, rem - 15)
            print(f"  Final LP-FW (t={t_alloc:.0f}s)")
            p_final, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
            try_seq(p_final, "p3_final_fw")
    
    print(f"  Phase 3 done. Best: {best_score:.6f}")
    
    # === Phase 4: Multi-length exploration (400s) ===
    print(f"\n=== Phase 4: Multi-length ===")
    t4_start = time.monotonic()
    phase4_budget = 400.0
    
    if best_seq is not None:
        n_best = len(best_seq)
        f_best = best_seq / best_seq.sum()
        
        # Try nearby lengths
        for delta in [8, -8, 16, -16, 4, -4, 32, -32]:
            rem = phase4_budget - (time.monotonic() - t4_start)
            if rem < 40:
                break
            new_n = n_best + delta
            if new_n < 128 or new_n > 600:
                continue
            
            # Resample
            px = np.interp(np.linspace(0, 1, new_n),
                           np.linspace(0, 1, n_best), f_best)
            px = project_simplex(px)
            
            # L-BFGS
            conv_px = fast_convolve(px)
            T0 = 0.008 * conv_px.max()
            T_end = 0.0004 * conv_px.max()
            t_alloc = min(60.0, rem - 30)
            p_opt, _ = lbfgs_optimize(px, T0, T_end, t_alloc)
            try_seq(p_opt, f"p4_n{new_n}_lbfgs")
            
            # Quick LP-FW
            rem = phase4_budget - (time.monotonic() - t4_start)
            if rem > 30:
                t_fw = min(20.0, rem - 20)
                p_fw, _ = fw_lp_optimize(p_opt, time_limit=t_fw)
                try_seq(p_fw, f"p4_n{new_n}_fw")
    
    # Final polish on the overall best
    rem = phase4_budget - (time.monotonic() - t4_start)
    if rem > 20 and best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(f_work)
        
        # Tiny L-BFGS
        conv_w = fast_convolve(f_work)
        T0 = 0.001 * conv_w.max()
        T_end = 0.00005 * conv_w.max()
        t_alloc = min(40.0, rem - 10)
        p_final, _ = lbfgs_optimize(f_work, T0, T_end, t_alloc)
        try_seq(p_final, "final_lbfgs")
        
        # Tiny SG
        rem = phase4_budget - (time.monotonic() - t4_start)
        if rem > 10:
            p_sg, _ = sg_polish(p_final, time_limit=min(10.0, rem - 5),
                                lr=0.0001, top_k_frac=0.02, temp=0.000005)
            try_seq(p_sg, "final_sg")
    
    print(f"  Phase 4 done. Best: {best_score:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
