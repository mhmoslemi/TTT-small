"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Diverse structured initializations (U-shaped, sine, parent)
2. Nesterov-accelerated log-sum-exp gradient descent with adaptive beta
3. Top-k subgradient method for the true max
4. Frank-Wolfe with LP oracle for structural refinement
5. Multi-length exploration
Key improvements: correct FFT gradient, better initialization, 
adaptive learning rate with backtracking, and structured restarts.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_weights(conv, beta):
    """Compute softmax weights for LSE smoothing."""
    bc = beta * conv
    bc_max = bc.max()
    # Protect against underflow
    exp_bc = np.exp(bc - bc_max)
    w_sum = exp_bc.sum()
    if w_sum < 1e-300:
        # All weights underflow; use argmax
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    return exp_bc / w_sum


def compute_grad_from_weights(w, a):
    """Compute gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (where valid).
    Uses the identity: grad[j] = 2 * (a * w_rev)[2n-2-j]
    """
    n = len(a)
    total_len = n + len(w) - 1  # = 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w[::-1], fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:total_len]
    # grad[j] = 2 * conv[2n-2-j] for j=0,...,n-1
    return 2.0 * conv[2*n-2 : 2*n-2-n : -1]


def subgrad_max(a, top_k=1):
    """Compute subgradient of max_k (a*a)_k using top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        # Get top-k indices
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        # Weight by softmax for smoothness
        pv = conv[peaks]
        scale = (pv.max() - pv.min()) * 50 if pv.max() > pv.min() else 1.0
        weights = np.exp(pv / scale)
        weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        if weights[idx] < 1e-12:
            continue
        # grad_k[j] = 2*a[k-j] if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * weights[idx] * a[k - j]
    
    return grad, conv


def lse_optimize(a0, time_limit, beta_start=30.0, beta_end=30000.0,
                 lr_factor=2.0, momentum_init=0.85, momentum_max=0.97):
    """Nesterov-style LSE smoothed optimization with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    beta = beta_start
    lr = lr_factor / np.sqrt(n)
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    prev_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        
        # Nesterov lookahead
        y = a + (momentum * (a - prev_a))
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        
        # Compute weights and gradient at y
        conv_y = fast_convolve(y)
        w = lse_weights(conv_y, beta)
        grad = compute_grad_from_weights(w, y)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Update velocity
        v = momentum * v - lr * grad
        
        # Step and project
        a_new = project_simplex(a + v)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        max_new = conv_new.max()
        
        if max_new < best_max - 1e-12:
            best_max = max_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        prev_a = a
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 10:
            lr *= 0.95
            momentum = max(0.85, momentum - 0.01)
        else:
            lr = min(lr * 1.001, lr_factor / np.sqrt(n))
        
        # Increase beta
        if beta < beta_end:
            beta *= 1.004
        
        # Periodic restart to escape local minima
        if no_improve > 200:
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr_factor / np.sqrt(n)
            momentum = momentum_init
            beta = max(beta_start, beta * 0.5)
            no_improve = 0
            # Small perturbation
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(a + rng.standard_normal(n) * 0.001)
            prev_a = a.copy()
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2 * n * best_max
            print(f"    LSE it={it}, score={score:.8f}, beta={beta:.0f}, "
                  f"lr={lr:.6f}, mom={momentum:.3f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_max


def subgrad_optimize(a0, time_limit, top_k=8, lr_init=0.05):
    """Subgradient method on the true max function."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        
        grad, conv = subgrad_max(a, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        a_new = project_simplex(a - lr * grad)
        
        conv_new = fast_convolve(a_new)
        max_new = conv_new.max()
        
        if max_new < best_max - 1e-12:
            best_max = max_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, lr_init)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.97
        
        a = a_new
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2 * n * best_max

    return best_a, best_max


def build_lp_matrix(f):
    """Build sparse constraint matrix A where A[k,j] = f[|k-j|]... 
    Actually A[k,j] = f[k-j] if 0<=k-j<n else 0.
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_step(f, M, A_mat=None):
    """Solve LP: max sum(g) s.t. A@g <= M*ones, g >= 0. Return g."""
    n = len(f)
    if A_mat is None:
        A_mat = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(a0, time_limit):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_max = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    m_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        A_mat = build_lp_matrix(a)
        g = fw_lp_step(a, M * m_scale, A_mat)
        
        if g is None:
            m_scale *= 1.15
            if m_scale > 5.0:
                m_scale = 1.0
            continue
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            m_scale = 1.0
            if M < best_max:
                best_max = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_scale = min(m_scale * 1.1, 5.0)
        
        if no_improve > 30:
            break
    
    return best_a, best_max


def make_initialization(n, method, rng):
    """Create an initial sequence of length n using the given method."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'inverse_sqrt':
        i = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / np.sqrt(i * (n + 1 - i))
        a /= a.sum()
        return a
    elif method == 'u_shaped':
        t = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + np.abs(t - 0.5) * 2)
        a /= a.sum()
        return a

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:

2. Nesterov-accelerated log-sum-exp gradient descent with adaptive beta and LR

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_weights(conv, beta):
    """Compute softmax weights for LSE smoothing (numerically stable)."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w_sum = exp_bc.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    return exp_bc / w_sum


def compute_grad_from_weights(w, a):
    """Compute gradient of LSE: grad[j] = 2*sum_k w[k]*a[k-j].
    Uses FFT: grad[j] = 2*(a * w_rev)[2n-2-j]
    """
    n = len(a)
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w[::-1], fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:total_len]
    return 2.0 * conv[2 * n - 2:2 * n - 2 - n:-1]


def subgrad_at_peaks(a, top_k=8, temperature=0.01):
    """Compute weighted subgradient using top-k peaks of convolution."""
    n = len(a)
    conv = fast_convolve(a)
    
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        if pv.max() > pv.min():
            scaled = (pv - pv.min()) / temperature
            weights = np.exp(scaled)
            weights /= weights.sum()
        else:
            weights = np.ones(len(peaks)) / len(peaks)
    
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        if weights[idx] < 1e-12:
            continue
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        j_arr = np.arange(js, je + 1)
        grad[j_arr] += 2.0 * weights[idx] * a[k_int - j_arr]
    
    return grad, conv


def lse_optimize(a0, time_limit, beta_start=30.0, beta_end=50000.0,

            y = a.copy()

def build_lp_matrix(f):
    """Build sparse A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    a = np.maximum(a0.astype(np.float64), 0.0)

        if gs < 1e-15:

            if no_improve > 20:

            m_scale = min(m_scale * 1.05, 3.0)

        # Line search

def _solve_lp(f, M, A_mat):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def block_optimize(a0, time_limit, n_blocks=16):
    """Optimize with block structure: a is piecewise constant with n_blocks segments."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize block boundaries evenly
    boundaries = np.linspace(0, n, n_blocks + 1).astype(int)
    block_widths = np.diff(boundaries)
    
    # Initial block values from a
    block_vals = np.array([a[boundaries[i]:boundaries[i+1]].mean() for i in range(n_blocks)])
    
    best_score = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.1
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        
        # Build a from blocks
        a_build = np.zeros(n)
        for i in range(n_blocks):
            a_build[boundaries[i]:boundaries[i+1]] = max(block_vals[i], 0)
        
        s = a_build.sum()
        if s < 1e-15:
            break
        a_build = a_build / s
        
        # Compute gradient w.r.t. block_vals
        conv = fast_convolve(a_build)
        max_k = np.argmax(conv)
        # Subgradient for block i: sum over positions in block i of 2*a[max_k - pos]
        bgrad = np.zeros(n_blocks)
        for i in range(n_blocks):
            pos = np.arange(boundaries[i], boundaries[i+1])
            valid = (max_k - pos >= 0) & (max_k - pos < n)
            bgrad[i] = 2.0 * np.sum(a_build[max_k - pos[valid]]) * (block_widths[i] / s)
        
        gnorm = np.linalg.norm(bgrad)
        if gnorm < 1e-15:
            break
        bgrad = bgrad / gnorm
        
        block_vals = np.maximum(block_vals - lr * bgrad, 0)
        
        # Rebuild and check
        a_build = np.zeros(n)
        for i in range(n_blocks):
            a_build[boundaries[i]:boundaries[i+1]] = block_vals[i]
        s = a_build.sum()
        if s > 1e-15:
            a_build = a_build / s
            conv = fast_convolve(a_build)
            if conv.max() < best_score:
                best_score = conv.max()
                best_a = a_build.copy()
        
        lr *= 0.999
    
    return best_a, best_score

    for idx, (score_est, n, a_c, method) in enumerate(top_candidates):
        t_remaining = deadline - time.monotonic() - 500
        if t_remaining < 100:
            break
        t_alloc = min(400, t_remaining // max(1, len(top_candidates) - idx))
        if t_alloc < 50:
            t_alloc = 50
        
        print(f"  Optimizing candidate {idx+1} (n={n}, {method}, est={score_est:.6f}), t={t_alloc:.0f}s")
        a_opt, obj = lse_optimize(a_c, time_limit=t_alloc,
                                  beta_start=20.0, beta_end=50000.0,
                                  lr_factor=3.0, momentum_init=0.88,
                                  momentum_max=0.97, restart_interval=250)
        try_eval(a_opt, best_tracker)
    
    # === Phase 3: Subgradient polishing ===
    print("\n=== Phase 3: Subgradient polishing ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 350
    if t_remaining > 50:
        t_alloc = min(200, t_remaining)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = subgrad_optimize(current_best, time_limit=t_alloc,
                                      top_k=16, lr_init=0.02, temperature=0.003)
        try_eval(a_opt, best_tracker)
    
    # === Phase 4: FW-LP refinement ===
    print("\n=== Phase 4: FW-LP refinement ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 200
    if t_remaining > 60:
        t_alloc = min(200, t_remaining)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc)
        try_eval(a_opt, best_tracker)
    
    # === Phase 5: Final high-beta LSE ===
    print("\n=== Phase 5: Final high-beta LSE ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 100
    if t_remaining > 50:
        t_alloc = min(150, t_remaining)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_optimize(current_best, time_limit=t_alloc,
                                  beta_start=200.0, beta_end=100000.0,
                                  lr_factor=1.5, momentum_init=0.80,
                                  momentum_max=0.92, restart_interval=150)
        try_eval(a_opt, best_tracker)
    
    # === Phase 6: Block optimization ===
    print("\n=== Phase 6: Block optimization ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 80
    if t_remaining > 40:
        t_alloc = min(80, t_remaining)
        for nb in [8, 16, 32]:
            if time.monotonic() > deadline - 60:
                break
            a_opt, obj = block_optimize(current_best, time_limit=t_alloc / 3, n_blocks=nb)
            try_eval(a_opt, best_tracker)
    
    # === Phase 7: Final subgradient + LSE combo ===
    print("\n=== Phase 7: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 40
    if t_remaining > 30:
        t1 = t_remaining // 2
        t2 = t_remaining - t1
        a_opt, _ = subgrad_optimize(current_best, time_limit=t1, top_k=20, lr_init=0.01)
        try_eval(a_opt, best_tracker)
        current_best = best_tracker[1]
        a_opt, _ = lse_optimize(current_best, time_limit=t2,
                                beta_start=500.0, beta_end=200000.0,
                                lr_factor=1.0, momentum_init=0.75)
        try_eval(a_opt, best_tracker)
    
    # === Final ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
