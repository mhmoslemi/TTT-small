import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def weighted_gradient(f, conv, top_k, temperature):
    """Compute weighted gradient using top-k peaks."""
    n = len(f)
    # Get top-k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        pv = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
    
    # Softmax weights
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)

    """
    f: array of length n (the sequence)
    peak_weights: array of length 2n-1 (weights on each lag)

    u = peak_weights[::-1]

    grad = 2.0 * c[n-1:2*n-1][::-1]

"""
Optimized solver for minimizing 2*n*max(autocorr(a))/(sum(a))^2 using
FFT-based weighted gradient descent on the simplex with adaptive temperature,
multi-start exploration, and careful time management.
Key improvement: correct O(n log n) gradient via FFT cross-correlation,
adaptive peak weighting, and multi-phase strategy.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(f, conv, top_k, temperature):
    """
    Compute weighted gradient of sum_k w_k * (f*f)[k] w.r.t. f.
    Uses only top_k peaks with softmax weights at given temperature.
    O(n log n) via FFT.
    
    Key insight: d/d f[m] of (f*f)[k] = 2*f[k-m] if 0<=k-m<n, else 0.
    So grad[m] = 2 * sum_k w[k] * f[k-m] * I(0<=k-m<n)
    This equals 2 * (f * flip(w))[2n-2-m] taking first n elements.
    """
    n = len(f)
    n_conv = 2 * n - 1
    
    if top_k >= n_conv:
        peaks = np.arange(n_conv)
        pv = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
    
    # Softmax weights
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Build peak weight vector of length 2n-1
    p = np.zeros(n_conv)
    p[peaks] = w
    
    # Compute grad[m] = 2 * sum_l f[l] * p[m+l] for m = 0,...,n-1
    # This is 2 * (f * flip(p))[2n-2-m], take first n
    u = p[::-1]
    fft_size = 1 << (n_conv + n - 1 - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    U = np.fft.rfft(u, fft_size)
    c = np.fft.irfft(F * U, fft_size)[:n_conv + n - 1]
    
    # grad[m] = 2 * c[2n-2-m] for m=0,...,n-1
    # indices: 2n-2, 2n-3, ..., n-1
    grad = 2.0 * c[n - 1:2 * n - 1][::-1]
    
    return grad

def mirror_symmetrize(f, alpha=0.5):
    """Blend with mirror-symmetric version."""
    n = len(f)
    mirrored = f[::-1]
    sym = alpha * f + (1 - alpha) * mirrored
    return project_simplex(sym)

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edge-concentrated (U-shape)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.7
    inits.append(edge_w / edge_w.sum())
    
    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    two_hump = h1 + h2
    inits.append(two_hump / two_hump.sum())
    
    # Cosine
    c = np.abs(np.cos(np.pi * x / (n - 1))) + 0.1
    inits.append(c / c.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.2))
    
    # Mirror-symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.3)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] = half[half_n - 1] + half[half_n]
    inits.append(sym / sym.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
