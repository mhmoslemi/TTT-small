   ```
   max  1^T g
   s.t. (a*g)_k <= M  for k in S

   ```

   If the optimal `sum(g) > 1` (since `sum(a) = 1`), then we can take the direction `d = g - a` and it should decrease the objective (to first order).

   - `d = g_norm - a` (this has `1^T d = 0` since both have sum 1)

   The step size `t` should be chosen to actually decrease the objective. We can use bisection:

   ```python
       n = len(a)

       A = np.fft.rfft(a, fft_size)

       return np.fft.irfft(A * G, fft_size)[:size]

       So C @ g gives the cross-convolution values at the active lags.
       """

       K = len(active_lags)
       C = np.zeros((K, n))
       for i, k in enumerate(active_lags):
           for j in range(n):
               kj = k - j
               if 0 <= kj < n:

       return C
   ```

```python
"""
Active-set LP optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Only constrain near-peak lags in the LP (not all 2n-1), yielding
much less constrained LPs that find better descent directions. Combined with
adaptive active-set management, bisection step sizing, and multi-scale restarts.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_constraint_matrix(a, active_lags):
    """Build C such that C @ g = cross-conv(a, g) at active lags."""
    n = len(a)
    K = len(active_lags)
    C = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            C[i, j] = a[k - j]
    return C


def solve_active_lp(a, M, active_lags):
    """
    Solve: max sum(g) s.t. cross_conv(a,g)_k <= M for k in active, g >= 0.
    Returns (g, sum_g, success).
    """
    n = len(a)
    C = build_constraint_matrix(a, active_lags)
    c = -np.ones(n)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            return g, g.sum(), True
    except Exception:
        pass
    return None, 0.0, False


def bisection_step(a, d, n, current_obj, max_iter=10):
    """Find best step t in (0,1] for a + t*d using bisection."""
    best_t = 0.0
    best_obj = current_obj
    lo, hi = 0.0, 1.0

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        if mid < 1e-10:
            break
        a_try = np.maximum(a + mid * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            hi = mid
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = mid
            lo = mid
        else:
            hi = mid

    if best_t > 0:
        a_new = np.maximum(a + best_t * d, 0.0)
        return normalize(a_new)
    return None


def active_set_optimize(a0, time_limit, top_k_init=10, verbose=False):
    """
    Main active-set LP optimizer with adaptive parameters.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    top_k = top_k_init
    eps_frac = 0.02  # active set threshold: conv >= M*(1-eps_frac)

    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # Find active set
        threshold = M * (1.0 - eps_frac)
        active = np.where(conv >= threshold)[0]

        # Limit active set size
        if len(active) > top_k:
            idx = np.argsort(conv[active])[-top_k:]
            active = active[idx]
        elif len(active) < 3:
            # If too few active, add next peaks
            all_idx = np.argsort(conv)[::-1][:top_k]
            active = all_idx

        # Solve LP
        g, sum_g, success = solve_active_lp(a, M, active)

        if not success or sum_g <= 1.0 + 1e-7:
            no_improve += 1
        else:
            # Compute direction
            g_norm = g / sum_g
            d = g_norm - a
            norm_d = np.linalg.norm(d)

            if norm_d < 1e-12:
                no_improve += 1
            else:
                # Bisection step sizing
                a_new = bisection_step(a, d, n, 2.0 * n * M, max_iter=12)

                if a_new is not None:
                    conv_new = fast_convolve(a_new)
                    M_new = conv_new.max()
                    obj_new = 2.0 * n * M_new

                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        no_improve = 0
                        perturb_count = 0
                    elif obj_new < 2.0 * n * M - 1e-13:
                        # Accept even if not new global best (local improvement)
                        no_improve = 0
                    else:
                        no_improve += 1

                    a = a_new if a_new is not None else a
                    conv = conv_new
                    M = M_new
                else:
                    no_improve += 1

        # Adaptive parameter updates
        if no_improve >= 30:
            # Perturb and adapt
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = normalize(best_a + noise)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1

            # Adapt active set parameters
            if perturb_count % 3 == 0:
                top_k = min(80, top_k + 5)
                eps_frac = min(0.15, eps_frac + 0.01)
            elif perturb_count % 3 == 1:
                top_k = max(5, top_k - 3)

            if perturb_count > 200:
                break

        if verbose and no_improve % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter, obj={best_obj:.8f}, t={elapsed:.1f}s, no_impr={no_improve}, K={top_k}")

    return best_a, best_obj


def gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.003):
    """Nesterov gradient descent with softmax peak approximation."""
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        conv_y = fast_convolve(y)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * a[k - idxs]

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        temp *= 1.005
        mom = min(0.97, mom + 0.0003)

        if no_improve >= 300:
            break

    return best_a, best_obj


def coordinate_refine(a0, time_limit):
    """Random pairwise mass transfer to refine the solution."""
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # Pick a peak lag
        peak_k = int(np.argmax(conv))
        # Pick indices that contribute to this peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        j1 = rng.integers(j_start, j_end + 1)
        j2 = rng.integers(0, n)
        if j1 == j2:
            continue

        delta = rng.choice([0.001, 0.002, 0.005, 0.01, 0.02]) * rng.choice([1, -1])
        if a[j1] + delta < 0 or a[j2] - delta < 0:
            continue

        a_try = a.copy()
        a_try[j1] += delta
        a_try[j2] -= delta
        a_try = normalize(a_try)
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()

        if obj_try < best_obj - 1e-12:
            a = a_try
            best_obj = obj_try
            best_a = a.copy()
            conv = conv_try

    return best_a, best_obj


def make_initializations(n, parent, rng):

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    s1 = np.sin(np.pi * x / n) ** 2 + 0.001
    inits['sine1'] = s1 / s1.sum()

    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.001
    inits['sine2'] = s2 / s2.sum()

    s3 = np.sin(3 * np.pi * x / n) ** 2 + 0.001
    inits['sine3'] = s3 / s3.sum()

    # Sparse: ones at regularly spaced positions
    sparse = np.zeros(n)
    step = max(1, n // 30)
    sparse[::step] = 1.0
    inits['sparse'] = sparse / sparse.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
            # Slightly perturbed parent
            noise = rng.standard_normal(n) * 0.01
            pp2 = np.maximum(p / ps + noise, 0)
            inits['parent_noise'] = pp2 / pp2.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Active-set LP search with adaptive parameters,
    gradient descent, and coordinate refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}")
    except Exception:
        pass

    # Determine target length
    if parent is not None:
        n_target = len(parent)
    else:
        n_target = 768

    print(f"Target length: {n_target}")

    # =========================================================
    # Phase 1: Quick gradient warm-up (60s)
    # =========================================================
    print("=== Phase 1: Gradient Warm-up ===")
    t_start = time.monotonic()
    phase1_time = 60.0

    inits = make_initializations(n_target, parent, rng)
    # Prioritize parent-based
    priority = ['parent', 'parent_noise', 'parent_interp', 'sine1', 'sine3', 'sparse', 'random']
    init_names = [nm for nm in priority if nm in inits]
    init_names += [nm for nm in inits if nm not in priority]
    init_names = init_names[:4]

    t_per = phase1_time / max(1, len(init_names))
    warmup_results = []

    for name in init_names:
        if time.monotonic() > deadline - 1500:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 1400)
        if t_alloc < 3:
            break
        a0 = inits[name]
        a_opt, obj = gradient_optimize(a0, time_limit=t_alloc,
                                        lr_init=0.008, momentum=0.92,
                                        top_k=30, temp_init=0.002)
        score = float(evaluate_sequence(a_opt.tolist()))
        warmup_results.append((score, name, a_opt.copy()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Grad {name}: {score:.6f}")

    warmup_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Active-set LP deep optimization (main phase)
    # =========================================================
    print("=== Phase 2: Active-Set LP Optimization ===")
    t_start2 = time.monotonic()

    # Take top candidates from warmup
    candidates = []
    seen = set()
    for score, name, a_opt in warmup_results:
        if name not in seen:
            candidates.append(a_opt)
            seen.add(name)
        if len(candidates) >= 3:
            break

    # Always include the overall best
    if best_sequence is not None:
        if len(candidates) < 3:
            candidates.append(best_sequence.copy())
        elif not any(np.allclose(c, best_sequence, atol=1e-6) for c in candidates):
            candidates[-1] = best_sequence.copy()

    remaining = deadline - time.monotonic() - 200.0
    n_cand = max(1, len(candidates))
    t_per_lp = max(30.0, remaining * 0.65 / n_cand)

    for i, a0 in enumerate(candidates):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 10:
            break
        print(f"  LP candidate {i+1}/{n_cand}, t={t_alloc:.0f}s")
        a_opt, obj = active_set_optimize(a0, time_limit=t_alloc, top_k_init=10, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i+1}: {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i+1}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Multi-config LP refinement on best
    # =========================================================
    print("=== Phase 3: Multi-Config LP Refinement ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        configs = [
            {'top_k': 5, 't_frac': 0.3},
            {'top_k': 15, 't_frac': 0.3},
            {'top_k': 30, 't_frac': 0.2},
            {'top_k': 50, 't_frac': 0.2},
        ]

        for ci, cfg in enumerate(configs):
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(remaining * cfg['t_frac'], deadline - time.monotonic() - 40)
            if t_alloc < 8:
                break

            # Slightly perturb for diversity
            if ci > 0:
                noise = rng.standard_normal(len(best_sequence)) * 0.003
                a0 = normalize(best_sequence + noise)
            else:
                a0 = best_sequence.copy()

            a_opt, obj = active_set_optimize(a0, time_limit=t_alloc,
                                              top_k_init=cfg['top_k'])
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Refine {ci} (K={cfg['top_k']}): {score:.6f} *NEW BEST*")
            else:
                print(f"  Refine {ci} (K={cfg['top_k']}): {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Gradient + coordinate polish
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Gradient polish
        t_grad = min(remaining * 0.4, 60.0)
        if t_grad > 5:
            for gi in range(2):
                if time.monotonic() > deadline - 20:
                    break
                cfgs = [
                    {'lr_init': 0.004, 'momentum': 0.94, 'top_k': 50, 'temp_init': 0.001},
                    {'lr_init': 0.008, 'momentum': 0.9, 'top_k': 20, 'temp_init': 0.003},
                ]
                cfg = cfgs[gi % len(cfgs)]
                a_opt, obj = gradient_optimize(best_sequence, time_limit=t_grad / (2 - gi), **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad polish {gi}: {score:.6f} *NEW BEST*")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 10:
            a_opt, obj = coordinate_refine(best_sequence, time_limit=min(remaining * 0.5, 30.0))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

        # Final LP micro-polish
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_opt, obj = active_set_optimize(best_sequence, time_limit=remaining, top_k_init=20)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(n_target) / n_target
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
