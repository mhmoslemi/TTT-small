"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with efficient sparse LP oracle
2. QP polishing with active set method (scipy minimize)
3. Multi-start with structured initializations
4. Adaptive time management

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:2*n-1]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(n):
    """Build the sparse LP constraint matrix for the convolution oracle.
    Returns a sparse matrix A of shape (2n-1, n) where A[k,j] = 1 if 0 <= k-j < n else 0.
    Actually we need A[k,j] = f[k-j] which depends on f, so we build the structure here."""
    # The structure is: row k, col j is nonzero iff 0 <= k-j < n
    # i.e., k-n+1 <= j <= k and 0 <= j <= n-1
    # So j ranges from max(0, k-n+1) to min(n-1, k)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def solve_lp_oracle(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (normalized to sum to 1) or None."""
    n = len(f)
    # Build sparse matrix: A[k, j] = f[k-j] if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def qp_polish(a_init, n, active_k, max_time=30):
    """Polish using QP: min t s.t. conv_k(a) <= t for k in active_k, a>=0, sum(a)=1."""
    def objective(x):
        # x = [a (n vars), t (1 var)]
        a = x[:n]
        t = x[n]
        return t
    
    def conv_k(a, k):
        # conv_k(a) = sum_i a_i * a[k-i] for valid i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            return 0.0
        return np.dot(a[js:je+1], a[k-je:k-js+1][::-1])
    
    # Constraints: conv_k(a) - t <= 0
    cons = []
    for k in active_k:
        def make_constraint(k_val):
            def constraint(x):
                a = x[:n]
                t = x[n]
                return t - conv_k(a, k_val)
            def jacobian(x):
                a = x[:n]
                grad_a = np.zeros(n)
                js = max(0, k_val - n + 1)
                je = min(n - 1, k_val)
                for i in range(js, je+1):
                    grad_a[i] = 2.0 * a[k_val - i]
                grad_t = -1.0
                return np.append(grad_a, grad_t)
            return {'type': 'ineq', 'fun': constraint, 'jac': jacobian}
        cons.append(make_constraint(k))
    
    # sum(a) = 1
    def eq_constraint(x):
        return x[:n].sum() - 1
    def eq_jacobian(x):
        j = np.zeros(n+1)
        j[:n] = 1.0
        return j
    cons.append({'type': 'eq', 'fun': eq_constraint, 'jac': eq_jacobian})
    
    # Bounds: a >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    x0 = np.append(a_init, conv_fast_max(a_init))
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
        if res.success:
            a_opt = np.maximum(res.x[:n], 0)
            s = a_opt.sum()
            if s > 1e-10:

            return a_opt
    except Exception:
        pass
    return None

def conv_fast_max(a):
    """Quick max of auto-convolution."""
    return fast_convolve(a).max()

def subgradient(f, conv, top_k=64):
    """Subgradient of max(conv) using uniform weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    sg = np.zeros(n)
    for k in peaks:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += f[k - idx]
    sg /= len(peaks)
    return sg

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with efficient sparse LP oracle (precomputed structure)

3. Smooth-max optimization (softmax approximation) as alternative polish

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def precompute_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix.
    Returns (rows, cols) arrays for the (2n-1) x n convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, rows, cols):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(f)
    vals = f[rows - cols]  # f[k-j] for each (k,j) pair
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg

def qp_polish(a_init, n, active_k, max_iter=300, tol=1e-12):
    """Polish via SLSQP: min t s.t. conv_k(a) <= t, a>=0, sum(a)=1."""
    def obj_func(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = []
    for k in active_k:
        def make_cfunc(k_val):
            def cfunc(x):
                a = x[:n]
                t = x[n]
                js = max(0, k_val - n + 1)
                je = min(n - 1, k_val)
                if js > je:
                    return t
                return t - np.dot(a[js:je+1], a[k_val - je:k_val - js + 1][::-1])
            def cjac(x):
                a = x[:n]
                grad_a = np.zeros(n)
                js = max(0, k_val - n + 1)
                je = min(n - 1, k_val)
                for i in range(js, je + 1):
                    grad_a[i] = -2.0 * a[k_val - i]
                grad_a[-1]  # just to ensure length
                g_full = np.zeros(n + 1)
                g_full[:n] = grad_a
                g_full[n] = 1.0
                return g_full
            return cfunc, cjac
        cf, cj = make_cfunc(k)
        cons.append({'type': 'ineq', 'fun': cf, 'jac': cj})

    def eq_func(x):
        return x[:n].sum() - 1.0

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    cons.append({'type': 'eq', 'fun': eq_func, 'jac': eq_jac})
    bounds = [(0, None)] * n + [(0, None)]
    conv_init = fast_convolve(a_init)
    x0 = np.append(a_init.copy(), conv_init.max())
    try:
        res = minimize(obj_func, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': tol})
        if res.x is not None:
            a_opt = np.maximum(res.x[:n], 0)
            s = a_opt.sum()
            if s > 1e-10:
                a_opt = a_opt / s
                return a_opt
    except Exception:
        pass
    return None


def smooth_max_optimize(a_init, n, peak_indices, beta, max_iter=200):
    """Optimize smooth approximation of max using L-BFGS-B."""
    def obj_and_grad(x):
        a = np.maximum(x, 0)
        s = a.sum()
        if s < 1e-15:
            return 0, np.zeros(n)
        a = a / s
        conv = fast_convolve(a)
        pk = conv[peak_indices]
        scaled = pk / beta
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(pk))
            w[np.argmax(pk)] = 1.0
        else:
            w /= wsum
        obj = beta * (np.log(wsum) + scaled.max())
        grad = np.zeros(n)
        for i in range(len(peak_indices)):
            if w[i] < 1e-14:
                continue
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]
        return obj, grad

    x0 = a_init.astype(np.float64).copy()
    try:
        res = minimize(obj_and_grad, x0, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'gtol': 1e-8})
        a_opt = np.maximum(res.x, 0)
        s = a_opt.sum()
        if s > 1e-10:
            a_opt = a_opt / s
            return a_opt
    except Exception:
        pass
    return None


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    tri = np.minimum(x, n - 1.0 - x) + 1.0
    inits.append(tri / tri.sum())
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(c / c.sum())
    g = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
    inits.append(g / g.sum())
    sn = np.sin(np.pi * (x + 0.5) / n) + 0.05
    inits.append(sn / sn.sum())
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    if parent is not None:
        if len(parent) == n:
            inits.insert(0, parent.copy())
        else:
            px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.insert(0, px / ps)
    return inits


def sg_descent(f0, time_limit, rows, cols, initial_lr=0.01, top_k=64, temp=0.001):
    """Subgradient descent with momentum and LP escape."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = initial_lr
    cur_temp = temp
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        v = 0.9 * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        obj = conv.max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = project_simplex(f - 0.5 * v)
            no_improve += 1
        cur_lr *= 0.9995
        cur_temp *= 1.002
        if no_improve > 200 and rem > 30:
            g = solve_lp_oracle(f, best_obj * 1.01, rows, cols)
            if g is not None:
                for alpha in [0.3, 0.5, 0.7]:
                    ft = project_simplex((1 - alpha) * f + alpha * g)
                    val = fast_convolve(ft).max()
                    if val < best_obj:
                        best_obj = val
                        best_f = ft.copy()
                        f = ft
                        conv = fast_convolve(f)
                        break
            no_improve = 0
            cur_lr = initial_lr * 0.5
            v = np.zeros(n)
        if no_improve > 800:
            rng = np.random.default_rng(it * 7 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = initial_lr
            no_improve = 0
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Optimize parent via FW + QP polish
    Phase 2: Multi-length exploration with FW + SG + QP
    Phase 3: Deep iterative QP polishing with active set updates
    Phase 4: Smooth-max optimization and final micro-polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)}) (was {old:.10f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.10f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # Phase 1: Parent optimization
    if parent is not None:
        n_p = len(parent)
        print(f"\n=== Phase 1: Parent optimization (n={n_p}) ===")
        rows_p, cols_p = precompute_lp_structure(n_p)
        t1 = min(180, deadline - time.monotonic() - 500)
        if t1 > 30:
            print(f"  FW on parent, t={t1:.0f}s")
            a_opt, obj = fw_optimize(parent, time_limit=t1, rows=rows_p, cols=cols_p)
            try_seq(a_opt, "parent_fw")
            # QP polish
            if time.monotonic() < deadline - 100:
                conv = fast_convolve(a_opt)
                n_peaks = min(16, n_p)
                top_idx = sorted(np.argpartition(conv, -n_peaks)[-n_peaks:].tolist())
                a_qp = qp_polish(a_opt, n_p, top_idx, max_iter=400)
                if a_qp is not None:
                    try_seq(a_qp, "parent_qp1")
                # Second QP with updated peaks
                if a_qp is not None and time.monotonic() < deadline - 80:
                    conv2 = fast_convolve(a_qp)
                    top_idx2 = sorted(np.argpartition(conv2, -16)[-16:].tolist())
                    a_qp2 = qp_polish(a_qp, n_p, top_idx2, max_iter=300)
                    if a_qp2 is not None:
                        try_seq(a_qp2, "parent_qp2")
                # Smooth max
                if time.monotonic() < deadline - 60:
                    conv3 = fast_convolve(best_seq)
                    top3 = sorted(np.argpartition(conv3, -20)[-20:].tolist())
                    a_sm = smooth_max_optimize(best_seq, n_p, top3, beta=50, max_iter=100)
                    if a_sm is not None:
                        try_seq(a_sm, "parent_smooth")
                        # Final QP on smooth result
                        conv4 = fast_convolve(a_sm)
                        top4 = sorted(np.argpartition(conv4, -18)[-18:].tolist())
                        a_qp3 = qp_polish(a_sm, n_p, top4, max_iter=300)
                        if a_qp3 is not None:
                            try_seq(a_qp3, "parent_qp3")

    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-length exploration ===")
    lengths = [512, 384, 256, 640]
    if parent is not None:
        if len(parent) in lengths:
            lengths.remove(len(parent))
            lengths.insert(0, len(parent))

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        if parent is not None and n == len(parent):
            continue
        rows_n, cols_n = precompute_lp_structure(n)
        inits = generate_inits(n, parent, rng)
        n_inits = min(4, len(inits))
        rem_total = deadline - time.monotonic() - 150
        t_per = max(25, min(80, rem_total / (n_inits * max(1, len(lengths) - lengths.index(n)))))

        for i in range(n_inits):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 80)
            if t_alloc < 20:
                break
            f_init = inits[i]
            # FW
            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc * 0.6, rows=rows_n, cols=cols_n)
            try_seq(a_opt, f"n{n}_fw_{i}")
            # SG follow-up
            rem2 = deadline - time.monotonic() - 60
            if rem2 > 20:
                a_sg, _ = sg_descent(a_opt, time_limit=min(rem2 * 0.4, 40),
                                     rows=rows_n, cols=cols_n,
                                     initial_lr=0.008, top_k=max(32, n // 10), temp=0.002)
                try_seq(a_sg, f"n{n}_sg_{i}")
            # QP polish
            rem3 = deadline - time.monotonic() - 40
            if rem3 > 15:
                a_curr = a_opt if a_opt.sum() > 0 else a_sg
                conv = fast_convolve(a_curr)
                top_idx = sorted(np.argpartition(conv, -12)[-12:].tolist())
                a_qp = qp_polish(a_curr, n, top_idx, max_iter=300)
                if a_qp is not None:
                    try_seq(a_qp, f"n{n}_qp_{i}")

    # Phase 3: Deep iterative QP polishing on best
    print(f"\n=== Phase 3: Deep QP polishing ===")
    print(f"  Current best: {best_score:.10f}")
    if best_seq is not None:
        n_best = len(best_seq)
        rows_b, cols_b = precompute_lp_structure(n_best)
        for rnd in range(8):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            # QP with increasing peaks
            n_peaks = min(10 + rnd * 3, n_best)
            conv = fast_convolve(best_seq)
            top_idx = sorted(np.argpartition(conv, -n_peaks)[-n_peaks:].tolist())
            a_qp = qp_polish(best_seq, n_best, top_idx, max_iter=500)
            if a_qp is not None:
                try_seq(a_qp, f"deep_qp_{rnd}")
            # Smooth max
            rem = deadline - time.monotonic() - 20
            if rem > 15:
                beta = 30 + rnd * 10
                a_sm = smooth_max_optimize(best_seq, n_best, top_idx, beta=beta, max_iter=150)
                if a_sm is not None:
                    try_seq(a_sm, f"deep_smooth_{rnd}")
                    # QP after smooth
                    if time.monotonic() < deadline - 15:
                        conv_sm = fast_convolve(a_sm)
                        top_sm = sorted(np.argpartition(conv_sm, -n_peaks)[-n_peaks:].tolist())
                        a_qp2 = qp_polish(a_sm, n_best, top_sm, max_iter=300)
                        if a_qp2 is not None:
                            try_seq(a_qp2, f"deep_sm_qp_{rnd}")
            # FW escape with perturbation
            rem = deadline - time.monotonic() - 20
            if rem > 15 and rnd % 3 == 2:
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                a_fw, _ = fw_optimize(f_pert, time_limit=min(30, rem * 0.7),
                                      rows=rows_b, cols=cols_b)
                try_seq(a_fw, f"deep_fw_{rnd}")
                if time.monotonic() < deadline - 10:
                    conv_fw = fast_convolve(a_fw)
                    top_fw = sorted(np.argpartition(conv_fw, -12)[-12:].tolist())
                    a_qp3 = qp_polish(a_fw, n_best, top_fw, max_iter=200)
                    if a_qp3 is not None:
                        try_seq(a_qp3, f"deep_fw_qp_{rnd}")

    # Phase 4: Final micro-polish
    print(f"\n=== Phase 4: Final micro-polish ===")
    print(f"  Current best: {best_score:.10f}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            conv = fast_convolve(best_seq)
            vals = np.sort(conv)[::-1]
            n_peaks = min(np.sum(vals >= vals[0] * 0.999), n_best)
            n_peaks = max(n_peaks, 8)
            top_idx = sorted(np.argpartition(conv, -n_peaks)[-n_peaks:].tolist())
            a_final = qp_polish(best_seq, n_best, top_idx, max_iter=600, tol=1e-13)
            if a_final is not None:
                try_seq(a_final, "final_qp")
            # One more smooth max with high beta
            if time.monotonic() < deadline - 5:
                a_sm_final = smooth_max_optimize(best_seq, n_best, top_idx, beta=100, max_iter=200)
                if a_sm_final is not None:
                    try_seq(a_sm_final, "final_smooth")
                    if time.monotonic() < deadline - 3:
                        conv_fs = fast_convolve(a_sm_final)
                        top_fs = sorted(np.argpartition(conv_fs, -12)[-12:].tolist())
                        a_qp_final = qp_polish(a_sm_final, n_best, top_fs, max_iter=200)
                        if a_qp_final is not None:
                            try_seq(a_qp_final, "final_sm_qp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
