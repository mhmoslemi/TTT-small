    best=p; best_score=_score_p(p)

    for idx, spec in enumerate(specs):

```python
"""
Search for low-score nonnegative step functions by minimizing
2*n*max(conv(a,a))/(sum(a))^2.

The implementation combines:
  - L-BFGS polishing on softmax/square parameterizations of the simplex,
  - annealed smooth peak penalties (log-p-norm / excess-over-flat),
  - a cheap spectral Newton-style construction for square roots of flat
    convolution,
  - Frank-Wolfe steps with an exact LP linear-response oracle,
  - stochastic/mirror updates, coordinate moves, and modest length exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


class _Timeout(Exception):
    pass


_OPT = {
    "n": 0,
    "ref": 0,
    "beta": 16.0,
    "smode": "excess",
    "mode": "softmax",
}

_TIMEOUT = {"deadline": None}

_LASTX = {"x": None}


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _norm_p(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _floor_p(p, eps=1e-9):
    p = _norm_p(p)
    if eps <= 0.0:
        return p
    p = np.maximum(p, float(eps))
    return _norm_p(p)


def _conv_sq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64)
    c = c / s
    return c


def _conv_ab(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    L = int(a.size + b.size - 1)
    if L <= 0:
        return np.ones(1, dtype=np.float64)
    N = _next_pow2(L)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    return np.maximum(c, 0.0)


def _score_p(p):
    p = _norm_p(p)
    c = _conv_sq(p)
    return float(2.0 * len(p) * float(c.max()))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = int(v.size)
    if n == 0:
        return v

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * rho) > (cssv - 1.0)

    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)

    idx = int(np.nonzero(cond)[0][-1])
    theta = (cssv[idx] - 1.0) / float(idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def _gradient_weights(w, a):
    """
    Gradient of sum_k w_k (a*a)[k] wrt a.
    w length is 2n-1.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    L = int(w.size)
    if n == 0:
        return np.array([], dtype=np.float64)

    conv_len = L + n - 1
    N = _next_pow2(conv_len)
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a[::-1], N)
    g = np.fft.irfft(W * Ar, N)[:conv_len]
    g = 2.0 * g[n - 1:n - 1 + n]
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def _smooth(c, beta, smode="excess"):
    """
    Smooth penalty and weights.

    smode='power':  F = (1/beta) log sum_k (c_k * L)^beta
    smode='excess': F = (1/beta) log sum_k exp(beta*(c_k*L - 1))
    """
    c = np.asarray(c, dtype=np.float64).ravel()
    L = int(c.size)
    q = np.maximum(c, 0.0) * float(L)

    if smode == "power":
        q_safe = np.maximum(q, 1e-300)
        s = np.log(q_safe)
    else:
        s = q - 1.0

    s = np.where(np.isfinite(s), s, -1e300)
    m = float(np.max(s))
    z = float(beta) * (s - m)
    z = np.clip(z, -745.0, 0.0)
    ez = np.exp(z)
    zsum = float(ez.sum())

    if not np.isfinite(zsum) or zsum <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        val = m
    else:
        w = ez / zsum
        val = float(m + np.log(zsum) / float(beta))

    w = np.where(np.isfinite(w), w, 0.0)
    ws = float(w.sum())
    if ws <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ws

    return float(val), w


def _fun_softmax(x):
    n = int(_OPT.get("n", 0))
    ref = int(_OPT.get("ref", 0))
    beta = float(_OPT.get("beta", 16.0))
    smode = str(_OPT.get("smode", "excess"))

    x = np.asarray(x, dtype=np.float64).ravel()
    if n <= 0 or x.size != n or not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n, dtype=np.float64)

    z = x - x[ref]
    if not np.all(np.isfinite(z)):
        return 1e100, np.zeros(n, dtype=np.float64)

    m = float(np.max(z))
    ez = np.exp(z - m)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return 1e100, np.zeros(n, dtype=np.float64)

    p = ez / s
    c = _conv_sq(p)
    val, w = _smooth(c, beta, smode)
    L = int(c.size)

    g = _gradient_weights(L * w, p)
    dot = float(np.dot(p, g))
    gz = p * (g - dot)
    gx = gz - gz[ref]

    gx = np.where(np.isfinite(gx), gx, 0.0)
    return float(val), gx


def _fun_square(x):
    n = int(_OPT.get("n", 0))
    beta = float(_OPT.get("beta", 16.0))
    smode = str(_OPT.get("smode", "excess"))

    x = np.asarray(x, dtype=np.float64).ravel()
    if n <= 0 or x.size != n or not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n, dtype=np.float64)

    r = float(np.linalg.norm(x))
    if not np.isfinite(r) or r <= 1e-150:
        return 1e100, np.zeros(n, dtype=np.float64)

    u = x / r
    p = u * u
    ps = float(p.sum())
    if ps <= 1e-300:
        return 1e100, np.zeros(n, dtype=np.float64)
    p = p / ps

    c = _conv_sq(p)
    val, w = _smooth(c, beta, smode)
    L = int(c.size)

    g = _gradient_weights(L * w, p)
    dot = float(np.dot(p, g))
    h = 2.0 * u * (g - dot)
    h = h - u * float(np.dot(u, h))
    gx = h / r

    gx = np.where(np.isfinite(gx), gx, 0.0)
    return float(val), gx

def _lbfgs_run(p, beta, time_left, smode="excess", mode="softmax", ref=None, maxiter=None):

    if n == 0 or float(time_left) < 4.0:
        return p

    if ref is None:
        ref = int(np.argmax(p))
    ref = int(ref) % n

    p = _floor_p(p, 1e-9)

    if maxiter is None:
        maxiter = max(80, min(2500, int(float(time_left) * 5.0)))

    _OPT["n"] = n
    _OPT["ref"] = ref
    _OPT["beta"] = float(beta)
    _OPT["smode"] = str(smode)
    _OPT["mode"] = str(mode)

    if mode == "square":
        x0 = np.sqrt(p)
        r = float(np.linalg.norm(x0))
        if r <= 1e-150:
            x0 = np.ones(n, dtype=np.float64) / np.sqrt(float(n))
        else:
            x0 = x0 / r
        fun = _fun_square
    else:
        p_safe = np.maximum(p, 1e-300)
        with np.errstate(divide="ignore", invalid="ignore"):
            x0 = np.log(p_safe) - np.log(p_safe[ref])
        x0 = np.where(np.isfinite(x0), x0, 0.0)
        med = float(np.median(x0))
        if np.isfinite(med):
            x0 = x0 - med
        fun = _fun_softmax

    _TIMEOUT["deadline"] = time.monotonic() + max(2.0, float(time_left) - 2.0)
    _LASTX["x"] = x0.copy()

    options = {
        "maxiter": int(maxiter),
        "gtol": 1e-9,
        "ftol": 1e-14,
        "maxcor": 30,
        "maxls": 40,
    }

    x = None
    try:
        res = minimize(
            fun,
            x0,
            jac=True,
            method="L-BFGS-B",
            callback=_minimize_callback,
            options=options,
        )
        if hasattr(res, "x") and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    finally:
        _TIMEOUT["deadline"] = None

    if x is None:
        x = _LASTX.get("x")
        if x is not None:
            x = np.asarray(x, dtype=np.float64)
        else:
            x = None

    if x is None:
        return p
    if x.ndim != 1 or x.size != n or not np.all(np.isfinite(x)):
        return p

    if mode == "square":
        r = float(np.linalg.norm(x))
        if r <= 1e-150:
            return p
        u = x / r
        p_new = u * u
    else:
        z = x - x[ref]
        if not np.all(np.isfinite(z)):
            return p
        m = float(np.max(z))
        ez = np.exp(z - m)
        s = float(ez.sum())
        if not np.isfinite(s) or s <= 1e-300:
            return p
        p_new = ez / s

    p_new = _norm_p(p_new)
    return p_new


def _lbfgs_sequence(p_start, specs, deadline, mode="softmax", ref=None):
    p = _norm_p(p_start)
    best = p.copy()
    best_score = _score_p(best)

    if len(specs) == 0 or time.monotonic() > deadline - 8.0:
        return best, best_score

    for idx, spec in enumerate(specs):
        beta = float(spec[0])
        smode = str(spec[1])

        now = time.monotonic()
        rem = deadline - now - 5.0
        if rem < 8.0:
            break

        rem_specs = len(specs) - idx
        t = (rem * 0.95) / float(rem_specs)
        t = max(8.0, min(180.0, t))
        t = min(t, rem)

        p_new = _lbfgs_run(p, beta, t, smode=smode, mode=mode, ref=ref)
        s_new = _score_p(p_new)

        if s_new < best_score - 1e-12:
            best = p_new.copy()
            best_score = s_new
            p = best
        else:
            p = best

        print(
            "    L-BFGS mode=%s beta=%d smode=%s best=%.8f"
            % (mode, int(beta), smode, best_score)
        )

    return best, best_score


def _spectral_newton(p_start, max_iters=25):
    """
    Cheap spectral Newton-like iteration for solving A^2 ~= flat polynomial.
    """
    a = _norm_p(p_start)
    best = a.copy()
    best_score = _score_p(best)
    n = int(a.size)

    L = 2 * n - 1
    N = _next_pow2(L)

    c0 = np.zeros(N, dtype=np.float64)
    c0[:L] = 1.0 / float(L)
    C0 = np.fft.fft(c0)

    for _ in range(int(max_iters)):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        eps = 1e-10 * max(1.0, float(np.max(mag)))
        A_safe = np.where(mag > eps, A, np.complex128(eps) + 0.0j)

        R = C0 / A_safe
        rmax = float(np.max(np.abs(R)))
        if rmax > 1e6:
            R = R * (1e6 / rmax)

        cand_full = np.fft.ifft(R)[:n].real
        a_pad = np.zeros(N, dtype=np.float64)
        a_pad[:n] = a
        cand = 0.5 * a_pad[:n] + 0.5 * cand_full

        cand = np.where(np.isfinite(cand), cand, 0.0)
        cand = np.maximum(cand, 0.0)
        cs = float(cand.sum())
        if not np.isfinite(cs) or cs <= 1e-300:
            break
        cand = cand / cs

        improved = False
        for t in (0.1, 0.3, 0.5, 0.7, 1.0):
            mix = (1.0 - t) * a + t * cand
            mix = _norm_p(mix)
            s = _score_p(mix)
            if s < best_score - 1e-12:
                best = mix.copy()
                best_score = s
                a = mix
                improved = True

        if not improved:
            break

    return best, best_score


def _line_search_mix(c_a, c_g, c_ag):
    current = float(c_a.max())

    alphas = np.linspace(0.0, 1.0, 201, dtype=np.float64)
    aa = alphas[:, None]
    q = (
        ((1.0 - aa) ** 2) * c_a[None, :]
        + (2.0 * aa * (1.0 - aa)) * c_ag[None, :]
        + (aa ** 2) * c_g[None, :]
    )
    vals = q.max(axis=1)
    i = int(np.argmin(vals))
    best_alpha = float(alphas[i])
    best_val = float(vals[i])

    base = best_alpha
    lo = max(0.0, base - 0.05)
    hi = min(1.0, base + 0.05)
    if hi - lo > 1e-12:
        al2 = np.linspace(lo, hi, 81, dtype=np.float64)
        aa2 = al2[:, None]
        q2 = (
            ((1.0 - aa2) ** 2) * c_a[None, :]
            + (2.0 * aa2 * (1.0 - aa2)) * c_ag[None, :]
            + (aa2 ** 2) * c_g[None, :]
        )
        vals2 = q2.max(axis=1)
        j = int(np.argmin(vals2))
        if float(vals2[j]) < best_val:
            best_alpha = float(al2[j])
            best_val = float(vals2[j])

    if best_val > current:
        best_alpha = 0.0
        best_val = current

    return best_alpha, best_val


def _lp_direction(a, time_limit=6.0):
    a = _norm_p(a)
    n = int(a.size)
    if n == 0 or n > 700:
        return None, None

    L = 2 * n - 1
    nv = n + 1

    A_ub = np.zeros((L, nv), dtype=np.float64)
    A_ub[:, 0] = -1.0
    for j in range(n):
        A_ub[j:j + n, j + 1] = a

    A_eq = np.zeros((1, nv), dtype=np.float64)
    A_eq[0, 1:] = 1.0

    c = np.zeros(nv, dtype=np.float64)
    c[0] = 1.0
    b_ub = np.zeros(L, dtype=np.float64)
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * nv

    res = None
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(time_limit), "presolve": True},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except Exception:
            return None, None

    if res is None or not getattr(res, "success", False):
        return None, None

    x = np.asarray(res.x, dtype=np.float64)
    if x.size != nv:
        return None, None

    g = np.maximum(x[1:], 0.0)
    gs = float(g.sum())
    if not np.isfinite(gs) or gs <= 1e-15:
        return None, None

    g = g / gs
    t_opt = float(x[0])
    if not np.isfinite(t_opt):
        t_opt = float(np.max(_conv_sq(g)))

    return g, t_opt


def _fw_lp(p_start, deadline):
    a = _norm_p(p_start)
    best = a.copy()
    best_score = _score_p(best)
    n = int(a.size)

    if n > 600 or time.monotonic() > deadline - 10.0:
        return best, best_score

    c_a = _conv_sq(a)
    no_improve = 0

    while time.monotonic() < deadline - 8.0:
        g, t_opt = _lp_direction(a, time_limit=5.0)
        if g is None:
            break

        c_g = _conv_sq(g)
        c_ag = _conv_ab(a, g)
        alpha, val = _line_search_mix(c_a, c_g, c_ag)

        if alpha <= 1e-8:
            no_improve += 1
            if no_improve > 8:
                break
            alpha = 0.05

        a_new = (1.0 - alpha) * a + alpha * g
        a_new = _norm_p(a_new)
        s_new = _score_p(a_new)

        if s_new < best_score - 1e-12:
            best = a_new.copy()
            best_score = s_new
            a = a_new
            c_a = _conv_sq(a)
            no_improve = 0
            print("    LP-FW improved to %.8f" % best_score)
        else:
            no_improve += 1
            if no_improve > 10:
                break

    return best, best_score


def _sg_polish(p_start, deadline, beta=24.0, smode="excess", lr=0.004):
    p = _floor_p(p_start, 1e-9)
    best = p.copy()
    best_score = _score_p(best)
    n = int(p.size)
    no_improve = 0
    cur_lr = float(lr)

    while time.monotonic() < deadline - 3.0:
        c = _conv_sq(p)
        _, w = _smooth(c, float(beta), smode)
        L = int(c.size)

        g = _gradient_weights(L * w, p)
        dot = float(np.dot(p, g))
        g = g - dot

        logp = np.log(np.maximum(p, 1e-300))
        logp_new = logp - cur_lr * g
        m = float(np.max(logp_new))
        p_new = np.exp(logp_new - m)
        s = float(p_new.sum())
        if not np.isfinite(s) or s <= 1e-300:
            break
        p_new = _norm_p(p_new)

        s_new = _score_p(p_new)
        if s_new < best_score - 1e-12:
            best = p_new.copy()
            best_score = s_new
            p = p_new
            no_improve = 0
            cur_lr = min(0.03, cur_lr * 1.03)
        else:
            no_improve += 1
            cur_lr *= 0.75
            if no_improve > 90:
                idx = np.arange(n, dtype=np.float64)
                noise = 0.002 * np.sin(idx * 0.41 + 1.7)
                p = _project_simplex(best + noise)
                cur_lr = float(lr)
                no_improve = 0

            if cur_lr < 1e-9:
                break

    return best, best_score


def _coordinate_polish(p_start, deadline):
    p = _floor_p(p_start, 1e-9)
    best = p.copy()
    best_score = _score_p(best)
    n = int(p.size)
    rng = np.random.default_rng(424242)
    no_improve = 0

    while time.monotonic() < deadline - 2.5:
        c = _conv_sq(p)
        L = int(c.size)
        topk = min(12, L)
        top_idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(L, dtype=np.float64)
        w[top_idx] = 1.0 / float(topk)

        g = _gradient_weights(w, p)
        dot = float(np.dot(p, g))
        g = g - dot

        order = np.argsort(g)
        dsts = order[:8]
        srcs = order[-8:][::-1]

        improved = False
        for src in srcs:
            if improved:
                break
            for dst in dsts:
                if src == dst:
                    continue
                for frac in (0.01, 0.03, 0.10, 0.25, 0.50):
                    delta = p[src] * frac
                    if delta <= 0.0:
                        continue
                    p2 = p.copy()
                    p2[src] -= delta
                    p2[dst] += delta
                    p2 = _norm_p(p2)
                    s2 = _score_p(p2)
                    if s2 < best_score - 1e-13:
                        best = p2.copy()
                        best_score = s2
                        p = p2
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            for _ in range(5):
                src = int(rng.integers(n))
                dst = int(rng.integers(n))
                if src == dst:
                    continue
                delta = p[src] * 0.05
                p2 = p.copy()
                p2[src] -= delta
                p2[dst] += delta
                p2 = _norm_p(p2)
                s2 = _score_p(p2)
                if s2 < best_score - 1e-13:
                    best = p2.copy()
                    best_score = s2
                    p = p2
                    no_improve = 0
                    break

            if no_improve > 180:
                break

    return best, best_score


def _edge_inits(n, rng):
    out = []
    x = np.arange(n, dtype=np.float64) + 0.5
    right = float(n) + 1.0 - x

    out.append(np.ones(n, dtype=np.float64) / float(n))

    for alpha in (0.25, 0.5, 0.75, 1.0, 1.5):
        w = np.power(x, -alpha) + np.power(right, -alpha)
        out.append(_norm_p(w))

        w2 = np.power(x, -alpha) * np.power(right, -alpha)
        out.append(_norm_p(w2))

    edge = _norm_p(np.power(x, -0.5) + np.power(right, -0.5))
    for beta in (0.3, 0.6, 0.9):
        mix = (1.0 - beta) * (np.ones(n) / float(n)) + beta * edge
        out.append(_norm_p(mix))

    for conc in (0.3, 1.0, 2.5):
        try:
            out.append(_norm_p(rng.dirichlet(np.full(n, conc, dtype=np.float64))))
        except Exception:
            pass

    return out


def _perturb_simplex(p, sigma, seed=1234):
    p = _norm_p(p)
    rng = np.random.default_rng(seed)
    noise = rng.standard_normal(p.size) * float(sigma)
    return _project_simplex(p + noise)


def _interpolate(p, m):
    p = _norm_p(p)
    m = int(m)
    if m <= 0:
        return p
    if m == p.size:
        return p

    xo = np.linspace(0.0, 1.0, p.size, dtype=np.float64)
    xn = np.linspace(0.0, 1.0, m, dtype=np.float64)
    q = np.interp(xn, xo, p)
    return _norm_p(q)


def _update_state(state, seq, label):
    seq = _norm_p(seq)
    score = None
    try:
        score = float(evaluate_sequence([float(x) for x in seq]))
    except Exception:
        score = None

    if score is None or not np.isfinite(score):
        score = _score_p(seq)

    if not np.isfinite(score):
        return state

    if state["score"] is None or score < float(state["score"]) - 1e-15:
        state["score"] = score
        state["seq"] = seq.copy()
        print("  NEW BEST [%s]: %.8f (n=%d)" % (label, score, len(seq)))

    return state


def _explore_length(n, deadline, parent, state):
    n = int(n)
    if n <= 0 or n > 1200:
        return
    if time.monotonic() > deadline - 70.0:
        return
    if parent is not None and n == len(parent):
        return

    p_list = []
    if parent is not None:
        p_list.append(_interpolate(parent, n))

    rng = np.random.default_rng(987653 + n)
    p_list.extend(_edge_inits(n, rng)[:5])

    best_p = None
    best_s = float("inf")
    for p in p_list:
        p = _floor_p(p, 1e-9)
        s = _score_p(p)
        _update_state(state, p, "explore_init_n%d" % n)
        if s < best_s:
            best_s = s
            best_p = p

    if best_p is None:
        return

    specs = [(16, "excess"), (32, "excess")]
    block_end = min(deadline, time.monotonic() + 80.0)
    p_best, _ = _lbfgs_sequence(best_p, specs, block_end, mode="softmax")
    _update_state(state, p_best, "explore_lbfgs_n%d" % n)

    if time.monotonic() < deadline - 80.0:
        alt = None
        for p in reversed(p_list):
            if p is not best_p:
                alt = p
                break
        if alt is not None:
            block_end2 = min(deadline, time.monotonic() + 70.0)
            p_best2, _ = _lbfgs_sequence(_floor_p(alt, 1e-9), specs, block_end2, mode="softmax")
            _update_state(state, p_best2, "explore_lbfgs2_n%d" % n)

            break
        p_start = _floor_p(p_start, 1e-9)
        _update_state(state, p_start, "div_pert_%d" % idx)
        local_deadline = min(div_deadline, time.monotonic() + 70.0)
        if local_deadline < time.monotonic() + 20.0:
            continue
        p_out, _ = _lbfgs_sequence(p_start, div_specs, local_deadline, mode="softmax")
        _update_state(state, p_out, "div_lbfgs_%d" % idx)
        best_p = state["seq"].copy() if state["seq"] is not None else best_p

    print("=== Phase 7: modest length exploration ===")
    explore_deadline = min(deadline, t_start + 0.92 * max(60.0, budget))
    if explore_deadline < time.monotonic() + 90.0:
        explore_deadline = time.monotonic() + min(180.0, max(30.0, (deadline - time.monotonic()) * 0.50))

    explore_lengths = [512, 768, 1024]
    if base_n != 512:
        explore_lengths.append(512)
    if base_n != 640:
        explore_lengths.append(640)
    if base_n != 896:
        explore_lengths.append(896)

    seen_lengths = set()
    for n in explore_lengths:
        n = int(n)
        if n in seen_lengths or n == base_n:
            continue
        seen_lengths.add(n)
        if time.monotonic() > explore_deadline - 80.0:
            break
        _explore_length(n, explore_deadline, parent, state)
        best_p = state["seq"].copy() if state["seq"] is not None else best_p

    print("=== Phase 8: final high-beta squeeze ===")
    final_deadline = deadline - 70.0
    if final_deadline > time.monotonic() + 25.0:
        final_specs = [(64, "excess"), (96, "excess"), (128, "excess")]
        p_out, _ = _lbfgs_sequence(best_p, final_specs, final_deadline, mode="softmax")
        _update_state(state, p_out, "final_lbfgs")
        best_p = state["seq"].copy() if state["seq"] is not None else best_p

    print("=== Phase 9: subgradient / coordinate polish ===")
    sg_deadline = deadline - 35.0
    if sg_deadline > time.monotonic() + 15.0:
        p_out, _ = _sg_polish(best_p, sg_deadline, beta=32.0, smode="excess", lr=0.003)
        _update_state(state, p_out, "sg_polish")
        best_p = state["seq"].copy() if state["seq"] is not None else best_p

    coord_deadline = deadline - 3.0
    if coord_deadline > time.monotonic() + 10.0:
        p_out, _ = _coordinate_polish(best_p, coord_deadline)
        _update_state(state, p_out, "coord_polish")
        best_p = state["seq"].copy() if state["seq"] is not None else best_p

    if state["seq"] is None:
        fb = np.ones(max(32, base_n), dtype=np.float64) / float(max(32, base_n))
        state = _update_state(state, fb, "final_fallback")
        best_p = fb

    _update_state(state, best_p, "final_check")

    print("FINAL SCORE: %.10f (n=%d)" % (float(state["score"]), len(state["seq"])))
    return [float(x) for x in state["seq"]]
```
