```python
"""
Step-function coefficient minimization for:
    2*n*max(conv(a,a)) / (sum(a))^2

The program uses a combination of:
  * a weighted conditional-gradient (Frank-Wolfe-like) solver, whose linear
    oracle finds a direction g on the simplex minimizing a weighted overlap
    max_s w_s (p*g)_s (possibly with a first-order correction),
  * exact line search over convex combinations p -> (1-alpha)p + alpha*g,
  * Adam-style projected-gradient polishing with log-sum-exp smoothing,
  * occasional multiplicative/projected smooth steps and restarts,
  * exploration of a few alternative sequence lengths.
"""

import time
import numpy as np
from scipy.optimize import linprog


_LP_TABLES = {}


def as_norm_sequence(p):
    """Convert input to a finite, nonnegative, sum-to-one ndarray."""
    arr = np.asarray(p, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-18:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def fast_convolve(a):
    """Linear convolution a*a via FFT, with tiny negative roundoff clipped."""
    n = len(a)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    out = np.real(out)
    out[~np.isfinite(out)] = 0.0
    out = np.maximum(out, 0.0)
    return out


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        x = float(v[0])
        if not np.isfinite(x):
            x = 0.0
        return np.array([max(x, 0.0)], dtype=np.float64)

    u = np.sort(v)[::-1]
    u = np.where(np.isfinite(u), u, 0.0)
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    ind = int(np.where(cond)[0][-1])
    theta = (cssv[ind] - 1.0) / float(ind + 1)
    w = v.astype(np.float64) - theta
    w = np.where(np.isfinite(w), w, 0.0)
    np.maximum(w, 0.0, out=w)
    s = float(w.sum())
    if s <= 1e-18:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


def lse_weights(c, beta):
    """Softmax weights exp(beta*(c-max(c))) / sum, stable."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c.copy()
    m = float(np.max(c))
    if beta <= 1e-14:
        return np.full(c.size, 1.0 / c.size, dtype=np.float64)
    z = beta * (c - m)
    z = np.where(np.isfinite(z), z, -1000.0)
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-18:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def compute_grad(p, weights):
    """
    Gradient wrt p of sum_s weights[s] * (p*p)[s].
    Returns 2 * sum_s weights[s] * p_{s-i}.
    """
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    w = np.asarray(weights, dtype=np.float64).ravel()
    if w.size < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:w.size] = w
        w = tmp
    else:
        w = w[:L]
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(w.sum())
    if s <= 1e-18:
        return np.zeros(n, dtype=np.float64)
    rev = p[::-1].copy()
    total_len = L + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(rev, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    g = conv[n - 1:n - 1 + n]
    g = 2.0 * np.real(g)
    g[~np.isfinite(g)] = 0.0
    g = np.maximum(g, 0.0)
    return g


def make_weights(c, T, w_min=1e-6):
    """Weights peaked at large c, with a small floor."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c.copy()
    M = float(np.max(c))
    if T <= 1e-15 * max(M, 1e-30):
        w = (c >= M - 1e-12 * max(M, 1e-30)).astype(np.float64)
        if float(w.max()) <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
    else:
        z = (c - M) / max(T, 1e-300)
        z = np.where(np.isfinite(z), z, -1000.0)
        e = np.exp(z - np.max(z))
        mx = float(e.max())
        if mx <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / mx
    w = np.maximum(w, float(w_min))
    return w


def get_lp_tables(n):
    """Precompute column/p-index tables for (p*g)_s rows, independent of p."""
    if n in _LP_TABLES:
        return _LP_TABLES[n]
    L = 2 * n - 1
    cols_list = []
    pidx_list = []
    for s in range(L):
        start = max(0, s - n + 1)
        end = min(n - 1, s) + 1
        if start < end:
            cols = np.arange(start, end, dtype=np.int64)
            pidx = s - cols
        else:
            cols = np.empty(0, dtype=np.int64)
            pidx = np.empty(0, dtype=np.int64)
        cols_list.append(cols)
        pidx_list.append(pidx)
    _LP_TABLES[n] = (cols_list, pidx_list)
    return _LP_TABLES[n]


def solve_lp_direction(p, c, weights, alpha, lp_time, tiebreak=None):
    """
    Solve the linear oracle:
        minimize t

    best_v = float(vals[idx])
```

```python
"""
Step-function coefficient minimization for:
    2*n*max(conv(a,a)) / (sum(a))^2

The program uses a combination of:
  * a weighted conditional-gradient (Frank-Wolfe-like) solver, whose linear
    oracle finds a direction g on the simplex minimizing a weighted overlap
    max_s w_s (p*g)_s (possibly with a first-order correction),
  * exact line search over convex combinations p -> (1-alpha)p + alpha*g,
  * Adam-style projected-gradient polishing with log-sum-exp smoothing,
  * occasional multiplicative/projected smooth steps and restarts,
  * exploration of a few alternative sequence lengths.
"""

import time
import numpy as np
from scipy.optimize import linprog


_LP_TABLES = {}


def as_norm_sequence(p):
    """Convert input to a finite, nonnegative, sum-to-one ndarray."""
    arr = np.asarray(p, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-18:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def fast_convolve(a):
    """Linear convolution a*a via FFT, with tiny negative roundoff clipped."""
    n = len(a)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    out = np.real(out)
    out[~np.isfinite(out)] = 0.0
    out = np.maximum(out, 0.0)
    return out


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        x = float(v[0])
        if not np.isfinite(x):
            x = 0.0
        return np.array([max(x, 0.0)], dtype=np.float64)

    u = np.sort(v)[::-1]
    u = np.where(np.isfinite(u), u, 0.0)
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    ind = int(np.where(cond)[0][-1])
    theta = (cssv[ind] - 1.0) / float(ind + 1)
    w = v.astype(np.float64) - theta
    w = np.where(np.isfinite(w), w, 0.0)
    np.maximum(w, 0.0, out=w)
    s = float(w.sum())
    if s <= 1e-18:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


def lse_weights(c, beta):
    """Softmax weights exp(beta*(c-max(c))) / sum, stable."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c.copy()
    m = float(np.max(c))
    if beta <= 1e-14:
        return np.full(c.size, 1.0 / c.size, dtype=np.float64)
    z = beta * (c - m)
    z = np.where(np.isfinite(z), z, -1000.0)
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-18:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def compute_grad(p, weights):
    """
    Gradient wrt p of sum_s weights[s] * (p*p)[s].
    Returns 2 * sum_s weights[s] * p_{s-i}.
    """
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    w = np.asarray(weights, dtype=np.float64).ravel()
    if w.size < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:w.size] = w
        w = tmp
    else:
        w = w[:L]
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(w.sum())
    if s <= 1e-18:
        return np.zeros(n, dtype=np.float64)
    rev = p[::-1].copy()
    total_len = L + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(rev, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    g = conv[n - 1:n - 1 + n]
    g = 2.0 * np.real(g)
    g[~np.isfinite(g)] = 0.0
    g = np.maximum(g, 0.0)
    return g


def make_weights(c, T, w_min=1e-6):

            w[int(np.argmax(c))] = 1.0

        cols_list.append(cols)

    Line search for alpha in [0,1] minimizing
        max_k [(1-alpha)^2 c_k + 2 alpha(1-alpha) h_k + alpha^2 cg_k]
    where c=p*p, h=p*g, cg=g*g.

    The function is a maximum of quadratics and may be nonconvex, so we use a
    dense global grid plus a local refinement.
    """
    c = np.asarray(c, dtype=np.float64)
    h = np.asarray(h, dtype=np.float64)
    cg = np.asarray(cg, dtype=np.float64)
    lin = 2.0 * (h - c)
    quad = c - 2.0 * h + cg

    grid = np.linspace(0.0, 1.0, 513, dtype=np.float64)
    th = grid[:, None]
    vals = np.maximum(0.0, c[None, :] + th * lin[None, :] + th * th * quad[None, :])
    vals = vals.max(axis=1)
    idx = int(np.argmin(vals))
    best_a = float(grid[idx])
    best_v = float(vals[idx])

    lo = max(0.0, best_a - 0.012)
    hi = min(1.0, best_a + 0.012)
    grid2 = np.linspace(lo, hi, 129, dtype=np.float64)
    th2 = grid2[:, None]
    vals2 = np.maximum(0.0, c[None, :] + th2 * lin[None, :] + th2 * th2 * quad[None, :])
    vals2 = vals2.max(axis=1)
    idx2 = int(np.argmin(vals2))
    if float(vals2[idx2]) < best_v:
        best_a = float(grid2[idx2])
        best_v = float(vals2[idx2])

    # Ensure endpoints are considered (already in grid, but robust).
    v0 = float(np.max(c))
    v1 = float(np.max(cg))
    if v0 < best_v:
        best_a, best_v = 0.0, v0
    if v1 < best_v:
        best_a, best_v = 1.0, v1
    return best_a, best_v


def combine_sequences(p, g, alpha):
    alpha = float(alpha)
    if alpha <= 0.0:
        return p.copy()
    if alpha >= 1.0:
        return g.copy()
    out = (1.0 - alpha) * p + alpha * g
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-18:
        return p.copy()
    return out / s


def update_best_from_M(p, M, tracker):
    """Update tracker [best_score, best_sequence] using internal objective M."""
    p = as_norm_sequence(p)
    n = len(p)
    M = float(M)
    if not np.isfinite(M) or M < 0.0:
        M = 0.0
    score = 2.0 * float(n) * M
    if np.isfinite(score) and score < float(tracker[0]) - 1e-13:
        tracker[0] = float(score)
        tracker[1] = p.copy()
        print(f"  new best: {score:.10f}, n={n}")
    return tracker


def resample_sequence(p, n_new):
    """Interpolate a normalized sequence to a new length."""
    p = as_norm_sequence(p)
    n_old = len(p)
    n_new = int(max(2, n_new))
    if n_old == n_new:
        return p.copy()
    x_old = np.linspace(0.0, 1.0, n_old, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    q = np.interp(x_new, x_old, p)
    return as_norm_sequence(q)


def smooth_step(p, c, M, T):
    """A cheap projected-gradient step using weighted subgradient/LSE weights."""
    p = as_norm_sequence(p)
    c = np.asarray(c, dtype=np.float64)
    M = float(max(M, 1e-300))
    w = make_weights(c, T, w_min=1e-5)
    g = compute_grad(p, w)
    gs = g / M
    gs = gs - float(np.mean(gs))
    mx = float(np.max(np.abs(gs)))
    if not np.isfinite(mx) or mx <= 1e-18:
        return p, M

    d = -gs / mx
    best_p = p
    best_M = M
    for eta in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        q = project_simplex(p + eta * d)
        c_q = fast_convolve(q)
        Mq = float(np.max(c_q))
        if Mq < best_M - 1e-13:
            best_M = Mq
            best_p = q
    return best_p, best_M


def smooth_optimize(p0, time_limit, tracker, beta_start=30.0, beta_end=100000.0,
                    lr_start=0.08, seed=12345):
    """
    Projected Adam-style smoothing on the simplex.
    Uses LSE weights and descended on exact max coefficient.
    """
    p = as_norm_sequence(p0)
    n = len(p)
    if time_limit <= 0.0:
        return p

    c = fast_convolve(p)
    M = float(np.max(c))
    update_best_from_M(p, M, tracker)
    best_p = p.copy()
    best_M = M

    beta = float(beta_start)
    lr = float(lr_start)
    b1 = 0.90
    b2 = 0.999
    mom = np.zeros(n, dtype=np.float64)
    vv = np.zeros(n, dtype=np.float64)
    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while True:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit:
            break

        c = fast_convolve(p)
        M = float(np.max(c))
        if M < best_M - 1e-13:
            best_M = M
            best_p = p.copy()
            update_best_from_M(best_p, best_M, tracker)

        w = lse_weights(c, beta)
        g = compute_grad(p, w)
        gs = g / max(M, 1e-300)
        gs = gs - float(np.mean(gs))
        mx = float(np.max(np.abs(gs)))
        if not np.isfinite(mx) or mx <= 1e-18:
            beta = min(beta * 1.2, float(beta_end))
            no_improve += 1
            it += 1
            if it % 1000 == 0:
                print(f"    smooth it={it}, M={M:.10f}, beta={beta:.1f}, lr={lr:.6f}")
            continue

        gg = gs / mx
        mom = b1 * mom + (1.0 - b1) * gg
        vv = b2 * vv + (1.0 - b2) * (gg * gg)
        step = mom / (np.sqrt(vv) + 1e-8)

        accepted = False
        for fac in (1.0, 0.5, 0.25, 0.1, 0.05):

                    best_p = p.copy()
                    update_best_from_M(best_p, best_M, tracker)

                no_improve = 0
                accepted = True
                break
```

```python
"""
Step-function coefficient minimization for:
    2*n*max(conv(a,a)) / (sum(a))^2

The program uses:
  * a conditional-gradient (FW-like) solver with a weighted LP oracle that
    minimizes a weighted approximate first-order convolution penalty,
  * exact line search over convex combinations p -> (1-alpha)p + alpha*g,
  * projected Adam polishing with log-sum-exp weighting,
  * small projected-gradient steps and restarts,
  * occasional exploration of interpolated alternative lengths.
"""

import time
import numpy as np
from scipy.optimize import linprog


_LP_TABLES = {}


def as_norm_sequence(p):
    """Convert input to a finite, nonnegative, sum-to-one ndarray."""
    arr = np.asarray(p, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-18:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def fast_convolve(a):
    """Linear auto-convolution a*a by FFT. Clips tiny negative roundoff."""
    n = len(a)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    out = np.real(out)
    out[~np.isfinite(out)] = 0.0
    out = np.maximum(out, 0.0)
    return out


def fast_convolve2(a, b):
    """Linear convolution a*b by FFT. Clips tiny negative roundoff."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.zeros(max(0, n + m - 1), dtype=np.float64)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    B = np.fft.rfft(b.astype(np.float64), fft_size)
    out = np.fft.irfft(A * B, fft_size)[:size]
    out = np.real(out)
    out[~np.isfinite(out)] = 0.0
    out = np.maximum(out, 0.0)
    return out


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        x = float(v[0])
        if not np.isfinite(x):
            x = 0.0
        return np.array([max(x, 0.0)], dtype=np.float64)

    u = np.sort(v)[::-1]
    u = np.where(np.isfinite(u), u, 0.0)
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    ind = int(np.where(cond)[0][-1])
    theta = (cssv[ind] - 1.0) / float(ind + 1)
    w = v.astype(np.float64) - theta
    w = np.where(np.isfinite(w), w, 0.0)
    np.maximum(w, 0.0, out=w)
    s = float(w.sum())
    if s <= 1e-18:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


def lse_weights(c, beta):
    """Stable softmax weights exp(beta*(c-max(c))) / sum."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c.copy()
    m = float(np.max(c))
    if beta <= 1e-14:
        return np.full(c.size, 1.0 / c.size, dtype=np.float64)
    z = beta * (c - m)
    z = np.where(np.isfinite(z), z, -1000.0)
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-18:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def compute_grad(p, weights):
    """
    Gradient wrt p of sum_s weights[s] * (p*p)[s].
    Returns 2 * sum_s weights[s] * p_{s-i}.
    """
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    w = np.asarray(weights, dtype=np.float64).ravel()
    if w.size < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:w.size] = w
        w = tmp
    else:
        w = w[:L]
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(w.sum())
    if s <= 1e-18:
        return np.zeros(n, dtype=np.float64)

    rev = p[::-1].copy()
    total_len = L + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(rev, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    g = conv[n - 1:n - 1 + n]
    g = 2.0 * np.real(g)
    g[~np.isfinite(g)] = 0.0
    g = np.maximum(g, 0.0)
    return g


def make_weights(c, T, w_min=1e-6):
    """Weights peaked near the maximum coefficient, with a small floor."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c.copy()
    M = float(np.max(c))
    if T <= 1e-15 * max(M, 1e-30):
        w = (c >= M - 1e-12 * max(M, 1e-30)).astype(np.float64)
        if float(w.max()) <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
    else:
        z = (c - M) / max(T, 1e-300)
        z = np.where(np.isfinite(z), z, -1000.0)
        e = np.exp(z - np.max(z))
        mx = float(e.max())
        if mx <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / mx
    w = np.maximum(w, float(w_min))
    return w


def get_lp_tables(n):
    """Precompute column/p-index tables for rows of (p*g)_s."""
    if n in _LP_TABLES:
        return _LP_TABLES[n]
    L = 2 * n - 1
    cols_list = []
    pidx_list = []
    for s in range(L):
        start = max(0, s - n + 1)
        end = min(n - 1, s) + 1
        if start < end:
            cols = np.arange(start, end, dtype=np.int64)
            pidx = s - cols
        else:
            cols = np.empty(0, dtype=np.int64)
            pidx = np.empty(0, dtype=np.int64)
        cols_list.append(cols)
        pidx_list.append(pidx)
    _LP_TABLES[n] = (cols_list, pidx_list)
    return _LP_TABLES[n]


def solve_lp_direction(p, c, weights, alpha, lp_time, tiebreak=None):
    """
    Linear oracle:
        minimize t
        s.t.  (1-2a)*w_s*c_s + 2a*w_s*(p*g)_s <= t
              sum(g)=1, g>=0, t>=0

    LP form used:
        2a*w_s*(p*g)_s - t <= -(1-2a)*w_s*c_s

    For alpha=0.5, this minimizes max_s w_s*(p*g)_s.
    """
    try:
        p = np.asarray(p, dtype=np.float64)
        c = np.asarray(c, dtype=np.float64)
        weights = np.asarray(weights, dtype=np.float64)
        n = len(p)
        if n <= 0 or lp_time <= 0.0:
            return None
        L = 2 * n - 1
        if c.size != L or weights.size != L:
            return None

        cmax = float(np.max(c))
        if cmax <= 1e-18:
            return None
        if not np.any(weights > 1e-14):
            return None

        alpha = min(0.499999, max(1e-3, float(alpha)))
        coeff = 2.0 * alpha
        const = 1.0 - 2.0 * alpha

        # Scale weights so LP objective/cutoffs are O(100), not O(1/n).
        scale = max(1.0, 100.0 / cmax)

        cols_list, pidx_list = get_lp_tables(n)
        A_ub = np.zeros((L, n + 1), dtype=np.float64)
        b_ub = np.zeros(L, dtype=np.float64)

        for s in range(L):
            ws = float(weights[s]) * scale
            A_ub[s, n] = -1.0
            if ws > 1e-12:
                cols = cols_list[s]
                pidx = pidx_list[s]
                if cols.size > 0:
                    A_ub[s, cols] = coeff * ws * p[pidx]
                b_ub[s] = -const * ws * float(c[s])

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        bounds = [(0.0, None)] * n
        bounds.append((0.0, None))

        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[n] = 1.0
        if tiebreak is not None:
            tb = np.asarray(tiebreak, dtype=np.float64)
            tb = np.where(np.isfinite(tb), tb, 0.0)
            if tb.size == n and float(np.sum(np.abs(tb))) > 0.0:
                c_obj[:n] = -1e-3 * tb

        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            disp=False,
            options={"presolve": True, "time_limit": float(lp_time)},
        )

        if res is None or getattr(res, "success", False) is False:
            return None

        g = np.asarray(res.x[:n], dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        sg = float(g.sum())
        if sg <= 1e-15:
            return None
        return g / sg
    except Exception:
        return None


def line_search_combo(c, h, cg):
    """
    Line search for alpha in [0,1] minimizing
        max_k [(1-alpha)^2 c_k + 2 alpha(1-alpha) h_k + alpha^2 cg_k]
    where c=p*p, h=p*g, cg=g*g.

    The objective is a max of quadratics; it need not be globally convex,
    hence dense grid + local refinement.
    """
    c = np.asarray(c, dtype=np.float64)
    h = np.asarray(h, dtype=np.float64)
    cg = np.asarray(cg, dtype=np.float64)

    lin = 2.0 * (h - c)
    quad = c - 2.0 * h + cg

    grid = np.linspace(0.0, 1.0, 513, dtype=np.float64)
    th = grid[:, None]
    vals = np.maximum(0.0, c[None, :] + th * lin[None, :] + th * th * quad[None, :])
    vals = vals.max(axis=1)
    idx = int(np.argmin(vals))
    best_a = float(grid[idx])
    best_v = float(vals[idx])

    lo = max(0.0, best_a - 0.012)
    hi = min(1.0, best_a + 0.012)
    grid2 = np.linspace(lo, hi, 129, dtype=np.float64)
    th2 = grid2[:, None]
    vals2 = np.maximum(0.0, c[None, :] + th2 * lin[None, :] + th2 * th2 * quad[None, :])
    vals2 = vals2.max(axis=1)
    idx2 = int(np.argmin(vals2))
    if float(vals2[idx2]) < best_v:
        best_a = float(grid2[idx2])
        best_v = float(vals2[idx2])

    # Robust endpoint check.
    v0 = float(np.max(c))
    v1 = float(np.max(cg))
    if v0 < best_v:
        best_a, best_v = 0.0, v0
    if v1 < best_v:
        best_a, best_v = 1.0, v1
    return best_a, best_v


def combine_sequences(p, g, alpha):
    alpha = float(alpha)
    if alpha <= 0.0:
        return p.copy()
    if alpha >= 1.0:
        return g.copy()
    out = (1.0 - alpha) * p + alpha * g
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-18:
        return p.copy()
    return out / s


def update_best_from_M(p, M, tracker):
    """Update tracker [best_score, best_sequence] from M=max(conv(p,p))."""
    p = as_norm_sequence(p)
    n = len(p)
    M = float(M)
    if not np.isfinite(M) or M < 0.0:
        M = 0.0
    score = 2.0 * float(n) * M
    if np.isfinite(score) and score < float(tracker[0]) - 1e-13:
        tracker[0] = float(score)
        tracker[1] = p.copy()
        print(f"  new best: {score:.10f}, n={n}")
    return tracker


def resample_sequence(p, n_new):
    """Interpolate a normalized sequence to a new length."""
    p = as_norm_sequence(p)
    n_old = len(p)
    n_new = int(max(2, n_new))
    if n_old == n_new:
        return p.copy()
    x_old = np.linspace(0.0, 1.0, n_old, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    q = np.interp(x_new, x_old, p)
    return as_norm_sequence(q)


def smooth_step(p, c, M, T):
    """A cheap projected-gradient step using weighted subgradient."""
    p = as_norm_sequence(p)
    c = np.asarray(c, dtype=np.float64)
    M = float(max(M, 1e-300))
    w = make_weights(c, T, w_min=1e-5)
    g = compute_grad(p, w)
    gs = g / M
    gs = gs - float(np.mean(gs))
    mx = float(np.max(np.abs(gs)))
    if not np.isfinite(mx) or mx <= 1e-18:
        return p, M

    d = -gs / mx
    best_p = p
    best_M = M
    for eta in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        q = project_simplex(p + eta * d)
        c_q = fast_convolve(q)
        Mq = float(np.max(c_q))
        if Mq < best_M - 1e-13:
            best_M = Mq
            best_p = q
    return best_p, best_M


def smooth_optimize(p0, time_limit, tracker, beta_start=30.0, beta_end=100000.0,
                    lr_start=0.08, seed=12345):
    """Projected Adam polishing using LSE weights and exact max descent."""
    p = as_norm_sequence(p0)
    n = len(p)
    if time_limit <= 0.0:
        return p

    c = fast_convolve(p)
    M = float(np.max(c))
    update_best_from_M(p, M, tracker)
    best_p = p.copy()
    best_M = M

    beta = float(beta_start)
    lr = float(lr_start)
    b1 = 0.90
    b2 = 0.999
    mom = np.zeros(n, dtype=np.float64)
    vv = np.zeros(n, dtype=np.float64)
    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while True:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit:
            break

        c = fast_convolve(p)
        M = float(np.max(c))
        if M < best_M - 1e-13:
            best_M = M
            best_p = p.copy()
            update_best_from_M(best_p, best_M, tracker)

        w = lse_weights(c, beta)
        g = compute_grad(p, w)
        gs = g / max(M, 1e-300)
        gs = gs - float(np.mean(gs))
        mx = float(np.max(np.abs(gs)))

        if not np.isfinite(mx) or mx <= 1e-18:
            beta = min(beta * 1.2, float(beta_end))
            no_improve += 1
            it += 1
            if it % 1000 == 0:
                print(f"    smooth it={it}, M={M:.10f}, beta={beta:.1f}, lr={lr:.6f}")
            continue

        gg = gs / mx
        mom = b1 * mom + (1.0 - b1) * gg
        vv = b2 * vv + (1.0 - b2) * (gg * gg)
        step = mom / (np.sqrt(vv) + 1e-8)

        accepted = False
        for fac in (1.0, 0.5, 0.25, 0.1, 0.05):
            q = project_simplex(p - lr * fac * step)
            c_q = fast_convolve(q)
            Mq = float(np.max(c_q))
            if Mq < M - 1e-12:
                p = q
                M = Mq
                if M < best_M - 1e-13:
                    best_M = M
                    best_p = p.copy()
                    update_best_from_M(best_p, best_M, tracker)
                lr = min(lr * 1.02, 0.5)
                no_improve = 0
                accepted = True
                break

        if not accepted:
            no_improve += 1
            lr = max(lr * 0.75, 1e-4)
            if no_improve >= 40:
                scale = 0.002 + 0.004 * min(1.0, float(no_improve) / 200.0)
                p = project_simplex(best_p + rng.standard_normal(n) * scale)
                mom = np.zeros(n, dtype=np.float64)
                vv = np.zeros(n, dtype=np.float64)
                no_improve = 0
                beta = min(beta * 1.5, float(beta_end))
                c = fast_convolve(p)
                M = float(np.max(c))

        beta = min(beta * 1.0015, float(beta_end))
        it += 1
        if it % 1000 == 0:
            print(
                f"    smooth it={it}, M={M:.10f}, best_M={best_M:.10f}, "
                f"beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s"
            )

    return best_p


def lp_fw_search(p0, time_limit, tracker, seed=1, lp_time=2.0):
    """
    Main conditional-gradient search.

    Repeatedly:
      1. compute current c and M,
      2. build weights emphasizing large coefficients,
      3. solve an LP for a direction g on the simplex,
      4. line-search over the segment p to g exactly,
      5. if stuck, use smooth projected steps and random restarts.
    """
    p = as_norm_sequence(p0)
    n = len(p)
    if time_limit <= 0.0:
        return p

    c = fast_convolve(p)
    M = float(np.max(c))
    update_best_from_M(p, M, tracker)
    best_p = p.copy()
    best_M = M

    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()
    T_frac = 0.12
    alpha_lp = 0.38
    no_improve = 0
    it = 0

    while True:
        elapsed = time.monotonic() - t0
        rem = time_limit - elapsed
        if rem <= 3.0:
            break

        c = fast_convolve(p)
        M = float(np.max(c))
        if M < best_M - 1e-13:
            best_M = M
            best_p = p.copy()
            update_best_from_M(best_p, best_M, tracker)

        # If we drift far from the incumbent, snap back.
        if no_improve > 18 and M > best_M + 0.02 * max(best_M, 1e-12):
            p = best_p.copy()
            c = fast_convolve(p)
            M = best_M
            no_improve = 0
            T_frac = 0.12
            alpha_lp = 0.38
            continue

        T = T_frac * max(M, 1e-300)
        weights = make_weights(c, T, w_min=1e-6)

        improved = False
        trials = []

        # A mild tiebreak normally points toward p; when stuck, randomize.
        tie_a = p if no_improve <= 12 else rng.standard_normal(n)
        trials.append((alpha_lp, weights, tie_a))

        if no_improve < 4:
            trials.append((0.50, weights, tie_a))
        elif no_improve < 12:
            trials.append((0.22, weights, tie_a))
        else:
            T2 = min(0.25 * M, T * 1.8)
            w2 = make_weights(c, T2, w_min=1e-6)
            trials.append((0.50, w2, p))

        for tri in trials:
            elapsed = time.monotonic() - t0
            if time_limit - elapsed <= 4.0:
                break

            alpha_try, w_try, tie = tri
            g = solve_lp_direction(
                p, c, w_try, alpha_try,
                min(float(lp_time), time_limit - elapsed - 2.0),
                tiebreak=tie
            )
            if g is None:
                continue

            h = fast_convolve2(p, g)
            cg = fast_convolve(g)
            alpha_star, val = line_search_combo(c, h, cg)

            if val < best_M - 1e-13:
                p_new = combine_sequences(p, g, alpha_star)
                c_new = fast_convolve(p_new)
                M_new = float(np.max(c_new))
                if M_new < best_M - 1e-12:
                    p = p_new
                    M = M_new
                    best_M = M
                    best_p = p.copy()
                    update_best_from_M(best_p, best_M, tracker)

                    no_improve = 0
                    T_frac = max(0.025, T_frac * 0.97)
                    alpha_lp = alpha_try
                    improved = True
                    break

        if improved:
            it += 1
            if it % 100 == 0:
                print(
                    f"    LP-FW it={it}, M={M:.10f}, best_M={best_M:.10f}, "
                    f"T_frac={T_frac:.3f}, alpha={alpha_lp:.3f}, elapsed={time.monotonic()-t0:.0f}s"
                )
            continue

        no_improve += 1

        if no_improve <= 3:
            T_frac = min(0.25, T_frac * 1.25)
        elif no_improve <= 8:
            alpha_lp = 0.50
        elif no_improve <= 15:
            alpha_lp = 0.18
            T_frac = 0.05 + 0.10 * rng.random()
        elif no_improve <= 35:
            T_smooth = max(0.02 * M, T)
            p2, M2 = smooth_step(p, c, M, T_smooth)
            if M2 < M - 1e-13:
                p = p2
                M = M2
                if M2 < best_M - 1e-13:
                    best_M = M2
                    best_p = p.copy()
                    update_best_from_M(best_p, best_M, tracker)
                no_improve = 0
                T_frac = min(0.20, T_frac * 1.1)
            else:
                scale = 0.002 + 0.004 * min(1.0, float(no_improve) / 50.0)
                p = project_simplex(best_p + rng.standard_normal(n) * scale)
                c = fast_convolve(p)
                M = float(np.max(c))
                no_improve = 0
                T_frac = 0.12
                alpha_lp = 0.38
        else:
            scale = 0.004 + 0.008 * min(1.0, float(no_improve) / 200.0)
            p = project_simplex(best_p + rng.standard_normal(n) * scale)
            c = fast_convolve(p)
            M = float(np.max(c))
            no_improve = 0
            T_frac = 0.14
            alpha_lp = 0.38

        it += 1
        if it % 100 == 0:
            print(
                f"    LP-FW it={it}, M={M:.10f}, best_M={best_M:.10f}, "
                f"no_improve={no_improve}, elapsed={time.monotonic()-t0:.0f}s"
            )

    return best_p


def propose_candidate(seed=42, budget_s=1800):
    """Main entry point. Returns the best sequence found as a Python list."""
    t_start = time.monotonic()
    budget = max(10.0, float(budget_s) - 30.0)
    deadline = t_start + budget

    tracker = [float("inf"), None]

    # Load parent if available.
    parent = None
    try:
        seq = height_sequence_1
        p = np.asarray(seq, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and float(p.sum()) > 1e-12:
            parent = p / float(p.sum())
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(448, dtype=np.float64) / 448.0

    M0 = float(np.max(fast_convolve(parent)))
    update_best_from_M(parent, M0, tracker)
    print(f"Initial best: {tracker[0]:.10f}, n={len(parent)}")

    cur = tracker[1] if tracker[1] is not None else parent

    # Phase 1: main LP-FW on the best current length.
    rem = deadline - time.monotonic()
    main_t = min(800.0, rem - 500.0)
    if main_t > 60.0:
        print("Phase 1: main LP-FW")
        lp_fw_search(cur, main_t, tracker, seed=int(seed) + 1, lp_time=2.0)

    cur = tracker[1] if tracker[1] is not None else parent

    # Phase 2: LSE/Adam polishing.
    rem = deadline - time.monotonic()
    smooth_t = min(300.0, rem - 300.0)
    if smooth_t > 30.0:
        print("Phase 2: smooth polishing")
        smooth_optimize(
            cur, smooth_t, tracker,
            beta_start=30.0, beta_end=100000.0,
            lr_start=0.08, seed=int(seed) + 2
        )

    cur = tracker[1] if tracker[1] is not None else parent

    # Phase 3: second LP-FW refinement.
    rem = deadline - time.monotonic()
    lp2_t = min(250.0, rem - 150.0)
    if lp2_t > 20.0:
        print("Phase 3: LP-FW refinement")
        lp_fw_search(cur, lp2_t, tracker, seed=int(seed) + 3, lp_time=1.6)

    cur = tracker[1] if tracker[1] is not None else parent

    # Phase 4: explore one alternative length if still needed and time remains.
    rem = deadline - time.monotonic()
    if rem > 250.0 and tracker[0] > 1.5028:
        n_cur = len(cur)
        if n_cur <= 448:
            n_alt = 512
        else:
            n_alt = 640

        print(f"Phase 4: alternative length {n_alt}")
        alt = resample_sequence(cur, n_alt)
        M_alt = float(np.max(fast_convolve(alt)))
        update_best_from_M(alt, M_alt, tracker)

        cur = tracker[1] if tracker[1] is not None else alt
        rem = deadline - time.monotonic()
        alt_lp = min(120.0, rem - 120.0)
        if alt_lp > 30.0:
            lp_fw_search(cur, alt_lp, tracker, seed=int(seed) + 4, lp_time=1.2)

        cur = tracker[1] if tracker[1] is not None else cur
        rem = deadline - time.monotonic()
        alt_sm = min(100.0, rem - 60.0)
        if alt_sm > 20.0:
            smooth_optimize(
                cur, alt_sm, tracker,
                beta_start=50.0, beta_end=80000.0,
                lr_start=0.07, seed=int(seed) + 5
            )

    cur = tracker[1] if tracker[1] is not None else parent

    # Phase 5: final short LP polish on global best.
    rem = deadline - time.monotonic()
    final_t = min(45.0, rem - 8.0)
    if final_t > 10.0:
        print("Phase 5: final polish")
        lp_fw_search(cur, final_t, tracker, seed=int(seed) + 6, lp_time=1.0)

    fin = tracker[1] if tracker[1] is not None else parent
    fin = np.asarray(fin, dtype=np.float64)
    fin = np.where(np.isfinite(fin), fin, 0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    s = float(fin.sum())
    if s <= 1e-12:
        fin = np.ones(448, dtype=np.float64) / 448.0
        s = 1.0
    fin = fin / s

    # Final verification and safe fallback.
    try:
        fs = float(evaluate_sequence(fin.tolist()))
        if not np.isfinite(fs):
            raise ValueError("non-finite final score")
        print(f"Final verified score: {fs:.10f}, n={len(fin)}")
    except Exception:
        fin = np.ones(448, dtype=np.float64) / 448.0
        print("Final verification failed; returning uniform fallback.")

    return [float(x) for x in fin]
```
