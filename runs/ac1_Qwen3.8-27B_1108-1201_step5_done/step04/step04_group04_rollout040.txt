import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_sparse_lp(f, M, top_K=40):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for top K lags, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    top_idx = np.argpartition(conv, -top_K)[-top_K:]
    top_idx = np.sort(top_idx)

    c = -np.ones(n)
    A_ub = np.zeros((len(top_idx), n))
    for idx, k in enumerate(top_idx):
        ls = max(0, k - n + 1)
        le = min(n - 1, k)
        A_ub[idx, ls:le + 1] = f[le - ls::-1]
    b_ub = np.full(len(top_idx), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def solve_full_lp(f, M):
    """Solve LP with all constraints (fallback for small n)."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        ls = max(0, k - n + 1)
        le = min(n - 1, k)
        A_ub[k, ls:le + 1] = f[le - ls::-1]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def smoothed_gradient_fft(conv, a, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def fw_sparse_lp(f0, time_limit, top_K=40, verbose=False):
    """Frank-Wolfe with sparse LP oracle and adaptive K."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    cur_K = top_K

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_sparse_lp(f, M * m_relax, top_K=cur_K)
        if g is None:
            if cur_K < 2 * n - 1:
                cur_K = min(cur_K + 10, 2 * n - 1)
            else:
                m_relax = min(m_relax * 1.05, 2.0)
                if m_relax > 1.5:
                    g = solve_full_lp(f, M * 1.1)
                    if g is None:
                        m_relax = 1.0
                        cur_K = top_K
                        continue
            if g is not None:
                pass
            else:
                m_relax = 1.0
                cur_K = top_K
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.02, 1.5)
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.02:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            ss = f.sum()
            if ss < 1e-15:
                break
            f = f / ss
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            cur_K = top_K

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      FW sparse iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 1.5)

        if no_improve > 25:
            rng = np.random.default_rng((it * 7919 + n * 31) % (2**31))
            scale = 0.003 + 0.002 * min(no_improve // 25, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            cur_K = top_K
            no_improve = 0

    return best_f, best_obj


def mirror_descent(f0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) on the simplex."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eta = 0.5
    top_k = max(3, n // 16)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute weighted subgradient from top peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Soft weighting: weight proportional to how close to max
        weights = np.maximum(pv - M * 0.95, 0)
        if weights.sum() < 1e-15:
            weights = np.zeros(top_k)
            weights[np.argmax(pv)] = 1.0
        else:
            weights = weights / weights.sum()

        g = np.zeros(n)
        for wi, ki in zip(weights, peaks):
            if wi < 1e-12:
                continue
            k = int(ki)
            ls = max(0, k - n + 1)
            le = min(n - 1, k)
            g[ls:le + 1] += wi * 2.0 * f[le - ls::-1]

        # Multiplicative update
        f_new = f * np.exp(-eta * g)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"      MD iter {it}: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            f = f_new  # still move, just track best
            if no_improve % 100 == 0:
                eta *= 0.9

        if it % 50 == 0:
            eta = max(eta * 0.995, 0.01)

        if no_improve > 300:
            rng = np.random.default_rng((it * 131 + 7) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            eta = max(eta * 2, 0.1)
            no_improve = 0

    return best_f, best_obj


def smoothed_gd(f0, time_limit, T_init=0.002, T_min=0.00005, verbose=False):
    """Smoothed gradient descent with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        grad = smoothed_gradient_fft(conv, f, T)
        grad = grad - np.mean(grad)

        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"      SGD iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            f = f_new
            if no_improve % 30 == 0:
                lr = max(lr * 0.9, 0.0001)

        T = max(T * 0.996, T_min)

        if no_improve > 250:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = max(lr * 1.3, 0.002)
            T = max(T_init * 0.2, T_min)
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, top_k_frac=0.06, temp=0.00003):
    """Subgradient polish focused on top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(5, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            ls = max(0, k - n + 1)
            le = min(n - 1, k)
            sg[ls:le + 1] += 2.0 * w[i] * f[le - ls::-1]

        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 60 == 0:
            lr *= 0.995
            temp *= 1.01

        if no_improve > 200:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            lr = max(lr, 0.0003)
            no_improve = 0

    return best_f, best_obj


def symmetrize(f):
    """Make a sequence symmetric and normalize."""
    n = len(f)
    sym = 0.5 * (f + f[::-1])
    s = sym.sum()
    if s < 1e-15:
        return f.copy()
    return sym / s


def generate_inits(n, parent, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated (U-shape) - good for this problem
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Broader U-shape
    edge_w2 = 1.0 / (np.minimum(x, n - 1 - x) ** 0.5 + 1.0)
    inits.append(edge_w2 / edge_w2.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two humps (symmetric)
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    two_hump = h1 + h2
    inits.append(two_hump / two_hump.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Symmetrize all
    inits = [symmetrize(a) for a in inits]
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining sparse LP-FW,
    mirror descent, smoothed gradient, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, obj_estimate={fast_convolve(parent).max()*2*len(parent):.6f}")
    except Exception:
        pass

    # === Phase 1: Exploration (250s) ===
    print("=== Phase 1: Exploration ===")
    t_start = time.monotonic()
    phase1_budget = 250.0

    base_n = len(parent) if parent is not None else 368
    lengths_to_try = [base_n]
    for delta in [8, -8, 16, -16, 4, -4, 24, -24, 32]:
        cand = base_n + delta
        if 256 <= cand <= 512 and cand not in lengths_to_try:
            lengths_to_try.append(cand)

    rng_shuffled = lengths_to_try[1:]
    rng.shuffle(rng_shuffled)

    all_results = {}

    # Try parent length first with multiple inits
    if parent is not None:
        n = base_n
        inits = generate_inits(n, parent, rng)
        best_init = None
        best_init_obj = float('inf')
        for fi in inits:
            obj = fast_convolve(fi).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = fi
        t_alloc = min(60.0, phase1_budget - 30.0)
        print(f"  FW sparse n={n}, t={t_alloc:.0f}s")
        a_opt, _ = fw_sparse_lp(best_init, time_limit=t_alloc, top_K=50)
        try_seq(a_opt, "p1_parent_fw")
        all_results[n] = a_opt

    # Try other lengths
    for n in rng_shuffled[:3]:
        rem = phase1_budget - (time.monotonic() - t_start)
        if rem < 30:
            break
        t_alloc = min(40.0, rem - 15.0)
        if t_alloc < 20:
            break
        inits = generate_inits(n, parent, rng)
        best_init = None
        best_init_obj = float('inf')
        for fi in inits:
            obj = fast_convolve(fi).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = fi
        print(f"  FW sparse n={n}, t={t_alloc:.0f}s")
        a_opt, _ = fw_sparse_lp(best_init, time_limit=t_alloc, top_K=40)
        try_seq(a_opt, f"p1_n{n}")
        all_results[n] = a_opt

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Deep optimization (950s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()
    phase2_budget = 950.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        f_work = symmetrize(f_work)
        n_work = len(f_work)

        n_cycles = 8
        t_per_cycle = phase2_budget / n_cycles

        for cycle in range(n_cycles):
            rem = phase2_budget - (time.monotonic() - t_start2)
            if rem < 20:
                break

            # Sub-phase A: Sparse LP-FW
            t_a = min(t_per_cycle * 0.4, rem * 0.4)
            if t_a > 15:
                print(f"  Cycle {cycle}: FW sparse (t={t_a:.0f}s)")
                a_opt, _ = fw_sparse_lp(f_work, time_limit=t_a, top_K=50)
                try_seq(a_opt, f"p2_fw_{cycle}")
                if best_seq is not None:
                    f_work = symmetrize(best_seq / best_seq.sum())

            # Sub-phase B: Mirror descent
            rem = phase2_budget - (time.monotonic() - t_start2)
            t_b = min(t_per_cycle * 0.25, rem * 0.25)
            if t_b > 10:
                print(f"  Cycle {cycle}: Mirror descent (t={t_b:.0f}s)")
                a_opt, _ = mirror_descent(f_work, time_limit=t_b)
                try_seq(a_opt, f"p2_md_{cycle}")
                if best_seq is not None:
                    f_work = symmetrize(best_seq / best_seq.sum())

            # Sub-phase C: Smoothed gradient
            rem = phase2_budget - (time.monotonic() - t_start2)
            t_c = min(t_per_cycle * 0.25, rem * 0.25)
            if t_c > 10:
                T0 = max(0.001, 0.003 * fast_convolve(f_work).max())
                print(f"  Cycle {cycle}: Smoothed GD (t={t_c:.0f}s, T={T0:.5f})")
                a_opt, _ = smoothed_gd(f_work, time_limit=t_c, T_init=T0)
                try_seq(a_opt, f"p2_sgd_{cycle}")
                if best_seq is not None:
                    f_work = symmetrize(best_seq / best_seq.sum())

            # Sub-phase D: Try slight length change
            rem = phase2_budget - (time.monotonic() - t_start2)
            if rem > 25 and cycle % 3 == 2:
                for delta in [4, -4]:
                    new_n = n_work + delta
                    if 256 <= new_n <= 512:
                        px = np.interp(np.linspace(0, 1, new_n),
                                       np.linspace(0, 1, n_work), f_work)
                        px = project_simplex(px)
                        px = symmetrize(px)
                        t_d = min(20.0, rem / 2.0)
                        a_opt, _ = fw_sparse_lp(px, time_limit=t_d, top_K=40)
                        try_seq(a_opt, f"p2_len{new_n}_c{cycle}")
                        rem = phase2_budget - (time.monotonic() - t_start2)
                        if rem < 15:
                            break

    print(f"  Phase 2 best: {best_score:.6f}")

    # === Phase 3: Fine polish (350s) ===
    print(f"\n=== Phase 3: Fine Polish ===")
    t_start3 = time.monotonic()
    phase3_budget = 350.0

    if best_seq is not None:
        f_work = symmetrize(best_seq / best_seq.sum())

        # Multiple rounds of subgradient polish
        configs = [
            {"lr": 0.0008, "top_k_frac": 0.04, "temp": 0.00002},
            {"lr": 0.0005, "top_k_frac": 0.03, "temp": 0.00001},
            {"lr": 0.0003, "top_k_frac": 0.02, "temp": 0.000005},
            {"lr": 0.0002, "top_k_frac": 0.02, "temp": 0.000003},
        ]

        for i, cfg in enumerate(configs):
            rem = phase3_budget - (time.monotonic() - t_start3)
            if rem < 25:
                break
            t_alloc = min(60.0, rem - 15.0)
            print(f"  SG polish {i} (t={t_alloc:.0f}s)")
            a_opt, _ = sg_polish(f_work, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"p3_sg_{i}")
            if best_seq is not None:
                f_work = symmetrize(best_seq / best_seq.sum())

        # Very fine smoothed gradient
        rem = phase3_budget - (time.monotonic() - t_start3)
        if rem > 30:
            T_init = max(0.0002, 0.0008 * fast_convolve(f_work).max())
            t_alloc = min(80.0, rem - 15.0)
            print(f"  Fine SGD (t={t_alloc:.0f}s, T={T_init:.6f})")
            a_opt, _ = smoothed_gd(f_work, time_limit=t_alloc,
                                    T_init=T_init, T_min=0.00001)
            try_seq(a_opt, "p3_fine_sgd")
            if best_seq is not None:
                f_work = symmetrize(best_seq / best_seq.sum())

        # Final mirror descent
        rem = phase3_budget - (time.monotonic() - t_start3)
        if rem > 20:
            t_alloc = min(50.0, rem - 10.0)
            print(f"  Final MD (t={t_alloc:.0f}s)")
            a_opt, _ = mirror_descent(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_final_md")

    print(f"  Phase 3 best: {best_score:.6f}")

    # === Phase 4: Final tricks (150s) ===
    print(f"\n=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        f_work = symmetrize(best_seq / best_seq.sum())
        n_best = len(f_work)

        # Tiny perturbations + quick polish
        for i in range(4):
            if time.monotonic() > deadline - 10:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = symmetrize(project_simplex(f_work + noise))
            a_opt, _ = sg_polish(f_pert, time_limit=15,
                                  lr=0.0002, top_k_frac=0.02, temp=0.000003)
            try_seq(a_opt, f"p4_pert{i}")

        # Quick FW
        if time.monotonic() < deadline - 25:
            a_opt, _ = fw_sparse_lp(f_work, time_limit=20, top_K=60)
            try_seq(a_opt, "p4_fw")

        # Symmetrize + polish
        if time.monotonic() < deadline - 15:
            a_opt, _ = sg_polish(f_work, time_limit=10,
                                  lr=0.0001, top_k_frac=0.01, temp=0.000002)
            try_seq(a_opt, "p4_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
