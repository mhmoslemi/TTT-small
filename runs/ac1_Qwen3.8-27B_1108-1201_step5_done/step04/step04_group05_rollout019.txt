import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_active_lp(f, active_idx, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_idx, g >= 0."""
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = np.zeros((len(active_idx), n))
    for ii, k in enumerate(active_idx):
        for j in range(n):
            j2 = k - j
            if 0 <= j2 < n:
                A_ub[ii, j] = f[j2]
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def solve_full_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            best_obj = M
            best_f = f.copy()
        
        # Identify top peaks
        peak_idx = np.argsort(conv)[-n_peaks:]

        for k in peak_idx:

def sg_polish(f0, time_limit, lr=0.003, mom=0.93, top_k=30, temp=0.00005):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng_local = np.random.default_rng(7777)
    
    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        cur_temp *= 1.002
        
        if no_improve > 350:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.0005)
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent_polish(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)

        improved = False
        order = rng_local.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            candidates = [(j + 1) % n, (j - 1) % n, j ^ (n // 2), rng_local.integers(n)]
            for source in candidates:

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine taper
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())
    
    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)))
    inits.append(peak / peak.sum())
    
    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:

    inits.append(ramp / ramp.sum())

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set Frank-Wolfe with adaptive
LP oracle, peak-targeting heuristics, cosine-mode global exploration, and multi-length

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_active_lp(f, active_idx, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_idx, g >= 0."""
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = np.zeros((len(active_idx), n))
    for ii in range(len(active_idx)):
        k = int(active_idx[ii])
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A_ub[ii, m] = f[j]
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_full_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None
