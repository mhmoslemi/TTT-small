def soft_gradient(a, conv, top_k=50, temp=0.001):

def lp_solve_direction(a, M, C):

def generate_initializations(n, parent, rng):
    """Generate diverse initializations for optimization."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sine-based
    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine1'] = s1 / s1.sum()

    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.01
    inits['sine2'] = s2 / s2.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Tapered (raised cosine window)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc = np.maximum(rc, 0.01)
    inits['tapered'] = rc / rc.sum()

    # Two-lobe
    lobe1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    lobe2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    inits['twolobe'] = (lobe1 + lobe2) / (lobe1.sum() + lobe2.sum())

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits

        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  NEW BEST: score={score:.8f}, n={len(a_opt)}")
        return score

    # =========================================================
    # Phase 1: Quick gradient scan across lengths
    # =========================================================
    print("=== Phase 1: Multi-length Gradient Scan ===")
    t_start = time.monotonic()
    phase1_budget = 300.0

    lengths = [768, 1024, 1536]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)
    lengths = sorted(set(lengths))[:4]

    scan_results = []
    for n in lengths:
        if time.monotonic() > deadline - 800:
            break
        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'parent_interp', 'uniform', 'tapered', 'sine1', 'random', 'bimodal']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]

        t_per_init = max(8.0, (phase1_budget - 20) / (len(lengths) * max(1, len(init_names))))

        for name in init_names:
            if time.monotonic() > deadline - 700:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 600)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj = nesterov_phase(a0, t_alloc, 0.012, 0.91, 40, 0.002)
            score = update_best(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
            print(f"  Grad n={n}, {name}: obj={obj:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Hybrid deep optimization on top candidates
    # =========================================================
    print("=== Phase 2: Hybrid Deep Optimization ===")
    t_start2 = time.monotonic()

    # Pick top unique-length candidates
    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_n:
            top_configs.append((n, a_opt))
            seen_n.add(n)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None and len(best_sequence) not in seen_n:
        top_configs.append((len(best_sequence), best_sequence.copy()))
        seen_n.add(len(best_sequence))

    remaining = deadline - time.monotonic() - 400.0
    n_cand = max(1, len(top_configs))
    t_per = max(30.0, remaining * 0.7 / n_cand)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 300)
        if t_alloc < 10:
            break
        print(f"  Hybrid candidate {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = hybrid_optimize(a0, t_alloc)
        score = update_best(a_opt)
        print(f"  Hybrid {i} (n={n}): obj={obj:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP deep optimization on best
    # =========================================================
    print("=== Phase 3: LP Deep Optimization ===")
    t_start3 = time.monotonic()
    if best_sequence is not None and len(best_sequence) <= 1536:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 30:
            t_alloc = min(remaining * 0.7, deadline - time.monotonic() - 150)
            if t_alloc > 15:
                print(f"  LP on best (n={len(best_sequence)}), t={t_alloc:.0f}s")
                a_opt, obj = lp_optimize(best_sequence, t_alloc)
                update_best(a_opt)
                print(f"  LP done: obj={obj:.6f}")
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Intensive polishing (gradient + spectral + mirror)
    # =========================================================
    print("=== Phase 4: Intensive Polishing ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        if remaining > 20:
            # Multiple polishing strategies
            strategies = [
                ('nesterov_fine', lambda a, t: nesterov_phase(a, t, 0.003, 0.95, 100, 0.0003, 0.9999, 1.001, 200)),
                ('mirror', lambda a, t: mirror_descent_phase(a, t, 0.003, 60, 0.0005)),
                ('spectral', lambda a, t: spectral_phase(a, t, beta=0.5, max_stall=50)),
                ('nesterov_wide', lambda a, t: nesterov_phase(a, t, 0.008, 0.88, 80, 0.001, 0.9998, 1.003, 150)),
            ]
            t_per_strat = max(5.0, remaining * 0.6 / len(strategies))
            for sname, sfunc in strategies:
                if time.monotonic() > deadline - 40:
                    break
                t_alloc = min(t_per_strat, deadline - time.monotonic() - 30)
                if t_alloc < 5:
                    break
                print(f"  Polish [{sname}], t={t_alloc:.0f}s")
                a_opt, obj = sfunc(best_sequence, t_alloc)
                update_best(a_opt)
                print(f"  [{sname}] done: obj={obj:.6f}")

            # Final micro-gradient
            remaining = deadline - time.monotonic() - 20.0
            if remaining > 5:
                a_opt, obj = nesterov_phase(best_sequence, remaining, 0.001, 0.96, 200, 0.0001,
                                            0.99999, 1.0005, 100)
                update_best(a_opt)
                print(f"  Final micro-grad: obj={obj:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Try different lengths from best (interpolation)
    # =========================================================
    print("=== Phase 5: Length Exploration ===")
    if best_sequence is not None and time.monotonic() < deadline - 60:
        bn = len(best_sequence)
        # Try nearby lengths
        try_lengths = [bn - 32, bn + 32, bn + 64]
        try_lengths = [t for t in try_lengths if t > 64]
        t_per_try = max(5.0, (deadline - time.monotonic() - 30) / max(1, len(try_lengths)))
        for tn in try_lengths:
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(t_per_try, deadline - time.monotonic() - 20)
            if t_alloc < 5:
                break
            # Interpolate best to new length
            a_interp = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, bn), best_sequence / best_sequence.sum())
            a_interp = np.maximum(a_interp, 0)
            si = a_interp.sum()
            if si > 1e-15:
                a_interp = a_interp / si
            a_opt, obj = nesterov_phase(a_interp, t_alloc, 0.008, 0.92, 60, 0.001)
            score = update_best(a_opt)
            print(f"  Try n={tn}: obj={obj:.6f}")

    print(f"Phase 5 done. Best: {best_score:.6f}")

    # =========================================================
    # Fallback and final verification
    # =========================================================
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_sequence = [max(0.0, min(1000.0, float(x))) for x in best_sequence]

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
