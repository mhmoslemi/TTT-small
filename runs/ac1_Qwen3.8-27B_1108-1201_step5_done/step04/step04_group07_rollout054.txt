where M is the current peak (or M * expansion_factor).

This finds a non-negative direction that keeps the convolution bounded.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-based Frank-Wolfe,
adaptive multi-phase search with crossover, and structured perturbations.
Key improvements over baseline: top-k LP constraints, adaptive length search,
crossover between good solutions, and structured restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=None):
    """Solve linearized LP: min t s.t. t >= conv_f[k] + 2*(B*g)[k], g >= -f, sum(g)=0."""
    n = len(f)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        peaks = np.arange(size)
    peaks = np.sort(peaks)
    
    B = build_conv_matrix(f)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M, top_k=None):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k (or top_k), g >= 0."""
    n = len(f)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        # Use all constraints but that's expensive; use top_k of the binding ones
        # Actually for max-sum we need all constraints for correctness
        # Use a relaxed version with top_k most binding constraints
        # For now use all
        pass
    
    c = -np.ones(n)
    A_ub = np.zeros((size, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(size, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def line_search(f, target, current_obj, max_alpha=1.0):
    """Find best alpha for f_new = (1-alpha)*f + alpha*target."""
    best_alpha = 0.0
    best_obj = current_obj
    
    alphas = np.linspace(0.1, max_alpha, 10)
    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * target
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.7)
        hi = min(max_alpha, best_alpha * 1.3)
        for _ in range(10):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    
    return best_alpha, best_obj


def lp_fw_optimize(f0, time_limit, top_k=None, verbose=False):
    """Frank-Wolfe with linearized LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        g = solve_linearized_lp(f, conv, top_k=top_k)
        
        if g is None:
            no_improve += 3
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue
        
        alpha, obj_val = line_search(f, target, current_obj)
        
        if alpha > 1e-8:
            f = (1.0 - alpha) * f + alpha * target
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            
            if obj_val < best_obj - 1e-14:
                best_obj = obj_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LPFW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0
    
    return best_f, best_obj

                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def subgrad_optimize(f0, time_limit, lr=0.002, mom=0.93, top_k=30, temp=0.0001):
    """Smoothed subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9998
        cur_temp *= 1.005
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj


def local_search_opt(f0, time_limit, rng, base_scale=0.003):
    """Local search with adaptive scale."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    scale = base_scale
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()
                improved = True
        
        if improved:
            scale *= 0.98
        else:
            scale *= 0.9
            if scale < 5e-5:
                # Larger jump
                f_jump = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                obj = fast_convolve(f_jump).max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f_jump.copy()
                scale = base_scale * 0.5
    
    return best_f, best_obj


def crossover(f1, f2, alpha):
    """Element-wise crossover between two solutions."""
    n = max(len(f1), len(f2))
    # Interpolate both to length n
    x1 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(f1)), f1)
    x2 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(f2)), f2)
    f = (1 - alpha) * x1 + alpha * x2
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f1.copy()
    return f / s


def generate_structured_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Left-heavy ramp
    ramp = np.linspace(n, 1, n)
    inits.append(ramp / ramp.sum())
    
    # Right-heavy ramp
    ramp = np.linspace(1, n, n)
    inits.append(ramp / ramp.sum())
    
    # V-shape
    v = np.abs(x - n/2) + 1
    inits.append(v / v.sum())
    
    # Inverted V (Gaussian center)
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    inits.append(g / g.sum())
    
    # Squared sine
    s = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits.append(s / s.sum())
    
    # Log-uniform
    l = np.log(x + 1)
    inits.append(l / l.sum())
    
    # Power law
    p = (x + 1) ** 0.5
    inits.append(p / p.sum())
    
    # Two bumps
    b1 = np.exp(-0.5 * ((x - n*0.25) / (n*0.1))**2)
    b2 = np.exp(-0.5 * ((x - n*0.75) / (n*0.1))**2)
    two = b1 + b2
    inits.append(two / two.sum())
    
    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
