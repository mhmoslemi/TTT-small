import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_full_subgradient(f, conv):
    """Compute exact subgradient of max(conv) (using all peaks at max)."""
    n = len(f)
    max_val = conv.max()
    peaks = np.where(conv >= max_val * (1 - 1e-10))[0]
    
    sg = np.zeros(n)
    for k in peaks:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * f[k - idx]
    
    # Normalize by number of peaks (this is the average subgradient)
    sg /= len(peaks)
    return sg

def compute_soft_subgradient(f, conv, top_k=200, temp=0.001):
    """Compute subgradient with softmax weighting over top peaks."""
    n = len(f)
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k_idx = int(peaks[i])
        js = max(0, k_idx - n + 1)
        je = min(n - 1, k_idx)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k_idx - idx]
    
    return sg

def las_optimize(f0, time_limit, n_peaks=50, step_decay=0.999, verbose=False):
    """Linearized Active Set optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    step = 1.0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        # Find active peaks
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        pv = conv[peaks]
        
        # Build linear system: G @ delta = b
        # Row for each peak k: sum_j 2*f[k-j] * delta_j = M_target - c(k)
        # Plus: sum(delta) = 0
        m = len(peaks)
        G = np.zeros((m + 1, n))
        b = np.zeros(m + 1)
        
        for i, k in enumerate(peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            G[i, idx] = 2.0 * f[k - idx]
            b[i] = M - pv[i]  # We want c(k) + G[i]@delta = M, so G[i]@delta = M - c(k)
        
        G[m, :] = 1.0  # sum constraint
        b[m] = 0.0
        
        # Solve least squares
        try:
            delta, residuals, rank, sv = np.linalg.lstsq(G, b, rcond=None)
        except:
            delta = np.zeros(n)
        
        # Update

        f_new = np.maximum(f_new, 0.0)

        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            conv = conv_new
            M = conv.max()
        else:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with log-sum-exp smoothing,
diverse multi-length exploration, and adaptive restart strategies.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    """Compute the objective (before scaling back)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def fista_optimize(f0, time_limit, mu_start=0.01, mu_end=0.0005, 
                   n_peaks=128, top_k_grad=64, lr_base=0.05):
    """FISTA optimization with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    y = f.copy()
    t_k = 1.0
    mu = mu_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_base
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Compute gradient at y using top-k peaks
        conv_y = fast_convolve(y)
        
        # Softmax weights on top peaks
        k_eff = min(top_k_grad, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k_eff)[-k_eff:]
        pv = conv_y[peaks]
        
        scaled = pv / mu
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum
        
        # Gradient: grad_i = 2 * sum_k w_k * y[k-i]
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            ii = np.arange(js, je + 1)
            grad[ii] += 2.0 * w[idx] * y[k - ii]
        
        # FISTA step
        f_new = project_simplex(y - lr * grad)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = f_new + beta * (f_new - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / ys
        f = f_new
        t_k = t_new
        
        # Schedule
        mu = max(mu * 0.995, mu_end)
        lr = lr_base / (1.0 + 0.01 * it)
        
        # Restart if stuck

            y = f.copy()
            t_k = 1.0
            mu = max(mu_start * 0.1, mu_end * 10)
            no_improve = 0
    
    return best_f, best_obj

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute gradient contribution: sum over peaks of w_i * 2 * y[k-i]."""
