"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Spectral equalization initialization (Gerchberg-Saxton inspired)
2. p-norm annealing gradient descent (L2 -> L∞) with BB step sizes
3. Frank-Wolfe with active-set LP (top-K binding constraints only)
4. Coordinate descent polish
5. Multi-restart with diverse initializations and length exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_w(w, a):
    """Compute 2 * sum_k w[k] * a[k-i] for each i (the gradient wrt p-norm).
    w has length 2n-1, a has length n. Returns array of length n.
    """
    n = len(a)
    a_rev = a[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * Ar, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_init(n, n_iters=200, seed=0):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for _ in range(n_iters):
        # FFT of current a (zero-padded to 2n-1)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: flat magnitude (with DC term adjusted for sum=1)
        target_mag = np.ones_like(mag)
        target_mag[0] = np.sqrt(mag[0] ** 2)  # keep DC
        # Blend towards flat
        new_mag = 0.7 * mag + 0.3 * target_mag
        new_mag[0] = mag[0]  # preserve DC (sum)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-15:
            a_new = np.ones(n)
        a = a_new / a_new.sum()
    return a


def pnorm_optimize(f0, time_limit, p_start=2.0, p_end=200.0, lr_init=None, momentum=0.85):
    """p-norm annealing gradient descent with Barzilai-Borwein step sizes."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr = 1.0 / (n ** 0.5)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    p = p_start
    v = np.zeros(n)
    prev_f = None
    prev_grad = None
    bb_lr = lr
    cur_mom = momentum

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Compute p-norm gradient weights: w[k] = c_k^(p-1)
        if p <= 2.0 + 1e-9:
            w = conv.copy()  # p=2: w[k] = c_k
        else:
            # Avoid overflow
            conv_pos = np.maximum(conv, 1e-30)
            w = np.power(conv_pos, p - 1.0)

        grad = compute_grad_w(w, f)

        # BB step size
        if prev_f is not None and prev_grad is not None:
            sf = f - prev_f
            yg = grad - prev_grad
            sy = np.dot(sf, yg)
            ss = np.dot(sf, sf)
            yys = np.dot(yg, yg)
            if ss > 1e-30 and sy > 1e-30:
                bb_new = sy / ss  # BB1
                if yys > 1e-30:
                    bb_new2 = sy / yys  # BB2
                    bb_new = min(bb_new, bb_new2)
                bb_lr = np.clip(bb_new, 0.1 * lr, 10.0 * lr)
            # Blend with current lr
            bb_lr = 0.7 * bb_lr + 0.3 * lr

        # Momentum update
        v = cur_mom * v - bb_lr * grad
        f_new = project_simplex(f + v)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            prev_f = f.copy()
            prev_grad = grad.copy()
            f = f_new
            no_improve = 0
            # Slight lr increase on progress
            lr = min(lr * 1.005, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.93
                cur_mom = max(0.7, cur_mom - 0.005)
            if no_improve > 60:
                # Restart with higher p
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.01 / n)
                cur_mom = 0.82
                p = min(p * 1.8, p_end)
                no_improve = 0

        # Anneal p
        if p < p_end:
            p *= 1.005

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    pnorm it={it}, obj={best_obj:.10f}, p={p:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def fw_active_lp(f0, time_limit, K=80, m_exp_start=1.0):
    """Frank-Wolfe with active-set LP using only top-K binding constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = m_exp_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Find top-K active constraints
        conv = fast_convolve(f)
        M = conv.max()
        K_actual = min(K, 2 * n - 1)
        active_indices = np.argsort(conv)[::-1][:K_actual]

        # Build sparse constraint matrix for active constraints only
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_indices):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                rows.extend([idx] * len(js))
                cols.extend(js.tolist())
                vals.extend((f[k - j] for j in js))

        if not rows:
            break

        A = coo_matrix((vals, (rows, cols)), shape=(K_actual, n)).tocsr()
        c = -np.ones(n)
        b_ub = np.full(K_actual, M * m_exp)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 5})
            if res.success:
                g = res.x
            else:
                g = None
        except Exception:
            g = None

        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
                # Fallback: use subgradient direction
                peak_idx = np.argmax(conv)
                g = np.zeros(n)
                for j in range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)):
                    g[j] = f[peak_idx - j]
                if g.sum() < 1e-15:
                    g = np.ones(n)
            else:
                continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.03, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Refine with bisection
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                m_exp = min(m_exp * 1.04, 3.0)
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.001 + 0.002 * min(1.0, (no_improve - 50) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-LP it={it}, obj={best_obj:.10f}, K={K_actual}, elapsed={elapsed:.0f}s")

    return best_f, best_obj

    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit and sweep < n_sweeps_max:
        sweep += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Try several values for coordinate i
            old_val = f[i]
            best_v = old_val
            best_c = best_obj  # current max conv with this coordinate

            # Candidates: 0, half, double, and some perturbations
            candidates = [0.0, old_val * 0.5, old_val * 1.5,
                          old_val + 0.001, old_val - 0.001]
            # Also try values based on the "ideal" that would equalize peaks
            candidates = [max(0, c) for c in candidates]
            # Add a few random candidates
            rng_i = np.random.default_rng(i + sweep * n)
            candidates.extend([max(0, old_val + rng_i.normal(0, old_val * 0.3)) for _ in range(3)])

            for v in candidates:
                f[i] = v
                st = f.sum()
                if st < 1e-15:
                    f[i] = old_val
                    continue
                # Quick check: compute max conv
                conv_test = fast_convolve(f)
                mc = conv_test.max()
                if mc < best_c - 1e-12:
                    best_c = mc
                    best_v = v

            f[i] = best_v
            st = f.sum()
            if st < 1e-15:
                f[i] = old_val
                continue
            f = f / f.sum()

            if best_v != old_val:
                improved = True

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
        elif not improved:
            break

    return best_f, best_obj


def try_eval(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization with diverse strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_tracker = [float('inf'), None]

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_eval(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        try_eval(parent, best_tracker)

    n_parent = len(parent)

    # === Generate diverse initializations ===
    inits = [parent.copy()]

    # Spectral initialization
    n_spec = n_parent
    a_spec = spectral_init(n_spec, n_iters=300, seed=seed)
    inits.append(a_spec)
    try_eval(a_spec, best_tracker)

    # Cosine window
    x = np.linspace(0, np.pi, n_parent)
    a_cos = np.maximum(np.cos(x / 2) ** 2, 0)
    a_cos = a_cos / a_cos.sum()
    inits.append(a_cos)
    try_eval(a_cos, best_tracker)

    # Random smoothed
    a_rand = rng.dirichlet(np.ones(n_parent) * 0.5)
    # Smooth it
    kernel = np.ones(5) / 5
    a_rand_s = np.convolve(a_rand, kernel, mode='same')
    a_rand_s = np.maximum(a_rand_s, 0)
    a_rand_s = a_rand_s / a_rand_s.sum()
    inits.append(a_rand_s)
    try_eval(a_rand_s, best_tracker)

    # Truncated Gaussian
    xg = np.linspace(-2, 2, n_parent)
    a_gauss = np.exp(-xg ** 2 / 2)
    a_gauss = a_gauss / a_gauss.sum()
    inits.append(a_gauss)
    try_eval(a_gauss, best_tracker)

    # === Phase 1: p-norm annealing on best initialization ===
    print(f"\n=== Phase 1: p-norm annealing ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    t_remaining = deadline - time.monotonic() - 200
    t_alloc = min(400, t_remaining)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s for p-norm")
        a_opt, _ = pnorm_optimize(current_best, time_limit=t_alloc,
                                  p_start=2.0, p_end=100.0)
        try_eval(a_opt, best_tracker)

    # === Phase 2: FW with active LP ===
    print(f"\n=== Phase 2: FW Active LP ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    t_remaining = deadline - time.monotonic() - 150
    t_alloc = min(350, t_remaining)
    if t_alloc > 30:
        K_val = min(60, 2 * n_best - 1)
        print(f"  Allocating {t_alloc:.0f}s, K={K_val}")
        a_opt, _ = fw_active_lp(current_best, time_limit=t_alloc, K=K_val)
        try_eval(a_opt, best_tracker)

    # === Phase 3: p-norm again (higher p) on LP result ===
    print(f"\n=== Phase 3: p-norm polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    t_remaining = deadline - time.monotonic() - 100
    t_alloc = min(200, t_remaining)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = pnorm_optimize(current_best, time_limit=t_alloc,
                                  p_start=50.0, p_end=500.0, momentum=0.80)
        try_eval(a_opt, best_tracker)

    # === Phase 4: FW LP again ===
    print(f"\n=== Phase 4: FW LP refinement ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    t_remaining = deadline - time.monotonic() - 80
    t_alloc = min(200, t_remaining)
    if t_alloc > 20:
        K_val = min(100, 2 * n_best - 1)
        print(f"  Allocating {t_alloc:.0f}s, K={K_val}")
        a_opt, _ = fw_active_lp(current_best, time_limit=t_alloc, K=K_val)
        try_eval(a_opt, best_tracker)

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    t_remaining = deadline - time.monotonic() - 60
    t_alloc = min(120, t_remaining)
    if t_alloc > 15:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = coord_descent(current_best, time_limit=t_alloc)
        try_eval(a_opt, best_tracker)

    # === Phase 6: Multi-restart with perturbations ===
    print(f"\n=== Phase 6: Multi-restart ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    t_remaining = deadline - time.monotonic() - 40
    n_restarts = min(5, max(1, int(t_remaining / 60)))

    for r in range(n_restarts):
        if time.monotonic() > deadline - 30:
            break

        t_alloc = min(80, (deadline - time.monotonic() - 20) / max(1, n_restarts - r))
        if t_alloc < 20:
            break

        # Create perturbed version
        rng_r = np.random.default_rng(seed + 1000 + r * 77)
        scale = 0.003 + 0.004 * r
        f_pert = project_simplex(current_best + rng_r.standard_normal(n_best) * scale)

        # Short p-norm + short FW
        t1 = t_alloc * 0.5
        t2 = t_alloc * 0.5

        a_opt, _ = pnorm_optimize(f_pert, time_limit=t1, p_start=10.0, p_end=200.0)
        try_eval(a_opt, best_tracker)

        current_best = best_tracker[1] if best_tracker[1] is not None else f_pert
        a_opt, _ = fw_active_lp(current_best, time_limit=t2, K=min(50, 2 * n_best - 1))
        try_eval(a_opt, best_tracker)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    alt_lengths = [320, 448, 512, 384]
    if n_parent in alt_lengths:
        alt_lengths.remove(n_parent)
    # Filter out lengths that are too different
    alt_lengths = [al for al in alt_lengths if 0.5 * n_parent <= al <= 2.0 * n_parent][:2]

    for n_alt in alt_lengths:
        if time.monotonic() > deadline - 40:
            break

        t_alloc = min(100, deadline - time.monotonic() - 25)
        if t_alloc < 30:
            break

        print(f"  Trying length {n_alt}, {t_alloc:.0f}s")

        # Interpolate to new length
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        also_spec = spectral_init(n_alt, n_iters=100, seed=seed + n_alt)
        # Pick better of the two
        obj_interp = fast_convolve(f_interp).max()
        obj_spec = fast_convolve(also_spec).max()
        f_start = f_interp if obj_interp < obj_spec else also_spec

        # Optimize
        t1 = t_alloc * 0.45
        t2 = t_alloc * 0.45
        a_opt, _ = pnorm_optimize(f_start, time_limit=t1, p_start=5.0, p_end=100.0)
        try_eval(a_opt, best_tracker)

        cur = best_tracker[1] if best_tracker[1] is not None and len(best_tracker[1]) == n_alt else f_start
        a_opt, _ = fw_active_lp(cur, time_limit=t2, K=min(50, 2 * n_alt - 1))
        try_eval(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
