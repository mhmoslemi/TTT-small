Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smooth optimization:
1. Quick FW-LP initialization

3. Active-set focused multiplicative weight updates

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def convolve_grad(a, k):
    """
    Compute gradient of conv(a,a)[k] with respect to a.
    Result[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n, k + 1)
    # For j in [js, je): g[j] = 2*a[k-j]
    # k-j for j in [js,je) gives range [k-js, k-(je-1)] = [k-js, k-je+1]
    # So a[k-j] = a[k-je+1 : k-js+1] reversed
    g[js:je] = 2.0 * a[k - je + 1 : k - js + 1][::-1]
    return g

def smooth_max_gradient(conv, beta):
    """
    Compute gradient of (1/beta) * log(sum(exp(beta * conv))) with respect to conv.
    This is the softmax weights.
    """
    shifted = beta * conv - beta * conv.max()
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    return weights  # d/d(conv_k) of smooth_max = weights[k]

def solve_lp_for_fw(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[max(0,k-n+1):min(n,k+1)] * g[...]
    # For efficiency, build the matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None

def smooth_gradient_descent(f0, time_limit, beta_start=500, beta_end=5000, 
                            lr_start=0.01, lr_end=0.001):

    Gradient descent on the smooth approximation (1/β)log(sum(exp(β*conv))).
    This is differentiable and we can use the structure for efficient gradient computation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    total_time = time_limit
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < total_time:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        
        # Adaptive beta (temperature) and learning rate
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        conv = fast_convolve(f)
        
        # Compute smooth gradient: grad = sum_k w_k * grad_k where w_k = softmax(beta * conv)
        shifted = beta * conv - beta * conv.max()
        exp_vals = np.exp(shifted)
        weights = exp_vals / exp_vals.sum()
        
        # Compute total gradient
        total_grad = np.zeros(n)
        # Only process peaks with significant weight
        threshold = 1e-10
        active = weights > threshold
        for k in np.where(active)[0]:
            if weights[k] < threshold:
                continue
            total_grad += weights[k] * convolve_grad(f, k)

        # Use mirror descent (multiplicative) for better behavior
        f_new = f * np.exp(-lr * total_grad)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0

def active_set_optimize(f0, time_limit, top_k_range=(5, 30)):
    """
    Optimize focusing on the top-k active peaks.
    Minimizes the average of the top-k peak values using projected gradient.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.005
    it = 0
    v = np.zeros(n)  # momentum
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Determine adaptive top_k
        # Use peaks within a small fraction of the max
        tol = 0.001
        active_peaks = np.where(conv >= M - tol)[0]
        if len(active_peaks) < 3:
            active_peaks = np.argpartition(conv, -5)[-5:]
        elif len(active_peaks) > 50:
            active_peaks = np.argpartition(conv, -20)[-20:]
        
        # Compute gradient: average of gradients of active peaks, weighted by proximity to max
        diff = M - conv[active_peaks]
        w = np.exp(-diff / 0.0005)
        w = w / w.sum()
        
        total_grad = np.zeros(n)
        for i, k in enumerate(active_peaks):
            total_grad += w[i] * convolve_grad(f, int(k))
        
        # Momentum + projected gradient
        v = 0.9 * v - lr * total_grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:

            for factor in [0.5, 0.25, 0.1]:
                f_try = project_simplex(f + factor * v)
                obj_try = fast_convolve(f_try).max()
                if obj_try < conv.max() * (1 - 1e-8):
                    f = f_try
                    no_improve += 1
                    break
            else:
                no_improve += 1
                lr *= 0.95
        
        lr = max(lr * 0.9998, 0.0001)
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

            lr = 0.003
            no_improve = 0
    
    return best_f, best_obj

def multiplicative_weight_refine(f0, time_limit, lr_start=0.02, lr_end=0.002,
                                  temp_start=0.001, temp_end=0.0001):
    """
    Multiplicative weight (mirror descent) with adaptive temperature.
    Key: use smooth peak weighting and geometric learning rate decay.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        lr = lr_start * (lr_end / lr_start) ** progress
        temp = temp_start * (temp_end / temp_start) ** progress
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Softmax weights on peaks
        shifted = (conv - M) / temp
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-15:
            continue
        weights /= ws
        
        # Compute pressure (gradient of weighted sum of peaks)
        pressure = np.zeros(n)
        active = weights > 1e-8
        for k in np.where(active)[0]:
            if weights[k] < 1e-8:
                continue
            pressure += weights[k] * convolve_grad(f, int(k))
        
        # Multiplicative update
        f_new = f * np.exp(-lr * pressure)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                f_try = f * np.exp(-lr * factor * pressure)
                f_try = np.maximum(f_try, 1e-30)
                f_try = f_try / f_try.sum()
                obj_try = fast_convolve(f_try).max()
                if obj_try < conv.max() * (1 - 1e-9):
                    f_new = f_try
                    obj_new = obj_try
                    if obj_new < best_obj - 1e-15:
                        best_obj = obj_new
                        best_f = f_new.copy()
                    break
            else:
                f_new = f
        
        f = f_new
        
        if no_improve > 80:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 200.0)
            f = best_f * np.exp(rng.standard_normal(n) * scale)
            f = f / f.sum()
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent_refine(f0, time_limit, max_coords_per_iter=None):
    """
    Cyclic coordinate descent on the simplex.
    For each coordinate i, find the optimal value given all others fixed.
    The objective is max_k conv(k), and conv(k) is linear in a_i for fixed k
    (since conv(k) = a_i * a_{k-i} + sum_{j≠i} a_j * a_{k-j}).
    So for fixed i, conv(k) as a function of a_i is:
    - If k-i is valid: a_i * a_{k-i} + (rest)
    - This is linear in a_i!
    
    So max_k conv(k) is the max of linear functions of a_i, which is convex in a_i.
    We can find the minimizer of a convex piecewise-linear function of a single variable!
    
    Actually wait - we also have the constraint sum(a)=1, so changing a_i requires
    changing another coordinate. Let's use a pair swap: increase a_i and decrease a_j.
    
    For a swap (i, j) with delta: a_i -> a_i + delta, a_j -> a_j - delta.
    conv(k) changes by: delta * (a_{k-i} - a_{k-j}) for valid indices.
    So the change in conv(k) is linear in delta.
    The change in max_k conv(k) is the max of linear functions of delta, hence convex.
    We can find the optimal delta by minimizing this convex piecewise-linear function.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Precompute: for each pair (i,j), the sensitivity of each conv(k) to a swap
    # sensitivity[k] = a_{k-i} - a_{k-j} (for valid indices)
    
    # We'll cycle through random pairs and find the optimal swap
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Select a few random pairs to try
        num_pairs = min(50, n // 4)
        for _ in range(num_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Compute sensitivity vector: how conv(k) changes when we move delta from j to i
            # conv(k) = sum_m a_m * a_{k-m}
            # d(conv(k))/d(delta) where a_i += delta, a_j -= delta:
            # = a_{k-i} + a_i * 0 - a_{k-j} - a_j * 0  ... wait
            # Actually: conv(k) = sum_m a_m * a_{k-m}
            # d/d(delta): for m=i: da_i/d(delta) * a_{k-i} = 1 * a_{k-i}
            #             for m=j: da_j/d(delta) * a_{k-j} = -1 * a_{k-j}
            #             for m where k-m=i: a_m * da_{k-m}/d(delta) = a_m * (1 if k-m=i, 0 else 0)
            #             for m where k-m=j: a_m * da_{k-m}/d(delta) = a_m * (-1 if k-m=j, 0 else 0)
            
            # So: d(conv(k))/d(delta) = a_{k-i} - a_{k-j} + a_{k-i} - a_{k-j} ... no
            # Let me redo this carefully.
            # conv(k) = sum_m a_m * a_{k-m}
            # Let a_i(delta) = a_i + delta, a_j(delta) = a_j - delta, others unchanged.
            # d(conv(k))/d(delta) = sum_m [da_m/d(delta) * a_{k-m} + a_m * da_{k-m}/d(delta)]
            # da_m/d(delta) = 1 if m=i, -1 if m=j, 0 otherwise
            # = a_{k-i} - a_{k-j} + a_{k-i} - a_{k-j}  ... NO
            # Wait: sum_m da_m/d(delta) * a_{k-m} = a_{k-i} - a_{k-j}
            #     sum_m a_m * da_{k-m}/d(delta) = a_{k-i} - a_{k-j}  (same by symmetry! since k-m = i means m = k-i)
            # Hmm wait, that's only if k-i and k-j are both valid indices.
            # Let me be more careful:
            # sum_m da_m/d(delta) * a_{k-m} = (1)*a_{k-i} + (-1)*a_{k-j} = a_{k-i} - a_{k-j} (where a_m=0 if m out of range)
            # sum_m a_m * da_{k-m}/d(delta) = a_m where k-m has da/d(delta) nonzero
            #   k-m = i => m = k-i: contributes a_{k-i} * 1
            #   k-m = j => m = k-j: contributes a_{k-j} * (-1)
            # = a_{k-i} - a_{k-j}
            # Total: 2*(a_{k-i} - a_{k-j})
            
            # So sensitivity[k] = 2*(a_{k-i} - a_{k-j}) (with a_m=0 for m out of [0,n-1])
            
            sens = np.zeros(2 * n - 1)
            # For k in [0, 2n-2]:
            # a_{k-i} is valid if 0 <= k-i < n, i.e., i <= k <= i+n-1
            # a_{k-j} is valid if 0 <= k-j < n, i.e., j <= k <= j+n-1
            
            ks1_start = i
            ks1_end = i + n  # exclusive
            ks2_start = j
            ks2_end = j + n  # exclusive
            
            # a_{k-i} for k in [i, i+n-1] is a[0], a[1], ..., a[n-1]
            sens[ks1_start:ks1_end] += 2.0 * f  # a_{k-i} = a[k-i], so for k=i: a[0], k=i+1: a[1], etc.
            sens[ks2_start:ks2_end] -= 2.0 * f  # a_{k-j} = a[k-j]
            
            # Now max_k (conv(k) + delta * sens(k)) is a piecewise-linear convex function of delta.
            # We want to find delta in [-a_j, a_i] that minimizes this.
            # The minimum of max of linear functions occurs where the "upper envelope" is minimized.
            # This is equivalent to: find delta such that the maximum of (conv(k) + delta*sens(k)) is minimized.
            # = find delta such that the top values are as equal as possible.
            
            # Binary search or solve directly:
            # The subgradient of max_k (conv(k) + delta*sens(k)) at delta is:
            # sum_k w_k * sens(k) where w_k = indicator that k is at the max (averaged over ties)
            # At the optimum, 0 is in the subgradient, meaning there exist active k's with
            # weighted average of sens(k) = 0.
            
            # Simple approach: binary search for delta where the "breakpoint" occurs.
            # Or: evaluate at a few candidate delta values.
            
            # Candidate deltas: where two peaks cross: (conv(k1) + d*sens(k1)) = (conv(k2) + d*sens(k2))
            # => d = (conv(k2) - conv(k1)) / (sens(k1) - sens(k2))
            
            # For efficiency, just try a few values:
            best_delta = 0.0
            best_val = M
            
            # The key peaks are those near the maximum
            top_idx = np.argpartition(conv, -10)[-10:]
            for k1, k2 in [(top_idx[p], top_idx[q]) for p in range(min(5, len(top_idx))) for q in range(p+1, min(5, len(top_idx)))]:
                s1 = sens[k1]
                s2 = sens[k2]
                if abs(s1 - s2) > 1e-15:
                    d = (conv[k2] - conv[k1]) / (s1 - s2)
                    d = max(-f[j] * 0.99, min(f[i] * 0.99, d))
                    # Evaluate at this delta
                    new_conv_max = (conv + d * sens).max()
                    if new_conv_max < best_val:
                        best_val = new_conv_max
                        best_delta = d
            
            # Also try the "flatten" delta: where the two top peaks are equal
            top2 = np.argpartition(conv, -2)[-2:]
            if len(top2) == 2:
                s1 = sens[top2[0]]
                s2 = sens[top2[1]]
                if abs(s1 - s2) > 1e-15:
                    d = (conv[top2[1]] - conv[top2[0]]) / (s1 - s2)
                    d = max(-f[j] * 0.99, min(f[i] * 0.99, d))
                    new_conv_max = (conv + d * sens).max()
                    if new_conv_max < best_val:
                        best_val = new_conv_max
                        best_delta = d
            
            if best_delta != 0 and best_val < M - 1e-14:
                f[i] += best_delta
                f[j] -= best_delta
                f = np.maximum(f, 0.0)
                f = f / f.sum()
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Exact 1D coordinate descent (swap-based) with convex piecewise-linear minimization
3. Log-sum-exp smoothed gradient descent with adaptive temperature

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. Returns vector g where g[j] = 2*a[k-j] if valid."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n, k + 1)
    if je > js:
        g[js:je] = 2.0 * a[k - je + 1:k - js + 1][::-1]
    return g


def solve_lp_for_fw(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle for quick initialization."""
    f = np.maximum(f0.astype(np.float64), 0.0)

        if g is None:

            continue

        if g_sum < 1e-15:

def swap_sensitivity(f, i, j):
    """Compute sensitivity vector sens[k] = 2*(a[k-i] - a[k-j]) for swap (i,j)."""
    n = len(f)
    sens = np.zeros(2 * n - 1)
    sens[i:i + n] += 2.0 * f
    sens[j:j + n] -= 2.0 * f
    return sens


def optimal_swap_delta(conv, sens, delta_min, delta_max, top_k=15):
    """
    Find optimal delta in [delta_min, delta_max] minimizing max_k (conv[k] + delta*sens[k]).
    Uses the fact that the max of linear functions is convex piecewise-linear,
    so the minimum is at a breakpoint or boundary.
    """
    n_conv = len(conv)
    # Get top peaks
    if top_k >= n_conv:
        top = np.arange(n_conv)
    else:
        top = np.argpartition(conv, -top_k)[-top_k:]

    best_delta = 0.0
    best_val = conv.max()

    # Evaluate at boundaries
    for d in [delta_min, delta_max, 0.0]:
        val = (conv + d * sens).max()
        if val < best_val:
            best_val = val
            best_delta = d

    # Evaluate at breakpoints between pairs of top peaks
    for p in range(len(top)):
        for q in range(p + 1, len(top)):
            kp = top[p]
            kq = top[q]
            sp = sens[kp]
            sq = sens[kq]
            denom = sq - sp
            if abs(denom) < 1e-15:
                continue
            d = (conv[kp] - conv[kq]) / denom
            if d < delta_min or d > delta_max:
                continue
            val = (conv + d * sens).max()
            if val < best_val:
                best_val = val
                best_delta = d

    return best_delta, best_val


def coordinate_descent_optimize(f0, time_limit, num_pairs=30):
    """
    Cyclic coordinate descent using exact 1D swap optimization.
    At each step, try several (i,j) pairs and pick the best exact optimization.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks for sensitivity analysis
        top_k = min(20, 2 * n - 1)
        top_peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Smart pair selection: focus on coordinates that affect top peaks
        candidate_pairs = []

        # Strategy 1: For each top peak, find best source (large contribution) and target
        for k in top_peaks[:5]:
            k = int(k)
            # Contribution of position j to peak k is a[k-j]
            contrib = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n, k + 1)
            contrib[js:je] = f[je - 1 - js::-1]  # a[k-j] for j in [js, je)

            # Best source: j with largest contribution to this peak
            if contrib.max() > 1e-15:
                best_j = int(np.argmax(contrib))
                # Best target: i that contributes least to ALL top peaks
                # Approximate: i where sum of contributions to top peaks is small
                total_contrib = np.zeros(n)
                for k2 in top_peaks[:5]:
                    k2 = int(k2)
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n, k2 + 1)
                    total_contrib[js2:je2] += f[je2 - 1 - js2::-1]
                best_i = int(np.argmin(total_contrib))
                if best_i != best_j:
                    candidate_pairs.append((best_i, best_j))

        # Strategy 2: Random pairs
        for _ in range(num_pairs):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                candidate_pairs.append((i, j))

        # Try each candidate pair and keep the best improvement
        best_improvement = 0.0
        best_swap = None

        for (i, j) in candidate_pairs:
            if f[i] < 1e-15 or f[j] < 1e-15:
                continue
            sens = swap_sensitivity(f, i, j)
            delta_min = -f[j]
            delta_max = f[i]
            d_opt, v_opt = optimal_swap_delta(conv, sens, delta_min, delta_max, top_k=20)
            improvement = M - v_opt
            if improvement > best_improvement:
                best_improvement = improvement
                best_swap = (i, j, d_opt)

        if best_swap is not None and best_improvement > 1e-15:
            i, j, d_opt = best_swap
            f[i] += d_opt
            f[j] -= d_opt
            f = np.maximum(f, 0.0)
            f = f / f.sum()

            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 200:
            # Perturb and restart
            scale = 0.002 + 0.005 * min(1.0, no_improve / 500.0)
            f = best_f * np.exp(rng.standard_normal(n) * scale)
            f = f / f.sum()
            no_improve = 0

        if no_improve > 1000:
            break

    return best_f, best_obj


def smooth_gradient_descent(f0, time_limit, beta_start=300, beta_end=3000,
                            lr_start=0.008, lr_end=0.0005):
    """
    Gradient descent on log-sum-exp smooth approximation of max(conv).
    Uses mirror descent (multiplicative updates) for positivity preservation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        conv = fast_convolve(f)

        # Softmax weights (gradient of log-sum-exp)
        shifted = beta * conv - beta * conv.max()
        exp_vals = np.exp(shifted)
        weights = exp_vals / exp_vals.sum()

        # Compute gradient: sum_k w_k * grad_k
        total_grad = np.zeros(n)
        active = weights > 1e-10
        for k in np.where(active)[0]:
            if weights[k] < 1e-10:
                continue
            total_grad += weights[k] * convolve_grad_at_k(f, int(k))

        # Multiplicative update (mirror descent)
        f_new = f * np.exp(-lr * total_grad)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            # Backtrack
            improved = False
            for factor in [0.5, 0.25, 0.1, 0.05]:
                f_try = f * np.exp(-lr * factor * total_grad)
                f_try = np.maximum(f_try, 1e-30)
                f_try = f_try / f_try.sum()
                obj_try = fast_convolve(f_try).max()
                if obj_try < conv.max() * (1 - 1e-10):
                    f = f_try
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_f = f_try.copy()
                    no_improve += 1
                    improved = True
                    break
            if not improved:
                no_improve += 1

        if no_improve > 150:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 300.0)
            f = best_f * np.exp(rng.standard_normal(n) * scale)
            f = f / f.sum()
            no_improve = 0

    return best_f, best_obj

def active_set_subgradient(f0, time_limit, lr_start=0.004, mom=0.93):

    best_f = f.copy()

    no_improve = 0
    it = 0

        f_new = project_simplex(f + v)

        obj_new = conv_new.max()

            f = f_new
        else:
            no_improve += 1
            for factor in [0.5, 0.25, 0.1]:
                f_try = project_simplex(f + factor * v)
                obj_try = fast_convolve(f_try).max()
                if obj_try < conv.max() * (1 - 1e-10):
                    f = f_try
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_f = f_try.copy()
                    break

        lr *= 0.9997
        sigma *= 1.001

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = max(lr_start * 0.6, 0.0005)
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: FW-LP initialization (~120s)
    Phase 2: Coordinate descent with exact 1D optimization (~500s)
    Phase 3: Smooth gradient descent (~300s)
    Phase 4: Multiplicative weight refinement (~250s)
    Phase 5: Active set subgradient (~200s)
    Phase 6: Random restarts with local optimization (~250s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            sc = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(sc):
                best_score = sc
                best_seq = parent.copy()
            print(f"Parent: n={len(parent)}, score={sc:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        n_p = 384
        parent = np.ones(n_p) / n_p
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    n = len(best_seq)

    # Phase 1: FW-LP
    print("\n=== Phase 1: FW-LP ===")
    rem = deadline - time.monotonic()
    t1 = min(120, rem * 0.07)
    if t1 > 30:
        print(f"  t={t1:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t1)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [fw_lp]: {sc:.8f}")

    # Phase 2: Coordinate Descent (main workhorse)
    print("\n=== Phase 2: Coordinate Descent ===")
    rem = deadline - time.monotonic()
    t2 = min(500, rem * 0.3)
    if t2 > 50:
        print(f"  t={t2:.0f}s")
        a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=t2, num_pairs=40)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [coord_desc]: {sc:.8f}")
        else:
            print(f"  No improvement (score={sc:.8f})")

    # Phase 3: Smooth Gradient Descent
    print("\n=== Phase 3: Smooth GD ===")
    rem = deadline - time.monotonic()
    t3 = min(300, rem * 0.2)
    if t3 > 30:
        print(f"  t={t3:.0f}s")
        a_opt, _ = smooth_gradient_descent(
            best_seq, time_limit=t3,
            beta_start=300, beta_end=3000,
            lr_start=0.008, lr_end=0.0005
        )
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [smooth_gd]: {sc:.8f}")

    # Phase 4: Multiplicative Weight
    print("\n=== Phase 4: Multiplicative Weight ===")
    rem = deadline - time.monotonic()
    t4 = min(250, rem * 0.18)
    if t4 > 30:
        print(f"  MW pass 1, t={t4:.0f}s")
        a_opt, _ = multiplicative_weight_refine(
            best_seq, time_limit=t4,
            lr_start=0.015, lr_end=0.001,
            temp_start=0.0008, temp_end=0.00005
        )
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [mw1]: {sc:.8f}")

    # Second MW with different params
    rem = deadline - time.monotonic()
    t4b = min(150, rem * 0.12)
    if t4b > 20:
        print(f"  MW pass 2, t={t4b:.0f}s")
        a_opt, _ = multiplicative_weight_refine(
            best_seq, time_limit=t4b,
            lr_start=0.008, lr_end=0.0005,
            temp_start=0.0004, temp_end=0.00003
        )
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [mw2]: {sc:.8f}")

    # Phase 5: Active Set Subgradient
    print("\n=== Phase 5: Active Set SG ===")
    rem = deadline - time.monotonic()
    t5 = min(200, rem * 0.15)
    if t5 > 30:
        print(f"  t={t5:.0f}s")
        a_opt, _ = active_set_subgradient(
            best_seq, time_limit=t5,
            lr_start=0.004, mom=0.93
        )
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [active_sg]: {sc:.8f}")

    # Phase 6: More coordinate descent (second pass on improved sequence)
    print("\n=== Phase 6: Coord Descent Pass 2 ===")
    rem = deadline - time.monotonic()
    t6 = min(300, rem * 0.2)
    if t6 > 30:
        print(f"  t={t6:.0f}s")
        a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=t6, num_pairs=50)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [coord_desc2]: {sc:.8f}")

    # Phase 7: Random restarts
    print("\n=== Phase 7: Random Restarts ===")
    rem = deadline - time.monotonic()
    n_restarts = min(15, max(1, int(rem / 15)))
    for ri in range(n_restarts):
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        noise_scale = 0.001 + 0.003 * rng.random()
        perturbed = best_seq * np.exp(rng.standard_normal(n) * noise_scale)
        perturbed = perturbed / perturbed.sum()

        t_r = min(rem - 5, 20)
        if t_r > 5:
            a_opt, _ = coordinate_descent_optimize(perturbed, time_limit=t_r, num_pairs=30)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [restart_{ri}]: {sc:.8f}")

    # Phase 8: Final micro-polish with very fine parameters
    print("\n=== Phase 8: Final Micro Polish ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t8 = min(rem - 5, 30)
        a_opt, _ = smooth_gradient_descent(
            best_seq, time_limit=t8,
            beta_start=5000, beta_end=20000,
            lr_start=0.0005, lr_end=0.00005
        )
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [micro_polish]: {sc:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
