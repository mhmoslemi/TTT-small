Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe
with dual LP, golden section line search, palindromic symmetry,
frequency-domain initialization, and multi-scale coordinate descent.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_palindromic(a, n):
    """Create a palindromic sequence of length n from a shorter one."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])  # a followed by reversed a[:-1]

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def golden_section_search(f, a, g, conv_f, M_cur, n):
    """Golden section search for best alpha in [0,1] to mix f with g."""
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    lo, hi = 0.0, 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = fast_convolve(project_simplex(f + c * (g - f))).max()
    fd = fast_convolve(project_simplex(f + d * (g - f))).max()
    
    for _ in range(30):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = fast_convolve(project_simplex(f + c * (g - f))).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = fast_convolve(project_simplex(f + d * (g - f))).max()
    
    alpha = (lo + hi) / 2
    return alpha

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:

        g = g / g_sum

        f = np.ones(len(f0)) / len(f0)

        f = f / s

    conv = fast_convolve(f)

    best_f = f.copy()

    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= 1.005

            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0004)
            no_improve = 0
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:

    old_x = np.linspace(0, 1, old_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n from a (uses first half)."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def solve_lp_direction(f, M, n, active_k=None, max_constraints=None):

    return None

    d = lo + phi * (hi - lo)

    f_d = project_simplex(f + d * (g - f))
    fc = fast_convolve(f_c).max()
    fd = fast_convolve(f_d).max()

            d = c
            fd = fc
            c = hi - phi * (hi - lo)

            fc = fd
            d = lo + phi * (hi - lo)

            fd = fast_convolve(f_d).max()

                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=40, temp=0.00004):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.99985
        cur_temp *= 1.004

        if no_improve > 400:
            seed_val = (it_count * 7919 + 13) % 1000000
            rng = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0004)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum (good for low peak autocorrelation)."""
    rng = np.random.default_rng(seed)
    # Generate random phases (flat magnitude spectrum)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    # Inverse FFT to get time domain
    sig = np.fft.irfft(spectrum, fft_size)
    # Take first n samples, make non-negative
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        # Fallback: use absolute value
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: improved LP-Frank-Wolfe with adaptive constraints,
    palindromic symmetry, frequency-domain init, fine coordinate descent,
    and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: LP-FW on parent with adaptive constraints ===
    print("=== Phase 1: LP-FW on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(300, rem * 0.28)
    if t1 > 30:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True,
                                  adaptive_constraints=True, top_k_constraints=40)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Palindromic LP-FW ===
    print("\n=== Phase 2: Palindromic LP-FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        half_n = (n_best + 1) // 2
        pal_half = (best_seq[:half_n] + best_seq[n_best - 1: n_best - half_n - 1: -1]) / 2.0
        pal_full = project_simplex(make_palindromic(pal_half, n_best))
        try_seq(pal_full, "pal_init")

        rem = deadline - time.monotonic()
        t2 = min(180, rem * 0.18)
        if t2 > 30:
            a_opt, _ = lp_fw_optimize(pal_full, time_limit=t2, verbose=True,
                                      use_palindrome=True, adaptive_constraints=True,
                                      top_k_constraints=40)
            try_seq(a_opt, "palindromic_lpfw")

    # === Phase 3: Frequency-domain initialization + different lengths ===
    print("\n=== Phase 3: Multi-length & frequency init ===")
    base = best_seq if best_seq is not None else parent

    # Try flat spectrum init at several lengths
    for target_n in [512, 640, 896]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        f_init = interpolate_sequence(base, target_n)
        # Mix with flat spectrum
        fs_init = flat_spectrum_init(target_n, seed=target_n)
        f_mix = project_simplex(0.7 * f_init + 0.3 * fs_init)
        t_alloc = min(rem * 0.2, 150)
        print(f"  LP-FW n={target_n} (freq mix), t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_mix, time_limit=t_alloc, verbose=False,
                                  adaptive_constraints=True, top_k_constraints=30)
        try_seq(a_opt, f"freq_n{target_n}")

    # === Phase 4: Deep optimization with diverse restarts ===
    print(f"\n=== Phase 4: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        restart_configs = [
            ("direct", best_seq.copy()),
            ("small_noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)),
            ("rev_sym", project_simplex(0.65 * best_seq + 0.35 * best_seq[::-1])),
            ("flat_blend", project_simplex(0.82 * best_seq + 0.18 * np.ones(n_best) / n_best)),
            ("gaussian_center", project_simplex(
                np.exp(-3.0 * (np.arange(n_best) - n_best / 2.0) ** 2 / (0.5 * n_best) ** 2))),
        ]

        for i, (name, f_init) in enumerate(restart_configs):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem / max(1, len(restart_configs) - i), 200)
            print(f"  Restart [{name}], t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc,
                                      verbose=False, adaptive_constraints=True,
                                      top_k_constraints=35)
            try_seq(a_opt, f"deep_{name}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.35, 300)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coord_desc_fine(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

            # Another CD pass on the result
            rem = deadline - time.monotonic()
            if rem > 25:
                t_cd2 = min(rem * 0.25, 150)
                a_opt2, _ = coord_desc_fine(a_opt, time_limit=t_cd2)
                try_seq(a_opt2, "cd2")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sg = min(rem * 0.3, 180)
            n_best = len(best_seq)
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.0012, mom=0.94,
                                 top_k=max(25, n_best // 5),
                                 temp=0.00003)
            try_seq(a_opt, "sg")

            # Second SG with different params
            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt2, _ = sg_polish(a_opt, time_limit=min(rem * 0.3, 80),
                                      lr=0.0008, mom=0.95,
                                      top_k=max(15, n_best // 8),
                                      temp=0.00001)
                try_seq(a_opt2, "sg_fine")

    # === Phase 7: Final LP-FW + perturbations ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            t_final = min(rem * 0.4, 100)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final,
                                      adaptive_constraints=False)
            try_seq(a_opt, "final_lpfw_full")

        rem = deadline - time.monotonic()
        for i in range(4):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.001 * i)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.2, 40)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_pert,
                                      adaptive_constraints=True, top_k_constraints=25)
            try_seq(a_opt, f"final_perturb_{i}")

        # Tiny SG on final
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem * 0.5, 30),
                                 lr=0.0005, mom=0.95,
                                 top_k=max(10, n_best // 10),
                                 temp=0.00001)
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
