        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def solve_lp_oracle(f, M, time_limit=10):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def peak_shaving_coordinate(f, n_iters=500, step=0.002, top_peaks=5):
    """
    Coordinate descent that targets the top peaks.
    At each step, identify the peak lag, find the most contributing index,
    and transfer mass to the safest index.
    """
    n = len(f)
    f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    
    for it in range(n_iters):
        # Find top peaks
        if top_peaks >= len(conv):
            peak_lags = np.arange(len(conv))
        else:
            peak_lags = np.argsort(conv)[-top_peaks:]
        
        # For each peak, compute the "sensitivity" of each index
        sensitivity = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            weight = conv[k] - conv.min() + 1e-10  # relative importance
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                if j == i:
                    sensitivity[i] += 2.0 * weight * f[i]
                else:
                    sensitivity[i] += weight * f[j]
        
        # Normalize sensitivity
        s_max = sensitivity.max()
        if s_max < 1e-15:
            break
        
        # The index to decrease: highest sensitivity
        i_dec = int(np.argmax(sensitivity))
        
        # The index to increase: we want one that minimally increases the peaks
        # Compute "peak impact" for each index (how much it contributes to peaks)
        peak_impact = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                if j == i:
                    peak_impact[i] += f[i]
                else:
                    peak_impact[i] += f[j] * 0.5
        
        # Prefer increasing indices with low peak impact AND low current value
        # (low current value means less impact on conv(0) = sum f_i^2)
        i_inc_candidates = np.argsort(peak_impact + 0.1 * f)
        i_inc = -1
        for cand in i_inc_candidates:
            if cand != i_dec:
                i_inc = int(cand)
                break
        
        if i_inc == -1 or i_inc == i_dec:
            break
        
        # Transfer mass
        amt = min(step, f[i_dec] * 0.3)
        if amt < 1e-14:
            break
        
        f[i_dec] -= amt
        f[i_inc] += amt
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            f[i_dec] += amt
            f[i_inc] -= amt
            if no_improve > 100:
                step *= 0.95
                if step < 1e-6:
                    break
    
    return best_f, best_obj


def adaptive_sg_descent(f0, time_limit, initial_lr=0.01, mom=0.9, 
                        top_k_init=128, temp_init=0.005,
                        bb=True, lp_every=200, use_peak_shave=True):
    """Adaptive subgradient descent with momentum, BB steps, LP corrections, and peak shaving."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp_init
    cur_top_k = top_k_init
    
    # BB state
    s_prev = None
    y_prev = None
    f_prev = None
    sg_prev = None
    
    # SG averaging window
    sg_avg = np.zeros(n)
    sg_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        sg = compute_subgradient(f, conv, top_k=cur_top_k, temp=cur_temp)
        
        # Average subgradient
        sg_avg = 0.7 * sg_avg + 0.3 * sg
        sg_count += 1
        if sg_count > 10:
            sg_use = sg_avg
        else:
            sg_use = sg
        
        # BB step size
        if bb and s_prev is not None and y_prev is not None and sg_prev is not None:
            s_diff = f - f_prev
            y_diff = sg - sg_prev
            num = np.dot(s_diff, s_diff)
            den = np.dot(s_diff, y_diff)
            if den > 1e-12:
                bb_lr = num / den
                if 1e-4 < bb_lr < 0.5:
                    cur_lr = 0.6 * cur_lr + 0.4 * bb_lr
        
        f_prev = f.copy()
        sg_prev = sg.copy()
        
        # Momentum step
        v = mom * v - cur_lr * sg_use
        f_new = project_simplex(f + v)
        s_prev = f_new - f
        y_prev = sg - sg_prev
        
        f = f_new
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters: gradually focus on the true peak
        cur_temp *= 1.005
        cur_top_k = max(8, int(cur_top_k * 0.9995))
        cur_lr *= 0.9998
        
        # LP correction
        if no_improve > lp_every and rem > 30:
            g = solve_lp_oracle(f, best_obj * 1.01, time_limit=min(15, rem // 4))
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    # Line search towards g
                    best_alpha = 0
                    best_val = best_obj
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        ft = project_simplex((1 - alpha) * f + alpha * g)
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha
                    if best_alpha > 0:
                        f = project_simplex((1 - best_alpha) * f + best_alpha * g)
                        conv = fast_convolve(f)
                        obj = conv.max()
                        if obj < best_obj:
                            best_obj = obj
                            best_f = f.copy()
                        no_improve = 0
                        cur_lr = initial_lr * 0.5
                        cur_temp = temp_init
                        v = np.zeros(n)
                        sg_avg = np.zeros(n)
        
        # Random restart
        if no_improve > 800:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.7
            cur_temp = temp_init * 0.5
            sg_avg = np.zeros(n)
            no_improve = 0
            
    # Peak shaving at the end
    if use_peak_shave:
        rem = time_limit - (time.monotonic() - t0)
        if rem > 5:
            f_ps, obj_ps = peak_shaving_coordinate(best_f, n_iters=min(500, int(rem * 50)), step=0.001)
            if obj_ps < best_obj:
                best_f = f_ps
                best_obj = obj_ps
    
    return best_f, best_obj


def fine_polish(f0, time_limit, lr=0.001, mom=0.95, top_k=50, temp=0.0001):
    """Very fine subgradient polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9999
        cur_temp *= 1.002
        
        if no_improve > 300:
            rng = np.random.default_rng(it + 50)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0
    
    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Symmetric raised cosine
    rc = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(rc / rc.sum())
    
    # Symmetric Gaussian (wide)
    g1 = np.exp(-0.5 * ((x - mid) / (n / 4)) ** 2)
    inits.append(g1 / g1.sum())
    
    # Symmetric Gaussian (narrow)
    g2 = np.exp(-0.5 * ((x - mid) / (n / 8)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Triangular
    tri = 1.0 - np.abs(x - mid) / (mid + 0.5)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())
    
    # Inverse cosine (peaked at edges)
    ic = 1.0 / (np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.1)
    inits.append(ic / ic.sum())
    
    # Power law symmetric
    pl = 1.0 / (np.abs(x - mid) / mid + 1.0) ** 0.3
    inits.append(pl / pl.sum())
    
    # Two humps
    h1 = np.exp(-0.5 * ((x - n * 0.25) / (n / 10)) ** 2)
    h2 = np.exp(-0.5 * ((x - n * 0.75) / (n / 10)) ** 2)
    two_h = h1 + h2 + 0.01
    inits.append(two_h / two_h.sum())
    
    # Sine (single period)
    sn = np.sin(np.pi * x / (n - 1)) ** 2 + 0.01
    inits.append(sn / sn.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Random Dirichlet (sparse)
    inits.append(rng.dirichlet(np.ones(n) * 0.1))
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Slightly smoothed parent
            kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2) ** 2)
            px_smooth = np.convolve(px, kernel, mode='same')
            px_smooth = np.maximum(px_smooth, 0)
            inits.append(px_smooth / px_smooth.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive subgradient descent with momentum, BB step sizes, and SG averaging
2. Peak-shaving coordinate descent for local refinement
3. LP-based corrections (sparse usage)
4. Multi-length exploration with diverse structured initializations
5. Multi-phase time allocation with final polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def solve_lp_oracle(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def peak_shaving_coordinate(f, n_iters=500, step=0.002, top_peaks=5):
    """
    Coordinate descent targeting top peaks.
    Transfers mass from high-sensitivity indices to low-impact indices.
    """
    n = len(f)
    f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    cur_step = step

    for it in range(n_iters):
        if no_improve > 200:
            break

        if top_peaks >= len(conv):
            peak_lags = np.arange(len(conv))
        else:
            peak_lags = np.argsort(conv)[-top_peaks:]

        sensitivity = np.zeros(n)
        peak_impact = np.zeros(n)

        for k in peak_lags:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            w = (conv[k] - conv[peak_lags.min()]) + 1e-8
            for i in range(i_min, i_max + 1):
                j = k - i
                if j == i:
                    sensitivity[i] += 2.0 * w * f[i]
                    peak_impact[i] += f[i]
                else:
                    sensitivity[i] += w * f[j]
                    peak_impact[i] += 0.5 * f[j]

        s_max = sensitivity.max()
        if s_max < 1e-15:
            break

        i_dec = int(np.argmax(sensitivity))

        # Prefer indices with low peak impact and low current value
        score_inc = peak_impact + 0.05 * f
        order = np.argsort(score_inc)
        i_inc = -1
        for cand in order:
            if int(cand) != i_dec:
                i_inc = int(cand)
                break

        if i_inc == -1 or i_inc == i_dec:
            break

        amt = min(cur_step, f[i_dec] * 0.4)
        if amt < 1e-15:
            break

        f[i_dec] -= amt
        f[i_inc] += amt

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            f[i_dec] += amt
            f[i_inc] -= amt
            if no_improve % 20 == 0:
                cur_step *= 0.93

    return best_f, best_obj


def adaptive_sg_descent(f0, time_limit, initial_lr=0.01, mom=0.9,
                        top_k_init=128, temp_init=0.005,
                        bb=True, lp_every=200, use_peak_shave=True):
    """Adaptive subgradient descent with momentum, BB steps, SG averaging, LP corrections."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp_init
    cur_top_k = top_k_init

    s_prev = None
    y_prev = None
    f_prev = None
    sg_prev = None
    sg_avg = np.zeros(n)
    sg_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        sg = compute_subgradient(f, conv, top_k=cur_top_k, temp=cur_temp)

        sg_avg = 0.6 * sg_avg + 0.4 * sg
        sg_count += 1
        if sg_count > 5:
            sg_use = sg_avg
        else:
            sg_use = sg

        if bb and s_prev is not None and y_prev is not None and sg_prev is not None:
            s_diff = f - f_prev
            y_diff = sg - sg_prev
            num = np.dot(s_diff, s_diff)
            den = np.dot(s_diff, y_diff)
            if den > 1e-12:
                bb_lr = num / den
                if 1e-4 < bb_lr < 0.5:
                    cur_lr = 0.6 * cur_lr + 0.4 * bb_lr

        f_prev = f.copy()
        sg_prev = sg.copy()

        v = mom * v - cur_lr * sg_use
        f_new = project_simplex(f + v)
        s_prev = f_new - f
        y_prev = sg - sg_prev

        f = f_new
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= 1.004
        cur_top_k = max(8, int(cur_top_k * 0.9997))
        cur_lr *= 0.9998

        if no_improve > lp_every and rem > 25:
            g = solve_lp_oracle(f, best_obj * 1.01, time_limit=min(12, rem // 4))
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    best_alpha = 0
                    best_val = best_obj
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        ft = project_simplex((1 - alpha) * f + alpha * g)
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha
                    if best_alpha > 0:
                        f = project_simplex((1 - best_alpha) * f + best_alpha * g)
                        conv = fast_convolve(f)
                        obj = conv.max()
                        if obj < best_obj:
                            best_obj = obj
                            best_f = f.copy()
                        no_improve = 0
                        cur_lr = initial_lr * 0.4
                        cur_temp = temp_init * 0.5
                        v = np.zeros(n)
                        sg_avg = np.zeros(n)
                        sg_count = 0

        if no_improve > 600:
            rng = np.random.default_rng(it * 11 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.6
            cur_temp = temp_init * 0.5
            sg_avg = np.zeros(n)
            sg_count = 0
            no_improve = 0

    if use_peak_shave:
        rem = time_limit - (time.monotonic() - t0)
        if rem > 3:
            n_ps = min(400, int(rem * 40))
            f_ps, obj_ps = peak_shaving_coordinate(best_f, n_iters=n_ps, step=0.001)
            if obj_ps < best_obj:
                best_f = f_ps
                best_obj = obj_ps

    return best_f, best_obj


def fine_polish(f0, time_limit, lr=0.001, mom=0.96, top_k=40, temp=0.0001):
    """Very fine subgradient polishing with minimal perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

    n = len(f)

    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:

        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()

        if no_improve > 200:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.0004)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse structured initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    inits.append(np.ones(n) / n)

    rc = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 2 + 0.01
    inits.append(rc / rc.sum())

    g1 = np.exp(-0.5 * ((x - mid) / max(n / 4, 1)) ** 2)
    inits.append(g1 / g1.sum())

    g2 = np.exp(-0.5 * ((x - mid) / max(n / 8, 1)) ** 2)
    inits.append(g2 / g2.sum())

    tri = 1.0 - np.abs(x - mid) / (mid + 0.5)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    ic = 1.0 / (np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 2 + 0.1)
    inits.append(ic / ic.sum())

    pl = 1.0 / (np.abs(x - mid) / max(mid, 1) + 1.0) ** 0.4
    inits.append(pl / pl.sum())

    sn = np.sin(np.pi * x / max(n - 1, 1)) ** 2 + 0.01
    inits.append(sn / sn.sum())

    inits.append(rng.dirichlet(np.ones(n) * 0.4))
    inits.append(rng.dirichlet(np.ones(n) * 0.15))

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            kernel = np.exp(-0.5 * (np.arange(-7, 8) / 3) ** 2)
            px_s = np.convolve(px, kernel, mode='same')
            px_s = np.maximum(px_s, 0)
            if px_s.sum() > 1e-10:
                inits.append(px_s / px_s.sum())

    return inits
