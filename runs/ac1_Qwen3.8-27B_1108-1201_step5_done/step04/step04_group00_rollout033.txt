def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, k_indices):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. LP-based Frank-Wolfe with rational step size (global optimization)
2. Proximal subgradient with Nesterov momentum (fast polishing)
3. Adaptive coordinate descent (fine-tuning)
4. Multi-length and multi-restart exploration

Key differences from previous approach:
- Rational (optimal) FW step size instead of grid search
- Proximal subgradient method avoids LP bottleneck in polishing
- More aggressive coordinate descent with wider multiplier range
- Diverse restart strategy to escape local minima
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, peaks):
    """Compute weighted subgradient of max_k (f*f)[k]."""
    n = len(f)
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(k - j_hi, k - j_lo + 1)
            grad[j_lo:j_hi + 1] += 2.0 * f[idx]
    return grad


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with rational step size and adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 8:
            break

        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.12
            if m_relax > 8.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_relax = min(m_relax * 1.04, 3.0)
            continue

        # Rational step size with fallback candidates
        fg_dot = np.dot(f, f - g)
        fg_norm_sq = np.sum((f - g) ** 2)
        if fg_norm_sq > 1e-15:
            alpha_r = max(0.0, min(1.0, 2.0 * fg_dot / fg_norm_sq))
        else:
            alpha_r = 0.5

        candidates = sorted(set([
            alpha_r,
            max(0.01, alpha_r * 0.6),
            min(1.0, alpha_r * 1.4),
            0.3,
            0.5,
            0.7,
        ]))

        best_alpha = 0.0
        best_val = M

        for alpha in candidates:
            ft = f + alpha * (g - f)
            ft = project_simplex(ft)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = project_simplex(f_new)
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.08, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.004 * min(no_improve // 50, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def proximal_subgrad(f0, time_limit, lr0=0.008, top_k=30, verbose=False):
    """Proximal subgradient with Nesterov momentum - fast, no LP needed."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Nesterov state
    f_y = f.copy()  # momentum point
    t_nest = 1
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr0
    temp = 1.0 / (5000.0 * conv.max())

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Compute subgradient at momentum point
        conv_y = fast_convolve(f_y)
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k_actual)[-k_actual:]

        pv = conv_y[peaks]
        max_pv = pv.max()
        scaled = (max_pv - pv) / temp
        weights = np.exp(-scaled)
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.zeros(len(peaks))
            weights[np.argmax(pv)] = 1.0
        else:
            weights /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-13:
                continue
            k = int(peaks[i])
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = np.arange(k - j_hi, k - j_lo + 1)
                grad[j_lo:j_hi + 1] += 2.0 * weights[i] * f_y[idx]

        # Nesterov update
        t_nest += 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new

        f_old = f.copy()
        f = project_simplex(f_y - lr * grad)
        f_y = f + theta * (f - f_old)

        # Evaluate
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    ProxSG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Decay learning rate
        lr = lr0 / (1.0 + 0.005 * t_nest)
        # Slowly increase temperature (explore more peaks)
        temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng((it * 17 + 3) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            f_y = f.copy()
            f_old = f.copy()
            no_improve = 0
            lr = lr0 * 0.6
            t_nest = 1
            temp = 1.0 / (5000.0 * conv.max())

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Adaptive coordinate descent with wider multiplier range."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    passes = 0

    multipliers = [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0, 5.0]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                best_obj = best_j_obj
                best_f = f.copy()
                improved = True

        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

        if not improved and passes > 8:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
