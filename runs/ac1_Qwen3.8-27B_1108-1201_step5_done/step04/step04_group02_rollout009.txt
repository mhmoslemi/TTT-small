def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    b = fast_autoconv(a)

def coordinate_descent(a, n, time_limit=60, tol=1e-12):
    """Exact coordinate descent minimizing max(conv(a,a)) with sum(a)=1, a>=0."""
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    t0 = time.monotonic()
    b = fast_autoconv(a)
    best_obj = b.max()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    max_sweeps = 200
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit - 1:
            break
        improved = False
        
        for i in range(n):
            old_val = a[i]
            if old_val < 1e-15 and b.max() < 1.0/n:
                continue
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Unchanged max
            if k_lo > 0 and k_hi < 2 * n - 2:
                unchanged_max = max(b[:k_lo].max(), b[k_hi+1:].max())
            elif k_lo == 0:
                unchanged_max = b[k_hi+1:].max() if k_hi < 2*n-2 else 0.0
            else:
                unchanged_max = b[:k_lo].max()
            
            # Changed part: b[k] + delta * a[k-i] for k in [k_lo, k_hi]
            changed_indices = np.arange(k_lo, k_hi + 1)
            a_factors = a[changed_indices - i]
            b_changed = b[changed_indices].copy()
            
            # Golden section search for delta
            lo = -old_val
            hi = max(old_val * 3, 1.0/n)
            
            def obj_delta(delta):
                changed = b_changed + delta * a_factors
                return max(unchanged_max, changed.max())
            
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = obj_delta(c)
            fd = obj_delta(d)
            
            for _ in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_delta(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_delta(d)
            
            best_delta = (lo + hi) / 2.0
            new_val = max(0.0, old_val + best_delta)

                a[i] = new_val
                a = a / a.sum()
                b = fast_autoconv(a)
                obj = b.max()
                if obj < best_obj:
                    best_obj = obj
                improved = True
        
        if not improved:
            break
    
    return a


def gerchberg_saxton_init(n, n_iter=100, rng=None):
    """Generate initialization with approximately flat power spectrum."""
    if rng is None:
        rng = np.random.default_rng(0)

    a = rng.random(n) + 0.1
    a = a / a.sum()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iter):
        # Forward FFT
        A = np.fft.rfft(a, fft_size)
        # Set magnitude to constant (target flat spectrum)
        mag = np.abs(A)
        mag[mag < 1e-10] = 1e-10
        target_mag = np.median(mag)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Inverse FFT and keep first n elements
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
    
    return a

def softmax(x):

def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    exp_scaled = np.exp(scaled)
    return 2.0 * n * t * (bm / t + np.log(exp_scaled.sum()))


def _smooth_jac(x):

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
