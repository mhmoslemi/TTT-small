```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using SAMW (Away-step Frank-Wolfe)
with LP direction, multi-length search, and multi-phase polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_fast(f, M, n):
    """Faster LP solve using the banded structure."""
    c = -np.ones(n)
    # The constraint matrix has a banded/sparse structure
    # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f_{k-j} * g_j <= M
    # For each k, the non-zero entries in row k are at positions j where
    # f_{k-j} is defined, i.e., j in [max(0, k-n+1), min(n-1, k)]
    # The coefficient at position j in row k is f_{k-j}
    
    # Build sparse matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub_sparse = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub_sparse, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Away-step Frank-Wolfe (SAMW)

Key improvements over parent: SAMW away steps, longer length exploration (1024),

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_and_solve_lp(f, M, n):
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def samw_optimize(f0, time_limit, verbose=False):
    """Away-step Frank-Wolfe with LP direction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()

    it = 0

    while time.monotonic() - t0 < time_limit:

            f_proj[v_idx] = 0.0

            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

                        lo = mid
                    else:
                        hi = mid
        
        # Choose the better step
        if best_val_away < best_val_normal:
            # Use away step
            f_new = (1.0 - best_away_alpha) * f_proj + best_away_alpha * g
            f_new = np.maximum(f_new, 0.0)

        elif best_alpha > 1e-10:

def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        k_actual = max(k_actual, 5)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        cur_temp *= 1.003
        
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    
    return best_f, best_obj
