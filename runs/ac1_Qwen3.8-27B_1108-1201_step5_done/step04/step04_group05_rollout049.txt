"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved Frank-Wolfe with
asymmetry-aware LP oracle, multi-lag subgradient, and adaptive time allocation.
Key improvements: bias LP direction toward asymmetry, use top-k lag weighting,
and employ multiplicative updates for better simplex geometry.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_mat(f):
    """Build the 2n-1 by n matrix where A[k,j] = f[j] if k-j is in [0,n-1] else 0.
    So that A @ g = f * g (cross-convolution)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_v2(f, M, asym_bias=0.0, rng=None):
    """Solve LP: max sum(g) + asym_bias*(weighted asymmetry) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    if asym_bias != 0.0:
        # Bias toward making g more asymmetric (front-heavy or back-heavy)
        # Use a linear ramp as the bias direction
        ramp = np.arange(n, dtype=np.float64) / n
        ramp = ramp - 0.5  # centered
        c += asym_bias * ramp
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    no_improve = 0

def sg_polish_v2(f0, time_limit, seed=1):
    """Improved subgradient with multiplicative updates and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng_local = np.random.default_rng(seed)
    
    # Adaptive parameters
    lr = 0.005
    top_k = max(8, n // 4)
    temp = 0.0001  # start hot
    final_temp = 0.000001
    total_planned = 3000
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > total_planned:
            break
        
        # Adapt temperature (anneal)
        progress = min(1.0, it / total_planned)
        temp = temp * (1 - 0.001) + final_temp * 0.001  # slow anneal
        # Or exponential: temp = temp0 * (final/temp0)^(progress)
        
        # Get top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
        
        # Softmax weights
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Multiplicative update (mirror descent)
        # f_new[i] proportional to f[i] * exp(-lr * sg[i])
        log_f = np.log(f + 1e-30)
        log_f -= lr * sg
        f_new = np.exp(log_f - log_f.max())
        f_new = f_new / f_new.sum()
        
        # Check if we got zeros
        if not np.all(f_new > 0):
            # Fall back to additive
            f_new = project_simplex(f - lr * sg)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0

asymmetry-aware LP oracle, annealed subgradient polishing, pairwise swap
local search, and adaptive multi-phase time allocation.
Key improvements over previous: adaptive asymmetry bias in LP, multiplicative

with more focus on deep polishing of the best candidate.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f):
    """Build convolution matrix A where A[k,j] = f[j] if j <= k <= j+n-1."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, bias=0.0):
    """Solve LP: max sum(g) + bias*ramp@g s.t. A@... (f*g)[k]<=M, g>=0.
    Returns optimal g or None."""
    n = len(f)
    c = -np.ones(n)
    if abs(bias) > 1e-10:
        ramp = (np.arange(n, dtype=np.float64) - 0.5 * (n - 1)) / n
        c += bias * ramp
    A = build_A(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"presolve": True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_search(f0, time_limit, seed=42, verbose=False):
    """Frank-Wolfe with LP oracle, adaptive M, and asymmetry bias."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng_local = np.random.default_rng(seed)
    m_exp = 1.0
    bias = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        # Adapt asymmetry bias
        if no_improve > 25:
            bias = min(1.0, bias + 0.05)
        elif no_improve < 3:
            bias = max(0.0, bias * 0.85)

        g = solve_lp_direction(f, M * m_exp, bias=bias)
        if g is None:
            m_exp *= 1.08
            if m_exp > 8.0:
                m_exp = 1.0
                bias = 0.0
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        # Check if direction is useful
        direction_gain = M - np.dot(f, build_A(g).T @ f)  # rough
        diff = np.max(np.abs(g - f))
        if diff < 5e-8:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 2.0)
            if no_improve > 50:
                break
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M

        coarse_alphas = np.array([0.01, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25,
                                   0.35, 0.45, 0.55, 0.65, 0.75, 0.85, 0.95, 1.0])
        for alpha in coarse_alphas:

            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection around best coarse
        if 0.01 < best_alpha < 0.99:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(12):

                ft = (1.0 - mid) * f + mid * g

                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      FW iter {it}: obj={best_obj:.10f} bias={bias:.3f}")

                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 4.0)

        # Restart
        if no_improve > 60:

            strat = strategies[rng_local.integers(len(strategies))]

                f = project_simplex(best_f + rng_local.standard_normal(n) * sc)
            elif strat == "mirror":

                sc = rng_local.uniform(0.03, 0.1)

            elif strat == "dirichlet":

            else:
                ramp = np.linspace(0, 1, n)
                bl = rng_local.uniform(0.1, 0.4)
                f = project_simplex((1 - bl) * best_f + bl * ramp)

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            bias = 0.0

def subgradient_polish(f0, time_limit, seed=1):
    """Annealed subgradient with top-k peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng_local = np.random.default_rng(seed)

    # Parameters
    lr_init = 0.004
    mom = 0.92
    top_k = max(6, n // 5)
    temp_init = 0.0001
    temp_min = 0.000002

    v = np.zeros(n)
    lr = lr_init
    temp = temp_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Anneal temperature
        temp = max(temp * 0.9995, temp_min)
        lr = max(lr * 0.9998, 0.0003)

        # Top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]

        # Softmax weights
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Momentum update
        v = mom * v - lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        co = conv_new.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            sc = 0.002 + 0.002 * rng_local.random()
            f = project_simplex(best_f + rng_local.standard_normal(n) * sc)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def pairwise_swap_search(f0, time_limit, seed=7):
    """Local search: try moving mass between pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng_local = np.random.default_rng(seed)

    # Precompute which lags are "hot" (near the max)
    threshold = best_obj * 0.9
    hot_lags = np.where(conv >= threshold)[0]

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break

        improved = False
        # Try random pairs
        for _ in range(200):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            i = rng_local.integers(n)
            j = rng_local.integers(n)
            if i == j:
                continue

            # Try moving fraction of mass from j to i
            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = f[j] * frac
                if delta < 1e-15:
                    continue
                f_new = f.copy()
                f_new[i] += delta
                f_new[j] -= delta
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                if sn < 1e-15:
                    continue
                f_new = f_new / sn

                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-15:
                    f = f_new
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # No improvement in this round, try structured moves
            # Move mass from high-value positions to low-value ones
            sorted_idx = np.argsort(f)
            low_idx = sorted_idx[:n // 2]
            high_idx = sorted_idx[n // 2:]

            found = False
            for _ in range(50):
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                i = rng_local.integers(len(low_idx))
                j = rng_local.integers(len(high_idx))
                src = int(high_idx[j])
                dst = int(low_idx[i])
                for frac in [0.05, 0.1, 0.15]:
                    delta = f[src] * frac
                    if delta < 1e-15:
                        continue
                    f_new = f.copy()
                    f_new[dst] += delta
                    f_new[src] -= delta
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-15:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        found = True
                        break
                if found:
                    break

            if not found:
                break  # Converged

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Strong ramp (front-heavy)
    ramp_f = np.exp(-x / (n * 0.3))
    inits.append(ramp_f / ramp_f.sum())

    # Strong ramp (back-heavy)
    ramp_b = np.exp(-(n - 1 - x) / (n * 0.3))
    inits.append(ramp_b / ramp_b.sum())

    # Linear front-heavy
    lin_f = (n - x) / (n - x).sum()
    inits.append(lin_f)

    # Linear back-heavy
    lin_b = (x + 1) / (x + 1).sum()
    inits.append(lin_b)

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two peaks asymmetric
    p1 = np.exp(-((x - n * 0.3) ** 2) / (2 * (n * 0.1) ** 2))
    p2 = np.exp(-((x - n * 0.7) ** 2) / (2 * (n * 0.15) ** 2))
    two_peak = p1 + 0.6 * p2
    inits.append(two_peak / two_peak.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Cosine taper (front)
    cos_t = np.cos(np.pi * x / (2.0 * n))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.6))

    # Power law
    pl = (x + 1.0) ** (-0.8)
    inits.append(pl / pl.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Reversed parent
        px_r = px[::-1].copy()
        inits.append(px_r / px_r.sum())

    return inits

        a_opt, obj = fw_lp_search(state['best_seq'], time_limit=t_main, seed=seed + 100, verbose=True)
        try_sequence(a_opt, "p2_main", state)

    # Perturbation restarts
    pert_scales = [0.003, 0.008, 0.015, 0.03, 0.06, 0.1]
    for ps in pert_scales:
        rem = phase2_end - time.monotonic() - 20
        if rem < 15:
            break
        t_alloc = min(rem / 2.0, 80)
        if t_alloc < 10:
            break
        noise = rng.standard_normal(n_best) * ps
        f_init = project_simplex(state['best_seq'] + noise)
        a_opt, obj = fw_lp_search(f_init, time_limit=t_alloc, seed=seed + int(ps * 1000))
        try_sequence(a_opt, f"p2_pert{ps}", state)

    # Asymmetric blend
    rem = phase2_end - time.monotonic() - 15
    if rem > 20:
        for blend in [0.2, 0.4, 0.6]:
            if time.monotonic() > phase2_end - 15:
                break
            f镜 = project_simplex((1 - blend) * state['best_seq'] + blend * state['best_seq'][::-1])
            t_alloc = min(rem / 3.0, 60)
            if t_alloc < 10:
                break
            a_opt, obj = fw_lp_search(f镜, time_limit=t_alloc, seed=seed + 500 + int(blend * 10))
            try_sequence(a_opt, f"p2_mirror_{blend}", state)
            rem = phase2_end - time.monotonic() - 15

    # Additional diverse runs
    rem = phase2_end - time.monotonic() - 15
    if rem > 30:
        inits_extra = generate_inits(n_best, state['best_seq'], rng)
        for i in range(min(3, len(inits_extra))):
            if time.monotonic() > phase2_end - 15:
                break
            t_alloc = min(rem / 3.0, 50)
            if t_alloc < 10:
                break
            a_opt, obj = fw_lp_search(inits_extra[i + 4], time_limit=t_alloc, seed=seed + 200 + i)
            try_sequence(a_opt, f"p2_div{i}", state)

    print(f"  Phase 2 best: {state['best_score']:.8f}")

    # === Phase 3: Polishing (25% of budget) ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_time = total_budget * 0.25
    phase3_end = time.monotonic() + phase3_time

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        # Subgradient with multiple parameter settings
        rem = phase3_end - time.monotonic() - 30
        if rem > 20:
            t_sg = min(rem * 0.4, 80)
            a_opt, obj = subgradient_polish(state['best_seq'], time_limit=t_sg, seed=seed + 300)
            try_sequence(a_opt, "p3_sg1", state)

        # Second SG with different params
        rem = phase3_end - time.monotonic() - 20
        if rem > 15:
            t_sg = min(rem * 0.3, 50)
            a_opt, obj = subgradient_polish(state['best_seq'], time_limit=t_sg, seed=seed + 400)
            try_sequence(a_opt, "p3_sg2", state)

        # Pairwise swap local search
        rem = phase3_end - time.monotonic() - 10
        if rem > 10:
            t_ps = min(rem, 60)
            a_opt, obj = pairwise_swap_search(state['best_seq'], time_limit=t_ps, seed=seed + 500)
            try_sequence(a_opt, "p3_swap", state)

        # Another round of FW on the polished result
        rem = phase3_end - time.monotonic() - 5
        if rem > 15:
            t_fw = min(rem, 40)
            a_opt, obj = fw_lp_search(state['best_seq'], time_limit=t_fw, seed=seed + 600)
            try_sequence(a_opt, "p3_fw", state)

    print(f"  Phase 3 best: {state['best_score']:.8f}")

    # === Phase 4: Final squeeze (5% of budget) ===
    print(f"\n=== Phase 4: Final Squeeze ===")
    phase4_end = deadline

    if state['best_seq'] is not None:
        # Final short FW
        rem = phase4_end - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = fw_lp_search(state['best_seq'], time_limit=min(rem * 0.5, 30), seed=seed + 700)
            try_sequence(a_opt, "p4_fw", state)

        # Final swap
        rem = phase4_end - time.monotonic() - 3
        if rem > 8:
            a_opt, obj = pairwise_swap_search(state['best_seq'], time_limit=min(rem, 20), seed=seed + 800)
            try_sequence(a_opt, "p4_swap", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    # Final verification
    final_score = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(final_score):
        state['best_score'] = final_score

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    print(f"Time used: {time.monotonic() - (deadline - total_budget):.1f}s")

    return [float(x) for x in state['best_seq']]
