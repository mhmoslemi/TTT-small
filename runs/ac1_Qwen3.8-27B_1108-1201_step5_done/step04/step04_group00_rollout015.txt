"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smooth LSE-approximated gradient
descent with simulated annealing, multi-length search, and coordinate descent
polishing. Uses FFT-based convolution for O(n log n) gradient computation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n <= 1:
        return np.maximum(v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, T):
    """Compute gradient of LSE(conv(a,a)/T) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    conv_aa = fast_convolve(a)
    scaled = conv_aa / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[np.argmax(conv_aa)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1].copy()
    result_len = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (result_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(W * AR, fft_size)[:result_len]
    grad = (2.0 / T) * conv_result[n - 1:n - 1 + n]
    return grad

    """
    a = project_simplex(a_init.copy().astype(np.float64))

    conv = fast_convolve(a)
    cur_obj = conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    it = 0

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))

        T = max(T, cur_obj * 0.005)

                    best_mult = mult

            if best_mult != 1.0:
                a[j] = old_val * best_mult
                a /= a.sum()
                if best_local_obj < best_obj - 1e-14:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_raised_cosine(n):
    """Create a raised-cosine initial shape."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_gaussian(n, sigma_ratio=0.4, center=0.5):
    """Create a Gaussian initial shape."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma_ratio) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_asymmetric_bump(n, center=0.35, width=0.5):
    """Create an asymmetric bump (skewed) initial shape."""
    x = np.linspace(0, 1, n)
    # Asymmetric: sharper on one side, broader on the other
    left = (center - x) / (width * 0.6) for x < center
    right = (x - center) / (width * 1.2) for x >= center
    a = np.exp(-0.5 * np.where(x < center, left, right) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def propose_candidate(budget_s=1800):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LSE-smoothed gradient descent with simulated annealing,
    block coordinate optimization, multi-length search, and CD polishing.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    best_seq = None
    best_score = float('inf')

    def _try_seq(a, label=""):
        nonlocal best_seq, best_score
        a_flat = np.maximum(a, 0.0)
        if a_flat.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a_flat.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_flat.copy()
            print(f"  NEW BEST [{label}]: score={score:.6f} (n={len(a_flat)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Phase 1: SGD on parent length with multiple initializations
    print("\n=== Phase 1: SGD on parent ===")
    if parent is not None:
        n_parent = len(parent)
        conv = fast_convolve(parent)
        peak = conv.max()

        # Run SGD from parent
        t_alloc = min(400, (deadline - time.monotonic()) * 0.4)
        if t_alloc > 30:
            a_opt, _ = smooth_gradient_descent(
                parent, time_limit=t_alloc,
                T_ratio_init=2.0, T_ratio_final=0.03,
                lr_init=0.15, mom=0.85,
                noise_init=0.004, noise_decay=0.998, seed=1
            )
            _try_seq(a_opt, "sgd_parent")

        # SGD from slightly perturbed parent
        rem = deadline - time.monotonic()
        if rem > 60:
            t_alloc = min(200, rem * 0.3)
            rng = np.random.default_rng(77)
            a_pert = project_simplex(parent + rng.standard_normal(n_parent) * 0.02)
            a_opt, _ = smooth_gradient_descent(
                a_pert, time_limit=t_alloc,
                T_ratio_init=1.5, T_ratio_final=0.02,
                lr_init=0.1, mom=0.9,
                noise_init=0.003, noise_decay=0.999, seed=2
            )
            _try_seq(a_opt, "sgd_perturb")

    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else (parent if parent is not None else make_raised_cosine(512))

    candidate_lengths = sorted(set([len(base), 512, 640, 768, 896, 1024, 1152]))
    candidate_lengths = [l for l in candidate_lengths if 256 <= l <= 2048]

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break

        if target_n == len(base):
            a_init = base.copy()
        else:
            a_init = interpolate_sequence(base, target_n)

        t_alloc = min(rem * 0.25, 250)

        # SGD from interpolated
        a_opt, _ = smooth_gradient_descent(
            a_init, time_limit=t_alloc,
            T_ratio_init=2.0, T_ratio_final=0.03,
            lr_init=0.12, mom=0.88,
            noise_init=0.005, noise_decay=0.998, seed=target_n % 1000
        )
        _try_seq(a_opt, f"sgd_n{target_n}")

        # Block coordinate after SGD
        rem = deadline - time.monotonic()
        if rem > 30:
            t_bc = min(rem * 0.15, 80)
            a_opt2, _ = block_coordinate_optimize(a_opt, time_limit=t_bc, block_size=8, seed=target_n)
            _try_seq(a_opt2, f"block_n{target_n}")

    # Phase 3: Different shape initializations
    print(f"\n=== Phase 3: Shape exploration (best={best_score:.6f}) ===")
    n_try = len(best_seq) if best_seq is not None else 768

    shapes = [
        ("raised_cos", make_raised_cosine(n_try)),
        ("gauss_c030", make_gaussian(n_try, 0.35, 0.30)),
        ("gauss_c050", make_gaussian(n_try, 0.40, 0.50)),
        ("gauss_c070", make_gaussian(n_try, 0.35, 0.70)),
        ("asym_bump", make_asymmetric_bump(n_try, 0.35, 0.5)),
        ("asym_bump2", make_asymmetric_bump(n_try, 0.60, 0.4)),
    ]

    for name, a_shape in shapes:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.15, 120)
        a_opt, _ = smooth_gradient_descent(
            a_shape, time_limit=t_alloc,
            T_ratio_init=3.0, T_ratio_final=0.02,
            lr_init=0.15, mom=0.85,
            noise_init=0.006, noise_decay=0.997, seed=hash(name) % 10000
        )
        _try_seq(a_opt, f"shape_{name}")

    # Phase 4: Deep optimization on best
    print(f"\n=== Phase 4: Deep optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        for p in range(4):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem * 0.25, 180)

            if p == 0:
                a_init = best_seq.copy()
                lr_use = 0.03
                mom_use = 0.95
                noise_use = 0.001
            elif p == 1:
                rng = np.random.default_rng(100 + p)
                a_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
                lr_use = 0.05
                mom_use = 0.92
                noise_use = 0.003
            elif p == 2:
                # Mirror blend
                a_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                a_init = project_simplex(a_init)
                lr_use = 0.05
                mom_use = 0.93
                noise_use = 0.002
            else:
                # Flatten blend
                a_init = 0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best)
                a_init = project_simplex(a_init)
                lr_use = 0.04
                mom_use = 0.94
                noise_use = 0.002

            a_opt, _ = smooth_gradient_descent(
                a_init, time_limit=t_alloc,
                T_ratio_init=1.5, T_ratio_final=0.01,
                lr_init=lr_use, mom=mom_use,
                noise_init=noise_use, noise_decay=0.9995, seed=500 + p
            )
            _try_seq(a_opt, f"deep_{p}")

    # Phase 5: Coordinate descent polishing
    print(f"\n=== Phase 5: CD Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.5, 200)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd, seed=123)
            _try_seq(a_opt, "cd_polish")

    # Phase 6: Final combined polish
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Block coordinate
        rem = deadline - time.monotonic()
        if rem > 30:
            t_bc = min(rem * 0.4, 120)
            a_opt, _ = block_coordinate_optimize(best_seq, time_limit=t_bc, block_size=4, seed=999)
            _try_seq(a_opt, "block_final")

        # Tiny SGD
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sgd = min(rem * 0.4, 80)
            a_opt, _ = smooth_gradient_descent(
                best_seq, time_limit=t_sgd,
                T_ratio_init=0.5, T_ratio_final=0.005,
                lr_init=0.01, mom=0.97,
                noise_init=0.0005, noise_decay=0.9998, seed=7777
            )
            _try_seq(a_opt, "sgd_tiny")

        # Final CD
        rem = deadline - time.monotonic()
        if rem > 20:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem * 0.6, 60), seed=456)
            _try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = make_raised_cosine(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
