        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_grad(f, tau):
    """Compute LSE approximation of max(conv(f,f)) and its gradient via FFT."""
    n = len(f)
    conv = fast_convolve(f)
    max_c = conv.max()
    scaled = tau * (conv - max_c)
    # Clamp for numerical stability
    scaled = np.clip(scaled, -700, 700)
    exp_c = np.exp(scaled)
    z = exp_c.sum()
    if z < 1e-300:
        z = 1e-300
    weights = exp_c / z
    obj = max_c + np.log(z) / tau
    
    # Gradient via FFT cross-correlation
    # grad[i] = 2 * sum_j weights[i+j] * f[j]
    L = 1 << (3 * n).bit_length()  # >= 3n
    W = np.fft.rfft(weights, L)
    F = np.fft.rfft(f, L)
    corr = np.fft.irfft(np.conj(W) * F, L)
    grad = 2.0 * corr[:n]
    
    return obj, grad, conv

def nesterov_lse(f0, time_limit, tau_init=100.0, tau_max=5000.0, 
                  lr_init=0.01, verbose=False, seed=42):
    """Nesterov-accelerated optimization of LSE-smoothed auto-conv peak."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    y = f.copy()
    f_prev = f.copy()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    no_improve = 0
    it = 0
    momentum = 0.9
    
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        obj, grad, conv = lse_grad(y, tau)
        
        # Nesterov step
        f_new = project_simplex(y - lr * grad)
        y = f_new + momentum * (f_new - f)
        f = f_new
        
        # Check actual objective (true max, not LSE)
        true_obj = fast_convolve(f).max()
        
        if true_obj < best_obj - 1e-14:
            best_obj = true_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    Nesterov iter {it}: true_obj={best_obj:.10f}, tau={tau:.1f}")
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 20:
            lr *= 0.85
            if no_improve > 100:
                lr = max(lr, 0.001)
                # Increase tau to sharpen
                tau = min(tau * 1.05, tau_max)
        
        # Periodic momentum adjustment
        if it % 200 == 0:
            momentum = 0.9 + 0.05 * np.sin(it / 200.0)
        
        # Restart if stuck too long
        if no_improve > 300:
            print(f"    Restart at iter {it}, no_improve={no_improve}")
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            f_prev = f.copy()
            lr = lr_init * 0.5
            tau = tau_init * 2
            no_improve = 0
            momentum = 0.9
    
    return best_f, best_obj

def fw_lp_one_step(f, M):
    """One Frank-Wolfe step using LP oracle."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def coordinate_descent(f0, time_limit, top_k=20):
    """Coordinate descent targeting the peak of auto-convolution."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peak_k = np.argmax(conv)
        
        # Get the pairs contributing to the peak
        # c[peak_k] = sum_j f[j] * f[peak_k - j]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        pairs = []
        for j in range(js, je + 1):
            pairs.append((j, peak_k - j))
        
        # Find the highest contributing pair
        if not pairs:
            break
        
        contributions = [f[j] * f[k] for j, k in pairs]
        max_idx = int(np.argmax(contributions))
        j, k = pairs[max_idx]
        
        # Try to reduce f[j]*f[k] by spreading mass
        # Move small amount from the larger of f[j], f[k] to a random other position
        if j == k:
            source = j
        elif f[j] > f[k]:
            source = j
        else:
            source = k
        
        # Choose target: a position that doesn't contribute to the peak
        # (i.e., position t where (t, peak_k - t) doesn't involve source)
        # Simplified: just pick a random position far from the peak
        for attempt in range(10):
            target = rng.integers(n)
            if target == source:
                continue
            # Check if target contributes to peak
            if js <= target <= je and js <= peak_k - target <= je:
                if peak_k - target == source or target == source:
                    continue
            # Try the move
            for frac in [0.1, 0.05, 0.02, 0.01]:
                delta = f[source] * frac
                f_new = f.copy()
                f_new[source] -= delta
                f_new[target] += delta
                if f_new[source] < 0:
                    f_new[target] += f_new[source]
                    f_new[source] = 0.0
                f_new = np.maximum(f_new, 0.0)
                s_new = f_new.sum()
                if s_new < 1e-15:
                    continue
                f_new = f_new / s_new
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-15:
                    f = f_new
                    best_obj = val
                    best_f = f.copy()
                    break
        
        # Periodic full shuffle
        if it % 500 == 0:
            for _ in range(n):
                source = rng.integers(n)
                target = rng.integers(n)
                if source == target:
                    continue
                delta = f[source] * 0.05
                f_new = f.copy()
                f_new[source] -= delta
                f_new[target] += delta
                f_new = np.maximum(f_new, 0.0)
                s_new = f_new.sum()
                if s_new > 1e-15:
                    f_new = f_new / s_new
                    val = fast_convolve(f_new).max()
                    if val < best_obj:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
    
    return best_f, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    inits.append(cos_t / cos_t.sum())

        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

"""
Multi-stage Nesterov-optimized step function minimization.
Uses log-sum-exp smoothed auto-convolution peak with FFT-accelerated gradients,
Nesterov acceleration, adaptive temperature scheduling, FW-LP hybrid steps,
and targeted coordinate descent polishing to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum=1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def lse_objective_and_grad(f, tau):
    """
    Compute LSE approximation of max(conv(f,f)) and its gradient via FFT.
    Returns (lse_value, gradient, raw_convolution).
    """
    n = len(f)
    conv = fast_convolve(f)
    max_c = conv.max()
    scaled = tau * (conv - max_c)
    scaled = np.clip(scaled, -700, 700)
    exp_c = np.exp(scaled)
    z = exp_c.sum()
    if z < 1e-300:
        z = 1e-300
    weights = exp_c / z
    lse_val = max_c + np.log(z) / tau

    # Gradient via FFT cross-correlation:
    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * f[j]
    # This is 2*(weights *star f)[:n] where *star is cross-correlation
    L = 1 << (3 * n).bit_length()
    W = np.fft.rfft(weights, L)
    F = np.fft.rfft(f, L)
    corr = np.fft.irfft(np.conj(W) * F, L)
    grad = 2.0 * corr[:n]

    return lse_val, grad, conv


def nesterov_optimize(f0, time_limit, tau_init=100.0, tau_max=10000.0,
                      lr_init=0.01, momentum_init=0.9, verbose=False, seed=42):
    """
    Nesterov-accelerated projected gradient on LSE-smoothed auto-conv peak.
    Works on the simplex (sum f = 1, f >= 0).
    """
    n = len(f0)
    f = normalize_nonneg(f0)
    y = f.copy()
    f_prev = f.copy()

    conv0 = fast_convolve(f)
    best_obj = conv0.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    momentum = momentum_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        _, grad, conv_y = lse_objective_and_grad(y, tau)

        # Nesterov step
        f_new = project_simplex(y - lr * grad)
        y = f_new + momentum * (f_new - f)
        f = f_new
        f_prev = f_new

        # Evaluate true objective at current point f
        true_obj = fast_convolve(f).max()

        if true_obj < best_obj - 1e-14:
            best_obj = true_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Nesterov iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.5f}")
        else:
            no_improve += 1

        # Adaptive learning rate and temperature
        if no_improve > 25:
            lr *= 0.88
        if no_improve > 80:
            tau = min(tau * 1.03, tau_max)
        if no_improve > 200:
            tau = min(tau * 1.05, tau_max)

        # Adaptive momentum (slowly increasing then oscillating)
        if it % 500 == 0:
            momentum = 0.85 + 0.1 * min(1.0, it / 5000.0)

        # Restart if stuck
        if no_improve > 600:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            f_prev = f.copy()
            lr = max(lr_init * 0.5, 0.002)
            tau = min(tau * 0.5, tau_init * 5)
            momentum = 0.9
            no_improve = 0
            if verbose:
                print(f"    Restart at iter {it}")

    return best_f, best_obj


def fw_lp_step(f, M):
    """Single Frank-Wolfe step: solve LP max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def fw_line_search(f, g, M, n_alpha=12):
    """Line search: find best alpha in [0,1] for (1-alpha)*f + alpha*g."""
    best_alpha = 0.0
    best_val = M
    for alpha in np.linspace(0.05, 1.0, n_alpha):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_alpha, best_val


def fw_lp_optimize(f0, time_limit, n_lp_steps=10, verbose=False):
    """FW with LP oracle for a few steps, then return best found."""
    f = normalize_nonneg(f0)
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M

    t0 = time.monotonic()
    it = 0
    m_expand = 1.0

    while time.monotonic() - t0 < time_limit and it < n_lp_steps * 5:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        g = fw_lp_step(f, M * m_expand)
        if g is None:
            m_expand *= 1.1
            if m_expand > 5.0:
                m_expand = 1.0
            continue

        # Check fixed point
        if np.max(np.abs(g - f)) < 1e-10:
            if m_expand < 3.0:
                m_expand *= 1.1
                continue
            break

        alpha, val = fw_line_search(f, g, M)
        if alpha > 1e-10 and val < M - 1e-14:
            f = (1.0 - alpha) * f + alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            m_expand = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose:
                    print(f"    FW-LP iter {it}: obj={best_obj:.10f}")
        else:
            m_expand = min(m_expand * 1.05, 4.0)

    return best_f, best_obj


def coordinate_descent_targeted(f0, time_limit, n_shuffles=50):
    """
    Coordinate descent that directly targets the peak of auto-convolution.
    Identifies the peak, finds the highest contributing pair, and surgically
    redistributes mass to reduce the peak.
    """
    n = len(f0)
    f = normalize_nonneg(f0)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Find contributing pairs for this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            break

        # Contributions from each pair
        j_arr = np.arange(js, je + 1)
        k_arr = peak_k - j_arr
        contributions = f[j_arr] * f[k_arr]

        # Find top contributors
        n_top = min(5, len(j_arr))
        top_idx = np.argsort(contributions)[::-1][:n_top]

        for idx in top_idx:
            j = int(j_arr[idx])
            k_pos = int(k_arr[idx])

            # Source: the larger of f[j], f[k_pos]
            if f[j] >= f[k_pos]:
                source = j
            else:
                source = k_pos

            # Target: position that minimizes coupling at the peak
            # Prefer positions t where t and peak_k-t are both != source
            candidates = []
            for _ in range(20):
                t = int(rng.integers(n))
                if t == source:
                    continue
                partner = peak_k - t
                if 0 <= partner < n and partner == source:
                    continue
                # Score: prefer low f[t] (to spread mass) and low coupling
                score = f[t] + (f[partner] if 0 <= partner < n else 0)
                candidates.append((score, t))

            if not candidates:
                continue
            candidates.sort()
            _, target = candidates[0]

            # Try the move with different fractions
            for frac in [0.05, 0.02, 0.01, 0.005]:
                delta = f[source] * frac
                if delta < 1e-12:
                    break
                f_new = f.copy()
                f_new[source] -= delta
                f_new[target] += delta
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                if sn < 1e-15:
                    continue
                f_new = f_new / sn
                new_conv = fast_convolve(f_new)
                new_peak = new_conv.max()
                if new_peak < best_obj - 1e-14:
                    f = f_new
                    best_obj = new_peak
                    best_f = f.copy()
                    no_improve = 0
                    break

        # Periodic global shuffle to escape local minima
        if it % n_shuffles == 0:
            for _ in range(n // 2):
                s = int(rng.integers(n))
                t = int(rng.integers(n))
                if s == t:
                    continue
                delta = f[s] * 0.03
                f_new = f.copy()
                f_new[s] -= delta
                f_new[t] += delta
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                if sn > 1e-15:
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()

        if it % 200 == 0:
            no_improve += 1
            if no_improve > 3:
                # Larger perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0

    return best_f, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine taper
    cos_t = np.cos(np.pi * (x + 0.5) / (2.0 * n))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

                                       0.3 * state['best_seq'][::-1])
            a_opt, obj = nesterov_optimize(f_mirror, time_limit=min(rem, 60),
                                           tau_init=100.0, lr_init=0.01, seed=800)
            try_seq(a_opt, "mirror")

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    # === Phase 3: Polishing (30% of time) ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline - 5

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f} (n={n_best})")

        # Targeted coordinate descent (long)
        rem = phase3_end - time.monotonic() - 40
        if rem > 20:
            t_cd = min(rem, 120)
            print(f"  CD targeted polish, t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent_targeted(state['best_seq'], time_limit=t_cd)
            try_seq(a_opt, "cd_long")

        # Fine Nesterov with high tau (sharp)
        rem = phase3_end - time.monotonic() - 30
        if rem > 20:
            t_fine = min(rem, 80)
            print(f"  Fine Nesterov (high tau), t={t_fine:.0f}s")
            a_opt, obj = nesterov_optimize(state['best_seq'], time_limit=t_fine,
                                           tau_init=1000.0, tau_max=50000.0,
                                           lr_init=0.003, momentum_init=0.85,
                                           seed=900)
            try_seq(a_opt, "fine_nest")

        # Another CD pass
        rem = phase3_end - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = coordinate_descent_targeted(state['best_seq'],
                                                     time_limit=min(rem, 60))
            try_seq(a_opt, "cd2")

        # Final Nesterov squeeze
        rem = phase3_end - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = nesterov_optimize(state['best_seq'], time_limit=min(rem, 40),
                                           tau_init=2000.0, tau_max=100000.0,
                                           lr_init=0.001, momentum_init=0.8,
                                           seed=1000)
            try_seq(a_opt, "squeeze")

        # Ultra-fine CD
        rem = phase3_end - time.monotonic()
        if rem > 10 and state['best_seq'] is not None:
            a_opt, obj = coordinate_descent_targeted(state['best_seq'],
                                                     time_limit=min(rem, 30))
            try_seq(a_opt, "cd_final")

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
