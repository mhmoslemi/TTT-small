    if n == 1:
        return np.array([max(v[0], 0.0)])

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Nesterov-accelerated subgradient, reduced-constraint LP-FW,
coordinate descent, multi-length exploration, and diverse structured initialization.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, temp):
    """Compute weighted subgradient of max_k (f*f)_k using top peaks."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def nesterov_sg(f0, time_limit, lr0=0.02, top_k=30, temp=0.0001, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    f_prev = f.copy()
    t = 1
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temp
    cur_lr = lr0

    while time.monotonic() - t0 < time_limit:
        # Nesterov momentum
        beta = t / (t + 2.0)
        y = f + beta * (f - f_prev)
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k, cur_temp)

        f_new = y - cur_lr * sg
        f_new = project_simplex(f_new)

        f_prev = f
        f = f_new
        t += 1

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 200 == 0:
                print(f"    NSG iter {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive learning rate
        cur_lr = lr0 / (1.0 + 0.01 * t)
        cur_temp *= 1.005

        # Restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            f_prev = f.copy()
            t = 1
            cur_lr = lr0 * 0.5
            no_improve = 0

    return best_f, best_obj

def reduced_lp_fw_direction(f, conv, n, k_frac=0.3):
    """Solve reduced LP: max sum(g) s.t. (f*g)[k] <= M+k*tol for top-K peaks, g>=0."""
    M = conv.max()
    k_count = max(10, int(k_frac * (2 * n - 1)))
    top_k = min(k_count, 2 * n - 1)
    peak_indices = np.argsort(conv)[-top_k:]
    peak_vals = conv[peak_indices]
    # Use M + small tolerance for non-max peaks
    b_ub = np.zeros(top_k)
    for i, idx in enumerate(peak_indices):
        if np.isclose(conv[idx], M, atol=1e-10):
            b_ub[i] = M
        else:
            b_ub[i] = M + 0.01 * (M - conv[idx]) + 0.01 * M

    # Build constraint matrix (only for selected peaks)
    A_ub = np.zeros((top_k, n))
    for i, k in enumerate(peak_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je + 1] = f[js:je + 1]

    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        conv = fast_convolve(f)

        g = reduced_lp_fw_direction(f, conv, n, k_frac=k_frac)

            no_improve += 5

            continue

        if g_sum < 1e-15:

        g = g / g_sum

            no_improve += 1

                no_improve = 0
                continue

        # Line search

        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)

        # Refine with bisection

                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)

            f = f_new

            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def spectral_shaping(f0, n_iter=50):
    """Iterative spectral shaping: flatten the power spectrum of the padded sequence."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()

    for _ in range(n_iter):
        A = np.fft.rfft(f, fft_size)
        mag = np.abs(A)
        # Target: flat magnitude
        target = np.ones_like(mag)
        target[mag > 0] = np.mean(mag[mag > 0])
        # Blend: don't fully flatten, keep some structure
        new_mag = 0.7 * mag + 0.3 * target
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        # Project to non-negative
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss
        f = f_new

    return f


def interpolate_to_length(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_structured_init(n, kind='gaussian'):
    """Generate a structured initial sequence of length n."""
    if kind == 'gaussian':

        f = 0.5 + 0.4 * np.cos(np.pi * np.arange(n) / n)
    elif kind == 'triangle':
        f = 1.0 - 2.0 * np.abs(np.arange(n) - n / 2) / n
    elif kind == 'raised_cosine':
        x = np.arange(n) / n
        f = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif kind == 'parabolic':
        x = np.linspace(-1, 1, n)
        f = 1.0 - x * x
    elif kind == 'bimodal':
        x = np.linspace(-1, 1, n)
        f = np.exp(-4 * (x - 0.4) ** 2) + np.exp(-4 * (x + 0.4) ** 2)
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f

    for kind in ['gaussian', 'cosine']:
        f_struct = generate_structured_init(base_n, kind)
        f_blend = project_simplex(0.6 * parent + 0.4 * f_struct)
        candidates.append((f'blend_{kind}', f_blend))

    # Quick screen each candidate with short subgradient
    screen_time = 15  # seconds each
    screened = []
    for label, f_init in candidates:
        if time.monotonic() > t1_end - screen_time:
            break
        t_alloc = min(screen_time, t1_end - time.monotonic())
        if t_alloc < 5:
            break
        f_opt, obj = plain_sg(f_init, time_limit=t_alloc,
                              lr=0.003, momentum=0.9, top_k=min(30, base_n // 4),
                              temp=0.0001)
        screened.append((obj, f_opt, label))
        print(f"    Screen {label}: obj={obj:.8f}")

    screened.sort(key=lambda x: x[0])
    # Keep top candidates
    top_candidates = screened[:min(4, len(screened))]
    print(f"  Top after screening: {[f'{l}: {o:.8f}' for o, _, l in top_candidates]}")

    # Evaluate top candidates
    for obj, f_opt, label in top_candidates:
        try_seq(f_opt, f'screen_{label}')

    # === Phase 2: Deep hybrid optimization (45 min) ===
    print(f"\n=== Phase 2: Deep Hybrid Optimization ===")
    if best_seq is not None:
        start_best = best_seq
    else:
        start_best = top_candidates[0][1] if top_candidates else parent

    n_best = len(start_best)
    t2_end = time.monotonic() + min(2700, (deadline - time.monotonic()) * 0.35)

    # Run hybrid on best
    t_alloc = min(200, t2_end - time.monotonic())
    if t_alloc > 30:
        print(f"  Hybrid on best (n={n_best}), t={t_alloc:.0f}s")
        f_opt, _ = hybrid_optimize(start_best, time_limit=t_alloc, verbose=True)
        try_seq(f_opt, 'hybrid_best')

    # Run hybrid on second best if different
    for obj, f_opt, label in top_candidates[1:2]:
        if time.monotonic() > t2_end - 30:
            break
        rem = t2_end - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.8, 150)
        print(f"  Hybrid on {label}, t={t_alloc:.0f}s")
        f_opt2, _ = hybrid_optimize(f_opt, time_limit=t_alloc)
        try_seq(f_opt2, f'hybrid_{label}')

    # === Phase 3: LP-FW on best (30 min) ===
    print(f"\n=== Phase 3: LP-FW Deep (best={best_score:.6f}) ===")
    if best_seq is not None:
        t3_end = time.monotonic() + min(1800, (deadline - time.monotonic()) * 0.25)

        for p in range(3):
            rem = t3_end - time.monotonic()
            if rem < 20:
                break
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(len(best_seq)) * 0.003
                f_init = project_simplex(best_seq + noise)
            else:
                f_init = project_simplex(0.8 * best_seq + 0.2 * (np.ones(len(best_seq)) / len(best_seq)))

            t_alloc = min(rem * 0.7, 180)
            print(f"  LP-FW variant {p}, t={t_alloc:.0f}s")
            f_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, k_frac=0.25, verbose=(p == 0))
            try_seq(f_opt, f'lpfw_{p}')

    # === Phase 4: Coordinate descent (20 min) ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_cd = min(rem * 0.6, 180)
            print(f"  CD t={t_cd:.0f}s")
            f_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(f_opt, 'cd')

            # SG after CD
            rem = deadline - time.monotonic() - 60
            if rem > 10:
                f_opt, _ = plain_sg(best_seq, time_limit=min(rem, 100),
                                    lr=0.002, momentum=0.92, top_k=max(20, len(best_seq) // 5),
                                    temp=0.00003)
                try_seq(f_opt, 'sg_after_cd')

    # === Phase 5: Multi-length exploration (25 min) ===
    print(f"\n=== Phase 5: Multi-Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        base_f = best_seq
        t5_end = time.monotonic() + min(1500, (deadline - time.monotonic()) * 0.3)

        # Try several lengths around the current best
        current_n = len(base_f)
        target_lengths = []
        for factor in [0.75, 0.85, 1.0, 1.15, 1.3, 1.5]:
            target_n = int(round(current_n * factor / 32)) * 32
            target_n = max(64, min(target_n, 4096))
            if target_n != current_n and target_n not in target_lengths:
                target_lengths.append(target_n)

        for target_n in target_lengths:
            rem = t5_end - time.monotonic()
            if rem < 30:
                break
            f_interp = interpolate_to_length(base_f, target_n)
            t_alloc = min(rem * 0.5, 90)
            print(f"  Length {target_n}: quick opt, t={t_alloc:.0f}s")
            f_opt, _ = plain_sg(f_interp, time_limit=t_alloc,
                                lr=0.004, momentum=0.9, top_k=min(40, target_n // 4),
                                temp=0.0001)
            try_seq(f_opt, f'len{target_n}')

            # If this length improved, do deeper optimization
            if best_seq is not None and len(best_seq) == target_n:
                rem = t5_end - time.monotonic()
                if rem > 30:
                    t_alloc2 = min(rem * 0.5, 60)
                    f_opt2, _ = hybrid_optimize(best_seq, time_limit=t_alloc2)
                    try_seq(f_opt2, f'len{target_n}_deep')

    # === Phase 6: Final polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_final = len(best_seq)

        # Fine subgradient
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            f_opt, _ = plain_sg(best_seq, time_limit=min(rem, 80),
                                lr=0.001, momentum=0.95, top_k=max(15, n_final // 6),
                                temp=0.00001)
            try_seq(f_opt, 'final_sg')

        # Another LP-FW
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            f_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 60), k_frac=0.2)
            try_seq(f_opt, 'final_lpfw')

        # Small perturbations + re-optimize
        for _ in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_final) * 0.001
            f_pert = project_simplex(best_seq + noise)
            f_opt, _ = plain_sg(f_pert, time_limit=min(rem, 30),
                                lr=0.001, momentum=0.95, top_k=max(15, n_final // 6),
                                temp=0.00001)
            try_seq(f_opt, 'perturb')

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
