            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Golden section refinement
        if best_alpha > 0.02:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            ft1 = np.maximum(f + c1 * (g - f), 0)
            ft1 = ft1 / max(ft1.sum(), 1e-15)
            fc1 = fast_convolve(ft1).max()
            ft2 = np.maximum(f + c2 * (g - f), 0)
            ft2 = ft2 / max(ft2.sum(), 1e-15)
            fc2 = fast_convolve(ft2).max()
            for _ in range(8):
                if hi - lo < 1e-7:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    ft1 = np.maximum(f + c1 * (g - f), 0)
                    ft1 = ft1 / max(ft1.sum(), 1e-15)
                    fc1 = fast_convolve(ft1).max()
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    ft2 = np.maximum(f + c2 * (g - f), 0)
                    ft2 = ft2 / max(ft2.sum(), 1e-15)
                    fc2 = fast_convolve(ft2).max()
            cand = (lo + hi) / 2
            ft = np.maximum(f + cand * (g - f), 0)
            ft = ft / max(ft.sum(), 1e-15)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = cand

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 104729)
            scale = 0.002 + 0.001 * min(no_improve // 40, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def momentum_optimize(f0, time_limit, verbose=False):
    """Softmax subgradient with heavy-ball momentum and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    base_lr = 0.003

    T_start = 8.0 / n
    T_end = 0.3 / n
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        progress = min(1.0, it / max(1, int(time_limit / 0.02)))
        # Slower temperature decay: use power law
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        lr = base_lr * (1.0 - 0.3 * progress)

        grad, conv, max_c = compute_gradient_fft(f, T)

        # Heavy ball momentum
        momentum = beta * momentum - lr * grad
        f_new = project_simplex(f + momentum)

        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 41 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                momentum = np.zeros(n)
                no_improve = 0
                continue
            continue

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            momentum = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, verbose=False):
    """Mirror descent (multiplicative weight) with softmax gradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    T_start = 6.0 / n
    T_end = 0.2 / n
    base_eta = 0.4

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        progress = min(1.0, it / max(1, int(time_limit / 0.02)))
        T = T_start * (T_end / T_start) ** (progress ** 0.6)
        eta = base_eta * (1.0 - 0.4 * progress)

        grad, conv, max_c = compute_gradient_fft(f, T)

        # Multiplicative update: f_new[j] = f[j] * exp(-eta * grad[j]) / Z
        log_updates = -eta * grad
        log_updates -= log_updates.max()  # numerical stability
        updates = np.exp(log_updates)
        Z = updates.sum()
        if Z < 1e-30:
            break
        f_new = f * updates / Z

        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 53 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
                continue
            continue

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(it * 61 + 23)
            scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0

    return best_f, best_obj


def pairwise_optimize_fast(f0, time_limit, verbose=False):
    """
    Efficient pairwise coordinate descent using localized convolution updates.
    For pair (i,j), the change in conv at position k is:
      delta(k) = (t*c - f[i]) * (f[k-i] - f[k-j])
    where c = f[i]+f[j] and t is the new fraction for i.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    offsets = [1, 2, 3, 4, 5, 7, 11, 17, 29, 47, 89, 149, 251, 421]
    no_improve_passes = 0
    pass_num = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        pass_improved = False
        conv = fast_convolve(f)
        current_max = conv.max()

        if pass_num % 3 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = int(order[idx_i])
            for offset in offsets:
                idx_j = idx_i + offset
                if idx_j >= n:
                    idx_j = idx_i - offset
                if idx_j < 0:
                    idx_j = n - 1 - idx_i
                j = int(order[idx_j])
                if i == j:
                    continue

                c = f[i] + f[j]
                if c < 1e-12:
                    continue

                # Find affected range
                lo = max(0, min(i, j))
                hi = min(2 * n - 2, max(i, j) + n - 1)
                affected = conv[lo:hi + 1]

                # Current values at i and j
                fi = f[i]
                fj = f[j]

                # Try several t values
                best_t = fi / c
                best_val = current_max

                candidates = [best_t, 0.0, 1.0, 0.25, 0.5, 0.75]
                if 0.1 < best_t < 0.9:
                    candidates.extend([best_t - 0.15, best_t + 0.15,
                                       best_t - 0.05, best_t + 0.05])

                for t in candidates:
                    t = float(np.clip(t, 0.0, 1.0))
                    if abs(t - best_t) < 1e-10:
                        continue

                    # Compute change: delta(k) = (t*c - fi) * (f[k-i] - f[k-j])
                    delta_scale = t * c - fi
                    if abs(delta_scale) < 1e-15:
                        continue

                    # Only check affected positions
                    for k in range(lo, hi + 1):
                        val_ki = f[k - i] if 0 <= k - i < n else 0.0
                        val_kj = f[k - j] if 0 <= k - j < n else 0.0
                        new_val = affected[k - lo] + delta_scale * (val_ki - val_kj)
                        if new_val > best_val:
                            best_val = new_val

                    if best_val >= current_max:
                        # This t makes things worse or equal, skip
                        continue

                    # This t improves things
                    if best_val < current_max - 1e-13:
                        f[i] = t * c
                        f[j] = (1 - t) * c
                        f = f / f.sum()
                        conv = fast_convolve(f)
                        current_max = conv.max()
                        best_t = t
                        if current_max < best_obj - 1e-13:
                            best_obj = current_max
                            best_f = f.copy()
                            pass_improved = True
                        break

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break

    return best_f, best_obj


def peak_equalize(f0, time_limit, K=8, verbose=False):
    """
    Peak equalization: find directions that reduce the top-K peaks simultaneously.
    Uses a weighted subgradient focused on top peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.002
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        conv = fast_convolve(f)

        # Focus on top-K peaks with exponential weighting
        top_k = np.argpartition(conv, -K)[-K:]
        top_vals = conv[top_k]
        max_top = top_vals.max()
        # Weight: peaks closer to max get higher weight
        weights = np.exp(10.0 * (top_vals - max_top))
        weights = weights / weights.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w * f[k - idx]

        momentum = beta * momentum - lr * grad
        f_new = project_simplex(f + momentum)

        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 150:
                rng = np.random.default_rng(it * 97 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                momentum = np.zeros(n)
                no_improve = 0
                continue
            continue

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 250:
            rng = np.random.default_rng(it * 101 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            momentum = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()

        # Alternate methods
        if p % 3 == 0:

        elif p % 3 == 1:

    # Fallback

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))

        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
