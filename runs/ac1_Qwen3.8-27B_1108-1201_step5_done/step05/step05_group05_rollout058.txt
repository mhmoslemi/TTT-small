"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent with
log-sum-exp, active-set QP refinement, coordinate descent, and multi-length
exploration. Uses FFT-based convolution for efficiency.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_objective(a, beta):
    """Compute smoothed objective: 2n * logsumexp(beta * conv) / (sum a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    # logsumexp for numerical stability
    max_conv = conv.max()
    if max_conv < 0:
        lse = max_conv + np.log(np.exp(beta * (conv - max_conv)).sum()) / beta
    else:
        shifted = beta * (conv - max_conv)
        lse = max_conv + np.log(np.exp(shifted).sum()) / beta
    return 2.0 * n * lse / (S * S)


def smooth_gradient(a, beta):
    """Compute gradient of the smoothed objective with respect to a."""
    n = len(a)
    S = a.sum()
    conv = fast_convolve(a)
    
    # Softmax weights
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of logsumexp(beta * conv) w.r.t. a:
    # d/d a[i] logsumexp = sum_k w_k * d/d a[i] conv[k]
    # d/d a[i] conv[k] = 2 * a[k-i] if 0 <= k-i < n, else 0
    # So grad_i = 2 * sum_{j=0}^{n-1} a[j] * w[i+j] (where i+j ranges from i to i+n-1)
    # This is a cross-correlation between a and w
    
    # Use FFT for efficiency
    # grad_i = 2 * sum_j a[j] * w[i+j]
    # This is the cross-correlation of a (reversed) with w, or convolution of a with reversed w
    
    # Method: pad and use FFT
    # Let's define: for each i, grad_i = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is like a "sliding dot product" of a with shifted w
    
    # We can compute this as: reverse a, convolve with w, then reverse and shift
    a_rev = a[::-1]
    # conv(a_rev, w)[i + n - 1] = sum_j a_rev[j] * w[i+n-1-j] = sum_j a[n-1-j] * w[i+n-1-j]
    # Let m = n-1-j: = sum_m a[m] * w[i+m]
    # So conv(a_rev, w)[i + n - 1] = sum_m a[m] * w[i+m] = grad_i / 2
    
    total_len = n + 2 * n - 2  # length of conv(a_rev, w) = n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(A_rev * W, fft_size)[:n + len(w) - 1]
    
    # conv_result[i + n - 1] = sum_j a_rev[j] * w[i+n-1-j]
    # = sum_j a[n-1-j] * w[i+n-1-j]
    # Substituting m = n-1-j: = sum_m a[m] * w[i+m] (for m = 0 to n-1, i+m = 0 to 2n-2)
    
    grad = 2.0 * conv_result[n-1:n-1+n]
    
    # Now the full objective gradient:
    # f(a) = 2n * lse / S^2
    # df/da_i = 2n/S^2 * grad_i - 4n * lse * a_i / S^3
    # = 2n/S^2 * (grad_i - 2 * lse * a_i / S)
    
    lse = max_conv + np.log(exp_shifted.sum()) / beta
    
    full_grad = (2.0 * n / (S * S)) * (grad - 2.0 * lse * a / S)
    
    return full_grad
