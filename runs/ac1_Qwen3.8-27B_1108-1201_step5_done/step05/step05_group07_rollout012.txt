def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_shape(a, n_iters=200, alpha=0.5, beta=2.0):
    """
    Iterative spectral shaping: modify the amplitude spectrum to be flatter,
    then project back to non-negative time domain.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iters):
        # Pad to 2n-1 for linear convolution
        a_pad = np.zeros(2 * n - 1)
        a_pad[:n] = a
        
        # Compute spectrum
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Modify magnitude to be flatter
        # Apply a power compression to flatten
        mag_flat = mag ** alpha  # alpha < 1 compresses (makes flatter)
        
        # Also apply a frequency-dependent weight to penalize high magnitude
        # This further flattens
        mean_mag = mag_flat.mean()
        if mean_mag > 0:
            normalized = mag_flat / mean_mag
            # Apply exponential softening
            mag_flat = mag_flat * np.exp(-beta * (normalized - 1.0) ** 2)
        
        # Reconstruct
        A_new = mag_flat * np.exp(1j * phase)
        a_pad_new = np.real(np.fft.ifft(A_new))
        a_new = a_pad_new[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # Gradually increase alpha (less compression) and decrease beta (less softening)
        alpha = alpha + (1.0 - alpha) * 0.01
        beta = max(beta * 0.98, 0.1)
    
    return best_a, best_obj

def softmax_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

def subgradient_optimize(a0, time_limit, top_k=50, temperature=0.0001, lr=0.005,
                         momentum=0.9, verbose=False):
    """
    Subgradient optimization with Nesterov momentum and adaptive scheduling.
    Uses multiplicative weights to maintain non-negativity.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(0.5 * momentum * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        sg = softmax_subgradient(f_nest, conv_nest, cur_top_k, cur_temp)

        # Update momentum
        v = momentum * v + (1 - momentum) * sg

        # Multiplicative weights
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr *= 0.5
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9995
        if cur_lr < 0.0001:
            cur_lr = 0.0001
        cur_temp *= 1.005

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0005)
            cur_temp = temperature * 0.5
            no_improve = 0

    return best_f, best_obj

                f_trial[j] = new_val

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping (iterative amplitude compression in frequency domain with phase preservation)
2. Adaptive softmax subgradient with Nesterov momentum and multiplicative weights
3. Targeted peak-contributor reduction local search
4. Refined coordinate descent with peak-ordered sweeps
5. Multi-length exploration with spectral re-shaping
6. Final fine-tuning with alternating strategies
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_shape(a0, n_iters=300, alpha_start=0.3, beta_start=2.0, time_limit=30.0):
    """
    Iterative spectral shaping: compress the amplitude spectrum to flatten
    auto-correlation, then project back to non-negative time domain.
    Phase is preserved from the original signal.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    alpha = alpha_start
    beta = beta_start

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Pad to 2n-1
        a_pad = np.zeros(2 * n - 1)
        a_pad[:n] = a

        # Compute spectrum
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)

        # Amplitude compression: mag^alpha with alpha < 1 flattens peaks
        mag_flat = mag ** alpha

        # Additional Gaussian softening in log-magnitude domain
        log_mag = np.log(mag_flat + 1e-20)
        mean_lm = log_mag.mean()
        std_lm = log_mag.std()
        if std_lm > 1e-15:
            # Soft-clip: reduce extreme log-magnitudes
            z = (log_mag - mean_lm) / std_lm
            log_mag_soft = log_mag - beta * np.maximum(np.abs(z) - 1.0, 0.0) ** 2 * np.sign(z)
            mag_flat = np.exp(log_mag_soft)

        # Reconstruct with original phase
        A_new = mag_flat * np.exp(1j * phase)
        a_pad_new = np.real(np.fft.ifft(A_new))
        a_new = a_pad_new[:n]

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()

        # Schedule: gradually relax compression
        alpha = alpha + (1.0 - alpha) * 0.02
        beta = max(beta * 0.97, 0.1)

    return best_a, best_obj


def compute_softmax_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def subgradient_optimize(a0, time_limit, top_k=50, temperature=0.0001, lr=0.005,
                         momentum=0.9, verbose=False):
    """
    Subgradient optimization with Nesterov momentum and multiplicative weights.
    Adaptive learning rate, temperature, and top-k scheduling.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k
    best_no_improve_reset = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(0.5 * momentum * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        sg = compute_softmax_subgradient(f_nest, conv_nest, cur_top_k, cur_temp)

        # Update momentum
        v = momentum * v + (1.0 - momentum) * sg

        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr *= 0.5
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9993
        if cur_lr < 5e-5:
            cur_lr = 5e-5
        cur_temp *= 1.004

            cur_lr = max(lr * 0.6, 0.001)

                    f[j] -= amt

            if not improved:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj


def coordinate_descent_peak_ordered(a0, time_limit, verbose=False):
    """
    Coordinate descent processing indices in order of their contribution
    to the current dominant peak. Uses binary search for optimal value.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False

        # Get current peak and order by contribution
        conv = fast_convolve(f)
        peak_idx = int(np.argmax(conv))
        k = peak_idx
        js = max(0, k - n + 1)
        je = min(n - 1, k)

        if je >= js:
            contribs = np.zeros(n)
            for i in range(js, je + 1):
                contribs[i] = f[i] * f[k - i]
            order = np.argsort(-contribs)
        else:
            order = np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            # Try multipliers
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.4, 1.8, 2.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-weighted (reduces central auto-correlation peak)
    x = np.arange(n) / max(n - 1, 1)
    edge_w = 1.0 + 1.5 * (x - 0.5) ** 2 / 0.25
    edge_w = np.maximum(edge_w, 0.1)
    inits.append(edge_w / edge_w.sum())

    # Smoothed random
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 40)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Two-lobe (mass at 1/4 and 3/4 positions)
    c1 = n * 0.25
    c2 = n * 0.75
    sigma = n / 8.0
    lobe1 = np.exp(-0.5 * ((np.arange(n) - c1) / sigma) ** 2)
    lobe2 = np.exp(-0.5 * ((np.arange(n) - c2) / sigma) ** 2)
    two_lobe = lobe1 + lobe2
    two_lobe = np.maximum(two_lobe, 1e-10)
    inits.append(two_lobe / two_lobe.sum())

    # Anti-concentrated (inverse Gaussian)
    center = n / 2.0
    sigma = n / 5.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    anti = 1.0 / (gauss + 0.2)
    anti = np.maximum(anti, 1e-10)
    inits.append(anti / anti.sum())

    return inits[:count]


def evaluate_and_track(seq, best_state, label):
    """Evaluate a sequence and update best state if improved."""
    a = np.maximum(seq, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[0]:
        best_state[0] = score
        best_state[1] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining spectral shaping, adaptive subgradient,
    targeted peak reduction, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_score = float('inf')
    best_seq = None

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_score = p_score
            best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(640) / 640
        print("No parent found, using uniform n=640")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    # === Phase 1: Spectral shaping on diverse inits ===
    print("\n=== Phase 1: Spectral Shaping ===")
    candidates = [parent]
    for n_try in [640, 768, 1024]:
        candidates.extend(make_structured_inits(n_try, rng, count=2))

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - 500
        if rem < 20 or len(candidates) == 0:
            break
        t_alloc = min(rem * 0.07, 40)
        if t_alloc < 4:
            break
        n_cand = len(cand)
        print(f"  Spectral {idx}: n={n_cand}, t={t_alloc:.0f}s")
        a_opt, _ = spectral_shape(cand, n_iters=250, alpha_start=0.35,
                                  beta_start=1.5, time_limit=t_alloc)
        score = evaluate_and_track(a_opt, [best_score, best_seq], f"spec_{idx}")
        if a_opt is not None and np.isfinite(evaluate_sequence(a_opt.tolist())):
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()

    # Re-evaluate best
    bs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(bs):
        best_score = bs

    # === Phase 2: Subgradient optimization with multiple configs ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.8f}) ===")
    n_best = len(best_seq)

    configs = [
        (n_best, max(30, n_best // 4), 0.00008, 0.004, 0.88),
        (n_best, max(20, n_best // 8), 0.0003, 0.006, 0.82),
        (n_best, max(60, n_best // 2), 0.00002, 0.002, 0.93),
        (n_best, n_best, 0.000005, 0.001, 0.95),
    ]

    for p, (n_cfg, tk, temp, lr, mom) in enumerate(configs):
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(rem * 0.25, 100)

        if p == 0:
            f_init = best_seq.copy()
        elif p == 1:
            noise = rng.standard_normal(n_best) * 0.01
            f_init = project_simplex(best_seq + noise)
        elif p == 2:
            f_init = 0.75 * best_seq + 0.25 * (np.ones(n_best) / n_best)
            f_init = project_simplex(f_init)
        else:
            f_init = best_seq.copy()

        print(f"  SG{p}: n={n_cfg}, tk={tk}, temp={temp}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(f_init, time_limit=t_alloc,
                                         top_k=tk, temperature=temp,
                                         lr=lr, momentum=mom)
        score = evaluate_and_track(a_opt, [best_score, best_seq], f"sg_{p}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.8f}) ===")
    current_n = len(best_seq)
    target_lengths = [700, 850, 1024, 1280, 1600, 2048]
    target_lengths = [tn for tn in target_lengths if abs(tn - current_n) > 50]

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 25:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        # Spectral shape the interpolation
        a_spect, _ = spectral_shape(f_interp, n_iters=80, alpha_start=0.5,
                                    beta_start=0.8, time_limit=15)
        # Then subgradient polish
        t_sg = min(rem * 0.2, 50)
        a_opt, _ = subgradient_optimize(a_spect, time_limit=t_sg,
                                         top_k=max(25, target_n // 5),
                                         temperature=0.0001, lr=0.003,
                                         momentum=0.88)
        score = evaluate_and_track(a_opt, [best_score, best_seq], f"len_{target_n}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()

    # === Phase 4: Peak reduction search ===
    print(f"\n=== Phase 4: Peak Reduction (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 120
    if rem > 15:
        t_pr = min(rem * 0.5, 80)
        print(f"  Peak reduction t={t_pr:.0f}s")
        a_opt, _ = peak_reduction_search(best_seq, time_limit=t_pr)
        score = evaluate_and_track(a_opt, [best_score, best_seq], "peak_red")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 10:
        t_cd = min(rem * 0.7, 70)
        print(f"  Coord descent t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent_peak_ordered(best_seq, time_limit=t_cd)
        score = evaluate_and_track(a_opt, [best_score, best_seq], "cd")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()

    # === Phase 6: Final fine-tuning loop ===
    print(f"\n=== Phase 6: Final Fine-tuning (best={best_score:.8f}) ===")
    n_best = len(best_seq)

    while time.monotonic() < deadline - 10:
        # Fine subgradient
        rem = deadline - time.monotonic() - 10
        if rem < 8:
            break
        t_fine = min(rem * 0.4, 30)
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_fine,
                                         top_k=n_best, temperature=0.000002,
                                         lr=0.0005, momentum=0.96)
        score = evaluate_and_track(a_opt, [best_score, best_seq], "fine_sg")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()

        # Peak reduction
        rem = deadline - time.monotonic() - 8
        if rem < 5:
            break
        a_opt, _ = peak_reduction_search(best_seq, time_limit=min(rem * 0.4, 15))
        score = evaluate_and_track(a_opt, [best_score, best_seq], "fine_pr")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()

        # Coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem < 4:
            break
        a_opt, _ = coordinate_descent_peak_ordered(best_seq, time_limit=min(rem * 0.5, 12))
        score = evaluate_and_track(a_opt, [best_score, best_seq], "fine_cd")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()

        # Small perturbation + quick optimize
        rem = deadline - time.monotonic() - 5
        if rem < 6:
            break
        noise = rng.standard_normal(n_best) * 0.001
        f_pert = project_simplex(best_seq + noise)
        a_opt, _ = subgradient_optimize(f_pert, time_limit=min(rem * 0.4, 10),
                                         top_k=max(20, n_best // 4),
                                         temperature=0.00005, lr=0.002,
                                         momentum=0.90)
        score = evaluate_and_track(a_opt, [best_score, best_seq], "pert")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()

        # Check if still improving
        rem2 = deadline - time.monotonic()
        if rem2 < 20:
            break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
