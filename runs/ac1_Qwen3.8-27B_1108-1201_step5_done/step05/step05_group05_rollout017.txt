def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    # Using FFT: ifft(fft(w) * conj(fft(a))) gives circular cross-correlation
    # With sufficient padding, this equals linear cross-correlation.
    # cross_corr[i] = sum_k w[k] * a[k-i] for i >= 0
    W_full = np.fft.fft(w, fft_size)
    A_full = np.fft.fft(a, fft_size)
    cross_corr = np.fft.ifft(W_full * np.conj(A_full), fft_size)[:n].real
    
    return conv_wa + cross_corr

def softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()

def obj_and_grad(z, alpha):
    """
    Compute objective and gradient for f(z) = 2n * LSE_alpha(conv(softmax(z), softmax(z))).
    """
    n = len(z)
    a = softmax(z)
    conv = fast_convolve(a)
    
    # LSE
    conv_max = conv.max()
    shifted = alpha * (conv - conv_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    log_sum = conv_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    # Weights
    e = np.exp(shifted - shifted.max())
    w = e / e.sum()
    
    # d_obj_da
    d_obj_da = compute_gradient(a, w)
    
    # Chain rule for softmax: d_obj/dz[j] = sum_i d_obj_da[i] * da[i]/dz[j]
    # da[i]/dz[j] = a[i](delta_ij - a[j])
    # d_obj/dz[j] = d_obj_da[j]*a[j] - a[j]*sum_i d_obj_da[i]*a[i]
    dot = np.dot(d_obj_da, a)
    d_obj_dz = a * (d_obj_da - dot)
    
    obj = 2.0 * n * log_sum
    return obj, d_obj_dz * (2.0 * n)  # Scale gradient by 2n


def lse_optimize(z_init, time_limit, alpha0=5.0, alpha1=200.0, max_iter=500):
    """
    L-BFGS optimization with adaptive alpha.
    """
    n = len(z_init)
    z = z_init.copy().astype(np.float64)
    
    best_a = softmax(z)
    best_conv = fast_convolve(best_a)
    best_max = best_conv.max()
    best_obj = 2.0 * n * best_max
    
    t0 = time.monotonic()
    elapsed = 0
    
    # Phase 1: L-BFGS with increasing alpha
    current_alpha = alpha0
    n_phases = 5
    
    for phase in range(n_phases):
        if elapsed >= time_limit * 0.7:
            break
        
        target_alpha = alpha0 * (alpha1 / alpha0) ** (phase / max(1, n_phases - 1))
        phase_time = min((time_limit * 0.7 / n_phases), time_limit * 0.7 - elapsed)
        
        def fun(z_inner):
            return obj_and_grad(z_inner, target_alpha)[0]
        
        def grad(z_inner):
            return obj_and_grad(z_inner, target_alpha)[1]
        
        # L-BFGS
        result = minimize(fun, z, method='L-BFGS-B', jac=grad,
                         options={'maxiter': max_iter // n_phases, 
                                  'maxfun': max_iter // n_phases,
                                  'ftol': 1e-12, 'gtol': 1e-8})
        
        if result.success or result.fun < fun(z):
            z = result.x
        
        a = softmax(z)
        conv = fast_convolve(a)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_conv = conv.copy()
            best_max = cur_max
        
        elapsed = time.monotonic() - t0
        current_alpha = target_alpha
        print(f"    Phase {phase+1}/{n_phases}: alpha={target_alpha:.1f}, obj={best_obj:.8f}, time={elapsed:.1f}s")
    
    # Phase 2: High-alpha L-BFGS for final focus
    if time.monotonic() - t0 < time_limit * 0.85:
        for hi_alpha in [300, 500, 1000]:
            if time.monotonic() - t0 >= time_limit * 0.85:
                break
            def fun_hi(z_inner):
                return obj_and_grad(z_inner, hi_alpha)[0]
            def grad_hi(z_inner):
                return obj_and_grad(z_inner, hi_alpha)[1]
            
            result = minimize(fun_hi, z, method='L-BFGS-B', jac=grad_hi,
                            options={'maxiter': 100, 'maxfun': 200,
                                     'ftol': 1e-14, 'gtol': 1e-10})
            if result.success or result.fun < fun_hi(z):
                z = result.x
            
            a = softmax(z)
            conv = fast_convolve(a)
            cur_max = conv.max()
            cur_obj = 2.0 * n * cur_max
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    High-alpha: obj={best_obj:.8f}, time={elapsed:.1f}s")
    
    return best_a, best_obj


def coord_desc_refine(a_init, time_limit):
    """
    Coordinate descent refinement: for each coordinate, try to adjust it
    to reduce the max convolution value.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_conv = fast_convolve(a)
    best_max = best_conv.max()
    best_obj = 2.0 * n * best_max
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            # Try adjusting a[i] by various factors
            best_i_val = a[i]
            best_i_max = best_max
            
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 2.0, 0.3, 3.0]:
                new_val = a[i] * factor
                if new_val < 0:
                    continue
                # Redistribute: keep sum = 1
                remaining = 1.0 - new_val
                if remaining < 1e-15:
                    continue
                other_sum = s - a[i]
                if other_sum < 1e-15:
                    continue
                scale = remaining / other_sum
                
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] *= scale
                a_trial /= a_trial.sum()
                
                conv_trial = fast_convolve(a_trial)
                max_trial = conv_trial.max()
                
                if max_trial < best_i_max - 1e-15:
                    best_i_max = max_trial
                    best_i_val = new_val
            
            if best_i_val != a[i]:
                other_sum = s - a[i]
                scale = (1.0 - best_i_val) / other_sum
                a[i] = best_i_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a[mask] *= scale
                a /= a.sum()
                s = a.sum()
                
                conv_new = fast_convolve(a)
                max_new = conv_new.max()
                if max_new < best_max - 1e-15:
                    best_max = max_new
                    best_obj = 2.0 * n * best_max
                    best_a = a.copy()
                    best_conv = conv_new
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig[:n])
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def structured_init(n, seed=0):
    """Generate structured initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Exponential decay from left
    x = np.arange(n)
    a = np.exp(-x / (n * 0.3))
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    
    # Exponential decay from right
    a = np.exp(-x[::-1] / (n * 0.3))
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    
    # Quadratic (high in middle)
    a = 1.0 - (2.0 * x / (n - 1) - 1.0) ** 2
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    
    # Quadratic (high at edges)
    a = (2.0 * x / (n - 1) - 1.0) ** 2
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    
    # Sine-based
    a = (1.0 + np.sin(np.pi * x / n)) / 2.0
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    
    # Cosine-based (high at edges)
    a = (1.0 - np.cos(2.0 * np.pi * x / n)) / 2.0
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    
    # Random with exponential mixing
    a = 0.5 * np.ones(n) + 0.5 * rng.exponential(1.0, n)
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    
    # Flat spectrum
    inits.append(flat_spectrum_init(n, seed=seed + 100))
    
    return inits
