def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    ff = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, ff)
    return np.fft.irfft(A * A, ff)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_max_obj(a, conv, top_k, temp):
    """Smooth approximation of max(conv) using top-k soft-max."""
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    return temp * (np.log(np.sum(np.exp(scaled))) + scaled.max() / temp)
    # Wait, this is just the log-sum-exp of the top k values, scaled.
    # Actually: soft_max = temp * log(sum(exp(pv/temp)))
    # But I only use top k for efficiency.

def compute_smooth_grad(a, conv, top_k, temp):
    """Gradient of the soft-max approximation."""
    n = len(a)
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k_idx = int(peaks[i])
        js = max(0, k_idx - n + 1)
        je = min(n - 1, k_idx)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k_idx - idx]
    return grad

def interpolate_len(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, f)
    fi = np.maximum(fi, 0)
    s = fi.sum()
    if s > 1e-15:
        fi /= s
    return fi
