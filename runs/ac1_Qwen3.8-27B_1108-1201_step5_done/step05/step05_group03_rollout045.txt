def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(conv, f, top_k, temperature):
    """Compute smoothed subgradient of max(conv) using top-k softmax weighting."""
    n = len(f)
    conv_len = 2 * n - 1
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = temperature * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg

def adam_optimize(f0, time_limit, verbose=False, top_k_max=None, T_init=None, T_end=None):
    """Adam optimizer with adaptive temperature and cosine LR schedule."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if top_k_max is None:
        top_k_max = n
    if T_init is None:
        T_init = 8.0 / n
    if T_end is None:
        T_end = 0.3 / n
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    it = 0
    no_improve = 0
    best_lr = 0.005
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2:
            break
        
        # Cosine annealing for learning rate
        progress = min(1.0, elapsed / time_limit)
        lr = best_lr * 0.5 * (1 + np.cos(np.pi * progress))
        lr = max(lr, 0.0005)
        
        # Adaptive temperature and top-k
        top_k = max(5, int(top_k_max * (1 - 0.7 * progress)))
        T = T_init * (T_end / T_init) ** progress
        
        conv = fast_convolve(f)
        sg = compute_subgradient(conv, f, top_k, T)
        
        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 200:
                # Perturb and restart
                rng = np.random.default_rng(it * 41 + 13)
                scale = 0.003 * (1 + no_improve / 200)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                lr = best_lr  # Reset LR
                continue
            continue
        
        f = f_new
        new_M = fast_convolve(f).max()
        
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}, top_k={top_k}, lr={lr:.6f}")
        else:
            no_improve += 1
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, f, top_k, temperature):
    """Compute smoothed subgradient of max(conv) using top-k softmax weighting."""
    n = len(f)
    conv_len = 2 * n - 1
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = temperature * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg


def adam_optimize(f0, time_limit, verbose=False, top_k_max=None, T_init=None, T_end=None, lr_base=0.005):
    """Adam optimizer with adaptive temperature, cosine LR, and momentum restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if top_k_max is None:
        top_k_max = max(20, n)
    if T_init is None:
        T_init = 10.0 / n
    if T_end is None:
        T_end = 0.2 / n

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    no_improve = 0
    best_lr_at_restart = lr_base

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2:
            break

        # Cosine annealing for learning rate
        progress = min(1.0, elapsed / time_limit)
        lr = best_lr_at_restart * 0.5 * (1 + np.cos(np.pi * progress))
        lr = max(lr, 0.0003)

        # Adaptive temperature (exponential cooling) and top-k (decreasing)
        top_k = max(5, int(top_k_max * (1.0 - 0.75 * progress)))
        T = T_init * (T_end / T_init) ** progress

        conv = fast_convolve(f)
        sg = compute_subgradient(conv, f, top_k, T)

        # Adam update
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * sg
        v = beta2 * v + (1.0 - beta2) * sg * sg
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 300:
                rng = np.random.default_rng(it * 41 + 13)
                scale = 0.003 * (1.0 + min(no_improve / 100, 3.0))
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                best_lr_at_restart = lr_base * (1.0 + 0.2 * min(no_improve // 100, 2))
                continue
            continue

        f = f_new
        new_M = fast_convolve(f).max()

            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}, top_k={top_k}, lr={lr:.6f}")
        else:
            no_improve += 1

    return best_f, best_obj

        step = current_lr * m_hat / (np.sqrt(v_hat) + eps)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structured_init(n, kind, rng):
    """Create structured initializations."""
    if kind == "gaussian":
        x = np.arange(n, dtype=np.float64)
        center = n / 2.0
        sigma = n / 4.0
        a = np.exp(-(x - center) ** 2 / (2.0 * sigma ** 2))
    elif kind == "triangular":
        x = np.arange(n, dtype=np.float64)
        a = 1.0 - np.abs(2.0 * x / max(n - 1, 1) - 1.0)
    elif kind == "sine":
        x = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)
    elif kind == "sine2":
        x = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.5 * np.cos(4.0 * np.pi * x / n)
    elif kind == "bimodal":
        x = np.arange(n, dtype=np.float64)
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n / 8.0
        a = np.exp(-(x - c1) ** 2 / (2 * sigma ** 2)) + np.exp(-(x - c2) ** 2 / (2 * sigma ** 2))
    elif kind == "noise":
        a = rng.random(n)
    else:
        a = np.ones(n)

    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n)
    else:
        a = a / ss
    return a

            t_alloc = min(rem * 0.35, 120)
            if t_alloc < 15:
                break
            print(f"  Adam n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc, verbose=False,
                                      T_init=10.0 / target_n, T_end=0.15 / target_n)
            _try(a_opt, f"ml_n{target_n}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 25:
            t_cd = min(rem * 0.6, 200)
            print(f"  Time: {t_cd:.0f}s, n={len(best_seq)}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            _try(a_opt, "coord_desc")

            # Adam after coord descent
            rem = deadline - time.monotonic() - 350
            if rem > 15:
                a_opt, _ = adam_optimize(a_opt, time_limit=min(rem, 60), verbose=False,
                                          T_init=3.0 / len(a_opt), T_end=0.1 / len(a_opt))
                _try(a_opt, "adam_after_cd")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        scales = [0.01, 0.02, 0.03, 0.015, 0.025, 0.04, 0.05, 0.005]
        for p in range(8):
            rem = deadline - time.monotonic() - 200
            if rem < 12:
                break
            t_alloc = min(rem / max(1, 8 - p), 100)
            if t_alloc < 10:
                break

            rng = np.random.default_rng(5000 + p * 13)
            f_init = project_simplex(best_seq + rng.standard_normal(n_best) * scales[p])
            a_opt, _ = adam_optimize(f_init, time_limit=t_alloc, verbose=False,
                                      T_init=6.0 / n_best, T_end=0.1 / n_best)
            _try(a_opt, f"diverse_{p}")

    # === Phase 8: Final coordinate descent ===
    print(f"\n=== Phase 8: Final CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_cd2 = min(rem * 0.7, 100)
            print(f"  Time: {t_cd2:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd2)
            _try(a_opt, "cd_final")

    # === Phase 9: Final polish with low temperature ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = adam_optimize(best_seq, time_limit=min(rem, 60), verbose=True,
                                      top_k_max=max(10, n_best // 5),
                                      T_init=2.0 / n_best, T_end=0.05 / n_best,
                                      lr_base=0.003)
            _try(a_opt, "polish_adam")

        # Tiny perturbations
        for trial in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            rng = np.random.default_rng(9000 + trial)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            a_opt, _ = adam_optimize(f_pert, time_limit=min(rem, 20),
                                      T_init=1.0 / n_best, T_end=0.05 / n_best)
            _try(a_opt, f"tiny_{trial}")

    # === Phase 10: One more coord descent on absolute best ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 25))
            _try(a_opt, "cd_very_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
