def lse_convolution(sequence: List[float]) -> float:
    """
    Compute the convolution of a normalized sequence.
    
    Args:
        sequence: Input sequence of coefficients
        
    Returns:
        float: Maximum value of the convolution
    """
    # Convert to numpy array and handle clipppings
    seq_arr = np.array(sequence) / np.sum(sequence) if np.sum(sequence) > 0 else np.ones(len(sequence))
    seq_arr = np.maximum(seq_arr, 0.0)
    seq_arr = np.minimum(seq_arr, 1000.0)
    # Optimized convolution using FFT
    a = np.fft.rfft(seq_arr)
    conv = np.fft.irfft(a * a)[:2*len(seq_arr)-1]
    return np.max(conv)
