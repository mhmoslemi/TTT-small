"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on smoothed log-sum-exp
objective in softmax parameterization, combined with mirror descent (multiplicative
weights) refinement, full-gradient subgradient methods, local search polishing,
and multi-length exploration. Key innovation: analytical gradient formula for
log-sum-exp smoothing enabling fast L-BFGS convergence to local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_full_gradient(a, T):
    """
    Compute gradient of F(a) = 2n*T*logsumexp(conv(a,a)/T) w.r.t. a.
    Returns (grad_a, smooth_max, b) where grad_a is the projected gradient
    in the simplex tangent space.
    """
    n = len(a)
    b = fast_convolve(a)

    # Softmax weights on b
    scaled = b / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()

    # Cross-correlation: c[j] = sum_k p[k] * a[k-j]
    # Direct computation (O(n^2), fast for n < 3000)
    c = np.zeros(n)
    max_k = len(p)
    for j in range(n):
        ks = j
        ke = min(j + n, max_k)
        if ke > ks:
            c[j] = np.dot(p[ks:ke], a[:ke - ks])

    m_b = float(np.dot(p, b))
    grad_a = 4.0 * n * (c - m_b)
    # Project to tangent space
    grad_a -= grad_a.mean()

    # Smoothed max
    max_scaled = scaled.max()
    log_sum = np.log(p.sum())
    smooth_max = T * (max_scaled + log_sum)

    return grad_a, smooth_max, b


def lbfgs_optimize(a0, T, maxiter=3000, time_limit=30.0, verbose=False, label=""):
    """
    L-BFGS optimization on the smoothed objective using softmax parameterization.
    Gradient in x-space: 4n*a_j*(c_j - m_b)
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)

    t0 = time.monotonic()
    call_count = [0]

    def obj_and_grad(x):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            # Return current best
            a_cur = np.exp(x - x.max())
            a_cur /= a_cur.sum()
            b = fast_convolve(a_cur)
            obj = 2.0 * n * b.max()
            return obj, np.zeros(n)

        a_loc = np.exp(x - x.max())
        a_loc /= a_loc.sum()

        # Compute gradient in a-space
        b = fast_convolve(a_loc)
        scaled = b / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()

        # Cross-correlation
        c = np.zeros(n)
        max_k = len(p)
        for j in range(n):
            ks = j
            ke = min(j + n, max_k)
            if ke > ks:
                c[j] = np.dot(p[ks:ke], a_loc[:ke - ks])

        m_b = float(np.dot(p, b))

        # Gradient in x-space: 4n*a_j*(c_j - m_b)
        grad_x = 4.0 * n * a_loc * (c - m_b)

        # Objective
        max_scaled = scaled.max()
        log_sum = np.log(p.sum())
        smooth_max = T * (max_scaled + log_sum)
        obj = 2.0 * n * smooth_max

        return obj, grad_x

    try:
        result = minimize(obj_and_grad, x0, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxiter * 5,
                                   'iprint': -1})
    except Exception:
        a_opt = np.exp(x0 - x0.max())
        a_opt /= a_opt.sum()
        return a_opt, 2.0 * n * fast_convolve(a_opt).max()

    a_opt = np.exp(result.x - result.x.max())
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s < 1e-15:
        a_opt = a0.copy()
    else:
        a_opt /= s

    obj = 2.0 * n * fast_convolve(a_opt).max()
    if verbose:
        print(f"    L-BFGS {label}: {result.success}, func={result.fun:.8f}, "
              f"true_obj={obj:.8f}, iters={result.nit}, t={time.monotonic()-t0:.1f}s")
    return a_opt, obj


def mu_weights_optimize(f0, time_limit, top_k=50, temperature=0.00005, lr=0.003,
                        momentum=0.85, verbose=False, label=""):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum.
    Uses full gradient (all peaks weighted by temperature) for better signal.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = best_f.copy()
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)

        # Full gradient with temperature weighting
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_temp *= 1.01
        if it % 1000 == 0:
            cur_top_k = max(10, int(cur_top_k * 0.9))

        if no_improve > 400:
            rng_r = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.003 + 0.002 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng_r.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    MU {label} iter {it}, t={time.monotonic()-t0:.1f}s, "
                  f"obj={best_obj:.10f}, lr={cur_lr:.7f}, temp={cur_temp:.2e}")

    return best_f, best_obj


def pair_exchange_search(f0, time_limit, verbose=False):
    """Pair-exchange local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    conv = fast_convolve(f)
    n_peaks = min(20, 2 * n - 1)
    top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False
        conv = fast_convolve(f)
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

        rng = np.random.default_rng(hash((iteration, n)) % 1000000)
        trials = min(300, n)
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-10:
                continue
            for frac in [0.05, 0.15, 0.3, 0.5]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] = old_j + amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = old_j
                    break

        # Systematic peak-directed moves
        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            if contribs.max() < 1e-20:
                continue
            hi = idx[np.argmax(contribs)]
            lo = idx[np.argmin(contribs)]
            if hi == lo or f[hi] < 1e-10:
                continue
            for frac in [0.05, 0.1, 0.2]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi] += amt
                    f[lo] -= amt
                    break

        if not improved:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    mid = n // 2
    order = list(range(n))
    # Center-out ordering
    centered = []
    for d in range(mid + 1):
        if mid + d < n:
            centered.append(mid + d)
        if d > 0 and mid - d >= 0:
            centered.append(mid - d)
    if centered:
        order = centered

    no_improve_rounds = 0
    while time.monotonic() - t0 < time_limit and no_improve_rounds < 5:
        improved_round = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.3, 0.5, 0.7, 1.3, 1.7, 2.5]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved_round = True

        if not improved_round:
            no_improve_rounds += 1
        else:
            no_improve_rounds = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    for sigma_mult in [3.0, 5.0, 7.0]:
        sigma = n / sigma_mult
        gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        inits.append(gauss / gauss.sum())

    # Cosine window
    cos_w = np.cos(np.pi * x) ** 2
    cos_w = np.maximum(cos_w, 1e-10)
    inits.append(cos_w / cos_w.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    klen = max(3, n // 50)
    kernel = np.ones(klen) / klen
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Raised cosine with random asymmetry
    alpha = 0.3 + 0.4 * rng.random()
    asym = np.ones(n)
    asym[:n // 2] *= alpha
    asym = asym * (0.5 + 0.5 * np.cos(2 * np.pi * x))
    asym = np.maximum(asym, 1e-10)
    inits.append(asym / asym.sum())

    return inits[:count]
