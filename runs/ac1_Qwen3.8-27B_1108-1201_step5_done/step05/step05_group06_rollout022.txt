def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def parametric_init(n, alpha):
    """Generate a parametric initialization: a[j] ∝ (1 - x^2)^alpha where x = 2j/(n-1)-1."""
    if n == 1:
        return np.array([1.0])
    x = 2.0 * np.arange(n) / (n - 1) - 1.0
    x = np.clip(x, -0.999, 0.999)
    base = 1.0 - x * x
    a = base ** alpha
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smooth projected gradient,
mirror descent, exact coordinate bisection, and multi-length/multi-init search.
Focuses on precise convergence via extended coordinate bisection.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, epsilon, conv):
    """Gradient of log-sum-exp approximation of max(conv)."""
    n = len(a)
    conv_len = len(conv)
    scaled = epsilon * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(conv_len)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    
    # grad_j = 2 * sum_i a[i] * w[i+j] (cross-correlation)
    L = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = np.fft.rfft(a, L)
    W_fft = np.fft.rfft(w, L)
    R = np.fft.irfft(np.conj(A_fft) * W_fft, L)
    grad = 2.0 * R[:n]
    return grad


def smooth_ls_objective(a, epsilon, conv):
    """Log-sum-exp approximation of max(conv)."""
    scaled = epsilon * conv
    scaled -= scaled.max()
    return conv.max() + np.log(np.exp(scaled).sum()) / epsilon

def md_smooth(a0, time_limit, eps_start=5.0, eps_end=0.05, verbose=False):

        a_new = a * np.exp(exponent)

def coordinate_bisection(a0, time_limit, verbose=False, max_passes=500):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        max_improve = 0.0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-15:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-14:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-14:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break
            
            if abs(x_new - x_old) < 1e-15:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                conv = fast_convolve(a)
                continue
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-13:
                improvement = best_obj - obj
                if improvement > max_improve:
                    max_improve = improvement
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if verbose and passes % 10 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}, improved={pass_improved}")
        
        if not pass_improved:
            if passes > 5:
                rng = np.random.default_rng(passes * 41 + 13)
                noise_scale = 0.0005 * min(passes // 5, 5)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                conv = fast_convolve(a)
            else:
                break
    
    return best_a, best_obj


def parametric_init(n, alpha):
    """a[j] ∝ (1 - x^2)^alpha where x = 2j/(n-1) - 1."""
    if n == 1:
        return np.array([1.0])
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    x = np.clip(x, -0.9999, 0.9999)
    base = 1.0 - x * x
    a = base ** alpha
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
