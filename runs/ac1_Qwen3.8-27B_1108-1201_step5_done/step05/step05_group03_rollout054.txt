import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective(f):
    """Compute 2*n*max(conv(f,f))/(sum(f))^2"""
    conv = fast_convolve(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2 * len(f) * conv.max() / (s * s)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

    f = np.maximum(f, 0)
    f = f / f.sum()
    best_f = f.copy()

            order = np.arange(n)

            if time.monotonic() - t0 > time_limit:

                j = i + offset
                c = f[i] + f[j]

                t_cur = f[i] / c
                
                # Try a few candidates
                candidates = [t_cur, 0.0, 1.0, 0.5]

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max(conv, T, top_k=50):
    """Approximate max using log-sum-exp on top-k elements."""
    k = min(top_k, len(conv))
    idx = np.argpartition(conv, -k)[-k:]
    vals = conv[idx]
    v
max = vals.max()
    w = np.exp((vals - vmax) / T)
    return vmax + T * np.log(w.sum())

def adam_optimize(f, time_limit=60, verbose=False):
    n = len(f)
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    no_improve = 0
    
    # Temperature schedule
    T_start = 3.0 / n
    T_end = 0.1 / n
    lr_start = 0.003
    lr_end = 0.0001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 5000)
        T = T_start * (T_end / T_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        conv = fast_convolve(f)
        
        # Subgradient via softmax weighting
        top_k = min(50, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        w = np.exp(T * (pv - pv.max()))
        w /= w.sum()
        
        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            if w[i] < 1e-12:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2 * w[i] * f[k - idx]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        f = f_new
        
        new_max = fast_convolve(f).max()
        if new_max < best_obj - 1e-12:
            best_obj = new_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, 2 * n * best_obj

        improved = False
        conv = fast_convolve(f)
        
        if sweep % 3 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

            if time.monotonic() - t0 > time_limit:
                break
            i = int(order[pos])

                j = i + offset
                c = f[i] + f[j]
                if c < 1e-10:

                # delta_conv[k] = delta * (f[k-i] - f[k-j])
                # where delta = (t - t_cur) * c
                g = np.zeros(2 * n - 1)
                # f[k-i] is non-zero for k in [i, i+n-1]
                g[i:i+n] += f[:n]  # f[k-i] for k-i in [0, n-1]
                # f[k-j] is non-zero for k in [j, j+n-1]
                g[j:j+n] -= f[:n]  # f[k-j] for k-j in [0, n-1]
                
                # For candidate t, new_max = max(conv + delta*g) where delta = (t-t_cur)*c
                # Try a few candidates
                candidates = [0.0, 1.0, 0.5]
                if t_cur > 0.05:
                    candidates.append(t_cur - 0.15)
                    candidates.append(t_cur - 0.05)
                if t_cur < 0.95:
                    candidates.append(t_cur + 0.15)
                    candidates.append(t_cur + 0.05)
                
                best_t = t_cur
                best_val = best_max
                
                for t in candidates:
                    t = max(0.0, min(1.0, t))
                    if abs(t - t_cur) < 1e-10:
                        continue
                    delta = (t - t_cur) * c
                    new_conv = conv + delta * g
                    val = new_conv.max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                
                if best_t != t_cur:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    conv = fast_convolve(f)  # Recompute for consistency
                    best_max = conv.max()
                    if best_max < fast_convolve(best_f).max():
                        best_f = f.copy()
                    improved = True
        
        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            best_max = fast_convolve(f).max()
    
    return best_f, 2 * n * best_max

Wait, there's an issue with the pairwise approach. When I update `conv` after a successful change, the `g` vector for subsequent pairs is based on the old `f`, not the new one. I need to either:
1. Recompute `conv` and `g` after each successful change (expensive)
2. Or accept a small error and recompute `conv` at the end of each sweep

I'll go with option 2: recompute `conv` at the start of each sweep, and within a sweep, use the incremental update (which introduces small errors but is much faster).

Actually, the incremental update is exact if I update `conv` after each change:
`new_conv = conv + delta * g`

This is exact! Because `g` was computed from the current `f` before the change, and the change is only in positions i and j. So `new_conv` is exact.

But then for the next pair, I need to use the updated `f` and `conv`. So I should update both `f` and `conv` after each successful change.

Let me restructure:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiple diverse initializations (linear ramp, cosine, bump, random)
2. Fast pairwise coordinate descent with incremental convolution updates
3. Adam optimizer with log-sum-exp smoothing
4. Multi-length exploration
5. Iterated refinement
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pairwise_cd(f, max_offset=16, time_limit=120, verbose=False):
    """Fast pairwise coordinate descent with incremental conv updates."""
    n = len(f)
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    f = f / s
    conv = fast_convolve(f)
    best_f = f.copy()
    best_max = conv.max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved_count = 0
        
        if sweep % 2 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for pos in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(order[pos])
            max_off = min(max_offset, n - 1 - i)
            
            for offset in range(1, max_off + 1):
                j = i + offset
                c = f[i] + f[j]
                if c < 1e-10:
                    continue
                
                t_cur = f[i] / c
                
                # Compute sensitivity vector g[k] = f[k-i] - f[k-j]
                g = np.zeros(2 * n - 1)
                g[i:i+n] += f
                g[j:j+n] -= f
                
                # Try candidate transfer values
                best_t = t_cur
                best_val = best_max
                
                # Fine candidates around current
                for dt in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]:
                    t = t_cur + dt
                    if t < 0.0 or t > 1.0:
                        continue
                    delta = (t - t_cur) * c
                    val = (conv + delta * g).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                
                # Also try extremes
                for t in [0.0, 1.0]:
                    if abs(t - t_cur) < 1e-8:
                        continue
                    delta = (t - t_cur) * c
                    val = (conv + delta * g).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                
                if best_t != t_cur:
                    delta = (best_t - t_cur) * c
                    f[i] = best_t * c
                    f[j] = (1.0 - best_t) * c
                    conv += delta * g
                    best_max = conv.max()
                    improved_count += 1
                    if best_max < best_max * 0.999 or improved_count % 100 == 0:
                        best_f = f.copy()
        
        if improved_count > 0:
            no_improve = 0
            if verbose and sweep % 10 == 0:
                print(f"    PD sweep {sweep}: max={best_max:.10f}, improved={improved_count}")
        else:
            no_improve += 1
            if no_improve >= 3:
                break
            # Escape: perturb
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            best_max = conv.max()
    
    # Final verification
    conv_final = fast_convolve(best_f)
    final_max = conv_final.max()
    if final_max < best_max:
        best_f = f.copy()
    return best_f, 2.0 * n * final_max


def adam_optimize(f, time_limit=60, verbose=False):
    """Adam optimizer with smooth max objective."""
    n = len(f)
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    f = f / f.sum()
    
    best_f = f.copy()
    best_max = fast_convolve(f).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 3000)
        T = 2.0 / n * (0.05 ** progress)
        lr = 0.002 * (0.1 ** progress)
        
        conv = fast_convolve(f)

        top_k = min(32, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]

        w = np.exp((pv - pv.max()) / T)

        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it)

def golden_section_pair(f, i, j, max_iter=30):
    """Optimize the pair (i,j) using golden section search."""
    n = len(f)
    c = f[i] + f[j]
    if c < 1e-10:
        return f, fast_convolve(f).max()
    
    conv = fast_convolve(f)
    t_cur = f[i] / c
    
    g = np.zeros(2 * n - 1)
    g[i:i+n] += f
    g[j:j+n] -= f
    
    # The objective as function of t: max(conv + (t - t_cur)*c * g)
    # This is convex in t (max of linear functions)
    # Find minimum using golden section
    
    lo = 0.0
    hi = 1.0
    gr = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    
    def obj_at(t):
        delta = (t - t_cur) * c
        return (conv + delta * g).max()
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj_at(c1)
    f2 = obj_at(c2)
    
    for _ in range(max_iter):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj_at(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
