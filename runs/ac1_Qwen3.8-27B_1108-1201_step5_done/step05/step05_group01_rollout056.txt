gradient with Nesterov acceleration, multi-length interpolation, and

import numpy as np
import time


def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1.0))
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1.0))
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new


def compute_grad_smoothed(a, w, n):
    """
    Compute smoothed gradient: g[m] = 2 * sum_j w[m+j] * a[j] for m in [0,n-1].
    Uses FFT cross-correlation.
    w has length 2n-1.
    """
    # We need R[m] = sum_j w[m+j] * a[j] for m=0,...,n-1
    # This is cross-correlation of w and a
    # R = ifft(FFT(w) * conj(FFT(a))) (with sufficient zero-padding)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W = np.fft.rfft(w_full)
    Af = np.fft.rfft(a_full)
    
    R = np.fft.irfft(W * np.conj(Af), fft_size)
    return 2.0 * R[:n]


def momentum_pg(a0, time_limit, tau_init=0.01, lr_init=0.02, mom_init=0.9,
                restart_every=300, verbose=False):
    """
    Momentum projected gradient on the simplex.
    Optimizes max_k conv(a,a)[k] using soft-max smoothing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    mom = mom_init
    
    conv = fast_autocorr(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_autocorr(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Soft-max weights
        shifted = conv / tau
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        g = compute_grad_smoothed(a, w, n)
        
        # Momentum update
        v = mom * v - lr * g
        a_new = project_simplex(a + v)
        
        # Adaptive parameters
        lr *= 0.99997
        tau *= 0.99998
        mom = min(0.995, mom + 0.000002)
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            print(f"      PG it={it}, obj={best_obj:.10f}, lr={lr:.6f}, tau={tau:.6f}")
        
        # Restart
        if no_improve > restart_every:
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            scale = 0.001 + 0.001 * min(no_improve // restart_every, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, 0.005)
            tau = min(tau_init * 3, 0.1)
            no_improve = 0
    
    return best_a, best_obj


def anneal_pg(a0, time_limit, tau_start=0.05, tau_end=0.001, lr_start=0.05,
              lr_end=0.005, mom=0.9, verbose=False):
    """
    Simulated annealing style: start with large tau (smooth, global) and 
    large lr, then cool to small tau (sharp, local) and small lr.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    
    conv = fast_autocorr(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    # Pre-compute total iterations estimate
    # Each iteration takes roughly 2 * fft_time
    # For n=2048, fft_size=4096, rfft takes ~50us, so iteration ~200us
    # In 100s: ~500,000 iterations (optimistic) or ~100,000 (conservative)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Progress from 0 to 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        
        # Annealing schedule (exponential)
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Add noise proportional to temperature (annealing)
        if progress < 0.3:
            noise_scale = 0.001 * (1 - progress / 0.3)
            rng = np.random.default_rng(it % (2**31))
            a = project_simplex(a + rng.standard_normal(n) * noise_scale)
        
        conv = fast_autocorr(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        
        # Soft-max
        shifted = conv / tau
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        g = compute_grad_smoothed(a, w, n)
        
        # Momentum
        v = mom * v - lr * g
        a = project_simplex(a + v)
        
        it += 1
    
    return best_a, best_obj


def cosine_init(n, freq, amp, offset=0.0):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n + offset)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def exp_decay_init(n, rate, symmetric=True):
    """Exponential decay initialization."""
    j = np.arange(n)
    a = np.exp(-rate * j / n)
    if symmetric:
        a = np.concatenate([a, a[:n//2][::-1]])[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def random_init(n, seed=0):
    """Random positive initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s
