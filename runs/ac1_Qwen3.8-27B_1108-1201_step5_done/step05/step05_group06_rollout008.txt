def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)

    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max() * 2 * n  # 2n * max(conv) since sum = 1
    best_a = a.copy()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)
        grad = soft_max_gradient(a, conv, beta)
        
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad**2
        m_hat = m / (1 - 0.9**t)

            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 200:
            # Reset or add noise

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def sparse_softmax_gradient(a, conv, top_k=20, temp=0.01):
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-10:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            grad[js:je+1] += 2.0 * w[i] * a[k-je:k-js+1][::-1]
    
    return grad

    else:
        a = a / s
    n = len(a)

    best_obj = conv.max()
    best_a = a.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    total_iters_expected = 5000  # rough estimate
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(t / total_iters_expected, 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_base / (1 + 0.1 * t)
        temp = 0.005 + 0.02 * (1 - progress)

def interpolate_to_length(f, new_n):
    old_n = len(f)

    new_x = np.linspace(0, 1, new_n)

    ss = f_interp.sum()

def fast_convolve(a):

    n = len(a)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 31 + 17)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break

    return best_a, best_obj * 2 * n


def block_adam_polish(a0, time_limit, block_size=16, lr=0.005, verbose=False):
    """Block coordinate descent with Adam within each block."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    inner_iter = 10
    while time.monotonic() - t0 < time_limit:
        # Pick a random block
        start = rng.integers(0, max(1, n - block_size))
        end = min(start + block_size, n)
        block = slice(start, end)
        b_len = end - start

        # Get sum of block
        block_sum = a[block].sum()
        if block_sum < 1e-14:
            continue

        # Normalize block to sum = block_sum
        # Run Adam on just the block
        m_b = np.zeros(b_len)
        v_b = np.zeros(b_len)
        a_block = a[block].copy()

        for it in range(inner_iter):
            # Compute gradient for block coordinates
            conv_cur = fast_convolve(a)
            temp = 0.01
            k_actual = min(20, len(conv_cur))
            peaks = np.argpartition(conv_cur, -k_actual)[-k_actual:]
            pv = conv_cur[peaks]
            scaled = pv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()

            grad_full = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-10:
                    continue
                k = int(peaks[i])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js <= je:
                    grad_full[js:je + 1] += 2.0 * w[i] * a[k - je:k - js + 1][::-1]

            grad_b = grad_full[block]
            m_b = 0.9 * m_b + 0.1 * grad_b
            v_b = 0.999 * v_b + 0.001 * grad_b * grad_b
            m_hat = m_b / (1 - 0.9 ** (it + 1))
            v_hat = v_b / (1 - 0.999 ** (it + 1))

            a_block_new = a_block - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            # Project block onto {x >= 0, sum = block_sum}
            a_block_new = np.maximum(a_block_new, 0.0)
            bs = a_block_new.sum()
            if bs > 1e-15:
                a_block_new *= block_sum / bs
            else:
                a_block_new = a_block.copy()

            a[block] = a_block_new
            a_block = a_block_new

        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj * 2 * n


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def diverse_initializations(a_base, n, rng):
    """Generate diverse initializations from a base sequence."""
    inits = []
    # Direct
    inits.append(("direct", a_base.copy()))
    # Symmetric
    inits.append(("sym", symmetrize(a_base)))
    # Noise
    inits.append(("noise3", project_simplex(a_base + rng.standard_normal(n) * 0.003)))
    inits.append(("noise7", project_simplex(a_base + rng.standard_normal(n) * 0.007)))
    # Mix with flat
    inits.append(("flat50", project_simplex(0.5 * a_base + 0.5 * np.ones(n) / n)))
    # Mix with uniform blocks
    nblocks = rng.integers(2, 8)
    block_w = np.ones(nblocks) * rng.uniform(0.5, 2.0, nblocks)
    block_w /= block_w.sum()
    block_seq = np.repeat(block_w, n // nblocks + 1)[:n]
    block_seq = project_simplex(block_seq)
    inits.append(("blocks", project_simplex(0.5 * a_base + 0.5 * block_seq)))
    # Cosine modulation
    cos_mod = np.maximum(np.cos(np.pi * np.arange(n) / n), 0.0)
    cos_mod = project_simplex(cos_mod)
    inits.append(("cos", project_simplex(0.5 * a_base + 0.5 * cos_mod)))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: Adam with soft-max curriculum, block coordinate
    descent, exact coordinate bisection, multi-length exploration, and diverse
    restarts to minimize 2n*max(conv)/sum^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using flat sequence n=768")

    n_parent = len(parent)

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_try)})")

    # === Phase 1: Adam on parent ===
    print(f"\n=== Phase 1: Adam on parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    t_alloc = min(remaining * 0.30, 480)
    if t_alloc > 30:
        print(f"  Adam optimize, t={t_alloc:.0f}s")
        a_opt, obj = adam_simplex(parent, time_limit=t_alloc,
                                  beta_start=5.0, beta_end=200.0,
                                  top_k=40, lr_base=0.01, verbose=True, tag="P1")
        update_best(a_opt, "adam_parent")

    # === Phase 2: Block Adam polish ===
    print(f"\n=== Phase 2: Block Adam (best={best_score:.6f}) ===")
    if best_a is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.12, 120)
        if t_alloc > 10:
            print(f"  Block Adam, t={t_alloc:.0f}s")
            a_opt, obj = block_adam_polish(best_a, time_limit=t_alloc,
                                           block_size=16, lr=0.004)
            update_best(a_opt, "block_adam")

    # === Phase 3: Coordinate bisection ===
    print(f"\n=== Phase 3: Coordinate bisection (best={best_score:.6f}) ===")
    if best_a is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.15, 200)
        if t_alloc > 15:
            print(f"  CD polish, t={t_alloc:.0f}s")
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "cd_parent")

    # === Phase 4: Symmetric variant ===
    print(f"\n=== Phase 4: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            a_sym = symmetrize(best_a)
            t_alloc = min(remaining * 0.12, 150)
            print(f"  Symmetric Adam, t={t_alloc:.0f}s")
            a_opt, obj = adam_simplex(a_sym, time_limit=t_alloc,
                                      beta_start=5.0, beta_end=200.0,
                                      top_k=30, lr_base=0.008, verbose=False, tag="sym")
            update_best(a_opt, "sym_adam")
            # Then CD
            remaining = deadline - time.monotonic()
            t_alloc = min(remaining * 0.05, 60)
            if t_alloc > 10:
                a_opt2, obj2 = coordinate_bisection(a_opt, time_limit=t_alloc)
                update_best(a_opt2, "sym_cd")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    target_lengths = [1024, 1536, 2048]

    for target_n in target_lengths:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_adam = min(remaining * 0.15, 120)
        t_cd = min(remaining * 0.06, 50)
        print(f"  n={target_n}: Adam t={t_adam:.0f}s, CD t={t_cd:.0f}s")

        a_opt, obj = adam_simplex(f_interp, time_limit=t_adam,
                                  beta_start=5.0, beta_end=300.0,
                                  top_k=max(30, target_n // 30), lr_base=0.006,
                                  verbose=False, tag=f"n{target_n}")
        update_best(a_opt, f"adam_n{target_n}")

        remaining = deadline - time.monotonic()
        if remaining > t_cd + 10:
            a_opt2, obj2 = coordinate_bisection(a_opt, time_limit=t_cd)
            update_best(a_opt2, f"cd_n{target_n}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        inits = diverse_initializations(best_a, n_best, rng)

        for label, init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            t_alloc = min(remaining * 0.10, 100)
            print(f"  Restart [{label}], t={t_alloc:.0f}s")
            a_opt, obj = adam_simplex(init, time_limit=t_alloc,
                                      beta_start=5.0, beta_end=250.0,
                                      top_k=30, lr_base=0.008,
                                      verbose=False, tag=label)
            update_best(a_opt, f"restart_{label}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        # Coordinate bisection with more passes
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.25, 300)
        if t_alloc > 20:
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc,
                                              verbose=True, max_passes=300)
            update_best(a_opt, "final_cd")

        # Adam fine polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_alloc = min(remaining * 0.15, 120)
            print(f"  Final Adam, t={t_alloc:.0f}s")
            a_opt, obj = adam_simplex(best_a, time_limit=t_alloc,
                                      beta_start=50.0, beta_end=500.0,
                                      top_k=50, lr_base=0.003,
                                      verbose=False, tag="final")
            update_best(a_opt, "final_adam")

        # Tiny perturbations + CD
        remaining = time.monotonic() + t_alloc - time.monotonic()
        if remaining > 20:
            for p in range(4):
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break
                rng_p = np.random.default_rng(seed + 5000 + p * 7)
                f_pert = project_simplex(best_a + rng_p.standard_normal(len(best_a)) * 0.0008)
                t_alloc = min(remaining * 0.4, 30)
                a_opt, obj = coordinate_bisection(f_pert, time_limit=t_alloc)
                update_best(a_opt, f"perturb_{p}")

        # Block Adam final
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(remaining * 0.3, 60)
            a_opt, obj = block_adam_polish(best_a, time_limit=t_alloc,
                                           block_size=8, lr=0.002)
            update_best(a_opt, "final_block")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
