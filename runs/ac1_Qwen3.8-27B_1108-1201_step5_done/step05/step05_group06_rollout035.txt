```python
import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_gradient(a, tau):
    """Compute the gradient of the log-sum-exp approximation of max(a*a) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    # Weights
    c_shifted = c - c.max()
    w = np.exp(c_shifted / tau)
    w /= w.sum()
    
    # grad_a[j] = sum_k w[k] * (a[k-j]^+ + a[j-k]^+)
    # = conv(a, w_rev)[2n-2-j] + conv(a, w)[j]
    
    fft_size = 1 << (3*n - 2 - 1).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    W_fft = np.fft.rfft(w, fft_size)
    
    conv_aw = np.fft.irfft(A_fft * W_fft, fft_size)[:3*n-2]
    w_rev = w[::-1]
    Wr_fft = np.fft.rfft(w_rev, fft_size)
    conv_awr = np.fft.irfft(A_fft * Wr_fft, fft_size)[:3*n-2]
    
    grad_a = np.empty(n)
    for j in range(n):
        grad_a[j] = conv_awr[2*n-2-j] + conv_aw[j]
    
    return grad_a, c.max()

def lse_descent(a0, time_limit, tau_init=0.005, tau_final=0.0005, lr=0.02, mom=0.9, verbose=False):
    """Projected descent with log-sum-exp smoothed objective."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    v = np.zeros(n)
    it = 0
    cur_tau = tau_init
    cur_lr = lr
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, _ = smooth_max_gradient(a, cur_tau)
        v = mom * v - cur_lr * grad
        a_new = project_simplex(a + v)
        
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        # Anneal tau
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        cur_tau = tau_init * (tau_final / tau_init) ** progress
        # Decay learning rate
        cur_lr = lr * (0.8 + 0.2 * (1 - progress))
        
        if no_improve > 200:
            # Restart from best with noise
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            no_improve = 0
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE iter {it}: obj={best_obj:.10f}, tau={cur_tau:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj
```

```python
import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, tau):
    """
    Compute gradient of logsumexp(a*a, tau) w.r.t. a, and the current max.
    Returns (grad_a, max_c).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    # Softmax weights
    w = np.exp((c - c_max) / tau)
    w /= w.sum()

    # grad_a[j] = sum_k w[k]*(a[k-j]^+ + a[j-k]^+)
    # = conv(a, w_rev)[2n-2-j] + conv(a, w)[j]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    W_fft = np.fft.rfft(w, fft_size)
    conv_aw = np.fft.irfft(A_fft * W_fft, fft_size)[:3 * n - 2]
    w_rev = w[::-1]
    Wr_fft = np.fft.rfft(w_rev, fft_size)
    conv_awr = np.fft.irfft(A_fft * Wr_fft, fft_size)[:3 * n - 2]

    # Vectorized: grad_a[j] = conv_awr[2n-2-j] + conv_aw[j]
    # conv_awr[2n-2-j] for j=0..n-1 gives indices 2n-2, 2n-3, ..., n-1
    grad_a = conv_awr[n - 1:2 * n - 1][::-1] + conv_aw[0:n]
    return grad_a, c_max


def lse_descent(a0, time_limit, tau_init=0.004, tau_final=0.0004,
                lr=0.03, mom=0.92, restart_interval=300, verbose=False):
    """Log-sum-exp smoothed projected gradient descent with momentum."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    v = np.zeros(n)
    it = 0
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        cur_tau = tau_init * (tau_final / tau_init) ** progress
        cur_lr = lr * (1.0 - 0.5 * progress)

        grad, _ = lse_gradient(a, cur_tau)
        v = mom * v - cur_lr * grad
        a_new = project_simplex(a + v)

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > restart_interval:
            rng = np.random.default_rng(it * 13 + 5)
            noise = 0.002 + 0.003 * min(no_improve // restart_interval, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            no_improve = 0

        if verbose and it % 5000 == 0:
            print(f"        LSE iter {it}: obj={best_obj:.10f}, "
                  f"tau={cur_tau:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def freq_perturb(a0, scale, rng):
    """Perturb a sequence in the frequency domain, then project back to simplex."""
    n = len(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    # Add complex noise
    noise = (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A))) * scale
    noise[0] = 0.0  # Preserve DC (sum)
    A_new = A + noise
    # Enforce Hermitian symmetry
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-12:
        a_new /= s
    else:
        a_new = a0.copy()
    return a_new


def coordinate_bisection(a0, time_limit, verbose=False):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < 2000:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-15:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k = int(np.argmax(f_vals))
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-14:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-14:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-15:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_ramp(n, direction='down'):
    """Create a linear ramp sequence."""
    if direction == 'down':
        a = np.arange(n, 0, -1, dtype=np.float64)
    else:
        a = np.arange(1, n + 1, dtype=np.float64)
    a /= a.sum()
    return a


def make_exp_decay(n, rate=0.5):
    """Create an exponential decay sequence."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64) / n)
    a /= a.sum()
    return a


def make_cosine_window(n, cycles=2.0):
    """Create a cosine-modulated window."""
    x = np.arange(n, dtype=np.float64)
    envelope = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
    a = envelope * (1.0 + 0.3 * np.cos(2 * np.pi * cycles * x / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_superposition(n, n_components=3):
    """Create a sequence as superposition of random cosine components."""
    rng = np.random.default_rng(42 + n)
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) * 0.5
    for _ in range(n_components):
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.6)
        a += amp * np.cos(2 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

        ("cosine_3", make_cosine_window(n_ref, 3.0)),
        ("superpos", make_superposition(n_ref, 4)),
        ("mix_parent_ramp", project_simplex(0.6 * base + 0.4 * make_ramp(n_ref, 'down'))),
        ("mix_parent_cos", project_simplex(0.7 * base + 0.3 * make_cosine_window(n_ref, 2.5))),
    ]

    for label, init in inits:
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 100)
        print(f"  {label}, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(init, time_limit=t_alloc, verbose=False)
        update_best(a_opt, label)

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    if best_a is not None:
        target_lengths = [512, 1024, 1536]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 300
            if rem < 60:
                break
            f_interp = interpolate_to_length(best_a, target_n)
            t_alloc = min(rem * 0.35, 150)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = combined_optimize(f_interp, time_limit=t_alloc, verbose=True)
            update_best(a_opt, f"n{target_n}")

            # Also try ramp at this length
            rem = deadline - time.monotonic() - 200
            if rem > 40 and target_n != 512:
                ramp_init = make_ramp(target_n, 'down')
                t_alloc2 = min(rem * 0.3, 80)
                a_opt2, _ = combined_optimize(ramp_init, time_limit=t_alloc2, verbose=False)
                update_best(a_opt2, f"ramp_n{target_n}")

    # === Phase 4: Deep optimization with restarts ===
    print(f"\n=== Phase 4: Deep optimize (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restarts = [
            ("noise3", lambda: project_simplex(best_a + np.random.default_rng(1).standard_normal(n_best) * 0.003)),
            ("noise5", lambda: project_simplex(best_a + np.random.default_rng(2).standard_normal(n_best) * 0.005)),
            ("sym", lambda: project_simplex(0.5 * (best_a + best_a[::-1]))),
            ("freq_p1", lambda: freq_perturb(best_a, 0.008, np.random.default_rng(3))),
            ("freq_p2", lambda: freq_perturb(best_a, 0.004, np.random.default_rng(4))),
            ("ramp_mix", lambda: project_simplex(0.5 * best_a + 0.5 * make_ramp(n_best, 'down'))),
        ]

        for idx, (label, init_fn) in enumerate(restarts):
            rem = deadline - time.monotonic() - 200 + idx * 30
            if rem < 25:
                break
            t_alloc = min(rem, 120)
            print(f"  Restart {label}, t={t_alloc:.0f}s")
            f_init = init_fn()
            a_opt, _ = combined_optimize(f_init, time_limit=t_alloc, verbose=False)
            update_best(a_opt, f"deep_{label}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        # LSE fine-tune
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_alloc = min(rem, 80)
            print(f"  LSE fine, t={t_alloc:.0f}s")
            a_opt, _ = lse_descent(best_a, time_limit=t_alloc,
                                   tau_init=0.001, tau_final=0.0001,
                                   lr=0.01, mom=0.95, verbose=True)
            update_best(a_opt, "lse_fine")

        # CD final
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, min(200, (deadline - time.monotonic() - 20)))
            print(f"  CD final, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection(best_a, time_limit=t_alloc)
            update_best(a_opt, "cd_final")

        # Tiny perturbations + CD
        rem = deadline - time.monotonic() - 15
        if rem > 10 and best_a is not None:
            n_best = len(best_a)
            for p in range(4):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                rng_p = np.random.default_rng(seed + 2000 + p)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.001)
                t_alloc = min(rem, min(20, (deadline - time.monotonic() - 8)))
                a_opt, _ = coordinate_bisection(f_pert, time_limit=t_alloc)
                update_best(a_opt, f"perturb_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
```
