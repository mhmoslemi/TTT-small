Actually `conv(a,a)[k] = Σ_{j} a[j] * a[k-j]` where j ranges so that both j and k-j are in [0, n-1].

For k=0: j=0, so `a[0]*a[0] = a[0]^2`
For k=1: j=0 → a[0]*a[1], j=1 → a[1]*a[0]. So `2*a[0]*a[1]`.

For k = n-1: j can range from max(0, k-(n-1)) to min(n-1, k) = from 0 to n-1.
So `(a*a)[n-1] = Σ_{j=0}^{n-1} a[j] * a[n-1-j]`.

This is NOT `||a||_2^2` unless `a` is symmetric (i.e., `a[j] = a[n-1-j]`).

For a symmetric sequence, `(a*a)[n-1] = Σ_j a[j]^2 = ||a||_2^2`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Spectral initialization (random phases for flat DFT magnitude)
- Multi-peak subgradient descent with adaptive temperature
- Exact coordinate bisection polishing  
- Block coordinate descent for pairs
- Multi-length exploration with interpolation
- Simulated annealing restarts
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_init(n, rng):
    """Initialize with random phases in frequency domain for flat spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    Z = np.exp(1j * phases)
    a = np.real(np.fft.ifft(Z))
    a = np.abs(a)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def compute_subgrad(a, conv, top_k, temp):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

def subgrad_optimize(a0, time_limit, lr, mom, top_k, temp, decay):
    """Subgradient descent with momentum on the simplex."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgrad(a, conv, top_k, cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.002
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            noise = 0.003 + 0.003 * min(no_improve // 100, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.7
            no_improve = 0
    
    return best_a, best_obj

def coord_bisection(a0, time_limit, max_passes=200):
    """Coordinate descent with bisection for exact per-coordinate optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo = 0.0
            hi = 1.0
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                deriv_at_max = f_vals.max()
                k_max = int(np.argmax(f_vals))
                
                b_kj = b[k_max - j] if 0 <= k_max - j < n else 0.0
                B_k = B[k_max]
                ind = 1.0 if (2 * j < conv_len and k_max == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-12:
                    hi = mid
                elif deriv < -1e-12:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-13:
                best_obj = obj

        if not pass_improved:
            if passes > 3:

                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
                conv = fast_convolve(a)
            else:
                break
    
    return best_a, best_obj

def block_pair_optimize(a0, time_limit, max_iter=5000):
    """Optimize pairs of coordinates simultaneously."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        xi = a[i]
        xj = a[j]
        if xi < 1e-13 or xj < 1e-13:
            continue
        
        # Try several transfer amounts
        for frac in [0.1, 0.3, 0.5, 0.7]:
            delta = frac * xi
            a_trial = a.copy()
            a_trial[i] = xi - delta
            a_trial[j] = xj + delta
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial /= st
            
            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_obj

def symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_init(n, rng):
    """Initialize with random phases for near-flat DFT magnitude."""
    phases = rng.uniform(0, 2.0 * np.pi, n)
    Z = np.exp(1j * phases)
    a = np.real(np.fft.ifft(Z))
    a = np.abs(a)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def spectral_init_smooth(n, rng, alpha=0.3):

    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s > 1e-15:
        a_smooth /= s
    # Blend with original
    a_out = (1 - alpha) * a + alpha * a_smooth
    a_out = np.maximum(a_out, 0)
    s = a_out.sum()
    if s > 1e-15:
        a_out /= s
    return a_out


def compute_subgrad(a, conv, top_k, temp):
    """Weighted subgradient from top-k peaks using softmax."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def subgrad_optimize(a0, time_limit, lr, mom, top_k, temp, decay):
    """Projected subgradient descent with momentum on simplex."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgrad(a, conv, top_k, cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.003

        if no_improve > 150:
            rng = np.random.default_rng(it * 13 + 5)
            noise = 0.004 + 0.004 * min(no_improve // 100, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.65
            no_improve = 0

    return best_a, best_obj


def coord_optimize_active(a0, time_limit, max_passes=500):
    """Coordinate descent using active-set method for exact per-coordinate optimum."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            has_diag = (2 * j < conv_len)

            # Active-set method: iterate on which k is the bottleneck
            x_cur = x_old
            for _as in range(10):
                # Evaluate f_k(x_cur) for all k, find max
                om = 1.0 - x_cur
                f_vals = om * om * B + 2.0 * x_cur * om * b_valid
                if has_diag:
                    f_vals[2 * j] += x_cur * x_cur

                k_star = int(np.argmax(f_vals))

                # Minimize f_{k_star}(x) over [0, 1]
                Bk = B[k_star]
                bv = b_valid[k_star]
                ind = 1.0 if (has_diag and k_star == 2 * j) else 0.0

                A_coef = Bk - 2.0 * bv + ind
                B_coef = 2.0 * (bv - Bk)

                if A_coef > 1e-15:
                    x_new = -B_coef / (2.0 * A_coef)
                    x_new = max(0.0, min(1.0, x_new))
                elif A_coef < -1e-15:
                    f0 = Bk
                    f1 = Bk + B_coef + A_coef
                    x_new = 0.0 if f0 <= f1 else 1.0
                else:
                    x_new = 0.0 if B_coef >= 0 else 1.0

                if abs(x_new - x_cur) < 1e-14:
                    x_cur = x_new
                    break

                # Check if k_star is still active at x_new
                x_cur = x_new
                om_new = 1.0 - x_new
                f_check = om_new * om_new * B + 2.0 * x_new * om_new * b_valid
                if has_diag:
                    f_check[2 * j] += x_new * x_new
                k_check = int(np.argmax(f_check))

                if k_check == k_star:
                    x_cur = x_new
                    break

            if abs(x_cur - x_old) < 1e-14:
                continue

            a[j] = x_cur
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_cur) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 37 + 13)
                scale = 0.0006 + 0.0002 * min(passes - 3, 10)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                conv = fast_convolve(a)
            else:
                break

    return best_a, best_obj


def coord_bisection(a0, time_limit, max_passes=200, n_iter=30):
    """Coordinate descent with bisection (fallback, very robust)."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            has_diag = (2 * j < conv_len)

            lo = 0.0
            hi = 1.0
            for _ in range(n_iter):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if has_diag:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                bv = b_valid[k_max]
                Bk = B[k_max]
                ind = 1.0 if (has_diag and k_max == 2 * j) else 0.0
                deriv = 2.0 * (bv - Bk + mid * (Bk - 2.0 * bv + ind))
                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break

            x_new = (lo + hi) / 2.0
            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 41 + 17)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
                conv = fast_convolve(a)
            else:
                break

    return best_a, best_obj


def peak_reduction(a0, time_limit, threshold, max_iter=20000):
    """Iteratively reduce the maximum peak below threshold."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    lr = 0.002

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        peak_val = conv.max()
        if peak_val <= threshold:
            break

        # Find all peaks above threshold
        above = np.where(conv > threshold)[0]
        if len(above) == 0:
            break

        # Weighted subgradient from peaks above threshold
        pv = conv[above]
        w = (pv - threshold)
        w = w / w.sum()

        sg = np.zeros(n)
        for idx in range(len(above)):
            if w[idx] < 1e-12:
                continue
            k = int(above[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                ii = np.arange(js, je + 1)
                sg[ii] += 2.0 * w[idx] * a[k - ii]

        a = project_simplex(a - lr * sg)
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step size
        if it % 100 == 0:
            lr *= 0.98

        # Periodic noise
        if it % 500 == 499:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(a)

    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric and renormalize."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Combines spectral initialization, active-set coordinate descent,
    subgradient methods, peak reduction, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 25.0)

    best_a = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_try)})")

    # === Phase 1: Optimize parent with coordinate descent ===
    if parent is not None:
        print(f"\n=== Phase 1: Parent optimization (n={len(parent)}) ===")
        rem = deadline - time.monotonic() - 900
        if rem > 30:
            t_alloc = min(rem, 350)
            print(f"  Active-set CD, t={t_alloc:.0f}s")
            a_opt, _ = coord_optimize_active(parent, t_alloc, max_passes=300)
            update_best(a_opt, "parent_cd")

            rem = deadline - time.monotonic() - 800
            if rem > 20:
                t_alloc = min(rem, 100)
                print(f"  Bisection CD, t={t_alloc:.0f}s")
                a_opt, _ = coord_bisection(best_a if best_a is not None else parent, t_alloc, max_passes=100)
                update_best(a_opt, "parent_bis")

    # === Phase 2: Multi-length spectral initialization ===
    print(f"\n=== Phase 2: Multi-length spectral init (best={best_score:.6f}) ===")
    lengths_to_try = [768, 1024, 1536, 2048]

    for n_try in lengths_to_try:
        rem = deadline - time.monotonic() - 400
        if rem < 40:
            break
        print(f"  n={n_try}")

        for trial in range(2):
            rem = deadline - time.monotonic() - 300
            if rem < 25:
                break
            r2 = np.random.default_rng(seed * 13 + n_try * 7 + trial * 3)

            if trial == 0:
                a_init = spectral_init(n_try, r2)
            else:
                a_init = spectral_init_smooth(n_try, r2, alpha=0.4)

            # Quick subgradient to get in the basin
            t_sg = min(rem * 0.3, 60)
            a_opt, _ = subgrad_optimize(a_init, t_sg,
                                         lr=0.004, mom=0.92,
                                         top_k=max(15, n_try // 10),
                                         temp=0.0008, decay=0.999)
            update_best(a_opt, f"spec_n{n_try}_t{trial}")

            # Coordinate descent polish
            rem = deadline - time.monotonic() - 200
            if rem > 15 and best_a is not None:
                t_cd = min(rem, 80)
                a_opt, _ = coord_optimize_active(a_opt, t_cd, max_passes=80)
                update_best(a_opt, f"spec_n{n_try}_t{trial}_cd")

    # === Phase 3: Peak reduction on best ===
    print(f"\n=== Phase 3: Peak reduction (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            n_best = len(best_a)
            current_peak = fast_convolve(best_a).max()
            # Target: reduce peak by 5%
            target_peak = current_peak * 0.95
            t_alloc = min(rem, 120)
            print(f"  Current peak={current_peak:.8f}, target={target_peak:.8f}")
            a_opt, _ = peak_reduction(best_a, t_alloc, target_peak)
            update_best(a_opt, "peak_red")

            # Follow up with CD
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                t_cd = min(rem, 80)
                a_opt, _ = coord_bisection(a_opt, t_cd, max_passes=60)
                update_best(a_opt, "peak_red_cd")

    # === Phase 4: Symmetric variant ===
    print(f"\n=== Phase 4: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 100)
            a_opt, _ = coord_optimize_active(a_sym, t_alloc, max_passes=100)
            update_best(a_opt, "symmetric")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restart_configs = [
            ("noise3", 0.003),
            ("noise5", 0.005),
            ("noise8", 0.008),
            ("rev_mix", None),
            ("flat_mix", None),
        ]
        for label, scale in restart_configs:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break

            if label == "rev_mix":
                a_init = project_simplex(0.7 * best_a + 0.3 * best_a[::-1])
            elif label == "flat_mix":
                a_init = project_simplex(0.85 * best_a + 0.15 * np.ones(n_best) / n_best)
            else:
                r3 = np.random.default_rng(seed + hash(label) % 10000)
                a_init = project_simplex(best_a + r3.standard_normal(n_best) * scale)

            t_alloc = min(rem, 60)
            a_opt, _ = coord_optimize_active(a_init, t_alloc, max_passes=50)
            update_best(a_opt, f"restart_{label}")

    # === Phase 6: Final intensive coordinate bisection ===
    print(f"\n=== Phase 6: Final CD polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 180)
            print(f"  Active-set CD, t={t_alloc:.0f}s")
            a_opt, _ = coord_optimize_active(best_a, t_alloc, max_passes=500)
            update_best(a_opt, "final_active")

            rem = deadline - time.monotonic() - 20
            if rem > 15:
                t_alloc = min(rem, 100)
                print(f"  Bisection CD, t={t_alloc:.0f}s")
                a_opt, _ = coord_bisection(best_a, t_alloc, max_passes=200)
                update_best(a_opt, "final_bis")

            # Ultra-fine perturbations
            rem = deadline - time.monotonic() - 10
            if rem > 8 and best_a is not None:
                n_best = len(best_a)
                for p in range(5):
                    rem = deadline - time.monotonic() - 5
                    if rem < 5:
                        break
                    rp = np.random.default_rng(seed + 2000 + p * 11)
                    f_pert = project_simplex(best_a + rp.standard_normal(n_best) * 0.0008)
                    t_a = min(rem, 12)
                    a_opt, _ = coord_bisection(f_pert, t_a, max_passes=40)
                    update_best(a_opt, f"micro_{p}")

            # One more peak reduction at the very end
            rem = deadline - time.monotonic() - 5
            if rem > 8 and best_a is not None:
                t_a = min(rem, 30)
                curr_peak = fast_convolve(best_a).max()
                a_opt, _ = peak_reduction(best_a, t_a, curr_peak * 0.99)
                update_best(a_opt, "final_peakred")

    # Fallback
    if best_a is None:
        if parent is not None:
            best_a = parent.copy()
        else:
            best_a = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Something went wrong, use parent or default
        if parent is not None:
            best_a = parent.copy()
        else:
            best_a = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
