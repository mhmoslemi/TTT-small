"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smoothed L-BFGS on the simplex
(softmax parameterization), FFT-based gradients, and LP-Frank-Wolfe polishing.
Explores multiple sequence lengths and initializations.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(b):
    """Stable softmax."""
    b_max = b.max()
    e = np.exp(b - b_max)
    return e / e.sum()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, conv, tau, top_k=None):
    """
    Compute the gradient of the smoothed max objective.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    if top_k is not None and top_k < len(conv):
        # Use only top-k peaks
        idx = np.argpartition(conv, -top_k)[-top_k:]
        probs = np.zeros(len(conv))
        sub_conv = conv[idx]
        max_c = sub_conv.max()
        sub_probs = np.exp(tau * (sub_conv - max_c))
        sub_probs /= sub_probs.sum()
        probs[idx] = sub_probs
    else:
        max_c = conv.max()
        probs = np.exp(tau * (conv - max_c))
        probs /= probs.sum()
    
    # Gradient: grad_a[m] = 2 * sum_k probs[k] * a[k-m] (valid indices)
    # This is 2 * (probs cross-correlate a)[0:n]
    # Using FFT: (f ⋆ g)[m] = ifft(FFT(f) * conj(FFT(g)))[m]
    fft_size = 1 << (len(probs) + n - 1 - 1).bit_length()
    F = np.fft.rfft(probs, fft_size)
    G = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(F * np.conj(G), fft_size)[:n]
    
    return 2.0 * cross


def smoothed_lbfgs(f0, time_limit, tau_start=50.0, tau_end=5000.0, n_tau_steps=8, max_iter_per_tau=200, top_k=200):

    Gradually increase tau for increasingly sharp approximation.

    f = np.maximum(f0, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    b = np.log(f + 1e-30).copy()
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_conv_max = fast_convolve(f).max()
    
    # Create tau schedule (geometric)
    tau_ratio = (tau_end / tau_start) ** (1.0 / max(1, n_tau_steps - 1))
    taus = [tau_start * (tau_ratio ** i) for i in range(n_tau_steps)]
    
    for tau in taus:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_fn(b_val):
            a_val = softmax(b_val)
            conv_val = fast_convolve(a_val)
            max_c = conv_val.max()

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None

    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        
        g = solve_lp_direction(f, M, n)
        if g is None:
            # Relax M slightly
            M *= 1.01
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search with golden section
        best_alpha = 0.0
        best_val = M
        
        def line_obj(alpha):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return float('inf')
            ft = ft / st
            return fast_convolve(ft).max()
        
        # Golden section search
        lo, hi = 0.0, 1.0
        GR = (1.0 + np.sqrt(5.0)) / 2.0
        c1 = lo + (1.0 / GR) * (hi - lo)
        c2 = lo + (1.0 - 1.0 / GR) * (hi - lo)
        f1 = line_obj(c1)
        f2 = line_obj(c2)
        
        for _ in range(20):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = lo + (1.0 / GR) * (hi - lo)
                f1 = line_obj(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + (1.0 - 1.0 / GR) * (hi - lo)
                f2 = line_obj(c2)
        
        best_alpha = (lo + hi) / 2.0
        best_val = min(f1, f2)
        
        if best_val < M - 1e-14:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Perturbation to escape plateaus
        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def cosine_init(n, freq, amp, seed=0):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def hann_init(n):
    """Hann window initialization."""
    j = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / (n - 1)))
    s = a.sum()
    return a / s


def triangular_init(n):
    """Triangular (Bartlett) window initialization."""
    j = np.arange(n)
    mid = (n - 1) / 2.0
    a = np.maximum(0, 1.0 - np.abs(j - mid) / mid)
    s = a.sum()
    return a / s


def ramp_init(n, skew):
    """Ramp initialization with skew parameter."""
    j = np.arange(n)
    a = 1.0 + skew * j / (n - 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

We want: grad_a[m] = 2 * sum_{k=m}^{m+n-1} probs[k] * a[k-m] for m = 0, ..., n-1

Let l = k - m. Then: grad_a[m] = 2 * sum_{l=0}^{n-1} probs[m+l] * a[l]

This is the cross-correlation of probs and a at lag m: (probs ⋆ a)[m] = sum_l probs[m+l] * a[l]

In signal processing, the cross-correlation (f ⋆ g)[m] = sum_l f[m+l] * conj(g[l])

Using FFT: FFT(f ⋆ g) = FFT(f) * conj(FFT(g))

So: grad_a = 2 * ifft(FFT(probs_padded) * conj(FFT(a_padded)))[0:n]

where we need the FFT size to be at least len(probs) + len(a) - 1 = (2n-1) + n - 1 = 3n-2.
