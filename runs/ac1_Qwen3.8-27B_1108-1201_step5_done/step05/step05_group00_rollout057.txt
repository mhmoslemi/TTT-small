- We want a[k*-l] < a[k*-j] for the peak to decrease.
- But we also need to check that no other lag increases above the current peak.

This is a 1D optimization for delta (with the constraint that no other lag exceeds the current max). We can use bisection.

Let me code this up now. I'll aim for a clean, efficient implementation.

One more thought: the previous solution uses `project_simplex` which projects onto the probability simplex (sum = 1, all non-negative). This is correct but the projection might introduce errors. An alternative is to use the "water-filling" projection which is exact.

The `project_simplex` function in the previous solution uses the standard algorithm (sort, cumulative sum, find threshold), which is O(n log n). This should be fine.

Let me write the final code now.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multi-n exploration with diverse initializations
- High-precision smoothed gradient descent with bisection line search
- Peak-targeting subgradient mode for fine convergence
- Pairwise coordinate swap optimization
- FW-LP for global direction
"""
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_gradient_at_peak(a, k):
    """Subgradient of (a*a)[k] w.r.t. a. Returns 2*a flipped/shifted."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(max(0, k - n + 1), min(n - 1, k) + 1):
        grad[j] = 2.0 * a[k - j]
    return grad

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    if n > 2000:  # Too large for dense LP
        return None
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        A_ub[row_start:row_start + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def smoothed_gradient_step(a, T, lr, max_bisect=40):
    """
    One step of smoothed gradient descent with high-precision bisection line search.
    Returns (new_a, new_objective, old_objective).
    """
    n = len(a)
    conv = fast_convolve(a)
    old_obj = conv.max()
    
    # Compute softmax weights
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(conv)) / len(conv)
    else:
        p = p / p_sum
    
    # Gradient: 2 * correlate(p, a) -- efficient computation
    # grad[j] = 2 * sum_k p[k] * a[k-j] (where valid)
    # This is equivalent to 2 * convolve(p, reversed_a)[n-1-j]... let me think
    # Actually: grad_j = 2 * sum_k p[k] * (d/d a_j) (a*a)[k]
    # (a*a)[k] = sum_i a[i]*a[k-i], d/d a_j = a[k-j] + a[j-k] (if in range)
    # Wait, (a*a)[k] = sum_i a[i] * a[k-i]
    # d/d a_j: the term a[i]*a[k-i] depends on a_j when i=j (gives a[k-j]) or k-i=j (gives a[i]=a[k-j])
    # So d/d a_j (a*a)[k] = a[k-j] + a[k-j] = 2*a[k-j] if 0 <= k-j < n, else 0
    # Wait no: if i = j, we get a[k-j]. If k-i = j, i.e., i = k-j, we get a[i] = a[k-j].
    # But these are the same term! (a*a)[k] = sum_i a[i]*a[k-i]. The term at i=j is a[j]*a[k-j].
    # d/d a_j of this term = a[k-j] + a[j]*0 = a[k-j] (since k-j is fixed)
    # Wait, d/d a_j (a[i]*a[k-i]) for i=j: d/d a_j (a[j]*a[k-j]) = a[k-j] (if k-j != j) or 2*a[j] (if k-j = j)
    # Hmm, this is getting complicated. Let me just use the correlation form.
    # 
    # Actually, the gradient of sum_k p[k] * (a*a)[k] w.r.t. a[j] is:
    # sum_k p[k] * 2*a[k-j] (for valid k-j in [0,n-1])
    # = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # Let m = k-j, then k = j+m, m in [0, n-1], k in [j, j+n-1] ∩ [0, 2n-2]
    # = 2 * sum_m p[j+m] * a[m] (for valid j+m)
    # This is a cross-correlation of p and a!
    
    # Efficient computation via FFT:
    # grad[j] = 2 * sum_m a[m] * p[j+m]
    # This is the correlation of a and p, evaluated at lag j
    # = (a * reverse(p))[n-1+j] ... hmm
    # Actually, let's use np.correlate
    # np.correlate(p, a, mode='full')[j + n - 1] = sum_m p[m + j] * a[m]... let me verify
    # np.correlate(a, v, mode='full')[k] = sum_n a[n+k] * conj(v[n])... 
    # This is getting confusing. Let me just use FFT.
    
    # Method: pad a and p to size, FFT, multiply, IFFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F_a = np.fft.rfft(a, fft_size)
    F_p = np.fft.rfft(p, fft_size)
    # We want grad[j] = 2 * sum_m a[m] * p[j+m]
    # This is the "cross-correlation" type sum
    # In FFT terms: if we compute IFFT(FFT(a) * FFT(reverse(p))), we get...
    # Let me just do it directly for clarity, or use a trick.
    # 
    # Alternative: grad[j] = 2 * sum_m a[m] * p[j+m]
    # Let q[m] = a[m], r[k] = p[k]
    # grad[j] = 2 * sum_m q[m] * r[j+m]
    # = 2 * (q * r_reversed)[j] where r_reversed is p reversed... 
    # Actually this is just a correlation. Let me use np.correlate.
    # np.correlate(p, a, 'full')[j + (n-1)] should give sum_k p[k] * a[k-j]... 
    # I'm getting confused. Let me just compute it via FFT properly.
    
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m] for j = 0, ..., n-1
    # where p has length 2n-1 (indices 0 to 2n-2)
    # and we need j+m to be in [0, 2n-2]
    
    # This is a "sliding dot product" - can be computed as a correlation
    # Using FFT: 
    # Let A = FFT(a padded to fft_size)
    # Let P = FFT(p padded to fft_size)
    # Then the cross-correlation C[j] = sum_m a[m]*p[j+m] can be obtained from...
    
    # Actually the simplest way: 
    # C[j] = sum_m a[m] * p[j+m] = (a ⊛ p)[j] where ⊛ denotes cross-correlation
    # In numpy: np.correlate(p, a, mode='valid') gives sum_m p[m+j]*a[m] for j = 0,...,len(p)-len(a)
    # Since len(p) = 2n-1 and len(a) = n, 'valid' gives n-1 values... not quite what we want.
    
    # Let me use a different approach: padding
    # p has length 2n-1. a has length n.
    # We want for each j in [0, n-1]: sum_{m=0}^{n-1} a[m] * p[j+m]
    # This is like a sliding window of size n over p, dot-producted with a.
    # We can compute this as: correlate(p, a, mode='full')[n-1 : 2n-1]... let me verify.
    
    # np.correlate(p, a, mode='full')[k] = sum_m p[m] * a[m-k+n-1]... 
    # OK this is getting too confusing. Let me just use the direct FFT approach.
    
    # The sum S[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    # = sum_{m=0}^{n-1} a[m] * p[m+j]
    # This is the cross-correlation of a and p, at lag j.
    # In terms of convolution: S[j] = (a_reversed * p)[j + n - 1]
    # where a_reversed = a[::-1]
    
    a_rev = a[::-1]
    size2 = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size2 = 1 << (size2 - 1).bit_length()
    F_ar = np.fft.rfft(a_rev, fft_size2)
    F_p2 = np.fft.rfft(p, fft_size2)
    conv_full = np.fft.irfft(F_ar * F_p2, fft_size2)
    # S[j] = conv_full[j + n - 1] for j = 0, ..., n-1
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    
    # Line search with bisection
    best_alpha = 0.0
    best_val = old_obj
    
    # First try a few candidates
    alphas = [lr, lr*0.5, lr*0.2, lr*0.1, lr*2]
    for alpha in alphas:
        candidate = project_simplex(a - alpha * grad)
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha == 0:
        return a, old_obj, old_obj
    
    # Bisection refinement
    lo = 0.0
    hi = min(best_alpha * 2, 1.0)
    for _ in range(max_bisect):
        mid = (lo + hi) / 2.0
        candidate = project_simplex(a - mid * grad)
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val:
            best_val = val
            best_alpha = mid
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-10:
            break
    
    new_a = project_simplex(a - best_alpha * grad)
    return new_a, best_val, old_obj

def peak_targeting_step(a, max_bisect=30):
    """
    Subgradient step targeting the single highest peak.
    """
    n = len(a)
    conv = fast_convolve(a)
    old_obj = conv.max()
    k_peak = int(np.argmax(conv))
    
    # Subgradient at the peak
    grad = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    for j in range(js, je + 1):
        grad[j] = 2.0 * a[k_peak - j]
    
    # Multi-peak: also include peaks within 5% of max
    threshold = old_obj * 0.95
    near_peaks = np.where(conv >= threshold)[0]
    for k in near_peaks:
        if k == k_peak:
            continue
        js2 = max(0, k - n + 1)
        je2 = min(n - 1, k)
        for j in range(js2, je2 + 1):
            grad[j] += 2.0 * a[k - j]
    
    #Normalize gradient for stability
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, old_obj, old_obj
    grad = grad / gnorm
    
    # Line search
    best_alpha = 0.0
    best_val = old_obj
    
    for lr in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
        candidate = project_simplex(a - lr * grad)
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val:
            best_val = val
            best_alpha = lr
    
    if best_alpha == 0:
        return a, old_obj, old_obj
    
    # Bisection
    lo = best_alpha * 0.5
    hi = min(best_alpha * 2, 0.5)
    for _ in range(max_bisect):
        mid = (lo + hi) / 2.0
        candidate = project_simplex(a - mid * grad)
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val:
            best_val = val
            best_alpha = mid
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-10:
            break
    
    new_a = project_simplex(a - best_alpha * grad)
    return new_a, best_val, old_obj

def pairwise_swap_optimize(a, max_iter=200):
    """
    Targeted pairwise swaps to reduce the peak.
    For the current peak, move mass from high-contributing positions to low-contributing ones.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    a = a.copy()
    
    for _ in range(max_iter):
        conv = fast_convolve(a)
        peak_val = conv.max()
        k_peak = int(np.argmax(conv))
        
        # Find the "cost" of each position for this peak
        # cost[j] = a[k_peak - j] (contribution to the peak from position j)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = list(range(js, je + 1))
        
        if len(affected) < 2:
            break
        
        # For each position j in affected, its "cost" is a[k_peak - j]
        # We want to move mass from high-cost to low-cost positions
        costs = np.array([a[k_peak - j] for j in affected])
        vals = np.array([a[j] for j in affected])
        
        # Find the max-cost and min-cost positions
        i_max = int(np.argmax(costs))
        i_min = int(np.argmin(costs))
        
        if i_max == i_min or costs[i_max] <= costs[i_min] * 1.01:
            # No significant difference, try random pair
            break
        
        j_from = affected[i_max]  # position to reduce
        j_to = affected[i_min]    # position to increase
        
        # The change in peak when we move delta from j_from to j_to:
        # (a*a)[k_peak] changes by -delta * a[k_peak - j_from] + delta * a[k_peak - j_to]
        # = delta * (a[k_peak - j_to] - a[k_peak - j_from])
        # Since a[k_peak - j_from] > a[k_peak - j_to], this is negative (good!)
        
        # But we need to ensure other peaks don't exceed current max
        # For safety, use bisection on delta
        
        max_delta = min(a[j_from], 0.1) * 0.5  # Don't move more than 50% of the source
        
        lo_d = 0.0
        hi_d = max_delta
        best_d = 0.0
        best_v = peak_val
        
        for _ in range(25):
            mid_d = (lo_d + hi_d) / 2.0
            if mid_d < 1e-15:
                break
            a_candidate = a.copy()
            a_candidate[j_from] -= mid_d
            a_candidate[j_to] += mid_d
            if a_candidate[j_from] < 0:
                hi_d = mid_d
                continue
            c = fast_convolve(a_candidate)
            v = c.max()
            if v < best_v - 1e-15:
                best_v = v
                best_d = mid_d
                lo_d = mid_d
            else:
                hi_d = mid_d
        
        if best_d > 1e-12:
            a[j_from] -= best_d
            a[j_to] += best_d
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
        else:
            # Try a different pair
            # Pick random from top-cost and bottom-cost
            top_cost = np.argsort(costs)[::-1][:5]
            bot_cost = np.argsort(costs)[:5]
            found = False
            for ti in top_cost:
                for bi in bot_cost:
                    if ti == bi:
                        continue
                    jf = affected[ti]
                    jt = affected[bi]
                    max_d2 = min(a[jf], 0.05) * 0.3
                    for d in [max_d2, max_d2*0.5, max_d2*0.1]:
                        a_cand = a.copy()
                        a_cand[jf] -= d
                        a_cand[jt] += d
                        if a_cand[jf] < 0:
                            continue
                        c2 = fast_convolve(a_cand)
                        if c2.max() < best_obj - 1e-15:
                            a = a_cand
                            a = np.maximum(a, 0)
                            a = a / a.sum()
                            best_obj = fast_convolve(a).max()
                            best_a = a.copy()
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if not found:
                break
    
    return best_a, best_obj

def fw_lp_step(a, M_multiplier=1.0, lp_timeout=10):
    """One FW-LP step. Returns new a or None."""
    n = len(a)
    if n > 1500:
        return None
    conv = fast_convolve(a)
    M = conv.max() * M_multiplier
    g = solve_lp_oracle(a, M)
    if g is None:
        return None
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    g = g / g_sum
    
    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        candidate = (1 - alpha) * a + alpha * g
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha == 0:
        return None
    
    new_a = (1 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0)
    new_a = new_a / new_a.sum()
    return new_a

def make_initialization(n, method, rng):
    """Create initial sequence."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'dirichlet':
        alpha = 0.3 + rng.random() * 0.7
        return rng.dirichlet(np.ones(n) * alpha)
    elif method == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        center = n / 2
        sigma = n * (0.1 + rng.random() * 0.3)
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        g += 1e-10
        return g / g.sum()
    elif method == 'edge':
        x = np.arange(n, dtype=np.float64)
        w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
        return w / w.sum()
    elif method == 'cosine':
        x = np.arange(n, dtype=np.float64)
        c = np.abs(np.cos(np.pi * x / n)) ** (1 + rng.random() * 3) + 0.01
        return c / c.sum()
    elif method == 'two_peak':
        x = np.arange(n, dtype=np.float64)
        c1 = rng.random() * 0.5 + 0.25
        c2 = 1 - c1
        w = 0.1
        g = np.exp(-0.5 * ((x - c1*n) / (n*w))**2) + np.exp(-0.5 * ((x - c2*n) / (n*w))**2)
        g += 1e-10
        return g / g.sum()
    elif method == 'parent_interp':
        return None  # Handled separately
    else:
        return np.ones(n) / n

def main_optimize(a0, time_limit, T_init=0.001, verbose=True):
    """
    Main optimization loop combining smoothed gradient, peak targeting, and pairwise swaps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = 0.02
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break
        
        # Choose method based on iteration
        phase = (it % 100)
        
        if phase < 60:
            # Smoothed gradient
            a, new_obj, old_obj = smoothed_gradient_step(a, T, lr, max_bisect=35)
        elif phase < 80:
            # Peak targeting
            a, new_obj, old_obj = peak_targeting_step(a, max_bisect=25)
        elif phase < 90:
            # Pairwise swap
            a, new_obj = pairwise_swap_optimize(a, max_iter=50)
            old_obj = new_obj  # not used
        else:
            # Try FW-LP (expensive, do rarely)
            if n <= 1000:
                new_a = fw_lp_step(a, M_multiplier=1.0)
                if new_a is not None:
                    c = fast_convolve(new_a)
                    new_obj = c.max()
                    if new_obj < best_obj:
                        a = new_a
                        old_obj = new_obj
                else:
                    a, new_obj, old_obj = smoothed_gradient_step(a, T, lr, max_bisect=35)
            else:
                a, new_obj, old_obj = smoothed_gradient_step(a, T, lr, max_bisect=35)
        
        # Update best
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt parameters
        if no_improve > 10:
            T *= 1.03
        if no_improve > 30:
            T *= 1.05
            lr *= 0.95
        if no_improve > 80:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise_scale = 0.001 + 0.003 * min(1.0, (no_improve - 80) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = max(T_init * 0.5, T * 0.5)
            lr = 0.02
            no_improve = 0
        
        # Very slow decay of T and lr
        T = max(0.00001, T * 0.9999)
        lr = max(0.001, lr * 0.9999)
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.4f}, "
                  f"no_improve={no_improve}, elapsed={elapsed:.0f}s, remaining={remaining:.0f}s")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multi-n exploration with diverse initializations
- High-precision smoothed gradient descent with bisection line search and momentum
- Peak-targeting subgradient mode for fine convergence
- Pairwise coordinate swap optimization for targeted peak reduction
- Adaptive temperature scheduling and periodic restarts
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_gradient(a, p):
    """
    Compute gradient of sum_k p[k] * (a*a)[k] w.r.t. a.
    grad[j] = 2 * sum_m a[m] * p[j+m] = 2 * correlate(p, a, 'valid')[j]
    """
    return 2.0 * np.correlate(p, a, mode='valid')


def softmax_weights(conv, T):
    """Compute softmax weights from convolution values."""
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return p / p_sum


def smoothed_gradient_step(a, T, lr, momentum, v, max_bisect=30):
    """
    One step of smoothed gradient descent with momentum and bisection line search.
    Returns (new_a, new_v, new_obj).
    """
    n = len(a)
    conv = fast_convolve(a)
    old_obj = conv.max()

    p = softmax_weights(conv, T)
    grad = conv_gradient(a, p)

    # Momentum update
    v_new = momentum * v - lr * grad

    # Line search with bisection for the effective step
    best_alpha = 0.0
    best_val = old_obj
    best_candidate = a.copy()

    # Try several step sizes
    for alpha in [1.0, 0.5, 0.25, 0.1, 2.0]:
        candidate = project_simplex(a + alpha * v_new)
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
            best_candidate = candidate

    if best_alpha > 0 and best_val < old_obj:
        # Bisection refinement
        lo = max(0.0, best_alpha - 0.3)
        hi = min(best_alpha + 0.3, 5.0)
        for _ in range(max_bisect):
            mid = (lo + hi) / 2.0
            candidate = project_simplex(a + mid * v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = mid
                best_candidate = candidate
                lo = mid
            else:
                hi = mid
            if hi - lo < 1e-10:
                break
        return best_candidate, v_new, best_val
    else:
        return a, v_new, old_obj


def peak_targeting_step(a, lr_init=0.01, max_bisect=25):
    """
    Subgradient step targeting the top peaks (within 5% of max).
    Returns (new_a, new_obj).
    """
    n = len(a)
    conv = fast_convolve(a)
    old_obj = conv.max()

    # Find peaks within 5% of max
    threshold = old_obj * 0.95
    near_peaks = np.where(conv >= threshold)[0]

    # Compute weighted subgradient
    grad = np.zeros(n)
    for k in near_peaks:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je + 1] += 2.0 * a[k - js:je + 1][::-1]

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, old_obj
    grad = grad / gnorm

    # Line search
    best_alpha = 0.0
    best_val = old_obj
    best_candidate = a.copy()

    for lr in [lr_init, lr_init * 0.5, lr_init * 2, lr_init * 0.1]:
        candidate = project_simplex(a - lr * grad)
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = lr
            best_candidate = candidate

    if best_alpha == 0:
        return a, old_obj

    # Bisection
    lo = best_alpha * 0.5
    hi = min(best_alpha * 2.0, 0.5)
    for _ in range(max_bisect):

        candidate = project_simplex(a - mid * grad)
        c = fast_convolve(candidate)
        val = c.max()

            best_val = val
            best_alpha = mid
            best_candidate = candidate
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-10:
            break

    return best_candidate, best_val

                    break

    return best_a, best_obj


def solve_lp_oracle(f, M, timeout=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    if n > 1200:
        return None
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_step(a, m_multiplier=1.0):
    """One FW-LP step."""
    n = len(a)
    if n > 1200:
        return None
    conv = fast_convolve(a)
    M = conv.max() * m_multiplier
    g = solve_lp_oracle(a, M)
    if g is None:
        return None
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    g = g / g_sum

    best_alpha = 0.0
    best_val = M
    best_candidate = None
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        candidate = (1 - alpha) * a + alpha * g
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        c = fast_convolve(candidate)
        val = c.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
            best_candidate = candidate

    if best_alpha == 0:
        return None
    return best_candidate


def main_optimize(a0, time_limit, T_init=0.001, verbose=False):
    """
    Main optimization: combines smoothed gradient, peak targeting,
    pairwise swaps, and occasional FW-LP steps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        # Choose method based on cycle position
        phase = it % 100

        if phase < 55:
            # Smoothed gradient (primary)
            a, v, new_obj = smoothed_gradient_step(a, T, lr, momentum, v, max_bisect=30)
        elif phase < 75:
            # Peak targeting
            a, new_obj = peak_targeting_step(a, lr_init=0.005)
            v = np.zeros(n)
        elif phase < 90:
            # Pairwise swaps
            a, new_obj = pairwise_swap_optimize(a, max_iter=30)
            v = np.zeros(n)
        else:
            # FW-LP or fallback to smoothed gradient
            new_a = fw_lp_step(a, m_multiplier=1.0)
            if new_a is not None:
                c = fast_convolve(new_a)
                new_obj = c.max()
                if new_obj < best_obj:
                    a = new_a
                v = np.zeros(n)
            else:
                a, v, new_obj = smoothed_gradient_step(a, T, lr, momentum, v, max_bisect=30)

        # Update best
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt hyperparameters
        if no_improve > 15:
            T *= 1.04
        if no_improve > 40:
            lr *= 0.95
            momentum = min(0.95, momentum + 0.005)
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise_scale = 0.001 + 0.005 * min(1.0, (no_improve - 100) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = max(T_init * 0.5, T * 0.4)
            lr = 0.02
            momentum = 0.9
            v = np.zeros(n)
            no_improve = 0

        # Very gradual decay
        T = max(0.00001, T * 0.99995)
        lr = max(0.0005, lr * 0.9999)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.5f}, "
                  f"no_imp={no_improve}, rem={remaining:.0f}s")

    return best_a, best_obj


def make_initialization(n, method, rng):
    """Create initial sequence of given length."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'dirichlet':
        alpha = 0.3 + rng.random() * 0.7
        return rng.dirichlet(np.ones(n) * alpha)
    elif method == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        center = n / 2
        sigma = n * (0.1 + rng.random() * 0.3)
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 1e-10
        return g / g.sum()
    elif method == 'edge':
        x = np.arange(n, dtype=np.float64)
        w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
        return w / w.sum()
    elif method == 'cosine':
        x = np.arange(n, dtype=np.float64)
        power = 1.0 + rng.random() * 3.0
        c = np.abs(np.cos(np.pi * x / n)) ** power + 0.01
        return c / c.sum()
    elif method == 'two_peak':
        x = np.arange(n, dtype=np.float64)
        c1 = 0.25 + rng.random() * 0.2
        c2 = 0.55 + rng.random() * 0.2
        w = 0.08 + rng.random() * 0.1
        g = (np.exp(-0.5 * ((x - c1 * n) / (n * w)) ** 2) +
             np.exp(-0.5 * ((x - c2 * n) / (n * w)) ** 2)) + 1e-10
        return g / g.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Multi-n exploration with diverse starts (350s)
    Phase 2: Deep local optimization with varied hyperparams (1050s)
    Phase 3: Ultra-fine polish and final tweaks (330s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Multi-n Exploration (350s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        max(256, n_parent - 256),
        n_parent + 256,
        512, 640, 896, 1024,
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 25:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss < 1e-10:
                start = np.ones(n_try) / n_try
            else:
                start = start / ss

        t_alloc = min(remaining, 70)
        print(f"  n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = main_optimize(start, time_limit=t_alloc, T_init=0.002, verbose=False)
        try_seq(a_opt, f"explore_n{n_try}")

    # Try diverse initializations at best n
    if best_seq is not None:
        n_best = len(best_seq)
        for init_method in ['gaussian', 'edge', 'cosine', 'two_peak', 'dirichlet']:
            remaining = deadline - time.monotonic() - 1100
            if remaining < 15:
                break
            init = make_initialization(n_best, init_method, rng)
            t_alloc = min(remaining, 35)
            a_opt, _ = main_optimize(init, time_limit=t_alloc, T_init=0.003, verbose=False)
            try_seq(a_opt, f"init_{init_method}_n{n_best}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep Local Optimization (1050s) ===
    print("\n=== Phase 2: Deep Local Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple runs with different temperature settings
        configs = [
            {'T_init': 0.0003, 'budget': 250},
            {'T_init': 0.0008, 'budget': 250},
            {'T_init': 0.002, 'budget': 250},
            {'T_init': 0.005, 'budget': 200},
        ]

        for cfg in configs:
            remaining = deadline - time.monotonic() - 300
            if remaining < 30:
                break
            t_alloc = min(remaining, cfg['budget'])
            print(f"  T={cfg['T_init']}, t={t_alloc:.0f}s")
            a_opt, _ = main_optimize(best_seq, time_limit=t_alloc,
                                     T_init=cfg['T_init'], verbose=True)
            try_seq(a_opt, f"T={cfg['T_init']}")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01, 0.015, 0.03]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 80)
            a_opt, _ = main_optimize(perturbed, time_limit=t_alloc,
                                     T_init=0.001, verbose=False)
            try_seq(a_opt, f"perturb_{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Ultra-fine Polish (330s) ===
    print("\n=== Phase 3: Ultra-fine Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very low temperature, small steps
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            t_alloc = min(remaining, 180)
            print(f"  Fine polish T=0.00005, t={t_alloc:.0f}s")
            a_opt, _ = main_optimize(best_seq, time_limit=t_alloc,
                                     T_init=0.00005, verbose=True)
            try_seq(a_opt, "fine_T0.00005")

        # Pairwise swap intensive
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            n_iter = max(200, int(remaining * 40))
            print(f"  Pairwise swap, iter={n_iter}")
            a_opt, obj = pairwise_swap_optimize(best_seq, max_iter=n_iter)
            try_seq(a_opt, "swap_intensive")

        # Final peak targeting
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            a_opt, obj = peak_targeting_step(best_seq, lr_init=0.002, max_bisect=35)
            try_seq(a_opt, "peak_final")

        # Tiny perturbations with quick local opt
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            for _ in range(8):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.0003
                perturbed = project_simplex(best_seq + noise)
                a_opt, _ = main_optimize(perturbed, time_limit=min(remaining, 15),
                                         T_init=0.00005, verbose=False)
                try_seq(a_opt, "micro_perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
