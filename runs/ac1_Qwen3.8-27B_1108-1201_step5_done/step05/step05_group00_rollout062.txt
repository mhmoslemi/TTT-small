def solve_qp_subproblem(p, n):
    """Solve min a^T H_p a s.t. a >= 0, sum a = 1, where H_p[j,m] = p[j+m]."""
    import cvxpy as cp
    
    a = cp.Variable(n)
    
    # Build the quadratic form using the Hankel structure
    # a^T H_p a = sum_j,m a[j]*a[m]*p[j+m]
    # We can use cp.quad_form(a, H_p) if H_p is PSD
    
    # For efficiency, build H_p as a dense matrix
    H = np.array([[p[i+j] for j in range(n)] for i in range(n)])
    
    # Check PSD
    # If not PSD, add regularization
    eigvals = np.linalg.eigvalsh(H)
    if eigvals.min() < -1e-10:
        # Add regularization to make it PSD
        reg = -eigvals.min() + 1e-6
        H = H + reg * np.eye(n)
    
    objective = cp.Minimize(cp.quad_form(a, H))
    constraints = [a >= 0, cp.sum(a) == 1]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)
    
    if a.value is not None:
        return np.maximum(a.value, 0)
    return None

def fast_convolve(a):

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    """Compute H_p @ a where H_p[j,m] = p[j+m]."""
    n = len(a)
    # H_p @ a = [sum_m p[j+m]*a[m] for j in range(n)]
    # This is a valid correlation
    result = np.zeros(n)
    for j in range(n):
        # sum_m p[j+m] * a[m] for m=0..n-1
        # p has length 2n-1, so j+m ranges from j to j+n-1
        result[j] = np.dot(p[j:j+n], a)
    return result

def smooth_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.02, momentum=0.9):
    """Smoothed gradient descent with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        
        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient computation using hankel_mv
        # grad[i] = sum_k p[k] * (a[k-i] + a[i-k]) for valid indices
        # This equals 2 * hankel_mv(p, a) (by symmetry of the Hankel structure)
        # Actually: d(conv[k])/da[i] = a[k-i] (if valid) + a[i-k] (if valid)
        # sum_k p[k] * a[k-i] = (p * a)[correlation] = hankel_mv(p, a)[i] (for one direction)
        # sum_k p[k] * a[i-k] = same by symmetry
        # So grad = 2 * hankel_mv(p, a)
        grad = 2.0 * hankel_mv(p, a)
        
        # Momentum update
        v = momentum * v - lr * grad
        candidate = project_simplex(a + v)
        
        # Line search
        best_candidate = candidate
        best_val = fast_convolve(candidate).max()
        
        for backtrack in [0.5, 0.25, 0.1]:
            cand2 = project_simplex(a + backtrack * v)
            val2 = fast_convolve(cand2).max()
            if val2 < best_val:
                best_val = val2
                best_candidate = cand2
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if no_improve == 20:
            lr *= 0.8
        elif no_improve == 40:
            T *= 1.3
            lr = lr_init * 0.5
        elif no_improve == 80:
            T *= 0.7
            lr = lr_init * 0.3
            momentum = min(0.95, momentum + 0.01)
        elif no_improve > 150:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            momentum = 0.9
            no_improve = 0
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit):
    """Coordinate descent: for each peak, try to reduce it by adjusting nearby variables."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Variables that affect conv[k_peak]
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        
        if len(affected) == 0:
            break
        
        # Try to reduce the peak by moving mass from high values to low values within affected set
        a_affected = a[affected]
        max_idx = affected[np.argmax(a_affected)]
        min_idx = affected[np.argmin(a_affected)]
        
        # Try moving a small amount from max to min
        delta = a[max_idx] * 0.01
        for frac in [0.005, 0.01, 0.02, 0.05]:
            delta = a[max_idx] * frac
            if delta < a[min_idx] or delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[max_idx] -= delta
            a_trial[min_idx] += delta
            if a_trial[max_idx] < 0:
                continue
            val = fast_convolve(a_trial).max()
            if val < best_obj - 1e-15:
                a = a_trial
                best_obj = val
                best_a = a.copy()
                break
        
        # Also try random perturbations
        if it % 3 == 0:
            rng = np.random.default_rng((it * 12345 + k_peak) % (2**31))
            for scale in [0.001, 0.002, 0.005]:
                pert = rng.standard_normal(n) * scale
                pert[np.ones(n, dtype=bool)] *= 0  # zero out
                # Only perturb affected variables
                pert_full = np.zeros(n)
                pert_full[affected] = pert[:len(affected)]
                candidate = project_simplex(a + pert_full)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    break
    
    return best_a, best_obj

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute H_p @ a where H_p[j,m] = p[j+m]. p length 2n-1, a length n."""
    return np.correlate(p, a, mode='valid')

def smooth_grad(a0, time_limit, T_init=0.005, lr_init=0.015, mom_init=0.9):
    """Smoothed gradient descent with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    mom = mom_init
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Softmax weights with numerical stability
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum

        # Gradient: 2 * H_p @ a
        grad = 2.0 * hankel_mv(p, a)

        # Momentum + backtracking line search
        v = mom * v - lr * grad
        best_candidate = None
        best_val = fast_convolve(a).max()

        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        if best_candidate is not None:
            a = best_candidate
        else:
            # If no improvement, try a smaller direct step
            candidate = project_simplex(a - 0.001 * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                a = candidate
                best_val = val
            else:
                a = project_simplex(a + rng.standard_normal(n) * 0.0005)
                best_val = fast_convolve(a).max()

        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameter updates
        if no_improve == 15:
            lr *= 0.7
        elif no_improve == 30:
            T *= 1.5
            lr = lr_init * 0.4
            mom = min(0.97, mom + 0.02)
        elif no_improve == 60:
            T *= 0.6
            lr = lr_init * 0.2
        elif no_improve > 120:
            noise_scale = 0.002 + 0.003 * min(1.0, (no_improve - 120) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            mom = mom_init
            no_improve = 0

    return best_a, best_obj


def peak_targeted_grad(a0, time_limit, top_k=20, lr_init=0.003, mom_init=0.93):
    """Subgradient method targeting only the top-k peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99991 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Select top-k peaks
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        # Inverse-rank weighting
        order = np.argsort(pv)[::-1]
        weights = np.zeros(k)
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        weights = weights / ws

        # Compute subgradient
        grad = np.zeros(n)
        for i in range(k):
            if weights[i] < 1e-15:
                continue
            kk = int(peaks[i])
            sg_part = np.zeros(n)
            # d(conv[kk])/da[j] = a[kk-j] (if valid) + a[j-kk] (if valid)
            js_low = max(0, kk - n + 1)
            js_high = min(n - 1, kk)
            for j in range(js_low, js_high + 1):
                idx1 = kk - j
                idx2 = j - kk
                term = 0.0
                if 0 <= idx1 < n:
                    term += a[idx1]
                if 0 <= idx2 < n:
                    term += a[idx2]
                sg_part[j] = term
            grad += weights[i] * sg_part

        # Update
        v = mom * v - lr * grad
        best_candidate = a
        best_val = fast_convolve(a).max()
        for scale in [1.0, 0.5, 0.25]:
            candidate = project_simplex(a + scale * v)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 100:
            lr = lr_init * 0.5
            mom = min(0.97, mom + 0.01)
        if no_improve > 300:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0

    return best_a, best_obj


def game_iterate(a0, time_limit, T_init=0.01, use_cvxpy=True):
    """Game iterate: dual = softmax(conv/T), primal = QP solve."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(55555 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Solve QP: min a'^T H_p a' s.t. a' >= 0, sum(a') = 1
        H = None
        a_new = None

        if use_cvxpy:
            try:
                import cvxpy as cp
                # Build Hankel matrix
                idx = np.arange(n)
                H = p[idx[:, None] + idx[None, :]]

                # Check PSD
                min_eig = np.linalg.eigvalsh(H)[0]
                if min_eig < -1e-10:
                    H = H + (-min_eig + 1e-8) * np.eye(n)

                x = cp.Variable(n)
                objective = cp.Minimize(cp.quad_form(x, cp.psd_wrap(H)))
                constraints = [x >= 0, cp.sum(x) == 1]
                prob = cp.Problem(objective, constraints)

                # Use SCS with reasonable settings
                try:
                    prob.solve(solver=cp.SCS, verbose=False, max_iters=2000)
                except Exception:
                    prob.solve(solver=cp.ECOS, verbose=False)

                if x.value is not None:
                    x_val = np.maximum(x.value, 0)
                    xs = x_val.sum()
                    if xs > 1e-10:
                        x_val = x_val / xs
                        a_new = x_val
            except Exception:
                a_new = None

        if a_new is None:
            # Fallback: use gradient step
            grad = 2.0 * hankel_mv(p, a)
            a_new = project_simplex(a - 0.01 * grad)

        # Blend for stability
        alpha = 0.6 if no_improve == 0 else 0.4
        a_candidate = (1.0 - alpha) * a + alpha * a_new
        a_candidate = np.maximum(a_candidate, 0)
        cs = a_candidate.sum()
        if cs > 1e-10:
            a_candidate = a_candidate / cs
        else:
            a_candidate = a

        obj = fast_convolve(a_candidate).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_candidate.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try without blending
            obj_direct = fast_convolve(a_new).max()
            if obj_direct < best_obj - 1e-14:
                best_obj = obj_direct
                best_a = a_new.copy()
                no_improve = 0

        a = a_candidate if obj <= fast_convolve(a_new).max() else a_new

        # Adapt temperature
        if no_improve > 20:
            T *= 1.3
        if no_improve > 60:
            T *= 0.7
            no_improve = 20  # Reset counter partially
        if no_improve > 150:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            no_improve = 0

    return best_a, best_obj


def coordinate_refine(a0, time_limit):

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())

    # Triangular (peak at center)
    tri = 1.0 - 2.0 * np.abs(x / max(n - 1, 1) - 0.5)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Linearly decreasing
    lin = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(lin / lin.sum())

    # Two-Gaussian (antisymmetric-ish)
    g1 = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (0.15 * n)) ** 2)
    two_g = g1 + g2
    inits.append(two_g / two_g.sum())

    # Cosine bell
    cb = np.cos(np.pi * x / n) ** 2 + 0.05
    inits.append(cb / cb.sum())

    # Interpolated parent
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Random Dirichlet
    for alpha_val in [0.5, 1.0, 2.0]:
        d = rng.dirichlet(np.ones(n) * alpha_val)
        inits.append(d)

    return inits

                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        a_opt, _ = smooth_grad(start, time_limit=t_alloc, T_init=0.01, lr_init=0.01)
        try_sequence(a_opt, f"exp_n{n_try}", state)

    print(f"  Phase 1 best: {state['best_score']:.8f} (elapsed {time.monotonic()-t1_start:.0f}s)")

    # === Phase 2: Deep Optimization (~1000s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t2_start = time.monotonic()
    phase2_budget = 1000

    if state['best_seq'] is not None:
        best = state['best_seq'].copy()
        n_best = len(best)

        # 2a: Long smoothed gradient (low T, small lr)
        remaining = phase2_budget - (time.monotonic() - t2_start)
        if remaining > 60:
            t_alloc = min(remaining, 200)
            print(f"  Deep smooth T=0.002, lr=0.008, t={t_alloc:.0f}s")
            a_opt, _ = smooth_grad(best, time_limit=t_alloc, T_init=0.002, lr_init=0.008, mom_init=0.92)
            try_sequence(a_opt, "deep_smooth", state)

        # 2b: Game iterate
        remaining = phase2_budget - (time.monotonic() - t2_start)
        if remaining > 60:
            t_alloc = min(remaining, 250)
            print(f"  Game iterate T=0.01, t={t_alloc:.0f}s")
            a_opt, _ = game_iterate(best, time_limit=t_alloc, T_init=0.01, use_cvxpy=True)
            try_sequence(a_opt, "game", state)

        # 2c: Peak-targeted gradient
        remaining = phase2_budget - (time.monotonic() - t2_start)
        if remaining > 50:
            t_alloc = min(remaining, 150)
            print(f"  Peak-targeted k=10, t={t_alloc:.0f}s")
            a_opt, _ = peak_targeted_grad(best, time_limit=t_alloc, top_k=10, lr_init=0.002, mom_init=0.94)
            try_sequence(a_opt, "peak_k10", state)

            t_alloc = min(remaining - t_alloc, 100)
            if t_alloc > 20:
                print(f"  Peak-targeted k=30, t={t_alloc:.0f}s")
                a_opt, _ = peak_targeted_grad(best, time_limit=t_alloc, top_k=30, lr_init=0.003, mom_init=0.91)
                try_sequence(a_opt, "peak_k30", state)

        # 2d: Another smoothed gradient with different params
        remaining = phase2_budget - (time.monotonic() - t2_start)
        if remaining > 40:
            t_alloc = min(remaining, 150)
            print(f"  Smooth T=0.005, lr=0.005, mom=0.95, t={t_alloc:.0f}s")
            a_opt, _ = smooth_grad(best, time_limit=t_alloc, T_init=0.005, lr_init=0.005, mom_init=0.95)
            try_sequence(a_opt, "smooth_alt", state)

        # 2e: Game iterate with different T
        remaining = phase2_budget - (time.monotonic() - t2_start)
        if remaining > 40:
            t_alloc = min(remaining, 100)
            print(f"  Game T=0.03, t={t_alloc:.0f}s")
            a_opt, _ = game_iterate(best, time_limit=t_alloc, T_init=0.03, use_cvxpy=True)
            try_sequence(a_opt, "game_T3", state)

    # Try different n values around the best
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        remaining = phase2_budget - (time.monotonic() - t2_start)
        for n_try in [n_best - 64, n_best + 64]:
            remaining = phase2_budget - (time.monotonic() - t2_start)
            if remaining < 30 or n_try < 128 or n_try > 2048:
                continue
            t_alloc = min(remaining, 50)
            start = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), state['best_seq'])
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            a_opt, _ = smooth_grad(start, time_limit=t_alloc, T_init=0.01, lr_init=0.01)
            try_sequence(a_opt, f"adj_n{n_try}", state)

    print(f"  Phase 2 best: {state['best_score']:.8f} (elapsed {time.monotonic()-t2_start:.0f}s)")

    # === Phase 3: Final Polish (~350s) ===
    print("\n=== Phase 3: Final Polish ===")
    t3_start = time.monotonic()
    phase3_budget = 350

    if state['best_seq'] is not None:
        best = state['best_seq'].copy()
        n_best = len(best)

        # 3a: Coordinate refinement
        remaining = phase3_budget - (time.monotonic() - t3_start)
        if remaining > 30:
            t_alloc = min(remaining, 100)
            print(f"  Coord refine, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_refine(best, time_limit=t_alloc)
            try_sequence(a_opt, "coord", state)

        # 3b: Very fine smoothed gradient
        remaining = phase3_budget - (time.monotonic() - t3_start)
        if remaining > 30:
            t_alloc = min(remaining, 100)
            print(f"  Fine smooth T=0.0005, lr=0.002, t={t_alloc:.0f}s")
            a_opt, _ = smooth_grad(best, time_limit=t_alloc, T_init=0.0005, lr_init=0.002, mom_init=0.96)
            try_sequence(a_opt, "fine_smooth", state)

        # 3c: Short game iterate
        remaining = phase3_budget - (time.monotonic() - t3_start)
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  Final game T=0.005, t={t_alloc:.0f}s")
            a_opt, _ = game_iterate(best, time_limit=t_alloc, T_init=0.005, use_cvxpy=True)
            try_sequence(a_opt, "final_game", state)

        # 3d: Multiple tiny perturbations with quick polish
        remaining = phase3_budget - (time.monotonic() - t3_start)
        if remaining > 20:
            for scale in [0.0005, 0.001, 0.0015, 0.002]:
                remaining = phase3_budget - (time.monotonic() - t3_start)
                if remaining < 8:
                    break
                t_alloc = min(remaining, 12)
                f_pert = project_simplex(best + rng.standard_normal(n_best) * scale)
                a_opt, _ = smooth_grad(f_pert, time_limit=t_alloc, T_init=0.001, lr_init=0.003)
                try_sequence(a_opt, f"pert_{int(scale*10000)}", state)

        # 3e: One more coordinate refinement
        remaining = phase3_budget - (time.monotonic() - t3_start)
        if remaining > 15:
            t_alloc = min(remaining, 50)
            print(f"  Final coord, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_refine(best, time_limit=t_alloc)
            try_sequence(a_opt, "final_coord", state)

    print(f"  Phase 3 best: {state['best_score']:.8f} (elapsed {time.monotonic()-t3_start:.0f}s)")

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = parent
        state['best_score'] = float(evaluate_sequence(parent.tolist()))

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs) and fs < state['best_score']:
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
