    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_matvec(p, a):
    """Compute M(p)@a where M(p)_{j,m} = p[j+m]. p has length >= 2n-1."""
    return np.correlate(p, a, mode='valid')


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def logsumexp_smooth_max(conv, T):
    """Compute (1/T)*log(sum exp(T*x)) and the softmax weights."""
    scaled = T * conv
    scaled -= scaled.max()
    e = np.exp(scaled)
    p = e / e.sum()
    val = (scaled.max() + np.log(e.sum())) / T
    return val, p


def smooth_obj_grad(z, T, n, fft_size):
    """Smooth log-sum-exp objective and gradient in z-space (a=softmax(z))."""
    a = softmax(z)
    conv = fast_convolve(a)
    val, p = logsumexp_smooth_max(conv, T)
    obj = 2.0 * n * val
    
    # Gradient: df/da[m] = 2n * (sum_j p[m+j]*a[j] + sum_j a[j]*p[m-j])
    corr = hankel_matvec(p, a)  # sum_j p[m+j]*a[j], length n
    
    # conv_partial[m] = sum_j a[j]*p[m-j] = (a*p)[m], first n elements
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_ap = np.fft.irfft(A * P, fft_size)[:n]
    
    grad_a = 2.0 * n * (corr + conv_ap)
    
    # Chain rule for softmax: grad_z = a * (grad_a - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z


def smooth_lbfgs(a0, time_limit, T_start=0.5, T_end=0.01, n_T=30):

    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

        def fun(z_in):
            return smooth_obj_grad(z_in, T, n, fft_size)
        
        try:

                          options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-12})
            z = res.x
            a_new = softmax(z)
        except Exception:
            break

            print(f"    smooth_lbfgs T={T:.4f}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def game_iterate_nesterov(a0, time_limit, T_init=0.01):
    """
    Nesterov-accelerated game iteration for the min-max problem.
    Primal: minimize a^T M(lambda) a on simplex
    Dual: maximize a^T M(lambda) a over lambda in simplex
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Initialize dual
    conv = fast_convolve(a)
    T_dual = T_init
    scaled = conv / T_dual
    scaled -= scaled.max()
    lambda_ = np.exp(scaled)
    lambda_ = lambda_ / lambda_.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    t = 0
    T_nest = T_init  # Nesterov temperature (for dual softmax)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Primal step: solve min a'^T M(lambda_) a' on simplex
        # Use projected gradient with a few iterations
        M = lambda_  # the "matrix" is the Hankel with this vector
        
        # Nesterov momentum for primal
        if t == 1:
            a_nest = a.copy()
            v = np.zeros(n)
        else:
            beta = (1 + np.sqrt(1 + 4 * (t-1)**2)) / 2
            a_nest = a + (beta - 1) / beta * (a - a_prev)
        
        a_prev = a.copy()
        
        # Compute gradient at a_nest: grad = 2 * M @ a_nest = 2 * hankel_matvec(lambda_, a_nest)
        grad_nest = 2.0 * hankel_matvec(lambda_, a_nest)
        
        # Step size: eta = 1 / L where L is Lipschitz constant
        # L = 2 * ||M||_2 (approximate with 2 * max row sum)
        L = 2.0 * np.abs(lambda_).sum()  # crude bound
        eta = 1.0 / (L + 1e-10)
        
        # Projected gradient step
        a_new = a_nest - eta * grad_nest
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        a = a_new
        
        # Evaluate
        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Dual step: update lambda using softmax of (a^T H_k a) values
        T_dual *= 1.001  # slowly decrease temperature
        scaled = conv / T_dual
        scaled -= scaled.max()
        lambda_new = np.exp(scaled)
        lambda_new = lambda_new / lambda_new.sum()
        
        # Nesterov extrapolation for dual
        if t == 1:
            lambda_nest = lambda_new
        else:
            lambda_nest = lambda_new + (beta - 1) / beta * (lambda_new - lambda_)
        
        lambda_ = lambda_nest
    
    return best_a, 2.0 * n * best_obj

```python
def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_max = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)
    return np.maximum(v - theta, 0.0)
```
