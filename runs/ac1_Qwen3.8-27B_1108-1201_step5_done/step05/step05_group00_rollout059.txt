def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build A_ub sparsely - for each row k, A_ub[k, j] = f[j] if 0 <= k-j < n
    # Row k: sum_j f[k-j]*g[j] <= M for j in [max(0,k-n+1), min(k,n-1)]
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)

                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M, timeout=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle and adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp, timeout=max(5, int(min(15, remaining - 1))))
        if g is None:
            m_exp *= 1.2
            if m_exp > 4.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Fine-tune alpha with golden section
            if best_alpha < 0.99:
                lo = max(0.0, best_alpha * 0.7)
                hi = min(1.0, best_alpha * 1.5)
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    ft = (1.0 - mid) * f + mid * g
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-15:
                        break
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = mid
                        if mid > best_alpha * 1.01 if best_alpha > 0 else True:
                            lo = mid
                        else:
                            hi = mid
                    else:
                        if mid < best_alpha:
                            lo = mid
                        else:
                            hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            scale = 0.002 + 0.004 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def adaptive_smoothed_grad(a0, time_limit, T_init=0.0005, lr_init=0.05):
    """
    Smoothed gradient descent with adaptive temperature based on conv spread.
    Uses momentum for faster convergence.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    momentum = np.zeros(n)
    mom_factor = 0.9
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Adaptive temperature based on spread of conv values
        conv_mean = conv.mean()
        conv_std = conv.std()
        spread_ratio = conv_std / max(conv_mean, 1e-15)
        T_target = T_init * (1.0 + 2.0 * spread_ratio)
        T = 0.95 * T + 0.05 * T_target

        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = 2.0 * hankel_mv(p, a)

        # Momentum update
        momentum = mom_factor * momentum - lr * grad
        candidate = project_simplex(a + momentum)

        # Backtracking line search
        best_candidate = candidate
        best_val = fast_convolve(candidate).max()

        if best_val >= conv.max():
            # Try smaller steps
            for decay in [0.5, 0.25, 0.1]:
                candidate2 = project_simplex(a - decay * lr * grad)
                val2 = fast_convolve(candidate2).max()
                if val2 < best_val:
                    best_val = val2
                    best_candidate = candidate2

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            # Slightly increase lr on improvement
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            lr *= 0.995

        # Adapt
        if no_improve > 30:
            lr *= 0.95
        if no_improve > 80:
            T = T_init * 0.5
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.005 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            momentum = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def multi_peak_subgrad(a0, time_limit, top_k=10, lr_init=0.003, mom_factor=0.93):
    """Multi-peak subgradient with momentum and inverse-rank weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom_factor * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        mom_factor = min(0.98, mom_factor + 0.00005)

        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 300) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def spectral_refine(a0, time_limit):
    """
    Spectral domain refinement: iteratively adjust the power spectrum
    to flatten the autocorrelation, then project back to simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    blend = 0.05

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Current autocorrelation
        conv = fast_convolve(a)
        conv_mean = conv.sum() / len(conv)

        # Desired: flatten toward mean, but keep endpoints (they're constrained)
        # Actually, just reduce the peaks by blending toward the mean
        # in the spectral domain
        
        A = np.fft.rfft(a)
        P = np.abs(A) ** 2

        # Compute current conv spectrum
        P_conv = np.fft.rfft(conv)
        
        # Target: flat conv means P_conv should be concentrated at DC
        # But we can't make it perfectly flat. Instead, reduce the AC components
        # of P_conv (i.e., reduce the "texture" of the power spectrum)
        
        # Alternative: directly attenuate high-magnitude frequency components
        # of P to make conv smoother
        P_mean = P.mean()
        # Keep DC, reduce AC somewhat
        P_smooth = P.copy()
        P_smooth[1:] *= (1.0 - blend)
        # Rescale to preserve total energy
        P_smooth = P_smooth * (P.sum() / max(P_smooth.sum(), 1e-30))

        # Reconstruct with original phase
        phase = A / (np.abs(A) + 1e-300)
        A_new = np.sqrt(P_smooth) * phase
        a_new = np.fft.irfft(A_new, n * 2 - 1)
        a_new = a_new[:n]  # Take first n samples
        a_new = a_new.real
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            # Increase blend slowly
            blend = min(0.3, blend * 1.02)
        else:
            # Try smaller blend
            blend = max(0.01, blend * 0.9)
            # Don't update a, keep current

        if it % 100 == 0:
            # Periodic random restart from best
            rng = np.random.default_rng(it * 997 + 1)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            blend = 0.05

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """
    Coordinate-level optimization: for each element, try small adjustments
    that reduce the maximum convolution value.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Which variables affect this peak?
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        # Try to shift mass from high-contributing variables to low-contributing ones
        contributions = np.zeros(n)
        for j in range(js, je + 1):
            contributions[j] = 2.0 * a[j] * a[k_peak - j] if j != k_peak - j else a[j] ** 2

        # Sort by contribution (descending)
        affected = list(range(js, je + 1))
        if len(affected) < 3:
            continue

        contrib_in_affected = {j: contributions[j] for j in affected}
        sorted_aff = sorted(affected, key=lambda j: contrib_in_affected[j], reverse=True)
        high_vars = sorted_aff[:max(1, len(sorted_aff) // 4)]
        low_vars = sorted_aff[-max(1, len(sorted_aff) // 4):]

        # Try shifting mass from high to low
        improved = False
        for j_high in high_vars:
            for j_low in low_vars:
                for eps in [0.001, 0.0005, 0.0002]:
                    candidate = a.copy()
                    candidate[j_high] -= eps * a[j_high]
                    candidate[j_low] += eps * a[j_high]
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-14:
                        a = candidate
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and it % 100 == 0:
            # Broader random perturbation
            rng = np.random.default_rng(it * 12345 + 678)
            candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-14:
                a = candidate
                best_obj = val
                best_a = a.copy()
            elif it > 500:
                break

    return best_a, best_obj


def resample_sequence(parent, n_target):
    """Resample a sequence to a different length using interpolation."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    resampled = np.interp(x_new, x_orig, parent)
    resampled = np.maximum(resampled, 0.0)
    s = resampled.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return resampled / s
