"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_of_smoothed_max(f, T, top_k):
    """Compute subgradient of T * log(sum(exp(conv/T))) at top peaks."""
    n = len(f)
    conv = fast_convolve(f)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = T * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    w = w / ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg, conv

def nesterov_subgrad(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient on smoothed max objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    T_start = 3.0 / n
    T_end = 0.3 / n
    top_k_start = max(100, 2 * n)
    top_k_end = max(20, n // 4)
    
    lr0 = 0.01
    no_improve = 0
    it = 0
    max_iters = 50000
    
    # Nesterov state
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    beta = 0.9
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / max(1, max_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = lr0 * (1.0 + it * 1e-6) / (1.0 + it * 1e-4)
        
        sg, conv = subgradient_of_smoothed_max(y, T, top_k)
        
        # Nesterov update
        v = beta * v + (1 - beta) * sg
        y_new = project_simplex(y - lr * v)
        
        # Standard Nesterov: y_{k+1} = (1 - alpha)*x_{k+1} + alpha*y_k
        # But with projection, we simplify to just using momentum
        y = y_new
        
        new_M = fast_convolve(y).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            f_pert = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f_pert
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj

def peak_driven_coordinate(f0, time_limit, verbose=False):
    """Targeted coordinate moves based on convolution peak analysis."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            # Restart with perturbation
            rng_restart = np.random.default_rng(iteration * 3 + 1)
            f = project_simplex(best_f + rng_restart.standard_normal(n) * 0.005)
            no_improve = 0
            best_obj = fast_convolve(f).max()
        
        iteration += 1
        conv = fast_convolve(f)
        
        # Find top peaks
        num_peaks = min(10, 2 * n - 1)
        peak_indices = np.argpartition(conv, -num_peaks)[-num_peaks:]
        
        improved_this_iter = False
        
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            if conv[k] < best_obj * 0.95:
                continue  # Only focus on significant peaks
            
            k = int(k)
            # Find contributing pairs for this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Contribution of each pair (j, k-j) to conv[k]
            # We want to reduce the largest contributions
            contributions = f[js:je+1] * f[k-je:k-js+1][::-1]
            
            # Find the pair with the largest contribution
            local_idx = np.argmax(contributions)
            i1 = js + local_idx  # index of first element
            i2 = k - i1  # index of second element (k-j)
            
            if i1 == i2:
                # Single element squared - can't easily reduce without affecting others
                # Try moving some mass from i1 to a random other location
                j = rng.integers(0, n)
                if j == i1:
                    continue
                # Try transferring a fraction
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    f_trial = f.copy()
                    transfer = frac * f[i1]
                    f_trial[i1] -= transfer
                    f_trial[j] += transfer
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-13:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved_this_iter = True
                        break
            else:
                # Two distinct elements contributing
                # Try redistributing between i1 and i2
                c = f[i1] + f[i2]
                current_split = f[i1] / c if c > 0 else 0.5
                
                # Try different splits
                candidates = [current_split, 0.0, 1.0, 0.25, 0.5, 0.75,
                             max(0, current_split - 0.2), min(1, current_split + 0.2)]
                
                best_split = current_split
                best_val = conv[k]  # current value at this peak (proxy)
                
                for t in candidates:
                    t = np.clip(t, 0, 1)
                    f_trial = f.copy()
                    f_trial[i1] = t * c
                    f_trial[i2] = (1 - t) * c
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-13:
                        best_split = t
                        best_val = val
                
                if best_split != current_split:
                    f[i1] = best_split * c
                    f[i2] = (1 - best_split) * c
                    f = f / f.sum()
                    new_obj = fast_convolve(f).max()
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_f = f.copy()
                        improved_this_iter = True
        
        if improved_this_iter:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj

def spectral_shaping(f0, time_limit, verbose=False):
    """Try to flatten the power spectrum by adjusting the sequence."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            break
        
        it += 1
        # Compute current spectrum
        fft_size = 1 << (2 * n - 2).bit_length()
        A = np.fft.rfft(f, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flatten the power spectrum
        # Compute the "excess" at each frequency
        target_power = power.mean()
        
        # Generate a target amplitude
        target_amp = np.sqrt(target_power * np.ones_like(power))
        
        # Gradient in time domain: how to adjust f to flatten the spectrum
        # d(power[k])/d(f[i]) = 2*Re(A[k] * conj(dA[k]/df[i]))
        # A[k] = sum_j f[j] * exp(-2πijk/N)
        # dA[k]/df[i] = exp(-2πik*i/N)
        # So gradient of power w.r.t. f[i] = 2*Re(A[k] * exp(2πik*i/N))
        # Sum over k: gradient[i] = 2*Re(sum_k A[k]*exp(2πik*i/N))
        # = 2*Re(N * ifft(A) wrapped) ... this is getting complex
        
        # Simpler approach: use the fact that if we want |A| to be constant,
        # we can do: A_new = |A_target| * A/|A| = target_amp * phase(A)
        # Then f_new = ifft(A_new) (keeping it real and non-negative)
        
        # This is like one iteration of "spectral flattening"
        phase = A / np.maximum(np.abs(A), 1e-15)
        A_new = target_amp * phase
        f_spectral = np.fft.irfft(A_new, fft_size)[:n]
        f_spectral = np.maximum(f_spectral, 0)
        fs = f_spectral.sum()
        if fs > 1e-15:
            f_spectral = f_spectral / fs
        
        # Blend with current
        for alpha in [0.1, 0.3, 0.5, 0.7]:
            f_trial = (1 - alpha) * f + alpha * f_spectral
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-13:
                f = f_trial
                best_obj = val
                best_f = f.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                no_improve += 1
                if no_improve > 50:
                    break
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
                continue
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj

Let me derive this. Fix all a[k] for k≠i,j. Let c = a[i] + a[j]. Let a[i] = t*c, a[j] = (1-t)*c.

The convolution value at lag k is:
(a*a)[k] = sum_{valid l} a[l]*a[k-l]

The terms involving a[i] or a[j]:
- a[i]*a[k-i] (if k-i is valid and k-i ≠ i,j)
- a[j]*a[k-j] (if k-j is valid and k-j ≠ i,j)
- a[i]*a[j] if k = i+j
- a[i]^2 if k = 2i
- a[j]^2 if k = 2j

So (a*a)[k] = Q_k + t*c * B_k + (1-t)*c * C_k + (t^2 * c^2 if k=2i) + ((1-t)^2 * c^2 if k=2j) + (t*(1-t)*c^2 if k=i+j)

where B_k and C_k are the "background" contributions from other elements.

This is a quadratic in t. The max over k of these quadratics is a piecewise-quadratic function. We can find its minimum by checking the critical points.

This is a much more precise approach than the previous "try a few candidates" method!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient with adaptive temperature schedule
2. Informed pair coordinate descent (peak-guided)
3. LP-based Frank-Wolfe with line search
4. Simulated annealing with structured moves
5. Multi-resolution and multi-init exploration
6. Quadratic pair optimization for final polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, T, top_k):
    """Compute smoothed subgradient of max(conv) using log-sum-exp."""
    n = len(f)
    conv = fast_convolve(f)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = T * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg, conv


def nesterov_optimize(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    T_start = 4.0 / n
    T_mid = 1.0 / n
    T_end = 0.2 / n
    top_k_start = max(200, 2 * n)
    top_k_mid = max(50, n)
    top_k_end = max(10, n // 6)
    lr0 = 0.008
    beta = 0.93
    no_improve = 0
    it = 0
    max_iters = 80000
    y = f.copy()
    v = np.zeros(n)
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, it / max(1, max_iters * 0.9))
        if progress < 0.5:
            T = T_start * (T_mid / T_start) ** (progress / 0.5)
            top_k = int(top_k_start + (top_k_mid - top_k_start) * (progress / 0.5))
            lr = lr0 * (1.0 + 0.3 * progress)
        else:
            T = T_mid * (T_end / T_mid) ** ((progress - 0.5) / 0.5)
            top_k = int(top_k_mid + (top_k_end - top_k_mid) * ((progress - 0.5) / 0.5))
            lr = lr0 * (1.0 + 0.3 * 0.5) * (1.0 - 0.6 * (progress - 0.5))
        top_k = max(5, top_k)
        sg, conv = compute_subgradient(y, T, top_k)
        v = beta * v + (1.0 - beta) * sg
        y_new = project_simplex(y - lr * v)
        y = y_new
        new_M = fast_convolve(y).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = y.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Nesterov iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
        if no_improve > 300:
            rng = np.random.default_rng(it * 17 + 3)
            scale = 0.002 + 0.002 * min(no_improve // 100, 3)
            y = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
    return best_f, best_obj

                            best_t = t
                    if best_val < best_obj - 1e-13:
                        f[i1] = best_t * c
                        f[i2] = (1 - best_t) * c
                        f = f / f.sum()
                        new_obj = fast_convolve(f).max()
                        if new_obj < best_obj:
                            best_obj = new_obj
                            best_f = f.copy()
                            improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break
        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.005, n)
            if g is None:
                no_improve += 1
                if no_improve > 60:
                    break
                continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 23 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                continue
            continue
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 0.05 and best_alpha < 0.95:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            ft1 = np.maximum(f + c1 * (g - f), 0)
            ft1 = ft1 / max(ft1.sum(), 1e-15)
            fc1 = fast_convolve(ft1).max()
            ft2 = np.maximum(f + c2 * (g - f), 0)
            ft2 = ft2 / max(ft2.sum(), 1e-15)
            fc2 = fast_convolve(ft2).max()
            for _ in range(6):
                if hi - lo < 1e-7:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    ft1 = np.maximum(f + c1 * (g - f), 0)
                    ft1 = ft1 / max(ft1.sum(), 1e-15)
                    fc1 = fast_convolve(ft1).max()
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    ft2 = np.maximum(f + c2 * (g - f), 0)
                    ft2 = ft2 / max(ft2.sum(), 1e-15)
                    fc2 = fast_convolve(ft2).max()
            mid = (lo + hi) / 2
            ft = np.maximum(f + mid * (g - f), 0)
            ft = ft / max(ft.sum(), 1e-15)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing with structured moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    cur_obj = best_obj
    t0 = time.monotonic()
    rng = np.random.default_rng(88888)
    T_start = 0.008
    T_end = 0.0005
    step = 0
    est_steps = int(time_limit / 0.004)
    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(1.0, step / max(1, est_steps))
        T = T_start * (T_end / T_start) ** progress
        move_type = int(rng.integers(0, 5))
        f_trial = f.copy()
        if move_type == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.4, 0.4) * (f[i] + f[j])
                f_trial[i] += delta
                f_trial[j] -= delta
        elif move_type == 1:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.3, 0.3) * f[i]
                f_trial[i] -= delta
                f_trial[j] += delta
        elif move_type == 2:
            bs = int(rng.integers(3, min(15, n // 8 + 3)))
            start = int(rng.integers(0, max(1, n - bs)))
            end = min(start + bs, n)
            block_sum = f_trial[start:end].sum()
            if block_sum > 1e-12:
                f_trial[start:end] = np.maximum(
                    f_trial[start:end] + rng.standard_normal(end - start) * block_sum * 0.2, 0)
                bsum = f_trial[start:end].sum()
                if bsum > 1e-12:
                    f_trial[start:end] *= block_sum / bsum
        elif move_type == 3:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                c = f[i] + f[j]
                if c > 1e-12:
                    t = rng.uniform(0, 1)
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
        else:
            f_trial = f + rng.standard_normal(n) * T * 0.05
        f_trial = np.maximum(f_trial, 0)
        fsum = f_trial.sum()
        if fsum < 1e-15:
            continue
        f_trial = f_trial / fsum
        trial_obj = fast_convolve(f_trial).max()
        if trial_obj < cur_obj:
            f = f_trial
            cur_obj = trial_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_f = f.copy()
                if verbose and best_obj < 1.505:
                    print(f"    SA new best: {best_obj:.10f} step={step}")
        else:
            diff = trial_obj - cur_obj
            if diff > 0:
                accept_prob = np.exp(-diff / max(T, 1e-12))
                if rng.random() < accept_prob:
                    f = f_trial
                    cur_obj = trial_obj
    return best_f, best_obj
