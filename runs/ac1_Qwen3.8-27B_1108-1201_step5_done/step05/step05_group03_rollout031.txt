def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_1d_coord(conv, f, i):
    """
    Solve the 1D subproblem for coordinate i exactly.
    Returns optimal value of f[i] (>= 0) minimizing max_k (a*a)[k].
    """
    n = len(f)
    size = 2 * n - 1
    v = f[i]

    # Build A, B vectors
    # A[k] = 1 if k == 2*i, else 0
    # B[k] = 2*f[k-i] if i <= k <= i+n-1 and k != 2*i, else 0
    A = np.zeros(size)
    B = np.zeros(size)

    ki = 2 * i
    if 0 <= ki < size:
        A[ki] = 1.0

    lo = i
    hi = min(2 * n - 2, i + n - 1)
    if lo <= hi:
        B[lo:hi + 1] = 2.0 * f[0:(hi - lo + 1)]

    if 0 <= ki < size:
        B[ki] = 0.0

    # C[k] = conv[k] - A[k]*v^2 - B[k]*v
    C = conv - A * v * v - B * v

    # Evaluate at v=0
    h0 = C.max()
    tol = max(1e-13, 1e-10 * h0)
    active_0 = np.where(C >= h0 - tol)[0]
    if len(active_0) == 0:
        active_0 = np.array([np.argmax(C)])
    deriv0_max = B[active_0].max()

    if deriv0_max >= -1e-14:
        return 0.0

    # Function decreasing at 0, so v* > 0
    lo_v = 0.0
    hi_v = max(v * 2.0, 3.0 / n)

    # Expand upper bound
    for _ in range(30):
        vals = A * hi_v * hi_v + B * hi_v + C
        h_hi = vals.max()
        tol_hi = max(1e-13, 1e-10 * h_hi)
        active_hi = np.where(vals >= h_hi - tol_hi)[0]
        if len(active_hi) == 0:
            active_hi = np.array([np.argmax(vals)])
        derivs_hi = 2.0 * A[active_hi] * hi_v + B[active_hi]
        if derivs_hi.max() >= 0:
            break
        hi_v *= 2.0

    # Bisection
    for _ in range(30):
        mid = (lo_v + hi_v) / 2.0
        vals = A * mid * mid + B * mid + C
        h_mid = vals.max()
        tol_mid = max(1e-13, 1e-10 * h_mid)
        active_mid = np.where(vals >= h_mid - tol_mid)[0]
        if len(active_mid) == 0:
            active_mid = np.array([np.argmax(vals)])
        derivs_mid = 2.0 * A[active_mid] * mid + B[active_mid]

        if derivs_mid.max() > 0:
            hi_v = mid
        else:
            lo_v = mid

    return (lo_v + hi_v) / 2.0

                    if 0.1 < ratio < 0.9:

                best_t = f[i] / max(f[i] + f[j], 1e-15)
                best_val = best_obj
                c = f[i] + f[j]

                for t in candidates:
                    t = np.clip(t, 0.0, 1.0)

def adam_gradient(f0, time_limit, verbose=False):
    """
    Adam optimizer on smoothed max using top-k log-sum-exp.
    Provides a different optimization landscape for escaping CD local minima.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.003

    total_iters = int(time_limit / 0.02)
    it = 0
    no_improve = 0

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max(1, total_iters * 0.8))
        T = (4.0 / n) * (0.2 / 4.0) ** progress
        top_k = max(20, int(n * (1.0 - 0.7 * progress)))
        lr = base_lr * (1.0 - 0.4 * progress)

        conv = fast_convolve(f)
        h = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -h)[-h:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            indices = np.arange(js, je + 1)
            sg[indices] += 2.0 * w[idx] * f[k - indices]

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 200:
                if time.monotonic() - t0 > time_limit * 0.95:
                    break
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
            continue

        f = f_new
        new_M = fast_convolve(f).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0

            if verbose and it % 100 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.12f}, T={T:.6f}")
        else:
            no_improve += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
