def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

    m = conv.max()
    return T * (m / T + np.log(np.exp(conv / T - m / T).sum()))

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_grad(a, conv, T, top_k):
    """Compute gradient of LSE-smoothed max convolution w.r.t. a."""
    n = len(a)
    k_act = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / T
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # grad[lo..hi] += 2 * w[idx] * a[k-hi .. k-lo]
        grad[lo:hi + 1] += 2.0 * w[idx] * a[k - hi:k - lo + 1][::-1]
    return grad


def adam_softmax_optimize(a0, time_limit, verbose=False):
    """Adam on softmax parameterization with geometric T-cooling."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    # Initialize b (log a) for softmax parameterization
    b = np.log(np.maximum(a, 1e-15))
    b = b - b.mean()  # center for numerical stability

    best_obj = fast_convolve(softmax(b)).max()
    best_b = b.copy()
    best_a = softmax(b).copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    T_start = 3.0 / n
    T_end = 0.1 / n
    base_lr = 0.003
    top_k_start = max(50, n // 2)
    top_k_end = max(10, n // 8)

    no_improve = 0
    it = 0
    cooldown = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.4 * progress)

        a = softmax(b)
        conv = fast_convolve(a)
        current_M = conv.max()

        if current_M < best_obj - 1e-14:
            best_obj = current_M
            best_b = b.copy()
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"      Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1

        # Reset if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 131 + 7)
            b = best_b + rng.standard_normal(n) * 0.002
            b = b - b.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            continue

        # Compute gradient
        grad = lse_grad(a, conv, T, top_k)

        # Adam update on b
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        b_new = b - step
        b_new = b_new - b_new.mean()  # re-center for stability

        b = b_new

    return best_a, best_obj

def interpolate_sequence(f, new_n):

def fast_convolve(a):

    n = len(a)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute gradient of LSE-smoothed max convolution w.r.t. a."""
    n = len(a)

    pv = conv[peaks]
    scaled = (pv - pv.max()) / T
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for idx in range(len(peaks)):

        k = int(peaks[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:

        grad[lo:hi + 1] += 2.0 * w[idx] * a[k - hi:k - lo + 1][::-1]

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structured_init(n, kind="cosine"):

            a[n - 1 - i] = f * f

    if ss > 0:
        a /= ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
Adam with LSE smoothing, LP-Frank-Wolfe, coordinate descent,
spectral perturbations, multi-resolution, and diverse structured starts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad(a, conv, T, top_k):
    n = len(a)
    k_act = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / T
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        grad[lo:hi + 1] += 2.0 * w[idx] * a[k - hi:k - lo + 1][::-1]
    return grad


def adam_optimize(a0, time_limit, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    T_s, T_e = 5.0 / n, 0.03 / n
    base_lr = 0.006
    tk_s, tk_e = max(100, n), max(8, n // 6)
    no_improve, it = 0, 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_s * (T_e / T_s) ** progress
        top_k = max(5, int(tk_s + (tk_e - tk_s) * progress))
        lr = base_lr * (1.0 - 0.3 * progress)
        conv = fast_convolve(a)
        cur = conv.max()
        if cur < best_obj - 1e-15:
            best_obj = cur
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:

        else:
            no_improve += 1
        if no_improve > 250:
            rng = np.random.default_rng(it * 131 + 7)
            sc = 0.001 + 0.002 * min(no_improve // 100, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * sc)
            m, v = np.zeros(n), np.zeros(n)
            no_improve = 0
            continue
        grad = lse_grad(a, conv, T, top_k)
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        step = lr * (m / (1 - beta1 ** it1)) / (np.sqrt(v / (1 - beta2 ** it1)) + eps)
        a_new = project_simplex(a - step)
