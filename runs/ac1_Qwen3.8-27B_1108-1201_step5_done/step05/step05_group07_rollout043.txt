import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def conv_subgradient(a, conv, T):
    """Compute smooth gradient of log-sum-exp of convolution peaks."""
    n = len(a)
    Tc = T * conv
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / exp_Tc.sum()
    # grad_conv[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    grad_conv = 2.0 * correlate(w, a, mode='valid')
    return grad_conv, w

def lse_objective(x, n, T):
    """Log-sum-exp smoothed objective for sequence of length n.
    x has length n-1, last element is 1 - sum(x)."""
    a = np.empty(n)
    a[:n-1] = x
    a[n-1] = 1.0 - x.sum()
    if a[n-1] < -1e-10:
        return 1e10, np.zeros(n-1)
    a = np.maximum(a, 0)
    a /= a.sum()
    conv = fast_convolve(a)
    Tc = T * conv
    Tc -= Tc.max()
    f = conv.max() + np.log(np.exp(Tc).sum()) / T
    Tc2 = T * conv
    Tc2 -= Tc2.max()
    w = np.exp(Tc2) / np.exp(Tc2).sum()
    grad_conv = 2.0 * correlate(w, a, mode='valid')
    # da/dx: for i < n-1, da[i]/dx[i] = 1; da[n-1]/dx[i] = -1
    grad = grad_conv[:n-1] - grad_conv[n-1]
    return f, grad

def lbfqsb_optimize(a0, T, time_limit, max_iter=500):
    """Optimize using L-BFGS-B on the smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0, 0)
    a0 /= a0.sum()
    x0 = a0[:n-1].copy()
    
    t0 = time.monotonic()
    result = minimize(
        lse_objective, x0, args=(n, T),
        method='L-BFGS-B',
        jac=True,
        bounds=[(0, 1)] * (n-1),
        options={'maxiter': max_iter, 'maxfun': 2000,
                 'ftol': 1e-14, 'gtol': 1e-10,
                 'disp': False}

    x_opt = np.maximum(result.x, 0)
    a_opt = np.empty(n)
    a_opt[:n-1] = x_opt
    a_opt[n-1] = 1.0 - x_opt.sum()
    a_opt = np.maximum(a_opt, 0)
    a_opt /= a_opt.sum()
    return a_opt

def multiplicative_weights_optimize(a0, time_limit, T_start=100, T_end=10000,
                                     lr=0.01, momentum=0.8, verbose=False):
    """Multiplicative weights with temperature annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a /= a.sum()
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    total_time = time_limit
    while time.monotonic() - t0 < total_time:
        it += 1
        progress = (time.monotonic() - t0) / total_time
        T = T_start * (T_end / T_start) ** progress
        
        conv = fast_convolve(a)
        grad, w = conv_subgradient(a, conv, T)
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        
        # Multiplicative update
        exponent = -lr * v
        exponent -= exponent.max()
        a = a * np.exp(exponent)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-30:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        a /= s
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
            v = np.zeros(n)
            no_improve = 0
        if lr < 1e-6:
            break
        
        if verbose and it % 200 == 0:
            print(f"    MW iter {it}, T={T:.0f}, obj={best_obj:.10f}, lr={lr:.8f}")
    
    return best_a, best_obj

def targeted_local_search(a0, time_limit, verbose=False):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed
objective with L-BFGS-B optimization, multiplicative weights with
temperature annealing, peak-directed local search, and multi-length
exploration.

import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lse_obj_and_grad(x, n, T):
    """Log-sum-exp smoothed objective and gradient.
    x has length n-1 (first n-1 elements of a), a[n-1] = 1 - sum(x)."""
    a = np.empty(n)
    a[:n-1] = x
    a[n-1] = 1.0 - x.sum()
    if a[n-1] < -1e-9:
        # Infeasible; return large value
        return 1e10, np.zeros(n-1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n-1)
    a /= s
    
    conv = fast_convolve(a)
    Tc = T * conv
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    f = conv.max() + np.log(exp_Tc.sum()) / T
    
    w = exp_Tc / exp_Tc.sum()
    grad_conv = 2.0 * correlate(w, a, mode='valid')
    grad = grad_conv[:n-1] - grad_conv[n-1]
    
    return f, grad


def run_lbfqsb(a0, T, max_iter=300, maxfun=1000):
    """Run L-BFGS-B on the smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    x0 = a0[:n-1].copy()
    
    result = minimize(
        lse_obj_and_grad, x0, args=(n, T),
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1.0)] * (n-1),
        options={'maxiter': max_iter, 'maxfun': maxfun,
                 'ftol': 1e-15, 'gtol': 1e-12}
    )
    x_opt = np.maximum(result.x, 0.0)
    a_opt = np.empty(n)
    a_opt[:n-1] = x_opt
    a_opt[n-1] = 1.0 - x_opt.sum()
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s < 1e-15:
        return a0.copy()
    a_opt /= s
    return a_opt

def lse_obj_grad(x, n, T):
    a = np.empty(n)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lse_obj_grad(x, n, T):
    """Log-sum-exp smoothed objective and gradient.
    x has length n-1; a[n-1] = 1 - sum(x)."""
    a = np.empty(n)
    a[:n-1] = x
    a[n-1] = 1.0 - x.sum()
    if a[n-1] < -1e-6:
        pen = 1e4 * (x.sum() - 1.0) ** 2
        return 1e10 + pen, 2.0e4 * (x.sum() - 1.0) * np.ones(n - 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n - 1)
    a /= s
    conv = fast_convolve(a)
    Tc = T * conv
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    f = conv.max() + np.log(exp_Tc.sum()) / T
    w = exp_Tc / exp_Tc.sum()
    grad_conv = 2.0 * correlate(w, a, mode='valid')
    grad = grad_conv[:n-1] - grad_conv[n-1]
    return f, grad


def run_lbfqsb(a0, T, maxiter=200, maxfun=800):
    """Run L-BFGS-B on the log-sum-exp smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-15:

def igt_optimize(a0, time_limit, blend=0.4):
    """Iterative Fourier Transform: flatten spectrum to reduce auto-corr peaks."""

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        A = np.fft.fft(a)
        magnitude = np.abs(A)
        phase = np.angle(A)
        target_power = n * np.sum(a ** 2)
        if target_power > 1.0:
            c = np.sqrt((target_power - 1.0) / (n - 1))
        else:
            c = 0.0
        target_mag = np.full(n, c)
        target_mag[0] = 1.0
        new_mag = blend * target_mag + (1.0 - blend) * magnitude
        new_A = new_mag * np.exp(1j * phase)
        new_a = np.real(np.fft.ifft(new_A))
        new_a = np.maximum(new_a, 0.0)

            no_improve = 0
        else:
            no_improve += 1

def interp_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fn = np.interp(nx, ox, f)
    fn = np.maximum(fn, 0)
    s = fn.sum()
    if s > 1e-15:
        fn /= s
    return fn


def make_inits(n, rng, count=5):
    """Generate diverse initial sequences of length n."""
    inits = []
    inits.append(np.ones(n) / n)
    x = np.linspace(0, 1, n)
    # Increasing
    inc = np.linspace(0.1, 1.0, n)
    inits.append(inc / inc.sum())
    # Decreasing
    dec = np.linspace(1.0, 0.1, n)
    inits.append(dec / dec.sum())
    # Gaussian centered
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 4.0)
    g = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(g / g.sum())
    # Bimodal

    bim = (np.exp(-0.5 * ((np.arange(n) - c1) / (n * 0.1)) ** 2) +

    # Random smoothed

    r = np.convolve(r, kern, mode='same')

    inits.append(r / r.sum())
    return inits[:count]
