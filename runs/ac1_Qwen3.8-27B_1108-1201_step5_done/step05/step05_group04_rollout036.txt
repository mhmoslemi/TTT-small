```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
def fast_optimize(f0, time_limit, mode='auto'):
    """
    Fast optimization using subgradient descent with peak targeting.
    mode: 'explore' (fewer peaks, larger steps), 'exploit' (more peaks, smaller steps)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if mode == 'explore':
        n_peaks = 3
        eta_init = 5.0 / n
        momentum_decay = 0.85
    else:
        n_peaks = 8
        eta_init = 1.0 / n
        momentum_decay = 0.92
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    moment = np.zeros(n)
    eta = eta_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        
        no_improve += 1
        
        if no_improve > 500:
            # Perturb and restart
            rng = np.random.default_rng((it * 7919 + 13) % 1000003)
            scale = 0.005 + 0.01 * min(1.0, (no_improve - 500) / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            moment = np.zeros(n)
            eta = eta_init
            no_improve = 0
            continue
        
        # Find top peaks
        k_peak = min(n_peaks, 2 * n - 1)
        top_k = np.argpartition(c, -k_peak)[-k_peak:]
        
        # Weight: emphasize worst peaks
        peak_vals = c[top_k]
        spread = peak_vals.max() - peak_vals.min()
        if spread < 1e-15:
            w = np.ones(k_peak) / k_peak
        else:
            w = (peak_vals - peak_vals.min() + spread * 0.1) 
            w = w ** 2  # Emphasize the worst
            w /= w.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in range(k_peak):
            ki = int(top_k[idx])
            coeff = 2.0 * w[idx]
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo <= j_hi:
                idx_f = ki - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi+1] += coeff * f[idx_f]
        
        # Momentum update
        moment = momentum_decay * moment - eta * grad
        f_candidate = f + moment
        f_candidate = np.maximum(f_candidate, 0)
        s_cand = f_candidate.sum()
        if s_cand < 1e-15:
            eta *= 0.5
            continue
        f_candidate /= s_cand
        
        # Line search (simple)
        c_cand = fast_convolve(f_candidate)
        if c_cand.max() <= M:
            f = f_candidate
            eta = min(eta * 1.005, 10.0 / n)
        else:
            # Try smaller step
            for scale in [0.5, 0.25, 0.1]:
                f_try = f + scale * moment
                f_try = np.maximum(f_try, 0)
                s_try = f_try.sum()
                if s_try < 1e-15:
                    continue
                f_try /= s_try
                c_try = fast_convolve(f_try)
                if c_try.max() <= M:
                    f = f_try
                    eta *= 0.7
                    break
            else:
                eta *= 0.6
                moment = momentum_decay * moment
    
    return best_f, best_obj
```

Actually, one more thought: I should also try **coordinate descent** or **block coordinate descent**. If I fix all but one coordinate, the problem becomes:
$\min_{f_j} \max_k (f*f)[k]$ given all other $f_i$ ($i \neq j$).

Since $(f*f)[k]$ is linear in $f_j$ for each $k$ (because it's $\sum_i f_i f_{k-i}$ and $f_j$ appears in terms where $i=j$ or $k-i=j$), the problem is:
$\min_{f_j \geq 0} \max_k (a_k + b_k f_j)$

where $a_k = (f*f)[k] - f_j f_{k-j} - f_{j} f_{k-j} + f_{k-j} f_j$ (this is getting confusing).

Let me just do: if I change $f_j$ to $f_j + \delta$, then:

$\min_{\delta: f_j + \delta \geq 0} \max_k ((f*f)[k] + 2\delta f_{k-j})$

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast subgradient descent with peak targeting (O(n log n) per iteration)
2. Multi-length exploration (n = 512 to 2048)
3. Adaptive restarts and momentum
4. Final polishing with simulated annealing
5. Exploits the fact that grad of (f*f)[k] w.r.t. f is 2*f shifted by k
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_gradient(f, top_k, weights):
    """
    Compute subgradient of max(f*f) at the given top peaks.
    grad[j] = sum over active peaks k of: weights[k] * 2 * f[k-j]
    """
    n = len(f)
    grad = np.zeros(n)
    for idx, ki in enumerate(top_k):
        ki = int(ki)
        coeff = 2.0 * weights[idx]
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            idx_f = ki - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += coeff * f[idx_f]
    return grad


def optimize_subgrad(f0, time_limit, n_peaks=5, eta_init=None, mom_decay=0.9, 
                     restart_every=300, verbose=False):
    """
    Fast subgradient descent with momentum and adaptive step size.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if eta_init is None:
        eta = 2.0 / n
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    moment = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > restart_every:
            rng = np.random.default_rng((it * 7919 + 13) % 1000003)
            scale = 0.003 + 0.005 * min(1.0, (no_improve - restart_every) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            moment = np.zeros(n)
            eta = max(eta_init if eta_init else 2.0/n, 1.0/n)
            no_improve = 0
            if verbose and no_improve == 0:
                print(f"    Restart at it={it}, obj={fast_convolve(f).max():.10f}")
            continue
        
        # Find top peaks
        k_peak = min(n_peaks, 2 * n - 1)
        top_k = np.argpartition(c, -k_peak)[-k_peak:]
        
        # Peak weighting
        pv = c[top_k]
        med = np.median(c)
        excess = np.maximum(pv - med, 0)
        if excess.sum() < 1e-15:
            w = np.ones(k_peak) / k_peak
        else:
            w = excess ** 1.5
            w /= w.sum()
        
        # Compute gradient
        grad = compute_peak_gradient(f, top_k, w)
        
        # Momentum update
        moment = mom_decay * moment - eta * grad
        f_new = f + moment
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        f_new /= s_new
        
        # Evaluate
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()
        
        if obj_new <= M + 1e-14:
            f = f_new
            eta = min(eta * 1.003, 8.0 / n)
        else:
            # Try retry with half step
            f_try = f + 0.5 * moment
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try > 1e-15:
                f_try /= s_try
                c_try = fast_convolve(f_try)
                if c_try.max() <= M + 1e-14:
                    f = f_try
                    eta *= 0.8
                else:
                    eta *= 0.7
                    moment = mom_decay * moment
            else:
                eta *= 0.5
    
    return best_f, best_obj


def optimize_anneal(f0, time_limit, n_peaks=3, T_start=0.01, T_end=0.0001):
    """
    Simulated annealing for fine polishing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    cur_obj = best_obj
    
    t0 = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        progress = (time.monotonic() - t0) / total_time
        T = T_start * (T_end / T_start) ** progress
        
        # Perturbation: random perturbation of a few coordinates
        n_pert = max(3, n // 32)
        idx = rng.integers(0, n, size=n_pert)
        delta = rng.standard_normal(n_pert) * T * 0.1
        f_pert = f.copy()
        f_pert[idx] += delta
        f_pert = np.maximum(f_pert, 0)
        s_pert = f_pert.sum()
        if s_pert < 1e-15:
            continue
        f_pert /= s_pert
        
        c_new = fast_convolve(f_pert)
        obj_new = c_new.max()
        delta_obj = obj_new - cur_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / (T * 0.001)):
            f = f_pert
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f.copy()
    
    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
```

Wait, I need to double-check my gradient formula. Let me verify once more.

$(f*f)[k] = \sum_{i=0}^{n-1} f_i \cdot f[k-i]$ where $f[m] = 0$ for $m < 0$ or $m \geq n$.

$\frac{\partial (f*f)[k]}{\partial f_j} = \sum_i [\delta_{ij} f[k-i] + f_i \delta_{k-i,j}]$

First term: $\delta_{ij}$ means $i = j$, giving $f[k-j]$.
Second term: $\delta_{k-i,j}$ means $k-i = j$, i.e., $i = k-j$, giving $f[k-j]$.

So $\frac{\partial (f*f)[k]}{\partial f_j} = f[k-j] + f[k-j] = 2f[k-j]$.

(With the convention that $f[m] = 0$ for $m$ out of range.)

This is correct. So the subgradient of $\max_k (f*f)[k]$ (when peak is at $k^*$) is the vector with $j$-th component $2f[k^* - j]$.

Now let me also think about whether my code handles the edge cases correctly.

For peak at lag $k$:
- $f[k-j]$ is non-zero when $0 \leq k-j \leq n-1$
- This means $k-n+1 \leq j \leq k$
- Also $0 \leq j \leq n-1$
- So $j \in [\max(0, k-n+1), \min(n-1, k)]$

This is what I have in the code. Good.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast subgradient descent exploiting the structure that grad of (f*f)[k] is 2*f shifted by k
2. Multi-length exploration (n = 512 to 2048) with adaptive parameters
3. Peak-equality weighting to flatten the auto-convolution
4. Simulated annealing for final polishing
5. Periodic restarts to escape local minima
Key advantage: O(n log n) per iteration (FFT-dominated), enabling 10^5+ iterations in seconds.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_gradient(f, top_k, weights):
    """
    Compute subgradient of max_k (f*f)[k] at the given active peaks.
    For peak at lag k: d/dj (f*f)[k] = 2*f[k-j] (with f[m]=0 for m out of range).
    """
    n = len(f)
    grad = np.zeros(n)
    for idx in range(len(top_k)):
        ki = int(top_k[idx])
        coeff = 2.0 * weights[idx]
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            idx_f = ki - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += coeff * f[idx_f]
    return grad


def optimize_subgrad(f0, time_limit, n_peaks=5, eta_init=None, mom_decay=0.90,
                     restart_every=300, peak_power=1.5, verbose=False):
    """
    Fast subgradient descent with momentum and adaptive step size.
    Targets the top n_peaks of the auto-convolution with power-weighted emphasis.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if eta_init is None:
        eta = 2.0 / n
    else:
        eta = eta_init

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    t0 = time.monotonic()
    moment = np.zeros(n)
    no_improve = 0
    it = 0
    eta_max = 10.0 / n

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()

        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > restart_every:
            seed_val = (it * 7919 + 13) % 1000003
            rng = np.random.default_rng(seed_val)
            scale = 0.003 + 0.008 * min(1.0, (no_improve - restart_every) / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            moment = np.zeros(n)
            eta = max(eta_init if eta_init else 2.0 / n, 0.5 / n)
            no_improve = 0
            c = fast_convolve(f)
            if verbose:
                print(f"    Restart it={it}, obj={c.max():.10f}")
            continue

        # Find top peaks
        k_peak = min(n_peaks, 2 * n - 1)
        top_k = np.argpartition(c, -k_peak)[-k_peak:]

        # Peak weighting: emphasize peaks that are highest
        pv = c[top_k]
        med = np.median(c)
        excess = np.maximum(pv - med, 0)
        if excess.sum() < 1e-15:
            w = np.ones(k_peak) / k_peak
        else:
            w = excess ** peak_power
            w /= w.sum()

        # Compute gradient
        grad = compute_peak_gradient(f, top_k, w)

        # Momentum update
        moment = mom_decay * moment - eta * grad
        f_new = f + moment
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            moment *= 0.5
            continue
        f_new /= s_new

        # Line search: try full step, then half, then quarter
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()

        if obj_new <= M + 1e-14:
            f = f_new
            eta = min(eta * 1.005, eta_max)
        else:
            # Try half step
            f_half = f + 0.5 * moment
            f_half = np.maximum(f_half, 0)
            s_half = f_half.sum()
            accepted = False
            if s_half > 1e-15:
                f_half /= s_half
                c_half = fast_convolve(f_half)
                if c_half.max() <= M + 1e-14:
                    f = f_half
                    eta *= 0.85
                    accepted = True

            if not accepted:
                # Try quarter step
                f_qtr = f + 0.25 * moment
                f_qtr = np.maximum(f_qtr, 0)
                s_qtr = f_qtr.sum()
                if s_qtr > 1e-15:
                    f_qtr /= s_qtr
                    c_qtr = fast_convolve(f_qtr)
                    if c_qtr.max() <= M + 1e-14:
                        f = f_qtr
                        eta *= 0.75
                        accepted = True

            if not accepted:
                eta *= 0.6
                moment = mom_decay * moment

    return best_f, best_obj


def optimize_peak_equal(f0, time_limit, n_peaks=6, eta_init=None, mom_decay=0.92):
    """
    Specialized optimizer that tries to equalize the top peaks.
    Uses uniform weighting across top peaks to push them all down together.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if eta_init is None:
        eta = 1.5 / n
    else:
        eta = eta_init

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    t0 = time.monotonic()
    moment = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()

        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            rng = np.random.default_rng((it * 104729 + 7) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            moment = np.zeros(n)
            eta = max(eta * 0.8, 0.3 / n)
            no_improve = 0
            continue

        # Uniform weighting on top peaks (to equalize them)
        k_peak = min(n_peaks, 2 * n - 1)
        top_k = np.argpartition(c, -k_peak)[-k_peak:]
        w = np.ones(k_peak) / k_peak

        grad = compute_peak_gradient(f, top_k, w)
        moment = mom_decay * moment - eta * grad
        f_new = f + moment
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            continue
        f_new /= s_new

        c_new = fast_convolve(f_new)
        if c_new.max() <= M + 1e-14:
            f = f_new
            eta = min(eta * 1.003, 5.0 / n)
        else:
            f_half = f + 0.5 * moment
            f_half = np.maximum(f_half, 0)
            s_h = f_half.sum()
            if s_h > 1e-15:
                f_half /= s_h
                if fast_convolve(f_half).max() <= M + 1e-14:
                    f = f_half
                    eta *= 0.85
                else:
                    eta *= 0.7
                    moment = mom_decay * moment
            else:
                eta *= 0.5

    return best_f, best_obj


def optimize_anneal(f0, time_limit, T_start=0.005, T_end=0.00005, n_pert_frac=0.05):
    """
    Simulated annealing for fine polishing.
    Makes small random perturbations and accepts based on energy change.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    total_time = max(time_limit, 1.0)
    rng = np.random.default_rng(54321)
    n_pert = max(3, int(n * n_pert_frac))

    while time.monotonic() - t0 < time_limit:
        progress = (time.monotonic() - t0) / total_time
        T = T_start * (T_end / T_start) ** progress

        # Random perturbation of a subset of coordinates
        idx = rng.integers(0, n, size=n_pert)
        delta = rng.standard_normal(n_pert) * T
        f_pert = f.copy()
        f_pert[idx] += delta
        f_pert = np.maximum(f_pert, 0)
        s_pert = f_pert.sum()
        if s_pert < 1e-15:
            continue
        f_pert /= s_pert

        c_new = fast_convolve(f_pert)
        obj_new = c_new.max()
        delta_obj = obj_new - cur_obj

        if delta_obj < 0 or (delta_obj < 0.01 and rng.random() < np.exp(-delta_obj / (T * 0.0005))):
            f = f_pert
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f.copy()

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):

        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score
```
