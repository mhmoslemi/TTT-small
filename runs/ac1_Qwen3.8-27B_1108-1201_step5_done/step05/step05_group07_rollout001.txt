def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_v2(a, conv, top_k, temperature, flatness_weight=0.0):

    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    if temperature > 0:
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
    else:
        w = np.zeros(len(peaks))
        w[0] = 1.0
    
    sg = np.zeros(n)

            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

"""
Hybrid LP best-response and subgradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. LP-based best-response direction (more global than pure subgradient)
2. Adaptive step size with backtracking line search
3. Combined LP + subgradient hybrid with adaptive switching
4. Multi-length exploration with interpolation
5. Direct peak-equalization local search for final polishing
6. Simulated-annealing-style restarts with diverse perturbations
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective_normalized(f):
    """Compute max(conv(f,f)) for normalized f (sum=1)."""
    conv = fast_convolve(f)
    return conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def solve_lp_best_response(f, n):
    """
    Solve: min M s.t. (f*b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (M*, b*) or (None, None) on failure.
    """
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = f[j]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x[-1], res.x[:n]
    except Exception:
        pass
    return None, None


def lp_best_response_optimize(f0, time_limit, verbose=False, label=""):
    """
    LP best-response optimization with adaptive step size.
    At each iteration, solve the LP to find the best direction b,
    then take a step in that direction with backtracking line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_t = 0.3  # initial step size

    while time.monotonic() - t0 < time_limit:
        # Check time
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        # Solve LP
        M_star, b_star = solve_lp_best_response(f, n)
        if b_star is None:
            # LP failed, use subgradient as fallback
            conv_cur = fast_convolve(f)
            sg = compute_subgradient(f, conv_cur, min(30, n), 0.00005)
            b_star = project_simplex(np.ones(n) - 0.01 * sg)
            cur_t *= 0.5

        b_star = np.maximum(b_star, 0)
        bs = b_star.sum()
        if bs < 1e-15:
            b_star = np.ones(n) / n
        else:
            b_star /= bs

        # Backtracking line search
        found_step = False
        t_try = cur_t
        for _ in range(8):
            f_new = (1 - t_try) * f + t_try * b_star
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs < 1e-15:
                t_try *= 0.5
                continue
            f_new /= fs

            obj_new = compute_objective_normalized(f_new)
            if obj_new < best_obj - 1e-14:
                f = f_new
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                found_step = True
                cur_t = min(t_try * 1.1, 0.5)  # increase step slightly
                break
            else:
                t_try *= 0.5
                if t_try < 0.001:
                    break

        if not found_step:
            no_improve += 1
            cur_t *= 0.7
            if cur_t < 0.005:
                cur_t = 0.005

        # Restart if stuck
        if no_improve > 15:
            rng = np.random.default_rng((it * 104729 + 7) % 1000000)
            noise_scale = 0.005 + 0.003 * min(no_improve // 30, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            no_improve = 0
            cur_t = 0.2

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-BR iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, step={cur_t:.4f}")

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, top_k=50, temperature=0.00005,
                         lr=0.003, momentum=0.85, verbose=False, label=""):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        sg = compute_subgradient(f_nest, conv_nest, cur_top_k, cur_temp)

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.008
        if it % 500 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.95))

        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.002 + 0.001 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj

def peak_equalization_search(f0, time_limit, verbose=False):
    """
    Direct peak-equalization: identify the top peaks and try to
    reduce them by adjusting contributing coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        cur_max = conv.max()

        # Get top peaks
        k_peaks = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_vals = conv[peaks]
        # Only consider peaks that are "close" to the max
        threshold = cur_max * 0.95
        critical_peaks = peaks[peak_vals > threshold]

        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)

        for k in critical_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Find the coordinate with highest contribution
            local_max_idx = np.argmax(contribs)
            hi_coord = idx[local_max_idx]

            # Find a coordinate to transfer mass TO (one that doesn't contribute to this peak)
            # Random selection from non-critical coordinates
            for attempt in range(10):
                lo_coord = rng.integers(0, n)
                if lo_coord == hi_coord:
                    continue
                # Check if lo_coord contributes to any critical peak
                if f[lo_coord] < 1e-12:
                    continue

                for frac in [0.05, 0.1, 0.2, 0.3]:
                    amt = f[hi_coord] * frac
                    if f[hi_coord] - amt < 0:
                        continue
                    f[hi_coord] -= amt
                    f[lo_coord] += amt
                    val = compute_objective_normalized(f)
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi_coord] += amt
                        f[lo_coord] -= amt
                        break

        # Also try: for the global max peak, try redistributing among its contributors
        max_peak = int(np.argmax(conv))
        js = max(0, max_peak - n + 1)
        je = min(n - 1, max_peak)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[max_peak - idx]

        # Try equalizing the contributions
        target = contribs.mean()
        for i in range(len(idx)):
            if time.monotonic() - t0 > time_limit:
                break
            diff = target - contribs[i]
            if abs(diff) < 1e-12:
                continue
            # Adjust: if contrib is too high, reduce; if too low, increase
            coord_i = idx[i]
            partner = max_peak - coord_i
            if diff < 0:
                # Too high, reduce coord_i slightly
                for frac in [0.02, 0.05, 0.1]:
                    amt = f[coord_i] * frac
                    if amt > f[coord_i]:
                        continue
                    # Transfer to partner
                    f[coord_i] -= amt
                    f[partner] += amt
                    val = compute_objective_normalized(f)
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[coord_i] += amt
                        f[partner] -= amt
            # Recompute for next iteration
            contribs = f[idx] * f[max_peak - idx]

        if not improved:
            # Try random pair exchanges
            for _ in range(200):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[i] * frac
                    f[i] -= amt
                    f[j] += amt
                    val = compute_objective_normalized(f)
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[i] += amt
                        f[j] -= amt
                        break
            if not improved:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=4):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 100)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    return inits[:count]

        print("No parent found, using uniform n=768")

    # === Phase 1: Quick subgradient on parent and inits ===
    print("\n=== Phase 1: Quick Subgradient Initialization ===")
    candidates = [parent]
    for n_try in [512, 768, 1024]:
        candidates.extend(structured_inits(n_try, rng, count=2))

    for idx, cand in enumerate(candidates[:6]):
        rem = deadline - time.monotonic() - 800
        if rem < 15:
            break
        n_cand = len(cand)
        t_alloc = min(rem * 0.15, 60)
        if t_alloc < 5:
            t_alloc = 5
        print(f"  SG init {idx}: n={n_cand}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(cand, time_limit=t_alloc,
                                         top_k=max(20, n_cand // 8),
                                         temperature=0.00004, lr=0.003,
                                         momentum=0.85)
        try_seq(a_opt, f"sg_init_{idx}")

    # === Phase 2: LP Best-Response (main workhorse) ===
    print(f"\n=== Phase 2: LP Best-Response (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.6, 500)
            n_best = len(best_seq)
            print(f"  LP-BR: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lp_best_response_optimize(best_seq, time_limit=t_alloc,
                                                  verbose=True)
            try_seq(a_opt, "lp_br")

    # === Phase 3: Hybrid LP + Subgradient ===
    print(f"\n=== Phase 3: Hybrid Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.5, 400)
            n_best = len(best_seq)
            print(f"  Hybrid: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = hybrid_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "hybrid")

            # Second hybrid run with perturbation
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                noise = rng.standard_normal(n_best) * 0.005
                f_pert = project_simplex(best_seq + noise)
                t_alloc2 = min(rem * 0.5, 200)
                print(f"  Hybrid perturbed: t={t_alloc2:.0f}s")
                a_opt, _ = hybrid_optimize(f_pert, time_limit=t_alloc2)
                try_seq(a_opt, "hybrid_pert")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [640, 800, 896, 1024, 1280]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 100)
            if t_alloc < 10:
                continue
            print(f"  Length n={target_n}, t={t_alloc:.0f}s")
            # Use hybrid for length exploration
            a_opt, _ = hybrid_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 5: Peak equalization + local search ===
    print(f"\n=== Phase 5: Peak Equalization (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.5, 150)
            print(f"  Peak-eq: t={t_alloc:.0f}s")
            a_opt, _ = peak_equalization_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_eq")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_alloc = min(rem, 50)
            print(f"  Fine SG: t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc,
                                             top_k=n_best, temperature=0.000001,
                                             lr=0.0003, momentum=0.95)
            try_seq(a_opt, "fine_sg")

        # Peak equalization polish
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Peak-eq polish: t={t_alloc:.0f}s")
            a_opt, _ = peak_equalization_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_eq_polish")

        # Multiple perturbation attempts
        for attempt in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            for scale in [0.001, 0.003, 0.006]:
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(rem, 15)
                a_opt, _ = subgradient_optimize(f_pert, time_limit=t_alloc,
                                                 top_k=max(20, n_best // 5),
                                                 temperature=0.00002,
                                                 lr=0.001, momentum=0.92)
                try_seq(a_opt, f"pert_{scale}_{attempt}")

        # Final LP-BR attempt
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 20)
            print(f"  Final LP-BR: t={t_alloc:.0f}s")
            a_opt, _ = lp_best_response_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "lp_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
