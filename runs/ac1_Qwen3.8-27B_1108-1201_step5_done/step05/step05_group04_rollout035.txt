    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_hankel_sparse(n, active_set, lam):
    """Build sparse Hankel matrix B[i,j] = lam[m] if i+j = active_set[m]."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_set):
        lk = lam[idx]
        if abs(lk) < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                vals.append(lk)
    if not rows:
        return csr_matrix((n, n))
    return csr_matrix((vals, (rows, cols)), shape=(n, n))


def solve_kkt_system(n, active_set, lam):
    """Solve B(a) = mu*1, sum(a)=1 for given active set and lambda."""
    B = build_hankel_sparse(n, active_set, lam)
    try:
        v = spsolve(B.tocsc(), np.ones(n))
        if not np.all(np.isfinite(v)):
            return None
        denom = v.sum()
        if abs(denom) < 1e-12:
            return None
        a = v / denom
        return a
    except Exception:
        return None


def active_set_newton(n, active_set, max_iters=60, tol=1e-11):
    """
    Solve the KKT active-set problem using damped Newton on lambda.
    Returns (best_a, best_obj) where a minimizes max(c) with the given active set equalized.
    """
    p = len(active_set)
    if p == 1:
        a = np.ones(n) / n
        return a, fast_convolve(a).max()

    lam = np.ones(p) / p
    best_a = None
    best_obj = np.inf

    for it in range(max_iters):
        a = solve_kkt_system(n, active_set, lam)
        if a is None:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-14:
            a = np.ones(n) / n
        else:
            a /= a.sum()

        c = fast_convolve(a)
        obj = c.max()
        if best_a is None or obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Check if active peaks are equalized
        peak_vals = np.array([c[k] for k in active_set])
        spread = peak_vals.max() - peak_vals.min()
        if spread < tol:
            break

        # F vector: differences from last active peak
        F = np.array([c[active_set[m]] - c[active_set[p - 1]] for m in range(p - 1)])
        F_norm = np.linalg.norm(F)
        if F_norm < 1e-14:
            break

        # Numerical Jacobian
        eps = 1e-7
        J = np.zeros((p - 1, p - 1))
        for j in range(p - 1):
            lam_p = lam.copy()
            lam_p[j] += eps
            lam_p[p - 1] -= eps
            a_p = solve_kkt_system(n, active_set, lam_p)
            if a_p is None:
                J[:, j] = np.sign(F) * 1e-4
                continue
            a_p = np.maximum(a_p, 0.0)
            if a_p.sum() < 1e-14:
                a_p = np.ones(n) / n
            else:
                a_p /= a_p.sum()
            c_p = fast_convolve(a_p)
            F_p = np.array([c_p[active_set[m]] - c_p[active_set[p - 1]] for m in range(p - 1)])
            J[:, j] = (F_p - F) / eps

        # Damped Newton step
        try:
            step = np.linalg.lstsq(J, -F, rcond=None)[0]
        except Exception:
            step = -F * 0.01

        alpha = 1.0
        lam_new = lam.copy()
        improved = False
        for _ in range(25):
            trial = lam + alpha * step
            trial = np.maximum(trial, 0.0)
            trial_sum = trial.sum()
            if trial_sum < 1e-12:
                trial = np.ones(p) / p
            else:
                trial /= trial_sum

            a_trial = solve_kkt_system(n, active_set, trial)
            if a_trial is not None:
                a_trial = np.maximum(a_trial, 0.0)
                if a_trial.sum() > 1e-14:
                    a_trial /= a_trial.sum()
                    c_trial = fast_convolve(a_trial)
                    F_trial = np.array([c_trial[active_set[m]] - c_trial[active_set[p - 1]]
                                        for m in range(p - 1)])
                    if np.linalg.norm(F_trial) < F_norm:
                        lam_new = trial
                        improved = True
                        break
            alpha *= 0.5

        if not improved:
            # Small perturbation to escape
            rng = np.random.default_rng(it * 7919 + 13)
            lam = lam + rng.standard_normal(p) * 0.02
            lam = np.maximum(lam, 0.0)
            lam /= lam.sum()
        else:
            lam = lam_new

        if np.max(np.abs(lam - best_a is not None and 0 or 0)) < 1e-15:
            break

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = fast_convolve(best_a).max()
    return best_a, best_obj


def compute_grad_weighted(f, w, max_peaks=300):
    """Compute gradient of sum_k w[k] * (f*f)[k] w.r.t. f using top peaks."""
    n = len(f)
    conv_len = 2 * n - 1
    if len(w) <= max_peaks:
        sig_idx = np.argsort(w)[::-1]
    else:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        sig_idx = sig_idx[w[sig_idx] > 1e-15 * w.max() if w.max() > 0 else True]

    grad = np.zeros(n)
    for k in sig_idx:
        wk = w[k]
        if wk < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += wk * f[k - ms]
        ms2_lo = max(0, k - hi)
        ms2_hi = min(n - 1, k - lo)
        if ms2_lo <= ms2_hi:
            ms2 = np.arange(ms2_lo, ms2_hi + 1)
            grad[ms2] += wk * f[k - ms2]
    return grad


def lse_optimize(f0, time_limit, beta_start=50.0, beta_end=20000.0, max_peaks=300):
    """Log-sum-exp smoothed gradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.5 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_grad_weighted(f, w, max_peaks=max_peaks)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 6:
                lr *= 0.94
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 50:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(1.5 / n, 0.001)
                beta = min(beta * 1.5, beta_end)
                momentum = 0.90
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj
