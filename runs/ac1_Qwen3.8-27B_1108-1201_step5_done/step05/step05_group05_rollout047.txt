"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
Nesterov-accelerated smoothed subgradient for fast coarse optimization,
LP-Frank-Wolfe with constraint generation for accurate direction finding,
multi-length exploration, diverse structured restarts, and coordinate descent
polishing. Key improvements over prior art: constraint-generation LP for
nearly full-constraint accuracy at reduced cost, adaptive temperature
scheduling, Nesterov momentum for the subgradient phase, and systematic
multi-length search.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, n, top_k, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    k_act = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    scaled = pv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def nesterov_subgrad_optimize(f0, time_limit, tau0=0.005, lr0=0.003,
                               top_k=60, decay=0.9997):
    """Nesterov-accelerated smoothed subgradient optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    y = f.copy()
    conv_y = fast_convolve(y)
    best_obj = conv_y.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    tau = tau0
    lr = lr0
    m = 0.9
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, n, top_k, tau)

        y_new = project_simplex(y - lr * sg)
        theta = (1.0 + m) / 2.0
        f_new = project_simplex(y_new + (y_new - y) * (m - 1) / (1.0 + m))
        f_new = project_simplex((1.0 - theta) * f + theta * y_new +
                                (theta - 1.0) * (y - y_new))
        # Simpler Nesterov: y_{k+1} = x_{k+1} + (m_{k+1}-1)/(m_k) (x_{k+1} - x_k)
        # Use simpler momentum form:
        v = m * v - lr * sg
        f_new = project_simplex(f + v)

        y_new = f_new
        conv_new = fast_convolve(y_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        y = y_new
        tau *= decay
        lr = lr0 * (0.9999 ** it)
        lr = max(lr, 0.0001)

        if no_improve > 500:
            rng = np.random.default_rng(it * 104729 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            y = f.copy()
            tau = tau0 * (1.0 + 0.5 * (no_improve // 500))
            no_improve = 0

    return best_f, best_obj


def solve_lp_constraints(f, M, n, active_k):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_k, g >= 0."""
    c = -np.ones(n)
    K = len(active_k)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_step(f, n, top_k=60, use_generation=True, max_gen_iter=50):
    """One LP-FW step with optional constraint generation.
    Returns (g_normalized, success)."""
    conv = fast_convolve(f)
    M = conv.max()

    if use_generation:
        # Start with top-k constraints
        active_k = np.argsort(conv)[::-1][:top_k]
        g = None
        for gen_iter in range(max_gen_iter):
            g = solve_lp_constraints(f, M, n, active_k)
            if g is None:
                return None, False
            # Check all constraints
            fg = cross_convolve(f, g)
            violations = np.where(fg > M + 1e-9)[0]
            if len(violations) == 0:
                break
            # Add violated constraints
            new_k = violations[:30]
            active_k = np.unique(np.concatenate([active_k, new_k]))
            if len(active_k) > 400:
                # Cap constraints, keep worst violators + original top
                worst = np.argsort(fg)[::-1][:top_k]
                active_k = worst
        if g is None:
            return None, False
    else:
        active_k = np.argsort(conv)[::-1][:top_k]
        g = solve_lp_constraints(f, M, n, active_k)
        if g is None:
            return None, False

    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, False
    g = g / g_sum
    return g, True


def line_search_alpha(f, g, n, max_trials=20):
    """Find best mixing alpha via trials (faster than golden section for this use)."""
    best_alpha = 0.0
    best_obj = fast_convolve(f).max()
    alphas = np.concatenate([
        np.linspace(0, 1, max_trials),
        [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 0.99]
    ])
    alphas = np.unique(np.clip(alphas, 0, 1))
    for alpha in alphas:
        if alpha == 0:
            continue
        trial = project_simplex(f + alpha * (g - f))
        obj = fast_convolve(trial).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_alpha = alpha
    return best_alpha, best_obj


def lpfw_optimize(f0, time_limit, top_k=60, verbose=False,
                  use_generation=True, gen_iter=40):
    """LP-Frank-Wolfe optimization with constraint generation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g, success = lpfw_step(f, n, top_k=top_k,
                               use_generation=use_generation,
                               max_gen_iter=gen_iter)
        if not success:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue

        alpha, alpha_obj = line_search_alpha(f, g, n, max_trials=15)

        if alpha > 1e-8:
            f_new = project_simplex(f + alpha * (g - f))
            conv = fast_convolve(f_new)
            M_new = conv.max()

            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                m_exp = 1.0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.005 * min(no_improve // 100, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            m_exp = 1.0
            no_improve = 0
            if no_improve > 200:
                break

    return best_f, best_obj


def coord_desc_polish(f0, time_limit, n_mults=8):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    mults = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]
    if n_mults < len(mults):
        mults = mults[:n_mults]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in mults:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n) / n - 0.5
    sig = np.exp(-0.5 * (x / sigma_frac) ** 2)
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    return sig / ss


def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def diverse_restart(f_best, n, rng, mode):
    """Generate a diverse restart from the best solution."""
    if mode == 0:
        return project_simplex(f_best + rng.standard_normal(n) * 0.003)
    elif mode == 1:
        return project_simplex(0.7 * f_best + 0.3 * f_best[::-1])
    elif mode == 2:
        return project_simplex(0.8 * f_best + 0.2 * np.ones(n) / n)
    elif mode == 3:
        return project_simplex(f_best + rng.standard_normal(n) * 0.01)
    elif mode == 4:
        pal = make_palindromic(f_best, n)
        return project_simplex(0.5 * f_best + 0.5 * pal)
    elif mode == 5:
        return project_simplex(0.6 * f_best + 0.4 * gaussian_init(n))
    elif mode == 6:
        # Block perturbation
        f_new = f_best.copy()
        num_blocks = 4
        block_size = n // num_blocks
        for b in range(num_blocks):
            scale = rng.uniform(0.5, 2.0)
            f_new[b * block_size:(b + 1) * block_size] *= scale
        return project_simplex(f_new)
    elif mode == 7:
        # Sparse perturbation
        f_new = f_best.copy()
        mask = rng.random(n) < 0.3
        f_new[mask] *= rng.uniform(0.3, 1.5, mask.sum())
        return project_simplex(f_new)
    else:
        return project_simplex(f_best + rng.standard_normal(n) * 0.005)

                    try_seq(f_sg, f"restart_sg_{i}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [512, 640, 896, 1024, 1280]
        # Filter out lengths too close to current best
        n_best = len(best_seq)
        target_lengths = [tl for tl in target_lengths if abs(tl - n_best) > 64]
        # Also add n_best itself for re-optimization at different init
        rem = deadline - time.monotonic()
        if target_lengths:
            t_per_len = min(rem * 0.18 / len(target_lengths), 150)
            for tl in target_lengths:
                rem = deadline - time.monotonic()
                if rem < t_per_len + 10:
                    break
                print(f"  Length {tl}, t={t_per_len:.0f}s")
                # Interpolate best to target length
                f_interp = interpolate_sequence(best_seq, tl)
                # Mix with flat spectrum for diversity
                fs = flat_spectrum_init(tl, seed=tl)
                f_mix = project_simplex(0.75 * f_interp + 0.25 * fs)
                f_opt, _ = lpfw_optimize(f_mix, time_limit=min(t_per_len, rem - 10),
                                          top_k=40, use_generation=True, gen_iter=20)
                try_seq(f_opt, f"len_{tl}")

    # === Phase 5: Coordinate descent + final polish ===
    print(f"\n=== Phase 5: CD + Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_cd = min(rem * 0.3, 300)
            print(f"  CD t={t_cd:.0f}s")
            f_opt, _ = coord_desc_polish(best_seq, time_limit=t_cd, n_mults=8)
            try_seq(f_opt, "cd")

            # Second CD pass
            rem = deadline - time.monotonic()
            if rem > 30:
                t_cd2 = min(rem * 0.2, 150)
                f_opt2, _ = coord_desc_polish(f_opt, time_limit=t_cd2, n_mults=6)
                try_seq(f_opt2, "cd2")

    # === Phase 6: Final LP-FW on best with full constraint set ===
    print(f"\n=== Phase 6: Final LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            t_final = min(rem * 0.35, 200)
            print(f"  Final LP-FW t={t_final:.0f}s")
            f_opt, _ = lpfw_optimize(best_seq, time_limit=t_final, top_k=80,
                                      use_generation=True, gen_iter=50)
            try_seq(f_opt, "final_lpfw")

            # Tiny Nesterov polish
            rem = deadline - time.monotonic()
            if rem > 15:
                t_np = min(rem * 0.3, 60)
                f_np, _ = nesterov_subgrad_optimize(best_seq, time_limit=t_np,
                                                     tau0=0.001, lr0=0.0008,
                                                     top_k=40, decay=0.999)
                try_seq(f_np, "final_nesterov")

        # Final perturbations
        rem = deadline - time.monotonic()
        for i in range(3):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise_scale = 0.0005 + 0.001 * i
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_pert = min(rem * 0.2, 40)
            f_opt, _ = lpfw_optimize(f_pert, time_limit=t_pert, top_k=40,
                                      use_generation=True, gen_iter=20)
            try_seq(f_opt, f"final_perturb_{i}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
