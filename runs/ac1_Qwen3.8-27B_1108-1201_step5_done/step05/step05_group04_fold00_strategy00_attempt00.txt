def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    # --- Security Checks ---
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    
    # Convert to float and ensure nonnegative
    try:
        seq = np.array([float(x) for x in sequence])
    except:
        return np.inf
    
    # Reject booleans and NaN, infinity
    if np.any(np.isin(seq, [0.0, 1.0, -1.0, True, False]) | np.isnan(seq) | np.isinf(seq)):
        return np.inf
    
    # Clip negatives and huge values
    seq = np.maximum(seq, 0.0)
    seq = np.minimum(seq, 1000.0)
    
    # Reject low sum
    s = seq.sum()
    if s < 0.01:
        return np.inf
    
    # Compute normalized convolution
    conv = np.convolve(seq, seq, mode='full')
    max_val = np.max(conv)
    
    # Objective calculation
    n = len(seq)
    return float(2 * n * max_val / s**2)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    cond = (u * np.arange(1, n+1) > cssv)
    rho = np.nonzero(cond)[0][-1]
    theta = (np.cumsum(u)[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-correlation via FFT."""

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]
