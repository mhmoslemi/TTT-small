def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with proper security checks.
    Returns float('inf') if the input is invalid."""
    # Security Checks
    if not isinstance(sequence, list):

        if isinstance(x, bool) or not isinstance(x, (int, float)) or np.isnan(x):
            return np.inf
    min_val = min(sequence) if sequence else 0 # Avoid min(empty) error

        # In case the list cannot be summed due to element types
        return np.inf

def fast_convolve(seq):
    """Linear auto-convolution via FFT for a normalized/trimmed sequence."""
    seq_np = np.array(seq)
    if len(seq_np) == 0:
        raise ValueError("Attempting to convolve an empty sequence.")
    
    # Add zeros to even length for symmetry in convolution
    n = len(seq_np)
    size = 2*n - 1 if n >= 2 else min(n, 1)
    if n <= 1:
        size = 1
        if np.all(seq_np) >= 0:
            return np.array([seq_np[0]*seq_np[0]] if n>=1 else [0.0])
        else:
            raise ValueError("All elements in the sequence are non-positive.")
            
    # Use FFT with proper padding to avoid circular convolution
    pad_len = n * 2  # Safety padding
    while pad_len < n + n - 1:
        pad_len *= 2
    
    # Ensure no negative values in the sequence due to security checks
    seq_np = np.maximum(seq_np, 0.0)
    padded_seq = np.zeros(pad_len, dtype=np.float64)
    padded_seq[:n] = seq_np
    
    # Compute auto-convolution in four steps
    if np.sqrt(pad_len) < (1 << (size.bit_length())):
        fft_size = 1 << (size-1).bit_length()  # Proper power of two
    else:
        fft_size = 1 << (np.log(pad_len)/np.log(2) + 1)

    padded_fft = np.fft.rfft(padded_seq, fft_size)
    conv_fft = padded_fft * padded_fft
    conv = np.fft.irfft(conv_fft, fft_size)[:size]
    
    return conv


def project_simplex(v, tol=1e-15):
    """Euclidean projection on the probability simplex, returning normalized vector."""
    n = len(v)
    if n == 0:
        raise ValueError(" projecting empty vector.")
    s = np.sum(v)
    max_peak = np.max(v) if np.nonzero(v).size > 0 else 0
    if s < tol and max_peak < tol:
        raise ValueError("Sum and max below tolerance.")
    if s >= 1 and max_peak >= 1:
        return np.minimum(v / s, 1.0) if s >= max_peak else np.ones(n)/n
    sorted_indices = np.argsort(v)
    sorted_v = np.copy(v)
    sorted_v.sort()
    tmp = np.cumsum(sorted_v) / np.arange(1, n+1)
    t, s_sorted = 0.0, sorted_v
    for i in range(n-1, -1, -1):
        if s_sorted[i] - t <= 0:
            s_sorted = s_sorted[:i+1]
            break
        t += s_sorted[i]
    if len(s_sorted) == 0:
        return np.ones(n)/n
    u = s_sorted / t
    v = np.zeros(len(u), dtype=np.float64)
    v[:len(u)] = np.minimum(v + u * u.max() / (tmp * tmp).max(), 1.0)
    if np.sqrt(np.sum((v[:len(u)] - u)**2)) < tol:
        v = sorted_indices.argsort().argsort()[:len(v)]
        v = np.clip(v * 1000, 0, 1000)
        v / np.sum(v) = project_simplex(u)
    return v * (np.sqrt(len(v)) / np.sum(v))
