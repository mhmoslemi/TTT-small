"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with 
Nesterov acceleration, adaptive M targeting, multi-length exploration, 
and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix A_ub where A_ub[k,j] = f[k-j] if valid
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (f*g)[k] includes f[k-j]*g[j] for valid k-j
        # k-j is valid when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
        lo = j
        hi = j + n - 1
        A_ub[lo:hi+1, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):

        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    # Nesterov momentum
    f_nest = f.copy()
    momentum = 0.9
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    M_target_factor = 1.0  # Start with current M
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Solve LP at the Nesterov point
        g = solve_lp_direction(f_nest, M, n)
        if g is None:
            # Try with slightly relaxed M
            g = solve_lp_direction(f_nest, M * 1.01, n)
            if g is None:
                g = solve_lp_direction(f, M, n)
                if g is None:
                    g = solve_lp_direction(f, M * 1.02, n)
                    if g is None:
                        no_improve += 1
                        if no_improve > 100:

                            conv = fast_convolve(f)

def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Polish via cyclic coordinate descent with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve_passes >= 5:
            break

        improved = False
        # Cyclic order (or random every few passes)
        order = rng.permutation(n) if no_improve_passes == 0 else np.arange(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[i] < 1e-12:
                # Try to increase f[i] by taking from neighbor
                j = (i + 1) % n
                if f[j] < 1e-12:
                    continue
                c = f[j]
                for frac in [0.1, 0.25, 0.5, 0.75]:
                    f_trial = f.copy()
                    f_trial[i] = frac * c
                    f_trial[j] = (1 - frac) * c
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-15:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                continue

            # For each i, try moving mass to other positions
            # Try a few target positions
            targets = [(i+1)%n, (i-1+n)%n, (i+n//2)%n, rng.integers(n)]
            for j in targets:
                if i == j:
                    continue
                c = f[i] + f[j]
                if c < 1e-12:
                    continue
                # Current fraction
                t0_frac = f[i] / c
                candidates = [t0_frac, 0.0, 1.0, 0.25, 0.5, 0.75]
                if 0 < t0_frac < 1:
                    candidates.extend([max(0, t0_frac-0.1), min(1, t0_frac+0.1)])

                best_t = t0_frac
                best_val = best_obj

                for t in candidates:
                    t = np.clip(t, 0.0, 1.0)
                    if abs(t - t0_frac) < 1e-6:
                        continue
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-15:
                        best_val = val
                        best_t = t

                if best_val < best_obj - 1e-15:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True

        if improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1

    return best_f, best_obj

    momentum = 0.7
    vel = np.zeros(n)
    no_improve = 0

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with Nesterov momentum and adaptive M targeting
2. Subgradient descent with top-k peak weighting
3. Multi-length exploration with interpolation
4. Pairwise coordinate descent polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = j + n - 1
        A_ub[lo:hi + 1, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0)
    return None

def subgradient_optimize(f0, time_limit, verbose=False):
    """Subgradient descent with top-k peaks and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    eta = 0.008
    eta_min = 1e-7
    momentum = 0.7
    vel = np.zeros(n)
    no_improve = 0
    it = 0
    top_k = max(10, n // 10)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        conv = fast_convolve(f)
        M = conv.max()
        progress = min(1.0, it / max(1000, int(time_limit * 50)))

        # Top-k peaks with decaying temperature
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        T = 0.1 + 0.4 * (1 - progress)
        scaled = (pv - pv.max()) / T
        w = np.exp(scaled)
        w = w / w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Momentum + step
        vel = momentum * vel + (1 - momentum) * sg
        f_new = project_simplex(f - eta * vel)

        new_M = fast_convolve(f_new).max()
        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}")
        else:
            no_improve += 1
        f = f_new

        # Adaptive step size
        if no_improve > 300:
            eta *= 0.85
            if eta < eta_min:
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                vel = np.zeros(n)
                eta = 0.004
                no_improve = 0

    return best_f, best_obj

                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-15:
                        best_val = val
                        best_t = t

                if best_val < best_obj - 1e-15:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True
                    if verbose:
                        pass

        if improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def update_best(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence and update best if improved."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    if a.size == 0 or a.sum() < 1e-10:
        return best_seq, best_score
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW with Nesterov, subgradient, coordinate descent,
    multi-length exploration, and diverse restarts.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    # Gather initial candidates
    candidates = []

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            candidates.append(p_norm)
            print(f"Parent: n={len(p_norm)}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    # Structured initializations
    for n_init in [512, 768, 1024]:
        rng = np.random.default_rng(seed + n_init)
        a = rng.dirichlet(np.ones(n_init) * 0.3)
        candidates.append(a)

    # Bimodal
    n_bim = 768
    x = np.linspace(0, 1, n_bim)
    a_bim = np.exp(-((x - 0.3) ** 2) / 0.008) + np.exp(-((x - 0.7) ** 2) / 0.008)
    a_bim = a_bim / a_bim.sum()
    candidates.append(a_bim)

    # Evaluate all
    for i, c in enumerate(candidates):
        best_seq, best_score = update_best(c, best_seq, best_score, f"init_{i}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nInitial best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 1: LP-FW on best ===
    print("\n=== Phase 1: LP-FW ===")
    rem = deadline - time.monotonic() - 700
    t1 = min(rem, 450)
    if t1 > 30:
        print(f"  LP-FW: n={len(best_seq)}, time={t1:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t1, verbose=True)
        best_seq, best_score = update_best(a_opt, best_seq, best_score, "lpfw_1")

    # === Phase 2: Subgradient ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    t2 = min(rem, 200)
    if t2 > 30:
        print(f"  SG: n={len(best_seq)}, time={t2:.0f}s")
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t2, verbose=True)
        best_seq, best_score = update_best(a_opt, best_seq, best_score, "sg_2")

    # === Phase 3: LP-FW round 2 ===
    print(f"\n=== Phase 3: LP-FW round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    t3 = min(rem, 300)
    if t3 > 30:
        print(f"  LP-FW: n={len(best_seq)}, time={t3:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
        best_seq, best_score = update_best(a_opt, best_seq, best_score, "lpfw_3")

    # === Phase 4: Multi-length ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq
    base_n = len(base)
    for target_n in [512, 640, 896, 1024, 1280]:
        if target_n == base_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.3, 100)
        if t_alloc > 15:
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
            best_seq, best_score = update_best(a_opt, best_seq, best_score, f"lpfw_n{target_n}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 120
    if rem > 20:
        t_cd = min(rem * 0.6, 180)
        print(f"  CD: n={len(best_seq)}, time={t_cd:.0f}s")
        a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd, verbose=True)
        best_seq, best_score = update_best(a_opt, best_seq, best_score, "cd_5")

        # LP-FW after CD
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_fw = min(rem, 100)
            a_opt, _ = lp_fw_optimize(a_opt, time_limit=t_fw, verbose=False)
            best_seq, best_score = update_best(a_opt, best_seq, best_score, "lpfw_after_cd")

    # === Phase 6: Final diverse restarts ===
    print(f"\n=== Phase 6: Final (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for trial in range(8):
        rem = deadline - time.monotonic() - 8
        if rem < 15:
            break

        if trial == 0:
            f_init = best_seq.copy()
        elif trial == 1:
            f_init = best_seq[::-1].copy()
            f_init = f_init / f_init.sum()
        elif trial == 2:
            f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
            f_init = project_simplex(f_init)
        else:
            rng = np.random.default_rng(2000 + trial * 13)
            f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)

        # Alternate LP-FW and CD
        t_alloc = min(rem * 0.7, 80)
        if trial % 2 == 0:
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False)
        else:
            a_opt, _ = coordinate_descent_polish(f_init, time_limit=t_alloc)
        best_seq, best_score = update_best(a_opt, best_seq, best_score, f"final_{trial}")

        # Quick follow-up
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = lp_fw_optimize(a_opt, time_limit=min(rem, 30))
            best_seq, best_score = update_best(a_opt, best_seq, best_score, f"final2_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
