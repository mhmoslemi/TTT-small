def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    return np.concatenate([a, a[-2::-1]])

                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                remaining = 1.0 - new_val
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = remaining / denom
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj

def sg_polish(f0, time_limit, lr=0.001, mom=0.93, top_k=40, temp=0.00005):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9999
        cur_temp *= 1.003
        
        if no_improve > 300:
            rng = np.random.default_rng(it_count * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0
    
    return best_f, best_obj

def skewed_init(n, alpha=0.3, seed=0):
    """Create a skewed initial sequence (non-palindromic)."""
    rng = np.random.default_rng(seed)
    # Power law with skew
    t = np.linspace(0, 1, n)
    a = (t + 0.1) ** alpha * ((1-t) + 0.1) ** (alpha * 0.5)
    a = a + rng.standard_normal(n) * 0.005
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

def geometric_init(n, ratio=0.985, seed=0):
    """Geometric sequence init."""
    a = ratio ** np.arange(n, dtype=np.float64)
    a = a / a.sum()
    return a
