def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build dense matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - hi:k - lo + 1]
    return A


def lp_solve_direction(f, M, max_iter_lp=5000):
    """
    Solve the LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Returns normalized g (sum=1) or None if infeasible/unsuccessful.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'maxiter': max_iter_lp, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            sg = g.sum()
            if sg > 1e-10:
                return g / sg
    except Exception:
        pass
    return None


def lp_boosting_direction(f, M, n_iter=200, lr=0.05):
    """
    Approximate LP solution using dual boosting (multiplicative weights).
    Faster than exact LP solve.
    """
    n = len(f)
    m = 2 * n - 1
    y = np.ones(m) / m
    for _ in range(n_iter):
        # Compute c[j] = sum_k y[k] * f[k-j] for all j
        # This is A^T @ y where A[k,j] = f[k-j]
        c = np.zeros(n)
        for k in range(m):
            if y[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            c[lo:hi + 1] += y[k] * f[k - hi:k - lo + 1]
        # Find most violated constraint
        j_min = np.argmin(c)
        if c[j_min] >= 0.999:
            break
        # Update y: multiply by exp(lr * f[k - j_min])
        update = np.zeros(m)
        for k in range(m):
            val = k - j_min
            if 0 <= val < n:
                update[k] = f[val]
        y *= np.exp(lr * update)
        y /= y.sum()
    # Recover primal from dual (approximate)
    g = np.zeros(n)
    for k in range(m):
        if y[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += y[k] * f[k - hi:k - lo + 1]
    g = np.maximum(g, 0)
    sg = g.sum()
    if sg > 1e-10:
        return g / sg
    return None


def compute_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg
