def fast_grad(f, w):
    """Compute gradient of sum_k w[k] * (f*f)[k] wrt f using FFT.
    grad[m] = 2 * sum_{j=0}^{n-1} f[j] * w[m+j]
    This equals 2 * (conv(f_rev, w))[m + n - 1]
    """
    n = len(f)
    f_rev = f[::-1]
    # convolve(f_rev, w) has length n + len(w) - 1 = n + (2n-1) - 1 = 3n-2
    # We want elements at indices n-1, n, ..., 2n-2 (which are indices m+n-1 for m=0..n-1)
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.fft(f_rev, fft_size)
    W = np.fft.fft(w, fft_size)
    conv = np.fft.ifft(F * W).real
    return 2.0 * conv[n-1 : 2*n-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LSE optimization,
Nesterov acceleration, adaptive beta scheduling, and multi-length exploration.

import numpy as np
import time

def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_grad(f, w):
    """Compute gradient of sum_k w[k]*(f*f)[k] wrt f using FFT.
    grad[m] = 2 * sum_{j=0}^{n-1} f[j] * w[m+j]
    Using: grad = 2 * ifft(fft(f_rev) * fft(w))[n-1 : 2*n-1]
    """
    n = len(f)
    f_rev = f[::-1]
    conv_len = n + len(w) - 1  # = n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(F * W, fft_size)
    return 2.0 * conv[n-1 : 2*n-1]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT-accelerated gradient computation (exact, 30x faster than loop-based)
- LSE-smoothed optimization with momentum and adaptive step size
- Aggressive beta scheduling for progressive peak sharpening
- Multi-length exploration with interpolation
- Direct subgradient polishing at the argmax
- Random restarts for escaping local minima

import numpy as np
import time


def fast_conv(a):
    """Full linear auto-convolution via rFFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_grad(f, w):
    """
    Compute gradient of sum_k w[k]*(f*f)[k] with respect to f.
    grad[m] = 2 * sum_{j=0}^{n-1} f[j] * w[m+j]
    Uses FFT: grad = 2 * irfft(rfft(f_rev) * rfft(w))[n-1 : 2*n-1]
    """
    n = len(f)
    f_rev = f[::-1]
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    result = np.fft.irfft(F * W, fft_size)
    return 2.0 * result[n-1:2*n-1]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def lse_optimize(f0, time_limit, beta_start=30.0, beta_end=10000.0, lr_init=None,
                 momentum_init=0.90, verbose=False):
    """
    LSE-smoothed optimization with momentum and adaptive step size.
    Uses exact FFT-based gradient.
    """
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)
    if lr_init is None:
        lr = 4.0 / n
    else:
        lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    beta = beta_start

    conv = fast_conv(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv(f)

        # LSE weights (numerically stable)
        bc = beta * conv
        bc -= bc.max()
        exp_bc = np.exp(bc)
        w = exp_bc / exp_bc.sum()

        # Exact gradient via FFT
        grad = fast_grad(f, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        # Evaluate actual max objective
        conv_new = fast_conv(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 12.0 / n)
        else:
            no_improve += 1
            if no_improve == 20:
                lr *= 0.85
            elif no_improve == 50:
                lr *= 0.75
                momentum = max(0.82, momentum - 0.03)
            elif no_improve == 100:
                lr *= 0.6
                momentum = max(0.80, momentum - 0.05)
            elif no_improve > 150:
                rng = np.random.default_rng((it * 104729 + 7) % 999983)
                scale = 0.001 + 0.005 * min(1.0, (no_improve - 150) / 150.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)

                lr = max(3.0 / n, 0.001)

def subgrad_polish(f0, time_limit, verbose=False):
    """
    Direct subgradient descent at the argmax of convolution.
   Targets the single highest peak with a decaying step size.
    """
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)

    conv = fast_conv(f)
    best_obj = conv.max()
    best_f = f.copy()

    momentum = 0.85
    v = np.zeros(n)

    no_improve = 0

        it += 1
        conv = fast_conv(f)
        k_max = int(np.argmax(conv))

        # Subgradient of c_kmax(f): grad[m] = 2*f[k_max - m] if 0 <= k_max-m < n
        m_lo = max(0, k_max - n + 1)
        m_hi = min(n - 1, k_max)
        if m_lo > m_hi:
            break
        ms = np.arange(m_lo, m_hi + 1)

        grad[ms] = 2.0 * f[k_max - ms]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_conv(f_new)
        obj_new = conv_new.max()

def multi_peak_subgrad(f0, time_limit, n_peaks=5, verbose=False):
    """
    Subgradient targeting the top n_peaks peaks with exponential weighting.
    """
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv(f)

        # Select top peaks

        w_full[peaks] = w_peaks

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        grad /= gnorm

        f_new = project_simplex(f + v)

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def periodic_init(n, freq=3.0, amplitude=0.08):
    """Create a periodic perturbation of uniform for diverse initialization."""
    base = np.ones(n) / n
    x = np.arange(n)
    pert = np.cos(2.0 * np.pi * freq * x / n) * amplitude
    pert += np.cos(2.0 * np.pi * 2 * freq * x / n) * amplitude * 0.3
    f = base + pert
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f


def random_init(n, seed=0):
    """Random non-negative sequence normalized to sum 1."""
    rng = np.random.default_rng(seed)
    f = rng.exponential(1.0, size=n)
    f /= f.sum()
    return f
