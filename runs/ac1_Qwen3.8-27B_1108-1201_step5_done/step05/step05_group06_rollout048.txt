Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast subgradient descent with
exact subgradients, periodic coordinate bisection, monotone initializations,
and multi-length exploration.

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, k_star):
    """Compute subgradient of max(c) at peak k_star. ∂c[k_star]/∂a[j] = 2*a[k_star-j]"""
    n = len(a)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    if js <= je:
        indices = np.arange(js, je + 1)
        sg[indices] = 2.0 * a[k_star - indices]
    return sg

def fast_subgrad(a0, time_limit, lr_scale=0.01, mom=0.9, coord_every=200, n_coord=10):

    n = len(a)

    no_improve = 0

def optimize_coord(a, j):
    """Optimize a single coordinate a[j] via bisection, keeping sum=1."""
    n = len(a)
    x_old = a[j]
    if x_old < 1e-14:
        return a
    
    s_others = 1.0 - x_old
    if s_others < 1e-12:
        return a
    
    # b = a with a[j]=0, normalized
    b = a.copy()
    b[j] = 0.0
    b = b / s_others
    
    B = fast_convolve(b)
    conv_len = 2 * n - 1
    
    # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
    b_valid = np.zeros(conv_len)
    start = j
    end = min(j + n, conv_len)
    b_valid[start:end] = b[:end - start]
    
    lo = 0.0
    hi = 1.0
    x_new = x_old
    
    for _ in range(35):
        mid = (lo + hi) / 2.0
        om = 1.0 - mid
        f_vals = om * om * B + 2.0 * mid * om * b_valid
        if 2 * j < conv_len:
            f_vals[2 * j] += mid * mid
        k_max = int(np.argmax(f_vals))
        
        k = k_max
        b_kj = b[k - j] if 0 <= k - j < n else 0.0
        B_k = B[k]
        ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
        
        deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
        
        if deriv > 1e-13:
            hi = mid
            x_new = mid
        elif deriv < -1e-13:
            lo = mid
            x_new = mid
        else:
            x_new = mid
            break
    
    if abs(x_new - x_old) < 1e-14:
        return a
    
    a[j] = x_new
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    a[mask] *= (1.0 - x_new) / s_others
    a = np.maximum(a, 0.0)
    s_check = a.sum()
    if s_check > 1e-15:
        a /= s_check
    return a

def make_ramp(n, direction=1, p=1.0):
    """Create a monotone ramp sequence."""
    if direction == 1:
        x = np.arange(1, n + 1, dtype=np.float64)
    else:
        x = np.arange(n, 0, -1, dtype=np.float64)
    x = x ** p
    x = x / x.sum()
    return x

def make_exp_ramp(n, alpha=0.5, direction=1):
    """Create an exponential ramp."""
    x = np.arange(n, dtype=np.float64)
    if direction == 1:
        x = np.exp(alpha * x / n)
    else:
        x = np.exp(-alpha * x / n)
    x = x / x.sum()
    return x

def make_cosine_bump(n, center=0.3, width=0.4):
    """Create a cosine bump."""
    x = np.arange(n, dtype=np.float64) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - center) / width))
    a = np.clip(a, 0, None)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a.sum()
    return a

def symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
