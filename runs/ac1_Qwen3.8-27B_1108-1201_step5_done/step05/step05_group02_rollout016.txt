"""

Uses: subgradient descent with momentum, simulated annealing with directed moves,
peak-targeting coordinate descent, multi-length exploration, and multi-start
from diverse structured initializations. Key insight: the problem is non-convex
(individual quadratic forms a^T M_k a have indefinite Hessian), so we use
stochastic methods to escape local minima.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=1, temperature=0.0):
    """Compute (smoothed) subgradient of max_k (a*a)[k] using FFT-accelerated cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    k_actual = min(top_k, size)
    
    if temperature > 1e-12:
        idx = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[idx]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        w_full = np.zeros(size)
        w_full[idx] = w
    else:
        peaks = np.where(conv >= conv.max() - 1e-12)[0]
        w_full = np.zeros(size)
        w_full[peaks] = 1.0 / len(peaks)
    
    # Gradient: 2 * sum_k w_k * a[k-j] for each j
    # This is 2 * cross_correlation(w_full, a) restricted to [0, n-1]
    # cross[j] = sum_k w_full[k] * a[k-j] (with a[i]=0 for i outside [0,n-1])
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    
    return 2.0 * cross


def sg_momentum(a0, time_limit, lr=0.003, momentum=0.93, top_k=50, temp=0.0001,
                verbose=False, seed=42):
    """Subgradient descent with momentum and adaptive parameters."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        sg = compute_subgradient(f, conv, top_k=top_k, temperature=cur_temp)
        
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        new_conv = fast_convolve(f_new)
        obj = new_conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            conv = new_conv
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: max_conv={best_obj:.12f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1
            f = f_new
        
        # Adaptive parameters
        cur_lr *= 0.9997
        cur_temp *= 1.006
        
        if no_improve > 300:
            # Restart with perturbation
            scale = 0.001 + 0.002 * min(no_improve // 200, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            cur_temp = temp * 0.5
            no_improve = 0
            
        if cur_lr < 0.0002:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
            v = np.zeros(n)
            cur_lr = lr * 0.6
            cur_temp = temp * 0.3
            no_improve = 0
    
    return best_f, best_obj


def simulated_anneal(a0, time_limit, verbose=False, seed=123):
    """Simulated annealing with directed and random moves."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = conv.max()
    
    t0 = time.monotonic()
    total_time = time_limit
    initial_temp = 0.002
    final_temp = 0.00001
    rng = np.random.default_rng(seed)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min((time.monotonic() - t0) / total_time, 1.0)
        temp = initial_temp * (final_temp / initial_temp) ** progress
        
        # Choose a move type
        move_type = rng.random()
        
        f_trial = f.copy()
        
        if move_type < 0.3:
            # Directed perturbation: reduce elements contributing to peak
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            n_peaks = je - js + 1
            # Reduce random subset of contributing elements
            n_reduce = max(1, n_peaks // 3)
            indices = rng.choice(range(js, je + 1), size=min(n_reduce, n_peaks), replace=False)
            for j in indices:
                f_trial[j] *= (1.0 - rng.uniform(0.05, 0.3))
        elif move_type < 0.5:
            # Single coordinate change
            j = int(rng.integers(0, n))
            f_trial[j] *= (1.0 + rng.uniform(-0.5, 0.8))
        elif move_type < 0.65:
            # Block modification
            block_len = min(n // 8, int(rng.integers(3, max(4, n // 16))))
            start = int(rng.integers(0, n - block_len))
            direction = rng.choice([-1, 1])
            amount = rng.uniform(0.05, 0.4)
            f_trial[start:start + block_len] *= (1.0 + direction * amount * rng.uniform(0.5, 1.0, block_len))
        elif move_type < 0.8:
            # Swap two blocks
            block_len = min(n // 10, int(rng.integers(2, max(3, n // 20))))
            i1 = int(rng.integers(0, n - block_len))
            i2 = int(rng.integers(0, n - block_len))
            if abs(i1 - i2) > block_len:
                f_trial[i1:i1+block_len], f_trial[i2:i2+block_len] = f_trial[i2:i2+block_len].copy(), f_trial[i1:i1+block_len].copy()
        elif move_type < 0.9:
            # Gaussian smoothing on a region
            region_len = min(n // 4, int(rng.integers(10, max(11, n // 6))))
            start = int(rng.integers(0, max(1, n - region_len)))
            kernel = np.exp(-0.5 * (np.arange(region_len) - region_len / 2) ** 2 / (region_len / 6) ** 2)
            kernel /= kernel.sum()
            f_trial[start:start + region_len] = np.convolve(f_trial[start:start + region_len], kernel, 'same')
        else:
            # Random perturbation
            f_trial += rng.standard_normal(n) * temp * 0.5
        
        f_trial = np.maximum(f_trial, 0.0)
        ts = f_trial.sum()
        if ts < 1e-15:
            continue
        f_trial /= ts
        
        trial_conv = fast_convolve(f_trial)
        trial_obj = trial_conv.max()
        
        # Acceptance criterion
        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            f = f_trial
            cur_obj = trial_obj
            conv = trial_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_f = f.copy()
                if verbose and int((time.monotonic() - t0) * 10) % 50 == 0:
                    print(f"    SA iter {it}: obj={best_obj:.12f}, temp={temp:.6f}")
    
    return best_f, best_obj


def peak_coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent targeting coordinates that contribute to the current peaks."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        conv = fast_convolve(f)
        
        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-14:
                    continue
                
                # Try scaling this coordinate
                best_j_val = old_val
                best_j_obj = best_obj
                
                for mult in [0.3, 0.5, 0.7, 0.85, 0.95, 1.1, 1.3, 1.6]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    if f_trial.sum() < 1e-15:
                        continue
                    f_trial /= f_trial.sum()
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val
                
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj


def structured_initializations(n, seed=0):
    """Generate diverse structured initial sequences of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp (increasing)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Linear ramp (decreasing)
    inits.append(ramp[::-1] / ramp.sum())
    
    # Exponential decay
    exp_dec = np.exp(-3.0 * np.arange(n) / n)
    inits.append(exp_dec / exp_dec.sum())
    
    # Two bumps
    bump = np.zeros(n)
    c1 = int(n * 0.25)
    c2 = int(n * 0.75)
    sigma = n * 0.08
    for i in range(n):
        bump[i] = np.exp(-0.5 * (i - c1) ** 2 / sigma ** 2) + np.exp(-0.5 * (i - c2) ** 2 / sigma ** 2)
    inits.append(bump / bump.sum())
    
    # Sine-modulated
    x = np.linspace(0, 1, n)
    sine = 0.5 + 0.5 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    inits.append(sine / sine.sum())
    
    # Cosine window
    cos_win = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(cos_win / cos_win.sum())
    
    # Random sparse (50% non-zero)
    sparse = rng.random(n)
    sparse[rng.random(n) < 0.5] = 0
    if sparse.sum() > 0:
        inits.append(sparse / sparse.sum())
    
    # Random smooth
    smooth = np.cumsum(rng.standard_normal(n))
    smooth = np.maximum(smooth - smooth.min(), 0.01)
    inits.append(smooth / smooth.sum())
    
    # Triangular (peak at 1/3)
    tri = np.zeros(n)
    peak_pos = n // 3
    for i in range(n):
        tri[i] = max(0, 1 - abs(i - peak_pos) / (n * 0.4))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")
    
    base_n = len(parent)
    
    # === Phase 1: Subgradient from parent ===
    print(f"\n=== Phase 1: SG from parent (n={base_n}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(200, rem * 0.2)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = sg_momentum(parent, time_limit=t1, lr=0.004, momentum=0.93,
                               top_k=60, temp=0.00008, verbose=True, seed=seed)
        try_seq(a_opt, "sg_parent")
    
    # Second SG pass with different params
    rem = deadline - time.monotonic()
    t1b = min(150, rem * 0.15)
    if t1b > 20 and best_seq is not None:
        print(f"  SG pass 2: {t1b:.0f}s")
        a_opt, _ = sg_momentum(best_seq, time_limit=t1b, lr=0.002, momentum=0.94,
                               top_k=40, temp=0.00004, verbose=False, seed=seed + 1)
        try_seq(a_opt, "sg_parent2")
    
    # === Phase 2: Simulated Annealing ===
    print(f"\n=== Phase 2: Simulated Annealing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(300, rem * 0.25)
    if t2 > 30 and best_seq is not None:
        print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
        a_opt, _ = simulated_anneal(best_seq, time_limit=t2, verbose=True, seed=seed + 100)
        try_seq(a_opt, "sa_1")
        
        # Second SA from slightly perturbed start
        rem = deadline - time.monotonic()
        t2b = min(200, rem * 0.15)
        if t2b > 20:
            perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.003)
            a_opt, _ = simulated_anneal(perturbed, time_limit=t2b, verbose=False, seed=seed + 200)
            try_seq(a_opt, "sa_2")
    
    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Peak CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(150, rem * 0.15)
    if t3 > 20 and best_seq is not None:
        print(f"  Time: {t3:.0f}s")
        a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t3)
        try_seq(a_opt, "cd_1")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    base_len = len(base)
    
    candidate_lengths = set()
    for delta in [-128, -64, -32, 32, 64, 128, 256]:
        ln = base_len + delta
        if 256 <= ln <= 2048:
            candidate_lengths.add(ln)
    # Also try clean lengths
    for ln in [512, 640, 768, 832, 896, 1024, 1280]:
        if abs(ln - base_len) <= 512:
            candidate_lengths.add(ln)
    
    candidate_lengths = sorted(candidate_lengths)
    
    for target_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.25, 120)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  n={target_n}, time={t_alloc:.0f}s")
        
        # Quick SG + SA
        t_sg = t_alloc * 0.6
        t_sa = t_alloc * 0.4
        a_opt, _ = sg_momentum(f_interp, time_limit=t_sg, lr=0.003, momentum=0.93,
                               top_k=40, temp=0.00006, seed=seed + target_n)
        score = try_seq(a_opt, f"sg_n{target_n}")
        
        if t_sa > 10:
            a_opt, _ = simulated_anneal(a_opt, time_limit=t_sa, seed=seed + target_n + 500)
            try_seq(a_opt, f"sa_n{target_n}")
    
    # === Phase 5: Structured initializations ===
    print(f"\n=== Phase 5: Structured inits (best={best_score:.8f}) ===")
    for init_n in [base_len, base_len + 128, max(256, base_len - 128)]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_per_init = min(rem * 0.15, 60)
        if t_per_init < 15:
            break
        
        inits = structured_initializations(init_n, seed=seed + init_n)
        for ii, init in enumerate(inits[:5]):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_short = min(rem * 0.15, 40)
            if t_short < 10:
                break
            a_opt, _ = sg_momentum(init, time_limit=t_short, lr=0.005, momentum=0.92,
                                   top_k=50, temp=0.0001, seed=seed + init_n * 10 + ii)
            try_seq(a_opt, f"init_{init_n}_{ii}")
    
    # === Phase 6: Extended refinement on best ===
    print(f"\n=== Phase 6: Extended refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Long SG with very fine parameters
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_sg = min(rem * 0.4, 200)
            print(f"  Fine SG: {t_sg:.0f}s")
            a_opt, _ = sg_momentum(best_seq, time_limit=t_sg, lr=0.001, momentum=0.95,
                                   top_k=30, temp=0.00002, verbose=True, seed=seed + 1000)
            try_seq(a_opt, "sg_fine")
        
        # SA fine
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_sa = min(rem * 0.3, 100)
            if t_sa > 15:
                a_opt, _ = simulated_anneal(best_seq, time_limit=t_sa, seed=seed + 2000)
                try_seq(a_opt, "sa_fine")
        
        # CD fine
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_cd = min(rem * 0.3, 80)
            if t_cd > 10:
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_cd)
                try_seq(a_opt, "cd_fine")
    
    # === Phase 7: Final perturbation sweep ===
    print(f"\n=== Phase 7: Final perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(10):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            noise_scale = 0.0005 * (1 + p * 0.3)
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_short = min(rem * 0.2, 30)
            if t_short < 8:
                break
            a_opt, _ = sg_momentum(f_pert, time_limit=t_short, lr=0.0008, momentum=0.95,
                                   top_k=20, temp=0.00001, seed=seed + 3000 + p)
            try_seq(a_opt, f"perturb_{p}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
