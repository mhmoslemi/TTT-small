Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with
adaptive constraint selection, FW+ away steps, spectral randomization,
multi-length optimization, coordinate descent, and subgradient polishing.
Key improvements: better constraint management, length scaling, and diverse restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n from a (uses first half)."""
    half = (n + 1) // 2
    if len(a) >= half:
        a_use = a[:half]
    else:
        a_use = np.pad(a, (0, half - len(a)))
    return np.concatenate([a_use, a_use[-2::-1]])


def objective_simplex(f):
    """Compute 2*n*max_conv for f on the simplex (sum=1)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0. Returns x or None."""
    c = -np.ones(n)
    if active_k is None:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def find_best_alpha(f, g, n, max_iter=30):
    """Find optimal alpha in [0,1] minimizing max_conv(f + alpha*(g-f))."""
    # Use ternary search since the objective is the max of quadratics (unimodal-ish)
    lo = 0.0
    hi = 1.0
    # Golden section
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)

    f_c = np.maximum(f + c * (g - f), 0.0)
    f_c = f_c / f_c.sum()
    f_d = np.maximum(f + d * (g - f), 0.0)
    f_d = f_d / f_d.sum()
    fc = fast_convolve(f_c).max()
    fd = fast_convolve(f_d).max()

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            f_c = np.maximum(f + c * (g - f), 0.0)
            f_c = f_c / f_c.sum()
            fc = fast_convolve(f_c).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            f_d = np.maximum(f + d * (g - f), 0.0)
            f_d = f_d / f_d.sum()
            fd = fast_convolve(f_d).max()

    return (lo + hi) / 2.0


def lp_fw_optimize(f0, time_limit, verbose=False, top_k=50, use_away=True,
                   full_lp_every=20, seed=0):
    """LP-based Frank-Wolfe with away steps and adaptive constraints."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    # For FW+: keep track of vertices visited
    vertices = [f.copy()]
    v_coeff = np.array([1.0])

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Determine active constraints
        if it % full_lp_every == 0:
            active_k = None
        else:
            # Mix: top-k by value + some random
            k_sorted = np.argsort(conv)[::-1][:top_k]
            num_random = max(5, top_k // 4)
            random_k = rng.integers(0, 2 * n - 1, size=num_random)
            active_k = np.unique(np.concatenate([k_sorted, random_k]))

        g = solve_lp_direction(f, M * m_exp, n, active_k=active_k)
        if g is None:
            m_exp *= 1.15
            if m_exp > 4.0:
                m_exp = 1.0
                # Try with all constraints
                g = solve_lp_direction(f, M, n, active_k=None)
                if g is None:
                    continue
        else:
            m_exp = 1.0

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.08, 3.0)
            continue

        # Find best alpha
        alpha = find_best_alpha(f, g, n)

        if use_away and len(vertices) > 1 and alpha > 0.01:
            # FW+: try away steps from worst vertex
            # Find vertex with worst contribution
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()

            # Try removing each vertex (away step)
            best_obj_away = fast_convolve(f_new).max()
            best_away_f = f_new.copy()
            for vi in range(min(len(vertices), 5)):
                v = vertices[vi]
                # Move away from v: f_away = f_new + beta * (f_new - v)
                # Find beta that minimizes
                for beta_try in [0.1, 0.2, 0.3]:
                    f_away = f_new + beta_try * (f_new - v)
                    f_away = np.maximum(f_away, 0.0)
                    sa = f_away.sum()
                    if sa > 1e-10:
                        f_away = f_away / sa
                        obj_away = fast_convolve(f_away).max()
                        if obj_away < best_obj_away:
                            best_obj_away = obj_away
                            best_away_f = f_away

            f_new = best_away_f

            # Update vertices representation (simplified)
            vertices.append(g.copy())
            if len(vertices) > 8:
                vertices = vertices[-8:]

        else:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            if alpha > 0.01:
                vertices.append(g.copy())
                if len(vertices) > 8:
                    vertices = vertices[-8:]

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new
        conv = conv_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            M = M_new
            if verbose and it % 20 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        f = f_new

        if no_improve > 60:
            # Random restart with increasing perturbation
            scale = 0.002 + 0.004 * min(no_improve // 60, 8)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            vertices = [f.copy()]

    return best_f, best_obj / (2.0 * n)  # Return M only (normalized)


def coord_desc_fine(f0, time_limit, multipliers=None):
    """Coordinate descent with multi-scale local search per coordinate."""
    if multipliers is None:
        multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, mom=0.93, top_k=40, temp=0.00003):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.005

        if no_improve > 300:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n < 2:
        return np.full(new_n, f[0] / new_n * old_n)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, seed=0, n_iter=50):
    """Spectral phase retrieval: find non-negative signal with flat-ish convolution."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    # Start with flat magnitude
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        # Target: make A^2 have flat magnitude (so conv is flat)
        # A^2 magnitude = |A|^2. If |A| is flat, then |A^2| is flat.
        # So we want |A| to be flat.
        target_mag = np.ones(fft_size // 2 + 1)
        # But DC must be 1 (sum = 1)
        target_mag[0] = 1.0
        # Blend: keep some of the current magnitude
        blend = 0.7
        new_mag = blend * target_mag + (1 - blend) * np.abs(A)
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        # Ensure conjugate symmetry (rfft gives real result)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        else:
            break
        a = a_new
    return a


def spectral_random_step(f, n, rng, strength=0.3):
    """Randomize phase of spectrum while keeping magnitude, then project."""
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    # Perturb phase
    phase_pert = phase + rng.uniform(-strength * np.pi, strength * np.pi, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
        return a_new
    return f.copy()

                a_opt2, _ = sg_polish(a_opt, time_limit=t_sg2,
                                      lr=0.0005, mom=0.95,
                                      top_k=max(20, n_best // 6),
                                      temp=0.00001)
                try_seq(a_opt2, "sg_fine")

    # === Phase 7: Spectral randomization ===
    print(f"\n=== Phase 7: Spectral randomization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        bn = best_seq / best_seq.sum()
        rem = deadline - time.monotonic()
        attempts = 0
        while rem > 20 and attempts < 20:
            attempts += 1
            strength = 0.1 * (1 + attempts * 0.1)
            f_rand = spectral_random_step(bn, n_best, rng, strength=strength)
            # Quick optimization
            t_alloc = min(rem * 0.1, 30)
            a_opt, _ = lp_fw_optimize(f_rand, time_limit=t_alloc, verbose=False,
                                      top_k=40, use_away=False, full_lp_every=15,
                                      seed=seed + 400 + attempts)
            try_seq(a_opt, f"specrand_{attempts}")
            rem = deadline - time.monotonic()

    # === Phase 8: Final deep LP-FW ===
    print(f"\n=== Phase 8: Final Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 50:
            t_final = min(rem * 0.5, 200)
            print(f"  Final LP-FW t={t_final:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq / best_seq.sum(), time_limit=t_final,
                                      verbose=True, top_k=80, use_away=True,
                                      full_lp_every=15, seed=seed + 500)
            try_seq(a_opt, "final_lpfw")

        # Final CD + SG
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None:
            a_cd, _ = coord_desc_fine(best_seq / best_seq.sum(), time_limit=min(rem * 0.3, 60))
            try_seq(a_cd, "final_cd")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_sg, _ = sg_polish(best_seq / best_seq.sum(), time_limit=min(rem * 0.4, 40),
                                    lr=0.0004, mom=0.95,
                                    top_k=max(15, len(best_seq) // 8),
                                    temp=0.000005)
                try_seq(a_sg, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
