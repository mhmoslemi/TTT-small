def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def project_simplex_kl(v, ref, steps=50):
    """KL mirror descent projection: find g >= 0, sum(g)=1 minimizing KL(g||v/ref) + <c, g>
    Actually, for mirror descent, the update is:
    g[j] = ref[j] * exp(-eta * grad[j]) / sum(...)
    This is a multiplicative update.
    """
    # Simple exponential update
    g = ref * np.exp(np.minimum(v, 50.0) - np.maximum(v, -50.0))  # clip for stability
    # Actually, let me just do: g = ref * exp(v) normalized
    # But v could be large. Let's use a safer version.
    v_shifted = v - np.max(v)
    g = ref * np.exp(v_shifted)
    s = g.sum()
    if s < 1e-30:
        g = ref.copy()
    else:
        g /= s
    return g

def conjugate_subgrad(f0, time_limit, lr=0.005, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    # Target: gradually decrease M
    target = M * 0.95
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
            
        conv = fast_convolve(f)
        M = conv.max()
        
        # Hinge loss: focus on peaks above target
        excess = np.maximum(conv - target, 0.0)
        # Weight the subgradient by the excess
        # Subgradient of max(c) is 2*a[::-1] at the peak positions
        # For hinge: sum_k max(0, c[k]-target) * 2 * a_reversed_at_k
        
        # Find positions with excess

        if not active.any():
            # No active constraints, try to reduce target
            target = max(0.98 * M, target * 0.99)
            # Use uniform direction
            sg = np.zeros(n)
            # Small random perturbation

            sg = rng.standard_normal(n) * 0.001
        else:
            # Compute weighted subgradient

            # d/dc[k] * d(a*a)[k]/da[j] = excess[k] * a[k-j] (for valid k-j)
            # (a*a)[k] = sum_j a[j] * a[k-j]

def peak_reduction_cd(f0, time_limit, verbose=False):
    """Coordinate descent that specifically targets reducing the peak.
    For each coordinate j, compute how much it contributes to the peak,
    and try reducing it while redistributing to low-contribution coords."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        peak_pos = int(np.argmax(conv))
        
        # Find which coordinates j contribute most to the peak
        # c[peak_pos] = sum_j a[j] * a[peak_pos - j]
        js = max(0, peak_pos - n + 1)
        je = min(n - 1, peak_pos)
        
        contributions = np.zeros(n)
        idx = np.arange(js, je + 1)
        kg = peak_pos - idx
        mask = (kg >= 0) & (kg < n)
        contributions[idx[mask]] = f[idx[mask]] * f[kg[mask]]
        
        # Sort by contribution (highest first)
        sorted_idx = np.argsort(contributions)[::-1]
        
        improved = False
        # Try reducing top-contributing coordinates
        for j in sorted_idx[:min(20, n // 5)]:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            if f[j] < 1e-8:
                continue
            
            # Try reducing f[j] by various amounts
            for frac in [0.3, 0.5, 0.7, 0.9]:
                delta = f[j] * frac
                if delta < 1e-10:
                    continue
                
                # Remove delta from f[j], distribute to lowest-contributing coords
                f_trial = f.copy()
                f_trial[j] -= delta
                
                # Distribute to coords with lowest contribution to peak
                low_contrib = np.argsort(contributions)[:n//2]
                low_contrib = [lc for lc in low_contrib if lc != j and f[lc] >= 0]
                if not low_contrib:
                    continue
                
                # Distribute proportionally to current values (or uniformly)
                total_low = sum(f[lc] for lc in low_contrib)
                if total_low < 1e-15:
                    # Distribute uniformly among low-contrib coords
                    for lc in low_contrib:
                        f_trial[lc] += delta / len(low_contrib)
                else:
                    for lc in low_contrib:
                        f_trial[lc] += delta * f[lc] / total_low
                
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj_trial
                    best_f = f.copy()
                    improved = True
                    conv = fast_convolve(f)
                    M = conv.max()
                    peak_pos = int(np.argmax(conv))
                    # Recompute contributions
                    contributions = np.zeros(n)
                    js2 = max(0, peak_pos - n + 1)
                    je2 = min(n - 1, peak_pos)
                    idx2 = np.arange(js2, je2 + 1)
                    kg2 = peak_pos - idx2
                    mask2 = (kg2 >= 0) & (kg2 < n)
                    contributions[idx2[mask2]] = f[idx2[mask2]] * f[kg2[mask2]]
                    sorted_idx = np.argsort(contributions)[::-1]
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
        else:
            no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_f, best_obj

3. Targeted peak-reduction coordinate descent

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(f, target):
    """Compute subgradient of hinge-peak objective: sum_k max(0, c[k]-target)*2*f_reversed.
    Returns (subgradient, conv, peak) where subgradient has length len(f)."""
    n = len(f)
    c = fast_convolve(f)
    e = np.maximum(c - target, 0.0)
    if e.max() < 1e-15:
        return None, c, c.max()
    # Cross-correlation (e *_ f)[j] = sum_k e[k] * f[k-j]
    # DFT(e *_ f) = DFT(e) * conj(DFT(f))
    N = 1 << (3 * n - 1).bit_length()
    E = np.fft.rfft(e, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(E * np.conj(F), N)[:n]

    return sg, c, c.max()


def hinge_subgrad_nesterov(f0, time_limit, lr=0.006, momentum=0.9, verbose=False,
                           target_frac=0.95, target_decay=0.9997):
    """Nesterov-accelerated hinge-subgradient descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    target = M * target_frac

    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov step: evaluate at y = f + (t-1)/t * (f - f_prev)
        t_ns = 1 + momentum
        y_new = project_simplex(f + (t_ns - 1) / t_ns * (f - y))
        y = y_new

        sg, conv, M_cur = compute_subgradient_fft(y, target)

        if sg is None:
            # No active constraints - reduce target
            target = max(0.99 * M_cur, target * 0.999)
            # Small random kick to explore
            rng = np.random.default_rng(it * 31 + 7)
            sg = rng.standard_normal(n) * 0.001
        else:
            # Adaptive target: slowly decrease
            target = max(0.99 * M_cur, target * target_decay)

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 1000 == 0:
                print(f"    HSG iter {it}: obj={best_obj:.10f}, target={target:.10f}")
        else:
            no_improve += 1
            f = f_new
            if no_improve % 200 == 0:
                target = min(M_cur * 0.99, target * 1.005)

        cur_lr = max(0.0003, cur_lr * 0.99995)

        if no_improve > 800:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            y = f.copy()
            conv = fast_convolve(f)
            target = conv.max() * target_frac
            no_improve = 0
            cur_lr = max(lr * 0.4, 0.001)

    return best_f, best_obj


def peak_reduction_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting the peak: reduce high-contributing coords,
    redistribute to low-contributing ones."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        peak_pos = int(np.argmax(conv))

        # Contributions to peak
        js = max(0, peak_pos - n + 1)
        je = min(n - 1, peak_pos)
        idx = np.arange(js, je + 1)
        kg = peak_pos - idx
        mask = (kg >= 0) & (kg < n)
        contributions = np.zeros(n)
        contributions[idx[mask]] = f[idx[mask]] * f[kg[mask]]

        sorted_idx = np.argsort(contributions)[::-1]
        low_sorted = np.argsort(contributions)

        improved = False
        n_high = min(15, n // 5)
        for j in sorted_idx[:n_high]:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            if f[j] < 1e-8:
                continue

            best_frac = 0
            best_val = best_obj
            for frac in [0.2, 0.4, 0.6, 0.8, 0.95]:
                delta = f[j] * frac
                if delta < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] -= delta

                # Redistribute to lowest-contributing coords
                n_low = min(n // 3, 100)
                low_idx = low_sorted[:n_low]
                low_valid = [li for li in low_idx if li != j]
                if not low_valid:
                    continue
                total_fw = sum(f[li] for li in low_valid)
                if total_fw < 1e-15:
                    for li in low_valid:
                        f_trial[li] += delta / len(low_valid)
                else:
                    for li in low_valid:
                        f_trial[li] += delta * f[li] / total_fw

                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_val - 1e-14:
                    best_val = obj_trial
                    best_frac = frac
                    best_trial = f_trial

            if best_frac > 0:
                f = best_trial
                if best_val < best_obj - 1e-14:
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True
                    conv = fast_convolve(f)
                    # Recompute contributions
                    M = conv.max()
                    peak_pos = int(np.argmax(conv))
                    js2 = max(0, peak_pos - n + 1)
                    je2 = min(n - 1, peak_pos)
                    idx2 = np.arange(js2, je2 + 1)
                    kg2 = peak_pos - idx2
                    mask2 = (kg2 >= 0) & (kg2 < n)
                    contributions = np.zeros(n)
                    contributions[idx2[mask2]] = f[idx2[mask2]] * f[kg2[mask2]]
                    sorted_idx = np.argsort(contributions)[::-1]
                    low_sorted = np.argsort(contributions)
                    break

        if not improved:
            no_improve += 1
            if no_improve > 300:
                rng = np.random.default_rng(it * 104729 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                no_improve = 0
        else:
            no_improve = 0

        if no_improve > 800:
            break

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe: solve min-peak LP, line search, iterate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # LP: min t s.t. (f*g)[k] <= t, g >= 0, sum(g) = 1
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        n_con = 2 * n - 1
        A_ub = np.zeros((n_con, n + 1))
        for j in range(n):
            A_ub[j:j + n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_con)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = np.maximum(res.x[:n], 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = best_val
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Nesterov hinge-subgradient (FFT-accelerated) on parent
    2. Peak-reduction coordinate descent
    3. LP Frank-Wolfe for long-range moves
    4. Multi-length exploration (shorter preferred)
    5. Final polish with fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Nesterov Hinge Subgradient (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov Hinge-SG (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(450, rem * 0.35)
    if t1 > 30:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = hinge_subgrad_nesterov(parent, time_limit=t1,
                                          lr=0.005, momentum=0.92,
                                          target_frac=0.93, verbose=True)
        try_seq(a_opt, "nesterov_hsg")

        # Second run with different params
        rem = deadline - time.monotonic()
        t1b = min(200, rem * 0.15)
        if t1b > 20:
            a_opt, _ = hinge_subgrad_nesterov(best_seq, time_limit=t1b,
                                              lr=0.002, momentum=0.95,
                                              target_frac=0.97, verbose=False)
            try_seq(a_opt, "nesterov_hsg_fine")

    # === Phase 2: Peak Reduction CD ===
    print(f"\n=== Phase 2: Peak Reduction CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(300, rem * 0.25)
        if t2 > 30:
            print(f"  n={len(best_seq)}, t={t2:.0f}s")
            a_opt, _ = peak_reduction_cd(best_seq, time_limit=t2)
            try_seq(a_opt, "peak_cd")

    # === Phase 3: LP Frank-Wolfe ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(200, rem * 0.18)
        if t3 > 30:
            print(f"  n={len(best_seq)}, t={t3:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lp_fw")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = set()
    # Try nearby lengths (shorter preferred since score ~ n * peak)
    for delta in [-128, -64, -32, 32, 64, 128]:
        ln = len(base) + delta
        if 256 <= ln <= 3000:
            candidate_lengths.add(ln)
    # Powers of 2 and common values
    for pn in [384, 512, 640, 768, 1024, 1280]:
        candidate_lengths.add(pn)
    candidate_lengths = sorted(candidate_lengths)

    # Sort by closeness to base, prefer shorter
    def length_priority(ln):
        diff = abs(ln - len(base))
        penalty = max(0, ln - len(base)) * 0.5  # prefer shorter
        return diff + penalty

    candidate_lengths.sort(key=length_priority)

    for target_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.25, 150)
        if t_alloc < 25:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  HSG n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = hinge_subgrad_nesterov(f_interp, time_limit=t_alloc,
                                          lr=0.004, momentum=0.92,
                                          target_frac=0.93)
        try_seq(a_opt, f"multi_n{target_n}")

        # Quick peak CD on the result
        rem = deadline - time.monotonic()
        if rem > 40:
            t_cd = min(rem * 0.4, 60)
            if t_cd > 15:
                a_opt2, _ = peak_reduction_cd(a_opt, time_limit=t_cd)
                try_seq(a_opt2, f"multi_cd_n{target_n}")

    # === Phase 5: Final Polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick Nesterov SG
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = hinge_subgrad_nesterov(best_seq, time_limit=min(rem * 0.4, 80),
                                              lr=0.001, momentum=0.96,
                                              target_frac=0.99)
            try_seq(a_opt, "polish_hsg")

        # Quick peak CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = peak_reduction_cd(best_seq, time_limit=min(rem * 0.3, 60))
            try_seq(a_opt, "polish_cd")

        # Perturbations with quick refinement
        for p in range(5):
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0008 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = hinge_subgrad_nesterov(f_pert, time_limit=min(rem * 0.7, 30),
                                              lr=0.001, momentum=0.95,
                                              target_frac=0.98)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
