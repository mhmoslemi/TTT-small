    n = len(a)
    v = np.zeros(n)
    cur_t = t0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0_time = time.monotonic()
    
    while time.monotonic() - t0_time < time_limit:
        conv = fast_convolve(a)
        smooth_val, grad, weights = smooth_max_grad(a, conv, cur_t)
        
        # Projected gradient step with momentum
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Check objective
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        cur_t = max(t_end, cur_t * 0.999)
        # Decay lr
        lr *= 0.9999
    
    return best_a

with FFT-based gradient computation, momentum, temperature annealing,
coordinate bisection polishing, and multi-length exploration.
Key insight: the gradient of the smooth max of autocorrelation can be
computed in O(n log n) via FFT, enabling millions of fast iterations.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, conv, t):
    """
    Compute the gradient of the smooth max of conv w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv_len = 2 * n - 1
    scaled = conv / t
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()

    # grad_a[j] = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
    # = 2 * (a * w_rev)[2n-2-j] for j=0..n-1
    w_rev = w[::-1]
    total_len = n + conv_len - 1  # = 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_grad = np.fft.irfft(A * W, fft_size)[:total_len]
    grad_a = 2.0 * conv_grad[n - 1:2 * n - 1][::-1]

    return grad_a, w

def coordinate_bisection_fast(a0, time_limit, verbose=False):
    """
    Coordinate descent with bisection. Optimized for speed.
    Each pass tries each coordinate and finds the optimal value via bisection.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0

            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))

                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break

            x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
                conv = fast_convolve(a)
            else:
                break

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

    temperatures = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for idx, t_val in enumerate(temperatures):
        rem = total_deadline - time.monotonic() - (len(temperatures) - idx) * 30
        if rem < 10:
            break
        print(f"  L-BFGS t={t_val}, remaining budget ~{rem:.0f}s")
        try:
            a_opt = lbgfs_smooth(z_init, t_val, maxiter=200)
            _update_best(a_opt, f"lbfgs_t{t_val}")
            # Update z_init for next temperature
            z_init = np.log(a_opt + 1e-15)
        except Exception as e:
            print(f"  L-BFGS failed at t={t_val}: {e}")
            break

    # === Phase 3: Coordinate bisection polish ===
    print(f"\n=== Phase 3: CD polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = total_deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 250)
            print(f"  Allocated: {t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection_fast(best_a, time_limit=t_alloc, verbose=True)
            _update_best(a_opt, "cd_polish")

    # === Phase 4: Smooth PG on best with different parameters ===
    print(f"\n=== Phase 4: More smooth PG (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        configs = [
            (0.01, 0.0005, 0.03, 0.92),
            (0.005, 0.0002, 0.02, 0.95),
            (0.02, 0.001, 0.06, 0.85),
        ]
        for idx, (t_s, t_e, lr_s, m) in enumerate(configs):
            rem = total_deadline - time.monotonic() - (len(configs) - idx) * 50
            if rem < 20:
                break
            t_alloc = min(rem, 150)
            print(f"  Config {idx}: t=[{t_s},{t_e}], lr={lr_s}, mom={m}, t_alloc={t_alloc:.0f}s")
            # Add slight perturbation
            perturb = best_a / best_a.sum() + rng.standard_normal(n_best) * 0.002
            perturb = project_simplex(perturb)
            a_opt, _ = smooth_projected_gradient(
                perturb, time_limit=t_alloc, t_start=t_s, t_end=t_e,
                lr_start=lr_s, mom=m, verbose=False
            )
            _update_best(a_opt, f"spg_cfg{idx}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    if best_a is not None:
        target_lengths = [640, 896, 1024, 1280]
        for target_n in target_lengths:
            rem = total_deadline - time.monotonic() - 200
            if rem < 40:
                break
            f_interp = interpolate_to_length(best_a, target_n)
            t_alloc = min(rem * 0.4, 120)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_projected_gradient(
                f_interp, time_limit=t_alloc * 0.6, t_start=0.02, t_end=0.001,
                lr_start=0.03, mom=0.9, verbose=False
            )
            # Quick CD polish
            rem2 = total_deadline - time.monotonic() - 150
            if rem2 > 15:
                a_opt2, _ = coordinate_bisection_fast(a_opt, time_limit=min(rem2 * 0.3, 30))
                _update_best(a_opt2, f"multi_n{target_n}_cd")
            _update_best(a_opt, f"multi_n{target_n}")

    # === Phase 6: Symmetric variant ===
    print(f"\n=== Phase 6: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = total_deadline - time.monotonic() - 100
        if rem > 30:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 80)
            a_opt, _ = smooth_projected_gradient(
                a_sym, time_limit=t_alloc * 0.7, t_start=0.01, t_end=0.0005,
                lr_start=0.03, mom=0.92, verbose=False
            )
            rem2 = total_deadline - time.monotonic() - 60
            if rem2 > 15:
                a_opt2, _ = coordinate_bisection_fast(a_opt, time_limit=min(rem2, 40))
                _update_best(a_opt2, "sym_cd")
            _update_best(a_opt, "sym_spg")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = total_deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 200)
            print(f"  Allocated: {t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection_fast(best_a, time_limit=t_alloc, verbose=True)
            _update_best(a_opt, "final_cd")

            # Tiny perturbations + CD
            rem2 = total_deadline - time.monotonic() - 15
            if rem2 > 10 and best_a is not None:
                n_best = len(best_a)
                for p in range(5):
                    rem2 = total_deadline - time.monotonic() - 10
                    if rem2 < 8:
                        break
                    rng_p = np.random.default_rng(seed + 2000 + p * 13)
                    f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.001)
                    t_alloc2 = min(rem2, 15)
                    a_opt_p, _ = coordinate_bisection_fast(f_pert, time_limit=t_alloc2)
                    _update_best(a_opt_p, f"final_perturb_{p}")

    # === Phase 8: Very fine smooth gradient ===
    print(f"\n=== Phase 8: Fine SG (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = total_deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 100)
            a_opt, _ = smooth_projected_gradient(
                best_a, time_limit=t_alloc, t_start=0.005, t_end=0.0001,
                lr_start=0.01, mom=0.95, verbose=False
            )
            _update_best(a_opt, "fine_spg")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
