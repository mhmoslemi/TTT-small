import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g>=0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.92,
                         top_k=50, temp_scale=0.0001, verbose=False):
    """Momentum subgradient with softmax-weighted multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp_scale

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(k_actual)
            w[int(np.argmax(pv))] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            sg[js:je + 1] += 2.0 * w[i] * f[k - je:k - js - 1:-1]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.99992
        cur_temp *= 1.006

        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + 13) % 2**31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
            conv = fast_convolve(f)

        if verbose and it % 2000 == 0:
            print(f"      SG iter {it}: obj={best_obj:.10f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, max_iters=500, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        g, _ = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            g = np.maximum(g, 0)
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            f = f_new / sn if sn > 1e-15 else np.ones(n) / n
            M = best_val
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 10 == 0:
            print(f"      LP-FW iter {it}: obj={best_obj:.10f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    pass_num = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        improved = False

        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-12:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj

                for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ts = f_trial.sum()
                    if ts < 1e-10:
                        continue
                    f_trial /= ts
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f /= f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 5:
                rng = np.random.default_rng(pass_num * 104729 + 7)
                for _ in range(min(30, n // 5)):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 1.4, 1.8]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.7:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial /= f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve_passes > 15:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def multi_peak_lse_optimize(f0, time_limit, verbose=False):
    """
    Optimizes using a smooth log-sum-exp approximation to max(conv),
    with adaptive temperature annealing and heavy ball momentum.
    This provides smoother gradients than the exact subgradient.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = 0.91
    top_k = 80

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Adaptive temperature: start broad, narrow to peak
        progress = min(1.0, it / 10000)
        temp = 1e-5 + progress * 5e-5

        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(k_actual)
            w[0] = 1.0
        else:
            w /= ws

        # Compute weighted gradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            sg[js:je + 1] += 2.0 * w[i] * f[k - je:k - js - 1:-1]

        # Adaptive learning rate
        lr = 0.004 / (1.0 + 0.005 * it)

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new

        if no_improve > 500:
            rng = np.random.default_rng((it * 31 + 17) % 2**31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            no_improve = 0
            conv = fast_convolve(f)

        if verbose and it % 3000 == 0:
            print(f"      LSE iter {it}: obj={best_obj:.10f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def generate_diverse_starts(n, rng):
    """Generate diverse initial sequences of length n."""
    starts = []

    # Uniform
    starts.append(np.ones(n) / n)

    # Geometric (asymmetric)
    r = 0.98
    g = r ** np.arange(n)
    starts.append(g / g.sum())

    # Inverse geometric
    g2 = np.power(0.98, np.arange(n - 1, -1, -1))
    starts.append(g2 / g2.sum())

    # Two-hump
    center1 = n // 4
    center2 = 3 * n // 4
    sigma = n // 8
    x = np.arange(n)
    hump = np.exp(-0.5 * ((x - center1) / sigma) ** 2) + np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    starts.append(hump / hump.sum())

    # Cosine (emphasizing one end)
    x = np.arange(n)
    cos_seq = np.cos(np.pi * x / (2 * n)) ** 2
    cos_seq = np.maximum(cos_seq, 1e-8)
    starts.append(cos_seq / cos_seq.sum())

    # Random with smoothing
    r_seq = rng.random(n)
    kernel = np.ones(5) / 5
    r_smooth = np.convolve(r_seq, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 1e-8)
    starts.append(r_smooth / r_smooth.sum())

    return starts
