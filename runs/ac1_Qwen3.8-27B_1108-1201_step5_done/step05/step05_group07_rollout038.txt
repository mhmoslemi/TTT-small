import time

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_normalized(a):
    """Compute max(conv(a,a)) assuming sum(a)=1."""
    conv = fast_convolve(a)
    return conv.max(), conv

def compute_gradient(a, conv, temperature):
    """Compute gradient of log-sum-exp smoothed max of conv."""
    n = len(a)
    exp_conv = np.exp((conv - conv.max()) / temperature)
    w = exp_conv / exp_conv.sum()
    E_c = np.dot(w, conv)
    
    # cross-correlation of w and a
    # r[j] = sum_m w[j+m] * a[m] for m = 0..n-1 (where j+m < 2n-1)
    a_flip = a[::-1]

    r = conv_wa[n-1:2*n-1]  # this gives r[j] = sum_m w[j+m]*a[m]

    grad = 2.0 * (r - E_c)
    return grad, w, E_c

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (log-sum-exp) with adaptive step size and Nesterov momentum
2. Active-set coordinate descent with targeted peak shaving
3. Multi-length exploration with interpolation and re-optimization
4. Random perturbation restarts to escape local minima
Key improvement over previous: uses true gradient of smoothed objective (not subgradient),
enabling much faster convergence via second-order-like behavior of adaptive step sizes.
"""

import numpy as np
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, conv, temperature):
    """
    Compute gradient of F(a) = (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a.
    Returns (gradient, weights, expected_conv).
    """
    n = len(a)
    max_c = conv.max()
    exp_c = np.exp((conv - max_c) / temperature)
    w = exp_c / exp_c.sum()
    E_c = np.dot(w, conv)

    # Cross-correlation r[j] = sum_m w[j+m]*a[m]
    fft_size = 1 << (len(conv) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    r = np.fft.irfft(W * np.conj(A), fft_size)[:n]

    grad = 2.0 * (r - E_c)
    return grad, w, E_c


def smoothed_grad_descent(f0, time_limit, T_init=0.005, T_min=0.0005,
                          alpha_init=0.005, momentum_init=0.85,
                          verbose=False, label=""):
    """
    Smoothed gradient descent with Nesterov momentum and adaptive step size.
    Optimizes max(conv(a,a)) over a >= 0, sum(a) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    momentum = momentum_init
    alpha = alpha_init
    T = T_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute smoothed gradient
        grad, w, E_c = compute_smoothed_gradient(f, conv, T)
        # Project to tangent space of simplex (subtract mean)
        grad = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        v = v - np.mean(v)

        # Gradient step with adaptive alpha
        f_new = f - alpha * v
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new <= obj + 1e-14 * max(1.0, obj):
            f = f_new
            if obj_new < obj - 1e-16:
                alpha = min(alpha * 1.02, 0.1)
        else:
            alpha = max(alpha * 0.5, 1e-6)

        # Cool temperature
        T = max(T * 0.9995, T_min)

        # Adaptive momentum
        if no_improve > 200:
            momentum = max(momentum * 0.99, 0.7)
        elif no_improve == 0:
            momentum = min(momentum * 1.005, 0.95)

        # Restart if stuck
        if no_improve > 800:
            rng = np.random.default_rng((it * 104729 + 7) % 1000000)
            scale = 0.005 * min(1 + no_improve // 2000, 4.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            alpha = alpha_init
            T = T_init * 0.5
            momentum = momentum_init
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD[{label}] iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, "
                  f"alpha={alpha:.6f}, T={T:.6f}")

    return best_f, best_obj


def coordinate_descent_active(f0, time_limit, top_k=20, verbose=False):
    """
    Active-set coordinate descent: focus on reducing the top peaks of convolution.
    For each coordinate, try targeted perturbations that reduce peak contributions.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        conv = fast_convolve(f)
        current_obj = conv.max()
        improved_this_sweep = False

        # Find active peaks
        k_peaks = min(top_k, len(conv))
        peak_indices = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        # For each peak, identify the top contributing pairs
        # and try to reduce them
        rng = np.random.default_rng(sweep)

        for k in peak_indices[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Top contributors to this peak
            top_contrib = np.argsort(contribs)[::-1][:min(5, len(idx))]
            for ci in top_contrib:
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(idx[ci])
                partner = k - j
                old_j = f[j]
                old_partner = f[partner] if partner != j else 0

                if old_j < 1e-12:
                    continue

                # Try reducing a[j] and increasing a[partner] (or redistributing)
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    amt = old_j * frac
                    f[j] = old_j - amt
                    # Redistribute to partner or random positions
                    if partner != j and partner >= 0 and partner < n:
                        f[partner] = f[partner] + amt
                    else:
                        f[rng.integers(0, n)] += amt

                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_f = f.copy()
                        improved_this_sweep = True
                        break
                    else:
                        f[j] = old_j
                        if partner != j and partner >= 0 and partner < n:
                            f[partner] = f[partner] - amt
                        else:
                            # undo
                            pass

        # Also try random coordinate updates
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            j = rng.integers(0, n)
            old_j = f[j]
            if old_j < 1e-12:
                continue

            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 2.0]:
                new_j = old_j * mult
                if new_j < 0 or new_j >= 0.5:
                    continue
                denom = 1.0 - old_j
                if denom < 1e-10:
                    break
                scale = (1.0 - new_j) / denom
                f_trial = f.copy()
                f_trial[j] = new_j
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    break
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-13:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved_this_sweep = True
                    break
                elif val > old_j * mult * 0 + current_obj + 0.01 * (current_obj - best_obj + 1e-10):
                    break

        if not improved_this_sweep and sweep > 3:
            # Try a perturbation restart
            if sweep % 5 == 0:
                scale = 0.002 * (1 + sweep // 10)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
            else:
                break

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """
    Targeted peak shaving: iteratively identify the worst peak and reduce it
    by optimally redistributing mass among contributing elements.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_obj = conv.max()

        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        # Find the global peak
        k_max = int(np.argmax(conv))
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[k_max - idx]

        # Find the most "asymmetric" contributor pair (large * small)
        # Reducing the large one and increasing the small one
        # reduces the product more effectively
        products = contribs.copy()
        # Sort by product descending
        order = np.argsort(products)[::-1]

        improved = False
        for oi in order[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(idx[oi])
            partner = k_max - j
            if partner == j or partner < 0 or partner >= n:
                continue

            a_j = f[j]
            a_p = f[partner]
            if a_j < 1e-12 or a_p < 1e-12:
                continue

            # Try transferring mass from the larger to the smaller
            if a_j > a_p:
                donor, receiver = j, partner
            else:
                donor, receiver = partner, j

            d_val = f[donor]
            r_val = f[receiver]
            if d_val < 1e-12:
                continue

            # Optimal transfer: minimize (d_val - x)(r_val + x)*scale
            # The product is maximized when both are equal, so we want to
            # equalize them (subject to constraint that total is preserved)
            # Product = (d_val - x)(r_val + x) = d_val*r_val + x*(d_val - r_val) - x^2
            # Derivative: d_val - r_val - 2x = 0 => x = (d_val - r_val)/2
            # But we also need to account for other peaks...
            # For now, try a fraction of the equalizing transfer
            max_transfer = (d_val - r_val) * 0.5
            if max_transfer <= 0:
                continue

            for frac in [0.3, 0.5, 0.7, 1.0]:
                amt = max_transfer * frac
                f[donor] = d_val - amt
                f[receiver] = r_val + amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[donor] = d_val
                    f[receiver] = r_val

            if improved:
                break

        if not improved:
            # Try reducing the peak by spreading mass to non-peak-contributing positions
            # Find positions that don't contribute to any top peak
            top_peaks = np.argpartition(conv, -min(20, len(conv)))[-min(20, len(conv)):]
            peak_set = set(int(p) for p in top_peaks)

            # Find a "safe" position: one that doesn't appear in any peak's contributing range
            safe_positions = []
            for pos in range(n):
                # pos contributes to peaks at k = pos + i for i in range(n)
                # i.e., peaks from pos to pos + n - 1
                # If none of those are in peak_set, it's safe
                max_k = min(pos + n - 1, 2 * n - 2)
                min_k = pos
                is_safe = True
                for pk in peak_set:
                    if min_k <= pk <= max_k:
                        is_safe = False
                        break
                if is_safe:
                    safe_positions.append(pos)

            if not safe_positions:
                safe_positions = [0]

            # Transfer mass from peak-contributing to safe positions
            donor = int(idx[order[0]]) if len(order) > 0 else 0
            if f[donor] > 1e-10:
                target = safe_positions[0]
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[donor] * frac
                    f[donor] -= amt
                    f[target] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[donor] += amt
                        f[target] -= amt

            if not improved:
                no_improve += 50

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, seed=42):
    """Generate diverse structured initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Gaussian
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 3.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Raised cosine (squared)
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    rc = np.maximum(rc ** 2, 1e-10)
    inits.append(rc / rc.sum())

    # Smooth random
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.hanning(kernel_len)
    kernel /= kernel.sum()
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines smoothed gradient descent, active-set coordinate descent,
    peak shaving, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")

    # === Phase 1: Smoothed gradient descent on parent ===
    print(f"\n=== Phase 1: Smoothed Gradient Descent (parent n={len(parent)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 350)
    if t1 > 10:
        print(f"  Time: {t1:.0f}s")
        a_opt, obj_opt = smoothed_grad_descent(parent, time_limit=t1,
                                                T_init=0.003, T_min=0.0003,
                                                alpha_init=0.004, momentum_init=0.87,
                                                verbose=True, label="parent")
        try_seq(a_opt, "sgd_parent")

    # === Phase 2: Alternate initializations ===
    print(f"\n=== Phase 2: Alternate Inits (best={best_score:.6f}) ===")
    for n_init in [len(parent), 512, 1024]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        inits = structured_inits(n_init, seed=seed + n_init)
        t_alloc = min(rem * 0.12, 80)
        for i, init in enumerate(inits[:3]):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_sub = min(t_alloc * 0.3, 40)
            if t_sub < 5:
                break
            a_opt, _ = smoothed_grad_descent(init, time_limit=t_sub,
                                              T_init=0.005, T_min=0.0005,
                                              alpha_init=0.003, momentum_init=0.85,
                                              label=f"init_{n_init}_{i}")
            try_seq(a_opt, f"init_{n_init}_{i}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {"T": 0.001, "Tmin": 0.0001, "alpha": 0.002, "mom": 0.90, "pert": 0.003},
            {"T": 0.002, "Tmin": 0.0002, "alpha": 0.004, "mom": 0.85, "pert": 0.005},
            {"T": 0.0005, "Tmin": 0.00005, "alpha": 0.001, "mom": 0.92, "pert": 0.002},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.2, 150)
            if t_alloc < 10:
                break

            if ci == 0:
                f_init = best_seq.copy()
            elif ci == 1:
                rng = np.random.default_rng(seed + ci * 100)
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * cfg["pert"])
            else:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()

            a_opt, _ = smoothed_grad_descent(f_init, time_limit=t_alloc,
                                              T_init=cfg["T"], T_min=cfg["Tmin"],
                                              alpha_init=cfg["alpha"],
                                              momentum_init=cfg["mom"],
                                              label=f"deep_{ci}")
            try_seq(a_opt, f"deep_c{ci}")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_ps = min(rem * 0.2, 120)
            print(f"  Time: {t_ps:.0f}s")
            a_opt, _ = peak_shaving(best_seq, time_limit=t_ps)
            try_seq(a_opt, "peak_shave")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.25, 150)
            print(f"  Time: {t_cd:.0f}s")
            a_opt, _ = coordinate_descent_active(best_seq, time_limit=t_cd, top_k=15)
            try_seq(a_opt, "coord_desc")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = []
        for tn in [512, 768, 1024, 1280, 1536, 2048]:
            if tn != n_best:
                target_lengths.append(tn)

        for target_n in target_lengths:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 100)
            if t_alloc < 10:
                break
            print(f"  n={target_n}, t={talloc:.0f}s")
            a_opt, _ = smoothed_grad_descent(f_interp, time_limit=t_alloc,
                                              T_init=0.003, T_min=0.0003,
                                              alpha_init=0.003, momentum_init=0.87,
                                              label=f"len{target_n}")
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # Very fine gradient descent
        if rem > 20:
            t_fine = min(rem * 0.3, 80)
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_fine,
                                              T_init=0.0001, T_min=0.00001,
                                              alpha_init=0.0005, momentum_init=0.95,
                                              label="fine")
            try_seq(a_opt, "fine_grad")

        # Quick coordinate descent
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, _ = coordinate_descent_active(best_seq, time_limit=min(rem * 0.4, 40), top_k=10)
            try_seq(a_opt, "fine_cd")

        # Perturbation + quick optimize
        rem = deadline - time.monotonic()
        rng_final = np.random.default_rng(seed + 999)
        for _ in range(5):
            rem = deadline - time.monotonic()
            if rem < 12:
                break
            noise = rng_final.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_sub = min(rem * 0.3, 15)
            a_opt, _ = smoothed_grad_descent(f_pert, time_limit=t_sub,
                                              T_init=0.0005, T_min=0.00005,
                                              alpha_init=0.001, momentum_init=0.92,
                                              label="pert")
            try_seq(a_opt, "pert")

        # Final peak shave
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = peak_shaving(best_seq, time_limit=min(rem * 0.5, 30))
            try_seq(a_opt, "final_shave")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
