"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected subgradient descent
with softmax-weighted peaks, adaptive backtracking, diverse initializations,
and coordinate descent polishing. Exploits the structure that the optimal
non-negative sequence has a smooth profile balancing edge and center
auto-convolution values.

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max(conv)/(sum)^2 for non-negative a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def subgradient_direction(a, conv, top_k=30, temp=0.005):
    """Compute softmax-weighted subgradient of max(conv) at point a."""
    n = len(a)
    m = len(conv)
    k_act = min(top_k, m)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    scaled = pv / max(temp, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    # Gradient of (a*a)[k] w.r.t. a is 2*a[k-j] (with zero extension)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-10:
            continue
        k = int(peaks[i])
        # a[k-j] for j=0..n-1, with a[m]=0 if m<0 or m>=n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            sg[j] += 2.0 * w[i] * a[k - j]
    return sg


def projected_subgrad(a0, time_limit, lr0=0.01, mom=0.9, top_k=40,
                      temp0=0.01, temp_growth=1.003, decay=0.9997,
                      verbose=False, restart_interval=500, restart_scale=0.005):
    """Projected subgradient descent with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = compute_obj(a)
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr0
    temp = temp0
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sgrad = subgradient_direction(a, conv, top_k=top_k, temp=temp)
        v = mom * v - lr * sgrad
        a_new = project_simplex(a + v)

        # Backtracking line search
        obj_new = compute_obj(a_new)
        if not np.isfinite(obj_new):
            a_new = a
            obj_new = best_obj
            v = v * 0.5
        elif obj_new <= best_obj + 1e-10:
            a = a_new
            conv = fast_convolve(a)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    SG iter {it}: obj={best_obj:.8f}")
            else:
                no_improve += 1
            lr = min(lr * 1.002, 0.05)
        else:
            # Backtrack
            for _ in range(10):
                lr *= 0.5
                v = v * 1.0  # keep momentum
                a_try = project_simplex(a + v)
                obj_try = compute_obj(a_try)
                if obj_try <= obj_new * 1.01:
                    a = a_try
                    conv = fast_convolve(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    break
                else:
                    v = v * 0.5
            else:
                no_improve += 1
                a = a  # no change

        temp *= temp_growth
        lr = max(lr * decay, 1e-6)

        # Restart if stuck
        if no_improve > restart_interval:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * restart_scale)
            v = np.zeros(n)
            lr = max(lr0 * 0.4, 0.002)
            temp = max(temp0 * 0.5, 0.002)
            conv = fast_convolve(a)
            no_improve = 0
            restart_scale *= 0.8

    return best_a, best_obj


def lp_direction_solve(f, M, n, active_idx=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    from scipy.optimize import linprog

    c = -np.ones(n)
    if active_idx is None:
        K = 2 * n - 1
        A_ub = np.zeros((K, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(K, M)
    else:
        K = len(active_idx)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]

    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_step(f, n, top_k=30, M_expand=1.0):
    """One LP-Frank-Wolfe step. Returns new point on simplex or None."""
    conv = fast_convolve(f)
    M = conv.max() * M_expand
    if M < 1e-15:
        return None

    active_idx = None
    if top_k < 2 * n - 1:
        active_idx = np.argpartition(conv, -top_k)[-top_k:].tolist()

    g = lp_direction_solve(f, M, n, active_idx)
    if g is None:
        return None
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs


def lp_fw_loop(f0, time_limit, top_k=35, verbose=False):
    """LP-Frank-Wolfe with adaptive M expansion and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        g = lp_fw_step(f, n, top_k=top_k)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search: try a few alpha values
        best_alpha = 0.0
        best_trial_obj = best_obj
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_try = project_simplex(f + alpha * (g - f))
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_alpha = alpha

        if best_alpha > 0:
            f = project_simplex(f + best_alpha * (g - f))
            conv = fast_convolve(f)
            M_new = conv.max()
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f.copy()
                no_improve = 0
                if verbose:
                    print(f"    LP-FW obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            break

    return best_f, best_obj


def coord_descent(a0, time_limit, allowed_multipliers=None):
    """Coordinate descent: optimize one coordinate at a time, redistributing to others."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    no_improve_passes = 0

    if allowed_multipliers is None:
        allowed_multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.2, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                # Try setting it to a small value
                for mult in [0.01, 0.05]:
                    new_val = (1.0 - old_val) * mult
                    if new_val < 1e-12:
                        continue
                    scale = (1.0 - new_val) / (1.0 - old_val) if (1.0 - old_val) > 1e-10 else 1.0
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    a_trial = a_trial / a_trial.sum()
                    obj_try = compute_obj(a_trial)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                continue

            best_j_val = old_val
            best_j_obj = compute_obj(a)

            for mult in allowed_multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                obj_try = compute_obj(a_trial)
                if obj_try < best_j_obj - 1e-15:
                    best_j_obj = obj_try
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_a, best_obj


def raised_cosine_init(n, alpha=1.0):
    """Initialize with raised cosine: a[i] = (cos(pi*(i/(n-1)-0.5))**2)**alpha."""
    if n == 1:
        return np.array([1.0])
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** (2 * alpha)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def linear_ramp_init(n, direction=1):

def super_ellipse_init(n, p=2.0):
    """Initialize with super-ellipse: (1 - (2x-1)^(2p))^(1/p)."""
    if n == 1:
        return np.array([1.0])
    x = np.linspace(-1, 1, n)
    a = np.maximum(1.0 - np.abs(x) ** (2 * p), 0) ** (1.0 / p)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def interpolate_seq(a, new_n):
    """Interpolate a sequence to a new length via linear interpolation on [0,1]."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / 1.0)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp /= s
    else:
        a_interp = np.ones(new_n) / new_n
    return a_interp

                                     top_k=50, temp0=0.02, temp_growth=1.002,
                                     decay=0.9998, restart_interval=300,
                                     restart_scale=0.004)
        try_seq(a_opt, "subgrad_parent")

    # === Phase 3: Diverse initializations ===
    print(f"\n=== Phase 3: Diverse Initializations (best={best_score:.6f}) ===")
    init_configs = [
        ("raised_cos_1.0", raised_cosine_init(n_parent, 1.0)),
        ("raised_cos_1.5", raised_cosine_init(n_parent, 1.5)),
        ("raised_cos_0.5", raised_cosine_init(n_parent, 0.5)),
        ("super_ell_2", super_ellipse_init(n_parent, 2.0)),
        ("super_ell_4", super_ellipse_init(n_parent, 4.0)),
        ("ramp_up", linear_ramp_init(n_parent, 1)),
        ("ramp_down", linear_ramp_init(n_parent, -1)),
        ("mix_parent_rc", project_simplex(0.5 * base + 0.5 * raised_cosine_init(n_parent))),
        ("mix_parent_ramp", project_simplex(0.5 * base + 0.5 * linear_ramp_init(n_parent))),
    ]

    for name, init in init_configs:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.15, 120)
        if t_alloc < 20:
            break
        # Evaluate initial
        try_seq(init, f"init_{name}")
        # Optimize
        a_opt, _ = projected_subgrad(init, time_limit=t_alloc * 0.6, lr0=0.006,
                                     mom=0.93, top_k=40, temp0=0.015,
                                     temp_growth=1.003, decay=0.9998)
        try_seq(a_opt, f"sg_{name}")

        # LP-FW on the result
        rem = deadline - time.monotonic()
        if rem > 30:
            a_opt2, _ = lp_fw_loop(a_opt, time_limit=min(rem * 0.3, 60), top_k=25)
            try_seq(a_opt2, f"lpfw_{name}")

    # === Phase 4: Try different lengths ===
    print(f"\n=== Phase 4: Different Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [512, 896, 1024, 1280]:
            if best_seq is None or len(best_seq) == target_n:
                continue
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_interp = interpolate_seq(best_seq, target_n)
            t_alloc = min(rem * 0.15, 100)
            print(f"  Length {target_n}, t={t_alloc:.0f}s")
            a_opt, _ = projected_subgrad(a_interp, time_limit=t_alloc * 0.5,
                                         lr0=0.005, mom=0.93, top_k=35,
                                         temp0=0.01, temp_growth=1.003, decay=0.9998)
            try_seq(a_opt, f"len_{target_n}_sg")

            rem = deadline - time.monotonic()
            if rem > 30:
                a_opt2, _ = lp_fw_loop(a_opt, time_limit=min(rem * 0.3, 50), top_k=25)
                try_seq(a_opt2, f"len_{target_n}_lpfw")

    # === Phase 5: Deep optimization on best ===
    print(f"\n=== Phase 5: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 50:
            # Long subgradient run
            t_sg = min(rem * 0.3, 300)
            a_opt, _ = projected_subgrad(best_seq, time_limit=t_sg, lr0=0.003,
                                         mom=0.94, top_k=60, temp0=0.005,
                                         temp_growth=1.002, decay=0.9999,
                                         restart_interval=400, restart_scale=0.003)
            try_seq(a_opt, "deep_sg")

            # LP-FW full
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lpfw = min(rem * 0.2, 150)
                a_opt2, _ = lp_fw_loop(a_opt, time_limit=t_lpfw, top_k=40)
                try_seq(a_opt2, "deep_lpfw")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.35, 300)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd1")

            rem = deadline - time.monotonic()
            if rem > 25:
                t_cd2 = min(rem * 0.25, 150)
                a_opt2, _ = coord_descent(a_opt, time_limit=t_cd2)
                try_seq(a_opt2, "cd2")

            # CD on a perturbed version
            rem = deadline - time.monotonic()
            if rem > 20:
                n_best = len(best_seq)
                a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
                t_cd3 = min(rem * 0.3, 100)
                a_opt3, _ = coord_descent(a_pert, time_limit=t_cd3)
                try_seq(a_opt3, "cd_pert")

    # === Phase 7: Final perturbations and polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for i in range(6):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            noise_scale = 0.001 * (1 + i * 0.5)
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(rem * 0.2, 60)

            # Quick subgradient
            a_opt, _ = projected_subgrad(a_pert, time_limit=t_alloc * 0.5,
                                         lr0=0.002, mom=0.94, top_k=30,
                                         temp0=0.005, temp_growth=1.002, decay=0.9999)
            try_seq(a_opt, f"final_pert_{i}")

            # Quick LP-FW
            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt2, _ = lp_fw_loop(a_opt, time_limit=min(rem * 0.3, 30), top_k=20)
                try_seq(a_opt2, f"final_lpfw_{i}")

        # Final tiny subgradient
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = projected_subgrad(best_seq, time_limit=min(rem * 0.4, 40),
                                         lr0=0.001, mom=0.95, top_k=20,
                                         temp0=0.003, temp_growth=1.001, decay=0.9999)
            try_seq(a_opt, "final_sg_tiny")

    # Fallback
    if best_seq is None:
        best_seq = raised_cosine_init(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
