def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, top_k=50):
    """Compute subgradient of max_k conv(f,f)[k] using top-k peaks.
    Returns (gradient, M) where M is the max value."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    k_max = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_max)[-k_max:]
    
    # Softmax weights on peaks
    pv = conv[peaks]
    # Use a temperature that focuses on the top peaks
    T = 0.01 * M + 1e-10
    scaled = (pv - pv.max()) / T
    w = np.exp(scaled)
    ws = w.sum()
    w = w / ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg, M

def adam_subgrad_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    n = len(f)

    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    
    # Nesterov momentum
    y = f.copy()
    
    it = 0
    no_improve = 0
    best_lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / max(1, int(time_limit / 0.01)))
        lr = best_lr * (1.0 - 0.5 * progress ** 0.5)
        
        # Compute subgradient at current point (or Nesterov lookahead)
        eval_point = y if it > 5 else f
        sg, M = compute_subgradient(eval_point, top_k=min(100, n))
        
        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)

        else:
            y = y / ys
        
        f_new = y
        new_M = fast_convolve(f_new).max()
        
        if new_M < best_obj - 1e-14:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(f, top_k=30):
    """Compute smoothed subgradient of max_k conv(f,f)[k]."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    k_act = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    T = max(0.005 * M, 1e-12)
    scaled = (pv - pv.max()) / T
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg, M, conv


def fw_direction_approx(f, M, n, num_constraints=80):
    """Solve LP with only top binding constraints (faster than full LP)."""
    conv = fast_convolve(f)
    # Find the most binding constraints
    binding = np.argsort(conv)[-num_constraints:]
    c = -np.ones(n)
    A_ub = np.zeros((len(binding), n))
    for col, j in enumerate(range(n)):
        for row, k in enumerate(binding):
            if 0 <= k - j < n:
                A_ub[row, col] = f[j]
    b_ub = np.full(len(binding), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if res.success:
        return np.maximum(res.x, 0)
    return None

        else:
            no_improve_rounds += 1
            if no_improve_rounds >= 4:
                break

    return best_f, best_obj


def block_optimize(f0, time_limit, block_size=8):
    """Block coordinate optimization with random perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(27182)

    num_blocks = (n + block_size - 1) // block_size
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 150:
            break
        improved = False
        block_order = rng.permutation(num_blocks)

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            bs = end - start
            block_sum = f[start:end].sum()
            if block_sum < 1e-12:
                continue

            for _ in range(3):
                pert = rng.standard_normal(bs) * 0.015 * block_sum
                trial_block = np.maximum(f[start:end] + pert, 0)
                ts = trial_block.sum()
                if ts < 1e-12:
                    continue
                trial_block *= (block_sum / ts)
                f_trial = f.copy()
                f_trial[start:end] = trial_block
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-13:
                    f = f_trial.copy()
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def mirror_descent_step(f, sg, lr):
    """Multiplicative weight (mirror) descent step on the simplex."""
    n = len(f)
    log_ratio = -lr * sg
    log_ratio -= log_ratio.max()  # numerical stability
    weights = f * np.exp(log_ratio)
    ws = weights.sum()
    if ws < 1e-15:
        return f.copy()
    return weights / ws


def optimize_mirror(f0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) with subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    eta = 0.5  # base step size

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        progress = min(1.0, it / max(1, time_limit / 0.01))
        lr = eta / (1 + 0.3 * it ** 0.4)
        top_k = max(10, int(40 * (1 - progress * 0.5)))

        sg, M, _ = compute_subgrad(f, top_k=top_k)
        f_new = mirror_descent_step(f, sg, lr)
        new_M = fast_convolve(f_new).max()

        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    Mirror iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
            f = f_new

        if no_improve > 200:
            rng = np.random.default_rng(it * 53 + 29)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            no_improve = 0
            M = fast_convolve(f).max()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):

        if np.isfinite(sc) and sc < best_score:

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    # === Phase 1: Adam+FW optimization (30% time) ===
    print(f"\n=== Phase 1: Adam+FW (best={best_score:.8f}) ===")

    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")

        rem = deadline - time.monotonic()

        rem = deadline - time.monotonic()
        if rem > 20:

            try_seq(a_opt, "adamfw_final")

        # Momentum technique: N consecutive pairwise passes
        rem = deadline - time.monotonic()
        if rem > 15:

    # Fallback

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
