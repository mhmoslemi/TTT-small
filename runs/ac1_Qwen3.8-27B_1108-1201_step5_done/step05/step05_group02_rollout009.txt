```python
def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max(c, tau):
    """Smooth approximation of max using log-sum-exp."""
    c_max = c.max()
    return c_max + tau * np.log(np.sum(np.exp((c - c_max)/tau)))

def smooth_max_grad(c, tau):
    """Gradient of smooth_max with respect to c."""
    c_max = c.max()
    w = np.exp((c - c_max)/tau)
    return w / w.sum()

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
import numpy as np

def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z = z - z.max()
    ez = np.exp(z)
    return ez / ez.sum()

def grad_of_smooth_max_conv(a, w, fft_size):
    """
    Compute gradient of smooth_max(conv(a,a), tau) w.r.t. a.
    w: weights (softmax of conv/tau), length 2n-1
    Returns gradient of length n.
    """
    n = len(a)
    size = 2 * n - 1
    # Need fft_size >= (2n-2) + n = 3n-2 for linear cross-correlation
    fs = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fs)
    A = np.fft.rfft(a, fs)
    xcorr = np.fft.irfft(W * np.conj(A), fs)
    # xcorr[j] = sum_n w[n+j] * a[n] for j = 0, 1, ...
    return 2.0 * xcorr[:n]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def evaluate_fast(a):

    a = np.maximum(a, 0.0)

    a_norm = a / s

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Decay temperature
        tau *= tau_decay
        
        # Periodically check exact objective
        if t % 100 == 0:
            a_check = softmax(theta)
            c_check = fast_convolve(a_check)
            obj = c_check.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_theta = theta.copy()
                no_improve = 0
                if verbose and t % 1000 == 0:
                    print(f"      Adam t={t}: obj={obj:.10f}, tau={tau:.2e}")
            else:
                no_improve += 1
        
        if no_improve > 5000:
            # Restart with perturbation
            theta = best_theta + np.random.randn(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            tau = max(tau, tau_init * 0.1)
    
    a_best = softmax(best_theta)
    return a_best, best_obj

def optimize_projected_sg(a0, time_limit, lr_init=0.005, top_k=50, 
                          verbose=False):
    """
    Projected subgradient descent with top-k peak weighting.
    Works directly with a (simplex constraint).
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # momentum
    momentum = 0.9
    lr = lr_init
    temp = 0.0001
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Top-k peaks
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        pv = c[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Gradient
        fs = 1 << (3 * n - 2).bit_length()
        W = np.fft.rfft(w, fs)
        A = np.fft.rfft(a, fs)
        xcorr = np.fft.irfft(W * np.conj(A), fs)
        grad = 2.0 * xcorr[:n]
        
        # Momentum step
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        c = fast_convolve(a_new)
        obj = c.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= 0.9999
        temp *= 1.002
        
        if no_improve > 300:
            a = project_simplex(best_a + np.random.randn(n) * 0.001)
            v = np.zeros(n)
            lr = lr_init * 0.3
            no_improve = 0
    
    return best_a, best_obj

def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate in the top peak contributors,
    try to find the optimal value.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        # Find top 3 peaks
        top_k = min(3, len(c))
        peak_positions = np.argpartition(c, -top_k)[-top_k:]
        
        improved_this_round = False
        
        for pk in peak_positions:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            
            # For each coordinate that contributes to this peak, try to optimize it
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                if a[j] < 1e-12:
                    continue
                
                # Try a few multipliers
                best_val = a[j]
                best_o = best_obj
                for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                    new_val = a[j] * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    # Adjust others to maintain sum=1
                    others_sum = 1.0 - new_val
                    old_others_sum = 1.0 - a[j]
                    if old_others_sum < 1e-10:
                        continue
                    scale = others_sum / old_others_sum
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    
                    o = fast_convolve(a_trial).max()
                    if o < best_o - 1e-14:
                        best_o = o
                        best_val = new_val
                
                if best_val != a[j]:
                    old_others_sum = 1.0 - a[j]
                    others_sum = 1.0 - best_val
                    if old_others_sum > 1e-10:
                        scale = others_sum / old_others_sum
                        a[j] = best_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        a[mask] *= scale
                        a = a / a.sum()
                    if best_o < best_obj - 1e-14:
                        best_obj = best_o
                        best_a = a.copy()
                        improved_this_round = True
        
        if not improved_this_round:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def try_sequence(a, best_state):

    n = len(a_norm)

    if score < best_state[0]:
        best_state[0] = score
        best_state[1] = a_norm.copy()
        print(f"    NEW BEST: score={score:.8f}, n={n}")
```

def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
