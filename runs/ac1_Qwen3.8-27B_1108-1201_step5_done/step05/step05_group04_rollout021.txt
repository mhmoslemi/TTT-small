import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient FFT(a, w):
    """Compute 2 * sum_j a[j] * w[m+j] for m=0..n-1 using FFT."""
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A * W, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_objective(f, beta):
    """Compute LSE-smoothed max of auto-convolution and the weight vector."""
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    return conv.max(), w, conv


def backtracking_line_search(f, direction, fv, c1=1e-4, rho=0.5, max_iter=30):
    """Backtracking line search for LSE-smoothed objective.
    Returns (alpha, f_new, w_new, conv_new).
    """
    n = len(f)
    # We track the actual (non-smoothed) objective: max of conv
    # But use LSE for smoothness. Use LSE value for the Armijo condition.
    # Actually, let's just use the raw max objective for the check.
    best_alpha = 0.0
    best_val = fv
    alpha = 1.0
    for _ in range(max_iter):
        f_trial = np.maximum(f + alpha * direction, 0.0)
        s = f_trial.sum()
        if s < 1e-15:
            alpha *= rho
            continue
        f_trial /= s
        val = fast_convolve(f_trial).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
        if val <= fv - c1 * alpha * np.dot(-direction, -direction) * 0.5:
            # Satisfied Armijo-like condition (loose version)
            break
        alpha *= rho
    if best_alpha == 0.0:
        best_alpha = 1e-6
    f_new = np.maximum(f + best_alpha * direction, 0.0)
    f_new /= f_new.sum()
    conv_new = fast_convolve(f_new)
    return best_alpha, f_new, conv_new.max(), conv_new


def lse_gradient_descent(f0, time_limit, beta_start=50.0, beta_end=10000.0,
                         lr_init=0.5, momentum_init=0.9):
    """LSE-smoothed gradient descent with backtracking and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_beta = beta_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()

        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # FFT-based gradient
        grad = compute_gradient_fft(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad /= grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        val_new = fast_convolve(f_new).max()

        if val_new < best_obj - 1e-14:
            best_obj = val_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0

            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(0.1, lr * 0.5)
                beta = min(beta * 1.5, beta_end)
                best_beta = beta
                momentum = 0.88
                no_improve = 0

        # Gradually increase beta
        beta = min(beta * 1.008, beta_end)

    return best_f, best_obj


def single_peak_descent(f0, time_limit, n_peaks=1, lr_init=0.3, momentum_init=0.92):
    """Focused descent on the top n_peaks of the auto-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = lr_init
    moment = momentum_init
    v = np.zeros(n)

        it += 1
        conv = fast_convolve(f)

        pv = conv[peaks]

        if scaled.max() < 1e-15:
            w_peaks = np.ones(k_top) / k_top
        else:

        w = np.zeros(conv_len)
        w[peaks] = w_peaks

        grad = compute_gradient_fft(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad /= grad_norm

        v = moment * v - lr * grad
        f_new = project_simplex(f + v)
        val_new = fast_convolve(f_new).max()

        if val_new < best_obj - 1e-14:

            best_f = f_new.copy()

                lr *= 0.95
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(0.01, lr * 0.5)
                moment = min(0.96, moment + 0.01)
                no_improve = 0

    return best_f, best_obj

def multi_start_local(f0, time_limit, n_restarts=3, perturbation=0.01):
    """Try multiple perturbed starts and return the best."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    per_restart = time_limit / (n_restarts + 1)

    # First: optimize the original
    f_opt, obj = lse_gradient_descent(f, time_limit=per_restart,
                                       beta_start=50, beta_end=5000,
                                       lr_init=0.5, momentum_init=0.9)

        best_obj = obj
        best_f = f_opt

        # Perturb the best found so far

        if obj < best_obj:

def push_pull_step(f, peak_pos, strength=0.5):
    """Directly modify f to reduce the peak at peak_pos.
    The peak (a*a)[peak_pos] = sum_j a[j]*a[peak_pos-j].
    We reduce entries a[j] where a[peak_pos-j] is large.
    """
    n = len(f)
    f_new = f.copy()
    # Compute contributions to the peak
    lo = max(0, peak_pos - n + 1)
    hi = min(n - 1, peak_pos)
    if lo > hi:
        return f_new
    js = np.arange(lo, hi + 1)
    contribs = f[js] * f[peak_pos - js]
    # Reduce entries proportional to their contribution
    max_c = contribs.max()
    if max_c < 1e-15:
        return f_new
    reduction = strength * max_c * np.ones(len(js))
    f_new[js] -= reduction * 0.5
    f_new[js] = np.maximum(f_new[js], 0)
    f_new /= f_new.sum()
    return f_new


def try_sequence(a, best_tracker):

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def symmetrize(f):
    """Make the sequence palindromic (symmetric)."""
    n = len(f)
    f_sym = 0.5 * (f + f[::-1])
    f_sym = np.maximum(f_sym, 0)
    s = f_sym.sum()
    if s > 1e-15:
        f_sym /= s
    return f_sym

    t_alloc = max(60, t4_end - time.monotonic())
    if t_alloc > 30:
        n_restarts = 3
        print(f"  {n_restarts} restarts, t={t_alloc:.0f}s")
        f_opt, obj = multi_start_local(current, time_limit=t_alloc,
                                        n_restarts=n_restarts, perturbation=0.008)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1].copy() if best_tracker[1] is not None else current

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    t5_end = deadline - 150
    n_best = len(current)

    # Try nearby lengths
    offsets = [64, -64, 128, -128, 32, -32, 256, -256]
    tried_lengths = set()
    for off in offsets:
        n_try = n_best + off
        if n_try in tried_lengths or n_try < 128 or n_try > 2048:
            continue
        t_remaining = t5_end - time.monotonic()
        if t_remaining < 40:
            break
        t_alloc = min(80, t_remaining * 0.4)
        if t_alloc < 20:
            break
        tried_lengths.add(n_try)

        f_init = interpolate_to_length(current, n_try)
        print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
        f_opt, obj = lse_gradient_descent(f_init, time_limit=t_alloc,
                                           beta_start=50, beta_end=5000,
                                           lr_init=0.5, momentum_init=0.9)
        try_sequence(f_opt, best_tracker)
        # Quick single-peak follow-up
        t_sp = min(30, t5_end - time.monotonic() - 20)
        if t_sp > 15:
            f_opt2, _ = single_peak_descent(f_opt, time_limit=t_sp,
                                             n_peaks=1, lr_init=0.3, momentum_init=0.93)
            try_sequence(f_opt2, best_tracker)

    current = best_tracker[1].copy() if best_tracker[1] is not None else current

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polishing ===")
    current = best_tracker[1].copy() if best_tracker[1] is not None else current
    n_final = len(current)
    t_remaining = deadline - time.monotonic()
    print(f"  Final phase, n={n_final}, time={t_remaining:.0f}s")

    # Very high beta LSE
    t_alloc = max(20, t_remaining * 0.4)
    if t_alloc > 15:
        print(f"  High-beta LSE (300->20000), t={t_alloc:.0f}s")
        f_opt, obj = lse_gradient_descent(current, time_limit=t_alloc,
                                           beta_start=300, beta_end=20000,
                                           lr_init=0.3, momentum_init=0.95)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1].copy() if best_tracker[1] is not None else current

    # Single peak with very high momentum
    t_alloc = max(20, deadline - time.monotonic() - 60)
    if t_alloc > 15:
        print(f"  Single-peak high-momentum, t={t_alloc:.0f}s")
        f_opt, obj = single_peak_descent(current, time_limit=t_alloc,
                                          n_peaks=1, lr_init=0.2, momentum_init=0.96)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1].copy() if best_tracker[1] is not None else current

    # Final oscillation-aware pass
    t_alloc = max(10, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  Final oscillation-aware, t={t_alloc:.0f}s")
        f_opt, obj = oscillation_aware_descent(current, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using parent fallback")
    return [float(x) for x in parent]
