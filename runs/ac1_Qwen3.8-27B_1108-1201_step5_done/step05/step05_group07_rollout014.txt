"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent with FFT-based gradient (O(n log n) per step)
2. Adaptive temperature scheduling for progressive peak focus
3. Peak-directed coordinate descent for final polishing
4. Multi-length exploration with asymmetric initializations
5. Simulated annealing restarts to escape local minima
Key innovation: FFT-based gradient of the log-sum-exp smoothed objective,
giving a dense gradient that attacks all peaks proportionally, combined

import numpy as np
import time
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_fft(a, t, conv=None):
    """
    Compute gradient of t*log(sum(exp(conv/t))) w.r.t. a using FFT.
    Result: grad[i] = 2 * sum_j w[i+j] * a[j] where w = softmax(conv/t)
    This uses the identity: crosscorr(w, a)[i] = conv(w, a_rev)[(n-1)+i]
    """
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    w = np.exp(conv_shifted / t)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    total_len = 2 * n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a_rev, fft_size)
    cross = np.fft.irfft(W * A, fft_size)[:total_len]
    grad = 2.0 * cross[n - 1:2 * n - 1]
    return grad


def smooth_mirror_descent(f0, time_limit, t_init=0.5, t_min=0.001,
                          lr_init=0.05, momentum=0.7, verbose=False, label=""):
    """
    Mirror descent on simplex with log-sum-exp smoothed objective.
    Uses multiplicative update: f *= exp(-lr * grad), then normalize.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    cur_t = t_init
    cur_lr = lr_init
    no_improve = 0
    it = 0
    prev_obj = smooth_eval(f, cur_t)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        conv = fast_convolve(f)
        grad = smooth_grad_fft(f, cur_t, conv)

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad

        # Adaptive learning rate
        v_norm = np.sqrt(np.sum(v * v))
        if v_norm > 1e-15:
            v = v / v_norm

        # Multiplicative update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f_new = f * np.exp(exponent)
        fs = f_new.sum()
        if fs < 1e-30:
            f_new = best_f.copy()
            v = np.zeros(n)
            cur_lr *= 0.5
            continue
        f = f_new / fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        # Evaluate
        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_t *= 0.995
        if cur_t < t_min:
            cur_t = t_min
        cur_lr *= 0.999
        if cur_lr < 0.001:
            cur_lr = 0.001

        # Annealing restart
        if no_improve > 200:
            rng = np.random.default_rng(hash((it, int(time.monotonic()))) % 2**31)
            noise_scale = 0.003 + 0.002 * min(no_improve // 200, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = min(cur_lr * 2.0, lr_init)
            cur_t = max(cur_t * 2.0, t_init * 0.1)
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{label}] iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, temp={cur_t:.6f}")

    return best_f, best_obj


def smooth_eval(f, t):
    """Evaluate the smoothed objective (for internal use)."""
    conv = fast_convolve(f)
    cs = conv - conv.max()
    return conv.max() + t * np.log(np.sum(np.exp(cs / t)))


def peak_coordinate_descent(f0, time_limit, verbose=False, label=""):
    """
    Peak-directed coordinate descent:
    1. Find the max peak at lag k*
    2. Identify top contributors to that peak
    3. Try moving mass from high contributors to low-impact positions
    4. Use binary search for optimal transfer amount
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        current_max = conv[k_star]

        # Find contributors to peak at k_star
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        partners = k_star - idx
        contribs = f[idx] * f[partners]

        if len(contribs) == 0:
            break

        # Sort by contribution (descending)
        order = np.argsort(contribs)[::-1]
        top_indices = idx[order[:20]]
        top_partners = partners[order[:20]]

        improved = False
        rng = np.random.default_rng(hash(int(time.monotonic() * 1e6)) % 2**31)

        # Try moving mass from top contributors
        for ii in range(min(10, len(top_indices))):
            i = int(top_indices[ii])
            p = int(top_partners[ii])
            if f[i] < 1e-10:
                continue

            # Try moving some mass from i to a random low-impact position
            for attempt in range(5):
                # Choose destination: prefer positions not contributing to current peak
                j = int(rng.integers(0, n))
                if j == i or j == p:
                    continue

                # Binary search on transfer amount
                best_frac = 0.0
                best_test_obj = current_max
                lo, hi = 0.01, min(0.5, f[i] / (f.sum() - f[i] + 1e-15))
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    amt = f[i] * mid
                    f[i] -= amt
                    f[j] += amt
                    f /= f.sum()
                    test_obj = fast_convolve(f).max()
                    # Restore
                    f[j] -= amt
                    f[i] += amt
                    f /= f.sum()

                    if test_obj < best_test_obj - 1e-13:
                        best_test_obj = test_obj
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid

                if best_frac > 0.005:
                    amt = f[i] * best_frac
                    f[i] -= amt
                    f[j] += amt
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                    new_obj = fast_convolve(f).max()
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_f = f.copy()
                        improved = True

        # Also try moving mass TO the peak contributors from other locations
        # (sometimes adding mass to a small partner reduces the peak product)
        for ii in range(min(5, len(top_indices))):
            p = int(top_partners[ii])
            if f[p] < 1e-10 or f[p] > 0.5:
                continue
            j = int(rng.integers(0, n))
            if j == p or f[j] < 1e-10:
                continue
            for frac in [0.1, 0.05, 0.2]:
                amt = f[j] * frac
                f[j] -= amt
                f[p] += amt
                f = np.maximum(f, 0.0)
                f /= f.sum()
                test_obj = fast_convolve(f).max()
                if test_obj < best_obj - 1e-13:
                    best_obj = test_obj
                    best_f = f.copy()
                    improved = True
                else:
                    f[p] -= amt
                    f[j] += amt
                    f /= f.sum()
                    break

        if not improved:
            # Try random pair exchanges
            any_improve = False
            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    amt = f[i] * frac
                    f[i] -= amt
                    f[j] += amt
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                    test_obj = fast_convolve(f).max()
                    if test_obj < best_obj - 1e-13:
                        best_obj = test_obj
                        best_f = f.copy()
                        any_improve = True
                        improved = True
                        break
                    else:
                        f[j] -= amt
                        f[i] += amt
                        f /= f.sum()
                        break
            if not any_improve:
                break

        if verbose and (it % 50 == 0) if 'it' in dir() else False:
            pass

    return best_f, best_obj


def asymmetric_inits(n, rng, count=6):
    """Generate diverse asymmetric initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    # Exponential decay (mass on left)
    e1 = np.exp(-3.0 * x)
    inits.append(e1 / e1.sum())

    # Exponential growth (mass on right)
    e2 = np.exp(3.0 * x)
    inits.append(e2 / e2.sum())

    # Bipartite: high on left half, low on right
    bip = np.ones(n)
    bip[:n // 2] = 1.0 + rng.random() * 2.0
    bip[n // 2:] = 0.1 + rng.random() * 0.5
    inits.append(bip / bip.sum())

    # Sine-modulated
    sine = 1.0 + 0.8 * np.sin(np.pi * x + rng.random() * np.pi)
    sine = np.maximum(sine, 0.01)
    inits.append(sine / sine.sum())

    # Quadratic peak off-center
    center = n * (0.25 + rng.random() * 0.5)
    sigma = n * (0.1 + rng.random() * 0.15)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(5, n // 20)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Cosine window (asymmetric shift)
    shift = rng.random() * 0.3
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * (x - shift)))
    cos_w = np.maximum(cos_w, 0.01)
    inits.append(cos_w / cos_w.sum())

    return inits[:count]


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        if alt % 3 == 0:
            # Smoothed descent, very low temp
            a_opt, obj = smooth_mirror_descent(f, time_limit=chunk,
                                                t_init=0.01, t_min=0.001,
                                                lr_init=0.02, momentum=0.8,
                                                label="polish_sg")
        elif alt % 3 == 1:
            # Peak coordinate descent
            a_opt, obj = peak_coordinate_descent(f, time_limit=chunk, label="polish_cd")
        else:
            # Very aggressive: focus only on top few peaks
            t_very_low = max(0.001, 0.01 / (alt + 1))
            a_opt, obj = smooth_mirror_descent(f, time_limit=chunk,
                                                t_init=t_very_low, t_min=t_very_low * 0.1,
                                                lr_init=0.01, momentum=0.85,
                                                label="polish_aggr")

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = a_opt.copy()
            f = a_opt.copy()
        else:
            # Keep f as is or slightly perturb
            rng = np.random.default_rng(hash(alt) % 2**31)
            f = project_simplex(a_opt + rng.standard_normal(n) * 0.001)

        alt += 1

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining smoothed mirror descent,
    peak-directed coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = [float(x) for x in a]
        score = float(evaluate_sequence(a))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = np.array(a, dtype=np.float64)
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Smoothed mirror descent on parent and inits (50% time) ===
    print(f"\n=== Phase 1: Smoothed Mirror Descent ===")
    phase1_budget = (deadline - time.monotonic()) * 0.45
    candidates = [parent]
    candidates.extend(asymmetric_inits(n_parent, rng, count=4))

    # Also try a slightly different length
    if n_parent != 1024:
        candidates.append(interpolate_sequence(parent, 1024))

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - (budget_s - 20.0) * 0.3
        if rem < 5:
            break
        t_alloc = min(rem * 0.2, phase1_budget / (len(candidates) - idx), 80)
        if t_alloc < 5:
            t_alloc = 5

        n_cand = len(cand)
        t_init = max(0.01, n_cand / 5000.0)
        print(f"  Cand {idx}: n={n_cand}, t={t_alloc:.0f}s, t_init={t_init:.4f}")

        a_opt, obj = smooth_mirror_descent(cand, time_limit=t_alloc,
                                            t_init=t_init, t_min=0.001,
                                            lr_init=0.04, momentum=0.75,
                                            verbose=False, label=f"p1_{idx}")
        try_seq(a_opt, f"phase1_{idx}")

    # === Phase 2: Deep optimization on best found (25% time) ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.6f}) ===")
    phase2_budget = (deadline - time.monotonic()) * 0.30
    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {"t_init": 0.02, "t_min": 0.002, "lr": 0.03, "mom": 0.8},
            {"t_init": 0.05, "t_min": 0.005, "lr": 0.05, "mom": 0.7},
            {"t_init": 0.01, "t_min": 0.001, "lr": 0.02, "mom": 0.85},
            {"t_init": 0.1, "t_min": 0.01, "lr": 0.06, "mom": 0.6},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - (budget_s - 20.0) * 0.15
            if rem < 5:
                break
            t_alloc = min(rem * 0.25, phase2_budget / (len(configs) - p), 60)
            if t_alloc < 3:
                continue

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                # Blend with uniform
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                # Reverse blend (exploits asymmetry)
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)

            print(f"  Deep {p}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, obj = smooth_mirror_descent(f_init, time_limit=t_alloc,
                                                t_init=cfg["t_init"], t_min=cfg["t_min"],
                                                lr_init=cfg["lr"], momentum=cfg["mom"],
                                                label=f"deep_{p}")
            try_seq(a_opt, f"phase2_{p}")

    # === Phase 3: Peak coordinate descent (15% time) ===
    print(f"\n=== Phase 3: Peak Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - (budget_s - 20.0) * 0.10
        if rem > 10:
            t_pcd = min(rem * 0.7, 120)
            print(f"  Peak CD: t={t_pcd:.0f}s")
            a_opt, obj = peak_coordinate_descent(best_seq, time_limit=t_pcd, label="pcd")
            try_seq(a_opt, "phase3_pcd")

    # === Phase 4: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [n_parent, 1024, 1280, 640]
        # Remove duplicates and sort by proximity to n_parent
        target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:3]:
            rem = deadline - time.monotonic() - (budget_s - 20.0) * 0.05
            if rem < 15:
                break
            if target_n == n_parent:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 60)
            if t_alloc < 5:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = smooth_mirror_descent(f_interp, time_limit=t_alloc,
                                                t_init=0.03, t_min=0.002,
                                                lr_init=0.03, momentum=0.8,
                                                label=f"len_{target_n}")
            try_seq(a_opt, f"phase4_n{target_n}")

    # === Phase 5: Final polish (5% time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5.0
        if rem > 10:
            t_polish = min(rem, 60)
            print(f"  Polish: t={t_polish:.0f}s")
            a_opt, obj = final_polish(best_seq, time_limit=t_polish, label="final")
            try_seq(a_opt, "phase5_polish")

    # === Phase 6: Last-second attempts ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 2.0
        if rem > 5:
            n_best = len(best_seq)
            # Quick peak descent
            a_opt, obj = peak_coordinate_descent(best_seq, time_limit=min(rem, 15), label="last")
            try_seq(a_opt, "phase6_last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
