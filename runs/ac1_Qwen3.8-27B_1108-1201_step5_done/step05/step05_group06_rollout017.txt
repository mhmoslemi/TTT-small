import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    if old_n < 4:
        f_interp = np.interp(new_x, old_x, f)
    else:
        from scipy.interpolate import PchipInterpolator
        try:
            interp = PchipInterpolator(old_x, f, extrapolate=False)
            f_interp = interp(new_x)
        except Exception:
            f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def gs_spectral_step(a, target_ratio=0.99, iterations_per_call=200):
    """
    Gerchberg-Saxton style spectral shaping to reduce autocorrelation peak.
    Works by modifying the power spectrum to flatten the autocorrelation,
    then projecting back to non-negative time domain.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(iterations_per_call):
        A = np.fft.fft(a, N_fft)
        P = np.abs(A) ** 2
        c = np.fft.ifft(P).real[:N]

        # Find current peak
        c_max = c.max()
        if c_max < 1e-15:
            break

        # Create target: clip the autocorrelation to reduce peak
        T = c_max * target_ratio
        c_mod = c.copy()
        c_mod[c_mod > T] = T

        # Preserve the sum (must equal (sum a)^2 = 1)
        sum_c = c_mod.sum()
        if sum_c > 1e-15:
            c_mod *= 1.0 / sum_c

        # Ensure c_mod is non-negative (autocorrelation of non-negative seq should be)
        c_mod = np.maximum(c_mod, 0)
        sum_c2 = c_mod.sum()
        if sum_c2 > 1e-15:
            c_mod *= 1.0 / sum_c2

        # New power spectrum
        P_new = np.fft.fft(c_mod, N_fft)
        P_new = np.abs(P_new)

        # Update: new amplitude, keep phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)

        # Back to time domain
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj

        return a
    a = a / s

    N_fft = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.fft(a, N_fft)
    P = np.abs(A) ** 2

    # Blend with flat spectrum
    P_flat = np.ones_like(P)
    P_flat[0] = P[0]  # preserve DC
    P_new = (1 - alpha) * P + alpha * P_flat

    # New amplitude, keep phase
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a
    a_new /= s_new
    return a_new


def structured_perturbation(a, mode=0):
    """Apply structured perturbations based on sequence structure."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s

    if mode == 0:
        # Shift the mass distribution
        shift = np.random.default_rng(int(time.monotonic() * 1000)).integers(-3, 4)
        a = np.roll(a, shift)
        # Wrap-around creates artifacts, so blend with original
        a = 0.7 * a + 0.3 * a.copy()
    elif mode == 1:
        # Widen/narrow the effective window
        x = np.arange(n)
        center = np.argmax(a)
        width = np.sqrt(np.sum((x - center)**2 * a) / np.sum(a))
        new_width = width * np.random.default_rng(int(time.monotonic() * 1000)).uniform(0.8, 1.2)
        gauss = np.exp(-0.5 * ((x - center) / new_width) ** 2)
        a = 0.5 * a + 0.5 * gauss / gauss.sum()
    elif mode == 2:
        # Invert the shape (if it's peaked, make it more uniform)
        a = 0.6 * a + 0.4 * np.ones(n) / n
    elif mode == 3:
        # Asymmetric perturbation
        ramp = np.linspace(0.5, 1.5, n)
        a = a * ramp
    else:
        # Random smooth perturbation
        rng = np.random.default_rng(int(time.monotonic() * 1000))
        noise = rng.standard_normal(n)
        # Smooth the noise
        kernel = np.ones(5) / 5
        noise = np.convolve(noise, kernel, mode='same')
        a = a + 0.01 * noise

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def combined_optimize(a0, time_limit, verbose=False):
    """
    Combined optimization: coordinate bisection + spectral shaping + restarts.
    More aggressive than previous version.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    phase = 0
    cd_rounds_no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        if phase % 6 == 0:
            # Main coordinate bisection
            t_alloc = min(30, remaining)
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=False)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                cd_rounds_no_improve = 0
                if verbose:
                    print(f"  Phase {phase} CD: obj={best_obj:.10f}")
            else:
                cd_rounds_no_improve += 1

        elif phase % 6 == 1:
            # Spectral shaping (GS step)
            t_alloc = min(15, remaining)
            a_gs, obj_gs = gs_spectral_step(best_a, target_ratio=0.995, iterations_per_call=500)
            if obj_gs < best_obj - 1e-14:
                best_obj = obj_gs
                best_a = a_gs
                if verbose:
                    print(f"  Phase {phase} GS: obj={best_obj:.10f}")
            # Follow up with CD
            if remaining > 20:
                a_opt, obj = coordinate_bisection(best_a, time_limit=min(10, remaining - 5))
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_opt

        elif phase % 6 == 2:
            # Spectral shaped restart + CD
            t_alloc = min(25, remaining)
            alpha = 0.2 + 0.1 * (phase // 6)
            a_restart = spectral_shaped_restart(best_a, alpha=min(alpha, 0.7))
            a_opt, obj = coordinate_bisection(a_restart, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} SpectralRestart: obj={best_obj:.10f}")

        elif phase % 6 == 3:
            # Structured perturbation + CD
            t_alloc = min(25, remaining)
            mode = (phase // 6) % 4
            a_pert = structured_perturbation(best_a, mode=mode)
            a_opt, obj = coordinate_bisection(a_pert, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} Pert{mode}: obj={best_obj:.10f}")

        elif phase % 6 == 4:
            # Long CD run
            t_alloc = min(40, remaining)
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=False)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                cd_rounds_no_improve = 0
                if verbose:
                    print(f"  Phase {phase} CD-long: obj={best_obj:.10f}")
            else:
                cd_rounds_no_improve += 1

        else:
            # Aggressive spectral modification
            t_alloc = min(20, remaining)
            a_gs, obj_gs = gs_spectral_step(best_a, target_ratio=0.98, iterations_per_call=1000)
            if obj_gs < best_obj - 1e-14:
                best_obj = obj_gs
                best_a = a_gs
                if verbose:
                    print(f"  Phase {phase} GS-aggr: obj={best_obj:.10f}")
            if remaining > 10:
                a_opt, obj = coordinate_bisection(best_a, time_limit=min(10, remaining - 3))
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_opt

        phase += 1

    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining coordinate bisection, spectral shaping,
    structured restarts, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_try)})")

    # === Phase 1: Intensive coordinate bisection on parent ===
    print(f"=== Phase 1: CD on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 500)
        print(f"  CD, t={t_alloc:.0f}s")
        a_opt, obj = coordinate_bisection(parent, time_limit=t_alloc, verbose=True)
        update_best(a_opt, "cd_parent")

    # === Phase 2: Combined optimization (CD + spectral + restarts) ===
    print(f"\n=== Phase 2: Combined (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem, 350)
            print(f"  Combined, t={t_alloc:.0f}s")
            a_opt, _ = combined_optimize(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "combined")

    # === Phase 3: Spectral shaping exploration ===
    print(f"\n=== Phase 3: Spectral (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 20:
            # Try different alpha values for spectral flattening
            for alpha in [0.1, 0.2, 0.35, 0.5]:
                rem = deadline - time.monotonic() - 400
                if rem < 15:
                    break
                a_spec = spectral_shaped_restart(best_a, alpha=alpha)
                t_alloc = min(rem, 60)
                a_opt, obj = coordinate_bisection(a_spec, time_limit=t_alloc)
                update_best(a_opt, f"spec_a{alpha}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent

    target_lengths = [640, 768, 896, 1024, 1280]
    for target_n in target_lengths:
        if target_n == n_parent:
            continue
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.4, 120)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(f_interp, time_limit=t_alloc, verbose=False)
        update_best(a_opt, f"n{target_n}")

    # === Phase 5: Symmetric variant ===
    print(f"\n=== Phase 5: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 100)
            a_opt, _ = combined_optimize(a_sym, time_limit=t_alloc, verbose=False)
            update_best(a_opt, "symmetric")

    # === Phase 6: Deep optimization with diverse restarts ===
    print(f"\n=== Phase 6: Deep restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restart_strategies = [
            ("micro_noise", lambda: project_simplex(best_a + rng.standard_normal(n_best) * 0.002)),
            ("mid_noise", lambda: project_simplex(best_a + rng.standard_normal(n_best) * 0.008)),
            ("spec_flat", lambda: spectral_shaped_restart(best_a, 0.25)),
            ("smooth_perturb", lambda: structured_perturbation(best_a, mode=1)),
            ("asym_shift", lambda: structured_perturbation(best_a, mode=0)),
            ("ramp_mix", lambda: project_simplex(0.85 * best_a + 0.15 * np.linspace(0.5, 1.5, n_best) / np.linspace(0.5, 1.5, n_best).sum())),
        ]

        for idx, (label, init_fn) in enumerate(restart_strategies):
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_alloc = min(rem, 100)
            f_init = init_fn()
            a_opt, _ = combined_optimize(f_init, time_limit=t_alloc, verbose=False)
            update_best(a_opt, f"deep_{label}")

    # === Phase 7: Final intensive coordinate bisection ===
    print(f"\n=== Phase 7: Final CD polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 300)
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "final_cd")

            # Micro-perturbation + CD
            rem = deadline - time.monotonic() - 15
            if rem > 10 and best_a is not None:
                n_best = len(best_a)
                for p in range(5):
                    rem = deadline - time.monotonic() - 10
                    if rem < 8:
                        break
                    rng_p = np.random.default_rng(seed + 2000 + p * 7)
                    scale = 0.0005 * (p + 1)
                    f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * scale)
                    t_alloc = min(rem, 12)
                    a_opt, _ = coordinate_bisection(f_pert, time_limit=t_alloc)
                    update_best(a_opt, f"micro_{p}")

            # One more spectral + CD
            rem = deadline - time.monotonic() - 10
            if rem > 8 and best_a is not None:
                a_gs, _ = gs_spectral_step(best_a, target_ratio=0.999, iterations_per_call=500)
                t_alloc = min(rem, 8)
                a_opt, _ = coordinate_bisection(a_gs, time_limit=t_alloc)
                update_best(a_opt, "final_spec")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
