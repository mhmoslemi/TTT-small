import time

def fast_auto_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Compute gradient of T * log(sum(exp(conv/T))) w.r.t. a.
    Uses FFT for O(n log n) computation.

    conv = fast_auto_convolve(a)

    scaled = conv / T
    scaled -= scaled.max()

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using log-sum-exp smoothed
projected gradient descent with Nesterov momentum, diverse multi-scale
initializations, top-k peak refinement, and coordinate descent polishing.

import numpy as np
import time


def fast_auto_convolve(a):
    """Linear auto-convolution a*a via FFT. Returns array of length 2*n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, T):
    """
    Compute gradient of T * log(sum(exp(conv(a,a)/T))) w.r.t. a.
    O(n log n) via FFT.
    """
    n = len(a)
    conv = fast_auto_convolve(a)
    
    # Temperature-scaled softmax
    scaled = conv / T
    scaled -= scaled.max()
    s = np.exp(scaled)
    s_sum = s.sum()
    if s_sum < 1e-300:
        s = np.ones(len(conv)) / len(conv)
    else:
        s /= s_sum
    
    # grad[m] = 2 * sum_j s[m+j] * a[j] for j=0..n-1, m+j < 2n-1
    # Compute via FFT: conv(s, a_rev)[m + n - 1]
    a_rev = a[::-1]
    # Linear conv length: (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 3).bit_length()
    S = np.fft.rfft(s, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_sa = np.fft.irfft(S * AR, fft_size)
    
    grad = 2.0 * conv_sa[n - 1 : n - 1 + n]
    return grad

    cur_lr = lr

    v = np.zeros(n)
    beta = momentum
    
    conv = fast_auto_convolve(a)
    best_obj = float(conv.max())
    best_a = a.copy()
