"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
improved LP-Frank-Wolfe, log-sum-exp smoothed gradient descent with momentum,
fine coordinate descent, multi-length search, and structured restarts.
Key innovations: FFT-based LSE gradient, adaptive temperature scheduling,
Nesterov-like momentum, and diverse initialization strategies.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None or len(active_k) >= 2 * n - 1:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def golden_section_line(f, g, n, max_iter=20):
    """Golden section search for best alpha mixing f and g on simplex."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    f_c = project_simplex(f + c * (g - f))
    f_d = project_simplex(f + d * (g - f))
    fc = fast_convolve(f_c).max()
    fd = fast_convolve(f_d).max()
    for _ in range(max_iter):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - phi * (hi - lo)
            f_c = project_simplex(f + c * (g - f))
            fc = fast_convolve(f_c).max()
        else:
            lo, c, fc = c, d, fc
            d = lo + phi * (hi - lo)
            f_d = project_simplex(f + d * (g - f))
            fd = fast_convolve(f_d).max()
    return (lo + hi) / 2.0

    """Compute gradient of (1/q)*log(sum exp(q*conv_k)) w.r.t. f using FFT."""
    n = len(f)
    conv = fast_convolve(f)

        w[np.argmax(conv)] = 1.0
    else:
        w = exp_scaled / Z

    # Need N >= (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(f, N)
    W = np.fft.rfft(w, N)
    # cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    # In real FFT terms: we need the full complex version
    A_full = np.fft.fft(f, N)
    W_full = np.fft.fft(w, N)
    corr = np.fft.ifft(W_full * np.conj(A_full), N)
    grad = 2.0 * corr[:n].real
    return grad, conv


def lp_fw_optimize(f0, time_limit, verbose=False, top_k_constraints=35):
    """LP-based Frank-Wolfe with adaptive constraints and golden section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        active_k = None
        if it > 5 and top_k_constraints > 0:
            k_sorted = np.argsort(conv)[::-1][:top_k_constraints]
            active_k = k_sorted.tolist()
        if it % 50 == 0:
            active_k = None

        g = solve_lp_direction(f, M * m_exp, n, active_k=active_k)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.06, 3.0)
            continue

        alpha = golden_section_line(f, g, n)
        if alpha > 1e-10:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            conv = conv_new
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 30, 8)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def lse_gradient_descent(f0, time_limit, q0=5.0, q_max=300.0, verbose=False):
    """Log-sum-exp smoothed gradient descent with momentum and adaptive params."""
    f = project_simplex(np.maximum(f0.astype(np.float64), 0.0))
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    q = q0
    lr = 0.5 / n
    beta = 0.85
    v = np.zeros(n)
    no_improve = 0
    it = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive temperature: increase q over time (sharper max approximation)
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        q = q0 + (q_max - q0) * progress ** 1.5

        grad, conv = lse_gradient(f, q)

        # Heavy-ball momentum step
        v = beta * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1

        f = f_new

        # Adapt learning rate
        if no_improve > 30:
            lr *= 0.92
            beta = min(beta + 0.008, 0.98)
        if no_improve > 80:
            lr = min(lr * 1.3, 2.0 / n)
        if no_improve > 250:
            # Structured restart
            rng = np.random.default_rng(it * 1009 + 3)
            # Perturb in direction that reduces top peaks
            top_k = np.argsort(conv)[::-1][:10]
            anti_grad = np.zeros(n)
            for k in top_k:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    anti_grad[j] += f[k - j]
            anti_grad /= (anti_grad.max() + 1e-15)
            f = project_simplex(best_f - 0.01 * anti_grad + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = best_lr
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    LSE-GD iter {it}: obj={best_obj:.10f}, q={q:.1f}, lr={lr:.6f}")

    return best_f, best_obj


def coord_desc_fine(f0, time_limit):
    """Coordinate descent with multi-scale per-coordinate search."""
    f = project_simplex(np.maximum(f0.astype(np.float64), 0.0))
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.05, 1.1, 1.2, 1.3, 1.5, 1.8, 2.2, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (remaining / denom)
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= ((1.0 - best_j_val) / denom)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr=0.001, mom=0.94, top_k=40):
    """Subgradient polishing on the active peaks."""
    f = project_simplex(np.maximum(f0.astype(np.float64), 0.0))
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it_count = 0
    cur_lr = lr
    cur_mom = mom

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        # Softmax weights over active peaks
        temp = 0.05 / (n * 0.01)
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = cur_mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_mom = min(cur_mom + 0.0002, 0.98)

        if no_improve > 300:
            seed_val = (it_count * 7919 + 13) % 1000000
            rng = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def random_phase_init(n, seed=0):
    """Initialize using random phase (flat spectrum) approach."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig[:n])
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def chirp_init(n, seed=0):
    """Initialize with a chirp-like (frequency sweeping) structure."""
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    freq = rng.uniform(0.02, 0.3)
    phase = 2 * np.pi * freq * t * (t + 1) / (2 * n)
    sig = np.abs(np.cos(phase)) + 0.1
    sig = np.maximum(sig, 0)
    sig /= sig.sum()
    return sig


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-Frank-Wolfe, LSE gradient descent,
    coordinate descent, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        p_score = float(evaluate_sequence(parent.tolist()))
        best_score = p_score
        print(f"No parent, using uniform n=512, score={p_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t1 = min(250, rem * 0.25)
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True, top_k_constraints=40)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: LSE Gradient Descent ===
    print(f"\n=== Phase 2: LSE Gradient Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(300, rem * 0.30)
        if t2 > 30:
            print(f"  Allocating {t2:.0f}s, n={len(best_seq)}")
            a_opt, _ = lse_gradient_descent(best_seq, time_limit=t2,
                                            q0=5.0, q_max=200.0, verbose=True)
            try_seq(a_opt, "lse_gd")

    # === Phase 3: Fine Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(200, rem * 0.22)
        if t3 > 20:
            print(f"  Allocating {t3:.0f}s")
            a_opt, _ = coord_desc_fine(best_seq, time_limit=t3)
            try_seq(a_opt, "cd")
            # Second CD pass
            rem = deadline - time.monotonic()
            if rem > 20:
                t3b = min(100, rem * 0.12)
                a_opt2, _ = coord_desc_fine(a_opt, time_limit=t3b)
                try_seq(a_opt2, "cd2")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length search ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = []
        for ln in [512, 640, 768, 896, 1024, 1280, 1536, 1920]:
            if ln != base_n:
                target_lengths.append(ln)
        # Sort by proximity to base_n (interpolation is more accurate)
        target_lengths.sort(key=lambda x: abs(x - base_n))
        target_lengths = target_lengths[:5]  # Try at most 5 different lengths

        for i, target_n in enumerate(target_lengths):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            f_init = interpolate_sequence(best_seq, target_n)
            # Mix with a random-phase init for diversity
            rp_init = random_phase_init(target_n, seed=target_n + seed)
            f_mix = project_simplex(0.8 * f_init + 0.2 * rp_init)
            t_alloc = min(rem * 0.35, 150)
            print(f"  n={target_n} (mixed init), t={t_alloc:.0f}s")
            a_opt, _ = lse_gradient_descent(f_mix, time_limit=t_alloc,
                                            q0=10.0, q_max=150.0, verbose=False)
            try_seq(a_opt, f"multilen_{target_n}")
            # Quick LP-FW on the result
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lpfw = min(rem * 0.2, 60)
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw,
                                           top_k_constraints=30)
                try_seq(a_opt2, f"multilen_{target_n}_lpfw")

    # === Phase 5: Diverse restarts on best length ===
    print(f"\n=== Phase 5: Diverse restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = [
            ("noise_small", project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)),
            ("noise_med", project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)),
            ("rev_blend", project_simplex(0.7 * best_seq + 0.3 * best_seq[::-1])),
            ("flat_blend", project_simplex(0.85 * best_seq + 0.15 * np.ones(n_best) / n_best)),
            ("chirp_mix", project_simplex(0.6 * best_seq + 0.4 * chirp_init(n_best, seed))),
            ("rp_mix", project_simplex(0.7 * best_seq + 0.3 * random_phase_init(n_best, seed + 99))),
        ]

        for i, (name, f_init) in enumerate(restarts):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem / max(1, len(restarts) - i), 120)
            print(f"  Restart [{name}], t={t_alloc:.0f}s")
            a_opt, _ = lse_gradient_descent(f_init, time_limit=t_alloc,
                                            q0=8.0, q_max=180.0, verbose=False)
            try_seq(a_opt, f"restart_{name}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # LP-FW final
        rem = deadline - time.monotonic()
        if rem > 30:
            t_final = min(rem * 0.25, 80)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final,
                                      top_k_constraints=50)
            try_seq(a_opt, "final_lpfw")

        # Subgradient polish
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sg = min(rem * 0.25, 60)
            a_opt, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                          lr=0.0008, mom=0.95,
                                          top_k=max(30, n_best // 4))
            try_seq(a_opt, "final_sg")

        # LSE with very fine params
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lse = min(rem * 0.3, 60)
            a_opt, _ = lse_gradient_descent(best_seq, time_limit=t_lse,
                                            q0=50.0, q_max=400.0, verbose=False)
            try_seq(a_opt, "final_lse_fine")

        # Tiny perturbations + quick optimization
        rem = deadline - time.monotonic()
        for i in range(5):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0015 * i)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.25, 30)
            a_opt, _ = lse_gradient_descent(f_pert, time_limit=t_pert,
                                            q0=20.0, q_max=150.0, verbose=False)
            try_seq(a_opt, f"final_pert_{i}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(max(0, min(1000, x))) for x in best_seq]))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
