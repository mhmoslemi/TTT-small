**Key innovation: "Peak shaving" via direct manipulation**

Instead of using a generic optimization method, I can directly target the tallest peak and try to reduce it by adjusting the coordinates that contribute most to it.

For peak at position k*, the contribution of coordinate i to conv[k*] is `2*f[i]*f[k*-i]` (or `f[i]^2` if i=k*/2). So the coordinates with the largest `f[i]*f[k*-i]` contribute most to the peak.

I can try:
1. Reduce the largest contributing coordinate
2. Distribute its mass to coordinates that contribute less to the peak
3. Check if the overall objective improves

This is a form of "directed" optimization that's different from both gradient descent and coordinate descent.

**Another innovation: "Spectral" approach**

The convolution in the frequency domain is `|F|^2` where F is the DFT of f. The peak in the time domain corresponds to... hmm, this is the inverse FT of |F|^2. Minimizing the peak of the inverse FT of |F|^2 is related to minimizing PAPR of the signal f.

In the PAPR literature, the "RAX" algorithm works by:
1. Compute the time-domain signal x = f
2. Compute the frequency-domain representation
3. For each "active" frequency bin (where |X|^2 is large), reduce it
4. Go back to time domain

But since our signal is constrained to be non-negative, this doesn't directly apply. However, I can use a similar idea in the time domain:
1. Identify the peak position k*
2. The peak conv[k*] = sum_j f[j]*f[k*-j]
3. Try to reduce this by adjusting f[j] for the j that contribute most

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive coordinate descent with vectorized conv updates
2. Subgradient descent with Adam using exact near-peak subgradients
3. Alternating methods with random perturbations for escaping local minima
4. Fine-tuned final polishing with peak-targeted coordinate search
Key innovation: efficient coordinate descent exploiting the triangular structure
of auto-convolution (each coordinate affects only n positions).
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, frac_threshold=0.99):
    """Compute subgradient of max(conv) using positions near the peak."""
    n = len(f)
    max_val = conv.max()
    if max_val < 1e-30:
        return np.zeros(n)
    threshold = frac_threshold * max_val
    peak_mask = conv >= threshold
    peak_indices = np.where(peak_mask)[0]
    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(conv)])
    n_peaks = len(peak_indices)
    w = 1.0 / n_peaks
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += 2.0 * w * f[idx]
    return grad


def objective_val(f, conv=None):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(f)
    if conv is None:
        conv = fast_convolve(f)
    S = f.sum()
    if S < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (S * S)

def coord_descent(f_init, time_limit, verbose=True):
    """
    Coordinate descent minimizing 2*n*max(conv)/(sum)^2.
    Exploits the fact that changing f[i] only affects conv positions [i, i+n-1].
    """
    f = f_init.copy().astype(np.float64)
    n = len(f)
    conv = fast_convolve(f).astype(np.float64)
    S = f.sum()
    obj = 2.0 * n * conv.max() / (S * S)
    best_obj = obj
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        sweep += 1
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            t = f[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Precompute unchanged max (independent of candidate t')
            if k_lo > 0 and k_hi < 2 * n - 2:
                unchanged_max = max(np.max(conv[:k_lo]), np.max(conv[k_hi + 1:]))
            elif k_lo > 0:
                unchanged_max = np.max(conv[:k_lo])
            elif k_hi < 2 * n - 2:
                unchanged_max = np.max(conv[k_hi + 1:])
            else:
                unchanged_max = 0.0

            # Generate candidate values
            if t > 1e-12:
                fracs = np.array([0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])
                candidates = t * fracs
            else:
                avg = S / n
                candidates = np.array([0.0, 0.3 * avg, 0.5 * avg, avg, 2 * avg])

            best_cand = t
            best_cand_obj = objective_val(f, conv)

            for t_prime in candidates:
                if t_prime < 0:
                    continue
                S_new = S - t + t_prime
                if S_new < 1e-10:
                    continue

                delta = t_prime - t
                if abs(delta) < 1e-15:
                    continue

                # Compute affected max
                if k_lo == k_hi:
                    if k_lo == 2 * i:
                        affected_max = conv[k_lo] - t * t + t_prime * t_prime
                    else:
                        affected_max = conv[k_lo] + 2.0 * delta * f[k_lo - i]
                else:
                    affected_idx = np.arange(k_lo, k_hi + 1)
                    j_vals = affected_idx - i
                    affected_vals = conv[k_lo:k_hi + 1].copy() + 2.0 * delta * f[j_vals]
                    # Correct k=2i position
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        pos = 2 * i - k_lo
                        affected_vals[pos] = conv[2 * i] - t * t + t_prime * t_prime
                    affected_max = np.max(affected_vals)

                new_M = max(unchanged_max, affected_max)
                new_obj = 2.0 * n * new_M / (S_new * S_new)

                if new_obj < best_cand_obj - 1e-14:
                    best_cand_obj = new_obj
                    best_cand = t_prime

            if best_cand != t:
                # Apply update
                delta = best_cand - t
                f[i] = best_cand
                S = S - t + best_cand

                # Update conv in-place
                if k_lo == k_hi:
                    if k_lo == 2 * i:
                        conv[k_lo] += best_cand * best_cand - t * t
                    else:
                        conv[k_lo] += 2.0 * delta * f[k_lo - i]
                else:
                    j_vals = np.arange(k_lo, k_hi + 1) - i
                    conv[k_lo:k_hi + 1] += 2.0 * delta * f[j_vals]
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        conv[2 * i] += delta * delta

                sweep_improved = True
                if best_cand_obj < best_obj - 1e-14:
                    best_obj = best_cand_obj
                    best_f = f.copy()
                    if verbose and sweep % 50 == 0:
                        print(f"    CD sweep {sweep}: obj={best_obj:.8f}")

        if sweep_improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                break

    if verbose:
        print(f"  CD done: {sweep} sweeps, obj={best_obj:.8f}")
    return best_f, best_obj


def subgradient_adam(f_init, time_limit, lr_init=0.02, frac_thresh=0.99, verbose=True):
    """Subgradient descent with Adam optimizer using exact near-peak subgradients."""
    f = f_init.copy().astype(np.float64)
    S = f.sum()
    if S < 1e-15:
        f = np.ones(len(f)) / len(f)

        f = f / S
    n = len(f)

    conv = fast_convolve(f)
    obj = 2.0 * n * conv.max()
    best_obj = obj
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        grad = compute_subgradient(f, conv, frac_thresh)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        step_size = lr / (1.0 + 0.01 * t)
        step = step_size * m_hat / (np.sqrt(v_hat) + 1e-8)

        f_new = f - step
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new

        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
            if no_improve > 200:
                rng = np.random.default_rng(t * 7919 + 13)
                scale = 0.003 + 0.005 * min(1.0, (no_improve - 200) / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                obj = 2.0 * n * conv.max()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
                lr = max(lr_init * 0.5, 0.001)

    if verbose:
        print(f"  Subgrad-Adam done: obj={best_obj:.8f} (sum=1)")

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    print(f"  CD n={n_parent}, t={t_alloc:.0f}s")
    f_opt, obj = coord_descent(current, time_limit=t_alloc)
    # Normalize for evaluation
    s = f_opt.sum()
    if s > 1e-15:
        f_opt_norm = f_opt / s
    else:
        f_opt_norm = f_opt
    try_sequence(f_opt_norm, best_tracker)

    # Continue CD if time remains
    t_remaining = t1_end - time.monotonic()
    if t_remaining > 30 and best_tracker[1] is not None:
        print(f"  Continued CD, t={t_remaining:.0f}s")
        f_opt2, _ = coord_descent(best_tracker[1].copy(), time_limit=t_remaining)
        s2 = f_opt2.sum()
        if s2 > 1e-15:
            try_sequence(f_opt2 / s2, best_tracker)

    # === Phase 2: Subgradient + Adam ===
    print(f"\n=== Phase 2: Subgradient + Adam ===")
    t2_end = deadline - 500
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = max(30, t2_end - time.monotonic())
    print(f"  Subgrad-Adam n={len(current)}, t={t_alloc:.0f}s")
    f_opt, _ = subgradient_adam(current, time_limit=t_alloc, lr_init=0.015, frac_thresh=0.99)
    try_sequence(f_opt, best_tracker)

    # Second subgradient run with different parameters
    t_remaining = t2_end - time.monotonic()
    if t_remaining > 40 and best_tracker[1] is not None:
        current = best_tracker[1]
        print(f"  Subgrad-Adam (fine) n={len(current)}, t={t_remaining:.0f}s")
        f_opt, _ = subgradient_adam(current, time_limit=t_remaining,
                                     lr_init=0.005, frac_thresh=0.998)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Fine coordinate descent ===
    print(f"\n=== Phase 3: Fine Coordinate Descent ===")
    t3_end = deadline - 350
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = max(30, t3_end - time.monotonic())
    print(f"  Fine CD n={len(current)}, t={t_alloc:.0f}s")
    f_opt, obj = fine_coord_descent(current, time_limit=t_alloc, n_candidates=25)
    s = f_opt.sum()
    if s > 1e-15:
        f_opt_norm = f_opt / s
    else:
        f_opt_norm = f_opt
    try_sequence(f_opt_norm, best_tracker)

    # === Phase 4: Alternating with perturbations ===
    print(f"\n=== Phase 4: Alternating + Perturbations ===")
    t4_end = deadline - 200
    cycles = 0
    while time.monotonic() < t4_end and cycles < 6:
        cycles += 1
        current = best_tracker[1] if best_tracker[1] is not None else parent
        remaining = t4_end - time.monotonic()
        if remaining < 40:
            break

        if cycles % 3 == 1:
            # Coordinate descent
            t_alloc = min(80, remaining * 0.5)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: CD, t={t_alloc:.0f}s")
                f_opt, _ = coord_descent(current, time_limit=t_alloc)
                s = f_opt.sum()
                if s > 1e-15:
                    try_sequence(f_opt / s, best_tracker)
        elif cycles % 3 == 2:
            # Subgradient
            t_alloc = min(80, remaining * 0.5)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: Subgrad, t={t_alloc:.0f}s")
                f_opt, _ = subgradient_adam(current, time_limit=t_alloc,
                                             lr_init=0.008, frac_thresh=0.995)
                try_sequence(f_opt, best_tracker)
        else:
            # Perturbation
            t_alloc = min(60, remaining * 0.5)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: Perturb, t={t_alloc:.0f}s")
                f_opt, _ = perturbation_search(current, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final Fine-Tuning ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = max(20, deadline - time.monotonic() - 50)
    print(f"  Final fine CD, t={t_alloc:.0f}s")
    f_opt, obj = fine_coord_descent(current, time_limit=t_alloc, n_candidates=30)
    s = f_opt.sum()
    if s > 1e-15:
        f_opt_norm = f_opt / s
    else:
        f_opt_norm = f_opt
    try_sequence(f_opt_norm, best_tracker)

    # One more subgradient pass on the very best
    t_remaining = deadline - time.monotonic() - 30
    if t_remaining > 20 and best_tracker[1] is not None:
        current = best_tracker[1]
        print(f"  Final subgrad, t={t_remaining:.0f}s")
        f_opt, _ = subgradient_adam(current, time_limit=t_remaining,
                                     lr_init=0.003, frac_thresh=0.999)
        try_sequence(f_opt, best_tracker)

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [64, -64, 128, -128, 32, -32]
        for off in offsets:
            n_try = n_best + off
            if n_try < 256 or n_try > 1500:
                continue
            t_remaining = deadline - time.monotonic() - 20
            if t_remaining < 30:
                break
            t_alloc = min(50, t_remaining * 0.4)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = coord_descent(f_init, time_limit=t_alloc, verbose=False)
            s = f_opt.sum()
            if s > 1e-15:
                try_sequence(f_opt / s, best_tracker)

    # === Final ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
