import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            if symmetric:
                j2 = n - 1 - i
                f[i] = x_opt
                if j2 != i:
                    f[j2] = x_opt
            else:
                f[i] = x_opt

            # Renormalize
            f = f / f.sum()
            new_obj = fast_convolve(f).max()

            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

        if verbose and passes % 10 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    CoordDescent pass {passes}: obj={best_obj:.10f}, remaining={rem:.0f}s")

    return best_f, best_obj


def _remove_coord_contribution(conv, f, i, n):
    """Remove the contribution of f[i] to the convolution in-place."""
    fi = f[i]
    if fi < 1e-15:
        return
    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    for k in range(k_min, k_max + 1):
        if k == 2 * i:
            conv[k] -= fi * fi
        else:
            ki = k - i
            if 0 <= ki < n:
                conv[k] -= 2.0 * fi * f[ki]


def _eval_coord_obj(x, S, conv_base, f, i, n, symmetric):
    """Evaluate the normalized objective when f[i] (and f[n-1-i] if symmetric) = x."""
    if symmetric:
        j2 = n - 1 - i
        if j2 != i:
            total = S + 2.0 * x
            # Add contribution of both f[i]=x and f[j2]=x
            max_val = -1.0
            # Compute max over all k
            # For each k, conv[k] = conv_base[k] + contrib_i(k) + contrib_j2(k)
            # contrib_i(k) = 2*x*f[k-i] (if 0<=k-i<n and k!=2i) or x^2 (if k=2i)
            # contrib_j2(k) = 2*x*f[k-j2] (if 0<=k-j2<n and k!=2*j2) or x^2 (if k=2*j2)
            # Cross terms: f[i]*f[k-j2] and f[j2]*f[k-i] also contribute!
            # Actually this is getting complicated for symmetric case.
            # Let's just compute directly for affected range.
            # Affected k: those where k-i in [0,n) or k-j2 in [0,n)
            k_lo = max(0, i, j2 - n + 1)  # Hmm this is getting complex
            # Just do it simply:
            vals = conv_base.copy()
            # Add f[i] = x contribution
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        vals[k] += x * x
                    else:
                        vals[k] += 2.0 * x * f[ki]
            # Add f[j2] = x contribution (j2 != i)
            for k in range(max(0, j2), min(2 * n - 1, j2 + n)):
                kj = k - j2
                if 0 <= kj < n:
                    if k == 2 * j2:
                        vals[k] += x * x
                    else:
                        vals[k] += 2.0 * x * f[kj]
            # Add cross terms: f[i]*f[j2] contributes to conv at k = i+j2
            # conv[i+j2] gets x*x from f[i]*f[j2] and f[j2]*f[i]
            k_cross = i + j2
            if 0 <= k_cross <= 2 * n - 2:
                vals[k_cross] += 2.0 * x * x
            max_val = vals.max()
        else:
            total = S + x
            vals = conv_base.copy()
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        vals[k] += x * x
                    else:
                        vals[k] += 2.0 * x * f[ki]
            max_val = vals.max()
    else:
        total = S + x
        vals = conv_base.copy()
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    vals[k] += x * x
                else:
                    vals[k] += 2.0 * x * f[ki]
        max_val = vals.max()

    if total < 1e-15:
        return 1e15
    return max_val / (total * total)


def block_descent(f0, time_limit, block_size=4, verbose=False):
    """Optimize small blocks of coordinates jointly via local search."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    num_blocks = (n + block_size - 1) // block_size

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 200:
            break

        improved = False
        # Random block order
        perm = rng.permutation(num_blocks)

        for bi in perm:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            block_sum = f[start:end].sum()
            if block_sum < 1e-12:
                continue

            # Try several random perturbations
            for _ in range(4):
                pert = rng.standard_normal(end - start) * 0.05 * block_sum
                trial_block = np.maximum(f[start:end] + pert, 0)
                ts = trial_block.sum()
                if ts < 1e-12:
                    continue
                trial_block = trial_block * (block_sum / ts)
                f_trial = f.copy()
                f_trial[start:end] = trial_block
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-13:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 50 == 0:
            print(f"    Block it {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_init(n, seed):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    method = seed % 6
    if method == 0:
        # Uniform
        return np.ones(n) / n
    elif method == 1:
        # Gaussian (clipped)
        f = rng.standard_normal(n)
        f = np.maximum(f, 0)
        return f / f.sum()
    elif method == 2:
        # Triangular
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x)
        return f / f.sum()
    elif method == 3:
        # Bimodal
        x = np.linspace(0, 1, n)
        f = np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)
        return f / f.sum()
    elif method == 4:
        # Symmetric random
        half = n // 2
        f_half = np.maximum(rng.standard_normal(half), 0)
        if n % 2 == 0:
            f = np.concatenate([f_half, f_half[::-1]])
        else:
            f = np.concatenate([f_half, [f_half[half - 1]], f_half[::-1]])
        ss = f.sum()
        if ss < 1e-15:
            return np.ones(n) / n
        return f / ss
    else:
        # Sparse
        f = np.zeros(n)
        support = rng.choice(n, size=max(4, n // 4), replace=False)
        f[support] = 1.0
        return f / f.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Nesterov subgradient (diverse starts, multiple lengths)
    2. Coordinate descent (asymmetric and symmetric)
    3. Block descent
    4. Final polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Nesterov subgradient on parent ===
    print("\n=== Phase 1: Nesterov on parent ===")
    rem = deadline - time.monotonic() - 800
    t1 = min(rem, 300)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s, n={n_parent}")
        a_opt, _ = nesterov_subgrad(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "nesterov_p1")

    # === Phase 2: Coordinate descent on best ===
    print(f"\n=== Phase 2: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 600
        t_cd = min(rem, 350)
        if t_cd > 30:
            print(f"  Time: {t_cd:.0f}s, n={n_best}")
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd, symmetric=False, verbose=True)
            try_seq(a_opt, "coord_asym")

        # Symmetric version
        rem = deadline - time.monotonic() - 450
        if rem > 30:
            t_cd2 = min(rem, 200)
            a_opt2, _ = coord_descent(best_seq, time_limit=t_cd2, symmetric=True, verbose=True)
            try_seq(a_opt2, "coord_sym")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [512, 768, 1024, 1280]
    for target_n in target_lengths:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 300
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        base_score = float(evaluate_sequence(f_interp.tolist()))
        print(f"  Interpolated n={target_n}: score={base_score:.8f}")

        # Quick Nesterov
        t_alloc = min(rem * 0.3, 100)
        if t_alloc > 20:
            a_opt, _ = nesterov_subgrad(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"nest_n{target_n}")

        # Coordinate descent
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_cd = min(rem * 0.5, 120)
            a_opt, _ = coord_descent(a_opt if 'a_opt' in dir() else f_interp, time_limit=t_cd)
            try_seq(a_opt, f"coord_n{target_n}")

    # === Phase 4: Diverse multi-start ===
    print(f"\n=== Phase 4: Diverse starts (best={best_score:.8f}) ===")
    n_explore = len(best_seq) if best_seq is not None else n_parent

    for start_id in range(4):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(rem / max(1, 4 - start_id), 150)

        if start_id == 0:
            f_init = best_seq.copy() if best_seq is not None else parent
        elif start_id == 1:
            f_init = best_seq[::-1].copy() if best_seq is not None else parent
            f_init = f_init / f_init.sum()
        elif start_id == 2:
            rng = np.random.default_rng(200 + start_id)
            f_init = project_simplex(best_seq + rng.standard_normal(n_explore) * 0.01)
        else:
            f_init = 0.6 * (best_seq if best_seq is not None else parent) + 0.4 * (np.ones(n_explore) / n_explore)
            f_init = project_simplex(f_init)

        # Alternate: Nesterov then coord descent
        t_nest = min(t_alloc * 0.5, 80)
        t_cd = min(t_alloc * 0.5, 80)
        if t_nest > 15:
            a_opt, _ = nesterov_subgrad(f_init, time_limit=t_nest)
            try_seq(a_opt, f"div_nest_{start_id}")
        rem = deadline - time.monotonic() - 150
        if rem > 15 and t_cd > 15:
            a_opt, _ = coord_descent(a_opt if 'a_opt' in dir() else f_init, time_limit=min(rem, t_cd))
            try_seq(a_opt, f"div_coord_{start_id}")

    # === Phase 5: Block descent ===
    print(f"\n=== Phase 5: Block Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_blk = min(rem * 0.6, 150)
            bs = max(3, min(8, len(best_seq) // 64))
            print(f"  Time: {t_blk:.0f}s, block_size={bs}")
            a_opt, _ = block_descent(best_seq, time_limit=t_blk, block_size=bs, verbose=True)
            try_seq(a_opt, "block")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Coordinate descent on best
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_cd = min(rem, 80)
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd, symmetric=False)
            try_seq(a_opt, "coord_final")

        # Symmetric coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = coord_descent(best_seq, time_limit=min(rem, 50), symmetric=True)
            try_seq(a_opt, "coord_sym_final")

        # Tiny perturbations + coordinate descent
        for trial in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            rng = np.random.default_rng(800 + trial)
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.003)
            a_opt, _ = coord_descent(f_pert, time_limit=min(rem, 25))
            try_seq(a_opt, f"perturb_{trial}")

        # One more Nesterov pass
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "nest_final")

    # === Phase 7: Try different lengths from best ===
    print(f"\n=== Phase 7: Length tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        for ln in [len(best_seq) - 64, len(best_seq) + 64]:
            if ln < 128:
                continue
            rem = deadline - time.monotonic() - 5
            if rem < 10:
                break
            f_interp = interpolate_sequence(best_seq, ln)
            a_opt, _ = coord_descent(f_interp, time_limit=min(rem, 20))
            try_seq(a_opt, f"len{ln}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
