    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []

            cols.extend(js.tolist())

    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 35:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.001 + 0.004 * min(1.0, (no_improve - 35) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def compute_grad(f, w, max_peaks=400):
    """Compute gradient of sum_k w[k]*c_k w.r.t. f. Factor of 2 included."""
    n = len(f)
    max_w = w.max()
    if max_w < 1e-30:
        return np.zeros(n)

    if len(w) <= max_peaks:
        sig_idx = np.argsort(w)[::-1]
    else:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
    sig_idx = sig_idx[w[sig_idx] > 1e-15 * max_w]

    grad = np.zeros(n)
    for k in sig_idx:
        wk = 2.0 * w[k]
        if wk < 1e-18:
            continue
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        ms = np.arange(lo, hi + 1)
        grad[ms] += wk * f[ki - ms]
    return grad


def lse_nesterov(f0, time_limit, beta_start=10.0, beta_end=8000.0,
                  max_peaks=300, lr=None, verbose=False):
    """LSE-smoothed optimization with Nesterov acceleration and iterate averaging."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr is None:
        lr = 0.35

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    t_nest = 1.0
    y = f.copy()
    x_prev = f.copy()
    x_curr = f.copy()
    x_avg = f.copy()
    it_count = 0

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    momentum_coeff = 0.0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        t_nest_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        momentum_coeff = (t_nest - 1.0) / t_nest_new

        # Nesterov extrapolation
        y = x_curr + momentum_coeff * (x_curr - x_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = x_curr.copy()
            ys = 1.0
        else:
            y = y / ys

        # Compute LSE gradient
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        p = exp_bc / exp_bc.sum()

        grad = compute_grad(y, p, max_peaks=max_peaks)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            no_improve += 100
            break
        grad = grad / gn

        # Gradient step
        x_new = y - cur_lr * grad
        x_new = project_simplex(x_new)

        # Evaluate
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()

        # Iterate averaging
        x_avg = x_avg + (x_new - x_avg) / (it_count + 1)

        # Track best (check both current and averaged)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = x_new.copy()
            no_improve = 0
            x_prev = x_curr
            x_curr = x_new
            t_nest = t_nest_new
        else:
            # Also check averaged
            conv_avg = fast_convolve(x_avg)
            obj_avg = conv_avg.max()
            if obj_avg < best_obj - 1e-14:
                best_obj = obj_avg
                best_f = x_avg.copy()
                no_improve = 0
            else:
                no_improve += 1
                x_prev = x_curr
                x_curr = x_new
                t_nest = t_nest_new
                if no_improve >= 5:
                    cur_lr *= 0.95
                if no_improve > 50:
                    # Reset and increase beta
                    x_curr = best_f.copy()
                    x_prev = best_f.copy()
                    x_avg = best_f.copy()
                    t_nest = 1.0
                    cur_lr = max(lr * 0.5, 0.05)
                    beta = min(beta * 1.5, beta_end)
                    no_improve = 0

        # Increase beta slowly
        if no_improve == 0:
            beta *= 1.002
        else:
            beta *= 1.005
        beta = min(beta, beta_end)

        if verbose and it_count % 500 == 0:
            print(f"    LSE-N it={it_count}, obj={best_obj:.10f}, beta={beta:.1f}, lr={cur_lr:.4f}")

    return best_f, best_obj


def directed_subgrad(f0, time_limit, n_peaks=5, lr_init=0.3, momentum=0.0,
                     max_peaks=400, verbose=False):
    """Directed subgradient on top peaks with optional momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_len = len(conv)

        # Select top peaks with soft weighting
        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        # Exponential weighting emphasizing the highest
        scaled = pv - pv.min()
        if scaled.max() > 1e-15:
            s_max = scaled.max()
            w_rel = np.exp(5.0 * scaled / s_max - 5.0)
        else:
            w_rel = np.ones(k_top) / k_top
        w_rel /= w_rel.sum()

        w_full = np.zeros(conv_len)
        w_full[peaks] = w_rel

        grad = compute_grad(f, w_full, max_peaks=max_peaks)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            no_improve += 100
            break
        grad /= gn

        if momentum > 0:
            v = momentum * v - cur_lr * grad
            f_new = project_simplex(f + v)
        else:
            f_new = project_simplex(f - cur_lr * grad)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                cur_lr *= 0.97
            if no_improve > 200:
                rng = np.random.default_rng((it * 104729 + 3) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.3, 0.02)
                no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    DSG it={it}, obj={best_obj:.10f}, lr={cur_lr:.4f}")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining FW-LP, Nesterov LSE, and directed subgradient.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print(f"Using uniform init n=768")
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: FW-LP exploration ===
    print("\n=== Phase 1: FW-LP Exploration ===")
    t1_end = deadline - 1050

    # First: refine parent with FW
    t_alloc = min(120, t1_end - time.monotonic())
    if t_alloc > 20:
        print(f"  FW on parent n={n_parent}, t={t_alloc:.0f}s")
        f_opt, obj = fw_lp(parent, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # Try a couple of other lengths
    other_ns = [n_parent + 64, n_parent - 64, 896, 1024]
    other_ns = sorted(set([x for x in other_ns if 192 <= x <= 1280]), reverse=True)
    for nc in other_ns[:3]:
        t_remaining = t1_end - time.monotonic()
        if t_remaining < 30:
            break
        t_alloc = min(80, t_remaining * 0.35)
        if t_alloc < 20:
            break
        f_init = interpolate_to_length(parent, nc)
        print(f"  FW n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = fw_lp(f_init, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Nesterov LSE main optimization ===
    print(f"\n=== Phase 2: Nesterov LSE (beta 10->5000) ===")
    t2_end = deadline - 600
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    t_alloc = min(300, t2_end - time.monotonic())
    if t_alloc > 30:
        print(f"  LSE-N n={n_cur}, t={t_alloc:.0f}s, beta=10..5000")
        f_opt, _ = lse_nesterov(current, time_limit=t_alloc,
                                 beta_start=10.0, beta_end=5000.0,
                                 max_peaks=300, lr=0.35)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Second LSE pass at higher beta
    t_alloc = min(250, t2_end - time.monotonic())
    if t_alloc > 30:
        print(f"  LSE-N n={len(current)}, t={t_alloc:.0f}s, beta=100..8000")
        f_opt, _ = lse_nesterov(current, time_limit=t_alloc,
                                 beta_start=100.0, beta_end=8000.0,
                                 max_peaks=350, lr=0.25)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Directed subgradient + FW alternating ===
    print(f"\n=== Phase 3: Directed Subgrad + FW Alternation ===")
    t3_end = deadline - 350
    cycles = 0
    while time.monotonic() < t3_end and cycles < 10:
        cycles += 1
        current = best_tracker[1] if best_tracker[1] is not None else parent
        remaining = t3_end - time.monotonic()
        if remaining < 40:
            break

        if cycles % 3 == 1:
            t_alloc = min(80, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: DSG n={len(current)}, t={t_alloc:.0f}s, peaks=4")
                f_opt, _ = directed_subgrad(current, time_limit=t_alloc,
                                             n_peaks=4, lr_init=0.25, momentum=0.0)
                try_sequence(f_opt, best_tracker)
        elif cycles % 3 == 2:
            t_alloc = min(70, remaining * 0.35)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: FW n={len(current)}, t={t_alloc:.0f}s")
                f_opt, _ = fw_lp(current, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)
        else:
            t_alloc = min(90, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: LSE-N n={len(current)}, t={t_alloc:.0f}s, beta=500..10000")
                f_opt, _ = lse_nesterov(current, time_limit=t_alloc,
                                         beta_start=500.0, beta_end=10000.0,
                                         max_peaks=400, lr=0.15)
                try_sequence(f_opt, best_tracker)

    # === Phase 4: Intensive final polishing ===
    print(f"\n=== Phase 4: Final Polishing ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    # Very high beta LSE
    t_alloc = min(150, deadline - time.monotonic() - 80)
    if t_alloc > 20:
        print(f"  High-beta LSE n={n_cur}, t={t_alloc:.0f}s, beta=2000..20000")
        f_opt, _ = lse_nesterov(current, time_limit=t_alloc,
                                 beta_start=2000.0, beta_end=20000.0,
                                 max_peaks=500, lr=0.1)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Directed subgradient with 1-2 peaks
    t_alloc = min(100, deadline - time.monotonic() - 40)
    if t_alloc > 15:
        print(f"  DSG (1 peak) n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = directed_subgrad(current, time_limit=t_alloc,
                                     n_peaks=1, lr_init=0.15, momentum=0.0)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # One more FW pass
    t_alloc = min(60, deadline - time.monotonic() - 15)
    if t_alloc > 15:
        print(f"  Final FW n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = fw_lp(current, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Final micro LSE
    t_alloc = min(40, deadline - time.monotonic() - 5)
    if t_alloc > 10:
        print(f"  Micro LSE n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = lse_nesterov(current, time_limit=t_alloc,
                                 beta_start=10000.0, beta_end=30000.0,
                                 max_peaks=600, lr=0.05)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Try nearby lengths ===
    print(f"\n=== Phase 5: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        for off in [32, -32, 64, -64, 128, -128]:
            n_try = n_best + off
            if n_try < 192 or n_try > 1500:
                continue
            t_remaining = deadline - time.monotonic() - 10
            if t_remaining < 25:
                break
            t_alloc = min(50, t_remaining * 0.4)
            if t_alloc < 12:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Nearby n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_nesterov(f_init, time_limit=t_alloc,
                                     beta_start=50.0, beta_end=5000.0,
                                     max_peaks=300, lr=0.3)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    return [float(x) for x in (np.ones(768) / 768)]
