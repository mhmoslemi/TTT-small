    # (f + a*d) * (f + a*d) = f*f + a*f*d + a*d*f + a^2*d*d
    # = f*f + 2a*(f*d) + a^2*(d*d)  [since f*d = d*f for 1D convolution]
    
    conv_fd = np.convolve(f, d, 'full')
    conv_dd = np.convolve(d, d, 'full')
    
    size = 2 * n - 1
    conv_ff = conv_ff[:size]
    conv_fd = conv_fd[:size]
    conv_dd = conv_dd[:size]

    lo, hi = 0.0, 1.0

    for _ in range(35):
        mid = (lo + hi) / 2.0
        vals = conv_ff + 2.0 * mid * conv_fd + mid * mid * conv_dd
        k_max = np.argmax(vals)
        # Subgradient: 2*conv_fd[k_max] + 2*mid*conv_dd[k_max]
        sg = 2.0 * conv_fd[k_max] + 2.0 * mid * conv_dd[k_max]
        if sg < 0:
            lo = mid
        else:
            hi = mid

    best_alpha = (lo + hi) / 2.0
    # Also check endpoints
    vals_0 = conv_ff.max()
    vals_1 = (conv_ff + 2.0 * conv_fd + conv_dd).max()
    vals_best = (conv_ff + 2.0 * best_alpha * conv_fd + best_alpha * best_alpha * conv_dd).max()
    
    if vals_0 < vals_best:
        best_alpha = 0.0
        vals_best = vals_0
    if vals_1 < vals_best:
        best_alpha = 1.0
        vals_best = vals_1
    
    return best_alpha, float(vals_best)


def lp_fw_topk(f0, time_limit, K_init=15, K_max=50, verbose=False):
    """LP-FW with top-K active constraints and cutting-plane expansion."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    K = K_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Find top-K active constraints
        k_actual = min(K, len(conv))
        active_ks = np.argpartition(conv, -k_actual)[-k_actual:]
        active_ks = sorted(set(int(k) for k in active_ks))

        # Solve LP with these constraints
        g, lp_t = solve_lp_topk(f, active_ks, n)
        
        if g is None:
            # Fallback: use uniform
            g = np.ones(n) / n
        else:
            # Check if solution violates other constraints (cutting plane)
            conv_g = fast_convolve(g) * g.sum()  # (f*g) not (g*g)!
            # Actually we need (f*g), not (g*g)
            # (f*g)[k] = sum_j f[j]*g[k-j]
            # Let me compute it properly
            pass
        
        # Compute (f*g) to check violations
        fg_conv = np.convolve(f, g, 'full')
        fg_max = fg_conv.max()
        
        # If the LP solution has (f*g) violating constraints beyond the active set,
        # the direction might not be great, but we proceed anyway (the line search will handle it)
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                # Add perturbation
                rng = np.random.default_rng(it * 31337 + 7)
                noise = rng.standard_normal(n) * 0.003
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                no_improve = 0
            continue

        # Smart line search
        alpha, obj_val = smart_line_search(f, g, n)
        
        if alpha > 1e-10:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            
            conv_new = fast_convolve(f_new)
            obj_new = float(conv_new.max())
            
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                conv = conv_new
                no_improve = 0
                K = min(K + 1, K_max)
                if verbose and it % 50 == 0:
                    print(f"    LP-FW it={it}: obj={best_obj:.10f}, K={K}, elapsed={elapsed:.1f}s")
            else:
                no_improve += 1
                # Shrink K slightly to explore more
                K = max(K_init, K - 2)
        else:
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.003 + 0.004 * min(no_improve // 50, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            no_improve = 0
            K = K_init

    return best_f, best_obj


def mirror_descent_opt(f0, time_limit, lr_init=0.005, top_k=30, temp=5e-5, verbose=False):
    """Mirror descent (exponential gradient) for minimizing max convolution peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Softmax weighting
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted gradient: grad[j] = 2 * sum_k w[k] * f[k-j] (for valid k-j)
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # grad[j] += 2 * w[i] * f[k-j] for j in [j_start, j_end]
            kg = k - np.arange(j_start, j_end + 1)
            valid = (kg >= 0) & (kg < n)
            idx = np.arange(j_start, j_end + 1)[valid]
            grad[idx] += 2.0 * w[i] * f[kg[valid]]

        # Exponential gradient update
        log_ratio = -lr * grad
        # Clip to avoid overflow
        log_ratio -= log_ratio.max()
        log_ratios = np.exp(log_ratio)
        f_new = f * log_ratios
        fn = f_new.sum()
        if fn < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new /= fn

        conv = fast_convolve(f_new)
        co = float(conv.max())

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MD it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            lr = max(lr_init * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def subgradient_nesterov(f0, time_limit, lr=0.003, momentum=0.93, top_k=30, temp=5e-5, verbose=False):
    """Nesterov-accelerated subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    t = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t = t + 1
        momentum_t = (t + 1) / (t + 2)  # Nesterov momentum schedule

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            kg = k - np.arange(j_start, j_end + 1)
            valid = (kg >= 0) & (kg < n)
            idx = np.arange(j_start, j_end + 1)[valid]
            sg[idx] += 2.0 * w[i] * y[kg[valid]]

        v_new = momentum_t * v - cur_lr * sg
        y_new = y + (1.0 - momentum_t) * (f - y) + v_new - v
        # Simpler: y = f + v (standard Nesterov)
        y_new = f - cur_lr * sg  # gradient step
        v_new = y_new - f + momentum_t * (f - y)  # velocity update
        # Actually let me use the standard form:
        # y = x - v (or x + v depending on convention)
        # Let me just use: x_new = project(x - lr * grad + momentum * v_old)
        
        # Standard Nesterov:
        # y = x + v
        # grad = f'(y)
        # v_new = momentum * v - lr * grad
        # x_new = x + v_new
        
        # Let me redo this properly
        # At start of iteration: we have f (current), v (velocity), y = f + v
        y = f + v  # Nesterov lookahead
        conv_y = fast_convolve(y)
        
        # Compute subgradient at y
        peaks_y = np.argpartition(conv_y, -k_actual)[-k_actual:]
        pv_y = conv_y[peaks_y]
        scaled_y = pv_y / cur_temp
        scaled_y -= scaled_y.max()
        w_y = np.exp(scaled_y) / np.exp(scaled_y).sum()
        
        sg_y = np.zeros(n)
        for i in range(len(peaks_y)):
            if w_y[i] < 1e-12:
                continue
            k = int(peaks_y[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            kg = k - np.arange(j_start, j_end + 1)
            valid = (kg >= 0) & (kg < n)
            idx = np.arange(j_start, j_end + 1)[valid]
            sg_y[idx] += 2.0 * w_y[i] * y[kg[valid]]
        
        v = momentum * v - cur_lr * sg_y
        f_new = project_simplex(f + v)
        
        # Reduce v to avoid divergence
        v_norm = np.linalg.norm(v)
        if v_norm > 0.05:
            v = v * 0.05 / v_norm

        conv = fast_convolve(f_new)
        co = float(conv.max())

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SG it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
            if no_improve % 100 == 0:
                v *= 0.8  # dampen velocity

        cur_lr *= 0.9999
        cur_temp *= 1.002

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 7919) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0015)
            v = np.zeros(n)
            conv = fast_convolve(f)
            cur_lr = max(lr * 0.4, 0.0005)
            no_improve = 0

    return best_f, best_obj


def greedy_peak_cd(f0, time_limit, verbose=False):
    """Greedy coordinate descent: reduce coordinates contributing to the peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            j_start = max(0, peak_k - n + 1)
            j_end = min(n - 1, peak_k)
            
            # Get coordinates contributing to this peak, sorted by contribution
            contributions = np.zeros(n)
            for j in range(j_start, j_end + 1):
                kj = peak_k - j
                if 0 <= kj < n:
                    contributions[j] = f[j] * f[kj]
            
            # Try reducing the most contributing coordinates
            sorted_idx = np.argsort(contributions)[::-1]
            
            for j in sorted_idx[:20]:
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-12:
                    continue
                
                best_j_val = old_val
                best_j_obj = best_obj
                
                for mult in [0.3, 0.5, 0.7, 0.85, 1.2, 1.5, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    
                    val = float(fast_convolve(f_trial).max())
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                        
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
                    if improved and no_improve == 0:
                        break  # Move to next peak

        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Random exploration
                rng = np.random.default_rng(passes * 104729 + 3)
                for _ in range(min(30, n // 5)):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 1.5, 2.5]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.5:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = float(fast_convolve(f_trial).max())
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization with LP-FW (top-K), mirror descent,
    Nesterov subgradient, greedy CD, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(640) / 640
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    base = best_seq.copy()

    # === Phase 1: LP-FW with top-K on parent ===
    print("\n=== Phase 1: LP-FW top-K on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.30)
    if t1 > 40:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = lp_fw_topk(base, time_limit=t1, K_init=15, K_max=50, verbose=True)
        try_seq(a_opt, "lpfw_topk_parent")

    # === Phase 2: Mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(250, rem * 0.20)
        if t2 > 30:
            print(f"  n={len(best_seq)}, t={t2:.0f}s")
            a_opt, _ = mirror_descent_opt(best_seq, time_limit=t2, lr_init=0.004,
                                          top_k=40, temp=3e-5, verbose=True)
            try_seq(a_opt, "mirror_descent")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov SG (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(250, rem * 0.20)
        if t3 > 30:
            print(f"  n={len(best_seq)}, t={t3:.0f}s")
            a_opt, _ = subgradient_nesterov(best_seq, time_limit=t3, lr=0.0025,
                                            momentum=0.94, top_k=35, temp=4e-5, verbose=True)
            try_seq(a_opt, "nesterov_sg")

    # === Phase 4: Greedy peak CD ===
    print(f"\n=== Phase 4: Greedy Peak CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t4 = min(200, rem * 0.15)
        if t4 > 25:
            print(f"  n={len(best_seq)}, t={t4:.0f}s")
            a_opt, _ = greedy_peak_cd(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "greedy_cd")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.8f}) ===")
    if best_seq is not None:
        candidate_lengths = set()
        for delta in [-64, -128, 64, 128, 256]:
            ln = len(best_seq) + delta
            if 256 <= ln <= 2048:
                candidate_lengths.add(ln)
        candidate_lengths = sorted(candidate_lengths)

        for target_n in candidate_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.25, 150)
            if t_alloc < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_topk(f_interp, time_limit=t_alloc, K_init=15, K_max=40)
            try_seq(a_opt, f"multilen_{target_n}")

    # === Phase 6: Structured perturbations + quick re-optimize ===
    print(f"\n=== Phase 6: Perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Try different shaped perturbations
        perturbations = []
        # Sinusoidal
        x = np.linspace(0, 2 * np.pi, n_best)
        perturbations.append(0.005 * np.sin(x))
        # Cosine bump (boost center)
        bump = np.exp(-((np.arange(n_best) - n_best / 2) ** 2) / (2 * (n_best / 4) ** 2))
        perturbations.append(0.004 * (bump - bump.mean()))
        # Two bumps
        bump2 = (np.exp(-((np.arange(n_best) - n_best * 0.3) ** 2) / (2 * (n_best / 6) ** 2)) +
                 np.exp(-((np.arange(n_best) - n_best * 0.7) ** 2) / (2 * (n_best / 6) ** 2)))
        perturbations.append(0.003 * (bump2 - bump2.mean()))
        # Pure noise
        perturbations.append(rng.standard_normal(n_best) * 0.003)
        # More aggressive noise
        perturbations.append(rng.standard_normal(n_best) * 0.008)

        for pi, pert in enumerate(perturbations):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            f_pert = project_simplex(best_seq + pert)
            t_pert = min(80, rem * 0.4)
            if t_pert < 15:
                break
            a_opt, _ = lp_fw_topk(f_pert, time_limit=t_pert, K_init=10, K_max=30)
            try_seq(a_opt, f"perturb_{pi}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Quick LP-FW
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_opt, _ = lp_fw_topk(best_seq, time_limit=min(rem, 100), K_init=20, K_max=50)
            try_seq(a_opt, "polish_lpfw")

        # Quick mirror descent
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = mirror_descent_opt(best_seq, time_limit=min(rem, 60),

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-FW with top-K active constraints (fast, many iterations)
2. Smart line search via bisection on max-of-quadratics
3. Mirror descent (exponential gradient) for geometric diversity
4. Nesterov-accelerated subgradient descent
5. Greedy peak reduction coordinate descent
6. Structured perturbations and multi-length exploration
7. Final polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def full_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, normalize to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_topk(f, active_ks, n):
    """Solve: min t s.t. (f*g)[k] <= t for k in active_ks, sum(g)=1, g>=0.
    Returns (g, t) or (None, None)."""
    K = len(active_ks)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((K, n + 1))
    for i, k in enumerate(active_ks):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        if length > 0:
            A_ub[i, j_start:j_end + 1] = f[j_start:j_end + 1]
        A_ub[i, -1] = -1.0
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
            return g, float(res.x[-1])
    except Exception:
        pass
    return None, None


def smart_line_search(f, g, n):
    """Find alpha in [0,1] minimizing max_k ((f+alpha*(g-f)) * (f+alpha*(g-f)))[k]."""
    d = g - f
    conv_ff = full_convolve(f, f)
    conv_fd = full_convolve(f, d)
    conv_dd = full_convolve(d, d)
    # (f + a*d) * (f + a*d) = ff + 2a*fd + a^2*dd  (commutativity of convolution)

    lo, hi = 0.0, 1.0
    for _ in range(35):
        mid = (lo + hi) / 2.0
        vals = conv_ff + 2.0 * mid * conv_fd + mid * mid * conv_dd
        k_max = np.argmax(vals)
        sg = 2.0 * conv_fd[k_max] + 2.0 * mid * conv_dd[k_max]
        if sg < 0:
            lo = mid
        else:
            hi = mid

    best_alpha = (lo + hi) / 2.0
    vals_best = float((conv_ff + 2.0 * best_alpha * conv_fd + best_alpha * best_alpha * conv_dd).max())

    # Check endpoints
    vals_0 = float(conv_ff.max())
    vals_1 = float((conv_ff + 2.0 * conv_fd + conv_dd).max())
    if vals_0 < vals_best:
        best_alpha = 0.0
        vals_best = vals_0
    if vals_1 < vals_best:
        best_alpha = 1.0
        vals_best = vals_1

    return best_alpha, vals_best

def mirror_descent_opt(f0, time_limit, lr_init=0.005, top_k=30, temp=5e-5, verbose=False):
    """Mirror descent (exponential gradient)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            kg = k - js
            valid = (kg >= 0) & (kg < n)
            grad[js[valid]] += 2.0 * w[i] * f[kg[valid]]

        log_ratio = -lr * grad
        log_ratio -= log_ratio.max()
        log_ratios = np.exp(log_ratio)
        f_new = f * log_ratios
        fn = f_new.sum()
        if fn < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new /= fn

        conv = fast_convolve(f_new)
        co = float(conv.max())

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MD it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            lr = max(lr_init * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def subgradient_nesterov(f0, time_limit, lr=0.003, momentum=0.93, top_k=30, temp=5e-5, verbose=False):
    """Nesterov-accelerated subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))

        # Nesterov lookahead
        y = f + v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = f.copy()
        else:
            y = y / ys

        conv_y = fast_convolve(y)
        peaks_y = np.argpartition(conv_y, -k_actual)[-k_actual:]
        pv_y = conv_y[peaks_y]
        scaled_y = pv_y / cur_temp
        scaled_y -= scaled_y.max()
        w_y = np.exp(scaled_y)
        ws_y = w_y.sum()
        if ws_y < 1e-15:
            w_y = np.zeros(len(peaks_y))
            w_y[np.argmax(pv_y)] = 1.0
        else:
            w_y /= ws_y

        sg = np.zeros(n)
        for i in range(len(peaks_y)):
            if w_y[i] < 1e-12:
                continue
            k = int(peaks_y[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            kg = k - js
            valid = (kg >= 0) & (kg < n)
            sg[js[valid]] += 2.0 * w_y[i] * y[kg[valid]]

        v_new = momentum * v - cur_lr * sg
        # Clip velocity
        vn = np.linalg.norm(v_new)
        if vn > 0.05:
            v_new = v_new * 0.05 / vn
        v = v_new

        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = float(conv.max())

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SG it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
            if no_improve % 100 == 0:
                v *= 0.7

        cur_lr *= 0.9999
        cur_temp *= 1.002

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 7919) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0015)
            v = np.zeros(n)
            conv = fast_convolve(f)
            cur_lr = max(lr * 0.4, 0.0005)
            no_improve = 0

    return best_f, best_obj
