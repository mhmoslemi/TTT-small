"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using mirror descent with 
full-convolution subgradients, peak-shaving coordinate descent, 
and multi-start optimization with length adaptation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        # Let's define c[e] = w[e] for e=0..2n-2, and c[e] = 0 otherwise
        # g_j/2 = sum_{d=0}^{n-1} c[j+d] * b[d]
        # This is the "uncorrelated" cross-correlation, which equals:
        # = (b * flip(c))[j + len(c) - 1] in numpy convolve terms
        # Or using FFT: = IFFT(FFT(b) * conj(FFT(c)))[j]... 
        # Actually the cleanest way: 
        # g_j/2 = sum_d b[d] * c[j+d]
        # Let's reverse c: c_rev[d] = c[len(c)-1-d]
        # Then sum_d b[d] * c[j+d] = sum_d b[d] * c_rev[len(c)-1-j-d]
        # = (b * c_rev)[len(c)-1-j] ... no that's not right either.

def peak_shaving_cd(f0, time_limit, verbose=False, seed=42):

        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        if passes % 50 == 0 and verbose:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    CD pass {passes}: obj={best_obj:.10f}, rem={rem:.1f}s")
        
        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in peak_positions:
            peak_k = int(pk)
            if conv[peak_k] < best_obj * 0.95:
                continue  # Skip non-dominant peaks

            # (a*a)[peak_k] = sum_j a_j * a_{peak_k - j} for valid j
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            contrib = np.zeros(n)
            for j in range(js, je + 1):
                d = peak_k - j
                if 0 <= d < n:
                    contrib[j] = a_j * a_d  # can't do this in vectorized way easily

            idx = np.arange(js, je + 1)
            d_vals = peak_k - idx
            valid = (d_vals >= 0) & (d_vals < n)
            idx_valid = idx[valid]
            contrib[idx_valid] = f[idx_valid] * f[d_vals[valid]]
            
            # Try reducing the top contributors
            top_contrib = np.argsort(contrib)[-min(10, n // 10):]
            
            for j in top_contrib:
                if time.monotonic() - t0 > time_limit:
                    break
                
                old_val = f[j]
                if old_val < 1e-12:
                    continue

                    f_trial[j] = new_val

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:

                        best_j_val = new_val

                    denom = 1.0 - old_val
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    conv = fast_convolve(f)

def subgradient_optimize(f0, time_limit, lr=0.005, momentum=0.9, top_k=50, temp=0.0001,
                         verbose=False, seed=42):
    """
    Subgradient method with Euclidean projection (as in previous program but refined).
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 500 == 0 and verbose:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}, rem={rem:.1f}s")
        
        # Compute subgradient using top-k peaks with softmax weighting
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            g[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
        
        # Nesterov-like update
        v = momentum * v - cur_lr * g
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new
        
        cur_lr *= 0.9997
        cur_temp *= 1.003
        
        if no_improve > 300:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0005)
            cur_temp = temp * 0.3
            no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a multi-scale optimization strategy:
1. Multi-scale warm-up (coarse to fine)
2. FFT-accelerated mirror descent with full-convolution subgradient
3. Subgradient descent with adaptive softmax peaks
4. Peak-shaving coordinate descent
5. Multi-length exploration and restart
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_subgradient(f, conv, peaks, weights):
    """Compute subgradient using FFT for efficiency."""
    n = len(f)
    # g_j = sum_i weights[i] * 2 * f[peaks[i] - j] (if 0 <= peaks[i] - j < n)
    # = 2 * sum_k w_k * f[k-j] * 1[0<=k-j<n] where w is placed at peak positions
    
    # Create full weight vector
    w = np.zeros(2 * n - 1)
    for i, (pk, wi) in enumerate(zip(peaks, weights)):
        w[int(pk)] = wi
    
    # g_j = 2 * sum_{d=0}^{n-1} w[j+d] * f[d]
    # This is a cross-correlation
    N = 1 << (3 * n - 1).bit_length()
    W_f = np.fft.rfft(w, N)
    F_f = np.fft.rfft(f, N)
    cc = np.fft.irfft(W_f * np.conj(F_f), N)
    return 2.0 * cc[:n]


def mirror_descent(f0, time_limit, lr=0.01, temp=0.001, verbose=False, seed=42):
    """
    Mirror descent (multiplicative weights) with full-convolution softmax subgradient.
    Uses FFT for efficient subgradient computation.
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Precompute FFT size for subgradient
    N_sg = 1 << (3 * n - 1).bit_length()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 2000 == 0 and verbose:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    MD iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}, temp={cur_temp:.6f}, rem={rem:.1f}s")
        
        # Compute softmax weights over full convolution
        scaled = conv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient via FFT
        W_f = np.fft.rfft(w, N_sg)
        F_f = np.fft.rfft(f, N_sg)
        cc = np.fft.irfft(W_f * np.conj(F_f), N_sg)
        g = 2.0 * cc[:n]
        
        # Mirror descent update
        # f_j <- f_j * exp(-eta * g_j) / Z
        # In log space: log(f_j) <- log(f_j) - eta * g_j - log(Z)
        # For numerical stability, subtract max
        updates = -cur_lr * g
        updates -= updates.max()
        new_f = f * np.exp(updates)
        ns = new_f.sum()
        if ns < 1e-30:
            new_f = np.ones(n) / n
        else:
            new_f /= ns
        
        new_conv = fast_convolve(new_f)
        new_obj = new_conv.max()
        
        f = new_f
        conv = new_conv
        
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.001

            conv = fast_convolve(f)
            cur_lr = max(lr * 0.1, 0.0003)
            cur_temp = max(temp * 0.3, 0.00001)
            no_improve = 0
            best_obj = conv.max()
            best_f = f.copy()
    
    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.004, momentum=0.92, top_k=50, temp=0.0001,
                verbose=False, seed=42):
    """Subgradient method with FFT-accelerated gradient and adaptive scheduling."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    N_sg = 1 << (3 * n - 1).bit_length()
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 1000 == 0 and verbose:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}, rem={rem:.1f}s")
        
        # Compute subgradient using top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        ws = w_peaks.sum()
        if ws < 1e-15:
            w_peaks = np.zeros(len(peaks))
            w_peaks[np.argmax(pv)] = 1.0
        else:
            w_peaks /= ws
        
        # Place weights in full vector and compute subgradient via FFT
        w_full = np.zeros(2 * n - 1)
        for i in range(len(peaks)):
            if w_peaks[i] > 1e-14:
                w_full[int(peaks[i])] = w_peaks[i]
        
        W_f = np.fft.rfft(w_full, N_sg)
        F_f = np.fft.rfft(f, N_sg)
        cc = np.fft.irfft(W_f * np.conj(F_f), N_sg)
        g = 2.0 * cc[:n]

        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0

            cur_lr = max(lr * 0.3, 0.0005)

    return best_f, best_obj
