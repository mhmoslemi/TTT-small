"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Gauss-Newton QP iterations
with peak-targeted optimization, combined with subgradient polish
and multi-n exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_hankel(a):
    """Build the (2n-1) x n Hankel matrix H where H[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    H = np.zeros((2*n-1, n))
    for j in range(n):
        H[j:j+n, j] = a
    return H

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    """Compute the objective (normalized: sum(a)=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    return 2.0 * n * fast_convolve(a).max()

def gn_qp_step(a, c_target, dampen=1.0):
    """One Gauss-Newton QP step."""
    n = len(a)
    c_current = fast_convolve(a)
    g = c_current - c_target
    
    # Build Hankel matrix H (2n-1) x n
    H = np.zeros((2*n-1, n))
    for j in range(n):
        H[j:j+n, j] = a
    
    # QP: min ||2*H*d + g||^2 s.t. a + d >= 0, sum(d) = 0
    # Hessian: 4*H^T*H, gradient: 4*H^T*g
    HtH = H.T @ H
    Htg = H.T @ g
    
    # Solve KKT (ignoring inequality constraints, will project later)
    # [4*HtH  1] [d]   [-4*Htg]
    # [1^T    0] [λ] = [  0   ]
    size = n + 1
    A_kkt = np.zeros((size, size))
    A_kkt[:n, :n] = 4 * HtH
    A_kkt[:n, n] = 1.0
    A_kkt[n, :n] = 1.0
    b_kkt = np.zeros(size)
    b_kkt[:n] = -4 * Htg
    
    try:
        sol = np.linalg.solve(A_kkt, b_kkt)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A_kkt, b_kkt, rcond=None)[0]
    
    d = sol[:n] * dampen
    a_new = project_simplex(a + d)
    return a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gauss-Newton QP iterations (linearized peak-targeting via Hankel matrix)
2. Smoothed subgradient descent with momentum for local refinement
3. Symmetric subspace optimization
4. Multi-n exploration with adaptive target shaping
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(w, a):
    """Compute weighted subgradient: grad[j] = 2 * sum_k w[k] * a[k-j]."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if end > j:
            grad[j] = 2.0 * np.dot(w[j:end], a[:end - j])
    return grad


def build_hankel(a):
    """Build (2n-1) x n Hankel matrix H where H[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    H = np.zeros((2 * n - 1, n))
    for j in range(n):
        H[j:j + n, j] = a
    return H


def gn_qp_step(a, c_target, dampen=1.0):
    """One Gauss-Newton QP step to reduce ||conv(a,a) - c_target||^2."""
    n = len(a)
    c_current = fast_convolve(a)
    g = c_current - c_target

    H = build_hankel(a)
    HtH = H.T @ H
    Htg = H.T @ g

    # Solve KKT: [4*HtH | 1][d] = [-4*Htg]
    #             [1^T   | 0][λ]   [  0   ]
    size = n + 1
    A_kkt = np.zeros((size, size))
    A_kkt[:n, :n] = 4.0 * HtH
    A_kkt[:n, n] = 1.0
    A_kkt[n, :n] = 1.0
    b_kkt = np.zeros(size)
    b_kkt[:n] = -4.0 * Htg

    try:
        sol = np.linalg.solve(A_kkt, b_kkt)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A_kkt, b_kkt, rcond=None)[0]

    d = sol[:n] * dampen
    a_new = project_simplex(a + d)
    return a_new


def gn_optimize(a0, time_limit, c_target=None, max_iter=300):
    """Gauss-Newton optimization with QP subproblems and adaptive damping."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    if c_target is None:
        c_target = np.ones(2 * n - 1) / (2 * n - 1)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    current_obj = best_obj

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        improved = False
        for dampen in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_candidate = gn_qp_step(a, c_target, dampen=dampen)
            obj_new = fast_convolve(a_candidate).max()
            if obj_new < current_obj - 1e-13:
                a = a_candidate
                current_obj = obj_new
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                no_improve = 0

                no_improve = 0

def smooth_grad_opt(a0, time_limit, T=0.001, lr=0.03, mom=0.93):
    """Smoothed gradient descent with momentum and adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_T = T
    current_lr = lr

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        scaled = conv / current_T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        grad = compute_subgradient(w, a)
        v = mom * v - current_lr * grad
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            current_lr *= 0.95
        if no_improve > 80:
            current_T *= 1.05
        if no_improve > 250:
            rng = np.random.default_rng((it * 48271 + 7) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 250) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            current_lr = lr * 0.5
            current_T = T
            no_improve = 0

    return best_a, best_obj


def top_peak_sg_opt(a0, time_limit, top_k=15, lr=0.003, mom=0.94):
    """Multi-peak subgradient with inverse-rank weighting."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        # Build weighted subgradient
        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = weights
        grad = compute_subgradient(w_full, a)

        v = mom * v - lr * grad
        a = project_simplex
