def adam_optimize(f0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  temperature=0.001, n_top_k=None, verbose=False, label=""):
    """
    Adam optimization with softmax parameterization.
    Minimizes log-sum-exp approximation of max(auto-conv).
    """
    n = len(f0)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    f = np.maximum(f0, 1e-10).astype(np.float64)
    f = f / f.sum()
    w = np.log(f)  # log-parameters for softmax
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        f = softmax_param(w)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        dconv = logsumexp_grad(conv, temperature, n_top_k)
        
        # Gradient w.r.t. f
        grad_f = compute_grad(dconv, f, fft_size)
        
        # Gradient w.r.t. w (chain rule through softmax)
        # f[i] = exp(w[i]) / sum(exp(w[j]))
        # d(f[i])/d(w[j]) = f[i]*(delta_ij - f[j])
        # grad_w[j] = sum_i grad_f[i] * f[i] * (delta_ij - f[j])
        # = grad_f[j]*f[j] - f[j]*sum_i grad_f[i]*f[i]
        dot = np.dot(grad_f, f)
        grad_w = grad_f * f - f * dot
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * grad_w ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        f = softmax_param(w)
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive: if stuck, increase temperature (explore more)
        if no_improve > 500:
            temperature *= 1.5
            no_improve = 0
        
        if verbose and t % 1000 == 0:
            print(f"    {label} iter {t}, t={time.monotonic()-t0:.1f}s, obj={best_obj:.10f}, temp={temperature:.6f}")
    
    return best_f, best_obj


def zadoff_chu_init(n, rng):
    """
    Generate a non-negative sequence inspired by Zadoff-Chu sequences.
    Take IFFT of a random phase sequence and take magnitudes.
    """
    # Random phase sequence (like Zadoff-Chu)
    phase = rng.uniform(0, 2 * np.pi, n)
    # Or use quadratic phase for Zadoff-Chu
    # phase = 2 * np.pi * (np.arange(n)**2) % n / n  # for n prime
    z = np.exp(1j * phase)
    # IFFT gives a signal, take magnitude
    a = np.abs(np.fft.ifft(z))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Compute grad_a where grad_a[j] = 2 * sum_k dconv[k] * a[k-j].
    Uses FFT cross-correlation.
    """
    n = len(a)
    D = np.fft.rfft(dconv, fft_size)
    A = np.fft.rfft(a, fft_size)
    h = np.fft.irfft(D * np.conj(A), fft_size)[:n]
    return 2.0 * h


def logsumexp_dconv(conv, temperature, top_k=None):
    """Compute d(logsumexp)/d(conv) via softmax weighting."""
    if top_k is not None and top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        cv = conv[peaks]
        shifted = cv / temperature
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        dconv = np.zeros(len(conv))
        if ws > 1e-30:
            dconv[peaks] = w / (ws * temperature)
        else:
            dconv[peaks[np.argmax(cv)]] = 1.0 / temperature
    else:
        shifted = conv / temperature
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws > 1e-30:
            dconv = w / (ws * temperature)
        else:
            dconv = np.zeros(len(conv))
            dconv[np.argmax(conv)] = 1.0 / temperature
    return dconv


def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                  temp_init=0.001, temp_min=0.00001, top_k_frac=0.3,
                  verbose=False, label=""):
    """
    Adam optimization with softmax parameterization on the simplex.
    """
    n = len(f0)
    fft_size = 1 << (2 * n - 2).bit_length()
    top_k = max(10, int(len(f0) * 2 * top_k_frac))
    
    f = np.maximum(f0, 1e-10).astype(np.float64)
    f = f / f.sum()
    w = np.log(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    no_improve = 0
    temperature = temp_init
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        f = softmax_param(w)
        
        conv = fast_convolve(f)
        dconv = logsumexp_dconv(conv, temperature, top_k)
        grad_f = compute_grad_from_dconv(dconv, f, fft_size)
        
        # Chain rule through softmax
        dot = np.dot(grad_f, f)
        grad_w = f * (grad_f - dot)
        
        # Adam
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * grad_w ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic re-centering for numerical stability
        if t % 1000 == 0:
            w = w - w.mean()
        
        f = softmax_param(w)
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        temperature = max(temp_min, temp_init * (0.995 ** t))
        if no_improve > 1000:
            temperature = min(temperature * 2, temp_init)
            no_improve = 0
        
        if verbose and t % 2000 == 0:
            print(f"    {label} t={t}, elapsed={time.monotonic()-t0:.1f}s, obj={best_obj:.10f}, temp={temperature:.2e}")
    
    return best_f, best_obj


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Zadoff-Chu inspired (multiple random phases)
    for _ in range(3):
        phase = rng.uniform(0, 2 * np.pi, n)
        z = np.exp(1j * phase)
        a = np.abs(np.fft.ifft(z))
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Quadratic phase (more Zadoff-Chu-like)
    if n > 2:
        q = np.arange(n)
        phase = 2 * np.pi * (q * q) % n / n
        z = np.exp(1j * phase)
        a = np.abs(np.fft.ifft(z))
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Gaussian window
    center = n / 2.0
    sigma = n * (0.1 + 0.2 * rng.random())
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Sine-modulated
    x = np.arange(n) / n
    for freq in [1, 2, 3]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Parent (if available)
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
    elif parent is not None:
        # Interpolate parent to length n
        old_n = len(parent)
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, n)
        p_interp = np.interp(new_x, old_x, parent)
        p_interp = np.maximum(p_interp, 1e-10)
        inits.append(p_interp / p_interp.sum())
    
    return inits


def local_search(f0, time_limit, verbose=False):
    """Gradient-guided local search."""
    n = len(f0)
    f = np.maximum(f0, 1e-10).astype(np.float64)
    f = f / f.sum()
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_idx = int(np.argmax(conv))
        current_max = conv[peak_idx]
        
        improved = False
        
        # Compute contributions to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[peak_idx - idx]
        
        # Try reducing the top contributor
        top_idx = idx[np.argmax(contribs)]
        for frac in [0.02, 0.05, 0.1, 0.2]:
            amt = f[top_idx] * frac
            if amt < 1e-12:
                continue
            f_trial = f.copy()
            f_trial[top_idx] -= amt
            # Distribute to random low-value positions
            low_mask = f < np.median(f)
            low_indices = np.where(low_mask)[0]
            if len(low_indices) == 0:
                continue
            targets = rng.choice(low_indices, size=min(5, len(low_indices)), replace=False)
            f_trial[targets] += amt / len(targets)
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f_trial.copy()
                f = f_trial.copy()
                improved = True
                break
        
        # Random pair moves
        if not improved:
            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[i] * frac
                    f_trial = f.copy()
                    f_trial[i] -= amt
                    f_trial[j] += amt
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        improved = True
                        break
        
        if not improved:
            break
    
    return best_f, best_obj
