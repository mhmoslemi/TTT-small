"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Frank-Wolfe with constraint
generation, peak-focused subgradient descent, multi-length exploration,
simulated annealing, and adaptive coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_xy(x, y):
    """Linear cross-convolution via FFT."""
    n = len(x)
    m = len(y)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, active_k, n):
    """Build A_ub matrix rows for given active constraints."""
    K = len(active_k)
    A_ub = np.zeros((K, n))
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    return A_ub


def solve_lp_direction(f, M, n, active_k):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_k, g >= 0."""
    if len(active_k) == 0:
        return None
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, active_k, n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, top_k=40, verbose=False):
    """LP-Frank-Wolfe with constraint generation and backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem_time = time_limit - (time.monotonic() - t0)
        if rem_time < 0.5:
            break

        # Select initial active constraints: top-k by convolution value
        k_sorted = np.argsort(conv)[::-1]
        init_k = k_sorted[:top_k].tolist()

        # Constraint generation: start with top-k, add violated ones
        M_curr = M * m_exp
        active_k = init_k[:]
        g = None

        for cg_round in range(4):
            g = solve_lp_direction(f, M_curr, n, active_k)
            if g is None:
                m_exp *= 1.15
                if m_exp > 6.0:
                    m_exp = 1.0
                break
            # Check for violated constraints not in active set
            fg = fast_convolve_xy(f, g)
            active_set = set(active_k)
            violations = []
            for k in range(len(fg)):
                if k not in active_set and fg[k] > M_curr + 1e-10:
                    violations.append((fg[k], k))
            if not violations:
                break
            violations.sort(reverse=True)
            for _, k in violations[:15]:
                active_k.append(k)

        if g is None:
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue

        # Backtracking line search
        alpha = 1.0
        f_trial = f + alpha * (g - f)
        f_trial = project_simplex(np.maximum(f_trial, 0))
        conv_trial = fast_convolve(f_trial)
        M_trial = conv_trial.max()

        if M_trial >= M:
            # Try smaller steps
            for shrunk in [0.5, 0.25, 0.1, 0.05]:
                alpha = shrunk
                f_trial = f + alpha * (g - f)
                f_trial = project_simplex(np.maximum(f_trial, 0))
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < M:
                    break
                alpha = 0.0

        if alpha > 1e-10:
            f_new = f + alpha * (g - f)
            f_new = project_simplex(np.maximum(f_new, 0))
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            conv = conv_new

            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 20, 8)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def peak_subgradient_descent(f0, time_limit, top_peaks=30, lr_init=0.001,
                             mom=0.93, temp_init=0.00005):
    """Peak-focused subgradient descent with momentum and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1

        # Select top peaks with softmax weighting
        k_act = min(top_peaks, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Update with momentum
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(it_count * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0003)
            cur_temp = temp_init * 2.0
            no_improve = 0

    return best_f, best_obj


def peak_coord_descent(f0, time_limit):
    """Coordinate descent focused on reducing peak autocorrelation values."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Use a fine grid of multipliers for coordinate updates
    multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved_in_pass = False
        conv = fast_convolve(f)
        M = conv.max()

        # Process coordinates in order of their contribution to the peak
        peak_k = int(np.argmax(conv))
        # Sort coordinates by |f[peak_k - j]| (their subgradient contribution)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        order = list(range(js, je + 1))
        order.sort(key=lambda j: -f[peak_k - j])

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved_in_pass = True

        if not improved_in_pass:
            # Try a few random coordinate moves
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            for _ in range(5):
                j = rng.integers(0, n)
                old_val = f[j]
                if old_val < 1e-14:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                for mult in [0.5, 1.5, 0.2, 2.0]:
                    new_val = old_val * mult
                    remaining = 1.0 - new_val
                    if remaining <= 1e-10 or new_val < 0:
                        continue
                    scale = remaining / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-15:
                        best_obj = val
                        best_f = f.copy()
                        improved_in_pass = True
                        break
            if not improved_in_pass:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    # Enforce Hermitian symmetry for real output
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(np.fft.irfft(spectrum, fft_size))[:n]
        sig = np.maximum(sig, 0)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def simulated_annealing(f0, time_limit, n_cool=1000, t_start=0.005, t_end=0.0005):
    """Simulated annealing with structured perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / n_cool)
        temperature = t_start * (t_end / t_start) ** progress

        # Generate perturbation
        pert_type = rng.integers(0, 4)
        f_trial = f.copy()

        if pert_type == 0:
            # Random Gaussian noise
            scale = temperature * 0.5
            f_trial = f_trial + rng.standard_normal(n) * scale
        elif pert_type == 1:
            # Move mass between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amt = f[i] * temperature * 2
                amt = min(amt, f[i] * 0.9)
                f_trial[i] -= amt
                f_trial[j] += amt
        elif pert_type == 2:
            # Block swap
            block_size = max(1, n // 32)
            i = rng.integers(0, n - block_size)
            j = rng.integers(0, n - block_size)
            if abs(i - j) > block_size:
                f_trial[i:i+block_size], f_trial[j:j+block_size] = \
                    f_trial[j:j+block_size].copy(), f_trial[i:i+block_size].copy()
        else:
            # Mirror perturbation (flip a random segment)
            seg_len = max(2, n // 16)
            i = rng.integers(0, max(1, n - seg_len))
            seg = f_trial[i:i+seg_len].copy()
            f_trial[i:i+seg_len] = seg[::-1]

        f_trial = np.maximum(f_trial, 0)
        s_trial = f_trial.sum()
        if s_trial < 1e-15:
            continue
        f_trial /= s_trial

        obj_trial = fast_convolve(f_trial).max()

        # Acceptance criterion
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_f = f_trial.copy()
            f = f_trial
        elif obj_trial < f.sum() * 0:  # always true, placeholder
            pass
        else:
            current_obj = fast_convolve(f).max()
            delta = obj_trial - current_obj
            if delta < 0:
                f = f_trial
            else:
                accept_prob = np.exp(-delta / max(temperature, 1e-10))
                if rng.random() < accept_prob:
                    f = f_trial

    return best_f, best_obj

            t3 = min(120, rem * 0.12)
            print(f"  Time: {t3:.0f}s")
            a_opt, obj = peak_coord_descent(best_seq, time_limit=t3)
            try_seq(a_opt, "peak_cd")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length ({best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [896, 1024, 1280]
        rem = deadline - time.monotonic()
        for target_n in target_lengths:
            if rem < 40:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            # Also try flat spectrum mix
            fs = flat_spectrum_init(target_n, seed=target_n)
            f_mix = project_simplex(0.75 * f_interp + 0.25 * fs)
            t_alloc = min(rem * 0.25, 120)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = lpfw_optimize(f_mix, time_limit=t_alloc, top_k=45, verbose=False)
            try_seq(a_opt, f"len_{target_n}")
            rem = deadline - time.monotonic()

    # === Phase 5: Multi-restart LP-FW ===
    print(f"\n=== Phase 5: Multi-restart ({best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed)

        restarts = [
            ("noise_small", project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)),
            ("rev_blend", project_simplex(0.7 * best_seq + 0.3 * best_seq[::-1])),
            ("gauss", project_simplex(
                np.exp(-2.0 * (np.arange(n_best) - n_best / 2.0) ** 2 / (0.6 * n_best) ** 2))),
            ("bimodal", project_simplex(
                np.exp(-4.0 * ((np.arange(n_best) - n_best * 0.3) ** 2) / (0.15 * n_best) ** 2) +
                np.exp(-4.0 * ((np.arange(n_best) - n_best * 0.7) ** 2) / (0.15 * n_best) ** 2))),
            ("flat_blend", project_simplex(0.85 * best_seq + 0.15 * np.ones(n_best) / n_best)),
        ]

        rem = deadline - time.monotonic()
        for name, f_init in restarts:
            if rem < 30:
                break
            t_alloc = min(rem / (len(restarts) + 1), 80)
            print(f"  Restart [{name}], t={t_alloc:.0f}s")
            a_opt, obj = lpfw_optimize(f_init, time_limit=t_alloc, top_k=40, verbose=False)
            try_seq(a_opt, f"restart_{name}")
            rem = deadline - time.monotonic()

    # === Phase 6: Simulated annealing ===
    print(f"\n=== Phase 6: Simulated Annealing ({best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t6 = min(100, rem * 0.12)
            print(f"  Time: {t6:.0f}s")
            a_opt, obj = simulated_annealing(best_seq, time_limit=t6,
                                             n_cool=500, t_start=0.003, t_end=0.0003)
            try_seq(a_opt, "sa")

    # === Phase 7: Final subgradient polish ===
    print(f"\n=== Phase 7: Final SG Polish ({best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t7 = min(80, rem * 0.10)
            print(f"  Time: {t7:.0f}s")
            a_opt, obj = peak_subgradient_descent(best_seq, time_limit=t7,
                                                   top_peaks=50, lr_init=0.0005,
                                                   mom=0.95, temp_init=0.00002)
            try_seq(a_opt, "final_sg")

    # === Phase 8: Final LP-FW polish ===
    print(f"\n=== Phase 8: Final LP-FW ({best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t8 = min(60, rem * 0.08)
            a_opt, obj = lpfw_optimize(best_seq, time_limit=t8, top_k=60, verbose=False)
            try_seq(a_opt, "final_lpfw")

            # One more SG pass
            rem = deadline - time.monotonic()
            if rem > 10:
                a_opt2, obj2 = peak_subgradient_descent(a_opt, time_limit=min(30, rem * 0.5),
                                                         top_peaks=60, lr_init=0.0003,
                                                         mom=0.96, temp_init=0.00001)
                try_seq(a_opt2, "final_sg2")

    # === Phase 9: Tiny perturbations around best ===
    print(f"\n=== Phase 9: Perturbations ({best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 999)
        rem = deadline - time.monotonic()
        for i in range(6):
            if rem < 5:
                break
            noise = rng.standard_normal(n_best) * (0.0005 + 0.0005 * i)
            f_pert = project_simplex(best_seq + noise)
            conv_pert = fast_convolve(f_pert)
            try_seq(f_pert, f"perturb_{i}")
            # Quick LP-FW if improved
            rem = deadline - time.monotonic()
            if rem > 10:
                t_p = min(15, rem * 0.3)
                a_opt, obj = lpfw_optimize(f_pert, time_limit=t_p, top_k=35, verbose=False)
                try_seq(a_opt, f"perturb_lpfw_{i}")
            rem = deadline - time.monotonic()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
