    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    n = len(v)

        return np.array([max(v[0], 0.0)])

"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, top_k=40, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        active_k = None
        if it > 3:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            active_k = k_sorted.tolist()
        if it % 30 == 0:
            active_k = None

        g = solve_lp_direction(f, M, n, active_k=active_k)
        if g is None:
            M *= 1.1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            M *= 1.05
            continue

        # Simple line search
        best_alpha = 0.0
        best_trial_obj = conv.max()
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            f = f / f.sum()
            conv = fast_convolve(f)
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_f = f.copy()
                M = conv.max()
                no_improve = 0
                if verbose and it % 20 == 0:

            else:
                no_improve += 1
        else:

            M *= 1.05

def enhanced_coord_desc(f0, time_limit, verbose=False):
    """Enhanced coordinate descent with proper 1D optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            # Try a range of multipliers
            best_mult = 1.0
            best_trial = best_obj

            # Coarse scan
            mults = np.concatenate([
                np.linspace(0.0, 0.5, 6),
                np.linspace(0.5, 1.5, 6),
                [2.0, 3.0, 5.0]
            ])

            for mult in mults:
                new_val = old_val * mult
                if new_val < 0 or new_val > 1.0 - 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                val = fast_convolve(f_trial).max()
                if val < best_trial:
                    best_trial = val
                    best_mult = mult

            if best_mult != 1.0:
                new_val = old_val * best_mult
                scale = (1.0 - new_val) / denom
                f[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                conv = fast_convolve(f)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_f = f.copy()
                    improved = True
                else:
                    # Revert
                    f[j] = old_val
                    f[mask] *= 1.0 / scale
                    f = f / f.sum()

        if not improved:
            stuck += 1
            if stuck > 3:
                break
        else:
            stuck = 0
            if verbose:
                print(f"    CD pass: obj={best_obj:.10f}")

def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing with structured pair moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    T = 0.001  # Initial temperature
    T_min = 1e-6
    cooling = 0.9997
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - t0 < time_limit and T > T_min:
        it += 1

        # Structured move: pick i from peak contributors, j from non-contributors
        # Find the max peak
        k_max = np.argmax(conv)
        # Coordinates that contribute to conv[k_max]
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        contributors = np.arange(i_start, i_end + 1)

        # Weight by contribution
        contrib_weights = np.abs(f[contributors] * f[k_max - contributors])
        total_w = contrib_weights.sum()
        if total_w < 1e-15:
            contributors = np.zeros(n, dtype=float)
            contributors[:] = 1.0
            total_w = 1.0
        else:
            contributors_w = np.zeros(n)
            contributors_w[contributors] = contrib_weights / total_w

        # Pick i from contributors
        i = rng.choice(n, p=contributors_w)

        # Pick j from non-contributors (or random if all are contributors)
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[contributors] = False
        # Also exclude i itself
        non_contrib[i] = False

        if non_contrib.sum() > 0 and rng.random() < 0.7:
            # Bias toward low-value non-contributors
            j_candidates = np.where(non_contrib)[0]
            j_weights = 1.0 + 10.0 * (1.0 - f[j_candidates] * n)  # favor low values
            j_weights = j_weights / j_weights.sum()
            j = j_candidates[rng.choice(len(j_candidates), p=j_weights)]
        else:
            j = rng.integers(n)
            if j == i:
                j = (j + 1) % n

        # Amount to move
        delta = min(f[i], f[j]) * rng.uniform(0.02, 0.15)

        # Apply move
        f_trial = f.copy()
        f_trial[i] -= delta
        f_trial[j] += delta
        if f_trial[i] < 0:
            f_trial[i] = 0
            f_trial[j] += delta
        f_trial = np.maximum(f_trial, 0)
        f_trial = f_trial / f_trial.sum()

        conv_trial = fast_convolve(f_trial)
        obj_trial = conv_trial.max()
        obj_curr = conv.max()

        # Metropolis acceptance
        delta_obj = obj_trial - obj_curr
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            f = f_trial
            conv = conv_trial
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_f = f.copy()
                if verbose and it % 500 == 0:
                    print(f"    SA iter {it}: obj={best_obj:.10f}, T={T:.6f}")

        T *= cooling

        # Restart from best if stuck
        if it % 2000 == 0 and f is not best_f:
            f = best_f.copy()
            conv = fast_convolve(f)

    return best_f, best_obj

def smooth_max_optimize(f0, time_limit, verbose=False):
    """Optimize using smooth max approximation with gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Anneal the temperature from high to low
    T_start = 50.0
    T_end = 5.0
    n_anneals = 5
    T_schedule = np.geomspace(T_start, T_end, n_anneals)

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        phase_time = min((time_limit - (time.monotonic() - t0)) / (n_anneals + 1), 60)

        # Gradient descent with momentum
        v = np.zeros(n)
        lr = 0.01
        mom = 0.9
        it = 0
        phase_t0 = time.monotonic()

        while time.monotonic() - phase_t0 < phase_time:
            it += 1
            conv = fast_convolve(f)

            # Softmax weights
            scaled = T * conv
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()

            ep_conv = np.dot(p, conv)

            # Compute S[j] = sum_k p[k] * a[k-j] for valid indices
            S = np.zeros(n)
            for j in range(n):
                k_min = max(0, j)
                k_max = min(2*n-2, j+n-1)
                if k_min <= k_max:
                    S[j] = np.dot(p[k_min:k_max+1], f[0:k_max-k_min+1])

            # Gradient of smooth max w.r.t. f (before constraint projection)
            # dF/df_j = 4n * f_j * (S[j] - ep_conv)
            grad = 4.0 * n * f * (S - ep_conv)

            # Project gradient to maintain sum(f) = 1
            grad_centered = grad - np.mean(grad)

            # Momentum update
            v = mom * v - lr * grad_centered
            f_new = f + v
            f_new = project_simplex(f_new)

            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()

            f = f_new
            lr *= 0.9999

            if verbose and it % 500 == 0:
                print(f"    SmoothMax T={T:.1f} iter={it}: obj={obj_new:.10f}")

            if it > 5000:
                break

    return best_f, best_obj
