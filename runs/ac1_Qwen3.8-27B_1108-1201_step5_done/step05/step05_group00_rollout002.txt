"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FW-LP with sparse convolution matrix and warm starts
2. Adaptive M relaxation for escaping local optima
3. Multi-n exploration (768, 1024, 1536)
4. Smoothing-based polishing
5. Stochastic perturbations for diversification
"""

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_structure(n):
    """Precompute structure for convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    return rows, cols

def solve_lp_direction(f, M, rows, cols):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns g normalized."""
    n = len(f)
    vals = f[rows - cols]
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def fw_optimize(f0, time_limit, rows=None, cols=None, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if rows is None:
        rows, cols = build_conv_structure(n)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        it += 1
        
        # Solve LP direction
        g = solve_lp_direction(f, M * m_scale, rows, cols)
        if g is None:
            m_scale = min(m_scale * 1.5, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
