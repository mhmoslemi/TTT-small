import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_peak(a):
    if a.sum() < 1e-15:
        return 0.0, np.zeros(2*len(a)-1)
    c = fast_convolve(a / a.sum())
    return c.max(), c

def objective_value(a):
    """Compute 2*n*max_b/sum_a^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

So for uniform, the objective is `2n * 1/n = 2`. ✓

Now, if we make the sequence non-uniform, we can reduce some peaks at the cost of increasing others. The key is to find a sequence where ALL peaks are below `1/n`.

But wait, `conv[n-1] = sum_j a[j]*a[n-1-j]`. By AM-GM or just basic inequality:
`sum_j a[j]*a[n-1-j] >= 0` (trivially, since all non-negative).
But can we make it less than `1/n`?

If `a` is not symmetric, say `a[j]` is large for small `j` and small for large `j`, then `a[j]*a[n-1-j]` would be small for all `j`, making `conv[n-1]` small. But then `conv[0] = a[0]^2` would be large.

So there's a fundamental tradeoff. The minimum of `max_k conv[k]` is achieved when all the `conv[k]` values are equalized (or as close as possible).

The sum of all `conv[k]` values is `(sum a)^2 = 1` (for normalized `a`). There are `2n-1` values. If they were all equal, each would be `1/(2n-1)`, and the objective would be `2n/(2n-1) ≈ 1`. But they can't all be equal due to the structure constraints.

"""

2. Stale-sequence Frank-Wolfe with simplified LP

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_gradient(k, a, n):
    """Exact gradient of (a*a)[k] w.r.t. a, using the formula:
    d(conv[k])/d(a[j]) = 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)], else 0."""
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    grad[js:je + 1] = 2.0 * a[k - je:k - js + 1][::-1]
    return grad


def conv_max_and_grad(a):
    """Return (max_value, peak_position, gradient_at_peak)."""
    n = len(a)
    conv = fast_convolve(a)
    k = int(np.argmax(conv))
    g = peak_gradient(k, a, n)
    return conv.max(), k, g, conv


def weighted_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient over top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[0] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        sg += w[i] * peak_gradient(int(peaks[i]), a, n)
    return sg, peaks, w


def solve_lp_min_peak(f, n, top_constraints=None):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k (or top_k), g >= 0, sum(g) = 1.
    Returns (g, t) or (None, None)."""
    if top_constraints is None:
        top_constraints = 2 * n - 1

    # To speed up, only use the top constraints (peaks of f*f)
    ff = fast_convolve(f)
    if top_constraints < len(ff):
        active_constraints = np.argpartition(ff, -top_constraints)[-top_constraints:]
    else:
        active_constraints = np.arange(len(ff))

    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = len(active_constraints)
    A_ub = np.zeros((n_constraints, n + 1))
    for idx, k in enumerate(active_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[idx, js:je + 1] = f[k - je:k - js + 1][::-1]
        A_ub[idx, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=30,
                         temp=0.0001, verbose=False):
    """Subgradient descent with backtracking line search and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    cur_mom = momentum
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, _, _ = weighted_subgradient(f, conv, top_k, cur_temp)

        v = cur_mom * v - cur_lr * sg
        f_trial = f + v

        # Backtracking line search
        best_alpha = 1.0
        best_val = fast_convolve(project_simplex(f_trial)).max()
        val = best_val
        alpha = 1.0
        for _ in range(12):
            f_new = project_simplex(f + alpha * v)
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn < 1e-15:
                break
            f_new /= sn
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.7
            if alpha < 1e-4:
                break

        f_new = f + best_alpha * v
        f_new = project_simplex(np.maximum(f_new, 0.0))
        f_new = f_new / f_new.sum()
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.005
        cur_mom = min(0.97, cur_mom + 0.0001)

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0005)
            no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, top_constraints=100, verbose=False):
    """Frank-Wolfe with LP oracle and stale sequence (multiple steps per LP solve)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP to get best direction
        g, lp_t = solve_lp_min_peak(f, n, top_constraints=top_constraints)
        if g is None:
            g = np.ones(n) / n
            lp_t = M

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Stale sequence: take multiple steps in the same direction
        direction = g - f
        stale_steps = 8
        alpha = 1.0
        for si in range(stale_steps):
            if time.monotonic() - t0 > time_limit:
                break
            # Line search for this step
            best_a = 0.0
            best_val = M
            current_alpha = alpha
            for _ in range(8):
                ft = f + current_alpha * direction
                ft = project_simplex(np.maximum(ft, 0.0))
                ft = ft / ft.sum()
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_a = current_alpha
                current_alpha *= 0.7

            if best_a > 1e-12:
                f = f + best_a * direction
                f = project_simplex(np.maximum(f, 0.0))
                f = f / f.sum()
                M = fast_convolve(f).max()
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            alpha *= 0.5

        if verbose and it % 5 == 0:
            print(f"      LP-FW iter {it}: obj={best_obj:.10f}")

        if no_improve > 40:
            rng = np.random.default_rng(it * 6271 + 7)
            scale = 0.001 + 0.002 * min(no_improve // 30, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def pairwise_cd(f0, time_limit, verbose=False):
    """Coordinate descent using pairwise exchanges on peak-contributing elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find top peaks
        top_k = min(4, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # For each pair of contributing elements, try adjusting
            indices = list(range(js, je + 1))
            if len(indices) > 40:
                # Subsample for efficiency
                rng_sub = np.random.default_rng(passes * 31 + peak_k)
                indices = list(rng_sub.choice(indices, size=40, replace=False))

            for ii in range(len(indices)):
                if time.monotonic() - t0 > time_limit:
                    break
                i = indices[ii]
                for jj in range(ii + 1, len(indices)):
                    j = indices[jj]
                    if time.monotonic() - t0 > time_limit:
                        break

                    # Try moving mass from i to j and vice versa
                    for frac in [0.1, 0.2, 0.3, 0.5]:
                        # Move mass from i to j
                        transfer = f[i] * frac
                        if transfer > f[i] * 0.9:
                            continue
                        f_trial = f.copy()
                        f_trial[i] -= transfer
                        f_trial[j] += transfer
                        f_trial = np.maximum(f_trial, 0)
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break

                        # Move mass from j to i
                        transfer2 = f[j] * frac
                        if transfer2 > f[j] * 0.9:
                            continue
                        f_trial2 = f.copy()
                        f_trial2[j] -= transfer2
                        f_trial2[i] += transfer2
                        f_trial2 = np.maximum(f_trial2, 0)
                        f_trial2 = f_trial2 / f_trial2.sum()
                        val2 = fast_convolve(f_trial2).max()
                        if val2 < best_obj - 1e-14:
                            f = f_trial2
                            best_obj = val2
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break

        if not improved:
            no_improve += 1
            if no_improve > 8:
                # Random exploration
                rng = np.random.default_rng(passes * 7919 + 13)
                for _ in range(10):
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j or f[i] < 1e-10 or f[j] < 1e-10:
                        continue
                    frac = float(rng.uniform(0.1, 0.4))
                    transfer = f[i] * frac
                    f_trial = f.copy()
                    f_trial[i] -= transfer
                    f_trial[j] += transfer
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and passes % 10 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def spectral_init(n, seed=42):
    """Create a diverse initialization using spectral reweighting."""
    rng = np.random.default_rng(seed)
    f = rng.dirichlet(np.ones(n))

    # Apply spectral flattening iterations
    A = np.fft.rfft(f)
    for _ in range(20):
        # Target: constant amplitude (CAZAC-like)
        target_amp = np.sqrt(n)
        phase = np.angle(A)
        A_new = target_amp * np.exp(1j * phase)
        # Blend with current
        A = 0.7 * A + 0.3 * A_new
        f_new = np.fft.irfft(A, 2 * n - 1)[:n]
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-10:
            f = f_new / s
        else:
            break

    return f


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print("\n=== Phase 2: LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(350, rem * 0.22)
        if t2 > 25:
            print(f"  n={len(best_seq)}, t={t2:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, top_constraints=80, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 3: Pairwise CD ===
    print("\n=== Phase 3: Pairwise CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(250, rem * 0.18)
        if t3 > 20:
            print(f"  n={len(best_seq)}, t={t3:.0f}s")
            a_opt, _ = pairwise_cd(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "pairwise_cd")

    # === Phase 4: Try different lengths ===
    print("\n=== Phase 4: Multi-length exploration ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = set()
        for delta in [-128, -64, 64, 128, 256, -256]:
            ln = len(base) + delta
            if 256 <= ln <= 2048:
                candidate_lengths.add(ln)
        # Also try power-of-2 lengths
        for pn in [384, 512, 768, 1024, 1536]:
            if 256 <= pn <= 2048:
                candidate_lengths.add(pn)
        candidate_lengths = sorted(candidate_lengths)

        for target_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_alloc = min(rem * 0.25, 180)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            # Quick subgradient
            a_opt, _ = subgradient_optimize(f_interp, time_limit=t_alloc * 0.6,
                                            lr=0.003, momentum=0.94,
                                            top_k=30, temp=0.00006)
            try_seq(a_opt, f"mg_n{target_n}")

    # === Phase 5: Spectral init + optimize ===
    print("\n=== Phase 5: Spectral initialization ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        for sp in range(3):
            if rem < 50:
                break
            si = min(rem * 0.2, 150)
            if si < 20:
                break
            f_spec = spectral_init(base_n, seed=100 + sp * 37)
            a_opt, _ = subgradient_optimize(f_spec, time_limit=si,
                                            lr=0.003, momentum=0.94,
                                            top_k=30, temp=0.00006)
            try_seq(a_opt, f"spectral_{sp}")
            rem = deadline - time.monotonic()

    # === Phase 6: Final SG polish ===
    print(f"\n=== Phase 6: Final SG polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.3, 300)
        if t_sg > 20:
            print(f"  n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                            lr=0.002, momentum=0.95,
                                            top_k=50, temp=0.00003, verbose=True)
            try_seq(a_opt, "sg_polish")

        # Quick LP-FW
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.2, 120)
        if t_lp > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp, top_constraints=60)
            try_seq(a_opt, "lpfw_polish")

    # === Phase 7: Perturbation + re-optimize ===
    print(f"\n=== Phase 7: Perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            noise_scale = 0.0008 + 0.0004 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_p = min(rem * 0.3, 40)
            a_opt, _ = subgradient_optimize(f_pert, time_limit=t_p,
                                            lr=0.001, momentum=0.95,
                                            top_k=25, temp=0.00002)
            try_seq(a_opt, f"perturb_{p}")

    # === Phase 8: Final LP-FW polish ===
    print(f"\n=== Phase 8: Final LP polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_final = min(rem, 150)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final, top_constraints=50)
            try_seq(a_opt, "final_lp")

        # Very short SG
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem, 30),
                                            lr=0.0008, momentum=0.96,
                                            top_k=20, temp=0.00001)
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
