def smoothed_grad_optimize(a0, time_limit, T_init=None):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones(len(a)) / len(a)

        a = a / s

        T_init = fast_convolve(a).max() * 0.05

    T = T_init
    lr = 0.03
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_M = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

        if remaining < 0.1:
            break

        M = conv.max()

        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-15:
            break
        p = p / p_sum

        # Gradient: 2 * sum_k p_k * (a shifted by k)
        grad = np.zeros(n)
        for k in range(len(conv)):
            if p[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            j_idx = np.arange(js, je + 1)
            grad[j_idx] += 2.0 * p[k] * a[k - j_idx]

        # Momentum update
        v = 0.9 * v - lr * grad
        candidate = project_simplex(a + v)

        # Backtracking
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            candidate = project_simplex(a - shrink * lr * (grad / (lr + 1e-30) * lr))
            # Simpler: just try the momentum step and a few alternatives
            break

        # Simpler approach: use the momentum step directly
        candidate = project_simplex(a + v)
        val = fast_convolve(candidate).max()

        if val < best_M - 1e-14:
            a = candidate
            best_M = val
            best_a = a.copy()
            no_improve = 0
        else:
            # Revert and try smaller step
            candidate = project_simplex(a - lr * 0.5 * grad)
            val2 = fast_convolve(candidate).max()
            if val2 < best_M - 1e-14:
                a = candidate
                best_M = val2
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            lr *= 0.98

            v = np.zeros(n)
            no_improve = 0

        if no_improve == 0 and lr < 0.03:

def fourier_subspace_optimize(a0, time_limit, n_modes=8):
    """
    Optimize in a Fourier subspace:
    a_i = softmax-like parameterization using cos/sin modes.
    Then do a full FW pass from the result.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    x = np.arange(n, dtype=np.float64)

    # Build Fourier basis
    basis = np.ones((1, n))  # DC
    for k in range(1, n_modes + 1):
        basis = np.vstack([basis, np.cos(2 * np.pi * k * x / n)])
        if k * 2 < n:
            basis = np.vstack([basis, np.sin(2 * np.pi * k * x / n)])

    m = basis.shape[0]  # number of basis vectors
    # We want a = basis @ c such that a >= 0 and sum(a) = 1
    # Since basis[0] = ones, sum(a) = n * c[0] + sum of other terms
    # Actually sum(a) = sum_i (basis @ c)_i = (sum_i basis[i,j]) * c[j]
    # For DC: sum = n. For cos/sin modes: sum ≈ 0.

    # Objective: max of conv(basis@c, basis@c)
    # This is complex to compute analytically, so use numerical evaluation.

    def objective(c):
        a_trial = basis @ c
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            return 1e10
        a_trial = a_trial / s
        return fast_convolve(a_trial).max()

    # Initial c: project a onto the Fourier basis
    # Use least squares with the constraint a >= 0 (approximate)
    c0 = np.zeros(m)
    c0[0] = 1.0 / n  # DC component
    # Add small perturbations from the current a
    for j in range(1, m):
        c0[j] = np.dot(a, basis[j]) / n

    best_c = c0.copy()
    best_obj = objective(c0)

    t0 = time.monotonic()
    lr = 0.001
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        a_cur = np.maximum(basis @ best_c, 0.0)
        s_cur = a_cur.sum()
        if s_cur < 1e-10:
            break
        a_cur = a_cur / s_cur

        conv = fast_convolve(a_cur)
        k_peak = int(np.argmax(conv))

        # Numerical gradient of objective w.r.t. c
        # Use the subgradient: 2 * (a_cur shifted by k_peak) projected onto basis
        grad_a = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        j_idx = np.arange(js, je + 1)
        grad_a[j_idx] = 2.0 * a_cur[k_peak - j_idx]

        grad_c = basis.T @ (grad_a / s_cur - grad_a * a_cur / s_cur)
        # Simplified: just use basis.T @ grad_a normalized
        grad_c = basis.T @ grad_a
        gn = np.linalg.norm(grad_c)
        if gn > 1e-15:
            grad_c = grad_c / gn

        candidate_c = best_c - lr * grad_c
        cand_obj = objective(candidate_c)

        if cand_obj < best_obj - 1e-14:
            best_c = candidate_c
            best_obj = cand_obj
            no_improve = 0
            lr = min(lr * 1.01, 0.01)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 50:
                # Try random direction
                rng = np.random.default_rng(it * 7 + 3)
                direction = rng.standard_normal(m)
                direction[0] = 0  # Keep DC roughly fixed
                for step in [lr * 0.1, lr * 0.5, lr]:
                    trial_c = best_c + step * direction / (np.linalg.norm(direction) + 1e-15)
                    trial_obj = objective(trial_c)
                    if trial_obj < best_obj - 1e-14:
                        best_c = trial_c
                        best_obj = trial_obj
                        no_improve = 0
                        break
                else:
                    if no_improve > 200:
                        break

    a_result = np.maximum(basis @ best_c, 0.0)
    a_result = a_result / a_result.sum()
    return a_result


def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian center bump (inverse: reduce center)
    for sigma_frac in [0.1, 0.15, 0.25, 0.35]:
        sigma = sigma_frac * n
        w = 1.0 - 0.5 * np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        w = np.maximum(w, 0.1)
        inits.append(w / w.sum())

    # Edge concentration
    for sigma_frac in [0.05, 0.1, 0.15]:
        sigma = sigma_frac * n
        w = np.exp(-0.5 * (x / sigma) ** 2) + np.exp(-0.5 * ((x - (n - 1)) / sigma) ** 2)
        inits.append(w / w.sum())

    # Cosine modulation
    for eps in [0.1, 0.3, 0.5, 0.7]:
        w = 1.0 + eps * np.cos(2 * np.pi * x / n)
        w = np.maximum(w, 0.01)
        inits.append(w / w.sum())

    # Two-frequency
    for eps1, eps2 in [(0.3, 0.2), (0.5, 0.3), (0.2, 0.5)]:
        w = 1.0 + eps1 * np.cos(2 * np.pi * x / n) + eps2 * np.cos(4 * np.pi * x / n)
        w = np.maximum(w, 0.01)
        inits.append(w / w.sum())

    # Two peaks
    for frac in [0.2, 0.25, 0.3]:
        center1 = frac * (n - 1)
        center2 = (1 - frac) * (n - 1)
        sigma = 0.1 * n
        w = np.exp(-0.5 * ((x - center1) / sigma) ** 2) + np.exp(-0.5 * ((x - center2) / sigma) ** 2)
        inits.append(w / w.sum())

    # Linear ramp
    w = np.linspace(0.5, 1.5, n)
    inits.append(w / w.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    return inits


def kkt_polish(a0, time_limit):
    """
    KKT-inspired polishing: identify active set, solve for multipliers,
    then do a targeted optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Find active set (peaks within 1% of max)
        threshold = M * 0.995
        active_k = np.nonzero(conv >= threshold)[0]

        if len(active_k) == 0:
            break

        # For each active k, the gradient contribution is 2 * p_k * (a shifted by k)
        # We want to find a direction that reduces all active peaks simultaneously
        # Use: grad = sum over active k of (a shifted by k), normalized

        grad = np.zeros(n)
        for k in active_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            j_idx = np.arange(js, je + 1)
            grad[j_idx] += a[k - j_idx]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        grad /= gn

        # Try small steps in the negative gradient direction
        improved = False
        for lr in [0.001, 0.002, 0.005, 0.01, 0.02]:
            candidate = project_simplex(a - lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_M - 1e-15:
                a = candidate
                best_M = val
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try positive direction (sometimes helps by filling in valleys)
            for lr in [0.001, 0.002, 0.005]:
                candidate = project_simplex(a + lr * grad)
                val = fast_convolve(candidate).max()
                if val < best_M - 1e-15:
                    a = candidate
                    best_M = val
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try random perturbation in the support of high values
            rng = np.random.default_rng((it * 12345 + 7) % (2**31))
            noise = rng.standard_normal(n) * 0.001
            candidate = project_simplex(a + noise)
            val = fast_convolve(candidate).max()
            if val < best_M - 1e-15:
                a = candidate
                best_M = val
                best_a = a.copy()
            elif it > 100:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid optimization.
    Returns the best sequence found minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Multi-n parametric + quick FW (~200s) ===
    print("\n=== Phase 1: Parametric + Quick FW ===")
    t_start = time.monotonic()

    # Candidate n values
    if n_parent in [256, 384, 512, 768, 1024]:
        candidate_ns = [n_parent, n_parent // 2, n_parent * 2]
    else:
        candidate_ns = [256, 384, 512, 768, 1024]
    candidate_ns = [n for n in candidate_ns if 64 <= n <= 2048]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1200:
            break

        # Generate initializations
        inits = parametric_init(n_try, rng)

        # If we have a parent, interpolate
        if parent is not None and n_try != n_parent:
            p_interp = np.interp(np.linspace(0, 1, n_try),
                                 np.linspace(0, 1, n_parent), parent)
            p_interp = np.maximum(p_interp, 0)
            ps = p_interp.sum()
            if ps > 1e-10:
                inits.insert(0, p_interp / ps)

        # Quick evaluate all inits
        best_init = None
        best_init_score = float('inf')
        for i, init in enumerate(inits):
            sc = try_seq(init, f"init_n{n_try}_{i}")
            if sc < best_init_score:
                best_init_score = sc
                best_init = init

        if best_init is None:
            best_init = np.ones(n_try) / n_try

        # Quick FW-LP
        remaining = deadline - time.monotonic() - 1100
        if remaining > 20:
            t_alloc = min(remaining, 80)
            a_opt, _ = fw_lp_optimize(best_init, time_limit=t_alloc)
            try_seq(a_opt, f"fw_quick_n{n_try}")

        # Quick smoothed gradient
        remaining = deadline - time.monotonic() - 1000
        if remaining > 15 and best_score < 2.0:
            a_opt2, _ = smoothed_grad_optimize(best_seq, time_limit=min(remaining, 30))
            try_seq(a_opt2, f"sg_quick_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Deep FW-LP on best n (~500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 700
        if remaining > 30:
            t_alloc = min(remaining, 400)
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_deep")

            # Multiple perturbed restarts
            for scale in [0.005, 0.01, 0.02, 0.05]:
                remaining = deadline - time.monotonic() - 500
                if remaining < 15:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=min(remaining, 60))
                try_seq(a_opt2, f"fw_pert_{scale}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Smoothed gradient with multiple T (~300s) ===
    print("\n=== Phase 3: Smoothed Gradient ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        T_values = [0.0001, 0.0005, 0.001, 0.003, 0.01]
        for T in T_values:
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            a_opt, _ = smoothed_grad_optimize(best_seq, time_limit=t_alloc, T_init=T)
            try_seq(a_opt, f"sg_T{T}")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Fourier subspace (~200s) ===
    print("\n=== Phase 4: Fourier Subspace ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 150)
            a_opt, _ = fourier_subspace_optimize(best_seq, time_limit=t_alloc, n_modes=10)
            try_seq(a_opt, "fourier")

            # Then do a quick FW from the Fourier result
            remaining = deadline - time.monotonic() - 200
            if remaining > 15:
                a_opt2, _ = fw_lp_optimize(a_opt, time_limit=min(remaining, 50))
                try_seq(a_opt2, "fourier_fw")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: KKT Polish + Final refinement (~250s) ===
    print("\n=== Phase 5: KKT Polish & Final ===")
    t_start = time.monotonic()

    if best_seq is not None:
        # KKT polish
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 100)
            a_opt = kkt_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "kkt_polish")

        # Final FW
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 60)
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        # Final smoothed gradient with very low T
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = smoothed_grad_optimize(best_seq, time_limit=t_alloc, T_init=0.00005)
            try_seq(a_opt, "final_sg_lowT")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            n_best = len(best_seq)
            for _ in range(5):
                remaining = deadline - time.monotonic() - 5
                if remaining < 2:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = smoothed_grad_optimize(f_pert, time_limit=min(remaining, 5),
                                                  T_init=0.0001)
                try_seq(a_opt, "micro_pert")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Extra: Try a different n if we have time ===
    remaining = deadline - time.monotonic() - 5
    if remaining > 60 and best_seq is not None:
        print("\n=== Extra: Cross-n Transfer ===")
        n_best = len(best_seq)
        # Try n_best * 2 or n_best // 2
        for n_try in [n_best * 2, n_best // 2]:
            if n_try < 64 or n_try > 2048:
                continue
            remaining = deadline - time.monotonic() - 5
            if remaining < 30:
                break
            p_interp = np.interp(np.linspace(0, 1, n_try),
                                 np.linspace(0, 1, n_best), best_seq)
            p_interp = np.maximum(p_interp, 0)
            p_interp = p_interp / p_interp.sum()
            try_seq(p_interp, f"xfer_n{n_try}")
            t_alloc = min(remaining, 40)
            a_opt, _ = fw_lp_optimize(p_interp, time_limit=t_alloc)
            try_seq(a_opt, f"xfer_fw_n{n_try}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
