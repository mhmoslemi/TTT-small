import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.maximum(np.fft.irfft(A * A, fft_size)[:size], 0.0)


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=1, temp=0.0):
    """Compute (sub)gradient of max_k conv[k] with softmax weighting over top peaks."""
    n = len(a)
    k_act = min(top_k, len(conv))
    if k_act == 1 or temp <= 0:
        k_star = int(np.argmax(conv))
        # grad[m] = 2 * a[k_star - m] for valid indices
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[k_star - j]
        return grad
    else:
        # Softmax over top k peaks
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * w[i] * a[k - j]
        return grad


def fista_optimize(f0, time_limit, lr_init=0.15, top_k=1, temp=0.0,
                   max_iter=None, verbose=False, restart_every=200, noise_scale=0.002):
    """FISTA (accelerated projected gradient) for minimizing max auto-correlation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    t_k = 1.0
    lr = lr_init
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        if max_iter and it >= max_iter:
            break
        it += 1

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, temp=temp)

        # Backtracking line search (simplified: use fixed lr with occasional adjustment)
        z = y - lr * grad
        z = project_simplex(z)

        # Check if step was too large (increase in objective)
        conv_z = fast_convolve(z)
        obj_z = conv_z.max()
        if obj_z > best_obj * 1.05:
            lr *= 0.8
            z = y - lr * grad
            z = project_simplex(z)
            conv_z = fast_convolve(z)
            obj_z = conv_z.max()
        else:
            lr = min(lr * 1.005, 0.4)

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - f)
        f = z
        t_k = t_new

        # Track best
        if obj_z < best_obj - 1e-15:
            best_obj = obj_z
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart to escape plateaus
        if no_improve > restart_every:
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            y = f.copy()
            t_k = 1.0
            lr = lr_init
            no_improve = 0
            if verbose:
                print(f"    FISTA restart at iter {it}, obj={best_obj:.10f}")

        if verbose and it % 100 == 0:
            print(f"    FISTA iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def mwu_optimize(f0, time_limit, lr_init=0.1, top_k=1, temp=0.0,
                 verbose=False, decay=0.9999):
    """Multiplicative Weights Update (mirror descent with entropy) for the problem."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    lr = lr_init
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        conv_f = fast_convolve(f)
        grad = compute_subgradient(f, conv_f, top_k=top_k, temp=temp)

        # Multiplicative update: f_new = f * exp(-lr * grad) / Z
        log_f = np.log(np.maximum(f, 1e-300))
        log_f_new = log_f - lr * grad
        # Normalize to simplex
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s_new = f_new.sum()
        if s_new < 1e-300:
            f_new = f.copy()
        else:
            f_new /= s_new

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            # Don't update f if no improvement (or use damped update)
            f = 0.9 * f + 0.1 * f_new
            f = project_simplex(f)

        lr *= decay
        if lr < 1e-6:
            lr = lr_init * 0.5

        if no_improve > 300:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            lr = lr_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    MWU: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def peak_suppress(f0, time_limit, n_peaks=5, verbose=False):
    """Iteratively identify top convolution peaks and suppress them via targeted optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Find top peaks
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        peak_values = conv[peak_indices]
        max_peak = peak_values[0]

        # For each peak, compute the "conflict" indices and try to reduce them
        improved = False
        for pi in range(min(n_peaks, len(peak_indices))):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(peak_indices[pi])
            # The peak at k involves pairs (j, k-j) for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            conflict_indices = list(range(js, je + 1))

            # Try reducing the contribution from the largest terms
            contributions = np.array([f[j] * f[k - j] for j in conflict_indices])
            # Sort by contribution
            order = np.argsort(contributions)[::-1]

            # Try reducing the top contributors
            for oi in range(min(5, len(order))):
                j = conflict_indices[order[oi]]
                j2 = k - j
                # Reduce max(f[j], f[j2]) and redistribute
                if f[j] < 1e-10 and f[j2] < 1e-10:
                    continue
                # Try moving mass from the larger to the smaller
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    f_trial = f.copy()
                    if f[j] >= f[j2]:
                        move = f[j] * frac
                        f_trial[j] -= move
                        f_trial[j2] += move
                    else:
                        move = f[j2] * frac
                        f_trial[j2] -= move
                        f_trial[j] += move
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    conv_trial = fast_convolve(f_trial)
                    if conv_trial.max() < best_obj - 1e-14:
                        f = f_trial
                        best_obj = conv_trial.max()
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            # Try random perturbations focused on peak regions
            for _ in range(10):
                if time.monотonic() - t0 > time_limit if hasattr(time, 'monотonic') else time.monotonic() - t0 > time_limit:
                    break
                noise = rng.standard_normal(n) * 0.005
                # Amplify noise at peak conflict regions
                for pi in range(min(3, len(peak_indices))):
                    k = int(peak_indices[pi])
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    noise[js:je+1] *= 2.0
                f_trial = project_simplex(best_f + noise)
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-14:
                    f = f_trial
                    best_obj = conv_trial.max()
                    best_f = f.copy()
                    improved = True
                    break

        # Check time
        if time.monotonic() - t0 > time_limit:
            break

    return best_f, best_obj


def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None or len(active_k) >= 2 * n - 1:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, top_k_constraints=30, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        active_k = None
        if it > 5:
            k_sorted = np.argsort(conv)[::-1][:top_k_constraints]
            active_k = k_sorted.tolist()
        if it % 50 == 0:
            active_k = None

        g = solve_lp_direction(f, M * m_exp, n, active_k=active_k)
        if g is None:
            m_exp *= 1.12
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_alpha_obj = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 1.0]:
            f_new = project_simplex(f + alpha * (g - f))
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_alpha_obj:
                best_alpha_obj = obj_new
                best_alpha = alpha
                conv = conv_new

        if best_alpha > 1e-8:
            f = project_simplex(f + best_alpha * (g - f))
            if best_alpha_obj < best_obj - 1e-14:
                best_obj = best_alpha_obj
                best_f = f.copy()
                no_improve = 0
                M = best_obj
                m_exp = 1.0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coord_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multi-scale per-coordinate search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.5, 4.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig[:n])
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def周期性_init(n, seed=0):
    """Periodic-like initialization."""
    rng = np.random.default_rng(seed)
    f = np.ones(n) / n
    # Add periodic modulation
    freq = rng.randint(3, 15)
    t = np.arange(n)
    modulation = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2*np.pi))
    f = f * modulation
    f = np.maximum(f, 0)
    f /= f.sum()
    return f


def gaussian_center_init(n, width=0.4):
    """Gaussian-centered initialization."""
    t = np.arange(n) - n / 2.0
    f = np.exp(-t**2 / (2.0 * (width * n / 2.0)**2))
    f = np.maximum(f, 0)
    f /= f.sum()
    return f

            return

        score = float(evaluate_sequence(a_norm.tolist()))

"""
Convex-structured optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA, MWU, peak-suppression, LP-FW, and multi-length search, leveraging
the convexity of the max-of-quadratics objective over the probability simplex.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.maximum(np.fft.irfft(A * A, fft_size)[:size], 0.0)


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=1, temp=0.0):
    """Compute (sub)gradient of max_k conv[k] with optional softmax over top peaks."""
    n = len(a)
    k_act = min(top_k, len(conv))
    if k_act == 1 or temp <= 0:
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[k_star - j]
        return grad
    else:
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * w[i] * a[k - j]
        return grad


def fista_optimize(f0, time_limit, lr_init=0.15, top_k=1, temp=0.0,
                   max_iter=None, verbose=False, restart_every=250, noise_scale=0.002):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    t_k = 1.0
    lr = lr_init
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        if max_iter and it >= max_iter:
            break
        it += 1

        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, temp=temp)

        z = y - lr * grad

        if obj_z > best_obj * 1.08:
            lr *= 0.7

            conv_z = fast_convolve(z)

        else:

            t_k = 1.0

def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None or len(active_k) >= 2 * n - 1:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coord_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multi-scale per-coordinate search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj
