```python
def lse_objective_and_grad(a, beta):
    """Compute LSE approximation of max(a*a) and its gradient."""
    n = len(a)
    c = fast_convolve(a)
    c_shifted = beta * (c - c.max())
    w = np.exp(c_shifted)
    w_sum = w.sum()
    w /= w_sum
    obj = (c.max() + np.log(w_sum) / beta)  # LSE value
    
    # Gradient
    corr = np.correlate(w, a, 'valid')
    diag = w[::2][:n] * a
    grad_a = corr + diag
    return obj, grad_a
```

```python
    n = len(v)
```

```python
def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp

                    best_val = val_mid
                    best_xp = xp_mid

            if abs(best_xp - x) < 1e-13:
                continue

            # Apply update
            a[j] = best_xp
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            s_others = 1.0 - x
            s_new = 1.0 - best_xp
            if s_others > 1e-12:
                a[mask] *= s_new / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                c = fast_convolve(a)
                break

            c = fast_convolve(a)
            obj = c.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 97 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
                c = fast_convolve(a)
            else:
                break

        if verbose and passes % 10 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.10f}")

    return best_a, best_obj


def subgradient_step(a, c, top_k=40, temp=0.001):
    """Compute multi-peak subgradient step (projected)."""
    n = len(a)
    conv_len = len(c)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(c, -k_actual)[-k_actual:]

    pv = c[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

    return sg


def subgradient_optimize(a0, time_limit=40.0, lr=0.01, mom=0.9, top_k=30, temp=0.0005):
    """Projected subgradient descent with momentum."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = subgradient_step(a, c, top_k=top_k, temp=temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        c = c_new
        cur_lr *= 0.9998

        if no_improve > 300:
            rng = np.random.default_rng(it * 41 + 3)
            noise = 0.003 + 0.002 * min(no_improve // 200, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            c = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.7
            no_improve = 0

    return best_a, best_obj


def symmetrize(a):
    """Make sequence palindromic."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LSE-Adam, coordinate descent,
    subgradient methods, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    def update_best(a_try, label):
        nonlocal best_a, best_score
        if a_try is None:
            return
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_try)})")

    # === Phase 1: LSE-Adam on parent ===
    print(f"\n=== Phase 1: LSE-Adam on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    if rem > 300:
        t_alloc = min(250, rem - 200)
        print(f"  LSE-Adam, t={t_alloc:.0f}s")
        a_opt, obj = lse_adam_optimize(parent, time_limit=t_alloc, beta_start=15, beta_end=200, lr=0.06)
        update_best(a_opt, "lse_parent")

    # === Phase 2: Coordinate descent polish ===
    print(f"\n=== Phase 2: CD polish ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 100:
            t_alloc = min(120, rem - 80)
            print(f"  CD, t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_a, time_limit=t_alloc)
            update_best(a_opt, "cd1")

    # === Phase 3: LSE-Adam with higher beta ===
    print(f"\n=== Phase 3: LSE-Adam high-beta ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 150:
            t_alloc = min(150, rem - 100)
            print(f"  LSE-Adam beta 50-500, t={t_alloc:.0f}s")
            a_opt, obj = lse_adam_optimize(best_a, time_limit=t_alloc, beta_start=50, beta_end=500, lr=0.03)
            update_best(a_opt, "lse_high")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length ===")
    base = best_a if best_a is not None else parent
    target_lengths = [512, 896, 1024, 1280]

    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if target_n == len(base):
            continue
        f_interp = interpolate_to_length(base, target_n)
        # LSE-Adam on new length
        t_alloc = min(100, rem // 2)
        print(f"  n={target_n} LSE-Adam, t={t_alloc:.0f}s")
        a_opt, obj = lse_adam_optimize(f_interp, time_limit=t_alloc, beta_start=15, beta_end=250, lr=0.05)
        update_best(a_opt, f"len{target_n}")

        # CD on result
        rem = deadline - time.monotonic()
        if rem > 40 and a_opt is not None:
            t_alloc = min(40, rem - 20)
            a_opt2, obj2 = coordinate_descent(a_opt, time_limit=t_alloc)
            update_best(a_opt2, f"len{target_n}_cd")

    # === Phase 5: Symmetric optimization ===
    print(f"\n=== Phase 5: Symmetric ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 80:
            a_sym = symmetrize(best_a)
            t_alloc = min(80, rem - 40)
            print(f"  Symmetric LSE-Adam, t={t_alloc:.0f}s")
            a_opt, obj = lse_adam_optimize(a_sym, time_limit=t_alloc, beta_start=20, beta_end=300, lr=0.04)
            update_best(a_opt, "sym")

    # === Phase 6: Subgradient from diverse starts ===
    print(f"\n=== Phase 6: Subgradient diverse ===")
    if best_a is not None:
        n_best = len(best_a)
        restarts_info = [
            ("noise3", 0.003),
            ("noise7", 0.007),
            ("flat_mix", -1),  # special
        ]
        for label, ns in restarts_info:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            if label == "flat_mix":
                f_init = project_simplex(0.85 * best_a + 0.15 * np.ones(n_best) / n_best)
            else:
                f_init = project_simplex(best_a + rng.standard_normal(n_best) * ns)
            t_alloc = min(60, rem - 20)
            print(f"  SG {label}, t={t_alloc:.0f}s")
            a_opt, obj = subgradient_optimize(f_init, time_limit=t_alloc, lr=0.008, mom=0.92, top_k=40, temp=0.0003)
            update_best(a_opt, f"sg_{label}")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final CD polish ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            t_alloc = min(100, rem - 30)
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_a, time_limit=t_alloc)
            update_best(a_opt, "final_cd")

    # === Phase 8: Tiny perturbations + CD ===
    print(f"\n=== Phase 8: Perturb + CD ===")
    if best_a is not None:
        n_best = len(best_a)
        for p in range(5):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            rng_p = np.random.default_rng(seed + 5000 + p * 7)
            f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * (0.001 + 0.001 * p))
            t_alloc = min(30, rem - 10)
            a_opt, obj = coordinate_descent(f_pert, time_limit=t_alloc)
            update_best(a_opt, f"perturb_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Try to fix
        best_a = np.clip(best_a, 0, 1000)
        s = best_a.sum()
        if s > 0.01:
            best_a = best_a / s * s  # keep same scale
            fs = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(fs):
                best_score = fs
            else:
                best_a = parent
                best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
```
