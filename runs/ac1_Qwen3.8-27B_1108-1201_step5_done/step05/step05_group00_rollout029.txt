import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def correlate_p_a(p, a):
    """Compute sum_k p[k]*a[k-i] for i = 0, ..., n-1.
    p has length 2n-1, a has length n."""
    n = len(a)

    result = np.zeros(n)

        def obj_and_grad(x_vec):

            x_shifted = x_vec - x_vec.max()  # numerical stability
            a_vec = np.exp(x_shifted)
            a_vec = a_vec / a_vec.sum()

            conv_shifted = conv - conv.max()
            log_sum_exp = conv.max() + np.log(np.sum(np.exp(conv_shifted)))

            # Actually, since sum=1, the objective is 2*n*max_conv ≈ 2*n*T*log_sum_exp

            # Gradient of log_sum_exp w.r.t. a: (1/T) * p where p = softmax(conv/T)
            p = np.exp(conv_shifted / T)
            p = p / p.sum()
            
            # grad_a = 2 * correlate(p, a) * (2*n) ... wait
            # d(2*n*T*log_sum_exp)/da[i] = 2*n*T * (1/T) * sum_k p[k] * d(conv[k])/da[i]

            best_x = x.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- L-BFGS with log-sum-exp smoothed objective and softmax (exponential) parameterization
- Multi-temperature continuation schedule for principled annealing
- Multi-scale compression for finding coarse structure
- Diverse initializations including structural patterns
- Perturbation-based escape from local optima
- Peak-targeted local refinement

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def correlate_p_a(p, a):
    """Compute g[i] = sum_m p[m+i]*a[m] for i=0..n-1. p has len 2n-1, a has len n."""
    n = len(a)
    result = np.zeros(n)
    for i in range(n):
        result[i] = np.dot(a, p[i:i + n])
    return result


def _lble_func(x_vec, n, T):
    """Objective for L-BFGS: smoothed peak of auto-convolution."""
    x_shifted = x_vec - x_vec.max()
    a_vec = np.exp(x_shifted)
    a_vec = a_vec / a_vec.sum()
    conv = fast_convolve(a_vec)
    cx = conv - conv.max()
    log_sum_exp = conv.max() + np.log(np.sum(np.exp(cx)))
    return 2.0 * n * T * log_sum_exp


def _lble_grad(x_vec, n, T):
    """Gradient for L-BFGS."""
    x_shifted = x_vec - x_vec.max()
    a_vec = np.exp(x_shifted)
    a_vec = a_vec / a_vec.sum()
    conv = fast_convolve(a_vec)
    cx = conv - conv.max()
    p = np.exp(cx / T)
    p = p / p.sum()
    g = correlate_p_a(p, a_vec)
    grad_a = 4.0 * n * g
    ga_weighted = np.dot(grad_a, a_vec)
    grad_x = a_vec * (grad_a - ga_weighted)
    return grad_x


def softmax_lbgfs(a0, T_start, T_end, T_ratio, max_iter, time_limit):
    """
    L-BFGS optimization with softmax parameterization and temperature continuation.
    Returns (best_a, best_true_obj).
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    eps = 1e-15
    x = np.log(a + eps)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    T = T_start
    t0 = time.monotonic()

    while T > T_end and time.monotonic() - t0 < time_limit - 1.0:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        try:

                           options={'maxiter': max_iter, 'gtol': 1e-10,
                                    'ftol': 1e-12, 'maxls': 50})
            if res.x is not None:
                x = res.x
        except Exception:
            pass

        x_shifted = x - x.max()
        a_new = np.exp(x_shifted)
        a_new = a_new / a_new.sum()
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()

def compressed_optimize(a0, time_limit, block_sizes=(4, 8, 16)):
    """
    Multi-scale: compress, optimize, upsample, refine.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

    for bs in block_sizes:
        if time.monotonic() - t0 > time_limit - 10:
            break
        if n % bs != 0 or n // bs < 16:
            continue

        n_comp = n // bs
        # Compress by averaging blocks
        a_comp = a.reshape(n_comp, bs).mean(axis=1)
        a_comp = a_comp / a_comp.sum()

        # Optimize compressed
        conv_comp = fast_convolve(a_comp)
        T_start = 0.05 * (conv_comp.max() - conv_comp.min() + 1e-10)
        a_comp_opt, _ = softmax_lbgfs(a_comp, T_start, 1e-5, 0.75, 100,
                                      time_limit=min(30.0, time_limit - (time.monotonic() - t0)))

        # Upsample
        a_up = np.repeat(a_comp_opt, bs)
        a_up = a_up / a_up.sum()
        obj = fast_convolve(a_up).max()
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_up.copy()
            a = a_up

    # Final refinement at full scale
    if time.monotonic() - t0 < time_limit - 15:
        conv = fast_convolve(a)
        T_start = 0.03 * (conv.max() - conv.min() + 1e-10)
        a_opt, obj = softmax_lbgfs(a, T_start, 1e-6, 0.75, 150,
                                   time_limit=time_limit - (time.monotonic() - t0))
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a, best_obj


def peak_targeted_refine(a0, time_limit, n_peaks=5):
    """
    For each of the top peaks, try local reductions via small perturbations
    focused on the contributing variables.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit - 1.0:
        conv = fast_convolve(a)
        # Get top peaks
        k_peaks = np.argsort(conv)[::-1][:n_peaks]

        for k_peak in k_peaks:
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            # Variables contributing to this peak
            js = max(0, int(k_peak) - n + 1)
            je = min(n - 1, int(k_peak))
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Try to reduce by shifting weight from high contributors to low
            a_affected = a[affected]
            if a_affected.max() < 1e-10:
                continue

            # Choose top and bottom contributors
            idx_max = affected[np.argmax(a_affected)]
            idx_min = affected[np.argmin(a_affected)]

            if idx_max == idx_min:
                continue

            # Try several shift amounts
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                candidate = a.copy()
                shift = candidate[idx_max] * frac
                candidate[idx_max] -= shift
                candidate[idx_min] += shift
                if candidate[idx_max] < 0:
                    continue
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-12:
                    best_obj = c.max()
                    best_a = candidate.copy()
                    a = candidate.copy()
                    break

        # Also try small random perturbations
        if len(k_peaks) > 0:
            perturb = rng.standard_normal(n) * 0.002
            candidate = project_simplex(a + perturb)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = candidate.copy()
                a = candidate

    return best_a, best_obj


def diverse_initializations(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine (single period)
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append(c1 / c1.sum())

    # Two cosine peaks
    c2 = 0.5 + 0.5 * np.cos(4 * np.pi * x / n)
    inits.append(c2 / c2.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())

    # Sawtooth
    saw = x / n
    inits.append(saw / saw.sum())

    # Anti-symmetric (low-high-low-high pattern)
    ah = np.zeros(n)
    ah[::2] = 1.0
    ah[1::2] = 0.3
    inits.append(ah / ah.sum())

    # Quadratic
    q = (x / (n - 1)) ** 2 + 0.1
    inits.append(q / q.sum())

    # Dirichlet with low concentration (spiky)
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Dirichlet with medium concentration
    inits.append(rng.dirichlet(np.ones(n) * 1.0))

    # Interpolated parent
    if parent is not None and len(parent) > 1:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits
