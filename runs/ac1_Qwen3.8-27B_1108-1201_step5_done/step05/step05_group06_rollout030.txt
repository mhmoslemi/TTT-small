def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

For coordinate j with current value x_j:
- Let `a_others = a` with `a[j] = 0`, normalized so sum = 1.
- If we set `a[j] = t`, then the other coordinates are scaled by `(1-t)/sum(a_others_unscaled)`.

  The convolution at position k is:
  `c[k] = (1-t)^2 * conv(b,b)[k] + 2*t*(1-t) * cross(b, e_j)[k] + t^2 * delta[k, 2j]`
  
  where `cross(b, e_j)[k] = b[k-j]` if `0 <= k-j < n`, else 0, and `delta[k, 2j] = 1` if `k = 2j`, else 0.

  Wait, let me be more careful. The convolution of `a` with itself, where `a[j] = t` and `a[k] = b[k] * (1-t)` for k ≠ j:
  
  `conv(a,a)[m] = sum_{p+q=m} a[p]*a[q]`
  
  = `sum_{p+q=m, p≠j, q≠j} b[p]*(1-t) * b[q]*(1-t)` 
    + `sum_{p+q=m, p=j, q≠j} t * b[q]*(1-t)`
    + `sum_{p+q=m, p≠j, q=j} b[p]*(1-t) * t`
    + `sum_{p+q=m, p=j, q=j} t * t`
  
  = `(1-t)^2 * conv(b,b)[m]` + `2*t*(1-t) * b[m-j]` (if 0 <= m-j < n) + `t^2 * [m == 2*j]`

  where `conv(b,b)` is the convolution of the normalized others with itself.

  So `f_m(t) = (1-t)^2 * B[m] + 2*t*(1-t) * C[m] + t^2 * D[m]`
  
  where `B[m] = conv(b,b)[m]`, `C[m] = b[m-j]` (if valid, else 0), `D[m] = [m == 2*j]`.

  We want to find `t` that minimizes `max_m f_m(t)`.

  This is a minimax problem in 1D. The function `max_m f_m(t)` is the max of quadratic functions, so it's convex (max of convex functions is convex). We can use bisection on the derivative.

  The derivative of `f_m(t)` with respect to t is:
  `f_m'(t) = -2(1-t)*B[m] + 2(1-2t)*C[m] + 2t*D[m]`
  `= 2*(t*B[m] - B[m] + C[m] - 2t*C[m] + t*D[m])`
  `= 2*((t - 1)*B[m] + (1 - 2t)*C[m] + t*D[m])`
  `= 2*(t*(B[m] - 2*C[m] + D[m]) + (-B[m] + C[m]))`

  So `f_m'(t) = 2*(B[m] - 2*C[m] + D[m]) * t + 2*(C[m] - B[m])`

  This is linear in t! So the derivative of each `f_m` is linear. The derivative of `max_m f_m(t)` is the derivative of the active constraint (the one achieving the max).

  For bisection:
  - At each iteration, find `m* = argmax_m f_m(t)`.
  - Compute `f_{m*}'(t)`.
  - If `f_{m*}'(t) > 0`, the max is increasing, so decrease t (move left).
  - If `f_{m*}'(t) < 0`, the max is decreasing, so increase t (move right).
  
  This is exactly what the previous code does. Good.

  The efficiency concern: at each bisection step, we need to:
  1. Compute `f_m(t)` for all m (O(n) vectorized).
  2. Find argmax (O(n)).
  3. Compute the derivative at argmax (O(1)).
  
  So each step is O(n), and with 45 steps, it's O(45n) per coordinate. For n=768, that's about 34K operations per coordinate, or 26M per pass. Very fast.

  The setup (computing B = conv(b,b)) is O(n log n) per coordinate, so the total setup per pass is O(n^2 log n). For n=768, that's about 5.9M. Also fast.

  So the total time per pass is about 30M operations, which at 100M ops/s is about 0.3 seconds. We can do about 6000 passes in 1800 seconds. That's plenty.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using intensive adaptive coordinate 
bisection with randomized ordering, pair redistribution escape, multi-start
initialization from diverse targets, and frequency-domain perturbation.
"""

import numpy as np
import time

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cd_pass(a, order=None, tight_range=False):
    """One pass of coordinate bisection descent.
    
    Args:
        a: input sequence (sum = 1, non-negative)
        order: permutation of coordinates to process (default: 0..n-1)
        tight_range: if True, use narrow initial bisection range
    
    Returns:
        (new_a, new_obj, improved) where new_obj = max(conv(new_a, new_a))
    """
    n = len(a)
    if order is None:
        order = np.arange(n)
    
    a = a.copy()
    conv = fast_convolve(a)
    obj = conv.max()
    improved = False
    
    for j in order:
        x_old = a[j]
        if x_old < 1e-14:
            continue
        
        s_others = 1.0 - x_old
        if s_others < 1e-12:
            continue
        
        # Build normalized others
        b = a.copy()
        b[j] = 0.0
        b = b / s_others
        
        # Precompute B = conv(b, b)
        B = fast_convolve(b)
        conv_len = 2 * n - 1
        
        # Precompute C[m] = b[m-j] if valid
        C = np.zeros(conv_len)
        start = j
        end = min(j + n, conv_len)
        C[start:end] = b[:end - start]
        
        # D[m] = 1 if m == 2j
        D = np.zeros(conv_len)
        if 2 * j < conv_len:
            D[2 * j] = 1.0
        
        # Bisection setup
        if tight_range:
            lo = max(0.0, x_old * 0.5)
            hi = min(1.0, x_old * 1.5)
            if hi - lo < 1e-6:
                lo = 0.0
                hi = 1.0
        else:
            lo = 0.0
            hi = 1.0
        
        # Precompute coefficients for f_m(t) = (1-t)^2 * B[m] + 2t(1-t)*C[m] + t^2*D[m]
        # = t^2 * (B[m] - 2C[m] + D[m]) + t * (2C[m] - 2B[m]) + B[m]
        # Let's write as: quad[m]*t^2 + lin[m]*t + const[m]
        quad = B - 2.0 * C + D
        lin = 2.0 * C - 2.0 * B
        const = B
        
        # Derivative: 2*quad[m]*t + lin[m]
        
        x_new = x_old
        for _ in range(50):
            mid = (lo + hi) / 2.0
            # Evaluate f_m(mid) for all m
            f_vals = quad * mid * mid + lin * mid + const
            m_star = np.argmax(f_vals)
            # Derivative at mid
            deriv = 2.0 * quad[m_star] * mid + lin[m_star]
            if deriv > 1e-12:
                hi = mid
            elif deriv < -1e-12:
                lo = mid
            else:
                break
            x_new = (lo + hi) / 2.0
        
        if abs(x_new - x_old) < 1e-15:
            continue
        
        # Apply update
        a[j] = x_new
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a[mask] *= (1.0 - x_new) / s_others
        a = np.maximum(a, 0.0)
        s_check = a.sum()
        if s_check < 1e-14:
            a[j] = x_old
            continue
        a /= s_check
        
        # Recompute objective
        conv = fast_convolve(a)
        new_obj = conv.max()
        if new_obj < obj:
            obj = new_obj
            improved = True
    
    return a, obj, improved

def pair_redistribute(a, i, j):
    """Redistribute mass between coordinates i and j.
    Finds optimal split of a[i]+a[j] between them.
    """
    n = len(a)
    s = a[i] + a[j]
    if s < 1e-14:
        return a.copy()
    
    s_others = 1.0 - s
    if s_others < 1e-12:
        return a.copy()
    
    # Others normalized
    b = a.copy()
    b[i] = 0.0
    b[j] = 0.0
    b = b / s_others
    
    # B = conv(b, b)
    B = fast_convolve(b)
    conv_len = 2 * n - 1
    
    # C_i[m] = b[m-i] if valid
    C_i = np.zeros(conv_len)
    start_i = i
    end_i = min(i + n, conv_len)
    C_i[start_i:end_i] = b[:end_i - start_i]
    
    # C_j[m] = b[m-j] if valid
    C_j = np.zeros(conv_len)
    start_j = j
    end_j = min(j + n, conv_len)
    C_j[start_j:end_j] = b[:end_j - start_j]
    
    # D_i[m] = 1 if m == 2i
    D_i = np.zeros(conv_len)
    if 2 * i < conv_len:
        D_i[2 * i] = 1.0
    
    # D_j[m] = 1 if m == 2j
    D_j = np.zeros(conv_len)
    if 2 * j < conv_len:
        D_j[2 * j] = 1.0
    
    # Cross term: a[i]*a[j] contributes to b[i+j]
    # In the convolution, a[i]*a[j] appears at position i+j (if i+j < 2n-1)
    # Actually, it appears at position i+j AND at position i+j (since conv is symmetric)
    # Wait: conv(a,a)[k] = sum_p a[p]*a[k-p]
    # The term a[i]*a[j] appears when p=i, k-p=j, so k=i+j
    # And when p=j, k-p=i, so k=i+j (same position)
    # So the cross term is 2*a[i]*a[j] at position i+j (if i+j < 2n-1)
    
    # Actually for the pair redistribution, let's parametrize:
    # a[i] = x * s, a[j] = (1-x) * s, others = b * (1-s)
    # conv[k] = (1-s)^2 * B[k] + 2*x*s*(1-s)*C_i[k] + 2*(1-x)*s*(1-s)*C_j[k]
    #          + x^2*s^2*D_i[k] + (1-x)^2*s^2*D_j[k] + 2*x*(1-x)*s^2*E[k]
    # where E[k] = 1 if k == i+j
    E = np.zeros(conv_len)
    if i + j < conv_len:
        E[i + j] = 1.0
    
    # f_k(x) = A_k * x^2 + B_k * x + C_k (quadratic in x)
    # (1-s)^2 * B[k] is constant in x
    # 2*x*s*(1-s)*C_i[k] + 2*(1-x)*s*(1-s)*C_j[k] = 2*s*(1-s)*(x*C_i[k] + (1-x)*C_j[k])
    #   = 2*s*(1-s)*(x*(C_i[k] - C_j[k]) + C_j[k])
    #   linear in x
    # x^2*s^2*D_i[k] + (1-x)^2*s^2*D_j[k] = s^2*(x^2*D_i[k] + (1-2x+x^2)*D_j[k])
    #   = s^2*(x^2*(D_i[k]+D_j[k]) - 2x*D_j[k] + D_j[k])
    # 2*x*(1-x)*s^2*E[k] = 2*s^2*E[k]*(x - x^2)
    #   = 2*s^2*E[k]*x - 2*s^2*E[k]*x^2
    
    # Total quadratic coeff: s^2*(D_i[k] + D_j[k]) - 2*s^2*E[k]
    # Total linear coeff: 2*s*(1-s)*(C_i[k] - C_j[k]) - 2*s^2*D_j[k] + 2*s^2*E[k]
    # Total constant: (1-s)^2*B[k] + 2*s*(1-s)*C_j[k] + s^2*D_j[k]
    
    quad_x = s * s * (D_i + D_j) - 2.0 * s * s * E
    lin_x = 2.0 * s * (1.0 - s) * (C_i - C_j) - 2.0 * s * s * D_j + 2.0 * s * s * E
    const_x = (1.0 - s) ** 2 * B + 2.0 * s * (1.0 - s) * C_j + s * s * D_j
    
    # Bisection on x in [0, 1]
    lo = 0.0
    hi = 1.0
    x_new = a[i] / s  # current fraction
    
    for _ in range(45):
        mid = (lo + hi) / 2.0
        f_vals = quad_x * mid * mid + lin_x * mid + const_x
        m_star = np.argmax(f_vals)
        deriv = 2.0 * quad_x[m_star] * mid + lin_x[m_star]
        if deriv > 1e-12:
            hi = mid
        elif deriv < -1e-12:
            lo = mid
        else:
            break
        x_new = (lo + hi) / 2.0
    
    a_new = a.copy()
    a_new[i] = x_new * s
    a_new[j] = (1.0 - x_new) * s
    a_new[i] = max(a_new[i], 0.0)
    a_new[j] = max(a_new[j], 0.0)
    # Others stay proportional
    return a_new

def freq_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    
    # Modify amplitude to be more uniform
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Mix with uniform amplitude
    target_mag = np.full_like(mag, mag[0])  # DC value as target
    # Actually, let's just add noise to the amplitude
    mag_new = mag + strength * rng.standard_normal(len(mag)) * mag.mean()
    mag_new = np.maximum(mag_new, 0.0)
    
    A_new = mag_new * np.exp(1j * phase)
    # Make sure DC is preserved (to maintain sum)
    A_new[0] = A[0]
    
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-14:
        a_new /= s
    else:
        a_new = a.copy()
    
    return a_new

def interpolate_to_length(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-14:
        f_interp /= s
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using intensive adaptive coordinate
bisection with randomized ordering, pair redistribution escape, frequency-domain
perturbation, multi-start initialization, and multi-length exploration.

import numpy as np

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cd_pass(a, order=None, tight_range=False, max_bisect=50):
    """One pass of coordinate bisection descent. Returns (new_a, new_obj, improved)."""
    n = len(a)
    if order is None:
        order = np.arange(n)

    a = a.copy()
    conv = fast_convolve(a)
    obj = conv.max()
    improved = False

    for j in order:
        x_old = a[j]
        if x_old < 1e-14:
            continue

        s_others = 1.0 - x_old
        if s_others < 1e-12:
            continue

        b = a.copy()
        b[j] = 0.0
        b = b / s_others

        B = fast_convolve(b)
        conv_len = 2 * n - 1

        C = np.zeros(conv_len)
        start = j
        end = min(j + n, conv_len)
        C[start:end] = b[:end - start]

        D = np.zeros(conv_len)
        if 2 * j < conv_len:
            D[2 * j] = 1.0

        if tight_range:
            lo = max(0.0, x_old * 0.3)
            hi = min(1.0, x_old * 2.0)

                lo = 0.0
                hi = 1.0
        else:
            lo = 0.0
            hi = 1.0

        quad = B - 2.0 * C + D
        lin = 2.0 * C - 2.0 * B
        const = B

        for _ in range(max_bisect):
            mid = (lo + hi) / 2.0
            f_vals = quad * mid * mid + lin * mid + const
            m_star = np.argmax(f_vals)

            if deriv > 1e-13:
                hi = mid
            elif deriv < -1e-13:
                lo = mid
            else:

        new_obj = conv.max()
        if new_obj < obj:

            improved = True

def pair_redistribute(a, i, j):
    """Redistribute mass between coordinates i and j via bisection."""
    n = len(a)
    s = a[i] + a[j]
    if s < 1e-14:
        return a.copy()

    s_others = 1.0 - s
    if s_others < 1e-12:
        return a.copy()

    b = a.copy()
    b[i] = 0.0
    b[j] = 0.0
    b = b / s_others

    B = fast_convolve(b)
    conv_len = 2 * n - 1

    C_i = np.zeros(conv_len)
    si = i
    ei = min(i + n, conv_len)
    C_i[si:ei] = b[:ei - si]

    C_j = np.zeros(conv_len)
    sj = j
    ej = min(j + n, conv_len)
    C_j[sj:ej] = b[:ej - sj]

    D_i = np.zeros(conv_len)
    if 2 * i < conv_len:
        D_i[2 * i] = 1.0

    D_j = np.zeros(conv_len)
    if 2 * j < conv_len:
        D_j[2 * j] = 1.0

    E = np.zeros(conv_len)
    if i + j < conv_len:
        E[i + j] = 1.0

    quad_x = s * s * (D_i + D_j) - 2.0 * s * s * E
    lin_x = 2.0 * s * (1.0 - s) * (C_i - C_j) - 2.0 * s * s * D_j + 2.0 * s * s * E
    const_x = (1.0 - s) ** 2 * B + 2.0 * s * (1.0 - s) * C_j + s * s * D_j

    lo = 0.0
    hi = 1.0

    for _ in range(45):
        mid = (lo + hi) / 2.0
        f_vals = quad_x * mid * mid + lin_x * mid + const_x
        m_star = np.argmax(f_vals)
        deriv = 2.0 * quad_x[m_star] * mid + lin_x[m_star]
        if deriv > 1e-13:
            hi = mid
        elif deriv < -1e-13:
            lo = mid
        else:
            break

    x_new = (lo + hi) / 2.0

    a_new = a.copy()
    a_new[i] = max(x_new * s, 0.0)
    a_new[j] = max((1.0 - x_new) * s, 0.0)
    return a_new

def freq_perturb(a, rng, strength=0.08):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    mag = np.abs(A)
    phase = np.angle(A)

    mag_noise = rng.standard_normal(len(mag)) * mag.mean() * strength
    mag_new = np.maximum(mag + mag_noise, 0.0)

    A_new = mag_new * np.exp(1j * phase)
    A_new[0] = A[0]

    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-14:
        a_new /= s
    else:
        a_new = a.copy()

    return a_new

def interpolate_to_length(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-14:
        f_interp /= s
    return f_interp

            t_alloc = min(rem, 200)

            a_opt, obj = optimize_from_start(a_sym, time_limit=t_alloc, seed=seed + 1, verbose=False)
            update_best(a_opt, "symmetric")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    base = best_a if best_a is not None else parent

    target_lengths = []
    for dn in [-128, -64, 64, 128, 256]:
        nl = n_parent + dn
        if 256 <= nl <= 4096:
            target_lengths.append(nl)

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.4, 150)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, obj = optimize_from_start(f_interp, time_limit=t_alloc, seed=seed + target_n, verbose=False)
        update_best(a_opt, f"n{target_n}")

    # === Phase 4: Diverse restarts on best ===
    print(f"\n=== Phase 4: Restarts (best={best_score:.8f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restart_configs = [
            ("jitter1", 0.003),
            ("jitter2", 0.008),
            ("jitter3", 0.015),
            ("freq1", "freq"),
            ("freq2", "freq2"),
            ("mix", "mix"),
        ]

        for idx, (label, param) in enumerate(restart_configs):
            rem = deadline - time.monotonic() - 200 + idx * 30
            if rem < 30:
                break
            t_alloc = min(rem * 0.7, 120)

            if param == 0.003:
                f_init = project_simplex(best_a + global_rng.standard_normal(n_best) * 0.003)
            elif param == 0.008:
                f_init = project_simplex(best_a + global_rng.standard_normal(n_best) * 0.008)
            elif param == 0.015:
                f_init = project_simplex(best_a + global_rng.standard_normal(n_best) * 0.015)
            elif param == "freq":
                f_init = freq_perturb(best_a, global_rng, strength=0.1)
            elif param == "freq2":
                f_init = freq_perturb(best_a, global_rng, strength=0.2)
            else:
                f_init = project_simplex(0.7 * best_a + 0.3 * np.ones(n_best) / n_best)

            print(f"  Restart {label}, t={t_alloc:.0f}s")
            a_opt, obj = optimize_from_start(f_init, time_limit=t_alloc, seed=seed + 100 + idx, verbose=False)
            update_best(a_opt, f"restart_{label}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 300)
            print(f"  CD polish, t={t_alloc:.0f}s")
            a_opt, obj = optimize_from_start(best_a, time_limit=t_alloc, seed=seed + 999, verbose=True)
            update_best(a_opt, "final_cd")

            # Ultra-tight final passes
            rem = deadline - time.monotonic() - 15
            if rem > 10 and best_a is not None:
                n_best = len(best_a)
                for p in range(4):
                    rem = deadline - time.monotonic() - 10
                    if rem < 8:
                        break
                    rng_p = np.random.default_rng(seed + 2000 + p)
                    f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.001)
                    t_alloc = min(rem, 12)
                    a_opt, obj = cd_pass(f_pert, order=rng_p.permutation(n_best), tight_range=True)
                    if obj < fast_convolve(best_a).max():
                        best_a = a_opt
                        best_score = 2.0 * n_best * obj / (best_a.sum() ** 2)
                        print(f"  Tiny perturb {p}: obj={obj:.10f}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
