def fast_convolve(a):
    n = len(a)
    if n == 0: return np.array([])
    c_len = 2*n - 1
    size = 1 << (c_len - 1).bit_length()
    fa = np.fft.rfft(a, size)

    return c

def smooth_adam(a0, time_limit, beta=1000.0, lr=0.03, mom1=0.9, mom2=0.999, max_iter=1000000, restart_stall=1200, seed=1):
   a=normalize; n; c_len; size; diag
   C=fast_convolve; best; best_a
   m=np.zeros; v=np.zeros; t=0; no_imp=0; noise=0.003; rng=default_rng(seed)
   t0=time.monotonic()
   while time - t0 < time_limit:
      t +=1
      A=np.fft.rfft(a,size)
      C=np.fft.irfft(A*A,size)[:c_len]
      np.maximum(C,0,out=C)
      cur_obj=float(C.max())
      if cur_obj < best_obj - 1e-12: best_obj=cur_obj; best_a=a.copy(); no_imp=0
      else: no_imp +=1
      if no_imp > restart_stall or cur_obj > best_obj*1.25:
         a=normalize(best_a + rng.normal(n)*noise)
         C=fast_convolve(a)
         m.fill(0); v.fill(0); t=0; no_imp=0
         noise=min(0.06, noise*1.2)
         continue
      # gradient
      mC=float(C.max())
      # if beta large, use scaled
      ex=np.exp(beta*(C-mC))
      sex=ex.sum()
      if sex <=0 or not finite: w=zeros; w[argmax]=1
      else: w=ex/sex
      W=np.fft.rfft(w,size)
      g=np.fft.irfft(W*np.conj(A),size)[:n]
      g += w[diag]*a
      g -= g.mean()
      gnorm=np.sqrt(np.mean(g*g))
      if gnorm < 1e-14:
         no_imp +=100
         continue
      gn=g/gnorm
      m = mom1*m + (1-mom1)*gn
      v = mom2*v + (1-mom2)*(gn*gn)
      bias1=1-mom1**t; bias2=1-mom2**t
      direction = (m/bias1) / (np.sqrt(v/bias2)+1e-8)
      a = project_simplex(a - lr*direction)
      # optional cap if too bad? no
   # final
   a=normalize(best_a); C=fast_convolve; return a, C.max()

"""
Search for a non-negative step function minimizing

    2*n*max(conv(a,a)) / (sum(a))^2

using:
- fast exact coordinate line-search descent (O(n^2) per pass, no per-coordinate FFT),
- randomized exact two-coordinate pair descent with peak-based pair selection,
- smoothed LSE Adam in log-simplex space,
- multi-length interpolation, seeded restarts, and final polishing.
"""

import numpy as np
import time


def normalize_sequence(a):
    """Normalize to non-negative, sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a /= s
    return a


def fast_convolve(a):
    """Full linear self-convolution via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    c_len = 2 * n - 1
    size = 1 << (c_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:c_len]
    np.maximum(c, 0.0, out=c)
    return c


def local_score(a):
    """Score equivalent to evaluate_sequence, but faster and assumes valid input."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return float("inf")
    c = fast_convolve(a)
    m = float(c.max()) if c.size else 0.0
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def update_best(best_a, best_score, a_try, label):
    """Update best sequence if local score improves."""
    if a_try is None:
        return best_a, best_score
    a = normalize_sequence(a_try)
    score = local_score(a)
    if np.isfinite(score) and score < best_score - 1e-12:
        print(f"NEW BEST {label}: {score:.10f} n={a.size}")
        return a, score
    return best_a, best_score


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    n = v.size
    if n == 0:
        return np.array([1.0])
    if n == 1:
        return np.array([1.0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length and renormalize."""
    f = np.asarray(f, dtype=np.float64).ravel()
    f = normalize_sequence(f)
    old_n = f.size
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    return normalize_sequence(g)


def coord_phi_value_out(x, B, v, delta_idx, out):
    """Evaluate one coordinate line max-convolution value, overwriting out."""
    if x < 0.0:
        x = 0.0
    if x > 1.0:
        x = 1.0
    om = 1.0 - x
    out[:] = (om * om) * B
    out += (2.0 * x * om) * v
    if delta_idx >= 0:
        out[delta_idx] += x * x
    return float(np.max(out))


def pair_phi_value_out(delta, C, A, Q, out):
    """Evaluate one pair-move max-convolution value, overwriting out."""
    out[:] = C
    out += delta * A
    out += (delta * delta) * Q
    return float(np.max(out))


def coordinate_cd_fast(a0, time_limit, grid_m=24, golden_iters=18,
                       max_passes=1000000, noise_scale=0.002, seed=123):
    """
    Fast coordinate descent.

    For coordinate j, keep the ratios of all other coordinates fixed.
    The new convolution is a quadratic in x = a_j. We sample a small 1D grid
    plus log-spaced points, then do golden-section refinement in the bracket
    around the best sample. Updates are exact O(conv_len), no FFT per coordinate.
    """
    a = normalize_sequence(a0)
    n = a.size
    if n <= 2 or time_limit <= 1e-3:
        c = fast_convolve(a)
        return a, float(c.max())

    conv_len = 2 * n - 1
    C = fast_convolve(a)
    best_obj = float(C.max())
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    v = np.zeros(conv_len, dtype=np.float64)
    B = np.empty(conv_len, dtype=np.float64)
    Cnew = np.empty(conv_len, dtype=np.float64)
    tmp = np.empty(conv_len, dtype=np.float64)

    grid_max = grid_m + 24
    mat = np.empty((grid_max, conv_len), dtype=np.float64)
    phi = np.empty(grid_max, dtype=np.float64)

    gamma = 0.6180339887498949
    passes = 0
    no_imp = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        cur_obj = float(C.max())
        pass_improved = False

        order = np.arange(n)
        if passes % 2 == 0:
            rng.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            x = float(a[j])
            s_other = 1.0 - x
            if s_other <= 1e-12:
                continue

            # Build v_k = b_{k-j}, where b are others normalized.
            v.fill(0.0)
            end = j + n
            if end > conv_len:
                end = conv_len
            length = end - j
            v[j:end] = a[:length] / s_other
            if 2 * j < conv_len:
                v[2 * j] = 0.0

            s2 = s_other * s_other
            # B = conv(b) from current C:
            # C = s_other^2 B + 2*x*s_other*v + x^2 * delta_{2j}
            B[:] = C
            B /= s2
            if x != 0.0:
                B -= (2.0 * x / s_other) * v
            delta_idx = 2 * j
            if delta_idx < conv_len:
                B[delta_idx] -= (x * x) / s2
            np.maximum(B, 0.0, out=B)

            # 1D search grid.
            x_hi = max(1.5 * x + 1e-9, 12.0 / float(n))
            if x_hi > 0.5:
                x_hi = 0.5
            if x > x_hi:
                x_hi = min(0.5, 1.3 * x + 1e-9)

            base = np.linspace(0.0, x_hi, grid_m)
            parts = [base]
            if x_hi > 1e-10:
                lo_log = max(1e-12, x_hi * 1e-5)
                if lo_log < x_hi:
                    logpts = np.logspace(np.log10(lo_log), np.log10(x_hi), 12)
                    parts.append(logpts)
            if 0.0 < x < 1.0:
                parts.append(np.array([x], dtype=np.float64))

            X = np.concatenate(parts)
            X = np.unique(X)
            mX = X.size
            if mX > grid_max:
                X = np.unique(np.concatenate([base, np.array([x], dtype=np.float64)]))
                mX = X.size
            if mX <= 1:
                X = np.array([0.0, min(1.0, max(x_hi, 1e-8))], dtype=np.float64)
                mX = 2

            omX = 1.0 - X
            mat[:mX, :] = (omX * omX)[:, None] * B[None, :]
            mat[:mX, :] += (2.0 * X * omX)[:, None] * v[None, :]
            if delta_idx < conv_len:
                mat[:mX, delta_idx] += X * X
            np.max(mat[:mX, :], axis=1, out=phi[:mX])

            idxb = int(np.argmin(phi[:mX]))
            x_sample = float(X[idxb])
            f_sample = float(phi[idxb])

            lo = float(X[max(0, idxb - 1)])
            hi = float(X[min(mX - 1, idxb + 1)])
            if idxb == mX - 1 and x_hi < 0.5:
                lo = float(X[max(0, mX - 2)])
                hi = float(min(0.5, 2.0 * x_hi))

            if lo > hi:
                lo, hi = hi, lo
            if hi - lo < 1e-14:
                cx = 0.5 * (lo + hi)
                lo = max(0.0, cx - 1e-8)
                hi = min(1.0, cx + 1e-8)

            x1 = hi - gamma * (hi - lo)
            x2 = lo + gamma * (hi - lo)
            f1 = coord_phi_value_out(x1, B, v, delta_idx, tmp)
            f2 = coord_phi_value_out(x2, B, v, delta_idx, tmp)

            for _ in range(golden_iters):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gamma * (hi - lo)
                    f1 = coord_phi_value_out(x1, B, v, delta_idx, tmp)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gamma * (hi - lo)
                    f2 = coord_phi_value_out(x2, B, v, delta_idx, tmp)

            x_gold = 0.5 * (lo + hi)
            f_gold = coord_phi_value_out(x_gold, B, v, delta_idx, tmp)

            if f_gold < f_sample:
                x_new = x_gold
            else:
                x_new = x_sample

            if x_new < 0.0:
                x_new = 0.0
            if x_new > 1.0:
                x_new = 1.0

            om_new = 1.0 - x_new
            Cnew[:] = (om_new * om_new) * B
            Cnew += (2.0 * x_new * om_new) * v
            if delta_idx < conv_len:
                Cnew[delta_idx] += x_new * x_new

            obj_new = float(np.max(Cnew))
            if obj_new < cur_obj - 1e-12:
                new_s_other = 1.0 - x_new
                if new_s_other <= 1e-15:
                    a.fill(0.0)
                    a[j] = 1.0
                else:
                    ratio = new_s_other / s_other
                    a *= ratio
                    a[j] = x_new
                    ss = float(a.sum())
                    if not np.isfinite(ss) or ss <= 1e-15:
                        a = best_a.copy()
                        C = fast_convolve(a)
                        break
                    a /= ss
                C = Cnew
                cur_obj = obj_new
                pass_improved = True

        # End of pass: recompute to remove drift.
        a = normalize_sequence(a)
        C = fast_convolve(a)
        obj = float(C.max())
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        if not pass_improved:
            no_imp += 1
            if passes > 2:
                scale = noise_scale * (1.0 + 0.4 * min(5, no_imp // 5))
                scale = min(0.05, scale)
                trial = best_a + rng.standard_normal(n) * scale
                a = normalize_sequence(trial)
                C = fast_convolve(a)
        else:
            no_imp = 0

    a = normalize_sequence(best_a)
    C = fast_convolve(a)
    return a, float(C.max())


def pair_cd_fast(a0, time_limit, grid_m=24, golden_iters=18,
                 start_temp=0.0, stop_temp=0.0, peak_frac=0.6,
                 max_top=64, seed=555):
    """
    Randomized exact pair moves.

    Move delta from coordinate j to i:
        a_i <- x + delta, a_j <- y - delta
    The convolution update is C + delta*A + delta^2*Q and can be built in O(n).
    """
    a = normalize_sequence(a0)
    n = a.size
    if n <= 2 or time_limit <= 1e-3:
        c = fast_convolve(a)
        return a, float(c.max())

    conv_len = 2 * n - 1
    C = fast_convolve(a)
    cur_obj = float(C.max())
    best_obj = cur_obj
    best_a = a.copy()

    A = np.zeros(conv_len, dtype=np.float64)
    Q = np.zeros(conv_len, dtype=np.float64)
    mat = np.empty((grid_m + 5, conv_len), dtype=np.float64)
    phi = np.empty(grid_m + 5, dtype=np.float64)
    tmp = np.empty(conv_len, dtype=np.float64)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    it = 0
    long_stall = 0
    no_accept_reset = 20000
    noise = 0.002

    top_ks = np.array([], dtype=np.int64)
    top_w = np.array([], dtype=np.float64)

    gamma = 0.6180339887498949

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Refresh active peaks occasionally.
        if it % 32 == 1:
            K = min(max_top, conv_len)
            if K >= conv_len:
                top_ks = np.arange(conv_len, dtype=np.int64)
            else:
                top_ks = np.argpartition(C, -K)[-K:]
                top_ks = top_ks[np.argsort(C[top_ks])[::-1]]
            vals = C[top_ks]
            vmax = float(vals.max()) if vals.size else 0.0
            temp = max(0.05 * vmax, 1e-12)
            ex = np.exp((vals - vmax) / temp)
            sex = float(ex.sum())
            if sex <= 0.0 or not np.isfinite(sex):
                top_w = np.ones(K, dtype=np.float64) / float(K)
            else:
                top_w = ex / sex

        elapsed = time.monotonic() - t0
        frac = max(0.0, 1.0 - elapsed / max(1e-9, time_limit))
        temp = stop_temp + (start_temp - stop_temp) * frac

        # Choose pair: peak-based or random.
        if len(top_ks) > 0 and rng.random() < peak_frac:
            pk = int(rng.choice(len(top_ks), p=top_w))
            k = int(top_ks[pk])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            l = int(rng.integers(lo, hi + 1))
            i = l
            j = k - l
            if i == j:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))

        if i == j:
            continue

        x = float(a[i])
        y = float(a[j])
        if x + y <= 1e-13:
            continue

        # Build linear and quadratic delta coefficients.
        A.fill(0.0)
        Q.fill(0.0)
        Q[2 * i] += 1.0
        Q[2 * j] += 1.0
        Q[i + j] -= 2.0

        A[i:i + n] += 2.0 * a
        A[j:j + n] -= 2.0 * a

        # Delta candidates in [-x, y].
        D = np.linspace(-x, y, grid_m)
        D = np.concatenate((D, [0.0]))
        D = np.unique(D)
        mD = D.size
        if mD > grid_m + 5:
            D = D[:grid_m + 5]
            mD = D.size

        D2 = D * D
        mat[:mD, :] = C[None, :] + D[:, None] * A[None, :] + D2[:, None] * Q[None, :]
        np.max(mat[:mD, :], axis=1, out=phi[:mD])

        idxb = int(np.argmin(phi[:mD]))
        dlo = float(D[max(0, idxb - 1)])
        dhi = float(D[min(mD - 1, idxb + 1)])
        dlo = max(dlo, -x)
        dhi = min(dhi, y)
        if dlo > dhi:
            dlo, dhi = dhi, dlo
        if dhi - dlo < 1e-13:
            cx = 0.0 if (-x <= 0.0 <= y) else float(D[0])
            dlo = max(-x, cx - 1e-7)
            dhi = min(y, cx + 1e-7)

        d1 = dhi - gamma * (dhi - dlo)
        d2 = dlo + gamma * (dhi - dlo)
        f1 = pair_phi_value_out(d1, C, A, Q, tmp)
        f2 = pair_phi_value_out(d2, C, A, Q, tmp)

        for _ in range(golden_iters):
            if f1 < f2:
                dhi = d2
                d2 = d1
                f2 = f1
                d1 = dhi - gamma * (dhi - dlo)
                f1 = pair_phi_value_out(d1, C, A, Q, tmp)
            else:
                dlo = d1
                d1 = d2
                f1 = f2
                d2 = dlo + gamma * (dhi - dlo)
                f2 = pair_phi_value_out(d2, C, A, Q, tmp)

        d_gold = 0.5 * (dlo + dhi)
        f_gold = pair_phi_value_out(d_gold, C, A, Q, tmp)
        d_sample = float(D[idxb])
        f_sample = float(phi[idxb])

        if f_gold <= f_sample:
            delta = d_gold
        else:
            delta = d_sample

        delta = max(-x, min(y, delta))

        tmp[:] = C
        tmp += delta * A
        tmp += (delta * delta) * Q
        obj_new = float(np.max(tmp))

        tol = 1e-12
        accept = False
        if obj_new < cur_obj - tol:
            accept = True
        elif temp > 1e-8 and obj_new <= cur_obj * (1.0 + temp) + 1e-12:
            accept = True

        if accept:
            a[i] = x + delta
            a[j] = y - delta
            C[:] = tmp
            cur_obj = obj_new
            long_stall = 0
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()

            # If annealing drifts too far, reset.
            if cur_obj > best_obj * 1.25:
                a = normalize_sequence(best_a + rng.standard_normal(n) * noise)
                C = fast_convolve(a)
                cur_obj = float(C.max())
                noise = min(0.05, noise * 1.2)
                long_stall = 0
        else:
            long_stall += 1
            if long_stall > no_accept_reset:
                noise = min(0.05, 0.002 * (1.0 + 0.25 * (long_stall // 10000)))
                a = normalize_sequence(best_a + rng.standard_normal(n) * noise)
                C = fast_convolve(a)
                cur_obj = float(C.max())
                noise = min(0.06, noise * 1.1)
                long_stall = 0

        # Periodically enforce exactness.
        if it % 5000 == 0:
            C = fast_convolve(a)
            cur_obj = float(C.max())
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()

    a = normalize_sequence(best_a)
    c = fast_convolve(a)
    return a, float(c.max())


def smooth_adam(a0, time_limit, beta=1000.0, lr=0.03,
                mom1=0.90, mom2=0.999, restart_stall=1200, seed=1):
    """
    Smoothed LSE Adam on the simplex.

    Uses a log-sum-exp surrogate for max(conv(a,a)) and Adam-like Adam updates
    with simplex projection. Restarts from the best point when stalled.
    """
    a = normalize_sequence(a0)
    n = a.size
    if n <= 1 or time_limit <= 1e-3:
        c = fast_convolve(a)
        return a, float(c.max())

    c_len = 2 * n - 1
    size = 1 << (c_len - 1).bit_length()
    diag = 2 * np.arange(n, dtype=np.int64)

    C = fast_convolve(a)
    best_obj = float(C.max())
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    t = 0
    no_imp = 0
    noise = 0.003

    while time.monotonic() - t0 < time_limit:
        t += 1

        A = np.fft.rfft(a, size)
        C = np.fft.irfft(A * A, size)[:c_len]
        np.maximum(C, 0.0, out=C)
        cur_obj = float(C.max())

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > restart_stall or cur_obj > best_obj * 1.25:
            a = normalize_sequence(best_a + rng.standard_normal(n) * noise)
            C = fast_convolve(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_imp = 0
            noise = min(0.06, noise * 1.2)
            continue

        cmax = float(C.max())
        ex = np.exp(beta * (C - cmax))
        sex = float(ex.sum())
        if sex <= 0.0 or not np.isfinite(sex):
            w = np.zeros(c_len, dtype=np.float64)
            w[int(np.argmax(C))] = 1.0
        else:
            w = ex / sex

        W = np.fft.rfft(w, size)
        g = np.fft.irfft(W * np.conj(A), size)[:n]
        g += w[diag] * a
        g -= g.mean()

        gnorm = float(np.sqrt(np.mean(g * g)))
        if gnorm <= 1e-14:
            no_imp += 100
            continue

        gn = g / gnorm
        m = mom1 * m + (1.0 - mom1) * gn
        v = mom2 * v + (1.0 - mom2) * (gn * gn)

        bias1 = 1.0 - mom1 ** t
        bias2 = 1.0 - mom2 ** t
        mh = m / bias1
        vh = v / bias2
        direction = mh / (np.sqrt(vh) + 1e-8)

        a = project_simplex(a - lr * direction)

    a = normalize_sequence(best_a)
    c = fast_convolve(a)
    return a, float(c.max())


def random_smooth_init(n, rng):
    """Random exponential sequence with mild smoothing, normalized."""
    x = rng.exponential(1.0, size=n)
    k = int(min(5, max(1, n // 20)))
    t = np.arange(-k, k + 1, dtype=np.float64)
    sig = max(0.8 * k, 1e-6)
    ker = np.exp(-0.5 * (t / sig) ** 2)
    s = float(ker.sum())
    if s > 0.0 and np.isfinite(s):
        ker /= s
    x = np.convolve(x, ker, mode="same")
    return normalize_sequence(x)


def pattern_init(n, rng):
    """A few structured positive patterns."""
    idx = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.array([1.0])

    kind = int(rng.integers(0, 4))
    if kind == 0:
        a = (np.exp(-((idx / n - 0.25) ** 2) / (2.0 * 0.06 ** 2)) +
             np.exp(-((idx / n - 0.75) ** 2) / (2.0 * 0.06 ** 2)) +
             0.08 * np.ones(n))
    elif kind == 1:
        a = np.exp(-((idx / n - 0.5) ** 2) / (2.0 * 0.22 ** 2)) + 0.08 * np.ones(n)
    elif kind == 2:
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 1.0 + 0.75 * np.sin(2.0 * np.pi * idx / n + phase)
    else:
        a = rng.exponential(1.0, size=n)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    return normalize_sequence(a)


def get_parent_sequence():
    """Load height_sequence_1 if available, normalized to sum 1."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        return None
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-10:
        return None
    return p / s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search entry point.

    Alternates fast coordinate descent, pair descent, and smoothed LSE Adam,
    then explores nearby lengths and restarts, and finishes with a burn-in.
    """
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s) - 30.0)

    parent = get_parent_sequence()
    if parent is None:
        parent = np.ones(768, dtype=np.float64) / 768.0

    best_a = normalize_sequence(parent)
    best_score = local_score(best_a)
    print(f"Initial: n={best_a.size} score={best_score:.10f}")

    rng = np.random.default_rng(seed)

    # Main alternating cycles.
    cycle = 0
    while True:
        rem = deadline - time.monotonic()
        if rem < 260.0 or cycle >= 10:
            break

        n_cur = best_a.size

        # Coordinate descent.
        rem = deadline - time.monotonic()
        t_cd = min(140.0, rem - 220.0)
        if t_cd > 20.0 and n_cur <= 1500:
            gm = 24 if (cycle % 2 == 0) else 30
            a_opt, _ = coordinate_cd_fast(best_a, t_cd,
                                          grid_m=gm,
                                          golden_iters=18,
                                          seed=1000 + cycle * 17)
            best_a, best_score = update_best(best_a, best_score, a_opt, f"cycle{cycle} cd")

        # Pair descent.
        rem = deadline - time.monotonic()
        t_pair = min(80.0, rem - 180.0)
        if t_pair > 10.0:
            stemp = max(0.0, 0.06 - 0.007 * cycle)
            a_opt, _ = pair_cd_fast(best_a, t_pair,
                                    start_temp=stemp,
                                    seed=2000 + cycle * 13)
            best_a, best_score = update_best(best_a, best_score, a_opt, f"cycle{cycle} pair")

        # Smoothed Adam.
        rem = deadline - time.monotonic()
        t_smooth = min(50.0, rem - 150.0)
        if t_smooth > 5.0:
            beta_table = [500.0, 1000.0, 3000.0, 1500.0, 700.0, 2000.0, 5000.0, 1000.0, 2500.0, 1500.0]
            beta = beta_table[cycle % len(beta_table)]
            a_opt, _ = smooth_adam(best_a, t_smooth,
                                   beta=beta,
                                   lr=0.03,
                                   seed=3000 + cycle * 11)
            best_a, best_score = update_best(best_a, best_score, a_opt, f"cycle{cycle} smooth")

        # Quick coordinate polish.
        rem = deadline - time.monotonic()
        n_cur = best_a.size
        t_cd2 = min(45.0, rem - 120.0)
        if t_cd2 > 10.0 and n_cur <= 1500:
            a_opt, _ = coordinate_cd_fast(best_a, t_cd2,
                                          grid_m=20,
                                          golden_iters=12,
                                          seed=4000 + cycle * 7)
            best_a, best_score = update_best(best_a, best_score, a_opt, f"cycle{cycle} cd2")

        cycle += 1

    # Explore nearby lengths by interpolation.
    targets = [640, 896, 1024, 1536, 2048]
    for target in targets:
        rem = deadline - time.monotonic()
        if rem < 90.0:
            break
        if target == best_a.size:
            continue

        f = interpolate_to_length(best_a, int(target))

        if target <= 1200:
            t1 = min(50.0, rem - 70.0)
            if t1 > 15.0:
                a_opt, _ = coordinate_cd_fast(f, t1,
                                              grid_m=20,
                                              golden_iters=12,
                                              seed=5000 + target)
                best_a, best_score = update_best(best_a, best_score, a_opt, f"len{target} cd")
                rem = deadline - time.monotonic()
                t2 = min(25.0, rem - 60.0)
                if t2 > 5.0:
                    a_opt, _ = pair_cd_fast(a_opt, t2,
                                            start_temp=0.0,
                                            seed=6000 + target)
                    best_a, best_score = update_best(best_a, best_score, a_opt, f"len{target} pair")
        else:
            t1 = min(40.0, rem - 70.0)
            if t1 > 10.0:
                a_opt, _ = smooth_adam(f, t1,
                                       beta=1000.0,
                                       lr=0.02,
                                       seed=5000 + target)
                best_a, best_score = update_best(best_a, best_score, a_opt, f"len{target} smooth")
                rem = deadline - time.monotonic()
                if target <= 1600 and rem - 60.0 > 20.0:
                    a_opt, _ = coordinate_cd_fast(a_opt, min(25.0, rem - 70.0),
                                                  grid_m=16,
                                                  golden_iters=10,
                                                  seed=6000 + target)
                    best_a, best_score = update_best(best_a, best_score, a_opt, f"len{target} cd")

    # Restart / neighborhood search at the current best length.
    for r in range(10):
        rem = deadline - time.monotonic()
        if rem < 60.0:
            break

        n_cur = best_a.size
        init = None
        if r % 4 == 0:
            init = normalize_sequence(best_a + rng.standard_normal(n_cur) * 0.004)
        elif r % 4 == 1:
            init = normalize_sequence(best_a + rng.standard_normal(n_cur) * 0.015)
        elif r % 4 == 2:
            init = pattern_init(n_cur, rng)
        else:
            init = random_smooth_init(n_cur, rng)

        if init is None:
            continue

        if n_cur <= 1200:
            t1 = min(20.0, rem - 35.0)
            if t1 > 5.0:
                a_opt, _ = coordinate_cd_fast(init, t1,
                                              grid_m=16,
                                              golden_iters=10,
                                              seed=7000 + r)
                rem = deadline - time.monotonic()
                t2 = min(12.0, rem - 25.0)
                if t2 > 3.0:
                    a_opt, _ = pair_cd_fast(a_opt, t2,
                                            start_temp=0.0,
                                            seed=8000 + r)
                best_a, best_score = update_best(best_a, best_score, a_opt, f"restart{r}")
        else:
            t1 = min(25.0, rem - 30.0)
            if t1 > 5.0:
                a_opt, _ = smooth_adam(init, t1,
                                       beta=1000.0,
                                       lr=0.02,
                                       seed=7000 + r)
                best_a, best_score = update_best(best_a, best_score, a_opt, f"restart{r}")

    # Final burn.
    rem = deadline - time.monotonic()
    n_cur = best_a.size
    if n_cur <= 1500 and rem > 30.0:
        a_opt, _ = coordinate_cd_fast(best_a, min(120.0, rem - 15.0),
                                      grid_m=32,
                                      golden_iters=22,
                                      seed=9001)
        best_a, best_score = update_best(best_a, best_score, a_opt, "final cd")

    rem = deadline - time.monotonic()
    if rem > 15.0:
        a_opt, _ = pair_cd_fast(best_a, min(50.0, rem - 10.0),
                                start_temp=0.0,
                                seed=9002)
        best_a, best_score = update_best(best_a, best_score, a_opt, "final pair")

    best_a = normalize_sequence(best_a)
    final_list = [float(x) for x in best_a]

    try:
        final_eval = float(evaluate_sequence(final_list))
    except Exception:
        final_eval = best_score

    # Safety fallback.
    if not np.isfinite(final_eval) or final_eval > 2.0:
        fallback = get_parent_sequence()
        if fallback is None:
            fallback = np.ones(768, dtype=np.float64) / 768.0
        fallback = normalize_sequence(fallback)
        final_list = [float(x) for x in fallback]
        try:
            final_eval = float(evaluate_sequence(final_list))
        except Exception:
            final_eval = local_score(fallback)

    print(f"FINAL SCORE: {final_eval:.10f} n={len(final_list)}")
    return final_list
