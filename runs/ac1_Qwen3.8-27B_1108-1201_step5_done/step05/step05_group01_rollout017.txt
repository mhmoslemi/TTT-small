Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with adaptive log-sum-exp
smoothing, FFT-based gradient computation, away-steps Frank-Wolfe hybrid,
and coordinate descent polish. Multi-length exploration from interpolated

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_gradient(a, conv, tau):
    """
    Compute gradient of tau * log(sum exp(conv[k]/tau)) with respect to a.
    Uses FFT for efficiency.
    grad[m] = 2 * sum_{l=0}^{n-1} p[m+l] * a[l]
    where p[k] = exp(conv[k]/tau) / sum_j exp(conv[j]/tau)
    """
    n = len(a)
    # Compute softmax weights
    conv_scaled = conv / tau
    conv_scaled -= conv_scaled.max()
    p = np.exp(conv_scaled)
    p /= p.sum()
    
    # grad[m] = 2 * sum_{l=0}^{n-1} p[m+l] * a[l]
    # = 2 * (p_ext * a_rev)[m+n-1] where a_rev[l] = a[n-1-l]
    a_rev = a[::-1].copy()
    fft_size = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(P * A, fft_size)
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad, p

def fista_smoothed(a0, time_limit, tau0=0.01, tau_min=0.0005, L0=3.0, verbose=True):
    """
    FISTA with adaptive log-sum-exp smoothing for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    t = 1.0
    tau = tau0
    L = L0
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute convolution and gradient
        conv = fast_convolve(y)
        grad, p = compute_smoothed_gradient(y, conv, tau)
        
        # FISTA update
        step = 1.0 / L
        z = project_simplex(y - step * grad)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = z + (t / t_new - 1.0) * (z - a)
        a = z
        y = y_new
        t = t_new
        
        # Evaluate
        conv_a = fast_convolve(a)
        obj = conv_a.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal tau
        if it % 50 == 0:
            tau = max(tau_min, tau * 0.95)
        
        # Restart if no improvement for a while
        if no_improve > 200:
            y = best_a.copy()
            t = 1.0
            no_improve = 0
            tau = max(tau_min, tau * 0.8)
            L = L0
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA iter {it}, obj={best_obj:.10f}, tau={tau:.6f}, L={L:.2f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def away_step_fw(a0, time_limit, tau=0.01, verbose=True):
    """
    Frank-Wolfe with away steps on the simplex.
    Subgradient FW: at each step, move towards the vertex with smallest gradient
    or away from the vertex with largest weight.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    t = 1.0
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(y)
        grad, p = compute_smoothed_gradient(y, conv, tau)
        
        # Find FW direction: vertex with smallest gradient
        j_fw = np.argmin(grad)
        
        # Find away vertex: support vertex with largest alpha
        # For simplex, the representation is a = sum_i a[i] * e_i
        # So the support is wherever a[i] > 0
        support = np.where(a > 1e-10)[0]
        if len(support) > 0:
            j_away = support[np.argmax(a[support])]
        else:
            j_away = -1
        
        # Try three candidate directions
        candidates = []
        
        # 1. Move towards FW vertex
        g1 = np.zeros(n)
        g1[j_fw] = 1.0
        candidates.append((1.0, 0.0, g1))  # (gamma towards, gamma_away, direction)
        
        # 2. Move away from away vertex
        if j_away >= 0 and j_away != j_fw:
            g2 = np.zeros(n)
            g2[j_away] = 1.0
            candidates.append((0.0, 1.0, g2))
        
        # 3. Move towards FW and away from away vertex (if different)
        if j_away >= 0 and j_away != j_fw:
            candidates.append((1.0, 1.0, g1 + g2))
        
        # Line search for best candidate and step size
        best_val = best_obj
        best_move = None
        
        for gamma_t, gamma_a, direction in candidates:
            # The update is: a_new = a + step * (gamma_t * e_fw - gamma_a * a[j_away] / (1 + gamma_a * a[j_away]) * e_away)
            # Simplified: try a few step sizes
            for step in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                if gamma_a > 0:
                    # Away step: a_new = (a - step * gamma_a * a[j_away] * e_away / (1 + ...)) 
                    # Standard away step: a_new = a + step * (e_fw - a[j_away] * e_away)
                    # Then project to simplex
                    a_new = a + step * (gamma_t * g1 - gamma_a * a[j_away] * g2)
                else:
                    a_new = a + step * gamma_t * g1
                
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    continue
                a_new = a_new / s_new
                val = fast_convolve(a_new).max()
                if val < best_val:
                    best_val = val
                    best_move = a_new
        
        if best_move is not None:
            a = best_move
            if best_val < best_obj - 1e-12:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Perturb
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter {it}, obj={best_obj:.10f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent with golden section search.
    For each coordinate j, optimize f[j] while keeping sum=1.
    """
    n = len(a0)
    f = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    pass_num = 0
    no_pass_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-15:
                # Try setting it to a small value
                candidates_x = [0.001, 0.005, 0.01, old_val + 0.001]
            else:
                # Golden section around current value
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                candidates_x = [0.0, old_val * 0.5, old_val, old_val * 1.5, min(0.999, old_val + 0.01)]
            
            best_x = old_val
            best_val = fast_convolve(f).max()
            
            for x in candidates_x:
                if x < 0 or x > 0.999:
                    continue
                ft = f.copy()
                ft[j] = x
                rem = 1.0 - x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                if denom < 1e-12:
                    ft[mask] = rem / (n - 1) if n > 1 else 0
                else:
                    ft[mask] = ft[mask] * (rem / denom)
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_x = x
            
            if best_x != old_val and best_val < best_obj - 1e-14:
                ft = f.copy()
                ft[j] = best_x
                rem = 1.0 - best_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                if denom < 1e-12:
                    ft[mask] = rem / (n - 1) if n > 1 else 0
                else:
                    ft[mask] = ft[mask] * (rem / denom)
                f = ft / ft.sum()
                new_obj = fast_convolve(f).max()
                pass_improved = True
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()
        
        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def cosine_init(n, freq, amp):
    """Generate a cosine-modulated initialization on the simplex."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
