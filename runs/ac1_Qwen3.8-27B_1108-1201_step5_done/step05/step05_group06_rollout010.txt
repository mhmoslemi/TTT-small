def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def softmax_max(a, tau):
    m = a.max()
    return m + np.log(np.sum(np.exp((a - m) / tau))) / tau

def compute_grad_smooth(conv, a, tau):
    n = len(a)
    m = conv.max()
    w = np.exp((conv - m) / tau)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    return grad

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f * g)[k] (linear convolution)."""
    n = len(f)
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            A[k, idx] = f[k - idx]
    return A


def lp_direction(f, M):
    """
    Solve the LP: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0.
    Returns the solution g_0 and the objective value.
    """
    n = len(f)
    conv_len = 2 * n - 1
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = M * np.ones(conv_len)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x, float(-res.fun)
    return None, None


def lp_based_optimize(f0, time_limit, t_init=0.15, verbose=True):
    """
    LP-based optimization: iteratively solve the LP to get a direction,
    then update f with a convex combination.
    """
    f = f0.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    t = t_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g0, lp_obj = lp_direction(f, M)
        if g0 is None:
            break
        s_g = g0.sum()
        if s_g < 1e-15:
            break

        # Rescale g to unit sum, then blend
        g = g0 / s_g
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new /= ss

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    LP iter {it}: obj={best_obj:.10f}, t={t:.4f}")
        else:
            no_improve += 1
            # Decrease step size if no improvement
            t *= 0.9

        # Reset M to current max
        M = obj_new
        f = f_new

        # Adaptive: if LP objective is very close to M * n (theoretical max when all b equal),
        # we're near optimal for this f
        ratio = lp_obj / (M * n) if M > 0 else 0
        if ratio > 0.99 and no_improve > 20:
            t *= 0.8

        if no_improve > 100:
            # Restart from best with fresh step size
            f = best_f.copy()
            conv = fast_convolve(f)
            M = conv.max()
            t = t_init
            no_improve = 0

        if t < 0.001:
            break

    return best_f, best_obj


def coordinate_bisection(a0, time_limit, verbose=False):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 >= time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            break

    return best_a, best_obj


def plateau_objective(a, k=10):
    """Compute a plateau objective: max + lambda * (mean of top k peaks - max)."""
    conv = fast_convolve(a)
    n = len(a)
    s = a.sum()
    sorted_conv = np.sort(conv)[::-1]
    k_actual = min(k, len(sorted_conv))
    top_mean = sorted_conv[:k_actual].mean()
    peak = sorted_conv[0]
    # Plateau penalty: how much do top peaks stick out above the mean?
    plateau = peak - top_mean
    return 2.0 * n * peak / (s * s), plateau, conv


def plateau_optimize(f0, time_limit, k=10, t_init=0.1, verbose=True):
    """
    Optimize a plateau objective using LP-based direction with modified
    constraint: focus on reducing the spread of top peaks.
    """
    f = f0.copy().astype(np.float64)
    f /= f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    t = t_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top k peaks and their mean
        k_actual = min(k, len(conv))
        top_peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        peak_vals = conv[top_peaks]
        peak_mean = peak_vals.mean()

        # Use a modified M: set to slightly above the mean of top peaks
        # This forces the LP to spread out the mass more evenly
        M_modified = max(peak_mean, M * 0.95)

        g0, lp_obj = lp_direction(f, M_modified)
        if g0 is None:
            t *= 0.9
            if t < 0.001:
                break
            continue

        s_g = g0.sum()
        if s_g < 1e-15:
            break

        g = g0 / s_g
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new /= ss

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 30 == 0:
                print(f"    Plateau iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            t *= 0.95

        f = f_new

        if no_improve > 80:
            f = best_f.copy()
            t = t_init * 0.7
            no_improve = 0

        if t < 0.001:
            break

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def alternating_lp_cd(f0, time_limit, verbose=True):
    """
    Alternate between LP-based steps and coordinate bisection passes.
    This combines the global exploration of LP with local precision of CD.
    """
    f = f0.copy().astype(np.float64)
    f /= f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # LP phase
        t_lp = min(remaining * 0.5, 60)
        f_lp, obj_lp = lp_based_optimize(best_f, time_limit=t_lp, t_init=0.1, verbose=False)
        if obj_lp < best_obj - 1e-12:
            best_obj = obj_lp
            best_f = f_lp.copy()
            if verbose:
                print(f"    Alt LP: obj={best_obj:.10f}")

        # CD phase
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        t_cd = min(remaining * 0.5, 60)
        f_cd, obj_cd = coordinate_bisection(best_f, time_limit=t_cd, verbose=False)
        if obj_cd < best_obj - 1e-12:
            best_obj = obj_cd
            best_f = f_cd.copy()
            if verbose:
                print(f"    Alt CD: obj={best_obj:.10f}")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy LP-based optimization with coordinate bisection,
    plateau objective, alternating phases, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 35.0)

    best_a = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        ss = a_try.sum()
        if ss < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_try)})")
        return score

    # === Phase 1: LP-based optimization on parent ===
    print(f"\n=== Phase 1: LP-based (n={n_parent}) ===")
    t0 = time.monotonic()
    rem = deadline - t0 - 700
    t_alloc = min(rem, 500)
    if t_alloc > 30:
        print(f"  t={t_alloc:.0f}s")
        a_opt, obj = lp_based_optimize(parent, time_limit=t_alloc, t_init=0.15, verbose=True)
        update_best(a_opt, "lp_parent")

    # === Phase 2: Coordinate bisection polish ===
    print(f"\n=== Phase 2: CD polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 500
        t_alloc = min(rem, 300)
        if t_alloc > 20:
            print(f"  t={t_alloc:.0f}s")
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "cd1")

    # === Phase 3: Plateau optimization ===
    print(f"\n=== Phase 3: Plateau (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 400
        t_alloc = min(rem, 200)
        if t_alloc > 20:
            print(f"  t={t_alloc:.0f}s")
            a_opt, obj = plateau_optimize(best_a, time_limit=t_alloc, k=10, t_init=0.08, verbose=True)
            update_best(a_opt, "plateau10")

            rem = deadline - time.monotonic() - 350
            t_alloc = min(rem, 100)
            if t_alloc > 15:
                a_opt, obj = plateau_optimize(best_a, time_limit=t_alloc, k=20, t_init=0.06, verbose=True)
                update_best(a_opt, "plateau20")

    # === Phase 4: Alternating LP + CD ===
    print(f"\n=== Phase 4: Alternating LP+CD (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 300
        t_alloc = min(rem, 250)
        if t_alloc > 30:
            print(f"  t={t_alloc:.0f}s")
            a_opt, obj = alternating_lp_cd(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "alt_lp_cd")

    # === Phase 5: Multi-start from diverse initializations ===
    print(f"\n=== Phase 5: Multi-start (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        starts = []

        # Symmetric
        a_sym = symmetrize(best_a)
        starts.append(("sym", a_sym))

        # Noisy
        for scale in [0.003, 0.008, 0.015]:
            a_noisy = project_simplex(best_a + rng.standard_normal(n_best) * scale)
            starts.append((f"noise{int(scale*1000)}", a_noisy))

        # Reversed mix
        a_rev = project_simplex(0.5 * best_a + 0.5 * best_a[::-1])
        starts.append(("rev_mix", a_rev))

        # Flat blend
        a_flat = project_simplex(0.7 * best_a + 0.3 * np.ones(n_best) / n_best)
        starts.append(("flat", a_flat))

        t0 = time.monotonic()
        rem = deadline - t0 - 200
        if rem > 50:
            t_per = min(rem / len(starts), 120)
            print(f"  {len(starts)} starts, t_each={t_per:.0f}s")
            for label, a_init in starts:
                rem2 = deadline - time.monotonic() - 180
                if rem2 < 15:
                    break
                t_alloc = min(t_per, rem2)
                # Quick LP then CD
                a_opt, obj = lp_based_optimize(a_init, time_limit=t_alloc * 0.4, t_init=0.1, verbose=False)
                rem3 = deadline - time.monotonic() - 170
                if rem3 > 10:
                    a_opt2, obj2 = coordinate_bisection(a_opt, time_limit=min(t_alloc * 0.6, rem3), verbose=False)
                    update_best(a_opt2, f"ms_{label}")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration (best={best_score:.6f}) ===")
    if best_a is not None:
        base = best_a
        target_lengths = [384, 512, 640, 896, 1024, 1280]
        target_lengths = [t for t in target_lengths if t != len(base)]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_to_length(base, target_n)
            t_alloc = min(rem * 0.35, 120)
            print(f"  n={target_n}, t={t_alloc:.0f}s")

            # LP then CD
            a_opt, obj = lp_based_optimize(f_interp, time_limit=t_alloc * 0.5, t_init=0.12, verbose=False)
            rem2 = deadline - time.monotonic() - 140
            if rem2 > 10:
                a_opt2, obj2 = coordinate_bisection(a_opt, time_limit=min(t_alloc * 0.5, rem2), verbose=False)
                update_best(a_opt2, f"len{target_n}")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 250)
            print(f"  t={t_alloc:.0f}s")
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "final_cd")

    # === Phase 8: Ultra-fine perturbations + CD ===
    print(f"\n=== Phase 8: Ultra-fine (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        for p in range(15):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            rng_p = np.random.default_rng(seed + 7777 + p * 13)
            scale = 0.0003 + 0.0002 * p
            f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * scale)
            t_alloc = min(rem, 25)
            a_opt, obj = coordinate_bisection(f_pert, time_limit=t_alloc)
            update_best(a_opt, f"fine_{p}")

    # === Phase 9: One more LP pass on the very best ===
    print(f"\n=== Phase 9: Final LP (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            t_alloc = min(rem, 80)
            print(f"  t={t_alloc:.0f}s")
            a_opt, obj = lp_based_optimize(best_a, time_limit=t_alloc, t_init=0.05, verbose=True)
            update_best(a_opt, "final_lp")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
