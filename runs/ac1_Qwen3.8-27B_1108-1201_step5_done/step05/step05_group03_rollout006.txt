def lp_direction_cutting_plane(f, M, n, max_rounds=10):
    """LP with cutting plane: start with fewest constraints, add violated ones."""
    # Start with only the peak constraint
    conv_f = fast_convolve(f)
    k_max = np.argmax(conv_f)
    
    active_ks = [k_max]
    
    for round_idx in range(max_rounds):
        # Solve LP with current active constraints
        c = -np.ones(n)
        A_ub = np.zeros((len(active_ks), n))
        for i, k in enumerate(active_ks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[i, js:je+1] = f[js:je+1]
        b_ub = np.full(len(active_ks), M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # Add more constraints
            top_k = min(2*n-1, 10 * (round_idx + 1))
            active_ks = np.argsort(conv_f)[-top_k:].tolist()
            continue
        
        g = np.maximum(res.x, 0)
        
        # Check all constraints
        fg = np.convolve(f, g)[:2*n-1]  # or use FFT
        violated = np.where(fg > M * (1 + 1e-8))[0]
        
        if len(violated) == 0:
            return g
        
        # Add violated constraints
        active_ks = list(set(active_ks + violated.tolist()))
    
    # If we still have violations, solve with all constraints
    c = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane LP-Frank-Wolfe (major speedup over full LP)
2. Adaptive M reduction (bisection-style tightening)
3. Multi-length exploration with interpolation
4. Adam subgradient with adaptive top-k peaks
5. Exact coordinate descent with 1D golden section
6. Diverse restarts with spectral and random perturbations
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_row(f, k, n):
    """Build row k of the constraint matrix for cross-conv f*g <= M."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    row = np.zeros(n)
    row[js:je + 1] = f[js:je + 1]
    return row


def lp_direction_full(f, M, n):
    """Solve full LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_direction_cutting(f, M, n, max_rounds=15):
    """LP with cutting plane method for speed."""
    conv_f = fast_convolve(f)
    # Start with top constraints
    initial_k = max(5, min(2 * n - 1, n // 4))
    top_idx = np.argsort(conv_f)[-initial_k:]
    active_ks = list(top_idx)

    for rnd in range(max_rounds):
        c = -np.ones(n)
        A_ub = np.zeros((len(active_ks), n))
        for i, k in enumerate(active_ks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[i, js:je + 1] = f[js:je + 1]
        b_ub = np.full(len(active_ks), M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            if len(active_ks) >= 2 * n - 1:
                return None
            # Add more constraints
            more_k = min(2 * n - 1, len(active_ks) * 2 + 10)
            top_idx = np.argsort(conv_f)[-more_k:]
            active_ks = list(set(active_ks + list(top_idx)))
            continue

        g = np.maximum(res.x, 0)
        if g.sum() < 1e-15:
            return None

        # Check all constraints via FFT
        fg = cross_convolve(f, g)
        violated = np.where(fg > M * (1 + 1e-7))[0]

        if len(violated) == 0:
            return g

        # Add violated constraints
        new_ks = set(active_ks)
        for v in violated:
            new_ks.add(int(v))
        active_ks = list(new_ks)

        if len(active_ks) >= 2 * n - 1:
            break

    # Fallback: full LP
    return lp_direction_full(f, M, n)

            ft2 = np.maximum(f + c2 * (g - f), 0)

                else:

            if val < best_val:
                best_val = val
                best_alpha = cand

def adam_sg_optimize(f0, time_limit, verbose=False):
    """Adam on log-sum-exp smoothed objective with adaptive top-k."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.005

    T_start = 8.0 / n
    T_end = 0.3 / n
    top_k_start = max(20, n // 2)
    top_k_end = max(5, n // 16)

    total_iters = max(100, int(time_limit / 0.02))
    no_improve = 0

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (total_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.6 * progress)

        conv = fast_convolve(f)
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 41 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
            continue

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Adam it {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Exact coordinate descent with golden section on 1D subproblems."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = 2 * n * fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    no_improve_passes = 0
    pass_num = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        pass_improved = False

        if pass_num % 4 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit - 5:
                break

            i = order[idx]
            vi = f[i]
            if vi < 1e-12:
                # Try adding mass to this coordinate
                delta_range = (0.0, 0.01 / n)
            else:
                delta_range = (-vi * 0.9, vi * 0.5)

            # Find optimal delta via golden section
            lo, hi = delta_range

            def obj_at_delta(delta):
                f_trial = f.copy()
                f_trial[i] = max(0.0, f_trial[i] + delta)
                st = f_trial.sum()
                if st < 1e-15:
                    return 1e15
                f_trial = f_trial / st
                return 2 * n * fast_convolve(f_trial).max()

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            fc1 = obj_at_delta(c1)
            fc2 = obj_at_delta(c2)

            for _ in range(12):
                if hi - lo < 1e-10:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = obj_at_delta(c1)
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    fc2 = obj_at_delta(c2)

            opt_delta = (lo + hi) / 2
            opt_val = obj_at_delta(opt_delta)

            if opt_val < best_obj - 1e-13:
                f[i] = max(0.0, f[i] + opt_delta)
                f = f / f.sum()
                best_obj = opt_val
                best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_restart(f, n, rng, scale=0.05):
    """Generate a diverse restart using spectral perturbation."""
    F = np.fft.rfft(f)
    # Random phase perturbation in frequency domain
    phase_pert = rng.standard_normal(len(F)) * scale
    F_new = np.abs(F) * np.exp(1j * phase_pert)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    else:
        f_new = project_simplex(f + rng.standard_normal(n) * 0.01)
    return f_new

                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = best_seq[::-1].copy()
                f_init = f_init / f_init.sum()
            elif p == 3:
                f_init = spectral_restart(best_seq, n_best, rng, scale=0.08)
            elif p == 4:
                f_init = 0.6 * best_seq + 0.4 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = spectral_restart(best_seq, n_best, rng, scale=0.15)

            if p % 2 == 0:
                a_opt, _ = lpfw_optimize(f_init, time_limit=min(t_alloc, 120))
            else:
                a_opt, _ = adam_sg_optimize(f_init, time_limit=min(t_alloc, 100))
                # Follow up with short LP-FW
                rem = deadline - time.monotonic() - 250
                if rem > 15:
                    a_opt, _ = lpfw_optimize(a_opt, time_limit=min(rem, 60))
            try_seq(a_opt, f"diverse_{p}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_cd = min(rem * 0.5, 120)
            print(f"  Allocating {t_cd:.0f}s", flush=True)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "coord_desc")

    # === Phase 6: Pairwise descent ===
    print(f"\n=== Phase 6: Pairwise (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_pw = min(rem * 0.6, 100)
            print(f"  Allocating {t_pw:.0f}s", flush=True)
            a_opt, _ = pairwise_descent(best_seq, time_limit=t_pw)
            try_seq(a_opt, "pairwise")

            # Quick Adam after pairwise
            rem = deadline - time.monotonic() - 100
            if rem > 10:
                a_opt, _ = adam_sg_optimize(a_opt, time_limit=min(rem, 50))
                try_seq(a_opt, "adam_after_pw")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        # LP-FW final
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_final = min(rem, 60)
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_final)
            try_seq(a_opt, "lpfw_final")

        # Tiny perturbations
        rng = np.random.default_rng(99999)
        for trial in range(5):
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.002)
            a_opt, _ = adam_sg_optimize(f_pert, time_limit=min(rem, 25))
            try_seq(a_opt, f"perturb_{trial}")

        # One more coordinate descent pass
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "coord_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
