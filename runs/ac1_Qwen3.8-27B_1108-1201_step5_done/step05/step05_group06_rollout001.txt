"""
Hybrid differential evolution + basin hopping + block/coordinate descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses multi-length exploration,
Nesterov-accelerated subgradient for fast local steps, and adaptive scheduling.

import numpy as np
import time
from scipy.optimize import minimize as sp_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_val(a):
    """Compute the objective 2*n*max(conv)/sum^2 for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a / s)
    return 2.0 * n * conv.max() / 1.0


def subgradient_max_conv(a, conv, top_k=40, temp_scale=0.005):
    """Compute weighted subgradient of max(conv[k]) w.r.t. a."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks].copy()
    temp = temp_scale * conv.max()
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def nesterov_local(a0, time_limit, lr=0.05, top_k=40, temp_scale=0.004, verbose=False):
    """Nesterov accelerated projected subgradient descent."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    t_nest = 1.0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new

        conv_y = fast_convolve(y)
        sg = subgradient_max_conv(y, conv_y, top_k=top_k, temp_scale=temp_scale)

        # Adaptive step size (BB-like)
        step = lr / np.sqrt(1.0 + 0.01 * it)
        y_new = project_simplex(y - step * t_nest * sg)
        a_new = y_new + beta * (y_new - y)
        a_new = project_simplex(a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new /= s_new

        y = y_new
        t_nest = t_new
        a = a_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            print(f"      Nest iter {it}: obj={best_obj:.10f}")

    return best_a, best_obj


def coordinate_bisection(a0, time_limit, verbose=False, max_passes=200):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        rng = np.random.default_rng(passes * 31 + 7)
        # Randomize coordinate order for diversity
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0

            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break

            x_new = (lo + hi) / 2.0
            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng2 = np.random.default_rng(passes * 47 + 11)
                noise = 0.001 + 0.002 * min(passes // 5, 5)
                a = project_simplex(best_a + rng2.standard_normal(n) * noise)
            else:
                break

    return best_a, best_obj


def block_descent(a0, time_limit, block_size=8, verbose=False):
    """Block coordinate descent using Nelder-Mead within blocks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n // block_size + 1)

        for blk_idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            start = blk_idx * block_size
            end = min(start + block_size, n)
            if end <= start:
                continue

            block = a[start:end].copy()
            if block.max() < 1e-12:
                continue

            def block_obj(x):
                a_trial = a.copy()
                a_trial[start:end] = np.maximum(x, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    return 1e10
                a_trial /= st
                return fast_convolve(a_trial).max()

            res = sp_minimize(block_obj, block, method='Nelder-Mead',
                              options={'maxiter': 300, 'xatol': 1e-10, 'fatol': 1e-12,
                                       'adaptive': True})
            if res.fun < best_obj - 1e-12:
                a[start:end] = np.maximum(res.x, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a /= st
                best_obj = res.fun
                best_a = a.copy()
                improved = True
                stall = 0
            else:
                stall += 1

        if not improved:
            if stall > 5:
                noise_scale = 0.005
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                stall = 0
            elif stall > 2:
                break

    return best_a, best_obj


def basin_hopping(a0, time_limit, refine_fn, refine_time=5.0,
                  noise_scale=0.05, max_iter=None, verbose=False):
    """Basin hopping: perturb, locally optimize, accept if better."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if max_iter and it >= max_iter:
            break

        # Perturb
        scale = noise_scale * (1.0 + 0.5 * np.sin(0.1 * it))
        a_pert = project_simplex(best_a + rng.standard_normal(n) * scale)

        # Local refinement
        t_rem = time_limit - (time.monotonic() - t0)
        if t_rem < 2.0:
            break
        t_ref = min(refine_time, t_rem - 1.0)
        if t_ref < 1.0:
            break
        a_ref, obj = refine_fn(a_pert, time_limit=t_ref)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_ref.copy()
            if verbose:
                print(f"      BH it {it}: obj={best_obj:.10f}")

    return best_a, best_obj


def differential_evolution(a0, time_limit, pop_size=30, verbose=False):
    """Differential evolution with periodic local refinement."""
    n = len(a0)
    rng = np.random.default_rng(42)

    # Initialize population
    pop = []
    pop[0] = a0.copy() / a0.sum()
    for i in range(1, pop_size):
        scale = 0.02 + 0.08 * rng.random()
        p = project_simplex(a0 / a0.sum() + rng.standard_normal(n) * scale)
        pop.append(p)

    # Evaluate all
    objs = []
    for p in pop:
        objs.append(fast_convolve(p).max())
    objs = np.array(objs)

    best_a = pop[int(np.argmin(objs))].copy()
    best_obj = objs.min()
    t0 = time.monotonic()
    gen = 0

    while time.monotonic() - t0 < time_limit:
        gen += 1
        F = 0.5 + 0.3 * rng.random()
        CR = 0.3 + 0.4 * rng.random()

        best_idx = int(np.argmin(objs))
        trial = np.zeros(n)

        # DE/best/1 mutation
        r1, r2 = rng.choice(pop_size, 2, replace=False)
        v = pop[best_idx] + F * (pop[r1] - pop[r2])

        # Binary crossover
        mask = rng.random(n) < CR
        if not mask.any():
            mask[rng.integers(n)] = True
        trial = np.where(mask, v, pop[best_idx])
        trial = project_simplex(trial)
        trial_obj = fast_convolve(trial).max()

        # Selection
        if trial_obj < objs[best_idx]:
            pop[best_idx] = trial
            objs[best_idx] = trial_obj
            if trial_obj < best_obj - 1e-14:
                best_obj = trial_obj
                best_a = trial.copy()

        # Every few generations, do local refinement on best
        if gen % 8 == 0:
            t_rem = time_limit - (time.monotonic() - t0)
            if t_rem > 15:
                t_ref = min(8.0, t_rem - 10.0)
                a_ref, obj_ref = nesterov_local(best_a, time_limit=t_ref,
                                                lr=0.03, top_k=50, temp_scale=0.003)
                if obj_ref < best_obj - 1e-14:
                    best_obj = obj_ref
                    best_a = a_ref.copy()
                    # Inject refined result into population
                    idx = rng.integers(pop_size)
                    pop[idx] = best_a.copy()
                    objs[idx] = best_obj

                # Also try a perturbed version
                a_pert = project_simplex(best_a + rng.standard_normal(n) * 0.03)
                a_ref2, obj_ref2 = nesterov_local(a_pert, time_limit=min(5.0, t_rem - 15.0),
                                                  lr=0.03, top_k=40, temp_scale=0.004)
                if obj_ref2 < best_obj - 1e-14:
                    best_obj = obj_ref2
                    best_a = a_ref2.copy()
                    idx2 = rng.integers(pop_size)
                    pop[idx2] = best_a.copy()
                    objs[idx2] = best_obj

        if verbose and gen % 20 == 0:
            print(f"      DE gen {gen}: best={best_obj:.10f}")

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

    # Try a different length
    alt_lengths = [1024, 1536]
    for alt_n in alt_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        base_cur = best_a if best_a is not None else base
        f_alt = interpolate_to_length(base_cur, alt_n)
        t_alloc = min(rem, 120)
        print(f"  Block descent n={alt_n}, t={t_alloc:.0f}s")
        a_opt, _ = block_descent(f_alt, time_limit=t_alloc, block_size=8)
        update_best(a_opt, f"block_n{alt_n}")

    # === Phase 5: Intensive polishing ===
    print(f"\n=== Phase 5: Polishing (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    n_base = len(base)

    # Symmetric variant
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        a_sym = symmetrize(base)
        t_alloc = min(rem, 80)
        print(f"  Symmetric CD, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_bisection(a_sym, time_limit=t_alloc, max_passes=100)
        update_best(a_opt, "sym_cd")

    # Multiple restarts with CD
    restarts_config = [
        ("noise_003", 0.003),
        ("noise_008", 0.008),
        ("noise_015", 0.015),
        ("flat_mix", 0.0),  # special
    ]
    for label, scale in restarts_config:
        rem = deadline - time.monotonic() - 120
        if rem < 20:
            break
        if label == "flat_mix":
            f_init = project_simplex(0.85 * base + 0.15 * np.ones(n_base) / n_base)
        else:
            f_init = project_simplex(base + rng.standard_normal(n_base) * scale)
        t_alloc = min(rem, 80)
        print(f"  Restart {label}, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_bisection(f_init, time_limit=t_alloc, max_passes=80)
        update_best(a_opt, f"restart_{label}")

    # === Phase 6: Final intensive CD ===
    print(f"\n=== Phase 6: Final CD (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 250)
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection(best_a, time_limit=t_alloc, max_passes=500)
            update_best(a_opt, "final_cd")

            # Nesterov fine polish
            rem = deadline - time.monotonic() - 30
            if rem > 10:
                t_alloc = min(rem, 60)
                a_opt, _ = nesterov_local(best_a, time_limit=t_alloc,
                                          lr=0.01, top_k=60, temp_scale=0.001)
                update_best(a_opt, "final_nest")

            # Tiny perturbations
            rem = deadline - time.monotonic() - 15
            if rem > 10 and best_a is not None:
                n_best = len(best_a)
                for p in range(4):
                    rem = deadline - time.monotonic() - 10
                    if rem < 8:
                        break
                    rng_p = np.random.default_rng(seed + 2000 + p * 13)
                    f_p = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.001)
                    t_alloc = min(rem, 12)
                    a_opt, _ = coordinate_bisection(f_p, time_limit=t_alloc, max_passes=50)
                    update_best(a_opt, f"perturb_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
