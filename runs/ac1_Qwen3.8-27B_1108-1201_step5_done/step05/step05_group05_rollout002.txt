"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy combining
LP-Frank-Wolfe, block coordinate optimization, stochastic local search with
simulated annealing, structural template search, and subgradient polishing.
Key improvements: block-structured LP subproblems, frequency-domain candidate
generation, two-valued structural search, and adaptive constraint management.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n from half."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None or len(active_k) >= 2 * n - 1:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def golden_section_line(f, g, n, max_iter=20):
    """Golden section search for best alpha mixing f and g."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    f_c = project_simplex(f + c * (g - f))
    f_d = project_simplex(f + d * (g - f))
    fc = fast_convolve(f_c).max()
    fd = fast_convolve(f_d).max()

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            f_c = project_simplex(f + c * (g - f))
            fc = fast_convolve(f_c).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            f_d = project_simplex(f + d * (g - f))
            fd = fast_convolve(f_d).max()

    return (lo + hi) / 2.0


def lp_fw_optimize(f0, time_limit, verbose=False, use_palindrome=False,
                   top_k_constraints=35, full_every=40):
    """LP-based Frank-Wolfe with golden section and adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        active_k = None
        if it > 3:
            k_sorted = np.argsort(conv)[::-1][:top_k_constraints]
            active_k = k_sorted.tolist()
        if it % full_every == 0:
            active_k = None

        g = solve_lp_direction(f, M * m_exp, n, active_k=active_k)
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.06, 3.5)
            continue

        alpha = golden_section_line(f, g, n)

        if alpha > 1e-8:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            if use_palindrome:
                f_new = project_simplex(make_palindromic(f_new, n))

            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            conv = conv_new

            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.5)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.005 * min(no_improve // 50, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            if use_palindrome:
                f = project_simplex(make_palindromic(f, n))
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj

            # And upper bound on each x[j]

            c_vec = -np.ones(bsize)
            bounds = [(0, max(4.0 * block_sum / bsize, f[start:end].max() + block_sum / bsize * 2))] * bsize

            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success and res.x is not None:
                x_new = np.maximum(res.x, 0.0)
                f_new = f.copy()
                f_new[start:end] = x_new
                s_new = f_new.sum()
                if s_new > 1e-12:
                    f_new /= s_new
                    obj_new = fast_convolve(f_new).max()
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_f = f_new.copy()
                        f = f_new
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, mom=0.94, top_k=30, temp=0.00003):
    """Subgradient polishing with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            seed_val = (it_count * 7919 + 13) % 1000000
            rng = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def coord_desc_fine(f0, time_limit, n_mults=8):
    """Coordinate descent with multi-scale local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.0]
    if n_mults < len(multipliers):
        multipliers = multipliers[:n_mults]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                remaining = 1.0 - new_val
                if remaining <= 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def stochastic_search(f0, time_limit, verbose=False):
    """Stochastic local search with simulated annealing acceptance."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    T = 0.005
    T_min = 0.0001
    cooling = 0.998

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Generate perturbation
        # Try different perturbation types
        ptype = rng.integers(0, 4)
        if ptype == 0:
            # Gaussian noise
            scale = 0.002 + 0.003 * T / 0.005
            f_cand = project_simplex(f + rng.standard_normal(n) * scale)
        elif ptype == 1:
            # Block swap
            blen = rng.integers(8, min(64, n // 2))
            b1 = rng.integers(0, n - blen)
            b2 = rng.integers(0, n - blen)
            f_cand = f.copy()
            f_cand[b1:b1+blen], f_cand[b2:b2+blen] = f_cand[b2:b2+blen].copy(), f_cand[b1:b1+blen].copy()
            f_cand = project_simplex(f_cand)
        elif ptype == 2:
            # Block reverse
            blen = rng.integers(8, min(48, n // 2))
            b1 = rng.integers(0, n - blen)
            f_cand = f.copy()
            f_cand[b1:b1+blen] = f[b1:b1+blen][::-1]
            f_cand = project_simplex(f_cand)
        else:
            # Sparse update
            n_sparse = rng.integers(3, 15)
            idx = rng.choice(n, size=n_sparse, replace=False)
            f_cand = f.copy()
            f_cand[idx] *= rng.uniform(0.3, 3.0, size=n_sparse)
            f_cand = project_simplex(f_cand)

        obj_cand = fast_convolve(f_cand).max()

        # Acceptance criterion
        if obj_cand < best_obj:
            best_obj = obj_cand
            best_f = f_cand.copy()
            f = f_cand
            T = max(T * 1.05, 0.001)  # Reheat slightly
        elif obj_cand < f_obj if (f_obj := fast_convolve(f).max()) else False:
            f = f_cand
        elif T > T_min:
            delta = obj_cand - fast_convolve(f).max()
            if rng.random() < np.exp(-delta / T):
                f = f_cand

        T *= cooling

        if it % 500 == 0 and verbose:
            print(f"    SA iter {it}: best={best_obj:.10f}, T={T:.6f}")

    return best_f, best_obj


def two_valued_search(f0, time_limit, verbose=False):
    """Search for optimal two-valued structure {h1, h2}."""
    n = len(f0)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()

    # Try various two-valued structures
    for k_frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        k = int(n * k_frac)
        h1 = 1.0 / k  # value for "on" positions
        h2 = 0.0      # value for "off" positions

        # Start with random subset of size k
        for trial in range(3):
            idx = rng.choice(n, size=k, replace=False)
            a = np.zeros(n)
            a[idx] = h1
            obj = fast_convolve(a).max()

            if obj < best_obj:
                best_obj = obj
                best_f = a.copy()

            # Local improvement: swap operations
            for si in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                # Pick a random on/off pair to swap
                on_idx = np.where(a > 0)[0]
                off_idx = np.where(a == 0)[0]
                if len(on_idx) == 0 or len(off_idx) == 0:
                    break
                i_on = rng.choice(on_idx)
                i_off = rng.choice(off_idx)

                a_test = a.copy()
                a_test[i_on] = 0.0
                a_test[i_off] = h1
                obj_test = fast_convolve(a_test).max()

                if obj_test < obj:
                    a = a_test
                    obj = obj_test
                    if obj < best_obj:
                        best_obj = obj
                        best_f = a.copy()

    if verbose:
        print(f"    Two-valued search: best={best_obj:.10f}")

    return best_f, best_obj


def frequency_init(n, seed=0, n_targets=5):
    """Generate initial candidates with flat-ish power spectrum."""
    rng = np.random.default_rng(seed)
    candidates = []
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for trial in range(n_targets):
        if trial == 0:
            # Completely flat
            phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
            spectrum = np.exp(1j * phases)
        elif trial == 1:
            # Slightly tilted
            freq = np.arange(fft_size // 2 + 1) / (fft_size // 2)
            tilt = rng.uniform(0.5, 1.5)
            mag = 1.0 + 0.3 * (freq ** tilt)
            phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
            spectrum = mag * np.exp(1j * phases)
        else:
            # Random magnitude with smooth envelope
            freq = np.arange(fft_size // 2 + 1)
            smooth = np.exp(-0.5 * ((freq - fft_size // 4) / (fft_size // 6)) ** 2)
            mag = 0.5 + 0.5 * smooth + rng.uniform(0, 0.2, size=len(freq))
            phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
            spectrum = mag * np.exp(1j * phases)

        sig = np.fft.irfft(spectrum, fft_size)
        sig = sig[:n]
        sig = np.maximum(sig, 0)
        ss = sig.sum()
        if ss > 1e-10:
            sig /= ss
        else:
            sig = np.abs(sig[:n])
            ss = sig.sum()
            if ss > 1e-10:
                sig /= ss
            else:
                sig = np.ones(n) / n
        candidates.append(sig)

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            best_score = score
            best_seq = a.copy()

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
