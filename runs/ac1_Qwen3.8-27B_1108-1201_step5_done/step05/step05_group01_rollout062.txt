    m = conv.max()
    return m + tau * np.log(np.sum(np.exp((conv - m) / tau)))

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_gradient(conv_weights, f, n, K=None):
    """Compute gradient of sum_k weights[k]*conv[k] with respect to f.
    Uses top-K peaks for efficiency."""
    if K is not None:
        idx = np.argpartition(conv_weights, -K)[-K:]
        conv_weights = np.zeros(len(conv_weights))
        conv_weights[idx] = np.sort(conv_weights[-K:])[::-1]
    grad = np.zeros(n)
    nonzero = conv_weights > 1e-14
    for k in np.where(nonzero)[0]:
        w = conv_weights[k]
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += 2.0 * w * f[k - indices]
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed projected gradient
with Nesterov acceleration, L-BFGS-B optimization in softmax space, and
exact 1D coordinate descent for final polish.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_gradient(f, top_k=100, tau=0.001):
    """Compute smoothed gradient of max(conv(f,f)) using log-sum-exp over top-k peaks."""
    n = len(f)
    conv = fast_convolve(f)
    
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    # Temperature-scaled softmax weights
    m = pv.max()
    scaled = (pv - m) / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Compute gradient: grad[m] = sum_k w_k * 2*f[k-m] * [0<=k-m<n]
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        wi = w[i]
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += 2.0 * wi * f[k - indices]
    
    return grad, conv

def objective_value(f):
    """Compute the actual (unsmoothed) objective: max(conv(f,f))."""
    return fast_convolve(f).max()

def l_bfgs_b_softmax(z0, time_limit, top_k=100, tau=0.005, maxiter=5000):
    """L-BFGS-B optimization in softmax space."""
    n = len(z0)
    
    def obj_grad(z):
        # Softmax with numerical stability
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        
        # Compute gradient of smoothed max
        grad_a, conv = get_gradient(a, top_k=top_k, tau=tau)
        
        # Transform gradient to z-space
        # da_j/dz_m = a_j*(delta_jm - a_m)
        # dz_m = a_m * (grad_a[m] - dot(grad_a, a))
        ga = grad_a - np.dot(grad_a, a) * a
        grad_z = ga  # This is already the gradient in z-space (up to the softmax Jacobian)
        # Actually: dL/dz_m = sum_j dL/da_j * da_j/dz_m
        # = sum_j grad_a[j] * a_j * (delta_jm - a_m)
        # = a_m * grad_a[m] - a_m * sum_j grad_a[j] * a_j
        # = a_m * (grad_a[m] - dot(grad_a, a))
        grad_z = a * (grad_a - np.dot(grad_a, a))
        
        # Value (smoothed)
        conv_max = conv.max()
        # Use actual max for the objective value
        val = conv_max
        
        return val, grad_z
    
    res = minimize(obj_grad, z0, method='L-BFGS-B', jac=True,
                   options={'maxiter': maxiter, 'maxfun': maxiter * n,
                            'ftol': 1e-15, 'gtol': 1e-12, 'gtol2': 1e-12})
    
    # Convert back
    z = res.x
    z_shifted = z - z.max()
    a = exp_z = np.exp(z_shifted)
    a = a / a.sum()
    
    return a, res.fun, res.success

def cosine_init(n, freq, amp):
    """Cosine-modulated initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed projected gradient with
Nesterov acceleration (log-sum-exp), L-BFGS-B in softmax space, exact 1D
coordinate descent, and multi-length exploration with cosine-modulated inits.
Exploits convexity of the objective for reliable convergence to global optimum.

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, top_k=100, tau=0.005):
    """Compute smoothed gradient of max(conv(f,f)) using log-sum-exp over top-k peaks."""
    n = len(f)
    conv = fast_convolve(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    m = pv.max()
    scaled = (pv - m) / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        wi = w[i]
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += 2.0 * wi * f[k - indices]
    return grad, conv

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s
    n = len(f)
    y = f.copy()
    v = np.zeros(n)
    f_old = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    tau = tau0
    it = 0
    while time.monotonic() - t0 < time_limit:

        y = project_simplex(y)

    n = len(a)
    elapsed = time.monotonic() - t0
    est_per_iter = max(0.001, elapsed / max(1, 10))
    remaining_iters = min(maxiter, int((time_limit - elapsed) / est_per_iter) - 100)
    if remaining_iters < 50:
        remaining_iters = min(maxiter, 2000)
    try:
        res = minimize(obj_grad, z0, method='L-BFGS-B', jac=True,
                       options={'maxiter': remaining_iters, 'maxfun': remaining_iters * n * 2,
                                'ftol': 1e-16, 'gtol': 1e-14, 'disp': False})
        z = res.x
        z_shifted = z - z.max()
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()
        val_new = fast_convolve(a_new).max()
        if val_new < best_val:
            best_a = a_new
            best_val = val_new
    except Exception as e:
        print(f"    L-BFGS-B error: {e}")
    return best_a, best_val


def coordinate_descent(f0, time_limit, tol=1e-14):
    """1D coordinate descent with golden section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_improve_passes = 0
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            lo = 0.0
            hi = min(0.9999, 2.0 * old_val + 0.001)
            if hi - lo < 1e-14:
                continue
            c1 = hi - (hi - lo) / GR
            c2 = lo + (hi - lo) / GR
            ft1 = f.copy()
            ft1[j] = c1
            scale1 = (1.0 - c1) / denom
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            ft1[mask] = f[mask] * scale1
            f1 = fast_convolve(ft1).max()
            ft2 = f.copy()
            ft2[j] = c2
            scale2 = (1.0 - c2) / denom
            ft2[mask] = f[mask] * scale2
            f2 = fast_convolve(ft2).max()
            for _ in range(18):
                if hi - lo < 1e-13:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - (hi - lo) / GR
                    ft = f.copy()
                    ft[j] = c1
                    sc = (1.0 - c1) / denom
                    ft[mask] = f[mask] * sc
                    f1 = fast_convolve(ft).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (hi - lo) / GR
                    ft = f.copy()
                    ft[j] = c2
                    sc = (1.0 - c2) / denom
                    ft[mask] = f[mask] * sc
                    f2 = fast_convolve(ft).max()
            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_obj = fast_convolve(f).max()
            if opt_val < cur_obj - tol:
                ft = f.copy()
                ft[j] = opt_x
                sc = (1.0 - opt_x) / denom
                ft[mask] = f[mask] * sc
                stot = ft.sum()
                if stot > 1e-15:
                    ft = ft / stot
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_f = f.copy()
        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
    return best_f, best_obj


def cosine_init(n, freq, amp):
    """Cosine-modulated uniform initialization on simplex."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase convex optimization:
    1. Quick Nesterov on parent
    2. Multi-length exploration with diverse inits
    3. L-BFGS-B in softmax space for second-order convergence
    4. Coordinate descent for exact 1D optimization
    5. Iterated refinement cycles
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent, using uniform n=768")

    parent_n = len(parent)

    # === Phase 1: Quick Nesterov on parent ===
    print("\n=== Phase 1: Quick Nesterov on parent ===")
    rem = deadline - time.monotonic() - 900
    t1 = min(rem, 120)
    if t1 > 15:
        print(f"  t={t1:.0f}s")
        a_opt, _ = nesterov_optimize(parent, time_limit=t1, lr=0.03, mom=0.85,
                                     top_k=100, tau0=0.03, tau_decay=1.003)
        try_seq(a_opt, "nest_parent")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length exploration (best={best_score:.8f}) ===")
    lengths = [512, 768, 960, 1024, 1280, 1536, 1800]
    # Ensure parent length is first
    if parent_n not in lengths:
        lengths.insert(0, parent_n)
    else:
        lengths.remove(parent_n)
        lengths.insert(0, parent_n)

    for n_try in lengths:
        rem = deadline - time.monotonic() - 500
        if rem < 25:
            break
        inits = []
        if n_try == parent_n and parent is not None:
            inits.append(parent)
        inits.append(cosine_init(n_try, 1, 0.3))
        inits.append(cosine_init(n_try, 2, 0.25))
        inits.append(cosine_init(n_try, 3, 0.15))
        if best_seq is not None:
            inits.append(interpolate_to_length(best_seq, n_try))

        best_local = None
        best_local_score = float('inf')
        for a_init in inits:
            s = try_seq(a_init, f"init_n{n_try}")
            if s < best_local_score:
                best_local_score = s
                best_local = a_init

        if best_local is not None and n_try < 1200:
            t_alloc = min(rem * 0.35, 60)
            if t_alloc > 10:
                a_opt, _ = nesterov_optimize(best_local, time_limit=t_alloc,
                                             lr=0.02, mom=0.88,
                                             top_k=100, tau0=0.02, tau_decay=1.002)
                try_seq(a_opt, f"nest_n{n_try}")

    # === Phase 3: L-BFGS-B in softmax space ===
    print(f"\n=== Phase 3: L-BFGS-B (best={best_score:.8f}) ===")
    if best_seq is not None and len(best_seq) <= 2000:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.5, 300)
            print(f"  L-BFGS-B n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = lbfegs_optimize(best_seq, time_limit=t_alloc,
                                       top_k=100, tau=0.005, maxiter=5000)
            try_seq(a_opt, "lbfegs")
            # Second pass with smaller tau
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                a_opt2, _ = lbfegs_optimize(a_opt, time_limit=min(rem, 120),
                                            top_k=200, tau=0.002, maxiter=3000)
                try_seq(a_opt2, "lbfegs_fine")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_cd = min(rem * 0.5, 300)
            print(f"  CD t={t_cd:.0f}s, n={len(best_seq)}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 5: Iterated refinement ===
    print(f"\n=== Phase 5: Iterated refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        cycle = 0
        while time.monotonic() < deadline - 100 and cycle < 8:
            cycle += 1
            rem = deadline - time.monotonic() - 100
            if rem < 25:
                break

            if cycle % 3 == 1:
                # Nesterov with tight params
                t_alloc = min(rem * 0.3, 80)
                a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc,
                                             lr=0.005, mom=0.93,
                                             top_k=200, tau0=0.005, tau_decay=1.0005)
                try_seq(a_opt, f"iter_nest_{cycle}")
            elif cycle % 3 == 2:
                # L-BFGS-B
                if len(best_seq) <= 1500:
                    t_alloc = min(rem * 0.3, 120)
                    a_opt, _ = lbfegs_optimize(best_seq, time_limit=t_alloc,
                                               top_k=200, tau=0.003, maxiter=3000)
                    try_seq(a_opt, f"iter_lbfegs_{cycle}")
            else:
                # Coordinate descent
                t_alloc = min(rem * 0.4, 150)
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"iter_cd_{cycle}")

    # === Phase 6: Try longer sequences via interpolation ===
    print(f"\n=== Phase 6: Long sequences (best={best_score:.8f}) ===")
    if best_seq is not None:
        for target_n in [1800, 2400]:
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            f_interp = interpolate_to_length(best_seq, target_n)
            s = try_seq(f_interp, f"interp_n{target_n}")
            t_alloc = min(rem * 0.3, 100)
            if t_alloc > 15:
                a_opt, _ = nesterov_optimize(f_interp, time_limit=t_alloc,
                                             lr=0.02, mom=0.87,
                                             top_k=100, tau0=0.02, tau_decay=1.002)
                try_seq(a_opt, f"nest_n{target_n}")
                rem = deadline - time.monotonic() - 60
                if rem > 20 and target_n <= 1800:
                    a_opt2, _ = coordinate_descent(a_opt, time_limit=min(rem, 80))
                    try_seq(a_opt2, f"cd_n{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Tiny perturbation + quick optimize
        for _ in range(5):
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.4, 15)
            a_opt, _ = nesterov_optimize(f_pert, time_limit=t_alloc,
                                         lr=0.002, mom=0.94,
                                         top_k=200, tau0=0.003, tau_decay=1.0003)
            try_seq(a_opt, "perturb")

        # Final CD
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
