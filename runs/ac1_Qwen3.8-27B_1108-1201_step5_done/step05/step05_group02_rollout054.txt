def project_simplex(v):
    n = len(v)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Entropic-smoothed subgradient descent (FFT-based gradient computation)
2. LP-based Frank-Wolfe with line search
3. Peak-targeting coordinate descent with 1D search
4. Multi-length exploration and adaptive polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = np.maximum(v, -1e10)  # numerical safety
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_gradient(b, a, tau):
    """Compute gradient of soft_max(b, tau) w.r.t. a.
    Returns 2*q where q[m] = sum_k p[k]*a[k-m].
    """
    n = len(a)
    # Compute softmax weights
    p = b / tau
    p -= p.max()
    p = np.exp(p)
    p /= p.sum()
    
    # Compute q[m] = sum_k p[k] * a[k-m] using correlate
    # q[m] = sum_j p[m+j] * a[j] for m = 0, ..., n-1
    q = np.correlate(p, a, mode='valid')[:n]
    
    return 2.0 * q


def entropic_sg_optimize(f0, time_limit, tau_init=0.005, tau_final=0.0005,
                         lr_init=0.01, momentum_init=0.85, verbose=False):
    """Subgradient descent with entropic smoothing of the max function."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        
        # Adaptive parameters
        tau = tau_init * (tau_final / tau_init) ** progress
        lr = lr_init * (1.0 - 0.7 * progress)
        mom = momentum_init + 0.1 * progress
        
        # Compute soft gradient
        grad = compute_soft_gradient(conv, f, tau)
        
        # Momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-12:

def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t ∀k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None


def solve_lp_weighted_peaks(f, n, peak_weights):
    """LP: min sum_k w[k]*t_k s.t. (f*g)[k] <= t_k, g >= 0, sum(g) = 1.
    peak_weights: array of length 2n-1 with non-negative weights.
    """
    # Variables: g[0..n-1], t[0..2n-2]
    n_vars = n + (2*n - 1)
    c = np.zeros(n_vars)
    c[n:] = peak_weights  # coefficients for t variables
    
    n_constraints = 2*n - 1
    A_ub = np.zeros((n_constraints, n_vars))
    for j in range(n):
        A_ub[j:j+n, j] = f  # (f*g)[k] part
    # t_k >= (f*g)[k] means: (f*g)[k] - t_k <= 0
    for k in range(2*n-1):
        A_ub[k, n+k] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n_vars
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n]
    return None
