def solve_lp(a, M):
    n = len(a)
    # Maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # A_ub * g <= b_ub
    # A_ub[k, j] = a_{k-j} if 0 <= k-j < n
    # b_ub = M for all k
    
    k_vals = range(2*n - 1)
    A_ub = np.zeros((2*n-1, n))
    for k in k_vals:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
    
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based interior-point iterations
with fast O(n) coordinate descent polishing exploiting the analytical structure
of the convolution update, combined with multi-length exploration and restarts.

Key innovation: At each step, solve an LP to find the best "direction" g that
maximizes sum(g) subject to (a*g)_k <= M (current peak), then take a convex
combination step. This is far more effective than subgradient methods because
it uses all active constraints simultaneously.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    out = np.maximum(out, 0.0)
    return out


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a_{k-j} if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, time_limit=3.0):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, M) or (None, M) if LP fails.
    """
    n = len(a)
    conv = fast_convolve(a)
    M = float(conv.max())

    if M < 1e-15:
        return None, M

    c = -np.ones(n)
    A_ub = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None, M

    if res.status != 0 or res.x is None:
        return None, M

    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s < 1e-15:
        return None, M
    g = g / s
    return g, M


def fast_coord_descent(a, time_limit=30.0, max_passes=200):
    """
    Fast coordinate descent using analytical O(n) convolution update.
    For each coordinate j, the new b_k = old_b_k + 2*(x'-x)*a_{k-j} for k != 2j
    and b_{2j} = old_b_{2j} + (x'^2 - x^2).
    Uses golden section search to minimize the max.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    conv = fast_convolve(a)
    best_obj = float(conv.max())
    best_a = a.copy()
    deadline = time.time() + time_limit
    passes = 0

    golden = (np.sqrt(5.0) - 1.0) / 2.0

    while time.time() < deadline and passes < max_passes:
        passes += 1
        any_improved = False
        conv_len = 2 * n - 1

        # Precompute prefix/suffix max for O(1) outside-max queries
        prefix_max = np.maximum.accumulate(conv)
        suffix_max = np.maximum.accumulate(conv[::-1])[::-1]

        for j in range(n):
            if time.time() > deadline:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            k_lo = j
            k_hi = min(j + n, conv_len) - 1
            if k_lo > k_hi:
                continue

            # Max of conv outside [k_lo, k_hi]
            max_outside = 0.0
            if k_lo > 0:
                max_outside = max(max_outside, float(prefix_max[k_lo - 1]))
            if k_hi < conv_len - 1:
                max_outside = max(max_outside, float(suffix_max[k_hi + 1]))

            # Golden section search for optimal x_new
            lo = 0.0
            hi = min(1.0, max(x_old * 3.0, x_old + 0.01))

            for _ in range(30):
                x1 = hi - golden * (hi - lo)
                x2 = lo + golden * (hi - lo)

                # Evaluate max for x1
                idx1 = np.arange(k_lo, k_hi + 1) - j
                a_idx = a[idx1]
                old_contrib = 2.0 * x_old * a_idx
                new_contrib_1 = 2.0 * x1 * a_idx
                new_contrib_2 = 2.0 * x2 * a_idx

                if 2 * j >= k_lo and 2 * j <= k_hi:
                    pos = 2 * j - k_lo
                    old_contrib[pos] = x_old * x_old
                    new_contrib_1[pos] = x1 * x1
                    new_contrib_2[pos] = x2 * x2

                new_b1 = conv[k_lo:k_hi + 1] - old_contrib + new_contrib_1
                new_b2 = conv[k_lo:k_hi + 1] - old_contrib + new_contrib_2

                obj1 = max(max_outside, float(new_b1.max()))
                obj2 = max(max_outside, float(new_b2.max()))

                if obj1 < obj2:
                    hi = x2
                else:
                    lo = x1

            x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-12:
                continue

            a[j] = x_new
            a[j] = max(a[j], 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = best_a.copy()
                conv = fast_convolve(a)
                continue
            a = a / s_new
            conv = fast_convolve(a)
            cur_obj = float(conv.max())

            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                any_improved = True

            # Update prefix/suffix
            prefix_max = np.maximum.accumulate(conv)
            suffix_max = np.maximum.accumulate(conv[::-1])[::-1]

        if not any_improved:

                    a = a / s

def lp_iterate(a, time_limit=100.0, step_sizes=None, verbose=False):
    """
    Main LP-based iteration loop.
    a: initial sequence (will be normalized internally)
    """
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = float(conv.max())
    best_a = a.copy()
    deadline = time.time() + time_limit
    it = 0

    if step_sizes is None:
        step_sizes = [0.3, 0.5, 0.7, 1.0, 0.4, 0.6, 0.8]

    while time.time() < deadline:
        it += 1
        g, M = lp_direction(a, time_limit=min(3.0, max(0.5, deadline - time.time())))

        if g is None:
            # LP failed, try smaller step
            g = np.ones(n) / n

        # Try several step sizes, pick the best
        best_step_obj = best_obj
        best_step_a = best_a

        for t in step_sizes:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            obj = float(max(fast_convolve(a_trial)))

            a = best_step_a.copy()
        else:

            a = np.maximum(a, 0.0)
            st = a.sum()
            if st > 1e-15:
                a = a / st

        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: obj={best_obj:.10f}")

def interpolate_to_length(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1.0, old_n)
    new_x = np.linspace(0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                f_init = 0.7 * best_a + 0.3 * best_a[::-1]
                f_init = np.maximum(f_init, 0.0)
                s = f_init.sum()
                if s > 1e-15:
                    f_init = f_init / s

            print(f"  Restart {label}, t={t_alloc:.0f}s")
            a_opt, obj = lp_iterate(f_init, time_limit=t_alloc,
                                    step_sizes=[0.2, 0.4, 0.6, 0.3, 0.5],
                                    verbose=False)
            update_best(a_opt, f"r_{label}")

            # Quick CD
            rem = deadline - time.monotonic() - 80.0
            if rem > 5:
                a_opt2 = fast_coord_descent(a_opt, time_limit=min(rem, 25.0))
                update_best(a_opt2, f"r_{label}_cd")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 30.0
        if rem > 10:
            # CD polish
            t_alloc = min(rem, 120.0)
            a_opt = fast_coord_descent(best_a, time_limit=t_alloc)
            update_best(a_opt, "final_cd")

            rem = deadline - time.monotonic() - 20.0
            if rem > 5:
                # Tiny perturbations + CD
                n_best = len(best_a)
                for p in range(3):
                    rem = deadline - time.monotonic() - 10.0
                    if rem < 5:
                        break
                    rng_p = np.random.default_rng(seed + 2000 + p * 37)
                    f_pert = best_a + rng_p.standard_normal(n_best) * 0.0008
                    f_pert = np.maximum(f_pert, 0.0)
                    sp = f_pert.sum()
                    if sp > 1e-15:
                        f_pert = f_pert / sp
                    t_alloc = min(rem, 20.0)
                    a_opt = fast_coord_descent(f_pert, time_limit=t_alloc)
                    update_best(a_opt, f"perturb{p}")

            rem = deadline - time.monotonic() - 10.0
            if rem > 5:
                # One more LP pass
                t_alloc = min(rem, 30.0)
                a_opt, obj = lp_iterate(best_a, time_limit=t_alloc,
                                        step_sizes=[0.05, 0.1, 0.15, 0.08],
                                        verbose=False)
                update_best(a_opt, "final_lp")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
