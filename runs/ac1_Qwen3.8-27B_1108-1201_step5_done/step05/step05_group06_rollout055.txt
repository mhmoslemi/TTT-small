def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    betas = np.geomspace(beta_start, beta_end, n_betas)

        )

            t = result.x.copy()

"""

exact coordinate bisection descent, frequency-domain structured restarts, and

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(t):
    t_shifted = t - t.max()
    e = np.exp(t_shifted)
    return e / e.sum()


def smooth_obj_and_grad(t, beta):
    """
    Compute the smoothed objective F_beta(t) = (1/beta)*log(sum_k exp(beta*c_k))
    where c = conv(softmax(t), softmax(t)), and its gradient wrt t.
    Returns (F, grad_t).
    """
    a = softmax(t)
    n = len(a)
    c = fast_convolve(a)
    
    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    sum_exp = exp_bc.sum()
    F = max_c + np.log(sum_exp) / beta
    
    p = exp_bc / sum_exp
    
    # grad_a[i] = 2 * sum_k p[k] * a[k-i]  (= 2 * cross-correlation of a and p)
    # g[i] = sum_j a[j] * p[i+j] = conv(a, p_rev)[2n-2-i]
    p_rev = p[::-1].copy()
    conv_len = n + len(p_rev) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    conv_result = np.fft.irfft(A * P, fft_size)[:conv_len]
    # g[i] = conv_result[2n-2-i] for i=0,...,n-1
    # i.e., indices 2n-2, 2n-3, ..., n-1 = conv_result[n-1:2n-1][::-1]
    g = conv_result[n - 1:2 * n - 1][::-1].copy()
    grad_a = 2.0 * g
    
    # dF/dt_m = a_m * (grad_a[m] - dot(grad_a, a))
    dot_val = np.dot(grad_a, a)
    grad_t = a * (grad_a - dot_val)
    
    return F, grad_t


def lbfags_smoothed(t0, betas, time_limit, max_iter_per_beta=300):
    """Run L-BFGS with a sequence of increasing beta values."""
    t = t0.copy()
    best_t = t.copy()
    best_F = smooth_obj_and_grad(t, betas[0])[0]
    t_start = time.monotonic()
    
    for beta in betas:
        t_remain = time_limit - (time.monotonic() - t_start)
        if t_remain < 5:
            break
        
        # Define wrapper
        def fun_and_grad(t_curr):
            return smooth_obj_and_grad(t_curr, beta)
        
        def fun_only(t_curr):
            return smooth_obj_and_grad(t_curr, beta)[0]
        
        result = minimize(
            fun_and_grad, t, method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_beta, 'ftol': 1e-18,
                     'gtol': 1e-12, 'maxls': 30, 'maxfun': 500}
        )
        
        if result.fun < best_F - 1e-15:
            best_F = result.fun
            best_t = result.x.copy()
            t = result.x.copy()
        
        print(f"    L-BFGS beta={beta:.0f}: F={best_F:.10f}")
    
    return best_t, best_F


def coordinate_bisection(a0, time_limit, max_passes=500):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo = 0.0
            hi = 1.0
            
            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                k_max = int(np.argmax(f_vals))
                
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-14:
                    hi = mid
                elif deriv < -1e-14:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-15:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            break
    
    return best_a, best_obj


def sinusoidal_perturb(a, freqs, amplitudes, phases, seed=0):
    """Add sinusoidal perturbations in the spatial domain."""
    n = len(a)
    i = np.arange(n)
    a_pert = a.copy().astype(np.float64)
    for f, amp, ph in zip(freqs, amplitudes, phases):
        a_pert += amp * np.cos(2.0 * np.pi * f * i / n + ph)
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s > 1e-10:
        a_pert /= s
    else:
        a_pert = a.copy()
    return a_pert


def dft_based_perturb(a, n_modes=5, amplitude=0.05, seed=0):
    """Perturb in DFT domain: add random phases/amplitudes to non-DC modes."""
    n = len(a)
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    A = np.fft.rfft(a, size)
    A_new = A.copy()
    n_fft = len(A)
    # Perturb a random subset of non-DC modes
    modes = list(range(1, min(n_modes + 1, n_fft)))
    rng.shuffle(modes)
    for m in modes:
        if m >= n_fft:
            continue
        ph = rng.uniform(0, 2 * np.pi)
        amp = amplitude * A[0] / n
        A_new[m] += amp * np.exp(1j * ph)
    a_new = np.fft.irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def multi_peak_subgrad(a0, time_limit, lr=0.01, mom=0.9, top_k=30, temp=0.001):

    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_len = len(conv)
        k_actual = min(top_k, conv_len)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)

        if obj_new < best_obj - 1e-14:

            no_improve += 1
        
        a = a_new
        conv = conv_new

        cur_temp *= 1.005
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)

def block_cd(a0, time_limit, block_size=3, n_blocks=200):
    """Block coordinate descent: optimize small blocks of consecutive coordinates."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    for _ in range(n_blocks):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Pick a random block start
        start = rng.integers(0, max(1, n - block_size))
        block = list(range(start, min(start + block_size, n)))
        
        if len(block) < 2:
            continue
        
        # Get current values in the block
        x_fixed_sum = 1.0 - sum(a[j] for j in block)
        if x_fixed_sum < 0 or x_fixed_sum > 1:
            continue
        
        x_block = np.array([a[j] for j in block])
        x_block_sum = x_block.sum()
        if x_block_sum < 1e-12:
            continue
        
        # Optimize the block: minimize max(conv(a,a)) over the block coords
        # subject to block coords >= 0, sum(block coords) = x_block_sum
        
        def block_obj(x_b):
            x_b = np.maximum(x_b, 0)
            x_b_sum = x_b.sum()
            if x_b_sum < 1e-14:
                return 1e6
            # Scale to maintain total sum = 1
            x_b_norm = x_b * (x_block_sum / x_b_sum) if x_b_sum > 0 else x_b
            a_trial = a.copy()
            for i, j in enumerate(block):
                a_trial[j] = x_b_norm[i]
            return fast_convolve(a_trial).max()
        
        # Use Nelder-Mead for the small block
        x_init = x_block.copy()
        res = minimize(block_obj, x_init, method='Nelder-Mead',
                       options={'maxiter': 50, 'xatol': 1e-10, 'fatol': 1e-14})
        
        x_opt = np.maximum(res.x, 0)
        x_opt_sum = x_opt.sum()
        if x_opt_sum > 1e-12:
            x_opt = x_opt * (x_block_sum / x_opt_sum)
        
        a_trial = a.copy()
        for i, j in enumerate(block):
            a_trial[j] = x_opt[i]
        
        obj = fast_convolve(a_trial).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
