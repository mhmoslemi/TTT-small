def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)

            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0

            w /= ws

        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs < 1e-15:
            f = best_f.copy()
        else:
            f = f / fs
    
    return best_f, best_obj


def accelerated_sg(f0, time_limit, lr=0.005, momentum=0.9, top_k=50, temp=0.0001,
                  nesterov_mu=0.0, verbose=False):
    """Accelerated subgradient method with Nesterov-type momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)  # velocity for momentum
    if nesterov_mu > 0:
        y = f.copy()  # Nesterov lookahead
        beta = 1.0
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at y (Nesterov) or f
        base = y if (nesterov_mu > 0) else f
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = k - np.arange(js, je + 1)
            sg[js:je+1] += 2.0 * w[i] * base[idx]
        
        if nesterov_mu > 0:
            # Nesterov: y = f + (beta_prev - 1)/beta * v
            beta = 1.0 / (1.0 + nesterov_mu * np.sqrt(it + 1))
            # Simplified: y = (1-theta)*f + theta*y
            theta = 0.5 / (1.0 + nesterov_mu * np.sqrt(it))
            y = (1.0 - theta) * f + theta * y
            v = momentum * v - cur_lr * sg
            f_new = project_simplex(y + v)
        else:
            v = momentum * v - cur_lr * sg
            f_new = project_simplex(f + v)
        
        f = f_new
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    AccSG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        cur_lr *= 0.99995
        cur_temp *= 1.001
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 41)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            if nesterov_mu > 0:
                y = f.copy()
            no_improve = 0
    
    return best_f, best_obj


def active_set_lp(f0, n_peaks=10, time_limit=60.0):
    """Solve LP using only the top n_peaks constraints for speed.
    Min t s.t. b[k_i] <= t for top peaks, sum(a)=1, a>=0.
    This is a QP, not LP, since b[k] is quadratic.
    Instead, we linearize around f0: b[k] ≈ b0[k] + 2*a^T shift
    """
    n = len(f0)
    f = f0.astype(np.float64)
    conv = fast_convolve(f)
    
    # Get top peaks
    k_actual = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Linearize: b[k] ≈ 2 * sum_j a0[j] * a[k-j] (first-order Taylor)
    # Actually: b[k] = sum_j a[j]*a[k-j]
    # db[k]/da[m] = 2*a[k-m] (at the current point)
    # So: b[k] ≈ conv[k] + 2 * sum_m dm * a[k-m]
    # Let dm = a_new[m] - a[m], then b[k] ≈ conv[k] + 2*sum_m (a_new[m]-a[m]) * a[k-m]
    
    # This is still quadratic. Let's just use the subgradient approach
    # with the active set.
    # Actually, let's use a different approach: solve the LP relaxation
    # where we fix the "direction" from the subgradient.
    
    # Find the direction g that minimizes max_k (conv[k] + 2*sum_m g[m]*a[k-m])
    # where g is on the tangent space of the simplex (sum g = 0, g >= -a)
    
    # This is equivalent to the LP-FW step:
    # min_t max_k (conv[k] + 2*sum_m g[m]*a[k-m])
    # s.t. sum g[m] = 0, g[m] >= -a[m]
    
    # Let's solve this as an LP with the constraint b_lin[k] <= t
    # b_lin[k] = conv[k] + 2*sum_m g[m]*a[k-m]
    # = conv[k] + 2 * (g * a_rev)(k)  where a_rev is appropriate reversal
    
    # For the LP: variables are g[0..n-1] and t
    # Constraints: for each peak k_i:
    #   2*sum_m g[m]*a[k_i - m] + conv[k_i] <= t
    #   (i.e., sum_m g[m]*a[k_i-m] <= (t - conv[k_i])/2)
    # Plus: sum g = 0, g >= -a, t free
    # Minimize t
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints for each peak
    A_ub = np.zeros((k_actual, n + 1))
    b_ub = np.zeros(k_actual)
    
    for i, k in enumerate(peaks):
        for m in range(n):
            km = k - m
            if 0 <= km < n:
                A_ub[i, m] = 2.0 * f[km]
        A_ub[i, -1] = -1.0
        b_ub[i] = 0.0  # conv[k_i] + 2*sum g[m]*a[k_i-m] - t <= 0
                         # But we already have conv[k_i] in the constant
                         # Actually: 2*sum_m g[m]*a[k_i-m] <= t - conv[k_i]
                         # So: 2*sum_m g[m]*a[k_i-m] - t <= -conv[k_i]
        b_ub[i] = -conv[k_i]
    
    # Wait, let me redo this.
    # We want: conv[k_i] + 2*sum_m g[m]*a[k_i-m] <= t
    # => 2*sum_m g[m]*a[k_i-m] - t <= -conv[k_i]
    
    A_ub = np.zeros((k_actual, n + 1))
    b_ub = np.zeros(k_actual)
    for i, k in enumerate(peaks):
        for m in range(n):
            km = int(k) - m
            if 0 <= km < n:
                A_ub[i, m] = 2.0 * f[km]
        A_ub[i, -1] = -1.0
        b_ub[i] = -conv[int(k)]
    
    # Equality: sum g = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: g[m] >= -f[m], t free
    bounds = [(-f[m], None) for m in range(n)] + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        # Full line search along f + alpha*g
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            f_trial = f + alpha * g
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f_new = f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            return f_new
    
    return f  # No improvement


def multi_peak_targeting(f0, time_limit, n_peaks=20, verbose=False):
    """Iteratively target the top peaks with LP linearization + line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        f = f / s

    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f_new = active_set_lp(f, n_peaks=n_peaks, time_limit=2.0)
        if np.allclose(f_new, f, atol=1e-10):
            no_improve += 1
        else:

            if new_conv.max() < best_obj - 1e-14:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines B-spline inspired initialization, mirror descent (multiplicative weights),
accelerated subgradient with Nesterov momentum, active-set LP linearization,
and diverse multi-start with length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def bspline_sequence(order, n):
    """Generate a B-spline-like sequence of length n via iterative box convolution.
    Higher order = flatter self-convolution peak."""
    # Start with a single box
    f = np.array([1.0])
    box = np.array([1.0])
    # Convolve 'order' times to get B-spline of that order
    for _ in range(order):
        f = np.convolve(f, box, 'full')
    # Now f has length order+1, resample to length n
    old_n = len(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def tapered_sequence(n, power=2.0):
    """Sine-power window sequence."""
    x = np.linspace(0, np.pi, n)
    f = np.sin(x) ** power
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def exponential_sequence(n, rate=0.5):
    """Exponential taper sequence."""
    j = np.arange(n)
    f = np.exp(-rate * j / n)
    s = f.sum()
    f /= s
    return f


def compute_subgradient(f, conv, top_k, temp):
    """Compute softmax-weighted subgradient of max_k b[k]."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = k - np.arange(js, je + 1)
        sg[js:je + 1] += 2.0 * w[i] * f[idx]
    return sg


def mirror_descent_optimize(f0, time_limit, eta=0.005, top_k=50, temp=0.0005,
                            verbose=False):
    """Mirror descent (multiplicative weights) on simplex with adaptive annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_eta = eta
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient(f, conv, top_k, cur_temp)

        # Multiplicative update: a_new[j] = f[j] * exp(-eta * sg[j]) / Z
        log_update = -cur_eta * sg
        log_max = log_update.max()
        if log_max < -700:
            # All underflow, use uniform
            f_new = np.ones(n) / n
        else:
            log_update -= log_max
            update = f * np.exp(log_update)
            update = np.maximum(update, 0.0)
            us = update.sum()
            if us < 1e-30:
                f_new = np.ones(n) / n
            else:
                f_new = update / us

        f = f_new
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, eta={cur_eta:.6f}")
        else:
            no_improve += 1

        # Adaptive annealing
        cur_eta *= 0.99995
        cur_temp *= 1.0015

        # Increase eta slightly if improving (explorate more)
        if no_improve == 0 and cur_eta > 0.0005:
            cur_eta = max(0.0005, cur_eta * 1.0002)

        if no_improve > 800:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(f)
            cur_eta = max(0.002, eta * 0.4)
            cur_temp = temp
            no_improve = 0

        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs < 1e-15:
            f = best_f.copy()
        else:
            f = f / fs

    return best_f, best_obj


def accelerated_sg(f0, time_limit, lr=0.003, momentum=0.93, top_k=50,
                  temp=0.0001, nesterov_mu=0.0, verbose=False):
    """Accelerated subgradient with Nesterov momentum and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead
    beta_prev = 1.0

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if nesterov_mu > 0 and it > 1:
            beta = 1.0 / (1.0 + nesterov_mu * np.sqrt(it))
            theta = (beta_prev - 1.0) / beta
            y = f + theta * (f - y)
            beta_prev = beta
            base = y
        else:
            base = f

        sg = compute_subgradient(base, conv, top_k, cur_temp)
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        f = f_new
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    AccSG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1

        cur_lr *= 0.99998
        cur_temp *= 1.001

        if no_improve > 400:
            rng = np.random.default_rng(it * 13 + 41)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0015)
            v = np.zeros(n)
            y = f.copy()
            cur_lr = max(lr * 0.4, 0.0005)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def active_set_step(f, conv, n_peaks=12):
    """LP-based Frank-Wolfe step using linearization around current point.
    Solves: min t s.t. b_lin[k_i] <= t (for top peaks), sum(a)=1, a>=0.
    Uses first-order approximation of b[k]."""
    n = len(f)
    k_actual = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:].astype(int)

    # Linearize b[k] around f: b[k] ≈ conv[k] + 2 * sum_m (a_new[m]-f[m]) * f[k-m]
    # Let d = a_new - f, then: b_lin[k] = conv[k] + 2*sum_m d[m]*f[k-m]
    # We want: min max_i b_lin[k_i] s.t. sum d = 0, d >= -f
    # Equivalently: min t s.t. 2*sum_m d[m]*f[k_i-m] - t <= -conv[k_i]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((k_actual, n + 1))
    b_ub = np.zeros(k_actual)

    for i, k in enumerate(peaks):
        # 2*sum_m d[m]*f[k-m] for valid m
        # f[k-m] is valid when 0 <= k-m < n => k-n+1 <= m <= k
        ms = max(0, k - n + 1)
        me = min(n - 1, k)
        for m in range(ms, me + 1):
            A_ub[i, m] = 2.0 * f[k - m]
        A_ub[i, -1] = -1.0
        b_ub[i] = -conv[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = [(-f[m], None) for m in range(n)] + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if not res.success:
        return None

    d = res.x[:n]
    if np.max(np.abs(d)) < 1e-12:
        return None

    # Line search
    best_alpha = 0.0
    best_val = conv.max()
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_trial = f + alpha * d
        f_trial = np.maximum(f_trial, 0.0)
        st = f_trial.sum()
        if st < 1e-15:
            continue
        f_trial = f_trial / st
        val = fast_convolve(f_trial).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha < 1e-10:
        return None

    f_new = f + best_alpha * d
    f_new = np.maximum(f_new, 0.0)
    f_new = f_new / f_new.sum()
    return f_new


def lp_fw_optimize(f0, time_limit, n_peaks=15, verbose=False):
    """LP-based Frank-Wolfe with active set."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        f_new = active_set_step(f, conv, n_peaks=n_peaks)
        if f_new is None:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        new_conv = fast_convolve(f_new)
        if new_conv.max() < best_obj - 1e-14:
            best_obj = new_conv.max()
            best_f = f_new.copy()
            f = f_new
            conv = new_conv
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
            conv = new_conv

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
