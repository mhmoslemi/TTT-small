def softmax_gradient_ascent(z, T, lr=0.01):
    """
    One step of gradient ascent on z (where a = softmax(z)).
    We want to MINIMIZE the objective, so we do gradient descent on z.
    
    d(obj)/d(z_j) = a_j * (grad_obj_j - sum_i a_i * grad_obj_i)
    """
    a = np.exp(z - z.max())
    a /= a.sum()
    
    obj, grad_obj, conv = compute_gradient_smoothing(a, T)
    
    # Chain rule through softmax
    a_dot_g = np.dot(a, grad_obj)
    grad_z = a * (grad_obj - a_dot_g)
    
    return a, obj, grad_z, conv

def adam_smoothed_descent(a0, time_limit, T_start=1.0, T_end=50.0, beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """
    Smoothed gradient descent with Adam optimizer on the log-parametrization.
    """
    n = len(a0)
    
    # Initialize z from a0
    a_init = a0.copy()
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / a_init.sum()
    z = np.log(a_init)
    z -= z.mean()  # Center for numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_obj = float('inf')
    best_a = a0.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Anneal temperature
        fraction = min(1.0, t / (time_limit * 0.7))  # Anneal over first 70% of time
        T_cur = T_start * (T_end / T_start) ** fraction
        
        a = np.exp(z - z.max())
        a /= a.sum()
        
        # Compute gradient
        obj, grad_obj, conv = compute_gradient_smoothing(a, T_cur)
        
        # Chain rule through softmax
        a_dot_g = np.dot(a, grad_obj)
        grad_z = a * (grad_obj - a_dot_g)
        
        # Handle numerical issues
        grad_z = np.nan_to_num(grad_z, nan=0.0, posinf=0.0, neginf=0.0)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        base_lr = 0.01 * (1.0 + 0.5 * (1.0 - fraction))  # Start faster, slow down
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        z = z - step
        
        # Periodic renormalization of z for numerical stability
        if t % 100 == 0:
            z -= z.mean()
        
        a = np.exp(z - z.max())
        a /= a.sum()
        
        # Track best (use true max, not smoothed)
        true_obj = 2.0 * n * conv.max() / (a.sum() ** 2)

            best_obj = true_obj
            best_a = a.copy()
        
        if verbose and t % 500 == 0:

        grad_z = np.nan_to_num(grad_z, nan=0.0, posinf=0.0, neginf=0.0)
        
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * grad_z ** 2
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        
        base_lr = 0.005
        step = base_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = z - step
        
        if t % 100 == 0:
            z -= z.mean()
        
        a = np.exp(z - z.max())
        a /= a.sum()
        
        true_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()
        
    return best_a, best_obj
