new_conv[k] = old_conv[k] + 2*delta*(a_{k-i} + a_{k-(n-1-i)}) + delta^2 * ([k=2i] + [k=2(n-1-i)])

This is also O(n) per candidate.

Great! So for coordinate descent on symmetric pairs:
- Precompute old_conv (O(n) or O(n log n) once)
- For each pair (i, n-1-i), the new conv as a function of delta is old_conv + 2*delta*(a_{k-i} + a_{k-(n-1-i)}) + delta^2 * extra
- We want to minimize max_k new_conv[k] / (sum + delta)^2 (for the normalized version)

Actually, for the sum=1 case: if we add delta to position i and delta to position n-1-i, the new sum is 1 + 2*delta. After normalization, all values are divided by (1+2*delta). So the new normalized conv is:
(new_conv) / (1+2*delta)^2

And we want to minimize the max of this.

This is a 1D problem: minimize f(delta) = max_k [old_conv[k] + 2*delta * w_k + delta^2 * e_k] / (1+2*delta)^2

where w_k = a_{k-i} + a_{k-(n-1-i)} and e_k = [k=2i] + [k=2(n-1-i)].

This is a ratio of a convex function (max of quadratics) to a quadratic. It's not convex in general, but we can do golden section search or ternary search over a reasonable range of delta.

The range of delta is bounded: we need 1+2*delta > 0 (so delta > -1/2) and the new values must be non-negative (so a_i + delta >= 0, i.e., delta >= -a_i).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric coordinate descent with exact 1D minimization
2. Nesterov-accelerated subgradient descent with adaptive step size
3. Multi-length exploration with interpolation
4. Block coordinate descent for local refinement
5. Periodic annealing to escape local minima

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Enforce palindromic symmetry: a[i] = a[n-1-i]."""
    return (a + a[::-1]) / 2.0


def symmetric_coord_descent(a, time_limit, verbose=False):
    """
    Coordinate descent on symmetric pairs (i, n-1-i).
    For each pair, find the optimal delta that minimizes max(conv)/(sum)^2.
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    a = symmetrize(a)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve_rounds = 0
    round_num = 0
    half = n // 2
    
    while time.monotonic() - t0 < time_limit:
        round_num += 1
        round_improved = False
        elapsed = time.monotonic() - t0
        
        # Adaptive number of sweep directions
        if round_num % 2 == 0:
            order = list(range(half))
        else:
            order = list(range(half - 1, -1, -1))
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            j = n - 1 - i
            
            # Compute w and e vectors
            w = np.zeros(2 * n - 1)
            w[i:i+n] += a
            w[j:j+n] += a
            
            e = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                e[2 * i] = 1
            if 2 * j < 2 * n - 1:
                e[2 * j] = 1
            
            ai = a[i]
            # delta range: [-ai, upper]
            lo = -ai + 1e-12
            hi = 0.05  # reasonable upper bound
            
            # Coarse search
            best_delta = 0.0
            best_val = best_obj  # current objective
            
            num_coarse = 16
            deltas = np.linspace(lo, hi, num_coarse)
            # Vectorized evaluation
            new_convs = conv[None, :] + 2 * deltas[:, None] * w[None, :] + (deltas * deltas)[:, None] * e[None, :]
            new_S_vals = S + 2 * deltas
            valid = new_S_vals > 1e-10
            if not valid.any():
                continue
            new_convs = new_convs[valid]
            new_S_vals = new_S_vals[valid]
            vals = new_convs.max(axis=1) / (new_S_vals * new_S_vals)
            valid_deltas = deltas[valid]
            
            min_idx = np.argmin(vals)
            best_val = vals[min_idx]
            best_delta = valid_deltas[min_idx]
            
            if best_val < best_obj - 1e-13:
                # Fine search around best_delta
                fine_lo = max(lo, best_delta - (hi - lo) / num_coarse)
                fine_hi = min(hi, best_delta + (hi - lo) / num_coarse)
                fine_deltas = np.linspace(fine_lo, fine_hi, 12)
                new_convs_f = conv[None, :] + 2 * fine_deltas[:, None] * w[None, :] + (fine_deltas * fine_deltas)[:, None] * e[None, :]
                new_S_f = S + 2 * fine_deltas
                valid_f = new_S_f > 1e-10
                if valid_f.any():
                    new_convs_f = new_convs_f[valid_f]
                    new_S_f = new_S_f[valid_f]
                    vals_f = new_convs_f.max(axis=1) / (new_S_f * new_S_f)
                    valid_f_deltas = fine_deltas[valid_f]
                    min_idx_f = np.argmin(vals_f)
                    best_val = vals_f[min_idx_f]
                    best_delta = valid_f_deltas[min_idx_f]
                
                a[i] += best_delta
                a[j] += best_delta
                S = a.sum()
                # Renormalize to keep S=1
                a = a / S
                S = 1.0
                conv = fast_convolve(a)
                obj = conv.max()
                
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                round_improved = True
        
        if round_improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds > 30:
                break
        
        if verbose and round_num % 10 == 0:
            print(f"    CoordDescent round {round_num}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, verbose=False):
    """
    Nesterov-accelerated subgradient descent on max(conv(a,a)) with sum(a)=1.
    Uses top-k peaks for subgradient computation.
    """
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Adaptive step size (Polyak-like)
    gap_best = fast_convolve(a).max()  # best gap
    # Lower bound: average of conv = 1/(2n-1) for sum=1
    lb = 1.0 / (2 * n - 1)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_convolve(y)
        M_y = conv_y.max()
        
        # Find top-k peaks
        k = min(20, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k)[-k:]
        pv = conv_y[peaks]
        
        # Weight: exponential of (pv - M_y) / T
        T = 0.01 / n  # temperature
        scaled = (pv - pv.max()) / max(T, 1e-15)
        wts = np.exp(scaled)
        wts = wts / wts.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if wts[idx] < 1e-10:
                continue
            k_val = int(peaks[idx])
            # Gradient of (a*a)[k_val] w.r.t. a:
            # d/d(a_i) (a*a)[k] = 2*a_{k-i} (if k != 2i) or 2*a_i (if k=2i)
            # But we need to sum over i:
            # grad[k-i] += 2 * a_i ... 
            # Actually: (a*a)[k] = sum_i a_i * a_{k-i}
            # d/d(a_m) = a_{k-m} + a_{k-m} = 2*a_{k-m} (if k-m != m)
            #           = 2*a_m (if k-m = m, i.e., m = k/2)
            
            js = max(0, k_val - n + 1)
            je = min(n - 1, k_val)
            idxs = np.arange(js, je + 1)
            # grad contribution: 2 * a[k - idxs] for each idx
            sg[idxs] += 2.0 * wts[idx] * a[k_val - idxs]
            # Handle the self-term (when idx = k_val/2)
            # Actually the formula 2*a_{k-m} already handles it correctly:
            # (a*a)[k] = sum_m a_m * a_{k-m}
            # d/d(a_m) = a_{k-m} + a_{k-m} = 2*a_{k-m}
            # (when m != k-m)
            # d/d(a_m) = 2*a_m (when m = k-m, i.e., k = 2m)
            # But 2*a_{k-m} = 2*a_m when k=2m, so the formula 2*a_{k-m} works for all cases.
            # Wait, that's not right. Let me re-derive.
            # (a*a)[k] = sum_m a_m * a_{k-m}
            # The partial derivative w.r.t. a_m is:
            #   a_{k-m} (from the term where the first factor is a_m)
            #   + a_{k-m} (from the term where the second factor is a_m, i.e., when the sum index equals k-m)
            #   But these are the same term when m = k-m, so it's just a_{k-m} in that case.
            # Wait no: the sum is over all valid m values. The term a_m * a_{k-m} appears
            # once for each m. The derivative w.r.t. a_m is a_{k-m} (from the m-th term).
            # But also, when the sum index is k-m, the term is a_{k-m} * a_m, whose
            # derivative w.r.t. a_m is also a_{k-m}.
            # So if m != k-m (i.e., m != k/2), both terms contribute: total = 2*a_{k-m}.
            # If m = k/2 (requires k even), only one term: total = a_{k/2} = a_m.
            
            # Actually I realize my formula sg[idxs] += 2.0 * wts[idx] * a[k_val - idxs] 
            # correctly gives 2*a_{k-m} for each m in the valid range.
            # For the case m = k/2, this gives 2*a_m, but the correct derivative is a_m.
            # So we need to subtract a_m for the self-term.
            
            if k_val % 2 == 0:
                m_half = k_val // 2
                if 0 <= m_half < n:
                    sg[m_half] -= wts[idx] * a[m_half]
        
        # Nesterov update
        # Step size: use Polyak-like: step = (M_y - lb) / (||sg||^2 + eps)
        sg_norm_sq = np.dot(sg, sg)
        if sg_norm_sq < 1e-20:
            step_size = 0.01 / n
        else:
            step_size = (M_y - lb) / (sg_norm_sq + 1e-10)
            step_size = min(step_size, 1.0 / n)  # cap
        
        y = a - step_size * sg
        y = project_simplex(y)
        
        # Update momentum
        beta_new = (1 + it + 1) / (2 + it + 1 + beta)  # standard Nesterov beta_t
        # Actually, standard Nesterov: beta_t = (1 + t) / (2 + t + beta_{t-1})... 
        # Let me use the simpler version: beta_t = (1+t)/(1+t+beta_{t-1}) where beta_0 = 1
        # Or just use a fixed momentum of 0.9 with Polyak step.
        
        a_new = (1 - beta) * y + beta * a
        a_new = project_simplex(a_new)
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    NSGD iter {it}: obj={best_obj:.10f}, step={step_size:.6f}")
    
    return best_a, best_obj


def multi_start_search(a0, n_target, time_limit, verbose=False):
    """
    Try multiple random perturbations of a0, optimized via coordinate descent.
    """
    n = n_target
    best_a = a0.copy() if len(a0) == n else interpolate_sequence(a0, n)
    best_obj = fast_convolve(best_a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    num_starts = 0
    while time.monotonic() - t0 < time_limit:
        num_starts += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        t_alloc = min(remaining / max(1, 10 - num_starts), 60)
        if t_alloc < 5:
            break
        
        if num_starts == 1:
            f_init = best_a.copy()
        elif num_starts == 2:
            f_init = best_a[::-1].copy()
            f_init = f_init / f_init.sum()
        else:
            noise_scale = 0.01 * min(num_starts, 5)
            f_init = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
        
        a_opt, obj = symmetric_coord_descent(f_init, time_limit=t_alloc)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    MultiStart {num_starts}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def block_coord_descent(a, time_limit, block_size=4, verbose=False):
    """
    Block coordinate descent: optimize small blocks of consecutive pairs together.
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    a = symmetrize(a)
    S = a.sum()
    a = a / S
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    half = n // 2
    no_improve = 0
    rng = np.random.default_rng(777)
    
    for iteration in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 50:
            break
        
        improved = False
        # Pick a random block
        start = rng.integers(0, max(1, half - block_size + 1))
        end = min(start + block_size, half)
        
        # Current values of the block
        block_vals = a[start:end].copy()
        block_sum = block_vals.sum()
        
        if block_sum < 1e-12:
            # Try to add mass from elsewhere
            other_sum = a.sum() - block_sum
            if other_sum > 1e-12:
                a[start:end] = other_sum * 0.01
                a = a / a.sum()
                conv = fast_convolve(a)
                best_obj = conv.max()
                best_a = a.copy()
                improved = True
            continue
        
        # Try several perturbations
        for _ in range(5):
            if time.monotonic() - t0 > time_limit:
                break
            pert = rng.standard_normal(end - start) * 0.02 * block_sum
            trial = np.maximum(block_vals + pert, 0)
            ts = trial.sum()
            if ts < 1e-12:
                continue
            trial = trial * (block_sum / ts)
            
            a_trial = a.copy()
            a_trial[start:end] = trial
            # Also update symmetric positions

                a_trial[n - 1 - k] = trial[k - start]
            a_trial = a_trial / a_trial.sum()
            
            obj = fast_convolve(a_trial).max()
            
            if obj < best_obj - 1e-13:
                a = a_trial
                best_obj = obj
                best_a = a.copy()
                conv = fast_convolve(a)
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_obj
