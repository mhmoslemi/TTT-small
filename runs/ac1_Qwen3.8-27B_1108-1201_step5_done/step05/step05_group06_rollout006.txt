def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):

    return np.maximum(v - theta, 0.0)

            #     So we need 0 <= i < n AND 0 <= k-i < n, i.e., k-n+1 <= i <= k.
            #     The terms involving a_j (i=j) exist when max(0, k-n+1) <= j <= min(n-1, k).
            #     The terms involving a_{k-j} (i=k-j) exist when max(0, k-n+1) <= k-j <= min(n-1, k),
            #     i.e., j is in [k-min(n-1,k), k-max(0,k-n+1)] = [max(0, 2k-n+1-j)... this is getting circular.
            
            # Let me just compute it directly:
            # c_k(x) = c_base[k] + x * a_kj + a_kj * x + (x^2 if k-j==j else 0)
            # where a_kj = a_noj[k-j] if 0<=k-j<n else 0
            # But we need to be careful: the cross term x * a_kj appears once for i=j 
            # and once for i=k-j (if k-j != j). If k-j == j (k=2j), it's x * a_j = x^2.
            
            # Actually, the simplest way: 
            # c_k(x) = c_base[k] + 2*x*b_k + x^2*d_k
            # where b_k = a_noj[k-j] if 0<=k-j<n else 0
            #       d_k = 1 if k==2j else 0
            # But we need to check: when i=j, the term is a_j * a_{k-j}. If k-j is valid, this is x * a_noj[k-j].
            # When i=k-j (and k-j != j), the term is a_{k-j} * a_j = a_noj[k-j] * x.
            # So total cross term: 2*x*a_noj[k-j] (when k!=2j and k-j valid)
            # When k=2j: only one term, x * a_j = x^2.
            # When k-j is not valid (k-j < 0 or k-j >= n): no cross term from i=j.
            #   But i=k-j: need 0<=k-j<n. If k-j<0, then i=k-j<0, invalid. If k-j>=n, then i=k-j>=n, invalid.
            #   So no contribution at all.
            
            # So: b_k = a_noj[k-j] if (0 <= k-j < n and k != 2*j) else 0
            #     d_k = 1 if k == 2*j else 0
            #     c_k(x) = c_base[k] + 2*x*b_k + x^2*d_k
            
            # Wait, I think I'm overcomplicating. Let me just use:
            # c_k(x) = c_base[k] + 2*x*a_noj[k-j] (if 0<=k-j<n) + x^2 (if k==2j)
            # because when k==2j, the "2*x*a_noj[k-j]" term becomes 2*x*a_noj[j] but a_noj[j]=0
            # (since we set a_noj[j]=0), so it's 0, and we add x^2.
            # When k!=2j but 0<=k-j<n: c_k = c_base[k] + 2*x*a_noj[k-j]
            # When k != 2j and k-j not in [0,n): c_k = c_base[k]
            
            # Actually wait: a_noj[j] = 0 (we set it to 0). So:
            # If k = 2j: b_k = a_noj[j] = 0, d_k = 1. c_k = c_base[k] + x^2. ✓
            # If k != 2j and 0<=k-j<n: b_k = a_noj[k-j], d_k = 0. c_k = c_base[k] + 2*x*b_k. ✓
            # If k-j not in [0,n): b_k = 0, d_k = 0. c_k = c_base[k]. ✓
            
            conv_len = 2 * n - 1
            b = np.zeros(conv_len)
            if 0 <= k - j < n:  # This would be for a specific k
                pass  # We need to do this for all k
            
            # For all k from 0 to 2n-2:
            # b[k] = a_noj[k-j] if 0 <= k-j < n else 0
            # d[k] = 1 if k == 2j else 0
            
            k_range = np.arange(conv_len)
            k_minus_j = k_range - j
            valid = (k_minus_j >= 0) & (k_minus_j < n)
            b_vec = np.where(valid, a_noj[np.clip(k_minus_j, 0, n-1)], 0.0)
            d_vec = np.zeros(conv_len)
            if 2 * j < conv_len:
                d_vec[2 * j] = 1.0
            
            # Now we want to minimize max_k (c_base[k] + 2*x*b_vec[k] + x^2*d_vec[k]) over x in [0, 1]
            # (since a >= 0 and sum(a) = 1, we have 0 <= x <= 1)
            # Actually x can be at most 1 (if all mass is at position j)
            
            # This is: minimize max of (quadratic in x) functions
            # The optimal x is where the maximum is minimized.
            # We can use bisection or ternary search on x.
            
            # Actually, max of convex functions is convex, so we can use ternary search or bisection on the derivative.
            # Let f(x) = max_k (c_base[k] + 2*x*b_vec[k] + x^2*d_vec[k])
            # f'(x) = 2*b_vec[k*] + 2*x*d_vec[k*] where k* = argmax
            # f is convex, so we can find the minimum by bisection on f'.
            
            # Bisection on x in [0, 1]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = c_base + 2 * mid * b_vec + mid * mid * d_vec
                k_max = np.argmax(vals)
                deriv = 2 * b_vec[k_max] + 2 * mid * d_vec[k_max]
                if deriv > 1e-14:
                    hi = mid
                elif deriv < -1e-14:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) / 2.0
            
            # Apply the update
            # We need to maintain sum(a) = 1. Currently sum(a) = 1.
            # If we change a[j] from x_old to x_new, the sum changes by (x_new - x_old).
            # We need to redistribute the difference among the other coordinates proportionally.
            
            if abs(x_new - x_old) < 1e-15:
                continue
            
            diff = x_new - x_old
            a_noj_sum = 1.0 - x_old
            if a_noj_sum < 1e-12:
                continue
            
            a_trial = a.copy()
            a_trial[j] = x_new
            a_trial[~np.eye(n, dtype=bool)[j]] *= (1.0 - x_new) / a_noj_sum
            # Simpler: 
            a_trial = a.copy()
            a_trial[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a_trial[mask] = a[mask] * (1.0 - x_new) / a_noj_sum
            a_trial = np.maximum(a_trial, 0.0)
            s_check = a_trial.sum()
            if s_check < 1e-12:
                continue
            a_trial /= s_check
            
            c_trial = fast_convolve(a_trial)
            val = c_trial.max()
            if val < best_val - 1e-14:
                best_val = val
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_val

def multi_scale_search(a_best, deadline, verbose=False):
    """Search over multiple sequence lengths."""
    n_best = len(a_best)
    s = a_best.sum()
    a_norm = a_best / s
    
    candidate_lengths = set()
    # Around the best length
    for delta in [-32, -16, -8, -4, -2, -1, 0, 1, 2, 4, 8, 16, 32]:
        candidate_lengths.add(n_best + delta)
    # Powers of 2 and other nice numbers
    for ln in [512, 640, 768, 896, 1024, 1280]:
        if 64 <= ln <= 4096:
            candidate_lengths.add(ln)
    
    candidate_lengths = sorted(l for l in candidate_lengths if l >= 32)
    
    return candidate_lengths
