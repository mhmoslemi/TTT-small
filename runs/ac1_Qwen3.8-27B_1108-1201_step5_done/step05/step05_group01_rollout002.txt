"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast projected subgradient with
Nesterov momentum, FFT-based gradient computation via shift formula, adaptive
peak weighting, periodic smoothing, and multi-length exploration.
Key innovation: exploit the structure grad_k[m] = 2*a[k-m] for O(n) gradient
at each peak, enabling thousands of iterations even at n=1800.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_at_peaks(a, peaks, weights):
    """
    Compute the weighted sum of gradients at given peaks.
    grad_k[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    for i, k in enumerate(peaks):
        k = int(k)
        w = weights[i]
        if w < 1e-15:
            continue
        # Valid m range: 0 <= k-m < n => k-n+1 <= m <= k
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += 2.0 * w * a[k - indices]
    return grad

    v = np.zeros(n)

    best_f = f.copy()

    no_improve = 0
    it = 0
    cur_lr = lr0

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, 2 * n - 1)
        # Use softmax weighting over all peaks (not just top-k) for smoother gradients
        if it % 200 == 0:
            # Recompute weighting from all peaks
            all_peaks = np.arange(len(conv))
            pv = conv[all_peaks]
            scaled = pv / temp0

            w_all = np.exp(scaled)
            ws = w_all.sum()
            if ws < 1e-10:

                w = np.ones(len(peaks)) / len(peaks)
            else:

                    top_in_sig = np.argsort(w_all[sig_indices])[::-1][:k_actual]

                    w = w_all[peaks]
                    w /= w.sum()
                else:

                    w = w_all[peaks]
                    w /= w.sum()
        else:

            pv = conv[peaks]

            w = np.exp(scaled)

        if no_improve <= 50:
            cur_lr *= 0.999
        elif no_improve <= 200:
            cur_lr *= 0.995
        else:
            cur_lr *= 0.99

        # Periodic restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.003 * min((no_improve - 300) // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.8, 0.002)
            no_improve = 0
            conv = fast_convolve(f)
            best_obj = conv.max()
            if best_obj < obj_new:
                pass  # keep best_f

    return best_f, best_obj


def smoothing_step(f, sigma):
    """Apply Gaussian smoothing in the spatial domain."""
    n = len(f)
    kernel_size = max(3, int(6 * sigma) | 1)
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    # Circular convolution for efficiency
    f_smooth = np.convolve(f, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0.0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def peak_shaving(f, n_shave=5, strength=0.3):
    """
    Reduce mass at positions that contribute most to the peak convolution lag.
    If peak is at lag k, positions j and k-j contribute. Reduce those and
    redistribute to positions that don't contribute (or contribute less).
    """
    n = len(f)
    conv = fast_convolve(f)
    # Find top peak lags
    peak_lags = np.argpartition(conv, -n_shave)[-n_shave:]
    peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

    f_new = f.copy()
    for k in peak_lags:
        k = int(k)
        # Positions contributing to lag k
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        contrib_positions = np.arange(m_lo, m_hi + 1)
        other_positions = np.setdiff1d(np.arange(n), contrib_positions)

        if len(other_positions) == 0:
            continue

        # Reduce contributing positions, increase others
        total_contrib = f_new[contrib_positions].sum()
        if total_contrib > 1e-12:
            # Move mass from high-contributing to low-contributing
            f_new[contrib_positions] *= (1.0 - strength)
            f_new[other_positions] += strength * total_contrib / len(other_positions)

    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    return f_new


def pairwise_redistribute(f, n_pairs=20, rng=None):
    """
    Try redistributing mass between pairs of positions to reduce peaks.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    conv = fast_convolve(f)
    peak = conv.max()
    f_new = f.copy()

    for _ in range(n_pairs):
        # Pick a peak lag
        k = int(rng.integers(0, len(conv)))
        # Weight towards actual peaks
        peak_lag = int(rng.choice(np.arange(len(conv)), p=conv / conv.sum()))
        m_lo = max(0, peak_lag - n + 1)
        m_hi = min(n - 1, peak_lag)
        # Pick two positions in the contributing range
        m1 = int(rng.integers(m_lo, m_hi + 1))
        # Pick a position not contributing
        other = np.setdiff1d(np.arange(n), np.arange(m_lo, m_hi + 1))
        if len(other) == 0:
            continue
        m2 = int(other[int(rng.integers(0, len(other)))])

        # Try to move mass from m1 to m2
        delta = 0.001 * f[m1]
        f_trial = f_new.copy()
        f_trial[m1] -= delta
        f_trial[m2] += delta
        f_trial = np.maximum(f_trial, 0.0)
        f_trial /= f_trial.sum()

        new_conv = fast_convolve(f_trial)
        if new_conv.max() < peak - 1e-14:
            f_new = f_trial
            peak = new_conv.max()

    return f_new


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Apply slight smoothing
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_interp = np.convolve(f_interp, kernel, mode='same')
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_init(n, freq, amp):
    """Generate cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Initialize from parent or cosine structures at target length.
    2. Fast subgradient with Nesterov momentum (main workhorse).
    3. Periodic smoothing and peak shaving.
    4. Multi-start exploration.
    5. Final fine-tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-6:
            return float('inf')
        a = a / a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # === Get parent sequence ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: Interpolate to target lengths and quick evaluate ===
    print("=== Phase 1: Interpolation to target lengths ===")
    target_lengths = [1024, 1536, 1800, 2048]

    candidates = []
    for tl in target_lengths:
        f_t = interpolate_sequence(parent, tl)
        s = try_seq(f_t, f"interp_n{tl}")
        if np.isfinite(s):
            candidates.append((f_t, tl, s))

    # Also try cosine inits at target length
    for tl in [1024, 1536, 1800]:
        for freq, amp in [(1, 0.3), (2, 0.2), (3, 0.15), (1, 0.5)]:
            f_c = cosine_init(tl, freq, amp)
            s = try_seq(f_c, f"cos_n{tl}_f{freq}")
            if np.isfinite(s) and s < best_score + 0.05:
                candidates.append((f_c, tl, s))

    # Sort by score
    candidates.sort(key=lambda x: x[2])

    # === Phase 2: Main subgradient optimization on best candidates ===
    print(f"\n=== Phase 2: Subgradient optimization (best={best_score:.6f}) ===")

    # Optimize the best candidate at its native length
    if candidates:
        f_best, n_best, s_best = candidates[0]
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem * 0.4, 300)
            print(f"  Main opt: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(f_best, time_limit=t_alloc,
                                             lr0=0.015, mom0=0.82,
                                             top_k=60, temp0=0.0008)
            try_seq(a_opt, f"subgrad_n{n_best}")

    # Optimize second best if time
    if len(candidates) > 1:
        f_2nd, n_2nd, s_2nd = candidates[1]
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem * 0.25, 150)
            print(f"  Second opt: n={n_2nd}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(f_2nd, time_limit=t_alloc,
                                             lr0=0.012, mom0=0.8,
                                             top_k=50, temp0=0.001)
            try_seq(a_opt, f"subgrad2_n{n_2nd}")

    # === Phase 3: Smoothing + re-optimization ===
    print(f"\n=== Phase 3: Smoothing refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        for sigma in [1.5, 2.5, 4.0]:
            rem = deadline - time.monotonic() - 400
            if rem < 20:
                break
            f_smooth = smoothing_step(best_seq, sigma)
            s = try_seq(f_smooth, f"smooth_s{sigma}")
            if np.isfinite(s) and s < best_score + 0.01:
                t_alloc = min(rem * 0.2, 80)
                if t_alloc > 10:
                    a_opt, _ = subgradient_optimize(f_smooth, time_limit=t_alloc,
                                                     lr0=0.008, mom0=0.85,
                                                     top_k=40, temp0=0.0005)
                    try_seq(a_opt, f"smooth+opt_s{sigma}")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        for strength in [0.1, 0.2, 0.3]:
            rem = deadline - time.monotonic() - 300
            if rem < 15:
                break
            f_shaved = peak_shaving(best_seq, n_shave=5, strength=strength)
            s = try_seq(f_shaved, f"shave_{strength}")
            if np.isfinite(s) and s < best_score + 0.01:
                t_alloc = min(rem * 0.15, 50)
                if t_alloc > 8:
                    a_opt, _ = subgradient_optimize(f_shaved, time_limit=t_alloc,
                                                     lr0=0.006, mom0=0.85,
                                                     top_k=30, temp0=0.0004)
                    try_seq(a_opt, f"shave+opt_{strength}")

    # === Phase 5: Multi-start from best with perturbations ===
    print(f"\n=== Phase 5: Multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        n_starts = min(6, int((deadline - time.monotonic() - 200) / 60))
        for i in range(max(1, n_starts)):
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 100)

            if i == 0:
                f_init = best_seq.copy()
            elif i == 1:
                noise = rng.standard_normal(n_b) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif i == 2:
                # Mix with cosine
                f_cos = cosine_init(n_b, 2, 0.2)
                f_init = 0.8 * best_seq + 0.2 * f_cos
                f_init = project_simplex(f_init)
            elif i == 3:
                # Smoothed version
                f_init = smoothing_step(best_seq, 2.0)
            elif i == 4:
                # Anti-smoothed (sharpened)
                f_sharp = 2.0 * best_seq - smoothing_step(best_seq, 1.5)
                f_sharp = np.maximum(f_sharp, 0.0)
                f_sharp /= f_sharp.sum()
                f_init = f_sharp
            else:
                # Reverse
                f_init = best_seq[::-1].copy()
                f_init /= f_init.sum()

            a_opt, _ = subgradient_optimize(f_init, time_limit=t_alloc,
                                             lr0=0.01, mom0=0.83,
                                             top_k=45, temp0=0.0007)
            try_seq(a_opt, f"start_{i}")

    # === Phase 6: Interpolate best to longer lengths ===
    print(f"\n=== Phase 6: Length extension (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        for target_n in [max(n_b + 256, 1280), 1800, 2048]:
            if target_n <= n_b:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            f_ext = interpolate_sequence(best_seq, target_n)
            s = try_seq(f_ext, f"extend_n{target_n}")
            if np.isfinite(s):
                t_alloc = min(rem * 0.25, 100)
                if t_alloc > 10:
                    a_opt, _ = subgradient_optimize(f_ext, time_limit=t_alloc,
                                                     lr0=0.012, mom0=0.82,
                                                     top_k=50, temp0=0.0008)
                    try_seq(a_opt, f"extend_opt_n{target_n}")

    # === Phase 7: Final fine-tuning ===
    print(f"\n=== Phase 7: Final fine-tuning (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_fine = min(rem * 0.4, 80)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_fine,
                                             lr0=0.003, mom0=0.9,
                                             top_k=30, temp0=0.0002)
            try_seq(a_opt, "final_fine")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 40
        for _ in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 8:
                break
            noise = rng.standard_normal(n_b) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_small = min(rem * 0.3, 25)
            a_opt, _ = subgradient_optimize(f_pert, time_limit=t_small,
                                             lr0=0.002, mom0=0.92,
                                             top_k=25, temp0=0.00015)
            try_seq(a_opt, "final_perturb")

        # Pairwise redistribution
        rem = deadline - time.monotonic() - 20
        if rem > 5:
            f_pair = pairwise_redistribute(best_seq, n_pairs=50, rng=rng)
            try_seq(f_pair, "pairwise")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
