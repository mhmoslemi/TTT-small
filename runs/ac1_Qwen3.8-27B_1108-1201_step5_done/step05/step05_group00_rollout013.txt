"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Edge-concentrated initializations
- Smoothed gradient descent with adaptive temperature
- Peak-targeted 2-coordinate moves
- Multi-n exploration
- Peak balancing heuristic

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    """Compute the gradient contribution: (p * a) in the valid convolution sense."""
    return np.correlate(p, a, mode='valid')

def compute_grad_smooth(a, T):
    """Compute smoothed gradient of max(conv(a,a)) using softmax."""
    c = fast_convolve(a)
    scaled = c / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    # Gradient of sum_k p[k] * c[k] w.r.t. a[j]
    # c[k] = sum_j a[j] * a[k-j]
    # d(c[k])/d(a[j]) = a[k-j] (when valid)
    # So grad_j = sum_k p[k] * a[k-j] (for valid k-j)
    # This is: convolve(p, a) evaluated at the right positions
    # Actually: grad_j = 2 * sum_k p[k] * a[k-j] for j in [0, n-1]
    # which is 2 * (reverse_convolve(p, a))[j]
    # Let me use correlate properly
    return 2.0 * hankel_mv(p, a)

def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent with adaptive temperature and learning rate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    patience_counter = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        grad = compute_grad_smooth(a, T)
        
        # Backtracking line search
        best_candidate = a
        best_val = fast_convolve(a).max()
        
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.125]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            patience_counter = 0
        else:
            no_improve += 1
            patience_counter += 1
        
        # Adaptive temperature: cool down slowly, reset when stuck
        if no_improve > 10:
            T *= 1.02
        if no_improve > 50:
            T *= 1.05
        if patience_counter > 200:
            # Restart from best with noise
            rng = np.random.default_rng((it * 104729) % (2**31))
            noise_scale = 0.002 + 0.003 * min(1.0, (patience_counter - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0
            patience_counter = 0
            best_obj = fast_convolve(a).max()
        
        # Slowly decay learning rate
        lr *= 0.9998
    
    return best_a, best_obj

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, T):
    """Compute smoothed subgradient of max(conv(a,a)) via softmax."""
    c = fast_convolve(a)
    scaled = c / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    # grad_j = 2 * sum_k p[k] * a[k-j] for valid k-j
    # This is 2 * (cross-correlation of p and a at the right offset)
    # np.correlate(p, a, 'valid') gives sum_j p[j+n-1-k] * a[j+k] -- not quite
    # We want: for each j in [0,n-1], grad[j] = 2 * sum_{k: 0<=k-j<n} p[k]*a[k-j]
    # Let m = k-j, then k = j+m, and m ranges over [max(0,-j), min(n-1-j, 2n-1-j)]
    # Since p has length 2n-1 and a has length n:

    # Which equals sum_i p[i+(n-1-l)] * a[i+l]... 
    # Actually let me just use: grad = 2 * correlate(p, a, 'valid')
    # correlate(p, a, 'valid')[j] = sum_i p[i + (len(p)-len(a)+1)*... 
    # Let me just use the direct formula
    return 2.0 * np.correlate(p, a, mode='valid')

        if patience > 300:
            rng = np.random.default_rng((it * 104729 + 17) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (patience - 300) / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            if nesterov:
                v = np.zeros(n)
            no_improve = 0
            patience = 0

        lr *= 0.99985

    return best_a, best_obj


def peak_targeted_moves(a0, time_limit, n_peaks=5):
    """
    For top peaks in conv(a,a), use 2-coordinate moves to reduce them.
    Moves mass from positions with large partners to positions with small partners.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        c = fast_convolve(a)
        peak_indices = np.argpartition(c, -n_peaks)[-n_peaks:]
        peak_indices = peak_indices[np.argsort(c[peak_indices])[::-1]]

        improved = False

        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Partner values: for each j in affected, partner is a[k-j]
            partner_vals = a[k - affected]

            # Find j1 with largest partner (most contributes to c[k])
            j1_idx = np.argmax(partner_vals)
            j1 = affected[j1_idx]
            # Find j2 with smallest partner (least contributes)
            j2_idx = np.argmin(partner_vals)
            j2 = affected[j2_idx]

            if j1 == j2:
                continue

            # Try transferring mass from j1 to j2
            # c[k] change ≈ -delta * a[k-j1] + delta * a[k-j2] = -delta*(partner[j1]-partner[j2]) < 0
            max_delta = a[j1] * 0.15
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
                delta = min(frac * a[j1], max_delta)
                if delta < 1e-15:
                    continue
                candidate = a.copy()
                candidate[j1] -= delta
                candidate[j2] += delta
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                new_c = fast_convolve(candidate)
                if new_c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = new_c.max()
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Try random 2-coordinate moves targeting the global max peak
            if stall > 50:
                rng = np.random.default_rng((it * 48271 + 3) % (2**31))
                k_max = int(np.argmax(fast_convolve(a)))
                js = max(0, k_max - n + 1)
                je = min(n - 1, k_max)
                affected = np.arange(js, je + 1)
                for _ in range(20):
                    idx = rng.choice(len(affected))
                    j = affected[idx]
                    # Move mass from j to a random position outside affected
                    outside = np.setdiff1d(np.arange(n), affected)
                    if len(outside) == 0:
                        continue
                    j2 = outside[rng.choice(len(outside))]
                    delta = 0.005 * a[j]
                    if delta < 1e-15:
                        continue
                    candidate = a.copy()
                    candidate[j] -= delta
                    candidate[j2] += delta
                    candidate = np.maximum(candidate, 0.0)
                    candidate = candidate / candidate.sum()
                    new_c = fast_convolve(candidate)
                    if new_c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = new_c.max()
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_obj


def antisymmetry_injection(a0, time_limit):
    """
    Explicitly try to increase the antisymmetric component of a.
    This reduces the center peak of the auto-convolution.
    a = s + d where s is symmetric, d is antisymmetric.
    Center peak = sum(s^2) - sum(d^2), so increasing |d| helps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Compute symmetric and antisymmetric parts
        a_rev = a[::-1].copy()
        s_part = 0.5 * (a + a_rev)
        d_part = 0.5 * (a - a_rev)

        # For each j, try to increase |d_part[j]| by pushing towards boundary
        # a[j] = s_part[j] + d_part[j] >= 0 means |d_part[j]| <= s_part[j]
        rng = np.random.default_rng((it * 31 + 7) % (2**31))

        improved = False
        for _ in range(50):
            # Pick a random j and try to increase its antisymmetric component
            j = rng.choice(n)
            if j >= n - 1 - j:
                j = n - 1 - j  # Work with the "first half"

            partner = n - 1 - j
            # Currently: a[j] = s[j] + d[j], a[partner] = s[j] - d[j]
            # To increase |d[j]|: if d[j] > 0, increase a[j] and decrease a[partner]
            # if d[j] < 0, decrease a[j] and increase a[partner]
            # The max we can go is |d[j]| = s[j], i.e., one of them hits 0

            current_d = d_part[j]
            margin = s_part[j] - abs(current_d)
            if margin < 1e-10:
                continue

            # Try to push d by a fraction of the available margin
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                push = frac * margin
                if push < 1e-15:
                    continue
                if current_d >= 0:
                    candidate = a.copy()
                    candidate[j] += push
                    candidate[partner] -= push
                else:
                    candidate = a.copy()
                    candidate[j] -= push
                    candidate[partner] += push

                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs

                new_c = fast_convolve(candidate)
                if new_c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = new_c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Fall back to small random antisymmetric perturbations
            rng2 = np.random.default_rng((it * 6700417 + 13) % (2**31))
            for _ in range(10):
                j = rng2.choice(n // 2)
                partner = n - 1 - j
                push = rng2.uniform(-1, 1) * 0.001 * a.max()
                candidate = a.copy()
                candidate[j] += push
                candidate[partner] -= push
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                new_c = fast_convolve(candidate)
                if new_c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = new_c.max()
                    best_a = a.copy()
                    improved = True
                    break

            if not improved and it > 200:
                break

    return best_a, best_obj


def make_init(n, kind='edge', alpha=2.0, seed=42):
    """Generate various initializations."""
    x = np.arange(n, dtype=np.float64)
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'edge':
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-alpha * (d / n) ** 2)
        return a / a.sum()
    elif kind == 'edge_soft':
        d = np.minimum(x, n - 1 - x)
        a = 1.0 / (1.0 + alpha * (d / n) ** 2)
        return a / a.sum()
    elif kind == 'asym_edge':
        # Asymmetric edge concentration
        d_left = x
        d_right = n - 1 - x
        a = np.exp(-alpha * (d_left / n) ** 2) * 1.5 + np.exp(-alpha * (d_right / n) ** 2) * 0.5
        return a / a.sum()
    elif kind == 'twist':
        # Strongly asymmetric
        a = np.exp(-alpha * (x / n - 0.3) ** 2) + 0.5 * np.exp(-alpha * (x / n - 0.7) ** 2)
        return a / a.sum()
    elif kind == 'multi':
        n_modes = 3
        a = np.zeros(n)
        for m in range(n_modes):
            center = (m + 0.5) * n / n_modes
            a += np.exp(-2.0 * ((x - center) / n) ** 2)
        return a / a.sum()
    elif kind == 'asym_multi':
        a = np.zeros(n)
        for m in range(3):
            center = (m + 0.3) * n / 3
            a += np.exp(-3.0 * ((x - center) / n) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining:
    - Multi-n exploration with diverse initializations
    - Smoothed gradient descent (with and without Nesterov)
    - Peak-targeted 2-coordinate moves
    - Antisymmetric component optimization
    - Iterative alternating refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 50.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = make_init(768, 'edge', 2.0)
        try_seq(parent, "init_edge")

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration (~300s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_start = time.monotonic()

    n_candidates = sorted(set([
        n_parent, max(256, n_parent // 2), min(2048, n_parent * 2),
        384, 512, 640, 768, 896, 1024, 1280
    ]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 1536]

    init_kinds = ['edge', 'asym_edge', 'twist', 'edge_soft']

    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 900
        if remaining < 30:
            break

        starts = []
        if n_try == n_parent and parent is not None:
            starts.append(parent.copy())
        else:
            if parent is not None:
                interp = np.interp(np.linspace(0, 1, n_try),
                                   np.linspace(0, 1, n_parent), parent)
                starts.append(np.maximum(interp, 0) / max(np.maximum(interp, 0).sum(), 1e-15))
            for kind in init_kinds[:2]:
                starts.append(make_init(n_try, kind))

        for si, start in enumerate(starts):
            remaining = deadline - time.monotonic() - 800
            if remaining < 20:
                break
            t_alloc = min(remaining, 35)
            a_opt, _ = smoothed_grad_descent(start, time_limit=t_alloc,
                                             T_init=0.001, lr_init=0.03)
            try_seq(a_opt, f"p1_n{n_try}_s{si}")

    print(f"  Phase 1: best={best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Deep refinement of best (~500s) ===
    print("\n=== Phase 2: Deep Refinement ===")
    t_start = time.monotonic()

    if best_seq is not None:
        # Long smoothed gradient with fine temperature
        remaining = deadline - time.monotonic() - 400
        if remaining > 40:
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=min(remaining, 200),
                                             T_init=0.0005, lr_init=0.02, nesterov=True)
            try_seq(a_opt, "p2_smooth_nest")

        # Peak-targeted moves
        remaining = deadline - time.monotonic() - 250
        if remaining > 30:
            a_opt, _ = peak_targeted_moves(best_seq, time_limit=min(remaining, 120), n_peaks=5)
            try_seq(a_opt, "p2_peak")

        # Antisymmetric injection
        remaining = deadline - time.monotonic() - 150
        if remaining > 30:
            a_opt, _ = antisymmetry_injection(best_seq, time_limit=min(remaining, 80))
            try_seq(a_opt, "p2_antisym")

        # Another smoothing pass from the best of the above
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=min(remaining, 60),
                                             T_init=0.0002, lr_init=0.01)
            try_seq(a_opt, "p2_smooth2")

    print(f"  Phase 2: best={best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Alternating methods (~500s) ===
    print("\n=== Phase 3: Alternating Refinement ===")
    t_start = time.monotonic()

    if best_seq is not None:
        for rnd in range(12):
            remaining = deadline - time.monotonic() - 150
            if remaining < 30:
                break

            # Smooth gradient
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=min(remaining, 60),
                                             T_init=0.0003 * (1.0 + 0.1 * rnd),
                                             lr_init=0.015, nesterov=(rnd % 2 == 0))
            try_seq(a_opt, f"p3_smooth_{rnd}")

            # Peak moves
            remaining = deadline - time.monotonic() - 120
            if remaining < 15:
                break
            a_opt, _ = peak_targeted_moves(a_opt, time_limit=min(remaining, 40), n_peaks=3)
            try_seq(a_opt, f"p3_peak_{rnd}")

            # Antisymmetric injection
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            a_opt, _ = antisymmetry_injection(a_opt, time_limit=min(remaining, 25))
            try_seq(a_opt, f"p3_anti_{rnd}")

            # Perturb + refine
            remaining = deadline - time.monotonic() - 80
            if remaining < 15:
                break
            rng2 = np.random.default_rng((seed + rnd * 7919) % (2**31))
            n_best = len(best_seq)
            perturbed = project_simplex(best_seq + rng2.standard_normal(n_best) * 0.002)
            a_opt, _ = smoothed_grad_descent(perturbed, time_limit=min(remaining, 30),
                                             T_init=0.0005, lr_init=0.02)
            try_seq(a_opt, f"p3_perturb_{rnd}")

    print(f"  Phase 3: best={best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Try different n values from current best (~200s) ===
    print("\n=== Phase 4: n Variation ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best - 64, n_best + 64, n_best // 2, n_best * 2]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 30 or n_try < 128 or n_try > 2048:
                continue
            interp = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, n_best), best_seq)
            interp = np.maximum(interp, 0)
            interp = interp / interp.sum()
            a_opt, _ = smoothed_grad_descent(interp, time_limit=min(remaining, 50),
                                             T_init=0.001, lr_init=0.03)
            try_seq(a_opt, f"p4_n{n_try}")

    print(f"  Phase 4: best={best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final fine-tuning (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine temperature smoothing
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=min(remaining, 60),
                                             T_init=0.00005, lr_init=0.005, nesterov=True)
            try_seq(a_opt, "p5_fine")

        # Peak moves with many peaks
        remaining = deadline - time.monotonic() - 15
        if remaining > 15:
            a_opt, _ = peak_targeted_moves(best_seq, time_limit=min(remaining, 30), n_peaks=8)
            try_seq(a_opt, "p5_peak8")

        # Tiny random perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            for _ in range(4):
                remaining = deadline - time.monotonic() - 3
                if remaining < 5:
                    break
                rng3 = np.random.default_rng()
                perturbed = project_simplex(best_seq + rng3.standard_normal(n_best) * 0.0005)
                a_opt, _ = smoothed_grad_descent(perturbed, time_limit=min(remaining, 8),
                                                 T_init=0.0001, lr_init=0.003)
                try_seq(a_opt, "p5_tiny")

    print(f"  Phase 5: best={best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
