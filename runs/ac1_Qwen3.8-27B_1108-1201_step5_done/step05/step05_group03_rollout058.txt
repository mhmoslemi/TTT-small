def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def eval_h_var(t, f, i, conv, S, old):
    """
    Evaluate objective with f[i] changed from 'old' to 't'.
    Uses precomputed conv for efficiency.
    """
    n = len(f)
    S_new = S - old + t
    if S_new < 1e-15:
        return float('inf')

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    if k_lo > k_hi:
        max_conv = conv.max()
    else:
        k_range = np.arange(k_lo, k_hi + 1)

        if 2 * i >= k_lo and 2 * i <= k_hi:
            k_quad = 2 * i
            k_lin = k_range[k_range != k_quad]
            quad_val = conv[k_quad] + t * t - old * old
            if len(k_lin) > 0:
                f_part = f[k_lin - i]
                lin_vals = (conv[k_lin] - 2.0 * old * f_part) + 2.0 * t * f_part
                lin_max = lin_vals.max()
            else:
                lin_max = 0.0

            lin_max = lin_vals.max() if len(k_range) > 0 else 0.0

            other_max = conv[mask].max() if mask.any() else 0.0

def optimize_coord(f, i, conv, S):
    """
    Optimize coordinate i of f (given current conv and sum S).
    Returns optimal t >= 0 minimizing the objective.
    """
    n = len(f)
    old = f[i]

    # Determine search bounds
    t_max = max(5.0 * old, 20.0 / n, 0.01)

    # Coarse grid search to find good interval
    grid = np.linspace(0.0, t_max, 25)
    best_grid_val = float('inf')
    best_grid_idx = 0
    for idx, t in enumerate(grid):
        val = eval_h_var(t, f, i, conv, S, old)
        if val < best_grid_val:
            best_grid_val = val
            best_grid_idx = idx

    # Check if we should expand
    if best_grid_idx == len(grid) - 1:
        # Minimum at upper bound, expand
        t_max2 = t_max * 2
        grid2 = np.linspace(t_max, t_max2, 10)
        for t in grid2:
            val = eval_h_var(t, f, i, conv, S, old)
            if val < best_grid_val:
                best_grid_val = val
                best_grid_idx = len(grid) + int((t - t_max) / (t_max2 - t_max) * 9)
        t_max = t_max2

    # Narrow to interval around best grid point
    if best_grid_idx == 0:
        lo = 0.0
        hi = grid[1]
    elif best_grid_idx == len(grid) - 1:
        lo = grid[-2]
        hi = t_max
    else:
        lo = grid[best_grid_idx - 1]
        hi = grid[best_grid_idx + 1]

    # Golden section search in [lo, hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_h_var(c, f, i, conv, S, old)
    fd = eval_h_var(d, f, i, conv, S, old)

    for _ in range(45):
        if hi - lo < 1e-13 * max(1.0, t_max):
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_h_var(c, f, i, conv, S, old)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_h_var(d, f, i, conv, S, old)

    return (lo + hi) / 2.0


def update_conv(f, i, old, new, conv):
    """Update conv in-place after changing f[i] from old to new."""
    n = len(f)
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Remove old contribution
    if 2 * i <= 2 * n - 2:
        conv[2 * i] -= old * old
    if k_hi >= k_lo:
        k_range = np.arange(k_lo, k_hi + 1)
        mask = k_range != 2 * i
        if mask.any():
            k_lin = k_range[mask]
            conv[k_lin] -= 2.0 * old * f[k_lin - i]

    # Add new contribution
    f[i] = new
    if 2 * i <= 2 * n - 2:
        conv[2 * i] += new * new
    if k_hi >= k_lo:
        k_range = np.arange(k_lo, k_hi + 1)
        mask = k_range != 2 * i
        if mask.any():
            k_lin = k_range[mask]
            conv[k_lin] += 2.0 * new * f[k_lin - i]


def coord_descent(f0, time_limit, seed=42, verbose=False):
    """
    Coordinate descent with golden section search.
    Returns (best_f, best_score).
    """
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
        s = 1.0
    else:
        f = f / s
        s = 1.0
    n = len(f)

    conv = fast_convolve(f)
    obj = 2.0 * n * conv.max() / (s * s)
    best_f = f.copy()
    best_obj = obj

    t0 = time.monotonic()
    no_improve_passes = 0
    rng = np.random.default_rng(seed)
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False
        S = f.sum()

        # Choose ordering: alternate between sequential and random
        if pass_count % 3 == 1:
            order = np.arange(n)
        elif pass_count % 3 == 2:
            order = rng.permutation(n)
        else:
            # Gauss-Southwell approximation: prioritize coords near peak
            peak_k = int(np.argmax(conv))
            # Coords that contribute to peak: j and peak_k - j
            priority = np.zeros(n)
            for j in range(n):
                k1 = peak_k - j
                if 0 <= k1 < n:
                    priority[j] += conv[peak_k] * f[j] * f[k1]
            # Add random component
            priority += rng.random(n) * 0.1
            order = np.argsort(priority)[::-1]

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            t_new = optimize_coord(f, i, conv, S)

            # Clamp to non-negative
            t_new = max(0.0, t_new)

            if abs(t_new - f[i]) > 1e-14:
                old = f[i]
                S_new = S - old + t_new

                # Update conv in place
                update_conv(f, i, old, t_new, conv)
                S = S_new

                new_obj = 2.0 * n * conv.max() / (S * S)
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                # Perturb to escape local minimum
                scale = 0.002 * min(no_improve_passes, 10)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                S = f.sum()
                conv = fast_convolve(f)
                no_improve_passes = 0

    return best_f, best_obj


def subgradient_adam(f0, time_limit, seed=123, verbose=False):
    """
    Adam optimizer on smoothed objective with simplex projection.
    Used for diversification.
    """
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.003

    T_start = 3.0 / n
    T_end = 0.3 / n
    no_improve = 0
    it = 0
    max_iters = int(time_limit / 0.01)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max(1, int(max_iters * 0.9)))
        T = T_start * (T_end / T_start) ** progress
        lr = base_lr * (1.0 - 0.4 * progress)

        conv = fast_convolve(f)
        k_actual = min(max(20, n // 4), 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[ii] * f[k - idx]

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 37 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                continue
            continue

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_step(f0, time_limit, seed=99):
    """One phase of LP-based Frank-Wolfe."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

        f = f / s

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                no_improve += 1
                if no_improve > 50:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Coarse line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Coordinate descent on parent (main workhorse) ===
    print(f"\n=== Phase 1: Coord Descent on parent (n={n_parent}) ===")
    rem = max(0, deadline - time.monotonic() - 600)
    t1 = min(rem, 400)
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")
        a_opt, _ = coord_descent(parent, time_limit=t1, seed=42, verbose=True)
        try_seq(a_opt, "cd_parent")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [384, 512, 768, 1024, 1280, 1536]
    if n_parent in target_lengths:
        target_lengths.remove(n_parent)

    for target_n in target_lengths:
        rem = max(0, deadline - time.monotonic() - 400)
        if rem < 50:
            break
        f_interp = interpolate_seq(base, target_n)

        # Quick coord descent
        t_alloc = min(rem * 0.3, 150)
        if t_alloc > 20:
            print(f"  Coord descent n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = coord_descent(f_interp, time_limit=t_alloc, seed=100 + target_n)
            try_seq(a_opt, f"cd_n{target_n}")

    # === Phase 3: Subgradient diversification ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = max(0, deadline - time.monotonic() - 300)
        if rem > 30:
            t_sg = min(rem * 0.5, 120)
            print(f"  Adam-SG t={t_sg:.0f}s, n={len(best_seq)}")
            a_opt, _ = subgradient_adam(best_seq, time_limit=t_sg, seed=777)
            try_seq(a_opt, "adam_div")

            # Coord descent after subgradient
            rem = max(0, deadline - time.monotonic() - 250)
            if rem > 30:
                t_cd = min(rem, 100)
                a_opt, _ = coord_descent(a_opt, time_limit=t_cd, seed=888)
                try_seq(a_opt, "cd_after_adam")

    # === Phase 4: LP-FW on best ===
    print(f"\n=== Phase 4: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = max(0, deadline - time.monotonic() - 200)
        if rem > 30:
            t_lpfw = min(rem * 0.6, 120)
            print(f"  LP-FW t={t_lpfw:.0f}s, n={len(best_seq)}")
            a_opt, _ = lp_fw_step(best_seq, time_limit=t_lpfw, seed=99)
            try_seq(a_opt, "lpfw")

            # Coord descent after LP-FW
            rem = max(0, deadline - time.monotonic() - 150)
            if rem > 20:
                a_opt, _ = coord_descent(a_opt, time_limit=min(rem, 80), seed=7777)
                try_seq(a_opt, "cd_after_lpfw")

    # === Phase 5: Diverse initializations ===
    print(f"\n=== Phase 5: Diverse inits (best={best_score:.8f}) ===")
    n_best = len(best_seq) if best_seq is not None else 768

    diverse_inits = [
        ("chirp", make_chirp_init(n_best, 0)),
        ("windowed", make_windowed_init(n_best, 0)),
        ("uniform", np.ones(n_best) / n_best),
    ]

    # Add perturbed versions of best
    rng = np.random.default_rng(42)
    diverse_inits.append(("pert1", project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)))
    diverse_inits.append(("pert2", project_simplex(best_seq + rng.standard_normal(n_best) * 0.02)))
    diverse_inits.append(("reversed", best_seq[::-1].copy() / best_seq.sum()))

    for name, f_init in diverse_inits:
        rem = max(0, deadline - time.monotonic() - 100)
        if rem < 20:
            break
        t_alloc = min(rem * 0.25, 60)
        if t_alloc < 10:
            break
        print(f"  {name}: n={len(f_init)}, t={t_alloc:.0f}s")
        a_opt, _ = coord_descent(f_init, time_limit=t_alloc, seed=1000 + hash(name) % 100)
        try_seq(a_opt, f"div_{name}")

    # === Phase 6: Intensive coordinate descent on best ===
    print(f"\n=== Phase 6: Intensive CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = max(0, deadline - time.monotonic() - 50)
        if rem > 20:
            t_cd = min(rem, 200)
            print(f"  Final CD t={t_cd:.0f}s, n={len(best_seq)}")
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd, seed=55555)
            try_seq(a_opt, "cd_final")

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Micro-polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = max(0, deadline - time.monotonic() - 10)
        if rem > 10:
            t_micro = min(rem, 30)
            a_opt, _ = coord_descent(best_seq, time_limit=t_micro, seed=99999)
            try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
