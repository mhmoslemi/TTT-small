import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective_and_grad(a, beta):
    """Compute smooth approximation to max(conv) and its gradient."""
    conv = fast_convolve(a)
    m = conv.max()
    shifted = beta * (conv - m)
    # Log-sum-exp
    lse = m + np.log(np.sum(np.exp(shifted))) / beta
    # Weights
    w = np.exp(shifted)
    w /= w.sum()
    # Gradient: dg/da_i = 2 * sum_k w_k * a_{k-i}
    # s_i = sum_k w_k * a_{k-i} = correlate(w, a, 'valid')[i]
    s = np.correlate(w, a, mode='valid')
    grad = 2.0 * s
    return lse, grad, conv

Actually, for the projected gradient method, I can use a simpler approach:
1. Compute the gradient.
2. Use a fixed or slowly decaying step size.
3. Don't do full backtracking; instead, just check if the objective improved, and if not, reduce the step.

Or even better, I can use the **FISTA** (Fast Iterative Shrinkage-Thresholding Algorithm) approach, which is an accelerated projected gradient method. It converges at rate O(1/k^2) instead of O(1/k).

FISTA for our problem:
```
y_k = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1})
x_{k+1} = P_simplex(y_k - alpha_k * grad(f(y_k)))
t_{k+1} = (1 + sqrt(1 + 4*t_k^2))/2
```

where `f` is the smooth objective and `P_simplex` is the projection onto the simplex.

The step size `alpha_k` should be `1/L` where `L` is the Lipschitz constant of the gradient. For our problem, `L` depends on beta and the values of a. A safe choice is `alpha = 1/(2 * beta * max_a)` or something like that. Actually, the gradient of the log-sum-exp is a convex combination of the gradients of the individual `conv_k`, and each `grad(conv_k)` has norm at most `2 * max(a)`. So `L ≈ 2 * beta * max(a)` (roughly).

**Another insight: The "flat" constraint.**

If we require `conv_k <= M` for all k, and `sum(a) = 1`, then:

The question is: what's the minimum M achievable with non-negative a?

This is equivalent to: minimize M such that there exists `a >= 0, sum(a) = 1` with `sum_j a_j * a_{k-j} <= M` for all k.

This is a second-order cone program (SOCP)! We can write:
- Variables: a_0, ..., a_{n-1}, M
- Constraints: `sum_j a_j * a_{k-j} <= M` for k = 0, ..., 2n-2
- `sum(a) = 1`
- `a >= 0`

The constraints `sum_j a_j * a_{k-j} <= M` are quadratic (non-convex in general because of the `a_j * a_{k-j}` terms). Wait, actually for k=0: `sum_j a_j^2 <= M` which is convex (it's the squared norm). For k>0: `sum_j a_j * a_{k-j} <= M` which is NOT convex in general (it's a bilinear term).

So it's not an SOCP. It's a non-convex quadratic constraint problem (QCQP).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient optimization (FISTA with log-sum-exp annealing)
2. Exact coordinate bisection polishing
3. Multi-length and multi-initialization search
4. Pair-swap based local refinement
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_smooth_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*conv_k)))."""
    conv = fast_convolve(a)
    m = conv.max()
    shifted = beta * (conv - m)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # grad_i = 2 * sum_k w_k * a_{k-i}
    s = np.correlate(w, a, mode='valid')
    grad = 2.0 * s
    return grad, conv, w


def fista_optimize(a0, time_limit, beta_start=20.0, n_anneal=15, verbose=False):
    """
    FISTA (accelerated projected gradient) with log-sum-exp smoothing and annealing.
    """
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    # Create annealing schedule for beta
    # For n=768, the conv values are around 1/n ≈ 0.0013
    # We want beta * conv >> 1 for the log-sum-exp to approximate max
    # So beta should be >> n, say beta in [50, 10000]
    
    betas = np.geomspace(beta_start, beta_start * n * 100, n_anneal)
    
    for ai in range(n_anneal):
        beta = betas[ai]
        epoch_end = t0 + time_limit * (ai + 1) / n_anneal
        if time.monotonic() >= epoch_end - 2:
            continue
        
        # FISTA parameters
        x = a.copy()
        y = a.copy()
        t_fista = 1.0
        # Step size: L is the Lipschitz constant of the gradient
        # grad(f) has components 2*w_k*a_{k-i}, so |grad| <= 2*max(a)
        # But the gradient of the LOG-SUM-EXP is a weighted average, so |grad| <= 2*max(a)
        # Actually, the Lipschitz constant of the log-sum-exp gradient is harder to bound.
        # Let's use a heuristic: lr = 1/(2*beta*max(a)*n^0.5) or just try and adjust.
        max_a = max(a.max(), 1e-10)
        lr = 1.0 / (2.0 * max_a * np.sqrt(n))
        
        it = 0
        while time.monotonic() < epoch_end:
            it += 1
            
            grad_y, conv_y, _ = compute_smooth_grad(y, beta)
            x_new = project_simplex(y - lr * grad_y)
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            y_new = x_new + (t_fista - 1.0) / t_new * (x_new - x)
            y_new = np.maximum(y_new, 0.0)
            # We don't strictly need to project y, but keep it in a reasonable range
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = x_new.copy()
            # Don't normalize y (FISTA theory requires it on the feasible set,
            # but we're using a heuristic anyway)
            
            x = x_new
            y = y_new
            t_fista = t_new
            
            # Periodically check the actual objective
            if it % 10 == 0:
                obj = fast_convolve(x).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = x.copy()
                    if verbose and ai % 3 == 0:
                        print(f"    FISTA ai={ai}, it={it}, obj={best_obj:.10f}")
        
        # End of epoch: update a
        obj = fast_convolve(x).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = x.copy()
        a = x.copy()
    
    return best_a, best_obj


def coordinate_bisection(a0, time_limit, verbose=False):
    """Exact coordinate descent with bisection per coordinate."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < 200:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-15:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            # b = a with a_j set to 0, normalized
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            # Bisection on x in [0, 1]
            lo = 0.0
            hi = min(1.0, x_old * 3.0)  # restrict search range
            x_new = x_old
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                k_max = int(np.argmax(f_vals))
                
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-14:
                    hi = mid
                elif deriv < -1e-14:
                    lo = mid
                else:
                    x_new = mid
                    break
            else:
                x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-15:
                continue
            
            # Apply update
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if passes > 1:
                # Try a small random perturbation
                rng = np.random.default_rng(passes * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break
    
    return best_a, best_obj


def pair_refine(a0, time_limit, n_pairs_per_iter=50, verbose=False):
    """Try to improve by adjusting pairs of coordinates."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        # Identify top peaks
        top_k = min(20, 2*n-1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # For each peak, identify the contributing pairs
        for peak in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            # The peak at position p: conv_p = sum_j a_j * a_{p-j}
            # We want to reduce this. One way: reduce a_j for j where a_{p-j} is large.
            # Try: for a random j in [0, n-1], compute the "contribution" a_j * a_{p-j}
            # and try to reduce it by transferring mass.
            
            j = rng.integers(0, n)
            k = peak - j
            if k < 0 or k >= n:
                continue
            if j == k:
                continue  # can't transfer to self easily
            
            xj = a[j]
            xk = a[k]
            if xj < 1e-12 or xk < 1e-12:
                continue
            
            # Try transferring a small amount from the larger to the smaller
            # (this reduces the product xj*xk)
            if xj > xk:
                source, dest = j, k
            else:
                source, dest = k, j
            
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = frac * a[source]
                a_trial = a.copy()
                a_trial[source] -= delta
                a_trial[dest] += delta
                a_trial = normalize(a_trial)
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break
    
    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def make_bump(n, width=0.3):
    """Create a bump (Gaussian-like) function of length n."""
    x = np.linspace(0, 1, n)
    bump = np.exp(-((x - 0.5) / width) ** 2)
    return normalize(bump)


def make_tent(n):
    """Create a tent (triangular) function."""
    x = np.linspace(0, 1, n)
    tent = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(tent)


def make_u_shape(n, power=0.5):
    """Create a U-shaped function (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    u = x ** power + (1 - x) ** power
    return normalize(u)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smooth projected gradient with log-sum-exp annealing (FISTA-accelerated)
- Exact coordinate bisection for high-precision polishing
- Multi-length, multi-initialization search with adaptive time allocation
- Peak-directed pair refinement
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_smooth_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*conv_k))) w.r.t. a.
    Returns (grad, conv, weights).
    """
    conv = fast_convolve(a)
    m = conv.max()
    shifted = beta * (conv - m)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    # grad_i = 2 * sum_k w_k * a_{k-i} (with zero padding)
    s = np.correlate(w, a, mode='valid')
    grad = 2.0 * s
    return grad, conv, w


def fista_optimize(a0, time_limit, beta_start=30.0, n_anneal=12):
    """
    FISTA (accelerated projected gradient) on the log-sum-exp smoothed objective
    with geometric beta annealing.
    """
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    beta_end = max(beta_start * n * 200.0, 5000.0)
    betas = np.geomspace(beta_start, beta_end, n_anneal)

    for ai in range(n_anneal):
        beta = float(betas[ai])
        epoch_end = t0 + time_limit * (ai + 1) / n_anneal
        if time.monotonic() >= epoch_end - 1:
            continue

        x = a.copy()
        y = a.copy()
        t_fista = 1.0

        # Heuristic step size: gradient magnitude is ~2*max(a), so L ~ 2*max(a)*sqrt(n)
        max_a = max(float(np.max(a)), 1e-8)
        lr = 0.8 / (max_a * np.sqrt(n))

        it = 0
        no_improve = 0
        while time.monotonic() < epoch_end:
            it += 1
            grad_y, _, _ = compute_smooth_grad(y, beta)
            x_new = project_simplex(y - lr * grad_y)

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            y_new = x_new + ((t_fista - 1.0) / t_new) * (x_new - x)
            y_new = np.maximum(y_new, 0.0)
            y_sum = y_new.sum()
            if y_sum < 1e-15:
                y_new = x_new.copy()

            x = x_new
            y = y_new
            t_fista = t_new

            # Periodic check of true objective
            if it % 8 == 0:
                obj = fast_convolve(x).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = x.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve > 50:
                    # Reset FISTA state
                    y = x.copy()
                    t_fista = 1.0
                    no_improve = 0

        # End of epoch
        obj = fast_convolve(x).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = x.copy()
        a = x.copy()

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.02, top_k=40, temp=0.001):
    """
    Multi-peak subgradient descent with momentum.
    """
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    v = np.zeros(n)
    mom = 0.9
    cur_lr = lr_init
    cur_temp = temp
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[int(np.argmax(pv))] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.002

        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            noise = 0.003 + 0.001 * min(no_improve // 200, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            cur_lr = lr_init * 0.7
            no_improve = 0

    return best_a, best_obj


def coordinate_bisection(a0, time_limit):
    """Exact coordinate descent with bisection per coordinate."""
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < 500:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-15:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = min(1.0, max(x_old * 2.5, 0.1))
            x_new = x_old

            for _ in range(42):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-14:
                    hi = mid
                elif deriv < -1e-14:
                    lo = mid
                else:
                    x_new = mid
                    break
            else:
                x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 2:
                rng = np.random.default_rng(passes * 43 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0004)
            else:
                break

    return best_a, best_obj


def peak_directed_refine(a0, time_limit):
    """
    Directly target the highest peaks: for each peak, find the pair (j, k-j)
    contributing most and try to rebalance.
    """
    a = normalize(np.asarray(a0, dtype=np.font64).copy())
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        top_k = min(15, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]

        for peak in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            # Find top contributing pairs for this peak
            peak_val = int(peak)
            js = max(0, peak_val - n + 1)
            je = min(n - 1, peak_val)
            if js > je:
                continue

            # Contributions: a[j] * a[peak - j] for j in [js, je]
            indices = np.arange(js, je + 1)
            contribs = a[indices] * a[peak_val - indices]
            # Find the largest contribution
            top_idx = indices[np.argmax(contribs)]
            partner = peak_val - top_idx

            if top_idx == partner:
                continue  # Can't easily split a self-pair

            # Try transferring mass from the larger to the smaller
            if a[top_idx] > a[partner]:
                src, dst = top_idx, partner
            else:
                src, dst = partner, top_idx

            improved = False
            for frac in [0.15, 0.25, 0.4, 0.6]:
                delta = frac * a[src]
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[dst] += delta
                a_trial = normalize(a_trial)
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

            if not improved:
                continue

    return best_a, best_obj


def symmetrize(a):
    """Make symmetric and renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return normalize(a_sym)


def interpolate_to_length(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def make_gauss_init(n, width):
    x = np.linspace(0, 1, n)
    return normalize(np.exp(-((x - 0.5) / width) ** 2))


def make_tent_init(n):
    x = np.linspace(0, 1, n)
    return normalize(1.0 - np.abs(2.0 * x - 1.0))


def make_flat_edges_init(n, p=0.5):
    x = np.linspace(0, 1, n)
    return normalize(x**p + (1.0 - x)**p)
