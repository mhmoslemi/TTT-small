```python
import numpy as np
from scipy.optimize import linprog
import time

def solve_lp_cross_corr(a):
    """Given fixed a (sum=1, a>=0), find b that minimizes max_k (a*b)[k]
    subject to sum(b)=1, b>=0."""
    n = len(a)
    # conv_len = 2*n - 1
    # variables: b[0], ..., b[n-1], t
    # x = [b_0, ..., b_{n-1}, t]
    
    # Objective: minimize t => c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (a*b)[k] - t <= 0 for each k
    # A_ub @ x <= b_ub
    # A_ub has shape (2n-1) x (n+1)
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = a[j]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # b[i] >= 0, t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        b_opt = res.x[:n]
        t_opt = res.x[n]
        return b_opt, t_opt
    return None, None
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP-based
optimization (Sinkhorn-like iteration), coordinate descent polishing,
and multi-start exploration with asymmetric initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def solve_lp_given_a(a):
    """Given fixed a (sum=1, a>=0), find b minimizing max_k (a*b)[k]
    subject to sum(b)=1, b>=0."""
    n = len(a)
    conv_len = 2 * n - 1
    
    # Build A_ub sparsely: row k, col i gets a[k-i] if valid, col n gets -1
    # For efficiency with moderate n, use dense
    A_ub = np.zeros((conv_len, n + 1))
    for k in range(conv_len):
        i_start = max(0, k - n + 1)
        i_end = min(k + 1, n)
        for i in range(i_start, i_end):
            A_ub[k, i] = a[k - i]
        A_ub[k, n] = -1.0
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    b_ub = np.zeros(conv_len)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, float('inf')

def lp_iteration(a0, time_limit, alpha=0.5, verbose=False):
    """Alternating LP optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_opt, t_lp = solve_lp_given_a(a)
        if b_opt is None:
            break
        
        # Damped update
        a_new = (1 - alpha) * a + alpha * b_opt
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = b_opt.copy()
        else:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.10f}, t_lp={t_lp:.10f}, elapsed={elapsed:.1f}s")
        
        # Convergence check
        if np.max(np.abs(a - best_a)) < 1e-10:
            break
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with exact 1D optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < 200:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            # Golden section or bisection on x_j
            # Fix all other coordinates, optimize x_j
            # a[j] = x, others scaled to maintain sum=1
            # (a*a)[k] = sum_i a[i]*a[k-i]
            # = a[j]*a[k-j] + a[k-j]*a[j] + (other terms involving a[j])
            # = 2*a[j]*a[k-j] - a[j]^2*(delta_{k,2j}) + (terms not involving a[j])... 
            # Actually let me be more careful.
            
            # Let b = a with a[j] removed, normalized: b[i] = a[i]/(1-a[j]) for i != j
            # Then a = (1-x)*b + x*e_j where e_j is the unit vector
            # (a*a)[k] = (1-x)^2 * (b*b)[k] + 2*x*(1-x)*b[k-j] + x^2*delta_{k,2j}
            # where b[k-j] = b[k-j] if 0 <= k-j < n else 0, with b as vector
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a / s_others  # temporary, will fix a[j]
            b_j = b[j]  # this is x_old / s_others = 1 actually
            # Wait, I need to be more careful
            # a[i] for i != j stays proportional. Let me redefine.
            # a_new[i] = a[i] for i != j (scaled), a_new[j] = x
            # To maintain sum = 1: a_new[i] = a[i] * (1-x)/(1-x_old) for i != j
            
            # Let c[i] = a[i] / (1-x_old) for i != j, c[j] = 0
            # Then a_new = (1-x) * c + x * e_j... no
            # a_new[i] = (1-x) * c[i] for i != j, a_new[j] = x
            # where c[i] = a[i] / (1-x_old) for i != j
            
            c = np.zeros(n)
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            c[mask] = a[mask] / s_others
            
            # B = c * c (convolution)
            B = fast_convolve(c)
            # b_valid[k] = c[k-j] if 0 <= k-j < n else 0
            b_valid = np.zeros(2*n-1)
            start = j
            end = min(j+n, 2*n-1)
            b_valid[start:end] = c[:end-start]
            
            # f(x) = max_k [(1-x)^2 * B[k] + 2*x*(1-x)*b_valid[k] + x^2 * delta_{k,2j}]
            # We want to minimize f(x) over x in [0, 1]
            # Use golden section search or ternary search
            
            # Actually, this is the max of quadratic functions, so it's convex in x.
            # We can use binary search on the derivative or golden section.
            
            # Let's use a simple approach: sample x at many points and refine
            # Or use the fact that it's convex and use ternary search
            
            lo, hi = 0.0, 1.0
            
            def f_of_x(x):
                om = 1.0 - x
                vals = om * om * B + 2.0 * x * om * b_valid
                if 2 * j < 2 * n - 1:
                    vals[2 * j] += x * x
                return vals.max()
            
            # Ternary search (convex function)
            for _ in range(60):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if f_of_x(m1) < f_of_x(m2):
                    hi = m2
                else:
                    lo = m1
            
            x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-12:
                continue
            
            # Apply update
            a[mask] = c * (1.0 - x_new)
            a[j] = x_new
            a = np.maximum(a, 0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

For any `a`, `(a*a)[0] = a[0]^2` and `(a*a)[n-1] = sum a[i]*a[n-1-i]`. Also `(a*a)[n-1] >= (sum a[i])^2 / n = 1/n` by Cauchy-Schwarz... wait no, that's `sum a[i]^2 >= 1/n`, not `sum a[i]*a[n-1-i]`.

Actually, `sum a[i]*a[n-1-i]` can be less than 1/n! For example, if `a` is concentrated on the first half, then `a[i]` and `a[n-1-i]` are on opposite sides and their product is small.

But then `(a*a)[k]` for small k (within the first half) will be large.

The fundamental trade-off: the sum of all convolution values is `(sum a)^2 = 1`. There are 2n-1 values. So the average is `1/(2n-1)`. The minimum possible maximum is at least `1/(2n-1)` (by the pigeonhole principle). So the objective is at least `2n * 1/(2n-1) ≈ 1` for large n.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP-based
optimization (Sinkhorn-like iteration), coordinate descent polishing,
and multi-start exploration with asymmetric initializations.

Key insight: The problem is non-convex (max of bilinear terms), but we can
use alternating minimization: fix one copy of a, solve LP for the other,
iterate. This is analogous to Sinkhorn's algorithm for matrix scaling.

Another key insight: Symmetric sequences have objective >= 2 (by Cauchy-Schwarz
on the center term), so we must use asymmetric sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_lp_matrix(a):
    """Build sparse A_ub for the LP: A_ub[k, i] = a[k-i] if valid, A_ub[k, n] = -1"""
    n = len(a)
    conv_len = 2 * n - 1
    
    rows = []
    cols = []
    vals = []
    
    for k in range(conv_len):
        i_start = max(0, k - n + 1)
        i_end = min(k + 1, n)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(conv_len, n + 1))
    return A_ub

def solve_lp_given_a(a):
    """Given fixed a (sum=1, a>=0), find b minimizing max_k (a*b)[k]
    subject to sum(b)=1, b>=0."""
    n = len(a)
    conv_len = 2 * n - 1
    
    A_ub = build_lp_matrix(a)
    b_ub = np.zeros(conv_len)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        b_opt = np.maximum(res.x[:n], 0)
        b_opt = b_opt / b_opt.sum()  # renormalize for numerical safety
        return b_opt, res.x[n]
    return None, float('inf')

    else:
        a = a / s
    n = len(a)

    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

        it += 1
        b_opt, t_lp = solve_lp_given_a(a)
        if b_opt is None:

        a_new = np.maximum(a_new, 0)

            print(f"    LP iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

def coordinate_descent(a0, time_limit, verbose=False):

        passes += 1
        improved = False

            x_old = a[j]
            if x_old < 1e-14:

            s_others = 1.0 - x_old
            if s_others < 1e-12:

            # c[i] = a[i]/s_others for i != j, c[j] = 0
            c = np.zeros(n)
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            c[mask] = a[mask] / s_others

            b_valid = np.zeros(2*n-1)
            start = j
            end = min(j+n, 2*n-1)
            if end > start:
                b_valid[start:end] = c[:end-start]
            
            # Ternary search for optimal x
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0

                if 2*j < 2*n-1:

def make_geometric_init(n, ratio=0.95):
    """Create a geometric (exponentially decaying) initialization."""
    x = np.arange(n)
    a = ratio ** x
    a = a / a.sum()
    return a

def make_random_init(n, seed=0):
    """Create a random asymmetric initialization."""
    rng = np.random.default_rng(seed)
    # Exponential distribution (asymmetric, concentrated on one side)
    a = rng.exponential(1.0, size=n)
    a = a / a.sum()
    return a

def make_bimodal_init(n, seed=0):
    """Create a bimodal initialization (mass at both ends)."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    n1 = n // 3
    n2 = n // 3
    a[:n1] = rng.dirichlet(np.ones(n1))
    a[-n2:] = rng.dirichlet(np.ones(n2))
    # Small mass in the middle
    a[n1:n-n2] = 0.1 * np.ones(n - n1 - n2)
    a = a / a.sum()
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP-based optimization
(Sinkhorn-like iteration), coordinate descent polishing, and multi-start
exploration with asymmetric initializations.

Key insights:
1. Symmetric sequences have objective >= 2 (by Cauchy-Schwarz on center term),
   so we MUST use asymmetric sequences.
2. The problem is non-convex (max of bilinear terms), but alternating LP
   minimization (fix one copy, solve LP for the other) provides exact subproblem
   solutions and drives the objective down.
3. The theoretical lower bound is ~1 (all convolution values equal), giving
   significant room below the current 1.505.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_lp_sparse(a):
    """Build sparse A_ub matrix for the LP.
    A_ub[k, i] = a[k-i] if 0<=k-i<n, else 0; A_ub[k, n] = -1
    """
    n = len(a)
    conv_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        i_start = max(0, k - n + 1)
        i_end = min(k + 1, n)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(conv_len, n + 1))
    return A_ub


def solve_lp_given_a(a):
    """Given fixed a (sum=1, a>=0), find b minimizing max_k (a*b)[k]
    subject to sum(b)=1, b>=0. Returns (b_opt, t_opt)."""
    n = len(a)
    conv_len = 2 * n - 1
    A_ub = build_lp_sparse(a)
    b_ub = np.zeros(conv_len)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            b_opt = np.maximum(res.x[:n], 0.0)
            s = b_opt.sum()
            if s < 1e-15:
                return None, float('inf')
            b_opt = b_opt / s
            return b_opt, float(res.x[n])
    except Exception:
        pass
    return None, float('inf')


def lp_iteration(a0, time_limit, alpha=0.5, verbose=False):
    """Alternating LP optimization: fix a, solve LP for b, update a."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_opt, t_lp = solve_lp_given_a(a)
        if b_opt is None:
            break
        # Damped update
        a_new = (1.0 - alpha) * a + alpha * b_opt
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = b_opt.copy()
        else:
            a_new = a_new / s_new
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")
        # Check convergence
        if it > 10 and np.max(np.abs(a - best_a)) < 5e-9:
            break
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent: for each coordinate j, find optimal value via ternary search."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    while time.monotonic() - t0 < time_limit and passes < 100:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            x_old = a[j]
            if x_old < 1e-14:
                continue
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            # Build c (a with j removed, normalized)
            c = np.zeros(n)
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            c[mask] = a[mask] / s_others
            B = fast_convolve(c)
            # b_valid[k] = c[k-j] if 0 <= k-j < n
            b_valid = np.zeros(2 * n - 1)
            start = j
            end = min(j + n, 2 * n - 1)
            if end > start:
                b_valid[start:end] = c[:end - start]
            # Ternary search on x in [0,1]
            lo = 0.0
            hi = 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                vals1 = (1.0 - m1) ** 2 * B + 2.0 * m1 * (1.0 - m1) * b_valid
                if 2 * j < 2 * n - 1:
                    vals1[2 * j] += m1 * m1
                f1 = vals1.max()
                vals2 = (1.0 - m2) ** 2 * B + 2.0 * m2 * (1.0 - m2) * b_valid
                if 2 * j < 2 * n - 1:
                    vals2[2 * j] += m2 * m2
                f2 = vals2.max()
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_new = (lo + hi) / 2.0
            if abs(x_new - x_old) < 1e-12:
                continue
            a[mask] = c * (1.0 - x_new)
            a[j] = x_new
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                break
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                improved = True
        if not improved:
            break
        if verbose:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")
    return best_a, best_obj


def make_geometric_init(n, ratio=0.95):
    """Exponentially decaying (asymmetric) initialization."""
    x = np.arange(n, dtype=np.float64)
    a = ratio ** x
    return a / a.sum()


def make_random_init(n, seed=0):
    """Random exponential-distributed initialization."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, size=n)
    return a / a.sum()


def make_bimodal_init(n, seed=0):
    """Bimodal: mass concentrated at both ends."""
    rng = np.random.default_rng(seed)
    n1 = n // 3
    n2 = n // 3
    a = np.zeros(n)
    a[:n1] = rng.dirichlet(np.ones(n1))
    a[-n2:] = rng.dirichlet(np.ones(n2))
    mid_len = n - n1 - n2
    if mid_len > 0:
        a[n1:n - n2] = 0.05 * np.ones(mid_len)
    return a / a.sum()


def make_shifted_init(n, seed=0):
    """Shifted Gaussian (asymmetric bump)."""
    rng = np.random.default_rng(seed)
    center = rng.uniform(0.1, 0.4) * n
    width = rng.uniform(0.1, 0.25) * n
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    return a / a.sum()

            f_pert = best_a + rng.standard_normal(n_best) * noise_scale
            f_pert = np.maximum(f_pert, 0.0)
            s = f_pert.sum()
            if s > 1e-15:
                f_pert = f_pert / s
            t_alloc = min(rem * 0.4, 60)
            a_opt, _ = combined_optimize(f_pert, time_limit=t_alloc, verbose=False)
            upd(a_opt, f"restart_{p}")

    # === Phase 6: Final intensive CD polish ===
    if best_a is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem, 100)
            print(f"\n=== Phase 6: Final CD (t={t_alloc:.0f}s) ===")
            a_opt, _ = coordinate_descent(best_a, time_limit=t_alloc, verbose=True)
            upd(a_opt, "final_cd")

    # Fallback
    if best_a is None or state[0] is None:
        best_a = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Use state tracking
    if state[0] is not None:
        best_a = state[0]
        best_score = state[1]

    # Final verification
    try:
        fs = float(evaluate_sequence(best_a.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
