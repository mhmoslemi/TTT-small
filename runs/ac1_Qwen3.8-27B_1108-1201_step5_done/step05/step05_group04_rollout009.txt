"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum x=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(f, w, max_peaks=200):
    """Compute gradient of LSE objective w.r.t. f.
    grad[m] = 2 * sum_k w[k] * f[k-m] for valid indices.
    Uses only top max_peaks entries of w for efficiency.
    """
    n = len(f)
    conv_len = 2 * n - 1
    grad = np.zeros(n)

    if w.max() < 1e-30:
        return grad

    # Select significant peaks
    if conv_len <= max_peaks:
        sig_idx = np.where(w > 1e-15 * w.max())[0]
    else:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        sig_idx = sig_idx[w[sig_idx] > 1e-12 * w.max()]

    for k in sig_idx:
        wk = w[k]
        if wk < 1e-15:
            continue
        k = int(k)
        # grad[m] += 2 * wk * f[k-m] for m where 0 <= k-m <= n-1
        # i.e., m in [max(0, k-(n-1)), min(n-1, k)]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            ms = np.arange(m_lo, m_hi + 1)
            grad[ms] += 2.0 * wk * f[k - ms]

    return grad


def nesterov_lse(f0, time_limit, beta_start=30.0, beta_end=20000.0,
                 max_peaks=300, lr_init=None, use_nesterov=True):
    """Nesterov-accelerated LSE gradient descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr = 4.0 / n
    else:
        lr = lr_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    y = f.copy()  # Nesterov's lookahead variable
    v = np.zeros(n)
    momentum = 0.9
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    step_size_decay = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        if use_nesterov:
            gamma = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
            prev_gamma = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) * (it - 1))) / 2.0 if it > 1 else 1.0
            y = (gamma * f - (gamma - 1.0) * f) / prev_gamma + (prev_gamma - 1.0) / prev_gamma * f
            # Simplified: y = f + (it/(it+1)) * (f - prev_f) -- use momentum form
            y = f + momentum * v

        if s_eval < 1e-15:

        f_eval = f_eval / s_eval

        conv_e = fast_convolve(f_eval)
        bc = beta * conv_e
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient
        grad = compute_lse_grad(f_eval, w, max_peaks=max_peaks)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Step
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if no_improve == 0 and it > 5:
                lr = min(lr * 1.001, 10.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.95
            if no_improve > 100:

                v = np.zeros(n)
                lr = max(3.0 / n, 0.002)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A_cache = build_lp_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 30})
            if res.success and res.x is not None:
                g = res.x
            else:
                continue
        except Exception:
            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.005 * min(1.0, (no_improve - 40) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def block_cd_polish(f0, time_limit, block_size=8, n_iter_max=5000):
    """Block coordinate descent: optimize small blocks of consecutive coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        improved = False
        # Cycle through blocks
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            idx = np.arange(start, end)
            # Fix all other coordinates, optimize the block
            # Objective: max_k conv(a,a)_k, subject to a[idx] >= 0, sum a[idx] = f[idx].sum()
            # This is non-convex (quadratic), so use a simple approach:
            # Try shifting mass within the block
            block_sum = f[idx].sum()
            if block_sum < 1e-15:
                continue

            # Try uniform allocation in the block
            a_trial = f.copy()
            a_trial[idx] = block_sum / len(idx)
            st = a_trial.sum()
            a_trial /= st
            val = fast_convolve(a_trial).max()

            if val < best_obj - 1e-12:
                f = a_trial
                best_obj = val
                best_f = f.copy()
                improved = True

            # Try concentrating mass at the position with lowest "load"
            # (position where changing it most reduces the peak)
            for pos in idx:
                if f[pos] < 1e-12:
                    continue
                # Move half the mass to a neighbor
                for nb in [pos - 1, pos + 1]:
                    if 0 <= nb < n and nb not in idx:
                        a_trial = f.copy()
                        transfer = f[pos] * 0.3
                        a_trial[pos] -= transfer
                        a_trial[nb] += transfer
                        if a_trial[pos] < 0:
                            continue
                        st = a_trial.sum()
                        a_trial /= st
                        val = fast_convolve(a_trial).max()
                        if val < best_obj - 1e-12:
                            f = a_trial
                            best_obj = val
                            best_f = f.copy()
                            improved = True
                            break

            if not improved and time.monotonic() - t0 > time_limit:
                break

    return best_f, best_obj


def directed_subgrad(f0, time_limit, n_peaks=20, lr_init=0.003, momentum=0.92):
    """Directly target top peaks with subgradient method."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Select top peaks
        k_top = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = pv - pv.min()
        threshold = scaled.max() * 0.3
        w = np.where(scaled > threshold, np.exp(scaled - scaled.max()), 0.0)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = w

        grad = compute_lse_grad(f, w_full, max_peaks=k_top)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 500
            break
        grad /= grad_norm

        v = momentum * v - cur_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                cur_lr *= 0.94
            if no_improve > 200:
                rng = np.random.default_rng((it * 6271 + 17) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0003)
                no_improve = 0

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def diverse_init(n, seed=0):
    """Create diverse initializations for exploration."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian-like
    x = np.arange(n)
    center = n / 2.0
    sigma = n / 4.0
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    g /= g.sum()
    inits.append(g)

    # Two-peaked
    sigma2 = n / 6.0
    g2 = np.exp(-0.5 * ((x - n * 0.25) / sigma2) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / sigma2) ** 2)
    g2 /= g2.sum()
    inits.append(g2)

    # Linear ramp (asymmetric)
    ramp = np.linspace(0.1, 1.0, n)
    ramp /= ramp.sum()
    inits.append(ramp)

    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Quadratic U-shape
    t = np.linspace(-1, 1, n)
    u = 1.0 + t * t
    u /= u.sum()
    inits.append(u)

    # Cosine (first lobe)
    c = np.cos(np.pi * x / n) ** 2
    c = np.maximum(c, 0.0)
    s = c.sum()
    if s > 0:
        c /= s
    inits.append(c)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization with Nesterov-accelerated LSE,
    FW-LP, block CD, and directed subgradient methods.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Global exploration with diverse inits at multiple n ===
    print("\n=== Phase 1: Global Exploration ===")
    t1_end = deadline - 700

    n_candidates = sorted(set([n_parent, 448, 512, 576, 640, 704, 768, 832, 896]), reverse=True)
    n_candidates = [nc for nc in n_candidates if 256 <= nc <= 1024]

    explored = 0
    for nc in n_candidates:
        t_remaining = t1_end - time.monotonic()
        if t_remaining < 40 or explored >= 5:
            break
        t_alloc = min(120, t_remaining * 0.25)
        if t_alloc < 25:
            break
        explored += 1

        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_to_length(parent, nc)

        print(f"  FW-LP n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = fw_lp(f_init, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

        # Quick LSE follow-up
        current = best_tracker[1] if best_tracker[1] is not None else f_opt
        t_lse = min(50, t1_end - time.monotonic() - 30)
        if t_lse > 20:
            f_opt2, _ = nesterov_lse(current, time_limit=t_lse,
                                      beta_start=30.0, beta_end=5000.0, max_peaks=200)
            try_sequence(f_opt2, best_tracker)

    # Try diverse initializations at the parent length
    if time.monotonic() < t1_end - 50:
        n_div = n_parent
        for i, f_init in enumerate(diverse_init(n_div, seed=seed + i)):
            t_remaining = t1_end - time.monotonic()
            if t_remaining < 30:
                break
            t_alloc = min(40, t_remaining * 0.3)
            if t_alloc < 15:
                break
            print(f"  Diverse init {i} n={n_div}, t={t_alloc:.0f}s")
            f_opt, _ = nesterov_lse(f_init, time_limit=t_alloc,
                                     beta_start=20.0, beta_end=3000.0, max_peaks=150)
            try_sequence(f_opt, best_tracker)

    # === Phase 2: Intensive Nesterov LSE on best ===
    print(f"\n=== Phase 2: Nesterov LSE ===")
    t2_end = deadline - 400
    current = best_tracker[1] if best_tracker[1] is not None else parent

    # Multiple rounds with increasing beta
    rounds = 3
    for r in range(rounds):
        t_remaining = t2_end - time.monotonic()
        if t_remaining < 60:
            break
        t_alloc = min(180, t_remaining * 0.45)
        if t_alloc < 30:
            break
        beta_start = 50.0 * (r + 1)
        beta_end = 10000.0 * (r + 1)
        print(f"  Nesterov LSE round {r+1}: beta=[{beta_start},{beta_end}], t={t_alloc:.0f}s")
        f_opt, _ = nesterov_lse(current, time_limit=t_alloc,
                                 beta_start=beta_start, beta_end=beta_end, max_peaks=300)
        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current

    # === Phase 3: Directed subgradient + FW alternation ===
    print(f"\n=== Phase 3: Alternating Direction ===")
    t3_end = deadline - 250
    cycles = 0
    while time.monotonic() < t3_end and cycles < 10:
        cycles += 1
        current = best_tracker[1] if best_tracker[1] is not None else parent
        remaining = t3_end - time.monotonic()
        if remaining < 50:
            break

        if cycles % 3 == 1:
            t_alloc = min(80, remaining * 0.4)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: Directed subgrad, t={t_alloc:.0f}s")
                f_opt, _ = directed_subgrad(current, time_limit=t_alloc,
                                             n_peaks=16, lr_init=0.002, momentum=0.93)
                try_sequence(f_opt, best_tracker)
        elif cycles % 3 == 2:
            t_alloc = min(70, remaining * 0.4)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: FW-LP, t={t_alloc:.0f}s")
                f_opt, _ = fw_lp(current, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)
        else:
            t_alloc = min(80, remaining * 0.4)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: Nesterov LSE, t={t_alloc:.0f}s")
                f_opt, _ = nesterov_lse(current, time_limit=t_alloc,
                                         beta_start=100.0, beta_end=15000.0, max_peaks=400)
                try_sequence(f_opt, best_tracker)

    # === Phase 4: Block coordinate descent polishing ===
    print(f"\n=== Phase 4: Block CD Polishing ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(150, deadline - time.monotonic() - 80)
    if t_alloc > 20:
        print(f"  Block CD (size=6), t={t_alloc:.0f}s")
        f_opt, _ = block_cd_polish(current, time_limit=t_alloc, block_size=6)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Final high-beta LSE + try nearby n ===
    print(f"\n=== Phase 5: Final Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent

    # Very high beta
    t_alloc = min(120, deadline - time.monotonic() - 60)
    if t_alloc > 20:
        print(f"  High-beta LSE, t={t_alloc:.0f}s")
        f_opt, _ = nesterov_lse(current, time_limit=t_alloc,
                                 beta_start=500.0, beta_end=50000.0, max_peaks=500)
        try_sequence(f_opt, best_tracker)

    # Directed with very few peaks
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(80, deadline - time.monotonic() - 40)
    if t_alloc > 15:
        print(f"  Directed (n_peaks=8), t={t_alloc:.0f}s")
        f_opt, _ = directed_subgrad(current, time_limit=t_alloc,
                                     n_peaks=8, lr_init=0.001, momentum=0.95)
        try_sequence(f_opt, best_tracker)

    # Try nearby lengths
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [32, -32, 64, -64, 128, -128]
        for off in offsets:
            n_try = n_best + off
            if n_try < 256 or n_try > 1200:
                continue
            t_remaining = deadline - time.monotonic() - 20
            if t_remaining < 30:
                break
            t_alloc = min(40, t_remaining * 0.5)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = nesterov_lse(f_init, time_limit=t_alloc,
                                     beta_start=30.0, beta_end=5000.0, max_peaks=200)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using fallback")
    fallback = np.ones(512) / 512
    return [float(x) for x in fallback]
