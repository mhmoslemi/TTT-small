4. Symmetric variant optimization

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_gradient_fft(f, w):
    """Compute gradient of LSE using FFT cross-correlation.
    grad[m] = 2 * sum_k w[k] * f[k-m] * I(0 <= k-m < n)
    = 2 * cross_correlation(w, f)[m]
    """
    n = len(f)
    conv_len = 2 * n - 1
    # Need FFT size >= conv_len + n - 1 for linear cross-correlation
    fft_size = 1 << (conv_len + n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    # Cross-correlation: IFFT(W * conj(F))
    corr = np.fft.irfft(W * np.conj(F), fft_size)
    return 2.0 * corr[:n]

    """FISTA with LSE-smoothed max-convolution objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    beta = beta_start
    t_k = 1.0
    v = f.copy()
    v_prev = f.copy()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # FISTA extrapolation
        if it > 1:
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = v + (t_k - 1.0) / t_next * (v - v_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 0:
                y = y / s_y
            else:
                y = v.copy()
        else:
            y = v.copy()

        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(np.minimum(bc - bc_max, 50.0))  # clip for numerical stability
        w = exp_bc / exp_bc.sum()

        # Adaptive Lipschitz: L ≈ 2 * n * max(w)  (rough upper bound)
        # More conservative: L = 2 * n
        L = 2.0 * n
        # Actually, let's use a tighter bound

        # len(u_k) = min(k+1, n, 2n-1-k)
        # For simplicity, use L = 2 * n
        step = 1.0 / L
        
        v_prev = v.copy()
        v = y - step * grad
        v = project_simplex(v)
        
        # Evaluate
        conv_v = fast_convolve(v)
        obj = conv_v.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = v.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:

                scale = 0.001 + 0.003 * min(1.0, no_improve / 200.0)
                f_pert = best_f + rng.standard_normal(n) * scale
                f_pert = project_simplex(f_pert)
                v = f_pert
                v_prev = f_pert
                t_k = 1.0
                no_improve = 0
        
        # Update t_k
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        
        # Increase beta
        if beta < beta_end:
            beta *= 1.005
        
        if verbose and it % 100 == 0:
            print(f"    FISTA it={it}, obj={obj:.10f}, beta={beta:.1f}, t_k={t_k:.1f}")
    
    return best_f, best_obj

    """Compute cross-correlation (a ⋆ b)[m] = sum_j a[j+m] * b[j] for m=0..len(b)-1."""
    n_a = len(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE gradient (O(n log n) per step)
2. FISTA-style accelerated proximal gradient with adaptive step size
3. Multi-objective alternation (LSE + weighted L2 + directed peak)
4. Adaptive beta scheduling with geometric increase
5. Multi-length exploration with smart interpolation
6.Perturbation-based restarts to escape local minima
Key innovation: all gradient computations use FFT cross-correlation,
making each iteration O(n log n) instead of O(n * n_peaks).

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_fft(a, b):
    """Compute cross-correlation (a ⋆ b)[m] = sum_j a[j+m]*b[j] for m=0..len(b)-1."""
    n_a = len(a)
    n_b = len(b)
    fft_size = 1 << (n_a + n_b - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    corr = np.fft.irfft(A * np.conj(B), fft_size)
    return corr[:n_b]


def lse_grad_fft(f, w):
    """Gradient of LSE: grad[m] = 2 * (w ⋆ f)[m]."""
    return 2.0 * cross_corr_fft(w, f)


def weighted_l2_grad_fft(f, conv, w):
    """Gradient of sum_k w[k]*(conv[k])^2: grad[m] = 4*(w*conv ⋆ f)[m]."""
    return 4.0 * cross_corr_fft(w * conv, f)


def normalized(f0):
    """Ensure f is non-negative with sum=1."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    return f

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

        if remaining < 50:
            break

        if cycles % 3 == 1:
            t_alloc = min(80.0, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: LSE-FISTA (beta=500-10000)")
                f_opt, _ = accelerate_lse(current, time_limit=t_alloc,
                                           beta_start=500.0, beta_end=10000.0)
                try_sequence(f_opt, best_tracker)
        elif cycles % 3 == 2:
            t_alloc = min(80.0, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: Directed peak (n_peaks=5)")
                f_opt, _ = directed_peak_fista(current, time_limit=t_alloc, n_peaks=5)
                try_sequence(f_opt, best_tracker)
        else:
            t_alloc = min(80.0, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: Mixed norm")
                f_opt, _ = mixed_norm_optimize(current, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration ===")
    t5_end = deadline - 100.0
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        length_candidates = sorted(set([
            n_best,
            n_best + 64, n_best - 64,
            n_best + 128, n_best - 128,
            1024, 1152, 1280,
            640, 896
        ]))
        length_candidates = [lc for lc in length_candidates if 256 <= lc <= 1536]

        for n_try in length_candidates:
            if n_try == n_best:
                continue
            t_remaining = t5_end - time.monotonic()
            if t_remaining < 40:
                break
            t_alloc = min(60.0, t_remaining * 0.35)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = accelerate_lse(f_init, time_limit=t_alloc,
                                       beta_start=50.0, beta_end=5000.0)
            try_sequence(f_opt, best_tracker)

            # Quick directed peak polish
            t_alloc2 = min(30.0, t5_end - time.monotonic() - 20)
            if t_alloc2 > 10:
                f_opt2, _ = directed_peak_fista(f_opt, time_limit=t_alloc2, n_peaks=8)
                try_sequence(f_opt2, best_tracker)

    # === Phase 6: Final micro-polish ===
    print(f"\n=== Phase 6: Final Micro-Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(60.0, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  Ultra-high beta LSE: t={t_alloc:.0f}s")
        f_opt, _ = accelerate_lse(current, time_limit=t_alloc,
                                   beta_start=2000.0, beta_end=50000.0)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(40.0, deadline - time.monotonic() - 5)
    if t_alloc > 8:
        print(f"  Final directed peak: t={t_alloc:.0f}s")
        f_opt, _ = directed_peak_fista(current, time_limit=t_alloc, n_peaks=3)
        try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using parent fallback")
    return [float(x) for x in parent]
