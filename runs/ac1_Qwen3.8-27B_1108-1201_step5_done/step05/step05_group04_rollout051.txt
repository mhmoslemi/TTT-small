from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute gradient of sum_k v[k] * conv(a,a)[k] w.r.t. a.
    Uses the formula: grad[i] = 2 * sum_j a[j] * v[i+j]
    via FFT cross-correlation.

    N = 1 << (n + conv_len - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    V = np.fft.rfft(v_flipped, N)
    conv_result = np.fft.irfft(A * V, N)
    # cross[i] = conv_result[conv_len-1-i] for i in [0, n-1]
    idx = np.arange(conv_len - 1, conv_len - n - 1, -1)
    cross = conv_result[idx]
    return 2.0 * cross

def high_power_lse_adam(f0, time_limit, p_start=8, p_end=2048, max_iters=None):
    """
    High-power LSE optimization with Adam.
    Minimizes sum_k v[k] * conv[k]^p where v[k] = 1 for all k (uniform weights).
    As p increases, this approaches minimizing max(conv).
    Uses Adam for per-coordinate adaptive learning rates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    p = p_start
    lr = 1.0 / np.sqrt(n)
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros(n)  # first moment
    v_adam = np.zeros(n)  # second moment (named v_adam to avoid conflict with v in gradient)
    t = 0
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        
        # Compute v[k] = p * conv[k]^(p-1) (the "weight" vector)
        # To avoid overflow, scale: v[k] = p * (conv[k]/conv_max)^(p-1) * conv_max
        # Actually, we can work with normalized weights:
        conv_max = conv.max()
        if conv_max < 1e-30:
            break
        # v[k] = p * conv[k]^(p-1), but scale to avoid overflow
        # Use log-scaling: v[k] = exp(log(p) + (p-1)*log(conv[k]))
        # But for the gradient, only the RELATIVE weights matter (up to a scale factor
        # that gets absorbed into the learning rate)
        # So we can use: v[k] = conv[k]^(p-1) (dropping the factor p and overall scale)
        # This is equivalent to minimizing sum_k conv[k]^p / p which has the same minimizer.
        
        # Numerically stable computation:
        # v[k] = (conv[k] / conv_max)^(p-1)
        # This keeps values in [0, 1]
        ratio = np.maximum(conv / conv_max, 0.0)
        v = ratio ** (p - 1)  # This might underflow for large p, which is fine
        
        # Gradient: grad[i] = 2 * sum_j a[j] * v[i+j]
        grad = compute_gradient_fft(f, v)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v_adam / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart
                f = best_f.copy()
                m = np.zeros(n)
                v_adam = np.zeros(n)
                t = 0
                # Slight perturbation
                rng = np.random.default_rng(t + no_improve * 12345)
                f = project_simplex(f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        
        # Increase p
        p = min(p * 1.02, p_end)
        
        # Adapt learning rate
        if no_improve == 0:
            lr = min(lr * 1.001, 2.0 / np.sqrt(n))
        elif no_improve > 20:
            lr *= 0.98
    
    return best_f, best_obj

def peak_coordinate_descent(f0, time_limit, n_peaks=10):
    """
    Coordinate descent targeting the top peaks.
    For each peak location k, find the coordinate i that most affects conv[k]
    and adjust it to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        
        # Find top peaks
        k_top = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]  # sort by value descending
        
        improved = False
        for k in peaks:
            k = int(k)
            # conv[k] = sum_j f[j] * f[k-j]
            # To reduce conv[k], we want to reduce f[j]*f[k-j] for pairs (j, k-j)
            # The derivative w.r.t. f[i] is 2*f[k-i] (if k-i in [0,n-1])
            # So the gradient at coordinate i w.r.t. conv[k] is 2*f[k-i]
            # We want to DECREASE f[i] where f[k-i] is large, and INCREASE f[i] where f[k-i] is small
            
            # Simple approach: find the i that most increases conv[k] and decrease it
            # grad_i = 2 * f[k-i] for the contribution to conv[k]
            contrib = np.zeros(n)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                contrib[lo:hi+1] = 2.0 * f[k - np.arange(lo, hi+1)]
            
            if contrib.max() < 1e-15:
                continue
            
            # Decrease the coordinate with highest contribution, increase one with lowest
            i_dec = np.argmax(contrib)
            # Find coordinate with lowest contribution (to increase)
            # But we need to be careful: increasing f[i_inc] also affects other conv[k']
            # Simple heuristic: increase the coordinate with lowest value
            i_inc = np.argmin(f)
            
            if i_dec == i_inc:
                continue
            
            # Small transfer
            delta = min(f[i_dec] * 0.01, 0.001 / n)
            f_new = f.copy()
            f_new[i_dec] -= delta
            f_new[i_inc] += delta
            
            conv_new = fast_convolve(f_new)
            if conv_new.max() < best_obj - 1e-15:
                best_obj = conv_new.max()
                best_f = f_new.copy()
                f = f_new
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
    
    return best_f, best_obj

def fw_lp(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle (same as before, kept for comparison)."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Build LP
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(js))
                cols.extend(js.tolist())
                vals.extend((f[k - js]).tolist())
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.001)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 20})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-15:
                    g /= gs
                    if np.max(np.abs(g - f)) > 1e-12:
                        # Line search
                        best_alpha = 0.0
                        best_val = M
                        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                            ft = np.maximum((1 - alpha) * f + alpha * g, 0.0)
                            st = ft.sum()
                            if st > 1e-15:
                                ft /= st
                                val = fast_convolve(ft).max()
                                if val < best_val:
                                    best_val = val
                                    best_alpha = alpha
                        if best_alpha > 0:
                            f = np.maximum((1 - best_alpha) * f + best_alpha * g, 0.0)
                            f /= f.sum()
                            conv = fast_convolve(f)
                            M = conv.max()
                            if M < best_obj - 1e-14:
                                best_obj = M
                                best_f = f.copy()
                                no_improve = 0
                            else:
                                no_improve += 1
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        if no_improve > 30:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
