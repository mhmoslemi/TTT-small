def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute sum_k p[k] * (P_k a) efficiently."""
    n = len(a)
    indices = np.arange(n)[:, None] + np.arange(n)[None, :]
    return np.sum(p[indices] * a[None, :], axis=1)


def pk_vector(a, k):
    """Compute (P_k a)[t] = a[k-t] if 0<=k-t<n else 0, for all t."""
    n = len(a)
    t = np.arange(n)
    idx = k - t
    mask = (idx >= 0) & (idx < n)
    vec = np.zeros(n)
    vec[mask] = a[idx[mask]]
    return vec


def multi_peak_gradient(a, p):
    """Compute 2 * hankel_mv(p, a) - the smoothed gradient of max(c)."""
    return 2.0 * hankel_mv(p, a)


def lp_peak_direction(a, top_k_peaks, eps=0.001):
    """Find direction d minimizing d^T g_1 subject to |d^T g_i| <= eps for i>1.
    
    Returns direction d or None if no useful direction found.
    """
    n = len(a)
    g0 = pk_vector(a, top_k_peaks[0])
    
    c_obj = g0.copy()
    
    A_ub_list = []
    b_ub_list = []
    for i in range(1, min(len(top_k_peaks), 8)):
        gi = pk_vector(a, top_k_peaks[i])
        A_ub_list.append(gi)
        b_ub_list.append(eps)
        A_ub_list.append(-gi)
        b_ub_list.append(eps)
    
    if A_ub_list:
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
    else:
        A_ub = None
        b_ub = None
    
    A_eq = np.array([[1.0] * n])
    b_eq = np.array([0.0])
    
    lower = np.maximum(-a, -1.0)
    upper = np.ones(n)
    bounds = list(zip(lower, upper))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.fun < -1e-8:
            return res.x
    except Exception:
        pass
    return None

def game_iterate(a0, time_limit, T_init=0.001):
    """Game-theoretic iteration with efficient Hankel products."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    inner_lr = 0.1
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Inner loop: projected gradient on a^T M(p) a
        a_inner = a.copy()
        for _ in range(3):
            Ma = hankel_mv(p, a_inner)
            grad_inner = 2.0 * Ma
            a_inner = project_simplex(a_inner - inner_lr * grad_inner)
        
        obj = fast_convolve(a_inner).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_inner.copy()
            no_improve = 0
            a = a_inner
        else:
            # Blend
            for alpha in [0.3, 0.5, 0.7]:
                candidate = project_simplex((1 - alpha) * a + alpha * a_inner)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_a = candidate.copy()
                    no_improve = 0
                    a = candidate
                    break
            else:
                no_improve += 1
                a = a_inner
        
        # Adapt
        if no_improve > 20:
            T *= 1.05
            inner_lr *= 0.9
        if no_improve > 80:
            T *= 0.7
            inner_lr = 0.1
        if no_improve > 200:
            rng = np.random.default_rng((it * 31627 + 11) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            inner_lr = 0.1
            no_improve = 0
    
    return best_a, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Slightly perturbed parent
            pp = px + rng.standard_normal(n) * 0.01
            pp = np.maximum(pp, 0)
            pp = pp / pp.sum()
            inits.append(pp)
    
    return inits

    print(f"  Phase 5 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
