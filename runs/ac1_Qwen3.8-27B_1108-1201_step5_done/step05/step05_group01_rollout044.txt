import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient of max_k conv[k] using top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w[i] * a[k - indices]
    return sg

        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        a = a_new

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 100:
            eta *= 0.8
            temp *= 1.5
            if no_improve > 500:
                # Restart with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(a)
                eta = eta0 * max(0.3, 500.0 / (no_improve + 1))
                temp = temp0 * 2.0
                no_improve = 0

        # Gradually increase temperature (explore more peaks)
        temp *= 1.002
        # Slowly decrease eta
        if no_improve == 0:
            eta = min(eta0, eta * 1.001)

    return best_a, best_obj


def pairwise_cd(a0, time_limit, n_pairs_per_step=10):
    """Pairwise coordinate descent with golden section search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    conv = fast_convolve(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(n_pairs_per_step):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            # Pick random pair
            i, j = rng.choice(n, size=2, replace=False)
            s_ij = a[i] + a[j]
            if s_ij < 1e-14:
                continue

            # Golden section search on a[i] in [0, s_ij]
            lo = 0.0
            hi = s_ij
            GR = (1.0 + np.sqrt(5.0)) / 2.0
            c1 = lo + (1.0 - 1.0/GR) * (hi - lo)
            c2 = lo + (1.0/GR) * (hi - lo)

            def eval_pair(x_i):
                a[i] = x_i
                a[j] = s_ij - x_i
                # Only need to check affected lags
                # Affected lags: those where i or j contributes
                k_affected = set()
                for m in [i, j]:
                    k_lo = m
                    k_hi = m + n - 1
                    k_affected.update(range(k_lo, min(k_hi + 1, 2 * n - 1)))
                # Also need to check if current max is affected
                # For safety, just compute full conv for correctness (slower but safe)
                return fast_convolve(a).max()

            f1 = eval_pair(c1)
            f2 = eval_pair(c2)

            for _ in range(20):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1.0 - 1.0/GR) * (hi - lo)
                    f1 = eval_pair(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0/GR) * (hi - lo)
                    f2 = eval_pair(c2)

            opt_x = (lo + hi) / 2.0
            a[i] = opt_x
            a[j] = s_ij - opt_x
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum

            cur_obj = fast_convolve(a).max()
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()

    return best_a, best_obj


def lp_fw_step(a, M, n, top_constraints=100):
    """Single LP-FW step using only top constraints."""
    conv = fast_convolve(a)
    # Select top constraints
    n_constraints = min(top_constraints, len(conv))
    top_idx = np.argpartition(conv, -n_constraints)[-n_constraints:]
    top_idx = sorted(top_idx)

    # Build LP: max sum(g) s.t. (a*g)[k] <= M for k in top_idx, g >= 0
    c = -np.ones(n)
    rows = []
    b_ub = []
    for k in top_idx:
        row = np.zeros(n)
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            row[m_lo:m_hi + 1] = a[k - np.arange(m_lo, m_hi + 1)]
        rows.append(row)
        b_ub.append(M)

    if not rows:
        return None

    A_ub = np.array(rows)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(a0, time_limit, top_constraints=200):
    """LP-Frank-Wolfe with top constraints only."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = lp_fw_step(a, M, n, top_constraints=top_constraints)
        if g is None:
            # Relax M
            M *= 1.05
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            M *= 1.02
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            at = a + alpha * (g - a)
            at = np.maximum(at, 0.0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a_new = a + best_alpha * (g - a)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            M = fast_convolve(a_new).max()
            a = a_new
            if M < best_obj - 1e-13:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_obj


def structured_init(n, method='cosine', freq=1.0, amp=0.3, seed=0):
    """Generate structured initializations."""
    j = np.arange(n, dtype=np.float64)
    if method == 'cosine':
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    elif method == 'gaussian':
        center = n / 2.0
        width = n / (freq * 2.0)
        a = np.exp(-0.5 * ((j - center) / width) ** 2)
    elif method == 'power':
        a = np.abs(np.sin(np.pi * j / n)) ** freq + 1e-6
    elif method == 'chirp':
        a = 1.0 + amp * np.cos(2.0 * np.pi * (freq * j / n + 0.5 * (j / n) ** 2 * amp))
    elif method == 'random_phase':
        rng = np.random.default_rng(seed)
        phases = rng.uniform(0, 2 * np.pi, size=min(10, n))
        a = np.ones(n)
        for k, ph in enumerate(phases):
            a += 0.1 * np.cos(2 * np.pi * (k + 1) * j / n + ph)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def accelerated_subgradient(a0, time_limit, lr0=0.003, mom0=0.9, top_k=50, temp0=0.0005):
    """Nesterov-accelerated projected subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr0
    mom = mom0
    temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        sg = compute_subgradient(a, conv, top_k=top_k, temp=temp)

        v = mom * v - lr * sg
        y = y + v
        y = project_simplex(y)
        a = y.copy()

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.8
            temp *= 1.5
            if no_improve > 600:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                y = a.copy()
                v = np.zeros(n)
                conv = fast_convolve(a)
                lr = max(lr0 * 0.5, 0.0005)
                no_improve = 0

        # Slowly adapt
        mom = min(0.98, mom * 1.0002)
        if no_improve == 0:
            temp *= 1.001

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining mirror descent (multiplicative weights),
    pairwise coordinate descent, LP-FW, and structured initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    print(f"Starting with n={len(parent)}")

    # === Phase 1: Mirror descent on parent (main workhorse) ===
    print("\n=== Phase 1: Mirror descent on parent ===")
    t_remaining = deadline - time.monotonic() - 600
    t1 = max(60, min(t_remaining * 0.35, 500))
    print(f"  Time: {t1:.0f}s")
    a_opt, _ = mirror_descent(parent, time_limit=t1, eta0=0.02, top_k=80, temp0=0.002)
    try_seq(a_opt, "mirror_parent")

    # === Phase 2: Try structured inits at various lengths ===
    print(f"\n=== Phase 2: Structured initializations (best={best_score:.8f}) ===")
    init_configs = [
        (768, 'cosine', 1, 0.3), (768, 'cosine', 2, 0.2), (768, 'cosine', 3, 0.15),
        (768, 'chirp', 1, 0.5), (768, 'chirp', 2, 0.3),
        (768, 'power', 2, 0), (768, 'power', 4, 0),
        (768, 'gaussian', 1.5, 0), (768, 'gaussian', 3, 0),
        (512, 'cosine', 1, 0.3), (512, 'chirp', 1, 0.5),
        (1024, 'cosine', 1, 0.3), (1024, 'chirp', 1, 0.5),
        (1024, 'power', 3, 0),
    ]

    for n_init, method, freq, amp in init_configs:
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        a_init = structured_init(n_init, method, freq, amp)
        s = try_seq(a_init, f"{method}_n{n_init}_f{freq}")
        if np.isfinite(s) and s < best_score + 0.05:
            t_alloc = min(rem * 0.2, 60)
            if t_alloc > 10:
                a_opt, _ = mirror_descent(a_init, time_limit=t_alloc, eta0=0.03, top_k=50, temp0=0.003)
                try_seq(a_opt, f"mirror_{method}_n{n_init}")

    # === Phase 3: Interpolate to different lengths ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        for target_n in [640, 900, 1024, 1280, 1536]:
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            try_seq(f_interp, f"interp_n{target_n}")
            t_alloc = min(rem * 0.3, 80)
            if t_alloc > 15:
                a_opt, _ = mirror_descent(f_interp, time_limit=t_alloc, eta0=0.015, top_k=60, temp0=0.002)
                try_seq(a_opt, f"mirror_interp_n{target_n}")

    # === Phase 4: Accelerated subgradient ===
    print(f"\n=== Phase 4: Accelerated subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_ns = min(rem * 0.5, 150)
            print(f"  n={n_best}, t={t_ns:.0f}s")
            a_opt, _ = accelerated_subgradient(best_seq, time_limit=t_ns,
                                               lr0=0.004, mom0=0.88,
                                               top_k=max(30, n_best // 10),
                                               temp0=0.001)
            try_seq(a_opt, "accel_sg")

            # Second pass finer
            rem = deadline - time.monotonic() - 120
            if rem > 20:
                a_opt, _ = accelerated_subgradient(a_opt, time_limit=min(rem, 80),
                                                   lr0=0.001, mom0=0.93,
                                                   top_k=max(40, n_best // 8),
                                                   temp0=0.0003)
                try_seq(a_opt, "accel_sg_fine")

    # === Phase 5: Pairwise coordinate descent ===
    print(f"\n=== Phase 5: Pairwise CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.5, 120)
            print(f"  n={n_best}, t={t_cd:.0f}s")
            a_opt, _ = pairwise_cd(best_seq, time_limit=t_cd, n_pairs_per_step=5)
            try_seq(a_opt, "pairwise_cd")

    # === Phase 6: LP-FW polish ===
    print(f"\n=== Phase 6: LP-FW polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_lpfw = min(rem * 0.6, 100)
            print(f"  n={n_best}, t={t_lpfw:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw, top_constraints=min(150, n_best))
            try_seq(a_opt, "lpfw_polish")

    # === Phase 7: Final mirror descent polish ===
    print(f"\n=== Phase 7: Final mirror descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_final = min(rem, 60)
            print(f"  t={t_final:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_final, eta0=0.005, top_k=100, temp0=0.0005)
            try_seq(a_opt, "final_mirror")

    # === Phase 8: Tiny perturbations + quick mirror descent ===
    print(f"\n=== Phase 8: Perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for trial in range(8):
            rem = deadline - time.monotonic() - 5
            if rem < 8:
                break
            noise_scale = 0.002 * (trial + 1) * 0.3
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem - 3, 30)
            if t_alloc > 5:
                a_opt, _ = mirror_descent(f_pert, time_limit=t_alloc, eta0=0.008, top_k=60, temp0=0.001)
                try_seq(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
