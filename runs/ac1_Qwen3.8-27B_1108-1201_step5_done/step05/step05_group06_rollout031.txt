Actually, I realize I should be more careful. The FISTA algorithm minimizes `f(x)` over the constraint set `C`. The objective is the smoothed log-sum-exp, and the constraint set is the simplex.

But actually, I don't need to project y onto the simplex in FISTA. The standard FISTA for constrained problems is:
- x_k = Proj_C(y_k - (1/L) * grad f(y_k))
- y_{k+1} = x_k + ((t_k - 1)/t_{k+1}) * (x_k - x_{k-1})

The projection of x is automatic (since we project onto C). The y point doesn't need to be in C.

But in practice, keeping y in C can help with convergence. Let me just not project y.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA (accelerated proximal gradient)
on log-sum-exp smoothed objective with multi-scale tau, coordinate descent polishing,
and multi-length exploration. Exploits convexity of the problem for fast convergence.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, tau):
    """
    Compute gradient of f_tau(a) = tau * log(sum_k exp(conv_k(a) / tau)).
    Returns (grad, smoothed_value).
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_len = len(conv)
    
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(conv_len)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum
    
    smoothed_val = max_c + tau * np.log(w_sum)
    
    # grad_j = 2 * sum_{c=0}^{n-1} a[c] * w[c+j]  for j=0,...,n-1
    # Compute via FFT: g[j] = conv_full(a, w_flipped)[2n-2-j]
    w_flipped = w[::-1]
    fft_size2 = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, fft_size2)
    W = np.fft.rfft(w_flipped, fft_size2)
    conv_aw = np.fft.irfft(A * W, fft_size2)[:3 * n - 2]
    g = conv_aw[n - 1 : 2 * n - 1][::-1]
    
    return 2.0 * g, smoothed_val


def fista_optimize(a0, tau, max_iter, time_limit, verbose=False):
    """
    FISTA on the smoothed problem over the simplex.
    Returns (best_a, best_true_obj).
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 0.0)
    
    # Adaptive Lipschitz constant
    L = 4.0 / tau + 4.0
    step = 1.0 / L
    
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    
    conv_best = fast_convolve(a)
    best_obj = conv_best.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    for k in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad_y, _ = compute_smoothed_grad(y, tau)
        x = project_simplex(y - step * grad_y)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        t = t_new
        x_prev = x.copy()
        
        # Periodic evaluation of true (unsmoothed) objective
        if k % 500 == 0:
            conv_check = fast_convolve(x)
            obj = conv_check.max()
            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()
            if verbose and k % 5000 == 0:

    # Final check
    conv_final = fast_convolve(x)
    obj_final = conv_final.max()
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = x.copy()
    
    return best_a, best_obj

def coordinate_bisection(a0, time_limit, verbose=False):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    max_passes = 1000
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        max_improvement = 0.0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break
                x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = a0.copy()
                continue
            
            conv = fast_convolve(a)
            obj = conv.max()
            improvement = best_obj - obj
            if improvement > max_improvement:
                max_improvement = improvement
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if passes > 2:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0004)
            else:
                break
        if verbose and passes % 10 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def make_hanning(n):
    """Create a Hanning window of length n, normalized to sum 1."""
    t = np.arange(n)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1))) if n > 1 else np.array([1.0])
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s > 1e-15:
        w = w / s
    return w


def make_uniform(n):
    return np.ones(n) / n


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    t = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    w = np.exp(-t * t / (2.0 * sigma * sigma))
    s = w.sum()
    if s > 1e-15:
        w = w / s
    return w


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym
