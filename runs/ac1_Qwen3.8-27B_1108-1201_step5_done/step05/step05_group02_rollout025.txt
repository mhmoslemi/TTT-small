def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max(obj, peaks, temp):
    """Compute softmax-weighted subgradient of smooth max."""
    w = np.exp(peaks / temp)
    w /= w.sum()
    sg = np.zeros(len(obj))
    n = len(obj)
    for i, k in enumerate(peaks):
        if w[i] < 1e-12:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            kg = k - j
            if 0 <= kg < n:
                sg[j] += 2.0 * w[i] * obj[kg]
    return sg

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, peaks, weights):
    """Compute weighted subgradient of max_k (f*f)(k)."""
    n = len(f)
    sg = np.zeros(n)
    for k, wk in zip(peaks, weights):
        if wk < 1e-12:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # f[k-j] for j in [js, je] => f[k-je : k-js+1] reversed
        sg[js:je + 1] += 2.0 * wk * f[k - je:k - js + 1][::-1]
    return sg


def softmax_weights(conv, peaks, temp):
    """Compute softmax weights for selected peaks."""
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= s
    return w


def mirror_descent_optimize(f0, time_limit, init_temp=0.002, init_step=0.5,
                            top_k_frac=0.05, verbose=False):
    """Mirror descent (multiplicative weights) with softmax subgradients."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    temp = init_temp
    step = init_step
    it = 0
    no_improve = 0
    top_k = max(10, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        peaks = np.argpartition(conv, -top_k)[-top_k:]
        weights = softmax_weights(conv, peaks, temp)
        sg = compute_subgradient(f, peaks, weights)

        # Multiplicative weight update
        scaled = -step * sg
        scaled -= scaled.max()
        g = f * np.exp(scaled)
        gs = g.sum()
        if gs < 1e-300:
            g = np.ones(n) / n
        else:
            g = g / gs

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    MD iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Schedule
        temp *= 1.003
        step *= 0.9995
        top_k = max(10, int(n * top_k_frac * (1.0 + 0.5 * min(it / 2000.0, 2.0))))

        if no_improve > 200:
            rng = np.random.default_rng(it * 104729 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            temp = init_temp * 0.5
            step = init_step * 0.7

    return best_f, best_obj


def cutting_plane_fw(f0, time_limit, max_active=300, verbose=False):
    """Frank-Wolfe with cutting-plane LP constraint management."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    # Initialize active set with top peaks
    active_list = list(np.argsort(conv)[-min(50, 2 * n - 1)])

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        m = len(active_list)
        if m == 0:
            active_list = list(np.argsort(conv)[-50])
            m = len(active_list)

        # Build and solve LP
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        for idx, k in enumerate(active_list):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[idx, j] = f[k - j]
            A_ub[idx, -1] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs')

        if not res.success:
            # Fallback: mirror descent step
            peaks = np.argpartition(conv, -20)[-20:]
            weights = softmax_weights(conv, peaks, 0.001)
            sg = compute_subgradient(f, peaks, weights)
            g = f * np.exp(-0.3 * sg)
            g = g / g.sum()
        else:
            g = res.x[:n]
            g = np.maximum(g, 0.0)
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        # Check for violated constraints
        fg = np.convolve(f, g)
        violated = np.where(fg > 1e-10)[0]
        new_violated = [int(v) for v in violated if v not in active_list]
        if new_violated:
            # Add most violated first
            new_violated.sort(key=lambda k: -fg[k])
            active_list.extend(new_violated[:min(20, max_active - m)])
            if len(active_list) > max_active:
                # Trim to most important
                conv_val = fast_convolve(f)
                active_set = set(active_list)
                scores = {k: conv_val[k] + fg[k] for k in active_set}
                active_list = sorted(scores, key=scores.get, reverse=True)[:max_active]

        # Line search
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Refine
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    CP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            active_list = list(np.argsort(conv)[-min(50, 2 * n - 1)])

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_initializations(n):
    """Generate various structured initializations of length n."""
    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Triangular (tent)
    x = np.arange(n)
    mid = n // 2
    tri = np.maximum(0, 1.0 - np.abs(x - mid) / (mid + 1))
    tri /= tri.sum()
    candidates.append(tri)

    # Cosine bump
    x = np.linspace(0, np.pi, n)
    cos_bump = np.sin(x) ** 2
    cos_bump /= cos_bump.sum()
    candidates.append(cos_bump)

    # Exponential taper
    decay = np.exp(-np.abs(np.arange(n) - n // 2) / (n // 4))
    decay /= decay.sum()
    candidates.append(decay)

    # Binomial-like (centered)

        m = n // 2
        binom = np.array([comb(2 * m, m + i - m) for i in range(n)] if n <= 2 * m + 1 else [comb(2 * m, min(i, 2 * m - i)) for i in range(n)])

                                               top_k_frac=0.03, verbose=True)
            try_seq(a_opt, "sg_refine")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 25:
            t_cd = min(150, rem * 0.4)
            print(f"  n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Multiple short CP-FW runs with different restarts
        for trial in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            if trial == 0:
                start = best_seq
            else:
                noise = rng.standard_normal(n_best) * (0.001 + 0.0005 * trial)
                start = project_simplex(best_seq + noise)
            t_t = min(rem * 0.5, 60)
            a_opt, _ = cutting_plane_fw(start, time_limit=t_t, max_active=150)
            try_seq(a_opt, f"cpfw_final_{trial}")
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            t_t2 = min(rem * 0.4, 40)
            a_opt, _ = coordinate_descent_polish(a_opt, time_limit=t_t2)
            try_seq(a_opt, f"cd_final_{trial}")

        # Very fine mirror descent
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=min(rem, 50),
                                               init_temp=0.0005, init_step=0.1,
                                               top_k_frac=0.02)
            try_seq(a_opt, "md_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
