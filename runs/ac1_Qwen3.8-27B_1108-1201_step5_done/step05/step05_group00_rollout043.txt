```python
"""
Hybrid search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Main ingredients:
- Full-dimensional log-domain L-BFGS for fixed dual top-k / softmax quadratic
  subproblems.  The sequence is parameterized by softmax to stay nonnegative
  and normalized.
- Repeated dual updates: top-k peaks and softmax peak weights.
- Low-dimensional spline/knot soft-log-sum-exp optimization to escape local
  basins and discover global shape.
- Resampling and smooth random explorations.
"""

import numpy as np
import math
import time
from scipy.optimize import minimize


GLOBAL_BEST_SEQ = None
GLOBAL_BEST_SCORE = float("inf")

KNOT_N = 0
KNOT_LEFT = None
KNOT_RIGHT = None
KNOT_W = None
KNOT_T = 1.0


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    c = np.maximum(c, 0.0)
    return c


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def score_from_norm(a):
    a = normalize_seq(a)
    n = a.size
    if n == 0:
        return float("inf")
    C = fast_convolve(a)
    Cmax = float(C.max()) if C.size else float("inf")
    return float(2.0 * n * Cmax)


def update_global_with_norm(a, score, label):
    global GLOBAL_BEST_SEQ, GLOBAL_BEST_SCORE
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return
    if not np.isfinite(score):
        return
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return
    a = a / s
    if score < GLOBAL_BEST_SCORE - 1e-12:
        GLOBAL_BEST_SCORE = float(score)
        GLOBAL_BEST_SEQ = a.copy()
        print(f"NEW BEST [{label}] score={GLOBAL_BEST_SCORE:.8f} n={a.size}")


def consider_sequence(a, label):
    a = normalize_seq(a)
    if a.size == 0:
        return
    C = fast_convolve(a)
    Cmax = float(C.max()) if C.size else float("inf")
    score = float(2.0 * a.size * Cmax)
    update_global_with_norm(a, score, label)


def hankel_direct(p, a):
    """Return (H_p a)_i = sum_j p[i+j] a_j for i=0..n-1."""
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    full = np.convolve(p, a[::-1], mode="full")
    return full[n - 1:2 * n - 1]


def build_hankel(p, n):
    p = np.asarray(p, dtype=np.float64)
    H = np.empty((n, n), dtype=np.float64)
    for i in range(n):
        H[i, :] = p[i:i + n]
    return H


def make_p_softmax(C, ratio):
    C = np.asarray(C, dtype=np.float64)
    Cmax = float(C.max())
    T = max(1e-12, Cmax * max(1e-8, float(ratio)))
    if T <= 1e-12:
        p = np.zeros(C.size, dtype=np.float64)
        p[int(np.argmax(C))] = 1.0
        return p
    ex = np.exp((C - Cmax) / T)
    p = ex / float(ex.sum())
    return p


def make_p_topk(C, K, ratio):
    C = np.asarray(C, dtype=np.float64)
    L = C.size
    K = max(1, min(int(K), L))
    if K >= L:
        p = np.ones(L, dtype=np.float64) / float(L)
        return p
    idx = np.argpartition(C, L - K)[-K:]
    Cmax_on = float(C[idx].max())
    T = max(1e-12, Cmax_on * max(1e-8, float(ratio)))
    if K == 1:
        p = np.zeros(L, dtype=np.float64)
        p[int(idx[0])] = 1.0
        return p
    w = np.exp((C[idx] - Cmax_on) / T)
    p = np.zeros(L, dtype=np.float64)
    p[idx] = w
    p /= float(p.sum())
    return p


def resample_seq(a, n):
    a = normalize_seq(a)
    n = max(2, int(n))
    if a.size <= 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    b = np.interp(x_new, x_old, a)
    return normalize_seq(b)


def smooth_random_seq(n, seed, amp=0.5, max_freq=12):
    n = max(2, int(n))
    rng = np.random.default_rng(seed)
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, int(max_freq) + 1):
        ph = rng.uniform(0.0, 2.0 * np.pi)
        coeff = amp / float(m)
        if m % 2 == 0:
            coeff *= 0.6
        z += coeff * np.cos(2.0 * np.pi * m * t + ph)
    b = 1.0 + z
    b = np.clip(b, 1e-6, None)
    return normalize_seq(b)


def lbfgs_two_loop(g, S_mem, Y_mem):
    d = g.copy()
    alphas = []
    for i in range(len(S_mem) - 1, -1, -1):
        s = S_mem[i]
        yy = Y_mem[i]
        sst = float(np.dot(s, s))
        if sst < 1e-30:
            alpha = 0.0
        else:
            alpha = float(np.dot(s, d)) / sst
        alphas.append(alpha)
        d = d - alpha * yy

    for i in range(len(S_mem)):
        s = S_mem[i]
        yy = Y_mem[i]
        yyt = float(np.dot(yy, d))
        yytyy = float(np.dot(yy, yy))
        if yytyy < 1e-30:
            beta = 0.0
        else:
            beta = yyt / yytyy
        d = d + (beta - alphas[i]) * s
    return d


def yq_eval_with_args(y, free, r, H, n):
    shift = float(np.max(y))
    e = np.exp(y - shift)
    S = float(e.sum()) + 1.0
    b = np.empty(n, dtype=np.float64)
    b[free] = e
    b[r] = 1.0
    b *= n / S

    Hb = H.dot(b)
    phi = float(np.dot(Hb, b))
    g_b = 2.0 * Hb
    G = float(np.dot(g_b, b)) / float(n)
    grad_all = b * (g_b - G)
    grad = grad_all[free]
    return phi, grad, b


def lbfgs_y_quad(a_in, time_limit, p, memory=10, max_iter=5000, seed=123):
    a = normalize_seq(a_in)
    n = a.size
    if n <= 2 or time_limit <= 0.5:
        return a

    b = n * a
    r = int(np.argmax(b))
    mask = np.ones(n, dtype=bool)
    mask[r] = False
    free = np.nonzero(mask)[0]

    tiny = 1e-300
    ratio = b[free] / max(float(b[r]), tiny)
    y = np.log(np.maximum(ratio, tiny))
    y = y - np.mean(y)

    H = build_hankel(p, n)
    phi, g, b = yq_eval_with_args(y, free, r, H, n)

    best_b = b.copy()
    best_y = y.copy()
    best_score = score_from_norm(best_b / best_b.sum())

    t0 = time.monotonic()
    it = 0
    line_fail = 0
    no_improve = 0
    last_alpha = 0.5
    S_mem = []
    Y_mem = []
    rng = np.random.default_rng(seed)
    max_dnorm = 2.0
    check_every = 8
    restart_after = 500

    while it < max_iter and (time.monotonic() - t0) < time_limit:
        it += 1
        if not np.isfinite(phi) or not np.all(np.isfinite(g)):
            line_fail += 1
            if line_fail > 6:
                break
            y = best_y.copy()
            S_mem = []
            Y_mem = []
            phi, g, b = yq_eval_with_args(y, free, r, H, n)
            continue

        if it % check_every == 0:
            Cmax = float(fast_convolve(b).max())
            sc = 2.0 * Cmax / float(n)
            if np.isfinite(sc) and sc + 1e-12 < best_score:
                best_score = sc
                best_b = b.copy()
                best_y = y.copy()
                no_improve = 0
            else:
                no_improve += check_every

        if no_improve >= restart_after:
            noise = 0.015 + 0.025 * min(1.0, (no_improve - restart_after) / 2000.0)
            y = best_y + rng.standard_normal(y.size) * noise
            S_mem = []
            Y_mem = []
            last_alpha = 0.5
            line_fail = 0
            no_improve = 0
            phi, g, b = yq_eval_with_args(y, free, r, H, n)
            continue

        d = lbfgs_two_loop(g, S_mem, Y_mem)
        dnorm = float(np.linalg.norm(d))
        if dnorm < 1e-14:
            line_fail += 1
            if line_fail > 6:
                break
            continue
        if dnorm > max_dnorm:
            d *= max_dnorm / dnorm
            dnorm = max_dnorm

        gd = float(np.dot(g, d))
        if gd >= -1e-14:
            gnorm = float(np.dot(g, g))
            if gnorm <= 1e-24:
                line_fail += 1
                if line_fail > 8:
                    break
                continue
            d = -g
            gd = -gnorm

        alpha = max(1e-4, last_alpha)
        accepted = False
        for _ in range(35):
            y_new = y + alpha * d
            phi_new, g_new, b_new = yq_eval_with_args(y_new, free, r, H, n)
            if np.isfinite(phi_new) and np.all(np.isfinite(g_new)):
                if phi_new <= phi + 1e-4 * alpha * gd + 1e-12:
                    gnd = float(np.dot(g_new, d))
                    if gnd <= 0.9 * gd or phi_new < phi - 1e-7 * max(1.0, abs(phi)):
                        accepted = True
                        break
            alpha *= 0.5
            if alpha < 1e-10:
                break

        if accepted:
            s_vec = alpha * d
            y_vec = phi_new - phi
            if abs(y_vec) > 1e-12:
                S_mem.append(s_vec)
                Y_mem.append(y_vec)
                if len(S_mem) > memory:
                    S_mem.pop(0)
                    Y_mem.pop(0)
            y = y_new
            g = g_new
            b = b_new
            phi = phi_new
            last_alpha = max(1e-6, alpha)
            line_fail = 0
        else:
            line_fail += 1
            if line_fail > 12:
                y = best_y.copy()
                S_mem = []
                Y_mem = []
                last_alpha = 0.5
                line_fail = 0
                no_improve = min(no_improve, 100)
                phi, g, b = yq_eval_with_args(y, free, r, H, n)

    return best_b / max(float(best_b.sum()), 1e-300)


def dual_lbfgs(a_in, time_limit, seed=0):
    a = normalize_seq(a_in)
    n = a.size
    best_a = a.copy()
    best_score = score_from_norm(best_a)
    cur_a = a.copy()

    t0 = time.monotonic()
    stage = 0
    no_improve = 0
    rng = np.random.default_rng(seed + 1)

    while (time.monotonic() - t0) < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        b = n * cur_a
        C = fast_convolve(b)
        Cmax = float(C.max()) if C.size else 1e-300
        if not np.isfinite(Cmax) or Cmax <= 1e-12:
            Cmax = 1e-6

        sm = stage % 6
        if sm == 0:
            p = make_p_topk(C, max(3, int(0.005 * n)), 0.02)
        elif sm == 1:
            p = make_p_softmax(C, 0.05)
        elif sm == 2:
            p = make_p_topk(C, max(8, int(0.015 * n)), 0.04)
        elif sm == 3:
            p = make_p_softmax(C, 0.10)
        elif sm == 4:
            p = make_p_topk(C, max(20, int(0.04 * n)), 0.08)
        else:
            p = make_p_softmax(C, 0.20)

        chunk = min(45.0, max(5.0, rem - 8.0))
        cand = lbfgs_y_quad(cur_a, chunk, p, memory=12, max_iter=2500,
                            seed=seed + 17 * (stage + 1))
        cand = normalize_seq(cand)
        sc = score_from_norm(cand)

        if np.isfinite(sc) and sc + 1e-12 < best_score:
            best_a = cand.copy()
            best_score = sc
            cur_a = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        stage += 1

        if no_improve >= 3 and (no_improve % 2) == 1:
            scale = 0.02 + 0.04 * min(1.0, float(no_improve - 3) / 12.0)
            b_pert = best_a * n * np.exp(rng.standard_normal(n) * scale)
            cur_a = normalize_seq(np.clip(b_pert, 1e-300, None))
            no_improve = 0

        if time.monotonic() - t0 > time_limit - 2:
            break

    return best_a


def knot_prepare(n, K):
    global KNOT_N, KNOT_LEFT, KNOT_RIGHT, KNOT_W
    n = max(2, int(n))
    K = max(2, int(K))
    K = min(K, max(2, n))
    KNOT_N = n

    pos = np.linspace(0.0, 1.0, n)
    idx = pos * (K - 1)
    left = np.floor(idx).astype(int)
    right = left + 1
    w = idx - left
    right[right >= K] = K - 1
    left[left < 0] = 0
    left[left >= K] = K - 1
    w = np.clip(w, 0.0, 1.0)

    KNOT_LEFT = left
    KNOT_RIGHT = right
    KNOT_W = w


def knot_core(h, update_best):
    h = np.asarray(h, dtype=np.float64)
    n = KNOT_N
    if n <= 0:
        return float("inf"), np.zeros_like(h), np.array([]), float("inf")

    left = KNOT_LEFT
    right = KNOT_RIGHT
    w = KNOT_W

    t = h[left] + w * (h[right] - h[left])
    shift = float(np.max(t))
    e = np.exp(t - shift)
    S = float(e.sum())
    b = n * e / max(S, 1e-300)

    C = fast_convolve(b)
    Cmax = float(C.max()) if C.size else float("inf")
    T = max(1e-12, KNOT_T)

    if T <= 1e-12:
        p = np.zeros(C.size, dtype=np.float64)
        p[int(np.argmax(C))] = 1.0
        ex_sum = 1.0
    else:
        ex = np.exp((C - Cmax) / T)
        ex_sum = float(ex.sum())
        p = ex / max(ex_sum, 1e-300)

    hb = hankel_direct(p, b)
    g_b = 2.0 * hb
    G = float(np.dot(g_b, b)) / float(n)
    gt = b * (g_b - G)

    phi = Cmax
    if T > 1e-12 and ex_sum > 0:
        phi += T * math.log(ex_sum)

    gh = np.zeros(h.size, dtype=np.float64)
    if h.size > 1:
        np.add.at(gh, left, (1.0 - w) * gt)
        np.add.at(gh, right, w * gt)
        gh -= np.mean(gh)

    if update_best:
        s_b = float(b.sum())
        if s_b > 0:
            score = 2.0 * n * Cmax / (s_b * s_b)
            update_global_with_norm(b / s_b, score, f"knot_K{h.size}_T{KNOT_T:.3g}")

    return float(phi), gh, b, float(Cmax)


def knot_obj(h):
    val, gh, b, Cmax = knot_core(h, True)
    return val


def knot_grad(h):
    val, gh, b, Cmax = knot_core(h, False)
    return gh


def knot_optimize(a_in, K, T_ratio, maxiter=70, seed=0):
    global KNOT_T
    a = normalize_seq(a_in)
    n = a.size
    K = max(2, int(min(K, max(2, min(n, 128)))))
    knot_prepare(n, K)

    x_old = np.linspace(0.0, 1.0, a.size)
    x_knot = np.linspace(0.0, 1.0, K)
    loga = np.log(np.maximum(a, 1e-12))
    h0 = np.interp(x_knot, x_old, loga)
    h0 = h0 - np.mean(h0)

    rng = np.random.default_rng(seed)
    if seed % 2 == 0 and K > 2:
        h0 = h0 + rng.standard_normal(K) * 0.45
        h0 = h0 - np.mean(h0)

    b0 = n * a
    C0 = fast_convolve(b0)
    Cmax0 = float(C0.max()) if C0.size else 1e-3
    KNOT_T = max(1e-6, Cmax0 * max(1e-4, float(T_ratio)))

    if maxiter is None or maxiter <= 0:
        maxiter = 60 if n <= 1024 else 35

    try:
        minimize(knot_obj,
                 h0,
                 method="L-BFGS-B",
                 jac=knot_grad,
                 options={"maxiter": int(maxiter), "ftol": 1e-12, "gtol": 1e-8})
    except Exception:
        pass

    if GLOBAL_BEST_SEQ is not None:
        return normalize_seq(GLOBAL_BEST_SEQ)
    return a


def propose_candidate(seed=42, budget_s=1800):
    global GLOBAL_BEST_SEQ, GLOBAL_BEST_SCORE

    GLOBAL_BEST_SEQ = None
    GLOBAL_BEST_SCORE = float("inf")

    start_time = time.monotonic()
    total = max(120.0, float(budget_s))
    deadline = start_time + total - 30.0

    # Load previous state-of-the-art if available.
    parent = None
    try:
        parent = height_sequence_1
    except Exception:
        parent = None

    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, None)
        if parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 1e-12:
            consider_sequence(parent, "parent_initial")
    if GLOBAL_BEST_SEQ is None:
        consider_sequence(np.ones(768), "uniform768")

    best_a = normalize_seq(GLOBAL_BEST_SEQ)
    best_score = score_from_norm(best_a)
    best_n = best_a.size

    # Quick clean resampling to a small set of lengths.
    if time.monotonic() < deadline - 350:
        cand_ns = []
        for val in [best_n, best_n // 2, best_n * 2, 512, 768, 1024, 1536, 2048]:
            val = int(val)
            if 256 <= val <= 2048 and val not in cand_ns:
                cand_ns.append(val)
        cand_ns.sort()
        for nc in cand_ns:
            if time.monotonic() > deadline - 320:
                break
            a_try = resample_seq(best_a, nc)
            consider_sequence(a_try, f"resample_{nc}")

    best_a = normalize_seq(GLOBAL_BEST_SEQ)
    best_n = best_a.size

    # Main full-dimensional dual refinement.
    if time.monotonic() < deadline - 400:
        phase1 = min(700.0, max(120.0, (deadline - time.monotonic()) * 0.38))
        a_out = dual_lbfgs(best_a, phase1, seed=int(seed))
        consider_sequence(a_out, "dual_main")

    best_a = normalize_seq(GLOBAL_BEST_SEQ)
    best_n = best_a.size

    # Low-dimensional spline soft optimization.
    if time.monotonic() < deadline - 250:
        phase2_end = min(deadline - 260.0, time.monotonic() + 260.0)
        for K in [16, 32, 64]:
            if time.monotonic() > phase2_end:
                break
            for ratio in [0.20, 0.05, 0.02]:
                if time.monotonic() > phase2_end:
                    break
                a_out = knot_optimize(best_a, K, ratio,
                                      maxiter=65 if best_n <= 1024 else 40,
                                      seed=int(seed) + 997 + K + int(ratio * 1000))
                consider_sequence(a_out, f"knot_K{K}_r{ratio}")

    best_a = normalize_seq(GLOBAL_BEST_SEQ)
    best_n = best_a.size

    # A few smooth random-mixture escapes followed by short dual refinement.
    if time.monotonic() < deadline - 180:
        phase3_end = min(deadline - 190.0, time.monotonic() + 170.0)
        for rr in range(5):
            if time.monotonic() > phase3_end:
                break
            rand = smooth_random_seq(best_n, seed + 4000 + rr * 13, amp=0.65, max_freq=14)
            for mix in [0.65, 0.45, 0.25]:
                if time.monotonic() > phase3_end - 20:
                    break
                a_mix = normalize_seq((1.0 - mix) * best_a + mix * rand)
                limit = min(22.0, max(8.0, phase3_end - time.monotonic() - 10.0))
                a_out = dual_lbfgs(a_mix, limit, seed=int(seed) + 5000 + rr + int(mix * 100))
                consider_sequence(a_out, f"rand_m{mix}_{rr}")

    best_a = normalize_seq(GLOBAL_BEST_SEQ)

    # Final dual polish with whatever remains.
    remaining = deadline - time.monotonic() - 12.0
    if remaining > 25:
        a_out = dual_lbfgs(best_a, remaining, seed=int(seed) + 90909)
        consider_sequence(a_out, "dual_final")

    if GLOBAL_BEST_SEQ is None:
        seq = np.ones(768, dtype=np.float64) / 768.0
    else:
        seq = normalize_seq(GLOBAL_BEST_SEQ)

    # Final safety validation through provided evaluator.
    try:
        fs = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(fs):
            if not np.isfinite(GLOBAL_BEST_SCORE) or fs < GLOBAL_BEST_SCORE:
                GLOBAL_BEST_SCORE = fs
                GLOBAL_BEST_SEQ = seq.copy()
            print(f"FINAL VALIDATED SCORE: {fs:.8f} n={seq.size}")
        else:
            print("Final evaluator returned nonfinite; using uniform fallback.")
            seq = np.ones(768, dtype=np.float64) / 768.0
    except Exception as e:
        print(f"Final evaluator issue: {e}; using computed global best.")

    seq = np.asarray(GLOBAL_BEST_SEQ if GLOBAL_BEST_SEQ is not None else seq,
                     dtype=np.float64).ravel()
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if not np.isfinite(s) or s <= 1e-300:
        seq = np.ones(768, dtype=np.float64) / 768.0
    else:
        seq = seq / s

    return [float(x) for x in seq]
```
