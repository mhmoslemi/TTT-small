def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f * g)[k]."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        lo = i
        hi = min(m - 1, i + n - 1)
        A[lo:hi+1, i] = f[i]
    return A

def solve_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)

    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with guaranteed per-step improvement, subgradient polishing, adaptive line search,
multi-length exploration, and diverse restarts.
Key innovation: Uses the LP (max sum(g) s.t. ||f*g||_inf <= M) to find optimal
descent directions, alternating with subgradient methods for robustness.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f * g)[k] (full linear convolution)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        lo = i
        hi = min(m - 1, i + n - 1)
        A[lo:hi + 1, i] = f[i]
    return A


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None


def lp_refine(f0, time_limit, verbose=False):
    """LP-based iterative refinement with guaranteed improvement per step."""
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        g = solve_lp_direction(f, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive line search
        best_t = 0.1
        best_new_obj = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_try = normalize((1 - t) * f + t * g)
            obj = fast_convolve(f_try).max()
            if obj < best_new_obj:
                best_new_obj = obj
                best_t = t

        f = normalize((1 - best_t) * f + best_t * g)
        obj = fast_convolve(f).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 200 == 0:
            print(f"    LP iter {it}, elapsed={time.monotonic()-t0:.1f}s, obj={best_obj:.12f}")

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13 + n)
            noise = rng.standard_normal(n) * (0.001 + 0.001 * min(no_improve // 100, 3))
            f = project_simplex(best_f + noise)
            no_improve = 0

    return best_f, best_obj


def subgradient_refine(f0, time_limit, top_k=15, temperature=0.00005, lr=0.001, momentum=0.92):
    """Momentum subgradient with peak focusing."""
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest = f_nest / fn

        conv_nest = fast_convolve(f_nest)

        # Softmax-weighted subgradient on top peaks
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0001)
            continue
        f = normalize(f)

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7 + n)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = lr
            cur_temp = temperature
            no_improve = 0

    return best_f, best_obj

            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)

                    if val < best_obj - 1e-14:

def two_level_search(f0, time_limit):
    """Try to improve by adjusting only the ratio between high and low elements."""
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Classify elements as "high" or "low"
    median_val = np.median(f)
    high_mask = f > median_val
    low_mask = ~high_mask

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try scaling high elements up/down and compensating with low elements
        for scale_hi in [0.8, 0.9, 1.1, 1.2]:
            if time.monotonic() - t0 > time_limit:
                break
            f_try = f.copy()
            high_sum = f_try[high_mask].sum()
            low_sum = f_try[low_mask].sum()
            if low_sum < 1e-12:
                continue
            f_try[high_mask] *= scale_hi
            # Compensate to maintain sum=1
            new_high_sum = high_sum * scale_hi
            new_low_sum = 1.0 - new_high_sum
            if new_low_sum < 0:
                continue
            f_try[low_mask] *= new_low_sum / low_sum
            f_try = normalize(f_try)
            val = fast_convolve(f_try).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f_try.copy()
                f = f_try.copy()
                improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def try_and_update(a, best_seq, best_score, label=""):
    """Try a sequence, update best if improved. Returns (new_best_seq, new_best_score)."""
    a = np.maximum(a, 0.0)

    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return a.copy(), score
    return best_seq, best_score


def structured_inits(n, rng):

    alt = np.ones(n)

            a_opt, _ = lp_refine(f_pert, time_limit=t_quick, verbose=False)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"restart_{scale}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
