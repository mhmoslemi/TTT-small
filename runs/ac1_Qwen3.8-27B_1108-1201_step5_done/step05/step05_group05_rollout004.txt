def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v, kind='mergesort')[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Full constraints."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-8,
                           'primal_feasibility_tolerance': 1e-8})
    if res.success:
        return res.x
    return None

def solve_lp_topk(f, M, n, top_k):
    """Solve LP with only top-k binding constraints."""
    conv = fast_convolve(f)
    k_sorted = np.argsort(conv)[::-1][:top_k]
    K = len(k_sorted)
    c = -np.ones(n)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(k_sorted):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

        else:
            stall += 1
            if stall > 50:
                break
        
        # Find top peaks
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        # For each peak, compute which elements contribute
        # and try to reduce them
        improved = False
        for k in peak_indices:
            k = int(k)
            if k == 0:
                continue  # Can't easily reduce the energy peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # The peak value is sum(f[j] * f[k-j] for j in range(js, je+1))
            # To reduce it, we can try reducing f[j] or f[k-j] for large contributors
            contributions = np.array([f[j] * f[k - j] for j in range(js, je + 1)])
            top_contrib = np.argsort(contributions)[::-1][:3]

                j = js + tc
                # Try reducing f[j] slightly

                    if f[j] < delta * 0.5:
                        continue
                    f_trial = f.copy()
                    f_trial[j] -= delta * f[j]
                    if f_trial[j] < 0:
                        f_trial[j] = 0
                    # Redistribute to others
                    others = np.ones(n)
                    others[j] = 0
                    f_trial[others > 0] += f_trial[j] * 0  # Actually let's just renormalize

                    val = fast_convolve(f_trial).max()
                    if val < M - 1e-12:
                        f = f_trial
                        M = val
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 30:
                break
    
    return best_f, best_obj

import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v, kind='mergesort')[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_fast(a):
    """Fast evaluation matching evaluate_sequence semantics."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k=0..2n-2, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def solve_lp_topk(f, M, n, top_k):
    """Solve LP with only top-k most binding constraints."""
    conv = fast_convolve(f)
    k_sorted = np.argsort(conv, kind='mergesort')[::-1][:top_k]
    K = len(k_sorted)
    c = -np.ones(n)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(k_sorted):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def solve_lp_adaptive(f, M, n, top_k=50, use_full=False):

    if use_full or n <= 256:
        return solve_lp_full(f, M, n)
    else:
        return solve_lp_topk(f, M, n, top_k)

    it = 0
    m_exp = 1.0

                m_exp = 1.0
            no_improve += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Adaptive line search: try multiple step sizes

        best_val = best_obj
        # Start with finer steps near the previous good step

            fs = f_trial.sum()
            if fs < 1e-15:

            f_trial = f_trial / fs
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M_new = conv.max()

                best_f = f.copy()
                no_improve = 0

            no_improve += 1

        if no_improve > 25:

            scale = 0.002 + 0.004 * min(no_improve // 40, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

def slp_papr(f0, time_limit, max_iter=2000, threshold_clip=0.0):
    """
    SLP-based PAPR reduction adapted for non-negative sequences.
    Uses circular autocorrelation as a proxy, then projects back.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    fft_size = 1 << (n - 1).bit_length()
    if fft_size < 2 * n:
        fft_size <<= 1

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        # Compute circular autocorrelation via FFT
        F = np.fft.fft(f, fft_size)
        power = np.abs(F) ** 2
        autocorr = np.fft.ifft(power).real[:n]

        # Peak reduction: clip peaks above threshold
        tau = autocorr.max() * 0.85
        autocorr_clip = autocorr.copy()
        peak_mask = autocorr_clip > tau
        if np.any(peak_mask):
            excess = autocorr_clip[peak_mask] - tau
            autocorr_clip[peak_mask] = tau

        # Update in frequency domain
        delta = np.fft.fft(autocorr - autocorr_clip, fft_size)
        F_new = F * (1.0 - delta / np.maximum(np.abs(F), 1e-15))
        # Ensure |F_new| = |F| (maintain magnitude)
        mag = np.abs(F)
        phase = np.angle(F_new)
        F_new = mag * np.exp(1j * phase)

        f_new = np.fft.ifft(F_new).real[:n]
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / fs

        # Evaluate with linear convolution
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def peak_shave(f0, time_limit, n_peaks=5, n_trials=8):
    """
    Direct peak shaving: identify top autocorrelation peaks and
    try local modifications to reduce them.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

        # Find top peaks (excluding index 0 and 2n-2 which are just a[0]^2 and a[n-1]^2)
        peak_indices = np.argsort(conv, kind='mergesort')[::-1][:n_peaks]

        improved_this_round = False
        for k in peak_indices:
            k = int(k)
            if k <= 0 or k >= 2 * n - 2:
                continue

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js >= je:
                continue

            # Compute contributions at this peak
            js_arr = np.arange(js, je + 1)
            contribs = f[js_arr] * f[k - js_arr]
            
            # Try reducing the top contributors
            top_idx = np.argsort(contribs, kind='mergesort')[::-1][:4]
            for ti in top_idx:
                j = js + ti
                if f[j] < 1e-8:
                    continue

                for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                    f_trial = f.copy()
                    f_trial[j] *= (1.0 - frac)
                    f_trial = np.maximum(f_trial, 0.0)
                    fsum = f_trial.sum()
                    if fsum < 1e-15:
                        continue
                    f_trial = f_trial / fsum
                    val = fast_convolve(f_trial).max()
                    if val < M - 1e-12:
                        f = f_trial
                        M = val
                        improved_this_round = True
                        break
                if improved_this_round:
                    break

        if not improved_this_round:
            # Try increasing some elements to "spread out" the correlation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            for _ in range(n_trials):
                j1 = rng.integers(0, n)
                j2 = rng.integers(0, n)
                if j1 == j2:
                    continue
                for delta in [0.005, 0.01, 0.02]:
                    f_trial = f.copy()
                    amount = f[j1] * delta
                    if amount < 1e-10:
                        continue
                    f_trial[j1] -= amount
                    f_trial[j2] += amount
                    if f_trial[j1] < 0:
                        continue
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < M - 1e-12:
                        f = f_trial
                        M = val
                        improved_this_round = True
                        break
                if improved_this_round:
                    break

        if not improved_this_round:
            stall += 5
            if stall > 50:
                break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, multipliers=None):
    """Coordinate descent with multi-scale local search."""
    if multipliers is None:
        multipliers = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.0]

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                remaining = 1.0 - new_val
                if remaining <= 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr=0.001, mom=0.94, top_k=30, temp=5e-5):

        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)

    no_improve = 0
    cur_lr = lr
    cur_temp = temp

            break

        scaled = pv / cur_temp
        scaled -= scaled.max()

def interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length, maintaining normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def make_diverse_init(base, n, seed=0):
    """Create diverse initializations from a base sequence."""
    rng = np.random.default_rng(seed)
    inits = []
    names = []

    # Small noise
    f1 = project_simplex(base + rng.standard_normal(n) * 0.002)
    inits.append(f1)
    names.append("noise_s")

    # Medium noise
    f2 = project_simplex(base + rng.standard_normal(n) * 0.005)
    inits.append(f2)
    names.append("noise_m")

    # Reverse blend
    f3 = project_simplex(0.7 * base + 0.3 * base[::-1])
    inits.append(f3)
    names.append("rev_blend")

    # Flat blend
    f4 = project_simplex(0.85 * base + 0.15 * np.ones(n) / n)
    inits.append(f4)
    names.append("flat_blend")

    # Gaussian window
    x = np.arange(n) / n - 0.5
    f5 = project_simplex(np.maximum(np.exp(-4.0 * x * x * n), 0))
    # Mix with base
    f5 = project_simplex(0.6 * base + 0.4 * f5)
    inits.append(f5)
    names.append("gauss_mix")

    # Raised cosine
    f6 = project_simplex(np.maximum(0.5 * (1 + np.cos(2 * np.pi * (x + 0.5))), 0))
    f6 = project_simplex(0.7 * base + 0.3 * f6)
    inits.append(f6)
    names.append("cosine_mix")

    return inits, names


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-Frank-Wolfe (full constraints), SLP-PAPR,
    peak shaving, coordinate descent, subgradient polish, and diverse restarts.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = evaluate_fast(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_fast(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = evaluate_fast(parent)
        print(f"Using uniform init: n={n_init}, score={best_score:.6f}")

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, score={best_score:.6f}")
    print(f"Time budget: {budget_s - 20:.0f}s")

    # === Phase 1: LP-FW with full constraints on parent ===
    print("\n=== Phase 1: LP-FW (full LP) on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(300, rem * 0.28)
    if t1 > 30:
        print(f"  Time allocated: {t1:.0f}s")
        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, use_full=True, verbose=True)
        try_seq(a_opt, "lpfw_full")

    # === Phase 2: SLP-PAPR ===
    print(f"\n=== Phase 2: SLP-PAPR (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t2 = min(rem * 0.12, 120)
            print(f"  Time allocated: {t2:.0f}s")
            a_opt, obj = slp_papr(best_seq, time_limit=t2)
            try_seq(a_opt, "slp_papr")

    # === Phase 3: LP-FW with top-k (different constraint set) ===
    print(f"\n=== Phase 3: LP-FW (top-k=30) (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t3 = min(rem * 0.12, 120)
            print(f"  Time allocated: {t3:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3, use_full=False, top_k=30)
            try_seq(a_opt, "lpfw_topk30")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t4 = min(rem * 0.12, 150)
            print(f"  Time allocated: {t4:.0f}s")
            a_opt, obj = peak_shave(best_seq, time_limit=t4, n_peaks=5, n_trials=10)
            try_seq(a_opt, "peak_shave")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [640, 896, 1024]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.12, 100)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc,
                                        use_full=(target_n <= 1024), top_k=40)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        inits, names = make_diverse_init(best_seq, n_best, seed=seed)

        for i, (name, f_init) in enumerate(zip(names, inits)):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem / max(1, len(inits) - i), 80)
            print(f"  [{name}] t={t_alloc:.0f}s")
            use_f = (n_best <= 1024)
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc,
                                        use_full=use_f, top_k=35)
            try_seq(a_opt, f"restart_{name}")

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t7 = min(rem * 0.2, 200)
            print(f"  Time allocated: {t7:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t7)
            try_seq(a_opt, "coord_desc")

            # Second pass
            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt2, obj2 = coordinate_descent(a_opt, time_limit=min(rem * 0.3, 100))
                try_seq(a_opt2, "coord_desc2")

    # === Phase 8: Subgradient polish ===
    print(f"\n=== Phase 8: Subgradient Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            t8 = min(rem * 0.15, 120)
            print(f"  Time allocated: {t8:.0f}s")
            a_opt, obj = subgradient_polish(best_seq, time_limit=t8,
                                            lr=0.001, mom=0.94,
                                            top_k=max(20, n_best // 8),
                                            temp=5e-5)
            try_seq(a_opt, "sg_polish")

    # === Phase 9: Final combined polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            # LP-FW on best
            t9a = min(rem * 0.3, 80)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t9a, use_full=True)
            try_seq(a_opt, "final_lpfw")

            # Peak shave on result
            rem = deadline - time.monotonic()
            if rem > 10:
                a_opt2, _ = peak_shave(a_opt, time_limit=min(rem * 0.3, 40), n_peaks=3)
                try_seq(a_opt2, "final_shave")

            # Tiny perturbations
            rem = deadline - time.monotonic()
            rng = np.random.default_rng(99999)
            for i in range(4):
                rem = deadline - time.monotonic()
                if rem < 8:
                    break
                noise = rng.standard_normal(n_best) * (0.001 + 0.0005 * i)
                f_pert = project_simplex(best_seq + noise)
                t_pert = min(rem * 0.2, 25)
                a_opt3, _ = lp_fw_optimize(f_pert, time_limit=t_pert, use_full=True)
                try_seq(a_opt3, f"final_pert_{i}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = evaluate_fast(best_seq)

    # Final verification
    print(f"\nVerifying final answer with evaluate_sequence...")
    fs = evaluate_sequence([float(x) for x in best_seq])
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
