So H_LSE[i,j] = 2*p[i+j] + 4*beta*(h[i,j] - s[i]*s[j]) ✓

Now, h[i,j] = sum_k p[k] * f[k-i] * f[k-j] = (B^T @ B)[i,j] where B[k,i] = f[k-i]*sqrt(p[k]) (with zero-padding).

To compute B efficiently:
- B is (2n-1) x n
- B[k, i] = f[k-i] * sqrt(p[k]) if 0 <= k-i < n, else 0

For each row k, B[k, :] = sqrt(p[k]) * [0...0, f[k-(n-1)], ..., f[k-1], f[k], f[k+1], ..., f[k-0], 0...0]
(where the non-zero entries are f[k-i] for i in [max(0,k-n+1), min(n-1, k)])

This is a "sliding window" of f, scaled by sqrt(p[k]). We can build this matrix using a Toeplitz-like structure.

In NumPy, for n=768, B is 1535 x 768. Building it explicitly takes O(n * 2n) = O(2n^2) memory and time, which is fine.

Then h = B.T @ B is a 768x768 matrix multiply of (768 x 1535) @ (1535 x 768), which is about 908M flops. With BLAS, this is about 0.5-1 second.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Newton's method with
barrier-augmented LSE objective and continuation over smoothing parameter.
Exploits Hankel structure of Hessian for efficient computation.
"""

import numpy as np
import time


def fast_autocorr(f):
    """Linear autocorrelation via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    ac = np.fft.irfft(F * F, fft_size)[:size]
    return np.maximum(ac, 0.0)  # should be non-negative for non-negative f, but clamp for safety


def compute_quantities(f, beta):
    """Compute all quantities needed for Newton step.
    Returns: ac, p (softmax), s (cross-corr), grad, H (Hessian of LSE)
    """
    n = len(f)
    ac = fast_autocorr(f)
    
    # Softmax weights
    scaled = beta * ac
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    p = exp_s / exp_s.sum()
    
    # Cross-correlation: s[i] = sum_k p[k] * f[k-i]
    # Using the Toeplitz matrix approach
    idx = np.arange(n)
    T_idx = idx[:, None] + idx[None, :]  # (n, n)
    T = p[T_idx]  # T[i,j] = p[i+j]
    s = T @ f  # s[i] = sum_j p[i+j] * f[j]
    
    # Gradient of LSE
    grad = 2.0 * s
    
    # h[i,j] = sum_k p[k] * f[k-i] * f[k-j]
    # Compute via B^T @ B where B[k,i] = sqrt(p[k]) * f[k-i] (zero-padded)
    sqrt_p = np.sqrt(p)
    # Build B: (2n-1) x n
    # B[k, i] = sqrt_p[k] * f[k-i] if 0 <= k-i < n, else 0
    # For row k: non-zero entries at i in [max(0, k-n+1), min(n-1, k)]
    B = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        B[k, lo:hi + 1] = sqrt_p[k] * f[lo:hi + 1]
    
    h = B.T @ B  # (n, n)
    
    # Hessian of LSE: H[i,j] = 2*p[i+j] + 4*beta*(h[i,j] - s[i]*s[j])
    H = 2.0 * T + 4.0 * beta * (h - np.outer(s, s))
    
    return ac, p, s, grad, H

    n = len(f)
    ac, p, s, grad, H = compute_quantities(f, beta)

    barrier_grad = -mu / f  # d(-mu * sum(log f))/df = -mu/f
    barrier_H = mu / (f * f)  # d²(-mu * sum(log f))/df² = mu/f²
    
    total_grad = grad + barrier_grad
    total_H = H + np.diag(barrier_H)
    
    # KKT system: [H  1; 1^T 0] [dx; dlambda] = [-total_grad; 0]
    # Size: (n+1) x (n+1)
    KKT = np.zeros((n + 1, n + 1))
    KKT[:n, :n] = total_H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n + 1)
    rhs[:n] = -total_grad
    
    try:
        # Try Cholesky for PD check
        # Add small regularization for numerical stability
        KKT_reg = KKT + 1e-12 * np.eye(n + 1)
        sol = np.linalg.solve(KKT_reg, rhs)
        dx = sol[:n]
    except np.linalg.LinAlgError:
        # Fallback: use gradient direction
        dx = -total_grad
        dx -= np.mean(dx)  # project to sum=0 subspace
        dx = dx / np.linalg.norm(dx) * 0.01
    
    # Line search
    best_alpha = 0.0
    best_f = f.copy()
    best_obj = None
    cur_obj = (1.0 / beta) * np.log(np.sum(np.exp(beta * ac - beta * ac.max()))) + mu * np.sum(-np.log(f))
    # Actually, for line search, just use the actual objective (max of autocorrelation)
    cur_max = ac.max()
    
    alpha = 0.5
    for _ in range(max_ls):
        f_new = f + alpha * dx
        if np.any(f_new <= 0):
            alpha *= 0.5
            continue
        f_new = f_new  # already sums to 1 (since sum(dx) = 0 from KKT)
        # Check sum
        s_new = f_new.sum()
        if abs(s_new - 1.0) > 1e-6:
            f_new = f_new / s_new
        ac_new = fast_autocorr(f_new)
        obj_new = ac_new.max()
        if obj_new < cur_max * (1.0 - 1e-10):
            best_alpha = alpha
            best_f = f_new.copy()
            best_obj = obj_new
            break
        alpha *= 0.5
    
    if best_alpha > 0:
        return best_f, best_obj, True
    else:
        return f, cur_max, False

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Newton's method with
LSE-smoothed objective and continuation over beta.
Exploits Hankel structure of Hessian for O(n^2) computation.
Combines with projected subgradient fallback and multi-length exploration.
"""

import numpy as np
import time


def fast_autocorr(f):
    """Linear autocorrelation via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    ac = np.fft.irfft(F * F, fft_size)[:size]
    return np.maximum(ac, 0.0)


def compute_t_mat(n, p):
    """Build the Toeplitz matrix T[i,j] = p[i+j] of shape (n,n)."""
    idx = np.arange(n)
    T_idx = idx[:, None] + idx[None, :]
    return p[T_idx]


def compute_cross_corr(f, p, n):
    """s[i] = sum_j p[i+j] * f[j] using Toeplitz matrix."""
    idx = np.arange(n)
    T_idx = idx[:, None] + idx[None, :]
    T = p[T_idx]
    return T @ f


def compute_h_matrix(f, p, n):
    """Compute h[i,j] = sum_k p[k] * f[k-i] * f[k-j] via B^T @ B."""
    sqrt_p = np.sqrt(p)
    size = 2 * n - 1
    B = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            B[k, lo:hi + 1] = sqrt_p[k] * f[lo:hi + 1]
    return B.T @ B


def lse_gradient(f, beta):
    """Compute gradient of LSE objective. Returns (ac, p, grad)."""
    n = len(f)
    ac = fast_autocorr(f)
    scaled = beta * ac
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    p = exp_s / exp_s.sum()
    s = compute_cross_corr(f, p, n)
    grad = 2.0 * s
    return ac, p, grad


def lse_hessian(f, p, beta, grad):
    """Compute Hessian of LSE: H[i,j] = 2*p[i+j] + 4*beta*(h[i,j] - s[i]*s[j])."""
    n = len(f)
    s = grad / 2.0
    T = compute_t_mat(n, p)
    h = compute_h_matrix(f, p, n)
    H = 2.0 * T + 4.0 * beta * (h - np.outer(s, s))
    # Symmetrize for numerical stability
    H = 0.5 * (H + H.T)
    return H


def projected_newton_step(f, beta, max_ls=50):
    """One projected Newton step on the simplex with LSE objective.
    Returns (f_new, obj_new, success)."""
    n = len(f)
    ac, p, grad = lse_gradient(f, beta)
    cur_obj = ac.max()

    # Compute Hessian
    try:
        H = lse_hessian(f, p, beta, grad)
    except Exception:
        # Fallback to gradient step
        d = -grad
        d -= d.mean()
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            return f, cur_obj, False
        d = d / d_norm
        return _line_search(f, d, cur_obj, max_ls)

    # Newton direction: solve H @ d = -grad
    # Add regularization for stability
    reg = 1e-8 * np.trace(H) / n
    H_reg = H + reg * np.eye(n)

    try:
        d = np.linalg.solve(H_reg, -grad)
    except np.linalg.LinAlgError:
        # Fallback to gradient
        d = -grad

    # Project onto sum=0 subspace
    d -= d.mean()

    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return f, cur_obj, False

    return _line_search(f, d, cur_obj, max_ls)


def _line_search(f, direction, cur_obj, max_ls=50):
    """Backtracking line search along a direction on the simplex."""
    n = len(f)
    # Find max feasible step
    neg_mask = direction < 0
    if neg_mask.any():
        alpha_max = np.min(-f[neg_mask] / direction[neg_mask]) * 0.95
    else:
        alpha_max = 1.0
    alpha_max = min(alpha_max, 1.0)

    best_alpha = 0.0
    best_f = f.copy()
    best_obj = cur_obj

    alpha = min(alpha_max * 0.8, 1.0)
    for _ in range(max_ls):
        if alpha < 1e-12:
            break
        f_new = f + alpha * direction
        if np.any(f_new < -1e-12):
            alpha *= 0.5
            continue
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        f_new = f_new / s
        ac_new = fast_autocorr(f_new)
        obj_new = ac_new.max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            best_alpha = alpha
            # Try to go a bit further
            if alpha < alpha_max * 0.9:
                alpha *= 1.3
            else:
                break
        else:
            alpha *= 0.5

    return best_f, best_obj, best_alpha > 0


def directed_subgrad_step(f, n_peaks=5, lr=None, max_ls=50):
    """Projected subgradient step targeting top peaks."""
    n = len(f)
    ac = fast_autocorr(f)
    cur_obj = ac.max()

    # Top peaks
    top_k = min(n_peaks, len(ac))
    peaks = np.argpartition(ac, -top_k)[-top_k:]
    # Weight by height (emphasize higher peaks)
    w = ac[peaks] - ac[peaks].min() + 1e-10
    w = w ** 3  # Sharpen weights
    w /= w.sum()

    # Weighted gradient
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += w[idx] * 2.0 * f[k_int - ms]
            # Also the "other" contribution (symmetric)
            if k_int - ms != ms:
                grad[k_int - ms] += w[idx] * 2.0 * f[ms]
            # Actually this double counts. Let me just use the formula:
            # d g[k]/d f[i] = 2*f[k-i] if 0<=k-i<n
            # So grad[i] += 2*w*f[k-i] for valid k-i

    # Redo properly
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        k_int = int(k)
        for i in range(n):
            ki = k_int - i
            if 0 <= ki < n:
                grad[i] += w[idx] * 2.0 * f[ki]

    grad /= max(np.linalg.norm(grad), 1e-15)

    if lr is None:
        lr = 1.0 / n

    d = -grad
    d -= d.mean()
    return _line_search(f, d, cur_obj, max_ls)

def gradient_phase(f0, time_limit, beta_start=20.0, beta_end=8000.0):
    """Gradient-based optimization (fallback/supplement)."""
    f = f0.copy() / f0.sum()
    n = len(f)

    best_obj = fast_autocorr(f).max()
    t_start = time.monotonic()

    v = np.zeros(n)

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad_normed = grad / g_norm

        f_new = np.maximum(f + v, 0.0)
        s = f_new.sum()

        ac_new = fast_autocorr(f_new)

                v = np.zeros(n)
                lr = 2.0 / n
                beta = min(beta * 1.5, beta_end)

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, sum={p.sum():.4f}", flush=True)
            try_eval(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(448) / 448
        try_eval(parent, best_tracker)

    n_par = len(parent)
    print(f"Starting optimization, target n={n_par}", flush=True)

    # Phase 1: Newton optimization on parent length
    print("\n=== Phase 1: Newton on parent length ===", flush=True)
    t1_end = deadline - 600
    t_alloc = min(t1_end - time.monotonic(), 700)
    if t_alloc > 60:
        print(f"  Newton: n={n_par}, t={t_alloc:.0f}s", flush=True)
        f_opt, obj = newton_phase(parent, time_limit=t_alloc,
                                   beta_start=5.0, beta_end=8000.0, beta_factor=1.6)
        try_eval(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # Phase 2: Gradient polishing
    print("\n=== Phase 2: Gradient polishing ===", flush=True)
    t2_end = deadline - 400
    t_alloc = min(t2_end - time.monotonic(), 300)
    if t_alloc > 30:
        print(f"  Gradient: n={len(current)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = gradient_phase(current, time_limit=t_alloc,
                                   beta_start=30.0, beta_end=10000.0)
        try_eval(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Phase 3: Newton polishing with higher starting beta
    print("\n=== Phase 3: High-beta Newton polish ===", flush=True)
    t3_end = deadline - 200
    t_alloc = min(t3_end - time.monotonic(), 350)
    if t_alloc > 30:
        print(f"  Newton (high beta): n={len(current)}, t={t_alloc:.0f}s", flush=True)
        f_opt, obj = newton_phase(current, time_limit=t_alloc,
                                   beta_start=50.0, beta_end=20000.0, beta_factor=1.5)
        try_eval(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Phase 4: Directed subgradient refinement
    print("\n=== Phase 4: Directed subgradient ===", flush=True)
    t4_end = deadline - 100
    t_alloc = min(t4_end - time.monotonic(), 150)
    if t_alloc > 20:
        print(f"  Directed subgrad: n={len(current)}, t={t_alloc:.0f}s", flush=True)
        f = current.copy()
        n = len(f)
        best_f = f.copy()
        best_obj = fast_autocorr(f).max()
        t_s = time.monotonic()
        it = 0
        while time.monotonic() - t_s < t_alloc:
            it += 1
            lr = max(0.02 / n, 0.001 / n * max(1, 100 // it))
            f_new, obj, success = directed_subgrad_step(f, n_peaks=5, lr=lr)
            if success and obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_new.copy()
                f = f_new
            elif it % 50 == 0:
                # Small perturbation
                f = best_f + rng.standard_normal(n) * (0.002 / n)
                f = np.maximum(f, 0)
                f /= f.sum()
        try_eval(best_f, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current

    # Phase 5: Try nearby lengths
    print("\n=== Phase 5: Nearby lengths ===", flush=True)
    n_best = len(current)
    offsets = [64, -64, 32, -32, 128, -128, 16, -16]
    for off in offsets:
        n_try = n_best + off
        if n_try < 200 or n_try > 2000:
            continue
        t_remaining = deadline - time.monotonic() - 20
        if t_remaining < 40:
            break
        t_alloc = min(80, t_remaining * 0.35)
        if t_alloc < 20:
            break
        f_init = interpolate_seq(current, n_try)
        print(f"  Trying n={n_try}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = newton_phase(f_init, time_limit=t_alloc,
                                 beta_start=10.0, beta_end=5000.0, beta_factor=1.7)
        try_eval(f_opt, best_tracker)

    # Phase 6: Final micro-polish on best
    print("\n=== Phase 6: Final micro-polish ===", flush=True)
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = max(10, deadline - time.monotonic() - 5)
    if t_alloc > 15:
        print(f"  Final Newton: n={len(current)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = newton_phase(current, time_limit=t_alloc,
                                 beta_start=100.0, beta_end=50000.0, beta_factor=1.3)
        try_eval(f_opt, best_tracker)

    # Final
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}", flush=True)
            return [float(x) for x in final_seq]

    print("WARNING: Fallback to parent", flush=True)
    return [float(x) for x in parent]
