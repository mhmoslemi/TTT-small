                          method='highs', options={'presolve': True, 'time_limit': max(1.0, time_limit - (time.monotonic()-t0) - 1.0)})

            else:
                continue
        except Exception:
            continue

        if gs < 1e-15:
            break

            no_improve += 1
            if no_improve > 50:
                break
            continue

        best_alpha = 0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute sum_j p[j+m] * a[j] for all m (valid correlation)."""
    return np.correlate(p, a, mode='valid')

        # Preserve Hermitian symmetry
        mag_new[N - np.arange(1, (N + 1) // 2)] = mag_new[np.arange(1, (N + 1) // 2)]

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(np.ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            improved = False
            for t in [0.1, 0.25, 0.5, 0.75]:
                blend = (1.0 - t) * a + t * a_new
                blend = np.maximum(blend, 0)
                sb = blend.sum()
                if sb < 1e-15:
                    continue
                blend = blend / sb
                ob = fast_convolve(blend).max()
                if ob < best_obj - 1e-15:
                    best_obj = ob
                    best_a = blend.copy()
                    a = blend
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1

        # Adapt alpha
        if no_improve > 0:
            alpha = min(0.999, alpha + 0.0005)
        if no_improve > 500:
            rng2 = np.random.default_rng(it * 104729)
            a = project_simplex(best_a + rng2.standard_normal(n) * 0.004)
            alpha = alpha_init
            no_improve = 0

    return best_a, best_obj


def momentum_subgrad(a0, time_limit, T0=0.0003, lr0=0.015):
    """
    Heavy-ball momentum smoothed subgradient descent.
    Uses softmax-weighted peaks for smooth surrogate of max.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    T = T0
    lr = lr0
    momentum = 0.92

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Momentum update with backtracking
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            # Try smaller step
            v_half = momentum * v - (lr * 0.5) * grad
            a_half = project_simplex(a + v_half)
            obj_half = fast_convolve(a_half).max()
            if obj_half < best_obj - 1e-15:
                v = v_half
                a_new = a_half
                best_obj = obj_half
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

        a = a_new

        # Adaptive hyperparameters
        if no_improve > 20:
            lr *= 0.985
        if no_improve > 40:
            momentum = min(0.97, momentum + 0.003)
        if no_improve > 60:
            T *= 1.03
        if no_improve > 300:
            rng2 = np.random.default_rng(it * 48271 + 3)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 300) / 300.0)
            a = project_simplex(best_a + rng2.standard_normal(n) * scale)
            v = np.zeros(n)
            T = T0
            lr = lr0
            momentum = 0.92
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(a0, time_limit):
    """Frank-Wolfe with LP oracle and line search."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = M

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Build LP with all constraints
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            lp_time = max(1.0, remaining - 2.0)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': lp_time})
            if not res.success:
                continue
            g = np.maximum(res.x, 0)
        except Exception:
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * a + mid * g
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            rng2 = np.random.default_rng(it * 7919 + 7)
            a = project_simplex(best_a + rng2.standard_normal(n) * 0.002)
            M = fast_convolve(a).max()
            no_improve = 0

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Initialize with flat-magnitude spectrum and random phases."""
    N = 2 * n - 1
    n_freq = N // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    phases[0] = 0.0
    if N % 2 == 1 and n_freq > 1:
        phases[-1] = 0.0

    A = np.exp(1j * phases)
    a = np.fft.irfft(A, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def coordinate_polish(a0, time_limit):
    """Local coordinate descent on the peak-affecting variables."""
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Variables affecting peak k_peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = set(range(js, je + 1))

        # Try random perturbations on affected vars
        improved = False
        rng2 = np.random.default_rng(it * 314159 + k_peak)
        for attempt in range(10):
            scale = 0.001 / (1 + attempt * 0.5)
            pert = rng2.standard_normal(n) * scale
            for j in range(n):
                if j not in affected:
                    pert[j] = 0
            candidate = project_simplex(a + pert)
            c_val = fast_convolve(candidate).max()
            if c_val < best_obj - 1e-15:
                a = candidate
                best_obj = c_val
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break

    return best_a, best_obj
