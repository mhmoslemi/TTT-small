The constraint sum(a) = 1 means A(0) = 1 (DC component).
The constraint a >= 0 means IFFT(A) >= 0.

In the frequency domain, we want |A(ω)|^2 to be as flat as possible (so that its IFFT is as flat as possible), subject to the constraint that the IFFT of A (not |A|^2) is non-negative and has DC = 1.

This is a really interesting perspective! The "flat spectrum" sequences (like Barker, Golay, etc.) have flat |A(ω)|, but we also need a >= 0.

For the non-negative constraint, a sufficient condition is that A(ω) = B(ω)^2 for some B (then a = IFFT(B^2) and if B is chosen so that a >= 0). But this is very restrictive.

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Smoothed gradient descent with adaptive temperature, active-set
tracking, coordinate descent polishing, and diverse multi-restart.
Uses FFT-based convolution for efficiency.

import numpy as np
import time

def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_and_weights(conv, tau):
    """Compute smooth max and softmax weights."""
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    shifted -= shifted.max()  # numerical stability
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    smax = cmax + tau * np.log(sum_exp)
    weights = exp_vals / sum_exp
    return smax, weights

def conv_gradient(conv_weights, a):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Result: 2 * C where C[j] = sum_l a[l]*w[j+l] (cross-correlation).
    """
    n = len(a)
    N = 1 << (n + len(conv_weights) - 1).bit_length()
    A_f = np.fft.rfft(a, N)
    W_f = np.fft.rfft(np.zeros(N))
    W_f[:len(conv_weights)] = conv_weights
    # Cross-correlation: C[j] = sum_l a[l] * w[j+l]
    # = ifft(fft(a) * conj(fft(w)))
    C = np.fft.irfft(A_f * np.conj(W_f), N)[:n]
    return 2.0 * C

def smooth_gradient_descent(f0, time_limit, tau_init=0.01, tau_decay=0.998,
                             lr_init=0.01, lr_decay=0.999, momentum=0.9,
                             restart_interval=200):
    """
    Projected gradient descent on the smoothed objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_autocorr(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)  # momentum
    tau = tau_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        smax, weights = smooth_max_and_weights(conv, tau)
        grad = lr * n * conv_gradient(weights, f)
        
        v = momentum * v - grad
        f_new = project_simplex(f + v)
        conv = fast_autocorr(f_new)
        obj = conv.max()
        f = f_new

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameter adjustment
        if no_improve > restart_interval:
            # Restart from best with perturbation
            rng = np.random.default_rng(it * 1000003 + 7)
            noise_scale = 0.005 + 0.01 * min(no_improve // restart_interval, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            conv = fast_autocorr(f)
            tau = tau_init * 0.5
            lr = lr_init
            no_improve = 0
        else:
            tau *= tau_decay
            lr = max(lr * lr_decay, 1e-6)

    return best_f, best_obj

def active_set_subgradient(f0, time_limit, top_k=20, lr=0.005, momentum=0.92):
    """
    Subgradient method using only the top-k active constraints (largest conv values).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_autocorr(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find active constraints (top-k largest conv values)
        k_act = min(top_k, len(conv))
        peak_indices = np.argpartition(conv, -k_act)[-k_act:]
        peak_vals = conv[peak_indices]
        
        # Softmax weights over active peaks
        scaled = peak_vals / (peak_vals.max() * 0.1 + 1e-10)
        exp_vals = np.exp(scaled - scaled.max())
        weights_active = exp_vals / exp_vals.sum()
        
        # Build full weight vector
        weights = np.zeros(len(conv))
        for i, idx in enumerate(peak_indices):
            weights[int(idx)] = weights_active[i]
        
        # Subgradient
        sg = n * conv_gradient(weights, f)
        
        v = momentum * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_autocorr(f)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr = max(lr * 0.999, 1e-6)

        if no_improve > 300:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            lr = 0.005
            no_improve = 0
            conv = fast_autocorr(f)

    return best_f, best_obj

def coordinate_descent(f0, time_limit, multipliers=None):
    """Cycle coordinate descent with multiple step sizes per coordinate."""
    if multipliers is None:
        multipliers = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 1.8, 2.2, 3.0]
    
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_autocorr(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        pass_start = time.monotonic()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - pass_start > time_limit * 0.3:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                remaining = 1.0 - new_val
                if remaining <= 1e-12:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()

                val = fast_autocorr(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    return f_new

def random_init(n, seed):
    """Generate a random non-negative sequence (normalized)."""
    rng = np.random.default_rng(seed)
    # Mix different distributions
    method = seed % 5
    if method == 0:
        f = rng.uniform(0, 1, n)
    elif method == 1:
        f = rng.exponential(1, n)
    elif method == 2:
        t = np.arange(n) / n
        f = np.exp(-2 * (t - 0.5) ** 2)
    elif method == 3:
        f = np.abs(rng.standard_normal(n))
    else:
        f = np.ones(n) / n + 0.1 * rng.standard_normal(n)
        f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f

    w_full[:len(conv_weights)] = conv_weights
    W_f = np.fft.rfft(w_full, N)
    C = np.fft.irfft(A_f * np.conj(W_f), N)[:n]
    return 2.0 * C

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses continuation method (decreasing smoothing), exact FFT-based gradients,
cross-correlation gradient computation, Nesterov acceleration, paired coordinate
moves, and diverse multi-restart strategy.

import numpy as np
import time


def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_weights(conv, tau):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    smax = cmax + tau * np.log(sum_exp)
    weights = exp_vals / sum_exp
    return smax, weights


def conv_gradient(conv_weights, a):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Result: 2*C where C[j] = sum_m a[m]*w[m+j] (cross-correlation).
    Uses FFT for efficiency.
    """
    n = len(a)
    w_len = len(conv_weights)
    # Need N >= n + w_len - 1 for linear cross-correlation
    N = 1 << (n + w_len - 1).bit_length()
    w_full = np.zeros(N)
    w_full[:w_len] = conv_weights
    A_f = np.fft.rfft(a, N)
    W_f = np.fft.rfft(w_full, N)
    C = np.fft.irfft(A_f * np.conj(W_f), N)[:n]
    return 2.0 * C


def nesterov_smoothed_descent(f0, time_limit, tau_init=0.02, tau_end=0.001,
                               lr_init=0.05, momentum_init=0.85,
                               n_continuation=5, restart_every=200):
    """
    Nesterov-accelerated projected gradient on smoothed objective with
    continuation (decreasing tau).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_autocorr(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    total_time = time_limit

    # Continuation schedule
    tau_values = np.geomspace(tau_init, tau_end, n_continuation)
    phase_time = total_time / n_continuation

    for phase in range(n_continuation):
        tau = tau_values[phase]
        phase_deadline = t0 + (phase + 1) * phase_time
        v = np.zeros(n)
        lr = lr_init * (0.7 ** phase)
        mu = momentum_init + 0.02 * phase
        y = f.copy()
        no_improve = 0
        it = 0

        while time.monotonic() < phase_deadline:
            it += 1

            # Nesterov step: evaluate at y
            conv_y = fast_autocorr(y)
            smax, weights = smooth_max_and_weights(conv_y, tau)
            grad = n * conv_gradient(weights, y)

            # Update
            v_new = mu * v - lr * grad
            x_new = project_simplex(y + v_new)
            v = v_new

            # Nesterov extrapolation
            y = x_new + mu * (x_new - f)
            y = project_simplex(y)
            f = x_new

            conv = fast_autocorr(f)
            obj = conv.max()

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > restart_every:
                rng = np.random.default_rng(it * 1000033 + phase * 7)
                scale = 0.005 + 0.005 * min(no_improve // restart_every, 4)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                y = f.copy()
                v = np.zeros(n)
                lr = lr_init * (0.7 ** phase)
                no_improve = 0
                conv = fast_autocorr(f)

            # Adaptive lr
            if it % 50 == 0 and no_improve > 20:
                lr = max(lr * 0.9, 1e-5)

    return best_f, best_obj


def active_set_descent(f0, time_limit, top_k_range=(10, 50), lr=0.01,
                       momentum=0.9, adapt_k=True):
    """
    Projected gradient using top-k active constraints with adaptive k.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_autocorr(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_k = top_k_range[0]

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_act = min(cur_k, len(conv))
        peak_idx = np.argpartition(conv, -k_act)[-k_act:]
        peak_vals = conv[peak_idx]

        # Temperature-adaptive weights
        temp = max(peak_vals.max() * 0.05, 1e-6)
        exp_v = np.exp((peak_vals - peak_vals.max()) / temp)
        w_active = exp_v / exp_v.sum()

        weights = np.zeros(len(conv))
        weights[peak_idx] = w_active

        grad = n * conv_gradient(weights, f)
        v = momentum * v - lr * grad
        f = project_simplex(f + v)
        conv = fast_autocorr(f)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if adapt_k and it % 100 == 0:
                cur_k = max(top_k_range[0], cur_k // 2)
        else:
            no_improve += 1
            if adapt_k and no_improve % 50 == 0:
                cur_k = min(top_k_range[1], cur_k + 5)
            lr = max(lr * 0.998, 1e-5)

        if no_improve > 300:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = 0.01
            cur_k = top_k_range[0]
            no_improve = 0
            conv = fast_autocorr(f)

    return best_f, best_obj


def paired_coordinate_descent(f0, time_limit, n_pairs_per_pass=20):
    """
    Coordinate descent with paired moves: adjust two coordinates simultaneously.
    This can escape local minima that single-coordinate moves cannot.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_autocorr(f).max()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)

    # Single-coordinate multipliers
    single_mults = [0.7, 0.85, 0.92, 0.97, 1.03, 1.1, 1.2, 1.5]

    while time.monotonic() - t0 < time_limit:
        improved = False
        no_improve += 1

        # Single coordinate moves
        for _ in range(min(n, 100 * no_improve)):
            if time.monotonic() - t0 > time_limit:
                break
            j = rng.integers(n)
            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in single_mults:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-12 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()

                val = fast_autocorr(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        no_improve = 0

        # Paired moves (transfer mass between two coordinates)
        for _ in range(n_pairs_per_pass):
            if time.monotonic() - t0 > time_limit:
                break
            j1 = rng.integers(n)
            j2 = rng.integers(n)
            if j1 == j2:
                continue
            v1, v2 = f[j1], f[j2]
            if v1 < 1e-14 and v2 < 1e-14:
                continue

            for transfer in [0.05, 0.1, 0.2, 0.3]:
                # Try moving mass from j1 to j2
                for direction in [1, -1]:
                    amt = transfer * min(v1, v2)
                    new_v1 = v1 - direction * amt
                    new_v2 = v2 + direction * amt
                    if new_v1 < -1e-12 or new_v2 < -1e-12:
                        continue
                    new_v1 = max(new_v1, 0.0)
                    new_v2 = max(new_v2, 0.0)
                    total_change = (v1 + v2) - (new_v1 + new_v2)
                    if abs(total_change) < 1e-14:
                        scale_other = 1.0
                    else:
                        scale_other = (1.0 - new_v1 - new_v2) / (1.0 - v1 - v2)
                        if scale_other < 0 or scale_other > 2:
                            continue

                    f_trial = f.copy()
                    f_trial[j1] = new_v1
                    f_trial[j2] = new_v2
                    mask = np.ones(n, dtype=bool)
                    mask[j1] = False
                    mask[j2] = False
                    f_trial[mask] *= scale_other
                    f_trial /= f_trial.sum()

                    val = fast_autocorr(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break

        if not improved and no_improve > 5:
            break

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving simplicity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    return f_new


def make_palindromic(f, n):
    """Create palindromic sequence and project to simplex."""
    half = (n + 1) // 2
    a = f[:half]
    pal = np.concatenate([a, a[-2::-1]])
    return project_simplex(pal)


def diverse_init(n, seed, base=None):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    method = seed % 7
    if base is not None and len(base) == n:
        if method == 0:
            return project_simplex(base + rng.standard_normal(n) * 0.02)
        elif method == 1:
            return project_simplex(0.8 * base + 0.2 * rng.uniform(0, 1, n))
        elif method == 2:
            return make_palindromic(base, n)
        elif method == 3:
            # Shift and blend
            shift = rng.integers(1, n // 4)
            shifted = np.roll(base, shift)
            return project_simplex(0.6 * base + 0.4 * shifted)
        elif method == 4:
            # Spike perturbation
            f = base.copy()
            k = rng.integers(n)
            f[k] += 0.1
            f = np.maximum(f, 0)
            return project_simplex(f)
        elif method == 5:
            # Frequency perturbation
            A = np.fft.rfft(f, 2 * n - 1)
            phases = rng.uniform(-0.1, 0.1, len(A))
            A = A * np.exp(1j * phases)
            f_new = np.fft.irfft(A, 2 * n - 1)[:n]
            f_new = np.maximum(f_new, 0)
            ss = f_new.sum()
            if ss > 1e-10:
                return f_new / ss
            return project_simplex(f)
        else:
            # Power-law shaped perturbation
            t = np.arange(n) / n
            shape = t ** (rng.uniform(0.5, 3.0))
            f_new = 0.7 * base + 0.3 * shape / shape.sum()
            return project_simplex(f_new)
    else:
        # No base, generate from scratch
        if method == 0:
            f = rng.uniform(0.1, 1, n)
        elif method == 1:
            t = np.arange(n) / n
            f = np.exp(-2 * (t - 0.5) ** 2)
        elif method == 2:
            f = np.abs(rng.standard_normal(n))
        elif method == 3:
            f = 1.0 / (1.0 + rng.uniform(0.1, 5) * np.arange(n) / n)
        elif method == 4:
            f = np.ones(n) / n
        elif method == 5:
            f = rng.exponential(1, n)
        else:
            f = np.abs(np.sin(np.pi * np.arange(n) / n * rng.uniform(1, 4)))
        return project_simplex(np.maximum(f, 0))

                a_opt2, _ = paired_coordinate_descent(a_opt, time_limit=min(rem * 0.3, 80),
                                                       n_pairs_per_pass=20)
                try_seq(a_opt2, "paired_cd2")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [512, 640, 768, 896, 1024, 1280]
    target_lengths = [tn for tn in target_lengths if tn != len(base)]

    for tn in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 35:
            break
        f_init = interpolate_to_length(base, tn)
        # Add slight perturbation for diversity
        f_init = project_simplex(f_init + rng_main.standard_normal(tn) * 0.005)
        t_alloc = min(rem * 0.2, 70)
        print(f"  n={tn}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smoothed_descent(f_init, time_limit=t_alloc,
                                               tau_init=0.04, tau_end=0.003,
                                               lr_init=0.05, momentum_init=0.82,
                                               n_continuation=3, restart_every=120)
        try_seq(a_opt, f"len_{tn}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(a_opt) == best_seq.shape[0]:
            a_opt2, _ = active_set_descent(a_opt, time_limit=min(rem * 0.2, 30),
                                            top_k_range=(10, 40), lr=0.006, momentum=0.9)
            try_seq(a_opt2, f"len_{tn}_polish")

    # === Phase 6: Palindromic optimization ===
    print(f"\n=== Phase 6: Palindromic (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    pal_init = make_palindromic(base, n_base)
    pal_score = float(evaluate_sequence(pal_init.tolist()))
    if np.isfinite(pal_score) and pal_score < best_score:
        best_score = pal_score
        best_seq = pal_init.copy()
        print(f"  Palindromic init: {pal_score:.6f}")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_pal = min(rem * 0.2, 60)
        a_opt, _ = nesterov_smoothed_descent(pal_init, time_limit=t_pal,
                                               tau_init=0.03, tau_end=0.002,
                                               lr_init=0.04, momentum_init=0.83,
                                               n_continuation=3, restart_every=100)
        # Project back to palindromic
        a_opt_pal = make_palindromic(a_opt, n_base)
        try_seq(a_opt_pal, "palindromic_opt")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # Active-set with fine parameters
        if rem > 20:
            t_fin = min(rem * 0.25, 50)
            a_opt, _ = active_set_descent(best_seq, time_limit=t_fin,
                                           top_k_range=(20, 80),
                                           lr=0.004, momentum=0.93)
            try_seq(a_opt, "final_active")

        # Tiny paired CD
        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.25, 30)
            a_opt, _ = paired_coordinate_descent(best_seq, time_limit=t_cd,
                                                  n_pairs_per_pass=10)
            try_seq(a_opt, "final_cd")

        # Micro-perturbations with quick optimization
        rem = deadline - time.monotonic()
        for i in range(4):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise_scale = 0.002 + 0.002 * i
            f_pert = project_simplex(best_seq + rng_main.standard_normal(n_best) * noise_scale)
            t_pert = min(rem * 0.2, 20)
            a_opt, _ = nesterov_smoothed_descent(f_pert, time_limit=t_pert,
                                                   tau_init=0.01, tau_end=0.001,
                                                   lr_init=0.03, momentum_init=0.85,
                                                   n_continuation=2, restart_every=60)
            try_seq(a_opt, f"final_micro_{i}")

        # One more active-set pass on the very best
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = active_set_descent(best_seq, time_limit=min(rem * 0.5, 25),
                                           top_k_range=(30, 100),
                                           lr=0.003, momentum=0.94)
            try_seq(a_opt, "final_final")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
