def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

import numpy as np
import time
from scipy.optimize import minimize_scalar, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

    """Compute weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    for i in range(len(peaks)):
        if w[i] < 1e-12:

        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

    return sg


def subgradient_optimize(a0, time_limit=300, lr=0.01, mom=0.92, top_k=40,
                         temp=0.0005, decay=0.9998, verbose=False):
    """Projected subgradient descent with momentum."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_multi_peak_subgrad(a, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.004

        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.003 + 0.003 * min(no_improve // 200, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")

    return best_a, best_obj


def pair_coordinate_descent(a0, time_limit=300, verbose=False, max_passes=200):
    """Optimize pairs of coordinates simultaneously using golden section search."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate

    # Precompute sorted indices by value
    pass_count = 0

    while time.monotonic() - t0 < time_limit and pass_count < max_passes:
        pass_count += 1
        improved = False

        # Get active peaks
        conv = fast_convolve(a)
        top_peaks = np.argpartition(conv, -min(20, len(conv)))[-min(20, len(conv)):]

        # Try structured pairs: positions that affect the top peaks
        for peak_k in top_peaks[:5]:
            k = int(peak_k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            active = list(range(js, je + 1))
            if len(active) < 2:
                continue

            # Try a few pairs from the active set
            n_try = min(8, len(active))
            for _ in range(n_try):
                if time.monotonic() - t0 > time_limit:
                    break
                i = active[rng.integers(0, len(active))]
                j = active[rng.integers(0, len(active))]
                if i == j:
                    continue

                s_pair = a[i] + a[j]
                if s_pair < 1e-13:
                    continue

                # Golden section search over x in [0, s_pair]
                a_save_i = a[i]
                a_save_j = a[j]

                lo = 0.0
                hi = s_pair
                c1 = hi - gr * (hi - lo)
                d1 = lo + gr * (hi - lo)

                a[i] = c1
                a[j] = s_pair - c1
                fc1 = fast_convolve(a).max()

                a[i] = d1
                a[j] = s_pair - d1
                fd1 = fast_convolve(a).max()

                for _ in range(45):
                    if fc1 < fd1:
                        hi = d1
                        d1 = c1
                        fd1 = fc1
                        c1 = hi - gr * (hi - lo)
                        a[i] = c1
                        a[j] = s_pair - c1
                        fc1 = fast_convolve(a).max()
                    else:
                        lo = c1
                        c1 = d1
                        fc1 = fd1
                        d1 = lo + gr * (hi - lo)
                        a[i] = d1
                        a[j] = s_pair - d1
                        fd1 = fast_convolve(a).max()
                    if hi - lo < 1e-12 * max(s_pair, 1e-10):
                        break

                x_best = (lo + hi) / 2.0
                a[i] = x_best
                a[j] = s_pair - x_best
                obj = fast_convolve(a).max()

                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = a_save_i
                    a[j] = a_save_j

        # Also try some random pairs
        for _ in range(5):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            s_pair = a[i] + a[j]
            if s_pair < 1e-13:
                continue

            a_save_i = a[i]
            a_save_j = a[j]

            lo = 0.0
            hi = s_pair
            c1 = hi - gr * (hi - lo)
            d1 = lo + gr * (hi - lo)

            a[i] = c1
            a[j] = s_pair - c1
            fc1 = fast_convolve(a).max()

            a[i] = d1
            a[j] = s_pair - d1
            fd1 = fast_convolve(a).max()

            for _ in range(35):
                if fc1 < fd1:
                    hi = d1
                    d1 = c1
                    fd1 = fc1
                    c1 = hi - gr * (hi - lo)
                    a[i] = c1
                    a[j] = s_pair - c1
                    fc1 = fast_convolve(a).max()
                else:
                    lo = c1
                    c1 = d1
                    fc1 = fd1
                    d1 = lo + gr * (hi - lo)
                    a[i] = d1
                    a[j] = s_pair - d1
                    fd1 = fast_convolve(a).max()
                if hi - lo < 1e-12 * max(s_pair, 1e-10):
                    break

            x_best = (lo + hi) / 2.0
            a[i] = x_best
            a[j] = s_pair - x_best
            obj = fast_convolve(a).max()

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                improved = True
            else:
                a[i] = a_save_i
                a[j] = a_save_j

        if not improved:
            # Escape: add noise around best
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            if pass_count > 3:
                # Also try larger noise occasionally
                if rng.random() < 0.3:
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.01)

        if verbose and pass_count % 20 == 0:
            print(f"    PairCD pass {pass_count}: obj={best_obj:.10f}")

    return best_a, best_obj


def triple_move_descent(a0, time_limit=120, verbose=False):
    """Try moves that redistribute mass among 3 coordinates."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # Pick 3 coordinates
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        k = rng.integers(0, n)
        if i == j or i == k or j == k:
            continue

        s_ijk = a[i] + a[j] + a[k]
        if s_ijk < 1e-12:
            continue

        # Try a few random redistributions
        for trial in range(10):
            # Random barycentric coordinates
            u = rng.random(3)
            u /= u.sum()
            a_trial = a.copy()
            a_trial[i] = u[0] * s_ijk
            a_trial[j] = u[1] * s_ijk
            a_trial[k] = u[2] * s_ijk
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial /= st

            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_obj


def lbfgs_smoothed_optimize(a0, time_limit=120, T=50.0, verbose=False):
    """L-BFGS-B optimization on log-sum-exp smoothed objective.
    Uses the parameterization a = exp(z) / sum(exp(z)) = softmax(z).
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    # Initialize z from a: z = log(a) (with clipping for zeros)
    z = np.log(np.maximum(a, 1e-15))
    t0 = time.monotonic()

    obj_vals = []
    grad_calls = [0]

    def smoothed_obj(z_vec):
        grad_calls[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 0.0
        # Softmax
        z_shifted = z_vec - z_vec.max()
        aa = np.exp(z_shifted)
        aa = aa / aa.sum()
        conv = fast_convolve(aa)
        # Log-sum-exp
        Tc = T * conv
        max_cv = Tc.max()
        lse = (max_cv + np.log(np.sum(np.exp(Tc - max_cv)))) / T
        val = 2.0 * n * lse
        obj_vals.append(val)
        return val

    def smoothed_grad(z_vec):
        grad_calls[0] += 1
        z_shifted = z_vec - z_vec.max()
        aa = np.exp(z_shifted)
        aa = aa / aa.sum()
        conv = fast_convolve(aa)
        Tc = T * conv
        max_cv = Tc.max()
        exp_cv = np.exp(Tc - max_cv)
        weights = exp_cv / exp_cv.sum()

        # Compute grad w.r.t. a: sg_a[j] = sum_k weights[k] * (a[k-j] + a[j-k])
        sg_a = np.zeros(n)
        # Only iterate over significant weights for speed
        sig_idx = np.where(weights > 1e-10)[0]
        for k in sig_idx:
            wk = weights[k]
            if wk < 1e-12:
                continue
            # Contribution: a[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                sg_a[idx] += wk * aa[k - idx]
            # Contribution: a[j-k] for valid j
            j_start2 = max(0, k)
            j_end2 = min(n - 1, n - 1 + k)
            if j_start2 <= j_end2:
                idx2 = np.arange(j_start2, j_end2 + 1)
                sg_a[idx2] += wk * aa[idx2 - k]

        # Softmax gradient: d(obj)/dz_m = a_m * (sg_a[m] - sum_j sg_a[j]*a_j)
        dot = np.dot(sg_a, aa)
        grad_z = 2.0 * n * aa * (sg_a - dot)
        return grad_z

    # Run L-BFGS-B
    result = minimize(
        smoothed_obj, z, jac=smoothed_grad,
        method='L-BFGS-B',
        options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-6,
                 'ftol': 1e-12, 'maxls': 20}
    )

    a_opt = np.exp(result.x - result.x.max())
    a_opt = a_opt / a_opt.sum()

    # Evaluate actual objective
    conv = fast_convolve(a_opt)
    obj = conv.max()

    if verbose:
        print(f"    L-BFGS-B: grad_calls={grad_calls[0]}, success={result.success}, obj={obj:.10f}")

    return a_opt, obj


def single_coordinate_bisection(a0, time_limit=60, verbose=False):
    """Exact 1D coordinate optimization via bisection on the derivative."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    for _pass in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            # Binary search for optimal x_j
            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                # Compute conv with a[j] = mid, others scaled by om
                a_trial = a * om
                a_trial[j] = mid
                c = fast_convolve(a_trial)
                # Find the max and check derivative
                k_max = int(np.argmax(c))

                # Derivative of c[k_max] w.r.t. mid
                # c[k_max] = om^2 * B[k_max] + 2*mid*om * adj[k_max] + mid^2 * [k_max == 2j]
                # where B = conv(a_others), adj = cross term
                # dc/dmid = 2*om*adj[k_max] - 2*mid*om*B_others... 
                # Simpler: numerical derivative
                eps = 1e-8
                a_plus = a * om
                a_plus[j] = mid + eps
                c_plus = fast_convolve(a_plus).max()
                a_minus = a * om
                a_minus[j] = mid - eps
                c_minus = fast_convolve(a_minus).max()
                deriv = (c_plus - c_minus) / (2 * eps)

                if deriv > 1e-12:
                    hi = mid
                else:
                    lo = mid
                x_new = mid

            if abs(x_new - x_old) > 1e-10:
                a[j] = x_new
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= (1.0 - x_new) / s_others
                a = np.maximum(a, 0.0)
                s_check = a.sum()
                if s_check > 1e-15:
                    a /= s_check

                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining subgradient descent, pair coordinate
    descent with golden section, L-BFGS-B on smoothed objective, and multi-length
    exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform n=768")

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_try)})")

    # === Phase 1: Subgradient warm start ===
    print(f"\n=== Phase 1: Subgradient (n={len(parent)}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 300)
        print(f"  Subgradient, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(parent, time_limit=t_alloc, lr=0.01,
                                        top_k=40, temp=0.0004, verbose=True)
        update_best(a_opt, "sg_parent")

    # === Phase 2: Pair coordinate descent (main workhorse) ===
    print(f"\n=== Phase 2: Pair CD (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 400)
        print(f"  Pair CD, t={t_alloc:.0f}s")
        a_opt, _ = pair_coordinate_descent(base, time_limit=t_alloc, verbose=True)
        update_best(a_opt, "pair_cd")

    # === Phase 3: L-BFGS-B smoothed optimization ===
    print(f"\n=== Phase 3: L-BFGS-B (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  L-BFGS-B, t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_smoothed_optimize(best_a, time_limit=t_alloc, T=80.0, verbose=True)
            update_best(a_opt, "lbfgs")

    # === Phase 4: Single coordinate bisection polish ===
    print(f"\n=== Phase 4: CD bisection (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  CD bisection, t={t_alloc:.0f}s")
            a_opt, _ = single_coordinate_bisection(best_a, time_limit=t_alloc)
            update_best(a_opt, "cd_bisect")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    target_lengths = [640, 896, 1024, 1280]

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        # Quick SG then pair CD
        t_sg = min(30, rem * 0.2)
        t_pc = min(rem * 0.6, 150)
        print(f"  n={target_n}: SG={t_sg:.0f}s, PairCD={t_pc:.0f}s")

        a_sg, _ = subgradient_optimize(f_interp, time_limit=t_sg, lr=0.008, top_k=30)
        a_opt, _ = pair_coordinate_descent(a_sg, time_limit=t_pc)
        update_best(a_opt, f"len_{target_n}")

    # === Phase 6: Symmetric variant ===
    print(f"\n=== Phase 6: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 120)
            print(f"  Symmetric pair CD, t={t_alloc:.0f}s")
            a_opt, _ = pair_coordinate_descent(a_sym, time_limit=t_alloc)
            update_best(a_sym, "sym_init")
            update_best(a_opt, "sym_cd")

    # === Phase 7: Intensive final polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            # Alternate between pair CD and triple moves
            t_each = rem / 3.0

            print(f"  Pair CD final, t={t_each:.0f}s")
            a_opt, _ = pair_coordinate_descent(best_a, time_limit=t_each)
            update_best(a_opt, "final_pair")

            rem2 = deadline - time.monotonic() - 15
            if rem2 > 15:
                t2 = min(rem2 * 0.5, 60)
                print(f"  Triple moves, t={t2:.0f}s")
                a_opt, _ = triple_move_descent(best_a, time_limit=t2)
                update_best(a_opt, "final_triple")

            rem3 = deadline - time.monotonic() - 10
            if rem3 > 10:
                t3 = min(rem3 * 0.7, 60)
                print(f"  Pair CD round 2, t={t3:.0f}s")
                a_opt, _ = pair_coordinate_descent(best_a, time_limit=t3)
                update_best(a_opt, "final_pair2")

            # L-BFGS-B final
            rem4 = deadline - time.monotonic() - 5
            if rem4 > 10:
                t4 = min(rem4, 30)
                print(f"  L-BFGS-B final, t={t4:.0f}s")
                a_opt, _ = lbfgs_smoothed_optimize(best_a, time_limit=t4, T=100.0)
                update_best(a_opt, "final_lbfgs")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
