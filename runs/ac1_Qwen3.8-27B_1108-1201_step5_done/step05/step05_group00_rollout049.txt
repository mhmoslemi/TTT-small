"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- FW with LP oracle for globalStructure
- Spectral smoothing (power spectrum flattening)
- Coordinate descent on peak auto-correlation
- Multi-n search and adaptive restarts
- Newton-polish on the active peak set
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else -1
    if rho < 0:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute the sum over k of p[k] * a[k-j] for each j (valid correlation)."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses the banded structure of the convolution matrix."""
    n = len(f)
    c = -np.ones(n)
    # Build A_ub efficiently using the banded structure
    # A_ub[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k), else 0
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # i.e., A_ub[k, j] = f[j] for k in [j, j+n-1]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def spectral_smooth(a, time_limit, alpha=0.5):
    """
    Spectral approach: We want to flatten the power spectrum |A(f)|^2
    to reduce auto-correlation peaks. Use iterative reweighting in frequency domain.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, 2 * n - 1)
        power = np.abs(A) ** 2
        
        # Target: flatten power spectrum (make it uniform)
        # But we need to keep a non-negative
        # Use: a_new = IFFT(A * phase(A) * weight) where weight flattens power
        target_power = np.ones_like(power)
        # Geometric mean of current and target
        new_power = power ** (1 - alpha) * target_power ** alpha
        # Amplitude adjustment: new_A = A * sqrt(new_power / power)
        ratio = np.sqrt(new_power / np.maximum(power, 1e-30))
        new_A = A * ratio
        new_a = np.fft.irfft(new_A, 2 * n - 1)[:n]
        new_a = np.maximum(new_a, 0.0)
        s_new = new_a.sum()
        if s_new < 1e-15:
            break
        new_a = new_a / s_new
        
        obj = fast_convolve(new_a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = new_a.copy()
            a = new_a
        else:
            # Reduce alpha
            alpha = max(0.05, alpha * 0.95)
            if alpha < 0.06:
                break
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it = time.monotonic() - t0
        remaining = time_limit - it
        if remaining < 0.5:
            break
        
        g = solve_lp_oracle(f, M)
        if g is None:
            # Relax M
            g = solve_lp_oracle(f, M * 1.1)
            if g is None:
                no_improve += 5
                if no_improve > 50:
                    break
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mathematical analysis and hybrid optimization.

Key insight: For non-negative sequences, the auto-correlation peak is always at lag 0

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute sum_k p[k]*a[k-j] for each j (valid cross-correlation)."""
    return np.correlate(p, a, mode='valid')


def conv_gradient(a, k_peak):
    """Subgradient of conv[a][k_peak] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    for j in range(js, je + 1):
        grad[j] = 2.0 * a[k_peak - j]
    return grad

    """LP: max sum(g) s.t. (f*g)[k] <= M for all k=0..2n-2, g >= 0."""
    n = len(f)

    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 35:
            rnd = np.random.default_rng(int((time.monotonic() - t0) * 1000) + 17)
            f = project_simplex(best_f + rnd.standard_normal(n) * 0.004)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def smooth_gradient(a0, time_limit, T=0.001, lr=0.03):
    """Smoothed gradient with softmax-weighted peaks and backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()

        grad = 2.0 * hankel_mv(p, a)

        # Backtracking line search
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_a = candidate.copy()
                a = candidate
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1

        if no_improve > 20:
            T *= 1.08
            lr *= 0.96
        if no_improve > 80:
            rnd = np.random.default_rng(int((time.monotonic() - t0) * 1000) + 3)
            a = project_simplex(best_a + rnd.standard_normal(n) * 0.003)
            T = 0.001
            lr = 0.03
            no_improve = 0

    return best_a, best_obj


def peak_subgradient_descent(a0, time_limit, lr_init=0.005):
    """Subgradient descent targeting the single peak of the auto-correlation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        grad = conv_gradient(a, k_peak)

        candidate = project_simplex(a - lr * grad)
        val = fast_convolve(candidate).max()

        if val < best_obj - 1e-14:
            best_obj = val
            best_a = candidate.copy()
            a = candidate
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97

        if no_improve > 50:
            rnd = np.random.default_rng(int((time.monotonic() - t0) * 1000) + 11)
            a = project_simplex(best_a + rnd.standard_normal(n) * 0.002)
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def top_k_subgradient(a0, time_limit, top_k=10, lr_init=0.003, momentum=0.93):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        val = fast_convolve(a).max()
        if val < best_obj - 1e-14:
            best_obj = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        momentum = min(0.97, momentum + 0.00005)

        if no_improve > 150:
            rnd = np.random.default_rng(int((time.monotonic() - t0) * 1000) + 23)
            a = project_simplex(best_a + rnd.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def spectral_reweight(a0, time_limit, alpha=0.3, n_iter_max=5000):
    """
    Spectral flattening: iteratively adjust the amplitude of the DFT to
    flatten the power spectrum, which reduces auto-correlation sidelobes.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for _ in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, 2 * n - 1)
        power = np.abs(A) ** 2

        # Geometric interpolation toward flat spectrum
        target = np.full_like(power, max(power.min(), 1e-30))
        new_power = power ** (1.0 - alpha) * target ** alpha

        ratio = np.sqrt(new_power / np.maximum(power, 1e-30))
        new_A = A * ratio
        new_a = np.fft.irfft(new_A, 2 * n - 1)[:n]
        new_a = np.maximum(new_a, 0.0)
        s_new = new_a.sum()
        if s_new < 1e-15:
            break
        new_a = new_a / s_new

        obj = fast_convolve(new_a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = new_a.copy()
            a = new_a
        else:
            alpha = max(0.05, alpha * 0.95)
            if alpha < 0.06:
                break

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """
    For the peak lag, try pairwise weight transfers between contributing variables.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    rnd = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        if js >= je:
            break

        improved = False
        n_tries = min(30, je - js + 1)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            i1 = js + int(rnd.integers(0, je - js + 1))
            i2 = js + int(rnd.integers(0, je - js + 1))
            if i1 == i2:
                continue
            j1 = k_peak - i1
            j2 = k_peak - i2
            if j1 < 0 or j1 >= n or j2 < 0 or j2 >= n:
                continue

            for delta in [0.0005, 0.001, 0.002]:
                for direction in [1, -1]:
                    candidate = a.copy()
                    candidate[i1] -= direction * delta
                    candidate[i2] += direction * delta
                    if candidate[i1] < -1e-12 or candidate[i2] < -1e-12:
                        continue
                    candidate = np.maximum(candidate, 0.0)
                    s = candidate.sum()
                    if s < 1e-15:
                        continue
                    candidate = candidate / s
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = candidate.copy()
                        a = candidate
                        improved = True
                        break
                if improved:
                    break

        if not improved and it % 50 == 0:
            pert = project_simplex(a + rnd.standard_normal(n) * 0.001)
            val = fast_convolve(pert).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_a = pert.copy()
                a = pert
            elif it > 200:
                break

    return best_a, best_obj


def rescale_search(a, n_new):
    """Rescale/interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64)
    if n_new == len(a):
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
