"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LSE gradient (O(n log n) per iteration)
2. Nesterov-accelerated momentum
3. Adaptive beta scheduling
4. Frank-Wolfe with LP oracle for structural improvements
5. Multi-phase strategy with diverse initializations
6. Peak-targeted subgradient at high beta
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, beta, conv=None):
    """Compute gradient of LSE(beta * conv(a,a)) w.r.t. a using FFT.
    Returns the gradient (not normalized)."""
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    # Compute softmax weights
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Gradient = 2 * cross_correlate(w, a)
    # where cross_correlate(w, a)[m] = sum_j w[m+j] * a[j] for j=0..n-1
    # Computed via: convolve(w, flip(a))[m + n - 1]
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(a[::-1].copy(), fft_size)
    conv_w_a = np.fft.irfft(W * Af, fft_size)
    grad = 2.0 * conv_w_a[n-1 : 2*n-1]
    
    return grad, w


def lse_optimize_nesterov(f0, time_limit, beta_start=10.0, beta_end=20000.0, 
                           lr_init=None, momentum=0.9, verbose=False):
    """LSE-optimized with Nesterov momentum and adaptive beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr_init is None:
        lr_init = 2.0 / n
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_init
    mu = momentum  # Nesterov momentum parameter
    v = np.zeros(n)  # velocity for Nesterov
    f_prev = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta_growth = 1.003
    best_lse_val = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = f + mu * (f - f_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = f.copy()
        else:
            y = y / ys
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad, w = lse_gradient(y, beta, conv=conv_y)
        
        # Update
        f_prev = f.copy()
        f_new = project_simplex(f - lr * grad + mu * (f - f_prev))
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            lr = min(lr * 1.002, 10.0 / n)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.95
            if no_improve > 50:
                # Restart from best with smaller perturbation
                f = best_f.copy()
                f_prev = f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.0005)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
                mu = 0.85
        
        f = f_new
        
        # Adaptive beta
        if no_improve < 20:
            beta = min(beta * beta_growth, beta_end)
        elif no_improve < 40:
            beta = min(beta * 1.01, beta_end)
        # else: keep beta constant (handled by restart logic above)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Ne it={it}, t={elapsed:.1f}s, obj={obj_new:.10f}, "
                  f"best={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}")
    
    return best_f, best_obj


def peak_directed_nesterov(f0, time_limit, n_peaks=5, lr_init=None, momentum=0.92, verbose=False):

    conv_len = 2 * n - 1
    
    if lr_init is None:
        lr_init = 0.5 / sqrt(n)  # Hmm, need to import sqrt or use n**0.5
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

        if w_sum < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE gradient (O(n log n) per step)
2. Nesterov-accelerated momentum for faster convergence
3. Adaptive beta scheduling (smooth -> sharp -> peak-targeting)
4. Frank-Wolfe with LP oracle for structural escape
5. Multi-phase strategy with diverse initializations and restarts
6. Peak-directed subgradient at convergence

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient_fft(a, beta, conv=None):
    """Compute gradient of LSE(beta * conv(a,a)) w.r.t. a in O(n log n)."""
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    # Gradient = 2 * cross_correlate(w, a) via FFT
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(a[::-1].copy(), fft_size)
    conv_result = np.fft.irfft(W * Af, fft_size)
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]
    return grad, w

def peak_directed_optimize(f0, time_limit, n_peaks=5, lr_init=None,
                           momentum=0.93, verbose=False):
    """Subgradient method targeting the top peaks directly."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    if lr_init is None:
        lr_init = 1.0 / (n ** 0.5) * 2.0

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = lr_init
    mu = momentum
    f_prev = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        y = f + mu * (f - f_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = f.copy()
        else:
            y = y / ys

        conv_y = fast_convolve(y)
        k = min(n_peaks, conv_len)
        peak_idx = np.argpartition(conv_y, -k)[-k:]
        peak_vals = conv_y[peak_idx]

        # Weight: emphasize the maximum
        w_peaks = peak_vals ** 3
        w_sum = w_peaks.sum()
        if w_sum < 1e-15:
            w_peaks = np.ones(k) / k
        else:
            w_peaks = w_peaks / w_sum

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(k):
            ki = int(peak_idx[i])
            wk = w_peaks[i]
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo <= hi:
                # grad contribution: 2 * wk * y[ki - m] for m in [lo, hi]
                idx = ki - np.arange(lo, hi + 1)
                grad[lo:hi + 1] += wk * 2.0 * y[idx]

        f_prev = f.copy()
        f_new = project_simplex(f - lr * grad + mu * (f - f_prev))

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            lr = min(lr * 1.005, 10.0)
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.94
            if no_improve > 150:
                f = best_f.copy()
                f_prev = f.copy()
                lr = max(lr_init * 0.3, 0.001)
                no_improve = 0

        f = f_new

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Peak it={it}, t={elapsed:.1f}s, obj={obj_new:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

        it += 1
        A_cache = build_lp_matrix(f)

            m_exp *= 1.2
            if m_exp > 20.0:
                break
            continue

        gs = g.sum()
        if gs < 1e-15:

        g = g / gs

            m_exp = 1.0

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

        if t_alloc < 30:
            break
        print(f"  FW-LP cycle {cycles}, n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = fw_lp_optimize(current, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current

        # Quick LSE follow-up after FW
        t_lse = min(40, t3_end - time.monotonic() - 30)
        if t_lse > 15:
            f_opt2, _ = lse_optimize_nesterov(current, time_limit=t_lse,
                                               beta_start=200.0, beta_end=10000.0)
            try_sequence(f_opt2, best_tracker)
            current = best_tracker[1] if best_tracker[1] is not None else current

    # === Phase 4: Peak-directed refinement ===
    print(f"\n=== Phase 4: Peak-Directed Refinement ===")
    t4_end = deadline - 150
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    for npk in [3, 5, 8]:
        t_remaining = t4_end - time.monotonic()
        if t_remaining < 40:
            break
        t_alloc = min(100, t_remaining * 0.35)
        if t_alloc < 20:
            break
        print(f"  Peak-directed n_peaks={npk}, n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = peak_directed_optimize(current, time_limit=t_alloc,
                                           n_peaks=npk, lr_init=0.5 / (n_cur ** 0.5))
        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current

    # === Phase 5: Final micro-polish ===
    print(f"\n=== Phase 5: Final Micro-Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    # Very high beta LSE
    t_remaining = deadline - time.monotonic() - 30
    if t_remaining > 60:
        t_alloc = min(150, t_remaining * 0.5)
        print(f"  High-beta LSE t={t_alloc:.0f}s, beta_end=100000")
        f_opt, _ = lse_optimize_nesterov(current, time_limit=t_alloc,
                                          beta_start=5000.0, beta_end=100000.0,
                                          lr_init=1.5 / n_cur)
        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current

    # Directed peak with very few peaks
    t_remaining = deadline - time.monotonic() - 20
    if t_remaining > 30:
        t_alloc = min(80, t_remaining * 0.6)
        print(f"  Peak-directed (3 peaks) t={t_alloc:.0f}s")
        f_opt, _ = peak_directed_optimize(current, time_limit=t_alloc,
                                           n_peaks=3, lr_init=0.3 / (n_cur ** 0.5),
                                           momentum=0.95)
        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current

    # Try nearby lengths
    t_remaining = deadline - time.monotonic() - 15
    if t_remaining > 40 and best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        for off in [32, -32, 64, -64, 128, -128]:
            n_try = n_best + off
            if n_try < 256 or n_try > 2000:
                continue
            t_rem2 = deadline - time.monotonic() - 15
            if t_rem2 < 30:
                break
            t_alloc = min(50, t_rem2 * 0.4)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Nearby length n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize_nesterov(f_init, time_limit=t_alloc,
                                              beta_start=100.0, beta_end=20000.0)
            try_sequence(f_opt, best_tracker)

    # === Final: one more LSE pass on the best ===
    if best_tracker[1] is not None:
        t_remaining = deadline - time.monotonic() - 5
        if t_remaining > 20:
            current = best_tracker[1]
            n_cur = len(current)
            t_alloc = min(60, t_remaining * 0.8)
            print(f"  Final polish t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize_nesterov(current, time_limit=t_alloc,
                                              beta_start=5000.0, beta_end=80000.0,
                                              lr_init=1.0 / n_cur)
            try_sequence(f_opt, best_tracker)

    # Return best
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Fallback to parent")
    return [float(x) for x in parent]
