import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M at each lag, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def compute_lse_gradient(f, conv, beta, top_k=200):
    """Compute gradient of LSE approximation using top-k peaks."""
    n = len(f)
    conv_len = len(conv)
    if conv_len <= top_k:
        ks = np.arange(conv_len)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]

    bc = beta * conv[ks]
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()

    grad = np.zeros(n)
    for i, k in enumerate(ks):
        wk = w[i]
        if wk < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += wk * f[k - ms]
        ms2_lo = max(0, k - hi)
        ms2_hi = min(n - 1, k - lo)
        if ms2_lo <= ms2_hi:
            ms2 = np.arange(ms2_lo, ms2_hi + 1)
            grad[ms2] += wk * f[k - ms2]
    return grad


def peak_deflation(f0, time_limit, verbose=False):
    """
    Greedy peak-deflation coordinate descent.
    At each step, identify the dominant peak and reduce contributions to it.
    Uses a balanced approach: tries to equalize top-K peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            # Add noise to escape
            scale = 0.001 + 0.003 * min(1.0, (no_improve - 200) / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0
            continue

        # Identify top-K peaks and their contributing indices
        K = min(5, 2 * n - 1)
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[conv[peak_indices] > 0.5 * best_obj]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # For each peak, find contributing indices and try to reduce
        improved = False
        for k in peak_indices:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            contribs = f[js] * f[partners]
            total_contrib = contribs.sum()
            base = conv[k] - total_contrib

            # Try reducing the largest contributors
            order = np.argsort(contribs)[::-1]
            for idx in order[:min(3, len(order))]:
                j_try = js[idx]
                p_try = partners[idx]
                # Try reducing f[j_try]
                for frac in [0.5, 0.7, 0.85]:
                    dx = f[j_try] * (1.0 - frac)
                    if dx < 1e-15:
                        continue
                    # Redistribute to f[p_try] (the partner)
                    f_new = f.copy()
                    f_new[j_try] -= dx
                    f_new[p_try] += dx
                    if f_new[j_try] < 0:
                        # Redistribute uniformly
                        f_new = f.copy()
                        f_new[j_try] = 0
                        rem_dt = f[j_try]
                        f_new = f_new + rem_dt / n
                    f_new = np.maximum(f_new, 0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new /= sn
                    conv_new = fast_convolve(f_new)
                    if conv_new.max() < cur_max - 1e-12:
                        f = f_new
                        cur_max = conv_new.max()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and no_improve > 50:
            # Try random coordinate perturbation
            j = rng.integers(n)
            j2 = rng.integers(n)
            if j == j2:
                j2 = (j2 + 1) % n
            amt = f[j] * 0.1
            f_new = f.copy()
            f_new[j] -= amt
            f_new[j2] += amt
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn > 1e-15:
                f_new /= sn
                if fast_convolve(f_new).max() <= cur_max + 1e-10:
                    f = f_new

    return best_f, best_obj


def mirror_descent(f0, time_limit, beta_start=10.0, beta_end=5000.0,
                   top_k=150, verbose=False):
    """Mirror descent (multiplicative weights) with LSE gradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    step = 1.0 / np.sqrt(np.log(n))

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            f = best_f.copy()
            beta = min(beta * 2.0, beta_end)
            step = max(0.1 / np.sqrt(np.log(n)), 0.001)
            no_improve = 0
            if beta >= beta_end:
                # Add small noise
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                if beta >= beta_end * 10:
                    break

        grad = compute_lse_gradient(f, conv, beta, top_k=top_k)
        grad = grad / (np.linalg.norm(grad) + 1e-15)

        # Multiplicative update
        f_new = f * np.exp(-step * grad)
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new /= sn

        conv_new = fast_convolve(f_new)
        if conv_new.max() < conv.max() + 1e-12:
            f = f_new
            if conv_new.max() < best_obj:
                no_improve = 0
        else:
            # Try smaller step
            f_new2 = f * np.exp(-step * 0.5 * grad)
            f_new2 = np.maximum(f_new2, 0)
            sn2 = f_new2.sum()
            if sn2 > 1e-15:
                f_new2 /= sn2
                if fast_convolve(f_new2).max() < conv.max():
                    f = f_new2
                    step *= 0.7

        if beta < beta_end:
            beta *= 1.005

    return best_f, best_obj


def fw_lp(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and improved line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 5
            if no_improve > 30:
                break
            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Careful line search using golden section
        best_alpha = 0.0
        best_val = M
        # Try a grid first
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if best_alpha > 0.05:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31337 + 7)
            scale = 0.001 + 0.005 * min(1.0, (no_improve - 60) / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def balanced_peak_descent(f0, time_limit, K=5, beta=50.0, top_k=300):
    """
    Gradient descent that targets the top-K peaks simultaneously.
    Uses a softmax-weighted combination of peak gradients.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 1.0 / n
    momentum = 0.9
    v = np.zeros(n)
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            f = best_f.copy()
            v = np.zeros(n)
            lr = max(lr * 0.9, 0.1 / n)
            if no_improve > 200:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0
            if lr < 0.01 / n:
                break

        # Compute weights on top-K peaks
        conv_len = len(conv)
        k_sel = min(top_k, conv_len)
        peak_pos = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peak_pos]
        # Exponential weighting emphasizing the highest
        w_scaled = pv - pv.min()
        w_vals = np.exp(beta * w_scaled / (w_scaled.max() + 1e-15))
        w_vals /= w_vals.sum()

        w_full = np.zeros(conv_len)
        w_full[peak_pos] = w_vals

        # Compute weighted gradient
        grad = np.zeros(n)
        for i, k in enumerate(peak_pos):
            wk = w_vals[i]
            if wk < 1e-15:
                continue
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += wk * f[k - ms]
            ms2_lo = max(0, k - hi)
            ms2_hi = min(n - 1, k - lo)
            if ms2_lo <= ms2_hi:
                ms2 = np.arange(ms2_lo, ms2_hi + 1)
                grad[ms2] += wk * f[k - ms2]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        if conv_new.max() < conv.max() + 1e-12:
            f = f_new
        else:
            v = 0.5 * v
            f_new = project_simplex(f + v)
            if fast_convolve(f_new).max() < conv.max():
                f = f_new
                lr *= 0.95

    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing with coordinate moves and local polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    T = 0.005
    T_min = 0.0001
    cool_rate = 0.9995
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        it += 1
        # Choose a random coordinate pair to swap mass
        j = rng.integers(n)
        k = rng.integers(n)
        if j == k:
            k = (k + 1) % n

        # Try to move some mass from j to k (or vice versa)
        if f[j] > f[k]:
            src, dst = j, k
        else:
            src, dst = k, j

        amt = min(f[src] * 0.3, rng.uniform(0.001, 0.05))
        f_new = f.copy()
        f_new[src] -= amt
        f_new[dst] += amt
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            continue
        f_new /= sn

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        delta = obj_new - cur_obj

        if delta < 0 or rng.random() < np.exp(-delta / T):
            f = f_new
            cur_obj = obj_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                if verbose and it % 1000 == 0:
                    print(f"    SA it={it}, best={best_obj:.10f}")
        T *= cool_rate

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def fast_local_polish(f0, time_budget):
    """Rapid local optimization combining peak deflation and mirror descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    t0 = time.monotonic()

    # Peak deflation
    t1 = min(time_budget * 0.4, time_budget - 5)
    f, _ = peak_deflation(f, t1)

    # Mirror descent
    remaining = time_budget - (time.monotonic() - t0)
    t2 = min(remaining * 0.5, remaining - 2)
    if t2 > 3:
        f, _ = mirror_descent(f, t2, beta_start=20.0, beta_end=3000.0)

    return f
