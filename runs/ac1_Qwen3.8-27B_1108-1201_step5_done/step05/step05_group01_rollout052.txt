def convolve_fg(f, g):

    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def nesterov_subgrad(f0, time_limit, lr0=0.005, mom0=0.85, top_k=30, temp0=0.0001):
    """Nesterov-accelerated projected subgradient with adaptive multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        # Compute subgradient from top-k peaks
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        # Compute gradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * f[k - indices]
        
        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        f_new = project_simplex(y + v)
        y = project_simplex(f_new + cur_mom * (f_new - f))
        f = f_new
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        cur_temp *= 1.003
        cur_mom = min(0.98, cur_mom * 1.0002)
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.5, 0.001)
            no_improve = 0
            conv = fast_convolve(f)
    
    return best_f, best_obj


def gerchberg_saxton_init(n, n_iter=50, seed=0):
    """
    Generate initialization using Gerchberg-Saxton spectral flattening.
    Tries to make |DFT(a_padded)|^2 as flat as possible.
    """
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    # Start with random phase in frequency domain
    A = np.fft.rfft(np.ones(n, dtype=np.float64), fft_size)
    A *= (0.5 + 0.5 * rng.random(len(A)))  # random amplitude scaling
    
    a = None
    for it in range(n_iter):
        # Time domain: IDFT, take first n, clip, normalize
        a_full = np.fft.irfft(A, fft_size)
        a = a_full[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        # Frequency domain: DFT of padded a
        A_new = np.fft.rfft(a, fft_size)
        
        # Compute target amplitude: geometric mean scaled
        amp = np.abs(A_new)
        amp[amp < 1e-15] = 1e-15
        target = np.sqrt(np.median(amp))  # or some other target
        # Blend: partially flatten
        blend = 0.3
        new_amp = amp * (1 - blend) + target * blend
        A = A_new / amp * new_amp
    
    return a


def raised_cosine_init(n):
    """Raised cosine (Hann) window initialization."""
    j = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def blackman_init(n):
    """Blackman window initialization."""
    j = np.arange(n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * j / (n - 1)) + 0.08 * np.cos(4 * np.pi * j / (n - 1))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def gaussian_init(n, sigma_frac=0.4):
    """Gaussian window initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((j - center) / (sigma_frac * n / 2))**2)
    a = a / a.sum()
    return a


def random_init(n, seed=0):
    """Random non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    return a


def cosine_modulated_init(n, freq, amp, seed=0):
    """Cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
