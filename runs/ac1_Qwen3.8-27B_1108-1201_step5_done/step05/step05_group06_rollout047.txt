"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based local search,
smooth gradient refinement, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_local_search(a, M, max_iter=50, tol=1e-8):
    """
    LP-based local search. Given current a and M = max(a*a),
    find g that maximizes sum(g) subject to (a*g) <= M, g >= 0.
    Returns the rescaled g.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Build the constraint matrix A (conv_len x n)
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # i.e., (a * g)[k] = sum_j A[k,j] * g[j]
    
    # For efficiency, we'll use the fact that A is Toeplitz-like
    # A[k, j] = a[j] when max(0, k-n+1) <= j <= min(k, n-1)
    
    # Actually, let's just build it explicitly for moderate n
    # For n up to ~2000, this is feasible
    
    # c = -1 (we minimize -sum(g) to maximize sum(g))
    c = -np.ones(n)
    
    # A_ub: convolution matrix
    # For constraint k: sum_j a[j] * g[k-j] <= M (where g[i]=0 for i<0 or i>=n)
    # = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * g[k-j]
    # Let i = k-j, so j = k-i: sum_{i: 0<=i<n, 0<=k-i<n} a[k-i] * g[i]
    # = sum_i A[k,i] * g[i] where A[k,i] = a[k-i] if 0<=k-i<n, else 0
    
    A_ub = np.zeros((conv_len, n))
    for k in range(conv_len):
        # A[k, i] = a[k-i] if 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            A_ub[k, i_start:i_end+1] = a[k - i_end:k - i_start + 1][::-1]
    
    b_ub = np.full(conv_len, M)
    
    # Solve: minimize c @ g subject to A_ub @ g <= b_ub, g >= 0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if not res.success:
        return a.copy()
    
    g = res.x
    s = g.sum()
    if s < 1e-15:
        return a.copy()
    g /= s
    return g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based local search,
smooth gradient descent, coordinate bisection, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad(a, beta):
    """Compute smooth log-sum-exp objective and its gradient."""
    n = len(a)
    c = fast_convolve(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled -= scaled.max()
    log_sum = scaled.max() + np.log(np.exp(scaled).sum())  # log(sum exp(beta*c))
    F = (cm + log_sum / beta)  # (1/beta) * log(sum exp(beta * c))
    
    # Softmax weights
    s = np.exp(scaled)
    s /= s.sum()
    
    # Gradient computation via FFT cross-correlation
    N = 1 << (3 * n).bit_length()
    S = np.fft.rfft(s, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(S * np.conj(A), N)
    r = R[:n]  # r[i] = sum_k s[k] * a[k-i]
    
    grad = 2.0 * r
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < (2 * n - 1)
    grad[valid] -= s[diag_idx[valid]] * a[valid]
    
    return F, grad

def lp_direction(a, M, n_bounds=None):
    """
    Solve LP: maximize sum(g) subject to (a*g)_k <= M, 0 <= g <= 1.
    Returns the optimal g (rescaled to sum=1) or None if infeasible.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Build constraint matrix
    # (a*g)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * g[k-j]
    # Let i = k-j: (a*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[k-i] * g[i]
    # So A_ub[k, i] = a[k-i] if 0 <= k-i < n, else 0
    
    # For efficiency with larger n, only build the matrix if n is small enough
    if n > 3000:
        return None
    
    A_ub = np.zeros((conv_len, n))
    for k in range(conv_len):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            A_ub[k, i_start:i_end+1] = a[k - np.arange(i_end + 1 - i_start)][::-1]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(conv_len, M)
    bounds = [(0.0, 1.0)] * n  # 0 <= g <= 1
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-12:
        return None
    g /= s
    return g

def smooth_gradient_descent(a0, time_limit, beta_start=5, beta_end=100, 
                            lr=0.01, n_restarts=1, verbose=False):
    """Smooth gradient descent with annealing on beta."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for restart in range(n_restarts):
        cur_beta = beta_start
        cur_lr = lr
        no_improve = 0
        
        while time.monotonic() - t0 < time_limit:
            F, grad = smooth_obj_and_grad(a, cur_beta)
            
            # Projected gradient step
            a_new = project_simplex(a - cur_lr * grad)
            
            # Check actual objective (the true max, not the smooth approximation)
            c_new = fast_convolve(a_new)
            obj_new = c_new.max()
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            a = a_new
            
            # Anneal
            cur_beta *= 1.05
            cur_lr *= 0.9995
            
            if no_improve > 200:
                # Reset with noise
                rng = np.random.default_rng(int(time.monotonic() * 1000) + restart)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                cur_beta = beta_start
                cur_lr = lr
                no_improve = 0
        
        if restart > 0:
            break  # Only one restart in this function
    
    return best_a, best_obj

def coordinate_bisection(a0, time_limit, verbose=False):
    """Coordinate descent with bisection."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for _pass in range(1000):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue
            
            x_old = a[j]
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo, hi = 0.0, 1.0
            x_new = x_old
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-12:
                    hi = mid
                elif deriv < -1e-12:
                    lo = mid
                else:
                    break
                x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-13:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if _pass > 2:
                rng = np.random.default_rng(_pass * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break
    
    return best_a, best_obj

def lp_based_optimization(a0, time_limit, max_lp_iters=200, verbose=False):
    """
    LP-based optimization: at each step, solve the LP to find a direction,
    then do a line search along that direction.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    t0 = time.monotonic()
    
    for it in range(max_lp_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for direction
        g = lp_direction(a, M)
        if g is None:
            break
        
        # Line search: a_new = (1-t)*a + t*g, find t that minimizes max(a_new * a_new)
        best_t = 0.0
        best_f = M
        for t in np.linspace(0.01, 1.0, 30):
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial /= st
            c_trial = fast_convolve(a_trial)
            f = c_trial.max()
            if f < best_f:
                best_f = f
                best_t = t
        
        if best_t > 0 and best_f < best_obj - 1e-14:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a /= a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            best_obj = M
            best_a = a.copy()
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.10f}, t={best_t:.3f}")
        else:
            # Try smaller M (tighten the constraint)
            M *= 0.99
            # Or restart with noise
            if it > 20:
                rng = np.random.default_rng(it * 17 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(a)
                M = conv.max()
    
    return best_a, best_obj

def combined_optimize(a0, time_limit, verbose=False):
    """Alternate LP, smooth gradient, and coordinate bisection."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    phase = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        if phase % 5 == 0:
            # LP-based
            t_alloc = min(30, remaining)
            a_opt, obj = lp_based_optimization(best_a, time_limit=t_alloc, verbose=verbose)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} LP: obj={best_obj:.10f}")
        
        elif phase % 5 == 1:
            # Smooth gradient
            t_alloc = min(30, remaining)
            a_opt, obj = smooth_gradient_descent(best_a, time_limit=t_alloc, 
                                                  beta_start=10, beta_end=50)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} Smooth: obj={best_obj:.10f}")
        
        elif phase % 5 == 2:
            # Coordinate bisection
            t_alloc = min(60, remaining)
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} CD: obj={best_obj:.10f}")
        
        elif phase % 5 == 3:
            # Noise + CD
            rng = np.random.default_rng(phase * 103 + 11)
            noise_scale = 0.004 * max(0.3, 1.0 - phase * 0.03)
            a_noisy = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            t_alloc = min(40, remaining)
            a_opt, obj = coordinate_bisection(a_noisy, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} Noise+CD: obj={best_obj:.10f}")
        
        else:
            # Smooth gradient with higher beta
            t_alloc = min(30, remaining)
            a_opt, obj = smooth_gradient_descent(best_a, time_limit=t_alloc,
                                                  beta_start=30, beta_end=200)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} Smooth+: obj={best_obj:.10f}")
        
        phase += 1
    
    return best_a, best_obj

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
