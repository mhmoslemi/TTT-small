def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    A_sparse[rows, cols] = vals
    A_sparse = A_sparse.tocsr()
    
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def fw_lp_optimize(f0, time_limit, max_lp_time=8):

                # This gives g = uniform, which is not useful
                # Instead, try a random direction

                m_exp = 1.0

            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)

def game_iterate(a0, time_limit, T_schedule=None):
    """
    Game-theoretic iteration:
    - Dual: p = softmax(conv(a) / T)
    - Primal: solve min a^T M(p) a s.t. a in simplex
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    T = 0.001
    
    if T_schedule is None:
        T_schedule = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for T in T_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Inner loop at this temperature
        target_time = min(remaining * 0.8, 100)
        t_inner = time.monotonic()
        
        while time.monotonic() - t_inner < target_time:
            conv = fast_convolve(a)
            
            # Softmax dual
            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()
            
            # Solve QP: min a^T M(p) a s.t. sum(a)=1, a>=0
            # M(p)[j,m] = p[j+m]
            # Use scipy minimize
            def qp_obj(x):
                # Compute x^T M(p) x = sum_{j,m} x[j]*x[m]*p[j+m]
                # = sum_k p[k] * (sum_j x[j]*x[k-j])
                # = sum_k p[k] * (x*x)[k] in the "weighted" sense
                # Actually: sum_{j,m} x[j]*x[m]*p[j+m]
                # Let's compute it as: for each k, p[k] * sum_j x[j]*x[k-j]
                # But that's the dot product of p with the convolution of x with x
                c = fast_convolve(x)
                return np.dot(p, c)
            
            def qp_grad(x):
                # Gradient of sum_{j,m} x[j]*x[m]*p[j+m] with respect to x
                # = 2 * sum_m x[m]*p[.] ... 
                # d/dx[j] sum_{j',m} x[j']*x[m]*p[j'+m] = sum_m x[m]*p[j+m] + sum_{j'} x[j']*p[j'+m]
                # = sum_m x[m]*p[j+m] + sum_m x[m]*p[m+j]  (by symmetry of the sum)
                # Wait, let me be more careful.
                # f(x) = sum_{j,m} x[j]*x[m]*p[j+m]
                # df/dx[j] = sum_m x[m]*p[j+m] + sum_{j'} x[j']*p[j'+m] ... 
                # No: df/dx[j] = sum_m x[m]*p[j+m] + sum_{j'} x[j']*p[j'+j]... 
                # Actually: f = sum_{j,m} x[j] x[m] p[j+m]
                # df/dx[j0] = sum_m x[m] p[j0+m] + sum_{j} x[j] p[j+m0]... 
                # Hmm, this is getting confusing. Let me just use:
                # df/dx[j0] = 2 * sum_m x[m] p[j0+m]  (by symmetry of j,m in the sum)
                # Wait no, p[j+m] is symmetric in j+m, not in j and m separately.
                # Let me redo: f = sum_{j=0}^{n-1} sum_{m=0}^{n-1} x[j]*x[m]*p[j+m]
                # df/dx[j0] = sum_m x[m]*p[j0+m] + sum_j x[j]*p[j+m0]
                # The first term: for each m, we get x[m]*p[j0+m], m ranges 0 to n-1
                # The second term: for each j, we get x[j]*p[j+m0], j ranges 0 to n-1
                # But p has length 2n-1, indices 0 to 2n-2. j0+m ranges 0 to 2n-2. ✓
                # By the symmetry of the double sum (swap j and m), these two terms are equal:
                # sum_m x[m]*p[j0+m] = sum_j x[j]*p[j+m0] (just rename)
                # So df/dx[j0] = 2 * sum_m x[m] * p[j0+m]
                # This is 2 * (hankel product of p and x)
                
                # Compute: for each j, grad[j] = 2 * sum_m x[m] * p[j+m]
                # p has length 2n-1, x has length n
                # For j=0: sum_{m=0}^{n-1} x[m] * p[m]
                # For j=n-1: sum_{m=0}^{n-1} x[m] * p[n-1+m]
                
                # This is np.correlate or a sum
                # grad[j] = 2 * sum_{m=0}^{n-1} x[m] * p[j+m]
                # = 2 * (x * reverse(p[:n])) ... no
                # Let's just compute it directly
                # For efficiency, use the fact that this is a "sliding dot product"
                # Actually, let's just do it with a loop or vectorized
                result = np.zeros(n)
                for j in range(n):
                    result[j] = 2.0 * np.dot(x, p[j:j+n])
                return result
            
            # Use trust-constr or SLSQP
            try:
                res = minimize(qp_obj, a, jac=qp_grad, method='SLSQP',
                             constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - 1}],
                             bounds=[(0, None)] * n,
                             options={'maxiter': 50, 'ftol': 1e-10})
                if res.success:
                    a_new = np.maximum(res.x, 0)
                    a_new = a_new / a_new.sum()
                else:
                    a_new = a  # Keep old
            except Exception:
                a_new = a
            
            a = a_new
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
        
        # Update best
        if best_obj < fast_convolve(a).max() - 1e-10:
            a = best_a.copy()
    
    return best_a, best_obj

def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient: 2 * sum_m p[j+m] * a[m] for each j
        # Same as in game_iterate
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = 2.0 * np.dot(a, p[j:j+n])
        
        # Projected gradient step with backtracking
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Schedule
        if no_improve % 50 == 0 and no_improve > 0:
            T *= 1.02
            lr *= 0.995
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

def local_search(a0, time_limit):
    """
    Local search: try small perturbations and coordinate moves.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        improved = False
        rng = np.random.default_rng((it * 48271 + 1) % (2**31))
        
        # Try random small perturbations
        for scale in [0.0005, 0.001, 0.002, 0.005]:
            perturbation = rng.standard_normal(n) * scale
            candidate = project_simplex(a + perturbation)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        # Try coordinate moves: move mass from i to j
        if not improved and it % 3 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1]:
                candidate = a.copy()
                amount = candidate[i] * frac
                if candidate[j] + amount < 1.0:
                    candidate[i] -= amount
                    candidate[j] += amount
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved and it > 500:
            break
    
    return best_a, best_obj

Wait, I realize there's a problem with my game_iterate function. The QP objective computation using `fast_convolve` inside the minimize call will be very slow because it's called many times during the optimization. Let me optimize this.

Actually, the QP objective is `a^T M(p) a` where `M(p)[j,m] = p[j+m]`. This can be computed as:
`sum_{j,m} a[j]*a[m]*p[j+m] = sum_k p[k] * sum_j a[j]*a[k-j]`

The inner sum is related to the convolution but not exactly. Let me think...

Actually, `sum_{j,m} a[j]*a[m]*p[j+m]` where j,m range from 0 to n-1, and j+m ranges from 0 to 2n-2.

Let's substitute: for a fixed k = j+m, the sum over j,m with j+m=k is `sum_j a[j]*a[k-j]` (with valid j). This is exactly `(a*a)[k]` (the linear convolution)!

So the QP objective is `sum_k p[k] * (a*a)[k]` = `dot(p, convolve(a,a))`.

This is a quadratic form in a, and we can compute it efficiently using FFT:
`dot(p, ifft(|fft(a_padded)|^2))` where a_padded is zero-padded.

But actually, for the gradient, we need:
`d/da[j0] dot(p, conv(a,a)) = 2 * sum_m a[m] * p[j0+m]`

This is the "Hankel" product. For n=768, p has length 1535, and for each j we compute a dot product of length n. So the gradient computation is O(n^2) = O(600K) operations, which is fine.

The objective computation is O(n log n) via FFT.

So each QP iteration takes O(n^2) for the gradient + O(n log n) for the objective. With SLSQP doing maybe 50-100 iterations, that's manageable.

def solve_qp_active_set(p, n, max_iter=200):
    """
    Solve min a^T M(p) a s.t. sum(a)=1, a>=0.
    M(p)[j,m] = p[j+m].
    Uses active set method.
    """
    # Start with all variables active
    S = set(range(n))
    
    for _ in range(max_iter):
        if len(S) == 0:
            return np.ones(n) / n
        if len(S) == 1:
            a = np.zeros(n)
            a[list(S)[0]] = 1.0
            return a
        
        S_list = sorted(S)
        k = len(S_list)
        
        # Build the system: [2M_SS, 1; 1^T, 0] [a_S; lambda] = [0; 1]
        # M_SS[i,j] = p[S_list[i] + S_list[j]]
        M_SS = np.array([[p[S_list[i] + S_list[j]] for j in range(k)] for i in range(k)])
        
        # System matrix
        A = np.zeros((k+1, k+1))
        A[:k, :k] = 2.0 * M_SS
        A[:k, k] = 1.0
        A[k, :k] = 1.0
        # A[k, k] = 0
        
        b = np.zeros(k+1)
        b[k] = 1.0
        
        try:
            sol = np.linalg.solve(A, b)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(A, b, rcond=None)[0]
        
        a_S = sol[:k]
        lam = sol[k]
        
        # Check if any a_S[i] < 0: remove from active set
        min_idx = np.argmin(a_S)
        if a_S[min_idx] < -1e-12:
            S.remove(S_list[min_idx])
            continue
        
        # Build full solution
        a = np.zeros(n)
        for i, idx in enumerate(S_list):
            a[idx] = max(0.0, a_S[i])
        
        # Check KKT for inactive variables
        # For j not in S: need (2Ma + lambda)_j >= 0
        # 2Ma[j] + lambda = 2 * sum_m a[m] * p[j+m] + lambda
        # Compute Ma: for each j, Ma[j] = sum_m a[m] * p[j+m]
        Ma = np.zeros(n)
        for j in range(n):
            Ma[j] = np.dot(a, p[j:j+n])
        
        RHS = 2.0 * Ma + lam
        
        # Find most violated inactive constraint
        min_val = np.inf
        min_j = -1
        for j in range(n):
            if j not in S and RHS[j] < min_val:
                min_val = RHS[j]
                min_j = j
        
        if min_val >= -1e-10:
            # KKT satisfied
            return a
        else:
            S.add(min_j)
    
    # Fallback
    a = np.zeros(n)
    a[list(S)] = 1.0 / len(S)
    return a

"""
Hybrid optimization for minimum peak autocorrelation step function.
Uses FW-LP with sparse LP oracle, game-theoretic projected gradient,
smoothed gradient descent, and local search. Multi-n exploration with
adaptive time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Build sparse constraint matrix for LP oracle. Returns (row_idx, col_idx)."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns normalized g or None."""
    c = -np.ones(n)
    # Build values: for row k, col i, value is f[k-i]
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        vals[idx] = f[k - i]

    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()

    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def fw_lp_step(f, g, M, n):
    """Line search between f and g. Returns (best_alpha, best_val)."""
    best_alpha = 0.0
    best_val = M
    alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for alpha in alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    # Ternary search refinement
    if 0 < best_alpha < 1:
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(12):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.maximum((1 - m1) * f + m1 * g, 0)
            f1 = f1 / f1.sum()
            v1 = fast_convolve(f1).max()
            f2 = np.maximum((1 - m2) * f + m2 * g, 0)
            f2 = f2 / f2.sum()
            v2 = fast_convolve(f2).max()
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        ft = np.maximum((1 - mid) * f + mid * g, 0)
        ft = ft / ft.sum()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid

    return best_alpha, best_val

def compute_hankel_grad(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] with respect to a.
    grad[j] = 2 * sum_m a[m] * p[j+m]
    """
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, p[j:j + n])
    return grad


def game_projected_grad(a0, time_limit, T_start=0.001, lr_start=0.02):
    """
    Game-theoretic iteration with projected gradient on QP subproblem.
    Dual: p = softmax(conv(a)/T)
    Primal: a few steps of projected gradient on min a^T M(p) a
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    T = T_start
    lr = lr_start
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Inner loop: projected gradient on QP min a^T M(p) a
        a_inner = a.copy()
        for _ in range(5):
            grad = compute_hankel_grad(a_inner, p)
            candidate = project_simplex(a_inner - lr * grad)
            # Check if it helps
            new_obj = np.dot(p, fast_convolve(candidate))
            old_obj = np.dot(p, fast_convolve(a_inner))
            if new_obj < old_obj:
                a_inner = candidate
            else:
                break

        a = a_inner
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if no_improve > 30:
            T *= 1.03
            lr *= 0.98
        if no_improve > 100:
            T *= 0.7
            lr = lr_start * 0.7
        if no_improve > 300:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_start
            lr = lr_start
            no_improve = 0

    return best_a, best_obj

                    if ss < 1e-10:

                    candidate = candidate / ss
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True

        # Structured perturbation: add a sinusoidal pattern

    for n_try in [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]:
        if 128 <= n_try <= 2048:
            candidate_ns.append(n_try)
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns.sort()

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 350:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        # Quick FW-LP
        t_alloc = min(remaining * 0.04, 25)
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, f"fw_n{n_try}", (best_seq, best_score))

        # Quick smoothed gradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 300:
            a_opt2, _ = smoothed_grad(a_opt, time_limit=min(remaining2 * 0.02, 12),
                                      T_init=0.001, lr_init=0.03)
            best_seq, best_score = try_seq(a_opt2, f"sg_n{n_try}", (best_seq, best_score))

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best n ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 100:
        t_alloc = min(remaining * 0.35, 500)
        print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "deep_fw", (best_seq, best_score))

    # Perturbed FW-LP restarts
    for scale in [0.003, 0.008, 0.015, 0.03]:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc2 = min(remaining * 0.12, 120)
        a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2)
        best_seq, best_score = try_seq(a_opt2, f"fw_pert_{scale}", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Game-theoretic + Smoothed gradient ===
    print("\n=== Phase 3: Game & Smoothed Gradient ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 100:
        t_alloc = min(remaining * 0.25, 200)
        print(f"  Game projected grad, t={t_alloc:.0f}s")
        a_opt, _ = game_projected_grad(best_seq, time_limit=t_alloc,
                                       T_start=0.001, lr_start=0.015)
        best_seq, best_score = try_seq(a_opt, "game", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_alloc = min(remaining * 0.2, 150)
        print(f"  Smoothed grad T=0.0005, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc,
                                 T_init=0.0005, lr_init=0.015)
        best_seq, best_score = try_seq(a_opt, "smooth_fine", (best_seq, best_score))

    # Game with different T
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_best = len(best_seq)
        for T_init in [0.002, 0.005]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            a_opt, _ = game_projected_grad(a_pert, time_limit=min(remaining * 0.25, 80),
                                           T_start=T_init, lr_start=0.01)
            best_seq, best_score = try_seq(a_opt, f"game_T{T_init}", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Additional strategies ===
    print("\n=== Phase 4: Additional Strategies ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()

    # Try a different n for the best structure
    if remaining > 100:
        for n_try in [len(best_seq) * 2, len(best_seq) // 2]:
            if n_try < 128 or n_try > 2048:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(best_seq)), best_seq)
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining * 0.2, 80)
            a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
            best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", (best_seq, best_score))

    # Structured initializations
    if remaining > 80:
        n_best = len(best_seq)
        x = np.arange(n_best, dtype=np.float64)
        structures = []

        # Exponential decay
        for rate in [0.005, 0.01, 0.02]:
            exp_d = np.exp(-rate * x)
            structures.append(exp_d / exp_d.sum())

        # Two peaks (asymmetric)
        g1 = np.exp(-0.5 * ((x - 0.3 * (n_best - 1)) / (0.1 * n_best)) ** 2)
        g2 = np.exp(-0.5 * ((x - 0.7 * (n_best - 1)) / (0.15 * n_best)) ** 2)
        structures.append((g1 + g2) / (g1 + g2).sum())

        # Power law
        pw = 1.0 / (x + 1.0) ** 0.5
        structures.append(pw / pw.sum())

        for si, struct in enumerate(structures):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            # Blend with best
            for alpha in [0.3, 0.5, 0.7]:
                blend = (1 - alpha) * best_seq + alpha * struct
                blend = project_simplex(blend)
                t_alloc = min(remaining * 0.1, 30)
                a_opt, _ = fw_lp_optimize(blend, time_limit=t_alloc)
                best_seq, best_score = try_seq(a_opt, f"struct{si}_a{alpha}", (best_seq, best_score))

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Local search
    if remaining > 50:
        t_alloc = min(remaining * 0.3, 120)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, _ = local_search(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "local", (best_seq, best_score))

    # Very fine smoothed gradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.25, 60)
        print(f"  Fine smooth (T=0.0001), t={t_alloc:.0f}s")
        a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc,
                                 T_init=0.0001, lr_init=0.003)
        best_seq, best_score = try_seq(a_opt, "fine_smooth", (best_seq, best_score))

    # Final FW-LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.3, 40)
        print(f"  Final FW-LP, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "final_fw", (best_seq, best_score))

    # Tiny perturbations
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for _ in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            noise = rng.standard_normal(n_best) * 0.0005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smoothed_grad(f_pert, time_limit=min(remaining * 0.2, 8),
                                     T_init=0.0002, lr_init=0.003)
            best_seq, best_score = try_seq(a_opt, "micro_perturb", (best_seq, best_score))

    # Another local search pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, _ = local_search(best_seq, time_limit=min(remaining * 0.3, 25), seed=99999)
        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
