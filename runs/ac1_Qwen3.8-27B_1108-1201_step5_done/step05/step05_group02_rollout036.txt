Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses spectral (Gerchberg-Saxton) optimization, softmax-smoothed gradient descent,
LP-based Frank-Wolfe, and local coordinate descent.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    return best_f, best_obj

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: Spectral Gerchberg-Saxton, Softmax-smoothed gradient descent,
LP-based Frank-Wolfe, Local coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    if fft_size < 2:
        fft_size = 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve2(a, b):
    """Linear convolution of a and b via FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    if fft_size < 2:
        fft_size = 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0] / 1.0)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def hankel_vector_product(w, a):
    """Compute r[j] = sum_{m=0}^{n-1} w[m+j] * a[m] for j=0..n-1.
    w has length 2n-1, a has length n.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    w_rev = w[::-1]
    # Linear convolution of w_rev (len 2n-1) and a (len n) -> length 3n-2
    conv_full = fast_convolve2(w_rev, a)
    # r[j] = conv_full[2n-2-j] for j=0..n-1
    # So r = conv_full[n-1:2*n-1][::-1]
    r = conv_full[n - 1:2 * n - 1][::-1]
    return r


def spectral_gs_optimize(f0, time_limit, verbose=False):
    """Gerchberg-Saxton style spectral optimization for flat convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    if N < 2:
        N = 2

    # Target: flat convolution
    c_target = np.zeros(N)
    c_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    P_target = np.abs(np.fft.rfft(c_target)) ** 2

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lam = 0.5
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        A = np.fft.rfft(f, N)
        P = np.abs(A) ** 2

        ratio = P_target / (P + 1e-30)
        new_mag = np.abs(A) * np.power(ratio, lam)
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        f_new = np.fft.irfft(A_new, N)[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s_new

        obj = fast_convolve(f_new).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    GS iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        f = f_new

        if no_improve > 100:
            lam = max(0.05, lam * 0.9)
            no_improve = 0

        if it % 1000 == 0 and verbose:
            print(f"    GS iter {it}: cur={obj:.10f}, best={best_obj:.10f}, lam={lam:.3f}")

    return best_f, best_obj


def softmax_grad_optimize(f0, time_limit, verbose=False, beta_start=30.0, beta_end=800.0):
    """Gradient descent with softmax smoothing of the max operation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    lr = 0.05
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        progress = min(1.0, (it) / max(1, (time_limit * 15)))
        beta = beta_start * (beta_end / beta_start) ** progress

        conv = fast_convolve(f)

        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient: grad[j] = 2 * sum_m w[m+j] * a[m]
        r = hankel_vector_product(w, f)
        grad = 2.0 * r

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1

        f = f_new
        lr = max(0.001, lr * 0.9995)

        if no_improve > 600:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = 0.03
            no_improve = 0

        if it % 2000 == 0 and verbose:
            print(f"    SGD iter {it}: cur={obj_new:.10f}, best={best_obj:.10f}, beta={beta:.1f}")

    return best_f, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    M = fast_convolve(f).max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new

            if fast_convolve(f).max() < best_obj - 1e-15:
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.002 * min(no_improve // 40, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0

    return best_f, best_obj


def local_cd_optimize(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Focus on top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Try modifying coordinates that contribute to this peak
            n_coords = je - js + 1
            if n_coords > 30:
                # Randomly sample coordinates
                idx_sample = rng.integers(js, je + 1, size=min(30, n_coords))
            else:
                idx_sample = np.arange(js, je + 1)

            for j in idx_sample:
                if time.monotonic() - t0 > time_limit:
                    break

                old_val = f[j]
                if old_val < 1e-13:
                    continue

                best_j_obj = best_obj
                best_j_val = old_val

                for mult in [0.5, 0.7, 0.85, 1.15, 1.4, 1.8]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and passes % 50 == 0:
            print(f"    CD pass {passes}: best={best_obj:.10f}")

    return best_f, best_obj


def swap_optimize(f0, time_limit, verbose=False):
    """Try pairwise swaps/additions to reduce peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Find the peak
        pk = int(np.argmax(conv))
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)

        # Try to reduce the peak by moving mass from peak contributors to non-contributors
        if no_improve > 20:
            # Random perturbation
            idx1 = rng.integers(0, n)
            idx2 = rng.integers(0, n)
            if idx1 == idx2:
                continue
            amt = f[idx1] * rng.uniform(0.05, 0.3)
            f_trial = f.copy()
            f_trial[idx1] -= amt
            f_trial[idx2] += amt
            if f_trial[idx1] < 0:
                continue
            f_trial = f_trial / f_trial.sum()
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-15:
                f = f_trial
                best_obj = val
                best_f = f.copy()
                conv = fast_convolve(f)
                no_improve = 0
            else:
                no_improve += 1
            continue

        # Targeted: move mass from a peak contributor to a random non-contributor
        contributors = list(range(js, je + 1))
        non_contributors = [i for i in range(n) if i not in set(contributors)]
        if not non_contributors:
            no_improve += 1
            continue

        src = int(rng.choice(contributors))
        dst = int(rng.choice(non_contributors))

        for frac in [0.1, 0.2, 0.3, 0.5]:
            amt = f[src] * frac
            if amt < 1e-15:
                continue
            f_trial = f.copy()
            f_trial[src] -= amt
            f_trial[dst] += amt
            f_trial = f_trial / f_trial.sum()
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-15:
                f = f_trial
                best_obj = val
                best_f = f.copy()
                conv = fast_convolve(f)
                no_improve = 0
                break

        if time.monotonic() - t0 > time_limit:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Spectral GS (new approach)
    2. Softmax gradient descent
    3. LP-based Frank-Wolfe
    4. Swap/coordinate descent
    5. Multi-length exploration + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(640) / 640
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Spectral GS ===
    print(f"\n=== Phase 1: Spectral GS (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.25)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, _ = spectral_gs_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "gs_parent")

    # === Phase 2: Softmax Gradient Descent ===
    print(f"\n=== Phase 2: Softmax Gradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(350, rem * 0.22)
        if t2 > 30:
            print(f"  n={len(best_seq)}, Time budget: {t2:.0f}s")
            a_opt, _ = softmax_grad_optimize(best_seq, time_limit=t2,
                                             beta_start=20.0, beta_end=600.0, verbose=True)
            try_seq(a_opt, "sgd_main")

    # === Phase 3: LP-FW ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(350, rem * 0.22)
        if t3 > 30:
            print(f"  n={len(best_seq)}, Time budget: {t3:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lpfw_main")

    # === Phase 4: Swap/CD optimization ===
    print(f"\n=== Phase 4: Swap/CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t4a = min(150, rem * 0.12)
        if t4a > 20:
            print(f"  Swap opt, Time: {t4a:.0f}s")
            a_opt, _ = swap_optimize(best_seq, time_limit=t4a)
            try_seq(a_opt, "swap")

        rem = deadline - time.monotonic()
        t4b = min(100, rem * 0.08)
        if t4b > 15:
            print(f"  CD, Time: {t4b:.0f}s")
            a_opt, _ = local_cd_optimize(best_seq, time_limit=t4b)
            try_seq(a_opt, "cd")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    # Try a few nearby lengths
    target_lengths = []
    for delta in [-64, -32, 32, 64, 96, 128]:
        ln = n_base + delta
        if 300 <= ln <= 2000:
            target_lengths.append(ln)
    # Also try powers of 2
    for pn in [512, 640, 768, 832, 960, 1024]:
        if pn not in target_lengths and 300 <= pn <= 2000:
            target_lengths.append(pn)

    target_lengths = sorted(set(target_lengths))[:6]

    for tl in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, tl)
        t_alloc = min(rem * 0.25, 120)
        if t_alloc < 15:
            break
        print(f"  Trying n={tl}, time={t_alloc:.0f}s")

        # Quick GS + SGD
        a_opt, _ = spectral_gs_optimize(f_interp, time_limit=t_alloc * 0.4)
        score_gs = try_seq(a_opt, f"gs_n{tl}")

        rem2 = deadline - time.monotonic()
        t_alloc2 = min(rem2 * 0.2, 60)
        if t_alloc2 > 10:
            a_opt, _ = softmax_grad_optimize(a_opt, time_limit=t_alloc2,
                                             beta_start=50.0, beta_end=400.0)
            try_seq(a_opt, f"sgd_n{tl}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic()
        if rem > 30:
            # Quick GS
            t_pol = min(rem * 0.3, 80)
            a_opt, _ = spectral_gs_optimize(best_seq, time_limit=t_pol)
            try_seq(a_opt, "gs_polish")

            rem = deadline - time.monotonic()
            if rem > 20:
                # Quick SGD
                t_pol2 = min(rem * 0.3, 50)
                a_opt, _ = softmax_grad_optimize(best_seq, time_limit=t_pol2,
                                                 beta_start=100.0, beta_end=500.0)
                try_seq(a_opt, "sgd_polish")

            rem = deadline - time.monotonic()
            if rem > 15:
                # Quick LP-FW
                t_pol3 = min(rem * 0.4, 40)
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_pol3)
                try_seq(a_opt, "lpfw_polish")

        # Small perturbations
        for p in range(4):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0005 * p)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.3, 20)
            if t_pert < 5:
                break
            a_opt, _ = softmax_grad_optimize(f_pert, time_limit=t_pert,
                                             beta_start=80.0, beta_end=400.0)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
