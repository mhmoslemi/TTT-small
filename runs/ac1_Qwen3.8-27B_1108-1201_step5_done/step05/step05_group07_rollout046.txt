import numpy as np
import time
from scipy.optimize import minimize, line_search

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(a, tau):
    """Compute gradient of smoothed objective (1/tau)*log(sum exp(tau*b_k))."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient: grad_i = 2 * sum_j w_{i+j} * a_j
    size = 1 << (3 * n - 1).bit_length()
    W = np.zeros(size)
    W[:2*n-1] = w
    A = np.zeros(size)
    A[:n] = a
    corr = np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))).real
    grad = 2.0 * corr[:n]
    
    # Project gradient to tangent space of simplex (remove mean component)
    # Actually, for projected gradient, we don't need to project the gradient
    # The projection step handles the constraint
    return grad, conv.max()

def smoothed_optimize(f0, time_limit, tau_start=10.0, tau_end=0.01, 
                      lr_start=0.01, momentum=0.9, n_tau_steps=100):
    """Optimize using smoothed objective with continuation on tau."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    # Continuation schedule
    tau = tau_start
    lr = lr_start
    v = np.zeros(n)  # momentum
    
    while time.monotonic() - t0 < time_limit:
        grad, cur_max = compute_smoothed_grad(f, tau)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        
        # Gradient step
        f_new = f - lr * v
        
        # Project to simplex
        f_new = project_simplex(f_new)
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
        
        # Accept or reject (simple backtracking)
        if obj_new < cur_max + 1e-10:
            f = f_new
        else:
            # Try smaller step
            lr *= 0.5
            f = f  # Keep old
            # Recompute with smaller step
            f_new = f - lr * v
            f_new = project_simplex(f_new)
            f = f_new
        
        # Decrease tau (continuation)
        tau *= 0.995
        if tau < tau_end:
            tau = tau_end
        
        # Increase learning rate slowly
        lr *= 1.001
        lr = min(lr, lr_start * 2)
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed objective with L-BFGS-B,
continuation on tau, active-set refinement, and coordinate descent polishing.
Key innovations: exact O(n log n) gradient via FFT cross-correlation, homotopy
Continuation method, and systematic peak-targeted local search.

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_objective_and_grad(V, tau):
    """Compute smoothed objective and gradient w.r.t. V (softmax parameterization)."""
    n = len(V)
    # Compute a = softmax(V)
    V_shifted = V - V.max()
    exp_V = np.exp(V_shifted)
    S = exp_V.sum()
    a = exp_V / S

    # Compute convolution
    conv = fast_convolve(a)

    # Compute softmax weights w_k
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum

    # Objective: (1/tau) * log(sum_k exp(tau * b_k)) = max_b + (1/tau)*log(sum exp(tau*(b_k-max_b)))
    obj = conv.max() + (1.0 / tau) * np.log(np.exp(tau * (conv - conv.max())).sum())

    # Gradient: dF/dV_i = 2 * a_i * (g_i - t_w)
    # where g_i = sum_k w_k * a_{k-i} (cross-correlation)
    # and t_w = sum_k w_k * b_k
    t_w = np.dot(w, conv)

    # Cross-correlation: g_i = sum_j w_{i+j} * a_j
    size = 1 << (3 * n - 1).bit_length()
    W = np.zeros(size, dtype=np.complex128)
    W[:2 * n - 1] = w
    A = np.zeros(size, dtype=np.complex128)
    A[:n] = a
    corr = np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))).real
    g = corr[:n]

    grad = 2.0 * a * (g - t_w)
    return obj, grad


def lbfgsb_smoothed(f0, time_limit, tau_start=50.0, tau_end=5000.0,
                    max_iter_per_tau=30, tau_decay=0.97, verbose=False, label=""):
    """
    Optimize using L-BFGS-B on the softmax parameterization with continuation on tau.
    Starts with small tau (smooth) and increases to large tau (sharp, approaches true max).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    # Initialize V from f (log of f, with floor for small values)
    V = np.log(np.maximum(f, 1e-15))
    V -= V.max()

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    tau = tau_start
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        # L-BFGS-B step at current tau
        def func_and_grad(Vx):
            nx = len(Vx)
            Vx_shifted = Vx - Vx.max()
            exp_Vx = np.exp(Vx_shifted)
            a = exp_Vx / exp_Vx.sum()

            conv = fast_convolve(a)
            scaled = tau * conv
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = w / w_sum
            else:
                w = w / w_sum
            obj = conv.max() + (1.0 / tau) * np.log(np.exp(tau * (conv - conv.max())).sum())

            t_w = np.dot(w, conv)
            size = 1 << (3 * nx - 1).bit_length()
            W = np.zeros(size, dtype=np.complex128)
            W[:2 * nx - 1] = w
            A = np.zeros(size, dtype=np.complex128)
            A[:nx] = a
            corr = np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))).real
            g = corr[:nx]
            grad = 2.0 * a * (g - t_w)
            return obj, grad

        result = minimize(func_and_grad, V, method='L-BFGS-B',
                         options={'maxiter': max_iter_per_tau, 'gtol': 1e-6, 'maxfun': max_iter_per_tau * 3})

        V = result.x
        # Convert back to a
        V_shifted = V - V.max()
        exp_V = np.exp(V_shifted)
        f_new = exp_V / exp_V.sum()
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()

        # Evaluate true objective
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()

        f = f_new
        total_iters += result.nit if result.nit else 0

        # Decrease tau (make objective sharper)
        tau /= tau_decay
        if tau > tau_end:
            tau = tau_end

        if verbose and total_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    tau={tau:.1f}, obj={best_obj:.10f}, t={elapsed:.1f}s")

        if tau >= tau_end and result.success:
            # Convergence at highest tau
            break

    return best_f, best_obj


def projected_gradient_momentum(f0, time_limit, lr=0.005, momentum=0.9,
                                 n_peaks=50, tau=100.0, verbose=False, label=""):
    """Projected gradient descent with momentum using smoothed gradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    v = np.zeros(n)
    cur_lr = lr
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Compute smoothed gradient (projected onto simplex tangent)
        scaled = tau * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        t_w = np.dot(w, conv)
        size = 1 << (3 * n - 1).bit_length()
        W = np.zeros(size, dtype=np.complex128)
        W[:2 * n - 1] = w
        A = np.zeros(size, dtype=np.complex128)
        A[:n] = f
        corr = np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))).real
        g = corr[:n]

        grad = 2.0 * (g - t_w)
        # Project gradient to simplex tangent (subtract mean)
        grad -= grad.mean()

        # Momentum
        v = momentum * v + (1 - momentum) * grad

        # Step
        f_new = f - cur_lr * v
        f_new = project_simplex(f_new)

        # Evaluate
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive schedule
        cur_lr *= 0.9998
        tau *= 1.01
        if tau > 5000:
            tau = 5000

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 7) % 1000000)
            noise_scale = 0.003 + 0.002 * min(no_improve // 400, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = lr * 0.7
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_analytical(f0, time_limit, verbose=False, label=""):
    """
    Coordinate descent: for each coordinate j, find the optimal value by
    binary search on the 1D objective (treating other coordinates proportionally fixed).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Order: center-out for better initial progress
    mid = n // 2
    order = list(range(n))
    order.sort(key=lambda i: abs(i - mid))

    eps = 1e-14
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            # Try scaling a_j and proportionally adjusting others
            # New a_j = new_val, others scaled by (1-new_val)/(1-old_val)
            best_j_val = old_val
            best_j_obj = best_obj

            # Golden section search in [0, min(0.5, 3*old_val)]
            lo = max(0.0, old_val * 0.1)
            hi = min(0.499, old_val * 3.0)
            if hi - lo < 1e-8:
                continue

            # Try a few points
            candidates = np.linspace(lo, hi, 5)
            candidates = np.append(candidates, [old_val * 0.5, old_val * 1.5, old_val * 2.0])
            candidates = np.clip(candidates, 1e-8, 0.499)

            for new_val in candidates:
                if abs(new_val - old_val) < 1e-10:
                    continue
                scale = (1.0 - new_val) / (1.0 - old_val)
                if scale <= 0 or scale > 2.0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                others = np.ones(n)
                others[j] = 0
                f_trial *= np.where(others > 0, scale, 1.0)
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-10:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-13:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / (1.0 - old_val)
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = np.maximum(f, 0)
                f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def peak_targeted_search(f0, time_limit, verbose=False, label=""):
    """
    Peak-targeted local search: identify the top peaks, then for each peak,
    find the pair (i,j) whose transfer most reduces that peak, execute it.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        cur_max = conv.max()
        # Find all peaks within 0.5% of max
        threshold = cur_max * 0.995
        active_peaks = np.where(conv > threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])
        active_peaks = active_peaks[:30]  # Limit to 30

        best_improvement = 0
        best_move = None

        # For each active peak, try to reduce it
        for k in active_peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]

            # Find the top contributor
            hi_local = np.argmax(contribs)
            hi = idx[hi_local]
            hi_contrib = contribs[hi_local]
            if hi_contrib < 1e-10:
                continue

            # Try transferring a fraction of f[hi] to a low-contributor position
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = f[hi] * frac
                if amt < 1e-12:
                    continue

                # Find good target: minimize increase elsewhere
                # Simply try random targets
                for _ in range(3):
                    j = rng.integers(0, n)
                    if j == hi:
                        continue
                    f[hi] -= amt
                    f[j] += amt
                    val = fast_convolve(f).max()
                    improvement = cur_max - val
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (hi, j, amt, val)
                    f[hi] += amt
                    f[j] -= amt

        # Also try random pairs
        for _ in range(20):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-12:
                continue
            for frac in [0.1, 0.2, 0.4]:
                amt = f[i] * frac
                f[i] -= amt
                f[j] += amt
                val = fast_convolve(f).max()
                improvement = cur_max - val
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_move = (i, j, amt, val)
                f[i] += amt
                f[j] -= amt

        if best_move is not None and best_improvement > 1e-12:
            hi, j, amt, new_val = best_move
            f[hi] -= amt
            f[j] += amt
            f = np.maximum(f, 0)
            f /= f.sum()
            if new_val < best_obj - 1e-13:
                best_obj = new_val
                best_f = f.copy()
        else:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window (smooth taper)
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Gaussian
    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Smoothed random
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Logarithmic (taper toward edges)
    center = n / 2.0
    dist = np.abs(np.arange(n) - center) / center
    log_taper = 1.0 / (1.0 + dist * (2.0 + rng.random() * 3.0))
    log_taper = np.maximum(log_taper, 1e-10)
    inits.append(log_taper / log_taper.sum())

    # Cosine-raised with parameter
    alpha = 1.0 + rng.random() * 2.0
    cos_raised = np.cos(np.pi * x) ** (2 * alpha)
    cos_raised = np.maximum(cos_raised, 1e-10)
    inits.append(cos_raised / cos_raised.sum())

    return inits[:count]

                                    max_iter_per_tau=40, tau_decay=0.96,
                                    label=f"p1_init{idx}")
        try_seq(a_opt, f"init_{idx}")

    # === Phase 2: Projected gradient with momentum ===
    print(f"\n=== Phase 2: Projected Gradient (best={best_score:.6f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        pg_configs = [
            {"lr": 0.004, "mom": 0.90, "tau": 50.0, "frac": 0.2},
            {"lr": 0.006, "mom": 0.85, "tau": 200.0, "frac": 0.15},
            {"lr": 0.002, "mom": 0.93, "tau": 1000.0, "frac": 0.15},
        ]
        for idx, cfg in enumerate(pg_configs):
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(rem * cfg["frac"], 200)
            if t_alloc < 10:
                break
            print(f"  P2-{idx}: lr={cfg['lr']}, mom={cfg['mom']}, tau={cfg['tau']}, t={t_alloc:.0f}s")
            a_opt, _ = projected_gradient_momentum(best_seq, time_limit=t_alloc,
                                                    lr=cfg["lr"], momentum=cfg["mom"],
                                                    tau=cfg["tau"], label=f"pg_{idx}")
            try_seq(a_opt, f"pg_{idx}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [640, 896, 1024, 1280]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 120
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 150)
            if t_alloc < 10:
                break
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lbfgsb_smoothed(f_interp, time_limit=t_alloc,
                                        tau_start=30.0, tau_end=3000.0,
                                        max_iter_per_tau=40, tau_decay=0.96,
                                        label=f"len_{target_n}")
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 4: Peak-targeted local search ===
    print(f"\n=== Phase 4: Peak-Targeted Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_pks = min(rem * 0.5, 120)
            print(f"  Peak search t={t_pks:.0f}s")
            a_opt, _ = peak_targeted_search(best_seq, time_limit=t_pks, label="peak")
            try_seq(a_opt, "peak_search")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_cd = min(rem * 0.7, 150)
            print(f"  Coord descent t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_analytical(best_seq, time_limit=t_cd, label="cd")
            try_seq(a_opt, "coord_desc")

    # === Phase 6: Final refinement with very sharp tau ===
    print(f"\n=== Phase 6: Final Sharpening (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_final = min(rem * 0.5, 80)
            print(f"  Final L-BFGS-B t={t_final:.0f}s")
            a_opt, _ = lbfgsb_smoothed(best_seq, time_limit=t_final,
                                        tau_start=500.0, tau_end=20000.0,
                                        max_iter_per_tau=30, tau_decay=1.02,
                                        label="final")
            try_seq(a_opt, "final_sharp")

        # Peak search again
        rem = deadline - time.monotonic() - 30
        if rem > 8:
            t_pk2 = min(rem * 0.5, 50)
            a_opt, _ = peak_targeted_search(best_seq, time_limit=t_pk2, label="peak2")
            try_seq(a_opt, "peak_search2")

        # Coordinate descent again
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            t_cd2 = min(rem * 0.7, 40)
            a_opt, _ = coordinate_descent_analytical(best_seq, time_limit=t_cd2, label="cd2")
            try_seq(a_opt, "coord_desc2")

    # === Phase 7: Multi-start perturbation ===
    print(f"\n=== Phase 7: Multi-Start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for attempt in range(6):
            rem = deadline - time.monotonic() - 15
            if rem < 8:
                break
            noise_scale = 0.002 + attempt * 0.001
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.4, 30)
            if t_alloc < 5:
                break
            a_opt, _ = lbfgsb_smoothed(f_pert, time_limit=t_alloc,
                                        tau_start=100.0, tau_end=5000.0,
                                        max_iter_per_tau=20, tau_decay=1.03,
                                        label=f"pert{attempt}")
            try_seq(a_opt, f"pert_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
