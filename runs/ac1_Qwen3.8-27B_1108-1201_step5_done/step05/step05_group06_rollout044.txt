def lp_step(a, M):
    """Solve the LP step: find g that maximizes sum(g) subject to a*g <= M and g >= 0."""
    n = len(a)
    g = cp.Variable(n)
    constraints = [g >= 0]
    
    # Convolution constraints: (a*g)[k] <= M for k=0,...,2n-2
    for k in range(2*n - 1):
        # (a*g)[k] = sum_j a[j] * g[k-j] where 0<=j<n and 0<=k-j<n

        je = min(n - 1, k)

            constraints.append(conv_k <= M)

    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if g.value is None:

    return np.maximum(g.value, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with line search, combined with coordinate bisection polishing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_conv_matrix(a):
    """Create matrix C where (C@g)[k] = (a*g)[k] for linear conv."""
    n = len(a)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        # a[j] * g[k-j] contributes to C[k, j] = a[j] when k-j is in [0, n)
        # i.e., k in [j, j+n-1]
        k_start = j
        k_end = min(j + n, 2*n - 1)
        C[k_start:k_end, j] = a[:k_end - k_start]
    return C


def lp_direction(a, solver='ECOS'):
    """Solve LP: max sum(g) s.t. g>=0, conv(a,g) <= M."""
    n = len(a)
    M = fast_convolve(a).max()
    g = cp.Variable(n)
    C = make_conv_matrix(a)
    constraints = [g >= 0, C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=solver)
    if g.value is None:
        return a.copy(), M
    g_val = np.maximum(g.value, 0.0)
    return g_val, M


def find_step_size(a, g, M, tol=1e-10):
    """Find largest t in [0,1] such that max(conv((1-t)*a+t*g, (1-t)*a+t*g)) <= M."""
    n = len(a)
    
    def check(t):
        a_new = (1-t)*a + t*g
        return fast_convolve(a_new).max() - M
    
    # Binary search
    lo, hi = 0.0, 1.0
    # First check if t=1 works
    if check(1.0) <= tol:
        return 1.0
    # Find the boundary
    for _ in range(50):
        mid = (lo + hi) / 2
        if check(mid) <= tol:
            lo = mid
        else:
            hi = mid
    return lo


def lp_descent(a0, time_limit, verbose=False):
    """LP-based descent with line search."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

        if s_new < 1e-10:

            best_a = a_new.copy()
            a = a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with
line search, soft-max gradient descent, coordinate bisection polishing,
and multi-restart strategy. The LP approach finds the optimal direction g
maximizing sum(g) subject to conv(a,g) <= M and g >= 0.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_conv_matrix(a):
    """Create matrix C where (C@g)[k] = sum_j a[k-j]*g[j] (linear conv of a and g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        C[k_start:k_end, j] = a[:k_end - k_start]
    return C


def lp_direction(a, solver='CLARABEL'):
    """Solve LP: max sum(g) s.t. g>=0, conv(a,g) <= M. Returns (g, M)."""
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()
    g = cp.Variable(n)
    C = make_conv_matrix(a)
    constraints = [g >= 0, C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return a.copy(), M
    if g.value is None:
        return a.copy(), M
    g_val = np.maximum(g.value, 0.0)
    return g_val, M


def find_step_size(a, g, M, max_iter=40):
    """Find largest t in [0,1] such that max(conv((1-t)*a + t*g, itself)) <= M * (1+1e-10)."""
    tol = M * 1e-10

    def check(t):
        a_new = (1.0 - t) * a + t * g
        return fast_convolve(a_new).max() - M

    if check(1.0) <= tol:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        if check(mid) <= tol:
            lo = mid
        else:
            hi = mid
    return lo


def lp_descent(a0, time_limit, verbose=False):
    """LP-based descent with line search."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    solver = 'CLARABEL'

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_start = time.monotonic()

        try:
            g, M = lp_direction(a, solver=solver)
        except Exception:
            solver = 'ECOS'
            try:
                g, M = lp_direction(a, solver=solver)
            except Exception:
                break

        t_lp = time.monotonic() - t_start

        if g.sum() < 1e-10:
            break

        g = g / g.sum()

        if np.allclose(g, a, atol=1e-8):
            break

        t = find_step_size(a, g, M)
        if t < 1e-8:
            # Try a fraction
            t = 0.5 * t
            if t < 1e-10:
                break

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            break
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        a = a_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"    LP iter {it}: obj={best_obj:.10f}, t={t:.4f}, "
                      f"lp_time={t_lp:.2f}s")

    return best_a, best_obj


def softmax_subgrad(a, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def gradient_descent(a0, time_limit, lr=0.01, mom=0.9, top_k=50,
                     temp=0.0005, decay=0.9997, verbose=False):
    """Projected gradient descent with momentum on simplex."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        sg = softmax_subgrad(a, conv, top_k=top_k, temp=temp)
        v = mom * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= decay
        temp *= 1.002

        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            noise_scale = 0.003 * min(no_improve // 100, 3)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 0.0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = 0.005
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    GD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj


def coordinate_bisection(a0, time_limit, max_passes=200, verbose=False):
    """Coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            x_old = a[j]
            if x_old < 1e-14:
                continue
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo, hi = 0.0, 1.0
            x_new = x_old
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-12:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-12:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-13:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-12:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
                if verbose and passes <= 2:
                    print(f"    CD pass {passes}, coord {j}: obj={best_obj:.10f}")

        if not pass_improved:
            if passes > 1:
                rng = np.random.default_rng(passes * 31 + 7)
                a = np.maximum(best_a + rng.standard_normal(n) * 0.0003, 0.0)
                a = a / a.sum()
            else:
                break

    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric and renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                    f_pert = np.maximum(best_a + rng_p.standard_normal(n_best) * 0.001, 0)
                    f_pert = f_pert / f_pert.sum()
                    t_alloc = min(rem, 20)
                    a_opt, _ = coordinate_bisection(f_pert, time_limit=t_alloc)
                    update_best(a_opt, f"perturb_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
