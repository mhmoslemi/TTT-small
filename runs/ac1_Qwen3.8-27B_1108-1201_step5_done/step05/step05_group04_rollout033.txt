import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_grad(f, w, max_peaks=500):
    """Compute gradient of sum_k w[k]*conv(k) w.r.t. f.
    Gradient of conv(k) w.r.t. f[i] = 2*f[k-i] if 0<=k-i<n, else 0.
    """
    n = len(f)
    conv_len = 2 * n - 1

    if len(w) > max_peaks:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        sig_idx = sig_idx[w[sig_idx] > 1e-18 * (w[sig_idx].max() if w[sig_idx].max() > 0 else 1.0)]
    else:
        sig_idx = np.arange(len(w))
        sig_idx = sig_idx[w[sig_idx] > 1e-18]

    grad = np.zeros(n)
    for k in sig_idx:
        wk = w[k]
        if wk < 1e-18:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            grad[js] += 2.0 * wk * f[k - js]
    return grad


def nesterov_gradient(f0, time_limit, mode='lse', beta_start=50.0, beta_end=100000.0,
                      max_peaks=300, top_k=10, lr_init=None, momentum=0.92):
    """Nesterov-accelerated gradient optimization.
    mode: 'lse' for LSE surrogate, 'topk' for sum-of-top-K peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    if lr_init is None:
        lr = 3.0 / np.sqrt(n)
    else:
        lr = lr_init

    v = np.zeros(n)
    x_prev = f.copy()
    cur_beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        if it > 1:
            mu = (momentum * np.sqrt(1 + 4 * it) - 1) / 2.0
            mu = min(mu, 1.0)
            y = f + mu * (f - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = f.copy()
            else:
                y = y / sy
        else:
            y = f.copy()

        # Compute gradient at y
        conv_y = fast_convolve(y)
        if mode == 'lse':
            bc = cur_beta * conv_y
            bc_max = bc.max()
            exp_bc = np.exp(bc - bc_max)
            w = exp_bc / exp_bc.sum()
            grad = compute_weighted_grad(y, w, max_peaks)
        else:
            k_top = min(top_k, len(conv_y))
            top_idx = np.argpartition(conv_y, -k_top)[-k_top:]
            w = np.zeros(len(conv_y))
            w[top_idx] = 1.0
            grad = compute_weighted_grad(y, w, max_peaks)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 200
            break
        grad = grad / grad_norm

        # Update
        v = momentum * v + grad
        f_new = project_simplex(f - lr * v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        x_prev = f.copy()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.004, 8.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.94
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                x_prev = f.copy()
                lr = 3.0 / np.sqrt(n)
                cur_beta = min(cur_beta * 1.8, beta_end)
                no_improve = 0

        cur_beta = min(cur_beta * 1.006, beta_end)

        # Random restart if very stuck
        if no_improve > 150:
            rng = np.random.default_rng((it * 6151 + 13) % 999983)
            scale = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            x_prev = f.copy()
            lr = 2.0 / np.sqrt(n)
            no_improve = 0
            cur_beta = max(beta_start, cur_beta * 0.7)

    return best_f, best_obj


def subgradient_polyak(f0, time_limit, damping=0.3, n_peaks=1):
    """Subgradient method on true max with Polyak step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    # Lower bound on max_conv: 1/(2n-1)
    lb_max_conv = 1.0 / conv_len
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    curvature_sum = 1e-8

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_val = conv.max()

        # Gap for Polyak step
        gap = cur_val - lb_max_conv
        if gap < 1e-15:
            break

        # Subgradient: pick the argmax bin
        if n_peaks == 1:
            k_star = int(np.argmax(conv))
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            js = np.arange(lo, hi + 1)
            g = np.zeros(n)
            g[js] = 2.0 * f[k_star - js]
        else:
            top_ind = np.argpartition(conv, -n_peaks)[-n_peaks:]
            g = np.zeros(n)
            for k in top_ind:
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                js = np.arange(lo, hi + 1)
                g[js] += 2.0 * f[k - js]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-18:
            break

        # Polyak step
        curvature_sum = 0.9 * curvature_sum + 0.1 * g_norm
        t_polyak = (gap / (curvature_sum * curvature_sum)) * damping
        t_step = min(t_polyak, 0.5 / g_norm)

        f_new = project_simplex(f + t_step * g * (-1.0))  # descent: -grad
        # Wait, the gradient points in the direction of INCREASING conv at the peak.
        # We want to DECREASE it, so we go in the -gradient direction.
        # But the gradient of max_conv w.r.t. f is the subgradient pointing "up".
        # So descent is f - t * g (where g is the subgradient).
        # But we need to stay on the simplex. project_simplex handles that.

        f_new = project_simplex(f - t_step * g / g_norm)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                f = best_f.copy()
                no_improve = 0
                # Perturb
                rng = np.random.default_rng(it * 997)
                f = project_simplex(f + rng.standard_normal(len(f)) * 0.001)

    return best_f, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

                                     max_peaks=400, momentum=0.93,
                                     lr_init=1.0 / np.sqrt(n_cur))
        try_sequence(f_opt, best_tracker)

    # Subgradient polish
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(100, t4_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Subgrad n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_polyak(current, time_limit=t_alloc, damping=0.2, n_peaks=1)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Final micro-polish ===
    print(f"\n=== Phase 5: Final polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    # Very high beta, small steps
    t_alloc = min(150, deadline - time.monotonic() - 30)
    if t_alloc > 20:
        print(f"  Ultra-high beta LSE n={n_cur}, t={t_alloc:.0f}s, beta=[2000, 500000]")
        f_opt, _ = nesterov_gradient(current, time_limit=t_alloc, mode='lse',
                                     beta_start=2000.0, beta_end=500000.0,
                                     max_peaks=200, momentum=0.95,
                                     lr_init=0.3 / np.sqrt(n_cur))
        try_sequence(f_opt, best_tracker)

    # Directed top-1 subgradient with very small steps
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(80, deadline - time.monotonic() - 15)
    if t_alloc > 15:
        print(f"  Top-1 subgrad n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = nesterov_gradient(current, time_limit=t_alloc, mode='topk',
                                     top_k=1, momentum=0.95,
                                     lr_init=0.2 / np.sqrt(len(current)))
        try_sequence(f_opt, best_tracker)

    # Try a nearby n
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        for off in [64, -64, 128]:
            n_try = n_best + off
            if n_try < 256 or n_try > 1500:
                continue
            t_remaining = deadline - time.monotonic() - 10
            if t_remaining < 30:
                break
            t_alloc = min(60, t_remaining * 0.5)
            if t_alloc < 20:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = nesterov_gradient(f_init, time_limit=t_alloc, mode='lse',
                                         beta_start=100.0, beta_end=100000.0,
                                         max_peaks=200, momentum=0.92)
            try_sequence(f_opt, best_tracker)

    # === Final ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: fallback")
    fallback = np.ones(768) / 768
    return [float(x) for x in fallback]
