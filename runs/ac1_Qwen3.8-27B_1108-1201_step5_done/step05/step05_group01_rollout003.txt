            m_relax = max(1.0, m_relax * 0.95)

        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax *= 1.03
            continue

        g = g / g_sum

        # Check convergence
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                # Kick with noise
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.002 + 0.002 * min(no_improve // 80, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                no_improve = 0
                continue
            m_relax = min(m_relax * 1.02, 2.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        # FW step size + grid
        fw_alpha = 2.0 / (it + 2.0)
        alphas = sorted(set([0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5,
                             fw_alpha, 0.6, 0.7, 0.8, 0.9, 1.0]))

        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.12)
            hi = min(1.0, best_alpha + 0.12)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.04, 3.0)

    return best_f, best_obj


def smooth_optimize(f0, time_limit, beta0=50.0, beta_end=500.0, lr0=0.01):
    """
    Log-sum-exp smoothed optimization with adaptive step size.
    Minimizes (1/beta)*log(sum exp(beta*conv[k])) on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    beta = beta0
    lr = lr0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 1:
            break

        conv = fast_convolve(f)
        # Log-sum-exp gradient
        scaled = beta * conv
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        weights = exp_scaled / exp_scaled.sum()

        # Gradient of (1/beta)*log(sum exp(beta*conv[k])) wrt a
        # = sum_k w_k * grad conv[k]
        # grad conv[k] / grad a[m] = 2*a[k-m] if 0 <= k-m < n
        grad = np.zeros(n)
        # Use FFT for efficiency
        # The gradient at m is: 2 * sum_k w_k * a[k-m] (when 0<=k-m<n)
        # = 2 * (w * a_reversed)[m] convolved appropriately
        # Let's do it directly: for each k, grad[m] += 2 * w[k] * a[k-m] for m in [k-n+1, k]
        # This is a weighted cross-correlation of w and a
        # grad = 2 * ifft(fft(w, 2n) * conj(fft(a, 2n)))... 
        # Actually simpler: grad[m] = 2 * sum_{k: 0<=k-m<n} w[k] * a[k-m]
        # Let j = k-m, so grad[m] = 2 * sum_j w[m+j] * a[j] for j=0..n-1, m+j in [0, 2n-2]
        # This is the cross-correlation of w and a!
        # cross_corr[m] = sum_j w[m+j] * a[j]
        
        # Use FFT: cross_corr = ifft(fft(w, 2n-1) * conj(fft(a, 2n-1)))
        # Actually, let's use the direct formula with numpy
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = np.fft.rfft(conv * weights, fft_size)  # weighted conv
        A = np.fft.rfft(f, fft_size)
        # Cross-correlation: sum_j W[k] * A[k-m]... 
        # Let me just compute it directly for the non-zero weights
        # For efficiency, only use top-k weights
        k_threshold = 50
        top_k = np.argpartition(weights, -k_threshold)[-k_threshold:]
        for k in top_k:
            wk = weights[k]
            if wk < 1e-12:
                continue
            # grad[m] += 2 * wk * a[k-m] for m where 0 <= k-m < n
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                grad[m_lo:m_hi + 1] += 2.0 * wk * f[k - np.arange(m_lo, m_hi + 1)]

        # Step
        f_new = f - lr * grad
        f_new = project_simplex(f_new)
        
        new_conv = fast_convolve(f_new)
        new_obj = new_conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            # Adaptive learning rate
            lr *= 0.999
            if no_improve > 100:
                # Increase beta and restart
                beta = min(beta * 1.5, beta_end)
                lr = min(lr * 2.0, lr0)
                no_improve = 0
            no_improve += 1

        # Slowly increase beta
        progress = min(1.0, it / max(1, int(time_limit / 2)))
        beta = beta0 * (beta_end / beta0) ** progress

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate j, the objective max_k conv[k] as a function of f[j]
    (with others rescaled) is a piecewise quadratic.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break

            old_val = f[j]
            if old_val < 1e-15:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Binary search for optimal f[j]
            # f[j] = x, others scaled by (1-x)/denom
            # Objective: max_k conv(new_f)[k]
            
            def eval_coord(x):
                ft = f.copy()
                ft[j] = x
                if 1.0 - x < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * ((1.0 - x) / denom)
                return fast_convolve(ft).max()

            # Golden section search
            GR = (1.0 + np.sqrt(5.0)) / 2.0
            inv_gr = 1.0 / GR
            lo = 0.0
            hi = min(0.9999, 3.0 * old_val)
            
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(14):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(f).max()

            if opt_val < cur_val - 1e-13:
                ft = f.copy()
                ft[j] = opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * ((1.0 - opt_x) / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if not pass_improved:
            break

    return best_f, best_obj


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW + smooth optimization + coordinate descent.
    Starts from parent sequence and explores nearby landscape.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default uniform: n=768, score={p_score:.8f}")

    n_parent = len(parent)
    try_seq(parent, "parent")

    # === Phase 1: LP-FW on parent (majority of time) ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 600)
    print(f"  Time budget: {t1:.0f}s")
    a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True)
    try_seq(a_opt, "lpfw_main")

    # === Phase 2: Smooth optimization on best ===
    print(f"\n=== Phase 2: Smooth optimization ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t2 = min(rem * 0.2, 200)
            print(f"  Time budget: {t2:.0f}s")
            a_opt, obj = smooth_optimize(best_seq, time_limit=t2, beta0=30.0, beta_end=300.0, lr0=0.005)
            try_seq(a_opt, "smooth")

    # === Phase 3: Second LP-FW pass on best (with different settings) ===
    print(f"\n=== Phase 3: LP-FW pass 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t3 = min(rem * 0.25, 300)
            print(f"  Time budget: {t3:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lpfw_pass2")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coordinate descent polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t4 = min(rem * 0.3, 250)
            print(f"  Time budget: {t4:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t4)
            try_seq(a_opt, "coord_desc")

    # === Phase 5: Multi-start exploration ===
    print(f"\n=== Phase 5: Multi-start ===")
    if best_seq is not None:
        n_best = len(best_seq)
        starts = [
            ("noise_small", project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)),
            ("noise_med", project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)),
            ("cosine_mix", project_simplex(0.85 * best_seq + 0.15 * cosine_modulated_init(n_best, 2, 0.3))),
            ("reverse_mix", project_simplex(0.8 * best_seq + 0.2 * best_seq[::-1])),
        ]
        for name, f_init in starts:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 80)
            print(f"  {name}: t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"multi_{name}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            # Quick LP-FW
            t6 = min(rem * 0.4, 60)
            if t6 > 5:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t6)
                try_seq(a_opt, "final_lpfw")

            # Quick CD
            rem = deadline - time.monotonic()
            if rem > 10:
                t6b = min(rem * 0.6, 80)
                if t6b > 5:
                    a_opt, _ = coordinate_descent(best_seq, time_limit=t6b)
                    try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
