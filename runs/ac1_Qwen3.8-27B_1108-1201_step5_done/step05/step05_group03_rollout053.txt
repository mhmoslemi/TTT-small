def fast_convolve(a):

                best_val = best_M  # Use current best as baseline

                for t in [0.0, 1.0, best_t - 0.2, best_t - 0.1, best_t, best_t + 0.1, best_t + 0.2]:
                    t = max(0.0, min(1.0, t))
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    f_trial /= f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_val:
                        best_val = val
                        best_t = t

                if best_val < best_M - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f /= f.sum()
                    best_M = best_val
                    best_f = f.copy()
                    pass_improved = True
                    break  # Move to next i

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

    return best_f, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result /= ss
    return result


def parametric_init(n, seed=42):
    """Generate diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    x = np.linspace(0, 1, n)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (1 - x)**p * x**p  # Bump shape
        a = np.maximum(a, 1e-10)
        a /= a.sum()
        inits.append(a.copy())

    # Sine bump
    for p in [1, 2, 3, 5]:
        a = np.sin(np.pi * x) ** p
        a = np.maximum(a, 1e-10)
        a /= a.sum()
        inits.append(a.copy())

    # Gaussian
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        a /= a.sum()
        inits.append(a.copy())

    # Random
    for _ in range(5):
        a = rng.random(n)
        a /= a.sum()
        inits.append(a.copy())

    return inits
