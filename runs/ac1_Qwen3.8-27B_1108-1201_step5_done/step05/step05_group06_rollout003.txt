import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_multi_peak_subgrad(a, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:

            sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def peak_targeted_update(a, conv, n_peaks=10, frac=0.1):
    """
    For each of the top peaks in the auto-convolution, identify the
    coordinates that contribute most and try to redistribute mass
    from them to less-contributing coordinates.
    """
    n = len(a)
    a_new = a.copy()
    conv_len = len(conv)
    
    # Get top peaks
    top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
    top_peaks = top_peaks[np.argsort(conv[top_peaks])[::-1]]
    
    for k in top_peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        
        # Contributions: a[i] * a[k-i] for i in [js, je]
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k - idx]
        
        # Find the coordinate with the largest contribution
        max_contrib_idx = np.argmax(contribs)
        i_star = idx[max_contrib_idx]
        j_star = k - i_star
        
        # Choose which of i_star or j_star to reduce (the larger one)
        if a[i_star] >= a[j_star]:
            donor = i_star
        else:
            donor = j_star
        
        # Find a good receiver: a coordinate that doesn't contribute to this peak
        # and has small value
        in_peak = set(idx.tolist())
        candidates = [x for x in range(n) if x not in in_peak and a[x] < np.median(a)]
        
        if not candidates or a[donor] < 1e-8:
            continue
        
        # Pick the receiver with minimum contribution to this peak
        # (ideally 0)
        best_receiver = min(candidates, key=lambda x: a[x] if (x < js or x > je) else a[x])
        
        # Transfer a small fraction of mass
        delta = frac * a[donor]
        a_new[donor] -= delta
        a_new[best_receiver] += delta
    
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

def coordinate_descent_bisection(a0, time_limit, verbose=False, max_passes=200):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-14:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-14:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = a0.copy()
                continue
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 31 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
                conv = fast_convolve(a)
            else:
                break
    
    return best_a, best_obj


def block_coordinate_opt(a0, time_limit, block_size=4, verbose=False):
    """Optimize blocks of coordinates using local search."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        # Pick a random block of coordinates
        start = rng.integers(0, max(1, n - block_size))
        block = np.arange(start, start + block_size)
        
        # Get current values of the block
        x_block = a[block].copy()
        
        # Try random perturbations of the block
        best_block_obj = best_obj
        best_block = x_block.copy()
        
        for trial in range(30):
            # Random perturbation
            perturbation = rng.standard_normal(block_size) * 0.01
            new_block = np.maximum(x_block + perturbation, 0.0)
            
            # Normalize the block to maintain sum constraint
            # (keep sum of block the same)
            block_sum = new_block.sum()
            if block_sum < 1e-12:
                continue
            new_block = new_block / block_sum * x_block.sum()
            
            a_trial = a.copy()
            a_trial[block] = new_block
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial /= st
            
            obj = fast_convolve(a_trial).max()
            if obj < best_block_obj:
                best_block_obj = obj
                best_block = new_block.copy()
        
        if best_block_obj < best_obj - 1e-14:
            a[block] = best_block
            a = np.maximum(a, 0.0)
            st = a.sum()
            if st > 1e-12:
                a /= st
            conv = fast_convolve(a)
            best_obj = conv.max()
            best_a = a.copy()
    
    return best_a, best_obj

def generate_gaussian(n, sigma_frac=0.3):

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()

    return f_interp


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def asymmetric_shift(a, amount=0.3):
    """Create an asymmetric version by shifting mass."""
    n = len(a)
    a_new = a.copy()
    # Shift some mass from left to right
    shift = int(n * amount)
    s_left = a[:shift].sum()
    s_right = a[n-shift:].sum()
    if s_left > s_right:
        a_new[:shift] *= 0.7
        a_new[n-shift:] *= 1.3
    else:
        a_new[n-shift:] *= 0.7
        a_new[:shift] *= 1.3
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_multi_peak_subgrad(a, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def peak_targeted_update(a, conv, n_peaks=10, frac=0.08):
    """Redistribute mass from high-contributing coordinates at peak lags."""
    n = len(a)
    a_new = a.copy()
    top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
    top_peaks = top_peaks[np.argsort(conv[top_peaks])[::-1]]

    for k in top_peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        contribs = a_new[idx] * a_new[k - idx]
        max_idx = np.argmax(contribs)
        i_star = idx[max_idx]
        j_star = k - i_star
        if a_new[i_star] >= a_new[j_star]:
            donor = i_star
        else:
            donor = j_star
        if a_new[donor] < 1e-10:
            continue
        in_peak = set(idx.tolist())
        candidates = [x for x in range(n) if x not in in_peak]
        if not candidates:
            continue
        med_val = np.median(a_new)
        low_cands = [x for x in candidates if a_new[x] < med_val]
        if not low_cands:
            low_cands = candidates
        best_receiver = min(low_cands, key=lambda x: a_new[x])
        delta = frac * a_new[donor]
        a_new[donor] -= delta
        a_new[best_receiver] += delta

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

def spectral_shaping_step(a, n_iter=200, verbose=False):
    """
    Flatten the power spectrum in frequency domain to reduce autocorrelation peaks.
    Works by iteratively adjusting the phase of Fourier coefficients.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    a_work = a.copy().astype(np.float64)
    work_len = len(a_work)
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work /= s

    best_a = a_work.copy()
    best_obj = fast_convolve(best_a).max()
    rng = np.random.default_rng(999)

    for it in range(n_iter):
        # Get FFT
        A = np.fft.rfft(a_work, fft_size)
        magnitudes = np.abs(A)
        phases = np.angle(A)

        # Target: flat magnitude spectrum (except DC)
        # Perturb phases slightly
        phase_pert = rng.standard_normal(len(phases)) * 0.1
        new_phases = phases + phase_pert

        # Reconstruct
        A_new = magnitudes * np.exp(1j * new_phases)
        a_trial = np.fft.irfft(A_new, fft_size)[:work_len]
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-12:
            continue
        a_trial /= st

        obj = fast_convolve(a_trial).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_trial.copy()
            a_work = a_trial.copy()

    return best_a


def generate_rudin_shapiro(n):
    """Generate a Rudin-Shapiro-like non-negative sequence."""
    m = 1
    rs = np.array([1.0, 1.0])
    while len(rs) < n + n:
        rs = np.concatenate([rs, rs, rs, -rs])
    rs = np.abs(rs[:2 * n])
    if len(rs) >= n:
        seq = rs[:n]
    else:
        seq = rs
    seq = np.maximum(seq, 0.1)
    seq /= seq.sum()
    return seq


def generate_gaussian(n, sigma_frac=0.3):

    g = np.exp(-x**2 / (2 * sigma**2))

    return g


def generate_asymmetric(n, skew=0.4):
    """Generate an asymmetric sequence to break symmetry of autocorrelation."""
    x = np.linspace(0, 1, n)
    a = np.exp(-skew * x) + 0.1
    a = a / a.sum()
    return a


def generate_random_smooth(n, smoothness=3):

def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def asymmetric_shift(a, amount=0.3):
    """Create an asymmetric version by shifting mass."""
    n = len(a)
    a_new = a.copy()
    shift = int(n * amount)
    if shift < 1:
        return a_new
    s_left = a_new[:shift].sum()
    s_right = a_new[n - shift:].sum()
    if s_left > s_right:
        a_new[:shift] *= 0.7
        a_new[n - shift:] *= 1.3
    else:
        a_new[n - shift:] *= 0.7
        a_new[:shift] *= 1.3
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def combined_optimize(a0, time_limit, verbose=False):
    """Alternate optimization phases with adaptive time allocation."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    phase = 0
    no_improve_streak = 0
    prev_best = best_obj

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        if phase % 5 == 0:
            t_alloc = min(30, remaining * 0.2)
            lr = 0.006 / (1.0 + 0.2 * (phase // 5))
            top_k = max(20, min(n // 6, 80))
            temp = 0.0002 + 0.0001 * (phase // 5)
            a_opt, obj = subgradient_step(
                best_a, time_limit=t_alloc, lr=lr, mom=0.93,
                top_k=top_k, temp=temp, decay=0.9997
            )
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                no_improve_streak = 0
                if verbose:
                    print(f"  Phase {phase} SG: obj={best_obj:.10f}")
            else:
                no_improve_streak += 1

        elif phase % 5 == 1:
            t_alloc = min(50, remaining * 0.25)
            a_opt, obj = coordinate_descent_bisection(best_a, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                no_improve_streak = 0
                if verbose:
                    print(f"  Phase {phase} CD: obj={best_obj:.10f}")
            else:
                no_improve_streak += 1

        elif phase % 5 == 2:
            t_alloc = min(20, remaining * 0.1)
            a_opt, obj = block_coordinate_opt(best_a, time_limit=t_alloc, block_size=4)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                no_improve_streak = 0
                if verbose:
                    print(f"  Phase {phase} BC: obj={best_obj:.10f}")
            else:
                no_improve_streak += 1

        elif phase % 5 == 3:
            # Peak-targeted + quick CD
            a_pt = peak_targeted_update(best_a, fast_convolve(best_a), n_peaks=10, frac=0.06)
            t_alloc = min(20, remaining * 0.1)
            a_opt, obj = coordinate_descent_bisection(a_pt, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                no_improve_streak = 0
                if verbose:
                    print(f"  Phase {phase} PT+CD: obj={best_obj:.10f}")
            else:
                no_improve_streak += 1

        else:
            # Noise + restart
            if no_improve_streak > 3:
                rng = np.random.default_rng(phase * 103 + 7)
                noise_scale = 0.004 * max(0.5, 1.0 - phase * 0.03)
                a_noisy = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            else:
                a_noisy = best_a.copy()
            t_alloc = min(25, remaining * 0.12)
            a_opt, obj = subgradient_step(
                a_noisy, time_limit=t_alloc, lr=0.005, mom=0.90,
                top_k=max(15, n // 10), temp=0.0004, decay=0.9995
            )
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                no_improve_streak = 0
                if verbose:
                    print(f"  Phase {phase} Noise: obj={best_obj:.10f}")
            else:
                no_improve_streak += 1

        phase += 1

    return best_a, best_obj
