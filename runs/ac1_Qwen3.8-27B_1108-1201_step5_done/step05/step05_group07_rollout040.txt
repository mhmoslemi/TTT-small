"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated smooth-max optimization
with L-BFGS-B, multiplicative weights, and coordinate descent polishing.
Key: O(n log n) gradient via FFT, softmax parameterization for L-BFGS-B,
multi-length and multi-init exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, weights):
    """
    Compute grad_a[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j]
    using FFT for O(n log n) complexity.
    """
    n = len(a)
    # conv(a, reverse(weights))[2n-2-i] gives sum_j a[j] * weights[i+j]
    c_rev = weights[::-1]
    size = n + len(c_rev) - 1  # = n + 2n-1 - 1 = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    CR = np.fft.rfft(c_rev, fft_size)
    conv = np.fft.irfft(A * CR, fft_size)[:size]
    # grad_a[i] = 2 * conv[2n-2-i], for i=0,...,n-1
    # conv[2n-2-0], conv[2n-2-1], ..., conv[2n-2-(n-1)] = conv[2n-2], conv[2n-3], ..., conv[n-1]
    grad_a = 2.0 * conv[n-1:2*n-1][::-1]
    return grad_a


def smooth_max_obj_grad(a, temperature):
    """
    Compute the smooth max of conv(a,a) = T * logsumexp(conv/T) and its gradient.
    a is assumed to sum to 1 and be non-negative.
    Returns (objective_value, gradient_wrt_a).
    """
    conv = fast_convolve(a)
    scaled = conv / temperature
    s_max = scaled.max()
    shifted = scaled - s_max
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    log_sum = s_max + np.log(sum_exp)
    obj = temperature * log_sum
    
    weights = exp_shifted / sum_exp
    grad_a = compute_grad_fft(a, weights)
    
    return obj, grad_a


def softmax(z):
    z = z - z.max()
    ez = np.exp(z)
    return ez / ez.sum()


def lbfgs_optimize(f0, time_limit, temperature=0.0005, max_iter=2000):
    """
    Optimize using L-BFGS-B with softmax parameterization.
    a = softmax(z), minimize 2n * smooth_max(conv(a,a)).
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 1e-30)
    f = f / f.sum()
    
    z0 = np.log(f)
    
    best_z = z0.copy()
    best_a = f.copy()
    best_true_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    
    def obj_grad(z):
        a = softmax(z)
        obj, grad_a = smooth_max_obj_grad(a, temperature)
        # Chain rule for softmax
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        return 2.0 * n * obj, 2.0 * n * grad_z
    
    # Use callback to track best
    state = {'it': 0}
    
    def callback(zk):
        state['it'] += 1
        if state['it'] % 50 == 0:
            a_k = softmax(zk)
            obj_k = fast_convolve(a_k).max()
            nonlocal_best = obj_k < best_true_obj
            # Can't modify best_true_obj from here due to scoping
        return False
    
    # Run with time limit via maxiter and maxfun
    result = minimize(obj_grad, z0, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'maxfun': 10000,

    z_opt = result.x
    a_opt = softmax(z_opt)
    true_obj = fast_convolve(a_opt).max()
    
    # Check elapsed time
    elapsed = time.monotonic() - t0

def mu_weights_optimize(f0, time_limit, top_k=50, temperature=0.00005, lr=0.003,

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]
        
        v = momentum * v + (1 - momentum) * sg
        
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)

        f = f / f.sum()

            f = f / f.sum()
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0002)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular
    x = np.arange(n) / n
    tri = 1 - np.abs(2 * x - 1)
    tri = np.maximum(tri, 1e-10)
    inits.append(tri / tri.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())
    
    # Gaussian
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 3.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Bimodal
    c1 = n * (0.2 + 0.1 * rng.random())
    c2 = n * (0.7 + 0.1 * rng.random())
    s1 = n / (5.0 + rng.random() * 3.0)
    s2 = n / (5.0 + rng.random() * 3.0)
    bimodal = np.exp(-0.5 * ((np.arange(n) - c1) / s1) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / s2) ** 2)
    bimodal = np.maximum(bimodal, 1e-10)
    inits.append(bimodal / bimodal.sum())
    
    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated smooth-max optimization
with L-BFGS-B (softmax parameterization), improved multiplicative weights, coordinate
descent, and multi-length exploration. Key innovations: O(n log n) gradient via FFT,
softmax reparameterization for constrained optimization, and extensive length search
(n up to 8192) to exploit the asymptotic advantage of larger sequences.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, weights):
    """
    Compute grad_a[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j]
    using FFT for O(n log n) complexity.
    """
    n = len(a)
    c_rev = weights[::-1]
    size = n + len(c_rev) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    CR = np.fft.rfft(c_rev, fft_size)
    conv = np.fft.irfft(A * CR, fft_size)[:size]
    grad_a = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad_a


def smooth_max_obj_grad(a, temperature):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    conv = fast_convolve(a)
    scaled = conv / temperature
    s_max = scaled.max()
    shifted = scaled - s_max
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    log_sum = s_max + np.log(sum_exp)
    obj = temperature * log_sum
    weights = exp_shifted / sum_exp
    grad_a = compute_grad_fft(a, weights)
    return obj, grad_a


def softmax_func(z):
    """Compute softmax of z."""
    z = z - z.max()
    ez = np.exp(z)
    s = ez.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return ez / s


def lbfgs_optimize(f0, time_limit, temperature=0.0005, max_iter=3000):
    """
    Optimize using L-BFGS-B with softmax parameterization.
    a = softmax(z), minimize 2n * smooth_max(conv(a,a)).
    Returns (a_opt, true_objective, elapsed_time).
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 1e-30)
    f = f / f.sum()

    z0 = np.log(f)
    t0 = time.monotonic()

    best_a = f.copy()
    best_true_obj = fast_convolve(f).max()

    def obj_grad(z):
        a = softmax_func(z)
        obj, grad_a = smooth_max_obj_grad(a, temperature)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        # Track best
        nonlocal best_a, best_true_obj
        true_obj = fast_convolve(a).max()
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a.copy()
        return 2.0 * n * obj, 2.0 * n * grad_z

    try:
        result = minimize(obj_grad, z0, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                                   'gtol': 1e-9, 'maxls': 30})
    except Exception:
        result = None

    elapsed = time.monotonic() - t0
    return best_a, best_true_obj, elapsed

            state['done'] = True
            a = softmax_func(z0)

        a = softmax_func(z)
        obj, grad_a = smooth_max_obj_grad(a, temperature)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        true_obj = fast_convolve(a).max()
        if true_obj < state['best_obj']:
            state['best_obj'] = true_obj
            state['best_a'] = a.copy()
        return 2.0 * n * obj, 2.0 * n * grad_z

    try:
        minimize(obj_grad, z0, method='L-BFGS-B',
                 options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                          'gtol': 1e-9, 'maxls': 30})
    except Exception:
        pass

    elapsed = time.monotonic() - t0
    return state['best_a'], state['best_obj'], elapsed


def mu_weights_optimize(f0, time_limit, top_k=50, temperature=0.00005, lr=0.003,
                        momentum=0.85):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest = f_nest / fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f = f / fs
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.008
        if it % 500 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.95))

        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.002 + 0.001 * min(no_improve // 500, 5)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 0)
            f = f / f.sum()
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

    return best_f, best_obj


def mu_weights_full_softmax(f0, time_limit, temperature=0.001, lr=0.002, momentum=0.85):
    """
    Multiplicative weights using FULL softmax over all convolution peaks
    (not just top-k). More expensive per step but smoother gradient.
    Uses FFT for gradient computation.
    """
    f = np.maximum(f0.astype(np.float64), 1e-30)
    f = f / f.sum()
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature

    while time.monotonic() - t0 < time_limit:
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest = f_nest / fn

        conv_nest = fast_convolve(f_nest)
        scaled = conv_nest / cur_temp
        s_max = scaled.max()
        shifted = scaled - s_max
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        sg = compute_grad_fft(f_nest, w)

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f = f / fs
        f = np.maximum(f, 1e-30)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        if cur_lr < 0.0001:
            cur_lr = 0.0001
        cur_temp *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 7) % 1000000)
            noise_scale = 0.003 * min(1 + no_improve // 400, 4.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-10)
            f = f / f.sum()
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0002)
            no_improve = 0

    return best_f, best_obj


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
