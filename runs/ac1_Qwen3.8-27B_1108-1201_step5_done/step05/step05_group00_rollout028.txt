def smoothed_grad_descent(a0, time_limit, T_init=0.0003, lr_init=0.03, top_k=None):
    """Smoothed gradient descent with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    if top_k is None:
        top_k = max(10, n // 10)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)  # momentum
    mom = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Compute weighted subgradient using top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        # Softmax weights within the peaks
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w = w / w.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]

        # Momentum update
        v = mom * v - lr * grad
        candidate = project_simplex(a + v)

        # Backtracking
        best_candidate = a
        best_val = conv.max()
        for trial_frac in [1.0, 0.5, 0.25, 0.1]:
            c = project_simplex(a + trial_frac * v)
            val = fast_convolve(c).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = c

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if no_improve > 30:
            T *= 1.08
            lr *= 0.96
        if no_improve > 80:
            T *= 0.5
            lr *= 1.5
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.005 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def peak_target_descent(a0, time_limit):
    """
    Directly target the top peaks for reduction.
    For each peak, identify the contributing pairs and adjust them.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:

            break

        conv = fast_convolve(a)
        # Find top 5 peaks
        top_k = min(5, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        improved = False
        for k in peaks:
            if time.monotonic() - t0 > time_limit - 0.1:
                break

            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            pairs = [(j, k - j) for j in range(js, je + 1)]

            if not pairs:
                continue

            # Sort by contribution
            contributions = [(a[j] * a[k2], j, k2) for j, k2 in pairs]
            contributions.sort(reverse=True)

            # Try to reduce the top contributing pair
            for _ in range(3):
                if not contributions:
                    break
                _, j1, j2 = contributions[0]

                # Try shifting mass from j1 to a random other position
                # or from j2 to a random other position
                rng = np.random.default_rng((it * 1000 + int(k) + j1) % (2**31))
                
                # Option 1: reduce j1, increase a random other
                for _attempt in range(5):
                    j_target = rng.integers(0, n)
                    if j_target == j1 or j_target == j2:
                        continue
                    delta = a[j1] * 0.01  # move 1% of j1's mass
                    candidate = a.copy()
                    candidate[j1] -= delta
                    candidate[j_target] += delta
                    if candidate[j1] < 0:
                        continue
                    candidate = candidate / candidate.sum()
                    val = fast_convolve(candidate).max()
                    if val < best_obj - 1e-15:
                        a = candidate

def spectral_init(n, seed=42):
    """
    Generate an initial sequence using spectral methods.
    We want a probability vector with flat auto-convolution.
    In the frequency domain, flat auto-convolution means flat |A(w)|^2,
    which means |A(w)| is constant. For non-negative a, this is the
    all-pass filter problem.
    
    Approximation: start with a "tapered" window that has properties
    favorable for flat autocorrelation.
    """
    rng = np.random.default_rng(seed)
    
    # Various initializations
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine taper (Hann window)
    x = np.arange(n)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(hann / hann.sum())
    
    # Raised cosine with different beta
    for beta in [0.3, 0.5, 0.7]:
        t = x / (n - 1) * 2 - 1
        cos_t = np.cos(np.pi * t * beta)
        win = 0.5 * (1 + cos_t)
        inits.append(win / win.sum())
    
    # Gaussian taper
    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - (n-1)/2) / (sigma * n)) ** 2)
        inits.append(g / g.sum())
    
    # Power law taper
    for p in [0.5, 1.0, 1.5, 2.0]:
        t = np.abs(x - (n-1)/2) / ((n-1)/2)
        w = (1 - t) ** p
        w = np.maximum(w, 0.01)
        inits.append(w / w.sum())
    
    # Two-peak structure
    for gap in [0.1, 0.2, 0.3]:
        center1 = (1 - gap) * n / 2
        center2 = (1 + gap) * n / 2
        s1 = 0.1 * n
        g1 = np.exp(-0.5 * ((x - center1) / s1) ** 2)
        g2 = np.exp(-0.5 * ((x - center2) / s1) ** 2)
        inits.append((g1 + g2) / (g1 + g2).sum())
    
    # Random
    for _ in range(3):
        r = rng.dirichlet(np.ones(n) * 0.5)
        inits.append(r)
    
    return inits


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

**Idea: Use the fact that the objective is the max of quadratic functions.**

The objective (after normalizing sum=1) is `2n * max_k a^T H_k a` where `H_k` are Hankel matrices.

This is a D.C. (difference of convex) optimization problem in disguise, since `max` of convex functions is convex.

We can use the **proximal point method** or **ADMM** to solve this.

Actually, since it's convex (max of convex functions), we can use standard convex optimization! The problem is:

minimize t
subject to a^T H_k a <= t for all k
sum(a) = 1
a >= 0

This is a quadratically constrained program (QCP). The constraints are SOC-representable if H_k is PSD, but in general H_k is not PSD (it's a Hankel matrix which can have negative eigenvalues).

However, we can still solve this iteratively using:
- **Bisection on t**: For a given t, check if there exists a with a^T H_k a <= t for all k. This is still a QCP.
- **Successive convex approximation (SCA)**: Linearize the quadratic constraints at the current iterate.

**SCA approach:**
At iterate a^(m), the constraint a^T H_k a <= t can be approximated by:
2 * a^(m)^T H_k a - (a^(m))^T H_k a^(m) <= t (first-order approximation)

This is a linear constraint in a! So SCA reduces the problem to a sequence of LPs.

This is actually very similar to the Frank-Wolfe LP oracle! In fact, the LP oracle in the FW method is exactly the SCA step.

So the FW method is essentially SCA + line search. The key question is whether we can make it converge faster.

**Accelerated FW (AFW)**: There are accelerated versions of the FW method that converge at O(1/k²) instead of O(1/k). These use a "lookahead" point and a more sophisticated step size.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.02, top_k_frac=0.05):

    a = np.maximum(a0.astype(np.float64), 0.0)

        pv_max = pv.max()

        w_sum = w.sum()
        if w_sum < 1e-30:

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:

            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]

        # Momentum + backtracking

            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

            lr = lr_init

            stagnant = 0

    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

            t_alloc = min(remaining, 400)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.001, 0.003, 0.008, 0.015]:
            remaining = deadline - time.monotonic() - 350
            if remaining < 25:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 60)
            print(f"  FW perturb scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Subgradient refinement (250s) ===
    print("\n=== Phase 3: Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {'T_init': 0.0003, 'lr_init': 0.02, 'top_k_frac': 0.03},
            {'T_init': 0.001, 'lr_init': 0.015, 'top_k_frac': 0.05},
            {'T_init': 0.003, 'lr_init': 0.01, 'top_k_frac': 0.1},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 250
            if remaining < 20:
                break
            t_alloc = min(remaining, 80)
            print(f"  Subgrad config {ci} (T={cfg['T_init']}), t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak-targeted coordinate descent (200s) ===
    print("\n=== Phase 4: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Peak coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_coord")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final micro-polish (remaining) ===
    print("\n=== Phase 5: Final Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate between FW and subgrad with tiny perturbations
        polish_rounds = 5
        for ri in range(polish_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if ri % 2 == 0:
                # Small perturb + short FW
                noise = rng.standard_normal(n_best) * 0.0008
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 25)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{ri}")
            else:
                # Small perturb + subgrad
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 20)
                a_opt, _ = subgradient_optimize(start, time_limit=t_alloc,
                                                T_init=0.0002, lr_init=0.005,
                                                top_k_frac=0.02)
                try_seq(a_opt, f"polish_sg_{ri}")

        # One final peak coord descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_peak")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
