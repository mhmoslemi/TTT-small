"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B optimization with
softmax parameterization and adaptive log-sum-exp smoothing, combined with
coordinate descent polishing and multi-length exploration.
"""

import numpy as np

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective_and_grad_x(x, tau):
    """
    Compute the smooth objective and gradient with respect to x (softmax parameter).
    a = softmax(x), objective = (2*n/tau) * LSE(tau * conv(a,a))
    Returns (objective_value, grad_x).
    """
    n = len(x)
    # Compute a = softmax(x)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute convolution
    conv = fast_convolve(a)
    
    # Compute log-sum-exp of tau * conv
    tc = tau * conv
    tc_max = tc.max()
    lse = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    
    objective = (2.0 * n / tau) * lse
    
    # Compute gradient
    # softmax weights
    w = np.exp(tc - tc_max)
    w /= w.sum()
    
    # grad_a[j] = 4*n * sum_k w[k] * a[k-j] (for valid k-j)
    # = 4*n * np.correlate(w, a, mode='valid')[j]
    grad_a = 4.0 * n * np.correlate(w, a, mode='valid')
    
    # grad_x = a * (grad_a - dot(grad_a, a))
    c = np.dot(grad_a, a)
    grad_x = a * (grad_a - c)
    
    return objective, grad_x

def coordinate_descent_optimize(a_init, time_limit, verbose=False):

                continue

                a[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss < 1e-15:
                    a[j] = old_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] /= scale
                    break
                a /= ss
                
                val = fast_convolve(a).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_mult = mult
                else:
                    # Revert
                    a[j] = old_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] /= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    break

                    best_a = a.copy()
                    improved = True

    score = float(evaluate_sequence(best_a.tolist()))
    return best_a, score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-15)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def _objective(x, tau):
