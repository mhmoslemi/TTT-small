def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

from scipy.signal import correlate

def solve_inner_fw(f, temp, n_iters=50, verbose=False):
    """Solve: min_g sum_k exp(conv(f,g)[k]/temp) s.t. g in simplex."""
    n = len(f)
    fft_size = 1 << (2*n-1).bit_length()
    F = np.fft.rfft(f, fft_size)
    
    g = np.ones(n) / n
    
    # Estimate step size using Lipschitz constant
    conv_g = np.fft.irfft(F * np.fft.rfft(g, fft_size), fft_size)[:2*n-1]
    max_exp = np.exp(conv_g.max() / temp)
    # The Hessian of the objective is (1/temp^2) * C^T diag(exp(conv/temp)) C
    # ||Hessian||_2 <= (1/temp^2) * max(exp) * ||C||_2^2
    # ||C||_2^2 <= ||C||_F^2 = sum_{k,j} C[k,j]^2 = sum_j f[j]^2 * (2n-1) roughly
    # Actually ||C||_F^2 = sum_k sum_j C[k,j]^2 = sum_k sum_j f[k-j]^2 (valid) 
    # = sum_m f[m]^2 * (number of k such that k-m is valid)
    # For each m, the number of valid k is min(m+1, n, 2n-1-m) ... approximately n
    # So ||C||_F^2 ≈ n * ||f||_2^2
    # ||C||_2^2 <= ||C||_F^2 ≈ n * ||f||_2^2 <= n (since ||f||_1 = 1 implies ||f||_2 <= 1)
    L = max_exp * n / (temp * temp)
    step = 1.0 / L
    
    for t in range(n_iters):
        conv_g = np.fft.irfft(F * np.fft.rfft(g, fft_size), fft_size)[:2*n-1]
        exp_vals = np.exp(conv_g / temp)
        
        # Gradient
        grad = correlate(exp_vals, f, mode='valid') / temp
        
        # Projected gradient step
        g = project_simplex(g - step * grad)
        
        if verbose and t % 10 == 0:
            obj = np.sum(exp_vals)
            print(f"      Inner iter {t}: obj={obj:.6f}")
    
    return g

        it += 1
        improved = False
        
        # Find the current peak
        peak_pos = int(np.argmax(conv))
        
        # Find the dominant contributing pair
        # conv[peak_pos] = sum_j f[j] * f[peak_pos - j]
        js = range(max(0, peak_pos - n + 1), min(n, peak_pos + 1))
        contribs = {}
        for j in js:
            k = peak_pos - j
            if 0 <= k < n:
                contribs[j] = f[j] * f[k]
        
        if not contribs:
            break
        
        # Sort by contribution (descending)
        sorted_pairs = sorted(contribs.items(), key=lambda x: x[1], reverse=True)
        
        # Try to reduce the top contributing pairs
        for j_top, _ in sorted_pairs[:5]:
            k_top = peak_pos - j_top
            if k_top < 0 or k_top >= n:
                continue
            
            # Try reducing f[j_top] and redistributing
            for frac in [0.5, 0.7, 0.8, 0.9]:
                delta = f[j_top] * (1 - frac)
                if delta < 1e-12:
                    continue
                
                # Redistribute delta to coordinates that don't contribute to the peak
                # Choose the coordinate with minimum f that doesn't contribute to top peaks
                f_trial = f.copy()
                f_trial[j_top] -= delta
                
                # Find a good redistribution target

                rng = np.random.default_rng(it * 1000 + j_top * 10 + int(frac * 100))

                f_trial = f_trial / st

                break

                # Try a different approach: reduce both members of the top pair
                j1, j2 = sorted_pairs[0][0], sorted_pairs[0][1] if len(sorted_pairs) > 1 else None
                # Actually, let me try a different strategy

                # Subgradient step

                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 50:
                        break
        else:
            no_improve = 0

```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Softmax Frank-Wolfe with
accelerated inner solver, LP-FW diversity, multi-length exploration,
and subgradient/CD polishing. Key innovation: uses a smooth (softmax)
surrogate for the max in the FW direction-finding step, giving smoother
directions than the LP vertex solution.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.signal import correlate
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_fg(f, g, fft_size=None, F=None):
    """Compute conv(f, g) using precomputed FFT of f if available."""
    n = len(f)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1 << (size - 1).bit_length()
    if F is None:
        F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_inner_fw(f, temp, n_iters=80, fft_size=None, F=None):
    """
    Solve: min_g sum_k exp(conv(f,g)[k]/temp) s.t. g in simplex.
    Uses projected gradient with adaptive step size.
    """
    n = len(f)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1 << (size - 1).bit_length()
    if F is None:
        F = np.fft.rfft(f, fft_size)

    g = np.ones(n) / n

    # Estimate Lipschitz constant and step size
    conv_g = convolve_fg(f, g, fft_size, F)
    max_exp = float(np.exp(conv_g.max() / temp))
    # Hessian bound: (1/temp^2) * max_exp * ||C||^2 <= (1/temp^2) * max_exp * n * ||f||_1^2
    L = max_exp * (n + 1) / (temp * temp)
    step = 1.0 / L

    for t in range(n_iters):
        conv_g = convolve_fg(f, g, fft_size, F)
        exp_vals = np.exp(conv_g / temp)

        # Gradient: (1/temp) * correlate(exp_vals, f, 'valid')
        grad = correlate(exp_vals, f, mode='valid')
        grad /= temp

        # Adaptive step size (decay slightly)
        step_t = step / (1.0 + 0.005 * t)

        g = project_simplex(g - step_t * grad)

    return g


def soft_fw_optimize(f0, time_limit, temp_init=None, verbose=False):
    """
    Softmax Frank-Wolfe: outer FW loop with softmax surrogate inner solver.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    fft_size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, fft_size)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    if temp_init is None:
        temp = best_obj / (2.0 * np.log(max(2 * n, 4)))
    else:
        temp = temp_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Recompute F if f changed significantly
        F = np.fft.rfft(f, fft_size)

        # Solve inner problem
        n_inner = 100 if rem > 60 else 50
        g = solve_inner_fw(f, temp, n_iters=n_inner, fft_size=fft_size, F=F)
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        # Check convergence
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search with golden section
        best_alpha = 0.0
        best_val = float('inf')

        # Coarse
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine (golden section)
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            gr = (np.sqrt(5) - 1) / 2
            for _ in range(15):
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                f1 = np.maximum(f + x1 * (g - f), 0.0)
                s1 = f1.sum()
                if s1 > 1e-15:
                    f1 = f1 / s1
                    v1 = float(fast_convolve(f1).max())
                else:
                    v1 = float('inf')
                f2 = np.maximum(f + x2 * (g - f), 0.0)
                s2 = f2.sum()
                if s2 > 1e-15:
                    f2 = f2 / s2
                    v2 = float(fast_convolve(f2).max())
                else:
                    v2 = float('inf')
                if v1 < v2:
                    hi = x2
                else:
                    lo = x1
            mid = (lo + hi) / 2.0
            fm = np.maximum(f + mid * (g - f), 0.0)
            sm = fm.sum()
            if sm > 1e-15:
                fm = fm / sm
                vm = float(fast_convolve(fm).max())
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid

        if best_alpha > 1e-12:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            conv = fast_convolve(f)
            M = float(conv.max())

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    SFW iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adapt temperature: decrease to sharpen
        temp = max(0.00005, temp * 0.98)

        if no_improve > 30:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.002 * min(no_improve // 30, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            best_obj = float(conv.max())
            no_improve = 0
            temp = max(0.0001, best_obj / (2.0 * np.log(max(2 * n, 4))))

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe using min-peak LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Build and solve LP: min t s.t. conv(f,g)[k] <= t, g>=0, sum(g)=1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        for j in range(n):
            A_ub[j:j + n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_constraints)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            g = np.ones(n) / n
        else:
            g = res.x[:n]
            g = np.maximum(g, 0.0)
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = float('inf')
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            sn = f_new.sum()
            f = f_new / sn if sn > 1e-15 else np.ones(n) / n
            M = float(fast_convolve(f).max())
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng(it * 31337 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            best_obj = float(fast_convolve(f).max())
            no_improve = 0

    return best_f, best_obj


def sg_optimize(f0, time_limit, lr_init=0.003, momentum=0.94, top_k_frac=0.1,
                temp_init=0.00006, verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0
    top_k = max(10, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = float(conv.max())

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def peak_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        passes += 1
        improved = False

        peak_pos = int(np.argmax(conv))
        js = range(max(0, peak_pos - n + 1), min(n, peak_pos + 1))

        # Find top contributing coordinates
        contribs = []
        for j in js:
            k = peak_pos - j
            if 0 <= k < n:
                contribs.append((f[j] * f[k], j, k))
        contribs.sort(reverse=True)

        for _, j_top, k_top in contribs[:8]:
            if f[j_top] < 1e-12:
                continue
            for mult in [0.4, 0.6, 0.75, 0.9]:
                new_val = f[j_top] * mult
                delta = f[j_top] - new_val
                if delta < 1e-13:
                    continue
                f_trial = f.copy()
                f_trial[j_top] = new_val
                # Redistribute to k_top (which also contributes to peak) - reduce it
                f_trial[k_top] = max(0, f_trial[k_top] - delta * 0.5)
                # Add the remainder to a random non-peak coordinate
                rng = np.random.default_rng(passes * 1000 + j_top)
                for _ in range(5):
                    target = int(rng.integers(0, n))
                    if abs(target - j_top) > n // 4 and abs(target - k_top) > n // 4:
                        break
                f_trial[target] += delta
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial = f_trial / st
                val = float(fast_convolve(f_trial).max())
                if val < best_obj - 1e-13:
                    f = f_trial
                    conv = fast_convolve(f)
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 8:
                # Try subgradient step
                peaks = np.where(conv >= best_obj * 0.98)[0][:10]
                sg = np.zeros(n)
                for pk in peaks:
                    for j in range(max(0, int(pk) - n + 1), min(n, int(pk) + 1)):
                        k = int(pk) - j
                        if 0 <= k < n:
                            sg[j] += 2.0 * f[k]
                lr = 0.005
                f_new = project_simplex(f - lr * sg / len(peaks))
                val = float(fast_convolve(f_new).max())
                if val < best_obj - 1e-13:
                    f = f_new
                    conv = fast_convolve(f)
                    best_obj = val
                    best_f = f.copy()
                else:
                    no_improve += 1
                    if no_improve > 30:
                        break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    """

        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

        parent = np.ones(512) / 512
