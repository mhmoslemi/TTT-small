import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, conv, top_k, temperature):
    """Compute weighted subgradient of max(conv) using top-k peaks with softmax weighting."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    
    # Find top-k peaks
    if k_actual == conv_len:
        peaks = np.arange(conv_len)
    else:
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Softmax weighting
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Compute gradient directly (fast for small top_k)
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        wi = 2.0 * w[i]
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            grad[m_start:m_end + 1] += wi * a[k - m_end:k - m_start + 1][::-1]
    
    return grad

def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization."""
    j = np.arange(n)

    s = a.sum()
    if s < 1e-15:

    return a / s


def chirp_init(n, chirp_rate=1.0, seed=0):

def multi_band_init(n, num_bands=4, seed=0):
    """Generate a multi-band amplitude modulation."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    a = np.ones(n)
    band_width = n // num_bands
    for b in range(num_bands):
        amp = 0.5 + rng.random() * 1.0
        start = b * band_width
        end = min((b + 1) * band_width, n)
        a[start:end] *= amp
    a = np.maximum(a, 0.01)
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Interpolate a sequence to a new length and renormalize."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lp_direction(f, M, n):
    """Try to solve the LP direction problem. Returns None if too slow or infeasible."""
    from scipy.optimize import linprog
    # Only attempt for n <= 500 to keep it fast
    if n > 500:
        return None
    
    c = -np.ones(n)
    # Build constraint matrix A_ub where row k represents conv constraint at lag k
    # conv[k] = sum_j f[j] * g[k-j] <= M
    # For each k (0 to 2n-2), the constraint on g is:
    # sum_j f[j] * g[k-j] <= M (where 0 <= k-j < n)
    # = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} f[k-j] * g[j] <= M
    
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in conv[k] for k = j to k = j+n-1
        # In conv[k], the coefficient of g[j] is f[k-j]
        for k in range(j, min(j + n, 2 * n - 1)):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

                    a_opt_scaled = a_opt * best_seq.sum()
                    try_seq(a_opt_scaled, f"opt_{label}")
    
    # === Phase 6: Multi-start with perturbations ===
    print(f"\n=== Phase 6: Multi-start (best={best_score:.8f}) ===")
    if best_seq is not None:
        best_norm = best_seq / best_seq.sum()
        n_best = len(best_norm)
        
        for p in range(6):
            rem = deadline - time.monotonic() - 80
            if rem < 25:
                break
            
            t_alloc = min(rem * 0.3, 80)
            
            if p == 0:
                f_init = best_norm + rng.standard_normal(n_best) * 0.005
            elif p == 1:
                f_init = best_norm + rng.standard_normal(n_best) * 0.01
            elif p == 2:
                f_init = 0.8 * best_norm + 0.2 * cosine_modulated_init(n_best, 3, 0.3)
            elif p == 3:
                f_init = 0.9 * best_norm + 0.1 * (np.ones(n_best) / n_best)
            elif p == 4:
                f_init = 0.85 * best_norm + 0.15 * chirp_init(n_best, 1.5)
            else:
                # Reverse symmetry perturbation
                f_init = 0.5 * best_norm + 0.5 * best_norm[::-1]
            
            f_init = project_simplex(f_init)
            a_opt, _ = nesterov_optimize(f_init, time_limit=t_alloc,
                                          lr0=0.015, mom0=0.7, top_k=30,
                                          temp0=0.008, seed=100 + p)
            a_opt_scaled = a_opt * best_seq.sum()
            try_seq(a_opt_scaled, f"multistart_{p}")
    
    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        best_norm = best_seq / best_seq.sum()
        n_best = len(best_norm)
        
        # Very fine Nesterov
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem, 60)
            a_opt, _ = nesterov_optimize(best_norm, time_limit=t_alloc,
                                          lr0=0.003, mom0=0.9, top_k=100,
                                          temp0=0.001, seed=999)
            a_opt_scaled = a_opt * best_seq.sum()
            try_seq(a_opt_scaled, "final_nesterov")
        
        # Quick coordinate descent on all coords
        rem = deadline - time.monotonic() - 20
        if rem > 15 and n_best <= 1024:
            t_alloc = min(rem, 30)
            a_opt, _ = quick_coord_descent(best_norm, time_limit=t_alloc,
                                            n_coords=list(range(n_best)), seed=77)
            a_opt_scaled = a_opt * best_seq.sum()
            try_seq(a_opt_scaled, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
