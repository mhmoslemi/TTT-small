"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase strategy:
1. Diverse structured initializations (raised cosine, B-spline, binary smoothed)
2. LP-based Frank-Wolfe for quick convergence
3. Targeted peak-reduction local search with pair moves
4. Smoothing-based moves (averaging with neighbors)
5. Multi-length exploration with interpolation
6. Subgradient polish with adaptive softmax weighting
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0])
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def fw_optimize(f0, time_limit, verbose=False, max_iters=500):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            g = np.maximum(g, 0)
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            f = f_new / sn if sn > 1e-15 else np.ones(n) / n

            if best_val < best_obj - 1e-15:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 6271 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def peak_reduction_search(f0, time_limit, verbose=False):
    """Targeted local search: move mass from peak-contributing coords to safe coords."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f0)) / len(f0)
    n = len(f)
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(f)
        peak_val = conv.max()

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        # Identify "hot" coordinates (contributing to multiple peaks)
        hot_count = np.zeros(n)
        for pk in peak_positions:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for j in range(js, je + 1):
                hot_count[j] += f[j] * f[pk - j]

        # Sort coordinates by "hottness"
        sorted_idx = np.argsort(hot_count)[::-1]
        hot_coords = sorted_idx[:max(5, n // 10)]
        safe_coords = sorted_idx[-max(5, n // 10):]

        improved = False
        rng = np.random.default_rng(iteration * 3571 + 13)

        # Try moving mass from hot to safe
        for _ in range(min(50, n // 4)):
            if time.monotonic() - t0 > time_limit:
                break
            if len(hot_coords) < 2 or len(safe_coords) < 2:
                break

            i1 = int(hot_coords[rng.integers(0, len(hot_coords))])
            i2 = int(safe_coords[rng.integers(0, len(safe_coords))])
            if f[i1] < 1e-8:
                continue

            # Try transferring a small amount
            for frac in [0.05, 0.1, 0.15, 0.2]:
                transfer = f[i1] * frac
                if transfer < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[i1] -= transfer
                f_trial[i2] += transfer
                if f_trial[i1] < 0:
                    continue
                st = f_trial.sum()
                f_trial = f_trial / st
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break

        # Try pair swap: reduce two hot, increase two safe
        if len(hot_coords) >= 2 and len(safe_coords) >= 2:
            for _ in range(min(20, n // 8)):
                if time.monotonic() - t0 > time_limit:
                    break
                i1 = int(hot_coords[rng.integers(0, len(hot_coords))])
                i2 = int(hot_coords[rng.integers(0, len(hot_coords))])
                j1 = int(safe_coords[rng.integers(0, len(safe_coords))])
                j2 = int(safe_coords[rng.integers(0, len(safe_coords))])
                if i1 == i2 or j1 == j2:
                    continue
                transfer = min(f[i1], f[i2]) * 0.1
                if transfer < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[i1] -= transfer / 2
                f_trial[i2] -= transfer / 2
                f_trial[j1] += transfer / 2
                f_trial[j2] += transfer / 2
                if (f_trial[i1] < 0 or f_trial[i2] < 0):
                    continue
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True

        # Smoothing move: average with neighbor
        if not improved and iteration % 5 == 0:
            for _ in range(min(10, n // 10)):
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(rng.integers(1, n - 1))
                for avg_frac in [0.1, 0.2, 0.3]:

                    f_trial[j] = (1 - avg_frac) * f[j] + avg_frac * 0.5 * (f[j-1] + f[j+1])
                    f_trial = f_trial / f_trial.sum()

                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                # Larger perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * (0.002 + 0.002 * min(no_improve // 100, 5)))
                no_improve = 0
        else:
            no_improve = 0

    return best_f, best_obj


def smoothing_iteration(f0, time_limit, verbose=False):
    """Apply iterative smoothing that reduces peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f0)) / len(f0)
    n = len(f)
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    # Kernel sizes to try
    kernel_sizes = [2, 3, 5, 7, 11]
    fracs = [0.05, 0.1, 0.15, 0.2, 0.3]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for ks in kernel_sizes:
            if ks >= n:
                continue
            for frac in fracs:
                kernel = np.ones(ks) / ks
                pad = ks // 2
                f_padded = np.pad(f, pad, mode='edge')
                f_smooth = np.convolve(f_padded, kernel, mode='valid')[:n]
                f_smooth = np.maximum(f_smooth, 0)
                ss = f_smooth.sum()
                if ss < 1e-15:
                    continue
                f_smooth = f_smooth / ss

                # Blend
                f_trial = (1 - frac) * f + frac * f_smooth
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=40, temp=0.00005,
                verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f0)) / len(f0)
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        f = f_new
        cur_lr *= 0.9999
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def generate_diverse_starts(n, seed=42):
    """Generate diverse structured starting points of length n."""
    rng = np.random.default_rng(seed)
    starts = []

    # Uniform
    starts.append(np.ones(n) / n)

    # Raised cosine
    t = np.linspace(0, 1, n)
    rc = (1 + np.cos(2 * np.pi * t)) / 2
    rc[0] = 0.01
    rc[-1] = 0.01
    starts.append(rc / rc.sum())

    # Sine window
    sw = np.sin(np.pi * t) ** 2
    sw[0] = 0.01
    sw[-1] = 0.01
    starts.append(sw / sw.sum())

    # Triangular
    tri = 1 - 2 * np.abs(t - 0.5)
    tri[0] = 0.01
    tri[-1] = 0.01
    starts.append(tri / tri.sum())

    # B-spline-like (Gaussian)
    center = n // 2
    sigma = n / 4
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    starts.append(gauss / gauss.sum())

    # Smoothed random binary
    for _ in range(3):
        binary = rng.integers(0, 2, n).astype(float)
        # Smooth with a wide kernel
        ksize = max(3, n // 20)
        kernel = np.ones(ksize) / ksize
        pad = ksize // 2
        padded = np.pad(binary, pad, mode='wrap')
        smoothed = np.convolve(padded, kernel, mode='valid')[:n]
        smoothed = np.maximum(smoothed, 0.01)
        starts.append(smoothed / smoothed.sum())

    # Welch-like: equalize magnitudes in Fourier domain
    for _ in range(2):
        phases = rng.uniform(0, 2 * np.pi, n)
        real = np.cos(phases)
        real = np.maximum(real, 0)
        ss = real.sum()
        if ss > 1e-10:
            starts.append(real / ss)

    # Low-pass filtered random
    for _ in range(2):
        rand = rng.standard_normal(n)
        freq = np.fft.rfftfreq(n)
        cutoff = 0.3
        filt = np.exp(-freq * n * 0.5)
        spec = np.fft.rfft(rand) * filt
        lp = np.fft.irfft(spec, n)
        lp = np.maximum(lp, 0)
        ss = lp.sum()
        if ss > 1e-10:
            starts.append(lp / ss)

    return starts

    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20

        # Quick SG
        if rem > 20:
            a_opt, _ = sg_optimize(best_seq, time_limit=min(rem * 0.3, 60),
                                   lr=0.001, momentum=0.95,
                                   top_k=max(10, n_best // 12),
                                   temp=0.00002)
            try_seq(a_opt, "sg_polish")

        # Quick peak search
        rem = deadline - time.monotonic() - 15
        if rem > 15 and best_seq is not None:
            a_opt, _ = peak_reduction_search(best_seq, time_limit=min(rem * 0.3, 50))
            try_seq(a_opt, "peak_polish")

        # Quick FW
        rem = deadline - time.monotonic() - 15
        if rem > 10 and best_seq is not None:
            a_opt, _ = fw_optimize(best_seq, time_limit=min(rem * 0.4, 40))
            try_seq(a_opt, "fw_polish")

        # Perturbations
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise_scale = 0.001 + 0.001 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = sg_optimize(f_pert, time_limit=min(rem, 20),
                                   lr=0.0008, momentum=0.95,
                                   top_k=max(10, n_best // 15),
                                   temp=0.00002)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
