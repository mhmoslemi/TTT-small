4. Inverse FFT to get new `a`.
5. Project to `a >= 0` (clamp negatives to 0) and renormalize.
6. Repeat.

- Smoothed gradient descent (log-sum-exp) for local refinement

import numpy as np

# --- Core utilities ---

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_objective(a):
    """Compute the objective for a normalized (sum=1) non-negative sequence."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    conv = fast_convolve(a)
    return float(2 * len(a) * conv.max())

def spectral_shaping(a0, time_limit, target_flat=True, blend=0.5):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively: FFT -> flatten magnitude -> IFFT -> project to simplex.
    """
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # FFT size
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # FFT of padded a
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude
        if target_flat:
            # Flat target with fixed DC
            c = np.sqrt(max((N * np.sum(a**2) - 1.0) / (N - 1), 1e-10))
            target_mag = np.full(len(mag), c)
            target_mag[0] = 1.0  # Fix DC
        else:
            # Blend current with flat
            c = np.sqrt(max((N * np.sum(a**2) - 1.0) / (N - 1), 1e-10))
            target_mag = mag.copy()
            target_mag[0] = 1.0
            target_mag[1:] = blend * mag[1:] + (1 - blend) * c
        
        # Update phase spectrum
        A_new = target_mag * np.exp(1j * phase)
        
        # IFFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # No improvement: try blending
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
    
    return best_a, best_obj

# --- FW-LP ---

def solve_lp_oracle(f, M):
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # For each k, the constraint is: sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j] <= M
    # The coefficient of g[j] in constraint k is f[k-j] if 0<=k-j<n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
