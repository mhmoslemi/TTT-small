   h. Enforce a_new >= 0, sum(a_new) = 1.

4. Return a.

```python
def fourier_flat_autocorr(a0, time_limit, target_ratio=1.0, clip_strength=0.1):
    """
    Iteratively flatten the autocorrelation using Fourier domain methods.
    
    Parameters:
    - a0: initial sequence (will be normalized to sum=1)
    - time_limit: seconds to run
    - target_ratio: target max/average ratio (start above, decrease over time)
    - clip_strength: how aggressively to clip peaks
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    N = 1 << (2*n - 1).bit_length()  # next power of 2 >= 2n-1
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    best_a = a.copy()
    best_obj = 2*n*fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Adaptive target: start at current max, slowly decrease
    conv = fast_convolve(a)
    target = conv.max()
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # FFT
        A = np.fft.fft(a_padded)
        P = np.abs(A)**2
        
        # IDFT to get autocorrelation
        c = np.fft.ifft(P).real  # should be real
        c = c[:2*n-1]  # linear convolution part
        
        # Modify: reduce peaks
        # Soft clip: c_new = c - alpha * max(0, c - target)^2
        excess = np.maximum(0, c - target)
        c_new = c - clip_strength * excess**2
        c_new = np.maximum(c_new, 0)  # autocorrelation must be non-negative
        
        # FFT back to get new power spectrum
        c_padded = np.zeros(N)
        c_padded[:len(c_new)] = c_new
        P_new = np.fft.fft(c_padded)
        P_new = np.maximum(P_new.real, 0)  # must be non-negative real
        
        # Reconstruct: keep phase of A, use new magnitude
        phase = np.angle(A)
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)
        
        # Inverse FFT to get new time-domain signal
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Enforce constraints
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Evaluate
        obj = 2*n*fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            # Decrease target
            target = target * 0.999
        else:
            no_improve += 1
        
        # Update current (with some mixing to avoid oscillation)
        a = 0.5 * a + 0.5 * a_new
        a = np.maximum(a, 0)
        a /= a.sum()
        a_padded[:n] = a
        a_padded[n:] = 0
        
        # Adapt parameters
        if no_improve > 50:
            target = fast_convolve(a).max() * 0.99
            clip_strength = min(clip_strength * 1.1, 1.0)
            no_improve = 0
        
        # Gradually decrease target
        target = min(target, fast_convolve(a).max() * 0.995)
    
    return best_a, best_obj
```

```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)
```

Wait, there's a bug: when I update conv, I use `a[k_range - m]` but `a[m]` has already been changed to `best_x`. I need to use the OLD value of a[m] for the other entries, but k_range - m gives the indices of the OTHER entries (not m itself, unless m is in k_range-m... which it is when k=2m).

Actually, `a[k_range - m]` gives the values a[k-m] for k in k_range. Since k ranges from max(0,m) to min(2n-2, m+n-1), k-m ranges from max(0,m)-m to min(2n-2,m+n-1)-m = 0 to n-1. So k-m is always in [0, n-1], which are the valid indices.

The issue is: I'm updating a[m] BEFORE updating conv. So when I compute `a[k_range - m]`, the entry at index m (which corresponds to k=2m) will have the new value `best_x` instead of `old`. But that's actually fine because:
- For k ≠ 2m: the change is 2*(x-old)*a[k-m], and a[k-m] doesn't involve a[m] (since k-m ≠ m).
- For k = 2m: the change should be (x-old)*(x+old) = (x^2 - old^2). My formula gives 2*(x-old)*a[2m-m] + (x-old)^2 = 2*(x-old)*best_x + (x-old)^2. But a[m] is now best_x, so a[2m-m] = a[m] = best_x. So I get 2*(x-old)*best_x + (x-old)^2 = (x-old)(2*best_x + x - old) = (x-old)(x + best_x) = (x-old)(x+x) = 2*(x-old)*x. But the correct value is x^2 - old^2 = (x-old)(x+old).

So there's a discrepancy: I get 2*(x-old)*x but should get (x-old)(x+old). The difference is (x-old)(x+old) - 2*(x-old)*x = (x-old)(old - x) = -(x-old)^2.

So I need to subtract (x-old)^2 from my update at k=2m. Or better, I should save the old value before updating.
