3. Exact coordinate descent with bisection for local polish

import numpy as np
import time

def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    # Clip tiny negatives from FFT roundoff
    result = np.maximum(result, 0.0)
    return result

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_matvec(w, a):
    """
    Compute M @ a where M[i,j] = w[i+j] (w has length 2n-1, a has length n).
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    u = a[::-1].copy()  # reverse of a
    # Full convolution of w (len 2n-1) and u (len n)
    conv_len = (2*n - 1) + n - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    U = np.fft.rfft(u, fft_size)
    conv = np.fft.irfft(W * U, fft_size)
    # Result: (w*u)[i+n-1] for i=0,...,n-1
    return conv[n-1 : 2*n-1]

def compute_smoothed_gradient(a, c, tau):
    """
    Compute gradient of S(a) = (1/tau) * log(sum_k exp(tau * c[k])).
    Returns (gradient, objective_value, weights).
    """
    tc = tau * c
    tc_shifted = tc - tc.max()
    exp_vals = np.exp(tc_shifted)
    w = exp_vals / exp_vals.sum()
    
    # Objective
    log_sum_exp = tc.max() + np.log(exp_vals.sum())
    obj = log_sum_exp / tau
    
    # Gradient: g[i] = 2 * sum_j w[i+j] * a[j] = 2 * hankel(w) @ a
    g = 2.0 * hankel_matvec(w, a)
    
    return g, obj, w

def smoothed_gradient_descent(a0, tau_schedule, lr_schedule=None, time_limit=100, 
                                verbose=False, accel=True):
    """
    Nesterov-accelerated projected gradient descent on the smoothed objective.
    Continues through a schedule of temperatures.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    
    t0 = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autoconv(a)
        # Initialize Nesterov momentum
        m = np.zeros(n)
        t_counter = 1.0
        
        for it in range(2000):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov lookahead
            if accel:
                t_new = (1 + np.sqrt(1 + 4 * t_counter * t_counter)) / 2
                beta = (t_new - 1) / t_counter
                y = a + beta * m
                y = project_simplex(y)
            else:
                y = a
                t_new = t_counter
                beta = 0
                m = np.zeros(n)
            
            c_y = fast_autoconv(y)
            g, obj_s, w = compute_smoothed_gradient(y, c_y, tau)
            
            # Adaptive step size
            if lr_schedule is not None:
                lr = lr_schedule[it] if it < len(lr_schedule) else lr_schedule[-1]
            else:
                # Default: lr ~ 1/sqrt(iteration)
                lr = 1.0 / (np.sqrt(it + 1) * np.sqrt(n))
            
            # Gradient step
            a_new = project_simplex(y - lr * g)
            
            # Update momentum
            m = a_new - a
            a = a_new
            t_counter = t_new
            
            # Track best
            obj = c_y.max()  # Use the actual max (not smoothed) for tracking
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
        
        if verbose:
            print(f"    tau={tau:.4f}: obj={best_obj:.10f}")
    
    return best_a, best_obj

            hi = 1.0

                b_kj = b_valid[k]  # b[k-j] if valid, else 0

                    x_new = mid

                    x_new = mid
                else:
                    x_new = mid

            mask = np.ones(n, dtype=bool)

            a[mask] *= (1.0 - x_new) / s_others

                best_a = a.copy()

                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                conv = fast_autoconv(a)
            else:
                break
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with Nesterov acceleration and continuation
2. FFT-based Hankel matrix-vector products for O(n log n) gradient computation

7. Symmetric constraint exploitation

import numpy as np
import time


def fast_autoconv(a):
    """Fast linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    result = np.maximum(result, 0.0)
    return result


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(w, a):
    """
    Compute M @ a where M[i,j] = w[i+j] (w length 2n-1, a length n).
    Uses FFT for O(n log n) complexity.
    Result[i] = sum_j w[i+j] * a[j]
    """
    n = len(a)
    u = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    U = np.fft.rfft(u, fft_size)
    conv = np.fft.irfft(W * U, fft_size)
    return conv[n - 1: 2 * n - 1]


def compute_smoothed_gradient(a, c, tau):
    """
    Compute gradient and objective of S(a) = (1/tau)*log(sum_k exp(tau*c[k])).
    Returns (gradient, smoothed_obj, softmax_weights).
    """
    tc = tau * c
    tc_shifted = tc - tc.max()
    exp_vals = np.exp(tc_shifted)
    w = exp_vals / exp_vals.sum()
    log_sum_exp = tc.max() + np.log(exp_vals.sum())
    obj = log_sum_exp / tau
    g = 2.0 * hankel_matvec(w, a)
    return g, obj, w


def nesterov_smoothed_descent(a0, tau_schedule, time_limit=100.0, verbose=False,
                               base_lr=1.0, decay=0.9995):

    Nesterov-accelerated projected gradient descent on log-sum-exp smoothed objective.

        a = a / s
    n = len(a)

    best_obj = fast_autoconv(a).max()
    t0 = time.monotonic()
    total_iters = 0

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        m = np.zeros(n)
        t_counter = 1.0
        cur_lr = base_lr / np.sqrt(n)

        max_inner = 3000
        for it in range(max_inner):
            if time.monotonic() - t0 > time_limit:
                break
            total_iters += 1

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_counter * t_counter)) / 2.0
            beta = (t_new - 1.0) / t_counter
            y = a + beta * m
            y = project_simplex(y)

            c_y = fast_autoconv(y)
            g, obj_s, w = compute_smoothed_gradient(y, c_y, tau)

            a_new = project_simplex(y - cur_lr * g)

            a = a_new

                best_a = a.copy()

    """
    Multiplicative (natural gradient) update on the smoothed objective.
    a_new[i] proportional to a[i] * exp(-lr * g[i])
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    t0 = time.monotonic()

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        for it in range(5000):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autoconv(a)
            g, obj_s, w = compute_smoothed_gradient(a, c, tau)

            g_max = g.max()
            if g_max < 1e-15:
                break
            lr = 1.5 / (tau * g_max)

            a_new = a * np.exp(-lr * g)
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                break
            a_new /= sn

            obj_new = fast_autoconv(a_new).max()
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new

        if verbose:
            print(f"      tau={tau:.4f} (mult): obj={best_obj:.10f}")

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit=100.0, verbose=False):
    """
    Exact coordinate descent: for each coordinate, bisection-search the optimal
    value holding others fixed (with proportional rescaling).
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_autoconv(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    max_passes = 500

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        any_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_autoconv(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b_valid[k]
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = -2.0 * om * B_k + 2.0 * (1.0 - 2.0 * mid) * b_kj + 2.0 * mid * ind

                if deriv > 1e-14:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-14:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_autoconv(a)
            obj = conv.max()

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                any_improved = True

        if not any_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 31 + 7)
                noise = 0.001 + 0.0005 * min(passes - 3, 10)
                a = project_simplex(best_a + rng.standard_normal(n) * noise)
                conv = fast_autoconv(a)
            else:
                break

        if verbose and passes % 10 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.10f}")

    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i]) / 2, renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization centered in the sequence."""
    x = np.linspace(0, 1, n)
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_periodic_init(n, freq=3.0):
    """Periodic (positive) initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_tapered_init(n, power=2.0):
    """Tapered (raised cosine to the power) initialization."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** power
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def combined_optimize(a0, time_limit, verbose=False):
    """
    Multi-phase optimization: smoothed gradient (Nesterov + multiplicative)
    alternating with coordinate descent polish.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    t0 = time.monotonic()
    phase = 0
    max_phases = 50

    while time.monotonic() - t0 < time_limit and phase < max_phases:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        if phase % 3 == 0:
            # Nesterov smoothed descent
            t_alloc = min(30.0, remaining * 0.3)
            tau_sched = [0.01, 0.1, 1.0, 5.0, 20.0, 100.0, 500.0]
            if phase > 6:
                tau_sched = [10.0, 50.0, 200.0, 1000.0]
            t_new = 0.5 / (1.0 + 0.1 * (phase // 3))
            a_opt, obj = nesterov_smoothed_descent(
                best_a, tau_sched, time_limit=t_alloc, verbose=False,
                base_lr=t_new, decay=0.9997
            )
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Phase {phase} Nesterov: {best_obj:.10f}")

        elif phase % 3 == 1:
            # Multiplicative descent
            t_alloc = min(20.0, remaining * 0.2)
            tau_sched = [1.0, 10.0, 50.0, 200.0]
            if phase > 6:
                tau_sched = [50.0, 200.0, 1000.0]
            a_opt, obj = multiplicative_descent(
                best_a, tau_sched, time_limit=t_alloc, verbose=False
            )
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Phase {phase} Multi: {best_obj:.10f}")

        else:
            # Coordinate descent polish
            t_alloc = min(40.0, remaining * 0.4)
            a_opt, obj = coordinate_descent_polish(
                best_a, time_limit=t_alloc, verbose=False
            )
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Phase {phase} CD: {best_obj:.10f}")

        phase += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Combines smoothed gradient descent (continuation),
    coordinate descent polish, multi-start exploration, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 40.0

    best_a = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent available, using uniform n=768")

    n_parent = len(parent)

    def try_update(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        a_norm = a_try / s_try
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")

    # === Phase 1: Optimize parent with combined method ===
    print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 900
    if rem > 60:
        t_alloc = min(rem, 400)
        print(f"  Combined optimize parent, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(parent, time_limit=t_alloc, verbose=True)
        try_update(a_opt, "parent_SGD")

    # === Phase 2: Nesterov with aggressive continuation ===
    print(f"\n=== Phase 2: Aggressive Nesterov (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 700
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  Nesterov aggressive, t={t_alloc:.0f}s")
            tau_sched = [0.001, 0.01, 0.1, 1.0, 10.0, 100.0, 1000.0, 10000.0]
            a_opt, _ = nesterov_smoothed_descent(
                best_a, tau_sched, time_limit=t_alloc,
                base_lr=2.0, decay=0.9999
            )
            try_update(a_opt, "nesterov_agg")

    # === Phase 3: Symmetric variant ===
    print(f"\n=== Phase 3: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 200)
            print(f"  Symmetric, t={t_alloc:.0f}s")
            a_opt, _ = combined_optimize(a_sym, time_limit=t_alloc, verbose=False)
            try_update(a_opt, "symmetric")

    # === Phase 4: Multi-start with diverse initializations ===
    print(f"\n=== Phase 4: Multi-start (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        inits = [
            ("noise1", project_simplex(best_a + rng.standard_normal(n_best) * 0.01)),
            ("noise3", project_simplex(best_a + rng.standard_normal(n_best) * 0.03)),
            ("gauss", make_gaussian_init(n_best)),
            ("tapered", make_tapered_init(n_best, power=2.0)),
            ("tapered4", make_tapered_init(n_best, power=4.0)),
            ("periodic", make_periodic_init(n_best, freq=3.0)),
            ("rev_mix", project_simplex(0.7 * best_a + 0.3 * best_a[::-1])),
            ("flat_mix", project_simplex(0.85 * best_a + 0.15 * np.ones(n_best) / n_best)),
        ]

        for idx, (label, init) in enumerate(inits):
            rem = deadline - time.monotonic() - 400 + idx * 30
            if rem < 30:
                break
            t_alloc = min(rem, 120)
            print(f"  Start '{label}', t={t_alloc:.0f}s")
            a_opt, _ = combined_optimize(init, time_limit=t_alloc, verbose=False)
            try_update(a_opt, f"start_{label}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent

    for target_n in [512, 640, 896, 1024, 1280]:
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.4, 150)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(f_interp, time_limit=t_alloc, verbose=False)
        try_update(a_opt, f"n{target_n}")

    # === Phase 6: Intensive final polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        # Long coordinate descent
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 250)
            print(f"  CD intensive, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_a, time_limit=t_alloc, verbose=True)
            try_update(a_opt, "final_cd")

        # Nesterov fine-tune
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem, 80)
            tau_sched = [100.0, 500.0, 2000.0, 10000.0]
            a_opt, _ = nesterov_smoothed_descent(
                best_a, tau_sched, time_limit=t_alloc, base_lr=0.3, decay=0.99999
            )
            try_update(a_opt, "final_nest")

        # Tiny perturbations + CD
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            n_best = len(best_a)
            for p in range(4):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                rng_p = np.random.default_rng(seed + 5000 + p * 13)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.0005)
                t_alloc = min(rem, 12)
                a_opt, _ = coordinate_descent_polish(f_pert, time_limit=t_alloc)
                try_update(a_opt, f"perturb_{p}")

        # Multiplicative fine-tune
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 30)
            a_opt, _ = multiplicative_descent(
                best_a, [500.0, 2000.0, 10000.0], time_limit=t_alloc
            )
            try_update(a_opt, "final_mult")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
