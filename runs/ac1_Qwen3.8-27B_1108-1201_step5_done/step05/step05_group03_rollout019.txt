def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

3. Palindromic constraint exploration (halves effective search space)
4. Multi-length systematic exploration
5. Adaptive restarts with geometric noise schedule

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def exact_subgrad_at_peaks(f, conv, num_peaks=50):
    """Compute exact subgradient averaged over the top num_peaks peaks.
    Uses softmax weighting focused on the highest peaks."""
    n = len(f)
    size = 2 * n - 1
    
    if num_peaks >= size:
        peaks = np.arange(size)
    else:
        peaks = np.argpartition(conv, -num_peaks)[-num_peaks:]
    
    pv = conv[peaks]
    # Temperature: very low to focus on the top
    T = max(1e-15, (pv[0] - pv[-1]) * 0.05 + 1e-15) if len(pv) > 1 else 1e-15
    w = np.exp(-(pv - pv.max()) / T)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[0] = 1.0
    else:
        w = w / ws
    
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        f_idx = k - np.arange(lo, hi + 1)
        grad[lo:hi+1] += 2.0 * wi * f[f_idx]
    
    return grad

def nesterov_subgrad(f0, time_limit, num_peaks=50, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    y = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    step_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Nesterov momentum
        t_k = (np.sqrt(1 + 4.0 * it * it) - 1.0) / 2.0
        if it > 1:
            y = f + (t_k / (t_k + 1.0)) * (f - y)
        
        # Ensure y is on simplex
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / ys
        
        # Compute subgradient
        conv_y = fast_convolve(y)
        grad = exact_subgrad_at_peaks(y, conv_y, num_peaks)
        
        # Adaptive step size
        # Lipschitz constant estimate: ||grad||_2 * diameter
        g_norm = np.sqrt(np.dot(grad, grad))
        if g_norm < 1e-15:
            step = 0.0
        else:
            step = step_scale / (it + 1.0)
        
        f_new = f - step * grad
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    NSG iter {it}: obj={best_obj:.10f}, step={step:.6f}")
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                step_scale *= 0.8
        
        f = f_new
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            scale = 0.003 * min(1.0, no_improve / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            no_improve = 0
            step_scale = 1.0
    
    return best_f, best_obj

def pairwise_cd(f0, time_limit, num_pairs_per_pass=200, verbose=False):
    """Coordinate descent using exact 1D optimization on random pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Select random pairs
        pairs = rng.integers(0, n, size=(num_pairs_per_pass, 2))
        
        for idx in range(num_pairs_per_pass):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            i, j = int(pairs[idx, 0]), int(pairs[idx, 1])
            if i == j:
                continue
            
            c = f[i] + f[j]
            if c < 1e-12:
                continue
            
            conv = fast_convolve(f)
            # Slopes: s_k = 2*(f[k-j] - f[k-i])
            # (with f[m] = 0 for m < 0 or m >= n)
            kj = np.arange(2*n-1) - j  # k - j for k = 0..2n-2
            ki = np.arange(2*n-1) - i  # k - i for k = 0..2n-2
            
            fj_kj = np.zeros(2*n-1)
            valid_j = (kj >= 0) & (kj < n)
            fj_kj[valid_j] = f[kj[valid_j]]
            
            fi_ki = np.zeros(2*n-1)
            valid_i = (ki >= 0) & (ki < n)
            fi_ki[valid_i] = f[ki[valid_i]]
            
            slopes = 2.0 * (fj_kj - fi_ki)
            
            # Quadratic corrections
            quad = np.zeros(2*n-1)
            quad[2*i] += 1.0  # at k=2i: +t²
            quad[2*j] += 1.0  # at k=2j: +t²
            if i != j:
                quad[i+j] -= 2.0  # at k=i+j: -2t²
            
            # We want to find t in [0, f[i]] that minimizes max_k(conv[k] + slopes[k]*t + quad[k]*t²)
            # Since quad is mostly 0, this is approximately: minimize max(conv[k] + slopes[k]*t)
            # which is a convex piecewise-linear function of t.
            
            # Use ternary search on [0, f[i]]
            lo = 0.0
            hi = f[i]
            
            if hi < 1e-14:
                continue
            
            # Quick evaluation
            def eval_at(t_val):
                # Only need to check a subset of k values
                # The maximum will be near the current peak
                # Use top 200 + the peaks shifted by slopes
                vals = conv + slopes * t_val + quad * t_val * t_val
                return vals.max()
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_at(c1)
            f2 = eval_at(c2)
            
            for _ in range(20):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_at(c2)
            
            t_opt = (lo + hi) / 2.0
            best_t_val = eval_at(t_opt)
            
            if best_t_val < best_obj - 1e-12:
                f[i] -= t_opt
                f[j] += t_opt
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-15:
                    f = best_f.copy()
                    continue
                f = f / fs
                conv = fast_convolve(f)
                obj = conv.max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                break
    
    return best_f, best_obj
