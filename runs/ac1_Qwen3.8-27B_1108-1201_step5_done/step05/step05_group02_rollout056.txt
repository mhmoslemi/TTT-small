        v = beta2 * v + (1 - beta2) * grad ** 2

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based Frank-Wolfe for coarse optimization
2. Adam optimization with smooth max (LogSumExp) for fine optimization  
3. Targeted coordinate descent for peak reduction
4. Multi-length exploration
5. Final polish with multiple restart strategies

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n
        g = np.maximum(g, 0)

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        f_new = f_new / sn if sn > 1e-15 else np.ones(n) / n
        f = f_new
        M = best_val

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def compute_smooth_grad(a, T, top_k=400):
    """Compute gradient of smooth max objective: 2*n * (1/T)*logsumexp(T*conv)."""
    n = len(a)
    conv = fast_convolve(a)
    k = min(top_k, len(conv))
    idx = np.argpartition(conv, -k)[-k:]
    top = conv[idx] * T
    m = top.max()
    exp_vals = np.exp(top - m)
    p_top = exp_vals / exp_vals.sum()

    grad = np.zeros(n)
    for ii in range(k):
        pk = p_top[ii]
        if pk < 1e-14:
            continue
        kk = int(idx[ii])
        i_start = max(0, kk - n + 1)
        i_end = min(n - 1, kk)
        if i_start > i_end:
            continue
        seg = a[kk - i_end:kk - i_start + 1][::-1]
        grad[i_start:i_end + 1] += pk * seg

    return (4.0 * n) * grad, conv


def adam_smooth_optimize(a0, time_limit, T_init=2000.0, T_max=50000.0,
                         top_k=400, lr=0.0008, beta1=0.9, beta2=0.999,
                         eps=1e-8, verbose=True, seed=42):
    """Adam optimization with smooth max and simplex projection."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    T_cur = T_init
    lr_cur = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_smooth_grad(a, T_cur, top_k=top_k)

        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * grad * grad
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)

        a_new = project_simplex(a - lr_cur * m_hat / (np.sqrt(v_hat) + eps))

        conv_new = fast_convolve(a_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and t % 2000 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.10f}, T={T_cur:.0f}, lr={lr_cur:.6f}")
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        # Anneal temperature and adjust learning rate
        if no_improve > 300:
            T_cur = min(T_cur * 1.4, T_max)
            lr_cur *= 0.85
            no_improve = 0

        if no_improve > 5000:
            # Restart with perturbation
            noise_scale = 0.001 + 0.002 * min(no_improve // 5000, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m_vec = np.zeros(n)
            v_vec = np.zeros(n)
            t = 0
            T_cur = T_init
            lr_cur = lr
            no_improve = 0

        if no_improve > 20000:
            break

    return best_a, best_obj


def coord_desc_peaks(a0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False

        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        peak_positions = peak_positions[np.argsort(conv[peak_positions])[::-1]]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            i_start = max(0, peak_k - n + 1)
            i_end = min(n - 1, peak_k)

            # Find the most contributing coordinate
            contributions = np.abs(f[i_start:i_end + 1] * f[peak_k - i_end:peak_k - i_start + 1][::-1])
            if len(contributions) == 0:
                continue
            top_contrib = np.argsort(contributions)[::-1][:10]

            for tc in top_contrib:
                if time.monotonic() - t0 > time_limit:
                    break
                j = i_start + int(tc)
                old_val = f[j]
                if old_val < 1e-12:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj

                for mult in [0.3, 0.5, 0.7, 0.85, 1.2, 1.5, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-13:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            # Random exploration
            rng = np.random.default_rng(iteration * 104729)
            for _ in range(min(30, n // 20)):
                j = int(rng.integers(0, n))
                old_val = f[j]
                if old_val < 1e-11:
                    continue
                for mult in [0.4, 0.6, 1.4, 1.8]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-13:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break
                if improved:
                    break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=50,
                         temp=0.0001, verbose=False, seed=42):
    """Subgradient method with softmax-weighted peaks."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            sg[idx] += 2.0 * w[i] * f[kg]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        co = conv_new.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            conv = conv_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
            conv = conv_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-FW, Adam smooth optimization,
    subgradient, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Quick LP-FW for initial polish ===
    print(f"\n=== Phase 1: LP-FW (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(150, rem * 0.15)
    if t1 > 20:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_init")

    # === Phase 2: Adam smooth optimization (main phase) ===
    print(f"\n=== Phase 2: Adam Smooth (n={len(best_seq) if best_seq is not None else base_n}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_adam = min(rem * 0.35, 600)
        if t_adam > 30:
            print(f"  n={n_best}, time={t_adam:.0f}s")
            a_opt, _ = adam_smooth_optimize(
                best_seq, time_limit=t_adam,
                T_init=2000.0, T_max=40000.0,
                top_k=400, lr=0.001,
                verbose=True, seed=seed)
            try_seq(a_opt, "adam_main")

            # Second Adam run with different params
            rem = deadline - time.monotonic()
            t_adam2 = min(rem * 0.15, 250)
            if t_adam2 > 20:
                print(f"  n={len(best_seq)}, time={t_adam2:.0f}s (params v2)")
                a_opt, _ = adam_smooth_optimize(
                    best_seq, time_limit=t_adam2,
                    T_init=5000.0, T_max=80000.0,
                    top_k=600, lr=0.0004,
                    verbose=True, seed=seed + 1)
                try_seq(a_opt, "adam_v2")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.2, 300)
        if t_sg > 25:
            print(f"  n={n_best}, time={t_sg:.0f}s")
            a_opt, _ = subgradient_optimize(
                best_seq, time_limit=t_sg,
                lr=0.002, momentum=0.94,
                top_k=max(40, n_best // 8),
                temp=0.00005, verbose=True, seed=seed+2)
            try_seq(a_opt, "sg_main")

            rem = deadline - time.monotonic()
            t_sg2 = min(rem * 0.1, 150)
            if t_sg2 > 15:
                a_opt, _ = subgradient_optimize(
                    best_seq, time_limit=t_sg2,
                    lr=0.001, momentum=0.95,
                    top_k=max(20, n_best // 15),
                    temp=0.00002, verbose=False, seed=seed+3)
                try_seq(a_opt, "sg_fine")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.2, 250)
        if t_cd > 20:
            print(f"  n={len(best_seq)}, time={t_cd:.0f}s")
            a_opt, _ = coord_desc_peaks(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "coord_desc")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = []
        for delta in [-64, -32, 32, 64, 128, 192]:
            ln = len(base) + delta
            if 256 <= ln <= 2048:
                candidate_lengths.append(ln)
        # Also try some round numbers
        for ln in [768, 800, 832, 864, 896, 928, 960, 1024]:
            if ln not in candidate_lengths and 256 <= ln <= 2048:
                candidate_lengths.append(ln)
        candidate_lengths = sorted(set(candidate_lengths))

        for target_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.25, 180)
            if t_alloc < 15:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  Adam n={target_n}, time={t_alloc:.0f}s")
            a_opt, _ = adam_smooth_optimize(
                f_interp, time_limit=t_alloc,
                T_init=2000.0, T_max=40000.0,
                top_k=400, lr=0.0008,
                verbose=False, seed=seed + target_n % 100)
            try_seq(a_opt, f"multi_n{target_n}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick Adam with very fine params
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t = min(rem, 80)
            a_opt, _ = adam_smooth_optimize(
                best_seq, time_limit=t,
                T_init=10000.0, T_max=100000.0,
                top_k=800, lr=0.0002,
                verbose=False, seed=seed + 100)
            try_seq(a_opt, "adam_polish")

        # Quick coord descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = coord_desc_peaks(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "cd_polish")

        # Perturbed restarts
        for p in range(8):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            noise_scale = 0.0005 + 0.0003 * p
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t = min(rem, 30)
            a_opt, _ = adam_smooth_optimize(
                f_pert, time_limit=t,
                T_init=5000.0, T_max=50000.0,
                top_k=500, lr=0.0003,
                verbose=False, seed=seed + 200 + p)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
