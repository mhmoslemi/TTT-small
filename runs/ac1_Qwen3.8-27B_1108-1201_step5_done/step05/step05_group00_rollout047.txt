"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Primal-dual QP iteration with Hankel matrices (softmax & top-k weighting)
- Nesterov-accelerated subgradient for final polish

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_qp_hankel(w, a_init, n, max_iters=200, time_limit=5.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set method.
    W is Hankel: W_{i,j} = w[i+j] where w has length 2n-1.
    Warm-started from a_init.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    active = set(np.where(a > 1e-12)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)
        k = len(act)

        # Build submatrix W_A
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])

        # Regularize for numerical stability
        W_AA_reg = W_AA + 1e-12 * np.eye(k)

        # Solve W_AA x = 1
        try:
            x = np.linalg.solve(W_AA_reg, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(W_AA_reg, np.ones(k), rcond=None)[0]
            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Remove negative components
        removed = False
        min_val = float('inf')
        min_idx = -1
        for i, idx in enumerate(act):
            if a_act[i] < 0:
                if a_act[i] < min_val:
                    min_val = a_act[i]
                    min_idx = idx
        if min_idx >= 0:
            active.discard(min_idx)
            removed = True
            # Update a for next iteration
            a_new = np.zeros(n)
            for i, idx in enumerate(act):
                if idx != min_idx:
                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form full solution
        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        # Compute gradient 2*W*a
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa

        # Lagrange multiplier mu (average over active set)
        mu = np.mean([grad[idx] for idx in act])

        # Check KKT for inactive: grad[j] - mu >= 0
        inactive = set(range(n)) - active
        if inactive:
            min_viol = 0.0
            min_j = -1
            for j in inactive:
                viol = grad[j] - mu
                if viol < min_viol:
                    min_viol = viol
                    min_j = j
            if min_viol < -1e-8:
                active.add(min_j)
                continue

        # Converged
        return a

    return a


def qp_primal_dual(a0, time_limit, T_init=0.01, use_topk=False, topk=20,
                   blend=0.5, verbose=False):
    """
    Primal-dual iteration:
    - Dual: compute weights w from conv(a,a) (softmax or top-k)
    - Primal: solve QP min a^T W a s.t. sum(a)=1, a>=0
    - Blend: a = (1-blend)*a + blend*a_qp
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Compute weights
        if use_topk:
            k_use = min(topk, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        # Solve QP
        qp_time = min(3.0, remaining * 0.3)
        a_qp = solve_qp_hankel(w, a, n, max_iters=100, time_limit=qp_time)

        # Blend
        alpha = blend
        a_new = (1.0 - alpha) * a + alpha * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature
        if no_improve > 10:
            T *= 1.15
        if no_improve > 30:
            T *= 0.75  # Re-focus
        if no_improve > 80:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    qp_pd it={it}, obj={best_obj:.10f}, T={T:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum_init=0.9,
                     verbose=False):
    """
    Nesterov-accelerated subgradient method for min max_k conv(a,a)[k]
    subject to a in simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()  # Nesterov lookahead

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov lookahead
        y = a + mom * v

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        # Gradient of c[k_peak] w.r.t. a: (grad)[i] = 2*a[k_peak-i] (for valid i)
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        # Update
        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new

        # Project
        a = project_simplex(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        lr *= 0.9998
        mom = min(0.99, mom + 0.00005)

        if no_improve > 100:
            lr *= 0.8
            mom *= 0.95
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    nesterov it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj


def peak_push(a0, time_limit, verbose=False):
    """
    For the dominant peak, try to reduce it by shifting mass from
    contributing positions to non-contributing ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Variables that contribute to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)

        if len(contrib) == 0 or len(non_contrib) == 0:
            break

        # Find the largest contributing variable
        idx_max = contrib[np.argmax(a[contrib])]
        val = a[idx_max]
        if val < 1e-10:
            break

        # Try shifting a fraction to non-contributing positions
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
            shift = val * frac
            if shift < 1e-12:
                continue

            # Distribute shift to non-contrib positions proportional to their current values
            nc_vals = a[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                # Distribute uniformly
                delta = np.zeros(n)
                delta[idx_max] -= shift
                for j in non_contrib:
                    delta[j] += shift / len(non_contrib)
            else:
                delta = np.zeros(n)
                delta[idx_max] -= shift
                for j in non_contrib:
                    delta[j] += shift * nc_vals[j] / nc_sum

            candidate = project_simplex(a + delta)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-14:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved and it % 100 == 0:
            # Try random perturbations focused on contributing positions
            rng = np.random.default_rng((it * 31 + k_peak) % (2**31))
            for _ in range(3):
                delta = np.zeros(n)
                delta[contrib] = rng.standard_normal(len(contrib)) * 0.002
                candidate = project_simplex(a + delta)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-14:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 500:
            break

    return best_a, best_obj


def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.03, verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of log-sum-exp surrogate: sum_k p[k] * grad(c[k])
        # grad(c[k])[i] = a[k-i] + a[i+k-2(n-1)] ... simplified: 2*correlate(p,a)
        grad = 2.0 * hankel_mv(p, a)

        # Line search
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.1
            lr *= 0.95
        if no_improve > 50:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for exploration."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())
    # Cosine window
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):

    Main entry point. Multi-phase hybrid search optimized for the
    auto-correlation minimization problem.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Quick n-exploration (~200s) ===
    print("\n=== Phase 1: Quick n-exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 64),
        n_parent + 64,
        max(128, n_parent // 2),
        n_parent * 2 if n_parent * 2 <= 2048 else n_parent,
        512, 1024,
    ]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 30:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 40)
        print(f"  QP-PD n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = qp_primal_dual(start, time_limit=t_alloc, T_init=0.01, use_topk=False, topk=20)
        try_seq(a_opt, f"qp_n{n_try}")

        # Also try top-k variant
        remaining = deadline - time.monotonic() - 950
        if remaining > 15:
            a_opt2, _ = qp_primal_dual(a_opt, time_limit=min(remaining, 20),
                                       T_init=0.01, use_topk=True, topk=10)
            try_seq(a_opt2, f"qp_topk_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep QP primal-dual (~800s) ===
    print("\n=== Phase 2: Deep QP Primal-Dual ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple temperature settings
        configs = [
            {'T_init': 0.005, 'use_topk': False, 'topk': 20, 'blend': 0.5, 't_frac': 0.3},
            {'T_init': 0.01, 'use_topk': False, 'topk': 30, 'blend': 0.4, 't_frac': 0.25},
            {'T_init': 0.02, 'use_topk': True, 'topk': 10, 'blend': 0.6, 't_frac': 0.2},
            {'T_init': 0.003, 'use_topk': True, 'topk': 5, 'blend': 0.3, 't_frac': 0.15},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, int(budget_s * cfg['t_frac']))
            print(f"  QP-PD config {ci} (T={cfg['T_init']}, topk={cfg.get('topk')}, "
                  f"blend={cfg['blend']}), t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, **{k: v for k, v in cfg.items() if k != 't_frac'})
            try_seq(a_opt, f"qp_deep_{ci}")
            if best_seq is not None and len(best_seq) != n_best:
                n_best = len(best_seq)

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Nesterov subgradient + smoothed grad (~400s) ===
    print("\n=== Phase 3: Nesterov & Smoothed Gradient ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Nesterov with different LR settings
        for lr_i in [0.003, 0.008, 0.001]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            t_alloc = min(remaining, 100)
            print(f"  Nesterov lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=lr_i)
            try_seq(a_opt, f"nesterov_lr{lr_i}")

        # Smoothed gradient
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print(f"  Smoothed grad T=0.0005, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc, T_init=0.0005, lr_init=0.02)
            try_seq(a_opt, "smooth_T0.0005")

        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  Smoothed grad T=0.002, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc, T_init=0.002, lr_init=0.015)
            try_seq(a_opt, "smooth_T0.002")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Peak pushing + fine QP (~250s) ===
    print("\n=== Phase 4: Peak Pushing & Fine QP ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Peak pushing
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print(f"  Peak push, t={t_alloc:.0f}s")
            a_opt, _ = peak_push(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_push")

        # Fine QP with low temperature
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Fine QP T=0.001, t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, T_init=0.001,
                                      use_topk=False, topk=5, blend=0.3)
            try_seq(a_opt, "fine_qp")

        # Top-5 QP
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  QP top-5, t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual(best_seq, time_limit=t_alloc, T_init=0.01,
                                      use_topk=True, topk=5, blend=0.4)
            try_seq(a_opt, "qp_top5")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final perturbations + verification (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Tiny perturbations with quick refinement
        for _ in range(5):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise_scale = 0.0005 + 0.001 * _
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(remaining, 15)
            a_opt, _ = smoothed_grad(f_pert, time_limit=t_alloc, T_init=0.0003, lr_init=0.005)
            try_seq(a_opt, f"perturb_{_}")

        # One more Nesterov pass
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=0.002)
            try_seq(a_opt, "final_nesterov")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
