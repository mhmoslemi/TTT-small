import time
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    size = 1

    W = np.fft.rfft(w, size)
    g = np.fft.irfft(A * np.conj(W), size)[:n]
    return 2.0 * g

"""
Optimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Nesterov-accelerated projected subgradient with adaptive temperature (main engine)
2. Frequency-domain amplitude compression (auxiliary diversification)
3. Coordinate descent polish
4. Multi-start with diverse initializations and combination heuristics

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, w, n):
    """Compute 2*cross_correlation(a, w) which gives the subgradient direction."""
    size = 1
    while size < 3 * n - 2:
        size *= 2
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    g = np.fft.irfft(A * np.conj(W), size)[:n]
    return 2.0 * g


def soft_weights(conv, t):
    """Compute softmax weights from convolution values with temperature t."""
    scaled = conv / t
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def exact_subgradient_weights(conv):
    """Uniform weights on the argmax set (exact subgradient of the max)."""
    mx = conv.max()
    w = (conv >= mx - 1e-12).astype(np.float64)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    return w


def nesterov_optimize(a0, time_limit, t_init=0.01, t_min=1e-6, t_decay=0.9996,
                      lr_init=0.06, beta=0.85, batch=3, verbose=False):
    """
    Nesterov-accelerated projected subgradient method with adaptive temperature.
    Takes 'batch' momentum steps between objective evaluations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t = t_init
    lr = lr_init
    v = np.zeros(n)

    t_start = time.monotonic()

    it = 0

            conv_la = fast_convolve(a_la)

                w_la = soft_weights(conv_la, t)
            else:
                w_la = exact_subgradient_weights(conv_la)
            g = compute_gradient(a_la, w_la, n)
            v = beta * v - lr * g
            a = project_simplex(a + v)

        conv = fast_convolve(a)

def spectral_compress(a, alpha=0.3, n_iter=20):
    """
    Frequency-domain amplitude compression: reduce spectral peaks to flatten
    the auto-convolution. Iteratively compresses |FFT(a)| and projects back.
    """
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    size = 1
    while size < 2 * n - 1:
        size *= 2

    for _ in range(n_iter):
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compress magnitude: pull peaks toward mean
        mean_mag = mag.mean()
        max_mag = mag.max()
        if max_mag < 1e-10:
            break
        # Soft compression
        compressed = mag * (1.0 - alpha * (1.0 - mean_mag / mag[1:]))
        # More careful: compress bins above mean
        mask = mag > mean_mag
        if mask.any():
            excess = mag[mask] - mean_mag
            mag[mask] = mag[mask] - alpha * excess
            mag[mask] = np.maximum(mag[mask], 0.1 * mean_mag)

        # Reconstruct
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        a = a_new

    return a

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        # Focus on coordinates that contribute to current peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        # Collect candidate coordinates (those that appear in peak convolutions)
        candidate_coords = set()
        for pk in peak_positions:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)

            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = a.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False

                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()

                if best_j_obj < best_obj - 1e-13:

        else:
            no_improve_passes = 0

def msequence_init(n):
    """Create an initialization inspired by m-sequences (polar to non-negative mapping)."""
    # Use a simple algebraic construction: golden ratio based pseudo-random
    phi = (1.0 + np.sqrt(5.0)) / 2.0

    vals = np.zeros(n)

        x = (x * phi) % 1.0

        rem = deadline - time.monotonic()
        if rem > 30:
            a_opt2, obj2 = coordinate_descent_polish(a_opt, time_limit=min(rem * 0.2, 60))
            try_seq(a_opt2, f"cd_{label}")

    # === Phase 5: Combination and final polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Try averaging with parent (can smooth out peaks)
        rem = deadline - time.monotonic()
        if rem > 20:
            for blend in [0.3, 0.5, 0.7]:
                a_blend = project_simplex(blend * best_seq + (1 - blend) * parent)
                try_seq(a_blend, f"blend_{blend}")

        # Final Nesterov on best
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_final = min(rem, 120)
            print(f"  Final Nesterov: t={t_final:.0f}s")
            a_opt, obj = nesterov_optimize(best_seq, time_limit=t_final,
                                           t_init=0.003, t_min=1e-7,
                                           t_decay=0.9998, lr_init=0.02,
                                           beta=0.92, batch=5, verbose=True)
            try_seq(a_opt, "final_nest")

        # Final CD
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_cd = min(rem, 80)
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(832) / 832
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(832) / 832
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
