import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_conv_matrix_row(f, k, n):
    """Get the row of the convolution matrix for position k."""
    row = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        row[j] = f[k - j]
    return row


def solve_lp_multidirectional(f, M, n, active_ks, n_directions=3):
    """Solve LP multiple times with different constraint subsets, return best direction."""
    best_g = None
    best_sum = -1
    
    K = len(active_ks)
    if K == 0:
        active_ks = np.arange(2 * n - 1)
        K = len(active_ks)
    
    # Direction 1: Use all active constraints
    A_ub = np.zeros((K, n))
    for row, k in enumerate(active_ks):
        A_ub[row] = make_conv_matrix_row(f, k, n)
    b_ub = np.full(K, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        s = res.x.sum()
        if s > 1e-10:
            best_g = res.x / s
            best_sum = s
    
    # Additional directions with subsets
    if K > 10:
        for trial in range(min(n_directions - 1, 2)):
            # Random subset of constraints
            subset_size = max(5, K // (trial + 2))
            idx = np.random.default_rng(trial * 100 + K).choice(K, subset_size, replace=False)
            A_sub = A_ub[idx]
            b_sub = b_ub[idx]
            res2 = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds, method='highs',
                           options={'presolve': True, 'time_limit': 20})
            if res2.success and res2.x is not None:
                s2 = res2.x.sum()
                if s2 > best_sum * 1.01:
                    best_g = res2.x / s2
                    best_sum = s2
    
    return best_g

def peak_shaving(f0, time_limit, verbose=False):
    """Targeted peak shaving: identify peaks and reduce contributing pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Find top peaks
        top_k = min(10, 2 * n - 1)
        peak_positions = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[peak_positions]
        max_val = peak_vals[0]

        # For each peak, find the contributing pairs and try to reduce them
        improved = False
        for k in peak_positions[:5]:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Contribution of each pair
            contributions = np.array([f[j] * f[k - j] for j in range(js, je + 1)])
            
            # Find the largest contribution
            if len(contributions) > 0:
                max_idx = js + np.argmax(contributions)
                other_idx = k - max_idx
                
                # Try to reduce f[max_idx] by moving mass to a position that
                # doesn't contribute to any active peak
                if f[max_idx] > 1e-6:
                    # Find a "safe" position (low contribution to all peaks)
                    danger = np.zeros(n)
                    for pk in peak_positions:
                        pk = int(pk)
                        pjs = max(0, pk - n + 1)
                        pje = min(n - 1, pk)
                        for pje_j in range(pjs, pje + 1):
                            danger[pje_j] += f[pk - pje_j]
                    
                    # Move mass from max_idx to least dangerous position
                    safe_positions = np.argsort(danger)[:10]
                    for sp in safe_positions:
                        if sp == max_idx or sp == other_idx:
                            continue
                        if f[sp] < 0.01:  # Don't concentrate too much
                            # Try moving a small amount
                            for frac in [0.05, 0.1, 0.2, 0.3]:
                                amount = f[max_idx] * frac
                                f_trial = f.copy()
                                f_trial[max_idx] -= amount
                                f_trial[sp] += amount
                                f_trial = np.maximum(f_trial, 0)
                                f_trial /= f_trial.sum()
                                cm = fast_convolve(f_trial).max()
                                if cm < best_obj - 1e-12:
                                    best_obj = cm
                                    best_f = f_trial.copy()
                                    f = f_trial
                                    improved = True
                                    break
                                elif cm < max_val - 1e-12:
                                    f = f_trial
                                    improved = True
                                    break
                        if improved:
                            break
        if not improved:
            # Try random perturbations
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            cm = fast_convolve(f).max()
            if cm > best_obj * 1.01:
                f = best_f.copy()

    return best_f, best_obj


def coord_descent_targeted(f0, time_limit, verbose=False):
    """Coordinate descent that targets peak reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-10:
                continue

            old_val = f[j]
            best_j_val = old_val
            best_j_obj = best_obj

            # Try several new values
            for mult in [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                remaining = 1.0 - new_val
                denom = 1.0 - old_val
                if denom < 1e-10 or remaining < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-10:
                    continue
                f_trial /= ss

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, seed=0, mode='flat'):
    """Spectral-based initialization."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    if mode == 'flat':
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        spectrum = np.exp(1j * phases)
    elif mode == 'raised_cosine':
        freqs = np.linspace(0, 1, fft_size // 2 + 1)
        spectrum_magnitude = np.cos(np.pi * freqs / 2)**2
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        spectrum = spectrum_magnitude * np.exp(1j * phases)
    else:
        spectrum = np.ones(fft_size // 2 + 1, dtype=complex)

    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def barycentric_mix(f1, f2, t):
    """Mix two sequences on the simplex."""
    return project_simplex((1 - t) * f1 + t * f2)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n from a (uses first half)."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def solve_lp_direction(f, M, n, active_k=None, time_out=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None:
        active_k = np.arange(2 * n - 1)
    K = len(active_k)
    A_ub = np.zeros((K, n))
    for row, k in enumerate(active_k):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_out})
    if res.success and res.x is not None:
        return res.x
    return None

    rng_local = np.random.default_rng(42 + K)

    # Primary: use all given constraints
    g = solve_lp_direction(f, M, n, active_k, time_out)
    if g is not None:
        s = g.sum()
        if s > best_sum:
            best_sum = s
            best_g = g

    # Secondary: random subsets
    for trial in range(n_dir - 1):
        if K <= 8:
            break
        subset_size = max(5, K // (trial + 2))
        idx = rng_local.choice(K, subset_size, replace=False)
        g_sub = solve_lp_direction(f, M, n, active_k[idx], time_out // 2)
        if g_sub is not None:
            s_sub = g_sub.sum()
            if s_sub > best_sum * 1.02:
                best_sum = s_sub
                best_g = g_sub

    return best_g


def alpha_line_search(f, g, n, n_trial=8):
    """Try multiple mixing alphas, return best one and resulting max_conv."""
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    # Geometric spacing of alphas
    alphas = np.array([0.05, 0.1, 0.15, 0.25, 0.35, 0.5, 0.7, 1.0])
    for alpha in alphas[:n_trial]:
        f_t = f + alpha * (g - f)
        f_t = np.maximum(f_t, 0.0)
        ts = f_t.sum()
        if ts < 1e-10:
            continue
        f_t /= ts
        val = fast_convolve(f_t).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_alpha, best_val


def lp_fw_optimize(f0, time_limit, verbose=False, top_k=40, n_dir=2):
    """LP-based Frank-Wolfe with multi-direction and adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(31415)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        half_total = 2 * n - 1
        # Adaptive constraint selection
        if top_k >= half_total or it <= 2:
            active_k = np.arange(half_total)
        else:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            if it % 15 == 0:
                n_rand = max(3, top_k // 5)
                rand_k = rng.choice(half_total, n_rand, replace=False)
                active_k = np.unique(np.concatenate([k_sorted, rand_k]))
            else:
                active_k = k_sorted

        # Solve LP for direction
        g = multi_direction_lp(f, M, n, active_k, n_dir=n_dir)

        if g is None:
            # Try relaxed
            g = solve_lp_direction(f, M * 1.1, n, active_k[:min(12, len(active_k))], 10)
            if g is None:
                g = solve_lp_direction(f, M * 1.2, n, active_k[:min(8, len(active_k))], 8)

        if g is None:
            no_improve += 1
            if no_improve > 25:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                no_improve = 0
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                no_improve = 0
            continue

        # Line search
        best_alpha, new_M = alpha_line_search(f, g, n)

        if best_alpha > 1e-6:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            conv = fast_convolve(f_new)
            M_new = conv.max()

            if M_new < best_obj - 1e-13:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                if verbose and it % 25 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            f = f_new
            M = M_new
        else:
            no_improve += 1

        if no_improve > 50:
            pert_type = no_improve % 5
            if pert_type == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            elif pert_type == 1:
                f = project_simplex(best_f * rng.uniform(0.6, 1.4, n))
            elif pert_type == 2:
                shift = rng.integers(1, max(2, n // 15))
                f = project_simplex(np.roll(best_f, int(shift)))
            elif pert_type == 3:
                f = project_simplex(0.85 * best_f + 0.15 * np.ones(n) / n)
            else:
                half_n = (n + 1) // 2
                pal_half = (best_f[:half_n] + best_f[n-1:n-half_n:-1]) / 2.0
                f = project_simplex(make_palindromic(pal_half, n))
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def spectral_shaping(f0, time_limit, verbose=False):
    """Iterative spectral shaping to flatten power spectrum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    gamma = 0.15
    stall = 0

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(f, fft_size)
        mag = np.abs(A)
        # Skip DC component (index 0) to preserve sum
        mean_mag = mag[1:].mean() if np.any(mag[1:] > 1e-10) else 1.0
        target = np.maximum(mag, 1e-10)
        shaped_mag = target ** (1 - gamma) * (mean_mag ** gamma)
        phase = A / np.maximum(mag, 1e-10)
        A_new = shaped_mag * phase
        A_new[0] = A[0]  # Preserve DC

        f_new = np.fft.irfft(A_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-10:
            break
        f_new /= ss

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            stall = 0
            if verbose and it % 100 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.8f}")
        else:
            stall += 1
            gamma *= 1.03
            if gamma > 3.0:
                gamma = 0.05
            if stall > 200:
                break

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """Targeted peak shaving: reduce contributing pairs at active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 100:
            break

        top_k = min(8, 2 * n - 1)
        peak_positions = np.argsort(conv)[::-1][:top_k]

        improved = False
        for k in peak_positions[:4]:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find the dominant contributing pair
            contributions = np.array([f[j] * f[k - j] for j in range(js, je + 1)])
            if contributions.max() < 1e-12:
                continue
            dom_idx = js + int(np.argmax(contributions))
            other_idx = k - dom_idx

            if f[dom_idx] < 1e-6:
                continue

            # Compute danger map: where would adding mass increase peaks?
            danger = np.zeros(n)
            for pk in peak_positions:
                pk = int(pk)
                pjs = max(0, pk - n + 1)
                pje = min(n - 1, pk)
                for pj in range(pjs, pje + 1):
                    danger[pj] += f[pk - pj]

            # Try moving mass from dom_idx to safe positions
            safe_idx = np.argsort(danger)[:15]
            for sp in safe_idx:
                sp = int(sp)
                if sp == dom_idx or sp == other_idx:
                    continue
                if f[sp] > 0.05:
                    continue
                for frac in [0.08, 0.15, 0.25, 0.4]:
                    amount = f[dom_idx] * frac
                    f_trial = f.copy()
                    f_trial[dom_idx] -= amount
                    f_trial[sp] += amount
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    cm = fast_convolve(f_trial).max()
                    if cm < best_obj - 1e-12:
                        best_obj = cm
                        best_f = f_trial.copy()
                        f = f_trial
                        conv = fast_convolve(f)
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)

    return best_f, best_obj


def coord_descent(f0, time_limit, verbose=False):
    """Multi-scale coordinate descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.5]

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-11:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                remaining = 1.0 - new_val
                denom = 1.0 - old_val
                if denom < 1e-10 or remaining < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-10:
                    continue
                f_trial /= ss

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, mom=0.94, top_k=30, temp=0.00002):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it_count = 0
    rng = np.random.default_rng(999)
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        if time.monotonic() - t0 > time_limit:
            break

        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 400:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, seed=0, mode='flat'):
    """Spectral-based initialization for flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    if mode == 'flat':
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        spectrum = np.exp(1j * phases)
    elif mode == 'cosine':
        freqs = np.linspace(0, 1, fft_size // 2 + 1)
        mag = np.cos(np.pi * freqs / 2) ** 1.5
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        spectrum = mag * np.exp(1j * phases)
    else:
        spectrum = np.ones(fft_size // 2 + 1, dtype=complex)

    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig = sig / ss
    else:
        sig = np.ones(n) / n
    return sig


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining improved LP-Frank-Wolfe, spectral
    shaping, peak shaving, coordinate descent, and subgradient polishing
    with adaptive restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent from previous best
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: LP-FW on parent ===
    print("=== Phase 1: LP-FW on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(450, rem * 0.32)
    if t1 > 40:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True, top_k=50, n_dir=2)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Spectral shaping ===
    print(f"\n=== Phase 2: Spectral Shaping (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_sp = min(rem * 0.1, 150)
            a_opt, _ = spectral_shaping(best_seq, time_limit=t_sp, verbose=True)
            try_seq(a_opt, "spectral")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_ps = min(rem * 0.1, 130)
            a_opt, _ = peak_shaving(best_seq, time_limit=t_ps)
            try_seq(a_opt, "peak_shave")

    # === Phase 4: LP-FW on diverse inits at multiple lengths ===
    print(f"\n=== Phase 4: Multi-Length Spectral LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [512, 768, 1024]
    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)
        f_spec = spectral_init(target_n, seed=target_n + 7, mode='flat')
        f_mix = project_simplex(0.7 * f_interp + 0.3 * f_spec)

        t_alloc = min(rem * 0.16, 180)
        if t_alloc < 20:
            break
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_mix, time_limit=t_alloc, verbose=False, top_k=35, n_dir=1)
        try_seq(a_opt, f"multileng_n{target_n}")

    # === Phase 5: Deep restarts ===
    print(f"\n=== Phase 5: Deep Restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        restarts = [
            ("noise_tiny", project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)),
            ("noise_small", project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)),
            ("rev_sym", project_simplex(0.7 * best_seq + 0.3 * best_seq[::-1])),
            ("flat_blend", project_simplex(0.85 * best_seq + 0.15 * np.ones(n_best) / n_best)),
            ("gaussian", project_simplex(
                np.exp(-3.0 * (np.arange(n_best) - n_best / 2.0) ** 2 / (0.45 * n_best) ** 2))),
            ("palindromic", project_simplex(make_palindromic(

        ]

        for i, (name, f_init) in enumerate(restarts):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem / max(1, len(restarts) - i), 130)
            if t_alloc < 15:
                break
            print(f"  Restart [{name}], t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc,
                                      verbose=False, top_k=30, n_dir=1)

    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")

        rem = deadline - time.monotonic()

    print(f"\n=== Phase 7: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:

            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,

            # Fine polish
            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt2, _ = sg_polish(a_opt, time_limit=min(rem * 0.2, 70),
                                      lr=0.0004, mom=0.96,
                                      top_k=max(10, n_best // 10),
                                      temp=0.000008)
                try_seq(a_opt2, "sg_fine")

    # === Phase 8: Final LP-FW + micro perturbations ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        if rem > 20:
            t_final = min(rem * 0.3, 80)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final,
                                      verbose=False, top_k=min(200, 2 * n_best - 1),
                                      n_dir=1)
            try_seq(a_opt, "final_lpfw")

        for i in range(3):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.0003 + 0.0008 * i)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.15, 30)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_pert,
                                      verbose=False, top_k=25, n_dir=1)
            try_seq(a_opt, f"final_pert_{i}")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem * 0.35, 20),
                                 lr=0.0002, mom=0.96,
                                 top_k=max(8, n_best // 12),
                                 temp=0.000005)
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
