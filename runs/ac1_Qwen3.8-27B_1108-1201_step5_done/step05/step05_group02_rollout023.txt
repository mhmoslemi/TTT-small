import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak_active(f, n, k_frac=0.7):
    """Solve LP: min t s.t. (f*g)[k] <= t for active k, g in simplex.
    Uses only top k_frac fraction of constraints for speed."""
    conv = fast_convolve(f)
    M = conv.max()
    
    n_constraints = 2 * n - 1
    k_active = max(10, int(n_constraints * k_frac))
    
    if k_active >= n_constraints:
        active = np.arange(n_constraints)
    else:
        active = np.argpartition(conv, -k_active)[-k_active:]
    
    active = np.sort(active)
    m = len(active)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    for idx, k in enumerate(active):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[idx, j] = f[j]
        A_ub[idx, -1] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

def solve_lp_min_peak_full(f, n):

    n_constraints = 2 * n - 1
    
    c = np.zeros(n + 1)

def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    use_full = False

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Choose LP variant

                g, lp_t = solve_lp_min_peak_active(f, n, k_frac=k_frac)
            else:
                g, lp_t = solve_lp_min_peak_full(f, n)
            if g is None:
                g, lp_t = solve_lp_min_peak_full(f, n)
            use_full = False

        if g is None:
            # Fallback to max-sum LP
            g2 = solve_lp_max_sum(f, M * 1.1, n)
            if g2 is not None:
                gs = g2.sum()
                g = g2 / gs if gs > 1e-15 else np.ones(n) / n
            else:
                g = np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:

                scale = 0.005 + 0.01 * min(no_improve // 50, 5)

                no_improve = 0

                else:

def nesterov_sg_optimize(f0, time_limit, lr=0.005, momentum=0.9, top_k=50,
                         temp=0.0001, verbose=False):
    """Nesterov-accelerated subgradient method with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    beta = momentum
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 - t0 > time_limit:
            break
        it += 1

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_actual = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_actual)[-k_actual:]

        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * y[kg[mask]]

        # Nesterov update
        v = beta * v - cur_lr * sg
        f_new = project_simplex(f + v)
        y_new = project_simplex(f_new + beta * (f_new - f))
        f = f_new
        y = y_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    NSG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 7919) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Fine-grained coordinate descent for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        improved = False

        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Try perturbing each contributing coordinate
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break

                old_val = f[j]
                if old_val < 1e-14:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj

                # Try several multipliers
                for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.7:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ts = f_trial.sum()
                    if ts < 1e-15:
                        continue
                    f_trial = f_trial / ts

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        ts = f.sum()
                        if ts > 1e-15:
                            f = f / ts

                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Random exploration
                rng = np.random.default_rng(pass_count * 31337)
                tried = 0
                for _ in range(min(50, n // 10)):
                    if time.monotonic() - t0 > time_limit:
                        break
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-12:
                        continue
                    for mult in [0.4, 0.6, 1.4, 2.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.7:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        ts = f_trial.sum()
                        if ts < 1e-15:
                            continue
                        f_trial = f_trial / ts
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            tried += 1
                            break
                    if tried >= 20:
                        break

            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def anneal_restart(f0, seed):
    """Create a perturbed version of f0 for restart."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    # Randomly select a subset of coordinates to modify
    n_modify = max(10, n // 5)
    idx = rng.choice(n, size=n_modify, replace=False)
    f_new = f0.copy()
    for j in idx:
        f_new[j] *= rng.uniform(0.3, 2.0)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f0.copy()
    return f_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW with active-set, Nesterov subgradient,
    multi-length exploration, annealing restarts, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    print(f"Starting best: {best_score:.8f}")

    # === Phase 1: LP-FW with active set on parent ===
    print("\n=== Phase 1: LP-FW (active) on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(300, rem * 0.30)
    if t1 > 30:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True,
                                  use_active=True, k_frac=0.6)
        try_seq(a_opt, "lpfw_active_p1")

    # === Phase 2: Nesterov subgradient refinement ===
    print(f"\n=== Phase 2: Nesterov SG (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t2 = min(200, rem * 0.20)
        if t2 > 30:
            print(f"  n={n_best}, t={t2:.0f}s")
            a_opt, _ = nesterov_sg_optimize(best_seq, time_limit=t2,
                                            lr=0.004, momentum=0.92,
                                            top_k=max(40, n_best // 6),
                                            temp=0.00008)
            try_seq(a_opt, "nesterov_p2")

    # === Phase 3: LP-FW full constraints ===
    print(f"\n=== Phase 3: LP-FW (full) (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(250, rem * 0.22)
        if t3 > 30:
            print(f"  n={len(best_seq)}, t={t3:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True,
                                      use_active=False)
            try_seq(a_opt, "lpfw_full_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    base_n_cur = len(base)

    # Generate candidate lengths around the current best
    candidate_lengths = set()
    for delta in [-200, -100, -50, 50, 100, 200, 400, 800]:
        ln = base_n_cur + delta
        if 300 <= ln <= 3000:
            candidate_lengths.add(ln)
    # Also add some power-of-2-ish lengths
    for cn in [400, 512, 640, 800, 1024, 1280, 1600, 2048]:
        candidate_lengths.add(cn)
    candidate_lengths = sorted(candidate_lengths)

    # Sort by proximity to current best length
    candidate_lengths.sort(key=lambda x: abs(x - base_n_cur))

    explored = 0
    for target_n in candidate_lengths:
        if target_n == base_n_cur:
            continue
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(120, rem * 0.15)
        if t_alloc < 25:
            break

        f_interp = interpolate_sequence(base, target_n)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")

        # LP-FW on interpolated
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc * 0.7,
                                  use_active=True, k_frac=0.5)
        try_seq(a_opt, f"lpfw_n{target_n}")

        # Quick Nesterov SG
        rem2 = deadline - time.monotonic()
        if rem2 > 30:
            a_opt2, _ = nesterov_sg_optimize(a_opt, time_limit=min(t_alloc * 0.3, rem2 - 5),
                                             lr=0.003, momentum=0.92,
                                             top_k=max(20, target_n // 10),
                                             temp=0.00005)
            try_seq(a_opt2, f"nsg_n{target_n}")

        explored += 1
        if explored >= 5:
            break

    # === Phase 5: Annealing restarts ===
    print(f"\n=== Phase 5: Annealing restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 0
        for r in range(8):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_r = min(80, rem * 0.12)
            if t_r < 20:
                break

            f_pert = anneal_restart(best_seq, seed + r * 7919)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_r,
                                      use_active=True, k_frac=0.6)
            try_seq(a_opt, f"anneal_{r}")
            n_restarts += 1

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(200, rem * 0.25)
        if t_cd > 20:
            print(f"  n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")

    # === Phase 7: Final rapid iterations ===
    print(f"\n=== Phase 7: Final (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        # Quick alternating LP-FW and SG
        while rem > 20:
            t_chunk = min(40, rem)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_chunk * 0.6,
                                      use_active=True, k_frac=0.5)
            s1 = try_seq(a_opt, "final_lpfw")
            rem = deadline - time.monotonic() - 10
            if rem > 15:
                a_opt2, _ = nesterov_sg_optimize(best_seq, time_limit=min(t_chunk * 0.4, rem),
                                                 lr=0.001, momentum=0.94,
                                                 top_k=max(20, len(best_seq) // 10),
                                                 temp=0.00003)
                try_seq(a_opt2, "final_nsg")
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break

    # === Phase 8: Ultra-fine polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "ultra_cd")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(512) / 512

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
