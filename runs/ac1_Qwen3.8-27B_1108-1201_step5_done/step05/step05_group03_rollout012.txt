def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if not len(rho):
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hamming_window(n):
    if n == 1:
        return np.array([1.0])
    w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    return w / w.sum()

def hanning_window(n):
    if n == 1:
        return np.array([1.0])
    w = 0.5 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    return w / w.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using spline."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lp_direction(f, M, n, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # (f*g)[k] = sum_j f[k-j] * g[j] for valid k-j
    # So A_ub[k, j] = f[k-j] when 0 <= k-j <= n-1
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to (f*g)[k] for k = j, j+1, ..., j+n-1
        # (f*g)[k] = sum_j f[k-j]*g[j], so for fixed j, k-j ranges over 0..n-1
        # i.e., k ranges from j to j+n-1
        A_ub[j:j+n, j] = f
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
    if res.success:
        g = np.maximum(res.x, 0)
        if g.sum() > 1e-15:
            return g / g.sum()
    return None

def nesterov_subgradient(f0, time_limit, verbose=False):
    """Nesterov-accelerated projected subgradient with adaptive log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    # Nesterov momentum
    y = f.copy()  # extrapolated point
    beta = 0.9  # momentum
    
    # Adaptive temperature
    T = 3.0 / n
    T_min = 0.3 / n
    
    # Step size schedule
    base_lr = 0.003
    total_time = time_limit
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = (time.monotonic() - t0) / total_time
        if progress > 1.0:
            break
        
        # Schedule
        T = T * (T_min / T) ** (0.001 * progress)
        lr = base_lr * (1.0 - 0.3 * progress)
        
        # Compute smoothed gradient at y
        conv_y = fast_convolve(y)
        max_cv = conv_y.max()
        
        # Log-sum-exp smoothing
        top_k = max(10, min(2 * n - 1, n))
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        pv = conv_y[peaks]
        scaled = T * (pv - max_cv)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]
        
        # Nesterov update
        f_old = f
        f = project_simplex(y - lr * sg)
        
        # Update extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        beta = min(beta_new, 2.0)
        y = f + ((beta - 1.0) / beta) * (f - f_old)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        
        new_M = fast_convolve(f).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 41 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            y = f.copy()
            no_improve = 0
    
    return best_f, best_obj

                        val = fast_convolve(f_trial).max()

"""

1. Window function initialization + multi-scale growth
2. LP-based Frank-Wolfe with golden-section line search (primary)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if not len(rho):
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hamming_window(n):
    if n == 1:
        return np.array([1.0])
    w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    return w / w.sum()


def hanning_window(n):
    if n == 1:
        return np.array([1.0])
    w = 0.5 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    return w / w.sum()

    if n == 1:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = np.maximum(res.x, 0)
        if g.sum() > 1e-15:
            return g / g.sum()
    return None

        if progress > 1.0:
            break

        T_cur = T * (T_min / T) ** (0.5 * progress)
        lr = base_lr * (1.0 - 0.3 * progress)

        conv_y = fast_convolve(y)
        max_cv = conv_y.max()

        top_k = max(20, min(2 * n - 1, n))
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        pv = conv_y[peaks]
        scaled = T_cur * (pv - max_cv)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]

        f_old = f.copy()
        f = project_simplex(y - lr * sg)

        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom * beta_mom)) / 2.0
        beta_mom = min(beta_new, 2.5)
        y = f + ((beta_mom - 1.0) / beta_mom) * (f - f_old)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys

        new_M = fast_convolve(f).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Nesterov iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(it * 41 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            y = f.copy()
            beta_mom = 0.9
            no_improve = 0

    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False):
    """Optimize pairs of coordinates by mass transfer with diverse pair selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    offsets = [1, 2, 3, 5, 8, 13, 21, 34]
    no_improve_passes = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        if passes % 5 == 1:
            order = rng.permutation(n)
        elif passes % 5 == 2:
            order = np.arange(n)
        elif passes % 5 == 3:
            order = rng.permutation(n)
        else:
            order = np.argsort(-f)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = order[idx_i]

            j_list = []
            for offset in offsets:
                if idx_i + offset < n:
                    j_list.append(order[idx_i + offset])
            j_list.append(n - 1 - i)
            j_list.append((i + n // 2) % n)
            j_list.append(i + 1 if i + 1 < n else i - 1)

            for j in j_list:
                if i == j or j < 0 or j >= n:
                    continue

                c = f[i] + f[j]
                if c < 1e-10:
                    continue

                best_t = f[i] / c
                best_val = best_obj

                candidates = [best_t, 0.0, 1.0, 0.1, 0.25, 0.5, 0.75, 0.9]
                if 0.1 < best_t < 0.9:
                    candidates.extend([best_t - 0.05, best_t + 0.05, best_t - 0.1, best_t + 0.1])

                for t in candidates:
                    if t < -0.02 or t > 1.02:
                        continue
                    t = np.clip(t, 0.0, 1.0)
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1.0 - t) * c
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-13:
                        best_val = val
                        best_t = t

                if best_val < best_obj - 1e-12:
                    f[i] = best_t * c
                    f[j] = (1.0 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 4:
                break

    return best_f, best_obj


def block_optimize(f0, time_limit, block_size=8):
    """Optimize blocks of coordinates with local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    num_blocks = (n + block_size - 1) // block_size
    no_improve = 0

    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit or no_improve > 150:
            break

        improved = False
        block_order = rng.permutation(num_blocks)

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            block_sum = f[start:end].sum()
            if block_sum < 1e-12:
                continue

            for _ in range(4):
                if time.monotonic() - t0 > time_limit:
                    break
                pert = rng.standard_normal(end - start) * 0.015 * block_sum
                trial_block = np.maximum(f[start:end] + pert, 0)
                ts = trial_block.sum()
                if ts < 1e-12:
                    continue
                trial_block = trial_block * (block_sum / ts)

                f_trial = f.copy()
                f_trial[start:end] = trial_block
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()

                if obj < best_obj - 1e-13:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_f, best_obj


def check_and_update(a, label, state):
    """Evaluate a candidate and update state if improved."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    if a.sum() < 0.01:
        return state
    a = np.clip(a, 0.0, 1000.0)
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    return state


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Initialization exploration with window functions
    Phase 2: Multi-scale growth
    Phase 3: Intensive LP-FW
    Phase 4: Nesterov subgradient
    Phase 5: Diverse restarts
    Phase 6: Pairwise + block polish
    Phase 7: Final refinement
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            state = check_and_update(parent, "parent", state)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = hamming_window(384)
        state = check_and_update(parent, "default", state)
        print(f"Default: n={len(parent)}, score={state['best_score']:.8f}")

    parent_n = len(parent)
    base = state['best_seq'].copy() if state['best_seq'] is not None else parent

    # === Phase 1: Initialization exploration ===
    print(f"\n=== Phase 1: Init exploration (best={state['best_score']:.8f}) ===")
    inits = {
        'parent': parent,
        'uniform': np.ones(parent_n) / parent_n,
        'hamming': hamming_window(parent_n),
        'hanning': hanning_window(parent_n),
        'blackman': blackman_window(parent_n),
        'flattop': flattop_window(parent_n),
    }

    rng = np.random.default_rng(seed)
    inits['rand1'] = project_simplex(rng.standard_normal(parent_n))
    inits['rand2'] = project_simplex(np.abs(rng.standard_normal(parent_n)))

    best_init_obj = fast_convolve(base).max()
    best_init = base

    for name, init in inits.items():
        rem = deadline - time.monotonic()
        if rem < 250:
            break
        t = min(25, rem * 0.04)
        if t < 5:
            break
        a_opt, obj = lp_fw_optimize(init, time_limit=t)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_opt
            print(f"  Better init: {name} (obj={obj:.8f})")
            state = check_and_update(a_opt, f"init_{name}", state)

    base = best_init

    # === Phase 2: Multi-scale growth ===
    print(f"\n=== Phase 2: Multi-scale (best={state['best_score']:.8f}) ===")
    for target_n in [512, 768, 1024, 1536, 2048]:
        if target_n <= parent_n and target_n <= len(base):
            continue
        rem = deadline - time.monotonic()
        if rem < 100:
            break

        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.25, 120)
        if t_alloc < 15:
            continue

        print(f"  Growing to n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        state = check_and_update(a_opt, f"growth_n{target_n}")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_nes = min(rem * 0.15, 50)
            if t_nes > 10:
                a_opt2, _ = nesterov_subgradient(a_opt, time_limit=t_nes)
                state = check_and_update(a_opt2, f"nesterov_n{target_n}")

    # === Phase 3: Intensive LP-FW on best ===
    print(f"\n=== Phase 3: Intensive LP-FW (best={state['best_score']:.8f}) ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            t = min(rem * 0.3, 250)
            print(f"  Time: {t:.0f}s, n={len(state['best_seq'])}")
            a_opt, _ = lp_fw_optimize(state['best_seq'], time_limit=t, verbose=True)
            state = check_and_update(a_opt, "lpfw_intensive")

    # === Phase 4: Nesterov on best ===
    print(f"\n=== Phase 4: Nesterov (best={state['best_score']:.8f}) ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t = min(rem * 0.2, 100)
            print(f"  Time: {t:.0f}s, n={len(state['best_seq'])}")
            a_opt, _ = nesterov_subgradient(state['best_seq'], time_limit=t)
            state = check_and_update(a_opt, "nesterov_main")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse restarts (best={state['best_score']:.8f}) ===")
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        rng2 = np.random.default_rng(seed + 777)

        for p in range(6):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t_alloc = min(rem / max(1, 6 - p), 120)
            if t_alloc < 10:
                continue

            if p == 0:
                f_init = state['best_seq'].copy()
            elif p == 1:
                f_init = project_simplex(state['best_seq'] + rng2.standard_normal(n_best) * 0.01)
            elif p == 2:
                f_init = state['best_seq'][::-1].copy()
                f_init /= f_init.sum()
            elif p == 3:
                w = [hamming_window, hanning_window, blackman_window, flattop_window][p % 4]
                f_init = 0.6 * state['best_seq'] + 0.4 * w(n_best)
                f_init = project_simplex(f_init)
            elif p == 4:
                f_sharp = state['best_seq'] ** 1.5
                f_init = project_simplex(f_sharp)
            else:
                f_init = project_simplex(state['best_seq'] + rng2.standard_normal(n_best) * 0.02)

            if p % 2 == 0:
                a_opt, _ = lp_fw_optimize(f_init, time_limit=min(t_alloc, 100))
            else:
                a_opt, _ = nesterov_subgradient(f_init, time_limit=min(t_alloc, 80))
            state = check_and_update(a_opt, f"diverse_{p}")

    # === Phase 6: Pairwise + Block polish ===
    print(f"\n=== Phase 6: Pairwise+Block (best={state['best_score']:.8f}) ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t = min(rem * 0.35, 150)
            print(f"  Pairwise t={t:.0f}s, n={len(state['best_seq'])}")
            a_opt, _ = pairwise_optimize(state['best_seq'], time_limit=t)
            state = check_and_update(a_opt, "pairwise")

        rem = deadline - time.monotonic()
        if rem > 20:
            t = min(rem * 0.3, 80)
            bs = max(4, min(16, len(state['best_seq']) // 48))
            print(f"  Block t={t:.0f}s, bs={bs}")
            a_opt, _ = block_optimize(state['best_seq'], time_limit=t, block_size=bs)
            state = check_and_update(a_opt, "block")

    # === Phase 7: Final refinement ===
    print(f"\n=== Phase 7: Final (best={state['best_score']:.8f}) ===")
    if state['best_seq'] is not None:
        # LP-FW final
        rem = deadline - time.monotonic()
        if rem > 20:
            t = min(rem * 0.4, 80)
            a_opt, _ = lp_fw_optimize(state['best_seq'], time_limit=t)
            state = check_and_update(a_opt, "lpfw_final")

        # Nesterov final
        rem = deadline - time.monotonic()
        if rem > 15:
            t = min(rem * 0.3, 50)
            a_opt, _ = nesterov_subgradient(state['best_seq'], time_limit=t)
            state = check_and_update(a_opt, "nesterov_final")

        # Pairwise final
        rem = deadline - time.monotonic()
        if rem > 15:
            t = min(rem * 0.4, 40)
            a_opt, _ = pairwise_optimize(state['best_seq'], time_limit=t)
            state = check_and_update(a_opt, "pairwise_final")

        # Tiny perturbations
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            rng3 = np.random.default_rng(500 + trial)
            f_pert = project_simplex(state['best_seq'] + rng3.standard_normal(len(state['best_seq'])) * 0.002)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 15))
            state = check_and_update(a_opt, f"perturb_{trial}")

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs) and fs < state['best_score']:
        state['best_score'] = fs
    elif not np.isfinite(fs):
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
