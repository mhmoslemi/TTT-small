"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth gradient descent (log-sum-exp), adaptive length changes,
diverse initializations, and coordinate descent polishing.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_to_length(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def smooth_gd(a0, time_limit, T_init=50.0, T_growth=1.0015, lr_init=0.02,
              lr_decay=0.99995, momentum_base=0.85, top_K=30,
              restart_interval=500, verbose=False):
    """
    Smooth gradient descent with momentum on the probability simplex.
    Minimizes smooth approximation of max(conv(a,a)).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    momenta = np.zeros(n)
    beta = momentum_base

    c = fast_convolve(a)
    best_val = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        c_max = c.max()

        # Softmax weights (focus on peaks)
        K = min(top_K, 2 * n - 1)
        peak_idx = np.argpartition(c, -K)[-K:]
        pw = c[peak_idx]
        scaled = T * (pw - pw.max())
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(K)
            w[np.argmax(pw)] = 1.0
        else:
            w /= w_sum

        # Compute gradient using only top K peaks
        grad = np.zeros(n)
        for i in range(K):
            wk = w[i]
            if wk < 1e-12:
                continue
            k = int(peak_idx[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            m_vals = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += wk * 2.0 * a[m_vals]

        # Momentum update
        momenta = beta * momenta + grad
        a_new = a - lr * momenta
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        val = fast_convolve(a_new).max()
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      SGD iter {it}: val={best_val:.10f}, T={T:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        a = a_new
        T *= T_growth
        lr = max(lr * lr_decay, 0.0001)

        # Adaptive restart
        if no_improve > restart_interval:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise_scale = min(0.01, 0.002 * (no_improve // restart_interval))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            momenta = np.zeros(n)
            lr = max(lr_init * 0.5, 0.001)
            T = max(T_init, T * 0.5)
            no_improve = 0
            if time.monotonic() - t0 > time_limit:
                break

    return best_a, best_val

def coord_descent_polish(a0, time_limit, verbose=False):
    """Coordinate descent: adjust individual coordinates to reduce peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_val = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        c = fast_convolve(a)

        # Find top peaks
        K = min(5, 2 * n - 1)
        peak_positions = np.argpartition(c, -K)[-K:]

        for pk in peak_positions:
            pk = int(pk)
            j_start = max(0, pk - n + 1)
            j_end = min(n - 1, pk)

            # For each coordinate contributing to this peak, try adjustments
            for j in range(j_start, j_end + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old = a[j]
                if old < 1e-14:
                    continue

                best_j = old
                best_jv = best_val

                for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]:
                    new_v = old * mult
                    if new_v < 0 or new_v > 0.5:
                        continue
                    # Adjust: set a[j]=new_v, scale others to maintain sum=1
                    denom = 1.0 - old
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_v) / denom
                    a_trial = a.copy()
                    a_trial[j] = new_v
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    a_trial = np.maximum(a_trial, 0)
                    ts = a_trial.sum()
                    if ts < 1e-10:
                        continue
                    a_trial /= ts

                    val = fast_convolve(a_trial).max()
                    if val < best_jv - 1e-14:
                        best_jv = val
                        best_j = new_v

                if best_j != old:
                    denom = 1.0 - old
                    if denom > 1e-10:
                        scale = (1.0 - best_j) / denom
                        a[j] = best_j
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        a[mask] *= scale
                        a = np.maximum(a, 0)
                        a /= a.sum()

                    if best_jv < best_val - 1e-14:
                        best_val = best_jv
                        best_a = a.copy()
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_a, best_val


def gen_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian-like
    x = np.linspace(-1, 1, n)
    g = np.exp(-2 * x**2)
    inits.append(g / g.sum())

    # Triangular
    tri = np.minimum(np.arange(n), n - np.arange(n)) + 1
    inits.append(tri.astype(float) / tri.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (np.arange(n) / n - 0.5)))
    rc = np.maximum(rc, 1e-6)
    inits.append(rc / rc.sum())

    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Two bumps
    centers = [n * 0.25, n * 0.75]
    two = np.zeros(n)
    for c in centers:
        two += np.exp(-((np.arange(n) - c)**2) / (n * 0.02))
    inits.append(two / two.sum())

    # Sparse-ish (every other)
    sp = np.zeros(n)
    sp[::2] = 1.0
    inits.append(sp / sp.sum())

    # Sine-modulated
    sine = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n) / n * 3)
    sine = np.maximum(sine, 0.01)
    inits.append(sine / sine.sum())

    return inits

                start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)

            a_opt, val = smooth_gd(start, time_limit=t_alloc, verbose=False, **cfg)
            try_seq(a_opt, f"final_sgd_{i}")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        explore_lengths = []
        for delta in [-256, -128, 128, 256, 384, 512]:
            ln = n_best + delta
            if 256 <= ln <= 3000:
                explore_lengths.append(ln)

        for target_n in explore_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.3, 120)
            if t_alloc < 15:
                break
            interp_a = interp_to_length(best_seq, target_n)
            a_opt, val = smooth_gd(interp_a, time_limit=t_alloc,
                                    T_init=50, T_growth=1.002,
                                    lr_init=0.015, lr_decay=0.99997,
                                    momentum_base=0.87,
                                    top_K=max(20, target_n // 20))
            try_seq(a_opt, f"len_explore_{target_n}")

    # === Phase 6: Final perturbation attempts ===
    print(f"\n=== Phase 6: Perturbations (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            scale = 0.002 + 0.002 * p
            noise = rng.standard_normal(n_best) * scale
            perturbed = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.4, 40)
            a_opt, val = smooth_gd(perturbed, time_limit=t_alloc,
                                    T_init=60, T_growth=1.003,
                                    lr_init=0.01, lr_decay=0.99999,
                                    momentum_base=0.9,
                                    top_K=max(15, n_best // 25))
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
