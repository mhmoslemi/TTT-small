def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def ls_exp_objective(a, t):
    """Compute log-sum-exp of convolution and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = t * (conv - max_c)
    exp_vals = np.exp(shifted)
    lse = max_c + np.log(exp_vals.sum()) / t
    weights = exp_vals / exp_vals.sum()
    
    # Gradient: 2 * T1 where T1[j] = sum_k a[k] * w[j+k]
    a_rev = a[::-1]
    # conv of a_rev with weights
    conv_aw = np.convolve(a_rev, weights, mode='full')
    T1 = conv_aw[n-1:2*n-1]
    grad = 2.0 * T1
    
    return lse, grad, conv

def mirror_descent_step(a, grad, step_size):
    """Mirror descent update on the simplex (multiplicative weights)."""
    # a_new[i] = a[i] * exp(-step_size * grad[i]) / sum(...)
    log_a = np.where(a > 1e-30, np.log(a), -30)
    log_new = log_a - step_size * grad
    log_new -= log_new.max()  # for numerical stability
    new_a = np.exp(log_new)

Now, for the mirror descent approach:

The Bregman divergence for the simplex with entropy is the KL divergence. The mirror descent update is:
`a_new[i] = a[i] * exp(-eta * grad[i]) / Z`

where Z is the normalization constant. This is equivalent to:
`log(a_new[i]) = log(a[i]) - eta * grad[i] + const`

This is a multiplicative weight update. It naturally stays in the interior of the simplex (all components positive) and is often more efficient than Euclidean projection for simplex-constrained problems.

However, mirror descent can be slow near the boundary (when some components are very small). We can add a small regularization to prevent this.

For the step size, we can use:
- Fixed step size: eta = 1/sqrt(t) (standard)
- Adaptive: backtracking line search
- Or: eta = c / (max|grad|) for some constant c

Now, let me also think about the spectral approach. The key insight is:

The autocorrelation is the inverse FFT of `|A(f)|^2` where `A(f) = FFT(a)`.

To minimize the peak of the autocorrelation, we want `|A(f)|^2` to be as flat as possible. But since `a >= 0`, we can't freely manipulate the phase of `A(f)`.

One approach:
1. Compute `A(f) = FFT(a)`
2. Modify `|A(f)|` to be flatter (e.g., `|A_new(f)| = |A(f)|^alpha` for alpha < 1)
3. Keep the phase: `A_new(f) = |A_new(f)| * exp(i * arg(A(f)))`
4. `a_new = IFFT(A_new)`
5. Project onto non-negative: `a_new = max(a_new, 0)`
6. Renormalize
7. Repeat

This is similar to the Gerchberg-Saxton algorithm.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 0:
        return np.array([0.0])
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def ls_exp_objective_grad(a, t):
    """
    Compute log-sum-exp approximation of max convolution and its gradient.
    Returns (lse_value, gradient_vector, full_convolution).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = t * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    lse = max_c + np.log(exp_vals.sum()) / t
    weights = exp_vals / exp_vals.sum()

    # Gradient: grad[j] = 2 * sum_k a[k] * w[j+k]
    # Using convolution trick: T1[j] = conv(a_rev, w)[j+n-1]
    a_rev = a[::-1]
    conv_aw = np.convolve(a_rev, weights, mode='full')
    T1 = conv_aw[n - 1:2 * n - 1]
    grad = 2.0 * T1

    return lse, grad, conv


def mirror_descent_update(a, grad, eta):
    """Mirror descent (multiplicative weights) update on simplex."""
    # a_new[i] proportional to a[i] * exp(-eta * grad[i])
    # Use log-space for numerical stability
    log_a = np.where(a > 1e-30, np.log(a), -30.0)
    log_new = log_a - eta * grad
    log_new -= log_new.max()
    new_a = np.exp(log_new)
    s = new_a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return new_a / s


def projected_gradient_update(a, grad, eta):
    """Projected gradient descent step on simplex."""
    return project_simplex(a - eta * grad)

def spectral_flatten(f0, time_limit, alpha_range=(0.3, 0.9), verbose=False):
    """
    Spectral flattening: modify |A(f)| to reduce peak autocorrelation.
    iteratively: flatten spectrum -> IFFT -> clip negatives -> normalize
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute FFT
        A = np.fft.rfft(f, n)
        mag = np.abs(A)
        phase = np.angle(A)

        # Apply flattening: mag_new = mag^alpha (alpha < 1 flattens peaks)
        for alpha in np.linspace(alpha_range[0], alpha_range[1], 7):
            mag_new = np.power(mag + 1e-15, alpha)
            A_new = mag_new * np.exp(1j * phase)
            f_new = np.fft.irfft(A_new, n)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn

            obj = 2.0 * n * fast_convolve(f_new).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_new.copy()
                if verbose and it % 20 == 0:
                    print(f"    Spec iter {it}: obj={best_obj:.10f}, alpha={alpha:.3f}")

        # Also try: reduce high-frequency components (smoothing)
        freq = np.arange(len(A))
        for beta in [0.5, 0.7, 0.85, 0.95]:
            smooth = np.exp(-beta * (freq / max(1, len(A) - 1)) ** 2)
            A_smooth = A * smooth
            f_new = np.fft.irfft(A_smooth, n)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            obj = 2.0 * n * fast_convolve(f_new).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_new.copy()

        # Check for convergence
        if best_f is not None and np.allclose(f, best_f, atol=1e-10) and it > 10:
            break

    return best_f, best_obj


def coordinate_descent_refine(f0, time_limit, multipliers=None):
    """
    Coordinate descent: for each coordinate, try scaling it and renormalizing.
    Fast and effective for fine-tuning.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if multipliers is None:
        multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]

    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale

                obj = 2.0 * n * fast_convolve(f_trial).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    f[j] = new_val
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def tp_iterations(f0, time_limit, num_iters=5000):
    """
    Time-Phase (TP) iterations for low-PAPR sequence design.
    Based on the Lee-Brakke algorithm:
    1. Compute autocorrelation
    2. Find peak locations
    3. Modify sequence to reduce peaks
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    no_improve = 0

    for it in range(num_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation
        conv = fast_convolve(f)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        # Find the peak lag
        k_peak = np.argmax(conv)
        peak_val = conv[k_peak]

        # Compute the "error" signal: which parts of f contribute most to the peak
        # conv[k_peak] = sum_j f[j] * f[k_peak - j]
        # We want to reduce this. The "gradient" w.r.t. f at the peak is:
        # d(conv[k_peak])/df[m] = f[k_peak-m] + f[m-k_peak] (where defined)

        # Apply a correction: reduce f at positions that contribute most to the peak
        # correction[m] = f[k_peak-m] (if valid) + f[m-k_peak] (if valid)
        correction = np.zeros(n)
        for m in range(n):
            idx1 = k_peak - m
            if 0 <= idx1 < n:
                correction[m] += f[idx1]
            idx2 = m - k_peak
            if 0 <= idx2 < n:
                correction[m] += f[idx2]

        # Scale correction to achieve desired reduction
        # We want to reduce the peak by a fraction
        target_reduction = 0.05  # reduce peak by 5%
        step = min(target_reduction * peak_val / max(1e-15, correction.sum()), 0.1)

        f_new = f - step * correction
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            f = f.copy()
            continue
        f = f_new / sn

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(np.fft.irfft(spectrum, fft_size))[:n]
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig

            np.exp(-3.0 * (np.arange(n_best) - 0.25 * n_best) ** 2 / (0.15 * n_best) ** 2) +
            np.exp(-3.0 * (np.arange(n_best) - 0.75 * n_best) ** 2 / (0.15 * n_best) ** 2))),
    ]

    for name, f_init in restart_configs:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.12, 80)
        if t_alloc < 10:
            continue
        print(f"  Restart [{name}], t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent_optimize(f_init, time_limit=t_alloc,
                                             t_param=40.0, eta0=0.005)
        try_seq(a_opt, f"restart_{name}")

    # === Phase 8: Final intensive coordinate descent ===
    print(f"\n=== Phase 8: Final CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_final_cd = min(rem * 0.4, 200)
            print(f"  Final CD budget: {t_final_cd:.0f}s")
            multipliers_final = [0.6, 0.7, 0.8, 0.88, 0.93, 0.97,
                                 1.03, 1.07, 1.12, 1.2, 1.35, 1.5]
            a_opt, obj = coordinate_descent_refine(best_seq, time_limit=t_final_cd,
                                                   multipliers=multipliers_final)
            try_seq(a_opt, "final_cd")

            # Ultra-fine CD
            rem = deadline - time.monotonic()
            if rem > 20:
                t_ufine = min(rem * 0.3, 100)
                multipliers_ufine = [0.92, 0.95, 0.97, 0.985, 1.015, 1.03, 1.05, 1.08]
                a_opt2, obj2 = coordinate_descent_refine(a_opt, time_limit=t_ufine,
                                                         multipliers=multipliers_ufine)
                try_seq(a_opt2, "final_cd_ultrafine")

    # === Phase 9: One more round of mirror descent on best ===
    print(f"\n=== Phase 9: Final Mirror Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_final_md = min(rem * 0.35, 100)
            a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t_final_md,
                                                 t_param=100.0, eta0=0.003)
            try_seq(a_opt, "final_md")

    # === Fallback ===
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Something went wrong, use parent
        best_seq = parent
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
