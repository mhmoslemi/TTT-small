- The gradient of the smoothed objective is a weighted sum of the individual lag gradients, with weights proportional to exp(beta * (a*a)[k]).

- It automatically satisfies a >= 0 and sum(a) = 1.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed (log-sum-exp) 
gradient optimization with softmax parameterization, coordinate descent 
polishing, and multi-initialization search.
"""

import numpy as np
import time
from scipy.optimize import minimize

def compute_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_obj_grad(x, beta, n):
    """
    Compute smoothed objective and gradient.
    x: parameter vector (length n)
    beta: smoothing parameter
    n: sequence length
    """
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    a = e / e.sum()
    
    conv = compute_conv(a)
    conv_len = len(conv)
    
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    exp_shifted = np.exp(beta_conv - max_bc)
    w_sum = exp_shifted.sum()
    w_norm = exp_shifted / w_sum
    
    smoothed_max = (max_bc + np.log(w_sum)) / beta
    obj = 2.0 * n * smoothed_max
    
    # Gradient of smoothed_max w.r.t. a[j] = 2 * sum_m a[m] * w_norm[j+m]
    # Using cross-correlation:
    grad_sm = 2.0 * np.correlate(w_norm, a, mode='valid')
    grad_a = 2.0 * n * grad_sm
    
    # Softmax gradient
    grad_x = grad_a * a - a * np.dot(grad_a, a)
    
    return obj, grad_x

def coordinate_descent_polish(a0, time_limit, n_max_bisections=50):
    """Exact coordinate descent with bisection."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = compute_conv(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            x_old = a[j]
            if x_old < 1e-15:
                # Try increasing from 0
                x_new = 1e-8
            else:
                s_others = 1.0 - x_old
                if s_others < 1e-12:
                    continue
                
                b = a.copy()
                b[j] = 0.0
                b = b / s_others
                B = compute_conv(b)
                
                b_valid = np.zeros(2*n-1)
                start = j
                end = min(j + n, 2*n-1)
                b_valid[start:end] = b[:end-start]
                
                # Bisection
                lo, hi = 0.0, 1.0
                for _ in range(n_max_bisections):
                    mid = (lo + hi) / 2.0
                    om = 1.0 - mid
                    f_vals = om*om*B + 2.0*mid*om*b_valid
                    if 2*j < 2*n-1:
                        f_vals[2*j] += mid*mid
                    k_max = int(np.argmax(f_vals))
                    b_kj = b[k_max - j] if 0 <= k_max - j < n else 0.0
                    B_k = B[k_max]
                    ind = 1.0 if (2*j < 2*n-1 and k_max == 2*j) else 0.0
                    deriv = 2.0*(b_kj - B_k + mid*(B_k - 2.0*b_kj + ind))
                    if deriv > 1e-13:
                        hi = mid
                    elif deriv < -1e-13:
                        lo = mid
                    else:
                        break
                x_new = (lo + hi) / 2.0
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            s_others = a[mask].sum()
            if s_others > 1e-15:
                a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = compute_conv(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

    Optimize using log-sum-exp smoothing with L-BFGS.
    a0: initial point (sum 1, non-negative)
    beta_schedule: list of beta values (increasing)
    time_per_beta: time to spend at each beta

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_conv_max = compute_conv(a).max()
    
    for beta in beta_schedule:
        # Convert to x-space
        a_safe = np.maximum(a, 1e-30)
        x0 = np.log(a_safe)
        
        t0 = time.monotonic()
        result = minimize(
            lambda x: softmax_obj_grad(x, beta, n),
            x0,
            jac=True,
            method='L-BFGS-B',
            options={'maxiter': 200, 'gtol': 1e-8, 'maxls': 50}
        )
        
        # Extract solution
        x_opt = result.x
        a_opt = np.exp(x_opt - x_opt.max())
```

    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_f_and_grad(x, beta, n):
    """
    Compute smoothed objective and gradient for softmax-parameterized sequence.
    Returns (objective, gradient) for scipy.optimize.minimize with jac=True.
    """
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    a = e / e.sum()

    conv = compute_conv(a)
    conv_len = len(conv)

    beta_conv = beta * conv
    max_bc = beta_conv.max()
    exp_shifted = np.exp(beta_conv - max_bc)
    w_sum = exp_shifted.sum()
    if w_sum < 1e-30:
        w_sum = 1e-30
    w_norm = exp_shifted / w_sum

    smoothed_max = (max_bc + np.log(w_sum)) / beta
    obj = 2.0 * n * smoothed_max

    # Gradient of smoothed_max w.r.t. a[j] = 2 * sum_m a[m] * w_norm[j+m]
    # Use cross-correlation for efficiency
    grad_sm = 2.0 * np.correlate(w_norm, a, mode='valid')
    grad_a = 2.0 * n * grad_sm

    # Softbox gradient: d obj / dx[j] = grad_a[j]*a[j] - a[j]*sum_i(grad_a[i]*a[i])
    dot_ga_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga_a)

    return obj, grad_x


def smooth_optimize(a0, beta_list, time_per_beta, verbose=False):
    """
    Successive smoothed optimization: for each beta in beta_list, run L-BFGS
    on the log-sum-exp smoothed objective using softmax parameterization.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = compute_conv(a).max()

    for beta in beta_list:
        if beta <= 0:
            continue
        # Convert to log-space (with floor to avoid log(0))
        a_safe = np.maximum(a, 1e-25)
        x0 = np.log(a_safe)

        t0 = time.monotonic()
        try:
            result = minimize(
                softmax_f_and_grad,
                x0,
                args=(beta, n),
                jac=True,
                method='L-BFGS-B',
                options={'maxiter': 300, 'gtol': 1e-10, 'maxls': 50,
                         'maxfun': 10000}
            )
            x_opt = result.x
            a_new = np.exp(x_opt - x_opt.max())
            a_new = a_new / a_new.sum()
        except Exception:
            a_new = a.copy()

        # Evaluate true objective
        conv_new = compute_conv(a_new)
        max_new = conv_new.max()

        if max_new < best_max - 1e-15:
            best_max = max_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    beta={beta:.0f}: max_conv={best_max:.10f} (t={elapsed:.1f}s)")

    return best_a, best_max


def cd_polish_grid(a0, time_limit, n_points=300):
    """
    Coordinate descent with grid search per coordinate.
    More robust than bisection: evaluates objective at many x values.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    conv = compute_conv(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    # Grid for x (the value of coordinate j)
    x_grid = np.linspace(0.0, 1.0, n_points)

    passed = 0
    while time.monotonic() - t0 < time_limit:
        passed += 1
        improved = False
        max_improve = 0.0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            # Normalize others
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = compute_conv(b)

            # b_valid[k] = b[k-j] if 0<=k-j<n else 0
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            # Indicators for 2j == k
            ind_2j = np.zeros(conv_len)
            if 2 * j < conv_len:
                ind_2j[2 * j] = 1.0

            # Coefficients: f_k(x) = A*x^2 + Bc*x + C
            A_coef = B - 2.0 * b_valid + ind_2j
            Bc_coef = 2.0 * b_valid - 2.0 * B
            C_coef = B

            # Evaluate at all grid points vectorized
            xg = x_grid
            vals = (xg[:, None] ** 2) * A_coef[None, :] + \
                   xg[:, None] * Bc_coef[None, :] + C_coef[None, :]
            max_vals = vals.max(axis=1)
            best_idx = int(np.argmin(max_vals))
            x_new = x_grid[best_idx]

            if abs(x_new - x_old) < 1e-12:
                continue

            # Apply update
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            s_others_new = a[mask].sum()
            if s_others_new > 1e-15:
                a[mask] *= (1.0 - x_new) / s_others_new
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = compute_conv(a)
            obj = conv.max()
            improvement = best_obj - obj
            if improvement > max_improve:
                max_improve = improvement

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                improved = True

        if not improved:
            # Try a small perturbation to escape
            if passed > 1:
                rng = np.random.default_rng(passed * 31 + 7)
                noise = 0.002 / (0.5 + passed * 0.1)
                a = project_simplex(best_a + rng.standard_normal(n) * noise)
            else:
                break

    return best_a, best_obj


def paired_swap(a0, time_limit, n_trials=50000):
    """Random pair-swap local search."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = compute_conv(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for _ in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        delta = rng.uniform(0.01, 0.4) * min(a[i], a[j])

        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-12:
            continue
        a_trial /= st

        obj = compute_conv(a_trial).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_obj
