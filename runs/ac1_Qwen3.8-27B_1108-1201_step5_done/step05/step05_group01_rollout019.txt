from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def lse_objective_and_grad(z, n, tau):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. z."""
    a = softmax(z)
    b = fast_convolve(a)  # length 2n-1
    
    # p = softmax(tau * b)
    tb = tau * b
    p = np.exp(tb - tb.max())
    p /= p.sum()
    
    # scalar = sum(p * b)
    scalar = np.dot(p, b)
    
    # c[j] = sum_k p[k] * a[k-j] for j in [0, n-1]
    # This is cross-correlation of p and a
    # c = ifft(rfft(p, N) * conjugate(rfft(a, N)))[:n]
    N = 1 << (3 * n - 2 - 1).bit_length()  # need N >= 3n-2
    P = rfft(p, N)
    A = rfft(a, N)
    c_full = irfft(P * np.conj(A), N)
    c = c_full[:n]
    
    # gradient = 4n * a * (c - scalar)
    grad = 4.0 * n * a * (c - scalar)
    
    # objective = 2n * scalar (since with log-sum-exp, the smoothed max ≈ scalar for large tau)
    # Actually, the true smoothed objective is (2n/tau) * log(sum(exp(tau*b)))
    # = (2n/tau) * (tau*max(b) + log(sum(exp(tau*(b - max(b))))) + ...)
    # ≈ 2n * max(b) + (2n/tau) * log(mass near max)
    # For the gradient, we computed d/dz of (1/tau)*log(sum(exp(tau*b)))
    # which equals sum_k p_k * b_k = scalar (the p-weighted average)
    # So the smoothed objective value is (2n/tau) * (tau*max(b) + log(sum(exp(tau*(b-max(b))))))
    # But for optimization, we can just use 2n * scalar as a proxy (monotone in the true objective)
    
    obj = 2.0 * n * scalar
    return obj, grad

    a = np.maximum(a_init, 1e-15)

    n = len(a)
    
    z = np.log(a)  # log parameterization
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max() * 2 * n
    t0 = time.monotonic()
    
    for step in range(n_steps):

            break
        
        # Anneal tau
        frac = step / max(1, n_steps - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        # Actually let's use a more gradual schedule
        tau = tau_start * (1 + frac * 10) ** 2  # quadratic growth
        
        # Run L-BFGS

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed L-BFGS
optimization in log-space, coordinate descent polishing, and multi-length
exploration. Uses the key insight that optimal sequences are non-palindromic,
exploiting asymmetry to reduce the convolution peak.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_obj_grad(z, n, tau):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. log-parameter z.
    Objective: (2n/tau) * log(sum_k exp(tau * b_k)) where b = conv(softmax(z), softmax(z))
    Returns (objective_value, gradient).
    """
    a = softmax_fn(z)
    b = fast_convolve(a)  # length 2n-1

    # p = softmax(tau * b)
    tb = tau * b
    p = np.exp(tb - tb.max())
    p /= p.sum()

    # s = sum(p * b) = p-weighted average of b
    s_val = np.dot(p, b)

    # c_j = sum_k p[k] * a[k-j] = cross-correlation of p and a, first n entries
    N = 1 << (3 * n - 1).bit_length()
    P = rfft(p, N)
    A = rfft(a, N)
    c_full = irfft(P * np.conj(A), N)
    c = c_full[:n]

    # Gradient: 4n * a * (c - s)
    grad = 4.0 * n * a * (c - s_val)

    # Smoothed objective value (for monitoring)
    obj = 2.0 * n * s_val

    return obj, grad


def lse_optimize(a_init, time_limit, tau_start=5.0, tau_end=2000.0, n_phases=30):
    """
    Run L-BFGS with tau annealing on log-parameterization.
    Returns (best_a, best_max_b_normalized) where best_max_b = max(conv(a,a)) with sum(a)=1.
    """
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)

    best_a = a.copy()
    best_max_b = fast_convolve(a).max()
    t0 = time.monotonic()

    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit - 10:
            break

        # Exponential tau schedule
        frac = phase / max(1, n_phases - 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        # Determine max iterations based on remaining time
        remaining = time_limit - (time.monotonic() - t0)
        max_iter = min(80, max(10, int(remaining * 3)))

        try:
            res = minimize(
                lse_obj_grad, z.copy(),
                args=(n, tau),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10, 'maxls': 50}
            )
            z = res.x
        except Exception:
            pass

        a_new = softmax_fn(z)
        b_new = fast_convolve(a_new)
        max_b_new = b_new.max()

        if max_b_new < best_max_b:
            best_max_b = max_b_new
            best_a = a_new.copy()

        if phase % 5 == 0:
            print(f"    LSE phase {phase}/{n_phases}: tau={tau:.1f}, max_b={max_b_new:.8f}, best={best_max_b:.8f}")

    return best_a, best_max_b


def coord_descent_polish(a_init, time_limit, n_rounds=50):
    """
    Golden-section coordinate descent on the original (non-smooth) objective.
    For each coordinate j, optimize a[j] with others proportionally scaled.
    """
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_max_b = fast_convolve(a).max()
    t0 = time.monotonic()

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Binary search for the interval containing the minimum
            # Try several candidate ranges
            candidates = [
                (0.0, min(2.0 * old_val, 0.999)),
                (max(0.0, 0.5 * old_val), min(1.5 * old_val, 0.999)),
                (0.0, 0.999),
            ]

            best_x = old_val
            best_v = fast_convolve(a).max()

            for lo, hi in candidates:
                if hi - lo < 1e-10:
                    continue
                # Evaluate a few points to narrow down
                mids = [lo + inv_gr * (hi - lo), lo + (1.0 - inv_gr) * (hi - lo), (lo + hi) / 2.0]
                for x in mids:
                    if x < 0 or x > 0.9999:
                        continue
                    ft = a.copy()
                    ft[j] = x
                    scale = (1.0 - x) / denom
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    ft[mask] *= scale
                    v = fast_convolve(ft).max()
                    if v < best_v:
                        best_v = v
                        best_x = x

            if best_x != old_val and best_v < fast_convolve(a).max() - 1e-14:
                a[j] = best_x
                scale = (1.0 - best_x) / denom
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()  # renormalize for numerical stability
                improved = True

                cur_max = fast_convolve(a).max()
                if cur_max < best_max_b:
                    best_max_b = cur_max
                    best_a = a.copy()

        if not improved:
            break

        if rnd % 5 == 0:
            print(f"    CD round {rnd}: max_b={best_max_b:.8f}")

    return best_a, best_max_b


def make_power_init(n, alpha):
    """a[j] = (j+1)^alpha, normalized."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** alpha
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def make_exp_init(n, beta):
    """a[j] = exp(beta*j), normalized."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(beta * j)
    s = a.sum()
    a /= s
    return a


def make_linear_init(n, gamma):
    """a[j] = 1 + gamma*(2j/(n-1) - 1), normalized."""
    if n == 1:
        return np.array([1.0])
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + gamma * (2.0 * j / (n - 1) - 1.0)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_sine_init(n, freq, amp, phase):
    """a[j] = 1 + amp*sin(2*pi*freq*j/n + phase), normalized."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * j / n + phase)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def score_sequence(a):
    """Compute the evaluation score for a sequence (sum=1 assumed)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a = a / s
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new
