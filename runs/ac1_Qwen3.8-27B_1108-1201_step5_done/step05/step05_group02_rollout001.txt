import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_con)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            return None, None
        g = g / gs
        return g, res.x[-1]
    return None, None


def solve_lp_full(f, n):
    """Solve full LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            return None, None
        g = g / gs
        return g, res.x[-1]
    return None, None


def golden_section_line_search(f, g, M, n):
    """Golden section search for optimal alpha in [0,1] for f + alpha*(g-f)."""
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618

    def obj(alpha):
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            return M + 1.0
        ft = ft / st
        return fast_convolve(ft).max()

    # First check endpoints
    val_0 = obj(0.0)
    val_1 = obj(1.0)
    if val_1 < val_0:
        return 1.0, val_1

    a, b = 0.0, 1.0
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = obj(c)
    fd = obj(d)

    for _ in range(40):
        if b - a < 1e-8:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = obj(d)

    best_alpha = (a + b) / 2
    best_val = obj(best_alpha)
    return best_alpha, best_val


def lp_fw_optimize(f0, time_limit, verbose=False, use_reduced=True, reduced_k=60):
    """LP-based Frank-Wolfe with golden-section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Choose LP type: use reduced for speed, full occasionally
        if use_reduced and it < 200:
            g, lp_t = solve_lp_reduced(f, n, top_k=reduced_k)
        else:
            g, lp_t = solve_lp_full(f, n)

        if g is None:
            g = np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Golden section line search
        best_alpha, best_val = golden_section_line_search(f, g, M, n)

        if best_alpha > 1e-10 and best_val < best_obj - 1e-14:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f} (t={elapsed:.0f}s)")
        else:
            no_improve += 1

        if no_improve > 40:
            # KKT-guided restart: perturb coordinates that contribute to the peak
            conv = fast_convolve(f)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            # Perturb coordinates in the peak region more
            rng = np.random.default_rng(it * 31337 + 7)
            noise = rng.standard_normal(n) * 0.002
            noise[js:je+1] *= 2.0  # Double perturbation at peak-contributing coords
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def peak_focus_sg(f0, time_limit, lr=0.002, momentum=0.92, verbose=False):
    """Subgradient descent focusing on the single worst peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Focus on the single worst peak
        peak_k = int(np.argmax(conv))
        # Compute subgradient for this peak: 2*f[peak_k - j] for valid j
        sg = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        kg = peak_k - idx
        mask = (kg >= 0) & (kg < n)
        sg[idx[mask]] = 2.0 * f[kg[mask]]

        # Also include a small contribution from the 2nd peak for diversity
        conv_sorted = np.sort(conv)[::-1]
        if len(conv_sorted) >= 2 and conv_sorted[1] > 0.95 * conv_sorted[0]:
            peak_k2 = int(np.argmax(np.where(conv == conv_sorted[0], 0, conv)))
            js2 = max(0, peak_k2 - n + 1)
            je2 = min(n - 1, peak_k2)
            idx2 = np.arange(js2, je2 + 1)
            kg2 = peak_k2 - idx2
            mask2 = (kg2 >= 0) & (kg2 < n)
            sg[idx2[mask2]] += 0.3 * f[kg2[mask2]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 1000 == 0:
                print(f"    PGF iter {it}: obj={best_obj:.10f} (lr={cur_lr:.6f})")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.99992
        if it % 5000 == 0:
            cur_lr = max(lr * 0.5, 0.0003)

        if no_improve > 300:
            rng = np.random.default_rng(it * 104729)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0005)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Exact 1D coordinate descent: for each coordinate, find optimal value via binary search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        elapsed = time.monotonic() - t0

        # Determine coordinate order by contribution to the peak
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        # Process coordinates in the peak region first, then others
        peak_coords = list(range(js, je + 1))
        other_coords = [j for j in range(n) if j not in set(peak_coords)]
        # Shuffle others for diversity
        rng = np.random.default_rng(passes * 7919)
        rng.shuffle(other_coords)
        coord_order = peak_coords + other_coords

        for j in coord_order:
            if elapsed > time_limit - 5:
                break
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            # Binary search for optimal value of f[j]
            # When f[j] changes to new_val, all other coords scale by (1-new_val)/(1-old_val)
            best_j_val = old_val
            best_j_obj = best_obj

            lo = 0.0
            hi = min(old_val * 3.0, 0.5)
            if lo >= hi:
                continue

            # Check a few specific points first
            candidates = [old_val * 0.3, old_val * 0.5, old_val * 0.7,
                          old_val * 1.3, old_val * 1.7, old_val * 2.5]
            for cv in candidates:
                if cv < 1e-14 or cv > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    break
                scale = (1.0 - cv) / denom
                f_trial = f * scale
                f_trial[j] = cv
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = cv

            # Binary search refinement
            if best_j_val != old_val:
                # Determine search interval around best_j_val
                if best_j_val > old_val:
                    lo = old_val
                    hi = min(best_j_val * 1.5, 0.5)
                else:
                    lo = 0.0
                    hi = best_j_val * 1.5
                    if hi > old_val:
                        hi = old_val

                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        break
                    scale = (1.0 - mid) / denom
                    f_trial = f * scale
                    f_trial[j] = mid
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = mid
                        if mid > old_val:
                            lo = mid
                        else:
                            hi = mid
                    else:
                        if mid > old_val:
                            hi = mid
                        else:
                            lo = mid

            if best_j_val != old_val and best_j_obj < best_obj - 1e-14:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f = f * scale
                    f[j] = best_j_val
                    f = f / f.sum()
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
                    conv = fast_convolve(f)

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 5:
                break
        else:
            no_improve_passes = 0

        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f} (t={elapsed:.0f}s)")

    return best_f, best_obj


def multi_peak_lp_fw(f0, time_limit, verbose=False):
    """LP-FW that alternates between min-peak LP and max-sum LP for diversity."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Alternate between formulations
        if it % 3 == 0:
            # Max-sum LP: max sum(g) s.t. (f*g)[k] <= M*1.01
            c = -np.ones(n)
            n_con = 2 * n - 1
            A_ub = np.zeros((n_con, n))
            for j in range(n):
                A_ub[j:j + n, j] = f
            b_ub = np.full(n_con, M * 1.01)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                g = np.maximum(res.x, 0)
                gs = g.sum()
                if gs < 1e-15:
                    g = np.ones(n) / n
                else:
                    g = g / gs
            else:
                g = solve_lp_reduced(f, n, top_k=40)
                if g is None:
                    g = np.ones(n) / n
        else:
            g, _ = solve_lp_reduced(f, n, top_k=50)
            if g is None:
                g = np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        best_alpha, best_val = golden_section_line_search(f, g, M, n)

        if best_alpha > 1e-10 and best_val < best_obj - 1e-14:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 30 == 0:
                print(f"    MPFW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 55501 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization with reduced-constraint LP-FW, peak-focus subgradient,
    exact coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    base = best_seq.copy()

    # === Phase 1: Quick LP-FW with reduced constraints ===
    print(f"\n=== Phase 1: Quick LP-FW (reduced) (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(120, rem * 0.1)
    if t1 > 20:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True, use_reduced=True, reduced_k=40)
        try_seq(a_opt, "lpfw_quick")

    # === Phase 2: Full LP-FW ===
    print(f"\n=== Phase 2: Full LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(300, rem * 0.25)
    if t2 > 30 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True, use_reduced=False)
        try_seq(a_opt, "lpfw_full")

    # === Phase 3: Multi-peak LP-FW ===
    print(f"\n=== Phase 3: Multi-peak LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(200, rem * 0.18)
    if t3 > 30 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t3:.0f}s")
        a_opt, _ = multi_peak_lp_fw(best_seq, time_limit=t3, verbose=True)
        try_seq(a_opt, "mpfw")

    # === Phase 4: Peak-focus subgradient ===
    print(f"\n=== Phase 4: Peak-focus SG (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t4 = min(250, rem * 0.22)
        if t4 > 30:
            print(f"  n={n_best}, t={t4:.0f}s")
            a_opt, _ = peak_focus_sg(best_seq, time_limit=t4, lr=0.0015, momentum=0.93, verbose=True)
            try_seq(a_opt, "peak_sg")

            # Second run with different params
            rem = deadline - time.monotonic()
            t4b = min(100, rem * 0.1)
            if t4b > 20:
                a_opt, _ = peak_focus_sg(best_seq, time_limit=t4b, lr=0.0005, momentum=0.95, verbose=True)
                try_seq(a_opt, "peak_sg_fine")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(200, rem * 0.2)
        if t5 > 25:
            print(f"  n={len(best_seq)}, t={t5:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "coord_cd")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        candidate_lengths = []
        for factor in [0.85, 0.9, 1.1, 1.15, 1.25]:
            ln = int(base_n * factor)
            if 300 <= ln <= 3000 and ln != base_n:
                candidate_lengths.append(ln)
        # Also try some "nice" lengths
        for ln in [768, 896, 1024, 1152, 1280]:
            if ln not in candidate_lengths and 300 <= ln <= 3000:
                candidate_lengths.append(ln)
        candidate_lengths = sorted(set(candidate_lengths))

        for target_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.35, 150)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, use_reduced=True, reduced_k=50)
            try_seq(a_opt, f"len_{target_n}")

            # Quick subgradient polish
            rem = deadline - time.monotonic()
            if rem > 40:
                a_opt2, _ = peak_focus_sg(a_opt, time_limit=min(rem * 0.4, 60),
                                           lr=0.001, momentum=0.94)
                try_seq(a_opt2, f"len_{target_n}_sg")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 80),
                                       use_reduced=True, reduced_k=30)
            try_seq(a_opt, "lpfw_polish")

        # Quick coordinate descent
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "cd_polish")

        # Perturbation-based exploration
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            scale = 0.0005 + 0.0005 * p
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = peak_focus_sg(f_pert, time_limit=min(rem, 30),
                                      lr=0.0004, momentum=0.95)
            try_seq(a_opt, f"pert_{p}")

        # Final coordinate descent on best
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
