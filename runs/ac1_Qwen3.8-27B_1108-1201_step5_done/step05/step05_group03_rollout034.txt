import numpy as np
import time

def auto_conv(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective_value(a):
    """Compute 2*n*max_b/S^2."""
    S = a.sum()
    if S < 1e-15:
        return np.inf
    C = auto_conv(a)
    return 2 * len(a) * C.max() / (S * S)

def subgradient_descend(a0, time_limit, verbose=False):
    """Projected subgradient descent with momentum and restarting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / S
    n = len(a)
    
    C = auto_conv(a)
    best_obj = 2 * n * C.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)  # momentum
    beta = 0.9
    lr = 0.05
    lr_decay = 0.9995
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        C = auto_conv(a)
        M = C.max()
        tol = 0.01 * M  # lags within 1% of max are considered "active"
        active = np.where(C >= M - tol)[0]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            sg[idx] += 2.0 * a[k - idx]
        sg /= len(active)
        
        # Momentum update
        m = beta * m + (1 - beta) * sg
        a_new = a - lr * m
        a_new = project_simplex(a_new)

        C_new = auto_conv(a_new)
        obj_new = 2 * n * C_new.max()

            best_a = a_new.copy()
            no_improve = 0

            j = k_star - i
            if j < 0 or j >= n or j == i:

            c = a[i] + a[j]
            if c < 1e-12:

            if a[i] >= a[j]:

            else:

            best_M = M
            
            for t in candidates:
                a_trial = a.copy()
                a_trial[i] = t * c
                a_trial[j] = (1 - t) * c
                C_trial = auto_conv(a_trial)
                M_trial = C_trial.max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_t = t
            
            if best_M < M - 1e-14:

                a[j] = (1 - best_t) * c
                improved = True

                    best_obj = best_M
                    best_a = a.copy()

```python
"""
Minimizes 2*n*max(auto_conv(a))/(sum(a))^2 using:
1. Log-sum-exp smoothed projected gradient (Adam) on the simplex
2. Sharp subgradient descent with momentum on the true active set
3. Targeted peak-shaving coordinate descent on the dominant lag
4. Multi-length exploration with interpolation-based restarts
5. Final coordinate polish
"""

import numpy as np
import time


def auto_conv(a):
    """Fast linear auto-convolution of a real sequence via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to sum 1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_active_subgrad(a, C, rel_tol=0.02):
    """Compute subgradient at the active (near-maximal) lags."""
    n = len(a)
    M = C.max()
    if M < 1e-15:
        return np.zeros(n)
    active = np.where(C >= M * (1 - rel_tol))[0]
    sg = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        sg[idx] += 2.0 * a[k - idx]
    return sg / len(active)


def logsumexp_optimize(a0, time_limit, seed=42, verbose=False):
    """Optimize log-sum-exp smoothed auto-conv max using Adam on the simplex."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    C = auto_conv(a)
    best_obj = C.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.008
    T_init = 8.0 / n
    T_final = 0.05 / n
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        T = T_final + (T_init - T_final) * (1.0 - frac) ** 2
        lr = base_lr * (1.0 - 0.4 * frac)
        
        C = auto_conv(a)
        M = C.max()
        
        # Compute softmax weights over the convolution
        scaled = (C - M) / T
        scaled = np.clip(scaled, -50.0, 0.0)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros_like(C)
            w[np.argmax(C)] = 1.0
        else:
            w = w / ws
        
        # Compute gradient: only process significant weights
        sg = np.zeros(n)
        significant = w > 1e-6
        for k in np.where(significant)[0]:
            lo = max(0, int(k) - n + 1)
            hi = min(n - 1, int(k))
            idx = np.arange(lo, hi + 1)
            sg[idx] += 2.0 * w[int(k)] * a[k - idx]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        it1 = it + 1
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        a_new = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))
        
        C_new = auto_conv(a_new)
        obj_new = C_new.max()
        a = a_new
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 800:
            rng = np.random.default_rng(seed + it)
            a = normalize(best_a + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj


def subgrad_optimize(a0, time_limit, seed=42, verbose=False):
    """Projected subgradient descent with momentum on the true active set."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    C = auto_conv(a)
    best_obj = C.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    beta = 0.85
    lr = 0.03
    lr_decay = 0.9997
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        C = auto_conv(a)
        M = C.max()
        
        sg = compute_active_subgrad(a, C, rel_tol=0.01)
        m = beta * m + (1 - beta) * sg
        a_new = project_simplex(a - lr * m)
        
        C_new = auto_conv(a_new)
        obj_new = C_new.max()
        a = a_new
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= lr_decay
        
        if no_improve > 300:
            rng = np.random.default_rng(seed + it * 397)
            pert_scale = 0.002 + 0.002 * min(no_improve // 100, 3)
            a = normalize(best_a + rng.standard_normal(n) * pert_scale)
            m = np.zeros(n)
            lr = 0.015
            no_improve = 0
    
    return best_a, best_obj


def peak_shave_optimize(a0, time_limit, seed=42, verbose=False):
    """Targeted coordinate descent to reduce the peak of the auto-convolution."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    C = auto_conv(a)
    best_obj = C.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for round_idx in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        C = auto_conv(a)
        M = C.max()
        k_star = int(np.argmax(C))
        
        improved = False
        
        # For the dominant lag, try to unbalance pairs (i, k_star - i)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            j = k_star - int(i)
            if j < 0 or j >= n or j == i:
                continue
            
            c = a[i] + a[j]
            if c < 1e-12:
                continue
            
            current_ratio = a[i] / c
            
            # Try extreme ratios to minimize the product a[i]*a[j]
            # which contributes 2*a[i]*a[j] to C[k_star]
            if a[i] >= a[j]:
                candidates = [1.0, 0.95, 0.9, 0.8, 0.7]
            else:
                candidates = [0.0, 0.05, 0.1, 0.2, 0.3]
            
            best_t = current_ratio
            best_M = M
            
            for t in candidates:
                if abs(t - current_ratio) < 0.01:
                    continue
                a[i] = t * c
                a[j] = (1.0 - t) * c
                C_trial = auto_conv(a)
                M_trial = C_trial.max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_t = t
                else:
                    break  # no point trying further extremes
            
            # Restore if no improvement
            a[i] = best_t * c
            a[j] = (1.0 - best_t) * c
            
            if best_M < M - 1e-14:
                M = best_M
                improved = True
                if M < best_obj:
                    best_obj = M
                    best_a = a.copy()
        
        if not improved:
            # Try random pair moves
            for _ in range(10):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                j = k_star - i
                if j < 0 or j >= n:
                    j = int(rng.integers(n))
                if i == j:
                    continue
                c = a[i] + a[j]
                if c < 1e-12:
                    continue
                t = float(rng.random())
                a[i] = t * c
                a[j] = (1.0 - t) * c
                C_trial = auto_conv(a)
                M_trial = C_trial.max()
                if M_trial > M * 1.02:
                    # Revert
                    a[i] = (1.0 - t) * c
                    a[j] = t * c
                elif M_trial < best_obj:
                    best_obj = M_trial
                    best_a = a.copy()
    
    return best_a, best_obj


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a)
    return normalize(result)


def coordinate_polish(a0, time_limit, seed=42):
    """Final polish using pairwise coordinate optimization."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    C = auto_conv(a)
    best_obj = C.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Focus on top few lags
    for round_idx in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        C = auto_conv(a)
        M = C.max()
        # Get top 3 lags
        top_lags = np.argsort(C)[::-1][:3]
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            for k in top_lags:
                j = int(k) - int(i)
                if j < 0 or j >= n or j == i:
                    continue
                
                c = a[i] + a[j]
                if c < 1e-12:
                    continue
                
                current_ratio = a[i] / c
                # Try to make the pair more unbalanced
                if a[i] >= a[j]:
                    t_try = min(1.0, current_ratio + 0.1)
                else:
                    t_try = max(0.0, current_ratio - 0.1)
                
                if abs(t_try - current_ratio) < 0.005:
                    continue
                
                a[i] = t_try * c
                a[j] = (1.0 - t_try) * c
                C_trial = auto_conv(a)
                M_trial = C_trial.max()
                
                if M_trial < best_obj:
                    best_obj = M_trial
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = current_ratio * c
                    a[j] = (1.0 - current_ratio) * c
        
        if not improved:
            # Perturb and continue
            rng2 = np.random.default_rng(round_idx * 31 + 7)
            a = normalize(best_a + rng2.standard_normal(n) * 0.001)
    
    return best_a, best_obj
