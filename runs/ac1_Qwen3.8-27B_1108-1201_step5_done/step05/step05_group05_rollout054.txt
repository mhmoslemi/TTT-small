def compute_smooth_gradient(f, conv, tau, fft_size):
    """Compute gradient of smooth approximation g_tau = tau * log(sum exp(conv_k / tau))."""
    n = len(f)
    # Softmax weights
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Cross-correlation: grad_i = 2 * sum_k w[k] * f[k-i] (with bounds)
    # = 2 * (w_flipped * f)[i + len(w) - 1]
    # Where * is linear convolution
    
    # Pad for FFT
    w_len = len(conv)  # 2n-1
    conv_size = w_len + n - 1  # 3n-2
    fft_sz = 1 << (conv_size - 1).bit_length()
    
    w_flipped = w[::-1]
    
    W = np.fft.rfft(w_flipped, fft_sz)
    F = np.fft.rfft(f, fft_sz)
    conv_result = np.fft.irfft(W * F, fft_sz)
    
    # The cross-correlation at lag i is conv_result[i + w_len - 1]
    grad = 2.0 * conv_result[w_len - 1: w_len - 1 + n]
    
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient optimization
with Nesterov acceleration, tau-annealing, multi-length exploration,
frequency-domain initialization, and final coordinate descent polishing.
Key innovation: replaces LP-based Frank-Wolfe with FFT-based smooth gradient
computation (O(n log n) per iteration vs O(n^2) for LP), enabling many more
iterations in the same time budget.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, tau):
    """Compute softmax weights for smoothing the max."""
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    return w


def gradient_smooth(f, weights, grad_fft_size):
    """Compute gradient of smooth objective: 2 * cross_corr(weights, f)[:n]."""
    n = len(f)
    W = np.fft.rfft(weights, grad_fft_size)
    F = np.fft.rfft(f, grad_fft_size)
    R = np.fft.irfft(W * np.conj(F), grad_fft_size)
    return 2.0 * R[:n]


def nesterov_optimize(f0, time_limit, tau_start, tau_end, label="",
                      verbose=False, use_adaptive_alpha=True):
    """
    Nesterov accelerated gradient descent on smooth approximation of max(conv),
    with simplex projection and tau annealing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv_fft_size = 1 << (2 * n - 2).bit_length()
    grad_fft_size = 1 << (3 * n - 3).bit_length()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    k = 1
    f_prev = f.copy()
    y = f.copy()

    # Initial step size
    alpha = 1.0 / np.sqrt(n) * 2.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        # Exponential annealing of tau
        if elapsed_frac < 0.05:
            tau = tau_start
        else:
            adj_frac = (elapsed_frac - 0.05) / 0.95
            tau = tau_start * (tau_end / tau_start) ** adj_frac

        # Compute gradient at extrapolated point y
        y_conv = fast_convolve(y)
        w = softmax_weights(y_conv, tau)
        grad = gradient_smooth(y, w, grad_fft_size)

        # Take step and project
        v = y - alpha * grad
        f_new = project_simplex(v)

        # Nesterov momentum
        t_k = 1.0 / (1.0 - 1.0 / np.sqrt(1.0 + k))
        t_k1 = 1.0 / (1.0 - 1.0 / np.sqrt(1.0 + k + 1))
        beta = (t_k - 1.0) / t_k1

        f_prev = f.copy()
        f = f_new
        y = f + beta * (f - f_prev)
        k += 1

        # Evaluate actual objective
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if use_adaptive_alpha and k % 50 == 0:
            if obj < best_obj + 1e-10:
                alpha = min(alpha * 1.02, 10.0 / n)
            else:
                alpha = max(alpha * 0.95, 0.1 / n)

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(k * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            f_prev = f.copy()
            alpha = 1.0 / np.sqrt(n) * 1.5
            no_improve = 0
            k = max(k, 100)  # keep momentum high

        if verbose and k % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov[{label}] iter={k}, best={best_obj:.10f}, "
                  f"tau={tau:.6f}, alpha={alpha:.4f}, t={elapsed:.1f}s")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Multi-scale coordinate descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < -1e-12 or new_val > 1.0 - 1e-10:
                    continue
                new_val = max(0.0, new_val)
                remaining = 1.0 - new_val
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD pass at t={elapsed:.1f}s: best={best_obj:.10f}")

    return best_f, best_obj


def subgradient_polish(f0, time_limit, verbose=False):
    """Subgradient method focusing on peak positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.003
    top_k = max(20, n // 6)
    temp = 0.00005
    momentum = np.zeros(n)
    mom_coef = 0.93

    while time.monotonic() - t0 < time_limit:
        # Peak selection via softmax weighting
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * w[i] * f[k - j]

        # Momentum update
        momentum = mom_coef * momentum - lr * sg
        f_new = project_simplex(f + momentum)

        conv = fast_convolve(f_new)
        obj = conv.max()
        f = f_new

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(no_improve) * 997 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            momentum = np.zeros(n)
            lr = 0.002
            no_improve = 0

        if verbose and (no_improve == 0) and int(t0 + time.monotonic()) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG at t={elapsed:.1f}s: best={best_obj:.10f}")

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving simplex constraint."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig[:n])
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def palindromic_symmetrize(f):
    """Symmetrize a sequence to be palindromic."""
    n = len(f)
    half = (n + 1) // 2
    left = f[:half]
    if n % 2 == 0:
        return project_simplex(np.concatenate([left, left[::-1]]))
    else:
        return project_simplex(np.concatenate([left, left[-2::-1]]))

    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
