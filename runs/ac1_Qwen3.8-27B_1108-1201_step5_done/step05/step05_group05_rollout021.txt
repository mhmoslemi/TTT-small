"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent with
log-sum-exp approximation, Nesterov momentum, full-constraint LP-Frank-Wolfe,
adaptive coordinate descent, frequency-domain initialization, and multi-restart
diversification across sequence lengths.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, conv, T):
    """Compute gradient of (1/T)*log(sum(exp(T*conv))) w.r.t. a using FFT."""
    n = len(a)
    shifted = T * conv - T * conv.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum
    fft_size = 1 << (len(conv) + n - 1 - 1).bit_length()
    W_full = np.fft.fft(w, fft_size)
    A_full = np.fft.fft(a, fft_size)
    corr = np.fft.ifft(np.conj(W_full) * A_full).real[:n]
    return 2.0 * corr


def nesterov_smooth_optimize(a0, time_limit, T_init=50.0, T_max=5000.0,
                              lr_init=0.005, momentum_init=0.85,
                              verbose=False):
    """Nesterov accelerated gradient with log-sum-exp smoothing and increasing T."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    T = T_init
    lr = lr_init
    momentum = momentum_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_y = fast_convolve(y)
        grad = smooth_gradient(y, conv_y, T)

        y_new = project_simplex(a + momentum * (y - a) - lr * grad)
        a_new = project_simplex(a - ((1.0 - momentum) / (1.0 + momentum)) * (y_new - y) - lr * grad)

        conv_a = fast_convolve(a_new)
        obj = conv_a.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        y = y_new

        T = min(T * 1.015, T_max)
        if no_improve > 200:
            lr *= 0.8
            T = max(T * 0.9, T_init)
            if lr < 1e-7:
                break
        elif no_improve % 50 == 0:
            lr = lr_init / (1.0 + 0.1 * it / 100.0)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG iter {it}: obj={best_obj:.10f}, T={T:.1f}, lr={lr:.2e}, t={elapsed:.0f}s")

    return best_a, best_obj

        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            M *= 1.05
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 100:
                break
            M *= 1.03
            continue

        # Exact line search via bisection on the quadratic-in-alpha max
        lo, hi = 0.0, 1.0
        for _ in range(40):
            mid = (lo + hi) / 2.0
            f_mid = f + mid * (g - f)
            f_mid = np.maximum(f_mid, 0.0)
            f_mid = f_mid / f_mid.sum()
            obj_mid = fast_convolve(f_mid).max()
            f_q = f + 0.5 * (g - f)
            f_q = np.maximum(f_q, 0.0)
            f_q = f_q / f_q.sum()
            obj_q = fast_convolve(f_q).max()
            if obj_mid <= obj_q and mid > 0.5:
                lo = mid
            elif obj_mid <= obj_q:
                hi = mid
            else:
                if mid > 0.5:
                    hi = mid
                else:
                    lo = mid

        alpha = (lo + hi) / 2.0
        if alpha < 1e-10:
            no_improve += 1
            M *= 1.02
            continue

        f_new = f + alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
            M = M_new
            if verbose and it % 20 == 0:
                print(f"    LPFW iter {it}: obj={best_obj:.10f}, M={M:.10f}")
        else:
            no_improve += 1
            M = max(M_new, M * 1.01)

        f = f_new
        conv = conv_new

        if no_improve > 150:
            break

    return best_f, best_obj


def adaptive_cd(f0, time_limit, verbose=False):
    """Adaptive coordinate descent with continuous multiplier search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Randomize coordinate order
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            # Try a range of multipliers (log-spaced)
            best_j_val = old_val
            best_j_obj = best_obj
            multipliers = np.logspace(-1.5, 1.5, 25)

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 1.0 - 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD pass: obj={best_obj:.10f}, t={elapsed:.0f}s")

    return best_f, best_obj


def frequency_init(n, seed=0, n_modes=3):
    """Initialize using sum of a few cosines with random phases (low peak structure)."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    sig = np.ones(n)
    for i in range(1, n_modes + 1):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        sig += 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    sig = np.maximum(sig, 0.0)
    ss = sig.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return sig / ss


def golay_like_init(n, seed=0):
    """Initialize using a decimated Golay-like sequence structure."""
    rng = np.random.default_rng(seed)
    # Start with a random ±1 sequence, take absolute value, smooth
    base = rng.choice([-1, 1], size=n)
    # Smooth with a small kernel
    kernel = np.ones(5) / 5.0
    smoothed = np.convolve(base.astype(float), kernel, mode='same')
    sig = np.maximum(smoothed, 0.0)
    ss = sig.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return sig / ss


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining smooth gradient descent (log-sum-exp),
    full-constraint LP-Frank-Wolfe, adaptive coordinate descent, and diverse
    restarts across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = frequency_init(768, seed=123, n_modes=5)
        print("Generating parent from frequency init, n=768")

    n_parent = len(parent)

    # === Phase 1: Smooth gradient descent on parent ===
    print("\n=== Phase 1: Nesterov Smooth Gradient on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.25)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, obj = nesterov_smooth_optimize(parent, time_limit=t1,
                                               T_init=30.0, T_max=8000.0,
                                               lr_init=0.008, momentum_init=0.82,
                                               verbose=True)
        try_seq(a_opt, "nsg_parent")

    # === Phase 2: LP-FW with full constraints ===
    print(f"\n=== Phase 2: LP-FW Full Constraints ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(250, rem * 0.18)
        if t2 > 30:
            print(f"  Time budget: {t2:.0f}s")
            a_opt, obj = lp_fw_full(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "lpfw_full")

    # === Phase 3: Adaptive coordinate descent ===
    print(f"\n=== Phase 3: Adaptive Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(200, rem * 0.15)
        if t3 > 20:
            print(f"  Time budget: {t3:.0f}s")
            a_opt, obj = adaptive_cd(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 4: Multiple diverse restarts ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = []

        # Small perturbation
        restarts.append(("perturb_small", project_simplex(
            best_seq + rng.standard_normal(n_best) * 0.002)))
        # Medium perturbation
        restarts.append(("perturb_med", project_simplex(
            best_seq + rng.standard_normal(n_best) * 0.008)))
        # Frequency blend
        fs = frequency_init(n_best, seed=99, n_modes=7)
        restarts.append(("freq_blend", project_simplex(0.7 * best_seq + 0.3 * fs)))
        # Golay-like blend
        gl = golay_like_init(n_best, seed=77)
        restarts.append(("golay_blend", project_simplex(0.6 * best_seq + 0.4 * gl)))
        # Reversal blend
        restarts.append(("rev_blend", project_simplex(0.5 * best_seq + 0.5 * best_seq[::-1])))
        # Anti-peak: reduce values at high-convolution positions
        conv_best = fast_convolve(best_seq)
        peak_positions = np.argsort(conv_best)[::-1][:n_best // 4]
        penalty = np.zeros(n_best)
        for k in peak_positions:
            js = max(0, k - n_best + 1)
            je = min(n_best - 1, k)
            for j in range(js, je + 1):
                penalty[j] += best_seq[k - j]
        penalty /= penalty.max() + 1e-15
        bladed = np.maximum(best_seq * (1.0 - 0.3 * penalty), 0)
        restarts.append(("anti_peak", project_simplex(bladed)))

        for i, (name, f_init) in enumerate(restarts):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem / max(1, len(restarts) - i), 120)
            print(f"  Restart [{name}], t={t_alloc:.0f}s")

            # Use smooth gradient first, then CD
            a_opt, _ = nesterov_smooth_optimize(f_init, time_limit=t_alloc * 0.6,
                                                 T_init=50.0, T_max=3000.0,
                                                 lr_init=0.004, momentum_init=0.85)
            try_seq(a_opt, f"restart_{name}_nsg")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_cd = min(rem * 0.4, 40)
                a_opt2, _ = adaptive_cd(a_opt, time_limit=t_cd)
                try_seq(a_opt2, f"restart_{name}_cd")

    # === Phase 5: Try different sequence lengths ===
    print(f"\n=== Phase 5: Different Lengths ===")
    base = best_seq if best_seq is not None else parent
    for target_n in [768, 760, 770, 752]:
        if target_n == n_parent:
            continue
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)
        # Also try fresh frequency init at this length
        f_fs = frequency_init(target_n, seed=target_n, n_modes=5)
        f_mix = project_simplex(0.6 * f_interp + 0.4 * f_fs)

        t_alloc = min(rem * 0.2, 120)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smooth_optimize(f_mix, time_limit=t_alloc * 0.7,
                                             T_init=40.0, T_max=5000.0,
                                             lr_init=0.005, momentum_init=0.84)
        try_seq(a_opt, f"len_{target_n}")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.3, 40)
            a_opt2, _ = adaptive_cd(a_opt, time_limit=t_cd)
            try_seq(a_opt2, f"len_{target_n}_cd")

    # === Phase 6: Final deep optimization on best ===
    print(f"\n=== Phase 6: Final Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            # Very long smooth gradient with high T
            t_sg = min(rem * 0.35, 200)
            print(f"  Deep NSG, t={t_sg:.0f}s")
            a_opt, _ = nesterov_smooth_optimize(best_seq, time_limit=t_sg,
                                                 T_init=200.0, T_max=20000.0,
                                                 lr_init=0.002, momentum_init=0.9)
            try_seq(a_opt, "deep_nsg")

            rem = deadline - time.monotonic()
            if rem > 25:
                # LP-FW
                t_lp = min(rem * 0.3, 120)
                print(f"  Deep LPFW, t={t_lp:.0f}s")
                a_opt2, _ = lp_fw_full(a_opt, time_limit=t_lp)
                try_seq(a_opt2, "deep_lpfw")

            rem = deadline - time.monotonic()
            if rem > 20:
                # Final CD
                t_cd = min(rem * 0.35, 100)
                print(f"  Deep CD, t={t_cd:.0f}s")
                a_opt3, _ = adaptive_cd(best_seq, time_limit=t_cd)
                try_seq(a_opt3, "deep_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
