import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_weighted(f, weights, n):
    """Solve LP: min sum_k w_k * (f*g)[k] s.t. g >= 0, sum(g) = 1."""
    c = np.zeros(n)
    # (f*g)[k] = sum_j f[j] * g[k-j], so sum_k w_k * (f*g)[k] = sum_j g[j] * (sum_k w_k * f[k-j])
    # Let's compute c[j] = sum_k w_k * f[k-j] for valid k
    for j in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            lj = k - j
            if 0 <= lj < n:
                s += weights[k] * f[lj]
        c[j] = s
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            if M < best_obj - 1e-15:

                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.002 * min(no_improve // 30, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def sg_optimize_nesterov(f0, time_limit, lr=0.003, momentum=0.9, top_k=30, 
                          temp=0.00005, use_nesterov=True, verbose=False):
    """Nesterov-accelerated subgradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Nesterov variables
    y = f.copy()
    v = np.zeros(n)
    beta = momentum

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient at y
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * y[kg[mask]]

        if use_nesterov:
            # Nesterov update
            v = beta * v - cur_lr * sg
            y_new = project_simplex(y + v)
            # Update f towards y
            t_k = (1.0 + beta) / (it + 1.0 + beta)
            f_new = (1.0 - t_k) * f + t_k * y_new
            f_new = project_simplex(f_new)
            y = y_new
        else:
            v = beta * v - cur_lr * sg
            f_new = project_simplex(f + v)
            y = f_new

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG-N iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        # Adaptive parameters
        cur_lr *= 0.9999
        cur_temp *= 1.003
        beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))  # Nesterov beta update

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            beta = momentum
            no_improve = 0

    return best_f, best_obj


def fine_cd(f0, time_limit, verbose=False):
    """Fine-grained coordinate descent with 1D exact search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    # Precompute peak positions
    peak_k = min(5, len(conv))
    peak_positions = set(np.argpartition(conv, -peak_k)[-peak_k:].tolist())

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Get contributing coordinates for top peaks
        contrib_coords = set()
        top_k = min(3, len(conv))
        pk_positions = np.argpartition(conv, -top_k)[-top_k:]
        for pk in pk_positions:
            peak_k_pos = int(pk)
            js = max(0, peak_k_pos - n + 1)
            je = min(n - 1, peak_k_pos)
            for j in range(js, je + 1):
                contrib_coords.add(j)

        # Process contributing coordinates first, then others
        coords = sorted(contrib_coords)
        remaining = [j for j in range(n) if j not in contrib_coords]
        all_coords = coords + remaining

        for j in all_coords:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            # Fine 1D search: try many multipliers
            multipliers = np.array([0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9,
                                    1.0, 1.1, 1.2, 1.3, 1.5, 1.7, 2.0])
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            # If improvement found, refine with binary search
            if best_j_val != old_val:
                # Determine direction
                if best_j_val < old_val:
                    lo, hi = best_j_val, old_val
                else:
                    lo, hi = old_val, best_j_val
                
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        break
                    scale = (1.0 - mid) / denom
                    f_trial = f.copy()
                    f_trial[j] = mid
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj:
                        best_j_obj = val
                        best_j_val = mid
                        if best_j_val < old_val:
                            hi = mid
                        else:
                            lo = mid
                    else:
                        if mid < old_val:
                            lo = mid
                        else:
                            hi = mid

                # Apply best
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
                    conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """Peak shaving: iteratively reduce the dominant peak by adjusting contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find the dominant peak
        pk = int(np.argmax(conv))
        pk_val = conv[pk]

        # Find contributing coordinates
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        coords = list(range(js, je + 1))

        if not coords:
            break

        # For each contributing coordinate, try reducing it
        best_improvement = None
        best_new_f = None
        best_new_obj = best_obj

        for j in coords:
            if f[j] < 1e-12:
                continue
            # Try reducing f[j] by a small amount
            for frac in [0.7, 0.8, 0.9]:
                new_val = f[j] * frac
                denom = 1.0 - f[j]
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_new_obj - 1e-14:
                    best_new_obj = val
                    best_new_f = f_trial.copy()
                    best_improvement = (j, frac)

        if best_new_f is not None:
            f = best_new_f
            conv = fast_convolve(f)
            if best_new_obj < best_obj - 1e-14:
                best_obj = best_new_obj
                best_f = f.copy()
        else:
            # Try increasing non-contributing coordinates
            non_contrib = [j for j in range(n) if j not in coords]
            if not non_contrib:
                break
            # Pick the largest non-contributing coord and try increasing it
            j = max(non_contrib, key=lambda i: f[i])
            for frac in [1.1, 1.2, 1.3]:
                new_val = f[j] * frac
                if new_val > 0.5:
                    continue
                denom = 1.0 - f[j]
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    f = f_trial
                    conv = fast_convolve(f)
                    best_obj = val
                    best_f = f.copy()
                    break
            else:
                # No improvement, try random perturbation
                if it > 50:
                    break

    return best_f, best_obj
