import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

    grad_b[m] = 2*f[m] * (sum_k w[k]*f[k-m] - sum_k w[k]*c[k])

    F = np.fft.rfft(f, fft_size)

    # This is the correlation of f and w, which is conv(f, reversed(w))
    # Actually: sum_j f[j]*w[m+j] is the cross-correlation at lag m

            if time.monotonic() - t0 > time_limit:
                break

            # c'[k] = c[k] - 2*f[j]*f[k-j] (for valid k-j)
            c = fast_convolve(f)  # This is expensive; let's precompute
            # Actually, let's compute c'[k] directly
            # c'[k] = c[k] - 2*a_j*f[k-j] for 0<=k-j<n, else c[k]
            c_prime = c.copy()
            k_lo = max(0, j)  # k-j >= 0 => k >= j

            valid_ks = np.arange(k_lo, k_hi+1)

            t_max = max(1e-15, 1.0 - rest_sum)

            # f(t) = max_k (2*t*f[k-j] + c'[k]) for valid k
            # This is convex in t
            
            # Quick: evaluate at current t and a few candidates
            # The optimal t is where the two "upper" lines cross
            # For speed, just try a small set of candidates
            
            # Key lines: those with f[k-j] > 0, i.e., k in [j, j+n-1]
            # c_val(k, t) = 2*t*f[k-j] + c'[k]
            
            # Use golden section
            golden = (np.sqrt(5) - 1) / 2
            a_gs = 0.0
            b_gs = t_max
            c_gs = b_gs - golden * (b_gs - a_gs)
            d_gs = a_gs + golden * (b_gs - a_gs)

                # Only need to check k in [j, j+n-1] (where f[k-j] > 0 contributes)
                # Plus any k where c'[k] is already large
                vals = c_prime[valid_ks] + 2.0 * t_val * f[valid_ks - j]
                return vals.max()
            
            fc = obj_at_t(c_gs)
            fd = obj_at_t(d_gs)
            
            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - golden * (b_gs - a_gs)
                    fc = obj_at_t(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + golden * (b_gs - a_gs)
                    fd = obj_at_t(d_gs)
            
            t_opt = (a_gs + b_gs) / 2.0
            t_opt = max(0.0, min(t_opt, t_max))
            
            f[j] = t_opt
            # Renormalize? No, we're already maintaining sum=1 by construction.
            
            new_M = obj_at_t(t_opt)
            if new_M < best_M - 1e-14:
                best_M = new_M
                best_f = f.copy()
                improved = True
        
        if not improved and pass_num > 2:
            break
    
    return best_f, best_M

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:

    return e / s


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def a_idx(ki, lo, hi, f):

def coordinate_descent(f0, time_limit, max_passes=5000, verbose=False):
    """
    Exact coordinate descent: for each element j, find the optimal value
    that minimizes max_k c[k] using golden section search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_M = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    pass_num = 0
    stalled = 0

    while time.monotonic() - t0 < time_limit and pass_num < max_passes:
        pass_num += 1
        improved_this_pass = False

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            a_j = f[j]
            if a_j < 1e-15:
                # Try to give it a small value
                f[j] = 1e-8
                continue

            rest_sum = 1.0 - a_j
            t_max = max(1e-15, rest_sum)

            # Compute c without a[j]'s contribution
            # c[k] = sum_i f[i]*f[k-i]
            # c'[k] = c[k] - 2*f[j]*f[k-j] for k where j <= k <= j+n-1
            c = fast_convolve(f)
            c_prime = c.copy()
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            if k_lo <= k_hi:
                c_prime[k_lo:k_hi + 1] -= 2.0 * a_j * f[k_lo - j:k_hi - j + 1]

            # Minimize max_k (c_prime[k] + 2*t*f[k-j]) over t in [0, t_max]
            # Only k in [k_lo, k_hi] are affected by t
            slopes = f[k_lo:k_hi + 1]  # 2*f[k-j] for k in range
            intercepts = c_prime[k_lo:k_hi + 1]

            # Golden section search
            golden_ratio = (np.sqrt(5.0) - 1.0) / 2.0
            ga = 0.0
            gb = t_max
            gc = gb - golden_ratio * (gb - ga)
            gd = ga + golden_ratio * (gb - ga)

            def obj_at(t_val):
                if t_val <= 0:
                    return c.max()
                if t_val >= t_max:
                    return c.max()
                vals = intercepts + 2.0 * t_val * slopes
                return max(vals.max(), np.delete(c_prime, range(k_lo, k_hi + 1)).max() if (k_hi - k_lo + 1) < len(c_prime) else vals.max())

            # Simpler: just compute max over all k
            # c_at_t[k] = c_prime[k] + 2*t*slope_k where slope_k = f[k-j] if valid else 0
            slope_full = np.zeros(2 * n - 1)
            slope_full[k_lo:k_hi + 1] = f[k_lo - j:k_hi - j + 1]

            def obj_full(t_val):
                return (c_prime + 2.0 * t_val * slope_full).max()

            fc = obj_full(gc)
            fd = obj_full(gd)

            for _ in range(25):
                if fc < fd:
                    gb = gd
                    gd = gc
                    fd = fc
                    gc = gb - golden_ratio * (gb - ga)
                    fc = obj_full(gc)
                else:
                    ga = gc
                    gc = gd
                    fc = fd
                    gd = ga + golden_ratio * (gb - ga)
                    fd = obj_full(gd)

            t_opt = (ga + gb) / 2.0
            t_opt = max(0.0, min(t_opt, t_max))

            if abs(t_opt - a_j) < 1e-12:
                continue

            f[j] = t_opt
            new_M = obj_full(t_opt)

            if new_M < best_M - 1e-13:
                best_M = new_M
                best_f = f.copy()
                improved_this_pass = True
            elif new_M > best_M + 1e-8:
                f[j] = a_j  # Revert if it got worse

        if not improved_this_pass:
            stalled += 1
            if stalled > 3:
                break
        else:
            stalled = 0

    return best_f, best_M


def lse_polish(f0, time_limit, beta_start=100.0, beta_end=5000.0, verbose=False):
    """High-beta LSE polish using simplex projection (additive steps)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_M = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta = beta_start
    lr = 1.0 / n
    momentum = 0.92
    v = np.zeros(n)
    no_improve = 0
    it = 0
    top_k = 20

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_len = 2 * n - 1

        # Top-k peaks with exponential weights
        w = np.zeros(conv_len)
        k_sel = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[k_sel]
        scaled = (pv - pv.min()) * beta
        sm = scaled.max()
        if sm > 500:
            scaled *= 500.0 / sm
        w[k_sel] = np.exp(scaled - sm)
        ws = w.sum()
        if ws > 0:
            w /= ws

        # Gradient of sum_k w[k]*c[k] w.r.t. a
        # dc[k]/da[j] = 2*a[k-j] for valid
        grad = np.zeros(n)
        for k in k_sel:
            wk = w[k]
            if wk < 1e-15:
                continue
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            grad[lo:hi + 1] += wk * f[ki - hi:ki - lo + 1][::-1]
            # Second term (same as first by symmetry for auto-conv)
            grad[lo:hi + 1] += wk * f[ki - hi:ki - lo + 1][::-1]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.93
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 0.5 / n
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.005
        if top_k < 60 and it % 200 == 0:
            top_k += 5

    return best_f, best_M


def try_sequence(a, best_tracker):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def structured_init(n, mode='sine', amp=0.3):
    """Create structured initialization."""
    base = np.ones(n) / n
    x = np.arange(n) / n
    if mode == 'sine':
        pert = amp * np.sin(2 * np.pi * 3 * x)
    elif mode == 'cos2':
        pert = amp * np.cos(2 * np.pi * 2 * x)
    elif mode == 'ramp':
        pert = amp * (2 * x - 1)
    elif mode == 'bump':
        pert = amp * np.exp(-((x - 0.5) ** 2) / 0.02)
    else:
        pert = np.zeros(n)
    f = base + pert
    f = np.maximum(f, 1e-10)
    f /= f.sum()
    return f

def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization:
    Phase 1: Multi-length/initialization exploration with quick softmax descent
    Phase 2: Intensive softmax descent with adaptive peaks
    Phase 3: Coordinate descent refinement
    Phase 4: LSE polish + final coordinate descent
    """
    global seed_global
    seed_global = seed
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Exploration (first ~250s) ===
    print("\n=== Phase 1: Exploration ===")
    t1_end = min(deadline - 1300, time.monotonic() + 300)

    # Try different lengths and initializations
    length_configs = [
        (n_parent, 'parent'),
        (max(640, n_parent - 64), 'interp'),
        (min(1024, n_parent + 128), 'interp'),
        (768, 'structured'),
        (896, 'structured'),
    ]

    tried_configs = set()
    for n_try, init_type in length_configs:
        if time.monotonic() >= t1_end:
            break
        if (n_try, init_type) in tried_configs:
            continue
        tried_configs.add((n_try, init_type))
        if n_try < 128 or n_try > 1200:
            continue

        t_alloc = min(60, t1_end - time.monotonic() - 10)
        if t_alloc < 15:
            break

        if init_type == 'parent' and parent is not None and n_try == n_parent:
            f_init = parent.copy()
        elif init_type == 'interp' and parent is not None:
            f_init = interpolate_to_length(parent, n_try)
        elif init_type == 'structured':
            mode = ['sine', 'cos2', 'ramp', 'bump'][hash((n_try, init_type)) % 4]
            f_init = structured_init(n_try, mode=mode, amp=0.2)
        else:
            f_init = interpolate_to_length(parent, n_try) if parent is not None else np.ones(n_try) / n_try

        print(f"  Exploring n={n_try}, init={init_type}, t={t_alloc:.0f}s")
        f_opt, _ = softmax_descent(f_init, time_limit=t_alloc, top_k_start=8, top_k_end=40, max_beta=50)
        try_sequence(f_opt, best_tracker)

        # Quick coordinate descent follow-up
        t_cd = min(30, t1_end - time.monotonic() - 10)
        if t_cd > 10:
            f_opt2, _ = coordinate_descent(f_opt, time_limit=t_cd, max_passes=200)
            try_sequence(f_opt2, best_tracker)

    # === Phase 2: Main optimization (next ~1000s) ===
    print(f"\n=== Phase 2: Main Optimization ===")
    t2_end = deadline - 400

    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)
    print(f"  Starting from n={n_cur}, score={best_tracker[0]:.8f}")

    # Alternating softmax descent and coordinate descent
    cycle = 0
    while time.monotonic() < t2_end:
        cycle += 1
        current = best_tracker[1] if best_tracker[1] is not None else current
        remaining = t2_end - time.monotonic()
        if remaining < 30:
            break

        if cycle % 3 != 0:
            # Softmax descent
            t_alloc = min(150, remaining * 0.5)
            if t_alloc > 20:
                print(f"  Cycle {cycle}: softmax n={len(current)}, t={t_alloc:.0f}s")
                f_opt, _ = softmax_descent(current, time_limit=t_alloc,
                                           top_k_start=5, top_k_end=60, max_beta=200)
                try_sequence(f_opt, best_tracker)
        else:
            # Coordinate descent
            t_alloc = min(120, remaining * 0.5)
            if t_alloc > 20:
                print(f"  Cycle {cycle}: coord_descent n={len(current)}, t={t_alloc:.0f}s")
                f_opt, _ = coordinate_descent(current, time_limit=t_alloc, max_passes=3000)
                try_sequence(f_opt, best_tracker)

    # === Phase 3: Intensive coordinate descent (next ~250s) ===
    print(f"\n=== Phase 3: Intensive Coordinate Descent ===")
    t3_end = deadline - 150
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = t3_end - time.monotonic()
    if t_alloc > 20:
        print(f"  Coord descent n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_descent(current, time_limit=t_alloc, max_passes=10000)
        try_sequence(f_opt, best_tracker)

    # Try at a slightly different length
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current)
    for n_off in [64, -64, 128]:
        n_try = n_best + n_off
        if n_try < 192 or n_try > 1200:
            continue
        t_remaining = t3_end - time.monotonic() - 10
        if t_remaining < 20:
            break
        f_init = interpolate_to_length(current, n_try)
        t_alloc = min(50, t_remaining * 0.5)
        if t_alloc > 10:
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = coordinate_descent(f_init, time_limit=t_alloc, max_passes=2000)
            try_sequence(f_opt, best_tracker)

    # === Phase 4: Final polish (last ~130s) ===
    print(f"\n=== Phase 4: Final Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent

    # High-beta LSE
    t_alloc = min(80, deadline - time.monotonic() - 50)
    if t_alloc > 15:
        print(f"  LSE polish n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = lse_polish(current, time_limit=t_alloc, beta_start=200.0, beta_end=8000.0)
        try_sequence(f_opt, best_tracker)

    # Final coordinate descent
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(60, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  Final coord descent n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_descent(current, time_limit=t_alloc, max_passes=5000)
        try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    fallback = np.ones(512) / 512
    return [float(x) for x in fallback]
