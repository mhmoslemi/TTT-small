```python
def softmax_lBfgs(a0, p_start=10, p_end=1000, time_limit=300, verbose=True):
    """Optimize using L-BFGS-B on softmax parameterization with log-sum-exp objective."""
    from scipy.optimize import minimize
    
    n = len(a0)
    # Initial z: a0 = softmax(z) => z = log(a0) + const
    # Use z = log(a0 + eps) - mean(log(a0 + eps))
    eps = 1e-15
    z0 = np.log(a0 + eps)
    z0 -= z0.mean()
    
    best_obj = float('inf')
    best_z = z0.copy()
    best_a = a0.copy()
    
    p = p_start
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        def obj_and_grad(z):
            a = softmax(z)
            conv = fast_convolve(a)
            # Log-sum-exp
            max_conv = conv.max()
            lse = max_conv + np.log(np.sum(np.exp(p * (conv - max_conv))))
            obj = lse / p  # this is the smoothed max
            
            # Gradient
            w = np.exp(p * (conv - max_conv))
            w /= w.sum()
            
            # c[j] = sum_k w[k] * a[k-j]
            c = np.zeros(n)
            for j in range(n):
                ks = max(0, j)
                ke = min(2*n-1, j+n)
                c[j] = np.dot(w[ks:ke], a[ks-j:ke-j])
            
            c_mean = np.dot(w, conv)
            grad = 2.0 * a * (c - c_mean)
            
            return obj, grad
        
        # Run L-BFGS-B
        result = minimize(obj_and_grad, z0, method='L-BFGS-B', 
                         jac=True, options={'maxiter': 100, 'gtol': 1e-8})
        
        a_new = softmax(result.x)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()  # actual max, not smoothed
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_z = result.x.copy()
            best_a = a_new.copy()
            if verbose:
                print(f"    p={p:.1f}, smoothed={result.fun:.6f}, actual_max={obj_new:.6f}")
        
        z0 = result.x
        p *= 2.0  # increase sharpness
        if p > p_end:
            break
    
    return best_a, best_obj
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B on softmax parameterization
with log-sum-exp annealing, combined with coordinate bisection polishing and
multi-start diversity.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def safe_softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient(a, conv, p):
    """Compute gradient of log-sum-exp(p*conv) w.r.t. z (softmax parameters)."""
    n = len(a)
    # Softmax weights over conv
    max_conv = conv.max()
    w = np.exp(p * (conv - max_conv))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # c[j] = sum_k w[k] * a[k-j] via FFT cross-correlation
    size = max(1, (2 * n - 1) + n - 1)  # >= 3n-2
    N = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    c = np.fft.ifft(np.conj(A) * W, N).real[:n]
    
    c_mean = np.dot(w, conv)
    grad = 2.0 * a * (c - c_mean)
    
    return grad


def logsumexp_obj_grad(z, p):
    """Objective and gradient for L-BFGS-B."""
    a = safe_softmax(z)
    conv = fast_convolve(a)
    max_conv = conv.max()
    lse = max_conv / p + np.log(np.sum(np.exp(p * (conv - max_conv)))) / p
    # Simplified: lse = (1/p) * log(sum(exp(p*conv)))
    # = (1/p) * (p*max_conv + log(sum(exp(p*(conv-max_conv)))))
    # = max_conv + (1/p)*log(sum(exp(p*(conv-max_conv))))
    
    grad = compute_conv_gradient(a, conv, p)
    return lse, grad


def lBfgs_softmax_optimize(a0, p_start=5.0, p_end=500.0, time_limit=200, 
                            max_total_iters=500, verbose=True):
    """
    L-BFGS-B optimization on softmax parameterization with annealed log-sum-exp.
    """
    n = len(a0)
    # Initialize z from a0
    eps = 1e-30
    z = np.log(np.maximum(a0, eps))
    z -= z.mean()
    
    best_a = a0.copy()
    best_max = fast_convolve(a0).max()
    t0 = time.monotonic()
    p = p_start
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and p < p_end and total_iters < max_total_iters:
        # Run L-BFGS-B at current p
        result = minimize(logsumexp_obj_grad, z, method='L-BFGS-B',
                         args=(p,), jac=True,
                         options={'maxiter': 50, 'gtol': 1e-7, 'factr': 1e7})
        
        total_iters += result.get('nit', 50)
        a_new = safe_softmax(result.x)
        conv_new = fast_convolve(a_new)
        max_new = conv_new.max()
        
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
            if verbose:
                print(f"      p={p:.1f}, iters={result.get('nit',0)}, max_conv={max_new:.8f}, obj={2*n*max_new:.6f}")
        
        z = result.x
        p *= 1.8  # anneal
        
        # Check time
        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.8 and p > p_end * 0.5:
            break
    
    return best_a, 2 * n * best_max


def coordinate_bisection(a0, time_limit, max_passes=200):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                deriv_at_max = _bisection_deriv(f_vals, b, B, j, mid, conv_len)
                
                if deriv_at_max > 1e-12:
                    hi = mid
                    x_new = mid
                elif deriv_at_max < -1e-12:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break
    
    return best_a, 2 * n * best_obj


def _bisection_deriv(f_vals, b, B, j, mid, conv_len):
    """Compute derivative at the maximizing point for bisection."""
    k = int(np.argmax(f_vals))
    b_kj = b[k - j] if 0 <= k - j < len(b) else 0.0
    B_k = B[k]
    ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
    om = 1.0 - mid

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B on softmax parameterization
with annealed log-sum-exp objective, FFT-accelerated gradients, coordinate bisection
polishing, and multi-start diversity with structured initializations.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def safe_softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_z_gradient(a, conv, p):
    """
    Compute gradient of (1/p)*log(sum(exp(p*conv))) w.r.t. softmax parameters z.
    Uses O(n log n) FFT-based cross-correlation.
    """
    n = len(a)
    max_conv = conv.max()
    w = np.exp(p * (conv - max_conv))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    # c[j] = sum_k w[k] * a[k-j] via cross-correlation (FFT)
    size = 3 * n - 2
    N = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    c = np.fft.ifft(np.conj(A) * W, N).real[:n]

    c_mean = np.dot(w, conv)
    grad = 2.0 * a * (c - c_mean)
    return grad


def lse_obj_grad(z, p):
    """Objective and gradient for L-BFGS-B (log-sum-exp of p*conv)."""
    a = safe_softmax(z)
    conv = fast_convolve(a)
    max_conv = conv.max()
    shifted = p * (conv - max_conv)
    max_shifted = 0.0
    lse = max_conv + (1.0 / p) * np.log(np.sum(np.exp(shifted)))
    grad = compute_z_gradient(a, conv, p)
    return lse, grad


def powermean_obj_grad(z, p):
    """Objective and gradient using power mean (sum conv^p)^(1/p)."""
    a = safe_softmax(z)
    conv = fast_convolve(a)
    conv_p = conv ** p
    pm = conv_p.sum() ** (1.0 / p)

    # Gradient of pm w.r.t. a: (1/p) * (sum conv^p)^(1/p - 1) * p * conv^(p-1) * dconv/da
    # = (sum conv^p)^(1/p - 1) * conv^(p-1) dot (dconv/da)
    # dconv[k]/da[j] = 2*a[k-j] (approx)
    # So dpm/da[j] = (sum conv^p)^(1/p-1) * sum_k conv[k]^(p-1) * 2*a[k-j]

    conv_p1 = conv_p.sum() ** (1.0 / p - 1.0)
    # c[j] = sum_k conv[k]^(p-1) * a[k-j]
    weights = np.zeros_like(conv)
    max_c = conv.max()
    if max_c > 0:
        weights = np.exp((p - 1) * (np.log(np.maximum(conv, 1e-300)) - np.log(max_c)))
    else:
        weights[:] = 1.0
    w_sum = weights.sum()
    if w_sum > 0:
        weights /= w_sum

    n = len(a)
    size = 3 * n - 2
    N = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, N)
    W = np.fft.fft(weights, N)
    c = np.fft.ifft(np.conj(A) * W, N).real[:n]

    c_mean = np.dot(weights, conv)
    grad = 2.0 * a * (c - c_mean) * conv_p
    return pm, grad


def lBfgs_optimize(a0, p_start=5.0, p_end=800.0, time_limit=200,
                   use_powermean=False, verbose=True, label=""):
    """
    L-BFGS-B optimization with annealed log-sum-exp (or power mean) objective.
    Returns best sequence and its actual max-conv objective.
    """
    n = len(a0)
    eps = 1e-30
    z = np.log(np.maximum(a0, eps))
    z -= z.mean()

    best_a = a0.copy()
    best_max = fast_convolve(a0).max()
    t0 = time.monotonic()
    p = p_start

    obj_fn = powermean_obj_grad if use_powermean else lse_obj_grad

    while time.monotonic() - t0 < time_limit and p < p_end:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        maxiter = max(20, min(100, int(remaining * 5)))
        try:
            result = minimize(obj_fn, z, method='L-BFGS-B',
                              args=(p,), jac=True,
                              options={'maxiter': maxiter, 'gtol': 1e-8, 'factr': 1e8})
            a_new = safe_softmax(result.x)
        except Exception:
            break

        conv_new = fast_convolve(a_new)
        max_new = conv_new.max()

        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
            if verbose:
                print(f"    {label} p={p:.1f}: max_conv={max_new:.10f}, 2n*max={2*n*best_max:.6f}")

        z = result.x
        p *= 1.7

    return best_a, 2.0 * n * best_max


def coordinate_bisection(a0, time_limit, max_passes=300, verbose=False):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(42):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))
                b_kj = b[k_max - j] if 0 <= k_max - j < n else 0.0
                B_k = B[k_max]
                ind = 1.0 if (2 * j < conv_len and k_max == 2 * j) else 0.0
                deriv = -2.0 * om * B_k + 2.0 * (1.0 - 2.0 * mid) * b_kj + 2.0 * mid * ind

                if deriv > 1e-12:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-12:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            no_improve_count += 1
            if no_improve_count > 2:
                rng = np.random.default_rng(passes * 43 + 13)
                noise_scale = 0.0002 * no_improve_count
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                no_improve_count = 0
            elif passes > 5:
                break
        else:
            no_improve_count = 0

        if verbose and passes % 20 == 0:
            print(f"    CD pass {passes}: 2n*max={2*n*best_obj:.6f}")

    return best_a, 2.0 * n * best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_init(n, freq=1.0, alpha=0.3, n_peaks=1):
    """Generate a cosine-modulated initial sequence."""
    x = np.arange(n)
    if n_peaks == 1:
        a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n)
    elif n_peaks == 2:
        a = 1.0 + alpha * np.cos(4.0 * np.pi * x / n)
    elif n_peaks == 3:
        a = 1.0 + alpha * np.cos(6.0 * np.pi * x / n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def gaussian_mixture_init(n, n_gauss=2, alpha=0.3, seed=42):
    """Generate a Gaussian mixture initial sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) * 0.1
    for _ in range(n_gauss):
        center = rng.uniform(0.1, 0.9) * n
        width = rng.uniform(0.05, 0.2) * n
        amp = 1.0 + alpha
        x = np.arange(n)
        a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

        noisy = project_simplex(mix * base + (1 - mix) * rng.dirichlet(np.ones(n_base)))
        inits.append(noisy)
    # Symmetric
    inits.append(symmetrize(base))
    # Reverse mix
    inits.append(project_simplex(0.5 * base + 0.5 * base[::-1]))

    # Also try different lengths
    for target_n in [512, 896, 1024, 1280]:
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        f_interp = interpolate_to_length(base, target_n)
        inits.append(f_interp)

    for idx, init_a in enumerate(inits):
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(rem * 0.7, 80)
        if t_alloc < 15:
            continue
        label = f"init{idx}"
        a_opt, obj = lBfgs_optimize(init_a, p_start=5.0, p_end=300.0,
                                    time_limit=t_alloc, verbose=False, label=label)
        update_best(a_opt, label)

        # Quick CD polish
        rem = deadline - time.monotonic() - 120
        if rem > 15:
            t_cd = min(rem, 25)
            a_opt2, obj2 = coordinate_bisection(a_opt, time_limit=t_cd)
            update_best(a_opt2, label + "_cd")

    # === Phase 5: Deep optimization from best ===
    print(f"\n=== Phase 5: Deep optimize (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restarts = [
            ("direct", best_a.copy()),
            ("noise3", project_simplex(best_a + rng.standard_normal(n_best) * 0.003)),
            ("noise1", project_simplex(best_a + rng.standard_normal(n_best) * 0.001)),
            ("sym", symmetrize(best_a)),
            ("flat_mix", project_simplex(0.85 * best_a + 0.15 * np.ones(n_best) / n_best)),
        ]

        for idx, (label, f_init) in enumerate(restarts):
            rem = deadline - time.monotonic() - 100 + idx * 30
            if rem < 20:
                break
            t_alloc = min(rem, 100)
            a_opt, obj = lBfgs_optimize(f_init, p_start=20.0, p_end=1000.0,
                                        time_limit=t_alloc, verbose=True, label=f"P5-{label}")
            update_best(a_opt, f"deep_{label}")

    # === Phase 6: Final intensive CD ===
    print(f"\n=== Phase 6: Final CD (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_alloc = min(rem, 200)
            print(f"  CD t={t_alloc:.0f}s")
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "final_cd")

        # Tiny perturbations + CD
        rem = deadline - time.monotonic() - 30
        if rem > 10 and best_a is not None:
            n_best = len(best_a)
            for p in range(4):
                rem = deadline - time.monotonic() - 20 + p * 10
                if rem < 8:
                    break
                rng_p = np.random.default_rng(seed + 2000 + p * 7)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.0008)
                t_alloc = min(rem, 15)
                a_opt, obj = coordinate_bisection(f_pert, time_limit=t_alloc)
                update_best(a_opt, f"perturb_{p}")

        # One more L-BFGS-B round with high p from best
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 40)
            a_opt, obj = lBfgs_optimize(best_a, p_start=50.0, p_end=2000.0,
                                        time_limit=t_alloc, verbose=True, label="final_lBfgs")
            update_best(a_opt, "final_lBfgs")
            rem = deadline - time.monotonic() - 10
            if rem > 5:
                a_opt2, obj2 = coordinate_bisection(a_opt, time_limit=min(rem, 10))
                update_best(a_opt2, "final_lBfgs_cd")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
