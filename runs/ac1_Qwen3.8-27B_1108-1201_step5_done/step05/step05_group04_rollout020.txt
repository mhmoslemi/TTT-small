import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad_fft(a, w):
    """Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a using FFT.
    grad[j] = 2 * sum_m a[m] * w[m+j] (for j = 0, ..., n-1)
    This is computed as cross-correlation via FFT.
    """
    n = len(a)
    # Need FFT size >= n + len(w) - 1
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    R = np.fft.ifft(np.conj(A) * W)
    return 2.0 * R[:n].real

def lse_weights(conv, beta):
    """Compute LSE softmax weights."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    return exp_bc / exp_bc.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based O(n log n) gradient computation for LSE-smoothed max-conv objective
2. Nesterov-accelerated projected gradient on the simplex
3. Adaptive beta schedule (LSE temperature)
4. Multi-length exploration with symmetry exploitation
5. Peak-targeting coordinate refinement
Key: much faster gradient via cross-correlation FFT, enabling many more iterations.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, w):
    """Gradient of sum_k w[k]*(a*a)[k] via cross-correlation FFT.
    grad[j] = 2 * sum_m a[m] * w[m+j]
    """
    n = len(a)
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    R = np.fft.ifft(np.conj(A) * W)
    return 2.0 * R[:n].real


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def accelerated_lse(a0, time_limit, beta_start=3.0, beta_end=10000.0,
                    lr_init=None, momentum_max=0.95, seed=0):
    """Nesterov-accelerated projected gradient with LSE smoothing.
    Uses FFT-based O(n log n) gradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr_init is None:
        lr = 4.0 / (n ** 0.5)
    
    conv = fast_convolve(a)
    peak0 = conv.max()
    best_obj = peak0
    best_a = a.copy()
    
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)
    momentum = 0.75
    beta = beta_start
    beta_growth = 1.003
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    
    # Scale gradient by peak for consistent step sizes
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov step
        a_new = y + momentum * (y - a)
        a_new = project_simplex(a_new)
        
        conv = fast_convolve(a_new)
        peak = conv.max()
        
        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # FFT gradient
        grad = compute_grad_fft(a_new, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        
        # Step
        a_next = project_simplex(a_new - lr * grad)
        
        conv_next = fast_convolve(a_next)
        obj_next = conv_next.max()
        
        if obj_next < best_obj - 1e-12:
            best_obj = obj_next
            best_a = a_next.copy()
            no_improve = 0
            # Slightly increase lr on success
            lr = min(lr * 1.001, 10.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.95
            if no_improve > 100:
                # Restart from best with perturbation
                a = best_a.copy()
                y = best_a.copy()
                v = np.zeros(n)
                lr = max(4.0 / n, 0.01)
                momentum = 0.75
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
        
        # Update Nesterov variables
        y = a_next
        a = a_next
        
        # Adapt beta
        if beta < beta_end:
            beta *= beta_growth
        
        # Periodically try to increase momentum
        if it % 500 == 0:
            momentum = min(momentum + 0.005, momentum_max)
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based O(n log n) gradient via cross-correlation (30x faster than loop-based)
2. Nesterov-accelerated projected gradient on simplex with LSE smoothing
3. Backtracking line search for adaptive step sizes
4. Multi-length exploration with symmetry exploitation
5. Peak-targeting coordinate refinement for final polish
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, w):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a via FFT cross-correlation.
    grad[j] = 2 * sum_m a[m] * w[m+j] for j=0..n-1
    Uses: C = IFFT(conj(FFT(w)) * FFT(a)), then grad = 2*C[:n]
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    C = np.fft.ifft(np.conj(W) * A)
    return 2.0 * C[:n].real


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_soft_obj(conv, beta):
    """Compute LSE approximation of max and its softmax weights."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    return bc_max / beta + np.log(exp_bc.sum()) / beta, w


def accelerated_lse_fft(a0, time_limit, beta_start=3.0, beta_end=8000.0,
                        lr_init=None, seed=0):
    """Nesterov-accelerated projected gradient with LSE smoothing and FFT gradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if lr_init is None:
        lr = 3.0 / np.sqrt(n)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    y = a.copy()
    t_k = 1.0
    beta = beta_start
    beta_growth = 1.004
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov extrapolation
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / alpha * (a - y)
        y = project_simplex(y)
        t_k = alpha

        conv = fast_convolve(y)
        cur_obj = conv.max()

        # LSE weights and gradient
        _, w = lse_soft_obj(conv, beta)
        grad = compute_grad_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        # Backtracking line search
        best_trial_obj = cur_obj
        best_trial_a = y.copy()
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step_lr = lr * factor
            x_new = project_simplex(y - step_lr * grad)
            conv_new = fast_convolve(x_new)
            obj_new = conv_new.max()
            if obj_new < best_trial_obj:
                best_trial_obj = obj_new
                best_trial_a = x_new
            if obj_new < cur_obj:
                break

        if best_trial_obj < best_obj - 1e-13:
            best_obj = best_trial_obj
            best_a = best_trial_a.copy()
            no_improve = 0
            lr = min(lr * 1.0005, 8.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.92
            if no_improve > 120:
                # Restart with perturbation
                scale = 0.001 + 0.003 * min(1.0, (no_improve - 120) / 100.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                y = a.copy()
                t_k = 1.0
                lr = max(3.0 / np.sqrt(n), 0.01)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0

        a = best_trial_a

        # Adapt beta
        if beta < beta_end:
            beta *= beta_growth

    return best_a, best_obj


def peak_targeting(a0, time_limit, n_peaks=8, lr_init=0.002, seed=0):
    """Directly target top peaks with focused gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    momentum = 0.9
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        # Focus on top peaks with exponential weighting
        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        # Weight: emphasize the absolute top
        wpeaks = 1.0 / (pv.max() - pv + 1e-10)
        wpeaks /= wpeaks.sum()

        w_full = np.zeros(conv_len)
        w_full[peaks] = wpeaks

        grad = compute_grad_fft(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve > 200:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0008)
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.0001)
                no_improve = 0

        a = a_new

    return best_a, best_obj


def multiplicative_lse(a0, time_limit, beta_start=5.0, beta_end=5000.0, seed=0):
    """Multiplicative weight (exponentiated gradient) with LSE smoothing."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    beta = beta_start
    eta = 2.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        _, w = lse_soft_obj(conv, beta)
        grad = compute_grad_fft(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        # Multiplicative update with line search
        best_trial = a.copy()
        best_trial_obj = best_obj
        for factor in [1.0, 0.5, 0.25, 0.1]:
            log_a = np.log(a)
            log_a -= log_a.max()
            log_a_new = log_a - eta * factor * grad
            log_a_new -= log_a_new.max()
            a_try = np.exp(log_a_new)
            a_try /= a_try.sum()
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial = a_try
            if obj_try < conv.max():
                break

        if best_trial_obj < best_obj - 1e-13:
            best_obj = best_trial_obj
            best_a = best_trial.copy()
            no_improve = 0
            eta = min(eta * 1.002, 5.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 10:
                eta *= 0.9
            if no_improve > 150:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                eta = max(2.0 / np.sqrt(n), 0.01)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0

        a = best_trial
        if beta < beta_end:
            beta *= 1.003

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def symmetric_init(n, freq=3.0, amplitude=0.08):
    """Create a symmetric sinusoidal perturbation of uniform."""
    base = np.ones(n) / n
    x = np.arange(n)
    sym = np.cos(2.0 * np.pi * freq * x / n)
    # Make it symmetric
    half = n // 2
    if n % 2 == 1:
        sym_full = np.concatenate([sym[:half], [0.0], sym[half:][::-1]])
    else:
        sym_full = np.concatenate([sym[:half], sym[half:][::-1]])
    f = base + amplitude * sym_full
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f

def coordinate_refine(a0, time_limit, seed=0):
    """Direct coordinate-wise refinement targeting the peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peak_k = np.argmax(conv)

        # For each candidate coordinate j, compute the effect of increasing it
        # d(a*a)[peak_k] / d(a[j]) = 2*a[peak_k - j] if 0 <= peak_k-j < n
        # We want to DECREASE the peak, so we want to DECREASE a[j] for j where
        # a[peak_k-j] is large, or INCREASE a[j] where it would redistribute mass.
        
        # Simple approach: try random coordinate perturbations
        n_trials = 20
        improved = False
        for _ in range(n_trials):
            j = rng.integers(0, n)
            # Try decreasing a[j] (if positive) and adding to a random other coordinate
            if a[j] < 1e-6:
                continue
            delta = a[j] * rng.uniform(0.01, 0.05)
            a_try = a.copy()
            a_try[j] -= delta
            # Add to a random coordinate far from peak influence
            k_add = rng.integers(0, n)
            a_try[k_add] += delta
            a_try = np.maximum(a_try, 0)
            a_try /= a_try.sum()
            conv_try = fast_convolve(a_try)
            if conv_try.max() < best_obj - 1e-12:
                best_obj = conv_try.max()
                best_a = a_try.copy()
                a = a_try
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 500:
                break

    return best_a, best_obj
