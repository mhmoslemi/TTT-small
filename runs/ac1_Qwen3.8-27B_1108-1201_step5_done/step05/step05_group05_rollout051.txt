"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp optimization
with Adam, fine coordinate descent, mirror descent, and multi-initialization search.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad(a, tau):
    """Compute smooth objective value and gradient w.r.t. z (where a = softmax(z))."""
    n = len(a)
    c = fast_convolve(a)
    c_shifted = (c - c.max()) / tau
    log_sum_exp = c.max() / tau + np.log(np.sum(np.exp(c_shifted)))
    obj = 2.0 * n * tau * log_sum_exp
    
    w = np.exp(c_shifted)
    w /= w.sum()
    
    # Compute h[m] = sum_j a[j] * w[m+j] via FFT cross-correlation
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    cc = np.fft.irfft(np.conj(A) * W, N)
    h = cc[:n]
    
    mean_h = np.dot(a, h)
    grad = 2.0 * n * a * (h - mean_h)
    
    return obj, grad

def adam_smooth_optimize(z0, time_limit, tau_start=0.005, tau_end=0.0001,
                          lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization on softmax parameterization with decreasing tau."""
    z = z0.copy().astype(np.float64)
    n = len(z)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = None
    best_obj = np.inf
    
    t0 = time.monotonic()
    
    # Adaptive schedule: total iterations estimated
    # Each iteration takes about 1ms for n=768, so ~1000 iterations per second
    # We'll run until time_limit
    it_per_sec = 500  # conservative estimate
    
    total_iter = int(time_limit * it_per_sec)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Decrease tau
        frac = min(1.0, t / max(1, total_iter))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        a = np.exp(z - z.max())
        a /= a.sum()
        
        obj, grad = smooth_grad(a, tau)
        
        # Also compute the true objective
        true_obj = 2.0 * n * fast_convolve(a).max()
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    if best_a is None:
        a = np.exp(z - z.max())
        best_a = a / a.sum()
    
    return best_a, best_obj

def coord_desc_fine(a0, time_limit, n_evals=8):
    """Coordinate descent with ternary search on each coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 3:
        no_improve += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                # Try setting to small positive value
                candidates = [1e-6, 1e-5, 1e-4, 0.01 / n]
            else:
                # Ternary search around old_val
                lo = max(0.0, old_val * 0.3)
                hi = old_val * 3.0
                # Ensure lo + hi scaling is valid
                # After setting a[j] = t, others scale by (1-t)/(1-old_val)
                # We need (1-old_val) > 0 and t < 1
                if old_val >= 1 - 1e-10:
                    continue
                
                # Evaluate at a few points for a coarse grid, then ternary
                best_t = old_val
                best_val = best_obj
                
                # Coarse grid
                grid = np.linspace(lo, min(hi, 0.999), n_evals)
                for t in grid:
                    if t < 0 or t > 1 - 1e-10:
                        continue
                    scale = (1 - t) / (1 - old_val)
                    a_trial = a.copy()
                    a_trial[j] = t
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    a_trial = a_trial / a_trial.sum()
                    val = 2.0 * n * fast_convolve(a_trial).max()
                    if val < best_val:
                        best_val = val
                        best_t = t
                
                if best_t != old_val:
                    scale = (1 - best_t) / (1 - old_val)
                    a[j] = best_t
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    if best_val < best_obj - 1e-15:
                        best_obj = best_val
                        best_a = a.copy()
                        improved = True
        
        if improved:
            no_improve = 0
    
    return best_a, best_obj

def mirror_descent(a0, time_limit, eta0=0.05, eta_decay=0.9995):
    """Mirror descent with subgradients (multiplicative weight updates)."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    eta = eta0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        k_star = np.argmax(c)
        
        # Subgradient: sg[j] = a[k_star - j] for valid j
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            sg[j] = a[k_star - j]
        
        # Mirror descent update: a_new[j] = a[j] * exp(-eta * sg[j]) / Z
        exponents = -eta * sg
        exponents -= exponents.max()  # for numerical stability
        weights = a * np.exp(exponents)
        Z = weights.sum()
        if Z < 1e-30:
            break
        a_new = weights / Z
        
        obj = 2.0 * n * fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        eta *= eta_decay
        
        if no_improve > 200:
            # Reset with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            eta = eta0 * 0.5
            no_improve = 0
    
    return best_a, best_obj

def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig
