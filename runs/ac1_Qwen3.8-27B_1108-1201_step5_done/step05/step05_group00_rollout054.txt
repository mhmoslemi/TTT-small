def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def logsumexp(x, T):
    """Compute log(sum(exp(x/T))) with numerical stability."""
    xmax = x.max()
    return xmax / T + np.log(np.sum(np.exp((x - xmax) / T)))

    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = v.astype(np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_shape(a, power=0.8, iterations=100, clip_frac=0.9):
    """
    Gerchberg-Saxton style spectral shaping.
    Iteratively flattens the PSD while maintaining non-negativity.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    
    # Zero-pad to at least 2n-1
    N = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(iterations):
        A = np.fft.fft(a, N)
        S = np.abs(A) ** 2
        S_mean = S.mean()
        
        # Modify S to be flatter
        # Method: S_new = S^power (compresses dynamic range)
        S_new = np.power(S + 1e-30, power)
        
        # Reconstruct with original phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = np.sqrt(S_new) * phase
        a_new = np.real(np.fft.ifft(A_new))[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj

def spectral_shape_clip(a, iterations=200, clip_level=0.95):
    """
    Spectral shaping with clipping: clip the top of the PSD and redistribute.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    
    N = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(iterations):
        A = np.fft.fft(a, N)
        S = np.abs(A) ** 2
        S_max = S.max()
        S_mean = S.mean()
        
        # Clip the PSD
        threshold = clip_level * S_max
        S_new = S.copy()
        excess = np.maximum(S - threshold, 0)
        deficit = np.maximum(threshold - S, 0)
        
        # Redistribute excess to deficit
        if deficit.sum() > 1e-15:
            S_new = S + excess * (deficit / deficit.sum())
            S_new = np.maximum(S_new, 0)
        
        # Reconstruct
        phase = A / (np.abs(A) + 1e-30)
        A_new = np.sqrt(S_new) * phase
        a_new = np.real(np.fft.ifft(A_new))[:n]
        
        # Project
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj

def lse_smoothed_descent(a0, time_limit, T_start=0.0005, T_end=0.00005, 
                          lr_start=0.05, momentum=0.9, verbose=True):
    """
    Smoothed gradient descent using log-sum-exp of auto-correlation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    T = T_start
    lr = lr_start
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # Compute annealing schedule
    total_iters_est = int(time_limit / 0.005)  # estimate ~5ms per iter
    anneal_rate = np.log(T_end / T_start) / max(1, total_iters_est)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break
        
        conv = fast_convolve(a)
        
        # Softmax weights
        scaled = (conv - conv.max()) / T
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient: 2 * correlate(p, a)
        grad = 2.0 * np.correlate(p, a, mode='valid')
        
        # Momentum
        v = momentum * v + grad
        
        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            trial_lr = lr * factor
            candidate = project_simplex(a - trial_lr * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        current_obj = best_val
        
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    LSE it={it}, obj={best_obj:.8f}, T={T:.7f}")
        else:
            no_improve += 1
        
        # Anneal
        T *= np.exp(anneal_rate)
        T = max(T, T_end)
        
        # Adapt
        if no_improve > 50:
            lr *= 0.95
        if no_improve > 200:
            T = max(T_end, T * 0.8)
        if no_improve > 500:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise = 0.002 + 0.002 * min(1.0, (no_improve - 500) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            T = T_start * 0.5
            lr = lr_start * 0.5
            no_improve = 0
    
    return best_a, best_obj

def parametric_init(n, alpha, beta, gamma):
    """
    Generate a parametric initialization.
    alpha: edge concentration (higher = more U-shaped)
    beta: center dip (higher = more dip at center)
    gamma: asymmetry parameter
    """
    x = np.linspace(0, 1, n)
    # U-shaped component
    edge = (x ** alpha + (1 - x) ** alpha)
    # Center dip
    center = 1.0 - beta * np.exp(-((x - 0.5) ** 2) / (0.05 ** 2))
    a = edge * center
    # Asymmetry
    a = a * (1.0 + gamma * (2 * x - 1.0))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

def try_parametric_grid(n, time_limit):
    """Try a grid of parametric initializations."""
    best_a = None
    best_obj = float('inf')
    
    t0 = time.monotonic()
    
    alphas = [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]
    betas = [0.0, 0.3, 0.5, 0.7, 0.9]
    gammas = [-0.3, -0.1, 0.0, 0.1, 0.3]
    
    for alpha in alphas:
        if time.monotonic() - t0 > time_limit:
            break
        for beta in betas:
            if time.monotonic() - t0 > time_limit:
                break
            for gamma in gammas:
                a = parametric_init(n, alpha, beta, gamma)
                obj = fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
    
    return best_a, best_obj

def fw_with_lp(f0, time_limit, verbose=True):
    """Frank-Wolfe with LP oracle (simplified version)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        # A_ub has shape (2n-1, n), A_ub[k, j] = f[max(0, k-n+1):min(n, k+1)] at position j
        # Actually: (f*g)[k] = sum_j f[j]*g[k-j] ... no wait
        # conv(f,g)[k] = sum_j f[min(n-1,j)] * g[k-j] ... 
        # Let me be more careful.
        # np.convolve(f, g)[k] = sum_j f[j] * g[k-j] for j where both indices are valid
        # = sum_j f[j] * g[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        # So A_ub[k, j] = f[j] if j in [max(0,k-n+1), min(n-1,k)] else 0
        # Wait no. conv(f,g)[k] = sum_j f[j]*g[k-j]. For this to be valid:
        # 0 <= j < n and 0 <= k-j < n, so k-n+1 <= j <= k and 0 <= j < n
        # So j is in [max(0, k-n+1), min(n-1, k)]
        # And the coefficient of g[j'] where j' = k-j is f[j] = f[k-j']
        # So conv(f,g)[k] = sum_{j'} g[j'] * f[k-j'] where j' in [max(0,k-n+1), min(n-1,k)]
        # And f[k-j'] is valid when 0 <= k-j' < n, i.e., k-n+1 <= j' <= k
        # So A_ub[k, j'] = f[k-j'] if max(0,k-n+1) <= j' <= min(n-1,k) else 0
        
        # Actually let me just build it directly
        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[k, j] = f[k - j]
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                continue
        except Exception:
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            no_improve = 0
    
    return best_f, best_obj

Hmm, wait. I need to reconsider. The `fw_with_lp` function builds an A_ub matrix of size (2n-1) x n. For n=2048, that's 4095 x 2048 = about 8 million entries. The LP will be very slow for large n. I need to be careful about the LP size.

Actually, for the LP to be tractable, n should be at most around 500-1000. For larger n, I should skip the LP and use only gradient-based methods.

Let me also reconsider the spectral approach. The issue is:
- We zero-pad a to length N >= 2n-1
- We take FFT of length N
- The PSD S(ω) = |A(ω)|^2 has the property that its inverse FFT gives the circular auto-correlation of the zero-padded sequence, which equals the linear auto-correlation
- So if we can make S(ω) flat, the linear auto-correlation will be flat (except at the center where it equals sum(a)^2 = 1)

Wait, that's the key issue! The DC component of S is S(0) = |sum(a)|^2 = 1 (when sum(a)=1). This corresponds to the "center" of the auto-correlation. So the auto-correlation at lag n-1 (the center) is always 1 when sum(a)=1? No, that's not right.

Let me be more careful. If we zero-pad a to length N:
a_pad = [a[0], ..., a[n-1], 0, ..., 0] (length N)

The circular auto-correlation of a_pad is:
r[k] = sum_j a_pad[j] * a_pad[(j+k) mod N]

For k=0: r[0] = sum_j a_pad[j]^2 = sum_j a[j]^2
For 0 < k < n: r[k] = sum_j a_pad[j] * a_pad[j+k] = sum_{j=0}^{n-1-k} a[j] * a[j+k] (linear auto-correlation at lag k)
For k = n-1: r[n-1] = a[0]*a[n-1]

Wait, this doesn't match np.convolve. Let me think again.

np.convolve(a, a) gives the linear auto-correlation:
conv[k] = sum_j a[j] * a[j+k] for k >= 0 (with appropriate bounds)

Actually, np.convolve(a, a) is:
result[k] = sum_j a[j] * a[k-j] for j where 0 <= j < n and 0 <= k-j < n

So result[k] = sum_j a[j] * a[k-j] for j in [max(0, k-n+1), min(n-1, k)]

This is the cross-correlation of a with itself, which for real sequences is the same as auto-correlation at lag k (up to sign convention).

The circular auto-correlation of the zero-padded sequence a_pad (length N >= 2n-1) is:
r_circ[k] = sum_{j=0}^{N-1} a_pad[j] * a_pad[(j+k) mod N]

For 0 <= k <= n-1: r_circ[k] = sum_{j=0}^{n-1-k} a[j] * a[j+k]
This is NOT the same as conv[k] = sum_j a[j] * a[k-j].

Let me check: conv[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * a[k-j]

For k >= 0: let i = k-j, so j = k-i. When j = max(0,k-n+1), i = k - max(0,k-n+1) = min(k, n-1). When j = min(n-1,k), i = k - min(n-1,k) = max(0, k-n+1).

So conv[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[k-i] * a[i] = sum_i a[i] * a[k-i]

And r_circ[k] for 0 <= k <= n-1 = sum_{j=0}^{n-1-k} a[j] * a[j+k]

Let me substitute: let i = j+k, so j = i-k. When j=0, i=k. When j=n-1-k, i=n-1.
r_circ[k] = sum_{i=k}^{n-1} a[i-k] * a[i] = sum_{i=k}^{n-1} a[i] * a[i-k]

And conv[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (with the understanding that a[m] = 0 for m < 0 or m >= n)

Hmm, these are related but not identical. Let me use a specific example.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = v.astype(np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gen_u_shaped(n, alpha):
    """U-shaped initialization: more mass at edges."""
    x = np.linspace(0, 1, n)
    a = x**alpha + (1 - x)**alpha
