def compute_sg_sparse(a, weights, peaks):

def softmax_weights(conv, temperature, top_k):
    """Compute softmax weights over top-k peaks."""
    m = len(conv)
    k = min(top_k, m)
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()

    w_full = np.zeros(m)
    w_full[peaks] = w
    return w_full, peaks


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_optimize(f0, time_limit, top_k=50, temperature=1e-5,
                         lr=0.004, momentum=0.85, avg_window=500, verbose=False):
    """
    Subgradient optimization with Nesterov momentum and Polyak-Ruppert averaging.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    # Polyak averaging buffer
    avg_sum = np.zeros(n)
    avg_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        w, peaks = softmax_weights(conv_nest, cur_temp, cur_top_k)
        sg = compute_sg_sparse(f_nest, w, peaks)

        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm

        # Update momentum
        v = momentum * v + (1 - momentum) * sg

        # Update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f_new = f * np.exp(exponent)
        fs = f_new.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f_new /= fs
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()

        # Accumulate for Polyak averaging
        avg_sum += f_new
        avg_count += 1

        conv = fast_convolve(f_new)
        obj = conv.max()

        # Check averaged solution too
        if avg_count >= avg_window:
            f_avg = avg_sum / avg_count
            f_avg = np.maximum(f_avg, 0)
            s_avg = f_avg.sum()
            if s_avg > 1e-15:
                f_avg /= s_avg
            conv_avg = fast_convolve(f_avg)
            obj_avg = conv_avg.max()
            if obj_avg < obj:
                obj = obj_avg
                f_new = f_avg

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive schedule
        cur_lr *= 0.9995
        if cur_lr < 0.0001:
            cur_lr = 0.0001
        cur_temp *= 1.005

        # Restart if stuck
        if no_improve > 400:
            rng = np.random.default_rng((it * 6151 + 7) % 10000000)
            noise_scale = 0.001 + 0.001 * min(no_improve // 600, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.5
            avg_sum = np.zeros(n)
            avg_count = 0
            no_improve = 0

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, lr={cur_lr:.7f}")

    return best_f, best_obj

                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val and best_j_obj < cur_peak - 1e-14:
                others_sum = 1.0 - old_val
                if others_sum > 1e-12:
                    scale = (1.0 - best_j_val) / others_sum
                    if scale > 0:
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = np.maximum(f, 0)
                        f /= f.sum()
                        if best_j_obj < best_obj - 1e-13:
                            best_obj = best_j_obj
                            best_f = f.copy()
                            improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def micro_perturbation_search(f0, time_limit, verbose=False):
    """
    Very fine-grained perturbation search: try tiny random changes and
    accept only if they strictly improve the objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    scale = 0.001
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Try a small random perturbation
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick 2-5 random indices to perturb
            n_pert = rng.integers(2, 6)
            indices = rng.integers(0, n, size=n_pert)
            deltas = rng.standard_normal(n_pert) * scale

            f_trial = f.copy()
            for idx, d in zip(indices, deltas):
                f_trial[idx] += d
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial /= st

            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f_trial.copy()
                f = f_trial.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Decay scale
        if no_improve > 200:
            scale *= 0.8
            if scale < 1e-5:
                # Try a slightly larger perturbation to escape
                scale = 0.003
                noise = rng.standard_normal(n) * 0.002
                f = project_simplex(f + noise)
                no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining subgradient with averaging,
    peak equalization, coordinate descent, and micro-perturbation search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform n=768")

    n_parent = len(parent)

    # === Phase 1: Quick subgradient on parent and variants ===
    print(f"\n=== Phase 1: Subgradient Optimization (best={best_score:.6f}) ===")

    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem * 0.4, 300)
        print(f"  Main SG on parent: n={n_parent}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(parent, time_limit=t_alloc,
                                         top_k=50, temperature=1e-5,
                                         lr=0.004, momentum=0.85, avg_window=500)
        try_seq(a_opt, "sg_parent")

    # Try a noisy variant
    rem = deadline - time.monotonic() - 500
    if rem > 20 and best_seq is not None:
        noise = rng.standard_normal(n_parent) * 0.005
        f_noisy = project_simplex(best_seq + noise)
        t_alloc = min(rem * 0.3, 150)
        print(f"  SG on noisy variant: t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(f_noisy, time_limit=t_alloc,
                                         top_k=40, temperature=5e-6,
                                         lr=0.003, momentum=0.88, avg_window=400)
        try_seq(a_opt, "sg_noisy")

    # === Phase 2: Different configurations ===
    print(f"\n=== Phase 2: Config Exploration (best={best_score:.6f}) ===")

    configs = [
        {"top_k": 100, "temp": 5e-6, "lr": 0.002, "mom": 0.90, "label": "wide"},
        {"top_k": 20, "temp": 1e-6, "lr": 0.002, "mom": 0.92, "label": "narrow"},
        {"top_k": 30, "temp": 2e-5, "lr": 0.005, "mom": 0.80, "label": "broad"},
        {"top_k": 200, "temp": 1e-7, "lr": 0.001, "mom": 0.93, "label": "ultra"},
    ]

    for cfg in configs:
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        if best_seq is None:
            break
        t_alloc = min(rem * 0.25, 120)
        if t_alloc < 10:
            continue

        # Use best as starting point with tiny perturbation
        noise = rng.standard_normal(n_parent) * 0.002
        f_init = project_simplex(best_seq + noise)

        print(f"  Config [{cfg['label']}]: t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(f_init, time_limit=t_alloc,
                                         top_k=cfg["top_k"], temperature=cfg["temp"],
                                         lr=cfg["lr"], momentum=cfg["mom"], avg_window=600)
        try_seq(a_opt, cfg["label"])

    # === Phase 3: Peak equalization ===
    print(f"\n=== Phase 3: Peak Equalization (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.5, 150)
            print(f"  Peak eq: t={t_alloc:.0f}s")
            a_opt, _ = peak_equalization_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_eq")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.6, 120)
            print(f"  Coord descent: t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [700, 800, 850, 900, 1024]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            if abs(target_n - n_parent) < 50:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 80)
            print(f"  n={target_n}: t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(f_interp, time_limit=t_alloc,
                                             top_k=max(30, target_n // 10),
                                             temperature=1e-5, lr=0.003,
                                             momentum=0.85, avg_window=400)
            try_seq(a_opt, f"n{target_n}")

    # === Phase 6: Micro-perturbation polish ===
    print(f"\n=== Phase 6: Micro Perturbation (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem * 0.5, 100)
            print(f"  Micro perturbation: t={t_alloc:.0f}s")
            a_opt, _ = micro_perturbation_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

    # === Phase 7: Final fine subgradient ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem * 0.4, 60)
            print(f"  Fine SG: t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc,
                                             top_k=n_best, temperature=1e-7,
                                             lr=0.0005, momentum=0.95, avg_window=1000)
            try_seq(a_opt, "fine_sg")

        rem = deadline - time.monotonic() - 15
        if rem > 8:
            t_alloc = min(rem * 0.5, 40)
            print(f"  Final coord descent: t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd_final")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 20)
            print(f"  Final micro: t={t_alloc:.0f}s")
            a_opt, _ = micro_perturbation_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
