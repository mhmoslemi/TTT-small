Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    return C


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix A for LP: A@g <= M means f*g <= M.
    Shape: (2n-1, n). A[k, j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, n):
    """Solve LP: max sum(g) s.t. f*g <= M (element-wise), g >= 0.
    Returns g normalized to sum 1, or None if infeasible."""
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def lse_optimize(f0, time_limit, beta_start=50.0, beta_end=10000.0, max_peaks=300):
    """LSE smoothed gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = lse_gradient_step(f, beta, max_peaks)
        conv_max = conv.max()
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad /= grad_norm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 8.0 / n)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.95
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 50:

                v = np.zeros(n)

                no_improve = 0

            beta *= 1.003
    
    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, n_trials_per_coord=5):

def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    return C


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse matrix A for LP constraint A@g <= M (i.e., f*g <= M element-wise)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M (elementwise), g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 12})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def fw_lp_improved(f0, time_limit, verbose=False):
    """Improved FW with LP oracle: tight M, multi-direction, adaptive step."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Try LP with progressively looser M
        best_dir = None
        best_dir_obj = best_obj
        best_dir_alpha = 0.0

        m_factors = [1.0, 1.002, 1.005, 1.01, 1.02, 1.04, 1.08]

                if val < best_dir_obj - 1e-15:
                    best_dir_obj = val
                    best_dir = g.copy()
                    best_dir_alpha = alpha

            # If we found improvement with tight M, no need for looser

            if best_dir is not None:

        # Fine-tune alpha with bisection around best_dir_alpha

def compute_lse_grad(f, beta, max_peaks):
    """Compute LSE-weighted gradient of max(conv) w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    conv_len = 2 * n - 1
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()

    if conv_len > max_peaks:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        sig_idx = sig_idx[w[sig_idx] > 1e-12 * w.max()]
    else:
        sig_idx = np.arange(conv_len)

    grad = np.zeros(n)
    for k in sig_idx:
        wk = w[int(k)]
        if wk < 1e-15:
            continue
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo > hi:
            continue

        # First part: m in [lo, hi], contribution = f[ki - m]
        ms1 = np.arange(lo, hi + 1)
        grad[ms1] += wk * f[ki - ms1]

        # Second part: m in [max(0,ki-hi), min(n-1,ki-lo)], contribution = f[ki - m]
        ms2_lo = max(0, ki - hi)
        ms2_hi = min(n - 1, ki - lo)
        if ms2_lo <= ms2_hi:
            ms2 = np.arange(ms2_lo, ms2_hi + 1)
            grad[ms2] += wk * f[ki - ms2]

    return grad, conv


def lse_optimize(f0, time_limit, beta_start=50.0, beta_end=20000.0, max_peaks=300):
    """LSE smoothed projected gradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 4.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = compute_lse_grad(f, beta, max_peaks)
        conv_max = conv.max()
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 10.0 / n)
        else:
            no_improve += 1
            if no_improve >= 4:
                lr *= 0.96
                momentum = max(0.82, momentum - 0.004)
            if no_improve > 40:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(4.0 / n, 0.002)
                beta = min(beta * 2.0, beta_end)
                momentum = 0.90
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj


def coord_descent_polish(f0, time_limit):
    """Polish via pairwise coordinate transfers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        n_pairs = max(80, n // 3)

        for _ in range(n_pairs):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue

            deltas = [0.002, 0.005, 0.01, 0.02, 0.04]
            for delta in deltas:
                if f[i] < delta:
                    continue
                f_new = f.copy()
                f_new[i] -= delta
                f_new[j] += delta
                conv_new = fast_convolve(f_new)
                obj_new = conv_new.max()
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new
                    improved = True
                    break

        if not improved:
            # Try smaller deltas and more pairs
            for _ in range(n_pairs // 2):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for delta in [0.001, 0.002, 0.003]:
                    if f[i] < delta:
                        continue
                    f_new = f.copy()
                    f_new[i] -= delta
                    f_new[j] += delta
                    conv_new = fast_convolve(f_new)
                    obj_new = conv_new.max()
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_f = f_new.copy()
                        f = f_new
                        improved = True
                        break

        if not improved:
            break

    return best_f, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
