        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0

            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 41)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                continue
            continue
        
        f = f_new
        new_M = fast_convolve(f).max()
        
        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj


def pairwise_descent(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    no_improve_passes = 0
    passes = 0
    
    # Precompute offsets for pair selection
    offsets = list(range(1, min(20, n)))
    if n > 50:
        offsets.extend([25, 50, 100, 200])
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        pass_improved = False
        order = rng.permutation(n)
        
        for idx_i in range(0, n, max(1, n // 200)):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx_i]
            fi = f[i]
            if fi < 1e-10:
                continue
            
            for offset in offsets:
                j_candidates = [(idx_i + offset) % n, (idx_i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    fj = f[j]
                    c = fi + fj
                    if c < 1e-10:
                        continue
                    
                    # Try several transfer ratios
                    current_t = fi / c
                    candidates = [current_t, 0.0, 1.0, 0.5,
                                  max(0, current_t - 0.3), min(1, current_t + 0.3),
                                  max(0, current_t - 0.1), min(1, current_t + 0.1)]
                    
                    best_t = current_t
                    best_val = best_M  # Use current global best as reference
                    
                    for t in candidates:
                        f_trial = f.copy()
                        f_trial[i] = t * c
                        f_trial[j] = (1 - t) * c
                        val = fast_convolve(f_trial).max()
                        if val < best_val:
                            best_val = val
                            best_t = t
                    
                    if best_val < best_M - 1e-13:
                        f[i] = best_t * c
                        f[j] = (1 - best_t) * c
                        best_M = best_val
                        best_f = f.copy()
                        pass_improved = True
                        fi = f[i]  # Update for next offset
        
        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
    
    return best_f, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP-FW with reduced constraint set (top-k binding lags)

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n, lags):
    A = np.zeros((len(lags), n))
    for i in range(len(lags)):
        k = int(lags[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[i, j_start:j_end + 1] = f[j_start:j_end + 1]
    return A


def lp_direction(f, M, n, top_k):
    conv = fast_convolve(f)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    A_ub = build_lp_matrix(f, n, peaks)
    b_ub = np.full(k_actual, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0), conv, peaks
    return None, conv, peaks

        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 150:
                rng = np.random.default_rng(it * 41)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                continue
            continue

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj


def pairwise_descent(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    no_improve_passes = 0
    passes = 0

    max_offset = min(n - 1, 50)
    offsets = list(range(1, max_offset + 1))

    while time.monotonic() - t0 < time_limit:
        passes += 1
        if time.monotonic() - t0 > time_limit:
            break

        pass_improved = False
        order = rng.permutation(n)

        step_size = max(1, n // 300)
        for idx_i in range(0, n, step_size):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx_i]
            fi = f[i]
            if fi < 1e-12:
                continue

            for offset in offsets:
                if offset >= n:
                    break
                j_fwd = order[(idx_i + offset) % n]
                j_bwd = order[(idx_i - offset) % n]
                for j in [j_fwd, j_bwd]:
                    if j == i:
                        continue
                    fj = f[j]
                    c = fi + fj
                    if c < 1e-12:
                        continue

                    current_t = fi / c
                    candidates = [current_t, 0.0, 1.0, 0.5,
                                  max(0.0, current_t - 0.25),
                                  min(1.0, current_t + 0.25),
                                  max(0.0, current_t - 0.1),
                                  min(1.0, current_t + 0.1)]

                    best_t = current_t
                    best_val = best_M

                    for t in candidates:
                        f_trial = f.copy()
                        f_trial[i] = t * c
                        f_trial[j] = (1.0 - t) * c
                        val = fast_convolve(f_trial).max()
                        if val < best_val:
                            best_val = val
                            best_t = t

                    if best_val < best_M - 1e-13:
                        f[i] = best_t * c
                        f[j] = (1.0 - best_t) * c
                        best_M = best_val
                        pass_improved = True
                        fi = f[i]

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

    return f, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_peak(f0, time_limit):
    """Targeted coordinate descent focusing on the peak lag."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 50:
            break

        conv = fast_convolve(f)
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]

        # Identify coordinates contributing to the peak
        j_start = max(0, peak_lag - n + 1)
        j_end = min(n - 1, peak_lag)
        contributors = np.arange(j_start, j_end + 1)
        products = f[contributors] * f[contributors + peak_lag - j_start]

        # Try reducing the largest product by shifting mass
        if len(contributors) == 0:
            break

        top_idx = np.argsort(-products)
        improved = False

        for ti in top_idx[:5]:
            j = contributors[ti]
            k = j + peak_lag - j_start
            if k >= n or j == k:
                continue
            c = f[j] + f[k]
            if c < 1e-12:
                continue

            best_t = f[j] / c
            best_val = best_M

            for t in [best_t, 0.0, 1.0, 0.2, 0.3, 0.5, 0.7, 0.8,
                      max(0, best_t - 0.2), min(1, best_t + 0.2)]:
                f_trial = f.copy()
                f_trial[j] = t * c
                f_trial[k] = (1 - t) * c
                val = fast_convolve(f_trial).max()
                if val < best_val:
                    best_val = val
                    best_t = t

            if best_val < best_M - 1e-13:
                f[j] = best_t * c
                f[k] = (1 - best_t) * c
                best_M = best_val
                improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return f, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)
    best_seq = [None]
    best_score = [float('inf')]

    def record(a, label):
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(sc) and sc < best_score[0]:
            best_score[0] = sc
            best_seq[0] = a.copy()
            print("  NEW BEST [{}]: {:.10f} (n={})".format(label, sc, len(a)))
        return sc

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = record(parent, "parent")
            print("Parent: n={}, score={:.10f}".format(len(parent), p_score))
    except NameError:
        pass
    except Exception as e:
        print("Error loading parent: {}".format(e))

    if parent is None:
        parent = np.ones(768) / 768
        record(parent, "default")
        print("Default parent: n={}, score={:.10f}".format(len(parent), best_score[0]))

    n_parent = len(parent)

    # Phase 1: LP-FW on parent
    print("\n=== Phase 1: LP-FW on parent (n={}) ===".format(n_parent))
    rem = deadline - time.monotonic() - 550
    t1 = min(rem, 350)
    if t1 > 30:
        print("  Time budget: {:.0f}s".format(t1))
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, top_k=80, verbose=False)
        record(a_opt, "lpfw_p1")

    # Phase 2: Adam-SG
    print("\n=== Phase 2: Adam-SG (best={:.10f}) ===".format(best_score[0]))
    if best_seq[0] is not None:
        rem = deadline - time.monotonic() - 450
        t2 = min(rem, 200)
        if t2 > 30:
            print("  Time budget: {:.0f}s".format(t2))
            a_opt, _ = adam_sg_optimize(best_seq[0], time_limit=t2)
            record(a_opt, "adam_p2")

    # Phase 3: Multi-length exploration
    print("\n=== Phase 3: Multi-length (best={:.10f}) ===".format(best_score[0]))
    base = best_seq[0] if best_seq[0] is not None else parent
    target_lengths = [512, 720, 800, 864, 1024, 1200]
    for target_n in target_lengths:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 250
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.5, 140)
        if t_alloc > 20:
            print("  LP-FW n={}, t={:.0f}s".format(target_n, t_alloc))
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, top_k=60)
            record(a_opt, "lpfw_n{}".format(target_n))
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc2 = min(rem * 0.4, 80)
            if t_alloc2 > 15:
                a_opt2, _ = adam_sg_optimize(a_opt, time_limit=t_alloc2)
                record(a_opt2, "adam_n{}".format(target_n))

    # Phase 4: Pairwise descent
    print("\n=== Phase 4: Pairwise (best={:.10f}) ===".format(best_score[0]))
    if best_seq[0] is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_pw = min(rem * 0.6, 250)
            print("  Time budget: {:.0f}s, n={}".format(t_pw, len(best_seq[0])))
            a_opt, _ = pairwise_descent(best_seq[0], time_limit=t_pw)
            record(a_opt, "pairwise")

    # Phase 5: Peak-targeted coordinate descent
    print("\n=== Phase 5: Peak CD (best={:.10f}) ===".format(best_score[0]))
    if best_seq[0] is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_pc = min(rem * 0.5, 100)
            a_opt, _ = coordinate_descent_peak(best_seq[0], time_limit=t_pc)
            record(a_opt, "peak_cd")

    # Phase 6: Alternating with diverse restarts
    print("\n=== Phase 6: Alternating + restarts (best={:.10f}) ===".format(best_score[0]))
    if best_seq[0] is not None:
        n_best = len(best_seq[0])
        for p in range(6):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem / max(1, 6 - p), 100)

            if p == 0:
                f_init = best_seq[0].copy()
            elif p == 1:
                rng = np.random.default_rng(200 + p)
                f_init = project_simplex(best_seq[0] + rng.standard_normal(n_best) * 0.005)
            elif p == 2:
                f_init = best_seq[0][::-1].copy()
                f_init = f_init / f_init.sum()
            elif p == 3:
                f_init = project_simplex(0.85 * best_seq[0] + 0.15 * (np.ones(n_best) / n_best))
            elif p == 4:
                rng = np.random.default_rng(400 + p)
                f_init = project_simplex(best_seq[0] + rng.standard_normal(n_best) * 0.008)
            else:
                rng = np.random.default_rng(600 + p)
                pert = rng.standard_normal(n_best)
                pert = pert - pert.mean()
                f_init = project_simplex(best_seq[0] + pert * 0.005)

            if p % 2 == 0:
                a_opt, _ = lp_fw_optimize(f_init, time_limit=min(t_alloc, 90), top_k=70)
            else:
                a_opt, _ = adam_sg_optimize(f_init, time_limit=min(t_alloc, 90))
            record(a_opt, "alt_{}".format(p))

    # Phase 7: Final LP-FW with more constraints
    print("\n=== Phase 7: Final LP-FW (best={:.10f}) ===".format(best_score[0]))
    if best_seq[0] is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_final = min(rem, 100)
            print("  Time budget: {:.0f}s".format(t_final))
            a_opt, _ = lp_fw_optimize(best_seq[0], time_limit=t_final, top_k=120)
            record(a_opt, "lpfw_final")

    # Phase 8: Final polish
    print("\n=== Phase 8: Final Polish (best={:.10f}) ===".format(best_score[0]))
    if best_seq[0] is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = pairwise_descent(best_seq[0], time_limit=min(rem, 50))
            record(a_opt, "pw_final")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq[0], time_limit=min(rem, 35), top_k=150)
            record(a_opt, "lpfw_polish")

    if best_seq[0] is None:
        best_seq[0] = np.ones(768) / 768
        best_score[0] = float(evaluate_sequence(best_seq[0].tolist()))

    fs = float(evaluate_sequence(best_seq[0].tolist()))
    if np.isfinite(fs):
        best_score[0] = fs

    print("\nFINAL SCORE: {:.10f} (n={})".format(best_score[0], len(best_seq[0])))
    return [float(x) for x in best_seq[0]]
