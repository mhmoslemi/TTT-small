import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_subgradient(a, conv, temperature):
    """Compute full softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[k] * a[k - idx]
    return sg


def compute_topk_subgradient(a, conv, top_k, temperature):
    """Compute top-k softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:

def adam_softmax_optimize(f0, time_limit, temperature=0.01, lr=0.01,
                          beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """
    Adam optimizer on the soft-max smoothed objective.
    Objective: 2*n*tau*logsumexp(conv(a,a)/tau)
    This is a smooth approximation to 2*n*max(conv(a,a)).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    cur_temp = temperature
    cur_lr = lr
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        # Soft-max weight
        scaled = conv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Gradient computation
        sg = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[k] * a[k - idx] if False else 2.0 * w[k] * f[k - idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v_hat = beta2 * v_hat + (1 - beta2) * sg ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat_hat) + eps)

        # Project to simplex
        f_new = project_simplex(f_new)
        if f_new.sum() < 1e-10:
            f_new = best_f.copy()
            continue
        f_new /= f_new.sum()
        f = f_new

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature: decrease slowly to sharpen
        cur_temp *= 0.9998
        if cur_temp < 0.001:
            cur_temp = 0.001

        # Learning rate decay
        cur_lr *= 0.9999
        if cur_lr < 0.0001:
            cur_lr = 0.0001

        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(t * 31 + 7)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            m = np.zeros(n)
            v_hat = np.zeros(n)
            no_improve = 0
            cur_lr = max(lr * 0.5, 0.001)

    return best_f, best_obj


def peak_targeting_search(f0, time_limit, verbose=False):
    """
    Local search that specifically targets the peak positions.
    For each peak, identifies the top contributing pairs and tries
    to reduce their product by redistributing mass.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()
        peak_pos = np.argmax(conv)

        # Get top contributing pairs at the peak
        js = max(0, peak_pos - n + 1)
        je = min(n - 1, peak_pos)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[peak_pos - idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-15:
            break

        # Find the best move: transfer mass from high contributor to low
        sorted_idx = idx[np.argsort(-contribs)]
        # Target positions: those with low contribution at peak
        low_positions = np.argsort(contribs)[:n // 4] + js

        improved_this_round = False
        trials = 200

        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            if improved_this_round:
                break

            # Pick source from top contributors
            src_candidates = sorted_idx[:max(5, n // 8)]
            src = int(rng.choice(src_candidates))

            # Pick target from low contributors or random
            if rng.random() < 0.7:
                tgt = int(rng.choice(low_positions))
            else:
                tgt = int(rng.integers(0, n))

            if src == tgt or f[src] < 1e-10:
                continue

            # Try different transfer amounts
            for frac in [0.05, 0.1, 0.2, 0.35]:
                amt = f[src] * frac
                if amt < 1e-12:
                    break

                old_src = f[src]
                old_tgt = f[tgt]
                f[src] = old_src - amt
                f[tgt] = old_tgt + amt

                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved_this_round = True
                    break
                else:
                    f[src] = old_src
                    f[tgt] = old_tgt

        if not improved_this_round:
            # Try swapping: find i, j where swapping reduces peak
            moved = False
            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                # Swap some mass
                amt = min(f[i], f[j]) * 0.3
                if amt < 1e-12:
                    continue
                f[i] -= amt
                f[j] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    moved = True
                    break
                else:
                    f[i] += amt
                    f[j] -= amt

            if not moved:
                # More aggressive: random perturbation + accept
                noise = rng.standard_normal(n) * 0.001
                f_trial = project_simplex(f + noise)
                val = fast_convolve(f_trial).max()
                if val < best_obj * 1.0001:  # accept slightly worse to explore
                    f = f_trial
                else:
                    f = best_f.copy()
                    if time.monotonic() - t0 > time_limit * 0.8:
                        break

    return best_f, best_obj


def coordinate_descent_v2(f0, time_limit, verbose=False):
    """Improved coordinate descent with better candidate values."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Order: start from peak contributors
    conv = fast_convolve(f)
    peak_pos = int(np.argmax(conv))
    js = max(0, peak_pos - n + 1)
    je = min(n - 1, peak_pos)
    idx = np.arange(js, je + 1)
    contribs = f[idx] * f[peak_pos - idx]

    # Sort by contribution (high first)
    order = list(idx[np.argsort(-contribs)])
    remaining = [i for i in range(n) if i not in set(order)]
    order.extend(remaining)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.3, 0.5, 0.7, 1.3, 1.6, 2.0, 0.8, 1.2]:
                new_val = old_val * mult
                if new_val < 1e-12 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 5.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 80)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Raised cosine variant
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())

    # Bipolar-like (absolute value of symmetric pattern)
    t = np.arange(n)
    pattern = np.abs(np.sin(2 * np.pi * t / n * (1 + rng.integers(1, 4))))
    pattern = np.maximum(pattern, 1e-10)
    inits.append(pattern / pattern.sum())

    return inits[:count]


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining multiplicative weights, Adam soft-max,
    peak-targeting local search, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")

    # === Phase 1: Multiplicative weights (multiple configs) ===
    print("\n=== Phase 1: Multiplicative-Weights Subgradient ===")
    n_parent = len(parent)

    candidates = [parent]
    # Add interpolation to nearby lengths
    for dl in [-64, 64, 128]:
        if n_parent + dl >= 128:
            candidates.append(interpolate_sequence(parent, n_parent + dl))

    configs = [
        {"top_k": max(20, n_parent // 8), "temp": 0.00004, "lr": 0.003, "mom": 0.85, "anneal": 0.0},
        {"top_k": max(30, n_parent // 5), "temp": 0.0001, "lr": 0.004, "mom": 0.80, "anneal": 0.02},
        {"top_k": max(15, n_parent // 10), "temp": 0.00002, "lr": 0.002, "mom": 0.90, "anneal": 0.0},
        {"top_k": max(40, n_parent // 4), "temp": 0.0002, "lr": 0.005, "mom": 0.75, "anneal": 0.03},
    ]

    for idx, cand in enumerate(candidates[:3]):
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 200)
        if t_alloc < 15:
            t_alloc = min(rem * 0.3, 100)
        cfg = configs[idx % len(configs)]
        n_cand = len(cand)
        print(f"  MU cand {idx}: n={n_cand}, t={t_alloc:.0f}s, cfg={cfg}")
        a_opt, _ = mu_weights_optimize(cand, time_limit=t_alloc,
                                        top_k=cfg["top_k"], temperature=cfg["temp"],
                                        lr=cfg["lr"], momentum=cfg["mom"],
                                        annealing_prob=cfg["anneal"])
        try_seq(a_opt, f"mu_{idx}")

    # Continue MU on best
    if best_seq is not None:
        rem = deadline - time.monotonic() - 450
        if rem > 30:
            t_alloc = min(rem * 0.3, 200)
            print(f"  MU deep: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(best_seq, time_limit=t_alloc,
                                            top_k=max(20, len(best_seq) // 6),
                                            temperature=0.00003, lr=0.002,
                                            momentum=0.88, annealing_prob=0.01)
            try_seq(a_opt, "mu_deep")

    # === Phase 2: Adam soft-max optimization ===
    print(f"\n=== Phase 2: Adam Soft-Max (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.35, 200)
            print(f"  Adam soft-max: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = adam_softmax_optimize(best_seq, time_limit=t_alloc,
                                              temperature=0.05, lr=0.02)
            try_seq(a_opt, "adam_sm_1")

            rem = deadline - time.monotonic() - 350
            if rem > 20:
                t_alloc = min(rem * 0.3, 150)
                a_opt, _ = adam_softmax_optimize(best_seq, time_limit=t_alloc,
                                                  temperature=0.02, lr=0.01,
                                                  beta1=0.85, beta2=0.99)
                try_seq(a_opt, "adam_sm_2")

    # === Phase 3: Peak-targeting local search ===
    print(f"\n=== Phase 3: Peak-Targeting Search (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            t_alloc = min(rem * 0.4, 200)
            print(f"  Peak search: t={t_alloc:.0f}s")
            a_opt, _ = peak_targeting_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_search")

            # Another round with the improved result
            rem = deadline - time.monotonic() - 300
            if rem > 15:
                t_alloc = min(rem * 0.3, 120)
                a_opt, _ = peak_targeting_search(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "peak_search_2")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 15:
            t_alloc = min(rem * 0.5, 150)
            print(f"  Coord descent: t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_v2(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd_v2")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = []
        for tl in [512, 640, 768, 896, 1024, 1152, 1280]:
            if tl != n_best and tl >= 256:
                target_lengths.append(tl)

        # Sort by proximity to current best length
        target_lengths.sort(key=lambda x: abs(x - n_best))
        target_lengths = target_lengths[:4]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 150)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(f_interp, time_limit=t_alloc,
                                            top_k=max(20, target_n // 6),
                                            temperature=0.00004, lr=0.003,
                                            momentum=0.87)
            try_seq(a_opt, f"len_{target_n}")

            # Quick peak search on the result
            rem = deadline - time.monotonic() - 180
            if rem > 10:
                a_opt, _ = peak_targeting_search(a_opt, time_limit=min(rem, 40))
                try_seq(a_opt, f"len_{target_n}_ps")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = 5
        for r in range(restarts):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem * 0.2, 80)

            # Different perturbation strategies
            if r == 0:
                noise = rng.standard_normal(n_best) * 0.01
                f_init = project_simplex(best_seq + noise)
            elif r == 1:
                # Flip part of the sequence
                f_init = best_seq.copy()
                half = n_best // 2
                f_init[half:] = f_init[half:][::-1] * 0.5
                f_init = project_simplex(f_init)
            elif r == 2:
                # Mix with uniform
                f_init = (1 - 0.3 * (r - 1)) * best_seq + 0.3 * (r - 1) * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            elif r == 3:
                # Frequency domain perturbation
                F = np.fft.rfft(f_best if False else best_seq)
                angle_noise = rng.standard_normal(len(F)) * 0.05
                F_pert = F * np.exp(1j * angle_noise)
                f_init = np.real(np.fft.irfft(F_pert, n_best))
                f_init = np.maximum(f_init, 0)
                fs = f_init.sum()
                if fs > 1e-10:
                    f_init /= fs
                else:
                    f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            else:
                # Aggressive random
                f_init = np.abs(rng.standard_normal(n_best))
                f_init = f_init + 0.1 * best_seq
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(f_init, time_limit=t_alloc,
                                            top_k=max(15, n_best // 8),
                                            temperature=0.00005, lr=0.004,
                                            momentum=0.82, annealing_prob=0.02)
            try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem * 0.4, 60)
            a_opt, _ = mu_weights_optimize(best_seq, time_limit=t_alloc,
                                            top_k=n_best, temperature=0.000001,
                                            lr=0.0002, momentum=0.97, annealing_prob=0.0)
            try_seq(a_opt, "sg_ultra_fine")

        # Peak search on ultra-fine
        rem = deadline - time.monotonic() - 25
        if rem > 8:
            a_opt, _ = peak_targeting_search(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "ps_ultra_fine")

        # Tiny coordinate nudges
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            a_opt, _ = coordinate_descent_v2(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "cd_ultra_fine")

        # One more Adam pass
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt, _ = adam_softmax_optimize(best_seq, time_limit=min(rem, 15),
                                              temperature=0.005, lr=0.005)
            try_seq(a_opt, "adam_ultra_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
