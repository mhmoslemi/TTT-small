Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smooth-max
Nesterov acceleration, coordinate descent, and multi-length exploration.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(conv, alpha):
    """Compute gradient of smooth max of conv w.r.t. the input sequence a."""
    # softmax weights
    shifted = conv * alpha - conv.max() * alpha
    weights = np.exp(shifted)
    ws = weights.sum()
    if ws < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= ws
    return weights


def compute_subgrad(a, weights, n=None):
    """Compute subgradient of weighted sum of conv values."""
    if n is None:
        n = len(a)
    nconv = 2 * n - 1
    sg = np.zeros(n)
    for k in range(nconv):
        if weights[k] < 1e-15:
            continue
        # d conv[k]/d a[m] = 2*a[k-m] if 0 <= k-m < n
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * weights[k] * a[k - indices]
    return sg


def tilted_init(n, tilt=0.5, seed=42):
    """Generate a tilted (decreasing) initialization."""
    j = np.arange(n)
    # Linear tilt from 1+tilt to 1-tilt
    a = 1.0 + tilt * (1.0 - 2.0 * j / (n - 1))
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def exponential_init(n, rate=0.5, seed=42):
    """Generate an exponentially decaying initialization."""
    j = np.arange(n)
    a = np.exp(-rate * j / n)
    a = a / a.sum()
    return a


def sine_tilt_init(n, phase=0.3, seed=42):
    """Generate a sine-modulated tilted initialization."""
    j = np.arange(n)
    t = j / (n - 1)
    a = 1.0 + phase * (1.0 - 2.0 * t) + 0.3 * phase * np.sin(np.pi * t)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def nesterov_smooth_max(a0, time_limit, alpha_init=50.0, alpha_max=5000.0,
                         lr0=0.01, mom0=0.9, verbose=False):
    """Nesterov-accelerated gradient descent on smooth max of convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    conv = fast_convolve(a)
    cur_alpha = alpha_init
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr0
    cur_mom = mom0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute smooth max gradient
        weights = smooth_max_grad(conv, cur_alpha)
        sg = compute_subgrad(a, weights, n)

        # Subtract mean to ensure sum-preserving direction (approximate)
        sg_centered = sg - sg.mean()

        # Nesterov update
        v = cur_mom * v - cur_lr * sg_centered
        y_new = project_simplex(y + v)
        a_new = y_new
        y = a_new + cur_mom * (a_new - a)
        y = project_simplex(y)
        a = a_new

        conv = fast_convolve(a)
        cur_obj = conv.max()

        # Adapt alpha: increase when making progress
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            cur_alpha = min(alpha_max, cur_alpha * 1.005)
            if verbose and it % 100 == 0:
                print(f"    NS iter {it}: obj={best_obj:.10f}, alpha={cur_alpha:.1f}")
        else:
            no_improve += 1
            cur_alpha = max(alpha_init, cur_alpha * 0..999)

        # Decay learning rate
        cur_lr *= 0.9998
        cur_mom = min(0.98, cur_mom * 1.0001)

        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.005
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            y = a.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.5, 0.002)
            cur_alpha = alpha_init
            no_improve = 0
            conv = fast_convolve(a)

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_inner=15):
    """Coordinate descent with inner line search on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_pass_improve = 0
    pass_count = 0

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j
If old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section search
            lo = 0.0
            hi = min(0.9999, 3.0 * old_val)

            def eval_at(x):
                ft = a.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(n_inner):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = best_obj  # approximate

            if opt_val < cur_val - 1e-14:
                ft = a.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = np.maximum(ft, 0.0)
                ss = ft.sum()
                if ss > 1e-15:
                    ft /= ss
                a = ft
                pass_improved = True
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)

    return f_interp
