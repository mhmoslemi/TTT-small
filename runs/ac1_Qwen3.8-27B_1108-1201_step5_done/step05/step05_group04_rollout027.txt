```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LSE gradient with Nesterov momentum
2. Backtracking line search on the simplex
3. Multi-length exploration with population restarts
4. Adaptive peak targeting and final polish
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project onto {x : sum x = target_sum, x >= 0}."""
    n = len(v)
    # Project v onto simplex with sum = target_sum
    # First project onto sum = 1, then scale
    if n == 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_fft(f, w):
    """Compute grad of sum_k w_k * c_k where c = autocorr(f).
    grad_j = 2 * sum_l f[l] * w[j+l]
    """
    n = len(f)
    conv_len = 2 * n - 1
    f_rev = f[::-1]
    fft_size = 1 << (n + conv_len - 1).bit_length()
    F = np.fft.rfft(f_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(F * W, fft_size)
    return 2.0 * conv[n-1:2*n-1]

def compute_gradient_sparse(f, w, top_k=50):
    """Sparse gradient: only use top_k peaks."""
    n = len(f)
    conv_len = 2 * n - 1
    idx = np.argpartition(w, -min(top_k, conv_len))[-min(top_k, conv_len):]
    idx = idx[w[idx] > 1e-18]
    grad = np.zeros(n)
    for k in idx:
        wk = w[k]
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += 2.0 * wk * f[k - ms]
    return grad

def lse_projected_grad(f0, time_limit, beta_start=5.0, beta_end=20000.0, seed=0):
    """
    Projected gradient descent with LSE smoothing, Nesterov momentum,
    and backtracking line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 1.0 / n  # initial learning rate
    momentum = 0.9
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Compute LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Compute gradient
        if beta > 100:
            top_k = max(10, int(500.0 / beta))
            grad = compute_gradient_sparse(f, w, top_k=top_k)
        else:
            grad = compute_gradient_fft(f, w)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm  # normalize for momentum
        
        # Nesterov step
        v = momentum * v - lr * grad
        f_trial = project_simplex(f + v)
        
        # Backtracking line search
        obj_trial = fast_convolve(f_trial).max()
        alpha = 1.0
        best_alpha = 0.0
        best_obj_trial = obj_trial
        
        if obj_trial < best_obj:
            best_alpha = 1.0
            best_obj_trial = obj_trial
            # Try to push further (already accepted)
        else:
            # Need to backtrack
            for bt in range(15):
                alpha *= 0.7
                f_bt = project_simplex(f + alpha * v)
                obj_bt = fast_convolve(f_bt).max()
                if obj_bt < best_obj - 1e-13:
                    best_alpha = alpha
                    best_obj_trial = obj_bt
                    break
                if obj_bt < best_obj_trial:
                    best_obj_trial = obj_bt
                    best_alpha = alpha
        
        if best_alpha > 1e-6:
            f_new = project_simplex(f + best_alpha * v)
            obj_new = fast_convolve(f_new).max()
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
                # Gradually increase learning rate if improving
                lr = min(lr * 1.002, 3.0 / n)
            else:
                no_improve += 1
                # Update f to the trial even if not improving (for momentum)
                if best_obj_trial < obj_trial + 1e-12:
                    f = project_simplex(f + best_alpha * v)
                    obj_new_f = fast_convolve(f).max()
                    if obj_new_f < best_obj:
                        best_obj = obj_new_f
                        best_f = f.copy()
                        no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive beta increase
        if no_improve < 30:
            beta *= 1.005
        elif no_improve < 100:
            beta *= 1.02
        else:
            beta *= 1.05
        beta = min(beta, beta_end)
        
        # Decay momentum
        momentum = max(0.85, momentum - 0.0001 * no_improve)
        
        # Restart logic
        if no_improve > 100:
            # Perturb around best
            scale = 0.001 + 0.003 * min(1.0, (no_improve - 100) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = max(1.0 / n, 0.001)
            beta = min(beta * 0.5, 200.0)  # reset beta to explore
            momentum = 0.9
            no_improve = 0
        
        # Periodic re-evaluation
        if it % 200 == 0:
            conv_best = fast_convolve(best_f)
            best_obj = conv_best.max()
    
    # Final polish with very high beta
    return best_f, best_obj
```

```python
import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 1:
        return np.array([max(target_sum, 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(f, w):
    """Compute grad_j = 2*sum_l f[l]*w[j+l] via FFT convolution."""
    n = len(f)
    conv_len = 2 * n - 1
    f_rev = f[::-1]
    total_len = n + conv_len - 1
    fft_size = 1 << (total_len - 1).bit_length()
    F = np.fft.rfft(f_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(F * W, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def compute_gradient_sparse(f, w, top_k=50):
    """Sparse gradient using only top_k significant weights."""
    n = len(f)
    conv_len = 2 * n - 1

        wk = w[int(k)]
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += 2.0 * wk * f[ki - ms]
    return grad

        if t_alloc > 25:
            print(f"  LSE reversed n={n_parent}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize(f_rev, time_limit=t_alloc,
                                    beta_start=10.0, beta_end=5000.0, seed=seed + 999)
            try_sequence(f_opt, best_tracker)

    # === Phase 2: Focused LSE on best ===
    print(f"\n=== Phase 2: Focused LSE optimization ===")
    t2_end = deadline - 450
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    t_alloc = min(350, t2_end - time.monotonic())
    if t_alloc > 60:
        print(f"  LSE n={n_cur}, t={t_alloc:.0f}s, beta_end=30000")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                beta_start=5.0, beta_end=30000.0, seed=seed + 1000)
        try_sequence(f_opt, best_tracker)

    # Second run with different init (slight perturbation)
    current = best_tracker[1] if best_tracker[1] is not None else current
    rng = np.random.default_rng(seed + 7777)
    f_pert = project_simplex(current + rng.standard_normal(len(current)) * 0.003)
    t_alloc = min(150, t2_end - time.monotonic())
    if t_alloc > 40:
        print(f"  LSE perturbed n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize(f_pert, time_limit=t_alloc,
                                beta_start=15.0, beta_end=20000.0, seed=seed + 2000)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Peak-directed refinement ===
    print(f"\n=== Phase 3: Peak-directed refinement ===")
    t3_end = deadline - 200
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(120, t3_end - time.monotonic())
    if t_alloc > 30:
        print(f"  Peak-directed n={len(current)}, t={t_alloc:.0f}s, peaks=5")
        f_opt, _ = peak_directed(current, time_limit=t_alloc, n_peaks=5, seed=seed + 3000)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(80, t3_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Peak-directed n={len(current)}, t={t_alloc:.0f}s, peaks=3")
        f_opt, _ = peak_directed(current, time_limit=t_alloc, n_peaks=3, seed=seed + 4000)
        try_sequence(f_opt, best_tracker)

    # Another LSE round after peak targeting
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(80, t3_end - time.monotonic())
    if t_alloc > 25:
        print(f"  LSE post-peak n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                beta_start=100.0, beta_end=50000.0, seed=seed + 5000)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    # Coordinate polish
    t_alloc = min(60, deadline - time.monotonic() - 60)
    if t_alloc > 15:
        print(f"  Coordinate polish n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_polish(current, time_limit=t_alloc, seed=seed + 6000)
        try_sequence(f_opt, best_tracker)

    # Very high beta LSE
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(80, deadline - time.monotonic() - 30)
    if t_alloc > 20:
        print(f"  High-beta LSE n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                beta_start=2000.0, beta_end=100000.0, seed=seed + 7000)
        try_sequence(f_opt, best_tracker)

    # Try nearby lengths
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        for off in [32, -32, 64, -64, 128, -128]:
            n_try = n_best + off
            if n_try < 256 or n_try > 1500:
                continue
            t_remaining = deadline - time.monotonic() - 15
            if t_remaining < 30:
                break
            t_alloc = min(50, t_remaining * 0.4)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Nearby n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize(f_init, time_limit=t_alloc,
                                    beta_start=50.0, beta_end=10000.0, seed=seed + n_try)
            try_sequence(f_opt, best_tracker)

    # Final coordinate polish
    if best_tracker[1] is not None:
        t_remaining = deadline - time.monotonic() - 5
        if t_remaining > 10:
            f_opt, _ = coordinate_polish(best_tracker[1], time_limit=t_remaining, seed=seed + 8000)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
```
