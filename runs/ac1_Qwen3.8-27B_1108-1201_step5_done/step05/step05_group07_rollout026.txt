def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if not len(rho):
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-guided optimization,
fast pair-based local search with incremental convolution updates,
simulated annealing for escaping local optima, and multi-scale exploration.
Key innovations: (1) O(n) incremental convolution updates for fast pair search,
(2) simulated annealing acceptance in local search, (3) scipy linprog LP direction
finding on the banded convolution matrix, (4) multi-scale coarse-to-fine strategy.

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if not len(rho):
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


class FastConvOL:
    """Maintains a sequence and its auto-convolution with fast incremental updates."""

    def __init__(self, a):
        self.a = np.maximum(a.astype(np.float64), 0.0).copy()
        s = self.a.sum()
        if s > 1e-15:
            self.a /= s
        self.n = len(self.a)
        self.conv = fast_convolve(self.a)
        self.max_val = float(self.conv.max())

    def set_a(self, a):
        self.a = np.maximum(a.astype(np.float64), 0.0).copy()
        s = self.a.sum()
        if s > 1e-15:
            self.a /= s
        self.n = len(self.a)
        self.conv = fast_convolve(self.a)
        self.max_val = float(self.conv.max())

    def change_one(self, i, new_val):
        """Change a[i] to new_val and update convolution incrementally."""
        old_val = self.a[i]
        delta = new_val - old_val
        if abs(delta) < 1e-15:
            return
        n = self.n
        self.a[i] = new_val
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo > k_hi:
            self.max_val = float(self.conv.max())
            return
        ks = np.arange(k_lo, k_hi + 1)
        mis = ks - i
        updates = 2.0 * delta * self.a[mis]
        if k_lo <= 2 * i <= k_hi:
            idx_2i = 2 * i - k_lo
            updates[idx_2i] = delta * (new_val + old_val)
        self.conv[k_lo:k_hi + 1] += updates
        np.maximum(self.conv, 0.0, out=self.conv)
        self.max_val = float(self.conv.max())

    def change_pair(self, i, j, new_i, new_j):
        """Change a[i] and a[j] simultaneously."""
        old_i = self.a[i]
        old_j = self.a[j]
        delta_i = new_i - old_i
        delta_j = new_j - old_j
        n = self.n

        if abs(delta_i) < 1e-15 and abs(delta_j) < 1e-15:
            return

        self.a[i] = new_i
        self.a[j] = new_j

        # Update from change at i
        k_lo_i = max(0, i)
        k_hi_i = min(2 * n - 2, i + n - 1)
        ks_i = np.arange(k_lo_i, k_hi_i + 1)
        mis_i = ks_i - i
        upd_i = 2.0 * delta_i * self.a[mis_i]
        if k_lo_i <= 2 * i <= k_hi_i:
            idx = 2 * i - k_lo_i
            upd_i[idx] = delta_i * (new_i + old_i)

        # Update from change at j
        k_lo_j = max(0, j)
        k_hi_j = min(2 * n - 2, j + n - 1)
        ks_j = np.arange(k_lo_j, k_hi_j + 1)
        mis_j = ks_j - j
        upd_j = 2.0 * delta_j * self.a[mis_j]
        if k_lo_j <= 2 * j <= k_hi_j:
            idx = 2 * j - k_lo_j
            upd_j[idx] = delta_j * (new_j + old_j)

        self.conv[k_lo_i:k_hi_i + 1] += upd_i
        self.conv[k_lo_j:k_hi_j + 1] += upd_j
        np.maximum(self.conv, 0.0, out=self.conv)
        self.max_val = float(self.conv.max())

    def try_pair_optimize(self, i, j, fracs):
        """Try different splits of a[i]+a[j] between i and j. Return improvement."""
        old_i = self.a[i]
        old_j = self.a[j]
        total = old_i + old_j
        if total < 1e-12:
            return 0.0

        best_obj = self.max_val
        best_frac = old_i / total

        for frac in fracs:
            new_i = total * frac
            new_j = total * (1.0 - frac)
            self.change_pair(i, j, new_i, new_j)
            if self.max_val < best_obj - 1e-14:
                best_obj = self.max_val
                best_frac = frac
            else:
                self.change_pair(i, j, old_i, old_j)

        if abs(best_frac - old_i / total) > 1e-10:
            new_i = total * best_frac
            new_j = total * (1.0 - best_frac)
            self.change_pair(i, j, new_i, new_j)
            return best_obj

        self.change_pair(i, j, old_i, old_j)
        return self.max_val


def mu_weights_optimize(f0, time_limit, top_k=50, temperature=0.00005, lr=0.003,
                        momentum=0.85, smooth_lambda=0.0):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum
    and optional L2 smoothing of the convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 < 0:
            break
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-13:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[ii] * f_nest[k - idx]

        if smooth_lambda > 1e-15:
            # Add gradient of sum(conv^2) for smoothing
            conv_flipped = conv_nest[::-1]
            fft_sz = 1 << (3 * n - 1).bit_length()
            A_f = np.fft.rfft(f_nest, fft_sz)
            C_f = np.fft.rfft(conv_flipped, fft_sz)
            corr = np.fft.irfft(A_f * C_f, fft_sz)[:3 * n - 2]
            sg_sq = np.zeros(n)
            for ii in range(n):
                idx2 = 2 * n - 2 - ii
                if 0 <= idx2 < len(corr):
                    sg_sq[ii] = 4.0 * corr[idx2]
            sg += smooth_lambda * sg_sq

        v = momentum * v + (1 - momentum) * sg
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_temp *= 1.01

        if no_improve > 400:
            rng = np.random.default_rng((it * 6271 + 17) % 1000003)
            noise_scale = 0.002 + 0.001 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

    return best_f, best_obj


def lp_direction(a, M, max_time=10.0):
    """
    Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (normalized to sum 1) or None if infeasible/too slow.
    """
    n = len(a)
    t0 = time.monotonic()

    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[j])

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': max_time, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
                return g
    except Exception:
        pass

    if time.monotonic() - t0 > max_time:
        return None
    return None


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 100)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Multi-lobe
    n_lobes = rng.integers(2, 5)
    centers = np.linspace(0.1, 0.9, n_lobes)
    multi = np.zeros(n)
    for c in centers:
        multi += np.exp(-0.5 * ((np.arange(n) - c * n) / (n / (2 * n_lobes))) ** 2)
    multi = np.maximum(multi, 1e-10)
    inits.append(multi / multi.sum())

    return inits[:count]


def annealed_pair_search(f0, time_limit, T0=0.01, T_min=0.0001, cooldown=0.9995,
                         n_fracs=12):

    Simulated annealing pair search with incremental convolution updates.
    Focuses on pairs that contribute to the current maximum convolution peaks.

    n = fc.n

    best_obj = fc.max_val
    t0 = time.monotonic()
    T = T0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it_count += 1

        contrib_set = set()
        for k in top_peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                if fc.a[j] > 1e-8:
                    contrib_set.add(j)

        contrib_list = list(contrib_set)

        if len(contrib_list) < 2:
            # Fall back to random
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            while j == i:
                j = int(rng.integers(0, n))
            fc.try_pair_optimize(i, j, fracs)
        else:

                if time.monotonic() - t0 > time_limit:
                    break

                j = contrib_list[rng.integers(0, len(contrib_list))]
                if i == j:
                    continue

                old_obj = fc.max_val
                fc.try_pair_optimize(i, j, fracs)
                new_obj = fc.max_val

                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = fc.a.copy()
                elif new_obj > old_obj + 1e-13:

                        fc.max_val = float(fc.conv.max())
                        # Actually we can't easily revert pair changes,
                        # so just continue

        # Occasionally try fully random pairs

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:

                if fc.max_val < best_obj - 1e-13:
                    best_obj = fc.max_val
                    best_a = fc.a.copy()

        # Cool down
        T *= cooldown
        if T < T_min:
            T = T_min

        # Periodic full re-convolution check (fix numerical drift)
        if it_count % 500 == 0:
            fc.conv = fast_convolve(fc.a)
            fc.max_val = float(fc.conv.max())
            if fc.max_val < best_obj - 1e-13:
                best_obj = fc.max_val

        # Periodic random perturbation for exploration

            fc.a = project_simplex(fc.a + noise)
            fc.conv = fast_convolve(fc.a)
            fc.max_val = float(fc.conv.max())

    return best_a, best_obj

                    new_j = total * frac
                    new_t = total * (1.0 - frac)
                    fc.change_pair(j, int(t), new_j, new_t)
                    if fc.max_val < best_j_obj - 1e-14:
                        best_j_obj = fc.max_val
                        best_target = int(t)
                        best_frac = frac
                    else:
                        fc.change_pair(j, int(t), old_val, fc.a[t] + (total - old_val - fc.a[t]) + old_val - fc.a[t])
                        # Revert properly
                        fc.change_pair(j, int(t), old_val, total - old_val)

            if best_target >= 0 and best_j_obj < best_obj - 1e-13:
                total = old_val + fc.a[best_target]
                fc.change_pair(j, best_target, total * best_frac, total * (1 - best_frac))
                best_obj = fc.max_val
                best_a = fc.a.copy()
                improved = True

        if not improved:
            # Perturb and try again
            pert = np.random.default_rng(int(time.monotonic() * 1e9) % 100000)
            noise = pert.standard_normal(n) * 0.002
            fc.a = project_simplex(fc.a + noise)
            fc.conv = fast_convolve(fc.a)
            fc.max_val = float(fc.conv.max())

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining MW subgradient, LP-guided direction,
    annealed pair search, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform n=768")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 1: Coarse MW optimization on diverse candidates ===
    print("\n=== Phase 1: Coarse MW Subgradient ===")

    candidates = [parent]
    n_parent = len(parent)

    # Structured inits at parent length
    candidates.extend(structured_inits(n_parent, rng, count=3))

    # Different lengths
    for n_try in [640, 1024]:
        if n_try != n_parent:
            candidates.append(interpolate_sequence(parent, n_try))
            candidates.extend(structured_inits(n_try, rng, count=2))

    n_cands = len(candidates)
    t_per_cand = min(60.0, 400.0 / max(n_cands, 1))

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - 600
        if rem < 15:
            break
        t_alloc = min(t_per_cand, rem * 0.15)
        if t_alloc < 5:
            t_alloc = 5

        n_c = len(cand)
        print(f"  Cand {idx}: n={n_c}, t={t_alloc:.0f}s")
        a_opt, _ = mu_weights_optimize(
            cand, time_limit=t_alloc,
            top_k=max(20, n_c // 8),
            temperature=0.00003, lr=0.004,
            momentum=0.87, smooth_lambda=0.01
        )
        try_seq(a_opt, f"mw_{idx}")

    # === Phase 2: Deep MW with different configs ===
    print(f"\n=== Phase 2: Deep MW (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {"top_k": max(30, n_best // 4), "temp": 0.00001, "lr": 0.001, "mom": 0.92, "sl": 0.05},
            {"top_k": max(10, n_best // 12), "temp": 0.000005, "lr": 0.0005, "mom": 0.95, "sl": 0.0},
            {"top_k": max(50, n_best // 2), "temp": 0.0001, "lr": 0.006, "mom": 0.80, "sl": 0.02},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 500
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            else:
                f_init = project_simplex(0.7 * best_seq + 0.3 * best_seq[::-1])

            print(f"  Deep {p}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(
                f_init, time_limit=t_alloc,
                top_k=cfg["top_k"], temperature=cfg["temp"],
                lr=cfg["lr"], momentum=cfg["mom"], smooth_lambda=cfg["sl"]
            )
            try_seq(a_opt, f"deep_{p}")

    # === Phase 3: LP-guided direction ===
    print(f"\n=== Phase 3: LP Direction (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try LP at the best sequence
        fc_tmp = FastConvOL(best_seq)
        M = fc_tmp.max_val

        rem = deadline - time.monotonic() - 400
        if rem > 30 and n_best <= 2048:
            print(f"  Solving LP (n={n_best}, M={M:.6f})...")
            t_lp_start = time.monotonic()
            g = lp_direction(best_seq, M, max_time=min(rem * 0.3, 60.0))
            t_lp = time.monotonic() - t_lp_start
            print(f"  LP took {t_lp:.1f}s")

            if g is not None:
                # Try different step sizes
                for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    f_new = project_simplex((1 - t) * best_seq + t * g)
                    try_seq(f_new, f"lp_t{t}")

                # Also try the LP direction as initialization for MW
                rem = deadline - time.monotonic() - 300
                if rem > 20:
                    a_opt, _ = mu_weights_optimize(
                        g, time_limit=min(rem * 0.3, 60),
                        top_k=max(20, n_best // 6),
                        temperature=0.00002, lr=0.002,
                        momentum=0.90, smooth_lambda=0.01
                    )
                    try_seq(a_opt, "lp_mw")

    # === Phase 4: Annealed pair search ===
    print(f"\n=== Phase 4: Annealed Pair Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.5, 300)
            print(f"  Pair search: n={n_best}, t={t_alloc:.0f}s")
            a_opt, obj = annealed_pair_search(
                best_seq, time_limit=t_alloc,
                T0=0.005, T_min=0.00005, cooldown=0.9997, n_fracs=10
            )
            try_seq(a_opt, "anneal_pair")

        # Second round with different params
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.5, 200)
            print(f"  Pair search 2: t={t_alloc:.0f}s")
            a_opt, obj = annealed_pair_search(
                best_seq, time_limit=t_alloc,
                T0=0.002, T_min=0.0001, cooldown=0.9993, n_fracs=14
            )
            try_seq(a_opt, "anneal_pair2")

    # === Phase 5: Fine coordinate descent ===
    print(f"\n=== Phase 5: Fine Coord Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.6, 150)
            print(f"  Coord descent: t={t_alloc:.0f}s")
            a_opt, _ = fine_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [512, 896, 1280, 1536]:
            if target_n == len(best_seq):
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 80)
            print(f"  n={target_n}, t={t_alloc:.0f}s")

            a_opt, _ = mu_weights_optimize(
                f_interp, time_limit=t_alloc * 0.6,
                top_k=max(20, target_n // 6),
                temperature=0.00003, lr=0.003,
                momentum=0.88, smooth_lambda=0.02
            )
            # Then pair search
            rem2 = deadline - time.monotonic() - 70
            if rem2 > 15:
                a_opt2, _ = annealed_pair_search(
                    a_opt, time_limit=min(rem2 * 0.5, 40),
                    T0=0.003, T_min=0.0001, cooldown=0.9995, n_fracs=8
                )
                try_seq(a_opt2, f"len{target_n}_pair")
            try_seq(a_opt, f"len{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine MW
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = mu_weights_optimize(
                best_seq, time_limit=min(rem, 40),
                top_k=n_best, temperature=0.000001,
                lr=0.0002, momentum=0.97, smooth_lambda=0.0
            )
            try_seq(a_opt, "final_mw")

        # Final pair search
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = annealed_pair_search(
                best_seq, time_limit=min(rem, 25),
                T0=0.0005, T_min=0.00005, cooldown=0.999, n_fracs=6
            )
            try_seq(a_opt, "final_pair")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
