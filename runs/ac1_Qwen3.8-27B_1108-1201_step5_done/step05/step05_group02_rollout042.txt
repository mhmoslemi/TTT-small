$\frac{\partial f_k}{\partial a[i]} = \frac{\partial}{\partial a[i]} \sum_j a[j] a[k-j]$

The term $a[i]$ appears in the sum when $j = i$ (giving $a[i] \cdot a[k-i]$) and when $k-j = i$ i.e. $j = k-i$ (giving $a[k-i] \cdot a[i]$).

If $i \neq k-i$ (i.e., $2i \neq k$): both terms are present, so $\frac{\partial f_k}{\partial a[i]} = a[k-i] + a[k-i] = 2a[k-i]$.
Wait no, if $j = i$ gives $a[i] \cdot a[k-i]$ and $j = k-i$ gives $a[k-i] \cdot a[i]$, these are the same term counted twice. So the derivative is $2a[k-i]$ if $i \neq k-i$, and $2a[i]$ if $i = k-i$ (since it's $a[i]^2$).

Actually, more carefully:
$f_k(a) = \sum_j a[j] \cdot a[k-j]$

$\frac{\partial f_k}{\partial a[i]} = \sum_j \frac{\partial a[j]}{\partial a[i]} \cdot a[k-j] + \sum_j a[j] \cdot \frac{\partial a[k-j]}{\partial a[i]}$

$= a[k-i] + \sum_j a[j] \cdot \delta_{k-j, i}$

$= a[k-i] + a[k-i]$ (if $0 \leq k-i < n$)

$= 2a[k-i]$ (if $0 \leq k-i < n$)

But wait, if $i = k-i$ (i.e., $k = 2i$), then the term $a[i] \cdot a[i] = a[i]^2$ has derivative $2a[i]$, which is consistent with $2a[k-i] = 2a[i]$.

So in all cases: $\frac{\partial f_k}{\partial a[i]} = 2a[k-i]$ if $0 \leq k-i < n$, and 0 otherwise.

Great, so the subgradient is $s[i] = 2 \sum_{k \in A} \lambda_k a[k-i]$ (where the sum is over valid $k-i$).

This is essentially a "blended" version of the reversed sequence $a[k-i]$ for the active lags $k$.

So the mirror descent update is:
$a_{t+1}[i] \propto a_t[i] \exp(-2\eta \sum_{k \in A} \lambda_k a_t[k-i])$

This is interesting! It's a multiplicative update that down-weights positions $i$ where the "reflected" active lags have high values.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase non-convex optimization:
1. Active-set LP Frank-Wolfe with adaptive constraint selection
2. Mirror descent with multi-objective switching (peak, top-K, quantile)
3. Sparse coordinate descent on peak-contributing coordinates
4. Multi-length exploration and structured initializations
Key insight: the problem is NON-convex (auto-convolution terms are bi-linear),
so we use multiple strategies to escape local minima.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak_sparse(f, n, active_indices):
    """Solve LP: min t s.t. (f*g)[k] <= t for k in active_indices, g >= 0, sum(g) = 1.
    Uses sparse matrix for efficiency."""
    m = len(active_indices)
    if m == 0:
        return np.ones(n) / n, 0.0

    rows = []
    cols = []
    vals = []

    for idx, k in enumerate(active_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            kj = k - j
            if 0 <= kj < n:
                rows.append(idx)
                cols.append(j)
                vals.append(f[kj])
        # t variable
        rows.append(idx)
        cols.append(n)
        vals.append(-1.0)

    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n + 1))
    c = np.zeros(n + 1)
    c[-1] = 1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_max_sum_sparse(f, n, M, active_indices):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    m = len(active_indices)
    if m == 0:
        return np.ones(n) / n

    rows = []
    cols = []
    vals = []

    for idx, k in enumerate(active_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            kj = k - j
            if 0 <= kj < n:
                rows.append(idx)
                cols.append(j)
                vals.append(f[kj])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n))
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        gs = res.x.sum()
        if gs > 1e-15:
            return res.x / gs
    return None

                g = g / gs if gs > 1e-15 else np.ones(n) / n

def mirror_descent_optimize(f0, time_limit, lr=0.08, top_k=30, verbose=False):
    """Mirror descent (exponentiated gradient) with multi-objective switching."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr
    obj_mode = 0  # 0=peak, 1=top-k sum, 2=quantile

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Switch objective every 200 iterations
        if it % 200 == 0:
            obj_mode = (obj_mode + 1) % 3

        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Softmax weights (adaptive temperature)
        temp = 0.005 * (pv.max() - pv.min()) + 1e-12
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[0] = 1.0
        else:
            w /= ws

        # Compute subgradient: s[i] = 2 * sum_k lambda_k * f[k-i]
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kj = k - j
                if 0 <= kj < n:
                    sg[j] += 2.0 * w[i] * f[kj]

        # Mirror descent: multiplicative update
        # Clip sg to avoid overflow
        sg_max = np.max(np.abs(sg))
        if sg_max > 1e-10:
            sg = sg / sg_max * 0.5

        f_new = f * np.exp(-cur_lr * sg)
        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / fs

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (f_new - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = f + best_alpha * (f_new - f)
            f = np.maximum(f, 0)
            fsum = f.sum()
            if fsum < 1e-15:
                f = best_f.copy()
            else:
                f /= fsum
            conv = fast_convolve(f)

        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            cur_lr *= 0.98

        if no_improve > 300:
            # Reset with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj

                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:

                        improved = True
                        conv = fast_convolve(f)

        if not improved:

            if no_improve > 5:
                # Random exploration
                rng = np.random.default_rng(it * 104729 + 3)

                    if time.monotonic() - t0 > time_limit:

                    j = int(rng.integers(0, n))
                    old_val = f[j]

def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=40,
                          temp=0.00005, verbose=False):
    """Subgradient with Nesterov momentum and adaptive softmax peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kj = k - j
                if 0 <= kj < n:
                    sg[j] += 2.0 * w[i] * f[kj]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.005

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def make_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine bump (higher at edges)
    x = np.linspace(0, 1, n)
    c = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # Sine (higher in middle)
    s = 1.0 + 0.5 * np.sin(np.pi * x)
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    # Bimodal
    b = 1.0 / (0.1 + np.abs(x - 0.5) * 4)
    inits.append(b / b.sum())

    # U-shaped (high at edges)
    u = 1.0 + 2.0 * (2 * x - 1)**2
    inits.append(u / u.sum())

    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Two-scale cosine
    c2 = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(2 * np.pi * x)
    c2 = np.maximum(c2, 0.01)
    inits.append(c2 / c2.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase non-convex optimization for minimizing the auto-convolution peak ratio.
    Uses LP-FW, mirror descent, subgradient, and coordinate descent in combination.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Quick exploration with multiple initializations ===
    print(f"\n=== Phase 1: Exploration (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t_explore = min(60, rem * 0.05)
    if t_explore > 10:
        per_init_time = max(2, t_explore / 6)
        inits = make_initializations(base_n, rng)
        # Also include parent
        inits.insert(0, parent.copy())

        for idx, init in enumerate(inits[:6]):
            if time.monotonic() > deadline - 30:
                break
            print(f"  Init {idx}: running quick LP-FW ({per_init_time:.0f}s)...")
            a_opt, _ = lp_fw_optimize(init, time_limit=per_init_time,
                                       max_k=50, min_k=5)
            try_seq(a_opt, f"init_{idx}")

    # === Phase 2: Main LP-FW optimization ===
    print(f"\n=== Phase 2: Main LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_lpfw = min(800, rem * 0.5)
        if t_lpfw > 30:
            print(f"  n={len(best_seq)}, t={t_lpfw:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw,
                                       verbose=True, max_k=150, min_k=15)
            try_seq(a_opt, "lpfw_main")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_md = min(250, rem * 0.2)
        if t_md > 20:
            print(f"  n={n_best}, t={t_md:.0f}s")
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md,
                                                lr=0.08, top_k=40, verbose=True)
            try_seq(a_opt, "mirror_descent")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(200, rem * 0.2)
        if t_sg > 20:
            print(f"  n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                             lr=0.003, momentum=0.94,
                                             top_k=max(30, n_best // 10),
                                             temp=0.00004, verbose=True)
            try_seq(a_opt, "sg_main")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(200, rem * 0.2)
        if t_cd > 15:
            print(f"  n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Second round LP-FW on best ===
    print(f"\n=== Phase 6: LP-FW Round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_lpfw2 = min(300, rem * 0.3)
        if t_lpfw2 > 30:
            print(f"  n={len(best_seq)}, t={t_lpfw2:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw2,
                                       verbose=True, max_k=200, min_k=20)
            try_seq(a_opt, "lpfw_r2")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        candidate_lengths = set()
        for delta in [-128, -64, 64, 128, 256]:
            ln = len(best_seq) + delta
            if ln >= 256 and ln <= 2048:
                candidate_lengths.add(ln)

        for target_n in sorted(candidate_lengths)[:3]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 120)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc,
                                       max_k=100, min_k=10)
            try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem, 60),
                                             lr=0.001, momentum=0.95,
                                             top_k=max(20, n_best // 12),
                                             temp=0.00002)
            try_seq(a_opt, "sg_final")

        # Quick CD
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "cd_final")

        # Perturbation + quick optimize
        for p in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            scale = 0.001 + 0.001 * p
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 30),
                                       max_k=50, min_k=10)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
