import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, x):
    """Compute H @ x where H_{ij} = p[i+j]. p length 2n-1, x length n."""
    return np.correlate(p, x, mode='valid')


def solve_qp_cg(p, n, x0=None, max_iter=100, ridge=1e-10):
    """
    Solve min x^T H x s.t. sum(x) = 1, x >= 0.
    Uses CG to solve (H + ridge*I) x = 1, then normalizes and projects.
    """
    if x0 is None:
        x = np.ones(n) / n
    else:
        x = x0.copy()
    
    b = np.ones(n)
    # r = b - (H + ridge*I) @ x = b - H@x - ridge*x
    r = b - hankel_matvec(p, x) - ridge * x
    p_vec = r.copy()
    rs = r @ r
    
    for _ in range(max_iter):
        Ap = hankel_matvec(p, p_vec) + ridge * p_vec
        pAp = p_vec @ Ap
        if abs(pAp) < 1e-15:
            break
        alpha = rs / pAp
        x += alpha * p_vec
        r -= alpha * Ap
        rs_new = r @ r
        if rs_new < 1e-30:
            break
        p_vec = r + (rs_new / rs) * p_vec
        rs = rs_new
    
    x_sum = x.sum()
    if x_sum < 1e-15:
        return np.ones(n) / n
    x = x / x_sum
    return project_simplex(x)


def game_iterate(a0, time_limit, T_init=None, T_end=None, qp_iters=80, 
                 ridge=1e-10, restart_every=300):
    """
    Game-theoretic primal-dual iteration.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv0 = fast_convolve(a)
    obj_range = conv0.max() - conv0.min()
    if T_init is None:
        T_init = max(0.1 * obj_range, 1e-4)
    if T_end is None:
        T_end = T_init * 0.01
    
    best_a = a.copy()
    best_obj = conv0.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_end + 0.5 * (T_init - T_end) * (1.0 + np.cos(np.pi * progress))
        T = max(T, 1e-6)
        
        conv = fast_convolve(a)
        
        # Dual update
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Primal update (QP solve)
        a_qp = solve_qp_cg(p, n, x0=a, max_iter=qp_iters, ridge=ridge)
        a = a_qp
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart
        if no_improve >= restart_every:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            scale = 0.002 + 0.004 * min(1.0, (no_improve - restart_every) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.001, decay=0.9995):
    """
    Subgradient method focused on the top peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    v = np.zeros(n)  # Momentum
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break
        
        conv = fast_convolve(a)
        
        # Weighted subgradient: weight each peak by its excess
        c_mean = conv.mean()
        excess = np.maximum(conv - c_mean, 0)
        if excess.sum() < 1e-15:
            break
        w = excess / excess.sum()
        
        # Gradient: sum_k w_k * 2 * a[k-j] (for valid j)
        # = 2 * correlate(w, a, 'valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')
        
        # Update with momentum
        v = 0.9 * v - lr * grad
        a = project_simplex(a + v)
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= decay
        
        if no_improve > 500:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj

    # Anti-uniform (center heavy)

    saw = np.abs(np.sin(np.pi * x / n * 4)) + 0.1

    if parent is not None:

        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
    
    return inits
