import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_shave(a, time_limit, target_papr=1.5, top_k=50):
    """
    Iterative peak shaving: repeatedly reduce the coordinates most responsible
    for the highest peaks in the auto-convolution.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    step_size = 0.1
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        cmax = c.max()
        
        # Find top K peaks
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]

        # Weighted sum of responsibilities
        blame = np.zeros(n)
        for i in range(len(peaks)):

            # Scale by how much this peak exceeds average
            excess = max(0, peak_vals[i] - cmax / (2 * n - 1))
            m_lo = max(0, k - n + 1)

            if m_lo <= m_hi:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light smoothing to reduce interpolation artifacts
    if new_n > 4:
        kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
        f_interp = np.convolve(f_interp, kernel, mode='same')
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_gradient_peaks(a, c, top_k):
    """Compute weighted subgradient from top-k peaks of auto-convolution."""
    n = len(a)
    k_actual = min(top_k, len(c))
    peaks = np.argpartition(c, -k_actual)[-k_actual:]
    pv = c[peaks]
    
    # Softmax weighting based on peak values
    scale = max(pv.max() - pv.min(), 1e-12)
    exp_vals = np.exp((pv - pv.max()) / scale)
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += 2.0 * w[i] * a[k - indices]
    return grad

        grad = compute_gradient_peaks(a, c, effective_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        grad = grad / gnorm
        
        # Backtracking line search
        a_new = a
        f_new = cmax
        step = alpha
        improved = False
        
        for bt in range(12):
            candidate = a - step * grad
            candidate = np.maximum(candidate, 0.0)

            if cs < 1e-15:

                continue

def spectral_compress(a0, time_limit, compress_pow=0.7, n_iters=200, verbose=False):
    """

    """

    else:
        a = a / s

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    relx = 1.0  # relaxation parameter
    
    while time.monotonic() - t0 < time_limit and it < n_iters:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        
        # Compress amplitude: |A_new| = |A|^compress_pow
        mag = np.abs(A)
        phase = np.angle(A)
        mag_new = mag ** compress_pow
        
        # Reconstruct and project
        A_new = mag_new * np.exp(1j * phase)
        a_candidate = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative simplex
        a_candidate = np.maximum(a_candidate, 0.0)
        s_new = a_candidate.sum()
        if s_new < 1e-15:

        # Check improvement
        c_new = fast_convolve(a_candidate)
        val_new = c_new.max()
        
        if val_new < best_obj - 1e-14:
            best_obj = val_new
            best_a = a_candidate.copy()
            a = a_candidate
            no_improve = 0
            # Slowly increase compression
            compress_pow = min(0.95, compress_pow + 0.005)
            if verbose and it % 10 == 0:
                print(f"    SC iter {it}: obj={best_obj:.10f}, pow={compress_pow:.3f}")
        else:
            # Use relaxation: blend old and new
            a_blend = (1.0 - relx) * a + relx * a_candidate
            a_blend = np.maximum(a_blend, 0.0)

            if sb > 1e-15:

                    a = a_blend

                else:
                    no_improve += 1
            else:
                no_improve += 1

def coordinate_descent(a0, time_limit, n_coarse=8):
    """
    Coordinate descent: optimize each coordinate via golden section search.
    Only optimize a subset per pass for speed.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    no_pass_improve = 0
    pass_num = 0
    
    # Cycle through coordinates, optimizing a subset each pass
    n_per_pass = max(16, n // n_coarse)
    
    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Which coordinates to optimize this pass
        start = (pass_num * n_per_pass) % n
        coords = list(range(start, start + n_per_pass))
        if start + n_per_pass > n:
            coords += list(range(0, start + n_per_pass - n))
        coords = coords[:n_per_pass]
        
        pass_improved = False
        
        for j in coords:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                # Try small positive values
                for test_val in [1.0/n, 2.0/n]:
                    a_test = a.copy()
                    a_test[j] = test_val
                    a_test = np.maximum(a_test, 0)
                    st = a_test.sum()
                    if st > 1e-15:
                        a_test /= st
                        c_test = fast_convolve(a_test)
                        if c_test.max() < best_obj - 1e-14:
                            a = a_test
                            best_obj = c_test.max()
                            best_a = a.copy()
                            pass_improved = True
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            # Golden section on x = a[j], others scaled
            lo = 0.0
            hi = min(0.999, 3.0 * old_val)
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            
            def eval_coord(x, j=j):
                ft = a.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < -1e-10:
                    return float('inf')
                ft2 = max(0.0, ft2)
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()
            
            f1 = eval_coord(c1)
            f2 = eval_coord(c2)
            
            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)
            
            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            
            cur_val = fast_convolve(a).max()
            if opt_val < cur_val - 1e-14:
                a[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] = a[mask] * (ft2 / denom)
                a = np.maximum(a, 0.0)
                sa = a.sum()
                if sa > 1e-15:
                    a /= sa
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                pass_improved = True
        
        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break
    
    return best_a, best_obj

                t_alloc2 = min(rem * 0.3, 60)
                a_opt2, _ = spectral_compress(a_opt, time_limit=t_alloc2, compress_pow=0.8)
                try_seq(a_opt2, f"spec_n{target_n}")
    
    # === Phase 4: Combined optimization on best ===
    print(f"\n=== Phase 4: Combined optimization (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem * 0.6, 200)
        a_opt, _ = combined_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "combined")
    
    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD polish (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_alloc = min(rem * 0.5, 100)
        a_opt, _ = coordinate_descent(base, time_limit=t_alloc)
        try_seq(a_opt, "cd_polish")
    
    # === Phase 6: Multi-start refinement ===
    print(f"\n=== Phase 6: Multi-start (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    rng = np.random.default_rng(seed)
    
    for p in range(6):
        rem = deadline - time.monotonic() - 60
        if rem < 15:
            break
        t_alloc = min(rem * 0.3, 60)
        
        if p == 0:
            f_init = base.copy()
        elif p == 1:
            noise = rng.standard_normal(n_base) * 0.008
            f_init = project_simplex(base + noise)
        elif p == 2:
            f_init = 0.6 * base + 0.4 * cosine_modulated_init(n_base, 2, 0.3)
            f_init = project_simplex(f_init)
        elif p == 3:
            f_init = 0.7 * base + 0.3 * cosine_modulated_init(n_base, 5, 0.2)
            f_init = project_simplex(f_init)
        elif p == 4:
            f_init = 0.8 * base + 0.2 * base[::-1]
            f_init = project_simplex(f_init)
        else:
            # Block-repeat a smaller optimal
            small_n = max(64, n_base // 4)
            block = base[:small_n]
            block = block / block.sum()
            n_blocks = n_base // small_n
            f_init = np.tile(block, n_blocks)
            if len(f_init) < n_base:
                f_init = np.concatenate([f_init, base[len(f_init):]])
            f_init = f_init[:n_base]
            f_init = project_simplex(f_init)
        
        a_opt, _ = peak_shave_bt(f_init, time_limit=t_alloc, top_k_init=25, top_k_max=80)
        try_seq(a_opt, f"ms_{p}")
    
    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    
    # Quick spectral + peak shave
    rem = deadline - time.monotonic() - 40
    if rem > 15:
        a_opt, _ = spectral_compress(base, time_limit=min(rem * 0.5, 40), compress_pow=0.85)
        try_seq(a_opt, "final_spec")
    
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        a_opt, _ = peak_shave_bt(base, time_limit=min(rem, 30), top_k_init=50, top_k_max=200)
        try_seq(a_opt, "final_ps")
    
    # Tiny perturbations
    rem = deadline - time.monotonic() - 15
    for _ in range(3):
        rem = deadline - time.monotonic() - 10
        if rem < 8:
            break
        noise = rng.standard_normal(len(base)) * 0.002
        f_pert = project_simplex(base + noise)
        a_opt, _ = peak_shave_bt(f_pert, time_limit=min(rem, 15))
        try_seq(a_opt, "perturb")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
