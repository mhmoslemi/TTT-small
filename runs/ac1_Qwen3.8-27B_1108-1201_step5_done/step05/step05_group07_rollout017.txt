def adam_subgradient(f0, time_limit, top_k=50, temperature=0.00005, lr=0.003,
                     beta1=0.9, beta2=0.999, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    cur_temp = temperature
    cur_top_k = top_k
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute subgradient
        conv = fast_convolve(f)
        sg = compute_subgradient(f, conv, cur_top_k, cur_temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Exponential update with Adam-scaled rate
        exponent = -lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        exponent -= exponent.max()  # for numerical stability
        f = f * np.exp(exponent)
        f = np.maximum(f, 0)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            continue
        f /= fs
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
        
        # Adaptive schedule
        if t % 100 == 0:
            cur_temp *= 1.01
            cur_lr = lr * (0.999 ** t)
    
    return best_f, best_obj

def lbfgs_smooth_optimize(f0, T, time_limit, max_iter=5000, verbose=False):
    """
    L-BFGS-B optimization using log-space parameterization and softmax-x1
    smooth approximation of max. Optimizes over x where a = softmax(x).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    x0 = np.log(f + 1e-30)
    x0 -= x0.mean()  # center for numerical stability

    t0 = time.monotonic()
    best_x = x0.copy()
    best_val = 2.0 * n * fast_convolve(f).max()

    def obj_and_grad(x):
        if time.monotonic() - t0 > time_limit:
            return 2.0 * n * fast_convolve(f0).max(), np.zeros(n)
        a = np.exp(x - x.max())
        a_sum = a.sum()
        a = a / a_sum
        conv = fast_convolve(a)
        C = conv.max()
        scaled = (conv - C) / T
        e = np.exp(scaled)
        e_sum = e.sum()
        if e_sum < 1e-30:
            return 2.0 * n * C, np.zeros(n)
        w = e / e_sum
        val = 2.0 * n * T * (C + np.log(e_sum))

        # Gradient w.r.t. a: g_a[i] = (4n/T) * sum_k w[k] * a[k-i]
        g_a = np.zeros(n)
        n_conv = len(conv)
        # Use only peaks with significant weight for efficiency
        threshold = 1e-10
        active = w > threshold
        if active.sum() == 0:
            active[np.argmax(w)] = True
        for k in range(n_conv):
            if not active[k]:
                continue
            wk = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_a[idx] += (4.0 * n / T) * wk * a[k - idx]

        # Convert to x-space: gx = a * (g_a - dot(g_a, a))
        ga_dot_a = np.dot(g_a, a)
        gx = a * (g_a - ga_dot_a)
        return val, gx

    # Run L-BFGS-B
    result = scipy_minimize(
        obj_and_grad, x0, method='L-BFGS-B',
        jac=True, options={'maxiter': max_iter, 'maxfun': 20000,
                          'ftol': 1e-15, 'gtol': 1e-8, 'iprint': -1}
    )
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    val_opt = 2.0 * n * fast_convolve(a_opt).max()

    if val_opt < best_val:
        best_val = val_opt
        best_x = x_opt

    if verbose:
        print(f"    L-BFGS-B: val={val_opt:.10f}, iters={result.nit}, msg={result.message}")

    return a_opt, best_val

    """Coordinate descent with multiplier search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)
    if mid - d >= 0:
        order.append(mid - d)

    pass_count = 0
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peaks = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peaks

            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break
    return best_f, best_obj


def pair_exchange_search(f0, time_limit, verbose=False):
    """Pair-exchange local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        top_peaks = np.argpartition(conv, -min(20, 2 * n - 1))[-min(20, 2 * n - 1):]

        for trial in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-10:
                continue
            for frac in [0.05, 0.15, 0.3, 0.5]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] = old_j + amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = old_j
                    break

        # Also try peak-directed transfers
        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            hi_pos = np.argmax(contribs)
            lo_pos = np.argmin(contribs)
            hi = idx[hi_pos]
            lo = idx[lo_pos]
            if hi == lo or f[hi] < 1e-10:
                continue
            for frac in [0.1, 0.2, 0.35]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi] += amt
                    f[lo] -= amt
                    break

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=4):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 80)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Block pattern
    block_size = max(2, n // (10 + int(rng.random() * 10)))
    n_blocks = n // block_size
    block_vals = 0.5 + rng.random(n_blocks)
    block_seq = np.repeat(block_vals, block_size)[:n]
    if len(block_seq) < n:
        block_seq = np.concatenate([block_seq, np.ones(n - len(block_seq))])
    block_seq = np.maximum(block_seq, 1e-10)
    inits.append(block_seq / block_seq.sum())

    return inits[:count]


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Adam subgradient, L-BFGS-B polishing,
    coordinate descent, pair exchange, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        p_score = 2.0
        print("No parent found, using uniform n=768")

    # === Phase 1: Coarse Adam optimization on multiple starts ===
    print(f"\n=== Phase 1: Coarse Adam (best={best_score:.6f}) ===")

    candidates = [parent]
    for n_try in [768, 1024]:
        candidates.extend(structured_inits(n_try, rng, count=2))

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - 600
        if rem < 15:
            break
        n_cand = len(cand)
        t_alloc = min(rem * 0.15, 80)
        if t_alloc < 5:
            t_alloc = min(rem * 0.3, 40)

        print(f"  Cand {idx}: n={n_cand}, t={t_alloc:.0f}s")
        a_opt, _ = adam_subgradient(cand, time_limit=t_alloc,
                                    top_k=max(30, n_cand // 6),
                                    temperature=0.0001, lr=0.004,
                                    beta1=0.85, beta2=0.999)
        try_seq(a_opt, f"adam_c{idx}")

        # Quick L-BFGS-B polish
        rem = deadline - time.monotonic() - 500
        if rem > 20 and best_seq is not None:
            a_lb, _ = lbfgs_smooth_optimize(a_opt, T=0.001, time_limit=min(rem, 30),
                                            max_iter=1000)
            try_seq(a_lb, f"lbfgs_c{idx}")

    # === Phase 2: Deep Adam optimization ===
    print(f"\n=== Phase 2: Deep Adam (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {"top_k": max(20, n_best // 5), "temp": 0.00003, "lr": 0.002,
             "b1": 0.9, "b2": 0.999},
            {"top_k": max(10, n_best // 10), "temp": 0.00001, "lr": 0.001,
             "b1": 0.93, "b2": 0.9995},
            {"top_k": max(40, n_best // 3), "temp": 0.0002, "lr": 0.005,
             "b1": 0.8, "b2": 0.998},
            {"top_k": max(15, n_best // 7), "temp": 0.00005, "lr": 0.0015,
             "b1": 0.95, "b2": 0.9999},
        ]

        for p_idx, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 350
            if rem < 10:
                break
            t_alloc = min(rem * 0.35, 120)

            if p_idx == 0:
                f_init = best_seq.copy()
            elif p_idx == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            elif p_idx == 2:
                f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = 0.8 * best_seq + 0.2 * best_seq[::-1]
                f_init = project_simplex(f_init)

            print(f"  Deep {p_idx}: t={t_alloc:.0f}s, cfg_topk={cfg['top_k']}")
            a_opt, _ = adam_subgradient(f_init, time_limit=t_alloc,
                                        top_k=cfg["top_k"], temperature=cfg["temp"],
                                        lr=cfg["lr"], beta1=cfg["b1"], beta2=cfg["b2"])
            try_seq(a_opt, f"deep_{p_idx}")

    # === Phase 3: L-BFGS-B polishing at multiple temperatures ===
    print(f"\n=== Phase 3: L-BFGS-B Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        temperatures = [0.01, 0.003, 0.001, 0.0003, 0.0001]
        for T in temperatures:
            rem = deadline - time.monotonic() - 250
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 40)
            print(f"  L-BFGS T={T}, t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_smooth_optimize(best_seq, T=T, time_limit=t_alloc,
                                             max_iter=3000)
            try_seq(a_opt, f"lbfgs_T{T}")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [1024, 1280, 1536, 2048]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 80)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_subgradient(f_interp, time_limit=t_alloc,
                                        top_k=max(40, target_n // 5),
                                        temperature=0.00008, lr=0.003,
                                        beta1=0.88, beta2=0.999)
            try_seq(a_opt, f"len_{target_n}")

            # Quick L-BFGS-B on the result
            rem = deadline - time.monotonic() - 100
            if rem > 15:
                a_lb, _ = lbfgs_smooth_optimize(a_opt, T=0.001, time_limit=min(rem, 25),
                                                max_iter=2000)
                try_seq(a_lb, f"lbfgs_len_{target_n}")

    # === Phase 5: Coordinate descent + pair exchange ===
    print(f"\n=== Phase 5: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_cd = min(rem * 0.4, 60)
            print(f"  Coord descent t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

            rem = deadline - time.monotonic() - 50
            if rem > 10:
                t_pex = min(rem * 0.7, 40)
                print(f"  Pair exchange t={t_pex:.0f}s")
                a_opt, _ = pair_exchange_search(best_seq, time_limit=t_pex)
                try_seq(a_opt, "pex")

                # L-BFGS-B after local search
                rem = deadline - time.monotonic() - 30
                if rem > 10:
                    a_lb, _ = lbfgs_smooth_optimize(a_opt, T=0.0005,
                                                    time_limit=min(rem, 20),
                                                    max_iter=2000)
                    try_seq(a_lb, "lbfgs_after_ls")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine Adam
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = adam_subgradient(best_seq, time_limit=min(rem, 40),
                                        top_k=n_best, temperature=0.000003,
                                        lr=0.0003, beta1=0.97, beta2=0.9999)
            try_seq(a_opt, "adam_fine")

        # L-BFGS-B with very low T
        rem = deadline - time.monotonic() - 25
        if rem > 8:
            a_lb, _ = lbfgs_smooth_optimize(best_seq, T=0.0001,
                                            time_limit=min(rem, 20), max_iter=5000)
            try_seq(a_lb, "lbfgs_fine")

        # Coordinate descent final
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "cd_final")

        # Perturbation restarts
        for _ in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 5:
                break
            for scale in [0.001, 0.002]:
                rem = deadline - time.monotonic() - 5
                if rem < 3:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = adam_subgradient(f_pert, time_limit=min(rem, 8),
                                            top_k=max(20, n_best // 5),
                                            temperature=0.00002, lr=0.001,
                                            beta1=0.92, beta2=0.999)
                try_seq(a_opt, f"pert_{scale}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
