def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*(a*a)[k])) w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns (gradient, convolution).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws

    # g[j] = 2 * sum_m w[j+m]*a[m] for j=0,...,n-1
    # This is the cross-correlation (w cross a) evaluated at lags 0..n-1
    fft_size = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)
    grad = 2.0 * cross[:n]
    return grad, conv

def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_step(f, n):
    """Single LP-FW step: solve LP, line search, return updated f."""
    g, lp_t = solve_lp_min_peak(f, n)
    if g is None:
        return f, fast_convolve(f).max()

    g = np.maximum(g, 0)
    gs = g.sum()
    if gs < 1e-15:
        return f, fast_convolve(f).max()
    g = g / gs

    conv = fast_convolve(f)
    M = conv.max()
    best_alpha = 0.0
    best_val = M

    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 1e-12:
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, best_val
    return f, M

def peak_cd_optimize(f0, time_limit, verbose=False):
    """Peak-targeting coordinate descent with rebalancing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    best_f = f.copy()

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        top_peak_k = int(np.argmax(conv))

        js = max(0, top_peak_k - n + 1)
        je = min(n - 1, top_peak_k)

        # Find the coordinate with maximum contribution to the peak
        contributions = np.zeros(je - js + 1)
        for i, j in enumerate(range(js, je + 1)):
            contributions[i] = f[j] * f[top_peak_k - j]

        # Try reducing the top contributor and distributing to others
        idx_max = np.argmax(contributions)
        j_star = js + idx_max

        improved = False
        for frac in [0.3, 0.5, 0.7, 0.85]:
            reduction = f[j_star] * (1 - frac)
            if reduction < 1e-12:
                continue
            f_trial = f.copy()
            f_trial[j_star] -= reduction
            # Distribute to positions that DON'T contribute to the peak
            mask = np.ones(n, dtype=bool)
            for pk in range(max(0, top_peak_k - n + 1), min(n, top_peak_k + 1)):
                mask[pk] = False
            if mask.sum() == 0:
                mask[:] = True
                mask[j_star] = False
            if mask.sum() > 0:
                f_trial[mask] += reduction / mask.sum()
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-12:
                f = f_trial
                best_obj = val
                best_f = f.copy()
                improved = True
                break

        if not improved:

            rng = np.random.default_rng((it + hash(str(f[:5]))) % 999983)

                j1 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if j1 == j2:
                    continue
                delta = 0.001 * f.sum()
                f_trial = f.copy()
                f_trial[j1] -= delta
                f_trial[j2] += delta
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial = f_trial / st
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-12:
                    f = f_trial
                    best_obj = val

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_diverse_initials(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
    inits.append(tri / tri.sum())

    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())

    # Gaussian-like
    center = (n - 1) / 2
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / (n / 6.0)) ** 2)
    inits.append(gauss / gauss.sum())

    # Flat top with tapered edges
    fftop = np.ones(n)
    taper = max(1, n // 8)
    for i in range(taper):
        fftop[i] = 0.5 * (1 - np.cos(np.pi * i / taper))
        fftop[n - 1 - i] = 0.5 * (1 - np.cos(np.pi * i / taper))
    inits.append(fftop / fftop.sum())

    # Sinusoidal modulation
    x = np.arange(n)
    sino = 1.0 + 0.3 * np.sin(2 * np.pi * x / n)
    sino = np.maximum(sino, 0.1)
    inits.append(sino / sino.sum())

    # Random smooth
    smooth = np.abs(np.concatenate([
        np.cumsum(rng.standard_normal(n // 4)),
        np.zeros(n - n // 4)
    ]))
    if smooth.sum() < 1e-10:
        smooth = np.ones(n)
    inits.append(smooth / smooth.sum())

    # Bimodal
    c1 = n // 4
    c2 = 3 * n // 4
    bim = np.exp(-0.5 * ((np.arange(n) - c1) / (n / 12.0)) ** 2) + \
          np.exp(-0.5 * ((np.arange(n) - c2) / (n / 12.0)) ** 2)
    inits.append(bim / bim.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LSE gradient descent, LP-FW,
    coordinate descent, and diverse initialization for minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.size == 0 or a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default init: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Diverse initialization exploration ===
    print(f"\n=== Phase 1: Diverse Initialization (n={base_n}) ===")
    inits = generate_diverse_initials(base_n, rng)

    # Also try the parent
    inits.insert(0, parent.copy())

    for i, init in enumerate(inits):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(60, rem * 0.15)
        if t_alloc < 10:
            break
        label = f"init_{i}"
        if i == 0:
            label = "parent_lse"
        f_opt, _ = lse_optimize(init, time_limit=t_alloc,
                                beta_start=5.0, beta_end=300.0,
                                lr_start=0.008)
        try_seq(f_opt, label)

    # === Phase 2: Try nearby lengths ===
    print(f"\n=== Phase 2: Length Exploration (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    for delta in [-64, 64, -128, 128]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        target_n = len(base) + delta
        if target_n < 256 or target_n > 2048:
            continue
        t_alloc = min(80, rem * 0.2)
        if t_alloc < 15:
            break
        f_interp = interpolate_sequence(base, target_n)
        f_opt, _ = lse_optimize(f_interp, time_limit=t_alloc,
                                beta_start=8.0, beta_end=350.0,
                                lr_start=0.007)
        try_seq(f_opt, f"len_{target_n}")

    # === Phase 3: Main hybrid optimization ===
    print(f"\n=== Phase 3: Hybrid Optimization (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_main = min(rem * 0.35, 450)
        if t_main > 30:
            print(f"  n={len(best_seq)}, t={t_main:.0f}s")
            f_opt, _ = multi_phase_optimize(best_seq, time_limit=t_main)
            try_seq(f_opt, "hybrid_main")

    # === Phase 4: Extended LSE with slow annealing ===
    print(f"\n=== Phase 4: Extended LSE (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_lse = min(rem * 0.25, 300)
        if t_lse > 20:
            print(f"  n={len(best_seq)}, t={t_lse:.0f}s")
            f_opt, _ = lse_optimize(best_seq, time_limit=t_lse,
                                    beta_start=3.0, beta_end=800.0,
                                    lr_start=0.005)
            try_seq(f_opt, "lse_extended")

    # === Phase 5: LP-FW refinement ===
    print(f"\n=== Phase 5: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.2, 250)
        if t_lp > 20:
            print(f"  n={len(best_seq)}, t={t_lp:.0f}s")
            f_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp, max_iters=150)
            try_seq(f_opt, "lpfw_refine")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.2, 200)
        if t_cd > 15:
            print(f"  n={len(best_seq)}, t={t_cd:.0f}s")
            f_opt, _ = peak_cd_optimize(best_seq, time_limit=t_cd)
            try_seq(f_opt, "cd_polish")

    # === Phase 7: Random restarts from best ===
    print(f"\n=== Phase 7: Random Restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(10):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            scale = 0.002 + 0.003 * p
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(40, rem * 0.3)
            if t_alloc < 8:
                break
            f_opt, _ = lse_optimize(f_pert, time_limit=t_alloc,
                                    beta_start=20.0, beta_end=600.0,
                                    lr_start=0.003)
            try_seq(f_opt, f"restart_{p}")

    # === Phase 8: Final short multi-phase ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_final = min(rem * 0.4, 120)
        if t_final > 10:
            f_opt, _ = multi_phase_optimize(best_seq, time_limit=t_final)
            try_seq(f_opt, "final_hybrid")

        # Very short LSE
        rem = deadline - time.monotonic()
        if rem > 10:
            f_opt, _ = lse_optimize(best_seq, time_limit=min(rem * 0.3, 40),
                                    beta_start=50.0, beta_end=1000.0,
                                    lr_start=0.002)
            try_seq(f_opt, "final_lse")

        # Very short CD
        rem = deadline - time.monotonic()
        if rem > 8:
            f_opt, _ = peak_cd_optimize(best_seq, time_limit=min(rem * 0.3, 30))
            try_seq(f_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    elapsed = time.monotonic() + 20.0 - (deadline + 20.0)
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (time.monotonic() - budget_s + (deadline - time.monotonic())):.0f}s")
    return [float(x) for x in best_seq]
