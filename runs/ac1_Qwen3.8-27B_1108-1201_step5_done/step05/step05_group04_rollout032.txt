"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact subgradient descent with heavy-ball momentum on the simplex
2. Frequency-domain spectral flattening
3. Adaptive peak-squeezing with coordinate descent

5. Multi-length exploration with interpolation
Key: leverages exact subgradients (no LSE approximation), FFT-based gradient,

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_fast_len(size)
    A = fft(a, N)
    return np.real(ifft(A * A, N))[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_fft(a, w):
    """
    Compute g[m] = 2 * sum_k w[k] * a[k-m] for 0 <= k-m < n.
    Uses FFT for efficiency.
    
    a: sequence of length n
    w: weights of length 2n-1 (one per convolution lag)
    Returns: gradient of length n
    """
    n = len(a)
    # y[m] = sum_j a[j] * w[j+m] for j=0..n-1, j+m < 2n-1
    # This is cross-correlation of a and w
    # Using FFT: y = ifft(conj(fft(a, N)) * fft(w, N))
    N = next_fast_len(2 * n)  # Should be enough
    A = fft(a, N)
    W = fft(w, N)
    y = ifft(np.conj(A) * W)
    return 2.0 * np.real(y[:n])


def active_peak_weights(conv, tolerance=0.0):
    """
    Compute weights for active peaks.
    tolerance: include peaks within this fraction of max.
    Returns: weight vector of length 2n-1
    """
    max_val = conv.max()
    # Active peaks: those within tolerance of max
    if tolerance == 0:
        # Only exact max
        active = (conv == max_val)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        return w
    else:
        # Soft weighting: exponential in (conv - max)
        diff = max_val - conv
        w = np.exp(-diff / (tolerance * max_val + 1e-15))
        w[conv < max_val * (1 - tolerance)] = 0  # Only near-max
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.zeros_like(conv)
            w[conv == max_val] = 1.0 / (conv == max_val).sum()
        return w


def subgrad_descent_momentum(a0, time_limit, lr_init=0.01, momentum_init=0.9,
                               tol_frac=0.01, max_iter=None):
    """
    Subgradient descent with heavy-ball momentum on the simplex.
    Minimizes max(conv(a, a)) subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        peak_val = conv.max()
        
        if peak_val < best_obj - 1e-12:
            best_obj = peak_val
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.95
            if no_improve > 300:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7919 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0
                continue
        
        # Compute weights for active peaks
        w = active_peak_weights(conv, tolerance=tol_frac)
        
        # Compute subgradient
        g = compute_subgrad_fft(a, w)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g /= g_norm
        
        # Heavy ball momentum update
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < peak_val:
            a = a_new
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = project_simplex(a + factor * v)
                c_try = fast_convolve(a_try)
                if c_try.max() < peak_val:
                    a = a_try
                    break
            else:
                # No improvement, just update v
                pass
        
        # Adapt momentum
        if no_improve > 100:
            momentum = max(0.8, momentum - 0.002)
    
    return best_a, best_obj


def spectral_flatten(a0, time_limit, n_iter=None):
    """
    Frequency-domain approach: flatten the power spectrum.
    Alternates between time domain (non-negativity constraint) and
    frequency domain (spectral flattening).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    N = next_fast_len(2 * n)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    gamma = 0.1  # Smoothing parameter for spectral flattening
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Time domain: compute FFT
        A = fft(a, N)
        power = np.abs(A) ** 2
        
        # Flatten: replace power with a smoothed version
        # Target: make power as flat as possible while keeping DC = 1
        # Use iterative approach: power_new = (1-gamma)*power + gamma*mean_power*(n/(2n-1))
        # But we need to keep phase
        
        # Actually, let's use a different approach:
        # We want to reduce the peaks in conv = ifft(power)
        # So we want to reduce power at frequencies where ifft has peaks
        
        # Compute current conv
        conv = np.real(ifft(power, N))[:2*n-1]
        conv = np.maximum(conv, 0)  # Numerical safety
        peak = conv.max()
        peak_idx = np.argmax(conv)
        
        if peak < best_obj - 1e-12:
            best_obj = peak
            best_a = a.copy()
            no_improve = 0
            gamma = min(gamma * 1.05, 0.5)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                gamma = max(0.02, gamma * 0.95)
            if no_improve > 200:
                rng = np.random.default_rng(it * 104729)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
                continue
        
        # Spectral flattening step:
        # Reduce power at frequencies that contribute to the peak
        # The peak in conv at lag k means sum_j a[j]*a[j+k] is large
        # In frequency domain, this relates to the "harmonics"
        
        # Simpler approach: reduce the variance of power spectrum
        # power_target = mean(power) * ones (but keep DC)
        power_mean = power.mean()
        power_smooth = (1 - gamma) * power + gamma * power_mean
        
        # Keep DC component fixed (sum = 1 means A[0] = 1)
        power_smooth[0] = power[0]
        # Keep conjugate symmetry (real signal)
        # power is already real and symmetric
        
        # New amplitude, keep phase
        new_amp = np.sqrt(np.maximum(power_smooth, 0))
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # Transform back
        a_new = np.real(ifft(A_new, N))[:n]
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        a = a_new
    
    return best_a, best_obj


def coordinate_squeeze(a0, time_limit, n_coords=10):
    """
    Coordinate descent: iteratively adjust groups of coordinates
    to reduce the peak convolution value.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.001
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peak = conv.max()
        peak_lag = np.argmax(conv)
        
        if peak < best_obj - 1e-12:
            best_obj = peak
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.01)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.97
            if no_improve > 500:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                no_improve = 0
                continue
        
        # Find which coordinates contribute most to the peak
        # conv[peak_lag] = sum_i a[i] * a[peak_lag - i]
        # The "most contributing" pairs are those with large products
        
        lo = max(0, peak_lag - n + 1)
        hi = min(n - 1, peak_lag)
        pairs = np.arange(lo, hi + 1)
        products = a[pairs] * a[peak_lag - pairs]
        
        # Pick the n_coords largest products
        top_idx = np.argsort(products)[::-1][:n_coords]
        active_pairs = pairs[top_idx]
        
        # For each active pair (i, peak_lag-i), try to reduce the product
        # by transferring mass from one to the other (or to a random other coordinate)
        for idx in active_pairs:
            other = peak_lag - idx
            if idx == other:
                continue
            # Try reducing a[idx] and increasing a[random_other]
            # Or try a local swap
            candidate = a.copy()
            transfer = 0.1 * min(candidate[idx], candidate[other])
            if transfer > 1e-8:
                # Move mass from the larger to the smaller (within the pair)
                if candidate[idx] > candidate[other]:
                    candidate[idx] -= transfer
                    candidate[other] += transfer
                else:
                    candidate[other] -= transfer
                    candidate[idx] += transfer
                # Also try moving to a random other coordinate
                r = rng.integers(n)
                candidate2 = a.copy()
                candidate2[idx] -= transfer * 0.5
                candidate2[r] += transfer * 0.5
                candidate2[r] = max(candidate2[r], 0)
                candidate2 = project_simplex(candidate2)
                
                # Evaluate both
                c1 = fast_convolve(project_simplex(candidate))
                c2 = fast_convolve(candidate2)
                if c1.max() < peak:
                    a = project_simplex(candidate)
                elif c2.max() < peak:
                    a = candidate2
    
    return best_a, best_obj


def wavelet_perturb(a0, n_scales=4):
    """
    Generate perturbations in a wavelet-like basis.
    Returns a random perturbation vector that is "structured".
    """
    n = len(a0)
    rng = np.random.default_rng()
    pert = np.zeros(n)
    
    # Multi-scale sinusoidal perturbation
    for s in range(n_scales):
        freq = rng.integers(2, max(3, n // (2 ** s) + 1))
        amp = rng.random() * (0.1 / (s + 1))
        phase = rng.random() * 2 * np.pi
        x = np.arange(n)
        pert += amp * np.sin(2 * np.pi * freq * x / n + phase)
    
    return pert

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact subgradient descent with heavy-ball momentum on the simplex
2. FFT-based spectral flattening (frequency domain peak reduction)
3. Adaptive peak-squeezing via targeted coordinate moves
4. Wavelet-structured random perturbations for exploration
5. Multi-length exploration with interpolation and refinement
Key innovations: exact subgradients (no LSE smoothing), dual-domain optimization,
adaptive restart strategy, and structured exploration.

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_fast_len(size)
    A = fft(a, N)
    return np.real(ifft(A * A, N))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, w, max_active=400):
    """Compute subgradient g[m] = 2 * sum_k w[k]*a[k-m] for active peaks."""
    n = len(a)
    active = np.where(w > 1e-18)[0]
    if len(active) == 0:
        return np.zeros(n)
    if len(active) > max_active:
        top = np.argpartition(w[active], -max_active)[-max_active:]
        active = active[top]
    g = np.zeros(n)
    for k in active:
        wk = w[k]
        if wk < 1e-18:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * wk * a[k - idx]
    return g


def peak_weights(conv, mode="sharp", tol=0.02):
    """Compute subgradient weights on active peaks."""
    max_val = conv.max()
    if max_val < 1e-15:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    if mode == "sharp":
        # Only peaks within tol fraction of max
        threshold = max_val * (1.0 - tol)
        active = conv >= threshold
        if not active.any():
            active = np.zeros_like(conv, dtype=bool)
            active[np.argmax(conv)] = True
        w = np.zeros_like(conv)
        w[active] = conv[active] - conv[active].min() + 1e-10
        w /= w.sum()
    elif mode == "smooth":
        # Exponential weighting
        diff = max_val - conv
        w = np.exp(-diff / (tol * max_val + 1e-15))
        w[conv < max_val * (1.0 - 5 * tol)] = 0.0
        s = w.sum()
        if s > 0:
            w /= s
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
    else:  # "adaptive"
        # Binary: only exact max
        w = np.zeros_like(conv)
        w[conv == max_val] = 1.0
        w /= w.sum()
    return w


def subgrad_momentum(a0, time_limit, lr_init=0.02, momentum_init=0.92,
                     tol_frac=0.02, wmode="smooth"):
    """Subgradient descent with heavy-ball momentum on simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    lr = lr_init
    mom = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        peak = conv.max()

        if peak < best_obj - 1e-14:
            best_obj = peak
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.1)
        else:
            no_improve += 1
            if no_improve % 80 == 0:
                lr *= 0.96
            if no_improve > 400:
                rng = np.random.default_rng(it * 7919 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
                v = np.zeros(n)
                lr = lr_init * 0.6
                mom = momentum_init
                no_improve = 0
                continue

        # Compute weights and subgradient
        w = peak_weights(conv, mode=wmode, tol=tol_frac)
        g = compute_subgrad(a, w)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g /= g_norm

        # Momentum update
        v = mom * v - lr * g
        a_new = project_simplex(a + v)

        # Line search
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < peak:
            a = a_new
        else:
            improved = False
            for frac in [0.7, 0.4, 0.2, 0.1]:
                a_try = project_simplex(a + frac * v)
                c_try = fast_convolve(a_try)
                if c_try.max() < peak:
                    a = a_try
                    improved = True
                    break
            if not improved:
                v = mom * v  # Keep momentum but don't move

    return best_a, best_obj


def spectral_flatten(a0, time_limit, gamma_init=0.05):
    """Frequency-domain spectral flattening via phase-preserving magnitude adjustment."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    N = next_fast_len(3 * n)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    gamma = gamma_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = fft(a, N)
        power = np.abs(A) ** 2
        phase = np.angle(A)

        # Current convolution (peaks in time domain)
        conv_cur = np.real(ifft(power, N))[:2 * n - 1]
        peak = conv_cur.max()

        if peak < best_obj - 1e-14:
            best_obj = peak
            best_a = a.copy()
            no_improve = 0
            gamma = min(gamma * 1.03, 0.4)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                gamma = max(0.02, gamma * 0.93)
            if no_improve > 300:
                rng = np.random.default_rng(it * 104729 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
                continue

        # Flatten: blend power with its moving average
        # Use a simple smoothing in frequency domain
        power_mean = power.mean()
        power_new = (1.0 - gamma) * power + gamma * power_mean
        # Preserve DC (sum=1 constraint)
        power_new[0] = power[0]

        # Reconstruct with new magnitude, same phase
        new_amp = np.sqrt(np.maximum(power_new, 0.0))
        A_new = new_amp * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, N))[:n]
        a = project_simplex(a_new)

    return best_a, best_obj


def peak_squeeze(a0, time_limit, squeeze_frac=0.05, n_target=8):
    """
    Targeted peak reduction: identify contributions to the peak lag,
    and redistribute mass away from high-contribution positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        peak = conv.max()
        peak_lag = int(np.argmax(conv))

        if peak < best_obj - 1e-14:
            best_obj = peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                pert_scale = 0.003 + 0.003 * min(1.0, (no_improve - 300) / 200.0)
                a = project_simplex(best_a + rng.standard_normal(n) * pert_scale)
                no_improve = 0
                continue

        # Find contributing pairs for the peak lag
        lo = max(0, peak_lag - n + 1)
        hi = min(n - 1, peak_lag)
        if lo > hi:
            lo, hi = 0, n - 1

        idx = np.arange(lo, hi + 1)
        products = a[idx] * a[peak_lag - idx]

        # Select top contributors
        n_sel = min(n_target, len(idx))
        top_positions = np.argsort(products)[::-1][:n_sel]
        active_idx = idx[top_positions]

        # For each active position, try moving mass to a low-contribution position
        for pos in active_idx:
            other = peak_lag - pos
            if pos == other:
                continue

            # Try moving a small amount from pos to a random position with low product
            transfer = squeeze_frac * min(a[pos], 0.1)
            if transfer < 1e-10:
                continue

            # Find a target position with low contribution to peak
            target_scores = products.copy()
            target_scores[top_positions] = np.inf  # Don't transfer to other active
            # Find positions among non-active with lowest scores
            if len(target_scores) > n_sel:
                non_active = np.argsort(target_scores)[:min(3, len(target_scores) - n_sel)]
            else:
                non_active = np.array([0])

            for na in non_active:
                target_pos = idx[na]
                if target_pos == pos or target_pos == other:
                    continue

                candidate = a.copy()
                candidate[pos] -= transfer / 2.0
                candidate[target_pos] += transfer / 2.0
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs > 1e-15:
                    candidate /= cs

                c_new = fast_convolve(candidate)
                if c_new.max() < peak:
                    a = candidate
                    break

    return best_a, best_obj

def multi_scale_perturb(a0, scale=0.05):
    """Generate multi-scale structured perturbation."""
    n = len(a0)
    rng = np.random.default_rng()
    pert = np.zeros(n)
    x = np.arange(n)
    for s in range(5):
        freq = rng.integers(2, max(3, n // (2 ** s) + 1))
        amp = scale / (s + 1)
        phase = rng.random() * 2.0 * np.pi
        pert += amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return pert


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

            print(f"  *** NEW BEST: {score:.8f} (n={len(a)}) ***")
        return score

    # === Phase 1: Primary subgradient descent (longest phase) ===
    print("\n=== Phase 1: Subgradient + Momentum ===")
    t_end1 = deadline - 900
    cur = best_seq if best_seq is not None else parent
    t_alloc = min(450, max(30, t_end1 - time.monotonic()))
    print(f"  n={len(cur)}, t={t_alloc:.0f}s, lr=0.025, mom=0.93")
    result, obj = subgrad_momentum(cur, time_limit=t_alloc,
                                    lr_init=0.025, momentum_init=0.93,
                                    tol_frac=0.02, wmode="smooth")
    try_update(result)

    # === Phase 2: Spectral flattening ===
    print("\n=== Phase 2: Spectral Flattening ===")
    t_end2 = deadline - 600
    cur = best_seq if best_seq is not None else cur
    t_alloc = min(250, max(30, t_end2 - time.monotonic()))
    if t_alloc > 20:
        print(f"  n={len(cur)}, t={t_alloc:.0f}s")
        result, obj = spectral_flatten(cur, time_limit=t_alloc, gamma_init=0.04)
        try_update(result)

    # === Phase 3: Peak squeezing ===
    print("\n=== Phase 3: Peak Squeeze ===")
    t_end3 = deadline - 350
    cur = best_seq if best_seq is not None else cur
    t_alloc = min(180, max(20, t_end3 - time.monotonic()))
    if t_alloc > 15:
        print(f"  n={len(cur)}, t={t_alloc:.0f}s")
        result, obj = peak_squeeze(cur, time_limit=t_alloc,
                                    squeeze_frac=0.04, n_target=12)
        try_update(result)

    # === Phase 4: Hybrid gradient (LSE + exact) ===
    print("\n=== Phase 4: Hybrid Gradient ===")
    t_end4 = deadline - 200
    cur = best_seq if best_seq is not None else cur
    t_alloc = min(150, max(20, t_end4 - time.monotonic()))
    if t_alloc > 15:
        print(f"  n={len(cur)}, t={t_alloc:.0f}s")
        result, obj = gradient_with_norm(cur, time_limit=t_alloc,
                                          lr_init=0.01, momentum=0.92,
                                          peak_tol=0.01, beta_lse=30.0)
        try_update(result)

    # === Phase 5: Alternating with perturbations ===
    print("\n=== Phase 5: Alternating + Exploration ===")
    t_end5 = deadline - 80
    cycle = 0
    while time.monotonic() < t_end5 and cycle < 15:
        cycle += 1
        remaining = t_end5 - time.monotonic()
        if remaining < 40:
            break

        cur = best_seq if best_seq is not None else parent
        n_cur = len(cur)

        # Structured perturbation
        pert = multi_scale_perturb(cur, scale=0.04)
        perturbed = project_simplex(cur + pert)
        try_update(perturbed)

        # Short subgradient from perturbed
        t_sg = min(50, remaining * 0.35)
        if t_sg > 15:
            result, _ = subgrad_momentum(perturbed, time_limit=t_sg,
                                          lr_init=0.012, momentum_init=0.93,
                                          tol_frac=0.015)
            try_update(result)

        # Short spectral step
        cur = best_seq if best_seq is not None else cur
        t_sp = min(35, t_end5 - time.monotonic() - 10)
        if t_sp > 12:
            result, _ = spectral_flatten(cur, time_limit=t_sp, gamma_init=0.06)
            try_update(result)

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-length ===")
    cur = best_seq if best_seq is not None else parent
    n_best = len(cur)
    offsets = [64, -64, 128, -128, 256, 32, -32]
    for off in offsets:
        n_try = n_best + off
        if n_try < 128 or n_try > 2000:
            continue
        t_remaining = deadline - time.monotonic() - 10
        if t_remaining < 35:
            break
        t_alloc = min(70, t_remaining * 0.3)
        if t_alloc < 15:
            break

        f_init = interpolate_sequence(cur, n_try)
        print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
        result, _ = subgrad_momentum(f_init, time_limit=t_alloc,
                                      lr_init=0.015, momentum_init=0.92,
                                      tol_frac=0.02)
        try_update(result)

    # === Phase 7: Final intensive polish ===
    print("\n=== Phase 7: Final Polish ===")
    cur = best_seq if best_seq is not None else parent
    t_remaining = deadline - time.monotonic() - 5
    if t_remaining > 20:
        print(f"  Final polish: n={len(cur)}, t={t_remaining:.0f}s")
        # Very fine subgradient
        result, _ = subgrad_momentum(cur, time_limit=t_remaining,
                                      lr_init=0.005, momentum_init=0.95,
                                      tol_frac=0.005, wmode="sharp")
        try_update(result)

    # Final verification
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        try:
            fs = float(evaluate_sequence(final.tolist()))
            if np.isfinite(fs):
                print(f"\nFINAL: score={fs:.10f}, n={len(best_seq)}")
                return [float(x) for x in final]
        except Exception:
            pass

    print("WARNING: Fallback uniform")
    fb = np.ones(512) / 512
    return [float(x) for x in fb]
