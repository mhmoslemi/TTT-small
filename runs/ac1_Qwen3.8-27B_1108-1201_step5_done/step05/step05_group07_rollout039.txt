"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with adaptive softmax
max approximation, geometric temperature annealing, multi-start exploration,
length interpolation, and coordinate descent polishing.
Key innovations: exact gradient via cross-correlation, scale-invariant
formulation, and progressive sharpening from smooth to sharp max.

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def soft_max_value(conv, T):
    """Compute softmax max: (1/T)*log(sum(exp(T*x)))."""
    scaled = T * conv
    mx = scaled.max()
    return mx / T + np.log(np.sum(np.exp(scaled - mx))) / T


def soft_max_weights(conv, T):
    """Compute softmax weights w[k] = T*exp(T*conv[k])/sum(T*exp(T*conv))."""
    scaled = T * conv
    mx = scaled.max()
    w = np.exp(scaled - mx)
    w /= w.sum()
    return w


def smooth_objective_and_grad(a, T):
    """
    Compute the scale-invariant smooth objective and its gradient.
    f(a) = 2n * softmax_max(conv(a,a), T) / (sum(a))^2
    """
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 0.0, np.zeros(n)

    conv = fast_convolve(a)
    M = soft_max_value(conv, T)
    w = soft_max_weights(conv, T)

    # corr[i] = sum_m w[i+m] * a[m] = correlate(w, a, 'valid')[i]
    corr = np.correlate(w, a, mode='valid')

    # df/da[i] = 4n * (corr[i] * S - M) / S^3
    grad = 4.0 * n * (corr * s - M) / (s ** 3)
    val = 2.0 * n * M / (s * s)

    return val, grad


def l_bfgs_at_temperature(a_init, T, max_iter=200, time_budget=30):
    """Run L-BFGS-B at a fixed temperature."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()

    def fun(x):
        if time.monotonic() - t0 > time_budget:
            raise STOP_EXCEPTION
        v, g = smooth_objective_and_grad(x, T)
        return v, g

    try:
        res = minimize(fun, a, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': max_iter, 'maxfun': 500,
                                'gtol': 1e-8, 'ftol': 1e-12,
                                'maxls': 50, 'iprint': -1})
        a_opt = np.maximum(res.x, 0)
        s2 = a_opt.sum()
        if s2 > 1e-15:
            a_opt = a_opt / s2
    except Exception:
        a_opt = a

    return a_opt

def geometric_anneal(a_init, T_start=0.5, T_end=0.001, total_time=60,
                     iters_per_temp=100):
    """Geometric temperature annealing with L-BFGS-B at each level."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n

    # Determine number of temperature levels
    ratio = (T_end / T_start) ** 0.5
    temps = []
    T = T_start
    while T > T_end * 1.01:
        temps.append(T)
        T *= ratio
    temps.append(T_end)

    t_start = time.monotonic()
    for i, T in enumerate(temps):
        elapsed = time.monotonic() - t_start
        if elapsed > total_time * 0.85:
            break
        remaining = total_time - elapsed - 5
        if remaining < 3:
            break
        time_this = min(remaining * 0.15, 20)
        if time_this < 2:
            break

        a = l_bfgs_at_temperature(a, T, max_iter=iters_per_temp,
                                  time_budget=time_this)
        if i % 3 == 0:
            conv = fast_convolve(a)
            print(f"    Anneal T={T:.5f}, max_conv={conv.max():.8f}, "
                  f"obj_approx={2*len(a)*conv.max()/a.sum()**2:.6f}")

    return a


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mu_weights_optimize(f0, time_limit, top_k=50, temperature=0.00005,
                        lr=0.003, momentum=0.85):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr = max(cur_lr * 0.9995, 0.00003)
        cur_temp *= 1.01
        if it % 300 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.9))

        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.003
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

    return best_f


def coordinate_descent_polish(f0, time_limit):
    """Coordinate descent with multiplicative adjustment."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)
    if mid - d >= 0:
        order.append(mid - d)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f

def make_decay_init(n, alpha, rng):

def make_bimodal_init(n, frac, rng):
    """Bimodal initialization mass at two locations."""
    a = np.zeros(n)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-10)
    return f_interp / f_interp.sum()

            a_opt = coordinate_descent_polish(best_seq, t_cd)
            try_seq(a_opt, "cd")

            rem = deadline - time.monotonic() - 20
            if rem > 8:
                t_pex = min(rem * 0.5, 40)
                print(f"  Pair exchange: budget={t_pex:.0f}s")
                a_opt = pair_exchange_search(best_seq, t_pex)
                try_seq(a_opt, "pex")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            # Very fine L-BFGS
            t_fine = min(rem * 0.4, 40)
            print(f"  Fine L-BFGS: budget={t_fine:.0f}s")
            a_opt = geometric_anneal(best_seq, T_start=0.05, T_end=0.0001,
                                     total_time=t_fine, iters_per_temp=200)
            try_seq(a_opt, "fine_anneal")

            # Perturbation + quick optimize
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                for scale in [0.001, 0.002, 0.003]:
                    rem = deadline - time.monotonic() - 8
                    if rem < 5:
                        break
                    noise = rng.standard_normal(n_best) * scale
                    f_pert = project_simplex(best_seq + noise)
                    a_opt = geometric_anneal(f_pert, T_start=0.1, T_end=0.001,
                                             total_time=min(rem * 0.4, 15),
                                             iters_per_temp=50)
                    try_seq(a_opt, f"pert_{scale}")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt = coordinate_descent_polish(best_seq, min(rem, 15))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
