```python
import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_step(a, top_k, temperature):
    """
    Compute subgradient of max(conv(a,a)) using softmax weighting.
    Returns the subgradient vector.

    scaled -= scaled.max()

    ws = w.sum()

        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        # conv[k] = sum_j a[j]*a[k-j]

        #                                + a[j] for the term where second index is j (if k-j != j)
        # Actually: conv[k] = sum_{j} a[j]*a[k-j]
        # d/d a[j] = a[k-j] + (if k-j != j) a[j] ... no
        # Let me be more careful:
        # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d conv[k] / d a[m] = a[k-m] (from the term j=m) + a[m] (from the term j=k-m, if k-m != m)
        # Wait: if we differentiate sum_j a[j]*a[k-j] with respect to a[m]:

def mu_optimize(a0, time_limit, top_k=50, temperature=0.00005, lr=0.003,
                momentum=0.85, verbose=False):
    """Multiplicative-weights subgradient with Nesterov momentum."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = compute_objective(f)
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn
        
        sg = subgradient_step(f_nest, cur_top_k, cur_temp)
        
        v = momentum * v + (1 - momentum) * sg
        
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        
        obj = compute_objective(f)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9998
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_temp *= 1.005
        
        if no_improve > 500:
            # Restart with noise
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise_scale = 0.003 + 0.002 * min(no_improve // 1000, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            cur_temp = temperature * 0.5
            no_improve = 0
    
    return best_f, best_obj


def annealing_optimize(a0, time_limit, n_steps_per_iter=5, verbose=False):
    """
    Simulated annealing on the simplex.
    At each step, perturb a few coordinates and accept/reject based on temperature.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)
    
    current_obj = compute_objective(f)
    best_f = f.copy()
    best_obj = current_obj
    
    T = 0.001  # initial temperature
    T_end = 0.00001
    t0 = time.monotonic()
    total_steps = 0
    
    while time.monotonic() - t0 < time_limit:
        # Do several perturbations before updating temperature
        for _ in range(n_steps_per_iter):
            if time.monotonic() - t0 >= time_limit:
                break
            
            rng = np.random.default_rng()
            # Pick 2-5 coordinates to perturb
            n_perturb = rng.integers(2, 6)
            indices = rng.integers(0, n, size=n_perturb)
            
            old_vals = f[indices].copy()
            # Multiplicative perturbation
            perturb = np.exp(rng.standard_normal(n_perturb) * 0.1)
            f[indices] *= perturb
            f = np.maximum(f, 0)
            
            new_obj = compute_objective(f)
            
            if new_obj < current_obj or rng.random() < np.exp(-(new_obj - current_obj) / T):
                current_obj = new_obj
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_f = f.copy()
            else:
                f[indices] = old_vals
        
        # Cool down
        T *= 0.999
        if T < T_end:
            T = T_end
        
        total_steps += n_steps_per_iter
        if verbose and total_steps % 1000 == 0:
            print(f"    Anneal step {total_steps}, T={T:.6f}, best={best_obj:.8f}")
    
    return best_f, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using SCA (Successive Convex Approximation)
with LP-based subproblem solving, combined with subgradient optimization,
simulated annealing, and coordinate descent for final polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def sca_iteration(a, top_k=100, damping=0.5, max_lp_iter=200):
    """
    One SCA iteration: solve LP to find direction that reduces max conv.
    Returns (new_a, lp_M) or (None, None) if LP fails.
    """
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Build LP
    # minimize M
    # subject to: sum_j 2*a*[k-j]*a[j] - M <= conv[k] for k in peaks
    #             sum(a) = 1
    #             a >= 0, M >= 0
    
    A_rows = []
    b_vals = []
    
    for k in peaks:
        k = int(k)
        row = np.zeros(n + 1)
        row[-1] = -1.0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = 2.0 * a[k - j]
        A_rows.append(row)
        b_vals.append(conv[k])
    
    A_ineq = np.array(A_rows)
    b_ineq = np.array(b_vals)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'maxiter': max_lp_iter})
    
    if not result.success:
        return None, None
    
    new_a = np.maximum(result.x[:n], 0)
    s = new_a.sum()
    if s < 1e-15:
        return None, None
    new_a /= s
    
    return new_a, result.x[-1]


def sca_optimize(a0, time_limit, top_k=100, damping=0.5, verbose=False):
    """SCA optimization loop."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = compute_objective(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_damping = damping
    cur_top_k = top_k
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Get SCA direction
        new_a, lp_M = sca_iteration(f, top_k=cur_top_k)
        
        if new_a is None:
            # LP failed, try with fewer peaks
            cur_top_k = max(20, cur_top_k // 2)
            if cur_top_k < 20:
                break
            continue
        
        # Damped update
        f_new = (1 - cur_damping) * f + cur_damping * new_a
        f_new = np.maximum(f_new, 0)
        fn = f_new.sum()
        if fn < 1e-15:
            f_new = f.copy()
        else:
            f_new /= fn
        
        obj_new = compute_objective(f_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            # Increase damping slightly
            cur_damping = min(1.0, cur_damping * 1.02)
        else:
            no_improve += 1
            # Decrease damping
            cur_damping = max(0.1, cur_damping * 0.95)
        
        f = f_new
        
        if no_improve > 100:
            # Restart with noise
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)
            v = None
            cur_damping = 0.5
            cur_top_k = top_k
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    SCA iter {it}, t={time.monotonic()-t0:.1f}s, obj={best_obj:.8f}, damp={cur_damping:.3f}")
    
    return best_f, best_obj
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using SCA (Successive Convex Approximation) 
with LP subproblems, combined with subgradient optimization, coordinate descent, 
and multi-length exploration.
"""

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sca_matrix(a, peaks=None):
    """Build the constraint matrix for SCA LP."""
    n = len(a)
    conv = fast_convolve(a)
    num_total = 2 * n - 1
    
    if peaks is None:
        peaks = np.arange(num_total)
    
    K = len(peaks)
    A = np.zeros((K, n + 1))
    b = np.zeros(K)
    
    for idx, k in enumerate(peaks):
        k = int(k)
        A[idx, -1] = -1.0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[idx, js:je+1] = 2.0 * a[k-je:k-js+1][::-1]
        b[idx] = conv[k]
    
    return A, b, conv

    new_a = np.maximum(result.x[:n], 0)
    s = new_a.sum()
    if s < 1e-15:

    new_a /= s
