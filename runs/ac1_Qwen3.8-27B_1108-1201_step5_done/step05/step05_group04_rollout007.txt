import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_for_peaks(f, peak_indices):
    """Compute gradient contribution from specific peak positions."""
    n = len(f)
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        # d/d(f[i]) (f*f)[k] = f[k-i] + f[i-k] (with zero-padding)
        for i in range(n):
            val = 0.0
            ki = k - i
            if 0 <= ki < n:
                val += f[ki]
            ik = i - k
            if 0 <= ik < n:
                val += f[ik]
            grad[i] += val
    return grad


def compute_grad_for_peaks_fast(f, peak_indices):
    """Fast computation using vectorized operations."""
    n = len(f)
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        # Contribution from f[k-i] for i in [max(0,k-n+1), min(n-1,k)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            # f[k-i] for i in [lo1, hi1] => indices k-hi1 to k-lo1
            idx = k - np.arange(hi1, lo1 - 1, -1)
            grad[np.arange(lo1, hi1 + 1)] += f[idx]
        
        # Contribution from f[i-k] for i in [max(0,k), min(n-1, k+n-1)]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            # f[i-k] for i in [lo2, hi2] => indices i-k
            idx = np.arange(lo2, hi2 + 1) - k
            grad[np.arange(lo2, hi2 + 1)] += f[idx]
    return grad

    # Constraints: for each k, sum_i G[i,k]*delta[i] - t <= -b[k]
    #              sum_i delta[i] = 0 (handled separately)
    #              delta[i] >= -f[i] (lower bound)
    
    # Actually, let's simplify: since sum(delta)=0, we can eliminate one variable.
    # Let delta[n-1] = -sum_{i=0}^{n-2} delta[i]
    # Then the constraints become:
    # For each k: sum_{i=0}^{n-2} (G[i,k] - G[n-1,k])*delta[i] - t <= -b[k]
    # delta[i] >= -f[i] for i = 0..n-2
    # -sum_{i=0}^{n-2} delta[i] >= -f[n-1], i.e., sum_{i=0}^{n-2} delta[i] <= f[n-1]
    
    n_vars = n  # delta[0..n-2] and t => n variables total (n-1 deltas + 1 t)
    # Actually let's just use n+1 variables and the equality constraint
    
    # Let's use the full formulation with n+1 variables
    # Objective: minimize t (variable index n)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: G @ delta - t <= -b  =>  G @ delta + (-1)*t <= -b
    # And: -delta[i] <= f[i] (i.e., delta[i] >= -f[i])
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    b_ub = []
    
    # Convolution constraints (2n-1 of them)
    for k in range(conv_len):
        # G[i,k] = f[k-i] + f[i-k] for valid indices
        for i in range(n):
            ki = k - i
            ik = i - k
            g = 0.0
            if 0 <= ki < n:
                g += f[ki]
            if 0 <= ik < n:
                g += f[ik]
            if abs(g) > 1e-15:
                rows.append(k)
                cols.append(i)
                vals.append(g)
        # -1 * t
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
        b_ub.append(-b[k])
    
    # Lower bound constraints: delta[i] >= -f[i] => -delta[i] <= f[i]
    for i in range(n):
        rows.append(conv_len + i)
        cols.append(i)
        vals.append(-1.0)
        b_ub.append(f[i])
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Bounds: t is free, delta[i] >= -f[i] (already in constraints)
    bounds = [(-f[i], None) for i in range(n)] + [(None, None)]
    
    A_ub_mat = csr_matrix((vals, (rows, cols)), shape=(2*conv_len - 1 + n, n + 1))
    
    try:
        res = linprog(c, A_ub=A_ub_mat, b_ub=np.array(b_ub),
                      A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_per_lp})
        if res.success and res.x is not None:
            delta = res.x[:n]
            return delta
    except Exception:
        pass
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Novel "cyclical peak shaving" local search (different from previous gradient methods)
2. Multi-length exploration with adaptive interpolation
3. Low-dimensional Fourier shape search via differential evolution
4. Final micro-polish with subgradient method
Key novelty: The peak-shaving method directly manipulates elements contributing
to peaks, rather than following gradient directions. This can escape local
minima that gradient methods get stuck in.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_to_n(f, new_n):
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def peak_shave(a, time_limit, verbose=False):
    """
    Novel local search: directly manipulate elements to reduce peaks.
    At each step, identify the top peak, find the most contributing elements,
    and redistribute mass to quiet positions.
    """
    f = np.maximum(a.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    
    b = fast_convolve(f)
    best_obj = b.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    step_scale = 0.02
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Find top peaks
        k_star = int(np.argmax(b))
        
        # Compute contribution of each element to the peak
        # contrib[i] = f[k_star - i] (if valid) + f[i - k_star] (if valid)
        contrib = np.zeros(n)
        for i in range(n):
            v = 0.0
            ki = k_star - i
            if 0 <= ki < n:
                v += f[ki]
            ik = i - k_star
            if 0 <= ik < n:
                v += f[ik]
            contrib[i] = v
        
        # Top contributors (to reduce)
        top_t = max(3, n // 50)
        top_idx = np.argsort(contrib)[::-1][:top_t]
        top_contrib = contrib[top_idx]
        total_top = top_contrib.sum()
        
        # Quiet positions (to increase)
        threshold = np.median(contrib) * 0.5
        quiet_idx = np.where(contrib < threshold)[0]
        if len(quiet_idx) < top_t:
            quiet_idx = np.argsort(contrib)[:top_t]
        
        # Reduce top contributors, increase quiet positions
        # Amount to move: proportional to the excess
        excess = b[k_star] - b.mean()
        if excess <= 0:
            break
        
        move_amount = min(excess * 0.5, total_top * step_scale * 0.1)
        if move_amount < 1e-15:
            move_amount = f[top_idx[0]] * 0.005
        
        # Proportional reduction
        for i in range(len(top_idx)):
            frac = top_contrib[i] / (total_top + 1e-30)
            f[top_idx[i]] -= move_amount * frac
            if f[top_idx[i]] < 0:
                f[top_idx[i]] = 0.0
        
        # Proportional increase (uniform among quiet)
        n_quiet = len(quiet_idx)
        if n_quiet > 0:
            f[quiet_idx] += move_amount / n_quiet
        
        # Renormalize
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-15:
            break
        f /= s
        
        b = fast_convolve(f)
        obj = b.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            step_scale = min(step_scale * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                step_scale *= 0.9
            if no_improve > 200:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                b = fast_convolve(f)
                best_obj = b.max()
                best_f = f.copy()
                no_improve = 0
                step_scale = 0.02
    
    return best_f, best_obj


def subgradient_search(a, time_limit, K_init=10, K_max=50, lr_init=0.005):
    """
    Subgradient method targeting top K peaks with adaptive K and step size.
    Different from previous: uses backtracking line search and K annealing.
    """
    f = np.maximum(a.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    
    b = fast_convolve(f)
    best_obj = b.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_init
    K = K_init
    v = np.zeros(n)
    momentum = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        
        # Adaptive K: start small, increase over time
        K_cur = min(K_max, K_init + it // 50)
        
        # Get top K_cur peaks
        k_peaks = np.argpartition(b, -K_cur)[-K_cur:]
        peak_vals = b[k_peaks]
        
        # Weights: based on excess above the (K_cur+1)-th peak
        if len(k_peaks) > 1:
            threshold = np.sort(b)[-K_cur-1] if K_cur+1 < conv_len else peak_vals.min()
        else:
            threshold = 0
        excess = np.maximum(peak_vals - threshold, 0)
        w = excess / (excess.sum() + 1e-30)
        
        # Compute gradient
        grad = np.zeros(n)
        for idx, k in enumerate(k_peaks):
            if w[idx] < 1e-12:
                continue
            k = int(k)
            wi = w[idx]
            # d/d(f[i]) b[k] = f[k-i] + f[i-k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += wi * f[k - i]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            for i in range(lo2, hi2 + 1):
                grad[i] += wi * f[i - k]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            no_improve += 100
            break
        grad /= gn
        
        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Line search
        b_new = fast_convolve(f_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.92
                momentum = max(0.8, momentum - 0.01)
            if no_improve > 150:
                rng = np.random.default_rng((it * 104729) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                lr = max(lr_init * 0.3, 0.0005)
                no_improve = 0
    
    return best_f, best_obj


def fourier_shape_search(n, time_limit, n_modes=8, pop_size=40, verbose=True):

    Global search using Fourier parameterization + differential evolution.
    a[j] = softmax(theta_0 + sum_m (theta_{2m-1}*cos(2pi*m*j/n) + theta_{2m}*sin(2pi*m*j/n)))
    """

    rng = np.random.default_rng(42)
    
    x = np.arange(n)
    cos_terms = np.stack([np.cos(2 * np.pi * m * x / n) for m in range(1, n_modes + 1)])
    sin_terms = np.stack([np.sin(2 * np.pi * m * x / n) for m in range(1, n_modes + 1)])
    
    n_params = 1 + 2 * n_modes
    # Population
    pop = rng.standard_normal((pop_size, n_params)) * 0.5
    pop[0, 0] = 0.0  # first individual: near-uniform
    
    def eval_params(params):
        logits = params[0] + np.sum(params[1:2*n_modes:2] * cos_terms.T + 
                                      params[2:2*n_modes+1:2] * sin_terms.T, axis=1)
        # Softmax
        logits = logits - logits.max()
        probs = np.exp(logits)
        probs /= probs.sum()
        # Evaluate
        b = fast_convolve(probs)
        return b.max()
    
    scores = np.array([eval_params(p) for p in pop])
    best_idx = np.argmin(scores)
    best_params = pop[best_idx].copy()
    best_score = scores[best_idx]
    
    if verbose:
        print(f"    Fourier init: score={best_score:.10f}")
    
    F = 0.8  # scale factor
    CR = 0.9  # crossover rate
    
    gen = 0
    while time.monotonic() - t0 < time_limit:
        gen += 1
        # For each individual (except best), create trial
        for i in range(pop_size):
            if time.monotonic() - t0 >= time_limit:
                break
            
            # Select 3 distinct random individuals (not i, not best)
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, 3, replace=False)
            
            # Mutation
            v_vec = pop[best_idx] + F * (pop[r1] - pop[r2])
            
            # Crossover
            trial = np.zeros(n_params)
            j_rand = rng.integers(n_params)
            for j in range(n_params):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v_vec[j]
                else:
                    trial[j] = pop[i][j]
            
            # Evaluate
            trial_score = eval_params(trial)
            
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_params = trial.copy()
                    best_idx = i
        
        if verbose and gen % 20 == 0:
            print(f"    Fourier gen {gen}: best={best_score:.10f}")
    
    # Reconstruct best sequence
    logits = best_params[0] + np.sum(
        best_params[1:2*n_modes:2] * cos_terms.T + 
        best_params[2:2*n_modes+1:2] * sin_terms.T, axis=1)
    logits = logits - logits.max()
    probs = np.exp(logits)
    probs /= probs.sum()
    return probs, best_score


def try_eval(a, best_tracker):
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score

    """
    Main optimization function with 4 phases:
    1. Novel peak-shaving + subgradient local search (50% time)
    2. Multi-length exploration (20% time)
    3. Fourier global shape search (15% time)
    4. Final micro-polish (15% time)
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            print(f"Parent: n={n_parent}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_eval(parent, best_tracker)
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        n_parent = 768
        try_eval(parent, best_tracker)
    
    # === Phase 1: Local search (50%) ===
    print("\n=== Phase 1: Local Search ===")
    t1_end = deadline - (budget_s - 30) * 0.5
    
    # Peak shaving
    t_alloc = min(250, t1_end - time.monotonic())
    if t_alloc > 30:
        print(f"  Peak shaving n={n_parent}, t={t_alloc:.0f}s")
        f_opt, obj = peak_shave(parent, time_limit=t_alloc)
        try_eval(f_opt, best_tracker)
    
    # Subgradient
    current = best_tracker[1] if best_tracker[1] is not None else parent

                                         K_init=5, K_max=30, lr_init=0.003)

    # Another round of peak shaving

    # === Phase 2: Multi-length exploration (20%) ===
