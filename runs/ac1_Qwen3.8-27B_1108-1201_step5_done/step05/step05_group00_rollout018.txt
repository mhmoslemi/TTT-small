```python
    n = len(a)
    
    theta = to_log(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    T = T_init

        if remaining < 1.0:
            break

        def make_obj(theta):
            a_local = from_log(theta)
            conv = fast_convolve(a_local)
            scaled = conv / T
            scaled -= scaled.max()
            return 2.0 * n * T * np.log(np.exp(scaled).sum())
        
        def make_grad(theta):
            a_local = from_log(theta)
            conv = fast_convolve(a_local)
            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()
            g = 2.0 * n * hankel_mv(p, a_local)
            mean_g = np.dot(a_local, g)
            return a_local * (g - mean_g)
        
        try:
            res = minimize(make_obj, theta, jac=make_grad, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_call, 'maxcor': 20, 
                                   'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxls': 50})
            theta_new = res.x
            a_new = from_log(theta_new)
            obj_new = fast_convolve(a_new).max()
            
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
            theta = theta_new
        except Exception as e:
            # If L-BFGS fails, keep current theta
            pass
        
        # Cool down
        T *= T_decay
        if T < T_min:
            T = T_min
    
    return best_a, best_obj


def simulated_annealing(a0, time_limit, T_init=1e-5, T_end=1e-8, 
                        step_init=1e-4, accept_prob_init=0.3):
    """
    Very fine simulated annealing on the simplex.
    Tiny random perturbations with acceptance/rejection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    current_obj = best_obj
    
    t0 = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = (time.monotonic() - t0) / total_time
        
        # Annealing schedule
        T = T_init * (T_end / T_init) ** progress
        step = step_init * (1.0 - 0.5 * progress)
        
        # Generate perturbation
        perturbation = rng.standard_normal(n) * step
        candidate = project_simplex(a + perturbation)
        
        # Evaluate
        candidate_obj = 2.0 * n * fast_convolve(candidate).max()
        
        # Accept/reject
        if candidate_obj < current_obj:
            a = candidate
            current_obj = candidate_obj
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
        else:
            # Metropolis acceptance
            delta = candidate_obj - current_obj
            if delta < 1e-10:  # Numerical noise
                a = candidate
                current_obj = candidate_obj
            else:
                accept_prob = np.exp(-delta / T)
                if rng.random() < accept_prob:
                    a = candidate
                    current_obj = candidate_obj
    
    return best_a, best_obj


def block_coordinate_descent(a0, time_limit, block_size=50):

        scaled = conv / T

        p = np.exp(scaled)
        p = p / p.sum()

        rng = np.random.default_rng(42)
        block_order = rng.permutation(n_blocks)

            end = min(start + block_size, n)
            if start >= end:
                continue

            # We want to minimize sum_k p_k * (conv(a,a))_k
            # which is a quadratic in a

            # sum_k p_k * sum_i a_i * a_{k-i}
            # = a^T M(p) a where M(p)_{i,j} = p_{i+j}

                    H[i_rel, j_rel] = p[i + j]

            # The last term is constant
            # The second term gives the linear part c
            # The first term gives the quadratic part H
            
            # Let me redo:
            H = np.zeros((end - start, end - start))
            c = np.zeros(end - start)

                for j_rel, j in enumerate(range(start, end)):
                    H[i_rel, j_rel] = 2.0 * p[i + j]  # factor of 2 since both i and j are free
                for j in range(start):
                    c[i_rel] += 2.0 * p[i + j] * a[j]
                for j in range(end, n):
                    c[i_rel] += 2.0 * p[i + j] * a[j]
            
            s_remaining = 1.0 - (a[:start].sum() + a[end:].sum())
            
            if s_remaining <= 1e-12:
                continue
            
            # Solve: min x^T H x + c^T x s.t. x >= 0, sum(x) = s_remaining
            # Use a simple projected gradient or the identity:
            # At optimum: 2Hx + c = lambda * 1 (on active set)
            # So x = (2H)^{-1} (lambda * 1 - c)
            # And sum(x) = s_remaining gives lambda
            
            try:
                H_sym = (H + H.T) / 2.0  # Should already be symmetric
                # eigendecomposition or direct solve
                # x = (2H)^{-1}(lambda - c)
                # sum(x) = lambda * sum((2H)^{-1} 1) - sum((2H)^{-1} c) = s_remaining
                # lambda = (s_remaining + sum((2H)^{-1} c)) / sum((2H)^{-1} 1)
                
                inv_H = np.linalg.inv(2.0 * H_sym + 1e-10 * np.eye(end - start))
                b_ones = np.ones(end - start)
                b_c = inv_H @ c
                
                lambda_val = (s_remaining + b_c.sum()) / (inv_H @ b_ones).sum()
                x = inv_H @ (lambda_val * b_ones - c)
                x = np.maximum(x, 0.0)
                
                # Re-normalize
                x_sum = x.sum()
                if x_sum < 1e-15:
                    x = np.ones(end - start) / (end - start) * s_remaining
                else:
                    x = x / x_sum * s_remaining
                
                # Check if this improves
                a_test = a.copy()
                a_test[start:end] = x
                obj_test = fast_convolve(a_test).max()
                
                if obj_test < best_obj - 1e-12:
                    a = a_test
                    best_obj = obj_test
                    best_a = a.copy()
                    improved = True
            except np.linalg.LinAlgError:
                continue
        
        if not improved:
            T *= 0.9
            if T < 1e-5:
                break
    
    return best_a, best_obj


def coarse_to_fine(a0, time_limit):
    """
    Multi-resolution approach:
    1. Downsample, optimize, upsample
    2. This captures large-scale structure
    3. Then refine at full resolution
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    # Try a few downsample levels
    for factor in [2, 4]:
        n_small = n // factor
        if n_small < 16:
            continue
        
        # Downsample
        a_small = np.zeros(n_small)
        for i in range(n_small):
            a_small[i] = a[i*factor:(i+1)*factor].sum()
        a_small = a_small / a_small.sum()
        
        # Optimize the small problem with L-BFGS
        t_alloc = min((time_limit - (time.monotonic() - t0)) * 0.3, 60)
        if t_alloc < 2:
            continue
        
        a_small_opt, _ = lbfqg_log_domain(a_small, time_limit=t_alloc, T_init=0.01)
        
        # Upsample
        a_up = np.zeros(n)
        for i in range(n_small):
            a_up[i*factor:(i+1)*factor] = a_small_opt[i] / factor
        a_up = a_up / a_up.sum()
        
        # Refine at full resolution
        remaining = time_limit - (time.monotonic() - t0) - 60
        if remaining < 5:
            continue
        t_alloc = min(remaining, 120)
        a_refined, _ = lbfqg_log_domain(a_up, time_limit=t_alloc, T_init=0.001)
        
        obj = 2.0 * n * fast_convolve(a_refined).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_refined.copy()
    
    # Final full-resolution L-BFGS
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a_final, _ = lbfqg_log_domain(best_a, time_limit=remaining, T_init=0.0005)
        obj = 2.0 * n * fast_convolve(a_final).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_final.copy()
    
    return best_a, best_obj
```

```python
def swap_sa(a0, time_limit, T_init=1e-6, amount_init=1e-5):
    """SA using swap moves (preserves sum constraint)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    current_obj = best_obj
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        progress = (time.monotonic() - t0) / time_limit
        T = T_init * (1e-4) ** progress
        amount = amount_init * (1.0 - 0.5 * progress)
        
        # Pick two random positions
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Move 'amount' from i to j (or vice versa)
        amt = rng.random() * amount
        direction = rng.integers(0, 2)  # 0: i->j, 1: j->i
        
        if direction == 0:
            if a[i] < amt:
                continue
            a[i] -= amt
            a[j] += amt
        else:
            if a[j] < amt:
                continue
            a[j] -= amt
            a[i] += amt
        
        # Evaluate
        new_obj = 2.0 * n * fast_convolve(a).max()
        
        if new_obj < current_obj:
            current_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            delta = new_obj - current_obj
            if delta < T:
                if rng.random() < np.exp(-delta / T):
                    current_obj = new_obj
            else:
                # Revert
                if direction == 0:
                    a[i] += amt
                    a[j] -= amt
                else:
                    a[j] += amt
                    a[i] -= amt
    
    return best_a, best_obj
```

This is a fixed-point iteration. It converges linearly (like power iteration). L-BFGS should converge much faster (quadratic in the final phase).

The `smoothed_grad_descent` does:
1. Compute p = softmax(conv/T)
2. Gradient step: a ← project(a - lr * grad)
3. Backtracking line search

This is projected gradient descent with backtracking. It converges sublinearly (O(1/k)).

L-BFGS should be significantly better than both of these for the smooth (convex) subproblem. The key advantage is that it uses approximate second-order information to take superlinear steps.

I'm fairly confident the L-BFGS approach will outperform the previous methods in the "final stretch" (when we're close to the optimum). The main risk is that L-BFGS might get stuck at a point where the smooth approximation (log-sum-exp) is not accurate enough to guide it to the true optimum. To mitigate this, we use a decreasing temperature schedule.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- L-BFGS on log-sum-exp smoothed objective in log-domain (primary optimizer)
- Swap-based simulated annealing for ultra-fine local polish
- Block coordinate descent with exact QP solves
- Multi-temperature L-BFGS rounds with restarts
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def from_log(theta):
    """Convert log-domain parameters to probability vector."""
    s = theta - theta.max()
    a = np.exp(s)
    return a / a.sum()


def objective_value(a):
    """Compute 2*n*max(conv(a,a)) for normalized a."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def lbfqg_log_domain(a0, time_limit, T_init=0.001, T_decay=0.65, T_min=1e-7,
                     max_iter_per_round=300, max_rounds=50):
    """
    L-BFGS optimization in log-domain on the smoothed (log-sum-exp) objective.
    Multiple rounds with decreasing temperature for progressive refinement.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    theta = np.log(np.maximum(a, 1e-18))
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    T = T_init

    t0 = time.monotonic()
    round_num = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and round_num < max_rounds:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        round_num += 1

        # Make objective and gradient closures
        def _obj(theta):
            a_local = from_log(theta)
            conv = fast_convolve(a_local)
            scaled = conv / T
            scaled -= scaled.max()
            return 2.0 * n * T * np.log(np.exp(scaled).sum())

        def _grad(theta):
            a_local = from_log(theta)
            conv = fast_convolve(a_local)
            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()
            g = 2.0 * n * hankel_mv(p, a_local)
            mean_g = np.dot(a_local, g)
            return a_local * (g - mean_g)

        try:
            res = minimize(_obj, theta, jac=_grad, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_round, 'maxcor': 20,
                                    'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
            a_new = from_log(res.x)
            obj_new = fast_convolve(a_new).max()

            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            theta = res.x
        except Exception:
            no_improve += 1

        # Cool down
        T *= T_decay
        if T < T_min:
            T = T_min

        # If stuck, do a small perturbation and reset T
        if no_improve > 5:
            rng = np.random.default_rng(round_num * 31 + 7)
            a_pert = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            theta = np.log(np.maximum(a_pert, 1e-18))
            T = max(T_init * 0.3, T_min * 10)
            no_improve = 0

    return best_a, best_obj


def swap_simulated_annealing(a0, time_limit, T_init=2e-7, T_factor=0.3,
                             amount_init=1e-5, jump_every=2000):
    """
    Simulated annealing using swap moves (preserves sum=1 automatically).
    Periodically does a "jump" (Gaussian perturbation) to escape local optima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = objective_value(a)
    current_obj = best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(99991)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_factor ** progress)
        amount = amount_init * (1.0 - 0.7 * progress)

        # Swap move (fast, preserves sum)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        amt = rng.random() * amount + 1e-7
        if rng.integers(0, 2) == 0:
            src, dst = i, j
        else:
            src, dst = j, i

        if a[src] < amt:
            continue

        a[src] -= amt
        a[dst] += amt

        new_obj = objective_value(a)

        if new_obj < current_obj:
            current_obj = new_obj
            no_improve = 0
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            delta = new_obj - current_obj
            if delta < T and rng.random() < np.exp(-delta / T):
                current_obj = new_obj
                no_improve += 1
            else:
                # Revert
                a[src] += amt
                a[dst] -= amt
                no_improve += 1

        # Periodic jump to escape local optima
        if it % jump_every == 0 and no_improve > jump_every // 2:
            jump_size = 1e-5 + 3e-5 * min(1.0, no_improve / (jump_every * 3))
            a = project_simplex(best_a + rng.standard_normal(n) * jump_size)
            current_obj = objective_value(a)
            no_improve = 0

        # If very stuck, do a larger jump
        if no_improve > jump_every * 5:
            a = project_simplex(best_a + rng.standard_normal(n) * 5e-5)
            current_obj = objective_value(a)
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_fine(a0, time_limit):
    """
    Fine coordinate descent: try adjusting individual coordinates and
    pairs of coordinates to reduce the peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = objective_value(a)
    rng = np.random.default_rng(55555)

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Find the current peak
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Determine affected indices for this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = list(range(js, je + 1))

        # Try single-coordinate adjustments
        for idx in affected[:min(len(affected), 20)]:
            # Try moving a tiny amount from idx to a random other position
            if len(affected) < 2:
                break
            other = rng.integers(0, n)
            if other == idx:
                continue

            for frac in [0.01, 0.05, 0.1, 0.2]:
                amt = a[idx] * frac
                if amt < 1e-15:
                    continue
                a[idx] -= amt
                a[other] += amt
                new_obj = objective_value(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    a[idx] += amt
                    a[other] -= amt

        # Try pair adjustments within affected set
        if len(affected) >= 2:
            for _ in range(3):
                i1 = affected[rng.integers(0, len(affected))]
                i2 = affected[rng.integers(0, len(affected))]
                if i1 == i2:
                    continue
                frac = 0.05 + 0.2 * rng.random()
                amt = a[i1] * frac
                if amt < 1e-15:
                    continue
                # Move from i1 to i2
                a[i1] -= amt
                a[i2] += amt
                new_obj = objective_value(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    a[i1] += amt
                    a[i2] -= amt

        if it % 100 == 0 and no_improve > 500:
            # Reset from best with small noise
            a = project_simplex(best_a + rng.standard_normal(n) * 1e-5)
            no_improve = 0
        elif no_improve > 2000:
            break

    return best_a, best_obj


def block_qp_descent(a0, time_limit, block_size=40):
    """
    Block coordinate descent: for each block, solve the exact QP
    min a^T M(p) a s.t. a in simplex, where p = softmax(conv/T).
    Uses the analytical solution for the block subproblem.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    T = 0.001
    rng = np.random.default_rng(42)

    t0 = time.monotonic()
    no_round_improve = 0
    n_blocks = (n + block_size - 1) // block_size

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        block_order = rng.permutation(n_blocks)
        round_improved = False

        for bi in block_order:
            start = bi * block_size
            end = min(start + block_size, n)
            if start >= end:
                continue
            bs = end - start

            # Build H and c for the block
            H = np.zeros((bs, bs))
            c = np.zeros(bs)

            for i_rel in range(bs):
                i = start + i_rel
                for j_rel in range(bs):
                    j = start + j_rel
                    H[i_rel, j_rel] = 2.0 * p[i + j]
                for j in range(n):
                    if j >= start and j < end:
                        continue
                    c[i_rel] += 2.0 * p[i + j] * a[j]

            s_remaining = 1.0 - (a[:start].sum() + a[end:].sum())
            if s_remaining <= 1e-12:
                continue

            # Solve: min x^T H x + c^T x s.t. x >= 0, sum(x) = s_remaining
            # KKT: 2Hx + c = lambda (on active set)
            # x = (2H)^{-1}(lambda - c), sum(x) = s_remaining
            try:
                A = H + 1e-12 * np.eye(bs)
                inv_A = np.linalg.inv(A)
                ones = np.ones(bs)
                sum_inv = (inv_A @ ones).sum()
                sum_inv_c = (inv_A @ c).sum()

                if abs(sum_inv) < 1e-15:
                    continue

                lambda_val = (s_remaining * sum_inv + sum_inv_c) / sum_inv
                x = inv_A @ (lambda_val * ones - c)
                x = np.maximum(x, 0.0)
                x_sum = x.sum()
                if x_sum < 1e-15:
                    continue
                x = x / x_sum * s_remaining

                a_test = a.copy()
                a_test[start:end] = x
                obj_test = fast_convolve(a_test).max()

                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    best_a = a.copy()
                    round_improved = True
            except np.linalg.LinAlgError:
                continue

        if not round_improved:
            no_round_improve += 1
            T *= 0.85
            if T < 1e-5:
                break
            if no_round_improve > 30:
                break
        else:
            no_round_improve = 0

    return best_a, best_obj


def multi_temperature_lbfqg(a0, time_limit):
    """
    Run L-BFGS with multiple temperature schedules and keep the best.
    Also tries restarting from small perturbations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed=777)

    temperatures = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    decays = [0.6, 0.65, 0.7, 0.5, 0.6, 0.5]

    for ti, (T_init, decay) in enumerate(zip(temperatures, decays)):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_alloc = min(remaining * 0.4, 200)

        a_opt, obj_opt = lbfqg_log_domain(best_a, time_limit=t_alloc,
                                          T_init=T_init, T_decay=decay)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"    L-BFGS T={T_init}: obj={best_obj:.10f}")

        # Try a perturbed restart
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5 or ti < 2:
            continue
        t_alloc = min(remaining * 0.3, 100)
        perturbed = project_simplex(best_a + rng.standard_normal(n) * 0.003)
        a_opt2, obj_opt2 = lbfqg_log_domain(perturbed, time_limit=t_alloc,
                                            T_init=T_init * 2, T_decay=decay)
        if obj_opt2 < best_obj:
            best_obj = obj_opt2
            best_a = a_opt2.copy()
            print(f"    L-BFGS perturbed T={T_init}: obj={best_obj:.10f}")

    return best_a, best_obj
```
