def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses all 2n-1 constraints."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_active(f, M, n, active_k):
    """Solve LP with only active constraints."""
    c = -np.ones(n)

    A_ub = np.zeros((K, n))

        je = min(n - 1, k)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

def interpolate_sequence(f, new_n):

    old_n = len(f)

    old_x = np.linspace(0, 1, old_n)

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_active(f, M, n, active_k):
    """Solve LP with only active constraints."""
    c = -np.ones(n)
    K = len(active_k)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def line_search_obj(f, g, alpha):
    """Evaluate objective at mixing parameter alpha."""
    f_new = f + alpha * (g - f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return np.inf
    f_new = f_new / s
    return fast_convolve(f_new).max()


def line_search(f, g, n, max_iter=25):

        v = line_search_obj(f, g, a)
        if v < best_val:
            best_val = v
            best_alpha = a

    if best_alpha == 0.0:
        return 0.0

    # Golden section refinement
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = max(0.0, best_alpha - 0.08)
    hi = min(1.0, best_alpha + 0.08)

    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = line_search_obj(f, g, c)
    fd = line_search_obj(f, g, d)

    for _ in range(max_iter):

def subgradient_peak(f0, time_limit, lr=0.003, mom=0.92, top_k=30, temp=0.00004):

def coord_descent_binary(f0, time_limit=60):
    """Coordinate descent with binary search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue

            old_val = f[j]
            best_j_val = old_val
            best_j_obj = best_obj

            # Binary search: try increasing and decreasing
            for direction in [1.0, -1.0]:
                lo, hi = 0.0, old_val * 0.8 if direction < 0 else min(old_val * 2.5, 0.4)
                if lo >= hi or hi - lo < 1e-12:
                    continue
                for _ in range(10):
                    mid = (lo + hi) / 2.0
                    if mid < 1e-14:
                        mid = hi
                    f_trial = f.copy()
                    f_trial[j] = mid
                    st = f_trial.sum()
                    f_trial = f_trial / st
                    obj = fast_convolve(f_trial).max()
                    if obj < best_j_obj - 1e-15:
                        best_j_obj = obj
                        best_j_val = mid
                        if direction > 0:
                            lo = mid
                        else:
                            hi = mid
                    else:
                        if direction > 0:
                            hi = mid
                        else:
                            lo = mid

            if best_j_val != old_val:
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 2:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_anti_palindromic(a, n):

    # Blend towards anti-palindromic
    result = 0.6 * f + 0.4 * rev
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, uniform init: score={best_score:.6f}")

    n_base = len(parent)
    print(f"Starting optimization from n={n_base}, score={best_score:.6f}")

    # === Phase 1: Full-constraint LP-FW on parent ===
    print("\n=== Phase 1: LP-FW full constraints on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(250, rem * 0.28)
    if t1 > 30:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True,
                                  use_full=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: {best_score:.6f}")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length (best={best_score:.6f}) ===")
    lengths_to_try = sorted(set([max(128, n_base - 64), n_base - 32, n_base,
                                  n_base + 32, n_base + 64, n_base + 96]))
    for target_n in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        f_init = interpolate_sequence(best_seq, target_n)
        t_alloc = min(rem * 0.18, 140)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc,
                                  verbose=False, use_full=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST n={target_n}: {best_score:.6f}")

    # === Phase 3: Peak-targeted subgradient ===
    print(f"\n=== Phase 3: Peak SG (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sg = min(rem * 0.18, 120)
        a_opt, _ = subgradient_peak(best_seq, time_limit=t_sg,
                                    lr=0.002, mom=0.92,
                                    top_k=max(20, n_best // 10),
                                    temp=0.00003)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  SG result: {best_score:.6f}")

        # Fine SG
        rem = deadline - time.monotonic()
        if rem > 20:
            a_opt2, _ = subgradient_peak(best_seq, time_limit=min(rem * 0.12, 80),
                                         lr=0.001, mom=0.94,
                                         top_k=max(15, n_best // 12),
                                         temp=0.00002)
            sc = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt2.copy()
                print(f"  SG fine: {best_score:.6f}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Binary CD (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.22, 180)
        print(f"  CD t={t_cd:.0f}s")
        a_opt, _ = coord_descent_binary(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  CD result: {best_score:.6f}")

    # === Phase 5: Diverse structural restarts ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    restart_inits = []

    # Random perturbations of increasing size
    for scale in [0.003, 0.005, 0.008, 0.012]:
        restart_inits.append((f"pert_{scale:.3f}",
                              project_simplex(best_seq + rng.standard_normal(n_best) * scale)))

    # Reverse blend (exploits non-palindromic structure)
    rev = best_seq[::-1]
    restart_inits.append(("rev_03", project_simplex(0.7 * best_seq + 0.3 * rev)))
    restart_inits.append(("rev_05", project_simplex(0.5 * best_seq + 0.5 * rev)))

    # Flat blend
    restart_inits.append(("flat_015",
                          project_simplex(0.85 * best_seq + 0.15 * np.ones(n_best) / n_best)))

    # Gaussian center
    center = np.arange(n_best) - n_best / 2.0
    gauss = np.exp(-2.0 * center**2 / n_best**2)
    gauss = gauss / gauss.sum()
    restart_inits.append(("gauss_02",
                          project_simplex(0.8 * best_seq + 0.2 * gauss)))

    # Edges emphasized
    edges = np.zeros(n_best)
    edges[:n_best//4] = 1.0
    edges[-n_best//4:] = 1.0
    edges = edges / edges.sum()
    restart_inits.append(("edges_02",
                          project_simplex(0.8 * best_seq + 0.2 * edges)))

    for i, (name, f_init) in enumerate(restart_inits):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem / max(1, len(restart_inits) - i + 1), 130)
        print(f"  Restart [{name}], t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc,
                                  verbose=False, use_full=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{name}]: {best_score:.6f}")

    # === Phase 6: Final alternating polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    for rnd in range(4):
        rem = deadline - time.monotonic()
        if rem < 15:
            break

        # LP-FW
        t_lpfw = min(rem * 0.25, 80)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw, use_full=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Round {rnd} LP-FW: {best_score:.6f}")

        # Subgradient
        rem = deadline - time.monotonic()
        if rem < 12:
            break
        t_sg = min(rem * 0.15, 50)
        a_opt, _ = subgradient_peak(best_seq, time_limit=t_sg,
                                    lr=0.0012 * (0.7 ** rnd), mom=0.93,
                                    top_k=max(10, n_best // 12),
                                    temp=0.00002 * (0.8 ** rnd))
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Round {rnd} SG: {best_score:.6f}")

        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem < 12:
            break
        t_cd = min(rem * 0.15, 50)
        a_opt, _ = coord_descent_binary(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Round {rnd} CD: {best_score:.6f}")

    # === Phase 7: Tiny perturbations for final squeeze ===
    print(f"\n=== Phase 7: Final squeeze (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    if rem > 15:
        for i in range(6):
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0008 * i)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.2, 25)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_pert,
                                      use_full=True)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Squeeze {i}: {best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
