    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    n = len(f)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based optimization
(dual sequence construction) with adaptive step sizes, multiple initializations,
multi-length exploration, and local search polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(f):
    """Build matrix A such that A @ g = convolve(f, g) (full mode)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = f[k - j_start:k - j_end + 1][::-1]
    return A

def solve_lp_for_complement(f, M):
    """Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def structured_inits(n, rng, count=5):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    inits.append(hann / hann.sum())
    
    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Sinusoidal perturbation of uniform
    freq = rng.integers(2, max(3, n // 50))

    inits.append(sine / sine.sum())

    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())
    
    return inits[:count]

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based complement sequence
construction combined with subgradient refinement, multi-length exploration,
and local search. The LP step finds a "dual" sequence with bounded
cross-correlation, enabling principled descent on the objective.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix A such that A @ g equals np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = f[k - js:k - je + 1][::-1]
    return A


def solve_lp_complement(f, M):
    """Solve: max 1^T g s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_optimize(f0, time_limit, step_sizes=None, label=""):
    """LP-based iterative optimization with adaptive step selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    if step_sizes is None:
        step_sizes = [0.5, 0.3, 0.15]

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1

        g = solve_lp_complement(f, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-15:
            no_improve += 1
            if no_improve > 30:
                break
            continue

            f_new = np.maximum(f_new, 0.0)

            cm = fast_convolve(f_new).max()
            if cm < best_new_M:
                best_new_M = cm
                best_t = t

        if best_t < 0:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        fs2 = f.sum()
        if fs2 < 1e-15:
            f = best_f.copy()
            conv = fast_convolve(f)
            M = conv.max()
            no_improve += 1
            continue
        f /= fs2
        conv = fast_convolve(f)
        M = conv.max()

            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            break

    return best_f, best_obj


def project_simplex(v):
    """Projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
