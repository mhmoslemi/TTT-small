"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral shaping + multi-method refinement.
Key innovation: Iterative spectral flattening (geometric interpolation of power spectrum)
alternated with time-domain non-negativity projection, combined with subgradient descent
and multi-n exploration. Targets the theoretical minimum of ~2n/(2n-1) ≈ 1.0.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    return np.fft.ifft(A * A, fft_size).real[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_shaping(a0, time_limit, alpha_init=0.4, alpha_min=0.02, verbose=False):
    """
    Iterative spectral flattening:
    1. Compute power spectrum P = |FFT(a)|^2
    2. Geometrically interpolate P toward uniform (reduce dynamic range)
    3. Reconstruct time-domain sequence with modified spectrum, keep phase
    4. Project to non-negative, sum=1
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    alpha = alpha_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2
        P_mean = P.sum() / N

        # Geometric interpolation toward uniform spectrum
        log_P = np.log(P + 1e-30)
        log_P_mean = np.log(P_mean + 1e-30)
        log_P_new = (1.0 - alpha) * log_P + alpha * log_P_mean
        P_new = np.exp(log_P_new)

        # Reconstruct with flattened spectrum, preserve phase
        ratio = np.sqrt(P_new / (P + 1e-30))
        A_new = A * ratio

        a_new = np.fft.ifft(A_new, N).real[:n]
        a_new = np.maximum(a_new, 0.0)

        a_new = a_new / s

            best_a = a_new.copy()
            a = a_new

        if verbose and it % 500 == 0:

def subgradient_descent(a0, time_limit, top_k=20, lr_init=0.01, mom=0.9, verbose=False):
    """
    Multi-peak subgradient descent with momentum and adaptive step size.
    Focuses on reducing the top-k peaks of the auto-convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        conv_len = len(conv)

        # Select top-k peaks
        k_actual = min(top_k, conv_len)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        # Weight by inverse rank
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        weights /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        # Momentum update
        v = mom * v - lr * sg
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack: revert and reduce lr
            lr *= 0.95
            v *= 0.5

        # Adaptive lr
        if no_improve > 50:
            lr = max(lr * 0.9, 1e-6)
        if no_improve > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * (0.002 + 0.005 * min(1.0, (no_improve - 200) / 500.0)))
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj

    it = 0

            best_a = a_new.copy()
            a = a_new
            no_improve = 0

            no_improve += 1
            lr *= 0.98
            T *= 1.02

        if no_improve > 100:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, step=0.002):
    """
    Cyclic coordinate descent: for each j, try increasing/decreasing a[j]
    to reduce the peak of the auto-convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        improved = False
        conv = fast_convolve(a)
        peak = conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            if a[j] < 1e-10:
                continue

            # Try reducing a[j]
            for delta_frac in [0.05, 0.1, 0.2]:
                delta = a[j] * delta_frac
                if delta > peak * 0.01:
                    delta = peak * 0.01
                a[j] -= delta
                # Redistribute to random other index
                k = rng.integers(0, n)
                if k == j:
                    k = (j + 1) % n
                a[k] += delta
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[j] += delta
                    a[k] -= delta
                    break
                a = a / s
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[j] += delta
                    a[k] -= delta
                    a = a / a.sum()

        if not improved:
            # Try a random pair swap
            i = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if i == j2:
                j2 = (i + 1) % n
            for amt in [0.001, 0.002, 0.005]:
                if a[i] < amt:
                    continue
                a[i] -= amt
                a[j2] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    a = best_a.copy()
                    break
                else:
                    a[j2] -= amt
                    a[i] += amt
                    a = a / a.sum()

    return best_a, best_obj


def fw_lp_step(f0, M, n):
    """Single Frank-Wolfe step: solve LP max sum(g) s.t. f*g <= M, g >= 0."""
    from scipy.optimize import linprog
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f0
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        g = fw_lp_step(f, M * 1.01, n)
        if g is None:
            g = fw_lp_step(f, M * 1.05, n)
        if g is None:
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        best_val = M
        best_alpha = 0.0
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            break

    return best_f, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / (n - 1))) ** 2 + 0.01
    inits.append(c1 / c1.sum())

    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1.0 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    gc = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(gc / gc.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def search_one_n(n, parent, time_budget, seed=42):
    """Search for best sequence of length n within time_budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + time_budget

    best_a = None
    best_obj = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_obj
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_n = a / s
        obj = fast_convolve(a_n).max()
        score = 2.0 * n * obj
        if score < best_obj:
            best_obj = score
            best_a = a_n.copy()
            print(f"      [{n}] {label}: score={score:.8f} (max_c={obj:.8e})")
        return score

    # Generate initializations
    inits = generate_initializations(n, parent, rng)

    # Try spectral shaping from each initialization (short time each)
    per_init_time = min(time_budget * 0.3 / len(inits), 30)

    for i, init in enumerate(inits):
        if time.monotonic() > deadline - time_budget * 0.4:
            break
        remaining = deadline - time.monotonic() - time_budget * 0.3
        t_alloc = min(per_init_time, max(remaining, 2))
        if t_alloc < 1:
            break
        a_opt, _ = spectral_shaping(init, time_limit=t_alloc, alpha_init=0.3 + 0.1 * (i % 3))
        try_seq(a_opt, f"spectral_init{i}")

    # Now refine the best with multiple methods
    if best_a is not None:
        # Spectral shaping (longer)
        remaining = deadline - time.monotonic() - time_budget * 0.2
        if remaining > 5:
            t_alloc = min(remaining, time_budget * 0.25)
            a_opt, _ = spectral_shaping(best_a, time_limit=t_alloc, alpha_init=0.2)
            try_seq(a_opt, "spectral_refine")

        # Subgradient
        remaining = deadline - time.monotonic() - time_budget * 0.15
        if remaining > 5:
            t_alloc = min(remaining, time_budget * 0.2)
            a_opt, _ = subgradient_descent(best_a, time_limit=t_alloc, top_k=30, lr_init=0.005, mom=0.92)
            try_seq(a_opt, "subgrad")

        # Smoothed gradient
        remaining = deadline - time.monotonic() - time_budget * 0.1
        if remaining > 5:
            t_alloc = min(remaining, time_budget * 0.15)
            a_opt, _ = smoothed_gradient_descent(best_a, time_limit=t_alloc, T_init=0.0005, lr_init=0.01)
            try_seq(a_opt, "smooth_grad")

        # Spectral shaping again (second pass with different params)
        remaining = deadline - time.monotonic() - time_budget * 0.05
        if remaining > 5:
            t_alloc = min(remaining, time_budget * 0.1)
            a_opt, _ = spectral_shaping(best_a, time_limit=t_alloc, alpha_init=0.15, alpha_min=0.005)
            try_seq(a_opt, "spectral_pass2")

        # Coordinate descent polish
        remaining = deadline - time.monotonic()
        if remaining > 3:
            t_alloc = min(remaining, time_budget * 0.05)
            a_opt, _ = coordinate_descent_polish(best_a, time_limit=t_alloc)
            try_seq(a_opt, "coord_polish")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-n search with spectral shaping as the primary method.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, None)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(256) / 256

    # Track global best
    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a_n = a / s
        score = float(evaluate_sequence(a_n.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_n.copy()
            print(f"  *** GLOBAL BEST: score={score:.8f} (n={len(a_n)}) ***")

    # Try parent directly
    if parent is not None:
        update_best(parent)

    # Select n values to try
    # Focus on n where 2n-1 is close to a power of 2 (efficient FFT, less waste)
    n_candidates = [128, 256, 512, 768, 1024, 1536, 2048]

    # Filter: skip n that are too similar to parent or too extreme
    n_parent = len(parent) if parent is not None else 256
    n_candidates = [n for n in n_candidates if 64 <= n <= 4096]

    # Always include parent length
    if n_parent not in n_candidates and 64 <= n_parent <= 4096:
        n_candidates.append(n_parent)
    n_candidates.sort()

    # Allocate time: more time to larger n (lower theoretical minimum)
    # But also weight by proximity to parent
    total_time = deadline - time.monotonic() - 60
    weights = {}
    for n in n_candidates:
        # Theoretical minimum score: 2n/(2n-1)
        theo_min = 2.0 * n / (2 * n - 1)
        # Weight by how much improvement potential there is
        current_best_at_n = 2.0 * n / n  # uniform gives score=2
        potential = current_best_at_n - theo_min
        # Prefer n close to parent for warm start
        proximity = 1.0 / (1.0 + abs(n - n_parent) / 512.0)
        weights[n] = potential * proximity

    total_w = sum(weights.values())
    time_alloc = {}
    for n in n_candidates:
        frac = weights[n] / total_w
        time_alloc[n] = total_time * frac

    # Sort by time allocation descending (do biggest first)
    n_sorted = sorted(n_candidates, key=lambda x: -time_alloc[x])

    print(f"\nTime allocation: {[(n, f\"{time_alloc[n]:.0f}s\") for n in n_sorted]}")

    # Search each n
    for idx, n in enumerate(n_sorted):
        remaining = deadline - time.monotonic() - 30
        if remaining < 10:
            break

        t_alloc = min(time_alloc[n], remaining)
        print(f"\n  Searching n={n} (budget {t_alloc:.0f}s, entry {idx+1}/{len(n_sorted)})")

        a_opt, obj = search_one_n(n, parent, time_budget=t_alloc, seed=seed + idx * 100)
        if a_opt is not None:
            update_best(a_opt)

    # Final: try a few more things with remaining time
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            n_best = len(best_seq)
            print(f"\n  Final polish on n={n_best} (budget {remaining:.0f}s)")

            # Spectral with very low alpha
            t1 = min(remaining * 0.4, 30)
            a_opt, _ = spectral_shaping(best_seq, time_limit=t1, alpha_init=0.08, alpha_min=0.003)
            update_best(a_opt)

            # FW-LP
            remaining2 = deadline - time.monotonic() - 10
            if remaining2 > 10:
                t2 = min(remaining2 * 0.5, 40)
                a_opt, _ = fw_lp_optimize(best_seq, time_limit=t2)
                update_best(a_opt)

            # Subgradient with different params
            remaining3 = deadline - time.monotonic() - 10
            if remaining3 > 10:
                t3 = min(remaining3 * 0.5, 30)
                a_opt, _ = subgradient_descent(best_seq, time_limit=t3, top_k=50, lr_init=0.003, mom=0.95)
                update_best(a_opt)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
