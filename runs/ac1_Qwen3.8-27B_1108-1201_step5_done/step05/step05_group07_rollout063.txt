**Key insight for local search**: At the optimum, the KKT conditions say that the subgradient of max(b) should be constant (equal to lambda) across all positive coordinates. So we can measure how "close to optimal" we are by looking at the variance of the subgradient.

**Better local search algorithm**:
1. Compute b = conv(a, a).
2. Find the peak lag k*.
3. Compute the subgradient: g_j = 2 * a[k*-j] (for valid j).
4. The "violation" at coordinate j is g_j - mean(g) (for positive a_j).
5. Coordinates with high g_j are "contributing too much" to the peak.
6. Coordinates with low g_j are "under-contributing".
7. Transfer mass from high-g to low-g coordinates.
8. Check if the max peak decreased.

Actually, let me think about this differently. The most effective local moves are:
- **Single coordinate**: Change a_j, rescale others to maintain sum=1.
- **Two coordinates**: Change a_i and a_j, rescale others.
- **Group moves**: Change a group of coordinates.

For single coordinate: if we increase a_j by δ (and decrease others proportionally), the change in b_{k*} is approximately `2 * a[k*-j] * δ - (sum of contributions from the rescaled part)`. The net effect depends on the full gradient.

Let me implement a "smart pair exchange":
1. Find peak lag k*.
2. For each pair (i, j) where i+j = k*: reducing either a_i or a_j helps.
3. But reducing a_i also affects other lags.
4. The "best" pair to reduce is the one with the largest a_i * a_j (largest contribution to the peak).
5. The "best" destination for the mass is a coordinate that doesn't contribute to any peak.

def smart_pair_exchange(f, time_limit):
    """Smart pair exchange: reduce peak by transferring mass to non-peak coordinates."""
    f = f.copy()
    n = len(f)
    best_f = f.copy()
    best_obj = auto_conv(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = auto_conv(f)
        max_b = b.max()
        
        if max_b < best_obj - 1e-15:
            best_obj = max_b
            best_f = f.copy()
        
        # Find all peak lags (within 1% of max)
        threshold = max_b * 0.99
        peak_lags = np.where(b >= threshold)[0]
        
        # For each peak lag, find the best pair to reduce
        found_improvement = False
        for k in peak_lags:
            k = int(k)
            # Valid range for i: max(0, k-n+1) <= i <= min(n-1, k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            
            if i_end < i_start:
                continue
            
            # Contributions: f[i] * f[k-i]
            idx = np.arange(i_start, i_end + 1)
            contribs = f[idx] * f[k - idx]
            
            # Best pair to reduce: largest contribution
            local_max_idx = np.argmax(contribs)
            i_best = idx[local_max_idx]
            j_best = k - i_best
            
            if i_best == j_best:
                # Can only reduce f[i_best]
                # Transfer to a coordinate that minimizes the increase in max(b)
                for frac in [0.01, 0.05, 0.1]:
                    amt = f[i_best] * frac

                    mask = np.ones(n, dtype=bool)
                    mask[i_best] = False
                    total_others = f[mask].sum()
                    if total_others < 1e-15:
                        f[i_best] += amt
                        continue
                    f[mask] = f[mask] * (total_others + amt) / total_others
                    # Or: normalize
                    f = np.maximum(f, 0)
                    f /= f.sum()
                    
                    new_obj = auto_conv(f).max()
                    if new_obj < best_obj - 1e-14:

                        f = best_f.copy()  # Revert (or undo the change)

                # Try reducing both and putting mass into a "safe" coordinate
                for frac in [0.01, 0.03, 0.05]:
                    amt_i = f[i_best] * frac
                    amt_j = f[j_best] * frac
                    if amt_i < 1e-15 or amt_j < 1e-15:
                        continue
                    f[i_best] -= amt_i
                    f[j_best] -= amt_j
                    total_transfer = amt_i + amt_j
                    # Distribute to other coordinates proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[i_best] = False
                    mask[j_best] = False
                    total_others = f[mask].sum()
                    if total_others < 1e-15:

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = (time.monotonic() - t0) / time_limit
        lr = lr_start * (lr_end / lr_start) ** min(progress * 1.2, 1.0)
        temp = temp_start * (temp_end / temp_start) ** min(progress * 1.5, 1.0)

        g, b = lse_gradient(f, temp)
        v = momentum * v + g
        f_new = f - lr * v
        f_new = simple_normalize(f_new)

        b_new = auto_conv(f_new)
        obj = b_new.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 300:
            rng = np.random.default_rng(it * 104729 + 37)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            no_improve = 0
            temp = max(temp * 10, temp_start * 0.3)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE iter {it}, t={elapsed:.1f}s, obj={best_obj:.12f}, T={temp:.2e}")

    return best_f, best_obj


def fine_subgradient(f0, time_limit, top_k=100, temperature=1e-6,
                     lr=0.002, momentum=0.88, verbose=False):
    """Fine subgradient on max with top-k peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = auto_conv(f)
    best_obj = conv.max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = auto_conv(f)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for pi in range(len(peaks)):
            if w[pi] < 1e-14:
                continue
            k = int(peaks[pi])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[pi] * f[k - idx]

        v = momentum * v + (1.0 - momentum) * sg
        f_new = f - lr * v
        f_new = simple_normalize(f_new)

        conv_new = auto_conv(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        if no_improve > 200:
            rng = np.random.default_rng(it * 6151 + 7)
            noise = rng.standard_normal(n) * 0.001
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.12f}")

    return best_f, best_obj


def local_search(f0, time_limit, verbose=False):
    """Smart local search: targeted pair exchanges to reduce peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    b = auto_conv(f)
    best_obj = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b = auto_conv(f)
        max_b = b.max()

        if max_b < best_obj - 1e-14:
            best_obj = max_b
            best_f = f.copy()

        # Find near-peak lags
        threshold = best_obj * 0.995
        peak_lags = np.where(b >= threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([int(np.argmax(b))])
        # Limit to top few
        if len(peak_lags) > 10:
            peak_lags = peak_lags[np.argsort(b[peak_lags])[-10:]]

        improved = False

        for k in peak_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end < i_start:
                continue

            idx = np.arange(i_start, i_end + 1)
            contribs = f[idx] * f[k - idx]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Try reducing the top contributors
            top_idx = idx[np.argsort(contribs)[-5:]]

            for i_try in top_idx:
                if time.monotonic() - t0 > time_limit:
                    break
                j_try = k - i_try

                if i_try == j_try:
                    # Single coordinate: reduce f[i_try], distribute to others
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = f[i_try] * frac
                        if amt < 1e-14:
                            continue
                        f[i_try] -= amt
                        mask = np.ones(n, dtype=bool)
                        mask[i_try] = False
                        tot_others = f[mask].sum()
                        if tot_others < 1e-14:
                            f[i_try] += amt
                            continue
                        f[mask] *= (tot_others + amt) / tot_others
                        f = np.maximum(f, 0.0)
                        f /= f.sum()
                        new_obj = auto_conv(f).max()
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_f = f.copy()
                            improved = True
                        else:
                            f = best_f.copy()
                            break
                else:
                    # Pair: reduce both f[i_try] and f[j_try]
                    for frac in [0.005, 0.01, 0.02, 0.04]:
                        amt_i = f[i_try] * frac
                        amt_j = f[j_try] * frac
                        if amt_i < 1e-14 or amt_j < 1e-14:
                            continue
                        f[i_try] -= amt_i
                        f[j_try] -= amt_j
                        mask = np.ones(n, dtype=bool)
                        mask[i_try] = False
                        mask[j_try] = False
                        tot_others = f[mask].sum()
                        if tot_others < 1e-14:
                            f[i_try] += amt_i
                            f[j_try] += amt_j
                            continue
                        f[mask] *= (tot_others + amt_i + amt_j) / tot_others
                        f = np.maximum(f, 0.0)
                        f /= f.sum()
                        new_obj = auto_conv(f).max()
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_f = f.copy()
                            improved = True
                        else:
                            f = best_f.copy()
                            break

            # Also try random pair exchanges
            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                i_r = rng.integers(0, n)
                j_r = rng.integers(0, n)
                if i_r == j_r or f[i_r] < 1e-10:
                    continue
                for frac in [0.01, 0.03, 0.1]:
                    amt = f[i_r] * frac
                    if amt < 1e-14:
                        break
                    old_i = f[i_r]
                    old_j = f[j_r]
                    f[i_r] -= amt
                    f[j_r] += amt
                    new_obj = auto_conv(f).max()
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[i_r] = old_i
                        f[j_r] = old_j
                        break

        if not improved:
            # Try a slight perturbation before giving up
            if time.monotonic() - t0 < time_limit - 10:
                f = best_f.copy()
                noise = rng.standard_normal(n) * 0.001
                f = project_simplex(f + noise)
                new_obj = auto_conv(f).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_f = f.copy()
                else:
                    break

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent: try adjusting each coordinate individually."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = auto_conv(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        b = auto_conv(f)
        max_b = b.max()

        if max_b < best_obj - 1e-14:
            best_obj = max_b
            best_f = f.copy()

        # Focus on coordinates near the peak
        peak_k = int(np.argmax(b))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        focus_range = range(i_start, i_end + 1)

        # Also add a few random coordinates
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
        extra = list(rng.integers(0, n, min(10, n)))

        coords = list(focus_range) + extra
        coords = list(set(coords))
        # Sort by contribution to peak (descending)
        coords.sort(key=lambda j: f[j] * f[peak_k - j] if 0 <= peak_k - j < n else 0, reverse=True)

        for j in coords:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue

            old_val = f[j]
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = auto_conv(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try global coordinate changes
            any_improved = False
            for _ in range(5):
                if time.monotonic() - t0 > time_limit:
                    break
                j = rng.integers(0, n)
                if f[j] < 1e-10:
                    continue
                old_val = f[j]
                for mult in [0.3, 0.6, 0.9, 1.1, 1.4, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.99:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = np.maximum(f_trial, 0.0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial /= st
                    val = auto_conv(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        any_improved = True
                        improved = True
                        break

            if not any_improved and not improved:
                break

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: L2 gradient descent (encourage flat autocorrelation)
    Phase 2: LSE gradient descent with temperature annealing
    Phase 3: Fine subgradient on max
    Phase 4: Local search (pair exchange + coordinate descent)
    Phase 5: Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform n=768")

    n_parent = len(parent)
    print(f"\n=== Phase 1: L2 Gradient Descent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    t1 = max(10.0, min(rem, 300.0))
    print(f"  Allocating {t1:.0f}s")
    a1, _ = l2_optimize(parent, time_limit=t1, lr_start=0.05, lr_end=0.001, momentum=0.9)
    try_seq(a1, "l2")

    # Phase 2: LSE gradient descent from L2 result
    print(f"\n=== Phase 2: LSE Gradient Descent (best={best_score:.6f}) ===")
    a_start = best_seq if best_seq is not None else parent
    n_start = len(a_start)
    rem = deadline - time.monotonic() - 400
    t2 = max(10.0, min(rem, 400.0))
    print(f"  n={n_start}, allocating {t2:.0f}s")

    # Run LSE with different temperature starts
    configs = [
        {"lr_s": 0.15, "lr_e": 0.003, "mom": 0.92},
        {"lr_s": 0.08, "lr_e": 0.002, "mom": 0.88},
    ]
    for ci, cfg in enumerate(configs):
        rem = deadline - time.monotonic() - 200
        if rem < 15:
            break
        t2c = max(10.0, min(rem * 0.5, 200.0))
        init = a_start if ci == 0 else simple_normalize(a_start + rng.standard_normal(n_start) * 0.005)
        a2, _ = lse_optimize(init, time_limit=t2c,
                             temp_start=None, temp_end=1e-7,
                             lr_start=cfg["lr_s"], lr_end=cfg["lr_e"],
                             momentum=cfg["mom"])
        try_seq(a2, f"lse_{ci}")

    # Phase 3: Fine subgradient
    print(f"\n=== Phase 3: Fine Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 15:
            t3 = max(10.0, min(rem * 0.6, 200.0))
            print(f"  n={n_best}, allocating {t3:.0f}s")
            a3, _ = fine_subgradient(best_seq, time_limit=t3,
                                     top_k=max(50, n_best // 4),
                                     temperature=1e-6, lr=0.002, momentum=0.9)
            try_seq(a3, "sg_fine")

    # Phase 4: Local search
    print(f"\n=== Phase 4: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 10:
            t4a = max(5.0, min(rem * 0.5, 120.0))
            print(f"  Pair exchange: {t4a:.0f}s")
            a4, _ = local_search(best_seq, time_limit=t4a)
            try_seq(a4, "local")

            rem = deadline - time.monotonic() - 100
            if rem > 10:
                t4b = max(5.0, min(rem * 0.6, 100.0))
                print(f"  Coord descent: {t4b:.0f}s")
                a4b, _ = coordinate_descent_polish(best_seq, time_limit=t4b)
                try_seq(a4b, "cd")

    # Phase 5: Multi-length exploration
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [1024, 1536, 512]
        for tl in target_lengths:
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            f_interp = interpolate_seq(best_seq, tl)
            t5 = max(5.0, min(rem * 0.3, 60.0))
            print(f"  n={tl}, t={t5:.0f}s")
            # Quick LSE on interpolated
            a5, _ = lse_optimize(f_interp, time_limit=t5,
                                 temp_start=None, temp_end=1e-7,
                                 lr_start=0.08, lr_end=0.003,
                                 momentum=0.9)
            try_seq(a5, f"len_{tl}")

            # Quick local search
            rem = deadline - time.monotonic() - 60
            if rem > 10 and best_seq is not None and len(best_seq) == tl:
                a5b, _ = local_search(best_seq, time_limit=min(rem * 0.5, 30))
                try_seq(a5b, f"local_{tl}")

    # Phase 6: Final polish on absolute best
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            # Very fine LSE
            t6 = min(rem * 0.5, 30)
            a6, _ = lse_optimize(best_seq, time_limit=t6,
                                 temp_start=None, temp_end=1e-9,
                                 lr_start=0.01, lr_end=0.0005,
                                 momentum=0.95)
            try_seq(a6, "final_lse")

            rem = deadline - time.monotonic() - 5
            if rem > 3:
                a6b, _ = local_search(best_seq, time_limit=min(rem, 15))
                try_seq(a6b, "final_local")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
