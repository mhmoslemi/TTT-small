The theoretical minimum for the peak of the self-convolution of a non-negative sequence with sum 1 is... I believe it's related to the "autocorrelation problem" in signal processing. For continuous non-negative functions with integral 1, the minimum peak of the autocorrelation is 1 (achieved by a constant on a set of measure 1, but that's the uniform case giving peak 1/n for the discrete case).

Actually, I think for the discrete case, the minimum is achieved by sequences that are "as close to flat as possible" in the autocorrelation sense. This is a hard combinatorial/optimization problem.

Let me just focus on practical improvements. The previous program got 1.504964. Let me try to improve upon it.

Here's my plan:
1. Use the analytical subgradient (shifted a_rev) for fast iterations
2. Use a "top-k peaks" weighted subgradient for focus
3. Add a "peak shaving" local search step
4. Try multiple lengths: 832 (parent), 800, 864, 896, 960
5. Use aggressive restarts when stuck
6. Allocate time: 40% to parent length, 30% to other lengths, 30% to polishing

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using analytical subgradient descent
with peak-shaving coordinate updates and multi-length exploration.
Key innovation: Uses the closed-form gradient of the self-convolution peak
(∇c[k] = 2*shifted reversed a) for fast O(n log n) iterations via FFT.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """
    Compute weighted subgradient of max c[k].
    ∇c[k] = 2 * a_k_shift where (a_k_shift)[j] = a[k-j] if valid.
    This equals 2 * (reversed_a shifted by k).
    """
    n = len(a)
    a_rev = a[::-1]
    
    # Find top peaks
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Softmax weights
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Weighted sum of shifted reversed_a
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        # (∇c[k])[j] = 2*a[k-j] for 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
        # a[k-j] = a_rev[n-1-(k-j)] = a_rev[n-1-k+j]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        # a[k - j] for j in idx
        kg = k - idx  # = k - js, k - js - 1, ..., k - je
        # a[kg] where kg is in [0, n-1]
        sg[idx] += 2.0 * w[i] * a[kg]
    
    return sg


def subgradient_optimize(a0, time_limit, lr=0.01, momentum=0.9, top_k=20, temp=0.001,
                         verbose=False, restart_every=2000):
    """
    Fast subgradient descent with analytical gradient, momentum, and adaptive LR.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # momentum
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Get subgradient
        sg = compute_subgradient(a, conv, top_k, cur_temp)
        
        # Update with momentum
        v = momentum * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}, t={elapsed:.0f}s")
        else:
            no_improve += 1
            a = a_new
            conv = conv_new
        
        # Adaptive LR
        cur_lr *= 0.9999
        if no_improve > 50:
            cur_lr = max(0.0001, cur_lr * 0.9)
        
        # Temperature annealing
        cur_temp = max(0.00001, temp * (1 + 0.001 * it))
        
        # Restart if stuck
        if no_improve > restart_every:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            scale = 0.001 + 0.002 * min(no_improve // restart_every, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            conv = fast_convolve(a)
            cur_lr = lr * 0.5
            no_improve = 0
        
        # Periodic peak shaving
        if it % 500 == 0:
            a = peak_shave_step(a, conv, iters=50)
            conv = fast_convolve(a)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a.copy()
    
    return best_a, best_obj


def peak_shave_step(a, conv, iters=100):
    """
    Local peak shaving: find the peak, and for each contributing pair,
    transfer mass from the larger to the smaller element to reduce the peak.
    """
    n = len(a)
    a = a.copy()
    
    for _ in range(iters):
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        
        best_improvement = 0
        best_j = -1
        best_delta = 0
        
        for j in range(js, je + 1):
            k_j = peak_k - j
            if k_j < 0 or k_j >= n:
                continue
            # Reducing a[j] and increasing a[k_j] (or vice versa)
            # The change in c[peak_k] is approximately:
            # -a[j]*a[k_j] + (a[j]-d)*(a[k_j]+d) = a[j]*a[k_j] - 2*d*a[j] + d*a[k_j] + d^2
            # = a[j]*a[k_j] + d*(a[k_j] - 2*a[j]) + d^2
            # To decrease: need a[k_j] - 2*a[j] < 0, i.e., a[j] > a[k_j]/2
            # Or the other direction: a[j]*a[k_j] + d*(a[j] - 2*a[k_j]) + d^2
            # To decrease: need a[j] - 2*a[k_j] < 0, i.e., a[k_j] > a[j]/2
            
            # Try moving mass from j to k_j
            d = min(a[j] * 0.1, 0.001)
            if d < 1e-12:
                continue
            if a[j] > a[k_j]:
                # Move from j to k_j
                new_a = a.copy()
                new_a[j] -= d
                new_a[k_j] += d
                new_conv = fast_convolve(new_a)
                new_peak = new_conv.max()
                if new_peak < peak_val - 1e-12:
                    if peak_val - new_peak > best_improvement:
                        best_improvement = peak_val - new_peak
                        best_j = j
                        best_delta = d
            else:
                # Move from k_j to j
                d2 = min(a[k_j] * 0.1, 0.001)
                if d2 < 1e-12:
                    continue
                new_a = a.copy()
                new_a[k_j] -= d2
                new_a[j] += d2
                new_conv = fast_convolve(new_a)
                new_peak = new_conv.max()
                if new_peak < peak_val - 1e-12:
                    if peak_val - new_peak > best_improvement:
                        best_improvement = peak_val - new_peak
                        best_j = j
                        best_delta = -d2
        
        if best_j < 0:
            break
        
        if best_delta > 0:
            a[best_j] -= best_delta
            a[peak_k - best_j] += best_delta
        else:
            a[best_j] += (-best_delta)
            a[peak_k - best_j] -= (-best_delta)
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
    
    return a


def peak_shave_fast(a, conv, iters=200):
    """
    Faster peak shaving using the analytical gradient.
    At the peak k*, the gradient direction to reduce c[k*] is to decrease
    the product a[i]*a[k*-i]. The optimal direction is to reduce the
    larger element in each pair.
    """
    n = len(a)
    a = a.copy()
    best_obj = conv.max()
    best_a = a.copy()
    
    for _ in range(iters):
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        # Compute the gradient of c[peak_k]
        # (∇c[peak_k])[j] = 2*a[peak_k - j] for valid j
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[peak_k - idx]
        
        # Project the negative gradient onto the tangent space of the simplex
        # (i.e., make it sum to 0)
        grad_sum = grad.sum()
        grad_tangent = grad - grad_sum / n
        
        # Take a step along the projected negative gradient
        step = 0.01
        a_new = a - step * grad_tangent
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
        else:
            # Try smaller step
            step *= 0.5
            a_new = a - step * grad_tangent
            a_new = project_simplex(a_new)
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < peak_val - 1e-14:
                a = a_new
                conv = conv_new
            else:
                break
    
    return best_a


def interchangeable_cd(a, time_limit, verbose=False):
    """
    Coordinate descent that tries to equalize the convolution values.
    For each coordinate, try to find the value that minimizes the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    pass_cnt = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_cnt += 1
        improved = False
        
        # Find top 3 peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            if je < js:
                continue
                
            # For each coordinate in the peak's support, try adjusting
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                
                old_a = a[j]
                if old_a < 1e-10:
                    continue
                
                k_j = peak_k - j
                partner = a[k_j]
                
                # Try to equalize a[j] and a[k_j] (reduces the product)
                if abs(old_a - partner) > 1e-6:
                    # Move mass from larger to smaller
                    d = (old_a - partner) * 0.3
                    a_trial = a.copy()
                    a_trial[j] = old_a - d
                    a_trial[k_j] = partner + d
                    
                    if a_trial[j] >= 0 and a_trial[k_j] >= 0:
                        conv_trial = fast_convolve(a_trial)
                        if conv_trial.max() < best_obj - 1e-13:
                            a = a_trial
                            conv = conv_trial
                            best_obj = conv.max()
                            best_a = a.copy()
                            improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Try random pairs
                rng = np.random.default_rng(pass_cnt * 7919)
                for _ in range(min(50, n // 5)):
                    j = int(rng.integers(0, n))
                    # Find an active peak
                    pk = int(np.argmax(conv))
                    k_j = pk - j
                    if 0 <= k_j < n and k_j != j:
                        d = (a[j] - a[k_j]) * 0.2
                        if abs(d) > 1e-8:
                            a_trial = a.copy()
                            a_trial[j] -= d
                            a_trial[k_j] += d
                            if a_trial[j] >= 0 and a_trial[k_j] >= 0:
                                conv_trial = fast_convolve(a_trial)
                                if conv_trial.max() < best_obj - 1e-13:
                                    a = a_trial
                                    conv = conv_trial
                                    best_obj = conv.max()
                                    best_a = a.copy()
                                    improved = True
                if not improved:
                    break
        else:
            no_improve = 0
    
    return best_a, best_obj


def spectral_optimize(a0, time_limit, verbose=False):
    """
    Optimize in the frequency domain. The idea is:
    - The convolution peak is related to the time-domain correlation of |A(ω)|^2
    - We want |A(ω)|^2 to be as "flat" as possible (in a correlation sense)
    - We parameterize a via its phases in frequency domain and optimize those
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_lr = 0.001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        
        # The peak of conv is the max of the inverse FFT of |A|^2
        # Gradient w.r.t. a of the peak: use the analytical formula
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        
        # Subgradient: 2 * shifted_a_rev
        a_rev = a[::-1]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        sg = np.zeros(n)
        idx = np.arange(js, je + 1)
        sg[idx] = 2.0 * a[peak_k - idx]
        
        # Project gradient onto tangent space
        sg = sg - sg.mean()
        
        # Update
        a_new = project_simplex(a - cur_lr * sg)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            cur_lr *= 0.95
        
        if verbose and it % 1000 == 0:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def multi_start_search(a_base, time_limit, n_starts=5, verbose=True):
    """
    Multiple random perturbations followed by optimization.
    """
    n = len(a_base)
    best_a = a_base.copy()
    best_obj = fast_convolve(a_base).max()
    rng = np.random.default_rng(12345)
    
    t0 = time.monotonic()
    time_per_start = time_limit / n_starts
    
    for s in range(n_starts):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        
        # Generate perturbation
        scale = 0.005 * (s + 1)
        a_pert = project_simplex(a_base + rng.standard_normal(n) * scale)
        
        # Run subgradient optimization
        t_alloc = min(time_per_start * (1 + 0.5 * (s == 0)), rem - 5)
        if t_alloc < 5:
            break
            
        a_opt, obj_opt = subgradient_optimize(
            a_pert, time_limit=t_alloc,
            lr=0.005 * (1 + 0.2 * s),
            momentum=0.9 + 0.01 * s,
            top_k=10 + 5 * s,
            temp=0.001 * (1 + 0.5 * s)
        )
        
        if obj_opt < best_obj - 1e-12:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"  Multi-start {s}: obj={best_obj:.10f}")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using analytical subgradient descent
with Barzibili-Borwein adaptive step size, peak-shaving local search,
multi-length exploration, and diverse restarts.
Key: uses closed-form gradient of self-convolution (∇c[k] = 2*shifted a)
for O(n log n) iterations via FFT-based convolution.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """
    Compute weighted subgradient of max c[k].
    (∇c[k])[j] = 2*a[k-j] for valid k-j in [0, n-1].
    """
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        kg = k - idx
        sg[idx] += 2.0 * w[i] * a[kg]
    return sg


def fast_subgradient(a0, time_limit, lr=0.01, momentum=0.9, top_k=20,

    Fast subgradient descent with analytical gradient, BB step size,

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    prev_a = a.copy()
    prev_grad = None
    conv = fast_convolve(a)
    best_obj = conv.max() + flatness_weight * conv.std()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient
        sg = compute_subgradient(a, conv, top_k, cur_temp)

        # Add flatness penalty (gradient of std is complex, use variance proxy)
        if flatness_weight > 0:
            mean_c = conv.mean()
            # Gradient of sum(c^2) / len(c) w.r.t. a: 2*mean_c gradient
            sg = sg + flatness_weight * 2.0 * (conv - mean_c).mean() * sg

        # Project gradient to tangent space (sum to 0)
        sg_tangent = sg - sg.mean()

        # BB step size
        if use_bb and prev_grad is not None and it > 1:
            s_vec = a - prev_a
            y_vec = sg_tangent - prev_grad
            sy = s_vec.dot(y_vec)

        peak_val = conv[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[peak_k - idx]

        # Project to tangent space
        grad_t = grad - grad.mean()

        # Line search: try several step sizes
        best_step = 0
        best_val = peak_val
        for step in [0.001, 0.003, 0.01, 0.03, 0.1]:
            a_trial = project_simplex(a - step * grad_t)
            c_trial = fast_convolve(a_trial)
            v = c_trial.max()
            if v < best_val - 1e-12:
                best_val = v
                best_step = step

        if best_step == 0:
            break
        a = project_simplex(a - best_step * grad_t)
        conv = fast_convolve(a)
        if conv.max() < best_obj:
            best_obj = conv.max()

    return a


def coordinate_polish(a, time_limit, verbose=False):
    """
    Coordinate descent: for the peak bin, try adjusting each contributing
    coordinate to reduce the peak. Uses the fact that reducing a[j] where
    a[peak_k - j] is large most effectively reduces c[peak_k].
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Top 3 peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            if je < js:
                continue

            peak_val = conv[peak_k]

            # Sort by contribution a[j]*a[peak_k-j], try biggest first
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1]

            for oi in order[:20]:
                j = int(idx[oi])
                k_j = peak_k - j
                if k_j < 0 or k_j >= n:
                    continue
                old_j = a[j]
                if old_j < 1e-10:
                    continue

                # Try reducing a[j] (mass goes to the rest proportionally)
                for frac in [0.5, 0.7, 0.85, 0.9]:
                    new_j = old_j * frac
                    denom = 1.0 - old_j
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_j) / denom
                    a_trial = a.copy()
                    a_trial[j] = new_j
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    a_trial = a_trial / a_trial.sum()

                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-13:
                        a = a_trial
                        conv = fast_convolve(a)
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_starts(base, n_starts, n):
    """Generate diverse starting points from base."""
    rng = np.random.default_rng(777)
    starts = [base.copy()]
    for i in range(n_starts - 1):
        scale = 0.003 * (i + 1)
        s = project_simplex(base + rng.standard_normal(n) * scale)
        starts.append(s)
    return starts

    2. Local peak shaving

    5. Multi-length search

        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            if label:
                print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
