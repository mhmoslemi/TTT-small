def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix A for the LP: A@g <= M, where (f*g)[k] <= M."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    best_f = f.copy()

    no_improve = 0
    it = 0
    m_exp = m_expand
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        A_cache = build_lp_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 20})
            if not (res.success and res.x is not None):
                m_exp = min(m_exp * 1.2, 10.0)
                if m_exp > 5.0:
                    m_exp = 1.0
                continue
            g = np.maximum(res.x, 0.0)
        except Exception:
            m_exp = min(m_exp * 1.2, 10.0)
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            break
        g /= gs
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = m_expand
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 8.0)
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.003 * min(1.0, (no_improve - 40) / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = m_expand
            no_improve = 0
    
    return best_f, best_obj


def lse_gradient_step(f, beta, max_active=300):
    """Compute the gradient of the LSE surrogate at point f."""
    n = len(f)
    conv = fast_convolve(f)
    conv_len = 2 * n - 1
    
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Use only top weights for efficiency
    if conv_len > max_active:
        top_idx = np.argpartition(w, -max_active)[-max_active:]
        top_idx = top_idx[w[top_idx] > 1e-10 * w.max()]
    else:
        top_idx = np.arange(conv_len)
    
    grad = np.zeros(n)
    for k in top_idx:
        wk = w[int(k)]
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            grad[js] += wk * f[k - js]
    
    return grad, conv


def lse_optimize(f0, time_limit, beta_start=50.0, beta_end=50000.0, 
                 max_active=300, lr_scale=1.0, momentum_init=0.9):
    """LSE gradient descent with momentum and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    v = np.zeros(n)
    momentum = momentum_init
    lr = lr_scale / (n * 2)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        grad, conv_cur = lse_gradient_step(f, beta, max_active=max_active)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            no_improve += 100
            break
        grad /= gn
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.94
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr = lr_scale / (n * 2)
                beta = min(beta * 1.5, beta_end)
                momentum = momentum_init
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.005
    
    return best_f, best_obj


def subgradient_peak(f0, time_limit, n_peaks=12, lr_init=None):
    """Subgradient method targeting the top peaks of the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    if lr_init is None:
        lr_init = 2.0 / n
    
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Select peaks with exponential weighting
        k_sel = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peaks]
        
        # Soft weighting emphasizing highest peaks
        scale = pv.max() - pv
        w = np.exp(-scale / (0.05 * pv.max() + 1e-15))
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k_sel)
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for i, k in enumerate(peaks):
            if w[i] < 1e-12:
                continue
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                grad[js] += w[i] * 2.0 * f[k - js]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        # Backtracking line search
        best_alpha = 0.0
        best_val = best_obj
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = lr * mult
            f_try = np.maximum(f - alpha * grad, 0.0)
            st = f_try.sum()
            if st < 1e-15:
                continue
            f_try /= st
            val = fast_convolve(f_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f = np.maximum(f - best_alpha * grad, 0.0)
            f /= f.sum()
            if best_val < best_obj - 1e-12:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 20:
            lr *= 0.85
        if no_improve > 150:
            f = best_f.copy()
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_f, best_obj


def local_peak_trim(f0, time_limit, n_trim=8, seed=42):
    """Local search: identify peak contributors and try targeted reductions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        js = np.arange(lo, hi + 1)
        products = f[js] * f[k_star - js]
        
        # Get top contributing pairs
        top_idx = np.argsort(products)[::-1][:n_trim]
        
        found = False
        for idx in top_idx:
            j = int(js[idx])
            partner = k_star - j
            
            # Try several random redistributions
            for _ in range(8):
                f_try = f.copy()
                frac = rng.uniform(0.02, 0.15)
                delta = f_try[j] * frac
                if delta < 1e-10:
                    continue
                f_try[j] -= delta
                
                # Redistribute to a random position far from j and partner
                candidates = np.arange(n)
                candidates = candidates[np.abs(candidates - j) > n * 0.1]
                candidates = candidates[np.abs(candidates - partner) > n * 0.1]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                target = candidates[rng.integers(0, len(candidates))]
                f_try[target] += delta
                
                f_try = np.maximum(f_try, 0.0)
                st = f_try.sum()
                if st < 1e-15:
                    continue
                f_try /= st
                
                val = fast_convolve(f_try).max()
                if val < best_obj - 1e-12:
                    f = f_try
                    best_obj = val
                    best_f = f.copy()
                    found = True
                    break
            if found:
                break
        
        if not found:
            stall += 1
            if stall > 50:
                # Add noise to escape
                noise_scale = min(0.005, 0.001 * stall / 50)
                f_try = project_simplex(f + rng.standard_normal(n) * noise_scale)
                val = fast_convolve(f_try).max()
                if val < best_obj:
                    f = f_try
                    best_obj = val
                    best_f = f.copy()
                    stall = 0
                elif stall > 200:
                    f = best_f.copy()
                    stall = 100
    
    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def generate_initializations(n, n_inits=15):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Parabolic (U-shaped)
    for alpha in [0.3, 0.5, 0.8, 1.5]:
        f = 1.0 + alpha * (2.0 * x - 1.0)**2
        f = np.maximum(f, 1e-6)
        f /= f.sum()
        inits.append(f.copy())
    
    # Cosine bumps
    for freq in [1, 2, 3]:
        for alpha in [0.3, 0.6]:
            f = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x)
            f = np.maximum(f, 1e-6)
            f /= f.sum()
            inits.append(f.copy())
    
    # Sinusoidal dip (inverse U)
    for alpha in [0.3, 0.5]:
        f = 1.0 - alpha * np.sin(np.pi * x)
        f = np.maximum(f, 1e-6)
        f /= f.sum()
        inits.append(f.copy())
    
    # Random
    rng = np.random.default_rng(99)
    while len(inits) < n_inits:
        f = rng.random(n)
        f = np.maximum(f, 1e-6)
        f /= f.sum()
        inits.append(f)
    
    return inits
