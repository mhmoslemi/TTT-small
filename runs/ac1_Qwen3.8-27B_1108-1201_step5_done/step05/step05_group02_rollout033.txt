3. The minimum possible value is related to the "autocorrelation" or "peak factor" of the sequence.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Softmax mirror descent (multiplicative weights) - primary new method
2. Nesterov-accelerated subgradient descent
3. Multi-length exploration via interpolation

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def compute_softmax_gradient(a, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a.
    Returns gradient vector of same length as a."""
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    # Softmax weights
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    exp_c = np.exp(c_scaled)
    w = exp_c / exp_c.sum()  # length 2n-1, sums to 1
    
    # grad[i] = 2 * sum_m a[m] * w[i+m] for m in [0,n), i+m in [0, 2n-2]
    # = 2 * cross_corr(a, w)[i]
    # cross_corr(a, w)[i] = sum_m a[m] * w[m+i]
    # Using FFT: ifft(conj(fft(a, N)) * fft(w, N))[:n]
    N = 1 << (n + (2*n-1) - 1 - 1).bit_length()  # >= n + (2n-1) - 1
    A_fft = np.fft.fft(a, N)
    W_fft = np.fft.fft(w, N)
    cross = np.fft.ifft(np.conj(A_fft) * W_fft).real[:n]
    
    return 2.0 * cross

def softmax_mirror_descent(f0, time_limit, T_init=0.01, T_final=0.001,
                           eta_init=0.1, verbose=False):
    """Softmax mirror descent (multiplicative weights) for minimizing max(conv)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    T = T_init
    eta = eta_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        
        # Anneal temperature
        T = T_init * (T_final / T_init) ** progress
        # Anneal step size
        eta = eta_init * 0.9995 ** it
        
        # Compute gradient
        grad = compute_softmax_gradient(f, T)
        
        # Mirror descent update: a_new[i] = a[i] * exp(-eta * grad[i])
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        f_new = f * np.exp(log_update)
        f_new = np.maximum(f_new, 1e-30)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s_new
        
        eval_obj = fast_convolve(f_new).max()
        
        if eval_obj < best_obj - 1e-14:
            best_obj = eval_obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    MDS iter {it}: obj={best_obj:.10f}, T={T:.6f}, eta={eta:.6f}")
        else:
            no_improve += 1
            f = f_new  # still update to explore
        
        if no_improve > 500:
            # Reset with perturbation
            rng = np.random.default_rng(it * 997 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0
            eta = eta_init * 0.5
    
    return best_f, best_obj

        diff = peak_vals - max_peak
        w = np.exp(diff / 0.01)

            if w[idx] < 1e-12:
                continue

        # This is complex. Let me use a simpler momentum approach.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Frank-Wolfe with coordinate direction (argmin gradient) - highly efficient O(n log n) per iter
2. Softmax mirror descent (multiplicative weights)  
3. Multi-length exploration via interpolation
4. Peak-targeting coordinate descent for final polish
Key insight: The FW direction for max(conv) is e_{argmin s[i]} where s is the subgradient,
making each iteration just a few FFTs.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum=1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def compute_cross_corr_f_w(f, w):
    """Compute cross-correlation: d[i] = sum_m f[m] * w[m+i] for i in [0, n).
    Uses FFT for O(n log n) computation."""
    n = len(f)
    len_w = len(w)
    N = 1 << (n + len_w - 2).bit_length()
    F_fft = np.fft.fft(f, N)
    W_fft = np.fft.fft(w, N)
    result = np.fft.ifft(np.conj(F_fft) * W_fft).real[:n]
    return result


def fw_coordinate_optimize(f0, time_limit, T_init=0.05, T_final=0.001, use_softmax=True, verbose=False):
    """Frank-Wolfe with coordinate direction. Each iter: find argmin of gradient, line search.
    If use_softmax=True, uses softmax-weighted gradient (smooth).
    If use_softmax=False, uses exact subgradient (only max peak)."""
    f = normalize_nonneg(f0)
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1), 1.0)
        
        # Adaptive temperature
        T = T_init * (T_final / T_init) ** progress
        
        conv = fast_convolve(f)
        
        if use_softmax:
            # Softmax weights
            c_scaled = conv / T
            c_scaled -= c_scaled.max()
            w = np.exp(c_scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(conv))
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum
        else:
            # Exact subgradient: only the max peak
            w = np.zeros(len(conv))
            max_idx = np.argmax(conv)
            w[max_idx] = 1.0
        
        # d[i] = sum_m f[m] * w[m+i] = cross_corr(f, w)[i]
        d = compute_cross_corr_f_w(f, w)
        
        # FW direction: e_j where j = argmin d
        j = int(np.argmin(d))
        
        # Line search: minimize f((1-t)*f + t*e_j)
        best_t = 0.0
        best_val = best_obj
        t_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        
        for t in t_candidates:
            f_trial = (1.0 - t) * f
            f_trial[j] += t
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_t = t
        
        # Fine line search around best_t
        if best_t > 0:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)

                val = fast_convolve(f_trial).max()
                if val < best_val:
                    best_val = val
                    best_t = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_t > 1e-8:
            f_new = (1.0 - best_t) * f
            f_new[j] += best_t
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s > 1e-15:
                f_new /= s
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f_new.copy()
                no_improve = 0
                f = f_new
            else:
                no_improve += 1
                f = f_new
        else:
            no_improve += 1
        
        if no_improve > 300:

            rng = np.random.default_rng(it * 7919 + 13)

def mirror_descent_optimize(f0, time_limit, T_init=0.05, T_final=0.001, eta_init=0.3, verbose=False):
    """Softmax mirror descent (multiplicative weights) for the smoothed problem."""
    f = normalize_nonneg(f0)
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1), 1.0)
        
        T = T_init * (T_final / T_init) ** progress
        eta = eta_init * (0.9999 ** it)
        
        conv = fast_convolve(f)
        c_scaled = conv / T
        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            continue
        w /= w_sum
        
        # Gradient: grad[i] = 2 * cross_corr(f, w)[i]
        d = compute_cross_corr_f_w(f, w)
        grad = 2.0 * d
        
        # Mirror descent update: f_new[i] = f[i] * exp(-eta * grad[i])
        log_update = -eta * grad
        log_update -= log_update.max()
        f_new = f * np.exp(np.clip(log_update, -50, 50))
        f_new = np.maximum(f_new, 1e-30)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new /= s
        
        obj = fast_convolve(f_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    MDS iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            f = f_new
        
        if no_improve > 500:
            rng = np.random.default_rng(it * 997 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0
    
    return best_f, best_obj

                best_j_obj = best_obj
                
                for mult in [0.3, 0.5, 0.7, 0.9, 1.1, 1.4, 1.8]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.9:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial /= ss
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val
                
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                # Random exploration
                rng = np.random.default_rng(passes * 104729)
                found = False
                for _ in range(min(30, n // 5)):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 1.5, 2.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.9:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial /= f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            found = True
                            no_improve = 0
                            break
                    if found:
                        break
                if not improved and no_improve > 15:
                    break
        else:
            no_improve = 0
        
        if verbose and passes % 50 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. FW-coordinate with softmax gradient (primary, very fast)
    2. Mirror descent (multiplicative weights)
    3. FW-coordinate with exact subgradient
    4. Multi-length exploration
    5. Coordinate descent polish
    6. Final verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")
    
    base_n = len(parent)
    
    # === Phase 1: FW-coordinate with softmax (main optimization) ===
    print(f"\n=== Phase 1: FW-CD softmax (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(450, rem * 0.28)
    if t1 > 30:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, obj = fw_coordinate_optimize(parent, time_limit=t1,
                                             T_init=0.1, T_final=0.001,
                                             use_softmax=True, verbose=True)
        try_seq(a_opt, "fw_softmax")
    
    # === Phase 2: Mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(250, rem * 0.18)
        if t2 > 30:
            print(f"  n={len(best_seq)}, t={t2:.0f}s")
            a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t2,
                                                  T_init=0.05, T_final=0.001,
                                                  eta_init=0.3, verbose=True)
            try_seq(a_opt, "mirror_descent")
    
    # === Phase 3: FW-coordinate with exact subgradient ===
    print(f"\n=== Phase 3: FW-CD exact (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(250, rem * 0.18)
        if t3 > 30:
            print(f"  n={len(best_seq)}, t={t3:.0f}s")
            a_opt, obj = fw_coordinate_optimize(best_seq, time_limit=t3,
                                                 T_init=0.01, T_final=0.0005,
                                                 use_softmax=False, verbose=True)
            try_seq(a_opt, "fw_exact")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = set()
    for delta in [-64, -32, 32, 64, 128, 256]:
        ln = len(base) + delta
        if 256 <= ln <= 4096:
            candidate_lengths.add(ln)
    candidate_lengths = sorted(candidate_lengths)
    
    for target_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.3, 150)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  FW-CD n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = fw_coordinate_optimize(f_interp, time_limit=t_alloc,
                                           T_init=0.05, T_final=0.001,
                                           use_softmax=True, verbose=False)
        try_seq(a_opt, f"fw_n{target_n}")
    
    # === Phase 5: Fine FW-coordinate with small T ===
    print(f"\n=== Phase 5: Fine FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.2, 200)
        if t5 > 25:
            print(f"  n={len(best_seq)}, t={t5:.0f}s")
            a_opt, _ = fw_coordinate_optimize(best_seq, time_limit=t5,
                                               T_init=0.01, T_final=0.0003,
                                               use_softmax=True, verbose=True)
            try_seq(a_opt, "fw_fine")
    
    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Peak CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t6 = min(rem * 0.25, 200)
        if t6 > 20:
            print(f"  n={len(best_seq)}, t={t6:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t6, verbose=True)
            try_seq(a_opt, "peak_cd")
    
    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = fw_coordinate_optimize(best_seq, time_limit=min(rem, 60),
                                               T_init=0.005, T_final=0.0003,
                                               use_softmax=True, verbose=False)
            try_seq(a_opt, "fw_polish")
        
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=min(rem, 50), verbose=False)
            try_seq(a_opt, "cd_polish")
        
        # Perturbations from best
        for p in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.0005 + 0.0003 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = fw_coordinate_optimize(f_pert, time_limit=min(rem, 25),
                                               T_init=0.005, T_final=0.0003,
                                               use_softmax=True, verbose=False)
            try_seq(a_opt, f"perturb_{p}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
