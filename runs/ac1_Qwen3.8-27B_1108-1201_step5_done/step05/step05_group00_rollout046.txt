And the RHS for the constraint "2*conv(a,a_t)[k] - conv(a_t,a_t)[k] <= t" is:
t + conv(a_t, a_t)[k]

So in standard LP form:
minimize t
subject to: 2*a^T A_k - b_k <= t for all k
          sum(a) = 1
          a >= 0

Rewriting: 2*a^T A_k - t <= b_k for all k

This is a standard LP. Let me use scipy's linprog.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using SCA (Successive Convex Approximation)
with LP solves, Frank-Wolfe with LP oracle, and stochastic local search.

Key innovations over previous approach:
1. SCA outer loop for fast global convergence
2. Asymmetric initializations exploiting the fact that optimal is asymmetric
3. Stochastic pairwise mass transfers for local refinement
4. Adaptive constraint set (only top-K peaks in SCA)
5. Multiple n exploration
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_sca_lp(a_t, top_k=None):
    """
    Solve SCA LP: minimize t
    subject to: 2*conv(a, a_t)[k] - conv(a_t, a_t)[k] <= t for all k in K
                sum(a) = 1
                a >= 0
    
    If top_k is given, only use the top_k largest conv(a_t, a_t) values as constraints.
    """
    n = len(a_t)
    conv_tt = fast_convolve(a_t)
    
    if top_k is not None and top_k < len(conv_tt):
        # Select top_k peaks
        peaks = np.argpartition(conv_tt, -top_k)[-top_k:]
        peaks = sorted(peaks)
    else:
        peaks = np.arange(len(conv_tt))
    
    n_constr = len(peaks)
    
    # Decision variables: [a_0, ..., a_{n-1}, t]
    # Objective: minimize t => c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: 2*conv(a, a_t)[k] - t <= conv(a_t, a_t)[k] for each k in peaks
    # => 2 * sum_j a[j]*a_t[k-j] - t <= conv_tt[k]
    # => [2*a_t[k-(n-1)], 2*a_t[k-(n-2)], ..., 2*a_t[k], ..., -1] @ [a_0,...,a_{n-1}, t] <= conv_tt[k]
    
    A_ub = np.zeros((n_constr, n + 1))
    b_ub = np.zeros(n_constr)
    
    for i, k in enumerate(peaks):
        # conv(a, a_t)[k] = sum_j a[j] * a_t[k-j]
        # For j in [max(0, k-n+1), min(n-1, k)]:
        #   coefficient of a[j] is a_t[k-j]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = 2.0 * a_t[k - j]
        A_ub[i, n] = -1.0  # coefficient of t
        b_ub[i] = conv_tt[k]
    
    # Equality constraint: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x[:n], 0.0)
    except Exception:
        pass
    return None

def solve_sca_lp_fast(a_t, top_k=50):
    """Faster SCA LP using only top_k constraints."""
    return solve_sca_lp(a_t, top_k=top_k)

def sca_optimize(a0, time_limit, top_k=50, verbose=False):
    """SCA optimization loop."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv_tt = fast_convolve(a)
        current_max = conv_tt.max()
        
        # Adaptive top_k: start small, grow over time
        if it < 10:
            tk = min(20, top_k)
        elif it < 50:
            tk = min(30, top_k)
        else:
            tk = top_k
        
        a_new = solve_sca_lp_fast(a, top_k=tk)
        if a_new is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Normalize
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        a_new = np.maximum(a_new, 0.0)
        
        new_obj = fast_convolve(a_new).max()
        
        # Damped update if new solution is bad
        if new_obj > current_max * 1.1:
            # Blend
            for alpha in [0.5, 0.3, 0.1]:
                a_blend = (1 - alpha) * a + alpha * a_new
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                obj_blend = fast_convolve(a_blend).max()
                if obj_blend < current_max:
                    a = a_blend
                    new_obj = obj_blend
                    break
            else:
                no_improve += 1
                if no_improve > 50:
                    break
                continue
        
        a = a_new
        obj = new_obj
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Random restart
            rng = np.random.default_rng(it * 12345 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
        
        if verbose and it % 20 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, no_improve={no_improve}")
    
    return best_a, best_obj

def fw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Solve LP: max sum(g) s.t. conv(a, g)[k] <= M*m_exp for all k, g >= 0
        # This is: max 1^T g s.t. A g <= M*m_exp * 1, g >= 0
        # where A[k,j] = a[k-j] (valid terms)
        
        n_constr = 2 * n - 1
        c = -np.ones(n)
        A_ub = np.zeros((n_constr, n))
        for j in range(n):
            A_ub[j:j+n, j] = a
        b_ub = np.full(n_constr, M * m_exp)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 15})
            if not res.success:
                g = None
        except Exception:
            g = None
        
        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
                no_improve += 5
            continue
        
        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 2
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_t = (1.0 - alpha) * a + alpha * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-15:
                continue
            a_t = a_t / s
            val = fast_convolve(a_t).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(1.0, no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            M = fast_convolve(a).max()
            m_exp = 1.0
            no_improve = 0
    
    return best_a, best_obj

def stochastic_local_search(a0, time_limit, verbose=False):
    """
    Stochastic local search using pairwise mass transfers.
    At each step, randomly pick two coordinates i, j and transfer a small
    amount of mass from one to the other. Accept if objective improves.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    temperature = 0.001
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Every 100 iterations, compute the convolution
        if it % 100 == 1 or it == 1:
            conv_a = fast_convolve(a)
        
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        
        # Transfer amount
        transfer = a[i] * rng.uniform(0.05, 0.3)
        if transfer > a[i]:
            transfer = a[i] * 0.5
        
        # Direction: from i to j or j to i (whichever seems better)
        # Try both directions
        for direction in [1, -1]:
            a_trial = a.copy()
            if direction == 1:
                a_trial[i] -= transfer
                a_trial[j] += transfer
            else:
                a_trial[j] -= transfer
                a_trial[i] += transfer
            
            if a_trial[i] < 0 or a_trial[j] < 0:
                continue
            
            # Quick check: only recompute affected convolution values
            # conv changes at positions affected by i and j
            # For efficiency, do full recompute every K steps
            if it % 50 == 0:
                obj_trial = fast_convolve(a_trial).max()
            else:
                # Approximate: just check the most affected peaks
                # (This is a heuristic; full check is done periodically)
                obj_trial = fast_convolve(a_trial).max()
            
            if obj_trial < best_obj - 1e-14:
                a = a_trial
                best_obj = obj_trial
                best_a = a.copy()
                no_improve = 0
                break
        
        else:
            no_improve += 1
        
        # Adapt temperature
        if no_improve > 500:
            # Reset with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * temperature)
            temperature *= 0.8
            no_improve = 0
        
        if no_improve > 2000:
            break
    
    return best_a, best_obj
