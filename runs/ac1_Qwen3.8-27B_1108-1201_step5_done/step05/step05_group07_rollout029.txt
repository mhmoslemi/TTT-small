Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth L-BFGS optimization
on softmax-parameterized sequences with multi-temperature scheduling,
diverse initializations, multi-length exploration, and exact local polishing.
Key innovation: uses a smooth softmax approximation of the max operator
with analytically correct gradients (O(n log n) via FFT), enabling fast
second-order convergence that subgradient methods cannot match.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (max(n, 1) - 1).bit_length()


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n + m - 1]


def _compute_conv(a):
    """Auto-convolution a * a (full mode)."""
    return _fft_convolve(a, a)


def _softmax(t):
    """Numerically stable softmax."""
    t = t - t.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(t)) / len(t)
    return e / s


def _smooth_obj_grad(t, temp):
    """
    Compute smoothed objective value and gradient.
    obj = 2n * soft_max(conv(softmax(t), softmax(t)), temp)
    Returns (obj, grad_t).
    """
    n = len(t)
    a = _softmax(t)
    conv = _compute_conv(a)

    # Soft-max weights
    max_conv = conv.max()
    s = (conv - max_conv) / temp
    e = np.exp(s)
    w = e / e.sum()

    # Smoothed objective
    soft_max_val = max_conv + temp * np.log(e.sum())
    obj = 2.0 * n * soft_max_val

    # Compute h[i] = sum_j a[j] * w[i+j] via FFT
    a_flip = a[::-1]
    conv_aw = _fft_convolve(a_flip, w)
    h = conv_aw[n - 1:2 * n - 1]

    # Gradient: grad_t[m] = 4n * a[m] * (h[m] - dot(h, a))
    dot_ha = np.dot(h, a)
    grad_t = 4.0 * n * a * (h - dot_ha)

    return obj, grad_t

def _interpolate_seq(f, new_n):
    """Linearly interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def _structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences (unit sum, non-negative)."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine bump
    for pw in [1.5, 2.0, 3.0]:
        c = (1.0 + np.cos(2 * np.pi * x)) ** pw
        c = np.maximum(c, 1e-10)
        inits.append(c / c.sum())

    # Gaussian
    for sigma in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Polynomials
    for p in [2, 3, 5]:
        poly = (x + 1e-10) ** p
        inits.append(poly / poly.sum())

    # Bimodal
    c1 = np.exp(-((x - 0.3) / 0.15) ** 2)
    c2 = np.exp(-((x - 0.7) / 0.15) ** 2)
    bi = c1 + c2
    inits.append(bi / bi.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    klen = max(3, n // 50)
    kernel = np.ones(klen) / klen
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    return inits[:count]

            # Another round after polish
            rem = deadline - time.monotonic() - 30
            if rem > 10 and best_seq is not None:
                conv_scale = _compute_conv(best_seq).max()
                temps_final = [conv_scale * f for f in [0.01, 0.005, 0.002, 0.001]]
                t_alloc = min(rem * 0.4, 30)
                a_opt, _ = _optimize_lbfgs(best_seq, time_limit=t_alloc, temps=temps_final, maxiter_per_temp=200)
                _try_seq(a_opt, "refine1")

    # === Phase 5: Final micro-optimization ===
    print(f"\n=== Phase 5: Final (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_fin = len(best_seq)
        conv_fin = _compute_conv(best_seq)
        scale_fin = conv_fin.max()

        # Very fine L-BFGS
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            temps_ult = [scale_fin * f for f in [0.005, 0.003, 0.001, 0.0005, 0.0002]]
            t_alloc = min(rem * 0.6, 40)
            print(f"  Ultra-fine L-BFGS t={t_alloc:.0f}s")
            a_opt, _ = _optimize_lbfgs(best_seq, time_limit=t_alloc, temps=temps_ult, maxiter_per_temp=300)
            _try_seq(a_opt, "ultra")

        # Final local polish
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            a_opt, _ = _local_polish(best_seq, time_limit=min(rem, 20))
            _try_seq(a_opt, "final_polish")

        # Try a few more Adam passes
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = _optimize_adam(best_seq, time_limit=min(rem, 15), lr=0.005)
            _try_seq(a_opt, "adam_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
