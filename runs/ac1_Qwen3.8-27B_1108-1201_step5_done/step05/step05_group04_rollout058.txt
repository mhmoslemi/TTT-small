import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_fft(f, w):
    """Compute grad[m] = 2 * sum_k w[k] * f[k-m] using FFT.
    w has length 2n-1, f has length n.
    Returns gradient of length n.
    """
    n = len(f)
    conv_len = 2 * n - 1
    # Cross-correlation: C[m] = sum_k w[k+m] * f[k] for m = 0, ..., n-1
    # Use FFT with padding
    L = 1 << (conv_len + n - 1).bit_length()  # ≥ 3n-2
    W = np.fft.fft(w, L)
    F = np.fft.fft(f, L)
    C = np.fft.ifft(W * np.conj(F)).real
    # C[m] = sum_{k=0}^{min(n-1, conv_len-1-m)} w[k+m] * f[k] for m = 0, ..., n-1
    return 2.0 * C[:n]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation (O(n log n) per step)
2. LSE smoothed objective with adaptive beta
3. Projected gradient descent with backtracking line search
4. Multi-scale exploration (different lengths)
5. Coordinate descent polishing
6. Diverse restarts and simulated annealing escapes

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(f, w):
    """grad[m] = 2 * sum_k w[k] * f[k-m], computed via FFT."""
    n = len(f)
    conv_len = 2 * n - 1
    L = 1 << (conv_len + n - 1).bit_length()
    W = np.fft.fft(w, L)
    F = np.fft.fft(f, L)
    C = np.fft.ifft(W * np.conj(F)).real
    return 2.0 * C[:n]


def lse_objective(f, beta):
    """Compute LSE(beta * conv) and the weights."""
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    obj = (bc_max + np.log(exp_bc.sum())) / beta
    w = exp_bc / exp_bc.sum()
    return obj, w, conv


def lse_gradient_descent(f0, time_limit, beta_start=10.0, beta_end=10000.0,
                         lr_init=None, max_iter=None):
    """LSE gradient descent with FFT-based gradients and backtracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr = 1.0 / n
    else:
        lr = lr_init

    beta = beta_start
    v = np.zeros(n)
    momentum = 0.9
    t0 = time.monotonic()
    best_obj = np.inf
    best_f = f.copy()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()
        if M < best_obj:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Compute gradient via FFT
        grad = compute_gradient_fft(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        step = v / np.linalg.norm(v) if np.linalg.norm(v) > 1e-15 else v

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = lr
        for _ in range(10):
            f_new = f + alpha * step
            f_new = project_simplex(f_new)
            conv_new = fast_convolve(f_new)
            val = conv_new.max()
            if val < best_val - 1e-12:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5
            if alpha < 1e-10:
                break

        if best_alpha > 0:
            f = f + best_alpha * step
            f = project_simplex(f)
            # Reset momentum on acceptance
            v = 0.5 * v
        else:
            v = 0.5 * v
            lr *= 0.9

        # Adaptive beta
        if no_improve > 20:
            beta = min(beta * 1.5, beta_end)
            no_improve = 0
            lr = max(lr * 0.8, 0.0001)

        if beta < beta_end and it % 50 == 0:
            beta *= 1.01

        # Restart if stuck
        if no_improve > 100:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            beta = beta_start
            lr = 1.0 / n
            no_improve = 0
            momentum = 0.9

    return best_f, best_obj


def subgradient_descent(f0, time_limit, n_peaks=5):
    """Subgradient method focusing on top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.5 / n
    no_improve = 0
    avg_grad = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks
        k_top = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        peaks = peaks[conv[peaks] > 0.9 * M]  # Only near-max peaks

        # Subgradient: average over top peaks
        grad = np.zeros(n)
        for k in peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += 2.0 * f[k - ms]
        grad /= len(peaks) if len(peaks) > 0 else 1

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        # Polyak step size or fixed
        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        val = conv_new.max()

        if val < best_obj - 1e-12:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.01, 2.0 / n)
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.95
            if no_improve > 200:
                f = best_f.copy()
                lr = 0.5 / n
                n_peaks = max(2, n_peaks // 2)
                no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_perturb=5):

            f[m] = 0.0

            if s_rest < 1e-15:
                f = base.copy()
                continue

            best_val_m = f[m]  # Keep current if nothing better
            # Current value
            trial = base.copy()
            trial[m] = base[m]
            # Try a few ratios
            for ratio in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                trial[m] = base[m] * ratio
                s_trial = trial.sum()
                if s_trial < 1e-15:
                    continue
                trial_norm = trial / s_trial
                # Quick check: only compute affected convolution entries
                # For speed, use full convolve if n is small enough
                conv_trial = fast_convolve(trial_norm)
                val = conv_trial.max()
                if val < best_local - 1e-12:
                    best_local = val
                    best_val_m = trial[m]
                    f = trial_norm.copy()

            if best_local < best_obj - 1e-12:
                best_obj = best_local
                best_f = f.copy()

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation for O(n log n) per iteration
2. LSE smoothed objective with adaptive beta sweep
3. Momentum-based projected gradient descent with backtracking
4. Subgradient polishing on top peaks
5. Fast coordinate descent for final micro-optimization
6. Multi-length exploration with diverse restarts
7. Simulated annealing-style escapes from local minima

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(f, w):
    """Compute grad[m] = 2 * sum_k w[k] * f[k-m] via FFT cross-correlation.
    f has length n, w has length 2n-1. Returns gradient of length n.
    """
    n = len(f)
    conv_len = 2 * n - 1
    L = 1 << (conv_len + n - 1).bit_length()
    W = np.fft.fft(w, L)
    F = np.fft.fft(f, L)
    C = np.fft.ifft(W * np.conj(F)).real
    return 2.0 * C[:n]


def lse_gradient_descent(f0, time_limit, beta_start=10.0, beta_end=50000.0,
                         lr_init=None, momentum_init=0.9):
    """LSE gradient descent with FFT-based gradients, momentum, and adaptive schedule."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    lr = lr_init if lr_init is not None else 1.0 / n
    beta = beta_start
    momentum = momentum_init
    v = np.zeros(n)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # FFT gradient
        grad = compute_gradient_fft(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break
        grad /= grad_norm

        # Momentum
        v = momentum * v - lr * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-16:
            continue
        direction = v / v_norm

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = min(lr * 2.0, 1.0)
        for _ in range(12):
            f_trial = f + alpha * direction
            f_trial = project_simplex(f_trial)
            conv_trial = fast_convolve(f_trial)
            val = conv_trial.max()
            if val < best_val - 1e-13:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5
            if alpha < 1e-12:
                break

        if best_alpha > 0:
            f = f + best_alpha * direction
            f = project_simplex(f)
            v = momentum * v * 0.5
        else:
            v = v * 0.3
            lr *= 0.92

        # Adaptive beta: increase as we progress
        if beta < beta_end:
            target_beta = beta_start * (beta_end / beta_start) ** progress
            if no_improve > 15:
                beta = min(beta * 1.8, beta_end)
            else:
                beta = min(beta * 1.02, target_beta)

        # Adaptive momentum
        if no_improve > 40:
            momentum = max(0.8, momentum - 0.01)

        # Restart if severely stuck
        if no_improve > 150:
            rng = np.random.default_rng((it * 7919 + 31) % 1000000)
            scale = 0.003 + 0.005 * min(1.0, no_improve / 300.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            beta = max(beta_start, beta * 0.3)
            lr = max(0.5 / n, lr * 0.5)
            momentum = momentum_init
            no_improve = 0

    return best_f, best_obj


def subgradient_descent(f0, time_limit, n_peaks_init=8):
    """Subgradient method focusing on the top few peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.3 / n
    no_improve = 0
    n_peaks = n_peaks_init
    v = np.zeros(n)
    momentum = 0.85

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Select top peaks
        k_sel = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        # Only use peaks close to the max
        threshold = 0.85 * M
        peaks = peaks[conv[peaks] >= threshold]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        # Subgradient (average over selected peaks)
        grad = np.zeros(n)
        for k in peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += 2.0 * f[k - ms]
        grad /= len(peaks)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad /= gnorm

        # Momentum + step
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        val = conv_new.max()

        if val < best_obj - 1e-13:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.005, 1.0 / n)
        else:
            no_improve += 1
            if no_improve > 20:
                lr *= 0.95
            if no_improve > 50:
                n_peaks = max(2, n_peaks - 1)
            if no_improve > 200:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 0.3 / n
                n_peaks = n_peaks_init
                momentum = 0.85
                no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent: try scaling each coordinate by various factors."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:

    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    ratios = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 0.3]

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        any_improve = False

        for m in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[m] < 1e-10:
                # Try setting a small positive value
                for val in [0.001/n, 0.01/n, 0.05/n]:
                    trial = f.copy()
                    trial[m] = val
                    cs = trial.sum()
                    if cs < 1e-15:
                        continue
                    trial /= cs
                    cv = fast_convolve(trial)
                    if cv.max() < best_obj - 1e-13:
                        best_obj = cv.max()
                        best_f = trial.copy()
                        f = trial
                        any_improve = True
                        break
                continue

            for ratio in ratios:
                trial = f.copy()
                trial[m] = f[m] * ratio
                cs = trial.sum()
                if cs < 1e-15:
                    continue
                trial /= cs
                cv = fast_convolve(trial)
                val = cv.max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = trial.copy()
                    f = trial
                    any_improve = True
                    break  # Move to next coordinate

        if not any_improve:
            # Try a few more random perturbations before giving up
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                m = rng.integers(n)
                trial = f.copy()
                trial[m] *= rng.uniform(0.3, 3.0)
                cs = trial.sum()
                if cs < 1e-15:
                    continue
                trial /= cs
                cv = fast_convolve(trial)
                if cv.max() < best_obj - 1e-13:
                    best_obj = cv.max()
                    best_f = trial.copy()
                    f = trial
                    any_improve = True
            if not any_improve:
                break

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def diverse_initialization(n, rng):
    """Create a diverse random initialization."""
    # Mix of uniform, sinusoidal, and random
    choice = rng.integers(4)
    if choice == 0:
        # Uniform + noise
        f = np.ones(n) / n + rng.standard_normal(n) * 0.05
    elif choice == 1:
        # Cosine wave + uniform
        x = np.arange(n)
        freq = rng.uniform(1, 5)
        f = np.ones(n) / n + 0.1 * np.cos(2 * np.pi * freq * x / n)
    elif choice == 2:
        # Two-level sequence
        n1 = rng.integers(n // 3, 2 * n // 3)
        f = np.zeros(n)
        f[:n1] = 1.0 / n1
        f[n1:] = 0.0
        # Shuffle positions
        perm = rng.permutation(n)
        f = f[perm]
    else:
        # Exponential decay + noise
        x = np.arange(n)
        f = np.exp(-x / n * rng.uniform(1, 5)) + 0.1
        f = np.concatenate([f, f[::-1]])[:n]

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    else:
        f = np.ones(n) / n
    return f

        f_opt, _ = subgradient_descent(current, time_limit=t_alloc, n_peaks_init=6)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Coordinate descent ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t4_end = deadline - 80.0
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(100.0, t4_end - time.monotonic())
    if t_alloc > 15:
        print(f"  n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_descent(current, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Multi-length exploration with diverse restarts ===
    print("\n=== Phase 5: Multi-length Exploration ===")
    t5_end = deadline - 40.0
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])

        # Try nearby lengths
        length_tries = sorted(set([
            n_best, n_best + 64, n_best - 64,
            n_best + 128, n_best - 128,
            n_best + 32, n_best - 32,
            max(128, n_best - 256), n_best + 256
        ]))
        length_tries = [lt for lt in length_tries if 96 <= lt <= 1500]

        # Also try some diverse lengths
        for d in [384, 448, 512, 640, 768, 896, 1024, 1280]:
            if d not in length_tries and 96 <= d <= 1500:
                length_tries.append(d)
        length_tries = sorted(set(length_tries))

        tried = 0
        for nt in length_tries:
            t_remaining = t5_end - time.monotonic()
            if t_remaining < 25 or tried >= 6:
                break
            try:
                t_per = min(60.0, t_remaining * 0.25)
            except Exception:
                t_per = min(60.0, t_remaining * 0.25)
            if t_per < 12:
                break

            if nt == n_best:
                f_init = best_tracker[1].copy()
            else:
                f_init = interpolate_sequence(best_tracker[1], nt)

            # Diverse restart: add noise
            if nt != n_best and rng.random() < 0.5:
                f_init = f_init + rng.standard_normal(nt) * 0.01
                f_init = project_simplex(f_init)

            print(f"  LSE-GD n={nt}, t={t_per:.0f}s")
            f_opt, _ = lse_gradient_descent(f_init, time_limit=t_per,
                                             beta_start=20.0, beta_end=20000.0,
                                             lr_init=1.0 / nt, momentum_init=0.9)
            try_sequence(f_opt, best_tracker)

            # Quick coord descent follow-up
            t_cd = min(20.0, t5_end - time.monotonic() - 10)
            if t_cd > 8 and best_tracker[1] is not None:
                f_opt2, _ = coordinate_descent(f_opt, time_limit=t_cd)
                try_sequence(f_opt2, best_tracker)

            tried += 1

    # === Phase 6: Final micro-polish ===
    print("\n=== Phase 6: Final Micro-Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic()

    if t_remaining > 30:
        t_cd = min(50.0, t_remaining * 0.5)
        if t_cd > 10:
            print(f"  CoordDesc n={len(current)}, t={t_cd:.0f}s")
            f_opt, _ = coordinate_descent(current, time_limit=t_cd)
            try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_remaining = deadline - time.monotonic()
    if t_remaining > 20:
        t_hp = min(30.0, t_remaining * 0.5)
        if t_hp > 8:
            print(f"  HighBeta n={len(current)}, t={t_hp:.0f}s")
            f_opt, _ = lse_high_beta_polish(current, time_limit=t_hp)
            try_sequence(f_opt, best_tracker)

    # One last coordinate descent squeeze
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_remaining = deadline - time.monotonic()
    if t_remaining > 10:
        t_cd = min(20.0, t_remaining * 0.6)
        if t_cd > 5:
            f_opt, _ = coordinate_descent(current, time_limit=t_cd)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
