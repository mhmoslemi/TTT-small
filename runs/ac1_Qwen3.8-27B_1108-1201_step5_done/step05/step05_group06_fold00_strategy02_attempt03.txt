def embed_sequence(seq, depth=3):
    """Embeds a base sequence into a higher-dimensional representation."""
    base = seq.copy()
    total_len = len(seq) * (2**depth)
    embedded = []
    for i in range(len(seq)):
        cur = base[i].copy()
        sub_seq = cur * (np.where(i < total_len/2, 0.7, 1))
        embedded.append(sub_seq)
    return np.array(embedded)

def multi_scale_projection(seq, alpha):
    """Projects coefficients to control sparsity while maintaining validity."""
    # First ensure coefficients are bounded
    bounded = np.clip(seq, 0, None)
    norm_factor = bounded.sum()
    if norm_factor < 0.01:
        # Reinforce minimal sum requirement
        reinforcement = np.ones_like(bounded) * (0.01 / len(bounded))
        bounded = np.maximum(bounded, reinforcement)
    
    # Apply directional scaling factor
    scaled = bounded * alpha
    # Ensure we exceed the minimal sum threshold if it was too small
    if scaled.sum() < 0.01:
        scaled += (0.01 - scaled.sum()) / len(scaled)
        
    return scaled.tolist()
