def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, top_k=20, temperature=1.0):
    """Compute smoothed subgradient of max(conv) with respect to a."""
    n = len(a)
    m = len(conv)
    k_actual = min(top_k, m)
    idx = np.argpartition(conv, -k_actual)[-k_actual:]
    vals = conv[idx]
    scaled = vals / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.zeros(len(idx))
        w[np.argmax(vals)] = 1.0
    else:
        w = w / wsum

    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-14:
            continue
        k = int(idx[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        jj = np.arange(js, je + 1)
        g[jj] += 2.0 * w[i] * a[k - jj]
    return g

def coordinate_descent(a0, time_limit, verbose=False):

    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()

        elapsed = time.monotonic() - t0

            break

        passed = False
        order = rng.permutation(n)
        # Every few passes, use natural order
        if pass_count % 5 == 0:
            order = np.arange(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[i] < 1e-12:
                continue

            # Try transferring mass from i to other coordinates
            # The effect of changing a[i] by delta and distributing -delta/(n-1) to others:
            # conv_new[k] = conv[k] + delta * a[k-i] - delta/(n-1) * (overlap_k - contribution_of_i)
            # Simplified: just try a few delta values

            ai = f[i]
            # Try reducing a[i] and spreading to others, or increasing
            candidates = []
            if ai > 0.01 * f.mean():
                candidates.extend([0.5, 0.7, 0.85, 0.95])  # multiply a[i] by these

                candidates.extend([1.5, 2.0, 3.0])

            best_delta = 0.0
            best_val = best_obj

            for mult in candidates:
                new_ai = ai * mult
                delta = new_ai - ai
                if abs(delta) < 1e-15:
                    continue
                # New convolution: conv[k] + delta * a[k-i] - delta/(n-1) * (sum_valid - a[k-i] if valid)
                # Approximation: conv[k] + delta * (a[k-i] - mean_a)
                # More precise: for each k, the valid j's contribute
                # Let's just compute directly for speed
                f_trial = f.copy()
                f_trial[i] = new_ai
                # Distribute the change to maintain sum
                remaining = n - 1
                if remaining <= 0:

                f_trial = np.maximum(f_trial, 0)

                if ts > 1e-15:

                conv = fast_convolve(f)

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def interpolate_sequence(f, new_n):
    """Linear interpolation to new length, projected to simplex."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, seed=0):
    """Initialize using a spectral approach: random Fourier features."""
    rng = np.random.default_rng(seed)
    # Use sum of cosines with random frequencies
    x = np.arange(n)
    f = np.ones(n)
    for _ in range(min(10, n // 10)):
        freq = rng.uniform(0.1, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        f += 0.3 * np.cos(2 * np.pi * freq * x / n + phase)
    f = np.maximum(f, 0)
    f = f / f.sum()
    return f

        rem = deadline - time.monotonic() - 450
        if rem < 30:
            break
        if target_n == len(base):
            continue

        f_interp = interpolate_sequence(base, target_n)

        # Quick Nesterov Adam
        t_alloc = min(rem * 0.25, 80)
        if t_alloc > 10:
            print(f"  NAdam n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_adam(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"nadam_n{target_n}")

        # Coordinate descent
        rem = deadline - time.monotonic() - 400
        if rem > 15:
            t_cd = min(rem * 0.3, 60)
            if t_cd > 8:
                a_opt, _ = coordinate_descent(a_opt, time_limit=t_cd, verbose=False)
                try_seq(a_opt, f"coord_n{target_n}")

    # === Phase 5: LP-FW on best ===
    print(f"\n=== Phase 5: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t5 = min(rem, 150)
        if t5 > 10:
            print(f"  n={len(best_seq)}, t={t5:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "lpfw_p5")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 200
            if rem < 12:
                break
            t_alloc = min(rem / max(1, 4 - p), 60)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                rng = np.random.default_rng(200 + p)
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
            elif p == 2:
                f_init = spectral_init(n_best, seed=500 + p)
            else:
                f_init = 0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)

            if p % 2 == 0:
                a_opt, _ = nesterov_adam(f_init, time_limit=min(t_alloc, 50))
            else:
                a_opt, _ = coordinate_descent(f_init, time_limit=min(t_alloc, 50))
            try_seq(a_opt, f"diverse_{p}")

    # === Phase 7: Simulated annealing polish ===
    print(f"\n=== Phase 7: SA Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 10:
            t_sa = min(rem * 0.6, 60)
            print(f"  n={len(best_seq)}, t={t_sa:.0f}s")
            a_opt, _ = simulated_annealing_polish(best_seq, time_limit=t_sa, verbose=True)
            try_seq(a_opt, "sa_p7")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick Nesterov Adam
        rem = deadline - time.monotonic() - 40
        if rem > 8:
            a_opt, _ = nesterov_adam(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "nadam_final")

        # Coordinate descent
        rem = deadline - time.monotonic() - 25
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "coord_final")

        # Paired coordinate
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            a_opt, _ = paired_coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "paired_final")

        # SA
        rem = deadline - time.monotonic() - 8
        if rem > 4:
            a_opt, _ = simulated_annealing_polish(best_seq, time_limit=min(rem, 12))
            try_seq(a_opt, "sa_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
