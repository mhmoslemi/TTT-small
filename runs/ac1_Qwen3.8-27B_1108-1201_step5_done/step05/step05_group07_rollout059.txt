def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

    w = np.exp(scaled)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimization in log-parameterization space
2. Exact pair-transfer local search with global max evaluation
3. Multi-length exploration and interpolation
4. Simulated annealing for escape from local minima

import numpy as np
import time
import math

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_subgrad(a, conv, top_k, temperature):
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = np.zeros(n)
    for idx in range(k_actual):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += 2.0 * w[idx] * a[k - idx_arr]
    return sg

def adam_log_optimize(a0, time_limit, lr=0.01, top_k=50, temperature=0.001):
    """Adam optimization in log-parameterization space."""
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    # Log-parameterization
    phi = np.log(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        one_over_beta1_power = 1 - beta1 ** t
        one_over_beta2_power = 1 - beta2 ** t
        
        conv = fast_convolve(a)
        sg = softmax_subgrad(a, conv, top_k, temperature)
        
        # Gradient w.r.t. phi: grad_phi[i] = a[i] * (sg[i] - sum(a*sg))
        mean_sg = np.sum(a * sg)
        grad_phi = a * (sg - mean_sg)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_phi
        v = beta2 * v + (1 - beta2) * grad_phi ** 2
        m_hat = m / one_over_beta1_power
        v_hat = v / one_over_beta2_power
        
        phi -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to a
        phi -= phi.max()  # numerical stability
        a_new = np.exp(phi)
        a_new /= a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        a = a_new
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        if no_improve > 200:
            temperature *= 1.5
            if no_improve > 1000:
                break
        
    return best_a, best_obj

def pair_transfer_eval(conv_old, a, p, q, t, n):
    """Evaluate max of new conv after transferring t from p to q."""
    # B[k] = 2*(a[k-q] - a[k-p])
    # C[k] = delta_{k,2q} - 2*delta_{k,p+q} + delta_{k,2p}
    
    # Compute B vector
    # a[k-q] is valid when 0 <= k-q < n, i.e., q <= k < q+n
    # a[k-p] is valid when 0 <= k-p < n, i.e., p <= k < p+n
    
    # Use vectorized computation
    # B[k] = 2*(a[k-q] - a[k-p]) where out-of-range a values are 0
    k_range = np.arange(2*n - 1)
    
    # a[k-q]: valid for k in [q, q+n-1]
    a_kq = np.zeros(2*n - 1)
    a_kq[q:q+n] = a
    
    # a[k-p]: valid for k in [p, p+n-1]
    a_kp = np.zeros(2*n - 1)
    a_kp[p:p+n] = a
    
    B = 2 * (a_kq - a_kp)
    
    # C is sparse
    new_conv = conv_old + B * t
    # Add quadratic terms
    if 2*q < 2*n - 1:
        new_conv[2*q] += t * t
    if p+q < 2*n - 1:
        new_conv[p+q] -= 2 * t * t
    if 2*p < 2*n - 1:
        new_conv[2*p] += t * t
    
    return new_conv.max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimization in log-parameterization space with adaptive temperature
2. Batch pair-transfer local search with gradient-guided pair selection
3. L2 spectrum flattening warm-up in Fourier domain
4. Multi-length exploration and interpolation
5. Simulated annealing restarts for escape from local minima

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_subgrad(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv) w.r.t. a."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(k_actual)
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for idx in range(k_actual):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += 2.0 * w[idx] * a[k - idx_arr]
    return sg


def compute_B(a, p, q, n):
    """Compute B[k] = 2*(a[k-q] - a[k-p]) for transfer pair (p->q)."""
    k_range = 2 * n - 1
    B = np.zeros(k_range)
    B[q:q + n] += 2.0 * a
    B[p:p + n] -= 2.0 * a
    return B


def eval_transfer(conv_old, B, t, p, q, n):
    """Evaluate max of new conv after transferring t from p to q."""
    k_range = 2 * n - 1
    base_max = (conv_old + B * t).max()
    result = base_max
    if 2 * q < k_range:
        val = conv_old[2 * q] + B[2 * q] * t + t * t
        if val > result:
            result = val
    if p + q < k_range:
        val = conv_old[p + q] + B[p + q] * t - 2.0 * t * t
        if val > result:
            result = val
    if 2 * p < k_range:
        val = conv_old[2 * p] + B[2 * p] * t + t * t
        if val > result:
            result = val
    return result


def adam_log_optimize(a0, time_limit, lr=0.01, top_k=50, temperature=0.001,
                      beta1=0.9, beta2=0.999, verbose=False):
    """Adam optimization in log-parameterization space."""
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    phi = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temperature
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        one_over_b1 = 1 - beta1 ** t
        one_over_b2 = 1 - beta2 ** t
        conv = fast_convolve(a)
        sg = softmax_subgrad(a, conv, top_k, cur_temp)
        mean_sg = np.sum(a * sg)
        grad_phi = a * (sg - mean_sg)
        m = beta1 * m + (1 - beta1) * grad_phi
        v = beta2 * v + (1 - beta2) * grad_phi ** 2
        m_hat = m / one_over_b1
        v_hat = v / one_over_b2
        phi = phi - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        phi = phi - phi.max()
        a = np.exp(phi)
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-30:
            a = best_a.copy()
            phi = np.log(np.maximum(a, 1e-15))
            m = np.zeros(n)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, 0.001)
            continue
        a /= ss
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            cur_temp *= 2.0
            cur_lr = max(cur_lr * 0.7, 0.0005)
            if no_improve > 2000:
                break
        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}, obj={best_obj:.10f}, lr={cur_lr:.6f}, temp={cur_temp:.8f}")
    return best_a, best_obj


def l2_spectrum_flatten(a0, time_limit, lr=0.01, iters_per_check=100):
    """Minimize L2 norm of autocorrelation (flatten spectrum) as warm-up."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2
        # Gradient of sum(|A|^4) w.r.t. a_real:
        # d/d(a) sum(|A|^4) = 2 * IDFT(A * |A|^2) * 2 (factor from real/imag)
        # Actually: sum |A|^4 = sum (A_re^2 + A_im^2)^2
        # d/d A_re = 4 * A_re * (A_re^2 + A_im^2) = 4 * A_re * |A|^2
        # d/d A_im = 4 * A_im * |A|^2
        # So gradient w.r.t. A is 4 * A * |A|^2 (complex)
        # Gradient w.r.t. a: IDFT(4 * A * |A|^2)
        grad_A = 4.0 * A * power
        grad_a = np.fft.irfft(grad_A, fft_size)[:n]
        a = a - lr * grad_a
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-30:
            a = best_a.copy()
            continue
        a /= ss
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    inits.append(np.ones(n) / n)
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())
    center = n / 2.0
    sigma = n / 5.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    # Cosine raised to power
    cos_seq = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2
    cos_seq = np.maximum(cos_seq, 1e-10)
    inits.append(cos_seq / cos_seq.sum())
    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Adam in log-space, spectrum flattening,
    batch pair search, exact pair polishing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a.tolist()
        score = float(evaluate_sequence(a))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = np.array(a, dtype=np.float64)
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")

    best_seq = parent.copy()
    try_seq(parent, "parent")

    # === Phase 0: L2 spectrum flattening (warm-up) ===
    print("\n=== Phase 0: L2 Spectrum Flattening ===")
    rem = deadline - time.monotonic() - 1200
    if rem > 30:
        t_alloc = min(rem, 60)
        print(f"  t={t_alloc:.0f}s, n={len(parent)}")
        a_opt, _ = l2_spectrum_flatten(parent, time_limit=t_alloc, lr=0.05)
        try_seq(a_opt, "l2_flatten")

    # === Phase 1: Adam optimization on multiple initializations ===
    print(f"\n=== Phase 1: Adam Log-Optimization (best={best_score:.6f}) ===")
    candidates = [best_seq]
    n_parent = len(best_seq)
    for init in structured_inits(n_parent, rng):
        candidates.append(init)
    # Also try different lengths
    for n_try in [640, 1024]:
        for init in structured_inits(n_try, rng)[:2]:
            candidates.append(init)

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - 800
        if rem < 20:
            break
        n_cand = len(cand)
        t_alloc = min(rem * 0.15, 80)
        if t_alloc < 10:
            t_alloc = min(rem * 0.3, 60)
        print(f"  Cand {idx}: n={n_cand}, t={t_alloc:.0f}s")
        a_opt, _ = adam_log_optimize(cand, time_limit=t_alloc,
                                      lr=0.01, top_k=max(30, n_cand // 8),
                                      temperature=0.001,
                                      beta1=0.9, beta2=0.999)
        try_seq(a_opt, f"adam_{idx}")

    # === Phase 2: Deep Adam with different hyperparams ===
    print(f"\n=== Phase 2: Deep Adam (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {"lr": 0.005, "top_k": max(20, n_best // 6), "temp": 0.0005, "mom": 0.95},
            {"lr": 0.02, "top_k": max(40, n_best // 4), "temp": 0.002, "mom": 0.85},
            {"lr": 0.003, "top_k": n_best, "temp": 0.0001, "mom": 0.97},
        ]
        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 600
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 120)
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = np.maximum(best_seq + noise, 0)
                f_init /= f_init.sum()
            else:
                f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                f_init = np.maximum(f_init, 0)
                f_init /= f_init.sum()
            print(f"  Deep {p}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = adam_log_optimize(f_init, time_limit=t_alloc,
                                          lr=cfg["lr"], top_k=cfg["top_k"],
                                          temperature=cfg["temp"],
                                          beta1=cfg["mom"], beta2=0.999)
            try_seq(a_opt, f"deep_adam_{p}")

    # === Phase 3: Batch pair search ===
    print(f"\n=== Phase 3: Batch Pair Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  Batch pair search: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = batch_pair_search(best_seq, time_limit=t_alloc, top_k=25)
            try_seq(a_opt, "batch_pair")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [640, 1024, 1280, 1536, 2048]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 300
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            # Quick Adam
            t_adam = min(rem * 0.4, 60)
            if t_adam > 10:
                a_opt, _ = adam_log_optimize(f_interp, time_limit=t_adam,
                                              lr=0.01, top_k=max(30, target_n // 6),
                                              temperature=0.001)
                try_seq(a_opt, f"len_{target_n}_adam")
            # Quick pair search
            rem = deadline - time.monotonic() - 250
            if rem > 15:
                a_opt, _ = batch_pair_search(a_opt, time_limit=min(rem, 50), top_k=20)
                try_seq(a_opt, f"len_{target_n}_pair")

    # === Phase 5: Exact pair polishing ===
    print(f"\n=== Phase 5: Exact Pair Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 150)
            print(f"  Exact pair: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = exact_pair_polish(best_seq, time_limit=t_alloc, top_k=15)
            try_seq(a_opt, "exact_pair")

    # === Phase 6: Final refinement loop ===
    print(f"\n=== Phase 6: Final Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for round_num in range(5):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break

            # Fine Adam
            t_a = min(rem * 0.3, 40)
            if t_a > 8:
                a_opt, _ = adam_log_optimize(best_seq, time_limit=t_a,
                                              lr=0.002, top_k=n_best,
                                              temperature=0.0001,
                                              beta1=0.95, beta2=0.999)
                try_seq(a_opt, f"final_adam_{round_num}")

            # Fine pair search
            rem = deadline - time.monotonic() - 20
            if rem > 10:
                a_opt, _ = batch_pair_search(best_seq, time_limit=min(rem, 40), top_k=20)
                try_seq(a_opt, f"final_pair_{round_num}")

            # Perturbation + re-optimize
            rem = deadline - time.monotonic() - 15
            if rem > 10:
                for scale in [0.002, 0.005]:
                    rem = deadline - time.monotonic() - 10
                    if rem < 8:
                        break
                    noise = rng.standard_normal(n_best) * scale
                    f_pert = np.maximum(best_seq + noise, 0)
                    f_pert /= f_pert.sum()
                    a_opt, _ = adam_log_optimize(f_pert, time_limit=min(rem, 15),
                                                  lr=0.005, top_k=max(20, n_best // 5),
                                                  temperature=0.0005)
                    try_seq(a_opt, f"pert_{scale}_{round_num}")

            # Exact pair polish
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                a_opt, _ = exact_pair_polish(best_seq, time_limit=min(rem, 20), top_k=10)
                try_seq(a_opt, f"exact_polish_{round_num}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
