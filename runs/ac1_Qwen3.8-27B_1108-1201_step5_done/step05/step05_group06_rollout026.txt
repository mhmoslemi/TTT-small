"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based projected iterations with
line search, coordinate descent polishing, and multi-restart exploration.
Key insight: with sum(a)=sqrt(2n), the objective equals max(a*a), and an LP
gives the exact optimal direction at each step.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix A (2n-1 x n) such that A @ g = linear_conv(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
            options={'presolve': True, 'dual_feasibility_tol': 1e-9,
                     'primal_feasibility_tol': 1e-9}
        )
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return f.copy()


def solve_lp_active(f, M, n_active):
    """Solve LP using only the tightest constraints (active set)."""
    n = len(f)
    ff = fast_convolve(f)
    active = np.argpartition(ff, -n_active)[-n_active:]
    active = np.sort(active)

    A = np.zeros((len(active), n))
    for i in range(n):
        for idx, k in enumerate(active):
            j = k - i
            if 0 <= j < n:
                A[idx, i] = f[j]

    c = -np.ones(n)
    b = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
            options={'presolve': True}
        )
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            # Verify full constraint
            conv_check = np.convolve(f, g)
            if conv_check.max() <= M * 1.0001 + 1e-10:
                return g
    except Exception:
        pass
    return solve_lp(f, M)


def eval_obj_normalized(a):
    """Compute objective for sequence with sum = sqrt(2n). Returns max(a*a)."""
    n = len(a)
    conv = fast_convolve(a)
    return conv.max()


def lp_optimize(f0, time_limit, verbose=False, use_active=False, n_active=200):
    """
    LP-based iterative optimization.
    f0 should be normalized so sum(f0) = sqrt(2*len(f0)).
    Returns (best_f, best_M) where best_M = max(best_f * best_f).
    """
    n = len(f0)
    target_sum = np.sqrt(2.0 * n)

    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.full(n, target_sum / n)
    else:
        f = f * (target_sum / s)

    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP
        if use_active:
            g = solve_lp_active(f, M, n_active)
        else:
            g = solve_lp(f, M)

        g = np.maximum(g, 0.0)
        sg = g.sum()
        if sg < 1e-15:
            break
        g_res = g * (target_sum / sg)

        # Line search
        best_t = 0.3
        best_trial_M = M
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            f_trial = (1.0 - t) * f + t * g_res
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial * (target_sum / st)
            conv_trial = fast_convolve(f_trial)
            trial_M = conv_trial.max()
            if trial_M < best_trial_M:
                best_trial_M = trial_M
                best_t = t

        f = (1.0 - best_t) * f + best_t * g_res
        f = np.maximum(f, 0.0)
        st = f.sum()
        if st < 1e-15:
            f = best_f.copy()
            M = best_M
            continue
        f = f * (target_sum / st)

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 200 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    LP iter {it}: M={best_M:.8f}, t={best_t:.2f}, rem={rem:.0f}s")

        if no_improve > 300:

                f = f * (target_sum / st)
            else:
                f = best_f.copy()
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_M


def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Scale to probability simplex first, then scale to target_sum
    v_pos = np.maximum(v, 0.0)
    s = v_pos.sum()
    if s < 1e-15:
        return np.full(n, target_sum / n)
    # Projection onto sum=1 simplex
    u = np.sort(v_pos)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    result = np.maximum(v_pos - theta, 0.0)
    rs = result.sum()
    if rs > 1e-15:
        result = result * (target_sum / rs)
    else:
        result = np.full(n, target_sum / n)
    return result


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate-wise optimization with line search per coordinate."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)

    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)

    if s < 1e-15:

    else:
        a = a * (target_sum / s)

    conv = fast_convolve(a)
    best_M = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                # Try small positive values
                for x_try in [target_sum / (10 * n), target_sum / (5 * n)]:
                    a_trial = a.copy()
                    a_trial[j] = x_try

                    if rest_sum <= 0:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    rest = a[mask]
                    rs = rest.sum()
                    if rs > 1e-15:
                        a_trial[mask] = rest * (rest_sum / rs)
                    else:
                        a_trial[mask] = rest_sum / (n - 1)
                    ct = fast_convolve(a_trial)
                    mt = ct.max()
                    if mt < best_M - 1e-12:
                        best_M = mt
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        pass_improved = True
                continue

            # Try several values for a[j]
            candidates = [
                x_old * 0.5, x_old * 0.75, x_old * 1.25, x_old * 1.5,
                x_old + target_sum / (10 * n), x_old - target_sum / (10 * n)

            candidates = [max(0.0, c) for c in candidates]

            best_x = x_old
            best_trial_M = np.inf

            for x_try in candidates:
                a_trial = a.copy()
                a_trial[j] = x_try

                mask = np.ones(n, dtype=bool)

                conv = fast_convolve(a)

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    base = best_a if best_a is not None else parent
    target_lengths = [640, 896, 1024]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.4, 100)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, M_opt = lp_optimize(f_interp, time_limit=t_alloc, verbose=False)
        update_best(a_opt, f"len_{target_n}")

    # === Phase 6: Final intensive CD polishing ===
    print(f"\n=== Phase 6: Final CD polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, M_opt = coordinate_descent(best_a, time_limit=t_alloc)
            update_best(a_opt, "final_CD")

    # === Phase 7: Final LP fine-tuning ===
    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"\n=== Phase 7: Final LP fine-tune (best={best_score:.6f}) ===")
            print(f"  t={t_alloc:.0f}s")
            a_opt, M_opt = lp_optimize(best_a, time_limit=t_alloc, verbose=True,
                                       use_active=True, n_active=100)
            update_best(a_opt, "final_LP")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
