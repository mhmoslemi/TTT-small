"""
Multi-stage optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses structured initialization, exact coordinate descent with bisection,
multi-peak subgradient descent, adaptive restarts, and multi-length exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def coordinate_descent(a0, time_limit, verbose=False, max_passes=500):
    """Exact coordinate descent with bisection per coordinate."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    best_obj = obj_val(a)
    best_a = a.copy()
    t0 = time.monotonic()

    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Process in order of importance (larger values first)
        order = np.argsort(-a)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            x_old = a[j]
            if x_old < 1e-12:
                continue

            # Bisection: find x in [0, 1] minimizing the objective
            lo = 0.0
            hi = 1.0
            best_x = x_old
            best_obj_j = best_obj

            for _ in range(32):
                mid = (lo + hi) / 2.0
                rest = 1.0 - mid
                if rest < 1e-14:
                    hi = mid
                    continue
                old_rest = 1.0 - x_old
                if old_rest < 1e-12:
                    break
                scale = rest / old_rest
                trial = a * scale
                trial[j] = mid
                trial = np.maximum(trial, 0.0)
                c = fast_convolve(trial)
                obj_j = c.max()
                if obj_j < best_obj_j:
                    best_obj_j = obj_j
                    best_x = mid
                    hi = mid
                else:
                    lo = mid

            if best_x > 0 and best_obj_j < best_obj - 1e-13:
                rest = 1.0 - best_x
                old_rest = 1.0 - x_old
                scale = rest / old_rest
                a = a * scale
                a[j] = best_x
                a = np.maximum(a, 0.0)
                best_obj = best_obj_j
                best_a = a.copy()
                improved = True

        if not improved:
            break
        if verbose and p % 5 == 0:
            print(f"    CD pass {p}: obj={2*n*best_obj:.8f}")

    return best_a, 2.0 * n * best_obj


def subgradient_descent(a0, time_limit, lr=0.01, mom=0.9, top_k=40,
                        temp=0.001, verbose=False):
    """Multi-peak subgradient descent with momentum."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    v = np.zeros(n)
    best_obj = obj_val(a)
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        conv_max = conv.max()

        # Multi-peak subgradient
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]

        cur_lr = lr / (1.0 + 0.001 * it)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            noise = 0.003 + 0.003 * min(no_improve // 100, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    SG iter {it}: obj={2*n*best_obj:.8f}")

    return best_a, 2.0 * n * best_obj


def pair_optimization(a0, time_limit, verbose=False):

    a = normalize(a0.astype(np.float64))

                break
            xi, xj = a[i], a[j]
            if xi < 1e-12 or xj < 1e-12:

            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                # Transfer from i to j
                delta = frac * xi

                trial[i] = xi - delta
                trial[j] = xj + delta
                trial = np.maximum(trial, 0.0)
                trial = normalize(trial)
                c = fast_convolve(trial)
                if c.max() < best_obj - 1e-13:

                    best_a = trial.copy()
                    a = trial.copy()

                trial2 = a.copy()

                    best_a = trial2.copy()

def structured_init(n, seed=42):
    """Generate multiple structured initializations and return the best."""
    rng = np.random.default_rng(seed)
    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Bimodal (U-shaped via cosine)
    j = np.arange(n)
    for alpha in [0.3, 0.5, 0.7]:
        c = 1.0 + alpha * np.cos(2.0 * np.pi * j / n)
        c = np.maximum(c, 0.01)
        candidates.append(c / c.sum())

    # Exponential decay
    for rate in [0.5, 1.0, 2.0]:
        e = np.exp(-rate * j / n)
        candidates.append(e / e.sum())

    # Random
    for _ in range(5):
        r = rng.random(n)
        candidates.append(r / r.sum())

    # Inverted uniform (try to be flat in convolution)
    for p in [0.5, 1.0, 1.5, 2.0]:
        center = (n - 1) / 2.0
        w = 1.0 + p * (j - center) ** 2 / (center ** 2 + 1)
        candidates.append(w / w.sum())
        # Also inverted:
        w2 = 1.0 + p - p * (j - center) ** 2 / (center ** 2 + 1)
        w2 = np.maximum(w2, 0.01)
        candidates.append(w2 / w2.sum())

    # Find the best
    best = None
    best_obj = np.inf
    for c in candidates:
        c = normalize(c)
        o = obj_val(c)
        if o < best_obj:
            best_obj = o
            best = c.copy()
    return best

            if rem < 20:
                break
            init = structured_init(n_parent, seed=100 + s * 17)
            t_alloc = min(40, rem * 0.5)
            a_opt, _ = combined_optimize(init * parent.sum(), t_alloc)
            update_best(a_opt * parent.sum(), f"struct_{s}")

    # === Stage 3: Multi-length exploration ===
    print(f"\n=== Stage 3: Multi-length (best={best_score:.6f}) ===")
    base = best_a / best_a.sum() if best_a is not None else parent_norm
    target_lengths = [512, 640, 896, 1024, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        scale = parent.sum() * target_n / n_parent  # scale to maintain similar magnitude
        t_alloc = min(80, rem * 0.5)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(f_interp * scale, t_alloc, verbose=False)
        update_best(a_opt * scale, f"n{target_n}")

    # === Stage 4: Intensive coordinate descent on best ===
    print(f"\n=== Stage 4: Intensive CD (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            n_best = len(best_a)
            a_norm_best = best_a / best_a.sum()
            total_sum = best_a.sum()
            t_alloc = min(200, rem)
            print(f"  CD on best, n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(a_norm_best, t_alloc, verbose=True)
            update_best(a_opt * total_sum, "intensive_cd")

    # === Stage 5: More perturbations and optimization ===
    print(f"\n=== Stage 5: Perturbation search (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        total_sum = best_a.sum()
        a_norm_best = best_a / best_a.sum()

        for p in range(10):
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            rng_p = np.random.default_rng(seed + 5000 + p * 13)
            noise_scale = 0.003 + 0.004 * (p / 10.0)
            f_pert = project_simplex(a_norm_best + rng_p.standard_normal(n_best) * noise_scale)
            t_alloc = min(40, rem * 0.5)
            a_opt, _ = combined_optimize(f_pert * total_sum, t_alloc)
            update_best(a_opt * total_sum, f"pert_{p}")

    # === Stage 6: Symmetric variant ===
    print(f"\n=== Stage 6: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_sym = 0.5 * (best_a + best_a[::-1])
            a_sym = np.maximum(a_sym, 0)
            t_alloc = min(60, rem)
            a_opt, _ = combined_optimize(a_sym, t_alloc)
            update_best(a_opt, "symmetric")

    # === Stage 7: Final intensive polish ===
    print(f"\n=== Stage 7: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            n_final = len(best_a)
            total_sum = best_a.sum()
            a_norm_f = best_a / best_a.sum()
            t_alloc = min(rem, 120)
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(a_norm_f, t_alloc)
            update_best(a_opt * total_sum, "final")

            # Tiny perturbations + quick CD
            rem = deadline - time.monotonic() - 5
            if rem > 10 and best_a is not None:
                for tp in range(3):
                    rem = deadline - time.monotonic() - 3
                    if rem < 8:
                        break
                    rng_tp = np.random.default_rng(seed + 9000 + tp)
                    f_tp = project_simplex(best_a / best_a.sum() + rng_tp.standard_normal(n_final) * 0.001)
                    t_alloc = min(10, rem)
                    a_opt, _ = coordinate_descent(f_tp, t_alloc)
                    update_best(a_opt * best_a.sum(), f"tiny_{tp}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
