import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the convolution matrix A where (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, edge_relax=0.0):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M_k for all k, g >= 0.
    M_k = M * (1 + edge_relax * ((k - center)/center)^2) for edge relaxation.
    """
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    
    if edge_relax > 0:
        center = n - 1
        k = np.arange(2 * n - 1, dtype=np.float64)
        b_ub = M * (1.0 + edge_relax * ((k - center) / center) ** 2)
    else:
        b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
        elif res.status == 9:
            return None  # infeasible (running time limit)
        else:
            return None
    except Exception:
        return None


def adaptive_fw_lp(f0, time_limit, edge_relax_init=0.1, verbose=False):

                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

def subgradient_polish(f0, time_limit, lr_init=0.005, mom=0.9, top_k=20):
    """
    Multi-peak subgradient with momentum.
    Uses the top-k peaks of the auto-convolution as subgradients.
    """
    a = np.maximum(f0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        weights /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]
        
        # Momentum update
        v = mom * v - lr * sg
        a_new = project_simplex(a + v)
        
        obj = float(fast_convolve(a_new).max())
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Don't update a if no improvement (keep best)
        
        lr *= 0.9995
        mom = min(0.97, mom + 0.00005)
        
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def coord_local_search(f0, time_limit, n_trials=5000):
    """
    Coordinate-level local search: try small random perturbations
    to 1-5 coordinates and accept if improvement.
    """
    a = np.maximum(f0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    trial = 0
    
    while time.monotonic() - t0 < time_limit and trial < n_trials:
        trial += 1
        
        # Random perturbation of 1-5 coordinates
        n_pert = rng.integers(1, 6)
        idx = rng.choice(n, size=n_pert, replace=False)
        scales = rng.uniform(-0.01, 0.01, size=n_pert)
        
        candidate = a.copy()
        for i, (ix, sc) in enumerate(zip(idx, scales)):
            candidate[ix] = max(0.0, a[ix] + sc)
        
        cs = candidate.sum()
        if cs < 1e-15:
            continue
        candidate = candidate / cs
        
        obj = float(fast_convolve(candidate).max())
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = candidate.copy()
            a = candidate
    
    return best_a, best_obj

        best_val = float(fast_convolve(a).max())
        
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = float(c.max())
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if no_improve > 30:
            T *= 1.03
            lr *= 0.98
        if no_improve > 200:
            rng = np.random.default_rng((it * 997 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick exploration of multiple n values with FW-LP
    Phase 2: Deep FW-LP on best candidate
    Phase 3: Subgradient polish
    Phase 4: Smoothed gradient descent
    Phase 5: Coordinate local search
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_list = a.tolist()
        score = float(evaluate_sequence(a_list))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = try_seq(parent, "parent")
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
    except Exception as e:
        print(f"No parent: {e}")
    
    if parent is None:
        parent = np.ones(512) / 512
    
    n_parent = len(parent)
    
    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t1 = time.monotonic()
    
    # Candidate n values
    n_candidates = sorted(set([
        n_parent,
        512, 768, 1024, 1536,
        max(256, n_parent // 2),
        min(2048, n_parent * 2),
    ]))
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 800
        if remaining < 30:
            break
        if n_try < 128 or n_try > 2048:
            continue
        
        if n_try == n_parent:
            start = parent.copy()
        else:
            # Interpolate parent to new length
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            start = np.interp(x_new, x_old, parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss < 1e-10:
                start = np.ones(n_try) / n_try
            else:
                start = start / ss
        
        # Quick FW-LP
        t_alloc = min(remaining, 120)
        print(f"  FW-LP n={n_try}, alloc={t_alloc:.0f}s")
        a_opt, _ = adaptive_fw_lp(start, time_limit=t_alloc, edge_relax_init=0.15)
        try_seq(a_opt, f"fw_n{n_try}")
        
        # Try with different edge relaxation
        remaining = deadline - time.monotonic() - 700
        if remaining > 20:
            a_opt2, _ = adaptive_fw_lp(start, time_limit=min(remaining, 60), edge_relax_init=0.3)
            try_seq(a_opt2, f"fw_edge0.3_n{n_try}")
    
    print(f"  Phase 1 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t1:.0f}s)")
    
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    n_best = len(best_seq)
    
    # === Phase 2: Deep FW-LP on best ===
    print(f"\n=== Phase 2: Deep FW-LP (n={n_best}) ===")
    t2 = time.monotonic()
    
    remaining = deadline - time.monotonic() - 400
    if remaining > 30:
        t_alloc = min(remaining, 300)
        print(f"  Deep FW-LP, alloc={t_alloc:.0f}s")
        a_opt, _ = adaptive_fw_lp(best_seq, time_limit=t_alloc, edge_relax_init=0.1)
        try_seq(a_opt, "deep_fw")
    
    # Perturbed restarts
    remaining = deadline - time.monotonic() - 300
    for scale in [0.002, 0.005, 0.01]:
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 80)
        a_opt, _ = adaptive_fw_lp(f_pert, time_limit=t_alloc, edge_relax_init=0.2)
        try_seq(a_opt, f"fw_pert_{scale}")
        remaining = deadline - time.monotonic() - 250
    
    # Try different edge relaxation on best
    remaining = deadline - time.monotonic() - 200
    if remaining > 20:
        for er in [0.05, 0.2, 0.4]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            a_opt, _ = adaptive_fw_lp(best_seq, time_limit=min(remaining, 60), edge_relax_init=er)
            try_seq(a_opt, f"fw_er{er}")
    
    print(f"  Phase 2 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t2:.0f}s)")
    
    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    t3 = time.monotonic()
    
    remaining = deadline - time.monotonic() - 200
    if remaining > 15:
        for cfg in [
            {'lr_init': 0.003, 'mom': 0.92, 'top_k': 15},
            {'lr_init': 0.005, 'mom': 0.88, 'top_k': 30},
            {'lr_init': 0.002, 'mom': 0.95, 'top_k': 10},
        ]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10:
                break
            t_alloc = min(remaining, 80)
            a_opt, _ = subgradient_polish(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"sg_k{cfg['top_k']}")
    
    print(f"  Phase 3 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t3:.0f}s)")
    
    # === Phase 4: Smoothed gradient descent ===
    print(f"\n=== Phase 4: Smoothed Gradient Descent ===")
    t4 = time.monotonic()
    
    remaining = deadline - time.monotonic() - 120
    if remaining > 10:
        for T_init in [0.0002, 0.0005, 0.001]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            a_opt, _ = smoothed_soft_max_descent(best_seq, time_limit=t_alloc, T_init=T_init, lr_init=0.02)
            try_seq(a_opt, f"smooth_T{T_init}")
    
    print(f"  Phase 4 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t4:.0f}s)")
    
    # === Phase 5: Coordinate local search + final FW ===
    print(f"\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    
    remaining = deadline - time.monotonic() - 40
    if remaining > 10:
        t_alloc = min(remaining, 60)
        print(f"  Coord search, alloc={t_alloc:.0f}s")
        a_opt, _ = coord_local_search(best_seq, time_limit=t_alloc, n_trials=20000)
        try_seq(a_opt, "coord")
    
    remaining = deadline - time.monotonic() - 20
    if remaining > 10:
        t_alloc = min(remaining, 30)
        a_opt, _ = adaptive_fw_lp(best_seq, time_limit=t_alloc, edge_relax_init=0.05)
        try_seq(a_opt, "final_fw")
    
    remaining = deadline - time.monotonic() - 5
    if remaining > 5:
        a_opt, _ = coord_local_search(best_seq, time_limit=remaining, n_trials=10000)
        try_seq(a_opt, "final_coord")
    
    print(f"  Phase 5 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t5:.0f}s)")
    
    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            print(f"\nFINAL VERIFIED SCORE: {fs:.8f} (n={len(best_seq)})")
    
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    return [float(x) for x in best_seq]
