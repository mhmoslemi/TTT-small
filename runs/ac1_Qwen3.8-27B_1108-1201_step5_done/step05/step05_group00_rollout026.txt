    n = len(a)

    # subject to: conv[k] + grad_k @ d <= M_new  for all k
    #             sum(d) = 0
    #             -r <= d_i <= r  for all i
    #             d_i >= -a_i  (i.e., a_i + d_i >= 0)
    
    # Variables: d (n), M_new (1) -> total n+1
    # Objective: minimize M_new -> c = [0, ..., 0, 1]

    #   -> grad_k @ d - M_new <= -conv[k]
    # Matrix shape: (2n-1) x (n+1)

    b = np.zeros(2*n-1)

        A[k, n] = -1

    #   -d_i <= a_i

    # Equality: 1

    A_full = np.zeros((5*n-1, n+1))

    # Conv constraints (already in A)

    b_full[idx:idx+2*n-1] = b

    for i in range(n):
        A_full[idx, i] = 1
        b_full[idx] = r
        idx += 1

    for i in range(n):
        A_full[idx, i] = -1
        b_full[idx] = r
        idx += 1

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-based trust region steps (linearized convolution constraints + box constraint on step)
- Adaptive trust region with line search

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(a):
    """Build the (2n-1) x n matrix where entry [k,m] = 2*a[k-m] if valid, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = 2.0 * a
    return A


def lp_trust_region_step(a, r):
    """
    Solve LP for a trust-region descent direction.
    Minimizes M_new subject to linearized conv constraints and box on d.
    Returns (d_opt, M_new) or (None, None).
    """
    n = len(a)
    conv = fast_convolve(a)

    A_conv = build_constraint_matrix(a)

    num_ineq = 5 * n
    A_ub = np.zeros((num_ineq, n + 1))
    b_ub = np.zeros(num_ineq)

    # Conv constraints: A_conv @ d - M_new <= -conv
    A_ub[:2 * n - 1, :n] = A_conv
    A_ub[:2 * n - 1, n] = -1.0
    b_ub[:2 * n - 1] = -conv

    idx = 2 * n - 1

    # d_i <= r
    I = np.eye(n)
    A_ub[idx:idx + n, :n] = I
    b_ub[idx:idx + n] = r
    idx += n

    # -d_i <= r
    A_ub[idx:idx + n, :n] = -I
    b_ub[idx:idx + n] = r
    idx += n

    # -d_i <= a_i (ensures a_i + d_i >= 0)
    A_ub[idx:idx + n, :n] = -I
    b_ub[idx:idx + n] = a
    idx += n

    # -M_new <= 0
    A_ub[idx, n] = -1.0
    b_ub[idx] = 0.0
    idx += 1

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * n + [(0.0, None)],
                      method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def smooth_grad_optimize(a0, time_limit, T_init=0.0003, lr_init=0.03):
    """Smoothed gradient descent with log-sum-exp surrogate and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    v = np.zeros(n)
    mom = 0.9
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Log-sum-exp gradient: weighted sum of peak gradients
        scaled = (conv - M) / T
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient = 2 * sum_k p_k * grad_k where grad_k[m] = 2*a[k-m]
        # = 4 * sum_k p_k * a[k-m] (for valid indices)
        # This is 4 * (p correlated with a)
        grad = 4.0 * np.correlate(p, a, mode='valid')

        # Momentum update
        v = mom * v - lr * grad
        candidate = project_simplex(a + v)

        val = fast_convolve(candidate).max()
        if val < M - 1e-15:
            a = candidate
            if val < best_obj:
                best_obj = val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            lr = min(lr * 1.02, 0.1)
        else:
            no_improve += 1
            v = np.zeros(n)
            lr *= 0.9
            T *= 1.05

        if no_improve > 100:
            T *= 0.7
            lr = lr_init * 0.5

            T = T_init
            lr = lr_init
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def active_peak_sg(a0, time_limit, top_k=5, lr_init=0.003, mom=0.93):

        candidate = project_simplex(a + v)

            best_obj = val

def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle (kept from previous for diversity)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        # Solve LP: max sum(g) s.t. (conv(f,g))_k <= M, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
            if res.success:
                g = np.maximum(res.x, 0.0)
                g_sum = g.sum()
                if g_sum < 1e-15:
                    no_improve += 1
                    continue
                g = g / g_sum

                if np.max(np.abs(g - f)) < 1e-10:
                    no_improve += 1
                    if no_improve > 50:
                        break
                    continue

                best_alpha = 0.0
                best_val = M
                for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    ft = (1.0 - alpha) * f + alpha * g
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha

                if best_alpha > 1e-10:
                    f = (1.0 - best_alpha) * f + best_alpha * g
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    M = fast_convolve(f).max()
                    if M < best_obj - 1e-15:
                        best_obj = M
                        best_f = f.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, elapsed={time.monotonic()-t_start:.0f}s)")
    return [float(x) for x in best_seq]
