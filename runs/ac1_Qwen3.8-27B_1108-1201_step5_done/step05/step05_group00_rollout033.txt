```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- SLP (Sequential LP) with direct peak minimization via linearized quadratic constraints
- Active-set Newton refinement for local convergence
- Multi-n exploration with diverse initializations
- Different from FW-LP: directly minimizes the peak via LP rather than finding a direction
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def slp_step(a0):
    """
    One SLP iteration: linearize the quadratic constraints at a0 and solve the LP.
    Returns (a_candidate, t_candidate).
    """
    n = len(a0)
    b0 = fast_convolve(a0)
    
    # Build LP: minimize t
    # subject to grad_k . a <= t + b0[k] - S2_k for all k
    # sum(a) = 1, a >= 0
    # Variables: a[0], ..., a[n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: grad_k . a - t <= b0[k] - S2_k
    # For each k, grad_k has n components:
    #   grad_k[j] = 2*a0[k-j] if 0 <= k-j < n and k != 2*j
    #              = a0[k-j] if k == 2*j
    #              = 0 otherwise
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for k in range(2 * n - 1):
        # Compute gradient
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if k == 2 * j:
                    A_ub[k, j] = a0[kj]  # = a0[j]
                else:
                    A_ub[k, j] = 2.0 * a0[kj]
        # t coefficient
        A_ub[k, n] = -1.0
        # RHS
        S2 = a0[k // 2] ** 2 if (k % 2 == 0 and 0 <= k // 2 < n) else 0.0
        b_ub[k] = b0[k] - S2
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return a0.copy(), b0.max()
```

```python
    b_ub = np.zeros(m)

        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        # grad values: 2*a0[k-j] for all j in range, except where k=2j
        grad_vals = 2.0 * a0[k - js]
```
