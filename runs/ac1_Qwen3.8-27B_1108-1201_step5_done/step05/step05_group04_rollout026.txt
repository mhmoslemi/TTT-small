def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f):

    cols = []

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    idx = np.arange(n)
    M = w[idx[:, None] + idx[None, :]]

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_weights(conv, beta):
    c = beta * conv
    c_max = c.max()
    e = np.exp(c - c_max)
    s = e.sum()
    if s < 1e-300:
        return np.zeros_like(conv)
    return e / s

def lse_grad(f, w):
    """grad[j] = 2 * sum_{i=0}^{n-1} f[i] * w[j+i]"""
    n = len(f)
    idx = np.arange(n)
    M = w[idx[:, None] + idx[None, :]]
    return 2.0 * (M @ f)

def lse_optimize(f0, time_limit, beta_start=5.0, beta_max=20000.0,
                 lr_init=0.02, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f)) / len(f)
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    
    # Polyak averaging
    polyak_f = f.copy()
    polyak_w = 0.0
    
    lr = lr_init
    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        w = lse_weights(conv, beta)
        grad = lse_grad(f, w)
        
        # Adam
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad**2
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        step = m_hat / (np.sqrt(v_hat) + 1e-8)
        
        f_new = project_simplex(f - lr * step)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            polyak_w += lr
            polyak_f = polyak_f * (1 - lr/(polyak_w)) + f_new * (lr/(polyak_w))
            # occasionally increase lr
            if it % 500 == 0:
                lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            if no_improve >= 3:
                lr *= 0.97
            if no_improve > 80:
                # restart
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                beta = min(beta * 2.0, beta_max)
                lr = max(0.005, lr * 0.5)
                no_improve = 0
        
        f = f_new
        t += 1
        beta = min(beta * 1.003, beta_max)
        
        if verbose and it % 1000 == 0:
            print(f"    it={it}, obj={best_obj:.10
