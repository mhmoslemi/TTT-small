"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses subgradient methods with FFT-based convolution, adaptive peak weighting,
Nesterov acceleration, coordinate descent polish, and multi-length exploration.
Key innovation: dual-scale optimization (coarse-to-fine) and simulated annealing
for escaping local minima, combined with structured initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """Compute weighted subgradient of max(conv) using top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    # Get top-k peak positions
    if k_actual == len(conv):
        peaks = np.arange(len(conv))
        vals = conv
    else:
        peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks = peak_indices
        vals = conv[peak_indices]
    
    # Softmax weighting with temperature
    scaled = vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            sg[indices] += 2.0 * w[i] * a[k - indices]
    
    return sg


def subgradient_optimize(a0, time_limit, lr=0.003, momentum=0.85,
                         top_k=30, temp=0.001, shake_interval=500,
                         shake_scale=0.003, verbose=False):
    """Nesterov-accelerated projected subgradient with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_mom = momentum
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        sg = compute_subgradient(a, conv, top_k, cur_temp)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Nesterov step
        v = cur_mom * v - cur_lr * sg / sg_norm
        y_new = project_simplex(y + v)
        a_new = y_new
        y = a_new + cur_mom * (a_new - a)
        y = project_simplex(y)
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        cur_lr *= 0.9997
        cur_temp *= 1.005

        # Shakeout: escape local minima

            scale = shake_scale * min(1.0 + no_improve // (2 * shake_interval), 3.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            y = a.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
            conv = fast_convolve(a)
        
        if verbose and it % 500 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, block_size=1, max_passes=50):
    """Block coordinate descent with golden section search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    
    passes = 0
    no_pass_improve = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        
        for j in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            j_end = min(j + block_size, n)
            old_block = a[j:j_end].copy()
            if old_block.sum() < 1e-14:
                
                continue
            
            denom = 1.0 - old_block.sum()
            if denom < 1e-12:
                continue
            
            if block_size == 1:
                # Single coordinate: golden section on a[j]
                old_val = a[j]
                lo = 0.0
                hi = min(0.999, max(3.0 * old_val, 2.0 / n))
                
                def eval_single(x):
                    at = a.copy()
                    at[j] = x
                    at2 = 1.0 - x
                    if at2 < 0:
                        return float('inf')
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    at[mask] = at[mask] * (at2 / denom)
                    return fast_convolve(at).max()
                
                c1 = lo + inv_gr * (hi - lo)
                c2 = lo + (1.0 - inv_gr) * (hi - lo)
                f1 = eval_single(c1)
                f2 = eval_single(c2)
                
                for _ in range(10):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = lo + inv_gr * (hi - lo)
                        f1 = eval_single(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + (1.0 - inv_gr) * (hi - lo)
                        f2 = eval_single(c2)
                
                opt_x = (lo + hi) / 2.0
                opt_val = min(f1, f2)
                cur_val = fast_convolve(a).max()
                
                if opt_val < cur_val - 1e-14:
                    at = a.copy()
                    at[j] = opt_x
                    at2 = 1.0 - opt_x
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    at[mask] = at[mask] * (at2 / denom)
                    at = at / at.sum()
                    a = at
                    pass_improved = True
                    new_obj = fast_convolve(a).max()
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
            else:
                # Block: try random directions in the block subspace
                rng = np.random.default_rng(j * 7919 + passes)
                for _ in range(3):
                    direction = rng.standard_normal(block_size)
                    direction -= direction.mean()  # Keep sum approximately constant
                    if np.abs(direction).max() < 1e-10:
                        continue
                    
                    # Line search along direction
                    best_t = 0.0
                    best_val = fast_convolve(a).max()
                    for t in [0.001, 0.005, 0.01, 0.02, 0.05, -0.005, -0.01]:
                        at = a.copy()
                        at[j:j_end] = old_block + t * direction
                        at = np.maximum(at, 0.0)
                        st = at.sum()
                        if st < 1e-10:
                            continue
                        at = at / st
                        val = fast_convolve(at).max()
                        if val < best_val:
                            best_val = val
                            best_t = t
                    
                    if best_t != 0.0 and best_val < fast_convolve(a).max() - 1e-14:
                        at = a.copy()
                        at[j:j_end] = old_block + best_t * direction
                        at = np.maximum(at, 0.0)
                        at = at / at.sum()
                        a = at
                        pass_improved = True
                        new_obj = fast_convolve(a).max()
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
        
        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coarse_to_fine(a_coarse, fine_n, time_per_stage):
    """Multi-resolution optimization: start coarse, refine to target length."""
    coarser_n = len(a_coarse)
    # Progressively upsample
    lengths = []
    cur = coarser_n
    while cur < fine_n:
        cur = min(cur * 2, fine_n)
        lengths.append(cur)
    if fine_n not in lengths:
        lengths.append(fine_n)
    
    a = a_coarse.copy()
    for target_n in lengths:
        if target_n == len(a):
            continue
        a = interpolate_sequence(a, target_n)
        a, _ = subgradient_optimize(a, time_per_stage, lr=0.005, momentum=0.85,
                                     top_k=20, temp=0.001, shake_interval=300)
    
    return a


def cosine_init(n, freq, amp, seed=0):
    """Generate cosine-modulated initialization."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    # Add small noise for diversity
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def ramp_init(n, seed=0):
    """Ramp-like initialization."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    a = 1.0 + 0.5 * (j / max(n - 1, 1))
    a += rng.standard_normal(n) * 0.02
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def bimodal_init(n, frac, seed=0):
    """Bimodal initialization: high at edges, low in middle (or vice versa)."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    center = n // 2
    dist = np.abs(j - center) / center
    a = 1.0 + frac * dist**2
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Generate diverse initializations at multiple lengths
    2. Run subgradient optimization on each
    3. Interpolate best to target lengths
    4. Coordinate descent polish
    5. Final fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    n_parent = len(parent)
    
    # === Phase 1: Quick subgradient on parent ===
    print(f"\n=== Phase 1: Quick SG on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 1200
    if rem > 10:
        t1 = min(rem, 60)
        a_opt, _ = subgradient_optimize(parent, time_limit=t1, lr=0.005,
                                         momentum=0.85, top_k=30, temp=0.001)
        try_seq(a_opt, "sg_parent")
    
    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length exploration (best={best_score:.6f}) ===")
    
    target_lengths = [512, 768, 1024, 1280, 1536, 1800]
    
    for n_target in target_lengths:
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        
        # Generate initializations for this length
        inits = []
        
        # From parent (interpolated)
        if len(parent) != n_target:
            inits.append(interpolate_sequence(parent, n_target))
        else:
            inits.append(parent.copy())
        
        # Cosine variants
        for freq in [1, 2, 3]:
            for amp in [0.2, 0.4]:
                inits.append(cosine_init(n_target, freq, amp, seed=freq * 100 + amp * 10))
        
        # Bimodal
        inits.append(bimodal_init(n_target, 0.5, seed=n_target % 100))
        
        # Ramp
        inits.append(ramp_init(n_target, seed=n_target % 50))
        
        # From best sequence if different length
        if best_seq is not None and len(best_seq) != n_target:
            inits.append(interpolate_sequence(best_seq, n_target))
        
        best_for_length = None
        best_for_length_score = float('inf')
        
        for idx, a_init in enumerate(inits):
            rem = deadline - time.monotonic() - 500
            if rem < 8:
                break
            
            t_alloc = min(rem * 0.15, 30)
            if t_alloc < 5:
                t_alloc = 5
            
            a_opt, obj = subgradient_optimize(a_init, time_limit=t_alloc,
                                               lr=0.004, momentum=0.87,
                                               top_k=max(15, n_target // 40),
                                               temp=0.0005, shake_interval=300)
            score = try_seq(a_opt, f"sg_n{n_target}_init{idx}")
            
            if np.isfinite(score) and score < best_for_length_score:
                best_for_length_score = score
                best_for_length = a_opt
        
        # If we found something good for this length, try more refinement
        if best_for_length is not None:
            rem = deadline - time.monotonic() - 400
            if rem > 15:
                t_ref = min(rem * 0.25, 60)
                a_opt, _ = subgradient_optimize(best_for_length, time_limit=t_ref,
                                                 lr=0.002, momentum=0.90,
                                                 top_k=max(20, n_target // 30),
                                                 temp=0.0002, shake_interval=400)
                try_seq(a_opt, f"refine_n{n_target}")
    
    # === Phase 3: Coarse-to-fine from parent ===
    print(f"\n=== Phase 3: Coarse-to-fine (best={best_score:.6f}) ===")
    
    # Make a very coarse version of parent
    coarse_n = max(64, n_parent // 8)
    a_coarse = interpolate_sequence(parent, coarse_n)
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_c2f = min(rem * 0.4, 80)
        a_cf = coarse_to_fine(a_coarse, 1024, t_c2f / 4)
        try_seq(a_cf, "coarse_to_fine_1024")
        
        rem = deadline - time.monotonic() - 200
        if rem > 20 and best_seq is not None and len(best_seq) <= 1024:
            a_cf2 = coarse_to_fine(best_seq, 1536, min(rem, 40) / 3)
            try_seq(a_cf2, "coarse_to_fine_1536")
    
    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            t_cd = min(rem * 0.5, 120)
            print(f"  CD n={n_best}, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, block_size=1)
            try_seq(a_opt, "cd_best")
            
            # Also try block CD
            rem = deadline - time.monotonic() - 150
            if rem > 10 and n_best <= 1024:
                a_opt2, _ = coordinate_descent(a_opt, time_limit=min(rem, 30), block_size=4)
                try_seq(a_opt2, "block_cd")
    
    # === Phase 5: Simulated annealing on best ===
    print(f"\n=== Phase 5: Simulated annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_sa = min(rem * 0.6, 120)
            a_sa = best_seq.copy()
            cur_obj = fast_convolve(a_sa).max()
            T = 0.01
            T_final = 0.0001
            t0 = time.monotonic()
            sa_it = 0
            
            while time.monotonic() - t0 < t_sa:
                sa_it += 1
                # Random perturbation: pick a few random positions and adjust
                n_perturb = max(3, n_best // 100)
                positions = rng.integers(0, n_best, size=n_perturb)
                deltas = rng.standard_normal(n_perturb) * 0.02 * T / 0.01
                
                a_try = a_sa.copy()
                for p, d in zip(positions, deltas):
                    a_try[p] += d
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-10:
                    continue
                a_try = a_try / st
                
                try_obj = fast_convolve(a_try).max()
                delta = try_obj - cur_obj
                
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    a_sa = a_try
                    cur_obj = try_obj
                    if cur_obj < best_obj_check(a_sa):
                        pass  # Will be caught by try_seq below
                
                T = max(T_final, T * 0.999)
            
            try_seq(a_sa, "sim_anneal")
    
    # === Phase 6: Final intensive optimization ===
    print(f"\n=== Phase 6: Final optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Multiple short runs with different hyperparameters
        configs = [
            (0.001, 0.92, 20, 0.0001),
            (0.003, 0.88, 40, 0.0003),
            (0.0005, 0.95, 10, 0.00005),
            (0.002, 0.90, 50, 0.0005),
        ]
        
        for cfg_idx, (lr_c, mom_c, tk_c, temp_c) in enumerate(configs):
            rem = deadline - time.monotonic() - 60
            if rem < 10:
                break
            t_alloc = min(rem * 0.3, 60)
            
            if cfg_idx == 0:
                a_start = best_seq.copy()
            else:
                noise = rng.standard_normal(n_best) * 0.003
                a_start = project_simplex(best_seq + noise)
            
            a_opt, _ = subgradient_optimize(a_start, time_limit=t_alloc,
                                             lr=lr_c, momentum=mom_c,
                                             top_k=tk_c, temp=temp_c,
                                             shake_interval=300,
                                             shake_scale=0.002)
            try_seq(a_opt, f"final_sg_{cfg_idx}")
        
        # Final CD
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def best_obj_check(a):
    """Helper to compute objective for best tracking."""
    conv = fast_convolve(a)
    return conv.max()
