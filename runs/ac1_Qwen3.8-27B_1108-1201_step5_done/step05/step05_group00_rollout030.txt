```python
def solve_qp(M, n):
    """Solve min a^T M a s.t. sum(a) = 1, a >= 0."""
    # Use active set method
    active = list(range(n))
    inactive = set()
    
    for iteration in range(1000):
        S = np.array(active)
        k = len(S)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            a = np.zeros(n)
            a[S[0]] = 1.0
            return a
        
        # Extract submatrix
        M_SS = M[np.ix_(S, S)]
        
        # KKT: 2*M_SS * a_S + lambda * 1 = 0, sum(a_S) = 1
        # => (2*M_SS, 1; 1^T, 0) [a_S; lambda] = [0; 1]
        KKT = np.zeros((k+1, k+1))
        KKT[:k, :k] = 2 * M_SS
        KKT[:k, k] = 1.0
        KKT[k, :k] = 1.0
        rhs = np.zeros(k+1)
        rhs[k] = 1.0
        
        try:
            sol = np.linalg.solve(KKT, rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(KKT, rhs, rcond=None)[0]
        
        a_S = sol[:k]
        lambda_val = sol[k]
        
        # Check if all a_S >= 0
        if np.all(a_S >= -1e-10):
            a_S = np.maximum(a_S, 0)
            # Check KKT for inactive set
            a_full = np.zeros(n)
            a_full[S] = a_S
            grad_full = 2 * M @ a_full
            # For inactive j: grad_full[j] + lambda_val >= 0
            min_inactive = min([grad_full[j] + lambda_val for j in inactive]) if inactive else 1.0
            if min_inactive >= -1e-8:
                a_full[S] = a_S
                return a_full
            else:
                # Add the most violated inactive variable
                worst = min(inactive, key=lambda j: grad_full[j] + lambda_val)
                inactive.remove(worst)
                active.append(worst)
        else:
            # Remove the most negative variable
            min_idx = np.argmin(a_S)
            if a_S[min_idx] < -1e-12:
                worst = S[min_idx]
                active.remove(int(worst))
                inactive.add(int(worst))
            else:
                # Numerical issues, break
                a_S = np.maximum(a_S, 0)
                a_S = a_S / a_S.sum()
                a_full = np.zeros(n)
                a_full[S] = a_S
                return a_full
    return None
```

This is a "minimax" problem. A standard approach is:
- Introduce t: minimize t s.t. f_k(a) <= t for all k, a in simplex
- Use the method of alternating: fix t, optimize a; fix a, optimize t (= max_k f_k(a))

For the inner problem (fix t, optimize a): minimize 0 s.t. a^T Q_k a <= t for all k, a in simplex. This is a feasibility problem with quadratic constraints - hard in general.

minimize sum_k max(0, f_k(a) - t) + mu * (t - t_min)
This is like a soft max.

minimize (1/β) * log(sum_k exp(β * f_k(a)))
This is a smooth approximation to the max, and for large β, it's close to the actual max.

The gradient of this smooth objective is:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smoothed gradient
descent with peak-balancing QP subproblems, parametric initializations,
and multi-start random restarts. Key improvements over previous approach:
- Adam optimizer with per-coordinate adaptive learning rates
- Aggressive temperature annealing with restart logic
- Active-set QP for exact peak balancing (not LP relaxation)
- Beta-distribution parametric family for diverse initializations
- Multiple n-exploration to find optimal length

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_gradient(a, p):
    """Gradient of sum_k p[k]*(a*a)[k] with respect to a.
    Result: 2 * (Q(p) @ a) where (Q(p)@a)[j] = sum_k p[k]*Q_k[j,*]@a
    Equivalent to 2 * correlate(p, a, mode='valid') for the Hankel structure.
    """
    n = len(a)
    # (a*a)[k] = sum_j a[j]*a[k-j], d/d a[m] = sum_k p[k] * (a[k-m] + a[m-k] if valid)
    # Efficiently: grad[m] = 2 * sum_k p[k] * a[k-m] (with bounds)
    # This is 2 * (p * reverse(a)) convolved... actually it's a correlation.
    # grad[m] = 2 * sum_{k} p[k] * a[k-m] where 0<=k-m<n and 0<=k<2n-1
    # Let l = k-m, then k = l+m, and we need 0<=l<n, 0<=l+m<2n-1
    # grad[m] = 2 * sum_l p[l+m] * a[l] for valid l, m
    # This is 2 * correlate(p, a, mode='valid')
    return 2.0 * np.correlate(p, a, mode='valid')


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def beta_param_sequence(n, s, t):
    """Generate a sequence shaped like Beta(s+1, t+1) PDF on {0,...,n-1}."""
    x = np.arange(n, dtype=np.float64)
    # Avoid zeros at boundaries
    x_safe = np.clip(x + 0.5, 0.01, n - 0.01)
    log_a = s * np.log(x_safe / n) + t * np.log((n - x_safe) / n)
    a = np.exp(log_a - log_a.max())
    s_sum = a.sum()
    if s_sum < 1e-15:
        return np.ones(n) / n
    return a / s_sum


def adaptive_smoothed_grad(a0, time_limit, T_init=0.001, T_min=0.00001):
    """
    Smoothed gradient descent with Adam optimizer and adaptive temperature.
    Uses log-sum-exp smoothing of the max objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t_step = 0

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    lr_base = 0.02

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Softmax weights (temperature-scaled)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum

        # Gradient
        grad = conv_gradient(a, p)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)

        # Adaptive learning rate with backtracking line search
        lr = lr_base / np.sqrt(t_step + 1)
        best_candidate = None
        best_val = current_max

        for lr_trial in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            step = lr_trial * m_hat / (np.sqrt(v_hat) + eps)
            candidate = project_simplex(a - step)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        if best_candidate is not None:
            a = best_candidate
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive temperature
        if no_improve > 50:
            T = max(T_min, T * 0.9)  # Cool down to focus
            lr_base *= 0.9
        elif no_improve > 150:
            T = max(T_min, T * 0.85)
            lr_base *= 0.85
        elif no_improve % 30 == 0 and no_improve < 50:
            T = min(T * 1.02, T_init)  # Slightly warm up

        # Random restart when very stuck
        if no_improve > 300:
            rng = np.random.default_rng((t_step * 7919 + 13) % (2**31))
            noise_scale = 0.001 + 0.003 * min(1.0, (no_improve - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init * 0.5
            lr_base = 0.02
            m = np.zeros(n)
            v = np.zeros(n)
            t_step = 0
            no_improve = 0

        if no_improve > 0 and no_improve % 100 == 0:
            print(f"    T={T:.6f}, lr={lr_base:.4f}, obj={best_obj:.8f}, no_improve={no_improve}")

    return best_a, best_obj


def qp_peak_balance(a0, time_limit, top_k=15, tol_frac=0.005):
    """
    Active-set QP: identify top peaks, build M = sum Q_k, solve min a^T M a s.t. simplex.
    Alternates with gradient steps for stability.
    """

    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Identify active peaks
        threshold = max_val * (1 - tol_frac)
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) > top_k * 3:
            # Take only top_k
            active_peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Build M (n x n) where M[i,j] encodes the sum of Q_k for active k
        # M[i,j] = 0.5 if i+j in active_set and i != j
        # M[i,i] = 1.0 if 2*i in active_set, 0 otherwise
        active_set = set(int(k) for k in active_peaks)
        M = np.zeros((n, n))
        for k in active_set:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if i == j:
                    M[i, i] += 1.0
                else:
                    M[i, j] += 0.5
                    M[j, i] += 0.5

        # Solve QP: min a^T M a s.t. sum(a) = 1, a >= 0
        a_qp = solve_qp_simplex(M, n, max_iter=500)

        if a_qp is not None:
            # Line search between a and a_qp
            best_alpha = 0.0
            best_val = max_val
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                candidate = (1 - alpha) * a + alpha * a_qp
                candidate = np.maximum(candidate, 0)
                s = candidate.sum()
                if s < 1e-15:
                    continue
                candidate = candidate / s
                val = fast_convolve(candidate).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

            if best_alpha > 0:
                a = (1 - best_alpha) * a + best_alpha * a_qp
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # If QP not improving, try a small gradient step
        if no_improve > 20:
            T = 0.001
            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()
            grad = conv_gradient(a, p)
            for lr in [0.01, 0.005, 0.002, 0.001]:
                candidate = project_simplex(a - lr * grad)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-14:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    no_improve = 0
                    break

        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_a, best_obj


def solve_qp_simplex(M, n, max_iter=500):
    """
    Solve min x^T M x s.t. sum(x) = 1, x >= 0 using active set method.
    Returns solution or None if failed.
    """
    active = list(range(n))
    inactive = set()

    for _ in range(max_iter):
        S = np.array(active)
        k = len(S)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            x = np.zeros(n)
            x[S[0]] = 1.0
            return x

        # Extract submatrix
        M_SS = M[np.ix_(S, S)]

        # KKT system: [2*M_SS, 1; 1^T, 0] [x_S; lam] = [0; 1]
        KKT = np.zeros((k + 1, k + 1))
        KKT[:k, :k] = 2.0 * M_SS
        KKT[:k, k] = 1.0
        KKT[k, :k] = 1.0
        rhs = np.zeros(k + 1)
        rhs[k] = 1.0

        try:
            sol = np.linalg.solve(KKT, rhs)
        except np.linalg.LinAlgError:
            # Add regularization
            KKT[:k, :k] += 1e-10 * np.eye(k)
            try:
                sol = np.linalg.solve(KKT, rhs)
            except np.linalg.LinAlgError:
                sol = np.linalg.lstsq(KKT, rhs, rcond=None)[0]

        x_S = sol[:k]
        lam = sol[k]

        # Check if all active variables are non-negative
        min_idx = np.argmin(x_S)
        if x_S[min_idx] < -1e-10:
            # Remove most negative variable from active set
            worst = int(S[min_idx])
            active.remove(worst)
            inactive.add(worst)
            continue

        # Clip small negatives
        x_S = np.maximum(x_S, 0.0)

        # Build full solution
        x_full = np.zeros(n)
        x_full[S] = x_S

        # Compute gradient at x_full: grad = 2*M*x_full
        grad_full = 2.0 * (M @ x_full)

        # Check KKT for inactive: grad[j] + lam >= 0 for all j in inactive
        if len(inactive) > 0:
            inactive_arr = np.array(list(inactive))
            kkt_inactive = grad_full[inactive_arr] + lam
            if np.min(kkt_inactive) < -1e-8:
                # Add most violated variable to active set
                worst_inact = inactive_arr[np.argmin(kkt_inactive)]
                inactive.remove(int(worst_inact))
                active.append(int(worst_inact))
                continue

        # KKT satisfied - optimal
        return x_full

    # Failed to converge; return best effort
    x = np.zeros(n)
    x[active] = 1.0 / len(active)
    return x


def multi_start_refine(a0, time_limit, n_starts=8):
    """Try multiple small perturbations and refine each briefly."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    per_start = time_limit / n_starts

    for start_idx in range(n_starts):
        if time.monotonic() - t0 > time_limit - 1:
            break

        if start_idx == 0:
            a_try = a.copy()
        else:
            rng = np.random.default_rng(start_idx * 104729 + 3)
            scale = 0.001 + 0.003 * start_idx / n_starts
            a_try = project_simplex(a + rng.standard_normal(n) * scale)

        t_alloc = min(per_start - (time.monotonic() - t0), per_start * 0.9)
        if t_alloc < 2:
            continue

        # Quick smoothed gradient
        a_ref, obj_ref = adaptive_smoothed_grad(a_try, time_limit=t_alloc, T_init=0.0005, T_min=0.00005)
        if obj_ref < best_obj - 1e-14:
            best_obj = obj_ref
            best_a = a_ref.copy()
            print(f"    Multi-start {start_idx}: improved to {best_obj:.8f}")

    return best_a, best_obj


def try_different_n(parent, time_limit):
    """Try a few different sequence lengths."""
    best_a = parent.copy()
    best_score = fast_convolve(parent).max() * 2 * len(parent) / (parent.sum() ** 2)

    n_parent = len(parent)
    candidates_n = []
    for n_try in [n_parent - 32, n_parent, n_parent + 32, n_parent + 64]:
        if 128 <= n_try <= 2048 and n_try not in candidates_n:
            candidates_n.append(n_try)

    t0 = time.monotonic()
    for n_try in candidates_n:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # Interpolate parent to new length
        a_new = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_parent), parent)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        t_alloc = min(remaining - 10, 60)
        if t_alloc < 5:
            continue

        a_opt, obj_opt = adaptive_smoothed_grad(a_new, time_limit=t_alloc, T_init=0.001)
        score = 2 * n_try * obj_opt / (a_opt.sum() ** 2)
        if score < best_score:
            best_score = score
            best_a = a_opt
            print(f"    n={n_try}: score={score:.8f}")

        print(f"  Different n exploration, t={t_alloc:.0f}s")
        a_opt, _ = try_different_n(best_seq if best_seq is not None else parent, time_limit=t_alloc)
        try_seq(a_opt, "diff_n")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Very fine smoothed gradient
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 60)
            a_opt, _ = adaptive_smoothed_grad(best_seq, time_limit=t_alloc, T_init=0.0001, T_min=0.00001)
            try_seq(a_opt, "final_smooth_fine")

        # QP with very tight tolerance
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = qp_peak_balance(best_seq, time_limit=t_alloc, top_k=5, tol_frac=0.001)
            try_seq(a_opt, "final_qp_tight")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            n_best = len(best_seq)
            for _ in range(5):
                remaining = deadline - time.monotonic() - 5
                if remaining < 2:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                a_pert = project_simplex(best_seq + noise)
                a_opt, _ = adaptive_smoothed_grad(a_pert, time_limit=min(remaining, 8), T_init=0.0002)
                try_seq(a_opt, "tiny_perturb")

    print(f"  Phase 5 best: {best_score:.8f}")

    # Final verification using evaluate_sequence
    if best_seq is None:
        best_seq = parent
        best_score = try_seq(best_seq, "fallback")

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
