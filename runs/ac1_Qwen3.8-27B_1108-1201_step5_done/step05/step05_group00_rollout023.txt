        p = np.exp(scaled)

            eta *= 0.999  # slowly decrease step size

IFFT(FFT(x) * conj(FFT(y)))[k] = sum_n x[n] * conj(y[(n-k) mod N]) (cross-correlation)

And IFFT(conj(FFT(x)) * FFT(y))[k] = sum_n conj(x[n]) * y[(n+k) mod N]

Since our sequences are real:
IFFT(conj(FFT(x)) * FFT(y))[k] = sum_n x[n] * y[(n+k) mod N]

So with x = a_padded, y = p_padded:
cross[k] = sum_n a[n] * p[(n+k) mod N]

For k in [0, n-1] and n+k < 2n-1 (which is true since n+k <= 2n-2 < N for N >= 2n-1):
cross[k] = sum_{n=0}^{n-1} a[n] * p[n+k]

This is exactly what we want! grad[j] = 2 * cross[j].

Great, so the FFT-based gradient computation works.

The idea: we want conv(a,a) to be flat. In the DFT domain:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast mirror descent (multiplicative weights) with O(n log n) per iteration
- Adaptive temperature and step size
- Frequency domain error reduction for spectral flattening
- Occasional LP oracle for global improvements
- Multi-n exploration
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_gradient(p, a):
    """Compute grad[j] = 2 * sum_m p[j+m] * a[m] using FFT.
    p has length 2n-1, a has length n. Returns array of length n.
    """
    n = len(a)
    fft_size = 2 * (2 * n - 1)  # large enough for linear correlation
    
    # Actually, we need the 'valid' correlation of p (len 2n-1) and a (len n)
    # Result length: (2n-1) - n + 1 = n
    # Use FFT: result = ifft(conj(FFT(a_padded)) * FFT(p_padded))[:n]
    
    pad_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, pad_size)
    P = np.fft.rfft(p, pad_size)
    
    # Cross-correlation: ifft(conj(A) * P)[k] = sum_n a[n] * p[n+k]
    # (for n+k < pad_size, which is guaranteed since n+k <= n-1+2n-2 = 3n-3 < pad_size for large enough pad)
    cross = np.fft.irfft(np.conj(A) * P, pad_size)
    return 2.0 * cross[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def mirror_descent(a0, time_limit, beta_init=5.0, eta_init=0.5, 
                   beta_growth=1.002, eta_decay=0.9995,
                   smooth_delta=0.001, verbose=False):
    """
    Fast mirror descent (exponentiated gradient) on the simplex.
    Minimizes max_k conv(a,a)[k] iteratively.
    
    Update rule: a_new[j] ∝ a[j] * exp(-eta * 2 * (p * a)[j])
    where p = softmax(beta * conv(a,a) / max(conv(a,a)))
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    beta = beta_init
    eta = eta_init
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        conv = fast_convolve(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Dual variables: softmax of scaled convolution
        scaled = beta * conv / max_c
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient via FFT
        grad = fast_gradient(p, a)

        # Mirror descent update in log-space
        # Add smoothing to prevent log(0)
        a_smooth = (1.0 - smooth_delta) * a + smooth_delta / n
        log_a = np.log(a_smooth)
        log_a -= eta * grad
        # Normalize in log-space
        log_max = log_a.max()
        log_a -= log_max
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt parameters
        if it % 50 == 0:
            eta *= eta_decay
            beta *= beta_growth

        # Periodic restart with noise
        if no_improve > 500:
            rng_noise = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise = rng_noise.standard_normal(n) * (0.001 + 0.002 * min(1.0, no_improve / 2000.0))
            a = project_simplex(best_a + noise)
            beta = beta_init
            eta = eta_init
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    MD it={it}, obj={best_obj:.10f}, beta={beta:.2f}, eta={eta:.6f}")

    return best_a, best_obj


def frequency_flatten(a0, time_limit, alpha_init=0.7, alpha_decay=0.999, verbose=False):
    """
    Frequency domain error reduction.
    Compresses the dynamic range of the power spectrum to flatten auto-convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    alpha = alpha_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        power = np.maximum(power, 1e-30)

        # Compress dynamic range
        mean_p = power.mean()
        new_power = power ** alpha * mean_p ** (1.0 - alpha)

        # Preserve phase, modify amplitude
        A_new = A / np.sqrt(power) * np.sqrt(new_power)

        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if it % 100 == 0:
            alpha = max(0.3, alpha * alpha_decay)

        if no_improve > 200:
            rng_noise = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng_noise.standard_normal(n) * 0.003)
            alpha = alpha_init
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    FF it={it}, obj={best_obj:.10f}, alpha={alpha:.4f}")

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 4.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_refine(a0, time_limit, radius=5, verbose=False):
    """
    Coordinate descent: for each element, try to shift mass to nearby elements
    to reduce the maximum of the auto-convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        improved = False
        for j in range(0, n, 2):  # Process every other element for speed
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            js = max(0, j - radius)
            je = min(n - 1, j + radius)
            
            # Try moving mass from j to neighbors
            for shift in range(1, radius + 1):
                for direction in [-1, 1]:
                    k = j + shift * direction
                    if k < 0 or k >= n:
                        continue
                    if a[k] < 1e-15:
                        continue
                    # Try moving a small fraction from j to k (or vice versa)
                    for frac in [0.1, 0.2, 0.5]:
                        a_test = a.copy()
                        amount = a[j] * frac
                        a_test[j] -= amount
                        a_test[k] += amount
                        if a_test[j] < 0:
                            continue
                        s = a_test.sum()
                        if s < 1e-15:
                            continue
                        a_test = a_test / s
                        obj = fast_convolve(a_test).max()
                        if obj < best_obj - 1e-12:
                            a = a_test
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and it > 5:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast mirror descent (multiplicative weights) with O(n log n) per iteration
- Adaptive temperature, step size, and momentum in log-space
- Frequency domain spectral compression for flattening auto-convolution
- Occasional LP oracle for global improvements
- Multi-n exploration with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_gradient(p, a):
    """Compute grad[j] = 2 * sum_m p[j+m] * a[m] using FFT.
    p has length 2n-1, a has length n. Returns array of length n.
    """
    n = len(a)
    pad_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, pad_size)
    P = np.fft.rfft(p, pad_size)
    cross = np.fft.irfft(np.conj(A) * P, pad_size)
    return 2.0 * cross[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def mirror_descent(a0, time_limit, beta_init=5.0, eta_init=0.5,
                   beta_growth=1.003, eta_decay=0.9995,
                   momentum_init=0.5, momentum_growth=0.0003,
                   smooth_delta=0.0005, verbose=False):
    """
    Fast mirror descent with momentum in log-space.
    Minimizes max_k conv(a,a)[k] on the probability simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    beta = beta_init
    eta = eta_init
    momentum = momentum_init
    v = np.zeros(n)  # momentum buffer in log-space

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        conv = fast_convolve(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Dual variables
        scaled = beta * conv / max_c
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = fast_gradient(p, a)

        # Momentum update in log-space
        v = momentum * v - eta * grad
        momentum = min(0.95, momentum + momentum_growth)

        # Mirror descent step
        a_smooth = (1.0 - smooth_delta) * a + smooth_delta / n
        log_a = np.log(a_smooth) + v
        log_max = log_a.max()
        log_a -= log_max
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt parameters
        if it % 20 == 0:
            eta *= eta_decay
            beta *= beta_growth

        # Reset when stuck
        if no_improve > 300:
            noise_scale = 0.002 + 0.003 * min(1.0, no_improve / 1500.0)
            rng_r = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale)
            beta = beta_init
            eta = eta_init
            momentum = momentum_init
            v = np.zeros(n)
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"    MD it={it}, obj={best_obj:.10f}, beta={beta:.1f}, eta={eta:.5f}")

    return best_a, best_obj


def frequency_flatten(a0, time_limit, alpha_init=0.75, alpha_decay=0.998, verbose=False):
    """
    Frequency domain spectral compression to flatten auto-convolution.
    Compresses the dynamic range of |A(omega)|^2 while preserving phase.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    alpha = alpha_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        power = np.maximum(power, 1e-30)

        mean_p = power.mean()
        new_power = power ** alpha * mean_p ** (1.0 - alpha)

        A_new = A / np.sqrt(power) * np.sqrt(new_power)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if it % 50 == 0:
            alpha = max(0.4, alpha * alpha_decay)

            rng_r = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng_r.standard_normal(n) * 0.003)

            no_improve = 0

    return best_a, best_obj

            m_exp = min(m_exp * 1.05, 3.0)

def generate_initializations(n, parent, rng):
    """Generate diverse initializations for the sequence."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.2
    inits.append(c1 / c1.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

            try_seq(a_opt2, "final_ff_low")

        # Quick perturbations + MD
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            for _ in range(3):
                remaining = deadline - time.monotonic() - 10
                if remaining < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_p = project_simplex(best_seq + noise)
                a_p, _ = mirror_descent(f_p, time_limit=min(remaining, 8),
                                        beta_init=12.0, eta_init=0.3)
                try_seq(a_p, "perturb")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
