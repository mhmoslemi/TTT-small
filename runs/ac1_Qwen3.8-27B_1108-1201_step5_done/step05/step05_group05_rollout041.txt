def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])

def dft_based_init(n, seed, alpha=1.0, mode="random"):
    """Generate sequence via DFT with controlled spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    half = fft_size // 2 + 1
    magnitude = np.ones(half)
    phase = rng.uniform(0, 2 * np.pi, half)

    if mode == "band":
        mask = rng.random(half) > 0.4
        magnitude[~mask] = 0.0
    elif mode == "tapered":
        freqs = np.linspace(0, 1, half)
        magnitude = np.exp(-alpha * freqs ** 2)

    spectrum = magnitude * np.exp(1j * phase)
    spectrum[0] = spectrum[0].real
    spectrum[-1] = spectrum[-1].real
    sig = np.fft.irfft(spectrum, fft_size)[:n]
    sig = np.maximum(sig, 0.0)
    s = sig.sum()
    if s < 1e-10:
        sig = np.abs(np.fft.irfft(spectrum, fft_size)[:n])
        s = sig.sum()
    if s > 1e-10:
        sig /= s
    else:
        sig = np.ones(n) / n
    return sig


def lbfw_smooth(a0, n, top_k_init=200, top_k_final=1, max_iters=300, time_limit=60.0):

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    while N_fft < conv_len + n - 1:

    def compute_grad(x):
        x_full = np.zeros(n, dtype=np.float64)
        x_full[1:] = x

        St = np.fft.rfft(S, N_fft)
        T = np.fft.irfft(Wt * np.conj(St), N_fft)[:n].real

                a = a / a.sum()
                conv = fast_convolve(a)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_a = a.copy()
                    improved = True
                else:
                    a[m] = 0.0
                    a = a / a.sum()
                continue

            rest = a.copy()
            rest[m] = 0.0
            rest_sum = rest.sum()
            if rest_sum < 1e-15:
                continue
            c = rest / rest_sum
            conv_bg = fast_convolve(rest)

            # Quadratic coefficients
            A = np.zeros(2 * n - 1)
            B = np.zeros(2 * n - 1)
            C = conv_bg.copy()

            lo = max(0, m)
            hi = min(2 * n - 2, m + n - 1)
            for k in range(lo, hi + 1):
                if k == 2 * m:
                    A[k] = 1.0
                else:
                    B[k] = 2.0 * c[k - m]

            def eval_t(t):
                vals = C + B * t
                vals[2 * m] += t * t
                return vals.max()

            # Golden section
            lo_t, hi_t = 1e-14, 1.0 - 1e-14
            phi = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_t - phi * (hi_t - lo_t)
            c2 = lo_t + phi * (hi_t - lo_t)
            f1 = eval_t(c1)
            f2 = eval_t(c2)

            for _ in range(55):
                if f1 < f2:
                    hi_t = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi_t - phi * (hi_t - lo_t)
                    f1 = eval_t(c1)
                else:
                    lo_t = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo_t + phi * (hi_t - lo_t)
                    f2 = eval_t(c2)

            t_opt = (lo_t + hi_t) / 2.0
            a[m] = t_opt
            rest2 = a.copy()
            rest2[m] = 0.0
            a = rest2 / rest2.sum() * (1.0 - t_opt / a[m]) if a[m] > 0 else a
            # Simpler renormalization:
            a[m] = t_opt * (1.0 - a[m]) / (1.0 - t_opt * a_sum_ratio) if False else 0
            # Let me just renormalize directly:
            pass

            # Proper renormalization
            a[m] = t_opt
            S_new = a.sum()
            if S_new < 1e-15:
                break
            a = a / S_new

            conv = fast_convolve(a)
            new_max = conv.max()
            if new_max < best_obj - 1e-13:
                best_obj = new_max
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def coord_desc_robust(a0, time_limit=60.0):
    """Robust coordinate descent with proper simplex handling."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        return np.ones(n) / n, 2.0
    a = a / S
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for m in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            am = a[m]

            # Build background (a with coordinate m zeroed)
            rest = a.copy()
            rest[m] = 0.0
            rest_sum = rest.sum()

            if rest_sum < 1e-14:
                # All mass at coordinate m; skip
                continue

            c = rest / rest_sum  # normalized rest, sum to 1
            conv_bg = fast_convolve(rest)  # = (rest_sum)^2 * fast_convolve(c)
            # Actually conv_bg[k] = sum_{i+j=k, i!=m, j!=m} rest[i]*rest[j]
            # And rest[j] = (1-am)*c[j], so conv_bg = (1-am)^2 * fast_convolve(c)
            # Let's just use conv_bg directly

            # For parameterization: a_m = t, a_j = (1-t)*c_j for j!=m
            # conv_k(t) = (1-t)^2 * C_k + 2*t*(1-t)*c_{k-m}*[valid, k!=2m] + t^2*[k=2m]
            # where C_k = sum_{i+j=k, i!=m, j!=m} c_i*c_j = conv(c)[k] minus terms with m
            # Since c[m] would be undefined (it's the variable), C_k = conv(c)[k] where c has a[m]=0
            conv_c = fast_convolve(c)

            # Coefficients for conv_k(t) as A_k*t^2 + B_k*t + Ck_k
            A = np.zeros(2 * n - 1)
            B = np.zeros(2 * n - 1)
            C = np.zeros(2 * n - 1)

            # C_k = conv_c[k] (since c[m] = 0 in our construction... wait, c has all n elements)
            # c = rest / rest_sum, where rest[m] = 0. So c[m] = 0.
            # conv_c[k] = sum_{i+j=k} c[i]*c[j] which already excludes m (since c[m]=0)
            C[:] = conv_c

            A[:] = 0.0
            B[:] = 0.0
            for k in range(max(0, m), min(2 * n - 1, m + n)):
                if k == 2 * m:
                    A[k] += 1.0
                else:
                    if 0 <= k - m <= n - 1 and c[k - m] > 0:
                        B[k] += 2.0 * c[k - m]

            def eval_t(t):
                vals = C + B * t + A * t * t
                return vals.max()

            # Golden section on [0, 1]
            lo_t, hi_t = 0.0, 1.0
            phi = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_t - phi * (hi_t - lo_t)
            c2 = lo_t + phi * (hi_t - lo_t)
            f1 = eval_t(c1)
            f2 = eval_t(c2)

            for _ in range(50):
                if f1 < f2:
                    hi_t = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi_t - phi * (hi_t - lo_t)
                    f1 = eval_t(c1)
                else:
                    lo_t = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo_t + phi * (hi_t - lo_t)
                    f2 = eval_t(c2)

            t_opt = (lo_t + hi_t) / 2.0

            # Update: a[m] = t_opt, a[j] = (1-t_opt)*c[j] for j!=m
            a_new = (1.0 - t_opt) * c
            a_new[m] = t_opt
            S_new = a_new.sum()  # Should be 1.0 by construction
            if abs(S_new - 1.0) > 1e-10:
                a_new = a_new / S_new

            conv_new = fast_convolve(a_new)
            new_max = conv_new.max()

            if new_max < best_obj - 1e-13:
                best_obj = new_max
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
            else:
                # Revert
                pass  # a unchanged

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def mirror_descent_polish(a0, time_limit=30.0, lr_init=0.005, top_k=20):
    """Mirror descent (exponential tilting) polish."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        return np.ones(n) / n
    a = a / S
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        w = np.zeros(len(conv))
        w[peaks] = 1.0 / k_act

        # Gradient: g[m] = 2 * sum_k w[k] * a[k-m] (valid k-m)
        conv_len = 2 * n - 1
        N_fft = 1
        while N_fft < conv_len + n - 1:
            N_fft <<= 1
        Wt = np.fft.rfft(w, N_fft)
        At = np.fft.rfft(a, N_fft)
        g = np.fft.irfft(Wt * np.conj(At), N_fft)[:n].real * 2.0

        # Mirror descent: a = exp(log(a) - lr * g) projected
        with np.errstate(divide='ignore'):
            log_a = np.where(a > 1e-30, np.log(a), -30.0)
        a_new = np.exp(log_a - lr * g)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / S_new

        conv = fast_convolve(a_new)
        new_max = conv.max()
        if new_max < best_obj - 1e-14:
            best_obj = new_max
            best_a = a_new.copy()
            a = a_new.copy()
            lr *= 0.999
        else:
            lr *= 0.998

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS (smooth), coordinate descent,
    mirror descent, and diverse DFT initializations.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n = len(parent)
    print(f"Working with n={n}")

    # === Phase 1: L-BFGS smooth optimization on parent ===
    print("\n=== Phase 1: L-BFGS on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(200, rem * 0.25)
    if t1 > 30:
        print(f"  t={t1:.0f}s, top_k 300->1")
        a_opt, m_opt = lbfw_smooth(parent, n, top_k_init=300, top_k_final=1,
                                    max_iters=200, time_limit=t1)
        try_seq(a_opt, "lbfw_parent")

    # === Phase 2: Coordinate descent on best ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(300, rem * 0.3)
        if t2 > 30:
            print(f"  t={t2:.0f}s")
            a_opt, m_opt = coord_desc_robust(best_seq, time_limit=t2)
            try_seq(a_opt, "cd_pass1")

        # Second CD pass
        rem = deadline - time.monotonic()
        if rem > 30:
            t2b = min(150, rem * 0.2)
            a_opt2, m_opt2 = coord_desc_robust(a_opt, time_limit=t2b)
            try_seq(a_opt2, "cd_pass2")

    # === Phase 3: Diverse DFT initializations ===
    print(f"\n=== Phase 3: Diverse Inits (best={best_score:.6f}) ===")
    inits = []
    # Parent variants
    inits.append(("parent", parent.copy()))
    inits.append(("parent_flipped", parent[::-1].copy()))
    inits.append(("parent_smooth", np.maximum(np.convolve(parent, np.ones(3)/3, 'same'), 0)))

    # DFT-based
    for i in range(6):
        mode = ["random", "band", "tapered"][i % 3]
        alpha = 1.0 + i * 0.5
        f = dft_based_init(n, seed=100 + i, alpha=alpha, mode=mode)
        # Mix with parent
        mix = 0.5 * parent + 0.5 * f
        mix = project_simplex(mix)
        inits.append((f"dft_{mode}_{i}", mix))

    for i, (name, f_init) in enumerate(inits):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem / max(1, len(inits) - i), 120)
        if t_alloc < 10:
            continue
        print(f"  [{name}] t={t_alloc:.0f}s")
        # L-BFGS
        a_opt, _ = lbfw_smooth(f_init, n, top_k_init=200, top_k_final=1,
                               max_iters=150, time_limit=t_alloc * 0.7)
        try_seq(a_opt, f"lbfw_{name}")
        # Quick CD
        rem = deadline - time.monotonic()
        if rem > 20:
            a_cd, _ = coord_desc_robust(a_opt, time_limit=min(rem * 0.3, 60))
            try_seq(a_cd, f"cd_{name}")

    # === Phase 4: Different lengths ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [640, 896, 1024]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            # Interpolate
            old_x = np.linspace(0, 1, len(best_seq))
            new_x = np.linspace(0, 1, target_n)
            f_interp = np.interp(new_x, old_x, best_seq)
            f_interp = np.maximum(f_interp, 0)
            ss = f_interp.sum()
            if ss > 1e-10:
                f_interp /= ss

            t_alloc = min(rem * 0.3, 150)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lbfw_smooth(f_interp, target_n, top_k_init=200,
                                    top_k_final=1, max_iters=150, time_limit=t_alloc * 0.6)
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cd, _ = coord_desc_robust(a_opt, time_limit=min(rem * 0.3, 60))
                try_seq(a_cd, f"multi_n{target_n}")
            else:
                try_seq(a_opt, f"lbfw_n{target_n}")

    # === Phase 5: Iterative refinement ===
    print(f"\n=== Phase 5: Iterative Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        for it in range(10):
            rem = deadline - time.monotonic()
            if rem < 30:
                break

            # Alternate L-BFGS and CD
            if it % 2 == 0:
                t_alloc = min(rem * 0.2, 80)
                a_opt, _ = lbfw_smooth(best_seq, n, top_k_init=100,
                                        top_k_final=1, max_iters=100,
                                        time_limit=t_alloc)
            else:
                t_alloc = min(rem * 0.2, 100)
                a_opt, _ = coord_desc_robust(best_seq, time_limit=t_alloc)

            score = try_seq(a_opt, f"iter_{it}")
            if score >= best_score - 1e-8:
                # No improvement, try perturbation
                noise = rng.standard_normal(n) * (0.002 + 0.001 * it)
                a_pert = project_simplex(best_seq + noise)
                rem = deadline - time.monotonic()
                if rem > 15:
                    a_opt2, _ = coord_desc_robust(a_pert, time_limit=min(rem * 0.3, 40))
                    try_seq(a_opt2, f"perturb_{it}")

    # === Phase 6: Mirror descent polish ===
    print(f"\n=== Phase 6: Mirror Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_md = min(rem * 0.2, 60)
            a_opt = mirror_descent_polish(best_seq, time_limit=t_md,
                                           lr_init=0.002, top_k=30)
            try_seq(a_opt, "mirror")
            # CD after mirror
            rem = deadline - time.monotonic()
            if rem > 15:
                a_cd, _ = coord_desc_robust(a_opt, time_limit=min(rem * 0.3, 40))
                try_seq(a_cd, "mirror_cd")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_final_cd = min(rem * 0.5, 200)
            print(f"  t={t_final_cd:.0f}s")
            a_opt, _ = coord_desc_robust(best_seq, time_limit=t_final_cd)
            try_seq(a_opt, "final_cd")

            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt2, _ = coord_desc_robust(a_opt, time_limit=min(rem * 0.4, 80))
                try_seq(a_opt2, "final_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
