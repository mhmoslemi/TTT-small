```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient_fgf(a, weights, peaks):
    """Compute weighted subgradient using FFT.
    
    a: sequence of length n
    weights: array of same length as peaks
    peaks: array of peak indices (positions in the convolution)
    
    Returns subgradient vector of length n.
    sg[m] = 2 * sum_i weights[i] * a[peaks[i] - m] (when 0 <= peaks[i]-m < n)
    """
    n = len(a)
    # Build weight vector
    L = 1 << (3*n - 2 - 1).bit_length()  # FFT size for linear cross-correlation
    W = np.zeros(L)
    for i, k in enumerate(peaks):
        W[k] = weights[i]
    A = np.zeros(L)
    A[:n] = a
    # Cross-correlation: corr[m] = sum_k W[k+m] * A[k]
    # Using FFT: corr = ifft(fft(W) * conj(fft(A)))
    corr = np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))).real
    return 2.0 * corr[:n]
```

**Algorithm: Adaptive Mirror Descent with FFT Subgradient**

```
Initialize a (from parent or cosine init), normalize to sum=1
Set temperature T = T_init, learning rate lr = lr_init
Track best_f, best_obj

For each iteration:
  1. Compute conv = fast_convolve(a)
  2. Compute objective = conv.max()
  3. Update best if improved
  4. Select peaks: use softmax weighting over top-k peaks with temperature T

  5. Compute subgradient sg using FFT
  6. Mirror descent update: a_new = a * exp(-lr * sg) / sum(a * exp(-lr * sg))
  7. Line search: try a few lr values, pick the one that minimizes the objective
  8. Update a, T, lr
```

Let me also think about the step size. For mirror descent on the simplex:
- The Bregman divergence is KL divergence
- The mirror map is exp(log(x)) = x (trivially)
- The update is: x_new = argmin_{y in simplex} <sg, y> + (1/lr) * KL(y || x)
  = argmin_{y in simplex} <sg, y> + (1/lr) * sum y_i * (log(y_i/x_i) - 1 + 1)
  = argmin_{y in simplex} <sg, y> + (1/lr) * sum y_i * log(y_i/x_i)

The solution to this is: y_i = x_i * exp(-lr * sg_i) / sum_j x_j * exp(-lr * sg_j)

This is the exponential weights / multiplicative weights update. ✓

For the step size: by theory, lr = sqrt(2 * log(n) / (L^2 * T)) where L is the Lipschitz constant and T is the total number of iterations. But in practice, I'll use a heuristic:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, weights, peaks):
    """Compute weighted subgradient of max conv using FFT cross-correlation.
    
    sg[m] = 2 * sum_i weights[i] * a[peaks[i] - m] (valid indices only)
    
    Parameters:
    a: array of length n
    weights: array of peak weights
    peaks: array of peak indices (integers in [0, 2n-2])
    
    Returns: subgradient array of length n
    """
    n = len(a)
    L = 1 << (3 * n - 1).bit_length()
    W = np.zeros(L)
    W[peaks] = weights
    A = np.zeros(L)
    A[:n] = a
    corr = np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))).real
    return 2.0 * corr[:n]

def softmax_peaks(conv, top_k, temperature):
    """Select top-k peaks with softmax weighting.
    
    Returns (weights, peak_indices)
    """
    k = min(top_k, len(conv))
    # Get top-k values and their indices
    idx = np.argpartition(conv, -k)[-k:]
    vals = conv[idx]
    # Softmax weighting
    scaled = vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    return w, idx

def mirror_descent_update(a, sg, lr):
    """Exponential weights update: a_new ∝ a * exp(-lr * sg)."""
    # Use log-space for numerical stability
    log_a = np.log(np.maximum(a, 1e-30))
    log_new = log_a - lr * sg
    # Subtract max for stability
    log_new -= log_new.max()
    new_a = np.exp(log_new)
    s = new_a.sum()
    if s < 1e-30:

    return new_a / s

def lp_fw_step(f, M, n):
    """Solve LP for Frank-Wolfe direction."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None
