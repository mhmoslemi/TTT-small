"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Natural gradient descent on softmax parameterization with annealed log-sum-exp
2. Active-set equalization via projected gradient with backtracking line search
3. Multi-scale coordinate descent (pairwise + group) for final polish
4. Structured initializations exploiting edge-concentration and symmetry
5. FFT-based efficient gradient computation exploiting shift structure
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, T, top_k=None):
    """
    Compute gradient of the log-sum-exp smoothed objective.
    F_smooth(a) = (1/T) * log(sum_k exp(T*(a*a)[k]))
    Returns (gradient, smoothed_value, peak_positions, weights)
    """
    n = len(a)
    conv = fast_convolve(a)
    if top_k is not None and top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))
    pv = conv[peaks]
    scaled = T * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    w = w / ws

    # Smoothed value
    smooth_val = conv[peaks].max() + np.log(ws) / T

    # Gradient: g[i] = sum_k w_k * 2*a[k-i] (if 0<=k-i<n)
    grad = np.zeros(n)
    for pk, wk in zip(peaks, w):
        k = int(pk)
        if wk < 1e-14:
            continue
        start = max(0, k - n + 1)
        end = min(n, k + 1)
        idx = k - np.arange(start, end)
        grad[start:end] += 2.0 * wk * a[idx]

    return grad, smooth_val, peaks, w


def natural_gradient_objective(x, a, T, top_k):
    """
    Compute the natural gradient (Fisher metric gradient) of the smoothed objective.
    a = softmax(x), so da_i/dx_j = a_i*(delta_ij - a_j)
    dF/dx_j = a_j * (dF/da_j - sum_i dF/da_i * a_i)
    """
    grad_a, smooth_val, peaks, w = compute_smoothed_gradient(a, T, top_k)
    mean_grad = np.dot(grad_a, a)
    nat_grad = a * (grad_a - mean_grad)
    return smooth_val, nat_grad


def adam_softplus_optimize(f0, time_limit, verbose=False):
    """
    Adam optimizer on log-sum-exp smoothed objective with softmax parameterization.
    Uses natural gradient for better conditioning.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-9
    base_lr = 0.02

    T_start = 8.0 / n
    T_end = 0.1 / n
    top_k_start = min(n, 128)
    top_k_end = min(n, 16)

    it = 0
    no_improve = 0
    max_iters = int(time_limit / 0.012)
    m = np.zeros(n)
    v = np.zeros(n)

    while it < max_iters:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        progress = min(1.0, (it - 1) / (max_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        top_k = max(4, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.4 * progress)

        smooth_val, nat_grad = natural_gradient_objective(None, f, T, top_k)

        it1 = it
        m = beta1 * m + (1 - beta1) * nat_grad
        v = beta2 * v + (1 - beta2) * nat_grad * nat_grad
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = f - step
        f_new = project_simplex(f_new)

        new_M = fast_convolve(f_new).max()

        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Softmax-Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 17 + 3)
                scale = 0.002 + 0.003 * min(no_improve // 100, 4)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                continue
            f = f_new

    return best_f, best_obj


def active_set_optimize(f0, time_limit, verbose=False):
    """
    Active-set method: identify top peaks, equalize them via projected gradient.
    Uses backtracking line search with Armijo condition.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    c_armijo = 1e-4
    rho = 0.5

    while time.monotonic() - t0 < time_limit:
        if no_improve > 150:
            break
        it += 1

        conv = fast_convolve(f)
        M = conv.max()

        # Identify active peaks (within 5% of max)
        active_thresh = M * 0.95
        active_peaks = np.where(conv >= active_thresh)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])
        # Limit to top 20
        if len(active_peaks) > 20:
            top_idx = np.argsort(conv[active_peaks])[-20:]
            active_peaks = active_peaks[top_idx]

        # Compute equalization gradient (average of active peak gradients)
        grad = np.zeros(n)
        for k in active_peaks:
            k = int(k)
            start = max(0, k - n + 1)
            end = min(n, k + 1)
            idx = k - np.arange(start, end)
            grad[start:end] += 2.0 * f[idx]
        grad /= len(active_peaks)

        # Remove mean component (stay on simplex)
        grad = grad - grad.mean()

        # Backtracking line search
        alpha = 0.5
        for _ls in range(20):
            f_trial = f - alpha * grad
            f_trial = project_simplex(f_trial)
            conv_trial = fast_convolve(f_trial)
            M_trial = conv_trial.max()

            if M_trial <= M - c_armijo * alpha * np.dot(grad, f - best_f) + 1e-15:
                break
            alpha *= rho

        else:
            alpha = max(alpha, 1e-8)
            f_trial = project_simplex(f - alpha * grad)
            M_trial = fast_convolve(f_trial).max()

        f = f_trial
        M = M_trial

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    ActiveSet iter {it}: obj={best_obj:.10f}, nactives={len(active_peaks)}")
        else:
            no_improve += 1

        # Periodic restart
        if no_improve > 60:
            rng = np.random.default_rng(it * 23 + 11)
            scale = 0.001 + 0.002 * min(no_improve // 30, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_optimize(f0, time_limit, verbose=False):
    """
    Multi-scale coordinate descent: pairwise optimization at multiple offsets.
    Uses ternary search for 1D subproblems.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    # Multiple offset scales for diverse pairings
    offset_sets = [
        [1, 2, 3, 5, 8, 13, 21],
        [7, 15, 31, 63, 127, 255],
        [2, 4, 8, 16, 32, 64, 128],
        [3, 9, 27, 81, 243],
    ]

    pass_num = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve_passes > 6:
            break
        pass_num += 1
        offset_set = offset_sets[pass_num % len(offset_sets)]
        pass_improved = False

        order = rng.permutation(n) if pass_num % 2 == 0 else np.arange(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx_i]
            if f[i] < 1e-12:
                continue

            for offset in offset_set:
                j_candidate = order[idx_i + offset] if idx_i + offset < n else order[(idx_i + offset) % n]
                j = j_candidate
                if i == j:
                    continue

                c_total = f[i] + f[j]
                if c_total < 1e-12:
                    continue

                # Ternary search on t = f[i] / c_total
                t_lo, t_hi = 0.0, 1.0
                best_t = f[i] / c_total
                best_val = fast_convolve(f).max()

                for _ in range(10):
                    t1 = t_lo + (t_hi - t_lo) / 3
                    t2 = t_hi - (t_hi - t_lo) / 3

                    f1 = f.copy()
                    f1[i] = t1 * c_total
                    f1[j] = (1 - t1) * c_total
                    f1 = f1 / f1.sum()
                    v1 = fast_convolve(f1).max()

                    f2 = f.copy()
                    f2[i] = t2 * c_total
                    f2[j] = (1 - t2) * c_total
                    f2 = f2 / f2.sum()
                    v2 = fast_convolve(f2).max()

                    if v1 < v2:
                        t_hi = t2
                        if v1 < best_val:
                            best_val = v1
                            best_t = t1
                    else:
                        t_lo = t1
                        if v2 < best_val:
                            best_val = v2
                            best_t = t2

                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c_total
                    f[j] = (1 - best_t) * c_total
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
            if verbose and pass_num % 5 == 0:
                print(f"    CoordDescent pass {pass_num}: obj={best_obj:.10f}")
        else:
            no_improve_passes += 1

    return best_f, best_obj

            no_improve += 1
            continue

        # Try several random directions within the group simplex
        improved = False
        for _trial in range(5):
            direction = rng.standard_normal(g)
            direction -= direction.mean()  # zero mean to preserve sum
            if np.abs(direction).max() < 1e-10:
                continue

            # Line search along this direction
            best_alpha = 0.0
            best_trial_val = fast_convolve(f).max()

            for alpha in [0.01, 0.02, 0.05, 0.1, 0.2]:
                trial_vals = group_vals + alpha * direction * total
                if (trial_vals < -1e-10).any():
                    continue
                trial_vals = np.maximum(trial_vals, 0)
                vs = trial_vals.sum()
                if vs < 1e-12:
                    continue
                trial_vals = trial_vals * (total / vs)

                f_trial = f.copy()
                f_trial[indices] = trial_vals
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()

                if val < best_trial_val - 1e-14:
                    best_trial_val = val
                    best_alpha = alpha
                    best_trial_vals = trial_vals.copy()

            if best_alpha > 0:
                f[indices] = best_trial_vals
                f = f / f.sum()
                if best_trial_val < best_obj - 1e-14:
                    best_obj = best_trial_val
                    best_f = f.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 100 == 0 and it < 500:
            print(f"    GroupDescent iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def structured_init(n, variant=0, rng_seed=42):
    """Generate structured initializations."""
    rng = np.random.default_rng(rng_seed)

    if variant == 0:
        # Uniform
        return np.ones(n) / n
    elif variant == 1:
        # Edge-concentrated (cosine window inverted)
        x = np.arange(n) / (n - 1)
        a = 1.0 - np.cos(np.pi * x)  # high at edges, low at center
        a = np.maximum(a, 0.0)
        return a / a.sum()
    elif variant == 2:
        # Two peaks at 1/4 and 3/4
        x = np.arange(n) / (n - 1)
        a = np.exp(-n * (x - 0.25) ** 2) + np.exp(-n * (x - 0.75) ** 2)
        return a / a.sum()
    elif variant == 3:
        # Periodic-like: high at multiples of n/3
        x = np.arange(n) / (n - 1)
        periods = [0.0, 1.0/3.0, 2.0/3.0]
        a = np.zeros(n)
        for p in periods:
            a += np.exp(-n * 2 * (x - p) ** 2)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif variant == 4:
        # Random with edge bias
        x = np.arange(n) / (n - 1)
        bias = 1.0 + 2.0 * (x - 0.5) ** 2 / 0.25
        a = rng.random(n) * bias
        return a / a.sum()
    elif variant == 5:
        # Anti-symmetric perturbation of uniform
        x = np.arange(n) / (n - 1)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif variant == 6:
        # Triangular (peak at center)
        x = np.arange(n) / (n - 1)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining natural gradient, active set, and coordinate methods.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = a / max(a.sum(), 1e-10)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick structured initialization exploration ===
    print(f"\n=== Phase 1: Structured Inits (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        for variant in range(7):
            if time.monotonic() > deadline - 600:
                break
            a_init = structured_init(n_parent, variant=variant, rng_seed=seed + variant)
            sc = float(evaluate_sequence(a_init.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_init.copy()
                print(f"  Structured init {variant} at n={n_parent}: {sc:.8f}")

    # === Phase 2: Natural gradient on parent ===
    print(f"\n=== Phase 2: Natural Gradient on parent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t2 = min(rem, 180)
        print(f"  Time: {t2:.0f}s, n={n_parent}")
        a_opt, _ = adam_softplus_optimize(best_seq, time_limit=t2, verbose=True)
        try_seq(a_opt, "natgrad_p2")

    # === Phase 3: Active set optimization ===
    print(f"\n=== Phase 3: Active Set (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t3 = min(rem, 150)
        print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
        a_opt, _ = active_set_optimize(best_seq, time_limit=t3, verbose=True)
        try_seq(a_opt, "active_p3")

    # === Phase 4: Natural gradient on best ===
    print(f"\n=== Phase 4: Natural Gradient on best (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t4 = min(rem, 120)
        a_opt, _ = adam_softplus_optimize(best_seq, time_limit=t4, verbose=False)
        try_seq(a_opt, "natgrad_p4")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    for target_n in [512, 768, 1024, 1536]:
        if time.monotonic() > deadline - 200:
            break
        if target_n == len(base):
            continue
        f_interp = interpolate_sequence(base, target_n)

        rem = deadline - time.monotonic() - 150
        if rem < 20:
            break

        # Natural gradient
        t_alloc = min(rem * 0.4, 80)
        if t_alloc > 10:
            a_opt_n, _ = adam_softplus_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt_n, f"natgrad_n{target_n}")

        # Active set
        rem = deadline - time.monotonic() - 120
        if rem > 10:
            t_as = min(rem * 0.5, 60)
            a_opt_n, _ = active_set_optimize(a_opt_n, time_limit=t_as, verbose=False)
            try_seq(a_opt_n, f"active_n{target_n}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts (best={best_score:.8f}) ===")
    n_best = len(best_seq)

    restart_configs = [
        ("noise_small", lambda: project_simplex(best_seq + np.random.default_rng(777).standard_normal(n_best) * 0.003)),
        ("reverse", lambda: best_seq[::-1] / best_seq[::1].sum()),
        ("blend_uniform", lambda: project_simplex(0.8 * best_seq + 0.2 * np.ones(n_best) / n_best)),
        ("noise_med", lambda: project_simplex(best_seq + np.random.default_rng(888).standard_normal(n_best) * 0.008)),
        ("edge_boost", lambda: _edge_boost_restart(best_seq)),
    ]

    for name, gen_fn in restart_configs:
        if time.monotonic() > deadline - 100:
            break
        rem = deadline - time.monotonic() - 80
        if rem < 10:
            break
        t_alloc = min(rem / 2, 60)
        f_init = gen_fn()
        # Alternate between methods
        if hash(name) % 2 == 0:
            a_opt, _ = adam_softplus_optimize(f_init, time_limit=t_alloc, verbose=False)
        else:
            a_opt, _ = active_set_optimize(f_init, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"restart_{name}")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t7 = min(rem * 0.7, 150)
        print(f"  Time: {t7:.0f}s, n={len(best_seq)}")
        a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=t7, verbose=True)
        try_seq(a_opt, "coord_p7")

        # Natural gradient after coord descent
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = adam_softplus_optimize(a_opt, time_limit=min(rem, 40))
            try_seq(a_opt, "natgrad_after_coord")

    # === Phase 8: Group descent polish ===
    print(f"\n=== Phase 8: Group Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t8 = min(rem * 0.6, 80)
        gs = max(3, min(8, len(best_seq) // 200))
        a_opt, _ = group_descent_optimize(best_seq, time_limit=t8, group_size=gs, verbose=False)
        try_seq(a_opt, "group_p8")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine coordinate descent
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=min(rem, 40), verbose=True)
            try_seq(a_opt, "coord_final")

        # Tiny perturbations + quick optimization
        for trial in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 5:
                break
            rng = np.random.default_rng(900 + trial)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
            a_opt, _ = active_set_optimize(f_pert, time_limit=min(rem, 15))
            try_seq(a_opt, f"perturb_final_{trial}")

        # One more coordinate pass
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=min(rem, 10))
            try_seq(a_opt, "coord_very_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def _edge_boost_restart(f):
    """Create a restart that boosts the edges relative to the center."""
    n = len(f)
    x = np.arange(n) / (n - 1)
    edge_weight = 1.0 + 0.5 * (2 * x - 1) ** 2
    boosted = f * edge_weight
    return boosted / boosted.sum()
