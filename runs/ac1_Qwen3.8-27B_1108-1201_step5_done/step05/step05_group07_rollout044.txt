def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with log-sum-exp continuation (multi-beta annealing)
2. Multiplicative-weights subgradient for high-temperature final refinement  
3. Systematic coordinate descent for final polish
4. Multi-length and multi-initialization exploration
Key innovation: Second-order optimization (L-BFGS-B) via softmax parameterization
converges dramatically faster than the first-order subgradient method used previously.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv_gradient(a, conv, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*conv))) wrt a."""
    n = len(a)
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    e = np.exp(shifted)
    w = e / e.sum()
    
    grad = np.zeros(n)
    active = np.nonzero(w > 1e-14)[0]
    for l in active:
        l = int(l)
        js = max(0, l - n + 1)
        je = min(n - 1, l)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[l] * a[l - idx]
    return grad

def lbgfbs_run(a_init, beta, max_iter=5000, gtol=1e-9, ftol=1e-13):
    """Run L-BFGS-B with softmax parameterization at given beta."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    z = np.log(a) - np.log(a).mean()
    
    def obj_grad(z):
        a_local = softmax(z)
        conv = fast_convolve(a_local)
        grad_a = compute_conv_gradient(a_local, conv, beta)
        mean_g = np.dot(a_local, grad_a)
        grad_z = a_local * (grad_a - mean_g)
        
        max_c = conv.max()
        shifted = beta * (conv - max_c)
        e2 = np.exp(shifted)
        obj_val = 2.0 * n * (max_c + np.log(e2.sum())) / beta
        return obj_val, grad_z
    
    result = scipy_minimize(obj_grad, z, method='L-BFGS-B', jac=True,
                            options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                     'gtol': gtol, 'ftol': ftol, 'iprint': 0})
    a_out = softmax(result.x)
    return a_out, result.fun, result.iter, result.success

def subgradient_optimize(f0, time_limit, top_k=50, temperature=0.00005, lr=0.003,
                         momentum=0.85):
    """Multiplicative-weights subgradient optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn
        
        conv_nest = fast_convolve(f_nest)
        k_actual = min(top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]
        
        v = momentum * v + (1 - momentum) * sg
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            no_improve = 0
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9998
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_temp *= 1.005
        
        if no_improve > 500:
            rng = np.random.default_rng((it * 31 + 7) % 1000000)
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit):
    """Coordinate descent optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)
    if mid - d >= 0:
        order.append(mid - d)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()
            
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

def pair_exchange(f0, time_limit, n_trials_per_round=500):
    """Pair exchange local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        top_k = min(20, 2*n-1)
        top_peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        rng = np.random.default_rng(int(time.monotonic() * 1e6) % 10000000)
        
        for _ in range(n_trials_per_round):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-10:
                continue
            for frac in [0.1, 0.3, 0.5]:
                amt = f[i] * frac
                f[i] -= amt
                f[j] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved
