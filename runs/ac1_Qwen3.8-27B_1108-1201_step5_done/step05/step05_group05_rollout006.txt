from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A, fft_size).real[:size]

def smoothed_obj_and_grad(x, t, n):
    """
    Compute smoothed objective and gradient.
    x: parameters (length n), a = softmax(x)
    t: smoothing parameter
    """
    # Compute a = softmax(x)
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    
    # Compute convolution
    conv = fast_convolve(a)
    
    # Compute log-sum-exp
    conv_scaled = conv / t
    conv_max = conv_scaled.max()
    log_Z = conv_max + np.log(np.sum(np.exp(conv_scaled - conv_max)))
    obj = t * log_Z
    
    # Compute gradient
    w = np.exp(conv_scaled - conv_max)  # = exp(conv/t) / exp(conv_max/t)
    Z_scaled = w.sum()  # = Z / exp(conv_max/t)
    
    # grad_a[i] = 2 * sum_j a[j] * w[i+j] / Z_scaled
    # (the exp(conv_max/t) cancels)
    grad_a = 2.0 * np.correlate(w, a, mode='valid') / Z_scaled
    
    # grad_x[i] = grad_a[i] * a[i] - a[i] * sum_j grad_a[j] * a[j]
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)
    
    return obj, grad_x

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A, fft_size).real[:size]


def compute_grad_a(a, w):
    """Compute grad_a[i] = sum_j a[j] * w[i+j] via FFT."""
    n = len(a)
    # We want: g[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1
    # This is cross-correlation of a and w
    N = 1 << (2 * n - 1 - 1).bit_length()  # Need N >= (2n-1) + n - 1 = 3n-2

def obj_grad(x, t, n):
    """Objective and gradient for L-BFGS."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    
    conv = fast_convolve(a)

    cmax = conv_scaled.max()
    exp_vals = np.exp(conv_scaled - cmax)
    Z = exp_vals.sum()

    w = exp_vals  # normalized weights (sum to Z, not 1)
    grad_a = 2.0 * np.correlate(w, a, mode='valid') / Z
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)

    conv = fast_convolve(a)
    conv_scaled = conv / t
    cmax = conv_scaled.max()
    exp_vals = np.exp(conv_scaled - cmax)
    Z = exp_vals.sum()
    obj = t * (cmax + np.log(Z))
    w = exp_vals
    grad_a = 2.0 * np.correlate(w, a, mode='valid') / Z
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)
    return obj, grad_x


def _obj_wrapper(x, t, n):
    return _compute_obj_grad(x, t, n)[0]


def _grad_wrapper(x, t, n):
    return _compute_obj_grad(x, t, n)[1]


def true_objective(a):
    """Compute the true (unsmoothed) objective for a sequence a."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def lbfgs_continue(x_init, t_start, t_end, time_limit, n, verbose=False):
    """
    Run L-BFGS with geometric continuation on t (smoothing parameter).
    Returns (best_x, best_smooth_obj).
    """
    best_x = x_init.copy()
    _, best_obj = _compute_obj_grad(best_x, t_start, n)
    t0 = time.monotonic()

    n_levels = 25
    t_schedule = np.geomspace(t_start, t_end, n_levels)

    for level, t_val in enumerate(t_schedule):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        max_iter = int(max(5, min(100, remaining * 8)))

        try:
            res = minimize(
                fun=_obj_wrapper,
                x0=best_x,
                jac=_grad_wrapper,
                method='L-BFGS-B',
                args=(t_val, n),
                options={'maxiter': max_iter, 'maxls': 50,
                         'ftol': 1e-16, 'gtol': 1e-10, 'iprint': 0}
            )
            if res.fun < best_obj:
                best_obj = res.fun
                best_x = res.x.copy()
        except Exception:
            pass

        if verbose and (level % 5 == 0 or level == n_levels - 1):
            elapsed = time.monotonic() - t0
            a = softmax(best_x)
            conv = fast_convolve(a)
            print(f"    Level {level}/24: t={t_val:.2e}, smooth={best_obj:.8f}, "
                  f"true_max={conv.max():.8f}, elapsed={elapsed:.1f}s")

    return best_x, best_obj


def lbfgs_refine(a_init, time_limit, n, t_div_start=10.0, t_div_end=5000.0, verbose=False):
    """
    Full L-BFGS continuation refine starting from sequence a_init (sum=1, non-neg).
    """
    conv0 = fast_convolve(a_init)
    t_start = conv0.max() / t_div_start
    t_end = conv0.max() / t_div_end
    x0 = np.log(a_init + 1e-15)
    x_opt, _ = lbfgs_continue(x0, t_start, t_end, time_limit=time_limit, n=n, verbose=verbose)
    return softmax(x_opt)


def coord_desc_refine(a, time_limit, multipliers=None):
    """Coordinate descent refinement on simplex."""
    if multipliers is None:
        multipliers = [0.5, 0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.7, 2.0, 2.5, 3.0]

    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = true_objective(a)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-12:
                continue

            best_j_val = a[j]
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = a[j] * mult
                if new_val < 0 or (1.0 - new_val) <= 1e-10:
                    continue
                denom = 1.0 - a[j]
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[j] = new_val
                trial[mask] = a[mask] * scale
                trial_sum = trial.sum()
                if trial_sum < 1e-15:
                    continue
                trial = trial / trial_sum

                obj = true_objective(trial)
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != a[j]:
                denom = 1.0 - a[j]
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()

                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0

    return best_a


def init_from_parent(parent, target_n):
    """Interpolate parent sequence to target_n length."""
    old_n = len(parent)
    if old_n == target_n:
        a = parent.copy()
    else:
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, target_n)
        a = np.interp(new_x, old_x, parent)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def init_flat_spectrum(n, seed=0):
    """Initialize with flat power spectrum (random phases)."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    n_half = fft_size // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, n_half)
    spectrum = np.exp(1j * phases)
    spec_full = np.zeros(fft_size, dtype=complex)
    spec_full[:n_half] = spectrum
    spec_full[n_half:] = np.conj(spectrum[1:n_half][::-1])
    sig = ifft(spec_full, fft_size).real
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    s = sig.sum()
    if s < 1e-10:
        sig = np.abs(sig)
        s = sig.sum()
    if s > 1e-10:
        sig /= s
    else:
        sig = np.ones(n) / n
    return sig


def init_random(n, seed=0):
    """Random exponential initialization on simplex."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    a /= a.sum()
    return a


def init_gaussian(n, center=0.5, sigma=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a /= a.sum()
    return a


def init_bell(n, width=0.4):
    """Raised cosine bell initialization."""
    x = np.arange(n) / n
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5) / width))
    a[x < 0.5 - width / 2] = 0
    a[x > 0.5 + width / 2] = 0
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS continuation on parent
    Phase 2: Refine best with tighter smoothing
    Phase 3: Diverse initializations (flat spectrum, random, gaussian, etc.)
    Phase 4: Multi-length exploration
    Phase 5: Coordinate descent refinement
    Phase 6: Final polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_a = None
    best_score = float('inf')

    # Helper to try a sequence (inline to avoid closure issues)
    def _try(a, label):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("Using default parent (uniform 512)")

    _try(parent, "parent_init")

    # ============ Phase 1: L-BFGS on parent ============
    print("\n=== Phase 1: L-BFGS on parent ===")
    n_p1 = len(parent)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400)
    print(f"  n={n_p1}, time={t_alloc:.0f}s")
    a_opt = lbfgs_refine(parent, time_limit=t_alloc, n=n_p1,
                         t_div_start=8.0, t_div_end=3000.0, verbose=True)
    _try(a_opt, "lbfgs_parent")

    # ============ Phase 2: Tighter refinement ============
    print(f"\n=== Phase 2: Tighter Refinement (score={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 300)
        n_p2 = len(best_a)
        print(f"  n={n_p2}, time={t_alloc:.0f}s")
        a_opt = lbfgs_refine(best_a, time_limit=t_alloc, n=n_p2,
                             t_div_start=20.0, t_div_end=10000.0, verbose=True)
        _try(a_opt, "lbfgs_tight")

    # ============ Phase 3: Diverse initializations ============
    print(f"\n=== Phase 3: Diverse Initializations ===")
    n_base = len(best_a) if best_a is not None else n_p1
    base_a = best_a if best_a is not None else parent

    inits = []
    inits.append(("flat_spec", init_flat_spectrum(n_base, seed=100)))
    inits.append(("random", init_random(n_base, seed=200)))
    inits.append(("gaussian", init_gaussian(n_base, 0.5, 0.35)))
    inits.append(("bell", init_bell(n_base, 0.5)))
    inits.append(("blend_flat", 0.6 * base_a + 0.4 * init_flat_spectrum(n_base, seed=300)))
    inits.append(("blend_gauss", 0.6 * base_a + 0.4 * init_gaussian(n_base, 0.5, 0.25)))
    inits.append(("blend_rand", 0.7 * base_a + 0.3 * init_random(n_base, seed=400)))

    for name, a_init in inits:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        a_init = np.maximum(a_init, 0)
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_init = a_init / s
        t_alloc = min(rem * 0.1, 150)
        print(f"  [{name}] time={t_alloc:.0f}s")
        a_opt = lbfgs_refine(a_init, time_limit=t_alloc, n=n_base,
                             t_div_start=12.0, t_div_end=4000.0, verbose=False)
        _try(a_opt, f"init_{name}")

    # ============ Phase 4: Multi-length exploration ============
    print(f"\n=== Phase 4: Multi-Length (score={best_score:.6f}) ===")
    for target_n in [384, 512, 640, 1024, 1280, 1600]:
        if target_n == len(base_a):
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        a_interp = init_from_parent(base_a, target_n)
        t_alloc = min(rem * 0.1, 120)
        print(f"  n={target_n}, time={t_alloc:.0f}s")
        a_opt = lbfgs_refine(a_interp, time_limit=t_alloc, n=target_n,
                             t_div_start=10.0, t_div_end=4000.0, verbose=False)
        _try(a_opt, f"len_{target_n}")

    # ============ Phase 5: Coordinate descent ============
    print(f"\n=== Phase 5: CD Refinement (score={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.25, 180)
            print(f"  CD time={t_cd:.0f}s")
            a_cd = coord_desc_refine(best_a, time_limit=t_cd)
            _try(a_cd, "cd")

            rem = deadline - time.monotonic()
            if rem > 20:
                t_cd2 = min(rem * 0.15, 100)
                a_cd2 = coord_desc_refine(a_cd, time_limit=t_cd2)
                _try(a_cd2, "cd2")

    # ============ Phase 6: Perturbation + re-optimize ============
    print(f"\n=== Phase 6: Perturbations (score={best_score:.6f}) ===")
    if best_a is not None:
        rng = np.random.default_rng(seed + 999)
        n_best = len(best_a)
        for i in range(5):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            noise_scale = 0.002 + 0.003 * i
            a_pert = best_a + rng.standard_normal(n_best) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s < 1e-10:
                continue
            a_pert = a_pert / s
            t_alloc = min(rem * 0.12, 80)
            a_opt = lbfgs_refine(a_pert, time_limit=t_alloc, n=n_best,
                                 t_div_start=15.0, t_div_end=5000.0, verbose=False)
            _try(a_opt, f"perturb_{i}")

    # ============ Phase 7: Final polish ============
    print(f"\n=== Phase 7: Final Polish (score={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_final = min(rem * 0.4, 80)
            a_f = lbfgs_refine(best_a, time_limit=t_final, n=len(best_a),
                               t_div_start=50.0, t_div_end=20000.0, verbose=False)
            _try(a_f, "final_lbfgs")

        rem = deadline - time.monotonic()
        if rem > 10 and best_a is not None:
            t_cd = min(rem * 0.3, 30)
            a_f = coord_desc_refine(best_a, time_limit=t_cd)
            _try(a_f, "final_cd")

    # Fallback
    if best_a is None:
        best_a = np.ones(256) / 256

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
