    w_max = w.max()
    a = np.exp(w - w_max)
    a = a / a.sum()
    
    conv = fast_convolve(a)

    weights = weights / weights.sum()
    
    # Gradient of log-sum-exp wrt a[m]:
    # d/da[m] log(sum_k exp(conv[k]/eps)) = (1/eps) * sum_k weights[k] * 2*a[k-m] * [0<=k-m<n]

        grad_a[m_lo:m_hi+1] += (2.0 / epsilon) * weights[k] * a[k - np.arange(m_lo, m_hi+1)]
    
    # Chain rule: a = softmax(w), so da[i]/dw[j] = a[i]*(delta[ij] - a[j])

    grad_a_weighted = grad_a * a
    grad_w = grad_a_weighted - a * grad_a_weighted.sum()
    
    return grad_w

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed objective with L-BFGS-B (softmax parameterization)
2. Temperature annealing
3. LP-FW refinement

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def structured_init(n, pattern='bump', seed=0):
    """Generate various structured initializations."""
    j = np.arange(n)
    rng = np.random.default_rng(seed)
    
    if pattern == 'uniform':
        a = np.ones(n)
    elif pattern == 'bump':
        # Gaussian bump in the middle
        center = n / 2
        width = n / 4
        a = np.exp(-0.5 * ((j - center) / width) ** 2)
    elif pattern == 'edges':
        # Concentrated at edges
        center = n / 2
        a = np.exp(-0.5 * ((j - center) / (n/8)) ** 2)
        a = a + 0.5 * np.exp(-0.5 * ((j - 0) / (n/16)) ** 2)
        a = a + 0.5 * np.exp(-0.5 * ((j - (n-1)) / (n/16)) ** 2)
    elif pattern == 'sine':
        a = 1.0 + 0.3 * np.sin(2 * np.pi * j / n)
    elif pattern == 'sine2':
        a = 1.0 + 0.3 * np.sin(4 * np.pi * j / n)
    elif pattern == 'random_smooth':
        # Random smooth via low-pass filter
        a = rng.random(n)
        # Apply simple moving average
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif pattern == 'two_bumps':
        c1, c2 = n//3, 2*n//3
        a = np.exp(-0.5*((j-c1)/(n/10))**2) + np.exp(-0.5*((j-c2)/(n/10))**2) + 0.1
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def smoothed_objective_and_grad(w, epsilon, n):
    """
    Compute the smoothed objective and its gradient.
    w: parameter vector (length n), a = softmax(w)
    Returns: (objective_value, gradient_wrt_w)
    """
    # Numerically stable softmax
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    a = exp_w / exp_w.sum()
    
    # Compute convolution
    conv = fast_convolve(a)
    
    # Log-sum-exp
    conv_scaled = conv / epsilon
    conv_max = conv_max = conv_scaled.max()
    log_sum_exp = conv_max + np.log(np.sum(np.exp(conv_scaled - conv_max)))
    
    obj = 2.0 * n * epsilon * log_sum_exp
    
    # Softmax weights over convolution
    softmax_conv = np.exp(conv_scaled - conv_max)
    softmax_conv = softmax_conv / softmax_conv.sum()
    
    # Gradient of smoothed max wrt a:
    # grad_a[m] = (2*n/epsilon) * sum_k softmax_conv[k] * 2 * a[k-m] * [0<=k-m<n]
    #           = (4*n/epsilon) * sum_k softmax_conv[k] * a[k-m] * [0<=k-m<n]
    #
    # This is a cross-correlation. Define w_conv[k] = (4*n/epsilon) * softmax_conv[k]
    # Then grad_a[m] = sum_k w_conv[k] * a[k-m] for valid k-m
    #
    # Let's compute this via FFT.
    # grad_a[m] = sum_{k} w_conv[k] * a[k-m] where 0<=k-m<n and 0<=k<2n-1
    # = sum_{j=0}^{n-1} w_conv[m+j] * a[j] (where w_conv is defined for 0..2n-2)
    
    w_conv = (4.0 * n / epsilon) * softmax_conv  # length 2n-1
    # We need: grad_a[m] = sum_{j=0}^{n-1} w_conv[m+j] * a[j]
    # This is the cross-correlation of w_conv and a, evaluated at lags 0..n-1
    
    # Using FFT: cross-correlation can be computed as ifft(conj(fft(a)) * fft(w_conv))
    # But we need to be careful about the definition.
    # Actually: (w_conv ⊛ a)[m] = sum_j w_conv[m+j] * a[j]
    # This is the "unconventional" cross-correlation.
    # Standard cross-correlation: (a ⋆ w_conv)[m] = sum_j a[j] * w_conv[j+m]
    # Which is the same thing!
    # 
    # To compute via FFT: 
    # Let's pad both to length >= 3n-2 (to avoid circular effects)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_conv, fft_size)
    # Cross-correlation: ifft(conj(A) * W) gives sum_j a[j] * w_conv[j+m]
    # Wait, let me think again.
    # If A = fft(a) and W = fft(w_conv), then ifft(A * conj(W)) gives circular cross-correlation.
    # Specifically: ifft(A * conj(W))[m] = sum_j a[j] * w_conv[(j-m) mod fft_size]
    # That's not quite what we want.
    #
    # We want: grad_a[m] = sum_{j=0}^{n-1} w_conv[m+j] * a[j]
    # Let's define b[j] = a[n-1-j] (reversed a).
    # Then: sum_j w_conv[m+j] * a[j] = sum_j w_conv[m+j] * b[n-1-j]
    # = (w_conv * b)[m + n - 1] (convolution)
    # where * denotes linear convolution.
    
    b = a[::-1]  # reversed
    # Linear convolution of w_conv (length 2n-1) and b (length n) has length 3n-2
    # We want elements at indices m + n - 1 for m = 0, ..., n-1
    # i.e., elements at indices n-1, n, ..., 2n-2
    
    fft_size2 = 1 << (3 * n - 2 - 1).bit_length()
    W2 = np.fft.rfft(w_conv, fft_size2)
    B2 = np.fft.rfft(b, fft_size2)
    conv_wb = np.fft.irfft(W2 * B2, fft_size2)[:3*n-2]
    
    grad_a = conv_wb[n-1:2*n-1]  # length n
    
    # Chain rule for softmax:
    # a = softmax(w), so da[i]/dw[j] = a[i]*(delta[ij] - a[j])
    # grad_w[j] = sum_i grad_a[i] * da[i]/dw[j]
    #           = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
    
    grad_a_dot_a = np.dot(grad_a, a)
    grad_w = grad_a * a - a * grad_a_dot_a
    
    return obj, grad_w


def optimize_smoothed(a0, epsilon_schedule, max_iter_per_eps=200, verbose=False):
    """
    Optimize the smoothed objective with temperature annealing.
    a0: initial point on simplex (length n)
    epsilon_schedule: list of epsilon values (decreasing)
    Returns: optimized point on simplex
    """
    n = len(a0)
    # Initialize w from a0: w = log(a) (with Small offset for zeros)
    a_safe = np.maximum(a0, 1e-300)
    w = np.log(a_safe / a_safe.sum())
    # Center w for numerical stability
    w = w - w.mean()
    
    best_obj = fast_convolve(a0).max()
    best_a = a0.copy()
    
    for eps_idx, epsilon in enumerate(epsilon_schedule):
        # L-BFGS-B optimization
        res = minimize(
            lambda ww: smoothed_objective_and_grad(ww, epsilon, n)[0],
            w,
            jac=lambda ww: smoothed_objective_and_grad(ww, epsilon, n)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_eps, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        w = res.x
        a = np.exp(w - w.max())
        a = a / a.sum()
        
        # Check true (unsmoothed) objective
        true_obj = fast_convolve(a).max()
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()
        
        if verbose:
            print(f"    eps={epsilon:.6f}: smoothed_obj={res.fun:.6f}, true_obj={true_obj:.10f}")
        
        # Update w for next epsilon (re-initialize from current a)
        a_safe = np.maximum(a, 1e-300)
        w = np.log(a_safe / a_safe.sum())
        w = w - w.mean()
    
    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    if res.success:
        return res.x
    return None

            no_improve = 0
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr0=0.01, top_k=50, temp0=0.001):
    """
    Subgradient method with softmax-weighted peaks and simplex projection.
    Uses the full gradient (not just top-k) for better accuracy.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_lr = lr0
    cur_temp = temp0
    no_improve = 0
    momentum = 0.85
    v = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax-weighted subgradient
        conv_scaled = conv / cur_temp
        conv_max = conv_scaled.max()
        weights = np.exp(conv_scaled - conv_max)
        weights = weights / weights.sum()
        
        # Gradient: grad_a[m] = 2 * sum_k weights[k] * a[k-m] * [0<=k-m<n]
        # Use FFT-based cross-correlation
        w_fft = (2.0) * weights  # length 2n-1
        b = f[::-1]
        fft_size = 1 << (3 * n - 2 - 1).bit_length()
        W = np.fft.rfft(w_fft, fft_size)
        B = np.fft.rfft(b, fft_size)
        grad_a = np.fft.irfft(W * B, fft_size)[:3*n-2][n-1:2*n-1]
        
        # Nesterov-style momentum update
        v = momentum * v - cur_lr * grad_a
        f_new = project_simplex(f + v)
        
        # Update
        f = f_new
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        cur_lr *= 0.999
        cur_temp *= 1.005
        momentum = min(0.95, momentum + 0.0001)
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            conv = fast_convolve(f)
            cur_lr = max(lr0 * 0.5, 0.002)
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
