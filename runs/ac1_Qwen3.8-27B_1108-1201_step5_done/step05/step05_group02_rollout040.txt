def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

**Spectral shaping (GS-like):**
1. Compute `F = DFT(a, N)` where `N` is the FFT size (zero-padded)
2. Compute `P = |F|^2` (power spectrum)
3. Modify `P` to be flatter: `P_new[k] = P[k] * alpha + beta * mean(P)` or `P_new[k] = min(P[k], gamma * median(P))`
4. Reconstruct: `a_new = IDFT(exp(i*angle(F)) * sqrt(P_new))`
5. Project to simplex: `a_new = max(a_new, 0)`, normalize to sum 1
6. Repeat

Combines: FFT-accelerated mirror descent with entropy regularization,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def entropy_project(v, beta, n_inner=10):
    """Entropy-regularized projection onto simplex.
    Minimizes (1/2)||x - v||^2 + beta * sum(x*log(x)) s.t. x>=0, sum=1.
    Uses bisection on the dual variable."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    lo, hi = -1e-10, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        x = np.zeros(n)
        for _ in range(n_inner):
            x_new = np.maximum(mid * v - beta, 0.0)
            s = x_new.sum()
            if s < 1e-300:
                x = np.ones(n) / n
                break
            x = x_new / s
        s = x.sum()
        if abs(s - 1.0) < 1e-12:
            break
        if s < 1.0:
            lo = mid
        else:
            hi = mid
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
    else:
        x = x / s
    return x


def compute_subgradient_fft(f, w):
    """Compute weighted subgradient of sum_k w[k]*conv(f,f)[k] using FFT.
    Returns gradient vector of length n. O(n log n)."""
    n = len(f)
    size = len(w)  # 2n-1
    total = n + size - 1
    fft_size = 1 << (total - 1).bit_length()
    f_flip = f[::-1]
    F = np.fft.rfft(f_flip, fft_size)
    W = np.fft.rfft(w.astype(np.float64), fft_size)
    conv_result = np.fft.irfft(F * W, fft_size)
    sg = 2.0 * conv_result[n - 1:2 * n - 2]
    return sg


def softmax_weights(conv, temp):
    """Compute softmax weights over convolution values."""
    if temp < 1e-15:
        w = np.zeros(len(conv))
        w[conv.argmax()] = 1.0
        return w
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[conv.argmax()] = 1.0
    else:
        w /= ws
    return w

def spectral_shaping(f0, n_iters=200, gamma_init=2.0, gamma_min=1.05,
                     decay_rate=0.995, verbose=False):
    """GS-like spectral shaping to flatten the power spectrum."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    fft_size = 1 << (n - 1).bit_length() * 2
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    gamma = gamma_init
    for it in range(n_iters):
        F = np.fft.fft(f, fft_size)
        P = np.abs(F) ** 2
        phase = np.angle(F)

        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        P_shrunk = P.copy()
        upper = P_mean * gamma
        lower = P_mean / gamma
        mask_hi = P_shrunk > upper
        mask_lo = P_shrunk < lower
        P_shrunk[mask_hi] = upper + (P_shrunk[mask_hi] - upper) * 0.05
        P_shrunk[mask_lo] = lower + (P_shrunk[mask_lo] - lower) * 0.05

        F_new = np.exp(1j * phase) * np.sqrt(P_shrunk)
        f_new = np.real(np.fft.ifft(F_new, fft_size))[:n]

        if s_new < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    print(f"\nStarting: best={best_score:.8f}, n={base_n}")

    # === Phase 1: Diverse initialization at base length ===
    print("\n=== Phase 1: Diverse initialization ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        init_seqs = diverse_initializations(base_n, rng)
        if parent is not None:
            init_seqs.insert(0, parent)
        for i, init in enumerate(init_seqs):
            obj = fast_convolve(init).max()
            print(f"    Init {i}: conv_max={obj:.8f}, obj={2*base_n*obj:.6f}")
            # Quick MD on this init
            t_alloc = min(8, rem * 0.05)
            if t_alloc > 2:
                a_opt, _ = mirror_descent(init, time_limit=t_alloc,
                                          lr_init=0.2, decay=0.999)
                try_seq(a_opt, f"init_{i}_md")
        rem = deadline - time.monotonic()

    # === Phase 2: Spectral shaping on best ===
    print(f"\n=== Phase 2: Spectral shaping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_ss = min(rem * 0.12, 180)
        if t_ss > 10:
            print(f"  n={len(best_seq)}, t={t_ss:.0f}s")
            a_opt = spectral_shaping(best_seq, n_iters=300, gamma_init=2.5,
                                     gamma_min=1.02, verbose=True)
            try_seq(a_opt, "spectral")
            # Also try on parent if different
            if parent is not None and not np.allclose(parent, best_seq):
                a_opt2 = spectral_shaping(parent, n_iters=100, gamma_init=3.0,
                                          gamma_min=1.05)
                try_seq(a_opt2, "spectral_parent")

    # === Phase 3: Mirror descent (main optimization) ===
    print(f"\n=== Phase 3: Mirror descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_md = min(rem * 0.25, 400)
        if t_md > 30:
            print(f"  n={len(best_seq)}, t={t_md:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_md,
                                      lr_init=0.4, lr_min=0.001,
                                      decay=0.9997, entropy_init=0.005,
                                      entropy_growth=1.005, verbose=True)
            try_seq(a_opt, "md_main")

            # Second MD with different params
            rem = deadline - time.monotonic()
            t_md2 = min(rem * 0.15, 200)
            if t_md2 > 20:
                a_opt2, _ = mirror_descent(best_seq, time_limit=t_md2,
                                           lr_init=0.15, lr_min=0.0005,
                                           decay=0.9999, entropy_init=0.05,
                                           entropy_growth=1.01, verbose=True)
                try_seq(a_opt2, "md_fine")

    # === Phase 4: LP-FW refinement ===
    print(f"\n=== Phase 4: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None and len(best_seq) <= 2000:
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.2, 250)
        if t_lp > 20:
            print(f"  n={len(best_seq)}, t={t_lp:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp, verbose=True)
            try_seq(a_opt, "lpfw_main")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        candidate_lengths = set()
        for delta in [-256, -128, 128, 256, 512, -512]:
            ln = len(best_seq) + delta
            if 256 <= ln <= 4096:
                candidate_lengths.add(ln)
        # Also try some powers of 2
        for p2 in [1024, 1280, 1536, 2048]:
            if 256 <= p2 <= 4096:
                candidate_lengths.add(p2)
        candidate_lengths = sorted(candidate_lengths)

        for target_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.2, 120)
            if t_alloc < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            # Quick spectral + MD
            a_ss = spectral_shaping(f_interp, n_iters=50, gamma_init=2.0,
                                    gamma_min=1.05)
            a_opt, _ = mirror_descent(a_ss, time_limit=t_alloc * 0.7,
                                      lr_init=0.3, decay=0.9995)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick MD polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t = min(rem * 0.3, 100)
            a_opt, _ = mirror_descent(best_seq, time_limit=t,
                                      lr_init=0.1, lr_min=0.0003,
                                      decay=0.9999, entropy_init=0.01,
                                      entropy_growth=1.01)
            try_seq(a_opt, "md_polish")

        # Quick LP-FW polish
        rem = deadline - time.monotonic() - 20
        if rem > 15 and len(best_seq) <= 2000:
            t = min(rem * 0.3, 80)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t)
            try_seq(a_opt, "lpfw_polish")

        # Perturbations + short MD
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise_scale = 0.003 + 0.003 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t = min(rem * 0.4, 30)
            a_opt, _ = mirror_descent(f_pert, time_limit=t,
                                      lr_init=0.2, decay=0.999,
                                      entropy_init=0.01)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
