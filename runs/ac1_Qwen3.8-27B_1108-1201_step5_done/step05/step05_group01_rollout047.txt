import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization on simplex."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def u_shape_init(n, sharpness=1.0):
    """U-shaped initialization: high at edges, low in middle."""
    j = np.arange(n)
    t = j / (n - 1)
    a = 1.0 + sharpness * (2.0 * t - 1.0) ** 2
    s = a.sum()
    return a / s


def w_shape_init(n, sharpness=1.0):
    """W-shaped: high at edges and at 1/3, 2/3 points."""
    j = np.arange(n)
    t = j / (n - 1)
    a = 1.0 + sharpness * np.cos(4.0 * np.pi * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_subgradient(f, peaks, weights):
    """Compute weighted subgradient of max conv at given peaks."""
    n = len(f)
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if weights[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        w = weights[idx]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w * f[k - indices]
    return sg


def nesterov_subgrad(f0, time_limit, lr0=0.01, mom0=0.8, top_k=10, temp0=0.0001):
    """
    Nesterov-accelerated projected subgradient with adaptive multi-peak weighting.
    Works on simplex (sum=1). Minimizes max(conv(f,f)).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(y)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks].copy()

        # Softmax weighting with adaptive temperature
        scaled = pv / max(cur_temp, 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = compute_subgradient(y, peaks, w)

        # Momentum update
        v = cur_mom * v - cur_lr * sg
        y_new = project_simplex(y + v)

        # Nesterov step
        t_it = 1.0 + np.sqrt(1.0 + 4.0 * it * it)
        beta = (1.0 - 1.0 / t_it) / (1.0 + cur_mom)
        f_new = y_new
        y = f_new + beta * (f_new - f)
        y = project_simplex(y)
        f = f_new

        co = fast_convolve(f).max()

        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive scheduling
        cur_lr *= 0.9997
        cur_temp *= 1.006
        cur_mom = min(0.95, cur_mom * 1.0002)

        # Restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            noise_scale = 0.002 + 0.003 * min((no_improve - 300) // 100, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.5, 0.002)
            cur_temp = max(temp0 * 0.5, 1e-6)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_coords=None):
    """
    Cyclic coordinate descent with golden-section search on simplex.
    For each coordinate j, optimize f[j] while rescaling others to maintain sum=1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    no_pass_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                # Try small positive value
                lo, hi = 0.0, 0.01
            else:
                lo, hi = 0.0, min(0.999, 2.5 * old_val)

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                rem = 1.0 - x
                if rem < 0:
                    return 1e10
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (rem / denom)
                return fast_convolve(ft).max()

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(14):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_obj = best_obj  # Use cached best to avoid recomputation each time

            if opt_val < best_obj - 1e-13:
                ft = f.copy()
                ft[j] = opt_x
                rem = 1.0 - opt_x
                if rem >= 0:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    ft[mask] = ft[mask] * (rem / denom)
                    st = ft.sum()
                    if st > 1e-15:
                        ft = ft / st
                        new_obj = fast_convolve(ft).max()
                        if new_obj < best_obj:
                            f = ft
                            best_obj = new_obj
                            best_f = f.copy()
                            pass_improved = True

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break

    return best_f, best_obj


def lp_fw_step(f, M, n):
    """Solve one LP-FW step: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    try:
        from scipy.optimize import linprog
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        if g_sum < 1e-15:

            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(rem * 0.5, 60),
                                         lr0=0.002, mom0=0.9,
                                         top_k=5,
                                         temp0=0.000005)
            try_seq(a_opt, "final_subgrad")

        # Short CD
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.7, 40))
            try_seq(a_opt, "final_cd")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for _ in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            n_b = len(best_seq)
            noise = rng.standard_normal(n_b) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_subgrad(f_pert, time_limit=20,
                                         lr0=0.001, mom0=0.92,
                                         top_k=3,
                                         temp0=0.000002)
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
