import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_subgradient(f, k_peak, n):
    """Exact subgradient of (f*f)[k_peak] w.r.t. f.
    g[m] = 2*f[k_peak - m] if 0 <= k_peak - m < n, else 0.
    """
    g = np.zeros(n)
    valid_start = max(0, k_peak - n + 1)
    valid_end = min(n - 1, k_peak)
    m_indices = np.arange(valid_start, valid_end + 1)
    j_indices = k_peak - m_indices
    g[m_indices] = 2.0 * f[j_indices]
    return g


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, min(j + n, 2 * n - 1))
        A_ub[rows, j] = f[:len(rows)]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30, 'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

def coordinate_descent(f0, time_limit, verbose=False):
    """Cyclic coordinate descent with golden section 1D search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        no_improve_passes += 1
        if no_improve_passes > 8:
            break

        pass_improved = False
        order = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = order[idx]
            c = f[i]
            if c < 1e-12:
                continue

            # Binary/golden search for optimal value of f[i]
            # Parameterize: f[i] = t, others scaled to maintain sum=1
            # f_new[i] = t, f_new[j] = f[j] * (1-t)/(1-c) for j != i
            lo = 0.0
            hi = min(0.99, 2.0 * c + 0.01)

            # Evaluate a few candidates
            candidates = [c, 0.0, hi, 0.5 * c, 1.5 * c, 0.25 * c, 2.0 * c]
            candidates = [max(0.0, min(0.999, x)) for x in candidates]

            best_t = c
            best_val = fast_convolve(f).max()

            for t in candidates:
                if abs(t - c) < 1e-10:
                    continue
                if t >= 1.0:
                    continue
                f_trial = f.copy()
                f_trial[i] = t
                others = f_trial.sum() - t
                if others < 1e-15:
                    continue
                # Scale others to sum to 1-t
                mask = np.arange(n) != i
                f_trial[mask] = f_trial[mask] * (1.0 - t) / others
                if f_trial.sum() < 1e-15:
                    continue
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t

            if best_val < best_obj - 1e-12:
                f[i] = best_t
                others_sum = f.sum() - best_t
                if others_sum > 1e-15:
                    mask = np.arange(n) != i
                    f[mask] = f[mask] * (1.0 - best_t) / others_sum
                f = f / f.sum()
                best_obj = best_val
                best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve_passes = 0

    return best_f, best_obj


def deflation_optimize(f0, time_limit, verbose=False):
    """Targeted peak deflation: explicitly reduce the peak lag."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    best_M = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))

        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()

        # Subgradient at peak
        g = peak_subgradient(f, k_peak, n)

        # Find the coordinate with largest gradient (most contributing to peak)
        # Reduce it and redistribute to least-contributing coordinates
        i_reduce = int(np.argmax(g))
        # Find coordinates with smallest gradient (safe to increase)
        sorted_g = np.argsort(g)
        i_increase = int(sorted_g[0])
        i_increase2 = int(sorted_g[1]) if n > 1 else 0

        if i_reduce == i_increase or i_reduce == i_increase2:
            # Try another pair
            if n > 2:
                i_increase = int(sorted_g[2])
            if i_reduce == i_increase:
                break

        # Try a few transfer amounts
        c = f[i_reduce]
        best_frac = 0.0
        best_val = M

        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            amount = frac * c
            if amount < 1e-10:
                continue
            f_trial = f.copy()
            f_trial[i_reduce] = c - amount
            # Distribute to increase coords proportional to their current values
            inc_total = f_trial[i_increase] + f_trial[i_increase2]
            if inc_total < 1e-15:
                f_trial[i_increase] += amount / 2
                f_trial[i_increase2] += amount / 2
            else:
                f_trial[i_increase] += amount * f_trial[i_increase] / inc_total
                f_trial[i_increase2] += amount * f_trial[i_increase2] / inc_total
            if f_trial[i_reduce] < 0:
                continue
            f_trial_sum = f_trial.sum()
            if f_trial_sum < 1e-15:
                continue
            f_trial = f_trial / f_trial_sum
            val = fast_convolve(f_trial).max()
            if val < best_val - 1e-13:
                best_val = val
                best_frac = frac

        if best_frac > 0:
            amount = best_frac * c
            f[i_reduce] = c - amount
            inc_total = f[i_increase] + f[i_increase2]
            if inc_total < 1e-15:
                f[i_increase] += amount / 2
                f[i_increase2] += amount / 2
            else:
                f[i_increase] += amount * f[i_increase] / inc_total
                f[i_increase2] += amount * f[i_increase2] / inc_total
            f = np.maximum(f, 0)
            f = f / f.sum()
        else:
            # No improvement from deflation, try subgradient step

            new_M = fast_convolve(f_new).max()
            if new_M < M:

            else:

                t = np.clip(t, 0.0, 1.0)
                f_trial = f.copy()
                f_trial[i] = t * c
                f_trial[j] = (1 - t) * c
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t

            if best_val < best_obj - 1e-12:
                f[i] = best_t * c
                f[j] = (1 - best_t) * c
                f = f / f.sum()
                best_obj = best_val
                best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_diverse_initializations(n, best_f):
    """Generate diverse initial sequences."""
    inits = []

    # The best found so far
    inits.append(best_f.copy())

    # Reversed
    inits.append(best_f[::-1].copy() / best_f[::-1].sum())

    # Alternating pattern blended with best
    alt = np.zeros(n)
    for i in range(n):
        alt[i] = 1.0 if i % 2 == 0 else 0.3
    alt = alt / alt.sum()
    inits.append(0.7 * best_f + 0.3 * alt)
    inits[-1] = inits[-1] / inits[-1].sum()

    # Random smooth
    rng = np.random.default_rng(123)
    smooth = np.abs(rng.standard_normal(n))
    # Apply simple low-pass
    kernel = np.ones(5) / 5.0
    smooth = np.convolve(smooth, kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    smooth = smooth / smooth.sum()
    inits.append(0.8 * best_f + 0.2 * smooth)
    inits[-1] = inits[-1] / inits[-1].sum()

    # Edge-concentrated
    edge = np.zeros(n)
    edge[:n//4] = 1.0
    edge[-n//4:] = 1.0
    edge = edge + 0.3
    edge = edge / edge.sum()
    inits.append(0.6 * best_f + 0.4 * edge)
    inits[-1] = inits[-1] / inits[-1].sum()

    # Noisy perturbation
    rng2 = np.random.default_rng(456)
    noisy = best_f + rng2.standard_normal(n) * 0.01
    noisy = project_simplex(noisy)
    inits.append(noisy)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining subgradient, LP-FW, coordinate descent,
    deflation, and diverse restarts to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})", flush=True)
        return sc

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}", flush=True)
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}", flush=True)

    if parent is None:
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}", flush=True)

    # Set global for subgradient function
    global T_initial_val
    T_initial_val = 3.0 / len(parent)

    n_parent = len(parent)

    # === Phase 1: Subgradient on parent ===
    print(f"\n=== Phase 1: Subgradient (n={n_parent}) ===", flush=True)
    rem = deadline - time.monotonic() - 700
    t1 = min(rem, 300)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s", flush=True)
        a_opt, _ = subgradient_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "subgrad_p1")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 550
        t2 = min(rem, 250)
        if t2 > 20:
            print(f"  Time: {t2:.0f}s, n={len(best_seq)}", flush=True)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "lpfw_p2")

    # === Phase 3: Deflation ===
    print(f"\n=== Phase 3: Deflation (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        t3 = min(rem, 150)
        if t3 > 15:
            print(f"  Time: {t3:.0f}s", flush=True)
            a_opt, _ = deflation_optimize(best_seq, time_limit=t3)
            try_seq(a_opt, "defl_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===", flush=True)
    base = best_seq if best_seq is not None else parent

    for target_n in [512, 1024]:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 350
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        T_initial_val = 3.0 / target_n

        # Subgradient
        t_alloc = min(rem * 0.4, 120)
        if t_alloc > 20:
            print(f"  Subgrad n={target_n}, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = subgradient_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"subgrad_n{target_n}")

        # LP-FW
        rem = deadline - time.monotonic() - 300
        if rem > 25:
            t_lpfw = min(rem * 0.5, 120)
            if t_lpfw > 15:
                print(f"  LP-FW n={target_n}, t={t_lpfw:.0f}s", flush=True)
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw)
                try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        T_initial_val = 3.0 / n_best
        inits = make_diverse_initializations(n_best, best_seq)

        for p_idx, f_init in enumerate(inits[:5]):
            rem = deadline - time.monotonic() - 250
            if rem < 20:
                break
            t_alloc = min(rem / max(1, 5 - p_idx), 150)

            # Alternate methods
            if p_idx % 2 == 0:
                a_opt, _ = subgradient_optimize(f_init, time_limit=min(t_alloc, 100))
            else:
                a_opt, _ = lp_fw_optimize(f_init, time_limit=min(t_alloc, 120))
            try_seq(a_opt, f"diverse_{p_idx}")

            # Quick polish
            rem = deadline - time.monotonic() - 200
            if rem > 15:
                a_opt2, _ = deflation_optimize(a_opt, time_limit=min(rem, 40))
                try_seq(a_opt2, f"diverse_{p_idx}_polish")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coord Descent (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_cd = min(rem * 0.6, 200)
            print(f"  Time: {t_cd:.0f}s, n={len(best_seq)}", flush=True)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "coord_desc")

    # === Phase 7: Pairwise ===
    print(f"\n=== Phase 7: Pairwise (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_pw = min(rem * 0.7, 150)
            print(f"  Time: {t_pw:.0f}s", flush=True)
            a_opt, _ = pairwise_optimize(best_seq, time_limit=t_pw)
            try_seq(a_opt, "pairwise")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===", flush=True)
    if best_seq is not None:
        n_final = len(best_seq)
        T_initial_val = 3.0 / n_final

        # Subgradient fine
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "subgrad_final")

        # Deflation fine
        rem = deadline - time.monotonic() - 30
        if rem > 8:
            a_opt, _ = deflation_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "defl_final")

        # Coordinate descent final
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "coord_final")

        # Tiny perturbations
        for trial in range(3):
            rem = deadline - time.monotonic() - 5
            if rem < 5:
                break
            rng = np.random.default_rng(700 + trial)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_final) * 0.003)
            a_opt, _ = subgradient_optimize(f_pert, time_limit=min(rem, 12))
            try_seq(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
