def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def general_convolve(a, b):
    """Linear convolution via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_obj_grad(f, T):
    """Compute smoothed objective and gradient."""
    conv = fast_convolve(f)
    c_max = conv.max()
    shifted = T * (conv - c_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    n = len(f)
    # Gradient: grad[j] = 2 * sum_k p[k] * f[k-j] for valid k-j
    # = 2 * convolve(f, p[::-1])[n-1:2n-1][::-1]
    p_rev = p[::-1]
    cp = general_convolve(f, p_rev)
    grad = 2.0 * cp[n-1:2*n-1][::-1]
    
    obj = c_max + np.log(exp_shifted.sum()) / T
    return obj, grad

def continuation_lbfgs(f0, time_limit, T_start=0.5, T_end=5000.0, n_T=40, max_iter=300):
    """Continuation method with L-BFGS on smoothed objective."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    T_values = np.geomspace(T_start, T_end, n_T)
    
    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_and_grad(z):
            z_shifted = z - z.max()
            fw = np.exp(z_shifted)
            fw_sum = fw.sum()
            if fw_sum < 1e-300:
                return 1e10, np.zeros(n)
            fw = fw / fw_sum
            obj, grad_f = smoothed_obj_grad(fw, T)
            inner = np.dot(grad_f, fw)
            grad_z = fw * (grad_f - inner)
            return obj, grad_z
        
        # Start from current f parameterized as log(f)
        f_safe = np.maximum(f, 1e-300)
        z0 = np.log(f_safe)
        
        try:
            result = minimize(obj_and_grad, z0, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'gtol': 1e-10,
                                    'ftol': 1e-12, 'maxls': 50})
            z_opt = result.x
            z_s = z_opt - z_opt.max()
            f_new = np.exp(z_s)
            f_new = f_new / f_new.sum()
        except Exception:
            f_new = f
        
        f_new = np.maximum(f_new, 0)
        f_s = f_new.sum()
        if f_s < 1e-15:
            f_new = f
        else:
            f_new = f_new / f_s
        
        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_obj

def lp_fw_optimize(f0, time_limit, top_k=50):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Select active constraints
        if it > 5 and top_k < 2*n-1:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            active_k = k_sorted
        else:
            active_k = None
        
        # Build and solve LP
        if active_k is None:
            K = 2*n - 1
            A_ub = np.zeros((K, n))
            for j in range(n):
                A_ub[j:j+n, j] = f
            b_ub = np.full(K, M)
        else:
            K = len(active_k)
            A_ub = np.zeros((K, n))
            b_ub = np.full(K, M)
            for row, k in enumerate(active_k):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    A_ub[row, j] = f[k - j]
        
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                no_improve += 1
                if no_improve > 100:
                    break
                continue
            g = res.x
        except Exception:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        # Line search (bisection)
        lo, hi = 0.0, 1.0
        for _ in range(20):
            mid = (lo + hi) / 2
            f_trial = project_simplex((1 - mid) * f + mid * g)
            conv_trial = fast_convolve(f_trial)
            if conv_trial.max() < conv.max():
                lo = mid
            else:
                hi = mid
        
        alpha = lo
        f_new = project_simplex((1 - alpha) * f + alpha * g)
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        conv = conv_new
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
            M = M_new
        else:
            no_improve += 1
            M = max(M * 1.01, M_new)
        
        if no_improve > 50:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            it = 0
    
    return best_f, best_obj

def coord_descent(f0, time_limit, n_multipliers=8):
    """Aggressive coordinate descent with golden section per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    n = len(f)

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int((time.monotonic() - t0) * 1000))
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-15:
                continue
            
            # Try various new values for f[j]
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Golden section search for optimal f[j]
            lo = 0.0
            hi = min(1.0 - 1e-15, old_val * 10 + 0.1)
            
            # Actually, let's just try a set of multipliers for efficiency
            multipliers = [0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0, 5.0]
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0 - 1e-10:
                    continue
                remaining = 1.0 - new_val
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj

def pair_swap_search(f0, time_limit):
    """Try swapping or partially swapping pairs of elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    fractions = [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]
    
    while time.monotonic() - t0 < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        for frac in fractions:
            # Move frac of f[j] to f[i] (and vice versa)
            delta = frac * (f[j] - f[i])
            if abs(delta) < 1e-15:
                continue
            
            f_trial = f.copy()
            f_trial[i] += delta
            f_trial[j] -= delta
            if f_trial[i] < 0 or f_trial[j] < 0:
                continue
            
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_trial.copy()
                f = f_trial.copy()
                break
    
    return best_f, best_obj

def generate_initialization_flat_spectrum(n, seed):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2*n - 2).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)[:n]
    sig = np.maximum(sig, 0)
    s = sig.sum()
    if s > 1e-10:
        sig /= s
    else:
        sig = np.abs(sig)
        s = sig.sum()
        if s > 1e-10:
            sig /= s
        else:
            sig = np.ones(n) / n
    return sig

def generate_initialization_binary(n, seed, weight=None):
    """Initialize with a random binary-like sequence."""
    rng = np.random.default_rng(seed)
    if weight is None:
        weight = n // 2
    positions = rng.choice(n, size=weight, replace=False)
    f = np.zeros(n)
    f[positions] = 1.0
    # Add some continuous variation
    f = f + rng.uniform(0, 0.3, n)
    f = f / f.sum()
    return f

def generate_initialization_gaussian(n, seed, center=0.5, width=0.2):
    """Gaussian-shaped initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    f = np.exp(-((x - center) / width) ** 2)
    f += rng.uniform(0, 0.1, n)
    f = np.maximum(f, 0)
    f = f / f.sum()
    return f

def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
