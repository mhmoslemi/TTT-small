**New approach: Use a "saddle point" / minimax formulation**

The problem `min_a max_k b_k(a) / (sum a)^2` can be written as a saddle point problem:

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, T, k_peaks=None):
    """Compute smoothed subgradient of max(b) using log-sum-exp with top-k."""
    conv = fast_convolve(f)
    n = len(f)
    
    if k_peaks is not None and k_peaks < len(conv):
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
    else:
        peaks = np.arange(len(conv))
    
    pv = conv[peaks]
    scaled = T * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg, w, peaks

def nesterov_smoothed(f0, time_limit, T_start, T_end, beta=0.9, lr_start=0.01, verbose=False):
    """Nesterov-accelerated log-sum-exp smoothed optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    y = f.copy()  # Nesterov's y
    v = np.zeros(n)  # momentum velocity
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # We'll use a total of max_iters steps
    # Each step should take about 0.005s for n~768

        if time.monotonic() - t0 > time_limit:

        lr = lr_start * (1.0 - 0.7 * progress)
        top_k = max(10, int(n * (1.0 - 0.8 * progress)))
        
        # Nesterov lookahead

def lp_with_dual(f, M, n):
    """Solve LP and return both primal and dual solutions."""
    c = -np.ones(n)
    # Build constraint matrix
    # (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # For each k (0 to 2n-2), the constraint is:
    #   sum_j f[j] * g[k-j] <= M (where k-j must be in [0,n-1])
    # So for variable g[j], its coefficient in constraint k is f[k-j] if 0 <= k-j < n, else 0.
    
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        dual = res.ineqlin_slack  # No, this is slack
        # In scipy's linprog with HiGHS, the dual values are in res.eqlin.marginal and res.ineqlin.marginal
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            dual_vals = np.abs(res.ineqlin.marginal)  # Dual variables (shadow prices)
        else:
            dual_vals = None
        return np.maximum(res.x, 0), dual_vals
    return None, None

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(f, T, top_k=None):
    """Compute log-sum-exp smoothed subgradient of max(conv(f,f))."""
    conv = fast_convolve(f)
    n = len(f)
    total = len(conv)
    
    if top_k is not None and top_k < total:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(total)
    
    pv = conv[peaks]
    shifted = T * (pv - pv.max())
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[int(np.argmax(pv))] = 1.0
    else:
        w = w / ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg, conv

def nesterov_softmax(f0, time_limit, verbose=False):
    """
    Nesterov-accelerated optimization using softmax parameterization.
    Optimizes in z-space where a = softmax(z).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Convert to z-space
    z = np.log(f + 1e-30)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    
    # Nesterov parameters
    beta = 0.9  # momentum
    # Temperature schedule
    T_start = 3.0 / n
    T_end = 0.3 / n
    # Learning rate schedule
    lr_start = 0.02
    lr_end = 0.001
    
    # Precompute timing
    # Each iteration: convolve (O(n log n)) + gradient (O(n * top_k)) + softmax (O(n))
    # For n=768, each iteration is maybe 0.3-0.5ms
    estimated_iters = int(time_limit / 0.0005)
    
    z_prev = z.copy()
    no_improve = 0
    
    for it in range(estimated_iters):
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        progress = min(1.0, it / max(1, estimated_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress * 0.5 + lr_start * 0.5
        # Actually let's use a simpler schedule
        lr = lr_start * (1.0 - 0.8 * progress)
        top_k = max(20, int(n * (1.5 - progress)))
        
        # Nesterov lookahead
        z_nest = z + beta * (z - z_prev)
        
        # Convert to a-space
        z_nest_shifted = z_nest - z_nest.max()  # for numerical stability
        a = np.exp(z_nest_shifted)
        a = a / a.sum()
        
        # Compute smoothed gradient w.r.t. a
        sg_a, conv = compute_smoothed_grad(a, T, top_k)
        
        # Convert gradient to z-space: dz = a * (sg_a - a @ sg_a)
        # This is the natural gradient / mirror gradient
        sg_z = a * (sg_a - np.dot(a, sg_a))
        
        # Update
        z_new = z - lr * sg_z
        z_prev = z
        z = z_new
        
        # Evaluate
        f_curr = np.exp(z - z.max())
        f_curr = f_curr / f_curr.sum()
        new_M = fast_convolve(f_curr).max()
        
        if new_M < best_obj - 1e-13:
            best_obj = new_M
            best_f = f_curr.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Nesterov iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        if no_improve > 500:
            rng = np.random.default_rng(it * 3 + 1)
            f_pert = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            z = np.log(f_pert + 1e-30)
            z_prev = z.copy()
            no_improve = 0
    
    return best_f, best_obj

def lp_fw(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    step = 1.0  # adaptive step
    
    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        # Build constraint matrix efficiently
        num_constraints = 2 * n - 1
        A_ub = np.zeros((num_constraints, n))
        for j in range(n):
            rows = np.arange(j, min(j + n, num_constraints))
            A_ub[rows, j] = f[:len(rows)]
        
        b_ub = np.full(num_constraints, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            # Relax constraint
            res = linprog(c, A_ub=A_ub, b_ub=np.full(num_constraints, M * 1.01), 
                         bounds=bounds, method='highs')
            if not res.success:
                no_improve += 1
                if no_improve > 40:
                    break
                continue
        
        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check if direction is meaningful
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                M = fast_convolve(f).max()
                no_improve = 0
                continue
            continue
        
        # Line search: try a few steps
        candidates = [0.1, 0.2, 0.3, 0.5, 0.7, step, 1.0]
        best_alpha = 0.0
        best_val = M
        
        for alpha in candidates:
            if alpha <= 0 or alpha > 1:
                continue
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            step = best_alpha  # adapt step size
            
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            step *= 0.7
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            M = fast_convolve(f).max()
            no_improve = 0
            step = 0.5
    
    return best_f, best_obj

        M = conv.max()

        num_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[-num_peaks:]
        
        # For each peak, find the most contributing pairs
        reduction_candidates = {}
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                # Contribution of pair (j, k-j) to b[k]
                contrib = f[j] * f[k - j]
                # We want to reduce this by moving mass from the larger of f[j], f[k-j]
                # to a position that doesn't contribute to any peak
                if contrib > 0:

                    for coord in [j, k - j]:
                        # Total contribution of this coord to all peaks
                        total_contrib = 0

                        reduction_candidates[coord] += total_contrib
        
        if not reduction_candidates:
            break
        
        # Sort by total contribution (most to least)
        sorted_coords = sorted(reduction_candidates.items(), key=lambda x: -x[1])

                        if js2 <= target <= je2:

                    if tc < best_target_contrib:

                        best_target = target

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def coordinate_polish(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
