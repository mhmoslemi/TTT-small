def fast_convolve(a):
    """Linear full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1.
    Returns (g, t) or (None, None)."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns g or None."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 1.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def coord_descent_polish(f0, time_limit):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        # Get top 5 peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # For each coordinate contributing to this peak, try adjusting
            candidates = list(range(js, je + 1))
            # Sort by contribution to peak (largest first)
            contribs = []
            for j in candidates:
                kj = peak_k - j
                if 0 <= kj < n:
                    contribs.append((f[j] * f[kj], j))
            contribs.sort(reverse=True)

            for _, j in contribs[:min(15, len(contribs))]:
                old_val = f[j]
                if old_val < 1e-12:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj

                # Try several multipliers
                for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.4, 1.8]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 8:
                # Random perturbation
                rng = np.random.default_rng(it * 31337 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                no_improve = 0
            if no_improve > 25:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW (primary), multi-length exploration,
    subgradient refinement, coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    print(f"Starting optimization. Parent n={base_n}, score={best_score:.8f}")

    # === Phase 1: LP-FW on parent (main workhorse) ===
    print(f"\n=== Phase 1: LP-FW on parent (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.42, 750)
    if t1 > 40:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Try nearby lengths ===
    print(f"\n=== Phase 2: Nearby lengths (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = []
    for delta in [-96, -64, -32, 32, 64, 96, 128, 192]:
        ln = len(base) + delta
        if ln >= 200 and ln <= 3000:
            candidate_lengths.append(ln)
    # Deduplicate and sort
    candidate_lengths = sorted(set(candidate_lengths))

    for target_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.18, 200)
        if t_alloc < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.25, 350)
        if t_sg > 30:
            print(f"  n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   lr=0.0025, momentum=0.945,
                                   top_k=max(40, n_best // 5),
                                   temp=0.00004)
            try_seq(a_opt, "sg_main")

            # Second pass with different params
            rem = deadline - time.monotonic()
            t_sg2 = min(rem * 0.15, 180)
            if t_sg2 > 25:
                a_opt, _ = sg_optimize(best_seq, time_limit=t_sg2,
                                       lr=0.001, momentum=0.955,
                                       top_k=max(60, n_best // 3),
                                       temp=0.00002)
                try_seq(a_opt, "sg_fine")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.25, 250)
        if t_cd > 20:
            print(f"  n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")

    # === Phase 5: Final LP-FW polish + perturbations ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick LP-FW
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_lp = min(rem * 0.4, 120)
            print(f"  LP-FW polish, t={t_lp:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp)
            try_seq(a_opt, "lpfw_polish")

        # Quick SG
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_sg = min(rem * 0.4, 80)
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   lr=0.0008, momentum=0.96,
                                   top_k=max(20, n_best // 8),
                                   temp=0.000015)
            try_seq(a_opt, "sg_polish")

        # Perturbations with quick optimization
        for p in range(5):
            rem = deadline - time.monotonic() - 12
            if rem < 15:
                break
            noise_scale = 0.0008 + 0.0004 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.3, 30)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_pert)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
