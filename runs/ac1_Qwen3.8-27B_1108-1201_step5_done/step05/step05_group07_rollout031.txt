def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def lse_obj_and_grad(a, beta=50.0):
    """Compute LSE objective and gradient w.r.t. a."""
    conv = fast_convolve(a)
    m = conv.max()
    exp_vals = np.exp(conv - m)
    p = exp_vals / exp_vals.sum()
    obj = m + np.log(exp_vals.sum()) / beta  # Actually this is log(sum(exp(conv))/exp(m)) / beta + m
    # Wait: LSE = (1/beta) * log(sum(exp(beta * conv)))
    # = (1/beta) * log(sum(exp(beta * conv - beta * m)) * exp(beta * m))
    # = m + (1/beta) * log(sum(exp(beta * conv - beta * m)))
    # Let me redo:
    # LSE = (1/beta) * log(sum_k exp(beta * conv_k))
    # = (1/beta) * log(exp(beta*m) * sum_k exp(beta*(conv_k - m)))
    # = m + (1/beta) * log(sum_k exp(beta*(conv_k - m)))
    # p_k = exp(beta*(conv_k - m)) / sum_j exp(beta*(conv_j - m))
    # So sum_k exp(beta*(conv_k - m)) = sum_k p_k * sum_j exp(beta*(conv_j - m))... no
    # sum_k exp(beta*(conv_k - m)) = S (say)
    # p_k = exp(beta*(conv_k - m)) / S
    # LSE = m + log(S) / beta = m + log(sum_k exp(beta*(conv_k - m))) / beta
    
    # Gradient w.r.t. a:
    # d(LSE)/d(a_i) = sum_k p_k * d(conv_k)/d(a_i) = sum_k p_k * 2*a[k-i] (with bounds)
    # = 2 * correlate(p, a)[i] (using numpy's correlate)
    
    grad_a = 2.0 * np.correlate(p, a, mode='valid')[:len(a)]
    
    return obj, grad_a, p, conv

def transform_grad_to_z(grad_a, a):
    """Transform gradient from a-space to z-space (where a = softmax(z))."""
    # d/dz_i = sum_j (d/da_j) * (da_j/dz_i)
    # da_j/dz_i = a_j * (delta_ij - a_i)
    # d/dz_i = grad_a_i * a_i - a_i * sum_j grad_a_j * a_j
    return a * grad_a - a * np.dot(grad_a, a)

def adam_log_optimize(a0, time_limit, lr=0.01, beta_lse=50.0, 
                      beta1=0.8, beta2=0.999, eps=1e-8,
                      lr_decay=0.9995, verbose=False):
    """
    Adam optimization on log-parameterization.
    Minimize LSE(beta * conv(a,a)) where a = softmax(z).
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    z = np.log(a0)  # log-parameterization
    
    m_ = np.zeros(n)
    v_ = np.zeros(n)
    t = 0
    
    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()
    
    t0 = time.monotonic()
    cur_lr = lr
    cur_beta_lse = beta_lse
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a = softmax(z)
        obj_lse, grad_a, p, conv = lse_obj_and_grad(a, cur_beta_lse)
        
        # Track actual max
        actual_max = conv.max()
        if actual_max < best_obj - 1e-12:
            best_obj = actual_max
            best_a = a.copy()
        
        # Transform gradient
        grad_z = transform_grad_to_z(grad_a, a)
        
        # Adam update
        m_ = beta1 * m_ + (1 - beta1) * grad_z
        v_ = beta2 * v_ + (1 - beta2) * grad_z ** 2
        m_hat = m_ / (1 - beta1 ** t)
        v_hat = v_ / (1 - beta2 ** t)
        z -= cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Schedule
        cur_lr *= lr_decay
        if t % 200 == 0:
            cur_beta_lse *= 0.9  # Anneal: make LSE sharper (closer to true max)
        
        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}, t={time.monotonic()-t0:.1f}s, max_conv={best_obj:.8f}, lr={cur_lr:.6f}, beta_lse={cur_beta_lse:.2f}")
    
    return best_a, best_obj

def peak_local_search(a0, time_limit, verbose=False):
    """
    Local search focused on reducing the top peaks of the autocorrelation.
    For each top peak, try to reduce it by adjusting the contributing elements.
    """
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        iteration += 1
        conv = fast_convolve(a)
        current_max = conv.max()
        
        # Get top peaks
        top_k = min(20, 2*n-1)
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        
        improved = False
        
        for k in peak_idx[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            peak_val = conv[k]
            
            # Get contributing indices
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            
            # Contributions: a[i] * a[k-i]
            contribs = a[idx] * a[k - idx]
            
            # Strategy: reduce the largest contributors
            # Try reducing the top contributor and distributing to others
            order = np.argsort(contribs)[::-1]
            
            for oi in order[:3]:
                i = idx[oi]
                j = k - i
                if i == j:
                    # Diagonal: reduce a[i]
                    for frac in [0.05, 0.1, 0.2]:
                        new_a = a.copy()
                        amt = a[i] * frac
                        new_a[i] = a[i] - amt
                        # Distribute to neighbors
                        neighbors = [x for x in [i-1, i+1] if 0 <= x < n]
                        if neighbors:
                            for nb in neighbors:
                                new_a[nb] += amt / len(neighbors)
                        new_a = np.maximum(new_a, 0)
                        ns = new_a.sum()
                        if ns < 1e-10:
                            continue
                        new_a /= ns
                        new_obj = fast_convolve(new_a).max()
                        if new_obj < best_obj - 1e-12:
                            a = new_a
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                else:
                    # Off-diagonal: transfer between a[i] and a[j]
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        # Make a[i] smaller, a[j] larger (or vice versa)
                        for direction in [1, -1]:
                            new_a = a.copy()
                            amt = min(a[i], a[j]) * frac
                            if direction == 1:
                                new_a[i] = a[i] - amt
                                new_a[j] = a[j] + amt
                            else:
                                new_a[i] = a[i] + amt
                                new_a[j] = a[j] - amt
                            new_a = np.maximum(new_a, 0)
                            ns = new_a.sum()
                            if ns < 1e-10:
                                continue
                            new_a /= ns
                            new_obj = fast_convolve(new_a).max()
                            if new_obj < best_obj - 1e-12:
                                a = new_a
                                best_obj = new_obj
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
            
            if not improved:
                # Try pairwise exchange with a random element
                rng = np.random.default_rng(iteration * 7919 + k)
                for _ in range(5):
                    j = int(rng.integers(0, n))
                    if j == i or j == (k - i):
                        continue
                    for frac in [0.05, 0.1]:
                        new_a = a.copy()
                        amt = a[idx[order[0]]] * frac
                        new_a[idx[order[0]]] -= amt
                        new_a[j] += amt
                        new_a = np.maximum(new_a, 0)
                        ns = new_a.sum()
                        if ns < 1e-10:
                            continue
                        new_a /= ns
                        new_obj = fast_convolve(new_a).max()
                        if new_obj < best_obj - 1e-12:
                            a = new_a
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best_a, best_obj

def pair_exchange_search(f0, time_limit, max_pairs=2000):
    """Pair-exchange local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        top_peaks = np.argpartition(conv, -min(30, 2 * n - 1))[-min(30, 2 * n - 1):]
        
        rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
        trials = min(max_pairs, 500)
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-10:
                continue
            for frac in [0.1, 0.3, 0.5]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] = old_j + amt
                f = np.maximum(f, 0)
                fs = f.sum()
                if fs < 1e-10:
                    f[i] = old_i
                    f[j] = old_j
                    break
                f /= fs
                val = fast_convolve(f).max()
                if val < best_obj - 1e-12:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = old_j
                    break
        
        if not improved:
            break
    
    return best_f, best_obj

def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: for each coordinate, find optimal value via golden section."""
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Process coordinates in order of potential impact (largest values first)
    order = np.argsort(a)[::-1]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            # Try several multipliers
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.5, 3.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                new_a = a.copy()
                new_a[j] = new_val
                # Renormalize others to keep sum = 1
                remaining = 1.0 - new_val
                if remaining < 1e-10:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = remaining / old_remaining
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] = a[mask] * scale
                new_a = np.maximum(new_a, 0)
                ns = new_a.sum()
                if ns < 1e-10:
                    continue
                new_a /= ns
                obj = fast_convolve(new_a).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                new_val = best_j_val
                remaining = 1.0 - new_val
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10 and remaining > 1e-10:
                    scale = remaining / old_remaining
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[j] = new_val
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    if best_j_obj < best_obj - 1e-12:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())
    
    # Gaussian centered
    center = n / 2.0
    sigma = n / 5.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Bump at edges (to minimize a[0]^2)
    edge = np.ones(n) * 0.5
    edge[0] = 0.1
    edge[-1] = 0.1
    edge = np.maximum(edge, 0.01)
    inits.append(edge / edge.sum())
    
    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())
    
    # Cosine raised (flat top)
    x = np.linspace(0, 1, n)
    flat_top = np.maximum(np.cos(np.pi * (x - 0.5)), 0.1)
    inits.append(flat_top / flat_top.sum())
    
    # Two peaks
    c1, c2 = n * 0.25, n * 0.75
    two_dip = np.exp(-((np.arange(n) - c1) / (n/10))**2) + np.exp(-((np.arange(n) - c2) / (n/10))**2)
    inits.append(two_dip / two_dip.sum())
    
    return inits

**Alternative: Spectral approach**

The autocorrelation is related to the power spectrum: by Wiener-Khinchin, the autocorrelation is the inverse Fourier transform of the power spectrum. So `conv(a,a)` is related to `|FFT(a)|^2`.

Specifically, for a zero-padded sequence, the circular autocorrelation is `ifft(|fft(a)|^2)`. The linear autocorrelation is a subset of this.

So minimizing the max of the autocorrelation is equivalent to making the power spectrum as "spread out" as possible (flat spectrum → flat autocorrelation). This is the "minimum peak-to-average sidelobe ratio" problem.

One approach: iterate between:
1. Compute power spectrum P = |FFT(a)|^2
2. Modify P to reduce peaks (e.g., P_new = 1/(P + epsilon) which spreads energy from peaks to valleys)
3. Recover a from P_new: a_new = ifft(sqrt(P_new) * phase(fft(a)))
4. Project onto simplex

This is similar to the Gerchberg-Saxton algorithm used in phase retrieval.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-parameterized Adam optimization on LSE-smoothed max autocorrelation
2. Iterative spectral flattening (Gerchberg-Saxton style)
3. Peak-targeted local search with coordinate updates
4. Multi-start with diverse structured initializations
5. Multi-length exploration
6. Coordinate descent and pair-exchange polish
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad(a, beta):
    """Compute LSE objective value and gradient w.r.t. a for conv(a,a)."""
    conv = fast_convolve(a)
    m = conv.max()
    scaled = beta * (conv - m)
    scaled -= scaled.max()
    e = np.exp(scaled)
    S = e.sum()
    p = e / S
    # LSE = m + log(S) / beta (approximately, for numerical stability)
    # Actual value: m + (scaled.max() + log(S)) / beta
    obj = m + (0.0 + np.log(S)) / beta
    # Gradient: d/d(a_i) sum_k p_k * conv_k = 2 * correlate(p, a)
    grad_a = 2.0 * np.correlate(p, a, mode='valid')[:len(a)]
    return obj, grad_a


def transform_grad_to_z(grad_a, a):
    """Transform gradient from a-space to z-space where a = softmax(z)."""
    return a * grad_a - a * np.dot(grad_a, a)


def adam_log_optimize(a0, time_limit, lr=0.01, beta_lse=50.0,
                      beta1=0.8, beta2=0.999, eps=1e-8,
                      lr_decay=0.9995, beta_anneal=0.995):
    """
    Adam optimization on log-parameterization.
    Minimize LSE(beta * conv(a,a)) where a = softmax(z).
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    z = np.log(a0)

    m_ = np.zeros(n)
    v_ = np.zeros(n)
    t = 0

    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()

    t0 = time.monotonic()
    cur_lr = lr
    cur_beta = beta_lse
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        a = softmax_fn(z)
        _, grad_a = lse_obj_grad(a, cur_beta)
        conv = fast_convolve(a)
        actual_max = conv.max()

        if actual_max < best_obj - 1e-12:
            best_obj = actual_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Transform gradient to z-space
        grad_z = transform_grad_to_z(grad_a, a)

        # Adam update
        m_ = beta1 * m_ + (1 - beta1) * grad_z
        v_ = beta2 * v_ + (1 - beta2) * grad_z ** 2
        m_hat = m_ / (1 - beta1 ** t)
        v_hat = v_ / (1 - beta2 ** t)
        z -= cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Schedule
        cur_lr *= lr_decay
        cur_beta = min(cur_beta * (1 + 0.001), 500.0)  # Gradually sharpen

        # Restart if stuck too long
        if no_improve > 500:
            noise_scale = 0.005
            a_pert = best_a + np.random.default_rng(t * 31).standard_normal(n) * noise_scale
            a_pert = project_simplex(a_pert)
            z = np.log(np.maximum(a_pert, 1e-10))
            m_ = np.zeros(n)
            v_ = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.001)
            no_improve = 0

    return best_a, best_obj


def spectral_flatten(a0, time_limit, alpha_init=0.6, alpha_min=0.3):
    """
    Iterative spectral flattening: shape the power spectrum to reduce autocorrelation peaks.
    """
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    alpha = alpha_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        H = np.fft.fft(a, N)
        abs_H = np.abs(H)
        abs_H[abs_H < 1e-30] = 1e-30
        phase = H / abs_H

        P = abs_H ** 2
        # Shape: compress dynamic range
        P_max = P.max()
        P_min = P.min()
        if P_max > 0:
            P_norm = P / P_max
            P_new = P_norm ** alpha * P_max

        H_new = np.sqrt(P_new) * phase
        a_new = np.real(np.fft.ifft(H_new, N)[:n])
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        alpha = max(alpha * 0.998, alpha_min)

        if no_improve > 300:
            # Perturb and restart
            rng = np.random.default_rng(it * 17)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            alpha = alpha_init
            no_improve = 0

    return best_a, best_obj


def peak_local_search(a0, time_limit):
    """
    Local search focused on reducing top autocorrelation peaks.
    For each peak, try to redistribute mass among contributing elements.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 80:
        it += 1
        conv = fast_convolve(a)
        current_max = conv.max()

        top_k = min(15, 2 * n - 1)
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        improved = False

        for k in peak_idx[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]

            # Try reducing top contributors
            order = np.argsort(contribs)[::-1]

            for oi in order[:4]:
                i = idx[oi]
                j = k - i
                if i == j:
                    # Diagonal: reduce a[i], spread to neighbors
                    for frac in [0.03, 0.07, 0.15]:
                        new_a = a.copy()
                        amt = a[i] * frac
                        new_a[i] = a[i] - amt
                        # Spread to nearby indices
                        spread = []
                        for d in [1, 2, -1, -2]:
                            nb = i + d
                            if 0 <= nb < n:
                                spread.append(nb)
                        if spread:
                            for nb in spread:
                                new_a[nb] += amt / len(spread)
                        new_a = np.maximum(new_a, 0)
                        ns = new_a.sum()
                        if ns < 1e-10:
                            continue
                        new_a /= ns
                        new_obj = fast_convolve(new_a).max()
                        if new_obj < best_obj - 1e-12:
                            a = new_a
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                else:
                    # Off-diagonal: transfer between a[i] and a[j]
                    for frac in [0.03, 0.07, 0.15, 0.25]:
                        for direction in [1, -1]:
                            new_a = a.copy()
                            amt = min(a[i], a[j]) * frac
                            if direction == 1:
                                new_a[i] = a[i] - amt
                                new_a[j] = a[j] + amt
                            else:
                                new_a[i] = a[i] + amt
                                new_a[j] = a[j] - amt
                            new_a = np.maximum(new_a, 0)
                            ns = new_a.sum()
                            if ns < 1e-10:
                                continue
                            new_a /= ns
                            new_obj = fast_convolve(new_a).max()
                            if new_obj < best_obj - 1e-12:
                                a = new_a
                                best_obj = new_obj
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break

            if not improved:
                # Random pair exchange with top-peak contributors
                rng = np.random.default_rng(it * 104729 + k)
                for _ in range(8):
                    j_rand = int(rng.integers(0, n))
                    i_src = idx[order[0]]
                    if j_rand == i_src:
                        continue
                    for frac in [0.03, 0.08]:
                        amt = a[i_src] * frac
                        new_a = a.copy()
                        new_a[i_src] = a[i_src] - amt
                        new_a[j_rand] = a[j_rand] + amt
                        new_a = np.maximum(new_a, 0)
                        ns = new_a.sum()
                        if ns < 1e-10:
                            continue
                        new_a /= ns
                        new_obj = fast_convolve(new_a).max()
                        if new_obj < best_obj - 1e-12:
                            a = new_a
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    order = np.argsort(a)[::-1]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = a[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0, 0.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                remaining = 1.0 - new_val
                old_remaining = 1.0 - old_val
                if remaining < 1e-10 or old_remaining < 1e-10:
                    continue
                scale = remaining / old_remaining
                new_a = a.copy()
                new_a[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] = a[mask] * scale
                new_a = np.maximum(new_a, 0)
                ns = new_a.sum()
                if ns < 1e-10:
                    continue
                new_a /= ns
                obj = fast_convolve(new_a).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                new_val = best_j_val
                remaining = 1.0 - new_val
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10 and remaining > 1e-10:
                    scale = remaining / old_remaining
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[j] = new_val
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    if best_j_obj < best_obj - 1e-12:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True

        if not improved:
            break

    return best_a, best_obj


def pair_exchange_search(f0, time_limit, max_trials_per_round=300):
    """Pair-exchange local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)

        # Focus on indices that contribute to top peaks
        top_k = min(10, 2 * n - 1)
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]

        # Candidate source indices (high contributors to peaks)
        src_set = set()
        for k in peak_idx:
            k = int(k)
            js = max(0, k - n + 1)

            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            top_contrib = idx[np.argsort(contribs)[-3:]]
            src_set.update(top_contrib.tolist())
        src_list = list(src_set)[:30]

        rng = np.random.default_rng(int(time.monotonic() * 1e6) % 10000000)

        for trial in range(max_trials_per_round):
            if time.monotonic() - t0 > time_limit:
                break
            if src_list:
                i = int(rng.choice(src_list))
            else:
                i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            old_i = f[i]
            if old_i < 1e-12:
                continue
            for frac in [0.1, 0.3, 0.5]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] += amt
                f = np.maximum(f, 0)
                fs = f.sum()
                if fs < 1e-10:
                    f[i] = old_i
                    f[j] -= amt
                    break
                f /= fs
                val = fast_convolve(f).max()
                if val < best_obj - 1e-12:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] -= amt
                    break

        if not improved:
            break

    return best_f, best_f if False else best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    inits.append(np.maximum(hann, 1e-10) / np.maximum(hann, 1e-10).sum())

    center = n / 2.0
    sigma = n / 5.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Edge-suppressed
    edge = np.ones(n)
    edge[0] = 0.3
    edge[-1] = 0.3
    inits.append(edge / edge.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kl = max(3, n // 50)
    kernel = np.ones(kl) / kl
    r = np.convolve(r, kernel, mode='same')
    inits.append(np.maximum(r, 1e-10) / np.maximum(r, 1e-10).sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(np.maximum(rc, 0.01) / np.maximum(rc, 0.01).sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Adam on LSE, spectral flattening,
    peak local search, coordinate descent, and pair exchange.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def _try(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print("  NEW BEST [{}]: {:.6f} (n={})".format(label, score, len(a)))

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent, using uniform n=512")

    # === Phase 1: Quick multi-start exploration ===
    print("\n=== Phase 1: Multi-Start Exploration ===")
    all_inits = [parent]
    for n_try in [512, 768, 1024]:
        if n_try == len(parent):
            continue
        all_inits.extend(structured_inits(n_try, rng)[:3])

    n_inits = len(all_inits)
    phase1_time = 0.30 * (budget_s - 25)
    per_init_time = phase1_time / n_inits

    for idx, init in enumerate(all_inits):
        rem = deadline - time.monotonic() - 0.4 * (budget_s - 25)
        if rem < 15:
            break
        t_alloc = min(per_init_time, rem / max(1, n_inits - idx), 60)
        t_alloc = max(t_alloc, 8)

        # Alternate between Adam and spectral
        if idx % 2 == 0:
            a_opt, _ = adam_log_optimize(init, time_limit=t_alloc,
                                          lr=0.02, beta_lse=30.0, lr_decay=0.998)
        else:
            a_opt, _ = spectral_flatten(init, time_limit=t_alloc, alpha_init=0.5)
        _try(a_opt, "p1_{}".format(idx))

    # === Phase 2: Deep optimization on best ===
    print("\n=== Phase 2: Deep Optimization (best={:.6f}) ===".format(best_score))
    if best_seq is not None:
        n_best = len(best_seq)

        # Adam with various configs
        configs = [
            {"lr": 0.01, "beta": 60.0, "decay": 0.999, "perturb": 0.0},
            {"lr": 0.005, "beta": 100.0, "decay": 0.9995, "perturb": 0.008},
            {"lr": 0.015, "beta": 40.0, "decay": 0.9985, "perturb": 0.005},
            {"lr": 0.003, "beta": 150.0, "decay": 0.9997, "perturb": 0.0},
        ]

        phase2_time = 0.35 * (budget_s - 25)
        per_run = phase2_time / (len(configs) + 2)

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 0.2 * (budget_s - 25)
            if rem < 15:
                break
            t_alloc = min(per_run, rem / max(1, len(configs) - p + 2), 180)

            if cfg["perturb"] > 0:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * cfg["perturb"])
            else:
                f_init = best_seq.copy()

            print("  Adam {}: n={}, t={:.0f}s".format(p, n_best, t_alloc))
            a_opt, _ = adam_log_optimize(f_init, time_limit=t_alloc,
                                          lr=cfg["lr"], beta_lse=cfg["beta"],
                                          lr_decay=cfg["decay"])
            _try(a_opt, "p2_adam_{}".format(p))

        # Spectral flattening
        rem = deadline - time.monotonic() - 0.15 * (budget_s - 25)
        if rem > 15:
            t_sp = min(per_run, rem, 120)
            print("  Spectral: t={:.0f}s".format(t_sp))
            a_opt, _ = spectral_flatten(best_seq, time_limit=t_sp, alpha_init=0.5)
            _try(a_opt, "p2_spectral")

            # Follow up with Adam on spectral result
            rem = deadline - time.monotonic() - 0.1 * (budget_s - 25)
            if rem > 10:
                a_opt2, _ = adam_log_optimize(a_opt, time_limit=min(rem, 60),
                                               lr=0.008, beta_lse=80.0)
                _try(a_opt2, "p2_spec_adam")

        # Peak local search
        rem = deadline - time.monotonic() - 0.1 * (budget_s - 25)
        if rem > 15:
            t_pls = min(rem * 0.5, 100)
            print("  Peak LS: t={:.0f}s".format(t_pls))
            a_opt, _ = peak_local_search(best_seq, time_limit=t_pls)
            _try(a_opt, "p2_peakls")

            # Fine Adam after peak LS
            rem = deadline - time.monotonic() - 0.05 * (budget_s - 25)
            if rem > 10:
                a_opt2, _ = adam_log_optimize(a_opt, time_limit=min(rem, 50),
                                               lr=0.004, beta_lse=120.0)
                _try(a_opt2, "p2_peakls_adam")

    # === Phase 3: Length exploration ===
    print("\n=== Phase 3: Length Exploration (best={:.6f}) ===".format(best_score))
    if best_seq is not None:
        target_lengths = [640, 896, 1024, 1280, 1536]
        for target_n in target_lengths:
            if target_n == len(best_seq):
                continue
            rem = deadline - time.monotonic() - 0.05 * (budget_s - 25)
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 90)
            print("  n={}, t={:.0f}s".format(target_n, t_alloc))
            a_opt, _ = adam_log_optimize(f_interp, time_limit=t_alloc,
                                          lr=0.01, beta_lse=50.0)
            _try(a_opt, "p3_n{}".format(target_n))

    # === Phase 4: Final polish ===
    print("\n=== Phase 4: Final Polish (best={:.6f}) ===".format(best_score))
    if best_seq is not None:
        # Coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_cd = min(rem * 0.4, 60)
            print("  Coord descent: t={:.0f}s".format(t_cd))
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            _try(a_opt, "p4_cd")

        # Pair exchange
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_pex = min(rem * 0.5, 40)
            print("  Pair exchange: t={:.0f}s".format(t_pex))
            a_opt, _ = pair_exchange_search(best_seq, time_limit=t_pex)
            _try(a_opt, "p4_pex")

        # Final fine Adam
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            a_opt, _ = adam_log_optimize(best_seq, time_limit=min(rem, 25),
                                          lr=0.002, beta_lse=200.0, lr_decay=0.999)
            _try(a_opt, "p4_fine_adam")

        # One more peak LS
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = peak_local_search(best_seq, time_limit=min(rem, 20))
            _try(a_opt, "p4_final_pls")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print("\nFINAL SCORE: {:.6f} (n={})".format(best_score, len(best_seq)))
    return [float(x) for x in best_seq]
