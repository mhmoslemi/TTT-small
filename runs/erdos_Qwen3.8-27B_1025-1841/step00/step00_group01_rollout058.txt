from scipy.optimize import minimize
import time


def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _obj(x, n):
    return np.max(np.correlate(x, 1.0 - x, mode="full")) * (2.0 / n)


def _eq(x, n):
    return np.sum(x) - n * 0.5


def _fix(h, n):
    t = n * 0.5

        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(m):

        else:
            h = h + d / n

    b = np.linspace(0.0, 2.0, n, endpoint=False)
