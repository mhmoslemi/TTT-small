def make_binary(h):

    n = len(h)

import numpy as np
import time

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def steepest_descent(h, deadline):
    n = len(h)
    dx = 2.0 / n
    c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    for _ in range(500):
        if time.time() > deadline - 30:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_val = c5
        best_move = None
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                val = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
                if val < best_val - 1e-15:
                    best_val = val
                    best_move = (i, j)
                h[j] = 0.0
            h[i] = 1.0
        if best_move is None:
            break
        i, j = best_move
        h[i] = 0.0
        h[j] = 1.0
        c5 = best_val
    return h, c5

    rng = np.random.RandomState(seed)

    h_init = None

        if time.time() > deadline - 60:

        h_work = best_h.copy()
