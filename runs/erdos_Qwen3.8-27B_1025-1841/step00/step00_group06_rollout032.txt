import numpy as np
import math
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_binary(h, n):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        idx = np.where(h == 1.0)[0]
        np.random.shuffle(idx)
        for i in idx[:num_ones - target]:
            h[i] = 0.0
    elif num_ones < target:
        idx = np.where(h == 0.0)[0]
        np.random.shuffle(idx)
        for i in idx[:target - num_ones]:
            h[i] = 1.0
    return h

def golden_ratio_seq(n):
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    fracs = (np.arange(n, dtype=float) * phi) % 1.0
    return (fracs < 0.5).astype(float)

def fibonacci_word(n):
    a, b = np.array([0.0]), np.array([1.0])
    while len(b) < n:
        a, b = b, np.concatenate([b, a])
    return b[:n].copy()

def quick_local_search(h, n, budget_s, rng):
    c5 = compute_c5(h, n)
    T = 0.003
    cooling = 0.9995

            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - c5
        if delta < 0 or (T > 1e-10 and rng.random() < math.exp(-delta / T)):
            c5 = new_c5
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling
    return h, c5
