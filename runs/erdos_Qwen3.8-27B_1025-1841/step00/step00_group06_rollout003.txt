def smooth_max(x, t):
    xmax = np.max(x)
    return xmax + math.log(np.sum(np.exp(t * (x - xmax)))) / t

def repair_h(h, n):

    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)

            if cnt > 0:
