            for delta in (0.012, -0.012, 0.035, -0.035, 0.006, -0.006):
                h_try = h_ld.copy()
                h_try[i1] += delta
                h_try = project_sum(np.clip(h_try, 0, 1), n_ld)
                c5 = compute_c5(h_try)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_try.copy()
                    h_ld = h_try.copy()
                    improved = True
                    break
            if not improved and it % 3 == 0:
                i2 = rng.randint(n_ld)
                if i2 != i1:
                    h_try = h_ld.copy()
                    tmp = h_try[i1]
                    h_try[i1] = h_try[i2]
                    h_try[i2] = tmp
                    h_try = project_sum(h_try, n_ld)
                    c5 = compute_c5(h_try)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_try.copy()
                        h_ld = h_try.copy()

    print(f"Final: n={best_n}, C5={best_c5:.7f}")
    return best_h, best_c5, best_n
