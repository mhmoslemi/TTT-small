            zeros = np.where(h == 0)[0]

            h[j] = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return np.max(corr) * dx

def fix_sum_binary(h, target):
    h = h.copy().astype(float)
    s = np.sum(h)
    while s > target:
        i = np.where(h == 1)[0]
        j = np.where(h == 0)[0]
        if len(i) == 0 or len(j) == 0:
            break
