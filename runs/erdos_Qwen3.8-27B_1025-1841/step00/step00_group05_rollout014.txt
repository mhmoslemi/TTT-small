import time

def _eval_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        h[:] = 0.5
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        n_fix = max(1, int(np.ceil(abs(diff))))
        if diff > 0:
            idx = np.argsort(-h)[:n_fix]
        else:
            idx = np.argsort(h)[:n_fix]
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def _cosine_h(n, a):

    return 0.5 + a * np.cos(2.0 * np.pi * x)
