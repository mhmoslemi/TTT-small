    return np.max(corr) * dx

def project_to_constraint(h, n):

def halton_sequence(n, base):
    """Generate Halton sequence."""
    result = np.zeros(n)
    for i in range(1, n + 1):
        f = 1.0
        r = 0.0
        j = i
        while j > 0:
            f /= base
            r += f * (j % base)
            j //= base
        result[i-1] = r
    return result

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full").astype(np.float64)

def delta_corr(h, i, j):
    n = len(h)
    delta = np.zeros(2 * n - 1)
    delta[n-1-i : 2*n-1-i] -= h
    delta[i : n+i] += (h[::-1] - 1.0)
    delta[n-1-j : 2*n-1-j] -= h
    delta[j : n+j] += (1.0 - h[::-1])
    return delta

def fix_sum_binary(h, n, rng):
    target = n // 2
    current = int(np.sum(h))
    while current > target:
        ones = np.where(h == 1)[0]
        h[ones[rng.integers(len(ones))]] = 0
        current -= 1
    while current < target:
        zeros = np.where(h == 0)[0]
        h[zeros[rng.integers(len(zeros))]] = 1
        current += 1
    return h

def halton_vocab(n, base):
    out = np.zeros(n)
    for i in range(1, n + 1):
        f = 1.0
        r = 0.0
        j = i
        while j > 0:
            f /= base
            r += f * (j % base)
            j //= base
        out[i - 1] = r
    return out

def make_halton_binary(n, seed):
    rng = np.random.default_rng(seed)
    base = rng.integers(2, 20)
    seq = halton_vocab(n, base)
    order = np.argsort(seq)
    h = np.zeros(n)
    h[order[:n // 2]] = 1
    return h

    """Create a quasi-structured binary sequence."""
    rng = np.random.default_rng(seed)

    dist = np.abs(positions - 0.5)
    order = np.argsort(dist)
    h = np.zeros(n)
    h[order[:n // 2]] = 1
    return h
