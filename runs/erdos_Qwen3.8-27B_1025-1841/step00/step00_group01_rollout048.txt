Actually for same-length arrays:
correlate(a, v, mode='full')[k] = sum_n a[n] * v[n + k - (N-1)]

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed())

    def eval_c5(h, n):
        dx = 2.0 / n
        return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    def fix_constraints(h, n):
        h = np.array(h, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        for _ in range(200):
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if not mask.any():
                break
            h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
        return h

    def interpolate_to(h_src, n_src, n_dst):
        if n_dst == n_src:
            return fix_constraints(h_src, n_src)
        x_src = np.arange(n_src, dtype=float)
        x_dst = np.linspace(0.0, n_src - 1.0, n_dst)
        h_dst = np.interp(x_dst, x_src, h_src)
        return fix_constraints(h_dst, n_dst)

    def slsqp_smooth(h_init, n, alpha, maxiter):
        target = n / 2.0
        dx = 2.0 / n

        def obj(h):
            c = np.correlate(h, 1 - h, mode='full') * dx
            m = float(np.max(c))
            shifted = c - m
            val = m + float(np.log(np.sum(np.exp(alpha * shifted)))) / alpha
            return val

        cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target,
                 'jac': lambda h: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            return fix_constraints(res.x, n)
        except Exception:
            return fix_constraints(h_init, n)

    def slsqp_max(h_init, n, maxiter):
        target = n / 2.0
        dx = 2.0 / n

        def obj(h):
            c = np.correlate(h, 1 - h, mode='full')
            return float(np.max(c)) * dx

        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_init, method='SLSQP', bounds=bounds,

            return fix_constraints(res.x, n)

            h[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            c = eval_c5(h, n)
            if c < best_c5:
                best_c5 = c
                stagnant = 0
            else:
                h[i] = 1.0
                h[j] = 0.0
                ones.remove(j)
                zeros.remove(i)
                zeros.append(j)
                ones.append(i)
                stagnant += 1
