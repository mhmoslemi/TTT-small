def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, dx):
    """Clip to [0,1] and scale to meet integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h) * dx
    if s < 1e-10:
        return h
    h = h * (1.0 / s)
    h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, dx, n, maxiter=200, ftol=1e-10):
    """Minimize C5 using SLSQP with t-auxiliary variable."""
    t0 = compute_c5(h0, dx) + 0.05
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    def obj(params):
        return params[n]
    
    def corr_ineq(params):
        h = params[:n]
        t = params[n]
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - c
    
    def sum_eq(params):
        return np.sum(params[:n]) * dx - 1.0
    
    constraints = [
        {'type': 'ineq', 'fun': corr_ineq},
        {'type': 'eq', 'fun': sum_eq}
    ]
    
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                  constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
    
    h_opt = res.x[:n]
    h_opt = repair_h(h_opt, dx)
    return h_opt, compute_c5(h_opt, dx)

def binary_local_search(h, dx, n, time_limit=120, max_rounds=50):
    """Swap-based local search for binary h."""
    t_start = time.time()
    best = h.copy()
    best_val = compute_c5(best, dx)

        if time.time() - t_start > time_limit:
            break
        improved = False
        ones = np.where(best >= 0.5)[0]
        zeros = np.where(best < 0.5)[0]

                    ones = np.where(best >= 0.5)[0]
                    zeros = np.where(best < 0.5)[0]
                else:
                    best[oi], best[zi] = best[zi], best[oi]
        
        if not improved:
            break

    s = np.sum(h) * dx

def repair(h, dx):
    h = np.clip(h, 0.0, 1.0)
    target = 1.0 / dx  # = n/2
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Scale, then adjust individual elements to fix sum
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # Fine-adjust: distribute remaining error
    diff = target - np.sum(h)
    if diff > 0:
        # Need to increase some elements
        room = 1.0 - h  # how much each can increase
        order = np.argsort(-room)
        for i in order:
            inc = min(diff, room[i])
            h[i] += inc
            diff -= inc
            if diff < 1e-12: break
    elif diff < 0:

        for i in order:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
            if diff > -1e-12: break
    return h
