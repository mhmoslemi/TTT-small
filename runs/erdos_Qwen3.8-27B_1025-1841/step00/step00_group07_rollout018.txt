                h_s = _project(h_base + rng.randn(n_pts) * 0.12, n_pts)
            else:
                h_s = _project(rng.rand(n_pts), n_pts)

            maxiter = 300 if n_pts <= 120 else 200
            h_cand = _slsqp_optimize(h_s, n_pts, maxiter)
            c5_cand = _compute_c5(h_cand)

            if c5_cand < best_c5:
                best_h = h_cand.copy()
                best_c5 = c5_cand
                h_base = h_cand.copy()

        if time.time() - t0 > budget_s * 0.8:
            break

        if len(best_h) == n_pts:
            h_ref, c5_ref = _gradient_refine(best_h, n_pts, n_steps=80)
            if c5_ref < best_c5:
                best_h = h_ref.copy()
                best_c5 = c5_ref

    return best_h, best_c5, len(best_h)
