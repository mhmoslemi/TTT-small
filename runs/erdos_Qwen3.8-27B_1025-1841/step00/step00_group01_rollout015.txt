    corr = np.correlate(h, 1.0 - h, mode="full")

    return np.correlate(h, 1.0 - h, mode="full")

def compute_subgradient(h, dx, n):
    corr = compute_full_corr(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    else:
        j = -k
        grad[j:] += (1.0 - h[:n-j])
        grad[:n-j] -= h[j:]
    return grad * dx

def project(h, n):

        if cnt == 0:
