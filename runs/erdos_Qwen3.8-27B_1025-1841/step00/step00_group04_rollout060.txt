import time

    corr = np.correlate(h, 1.0 - h, mode='full')

    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    # Log-sum-exp for smooth max
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

    return compute_c5(h, n)

    def c(h):
        return np.sum(h) - target
    return c

One important detail: the correlate includes all shifts from -(n-1) to (n-1). For shifts near the boundaries, the integral is over a shorter interval (not the full [0,2]). The max might occur at a boundary shift or at shift 0.

For shift n-1 (or -(n-1)): only one term: h[0]*(1-h[n-1]) * dx (or similar)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    exp_vals = np.exp(alpha * (vals - m))
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        j = k - (n - 1)
        if j >= 0:
            grad[j:] += wk * (1.0 - h[:n - j])
            grad[:n - j] -= wk * h[j:]
        else:
            jj = -j
            grad[:n - jj] += wk * (1.0 - h[jj:])
            grad[jj:] -= wk * h[:n - jj]
    grad *= dx
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h = h + diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h = h + diff * h / s
    return np.clip(h, 0.0, 1.0)

def make_obj(n, alpha):
    def obj(x):
        return smooth_max_val(x, n, alpha)
    return obj

def make_grad(n, alpha):
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    return grad
