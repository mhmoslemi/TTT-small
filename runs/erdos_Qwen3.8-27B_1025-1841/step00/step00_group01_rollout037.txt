```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, n):
    """Project h to satisfy sum(h) = n/2, keeping 0 <= h <= 1"""
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        frac = diff / n
        h_new = h + frac
        if np.all(h_new >= 0) and np.all(h_new <= 1):
            h = h_new
        else:
            # Water-filling approach
            if diff > 0:
                # Need to increase; add to elements with room
                mask = h < 1.0
                if not np.any(mask):
                    break
                h[mask] += diff / np.sum(mask)
            else:
                mask = h > 0.0
                if not np.any(mask):
                    break
                h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def resample(h, n_new):
    """Resample h from its current resolution to n_new"""
    n_old = len(h)
    x_old = np.arange(n_old) / n_old  # in [0, 1)
    x_new = np.arange(n_new) / n_new
    # Linear interpolation with wrap-around
    h_new = np.interp(x_new, x_old, h, period=1.0)
    return h_new

def smooth_max_value(corr, temperature):
    """Compute smooth max of corr array"""
    if temperature <= 0:
        return np.max(corr)
    scaled = corr / temperature
    a = np.max(scaled)
    return temperature * (a + np.log(np.mean(np.exp(scaled - a))))
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        frac = diff / n
        h2 = h + frac
        if np.all(h2 >= -1e-15) and np.all(h2 <= 1.0 + 1e-15):
            h = h2
        else:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
```
