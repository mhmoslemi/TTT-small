def smooth_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 0)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

        mid = (lo + hi) / 2.0

        else:

    h_val = np.clip(y - (lo + hi) / 2.0, 0, 1)

        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
