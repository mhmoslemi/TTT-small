from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_fn(x, n_points, alpha, pen_w):
    n = n_points
    dx = 2.0 / n
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    sm = (corr_max + np.log(np.sum(np.exp(alpha * (corr - corr_max))))) / alpha
    return sm * dx + pen_w * (np.sum(h) - n / 2.0) ** 2

def grad_fn(x, n_points, alpha, pen_w):
    n = n_points
    dx = 2.0 / n
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    exp_v = np.exp(alpha * (corr - corr_max))
    exp_sum = np.sum(exp_v)
    weights = exp_v / exp_sum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-16:
            continue
        offset = k - (n - 1)
        # First term: +w*(1-h[m-offset]) for valid m
        lo = max(0, offset)
        hi = min(n, n + offset)
        if lo < hi:
            idx = np.arange(lo, hi) - offset
            grad[lo:hi] += w * (1.0 - h[idx])
        # Second term: -w*h[m+offset] for valid m
        lo2 = max(0, -offset)
        hi2 = min(n, n - offset)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2) + offset
            grad[lo2:hi2] -= w * h[idx2]
    
    grad *= dx
    grad += 2.0 * pen_w * (np.sum(h) - n / 2.0)
    return grad

                h[i] += step
                remain -= step
                if remain < 1e-13:
                    break

            order = np.argsort(h)  # least value first
            remain = -diff
            for i in order:
                step = min(h[i], remain)
                h[i] -= step
                remain -= step
                if remain < 1e-13:
                    break
    return np.clip(h, 0.0, 1.0)

import numpy as np

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_fn(x, n_pts, alpha, pen_w):
    n = n_pts
    dx = 2.0 / n
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sm = (cmax + np.log(np.sum(np.exp(alpha * (corr - cmax))))) / alpha
    return sm * dx + pen_w * (np.sum(h) - n / 2.0) ** 2

def grad_fn(x, n_pts, alpha, pen_w):
    n = n_pts
    dx = 2.0 / n
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    ev = np.exp(alpha * (corr - cmax))
    es = np.sum(ev)
    w = ev / es
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        if wk < 1e-16:
            continue
        off = k - (n - 1)
