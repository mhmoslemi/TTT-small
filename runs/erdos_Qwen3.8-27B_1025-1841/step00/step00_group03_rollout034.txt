def compute_smoothed_grad(h, n, eps):

    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    g_max = np.max(g)
    exp_g = np.exp((g - g_max) / eps)
    p = exp_g / np.sum(exp_g)
    
    grad = np.zeros(n)
    # d = 0
    grad += p[n-1] * (1 - 2*h)
    # d > 0
    for d in range(1, n):
        w = p[n-1+d]
        if w < 1e-15:
            continue
        # term1: 1-h[i+d] for i=0..n-1-d
        grad[:n-d] += w * (1 - h[d:])
        # term2: -h[i-d] for i=d..n-1
        grad[d:] += w * (-h[:-d])
    # d < 0
    for d in range(1, n):
        w = p[n-1-d]
        if w < 1e-15:
            continue
        # d_actual = -d
        # term1: 1-h[i-d] for i=d..n-1
        grad[d:] += w * (1 - h[:-d])
        # term2: -h[i+d] for i=0..n-1-d
        grad[:n-d] += w * (-h[d:])
    
    grad *= dx
    return grad

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_grad(h, n, eps):
    """Compute gradient of smoothed objective using log-sum-exp."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    g_shifted = g - np.max(g)
    exp_g = np.exp(g_shifted / eps)
    Z = np.sum(exp_g)
    p = exp_g / Z  # length 2n-1
    
    grad = np.zeros(n)
    # d=0: contribution is (1-2h)
    grad += p[n-1] * (1.0 - 2.0 * h)
    # d>0
    for d in range(1, n):
        w1 = p[n-1+d]  # weight for shift +d
        w2 = p[n-1-d]  # weight for shift -d
        if w1 + w2 < 1e-15:
            continue
        # For shift d: grad[:n-d] += w1*(1-h[d:]), grad[d:] += w1*(-h[:-d])
        # For shift -d: grad[d:] += w2*(1-h[:-d]), grad[:n-d] += w2*(-h[d:])
        grad[:n-d] += w1 * (1.0 - h[d:]) + w2 * (-h[d:])
        grad[d:] += w1 * (-h[:-d]) + w2 * (1.0 - h[:-d])
    grad *= dx
    return grad

def local_search(h, n, max_iters=1000):
    """Coordinate descent / pairwise swap local search."""
    best_c5 = compute_c5(h)
    h = h.copy()
    for _ in range(max_iters):
        improved = False
        # Try pairwise adjustments
        for i in range(n):
            for j in range(i+1, n):
                if abs(h[i] - h[j]) < 1e-6:
                    continue
                # Try transferring small amount from larger to smaller
                # or vice versa
                hi, hj = h[i], h[j]
                # Try a few transfer amounts
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    delta = frac * (hi - hj)
                    h_new_i = hi - delta
                    h_new_j = hj + delta
                    if 0 <= h_new_i <= 1 and 0 <= h_new_j <= 1:
                        h[i], h[j] = h_new_i, h_new_j
                        c5 = compute_c5(h)
                        if c5 < best_c5 - 1e-12:
                            best_c5 = c5
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
                # Also try the reverse
                if hi > hj:
                    for frac in [0.1, 0.25, 0.5]:
                        delta = frac * (hi - hj)
                        h_new_i = hi - delta
                        h_new_j = hj + delta
                        if 0 <= h_new_i <= 1 and 0 <= h_new_j <= 1:
                            h[i], h[j] = h_new_i, h_new_j
                            c5 = compute_c5(h)
                            if c5 < best_c5 - 1e-12:
                                best_c5 = c5
                                improved = True
                            else:
                                h[i], h[j] = hi, hj
        if not improved:

    return h, best_c5
