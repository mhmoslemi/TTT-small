import numpy as np
from scipy.optimize import minimize
import time

def make_feasible(h, n_points):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n_points * 0.5
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def obj_func(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr)) * dx

def sum_constraint(x):
    return np.sum(x) - 0.5 * len(x)

def resample(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(80, 0.5)

    h_best = make_feasible(h_parent, len(h_parent))
    best_c5 = compute_c5(h_best, len(h_best))
    best_n = len(h_best)

    n = best_n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]

    # Optimize at parent resolution with multiple restarts
    n_restarts = 25
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 120:
            break

        if r == 0:
            x0 = h_best.copy()
        elif r == 1:
            x0 = h_best + rng.normal(0, 0.01, n)
        elif r == 2:
            x0 = h_best + rng.normal(0, 0.03, n)
        elif r == 3:
            x0 = h_best + rng.normal(0, 0.06, n)
        elif r == 4:
            x0 = h_best + rng.normal(0, 0.1, n)
        elif r == 5:
            # Binary random
            x0 = (rng.random(n) > 0.5).astype(float)
        elif r == 6:
            # Block pattern
            block = max(1, n // 16)
            x0 = np.zeros(n)
            for b in range(0, n, 2 * block):
                x0[b:b+block] = 1.0
        elif r == 7:
            # Sinusoidal
            x0 = 0.5 + 0.4 * np.sin(2 * np.pi * np.arange(n) / n * 3)
        elif r == 8:
            # Symmetric random
            half = n // 2
            xh = rng.random(half)
            x0 = np.concatenate([xh, xh[::-1]])
        elif r == 9:
            # Perturbed binary
            x0 = (rng.random(n) > 0.5).astype(float)
            x0 += rng.normal(0, 0.15, n)
        else:
            x0 = h_best + rng.normal(0, 0.02 * (1 + r * 0.3), n)

        x0 = make_feasible(x0, n)

        maxiter = 2000 if r < 5 else 800
        try:
            res = minimize(obj_func, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h_opt = make_feasible(res.x, n)
            c5_opt = compute_c5(h_opt, n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                h_best = h_opt.copy()
        except Exception:
            pass

    # Try higher resolutions
    for n_high in [160, 240, 320, 480]:
        if time.time() - t0 > budget_s - 150:
            break

        h_res = resample(h_best, n_high)
        h_res = make_feasible(h_res, n_high)
        c5_res = compute_c5(h_res, n_high)
        if c5_res < best_c5:
            best_c5 = c5_res
            h_best = h_res.copy()
            best_n = n_high

        bounds_h = [(0.0, 1.0)] * n_high
        n_tries = 3 if n_high <= 240 else 2

        for tr in range(n_tries):
            if time.time() - t0 > budget_s - 60:
                break

            if tr == 0:
                x0 = h_res
            else:
                x0 = h_res + rng.normal(0, 0.02, n_high)
            x0 = make_feasible(x0, n_high)

            maxiter_h = 800 if n_high <= 240 else 300
            try:
                res = minimize(obj_func, x0, method='SLSQP',
                             bounds=bounds_h, constraints=cons,
                             options={'maxiter': maxiter_h, 'ftol': 1e-12})
                h_opt = make_feasible(res.x, n_high)
                c5_opt = compute_c5(h_opt, n_high)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    h_best = h_opt.copy()
                    best_n = n_high
            except Exception:
                pass

    # Final polish at best resolution
    if time.time() - t0 < budget_s - 30:
        n_f = best_n
        bounds_f = [(0.0, 1.0)] * n_f
        x0 = h_best.copy()
        try:
            res = minimize(obj_func, x0, method='SLSQP',
                         bounds=bounds_f, constraints=cons,
                         options={'maxiter': 3000, 'ftol': 1e-14})
            h_opt = make_feasible(res.x, n_f)
            c5_opt = compute_c5(h_opt, n_f)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                h_best = h_opt.copy()
                best_n = n_f
        except Exception:
            pass

    h_best = make_feasible(h_best, best_n)
    best_c5 = compute_c5(h_best, best_n)
    return h_best, best_c5, best_n
