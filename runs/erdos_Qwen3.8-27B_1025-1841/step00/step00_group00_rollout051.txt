import numpy as np
from time import time

def project_feasible(g, target_sum):
    """Project g onto {0<=h<=1, sum(h)=target_sum}."""
    n = len(g)
    # Binary search for lambda
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(g - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - mid, 0.0, 1.0)
    # Fine adjustment
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # Spread the difference
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # Verify
        h2_diff = target_sum - h.sum()
        if abs(h2_diff) > 1e-12:
            # Fix individual elements
            idx = np.argsort(-(h if h2_diff > 0 else 1-h))[:int(abs(h2_diff))+1]
            for i in idx:
                adjust = min(abs(h2_diff), 1.0 - h[i] if h2_diff > 0 else h[i])
                if h2_diff > 0:
                    h[i] = min(1.0, h[i] + adjust)
                    h2_diff -= adjust
                else:
                    h[i] = max(0.0, h[i] - adjust)
                    h2_diff += adjust
    return h

def smooth_max(corr, tau):
    """Smoothed maximum of array."""
    m = corr.max()
    return m + np.log(np.exp((corr - m) / tau).sum()) * tau

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def gradient_smooth_max(h, tau):
    """Gradient of smooth max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    # tau for smooth max
    weights = np.exp((c - c.max()) / tau)
    weights /= weights.sum()
    
    # Compute gradient as weighted sum of individual gradients
    # f_k = sum_{valid i} h[i]*(1-h[i+k])
    # grad_j f_k = (1-h[j+k])*[j+k valid] - h[j-k]*[j-k valid]  for shift k
    # where shift k = m - (n-1) for full correlation index m
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        shift = k - (n - 1)
        w = weights[k]
        if w < 1e-10:
            continue
        if shift >= 0:
            # f = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])
            grad[:n-shift] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:n-shift]
        else:
            s = -shift
            # f = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
    
    return grad * dx

def project_feasible(g, target_sum):
    """Project g onto {0<=h<=1, sum(h)=target_sum} via binary search on lambda."""
    n = len(g)
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(g - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - mid, 0.0, 1.0)

    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

import numpy as np
from time import time

def project_feasible(g, target_sum):
    n = len(g)
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(g - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - mid, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_score(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def grad_smooth_max(h, tau, corr=None):
    n = len(h)
    dx = 2.0 / n
    if corr is None:
        corr = compute_corr(h)
    m = corr.max()
    weights = np.exp((corr - m) / tau)
    sw = weights.sum()
    if sw < 1e-300:
        weights = np.zeros_like(corr)
        weights[corr.argmax()] = 1.0
    else:
        weights /= sw
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            s = -k
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
    return grad * dx

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h.copy(), n_new / 2.0)
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new, n_new / 2.0)

def pgd_optimize(h, n, max_time, tau_start=0.05, tau_end=0.003, step_init=0.5):
    target_sum = n / 2.0
    best_h = h.copy()
    best_s = compute_score(h)
    h = h.copy()
    t_start = time()
    step = step_init
    prev_s = best_s
    it = 0
    while time() - t_start < max_time:
        it += 1
        frac = min(it / 2000.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        corr = compute_corr(h)
        grad = grad_smooth_max(h, tau, corr)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-12:
            break
        alpha = step
        improved = False
        for ls in range(15):
            h_new = project_feasible(h - alpha * grad, target_sum)
            s_new = compute_score(h_new)
            if s_new < prev_s or s_new < best_s:
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
        h = h_new
        prev_s = s_new
        if s_new < best_s:
            best_s = s_new
            best_h = h.copy()
            step = min(step * 1.1, 3.0)
        else:
            step = max(alpha * 1.2, step * 0.8)
    return best_h, best_s

def sa_refine(h, n, max_time, t_init=0.03):
    target_sum = n / 2.0
    best_h = h.copy()
    best_s = compute_score(h)
    h = h.copy()
    cur_s = best_s
    t = t_init
    t_start = time()
    while time() - t_start < max_time:
        n_perturb = max(2, n // 30)
        h_new = h.copy()
        idx = np.random.randint(0, n, size=n_perturb)
        h_new[idx] += np.random.uniform(-0.15, 0.15, size=n_perturb)
        h_new = project_feasible(h_new, target_sum)
        s_new = compute_score(h_new)
        if s_new < cur_s or np.random.random() < np.exp(-(s_new - cur_s) / max(t, 1e-10)):
            h = h_new
            cur_s = s_new
            if s_new < best_s:
                best_s = s_new
                best_h = h.copy()
        t *= 0.9997
    return best_h, best_s

def make_binary_seed(n, seed_val):
    rng = np.random.default_rng(seed_val)
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_golden_seed(n, offset=0.0):
    h = np.zeros(n)
    alpha = (1.0 + 5.0 ** 0.5) / 2.0
    for i in range(n):
        t = (i + offset) * alpha
        h[i] = 1.0 if (t - int(t)) < 0.5 else 0.0
    return project_feasible(h, n / 2.0)
