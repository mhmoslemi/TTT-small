import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(t * (corr - m)))) / t)

        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_max(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def obj_smooth(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(t * (corr - m)))) / t)

def con_fun(x, n):
    return np.sum(x) - n / 2.0

def con_jac(x, n):
    return np.ones(n)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def make_random_binary(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(80, 0.5)

    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = evaluate_c5(best_h)
    best_n = n_parent

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP at parent resolution
    n = n_parent
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n,)}]

    for r in range(25):
        if remaining() < 60:
            break

        if r == 0:
            h0 = best_h.copy()
        elif r == 1:
            h0 = best_h + 0.02 * rng.randn(n)
        elif r == 2:
            h0 = best_h + 0.05 * rng.randn(n)
        elif r == 3:
            h0 = best_h + 0.1 * rng.randn(n)
        elif r == 4:
            h0 = best_h + 0.2 * rng.randn(n)
        elif r == 5:
            h0 = make_random_binary(n, rng)
        elif r == 6:
            h0 = 0.5 + 0.3 * rng.randn(n)
            h0 = project_feasible(h0, n)
        else:
            scale = 0.03 + 0.25 * rng.rand()
            h0 = best_h + scale * rng.randn(n)
            if r % 7 == 0:
                h0 = np.sign(h0 - 0.5) * 0.4 + 0.1 * rng.randn(n)

        h0 = project_feasible(h0, n)

        if r % 3 == 0:
            fun, fargs = obj_smooth, (n, 40.0)
        else:
            fun, fargs = obj_max, (n,)

        try:
            res = minimize(fun, h0, args=fargs, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-12, 'iprint': 0})
            h_opt = project_feasible(res.x, n)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5 - 1e-12:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
        except Exception:
            pass

    # Coordinate descent polishing
    for sweep in range(40):
        if remaining() < 20:
            break
        improved = False
        step_sizes = [0.1, 0.05, 0.02, 0.01]
        for i in range(len(best_h)):
            for s in step_sizes:
                for sign in [-1, 1]:
                    h_trial = best_h.copy()
                    h_trial[i] += sign * s
                    h_trial = project_feasible(h_trial, len(best_h))
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < best_c5 - 1e-13:
                        best_h = h_trial
                        best_c5 = c5_trial
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break

    # Phase 2: Higher resolution
    for n2 in [160, 320]:
        if remaining() < 120:
            break

        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_feasible(h_init, n2)

        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n2,)}]

        n_restarts = 6 if n2 <= 160 else 4
        max_iter = 200 if n2 <= 160 else 100

        for r in range(n_restarts):
            if remaining() < 30:
                break

            if r == 0:
                h0 = h_init.copy()
            elif r == 1:
                h0 = h_init + 0.03 * rng.randn(n2)
            elif r == 2:
                h0 = h_init + 0.08 * rng.randn(n2)
            else:
                h0 = h_init + 0.15 * rng.randn(n2)
            h0 = project_feasible(h0, n2)

            try:
                res = minimize(obj_max, h0, args=(n2,), method='SLSQP',
                               bounds=bounds2, constraints=cons2,
                               options={'maxiter': max_iter, 'ftol': 1e-12, 'iprint': 0})
                h_opt = project_feasible(res.x, n2)
                c5_opt = evaluate_c5(h_opt)
                if c5_opt < best_c5 - 1e-12:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n2
            except Exception:
                pass

        # Quick coordinate descent at this resolution
        for sweep in range(10):
            if remaining() < 20:
                break
            improved = False
            for i in range(n2):
                for s in [0.05, 0.02]:
                    for sign in [-1, 1]:
                        h_trial = best_h.copy()
                        h_trial[i] += sign * s
                        h_trial = project_feasible(h_trial, n2)
                        c5_trial = evaluate_c5(h_trial)
                        if c5_trial < best_c5 - 1e-13:
                            best_h = h_trial
                            best_c5 = c5_trial
                            improved = True
                            break
                    if improved:
                        break
            if not improved:
                break

    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, len(best_h))
