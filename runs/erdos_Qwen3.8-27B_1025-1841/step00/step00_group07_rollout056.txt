import numpy as np
from scipy.optimize import minimize
import time

_P = {}

def _c5_val(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _sm_obj(h, n, beta):
    v = _c5_all(h)
    m = v.max()
    return float(m + np.log(np.sum(np.exp(beta * (v - m)))) / beta)

def _sm_grad(h, n, beta):
    v = _c5_all(h)
    m = v.max()
    ew = np.exp(beta * (v - m))
    w = ew / ew.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki] * dx
        if wi < 1e-15:
            continue
        k = ki - (n - 1)
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        if hi < lo:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx + k] += wi * (1.0 - h[idx])
        g[idx] -= wi * h[idx + k]
    return g

def _cfun(h, n):
    return float(np.sum(h) - n / 2.0)

def _cjac(h, n):
    return np.ones(n)

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    if abs(d) > 1e-12:
        m = (h < 1.0) if d > 0 else (h > 0.0)
        if m.any():
            h[m] += d / float(int(m.sum()))
            h = np.clip(h, 0.0, 1.0)
    return h

    if n == n_new:

    xo = (np.arange(n) + 0.5) * 2.0 / n
