```python
    """Ensure h is in [0,1] and sum(h) = n/2"""
    h = np.clip(h, 0, 1)

    target = n / 2.0
```

```python
    # corr has length 2n-1, indexed 0 to 2n-2
    # Lag k = idx - (n-1)
    # corr[idx] = sum h[a] * (1-h[b]) where b - a = k = idx - (n-1)
    # i.e., b = a + k = a + idx - (n-1)
    
    # For position i (changing from old_i to 1):
    # It appears as 'a' (first index) in lags where b = i+k is valid
    # k = b - i, so b ranges over [0, n-1], k ranges from -i to n-1-i
    # idx = k + (n-1) ranges from n-1-i to n-1-i + n-1-i... 
    # Actually let me think differently.
    
    # corr[idx] where idx = (n-1) + k, k = lag
    # contrib of (a, b=a+k) to corr[idx] is h[a]*(1-h[b])
    
    # When h[i] changes:
    # - As 'a' (first position): pairs (i, i+k) for valid k
    #   Change in contribution: (new_h[i] - old_h[i]) * (1-h[i+k])
    # - As 'b' (second position): pairs (i-k, i) for valid k  
    #   Change in contribution: h[i-k] * (0 - (1-new_h[i])) + h[i-k]*(1-old_h[i])
    #   = h[i-k] * (old_h[i] - new_h[i])... wait
    #   Old contribution: h[i-k]*(1-old_h[i])
    #   New contribution: h[i-k]*(1-new_h[i])
    #   Change: h[i-k]*(old_h[i] - new_h[i])
    
    delta_i = 1.0 - old_i  # new - old for position i (0→1, so delta=1)
    delta_j = 0.0 - old_j  # new - old for position j (1→0, so delta=-1)
    
    # For position i with delta_i:
    # As first position: for k from -i to n-1-i (where i+k in [0,n-1])
    #   corr[n-1+k] += delta_i * (1-h[i+k])
    # As second position: for k from -(n-1-i) to i (where i-k in [0,n-1])
    #   corr[n-1+k] += h[i-k] * (-delta_i)  [because (1-h[i]) changes by -delta_i]

    # corr_lag = sum_a h[a] * (1-h[a+lag])
    # d(corr_lag)/d(h[i]) = (1-h[i+lag]) [when i+lag valid] + (-h[i-lag]) [when i-lag valid]
    
    # So the change in corr for position i is:
    # delta_i * [(1-h[i+lag]) * 1_{0≤i+lag<n} + (-h[i-lag]) * 1_{0≤i-lag<n}]
    
    # For position j similarly with delta_j.
    
    # Let me just compute it directly for efficiency.
    for lag in range(-i, n-i):  # i+lag in [0, n-1]
        idx = n - 1 + lag
        corr[idx] += delta_i * (1.0 - h[i + lag])
    
    for lag in range(max(-(n-1-i), 0-i), min(i, n-1)):  # i-lag in [0, n-1]
        # i-lag in [0, n-1] means lag in [i-(n-1), i]
        idx = n - 1 + lag
        corr[idx] += -delta_i * h[i - lag]
    
    # Similarly for j
    for lag in range(-j, n-j):
        idx = n - 1 + lag
        corr[idx] += delta_j * (1.0 - h[j + lag])
    
    for lag in range(max(-(n-1-j), 0-j), min(j, n-1)):
        idx = n - 1 + lag
        corr[idx] += -delta_j * h[j - lag]
    
    return corr
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n * 0.5
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_bin(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    n_ones = int(np.sum(tm))
    target = n // 2
    if n_ones < target:
        idx = np.where(tm == 0)[0]
        np.random.shuffle(idx)
        tm[idx[:target - n_ones]] = 1.0
    elif n_ones > target:
        idx = np.where(tm == 1)[0]
        np.random.shuffle(idx)
        tm[idx[:n_ones - target]] = 0.0
    return tm

def lse_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / beta

def optimize_continuous(h, n, t0, budget_s):
    h = h.copy()
    c5 = compute_c5(h)
    for beta in [8, 25, 80, 200]:
        if time.time() - t0 > budget_s - 30:
            break
        try:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
            res = minimize(lambda x: lse_obj(x, beta), h,
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-10})
            h_cand = make_feasible(res.x, n)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5:
                h = h_cand
                c5 = c5_cand
        except Exception:
            pass
    return h, c5

def binary_local_search(h, n, t0, budget_s, max_iter=200000):
    h = h.round().astype(np.float64)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n_ones]] = 1.0
    elif n_ones > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n_ones - target]] = 0.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    for it in range(max_iter):
        if time.time() - t0 > budget_s - 10:
            break
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = zeros[np.random.randint(len(zeros))]
        j = ones[np.random.randint(len(ones))]
        h[i] = 1.0
        h[j] = 0.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
        else:
            h[i] = 0.0
            h[j] = 1.0
    return best_h, best_c5

def continuous_local_search(h, n, t0, budget_s, max_iter=50000):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() - t0 > budget_s - 10:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)
        scale = 0.05 + 0.1 * (1.0 - min(1.0, it / max_iter))
        delta = np.random.uniform(-scale, scale)
        h[i] += delta
        h[j] -= delta
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(96, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    resolutions = sorted(set(
        [len(h_parent)] +
        [n for n in [128, 160, 192, 200, 240, 256, 320, 384, 480, 512, 640]
         if n % 2 == 0 and n != len(h_parent)]
    ))

    for n in resolutions:
        if time.time() - t0 > budget_s - 120:
            break

        inits = []
        # Interpolate best
        if len(best_h) != n:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            inits.append(make_feasible(np.interp(x_new, x_old, best_h), n))
        else:
            inits.append(best_h.copy())
        # Thue-Morse binary
        inits.append(thue_morse_bin(n))
        # Thue-Morse smoothed
        tm = thue_morse_bin(n)
        inits.append(make_feasible(uniform_filter1d(tm, size=max(2, n // 32), mode='wrap'), n))
        # Smooth random
        inits.append(make_feasible(uniform_filter1d(np.random.rand(n), size=max(2, n // 24), mode='wrap'), n))

        for h0 in inits:
            if time.time() - t0 > budget_s - 60:
                break
            h, c5 = optimize_continuous(h0, n, t0, budget_s)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    # Binary local search on best
    if time.time() - t0 < budget_s - 60:
        n = len(best_h)
        h_bin, c5_bin = binary_local_search(best_h, n, t0, budget_s)
        if c5_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_bin

    # Continuous local search
    if time.time() - t0 < budget_s - 30:
        n = len(best_h)
        h_cl, c5_cl = continuous_local_search(best_h, n, t0, budget_s)
        if c5_cl < best_c5:
            best_h = h_cl.copy()
            best_c5 = c5_cl

    # Second round: go to higher resolution from best
    if time.time() - t0 < budget_s - 80:
        for n2 in [len(best_h) * 2, len(best_h) * 2 + 2]:
            if n2 % 2 != 0:
                n2 += 1
            if time.time() - t0 > budget_s - 40:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h_interp = make_feasible(np.interp(x_new, x_old, best_h), n2)
            h, c5 = optimize_continuous(h_interp, n2, t0, budget_s)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            h_bin, c5_bin = binary_local_search(best_h, n2, t0, budget_s, max_iter=50000)
            if c5_bin < best_c5:
                best_h = h_bin.copy()
                best_c5 = c5_bin
            if time.time() - t0 < budget_s - 20:
                h_cl, c5_cl = continuous_local_search(best_h, n2, t0, budget_s, max_iter=20000)
                if c5_cl < best_c5:
                    best_h = h_cl.copy()
                    best_c5 = c5_cl

    return best_h, best_c5, len(best_h)
```
