            best_h = h_res
    
    # Phase 2: Binary optimization at multiple resolutions
    time_left = budget_s - (time.time() - t0)
    n_list = [96, 192, 128, 256, 160]
    per_n = max(15, time_left / (len(n_list) + 1))
    
    for n in n_list:
        if time.time() - t0 > budget_s - 50:
            break
        t_end_n = min(t0 + budget_s - 50, time.time() + per_n)
        # Multiple random starts
        n_starts = max(2, int((t_end_n - time.time()) / max(5, per_n / 4)))
        for s in range(n_starts):
            if time.time() > t_end_n - 2:
                break
            h_init = np.zeros(n)
            ones = rng.choice(n, n // 2, replace=False)
            h_init[ones] = 1.0
            # Perturb slightly from random
            for _ in range(n // 4):
                a = rng.randint(n)
                b = rng.randint(n)
                h_init[a], h_init[b] = h_init[b], h_init[a]
            t_end_s = min(t_end_n - 1, time.time() + max(3, (t_end_n - time.time()) / (n_starts - s)))
            c5_res, h_res = sa_binary(h_init, t_end_s, rng)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res
    
    # Phase 3: Interpolate parent to even n and refine
    if parent is not None:
        for n in [96, 192, 288]:
            if time.time() - t0 > budget_s - 45:
                break
            x_old = np.linspace(0, 2, len(parent), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, parent)
            h_bin = (h_interp > 0.5).astype(float)
            n_ones = int(h_bin.sum())
            target_ones = n // 2
            if n_ones > target_ones:
                idxs = np.where(h_bin == 1)[0]
                remove = rng.choice(idxs, n_ones - target_ones, replace=False)
                h_bin[remove] = 0.0
            elif n_ones < target_ones:
                idxs = np.where(h_bin == 0)[0]
                add = rng.choice(idxs, target_ones - n_ones, replace=False)
                h_bin[add] = 1.0
            t_end3 = min(t0 + budget_s - 45, time.time() + 25)
            c5_res, h_res = sa_binary(h_bin, t_end3, rng)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res
    
    # Phase 4: Final polish on best
    if best_h is not None and time.time() - t0 < budget_s - 20:
        t_end4 = t0 + budget_s - 10
        n_best = len(best_h)
        if int(round(n_best / 2.0)) * 2 == n_best and np.allclose(best_h, np.round(best_h), atol=1e-6):
            c5_res, h_res = sa_binary(best_h.copy(), t_end4, rng)
        else:
            c5_res, h_res = sa_continuous(best_h.copy(), t_end4, rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res
    
    if best_h is None:
        n = 96
        best_h = np.zeros(n)
        best_h[0::2] = 1.0
        best_c5, _ = compute_c5(best_h)
    
    # Final validation
    final_c5, _ = compute_c5(best_h)
    n_final = len(best_h)
    if abs(best_h.sum() - n_final / 2.0) > 1e-8:
        best_h = repair_sum(best_h, n_final / 2.0)
        final_c5, _ = compute_c5(best_h)
    
    return best_h, final_c5, n_final
