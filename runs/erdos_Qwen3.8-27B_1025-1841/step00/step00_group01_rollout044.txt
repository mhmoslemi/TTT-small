        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, rng, n_iter, time_limit):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h)
    step_sizes = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        delta = step_sizes[rng.randint(len(step_sizes))]
        if rng.rand() < 0.5:
            if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                h[i] += delta
                h[j] -= delta
                c5_new = compute_c5(h)
                if c5_new < best - 1e-12:
                    best = c5_new
                else:
                    h[i] -= delta
                    h[j] += delta
        else:
            if h[i] - delta >= 0.0 and h[j] + delta <= 1.0:
                h[i] -= delta
                h[j] += delta
                c5_new = compute_c5(h)
                if c5_new < best - 1e-12:
                    best = c5_new
                else:
                    h[i] += delta
                    h[j] -= delta
    return h, best

def anneal_search(h, rng, n_iter, time_limit):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    temp = 0.02
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        delta = rng.uniform(0.02, 0.4)
        sign = 1.0 if rng.rand() < 0.5 else -1.0
        old_i, old_j = h[i], h[j]
        h[i] = np.clip(old_i + sign * delta, 0.0, 1.0)
        h[j] = np.clip(old_j - sign * delta, 0.0, 1.0)
        c5_new = compute_c5(h)
        acceptance = c5_new < best or (c5_new < best + temp * 0.001 and rng.rand() < 0.3)
        if acceptance and c5_new < best:
            best = c5_new
            best_h = h.copy()
        elif not acceptance:
            h[i], h[j] = old_i, old_j
        temp *= 0.999
    return best_h, best

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    else:
        parent = np.full(47, 0.5)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c5 = compute_c5(best_h)

    resolutions = [64, 96, 128, 160, 192]

    for n_pts in resolutions:
        if time.time() - start_time > budget_s * 0.75:
            break
        target = n_pts / 2.0

        x_old = np.linspace(0, 2, n_parent, endpoint=False)
        x_new = np.linspace(0, 2, n_pts, endpoint=False)
        h_start = np.interp(x_new, x_old, parent, right=parent[-1])
        h_start = project_sum(h_start, target)

        h_work = h_start.copy()
        c5_work = compute_c5(h_work)

        # SLSQP optimization
        remaining_slsqp = max(30, budget_s * 0.55 - (time.time() - start_time))
        max_iter = min(400, max(80, int(remaining_slsqp / 1.5)))
        bounds = [(0.0, 1.0)] * n_pts
        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n_pts)},)
        try:
            res = minimize(compute_c5, h_work, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-10})
            h_try = project_sum(res.x, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_work:
                h_work = h_try
                c5_work = c5_try
        except Exception:
            pass

        # Local search refinement
        if time.time() - start_time < budget_s * 0.8:
            t_alloc = max(5, (budget_s * 0.8 - (time.time() - start_time)))
            n_iter_ls = min(20000, int(t_alloc * 10000 / max(1, n_pts)))
            h_work, c5_work = local_search(h_work, rng, n_iter_ls, min(t_alloc, 120))

        # Annealing
        if time.time() - start_time < budget_s * 0.85:
            t_alloc = max(5, (budget_s * 0.85 - (time.time() - start_time)))
            n_iter_an = min(10000, int(t_alloc * 5000 / max(1, n_pts)))
            h_an, c5_an = anneal_search(h_work, rng, n_iter_an, min(t_alloc, 90))
            if c5_an < c5_work:
                h_work = h_an
                c5_work = c5_an

        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

        parent = best_h
        n_parent = len(parent)

    # Final polish at best resolution
    if time.time() - start_time < budget_s * 0.95:
        n_best = len(best_h)
        t_alloc = max(5, budget_s * 0.95 - (time.time() - start_time))
        h_pol, c5_pol = local_search(best_h, rng, int(t_alloc * 300), t_alloc * 0.7)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
