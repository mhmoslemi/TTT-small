                    pass

    # Phase 3: Nelder-Mead derivative-free
    if rem() > 40:
        try:
            r = minimize(nm_obj, best_h, args=(tgt,),
                         method='Nelder-Mead',
                         options={'maxiter': 8000, 'fatol': 1e-14, 'xatol': 1e-14})
            h_new = proj(r.x, tgt)
            c5n = c5_value(h_new)
            if c5n < best_c5:
                best_h, best_c5 = h_new, c5n
        except Exception:
            pass

    # Phase 4: Binary push + re-optimize
    if rem() > 40:
        for _ in range(4):
            if rem() < 15:
                break
            h_push = best_h.copy()
            h_push[h_push > 0.55] = 1.0
            h_push[h_push < 0.45] = 0.0
            h_push = proj(h_push, tgt)
            c5p = c5_value(h_push)
            if c5p < best_c5:
                best_h, best_c5 = h_push, c5p
            try:
                r = minimize(c5_lse, best_h, args=(500,),
                             method='SLSQP', bounds=bnds, constraints=cns,
                             options={'maxiter': 300, 'ftol': 1e-13})
                h_new = proj(r.x, tgt)
                c5n = c5_value(h_new)
                if c5n < best_c5:
                    best_h, best_c5 = h_new, c5n
            except Exception:
                pass

    # Phase 5: Random perturb + local optimize
    for trial in range(50):
        if rem() < 20:
            break
        h_r = best_h + np.random.randn(n) * 0.03
        h_r = proj(h_r, tgt)
        c5r = c5_value(h_r)
        if c5r < best_c5:
            best_h, best_c5 = h_r, c5r
        try:
            r = minimize(c5_lse, h_r, args=(200,),
                         method='SLSQP', bounds=bnds, constraints=cns,
                         options={'maxiter': 200, 'ftol': 1e-11})
            h_new = proj(r.x, tgt)
            c5n = c5_value(h_new)
            if c5n < best_c5:
                best_h, best_c5 = h_new, c5n
        except Exception:
            pass

    # Phase 6: Final SLSQP polish
    if rem() > 10:
        try:
            r = minimize(c5_lse, best_h, args=(2000,),
                         method='SLSQP', bounds=bnds, constraints=cns,
                         options={'maxiter': 500, 'ftol': 1e-14})
            h_new = proj(r.x, tgt)
            c5n = c5_value(h_new)
            if c5n < best_c5:
                best_h, best_c5 = h_new, c5n
        except Exception:
            pass

    best_h = proj(best_h, tgt)
    best_c5 = c5_value(best_h)
    return (best_h, best_c5, n)
