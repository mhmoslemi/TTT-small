            c5_new = compute_c5(h_new, n)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new.copy()

    # Phase 3: final polish at best resolution
    if time.time() - t0 < budget_s * 0.95:
        setup_globals(n)
        constraints = make_constraints()
        bounds = make_bounds(n)
        for r in range(5):
            if time.time() - t0 > budget_s * 0.97:
                break
            h_start = h_best + rng.normal(0, 0.01, n)
            h_start = project_feasible(h_start, n)
            try:
                res = minimize(_obj, h_start, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 800, 'ftol': 1e-15})
            except Exception:
                continue
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new.copy()

    h_best = project_feasible(h_best, len(h_best))
    c5_final = compute_c5(h_best, len(h_best))
    return h_best, c5_final, len(h_best)
