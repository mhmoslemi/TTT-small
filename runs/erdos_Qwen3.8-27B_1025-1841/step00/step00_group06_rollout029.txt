    """Optimize h[j] given all other coordinates fixed."""
    n = len(h)
    # Compute A_k and D_k for each shift k
    # R(k) = sum_{valid n} h[n+k]*(1-h[n])
    # h[j] contribution: h[j]*(1-h[j-k]) if j-k valid, plus h[j+k]*(1-h[j]) if j+k valid
    
    # For each shift k (from -(n-1) to n-1):
    # R(k) = D_k + A_k * h[j]
    # where A_k = (1-h[j-k]) if j-k in [0,n-1] else 0, minus h[j+k] if j+k in [0,n-1] else 0
    #       D_k = R(k) - A_k * h[j]
    
    corr_full = np.correlate(h, 1-h, mode='full')
    # corr_full[p] corresponds to shift k = p - (n-1)
    
    # For shift k, the coefficient of h[j] in R(k):
    # h[j] appears as h[n+k] when n = j-k (if 0 <= j-k < n): coefficient is (1-h[j-k])
    # h[j] appears in (1-h[n]) when n = j (if 0 <= j < n, always true): coefficient is -h[j+k] (if 0 <= j+k < n)
    
    A = np.zeros(2*n-1)
    for p in range(2*n-1):
        k = p - (n-1)
        coeff = 0.0
        # h[j] as h[n+k] where n = j-k
        n_idx = j - k
        if 0 <= n_idx < n:
            coeff += (1 - h[n_idx])
        # h[j] in (1-h[n]) where n = j, so term is h[j+k]*(1-h[j]), coeff of h[j] is -h[j+k]
        n_idx2 = j + k
        if 0 <= n_idx2 < n:
            coeff -= h[n_idx2]
        A[p] = coeff
    
    D = corr_full - A * h[j]
    
    # Now minimize max_p (D[p] + A[p] * t) for t in [0, 1]
    # This is a 1D convex problem. The optimal t is where the maximum is minimized.
    # We can use bisection on the subgradient, or just use scipy.optimize.minimize_scalar
    
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(lambda t: np.max(D + A * t), bounds=(0, 1), method='bounded')
    return res.x

        x_new = np.linspace(0, 2, n_try, endpoint=False)

        h_try = project_constraint(h_try, n_try / 2.0)
        max_sweeps = 200 if n_try <= 200 else 80
        optimize(h_try, n_try, max_sweeps, f"res_{n_try}")

    # Phase 3: Binary random starts
    for attempt in range(100):
        if time.time() > deadline - 30:
            break
        n_try = int(rng.choice([95, 100, 120, 150, 192, 200, 256]))
        h_rand = np.zeros(n_try)
        h_rand[rng.choice(n_try, n_try // 2, replace=False)] = 1.0
        h_rand = project_constraint(h_rand, n_try / 2.0)
        c5_r = compute_c5(h_rand)
        if c5_r < c5_best + 0.05:
            ks = np.arange(-(n_try - 1), n_try)
            for _ in range(40):
                if time.time() > deadline - 30:
                    break
                h_rand = coord_descent_sweep(h_rand, rng.permutation(n_try), ks)
                c5_r = compute_c5(h_rand)
                if c5_r < c5_best - 1e-13:
                    c5_best = c5_r
                    h_best = h_rand.copy()
                    n_best = n_try

    # Phase 4: Subgradient perturbations to escape local optima
    h = h_best.copy()
    n = n_best
    ks = np.arange(-(n - 1), n)
    step = 0.05
    for i in range(200):
        if time.time() > deadline - 10:
            break
        h = subgradient_step(h, step, rng)
        h = coord_descent_sweep(h, rng.permutation(n), ks)
        c5 = compute_c5(h)
        if c5 < c5_best - 1e-13:
            c5_best = c5
            h_best = h.copy()
            n_best = n
        step *= 0.98

    # Phase 5: Final polish
    h = h_best.copy()
    n = n_best
    ks = np.arange(-(n - 1), n)
    while time.time() < deadline:
        h = coord_descent_sweep(h, rng.permutation(n), ks)
        c5 = compute_c5(h)
        if c5 < c5_best - 1e-13:
            c5_best = c5
            h_best = h.copy()
        else:
            break

    # Ensure valid output
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_constraint(h_best, len(h_best) / 2.0)
    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final

    return h_best, c5_best, len(h_best)
