from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h_src, n_dst):
    n_src = len(h_src)
    x_dst = np.arange(n_dst, dtype=np.float64) * (n_src / float(n_dst))
    return np.interp(x_dst, np.arange(n_src, dtype=np.float64), h_src)

def make_beatty(n, alpha=1.618033988749895):
    idx = np.arange(n, dtype=np.float64)
    h = np.where((idx * alpha) % 1.0 < 0.5, 1.0, 0.0)
    return h

def make_alt(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return h

def local_search(h, n_iter, step):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(n_iter):
        ht = best.copy()
        i = np.random.randint(0, n)
        ht[i] += np.random.uniform(-step, step)
        ht = fix_sum(ht, target)
        v = compute_c5(ht)
        if v < best_val:
            best = ht
            best_val = v
    return best, best_val
