    _g_target = n * 0.5
    _g_dx = dx
    h_c = h_cur.copy()
    try:
        r = minimize(_nm_obj, h_c, method='Nelder-Mead',
                    options={'maxiter': 80000, 'xatol': 1e-11, 'fatol': 1e-13, 'adapt': True})
        h_c = _proj(r.x, n * 0.5)
        v = _c5(h_c, dx)
        if v < _g_best_c5:
            _g_best_c5 = v
            _g_best_h = h_c.copy()
    except Exception:
        pass
    return h_c

def _binary_ls(h_bin, n, dx, max_iters):
    global _g_best_c5, _g_best_h
    h = h_bin.astype(float).copy()
    cur = _c5(h, dx)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_iters):
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        new_c = _c5(h, dx)
        if new_c < cur:
            cur = new_c
            if cur < _g_best_c5:
                _g_best_c5 = cur
                _g_best_h = h.copy()
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, cur

def _round_to_01(h, n):
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        extra = np.random.choice(np.where(h_bin == 1)[0], n_ones - target, replace=False)
        h_bin[extra] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zeros = np.where(h_bin == 0)[0]
        pick = np.random.choice(zeros, deficit, replace=False)
        h_bin[pick] = 1.0
    return h_bin

def _coordinated(h, n, dx, max_iters):
    global _g_best_c5, _g_best_h
    h = h.copy()
    cur = _c5(h, dx)
    dx_val = 2.0 / n
    step = 0.05
    for _ in range(max_iters):
        i = np.random.randint(n)
        direction = 1 if h[i] < 0.99 else -1
        h[i] = np.clip(h[i] + direction * step, 0.0, 1.0)
        h = _proj(h, n * 0.5)
        new_c = _c5(h, dx)
        if new_c < cur:
            cur = new_c
            if cur < _g_best_c5:
                _g_best_c5 = cur
                _g_best_h = h.copy()
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    global _g_t0, _g_budget, _g_best_c5, _g_best_h, _g_n, _g_target, _g_dx, _g_smooth_t
    np.random.seed(seed)
    _g_t0 = time.time()
    _g_budget = budget_s

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.ones(42) * 0.5

    n0 = len(h_init)
    h0 = _proj(h_init, n0 * 0.5)
    _g_best_c5 = _c5(h0, 2.0 / n0)
    _g_best_h = h0.copy()

    # Phase 1: SLSQP at n0
    h_cur = _opt_slsqp(h0, n0, 2.0 / n0, budget_s * 0.30)

    # Phase 2: NM polish at n0
    if time.time() - _g_t0 < budget_s * 0.45:
        h_cur = _opt_nm(h_cur, n0, 2.0 / n0, budget_s * 0.45)

    # Phase 3: 0-1 local search at n0
    if time.time() - _g_t0 < budget_s * 0.55:
        h_bin = _round_to_01(h_cur, n0)
        _binary_ls(h_bin, n0, 2.0 / n0, 30000)

    # Phase 4: Coordinated search at n0
    if time.time() - _g_t0 < budget_s * 0.60:
        _coordinated(h_cur, n0, 2.0 / n0, 20000)

    # Phase 5: Higher resolutions
    for n_tr in [84, 112, 168, 224]:
        if time.time() - _g_t0 > budget_s * 0.85:
            break
        nb = len(_g_best_h)
        if n_tr <= nb:
            continue
        if n_tr % nb == 0:
            f = n_tr // nb
            h_up = np.repeat(_g_best_h, f)
        else:
            xo = np.arange(nb) * (2.0 / nb)
            xn = np.arange(n_tr) * (2.0 / n_tr)
            h_up = np.interp(xn, xo, _g_best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
        h_up = _proj(h_up, n_tr * 0.5)
        rem = min(budget_s * 0.85 - (_g_t0 if False else time.time() - _g_t0), budget_s * 0.15)
        if rem < 5.0:
            break
        h_up = _opt_slsqp(h_up, n_tr, 2.0 / n_tr, time.time() - _g_t0 + rem)
        if time.time() - _g_t0 < budget_s * 0.85:
            h_up = _opt_nm(h_up, n_tr, 2.0 / n_tr, budget_s * 0.85)

    # Phase 6: Final NM polish on best found
    if time.time() - _g_t0 < budget_s * 0.92:
        nf = len(_g_best_h)
        h_final = _opt_nm(_g_best_h.copy(), nf, 2.0 / nf, budget_s * 0.92)

    # Final feasibility check
    nf = len(_g_best_h)
    _g_best_h = _proj(_g_best_h, nf * 0.5)
    fc = _c5(_g_best_h, 2.0 / nf)

    if fc < _g_best_c5:
        _g_best_c5 = fc

    return (_g_best_h, _g_best_c5, nf)
