    c = _eval_c5(h_bin, dx)
    if c < best_c:
        best = h_bin.copy()
        best_c = c
    for it in range(20000):
        if time.time() > deadline:
            break
        h_t = best.copy()
        ones = np.where(h_t > 0.5)[0]
        zeros = np.where(h_t < 0.5)[0]
        if len(ones) > 0 and len(zeros) > 0:
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_t[a] = 0.0
            h_t[b] = 1.0
            c = _eval_c5(h_t, dx)
            if c < best_c:
                best = h_t
                best_c = c
    return best, best_c

def _slsqp_optimize(h, n, deadline):
    global _n_g
    _n_g = n
    dx = 2.0 / n
    def _obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode="full")) * dx
    constraints = []
    constraints.append({'type': 'eq', 'fun': _con_fun, 'jac': _con_jac})
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': 400, 'ftol': 1e-12})
    if time.time() > deadline:
        return h, _eval_c5(h, dx)
    return _repair(res.x, n), _eval_c5(_repair(res.x, n), dx)

_n_g = 70

def _con_fun(x):
    return np.sum(x) - _n_g / 2.0

def _con_jac(x):
    return np.ones(_n_g)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 25

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.ones(70) * 0.5

    n = len(h0)
    dx = 2.0 / n
    h = _repair(h0.copy(), n)
    best_h = h.copy()
    best_c = _eval_c5(best_h, dx)

    # Phase 1: SLSQP at initial resolution
    if time.time() < deadline:
        h_s, c_s = _slsqp_optimize(h, n, deadline)
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    # Phase 2: Gradient descent
    if time.time() < deadline:
        h_g, c_g = _gradient_descent(best_h, n, rng, deadline)
        if c_g < best_c:
            best_h, best_c = h_g, c_g

    # Phase 3: Local search
    if time.time() < deadline:
        h_ls, c_ls = _local_search(best_h, n, rng, deadline)
        if c_ls < best_c:
            best_h, best_c = h_ls, c_ls

    # Phase 4: Block moves
    if time.time() < deadline:
        h_bm, c_bm = _block_move(best_h, n, rng, deadline)
        if c_bm < best_c:
            best_h, best_c = h_bm, c_bm

    # Phase 5: Binarize
    if time.time() < deadline:
        h_b, c_b = _binarize(best_h, n, rng, deadline)
        if c_b < best_c:
            best_h, best_c = h_b, c_b

    # Phase 6: Higher resolution
    for n2 in [2 * n, 3 * n, 5 * n]:
        if time.time() > deadline - 60:
            break
        if n2 > 1000:
            break
        dx2 = 2.0 / n2
        h2 = _repair(_interp_h(best_h, n, n2), n2)
        n = n2

        if time.time() < deadline - 30:
            h2, c2 = _slsqp_optimize(h2, n, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2

        if time.time() < deadline - 20:
            h2, c2 = _gradient_descent(best_h, n, rng, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2

        if time.time() < deadline - 15:
            h2, c2 = _local_search(best_h, n, rng, deadline, n_max_iter=30000)
            if c2 < best_c:
                best_h, best_c = h2, c2

        if time.time() < deadline - 10:
            h2, c2 = _binarize(best_h, n, rng, deadline)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 7: Final refinement
    if time.time() < deadline:
        h_f, c_f = _local_search(best_h, len(best_h), rng, deadline, n_max_iter=20000)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    final_n = len(best_h)
    best_h = _repair(best_h, final_n)
    best_c = _eval_c5(best_h, 2.0 / final_n)

    try:
        off = evaluate_erdos_solution(best_h)
        if off is not None:
            best_c = float(off)
    except Exception:
        pass

    return (best_h, best_c, final_n)
