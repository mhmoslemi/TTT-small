import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_onto(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.argsort(-h)
        for i in idx:
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if diff < 1e-13:
                break
    else:
        idx = np.argsort(h)
        for i in idx:
            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
            if diff > -1e-13:
                break
    return h


class C5Problem:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n

    def objective(self, h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr)) * self.dx

    def con_eq(self, h):
        return np.sum(h) - self.n * 0.5

    def con_eq_jac(self, h):
        return np.ones(self.n)

    def make_bounds(self):
        return [(0.0, 1.0)] * self.n


def try_slsqp(h0, n, maxiter=200):

    bounds = prob.make_bounds()

        h_opt = project_onto(res.x, n)
        c5 = compute_c5(h_opt, n)
