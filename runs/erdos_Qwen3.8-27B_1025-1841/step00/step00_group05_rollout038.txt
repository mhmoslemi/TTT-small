            lo = mid
        else:
            hi = mid

    s = alpha * (corr - mc)
    val = mc + np.log(np.sum(np.exp(s))) / alpha

    v1 = (J - L >= 0) & (J - L < n)
    v2 = (J + L >= 0) & (J + L < n)
    
    idx1 = np.where(v1, J - L, 0)
    idx2 = np.where(v2, J + L, 0)
    
    term1 = np.where(v1, 1.0 - x[idx1], 0.0)
    term2 = np.where(v2, x[idx2], 0.0)
    
    g = (term1 - term2).T @ (w * dx)
    
    sd = x.sum() - tsum
    g += 2.0 * pen * sd
    return g
