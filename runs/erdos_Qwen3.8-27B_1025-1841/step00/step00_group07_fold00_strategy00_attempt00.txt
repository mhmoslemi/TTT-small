                    perturb_amount = 0.01 * (n_points / 80)
                    # Local search around current solution with Nelder-Mead
                    res = sopt.minimize(lambda x: objective(x), best_sol, method='Nelder-Mead',
                                        bounds=[(0,1)]*n_points, tol=1e-6)
                    if res.fun < best_c5:
                        best_c5 = res.fun
                        best_sol = res.x

            best_sol = x0  # original n_points=80 from parent
            best_c5 = objective(best_sol)

        # But we are not using the parent, so we try a full strategy.

        print("Running full strategy with no parent")

        try:

            # Full optimization by random forest or Bayesian? Not available, so do simple search.

            # Sample a range of n_points
            n_points_list = n_points_range  # a scattering of values
            best_c5 = 1.0
            best_h = None
            n_try = 0
            max_tries = 1000000

        except:
            return best_h, best_c5, n_points_list  # but not found

        # Use local search for n=80,81,...,80+Δ with random jumps

        # Since no parent is available, try all n_points candidates

        for n in n_points_list:
            print("Testing resolution", n)
            dx = 2 / n
            # Generate random h_vectors with sum=n/2
            while n_try < max_tries and time.time() - time_start < budget_s:
                # Random h vector generation
                # Two ways: (1) generate 0/1 vector with count n/2 (but n might be even or odd)
                # (2) generate real numbers, but better the latter is easier but more parameters.
                # (3) random uniform between 0 and 1, then norm to sum n/2
                try:
                    h_init = np.random.uniform(0,1,n)
                    sum_ = np.sum(h_init)
                    if abs(sum_-n/2) < 1e-1: continue  # reject by chance
                    h_init = h_init * (n/2) / sum_
                    h_init = np.clip(h_init, 0,1)
                    sum_val = np.sum(h_init)
                    score = objective(h_init)
                    if score < best_c5 or best_sol is None:
                        best_c5 = score
                        best_sol = h_init
                        bestn = n

                    # But this is random, may not converge.

                except: 
                    continue

                # Unable to find, so move to next

        # Since no improvement, return best found
        if best_sol is None:
            return initial_h_values, objective(initial_h_values), 80
        else:
            return best_sol, best_c5, bestn

    return best_sol, best_c5, n_points
