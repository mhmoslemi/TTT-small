def _proj(v, c):

    for _ in range(200):

def _slsqp_opt(h, n, tgt, dx, alpha, maxiter=800, ftol=1e-11):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return _smax(np.correlate(x, 1.0 - x, mode="full") * dx, alpha)
    def g(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt, 'jac': g}]
    r = minimize(f, h, method='SLSQP', bounds=bounds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return r
