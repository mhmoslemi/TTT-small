This is related to the "minimum overlap" problem in combinatorics.

Actually, the active set is only lag 0 (value 0.5), and all other lags have lower values. So the subgradient is just from lag 0, which is [0,0,0,0]. This means the gradient method would stop, but we're not at the optimum!

The issue is that the subgradient of the max function at a point where only one function is active is just the gradient of that function. If that gradient is 0, the method stops. But the true minimum might be in a direction that increases g_0 slightly while decreasing the other g_k significantly.

This is a well-known issue with subgradient methods for min-max problems. The solution is to use a "smoothed" version or to use a different algorithm.

2. Try to decrease g_{k*} while not increasing any other g_k too much

But L-BFGS-B doesn't handle non-smooth objectives well. The max function is not differentiable.

Let me use a "smoothing" trick: replace max_k g_k with the log-sum-exp approximation:
f_smooth(h) = (1/β) * log(sum_k exp(β * g_k))

As β → ∞, this approaches the true max. For finite β, it's smooth and I can use gradient-based methods.

The gradient of f_smooth is:
∂f_smooth/∂h[i] = (1/β) * sum_k exp(β*g_k) * β * ∂g_k/∂h[i] / sum_k exp(β*g_k)
                  = sum_k p_k * ∂g_k/∂h[i]

where p_k = exp(β*g_k) / sum_j exp(β*g_j) is a softmax over the lags.

This is a weighted average of the individual gradients, which is much better than the subgradient approach!

```python
def smoothed_objective_and_grad(h, beta=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    # Log-sum-exp for numerical stability
    max_corr = np.max(corr)
    log_sum = max_corr + np.log(np.sum(np.exp(corr - max_corr)))
    f = log_sum / beta  # Wait, this is (1/beta)*log(sum(exp(beta*g_k)))
    # Actually: f = (1/beta) * log(sum exp(beta * corr))
    # = (1/beta) * [beta*max + log(sum exp(beta*(corr - max)))]
    # = max + (1/beta) * log(sum exp(beta*(corr - max)))
    
    # Hmm, let me redo. f_smooth = (1/beta) * log(sum_k exp(beta * g_k))
    # For numerical stability:
    # = (1/beta) * log(exp(beta*max) * sum_k exp(beta*(g_k - max)))
    # = (1/beta) * [beta*max + log(sum_k exp(beta*(g_k - max)))]
    # = max + (1/beta) * log(sum_k exp(beta*(g_k - max)))
    
    beta_corr = beta * (corr - max_corr)
    log_sum_exp = np.log(np.sum(np.exp(beta_corr)))
    f = max_corr + log_sum_exp / beta
    
    # Gradient: (1/beta) * sum_k exp(beta*g_k) * beta * grad_g_k / sum_j exp(beta*g_j)
    # = sum_k p_k * grad_g_k where p_k = exp(beta*(g_k - max)) / sum_j exp(beta*(g_j - max))
    weights = np.exp(beta_corr)
    weights /= np.sum(weights)
    
    # Compute weighted sum of gradients
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        j_lo = max(0, lag)
        j_hi = min(n, n + lag)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += weights[k] * (1 - h[j_lo - lag:j_hi - lag])
        j_lo2 = max(0, -lag)
        j_hi2 = min(n, n - lag)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= weights[k] * h[j_lo2 + lag:j_hi2 + lag]
    grad *= dx
    
    return f, grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    n = len(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        rem = abs(diff)
        for i in order:
            if diff > 0:
                s = min(1.0 - h[i], rem)
                h[i] += s
            else:
                s = min(h[i], rem)
                h[i] -= s
            rem -= s
            if rem < 1e-13:
                break
    return h

def smoothed_c5_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = corr.max()
    shifted = beta * (corr - mx)
    exp_shifted = np.exp(shifted)
    wsum = exp_shifted.sum()
    f = mx + np.log(wsum) / beta
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    significant = np.where(weights > 1e-6)[0]
    for k in significant:
        lag = k - (n - 1)
        wk = weights[k]
        jlo = max(0, lag)
        jhi = min(n, n + lag)
        if jhi > jlo:
            grad[jlo:jhi] += wk * (1.0 - h[jlo - lag:jhi - lag])
        jlo2 = max(0, -lag)
        jhi2 = min(n, n - lag)
        if jhi2 > jlo2:
            grad[jlo2:jhi2] -= wk * h[jlo2 + lag:jhi2 + lag]
    grad *= dx
    return f, grad

def coordinate_descent(h, target, deadline, max_rounds=100):
    n = len(h)
    c5 = compute_c5(h)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            best_val = old
            best_c = c5
            candidates = [0.0, 1.0, max(0.0, old - 0.15), min(1.0, old + 0.15),
                          max(0.0, old - 0.05), min(1.0, old + 0.05)]
            for v in candidates:
                if abs(v - old) < 1e-7:
                    continue
                h[i] = v
                h = project_constraint(h, target)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_val = v
            if best_val != old:
                h[i] = best_val
                h = project_constraint(h, target)
                c5 = best_c
                improved = True
            else:
                h[i] = old
                h = project_constraint(h, target)
        if not improved:
            break
    return h, c5

def gradient_descent(h, target, deadline, max_iters=3000):
    n = len(h)
    c5 = compute_c5(h)
    lr = 0.3
    beta = 30.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        f, grad = smoothed_c5_and_grad(h, beta)
        h_new = h - lr * grad
        h_new = project_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            lr = min(lr * 1.1, 2.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break
    return h, c5
```
