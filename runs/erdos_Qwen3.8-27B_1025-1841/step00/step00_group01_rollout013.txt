from scipy.optimize import minimize

    N = len(h)
    dx = 2.0 / N
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def project_to_simplex_sum(h, N):
    """Project h to [0,1]^N with sum = N/2 (approximately)"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target = N / 2.0
    if abs(current_sum - target) < 1e-10:
        return h
    # Simple adjustment
    diff = target - current_sum
    # Adjust elements not at bounds
    not_at_bound = (h > 1e-10) & (h < 1 - 1e-10)
    if not np.any(not_at_bound):
        # All at bounds, need to adjust
        if diff > 0:
            # Need to increase sum, increase the smallest values
            idx = np.argsort(h)
            for i in idx:
                if h[i] < 1:
                    inc = min(diff, 1 - h[i])
                    h[i] += inc
                    diff -= inc
                    if abs(diff) < 1e-10:
                        break
        else:
            idx = np.argsort(-h)
            for i in idx:
                if h[i] > 0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
                    if abs(diff) < 1e-10:
                        break
    else:
        # Distribute proportionally
        n_free = np.sum(not_at_bound)
        adj = diff / n_free
        h[not_at_bound] += adj
        h = np.clip(h, 0, 1)
        # Re-normalize if clipping changed sum
        if abs(np.sum(h) - target) > 1e-8:
            h = project_to_simplex_sum(h, N)
    return h
