    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(h, target):
    h = h.astype(float).copy()
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break

        else:

        else:

from scipy.optimize import minimize
import time

_G_N = 0
_G_TAU = 1.0

def _smooth_obj(x):
    n = _G_N
    tau = _G_TAU
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(corr)
    if cmax > 50.0:
        return cmax
    return cmax + np.log(np.sum(np.exp(tau * (corr - cmax)))) / tau

    return float(np.max(np.correlate(x, 1.0 - x, mode='full')) * (2.0 / n))

def _sum_con(x):

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(h, target):
    h = h.astype(float).copy()
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = int(m.sum())
        if c == 0:
            break
        s = d / c
        if d > 0:
            s = min(s, 1.0 - h[m].min())
        else:
            s = max(s, -h[m].min())
        h[m] += s
    return np.clip(h, 0.0, 1.0)

def _interp(h, nf, nt):
    if nf == nt:
        return h.copy()
    xf = np.arange(nf) + 0.5
    xt = np.arange(nt) + 0.5
    vv = np.concatenate([h, h, h])
    xc = np.concatenate([np.arange(nf)+0.5, np.arange(nf)+nf+0.5, np.arange(nf)+2*nf+0.5])
    hn = np.interp(xt, xc, vv)
    return _proj(np.clip(hn, 0, 1), 0.5 * nt)

def _thue_morse(n):
    seq = np.zeros(n, dtype=float)
    for i in range(1, n):
        seq[i] = (seq[i >> 1] ^ (i & 1))
    return seq

def _optimize(h0, n, t0, budget):

    cons = [{'type': 'eq', 'fun': _sum_con}]
    bounds = [(0.0, 1.0)] * n
    best = h0.copy()
    best_c = _c5(best, n)

    for tau in [100.0, 500.0]:
        if time.time() - t0 > budget - 30:
            break
        _G_N = n
        _G_TAU = tau
        try:
            r = minimize(_smooth_obj, best.copy(), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 800, 'ftol': 1e-12})
            nh = _proj(np.clip(r.x, 0, 1), 0.5 * n)
            nc = _c5(nh, n)
            if nc < best_c - 1e-10:

        except Exception:

                     bounds=bounds, constraints=cons,
                     options={'maxiter': 400, 'ftol': 1e-12})
        nh = _proj(np.clip(r.x, 0, 1), 0.5 * n)
        nc = _c5(nh, n)
        if nc < best_c - 1e-10:
            best, best_c = nh, nc
    except Exception:
        pass

    return best, best_c
