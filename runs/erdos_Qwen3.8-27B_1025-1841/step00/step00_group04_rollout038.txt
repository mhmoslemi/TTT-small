        n_new = n0 * factor
        if n_new > 800 or remaining() < 50:
            break
        h_up = upsample_h(best_h, n_new)
        for t in [30, 80, 200]:
            if remaining() < 20:
                break
            h_opt = optimize_h(h_up, n_new, t=t, penalty=500, maxiter=300)
            c5 = compute_c5(h_opt, n_new)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

    # Phase 4: Random restarts at current best resolution
    n_best = len(best_h)
    attempt = 0
    while remaining() > 40:
        attempt += 1
        strategy = attempt % 5
        if strategy == 0:
            h_mut = best_h + np.random.randn(n_best) * 0.08
        elif strategy == 1:
            h_mut = best_h + np.random.randn(n_best) * 0.2
        elif strategy == 2:
            mask = np.random.rand(n_best) < 0.25
            h_mut = best_h.copy()
            h_mut[mask] = np.random.uniform(0, 1, int(mask.sum()))
        elif strategy == 3:
            # Block randomization
            block_size = max(1, n_best // 20)
            h_mut = best_h.copy()
            for i in range(0, n_best - block_size, block_size):
                if np.random.rand() < 0.4:
                    h_mut[i:i+block_size] = np.random.uniform(0, 1, block_size)
        else:
            # New random solution
            h_mut = np.random.rand(n_best)

        h_mut = feasible_project(np.clip(h_mut, 0, 1), n_best)
        t = min(100 + attempt * 20, 800)
        h_opt = optimize_h(h_mut, n_best, t=t, penalty=500, maxiter=250)
        c5 = compute_c5(h_opt, n_best)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Phase 5: Final polish with very high t
    if remaining() > 25:
        for t in [500, 1000, 2000]:
            if remaining() < 10:
                break
            h_opt = optimize_h(best_h, len(best_h), t=t, penalty=1000, maxiter=500)
            c5 = compute_c5(h_opt, len(best_h))
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

    # Ensure feasibility
    best_h = feasible_project(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
