                    if c5_trial < best_c5 - 1e-13:
                        best_h = trial
                        best_c5 = c5_trial
                        improved = True
                        break

    # Final fix
    best_h = fix_constraint(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
