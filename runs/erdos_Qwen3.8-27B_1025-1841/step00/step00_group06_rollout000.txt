import numpy as np
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

    n = len(h)

    # h[i] as h[n+k] where n=i-k: contributes (1-h[i-k]) if valid

    start = time.time()
    alpha = 0.01
    best_h = h.copy()
    best_c5, _ = compute_c5_full(h)
    m_lags = min(5, n - 1)
    stagnant = 0
    
    for it in range(200000):

            break

        if c5 < best_c5 - 1e-12:

            best_h = h.copy()

        else:
            stagnant += 1

        top_idx = np.argsort(corr)[-m_lags:]

        h_new = project_feasible(h - alpha * grad, n)
        c5_new, _ = compute_c5_full(h_new)

            h = h_new
            alpha = min(alpha * 1.03, 0.05)
        else:
            alpha *= 0.95
            if alpha < 0.001:
                alpha = 0.01

            idxs = rng.choice(n, n_pert, replace=False)
            h[idxs] += rng.uniform(-0.3, 0.3, n_pert)
            h = project_feasible(h, n)
            alpha = 0.02
            stagnant = 0
    
    return best_h, best_c5

def upscale(h_old, n_new):
    """Upscale h from its length to n_new using linear interpolation."""
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old, period=2.0)  # Hmm, period might not be right
    # Actually, just linear interp on [0, 2]
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # midpoints
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def make_sawtooth(n, period=4):
    """Create a sawtooth-like pattern with integral = 1."""
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        h[i] = 1.0 - phase  # decreases from 1 to 0
    # Scale to get integral = 1
    dx = 2.0 / n
    h *= 1.0 / (h.sum() * dx)
    h = np.clip(h, 0, 1)
    h = project_feasible(h, n)
    return h

def make_bipolar(n, rng, n_blocks=4):
    """Create a random bipolar (0/1) pattern."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        if b % 2 == 0:
            start = b * block_size
            end = min((b+1) * block_size, n)
            h[start:end] = 1.0
    # Adjust to get sum = n/2
    h = project_feasible(h, n)
    return h

def _c5_and_corr(h):

def _subgrad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:n] += 1.0 - h[0:n-k]
            g[0:n-k] -= h[k:n]
    else:
        m = -k
        if m < n:
            g[0:n-m] += 1.0 - h[m:n]
            g[m:n] -= h[0:n-m]
    return g * dx

def _project(h, n):

    for _ in range(100):

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

import numpy as np
import time

def _c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def _sg_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:n] += 1.0 - h[0:n-k]
        if n - k > 0:
            g[0:n-k] -= h[k:n]
    else:
        m = -k
        if m < n:
            g[0:n-m] += 1.0 - h[m:n]
        if n - m > 0:
            g[m:n] -= h[0:n-m]
    return g * dx

def _proj(h, n):
    tgt = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _upscale(h0, n):
    if len(h0) == n:
        return h0.copy()
    x0 = (np.arange(len(h0)) + 0.5) / len(h0)
    x1 = (np.arange(n) + 0.5) / n
    return np.interp(x1, x0, h0)

def _sawtooth(n, p):
    h = 1.0 - (np.arange(n, dtype=float) % p) / p
    return _proj(h, n)

def _rand_h(n, rng):
    return _proj(rng.rand(n), n)

def _optimize(h, n, rng, t_limit):
    t0 = time.time()
    alpha = 0.008
    beta = 0.85
    vel = np.zeros(n)
    bh = h.copy()
    bc, _ = _c5_corr(h)
    topm = min(7, 2 * n - 1)
    for it in range(500000):
        if time.time() - t0 > t_limit:
            break
        c, corr = _c5_corr(h)
        if c < bc - 1e-13:
            bc, bh = c, h.copy()
        top = np.argsort(corr)[-topm:]
        g = np.zeros(n)
        for idx in top:
            k = int(idx) - (n - 1)
            g += _sg_lag(h, k)
        g /= topm
        vel = beta * vel + (1.0 - beta) * g
        h2 = _proj(h - alpha * vel, n)
        c2, _ = _c5_corr(h2)
        if c2 < bc + 1e-13:
            h, alpha = h2, min(alpha * 1.04, 0.045)
        else:
            alpha *= 0.93
            if alpha < 4e-4:
                alpha = 0.012
                ns = max(3, n // 12)
                idxs = rng.choice(n, ns, replace=False)
                h[idxs] += rng.uniform(-0.5, 0.5, ns)
                h = _proj(h, n)
                vel = np.zeros(n)
    return bh, bc
