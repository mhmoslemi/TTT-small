import numpy as np
import time
import random

def compute_c5(h, n):
    """Compute C5 for a binary or general h of length n."""
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), (1 - h).astype(np.float64), mode='full')
    return np.max(corr) * dx

def binarize(h, n):
    """Convert h to binary, keeping n/2 ones."""
    h = np.array(h, dtype=np.float64)
    idx = np.argsort(-h)  # indices of largest values first
    result = np.zeros(n, dtype=np.float64)
    result[idx[:n//2]] = 1.0
    return result

def local_search(h, n, max_iters=10000):
    """Local search by swapping random 0-1 pairs."""
    h = h.copy().astype(np.float64)
    current_c5 = compute_c5(h, n)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for _ in range(max_iters):
        i = random.choice(ones)
        j = random.choice(zeros)
        h[i], h[j] = 0, 1
        new_c5 = compute_c5(h, n)
        if new_c5 < current_c5:
            current_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    
    return best_h, best_c5

def simulated_annealing(h, n, start_temp=0.01, end_temp=0.001, max_iters=50000):
    """Simulated annealing for binary h."""
    h = h.copy().astype(np.float64)
    current_c5 = compute_c5(h, n)
    best_c5 = current_c5
    best_h = h.copy()
    
    for it in range(max_iters):
        T = start_temp * (end_temp / start_temp) ** (it / max_iters)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = random.choice(ones)
        j = random.choice(zeros)
        
        h[i], h[j] = 0, 1
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        
        if delta < 0 or random.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    
    return best_h, best_c5

                best_h = h_fin.copy()
                best_n = n

    if best_h is None:
        if parent is not None:
            pn = len(parent)
            return parent, _compute_c5(parent, pn), pn
        n = 70
        best_h = _random_binary(n, rng)
        best_c5 = _compute_c5(best_h, n)
        best_n = n

    return best_h, best_c5, best_n
