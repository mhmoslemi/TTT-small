def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_single_shift(h, k, n):
    grad = np.zeros(n)
    j = np.arange(n)
    idx1 = j - k + (n - 1)
    v1 = (idx1 >= 0) & (idx1 < n)
    if np.any(v1):
        grad[v1] += (1.0 - h[idx1[v1]])
    idx2 = j + k - (n - 1)
    v2 = (idx2 >= 0) & (idx2 < n)
    if np.any(v2):
        grad[v2] -= h[idx2[v2]]
    return grad

def grad_smooth_max(h, n, t):
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode='full')
    dx = 2.0 / n
    c = corr * dx
    M = np.max(c)
    w = np.exp(t * (c - M))
    w_sum = np.sum(w)

    g = np.zeros(n)
    for k in range(ng):
        g += w[k] * grad_single_shift(h, k, n) * dx
