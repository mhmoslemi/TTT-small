        h2, c5_2 = h_bin, c5_bin

    # Phase 4: Short second SA from best with different seed
    if time.time() < deadline - 20:
        h3, c5_3 = sa_optimize(h2, n, t0, deadline - 20, seed + 7)
        if c5_3 < c5_2:
            h2, c5_2 = h3, c5_3
        h_bin2, c5_bin2 = try_binary(h2, n)
        if c5_bin2 < c5_2:
            h2, c5_2 = h_bin2, c5_bin2

    # Phase 5: Quick polish
    if time.time() < deadline - 5:
        h2, c5_2 = polish(h2, n, deadline - 5)

    # Phase 6: Try resolution refinement if time remains
    if (deadline - time.time()) > 200 and n < 200:
        n_new = n * 2
        h_fine = interp_up(h2, n_new)
        c5_fine = compute_c5(h_fine)
        if time.time() < deadline - 20:
            h_fine, c5_fine = sa_optimize(h_fine, n_new, t0, deadline - 20, seed + 13)
            if c5_fine < c5_2:
                h2, c5_2, n = h_fine, c5_fine, n_new

    return h2, c5_2, n
