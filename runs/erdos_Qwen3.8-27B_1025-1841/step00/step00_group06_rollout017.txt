import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, np.argmax(corr)

def repair_h(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        h += diff / n_points
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argsort(np.abs(h - 0.5))[::-1]
        for i in idx:
            step = min(abs(diff), 1.0 - h[i] if diff > 0 else h[i])
            if step <= 0:
                continue
            h[i] += np.sign(diff) * step
            diff -= np.sign(diff) * step
            if abs(diff) < 1e-10:
                break
    return h

def evaluate(h, n_points):
    return compute_c5(h, n_points)[0]

def local_search(h, n_points, time_limit, rng):
    """Swap-based local search"""
    best_h = h.copy()
    best_val = evaluate(h, n_points)
    t0 = time.time()
    dx = 2.0 / n_points
    while time.time() - t0 < time_limit:
        i = rng.integers(0, n_points)
        j = rng.integers(0, n_points)
        if i == j:
            continue
        # Try swapping
        h_new = h.copy()
        h_new[i], h_new[j] = h[j], h[i]
        h_new = repair_h(h_new, n_points)
        val = evaluate(h_new, n_points)
        if val < best_val - 1e-12:
            best_h = h_new.copy()
            best_val = val
            h = h_new  # accept
    return best_h, best_val

def subgradient_optimize(h, n_points, time_limit, rng):
    """Projected subgradient descent"""
    best_h = h.copy()
    best_val = evaluate(h, n_points)
    t0 = time.time()
    alpha = 0.01
    
    while time.time() - t0 < time_limit:
        val, k_star = compute_c5(h, n_points)
        if val < best_val:
            best_val = val
            best_h = h.copy()

        lag = k_star - (n_points - 1)
        sg = np.zeros(n_points)

            sg[:n_points - lag] += (1.0 - h[lag:])
            sg[lag:] -= h[:n_points - lag]
        
        h = h - alpha * sg
        h = repair_h(h, n_points)
    
        # Anneal alpha
        alpha *= 0.999
    
    return best_h, best_val

    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()

        sgn = np.sign(diff)
        rem = abs(diff)
        order = np.argsort(-h) if sgn > 0 else np.argsort(h)
        for idx in order:
            room = (1.0 - h[idx]) if sgn > 0 else h[idx]
            step = min(rem, room)
            if step < 1e-15:
                break
            h[idx] += sgn * step
            rem -= step
            if rem < 1e-14:
                break
    return h

    sg = np.zeros(n)

        sg[:n - lag] -= h[lag:]
    else:

        sg[:n - m] += 1.0 - h[m:]
        sg[m:] -= h[:n - m]
    return sg
