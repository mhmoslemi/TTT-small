```
def project_polytope(h, c):
    """Project h onto {x: 0<=x<=1, sum(x)=c} using bisection on lambda"""
    lo, hi = -2, 2
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > c:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)
```

```
        k_max = np.argmax(corr)

        grad = np.zeros(n)
```

```python
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * corr * dx
    mx = np.max(vals)
    return float(mx + np.log(np.sum(np.exp(vals - mx))) / alpha)

def soft_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * corr * dx
    mx = np.max(vals)
    weights = np.exp(vals - mx)
    weights /= np.sum(weights)
    # grad of soft max = weighted average of gradients of each corr[k]*dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        shift = n - 1 - k
        # d(corr[k]*dx)/dh[j] = dx * [(1-h[j+shift])*[0<=j+shift<n] - h[j-shift]*[0<=j-shift<n]]
        # Term 1
        i1 = np.arange(np.maximum(0, -shift), min(n, n - shift))
        grad[i1] += weights[k] * (1.0 - h[i1 + shift])
        # Term 2
        i2 = np.arange(np.maximum(0, shift), min(n, n + shift))
        grad[i2] -= weights[k] * h[i2 - shift]
    grad *= dx
    return grad

def project_polytope(h, c):
    """Project h onto {x: 0<=x<=1, sum(x)=c}"""
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    """One projected gradient step using the active (max) correlation"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))

    i2 = np.arange(np.maximum(0, shift), min(n, n + shift))
    grad[i2] -= h[i2 - shift]
    grad *= dx
    
    h_next = h - lr * grad
    h_next = project_polytope(h_next, n / 2.0)
    return h_next, k_max
```

```python
    elif mode == 3:
        # binary-like
        h = rng.uniform(0.0, 1.0, n)
        h = (h > 0.5).astype(float)
    elif mode == 4:
        # sine pattern
        t = np.linspace(0, 2 * np.pi * rng.randint(1, 6), n)
        h = 0.5 + 0.35 * np.sin(t) + 0.1 * rng.randn(n)
        h = np.clip(h, 0.0, 1.0)
    else:
        h = rng.uniform(0.0, 1.0, n)
    return project_polytope(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.ones(80) * 0.5

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    n_base = len(h_init)

    rng = np.random.RandomState(seed)

    resolutions = sorted(set([n_base, 2 * n_base, 3 * n_base, 4 * n_base]))
    resolutions = [r for r in resolutions if r <= 400]

    for n in resolutions:
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.82:
            break

        h_cur = upsample(best_h, n)
        # Estimate per-run time
        per_run = max(3.0, 0.02 * n) if n <= 200 else max(8.0, 0.04 * n)
        remaining = budget_s * 0.82 - elapsed
        max_runs = max(2, int(remaining / per_run))
        max_runs = min(max_runs, 60 if n <= 120 else 25)

        for attempt in range(max_runs):
            elapsed = time.time() - t_start
            if elapsed > budget_s * 0.86:
                break
            leftover = budget_s * 0.86 - elapsed
            if leftover < 5.0:
                break

            mode = attempt % 5
            h_start = make_start(n, h_cur, rng, mode)

            slsqp_budget = min(per_run * 1.5, leftover * 0.6)
            h_opt, val = slsqp_opt(h_start, n, time_limit=slsqp_budget)

            h_opt = project_polytope(h_opt, n / 2.0)
            val = compute_c5(h_opt)

            if val < best_c5:
                best_c5 = val
                best_h = h_opt.copy()
                h_cur = upsample(best_h, n)

    # Final: try a few more PG polishes on the best solution
    n_best = len(best_h)
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.92:
        pg_budget = min(60.0, (budget_s * 0.92 - elapsed))
        for _ in range(5):
            h_pg, v_pg = proj_grad_opt(best_h + rng.randn(n_best) * 0.02,
                                       n_best, budget=pg_budget / 6.0)
            h_pg = project_polytope(h_pg, n_best / 2.0)
            v_pg = compute_c5(h_pg)
            if v_pg < best_c5:
                best_c5 = v_pg
                best_h = h_pg.copy()

    best_h = project_polytope(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
