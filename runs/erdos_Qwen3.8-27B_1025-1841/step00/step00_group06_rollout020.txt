For other lags, it's less clear. But intuitively, binary should be near-optimal.

import numpy as np
import math
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_full(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def binary_from_h(h, n_points, rng):
    n = n_points
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_improve(h, n_points, rng, max_iters, deadline):
    n = n_points
    h = h.astype(float).copy()
    current = compute_c5(h, n)
    best = current
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        idx_ones = np.where(h == 1)[0]
        idx_zeros = np.where(h == 0)[0]
        i = idx_ones[rng.randint(len(idx_ones))]
        j = idx_zeros[rng.randint(len(idx_zeros))]
        h[i], h[j] = 0.0, 1.0
        new = compute_c5(h, n)
        if new <= current:
            current = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best

def anneal(h, n_points, rng, max_iters, t_start, t_end, deadline):
    n = n_points
    h = h.astype(float).copy()
    current = compute_c5(h, n)
    best = current
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max_iters)
        idx_ones = np.where(h == 1)[0]
        idx_zeros = np.where(h == 0)[0]
        i = idx_ones[rng.randint(len(idx_ones))]
        j = idx_zeros[rng.randint(len(idx_zeros))]
        h[i], h[j] = 0.0, 1.0
        new = compute_c5(h, n)
        delta = new - current
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-15)):
            current = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best

def systematic_ls(h, n_points, deadline, max_passes=50):
    n = n_points
    h = h.astype(float).copy()
    current = compute_c5(h, n)
    best = current
    best_h = h.copy()
    one_indices = np.where(h == 1)[0]
    zero_indices = np.where(h == 0)[0]
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(one_indices)):
            if time.time() > deadline:
                break
            i = one_indices[ii]
            for jj in range(len(zero_indices)):
                j = zero_indices[jj]
                h[i], h[j] = 0.0, 1.0
                new = compute_c5(h, n)
                if new < current - 1e-15:
                    current = new
                    if new < best:
                        best = new
                        best_h = h.copy()
                    one_indices = np.where(h == 1)[0]
                    zero_indices = np.where(h == 0)[0]
                    improved = True
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return best_h, best

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h_init = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h_init = np.ones(96)
        h_init[::2] = 0.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_obj = compute_c5(h_init, n_init)

    # Phase 1: Quick optimization at various even resolutions
    resolutions = [96, 128, 160, 192, 256, 320, 384]

    for n_pts in resolutions:
        if time.time() > deadline - 60:
            break

        # Resample
        x_old = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n_pts, endpoint=False)
        h_res = np.interp(x_new, x_old, np.concatenate([h_init, [h_init[0]]]))
        h_bin = binary_from_h(h_res, n_pts, rng)

        # Systematic local search for small n, random for large n
        if n_pts <= 160:
            h_opt, obj = systematic_ls(h_bin, n_pts, deadline)
        else:
            h_opt, obj = swap_improve(h_bin, n_pts, rng, 30000, deadline)

        if obj < best_obj:
            best_obj = obj
            best_h = h_opt.copy()

        # Simulated annealing from the locally optimal solution
        if time.time() < deadline - 30:
            h_sa, obj_sa = anneal(h_opt, n_pts, rng, 50000, 0.008, 0.0001, deadline)
            if obj_sa < best_obj:
                best_obj = obj_sa
                best_h = h_sa.copy()

    # Phase 2: Random restarts at best resolution
    if time.time() < deadline - 60:
        best_n = len(best_h)
        n_restarts = 8
        for r in range(n_restarts):
            if time.time() > deadline - 10:
                break
            h_rand = np.zeros(best_n)
            h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
            h_r, obj_r = anneal(h_rand, best_n, rng, 40000, 0.02, 0.0001, deadline)
            if obj_r < best_obj:
                best_obj = obj_r
                best_h = h_r.copy()

    # Phase 3: Final polish with systematic search at best resolution
    if time.time() < deadline - 5 and len(best_h) <= 160:
        h_final, obj_final = systematic_ls(best_h, len(best_h), deadline, max_passes=100)
        if obj_final < best_obj:
            best_obj = obj_final
            best_h = h_final.copy()
    elif time.time() < deadline - 5:
        h_final, obj_final = swap_improve(best_h, len(best_h), rng, 20000, deadline)
        if obj_final < best_obj:
            best_obj = obj_final
            best_h = h_final.copy()

    # Verify
    final_obj = compute_c5(best_h, len(best_h))
    best_obj = min(best_obj, final_obj)

    return best_h, best_obj, len(best_h)
