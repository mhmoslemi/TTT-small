```python
def project(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    # Bisection on threshold
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        # Add or remove from elements that are strictly between 0 and 1
        mask = (x > 0) & (x < 1)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, 1)
    return x
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - x.sum()
    if abs(d) > 1e-12:
        m = (x > 1e-10) & (x < 1.0 - 1e-10)
        if m.any():
            x[m] += d / m.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def subgrad_step(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)
    ar = np.arange(n)
    grad = np.zeros(n)
    v1 = (ar - lag >= 0) & (ar - lag < n)
    v2 = (ar + lag >= 0) & (ar + lag < n)
    grad[v1] += 1.0 - h[ar[v1] - lag]
    if np.any(v2):
        grad[v2] -= h[ar[v2] + lag]
    grad *= dx
    return grad

def opt_subgrad(h, n, n_iter, t0, time_limit):

    v_mom = np.zeros(n)
    beta = 0.85
    best_local = h.copy()

        if time.time() - t0 > time_limit:
            break
        grad = subgrad_step(h, n)
        lr = 0.08 / (1.0 + 5e-5 * it)
        v_mom = beta * v_mom + (1.0 - beta) * grad
        h = proj(h - lr * v_mom, target)
        if it % 100 == 99:
            val = compute_c5(h, n)

            h = proj(h, target)
    val = compute_c5(h, n)
    if val < best_local_val:
        best_local_val = val
        best_local = h.copy()
    return best_local, best_local_val

def coord_descent(h, n, t0, time_limit):
    target = n / 2.0
    val = compute_c5(h, n)
    for sweep in range(500):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            best_local = h.copy()
            best_v = val
            trials = [0.0, 1.0, 0.25, 0.75, max(0.0, old - 0.1), min(1.0, old + 0.1),
                      max(0.0, old - 0.05), min(1.0, old + 0.05)]
            for t in trials:
                h[i] = t
                h = proj(h, target)
                v = compute_c5(h, n)
                if v < best_v - 1e-12:
                    best_v = v
                    best_local = h.copy()
            if best_v < val - 1e-12:
                h = best_local.copy()
                val = best_v
                improved = True
            else:
                h[i] = old
                h = proj(h, target)
        if not improved:
            break
    return h, val

    h_new = np.interp(np.linspace(0, 2, n_new, endpoint=False),
                      np.linspace(0, 2, len(h_old), endpoint=False), h_old)
    return h_new
```
