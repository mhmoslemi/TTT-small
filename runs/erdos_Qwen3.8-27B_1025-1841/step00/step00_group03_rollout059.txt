                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        T = max(T * cooldown, T_min)
        
        if no_improve > max_no_improve:
            break
    
    return h, best_c5, best_h

def directed_search(h, n, rng, deadline, best_c5, best_h, max_iter):
    target = n // 2
    current_c5 = compute_c5_eval(h, n)
    iters = 0
    
    while time.time() < deadline and iters < max_iter:
        iters += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        d = max_idx - (n - 1)
        
        if d >= 0:
            valid = np.arange(0, n - d)
        else:
            valid = np.arange(-d, n)
        
        if len(valid) == 0:
            break
        
        conflict = []
        for ii in valid:
            jj = ii + d
            if h[ii] == 1.0 and h[jj] == 0.0:
                conflict.append((ii, jj))
        
        if not conflict:
            break
        
        pick = conflict[rng.integers(len(conflict))]
        i, j = pick[0], pick[1]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5_eval(h, n)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return h, best_c5, best_h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
    except:
        n = 70
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0
    
    target = n // 2
    h = make_binary(h0, n, target, rng)
    
    best_c5 = compute_c5_eval(h, n)
    best_h = h.copy()
    best_n = n
    
    # Phase 1: SA at original resolution (50% of time)
    phase1_end = t0 + (budget_s - 3) * 0.45
    h, best_c5, best_h = sa_run(h, n, rng, phase1_end, best_c5, best_h, 0.04, 80000)
    
    # Phase 2: Directed search (15% of time)
    phase2_end = t0 + (budget_s - 3) * 0.60
    h, best_c5, best_h = directed_search(h, n, rng, phase2_end, best_c5, best_h, 500000)
    
    # Phase 3: More SA with restarts (20% of time)
    phase3_end = t0 + (budget_s - 3) * 0.80
    while time.time() < phase3_end:
        h = make_binary(best_h + rng.random(n) * 0.1, n, target, rng)
        sub_end = min(phase3_end, time.time() + 5.0)
        h, best_c5, best_h = sa_run(h, n, rng, sub_end, best_c5, best_h, 0.06, 30000)
    
    # Phase 4: Try higher resolutions (15% of time)
    remaining_deadline = deadline
    for mult in [2, 3, 4]:
        if time.time() > remaining_deadline - 20:
            break
        n_try = n * mult
        target_try = n_try // 2
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_try = make_binary(h_interp, n_try, target_try, rng)
        c5_try = compute_c5_eval(h_try, n_try)
        
        sub_deadline = min(remaining_deadline, time.time() + (remaining_deadline - time.time()) * 0.5)
        h_try, c5_best, best_try = sa_run(h_try, n_try, rng, sub_deadline, c5_try, h_try.copy(), 0.02, 40000)
        
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = best_try.copy()
            best_n = n_try
    
    return best_h, best_c5, best_n
