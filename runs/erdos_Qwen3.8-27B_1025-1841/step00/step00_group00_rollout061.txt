import numpy as np
from time import time

def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / n)

def project(x, n):
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    diff = target - x.sum()
    if abs(diff) < 1e-10:
        return x
    if diff > 0:
        mask = x < 1.0
        cap = 1.0 - x[mask]
        tot = cap.sum()
        if tot > 0:
            x[mask] += cap * (diff / tot)
    else:
        mask = x > 0.0
        cap = x[mask]
        tot = cap.sum()
        if tot > 0:
            x[mask] -= cap * (-diff / tot)
    x = np.clip(x, 0.0, 1.0)
    # Fixup
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        idx = np.arange(n)
        np.random.shuffle(idx)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) < 1e-10:
                break
    return x

def upsample(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = np.arange(old_n) / old_n
    x_new = np.arange(new_n) / new_n
    new_h = np.interp(x_new, x_old, h, left=h[-1], right=h[0])
    # Actually, for the step function interpretation, we should be careful
    # But linear interp should be fine as a starting point
    return project(new_h, new_n)

def local_refine(h, n, max_iter, rng):
    """Coordinate descent: try changing one coordinate at a time"""
    best = h.copy()
    best_val = eval_c5(best)
    for it in range(max_iter):
        improved = False
        # Random order
        order = rng.permutation(n)
        for i in order:
            # Try a few values for h[i]
            for val in [0.0, best[i], 1.0]:
                if val == best[i]:
                    continue
                trial = best.copy()
                trial[i] = val
                trial = project(trial, n)
                val_new = eval_c5(trial)
                if val_new < best_val - 1e-12:
                    best = trial
                    best_val = val_new
                    improved = True
                    break
        if not improved:
            break
    return best, best_val

def project(x, n):
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        mask = x < 1.0
        cap = 1.0 - x[mask]
        tot = cap.sum()
        if tot > 1e-15:
