import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def correct_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = min(diff / cnt, 1.0 - h[mask].max())
            h[mask] += adj
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = min(-diff / cnt, h[mask].min())
            h[mask] -= adj
    return h

def binary_local_search(h, n, max_iters=5000, rng=None):
    if rng is None:
        rng = np.random
    h_b = (h > 0.5).astype(float).copy()
    h_b = correct_sum(h_b, n)
    if abs(np.sum(h_b) - n / 2.0) > 1e-6:
        return h, compute_c5(h, n)
    best = compute_c5(h_b, n)
    ones = np.where(h_b == 1.0)[0]
    zeros = np.where(h_b == 0.0)[0]
    for _ in range(max_iters):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_c = compute_c5(h_b, n)
        if new_c < best - 1e-12:
            best = new_c
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
            ones = np.where(h_b == 1.0)[0]
            zeros = np.where(h_b == 0.0)[0]
    return h_b, best

def smooth_obj(h, n, alpha=15.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - m)))) + m

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')
    best_n = 80

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 80
        h_init = np.full(n_init, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_init)
    best_n = n_init

    n = n_init
    bounds = [(0.0, 1.0)] * n

    def obj_raw(h):
        return compute_c5(h, n)

    def cons_fun(h):
        return np.sum(h) - n / 2.0

    cons_jac = lambda h: np.ones(n)

    # Phase 1: SLSQP from initial
    if tl() > 90:
        try:
            res = minimize(obj_raw, best_h.copy(), method='SLSQP',
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          bounds=bounds, options={'maxiter': 800, 'ftol': 1e-10})
            h_opt = correct_sum(res.x, n)
            c5 = compute_c5(h_opt, n)
            if c5 < best_c5:

        except Exception:
            pass

    if tl() > 90:

            except Exception:
                pass

    if tl() > 30:
        h_bin, c5_bin = binary_local_search(best_h, n, max_iters=8000, rng=rng)
        if c5_bin < best_c5:

        x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
        h_up = np.interp(x_new, x_old, best_h)
        h_up = correct_sum(h_up, n_new)

        def obj_new(h):
            return compute_c5(h, n_new)

            return np.sum(h) - n_new / 2.0
