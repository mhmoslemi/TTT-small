import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_lag(h, shift, n):
    g = np.zeros(n)
    a = max(0, -shift)
    b = min(n, n - shift)
    if b > a:
        g[a:b] += 1.0 - h[a + shift:b + shift]
    a2 = max(0, shift)
    b2 = min(n, n + shift)
    if b2 > a2:
        g[a2:b2] -= h[a2 - shift:b2 - shift]
    return g

def do_grad_desc(h, n, budget_s, t0, rng):
    dx = 2.0 / n
    lr = 0.05
    stall = 0
    c5_cur = c5_value(h)
    for _ in range(8000):
        if time.time() - t0 > budget_s - 20:
            break
        g1 = 1.0 - h
        cor = np.correlate(h, g1, mode='full')
        K = min(5, len(cor))
        top_idx = np.argpartition(cor, -K)[-K:]
        total_g = np.zeros(n)
        for idx in top_idx:
            total_g += grad_lag(h, idx - (n - 1), n)
        total_g *= dx / K
        h_new = project_feasible(h - lr * total_g, n)
        c5_new = c5_value(h_new)
        if c5_new < c5_cur - 1e-12:
            h = h_new
            c5_cur = c5_new
            stall = 0
        else:
            stall += 1
            lr *= 0.98
            if stall % 40 == 0:
                h = h + rng.standard_normal(n) * 0.03
                h = project_feasible(h, n)
                c5_cur = c5_value(h)
                lr = max(lr, 0.02)
                stall = 0
        if stall > 300:
            break
    return h, c5_cur

def do_binary_search(h_init, n, budget_s, t0, rng):
    if n % 2 != 0:
        return h_init, c5_value(h_init)
    h_bin = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        remove = rng.choice(one_idx, n_ones - target, replace=False)
        h_bin[remove] = 0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        add = rng.choice(zero_idx, target - n_ones, replace=False)
        h_bin[add] = 1
    c5_cur = c5_value(h_bin)
    for _ in range(80000):
        if time.time() - t0 > budget_s - 15:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c5_new = c5_value(h_bin)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return h_bin, c5_cur
