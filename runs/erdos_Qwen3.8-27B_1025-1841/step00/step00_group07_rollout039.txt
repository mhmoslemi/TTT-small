from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max()

def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_linear(h, new_n):
    """Upsample h to new_n points using linear interpolation"""
    n = len(h)
    old_x = np.linspace(0, 2, n, endpoint=False) + 1.0/(2*n)  # cell centers
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0/(2*new_n)
    # Use interpolation
    from scipy.interpolate import interp1d
    f = interp1d(old_x, h, kind='linear', fill_value=(h[0], h[-1]), bounds_error=False)
    new_h = f(new_x)
    return project_onto_constraint(new_h, new_n / 2.0)

def objective_and_grad(h, n):
    """Objective: max of correlate(h, 1-h, 'full') * dx
    Returns (value, gradient)"""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, mode="full") * dx
    idx = np.argmax(c)
    value = c[idx]
    # Gradient: dc[idx]/dh for the active constraint
    # c[idx] = dx * sum_i h[i] * (1-h[i+idx-n+1]) for valid i
    # Let k = idx - n + 1
    k = idx - n + 1
    # c[idx] = dx * sum_{i: 0<=i<n, 0<=i+k<n} h[i](1-h[i+k])
    # d/dc dh[j] = dx * [(1-h[j+k]) if 0<=j+k<n] + [-h[j-k] if 0<=j-k<n]
    grad = np.zeros(n)
    # First term: for j where j+k is valid (0<=j+k<n), contribution (1-h[j+k])
    if k >= 0:
        # j ranges from 0 to n-1-k

        grad[valid_j] += (1.0 - h[k:n])
        # Second term: for j where j-k is valid (0<=j-k<n), i.e., j >= k
        # contribution -h[j-k]
        valid_j2 = slice(k, n)
        grad[valid_j2] += -h[0:n-k]
    else:

        kk = -k
        # j ranges from kk to n-1
        valid_j = slice(kk, n)
        grad[valid_j] += (1.0 - h[0:n-kk])
        # j ranges from 0 to n-1-kk
        valid_j2 = slice(0, n - kk)
        grad[valid_j2] += -h[kk:n]
    grad *= dx
    return value, grad

import numpy as np
from scipy.optimize import minimize
import time

_target_sum = 0.0
_tau = 1.0

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def _project(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _upsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    old_spacing = 2.0 / n
    positions = np.arange(new_n) * (2.0 / new_n)
    old_idx = positions / old_spacing
    i0 = np.floor(old_idx).astype(int)
    i1 = np.clip(i0 + 1, 0, n - 1)
    i0 = np.clip(i0, 0, n - 1)
    frac = old_idx - i0
    new_h = h[i0] * (1.0 - frac) + h[i1] * frac
    return _project(new_h, new_n / 2.0)

def _smooth_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    z = _tau * c
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    sum_exp = exp_z.sum()
    value = (z_max + np.log(sum_exp)) / _tau
    p = exp_z / sum_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:] += pm * (1.0 - h[:n - k])
                grad[:n - k] += -pm * h[k:]
        else:
            kk = -k
            if n - kk > 0:
                grad[:n - kk] += pm * (1.0 - h[kk:])
                grad[kk:] += -pm * h[:n - kk]
    grad *= dx
    return value, grad

def _eq_fun(h):
    return h.sum() - _target_sum

def _eq_jac(h):
    return np.ones(len(h))

def _slsqp_solve(h0, n, tau, max_iter=400):
    global _target_sum, _tau
    _tau = tau
    _target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        result = minimize(_smooth_obj, h0, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
        h_out = _project(result.x, n / 2.0)
        return h_out
    except Exception:
        return h0

def run(seed=42, budget_s=1300, **kwargs):

    np.random.seed(seed)
    t_start = time.time()

        else:

            elif r == 2:

            else:
                h_r = np.random.rand(n)
            h_r = _project(h_r, n / 2.0)

            h_r = _slsqp_solve(h_r, n, 60.0, 300)
            c5_r = evaluate_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    n_best = len(best_h)
    best_h = _project(best_h, n_best / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n_best
