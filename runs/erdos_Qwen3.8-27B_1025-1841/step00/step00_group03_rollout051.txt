For a swap: remove a from A (add to B), remove b from B (add to A).

Change in c_k:

- Terms where m+k=a, i.e., m=a-k: was h[m]*(1-h[a]) = h[m]*0 = 0 (since a was in A). Now h[m]*(1-h[a]) = h[m]*1 (since a is now in B). So if m=a-k ∈ A_new (i.e., a-k ∈ A_old and a-k ≠ b), we gain 1.

So total change in (c_k/dx) for the swap:
delta_k = -[a+k ∈ B_old, a+k≠b, in range] + [a-k ∈ A_old, a-k≠b, in range] + [b+k ∈ B_old, b+k≠a, in range] - [b-k ∈ A_old, b-k≠a, in range]

```python
def swap_delta(h, i, j, n):
    """Compute change in each c_k when swapping h[i] (1->0) and h[j] (0->1).
    Returns array of length 2n-1 with delta for each lag in 'full' correlation order."""
    # In full correlation: result[k] = sum_m h[m] * (1-h[m + k - (n-1)])
    # for valid m and m+k-(n-1)
    # So lag in "shift" terms: shift = k - (n-1), ranging from -(n-1) to n-1
    
    delta = np.zeros(2*n-1)
    for k in range(2*n-1):
        shift = k - (n-1)  # actual shift
        # Four terms:
        # 1. m=i, m+shift in B: i+shift in B_old, i+shift ≠ j, both in [0,n-1]
        m1 = i + shift
        if 0 <= m1 < n and m1 != j and h[m1] == 0.0:
            delta[k] -= 1.0
        # 2. m+shift=i, i.e., m=i-shift: (i-shift) in A_old, i-shift≠j, in [0,n-1]
        m2 = i - shift
        if 0 <= m2 < n and m2 != j and h[m2] == 1.0:
            delta[k] += 1.0
        # 3. m=j, m+shift in B: j+shift in B_old, j+shift≠i, in [0,n-1]
        m3 = j + shift
        if 0 <= m3 < n and m3 != i and h[m3] == 0.0:
            delta[k] += 1.0
        # 4. m+shift=j, i.e., m=j-shift: (j-shift) in A_old, j-shift≠i, in [0,n-1]
        m4 = j - shift
        if 0 <= m4 < n and m4 != i and h[m4] == 1.0:
            delta[k] -= 1.0
    
    return delta
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def to_binary(h, n):
    target_counts = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target_counts]] = 1.0
    return b

def repair_continuous(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 0 and abs(s - target) > 1e-8:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        if diff > 0:
            for i in np.random.permutation(n):
                if h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                    if abs(diff) < 1e-9:
                        break
        else:
            for i in np.random.permutation(n):
                if h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
                    if abs(diff) < 1e-9:
                        break
    return h
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 0 and abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        order = np.random.permutation(n)
        for i in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def hill_climb(h, n, time_budget):
    """Full-scan hill climbing on binary sequence"""
    t0 = time.time()
    c5 = compute_c5(h)
    improved = True
    while improved and time.time() - t0 < time_budget:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0
        best_ij = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                delta = new_c5 - c5
                h[i] = 1.0
                h[j] = 0.0
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_ij = (i, j)
        
        if best_ij is not None:
            i, j = best_ij
            h[i] = 0.0
            h[j] = 1.0
            c5 += best_delta
            improved = True
    return h, c5

def anneal(h, n, t_start, end_time):
    """Simulated annealing with single swaps"""
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    temp = 0.5

                temp *= 0.97

            h[j] = 0.0
