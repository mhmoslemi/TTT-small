        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= h[idx + shift] * w
    return grad

def slsqp_optimize(h_init, n, T=40, maxiter=300, time_limit=60):
    h_init = project_sum(h_init, n)
    t0 = time.time()

    def obj(v):
        return smooth_obj(v, n, T)

    def grad(v):
        return smooth_grad(v, n, T)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0,
                   'jac': lambda v: np.ones(n)}]
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-11, 'disp': False})
        if (time.time() - t0) > time_limit:
            return None, None
        h_opt = project_sum(res.x, n)
        c5 = eval_c5(h_opt, n)
        return h_opt, c5
    except Exception:
        return None, None

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, UnboundLocalError):
        h0 = np.full(80, 0.5)

    n0 = len(h0)
    best_h = project_sum(h0, n0)
    best_n = n0
    best_c5 = eval_c5(best_h, best_n)

    def time_up(frac=0.92):
        return (time.time() - start_time) >= budget_s * frac

    # Phase 1: Optimize parent
    h_opt, c5_opt = slsqp_optimize(h0, n0, T=40, maxiter=400, time_limit=80)
    if c5_opt is not None and c5_opt < best_c5:
        best_h, best_c5, best_n = h_opt, c5_opt, n0

    # Phase 2: Random restarts at n0
    for i in range(40):
        if time_up():
            break
        h_rand = rng.rand(n0)
        h_rand = project_sum(h_rand, n0)
        # Perturb parent too
        if i % 2 == 0:
            h_rand = 0.7 * best_h + 0.3 * h_rand
            h_rand = project_sum(h_rand, n0)
        T_use = 30 + 10 * (i % 3)
        h_opt, c5_opt = slsqp_optimize(h_rand, n0, T=T_use, maxiter=250, time_limit=30)
        if c5_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n0

    # Phase 3: Higher resolutions
    for target_n in [160, 320, 480]:
        if time_up():
            break
        if (time.time() - start_time) > budget_s * 0.75:
            break
        # Interpolate best solution to target_n
        x_src = np.arange(best_n) * 2.0 / best_n
        x_dst = np.arange(target_n) * 2.0 / target_n
        h_ext = np.concatenate([best_h, [best_h[0]]])
        x_ext = np.concatenate([x_src, [2.0]])
        h_interp = np.interp(x_dst, x_ext, h_ext)
        h_interp = project_sum(h_interp, target_n)

        elapsed = time.time() - start_time
        remaining = budget_s * 0.92 - elapsed
        tl = max(10, min(120, remaining / 3))
        h_opt, c5_opt = slsqp_optimize(h_interp, target_n, T=35, maxiter=200, time_limit=tl)
        if c5_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, target_n

        # One more random restart at this resolution
        if not time_up():
            h_rand = rng.rand(target_n)
            h_rand = 0.5 * project_sum(h_interp, target_n) + 0.5 * h_rand
            h_rand = project_sum(h_rand, target_n)
            h_opt, c5_opt = slsqp_optimize(h_rand, target_n, T=35, maxiter=150, time_limit=tl)
            if c5_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, target_n

    # Final polish at best resolution
    if not time_up():
        elapsed = time.time() - start_time
        remaining = budget_s * 0.90 - elapsed
        if remaining > 10:
            h_opt, c5_opt = slsqp_optimize(best_h, best_n, T=60, maxiter=500, time_limit=remaining)
            if c5_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    return best_h, best_c5, best_n
