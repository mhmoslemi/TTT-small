import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_feasible(h, N):
    target = N / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_grad_smooth(h, N, temp, penalty):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = (corr - max_c) / temp
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = dx * (max_c + temp * np.log(sum_exp)) + penalty * (np.sum(h) - N / 2.0) ** 2
    weights = exp_s / sum_exp
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        l = k - (N - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        a, b = max(0, l), min(N, N + l)
        if a < b:
            j = np.arange(a, b)
            grad[j] += w * (1.0 - h[j - l])
        c, d = max(0, -l), min(N, N - l)
        if c < d:
            j = np.arange(c, d)
            grad[j] -= w * h[j + l]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(h) - N / 2.0)
    return obj, grad

def interpolate_h(h_old, N_new):
    N_old = len(h_old)
    if N_old == N_new:
        return h_old.copy()
    x_old = np.arange(N_old) / float(N_old)
    x_new = np.arange(N_new) / float(N_new)
    return np.interp(x_new, x_old, h_old)

def optimize_h(h0, N, t0, budget_s):
    temps = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    penalties = [50.0, 100.0, 500.0, 1000.0, 5000.0, 10000.0, 50000.0, 100000.0]
    time_limit = budget_s * 0.82
    for ti in range(len(temps)):
        if time.time() - t0 > time_limit:
            break
        temp = temps[ti]
        pen = penalties[ti]
        maxiter = 300 if N <= 140 else 200
        res = minimize(
            obj_grad_smooth, h0, args=(N, temp, pen),
            jac=True, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * N,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12}
        )
        h0 = project_feasible(res.x, N)
    return h0

def perturb_h(h, N, rng, scale):
    h_pert = h + rng.normal(0.0, scale, N)
    return project_feasible(h_pert, N)

def local_search(h, N, t0, budget_s, rng, n_trials=50):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    time_limit = budget_s * 0.95
    for trial in range(n_trials):
        if time.time() - t0 > time_limit:
            break
        for scale in [0.05, 0.02, 0.01]:
            if time.time() - t0 > time_limit:
                break
            h_pert = perturb_h(best_h, N, rng, scale)
            h_opt = optimize_h(h_pert, N, t0, budget_s)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values

    rng = np.random.default_rng(seed)

    best_c5 = float('inf')
    best_h = None
    best_N = 70

    try:

        pass

    grid_sizes = [70, 100, 140, 200, 280]

    for N in grid_sizes:

            break
        remaining = budget_s * 0.8 - (time.time() - t0)
        n_restarts = max(2, min(int(remaining / 40), 6))

        for restart in range(n_restarts):
            if time.time() - t0 > budget_s * 0.8:
                break

            if restart == 0 and parent_h is not None:
                h0 = interpolate_h(parent_h, N)
            elif restart == 1 and best_h is not None:
                h0 = interpolate_h(best_h, N)
            elif restart == 2:
                # Symmetric step-like pattern
                h0 = np.zeros(N)
                w = N // 4
                h0[w:2*w] = 0.8
                h0[3*w:4*w] = 0.2
                h0 += rng.normal(0, 0.1, N)

                period = max(3, N // 12)
                h0 = 0.5 + 0.35 * np.sin(2.0 * np.pi * np.arange(N) / period)

            else:
                h0 = rng.uniform(0.2, 0.8, N)

            h0 = project_feasible(h0, N)
            h0 = optimize_h(h0, N, t0, budget_s)
            c5 = compute_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
                best_N = N

    # Local search on best solution
    if best_h is not None and time.time() - t0 < budget_s * 0.9:

    if best_h is None:
        N = 70
        best_h = project_feasible(np.full(N, 0.5), N)

        best_N = N

    # Final verification
    best_h = project_feasible(best_h, best_N)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_N
