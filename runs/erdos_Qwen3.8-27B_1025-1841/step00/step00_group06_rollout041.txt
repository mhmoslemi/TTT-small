    best_h = h.copy()

    time_per_n = (budget_s * phase1_frac) / len(n_candidates)
    n_results = []
    
    for n in n_candidates:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 3:
            break
        t_alloc = min(time_per_n, remaining * 0.2)
        
        for start_idx in range(2):
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            if remaining < 2:
                break
            
            t_this = min(t_alloc / 2, remaining * 0.3)
            
            if start_idx == 0 and parent is not None:
                pn = len(parent)
                x_p = np.linspace(0, 2, pn)
                x_n = np.linspace(0, 2, n)
                h_cont = np.interp(x_n, x_p, parent)
                h = binary_from_continuous(h_cont, n)
            else:
                h = structured_init(n, rng)
            
            h_opt, c5_opt = simulated_annealing(h, n, t_this, rng)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
            n_results.append((c5_opt, n, h_opt.copy()))
    
    # Phase 2: Deep optimization
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 10 and n_results:
        n_results.sort(key=lambda x: x[0])
        seen_n = set()
        top_results = []
        for c5, n, h in n_results:
            if n not in seen_n:
                top_results.append((c5, n, h))
                seen_n.add(n)
            if len(top_results) >= 3:
                break
        
        time_per_deep = remaining * 0.75 / len(top_results)
        
        for c5_init, n, h_init in top_results:
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            if remaining < 5:
                break
            
            t_deep = min(time_per_deep, remaining * 0.5)
            n_restarts = max(2, int(t_deep / 30))
            time_per_restart = t_deep / n_restarts
            
            for r in range(n_restarts):
                elapsed = time.time() - start_time
                remaining = budget_s - elapsed
                if remaining < 3:
                    break
                
                if r == 0:
                    h = h_init.copy()
                else:
                    h = structured_init(n, rng)
                
                t_r = min(time_per_restart, (budget_s - start_time - 2) * 0.8)
                if t_r < 1:
                    break
                h_opt, c5_opt = simulated_annealing(h, n, t_r, rng)
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
    
    # Phase 3: Final polish on best solution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5 and best_h is not None:
        h_opt, c5_opt = simulated_annealing(best_h, best_n, remaining * 0.8, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
