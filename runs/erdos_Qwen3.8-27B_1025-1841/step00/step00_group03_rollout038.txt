```python
import numpy as np
import time
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def solve_milp(n, time_limit=120, seed=42):
    """Solve the MILP for a given n."""
    target = n // 2
    
    # Decision variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)
    
    # z variables: z[m, i] = h[i] * h[i+m] for i=0..n-1-m, m=1..n-1
    z = {}
    for m in range(1, n):
        z[m] = cp.Variable(n - m)
    
    constraints = [cp.sum(h) == target]
    
    # Linearization constraints
    for m in range(1, n):
        for i in range(n - m):
            constraints += [
                z[m][i] <= h[i],
                z[m][i] <= h[i + m],
                z[m][i] >= h[i] + h[i + m] - 1
            ]
    
    # Overlap constraints
    for m in range(1, n):
        # Positive lag: sum_{i=0}^{n-1-m} h[i]*(1-h[i+m]) <= t
        pos_overlap = cp.sum(h[:n-m]) - cp.sum(z[m])
        constraints.append(pos_overlap <= t)
        
        # Negative lag: sum_{i=0}^{n-1-m} h[i+m]*(1-h[i]) <= t
        neg_overlap = cp.sum(h[m:]) - cp.sum(z[m])
        constraints.append(neg_overlap <= t)
    
    # Objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.SCS, time_limit_sec=time_limit)
    except:
        try:
            prob.solve(solver=cp.CBC, time_limit=time_limit)
        except:
            prob.solve(solver=cp.SCIP, time_limit=time_limit)
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        h_val = h.value
        if h_val is not None:
            h_val = np.round(h_val)
            h_val = np.clip(h_val, 0, 1)
            return h_val, t.value
    return None, None
```

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n, target, rng):
    """Convert to binary with exactly target ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        to_flip = rng.choice(ones, n_ones - target, replace=False)
        h_bin[to_flip] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        to_flip = rng.choice(zeros, target - n_ones, replace=False)

def steep_descent_1swap(h, max_time=5.0):

    c5 = evaluate_c5(h)
    start = time.time()
    while time.time() - start < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_c5 = evaluate_c5(h)
                delta = new_c5 - c5
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h[i] = 1
                h[j] = 0
        if best_i == -1:

        h[best_i] = 0
        h[best_j] = 1
        c5 += best_delta
    return h, c5
