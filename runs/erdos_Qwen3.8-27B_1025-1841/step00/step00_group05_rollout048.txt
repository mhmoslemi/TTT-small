```python
import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, S):
    """Project h onto {x: 0 <= x <= 1, sum(x) = S}"""
    n = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - S) < 1e-12:
        return h
    # Need to adjust by delta = S - current_sum
    delta = S - current_sum
    # Entries that can still change
    if delta > 0:
        # Need to increase sum; can increase entries < 1
        room = 1 - h
        total_room = np.sum(room)
        if total_room < 1e-12:
            return h
        h += delta * room / total_room
    else:
        # Need to decrease sum; can decrease entries > 0
        room = h
        total_room = np.sum(room)
        if total_room < 1e-12:
            return h
        h += delta * room / total_room
    # Might need to clip again if the adjustment pushed some out of range
    for _ in range(10):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - S) < 1e-10:
            break
        delta = S - current_sum
        if delta > 0:
            mask = h < 1 - 1e-12
            if not np.any(mask):
                break
            h[mask] += delta / np.sum(mask)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += delta / np.sum(mask)
    return np.clip(h, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def smooth_max(c, beta):
    """Smooth max using log-sum-exp"""
    m = np.max(c)
    if beta < 1e-6:
        return m
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta
```

For v(x) = sum_j a_j cos(2π f_j x + φ_j), the autocorrelation is:
(v⊛v)(k) = sum_j a_j²/2 cos(2π f_j k) + cross terms

If we choose frequencies that are incommensurate, the cross terms average out, and the autocorrelation is approximately a sum of cosines. To make it flat (close to 0 for k≠0), we need the cosines to cancel.
