    conv = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(conv)

    epsilon = 1e-6
    active = np.where(conv >= max_val - epsilon)[0]
    
    # Compute subgradient
    g = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # Actual shift
        # c_k = sum_i h[i+k] * (1-h[i])
        # dc_k/dh[j] = (1-h[j-k]) if j is in the "first" position (h[j+k] where j+k is valid)
        #             + (-h[j+k]) if j is in the "second" position (1-h[j] where j is valid)
        # More carefully:
        # c_k = sum_i h[i+k] * (1-h[i]) for i where 0<=i<n and 0<=i+k<n
        # dc_k/dh[j] = sum over terms involving h[j]
        #   If j = i+k (j is the first index): contributes (1-h[i]) = (1-h[j-k])
        #   If j = i (j is the second index): contributes -h[i+k] = -h[j+k]

        # Contribution: (1-h[j-k]) if 0 <= j-k < n
        # For the second role: j = i, so i+k = j+k. Valid when 0<=j<n and 0<=j+k<n

        for j in range(n):

    """Project h to [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0, 1)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to decrease sum by diff
        # Decrease values that are > 0, proportional to their values
        mask = h > 0
        adj = diff / mask.sum()
        h[mask] -= adj
    else:
        # Need to increase sum by -diff
        mask = h < 1
        adj = (-diff) / mask.sum()

    remaining = h.sum() - target_sum
    if abs(remaining) > 1e-12:

        else:
            mask = h < 1 - 1e-10

            h[mask] -= remaining / mask.sum()
            h = np.clip(h, 0, 1)

    return h_bin

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 1e-10
        else:
            mask = h < 1.0 - 1e-10
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sa_binary(h, t0=0.015, cool_rate=0.9998, time_limit=300, rng=None):
    if rng is None:
        rng = np.random
    n = len(h)
    h = h.astype(float).copy()
    best_h = h.copy()
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    t_start = time.time()
    T = t0
    it = 0
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    stall = 0
    while time.time() - t_start < time_limit:
        it += 1
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 50000 and T > 0.001:
                T *= 0.5
                stall = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
        if it % 5000 == 0:
            T *= cool_rate
    return best_h, best_c5

def sa_binary_double(h, t0=0.01, cool_rate=0.9995, time_limit=100, rng=None):
    if rng is None:
        rng = np.random
    n = len(h)
    h = h.astype(float).copy()
    best_h = h.copy()
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    t_start = time.time()
    T = t0
    while time.time() - t_start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if i1 == i2 or j1 == j2:
            continue
        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
        if np.random.random() < 0.001:
            T *= cool_rate
    return best_h, best_c5

def subgradient_method(h, time_limit=100, alpha0=0.05, rng=None):
    if rng is None:
        rng = np.random
    n = len(h)
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    alpha = alpha0
    target_sum = n / 2.0
    while time.time() - t_start < time_limit:
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(conv)
        epsilon = max(1e-4, 0.01 * max_val)
        active = np.where(conv >= max_val - epsilon)[0]
        g = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)
            for j in range(n):
                c = 0.0
                if 0 <= j - k < n:
                    c += (1.0 - h[j - k])
                if 0 <= j + k < n:
                    c += (-h[j + k])
                g[j] += c
        g /= max(len(active), 1)
        h_new = h - alpha * g
        h_new = project_sum(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
        else:
            alpha *= 0.95
            if alpha < 1e-4:
                h_new = h + 0.01 * rng.randn(n)
                h_new = project_sum(h_new, target_sum)
                h = h_new
                alpha = alpha0 * 0.5
        if int(time.time() - t_start) % 2 == 0:
            alpha = max(alpha * 0.99, 1e-5)
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.append(h, h[0]))
    h_new = np.clip(h_new, 0, 1)
    h_new = project_sum(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_global_start = time.time()
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    def time_left():
        return budget_s - (time.time() - t_global_start)
    
    # Phase 1: Try binary version
    h_bin = to_binary(h_init)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin
    
    # Phase 2: Binary SA at initial resolution
    tl = time_left() - 400
    if tl > 30:
        h_sa, c5_sa = sa_binary(best_h, t0=0.015, cool_rate=0.9998, time_limit=tl, rng=rng)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa
    
    # Phase 3: Double-swap SA
    tl = time_left() - 200
    if tl > 20:
        h_d, c5_d = sa_binary_double(best_h, t0=0.01, time_limit=tl, rng=rng)
        if c5_d < best_c5:
            best_h, best_c5 = h_d, c5_d
    
    # Phase 4: Fractional subgradient refinement
    tl = time_left() - 100
    if tl > 20:
        h_sub, c5_sub = subgradient_method(best_h, time_limit=tl, alpha0=0.03, rng=rng)
        if c5_sub < best_c5:
            best_h, best_c5 = h_sub, c5_sub
    
    # Phase 5: Try nearby resolutions
    for n_try in [n_init + 2, n_init - 2, n_init + 10, n_init + 20, 2 * n_init]:
        if n_try < 10 or n_try % 2 != 0:
            continue
        tl = time_left() - 60
        if tl < 15:
            break
        h_try = upsample(best_h, n_try)
        h_try = to_binary(h_try)
        alloc = min(tl * 0.4, 60)
        h_try, c5_try = sa_binary(h_try, t0=0.012, cool_rate=0.9997, time_limit=alloc, rng=rng)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try
    
    # Phase 6: Final SA polish on best solution
    tl = time_left() - 10
    if tl > 10:
        h_final = to_binary(best_h)
        h_final, c5_final = sa_binary(h_final, t0=0.008, cool_rate=0.9999, time_limit=tl, rng=rng)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final
        else:
            # Also try fractional
            h_frac = best_h.astype(float)
            h_frac, c5_frac = subgradient_method(h_frac, time_limit=max(tl - 20, 5), alpha0=0.02, rng=rng)
            if c5_frac < best_c5:
                best_h, best_c5 = h_frac, c5_frac
    
    # Ensure feasibility
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_final)
