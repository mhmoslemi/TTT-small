```python
def eval_obj(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            c = int(m.sum())
            if c == 0:
                break
            increase = d / c
            h[m] = np.minimum(h[m] + increase, 1.0)
        else:
            m = h > 0.0
            c = int(m.sum())
            if c == 0:
                break
            decrease = d / c
            h[m] = np.maximum(h[m] + decrease, 0.0)
    return h

                    bf, bx = f, x

                best = bf
                imp = True
        if not imp:
            break

    elif n_ones < n // 2:

        flip = rng.choice(zeros, size=n // 2 - n_ones, replace=False)
        h[flip] = 1

        if i == j or h[i] == h[j]:

        h[i], h[j] = h[j], h[i]
        f = eval_obj(h, n)
        if f < best:
            best = f
            h_best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
```

```python
def compute_corr(h, n):
    """Compute full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def update_corr_after_change(corr, h, i, new_val, n):
    """Update correlation after changing h[i] to new_val."""
    old_val = h[i]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return corr
    
    corr = corr.copy()
```

```python
def make_update_vectors(h, i, n):
    """Precompute vectors for incremental correlation update at index i."""
    m0 = n - 1
    s = np.arange(2*n-1) - m0  # shifts
    
    # left[m] = 1 - h[i - s[m]] if 0 <= i - s[m] < n
    idx_left = i - s
    left = np.zeros(2*n-1)
    valid_l = (idx_left >= 0) & (idx_left < n)
    left[valid_l] = 1.0 - h[idx_left[valid_l]]
    
    # right[m] = h[i + s[m]] if 0 <= i + s[m] < n
    idx_right = i + s
    right = np.zeros(2*n-1)
    valid_r = (idx_right >= 0) & (idx_right < n)
    right[valid_r] = h[idx_right[valid_r]]
    
    return left, right, m0
```
