from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -50, 50)
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(s)))

def repair_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    scale = target / max(s, 1e-15)
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if diff > 0:
        room = 1.0 - h
        tr = np.sum(room)
        if tr > 1e-15:
            h += diff * room / tr
    elif diff < 0:
        mass = h.copy()
        tm = np.sum(mass)
        if tm > 1e-15:
            h += diff * mass / tm
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h, new_n):
    old_n = len(h)

    new_h = np.interp(new_x, old_x_ext, h_ext)
