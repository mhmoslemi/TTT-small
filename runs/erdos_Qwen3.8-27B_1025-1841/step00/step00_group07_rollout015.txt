```python
def compute_c5(h):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Need to add diff/total_spread to make sum correct
    # Use bisection on the shift
    # h_proj[i] = clip(h[i] + s, 0, 1)
    # Find s such that sum(h_proj) = target
    
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_try = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_try)
        if s < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    return h_proj

def smooth_max_and_grad(h, tau=30.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj_vals = corr * dx
    m = np.max(obj_vals)
    shifted = (obj_vals - m) * tau
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    obj = m + np.log(total) / tau
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            j1 = np.arange(0, n - lag)
            j2 = np.arange(lag, n)
        else:
            j1 = np.arange(-lag, n)
            j2 = np.arange(n + lag, n)
        grad[j1] += w * (1.0 - h[j1 + lag])
        grad[j2] -= w * h[j2 - lag]
    
    return obj, grad

def slsqp_objective(h, tau):
    obj, _ = smooth_max_and_grad(h, tau)
    return obj

def slsqp_gradient(h, tau):
    _, grad = smooth_max_and_grad(h, tau)
    return grad

def optimize_slsqp(h0, n, tau=30.0, time_limit=None, start_time=None):
    x0 = h0.copy()
    x0 = project_feasible(x0, n)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    if time_limit is not None:
        remaining = time_limit - (time.time() - start_time)
        if remaining < 5:
            return None
        maxiter = max(50, int(remaining * 2))
    else:
        maxiter = 500
    
    res = minimize(slsqp_objective, x0, jac=slsqp_gradient,
                  method='SLSQP', bounds=bounds, constraints=constraints,
                  args=(tau,), options={'maxiter': maxiter, 'ftol': 1e-9})
    
    if res.success or res.status == 0:
        h_opt = project_feasible(res.x, n)
        return h_opt
    # Even if not fully successful, check the result
    if hasattr(res, 'x') and res.x is not None:
        h_opt = project_feasible(res.x, n)
        return h_opt
    return None

def random_binary_h(n, rng):
    h = np.zeros(n)
    indices = rng.choice(n, size=n//2, replace=False)
    h[indices] = 1.0
    return h

def random_feasible(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def subgradient_polish(h, n, n_iters=2000, alpha_start=0.1, alpha_end=0.001):
    """Polish using subgradient method on the hard max"""
    h = h.copy()
    for it in range(n_iters):
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / n_iters)
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_best = np.argmax(corr)
        lag = k_best - (n - 1)
        
        # Gradient at the active lag
        if lag >= 0:
            j1 = np.arange(0, n - lag)
            j2 = np.arange(lag, n)
        else:
            j1 = np.arange(-lag, n)
            j2 = np.arange(n + lag, n)
        
        grad = np.zeros(n)
        grad[j1] = 1.0 - h[j1 + lag]
        grad[j2] -= h[j2 - lag]
        
        h = h - alpha * grad
        h = project_feasible(h, n)
    
    return h
```

```python
def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def _corr_and_weights(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = (corr - m) * tau
    s -= np.max(s)
    e = np.exp(s)
    w = e / np.sum(e)
    obj = m + np.log(np.sum(e)) / tau
    return obj, w, corr

def smooth_obj(h, tau):
    obj, _, _ = _corr_and_weights(h, tau)
    return obj

def smooth_grad(h, tau):
    obj, w, corr = _corr_and_weights(h, tau)
    n = len(h)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            j1 = np.arange(0, n - lag)
            j2 = np.arange(lag, n)
        else:
            j1 = np.arange(-lag, n)
            j2 = np.arange(n + lag, n)
        grad[j1] += wi * (1.0 - h[j1 + lag])
        grad[j2] -= wi * h[j2 - lag]
    return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, x0, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons, args=(tau,),
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            return project_feasible(res.x, n)
    except:
        pass
    return None

        corr = np.correlate(h, 1.0 - h, mode="full")
        k = np.argmax(corr)
        lag = k - (n - 1)
        g = np.zeros(n)

            j2 = np.arange(lag, n)
        else:

            j2 = np.arange(n + lag, n)
        g[j1] = 1.0 - h[j1 + lag]
        g[j2] -= h[j2 - lag]
        h = project_feasible(h - a * g, n)
    return h
```

    target = n / 2.0
    h = np.asarray(h, dtype=float)
