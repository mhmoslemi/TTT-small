import numpy as np
from scipy.optimize import minimize
import time

def _c5_fft(h, n):
    """Compute C5 using FFT-based correlation."""
    N = 2 * n - 1
    H = np.fft.rfft(h, N)
    C = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(C))
    dx = 2.0 / n
    return np.max(corr) * dx

def _grad_c5(h, n):
    """Compute subgradient of C5."""
    N = 2 * n - 1
    H = np.fft.rfft(h, N)
    C = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(C))
    k = np.argmax(corr)
    m = k - (n - 1)
    g = np.zeros(n)
    # For each j, the contribution to corr[k] from h[j]:
    # corr[k] = sum_i h[i] * (1 - h[i-m]) for valid i
    # d/dh[j]: (1-h[j-m]) if j-m in [0,n), plus -h[j+m] if j+m in [0,n)
    # But we need to be careful: h[j] appears as h[i] when i=j, and as h[i-m] when i=j+m
    valid1 = (j_idx >= 0) & (j_idx < n)  # will compute below
    for j in range(n):
        jm = j - m
        jp = j + m
        if 0 <= jm < n:
            g[j] += 1.0 - h[jm]
        if 0 <= jp < n:
            g[j] -= h[jp]
    dx = 2.0 / n
    return g * dx

        return budget_s - (time.time() - t0)

    def do_slsqp(h_init, n, max_iter=300):
        dx = 2.0 / n
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
                 'jac': lambda h: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        res = minimize(_c5_fft, h_init.copy(), args=(n,),
                       method='SLSQP', jac=lambda h: _grad_c5(h, n),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        return res.x, res.fun

            mask = rng.rand(n) < 0.3

import numpy as np
from scipy.optimize import minimize
import time

def _c5_fft(h, n):
    N = 2 * n - 1
    H = np.fft.rfft(h, N)
    C = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(C))
    return np.max(corr) * (2.0 / n)

def _grad_c5(h, n):
    N = 2 * n - 1
    H = np.fft.rfft(h, N)
    C = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(C))
    k = np.argmax(corr)
    m = k - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, m)
    hi1 = min(n, n + m)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        g[idx] += 1.0 - h[idx - m]
    lo2 = max(0, -m)
    hi2 = min(n, n - m)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        g[idx] -= h[idx + m]
    return g * (2.0 / n)

def _fix_integral(h, n):
    s = np.sum(h)
    if s < 1e-15:
        return np.ones(n) * 0.5
    h = h * (n / 2.0) / s
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        nm = np.sum(mask)
        if nm > 0:
            h[mask] += diff / nm
            h = np.clip(h, 0.0, 1.0)
    return h

def _do_slsqp(h_init, n, max_iter=300, t0=None, budget_s=None):
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def obj(h):
        return _c5_fft(h, n)

    def jac(h):
        return _grad_c5(h, n)

    res = minimize(obj, h_init.copy(), method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x, res.fun
