import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    h1m = 1.0 - h
    corr = np.correlate(h, h1m, mode='full')
    m_star = int(np.argmax(corr))
    k = m_star - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = h1m - h
    elif k > 0:
        grad[k:] += h1m[:n - k]
        grad[:n - k] -= h[k:]
    else:
        kk = -k
        grad[:n - kk] += h1m[kk:]
        grad[kk:] -= h[:n - kk]
    return grad * dx, m_star, k

def project_feasible(v, target_sum):
    v = np.asarray(v, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    proj = np.clip(v - t, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def subgrad_optimize(h0, max_time, alpha0=0.3, decay=0.999):
    h = h0.astype(float).copy()
    n = len(h)
    target = n * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    alpha = alpha0
    stagnant = 0
    while time.time() - t0 < max_time:
        grad, _, _ = compute_subgradient(h)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = alpha / gnorm
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 300:
                alpha *= 0.5
                stagnant = 0
        h = h_new
        alpha = max(alpha * decay, 1e-8)
    return best_h, best_c5

def multi_scale_init(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for scale in [2, 4, 8, 16, 32]:
        if scale > n:
            break
        for i in range(n):
            if (i // scale) % 2 == 0:
                h[i] = min(1.0, h[i] + 1.0 / scale)
            else:
                h[i] = max(0.0, h[i] - 1.0 / scale)
    h += rng.uniform(-0.1, 0.1, n)
    return project_feasible(h, n * 0.5)
