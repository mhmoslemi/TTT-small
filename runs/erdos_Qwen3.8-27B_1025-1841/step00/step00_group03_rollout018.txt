```python
        if diff > 0:
            idx = np.where(h < 1.0)[0]

                break
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

    def do_optimize(h_start, n, maxiter=300, ftol=1e-10):
        dx = 2.0 / n
        
        def obj(hv):
            corr = np.correlate(hv, 1.0 - hv, mode="full")
            return float(np.max(corr) * dx)

            k = int(np.argmax(corr))

            eps = 1e-6
            for i in range(n):

                hp[i] = np.clip(hp[i] + eps, 0, 1)
```

```python
def grad_active_lag(h, k, n):
    """Gradient of correlation at index k w.r.t. h, where k is in [0, 2n-2]"""
    m = k - (n - 1)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0:
        # i in [m, n-1]: grad += (1-h[i-m])
        g[m:] += (1.0 - h[:-m]) if m < n else 0
        # Wait, h[i-m] for i in [m, n-1] is h[0] to h[n-1-m]
        g[m:] += (1.0 - h[:n-m])  # h[i-m] where i goes m to n-1, so i-m goes 0 to n-1-m
        # i in [0, n-1-m]: grad -= h[i+m]
        g[:n-m] -= h[m:]  # h[i+m] where i goes 0 to n-1-m, so i+m goes m to n-1
    else:
        mp = -m
        # i in [0, n-1-mp]: grad += (1-h[i+mp])
        g[:n-mp] += (1.0 - h[mp:])  # h[i+mp] where i goes 0 to n-1-mp
        # i in [mp, n-1]: grad -= h[i-mp]
        g[mp:] -= h[:n-mp]  # h[i-mp] where i goes mp to n-1, so i-mp goes 0 to n-1-mp
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_active(h, k, n):
    m = k - (n - 1)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        mp = -m
        g[:n - mp] += (1.0 - h[mp:])
        g[mp:] -= h[:n - mp]
    return g

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_init = repair(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    def time_left():
        return budget_s - 20 - (time.time() - t_start)

    def do_optimize(h_start, n, maxiter=400, ftol=1e-11):
        dx = 2.0 / n

        def obj(hv):
            corr = np.correlate(hv, 1.0 - hv, mode="full")
            return float(np.max(corr) * dx)

        def grad(hv):
            corr = np.correlate(hv, 1.0 - hv, mode="full")
            k = int(np.argmax(corr))
            return grad_active(hv, k, n) * dx

        def sum_eq(hv):
            return hv.sum() - n / 2.0

        cons = [{'type': 'eq', 'fun': sum_eq}]
        bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(obj, h_start, jac=grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_out = repair(res.x, n)
            c5_val = compute_c5(h_out)

            h_out = repair(h_start, n)
            c5_val = compute_c5(h_out)
        return h_out, c5_val

        h_opt, c5 = do_optimize(h, n_init, maxiter=300)
        if c5 < best_c5:
```
