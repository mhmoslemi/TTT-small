import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    return float(np.max(np.correlate(h, 1-h, mode="full") * dx))

def project_to_set(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h |= h + (diff / tr) * room  # No, use | for bitwise? No.
            h = h + (diff / tr) * room
        else:
            tr = h.sum()
            if tr < 1e-15:
                break
            h = h + (diff / tr) * h
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where(0 < h < 1)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0, 1)
    return h

def compute_grad_c(h, dx):
    """Compute gradients of all correlation values w.r.t. h. Returns (2n-1, n) array."""
    n = len(h)
    np_ = 2 * n - 1
    grads = np.zeros((np_, n))
    for p in range(np_):
        k = p - (n - 1)
        if k >= 0:
            # First term: (1-h[j-k]) * I[k <= j <= n-1]
            lo, hi = k, n - 1
            if lo <= hi:
                j_idx = np.arange(lo, hi + 1)
                grads[p, j_idx] += (1.0 - h[j_idx - k])
            # Second term: -h[j+k] * I[0 <= j <= n-1-k]
            lo2, hi2 = 0, n - 1 - k
            if lo2 <= hi2:
                j_idx2 = np.arange(lo2, hi2 + 1)
                grads[p, j_idx2] -= h[j_idx2 + k]
        else:
            m = -k
            # First term: (1-h[j+m]) * I[0 <= j <= n-1-m]
            lo, hi = 0, n - 1 - m
            if lo <= hi:
                j_idx = np.arange(lo, hi + 1)
                grads[p, j_idx] += (1.0 - h[j_idx + m])
            # Second term: -h[j-m] * I[m <= j <= n-1]
            lo2, hi2 = m, n - 1
            if lo2 <= hi2:
                j_idx2 = np.arange(lo2, hi2 + 1)
                grads[p, j_idx2] -= h[j_idx2 - m]
    grads *= dx
    return grads

def lse_grad(h, dx, tau):
    """LSE-smoothed max objective and its gradient"""
    n = len(h)
    corrs = np.correlate(h, 1-h, mode="full") * dx
    # Softmax weights
    shifted = tau * (corrs - corrs.max())
    exps = np.exp(shifted)
    w = exps / exps.sum()
    # Objective value
    obj = corrs.max() + np.log(exps.sum() / len(corrs)) / tau
    # Gradient
    grads = compute_grad_c(h, dx)  # shape (2n-1, n)
    grad = w @ grads  # shape (n,)
    return obj, grad

    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def project_to_set(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target_sum:

        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    return np.clip(h - mu, 0.0, 1.0)

def compute_all_grads(h, dx):
    n = len(h)
    np_ = 2 * n - 1
    grads = np.zeros((np_, n))
    for p in range(np_):
        k = p - (n - 1)
        if k >= 0:
            j1 = np.arange(k, n)
            grads[p, j1] += (1.0 - h[j1 - k])
            j2 = np.arange(0, n - k)
            if len(j2) > 0:
                grads[p, j2] -= h[j2 + k]
        else:
            m = -k
            j1 = np.arange(0, n - m)
            if len(j1) > 0:
                grads[p, j1] += (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            grads[p, j2] -= h[j2 - m]
    grads *= dx
    return grads

def lse_objective_and_grad(h, dx, tau):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = tau * (corrs - corrs.max())
    exps = np.exp(shifted)
    s = exps.sum()
    w = exps / s
    obj = corrs.max() + np.log(s) / tau - np.log(len(corrs)) / tau
    # Actually: (1/tau)*log(sum(exp(tau*c))) = (1/tau)*log(sum(exp(tau*(c-cmax)))) + cmax
    # = cmax + (1/tau)*log(sum(exp(tau*(c-cmax))))
    # The log(s) where s = sum(exp(tau*(c-cmax)))
    # So obj = cmax + log(s)/tau
    # But this overestimates max by log(n)/tau for uniform weights
    # For our purpose, we just need the gradient to be correct
    grads = compute_all_grads(h, dx)
    grad = w @ grads
    return obj, grad

                     time_budget=None, t_start=None, seed=42):

        h_new = project_to_set(h_new, target)

            if tau > 50:

    return h_cur, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def project_to_set(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    return np.clip(h - mu, 0.0, 1.0)

def compute_all_grads(h, dx):
    n = len(h)
    np_ = 2 * n - 1
    grads = np.zeros((np_, n))
    for p in range(np_):
        k = p - (n - 1)
        if k >= 0:
            j1 = np.arange(k, n)
            if len(j1) > 0:
                grads[p, j1] += (1.0 - h[j1 - k])
            j2 = np.arange(0, n - k)
            if len(j2) > 0:
                grads[p, j2] -= h[j2 + k]
        else:
            m = -k
            j1 = np.arange(0, n - m)
            if len(j1) > 0:
                grads[p, j1] += (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            if len(j2) > 0:
                grads[p, j2] -= h[j2 - m]
    grads *= dx
    return grads

def lse_objective_and_grad(h, dx, tau):
    corrs = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = tau * (corrs - corrs.max())
    exps = np.exp(shifted)
    s = exps.sum()
    w = exps / s
    obj = corrs.max() + np.log(s) / tau
    grads = compute_all_grads(h, dx)
    grad = w @ grads
    return obj, grad

def gradient_descent(h, dx, target, tau0, lr0, max_iter, t_start, time_budget, seed):
    tau = tau0
    lr = lr0
    best_c5 = eval_c5(h, dx)
    h_cur = h.copy()
    stagnant = 0
    for it in range(max_iter):
        if time.time() - t_start > time_budget - 5:
            break
        obj, grad = lse_objective_and_grad(h_cur, dx, tau)
        h_new = h_cur - lr * grad
        h_new = project_to_set(h_new, target)
        c5_new = eval_c5(h_new, dx)
        c5_cur = eval_c5(h_cur, dx)
        if c5_new < c5_cur + 1e-15:
            h_cur = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
            stagnant = 0
            lr = min(lr * 1.005, 0.15)
        else:
            stagnant += 1
            if stagnant > 20:
                lr *= 0.7
                stagnant = 0
        if it % 400 == 0 and it > 0:
            tau = min(tau * 1.4, 80.0)
    return h_cur, best_c5

def nm_optimize(h, dx, target, max_iter, t_start, time_budget):
    def obj(x):
        if time.time() - t_start > time_budget - 3:
            return eval_c5(project_to_set(x, target), dx) + 10.0
        xp = project_to_set(x, target)
        return eval_c5(xp, dx)
    try:
        res = minimize(obj, h.copy(), method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-10,
                                'fatol': 1e-12, 'adaptive': True})
        return project_to_set(res.x, target)
    except Exception:
        return h

def upscale(h, n_new):
    n_old = len(h)
    old_x = np.linspace(0.0, 2.0, n_old, endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h, left=h[0], right=h[-1])
