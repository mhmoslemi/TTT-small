import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def objective_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def constraint_sum(h, n):
    return np.sum(h) - n / 2.0


def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)


def project_simplex(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def slsqp_optimize(h_init, n, maxiter, ftol):
    dx = 2.0 / n
    h0 = project_simplex(h_init, n)
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(objective_c5, h0, args=(dx,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.x is not None:
        h_opt = project_simplex(np.clip(res.x, 0.0, 1.0), n)
        return h_opt, compute_c5(h_opt)
    return None, np.inf


def local_search_binary(h, n, passes=50):
    """Local search on binary-like vectors: swap pairs to maintain sum=n/2."""
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(passes):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            # Convert to binary first
            sorted_idx = np.argsort(-h)
            ones = sorted_idx[:n // 2]
            zeros = sorted_idx[n // 2:]
            h = np.zeros(n)
            h[ones] = 1.0
            h[zeros] = 0.0
            c5 = compute_c5(h)
        for i in range(len(ones)):
            for j in range(len(zeros)):
                h[ones[i]] = 0.0
                h[zeros[j]] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < c5 - 1e-12:
                    c5 = new_c5
                    improved = True
                    break
                else:
                    h[ones[i]] = 1.0
                    h[zeros[j]] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, c5


def perturb_and_optimize(h_init, n, maxiter, rng):
    """Add noise then optimize."""
    noise = rng.normal(0, 0.05, size=n)
    h_noisy = project_simplex(h_init + noise, n)
    return slsqp_optimize(h_noisy, n, maxiter, 1e-10)

    # Phase 4: If time remains, do a final polish at the best resolution

    # Final validation
