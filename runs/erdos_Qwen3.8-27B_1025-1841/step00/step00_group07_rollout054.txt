import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h += diff / n
    return np.clip(h, 0.0, 1.0)

def soft_gradient(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = corr / tau
    c -= c.max()

    grad = np.zeros(n)

            ip = i + s
            if 0 <= ip < n:

            im = i - s
            if 0 <= im < n:
                g -= h[im]
            grad[i] += pm * dx * g
    return grad

        h, c5 = gd_optimize(h_up, n_new, tau=0.01, max_iter=2000,
                           lr=0.003, time_limit=budget_s - 150, t0=t0)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, n_new

    # Phase 6: SLSQP at final resolution
    if remaining() > 100:
        h, c5 = slsqp_optimize(best_h, best_n, max_iter=200)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: Final GD polish
    if remaining() > 60:
        h, c5 = gd_optimize(best_h, best_n, tau=0.005, max_iter=3000,
                           lr=0.003, time_limit=budget_s - 10, t0=t0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Ensure feasibility
    best_h = proj_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
