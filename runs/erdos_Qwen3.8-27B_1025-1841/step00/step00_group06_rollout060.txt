def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-8) & (h < 1.0 - 1e-8)
    if np.sum(free) > 0:
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0, 1)
    # Final correction
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-10:
        idx = np.argmax(h if diff > 0 else 1-h)
        h[idx] += diff
        h = np.clip(h, 0, 1)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

            grad[js] += dx * wk * (-h[js - ell])

def slp_step(h, n):
    """One SLP iteration: linearize and solve LP"""
    dx = 2.0 / n
    # Compute current corr and gradients
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    K = 2 * n - 1
    
    # Build LP: minimize t
    # Variables: h[0..n-1], t (total n+1 variables)
    # Constraints:
    #   For each k: -grad_k(h)·h + corr_k_const ≤ t  (same as grad_k(h)·h + t ≥ corr_k_const)
    #   sum(h) = n/2
    #   0 ≤ h[i] ≤ 1
    #   t ≥ 0
    
    # Actually: linearize corr_k at h*: corr_k(h) ≈ corr_k(h*) + grad_k·(h-h*)
    # Want: corr_k(h) ≤ t, i.e., corr_k(h*) + grad_k·(h-h*) ≤ t
    # => -grad_k·h + t ≤ corr_k(h*) - grad_k·h*
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((K + n, n + 1))  # A_ub @ x ≤ b_ub
    b_ub = np.zeros(K + n)
    
    for k in range(K):
        ell = k - (n - 1)
        gk = np.zeros(n)
        lo1 = max(0, -ell)
        hi1 = min(n, n - ell)
        if hi1 > lo1:
            gk[lo1:hi1] += dx * (1.0 - h[np.arange(lo1, hi1) + ell])
        lo2 = max(0, ell)
        hi2 = min(n, n + ell)
        if hi2 > lo2:
            gk[lo2:hi2] += dx * (-h[np.arange(lo2, hi2) - ell])
        
        # -gk·h + t ≤ corr_k(h) - gk·h*
        A_ub[k, :n] = -gk
        A_ub[k, n] = 1.0
        b_ub[k] = corr[k] - np.dot(gk, h)
    
    # Bound constraints: -h[i] ≤ 0
    for i in range(n):
        A_ub[K+i, i] = -1.0
        b_ub[K+i] = 0.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h

        h4, c5_4 = gradient_descent(h_interp, n_try, time_limit=tl, beta_init=40, lr_init=0.05)
        if remaining() < 60:
            break
        tl2 = min(40, remaining() - 40)
        h4b, c5_4b = slp_optimize(h4, n_try, time_limit=tl2)
        if c5_4b < best_c5:
            best_c5, best_h, best_n = c5_4b, h4b, n_try
        elif c5_4 < best_c5:
            best_c5, best_h, best_n = c5_4, h4, n_try

    # Phase 5: Mutations + quick optimize
    for _ in range(8):
        if remaining() < 80:
            break
        h_mut = mutate(best_h, best_n, rng, strength=0.06)
        tl = min(50, remaining() - 40)
        h5, c5_5 = gradient_descent(h_mut, best_n, time_limit=tl, beta_init=60, lr_init=0.04)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5
        # Also try SLP from mutated
        if remaining() > 60:
            tl2 = min(30, remaining() - 40)
            h5b, c5_5b = slp_optimize(h5, best_n, time_limit=tl2)
            if c5_5b < best_c5:
                best_c5, best_h = c5_5b, h5b

    # Phase 6: Final SLP + gradient polish
    if remaining() > 60:
        tl = min(50, remaining() - 10)
        h6, c5_6 = slp_optimize(best_h, best_n, time_limit=tl)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6
    if remaining() > 30:
        tl = min(25, remaining() - 5)
        h7, c5_7 = gradient_descent(best_h, best_n, time_limit=tl, beta_init=150, lr_init=0.01)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    best_h = np.clip(best_h, 0, 1)

    return (best_h, best_c5, best_n)
