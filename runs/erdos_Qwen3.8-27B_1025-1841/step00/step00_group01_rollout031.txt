import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def interpolate_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old, n_new, endpoint=False)
    # Periodic interpolation
    h_ext = np.concatenate([h, h, h])
    f = interp1d(np.concatenate([np.arange(-n_old, 2*n_old)]), h_ext, kind='linear')
    h_new = f(x_new)
    return np.clip(h_new, 0, 1)

def project_to_constraint(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(100, 0.5)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Define objective and constraints for SLSQP
    def make_obj(n):
        def obj(x):
            dx = 2.0 / n
            return float(np.max(np.correlate(x, 1.0 - x, mode="full")) * dx)
        return obj
    
    def make_constraint(n):
        def con(x):
            return np.sum(x) - n / 2.0
        return con
    
    # Try different resolutions
    resolutions = [len(h_init)]
    for n in [48, 50, 100, 150, 200, 300, 400, 500, 1000]:
        if n not in resolutions:
            resolutions.append(n)
    resolutions = sorted(set(resolutions))
    
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.85:
            break
        
        # Interpolate to this resolution
        h = interpolate_h(h_init, n)
        h = project_to_constraint(h, n)
        
        # SLSQP optimization
        try:
            result = minimize(
                make_obj(n),
                h,
                method='SLSQP',
                bounds=[(0, 1)] * n,
                constraints=[{'type': 'eq', 'fun': make_constraint(n)}],
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            if result.x is not None:
                h_opt = np.clip(result.x, 0, 1)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt

            pass

    if time.time() - start_time < budget_s * 0.9:
        n = len(best_h)
        # Try coordinate descent
        for iteration in range(1000):
            if time.time() - start_time > budget_s * 0.95:
                break
            improved = False
            for i in range(n):
                for delta in [0.01, -0.01, 0.005, -0.005]:
                    h_try = best_h.copy()
                    h_try[i] = np.clip(h_try[i] + delta, 0, 1)
                    # Fix constraint
                    diff = n/2.0 - np.sum(h_try)
                    if abs(diff) < 1e-10:
                        c5_try = compute_c5(h_try)
                    else:
                        # Distribute correction to other elements
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        h_try[mask] += diff / (n - 1)
                        h_try = np.clip(h_try, 0, 1)
                        c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-10:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
            if not improved:
                break
    
    return (best_h, best_c5, len(best_h))

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def interp_periodic(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old, n_new, endpoint=False)
    h_ext = np.concatenate([h, h, h])
    x_ext = np.arange(-n_old, 2 * n_old, dtype=float)
    f = interp1d(x_ext, h_ext, kind='linear', assume_sorted=True)
    return np.clip(f(x_new), 0, 1)

def project_constraint(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    h = h + (target - np.sum(h)) / n
    return np.clip(h, 0, 1)

def objective_smooth(x, n, beta):

    corr = np.correlate(x, 1.0 - x, mode="full") * dx

def constraint_fn(x, n):
    return np.sum(x) - n / 2.0
