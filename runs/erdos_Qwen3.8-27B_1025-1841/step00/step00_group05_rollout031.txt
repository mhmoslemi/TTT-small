from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    return np.max(cor) * dx

def soft_max_obj(h, alpha):
    """Smooth approximation of max_k S_k * dx"""
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    # Soft max
    max_cor = np.max(cor)
    shifted = cor - max_cor
    log_sum_exp = max_cor + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    return log_sum_exp * dx

def soft_max_grad(h, alpha):
    """Gradient of soft max objective"""
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    max_cor = np.max(cor)
    weights = np.exp(alpha * (cor - max_cor))
    weights /= np.sum(weights)
    
    # Gradient is weighted sum of individual S_k gradients
    grad = np.zeros(n)
    j = np.arange(n)
    # For each lag k, the gradient of S_k is:
    # (1-h[j+k])*I[0<=j+k<n] - h[j-k]*I[0<=j-k<n]
    # We need to compute the weighted sum efficiently
    
    # Use FFT or direct computation
    # For each weight w_m (corresponding to lag k = m - (n-1)):
    # contribution = w_m * [(1-h[j+k])*I - h[j-k]*I]
    
    # Efficient: the first term is (1-h) convolved with weights (shifted)
    # The second term is h convolved with weights (shifted)
    
    # First term: sum_m w_m * (1-h[j + k_m]) * I[0<=j+k_m<n]
    # This is like a weighted sum of (1-h) at shifted positions
    # = (1-h) correlated with w (reversed)
    
    # Actually, let me think of it as:
    # term1[j] = sum_m w_m * (1-h[j+k_m]) for valid j+k_m
    # term2[j] = sum_m w_m * h[j-k_m] for valid j-k_m
    
    # These can be computed as convolutions
    # term1[j] = sum_m w_m * (1-h[j + (m-(n-1))])
    # Let l = m-(n-1) = k, so m = l + (n-1)
    # term1[j] = sum_k w_{k+(n-1)} * (1-h[j+k]) for valid j+k
    # This is the cross-correlation of w (padded) with (1-h)
    
    # Simpler: just do it directly for the number of non-negligible weights
    # For large alpha, only the top few weights matter
    
    # Direct computation (efficient for moderate n)
    for m in range(len(cor)):
        if weights[m] < 1e-10:
            continue
        k = m - (n - 1)
        w = weights[m]
        # First term: (1-h[j+k]) for valid j+k
        idx = j + k  # j+k for j=0..n-1
        mask = (idx >= 0) & (idx < n)
        grad[mask] += w * (1.0 - h[idx[mask]])

        idx2 = j - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] -= w * h[idx2[mask2]]
