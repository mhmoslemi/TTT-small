def fw_vertex(g, target_sum):
    """Find vertex of {h: 0≤h≤1, sum(h)=target_sum} minimizing <g,h>"""
    n = len(g)
    sorted_idx = np.argsort(g)  # ascending order
    h = np.zeros(n)
    k = int(target_sum)  # number of full 1s
    frac = target_sum - k  # fractional part
    
    for i in range(k):
        h[sorted_idx[i]] = 1.0
    if frac > 1e-10:
        h[sorted_idx[k]] = frac
    return h

    dx = 2.0 / n

    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)

        return (1.0 - 2.0 * h) * dx
    elif delta > 0:
        if delta < n:
            grad[delta:] += (1.0 - h[:n - delta])
            grad[:n - delta] += -h[delta:]
    else:
        d = -delta
        if d < n:
            grad[:n - d] += (1.0 - h[d:])
            grad[d:] += -h[:n - d]
    return grad * dx

    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)

    k = int(np.floor(target_sum))
    frac = target_sum - k
    for i in range(min(k, n)):
        h[sorted_idx[i]] = 1.0
    if frac > 1e-10 and k < n:
        h[sorted_idx[k]] = frac
    return h

def optimize_pgd(h0, time_limit, n_top=10, tau=100.0):
    n = len(h0)
    target_sum = n / 2.0
    h = project_constrained(h0.copy(), target_sum)
    best_c5 = compute_C5(h)
    best_h = h.copy()
    start_time = time.time()
    step = 0.2
    patience = 500
    no_improve = 0
    while time.time() - start_time < time_limit:
        f_vals = compute_all_f(h)
        current_c5 = np.max(f_vals)
        if current_c5 < best_c5 - 1e-9:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
            step = min(step * 1.03, 0.8)
        else:
            no_improve += 1
        if no_improve > patience:
            scale = min(0.3, 0.1 + 0.2 * (no_improve - patience) / patience)
            h = project_constrained(h + scale * np.random.randn(n), target_sum)
            no_improve = 0
            step = 0.15
            continue
        grad = weighted_gradient(h, f_vals, n_top, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        alpha = step
        for _ in range(20):
            h_new = project_constrained(h - alpha * grad, target_sum)
            c5_new = compute_C5(h_new)
            if c5_new < current_c5 - 1e-8 * alpha * gnorm:
                h = h_new
                break
            alpha *= 0.5
        else:
            h = project_constrained(h - 1e-6 * grad, target_sum)
    return best_h, best_c5

def optimize_fw(h0, time_limit, n_top=10, tau=100.0):
    n = len(h0)
    target_sum = n / 2.0
    h = project_constrained(h0.copy(), target_sum)
    best_c5 = compute_C5(h)
    best_h = h.copy()
    start_time = time.time()
    for t in range(1, 500000):
        if time.time() - start_time > time_limit:
            break
        f_vals = compute_all_f(h)
        current_c5 = np.max(f_vals)
        if current_c5 < best_c5 - 1e-9:
            best_c5 = current_c5
            best_h = h.copy()
        grad = weighted_gradient(h, f_vals, n_top, tau)
        v = fw_vertex(grad, target_sum)
        gamma = min(2.0 / (t + 2.0), 0.4)
        best_g = gamma
        best_c5_try = current_c5
        for mult in [0.5, 0.7, 1.0, 1.3, 1.6, 2.0]:
            g_try = min(gamma * mult, 1.0)
            h_try = (1.0 - g_try) * h + g_try * v
            h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = compute_C5(h_try)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_g = g_try
        h = (1.0 - best_g) * h + best_g * v
        h = np.clip(h, 0.0, 1.0)
    return best_h, best_c5

def try_round(h, n):
    target_sum = n / 2.0
    h_new = (h >= 0.5).astype(float)
    s = np.sum(h_new)
    diff = target_sum - s
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_new == 0)[0]
            if len(zeros) > 0:
                idx = zeros[np.argmax(h[zeros])]
                h_new[idx] = min(diff, 1.0)
        else:
            ones = np.where(h_new == 1)[0]
            if len(ones) > 0:
                idx = ones[np.argmin(h[ones])]
                h_new[idx] = max(1.0 + diff, 0.0)
    return h_new, compute_C5(h_new)

def local_search_binary(h_bin, n, time_limit):
    target_sum = n / 2.0
    start_time = time.time()
    best_h = h_bin.copy()
    best_c5 = compute_C5(h_bin)
    is_binary = np.allclose(h_bin * (1 - h_bin), 0)
    if not is_binary:
        return best_h, best_c5
    for _ in range(200000):
        if time.time() - start_time > time_limit:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if h_bin[i] == h_bin[j]:
            continue
        h_try = best_h.copy()
        h_try[i] = 1.0 - h_try[i]
        h_try[j] = 1.0 - h_try[j]
        c5_try = compute_C5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h_init = np.full(45, 0.5)

    n_init = len(h_init)
    h_init = project_constrained(h_init, n_init / 2.0)
    best_h = h_init.copy()
    best_c5 = compute_C5(h_init)

    def elapsed():
        return time.time() - t_start

    # Phase 1: PGD at initial resolution
    remaining = budget_s - elapsed()
    if remaining > 60:
        t1 = min(150, remaining * 0.12)
        h1, c5_1 = optimize_pgd(h_init, t1, n_top=10, tau=100.0)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1
        hr, cr = try_round(h1, n_init)
        if cr < best_c5:
            best_h = hr
            best_c5 = cr

    # Phase 2: FW at initial resolution
    remaining = budget_s - elapsed()
    if remaining > 60:
        t2 = min(150, remaining * 0.12)
        h2, c5_2 = optimize_fw(best_h, t2, n_top=10, tau=100.0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
        hr, cr = try_round(h2, n_init)
        if cr < best_c5:
            best_h = hr
            best_c5 = cr

    # Phase 3: Higher resolution optimization
    for factor in [2, 3]:
        n_new = n_init * factor
        if n_new > 400:
            continue
        remaining = budget_s - elapsed()
        if remaining < 60:
            break
        x_old = np.linspace(0, 1, n_init, endpoint=False)
        x_new = np.linspace(0, 1, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_constrained(h_up, n_new / 2.0)

        t_alloc = min(remaining * 0.22, 200)
        h_opt, c5_opt = optimize_pgd(h_up, t_alloc, n_top=12, tau=120.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        hr, cr = try_round(h_opt, n_new)
        if cr < best_c5:
            best_h = hr
            best_c5 = cr

        remaining = budget_s - elapsed()
        if remaining > 60:
            t_fw = min(remaining * 0.18, 180)
            h_fw, c5_fw = optimize_fw(h_opt, t_fw, n_top=12, tau=120.0)
            if c5_fw < best_c5:
                best_c5 = c5_fw
                best_h = h_fw
            hr, cr = try_round(h_fw, n_new)
            if cr < best_c5:
                best_h = hr
                best_c5 = cr

    # Phase 4: Random restarts at various resolutions
    remaining = budget_s - elapsed()
    if remaining > 50:
        resolutions = [n_init, n_init * 2, min(n_init * 3, 200)]
        n_attempts = max(4, int(remaining / 30))
        t_per = max(5.0, (remaining - 40) / n_attempts)
        for i in range(n_attempts):
            if elapsed() > budget_s - 30:
                break
            t_alloc = min(t_per, budget_s - elapsed() - 25)
            if t_alloc < 5:
                break
            n_r = resolutions[i % len(resolutions)]
            r = np.random.random()
            if r < 0.35:
                h_rand = np.zeros(n_r)
                n_ones = n_r // 2
                ones_idx = np.random.choice(n_r, n_ones, replace=False)
                h_rand[ones_idx] = 1.0
                if n_r % 2 == 1:
                    z_idx = np.where(h_rand == 0)[0]
                    if len(z_idx) > 0:
                        h_rand[z_idx[0]] = 0.5
            elif r < 0.65:
                h_rand = np.random.beta(2.0, 2.0, n_r)
            else:
                if n_r == len(best_h):
                    h_rand = best_h + 0.15 * np.random.randn(n_r)
                else:
                    h_rand = np.interp(
                        np.linspace(0, 1, n_r, endpoint=False),
                        np.linspace(0, 1, len(best_h), endpoint=False),
                        best_h
                    )
                    h_rand += 0.1 * np.random.randn(n_r)
            h_rand = project_constrained(h_rand, n_r / 2.0)
            if i % 2 == 0:
                h_opt, c5_opt = optimize_pgd(h_rand, t_alloc, n_top=10, tau=100.0)
            else:
                h_opt, c5_opt = optimize_fw(h_rand, t_alloc, n_top=10, tau=100.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            hr, cr = try_round(h_opt, n_r)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr

    # Phase 5: Local search on best binary solution
    remaining = budget_s - elapsed()
    if remaining > 10:
        n_best = len(best_h)
        if np.allclose(best_h * (1 - best_h), 0):
            hb, cb = local_search_binary(best_h, n_best, min(remaining - 20, 60))
            if cb < best_c5:
                best_c5 = cb
                best_h = hb

    # Final validation
    n = len(best_h)
    best_h = project_constrained(best_h, n / 2.0)
    final_c5 = compute_C5(best_h)
    return best_h, final_c5, n
