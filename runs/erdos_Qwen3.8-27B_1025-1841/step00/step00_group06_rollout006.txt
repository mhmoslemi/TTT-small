    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    max_val = np.max(corr)
    active = np.where(np.isclose(corr, max_val, atol=1e-10))[0]

        weights = np.exp(log_weights)
        weights /= np.sum(weights)
        max_val = max_c + eps * (np.log(np.sum(weights)) + np.max(log_weights) - np.max(log_weights))
        # Actually: log_sum_exp = max_c + eps * log(sum(exp((corr-max_c)/eps)))
        # = max_c + eps * (max((corr-max_c)/eps) + log(sum(exp((corr-max_c)/eps - max(...)))))
        # Let me just compute it simply
        shifted = (corr - max_c) / eps
        shifted -= np.max(shifted)
        w = np.exp(shifted)
        w /= np.sum(w)
        max_val = max(corr)  # approximately, the actual smooth max is slightly larger
        # For optimization purposes, the weight is what matters
        weights = w
    
    grad = np.zeros(n)
    for m_idx in range(len(corr)):
        w = weights[m_idx]
        if w < 1e-10:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - h[:n-k])
            if k > 0:
                grad[:n-k] -= w * h[k:]
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n-kp]
    
    return max_val, grad * dx

def project_box_sum(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        c = np.clip(h - mid, 0.0, 1.0)
        if np.sum(c) < target:
            hi = mid
        else:
            lo = mid
    c = np.clip(h - (lo+hi)*0.5, 0.0, 1.0)
    diff = target - np.sum(c)
    if abs(diff) > 1e-14:
        free = (c > 1e-12) & (c < 1.0 - 1e-12)
        if free.any():
            c[free] += diff / free.sum()
            np.clip(c, 0.0, 1.0, out=c)
    return c

        step *= decay
    return x

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def c5_and_grad(h, method="smooth", eps=0.008):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx

    if method == "subgrad":
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-9)[0]
        weights = np.zeros(len(corr))
        weights[active] = 1.0 / len(active)
    else:
        max_c = np.max(corr)
        shifted = (corr - max_c) / eps
        shifted -= np.max(shifted)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        max_val = corr[np.argmax(corr)]

    grad = np.zeros(n)

        w = weights[m_idx]
