import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_obj_soft(n, T):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp((corr - m) / T))) / T
    return obj

def make_obj_hard(n):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx
    return obj

def interp_to(parent, n_new):
    n_old = len(parent)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h = np.interp(x_new, x_old, parent)
    tgt = n_new / 2.0
    h += (tgt - h.sum()) / n_new
    h = np.clip(h, 0.0, 1.0)
    h += (tgt - h.sum()) / n_new
    h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    tgt = n / 2.0
    h = h + (tgt - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    return h

def opt_slsqp(h0, n, T=100, maxiter=400, ftol=1e-10, tlimit=90):
    t0 = time.time()
    tgt = n / 2.0
    if T is not None:
        obj = make_obj_soft(n, T)
    else:
        obj = make_obj_hard(n)
    def con_eq(h):
        return h.sum() - tgt
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       maxiter=maxiter,
                       options={'ftol': ftol, 'disp': False})
        h_out = res.x if res.fun < obj(h0) + 0.5 else h0
    except Exception:
        h_out = h0
    h_out = np.clip(h_out, 0.0, 1.0)
    h_out = project_feasible(h_out, n)
    return h_out, compute_c5_exact(h_out)

    b[idx] = 1.0
    return b
