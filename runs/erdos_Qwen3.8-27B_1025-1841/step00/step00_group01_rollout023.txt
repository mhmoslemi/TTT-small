    return np.max(corr)

def smooth_max_c5(h, n, beta=80.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    weights = np.exp(beta * (corr - max_c))
    sw = np.sum(weights)
    f_val = (np.log(sw) + beta * max_c) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] / sw
        if w < 1e-10:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif s > 0:
            contrib = np.zeros(n)
            contrib[0:n-s] += (1.0 - h[s:n])
            contrib[s:n] -= h[0:n-s]
            grad += w * contrib * dx
        else:
            ss = -s
            contrib = np.zeros(n)
            contrib[ss:n] += (1.0 - h[0:n-ss])
            contrib[0:n-ss] -= h[ss:n]
            grad += w * contrib * dx
    return f_val, grad

def project_onto_set(v, S):
    n = len(v)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        if np.sum(proj) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def projected_gradient_descent(h_init, n, budget_s=120.0):
    start_t = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 1.0 / (1.5 * n)
    beta = 60.0
    stagnant = 0
    for iteration in range(20000):
        if time.time() - start_t > budget_s:
            break
        if iteration % 500 == 0 and iteration > 0:
            beta = min(beta * 1.5, 200.0)
        f_val, grad = smooth_max_c5(h, n, beta=beta)
        h_new = h - alpha * grad
        h_new = project_onto_set(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        step = 0
        while c5_new > best_c5 + 1e-13 and step < 25:
            alpha *= 0.5
            h_new = h - alpha * grad
            h_new = project_onto_set(h_new, n / 2.0)
            c5_new = compute_c5(h_new)
            step += 1
        if c5_new < best_c5 - 1e-11:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stagnant = 0
            if step == 0:
                alpha = min(alpha * 1.08, 0.8)
        else:
            stagnant += 1
            alpha *= 0.7
        if stagnant > 500:
            break
    return best_h, best_c5

def coordinate_descent(h_init, n, budget_s=60.0):
    start_t = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    for iteration in range(200):
        if time.time() - start_t > budget_s:
            break
        improved = False
        for i in range(n):
            h_t = h.copy()
            h_t[i] = 0.0
            h_t = project_onto_set(h_t, n / 2.0)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-12:
                h = h_t
                best_c5 = c5_t
                improved = True
            h_t = h.copy()
            h_t[i] = 1.0
            h_t = project_onto_set(h_t, n / 2.0)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-12:
                h = h_t
                best_c5 = c5_t
                improved = True
        if not improved:
            break
    return h, best_c5

def local_perturbation(h_init, n, rng, budget_s=30.0):
    start_t = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    improvements = 0
    for _ in range(500):
        if time.time() - start_t > budget_s:
            break
        idx = rng.randint(0, n)
        step_size = 0.1 / (1.0 + 0.01 * best_c5)
        for delta in [step_size, -step_size, 0.5 * step_size, -0.5 * step_size]:
            h_t = h.copy()
            h_t[idx] += delta
            h_t = project_onto_set(h_t, n / 2.0)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-13:
                h = h_t
                best_c5 = c5_t
                improvements += 1
                break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(47) * 0.5

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    n0 = len(best_h)

    # Phase 1: Projected gradient at initial resolution
    time_left = budget_s - (time.time() - start_time)
    if time_left > 90:
        h_new, c5_new = projected_gradient_descent(best_h, n0, budget_s=min(300, time_left * 0.25))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Coordinate descent
    time_left = budget_s - (time.time() - start_time)
    if time_left > 60:
        h_new, c5_new = coordinate_descent(best_h, n0, budget_s=min(120, time_left * 0.15))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Local perturbation
    time_left = budget_s - (time.time() - start_time)
    if time_left > 60:
        h_new, c5_new = local_perturbation(best_h, n0, rng, budget_s=min(60, time_left * 0.1))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Try higher resolutions
    for target_n in [n0 * 2, n0 * 3, 96, 128, 160]:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 200:
            break
        if target_n > 256 or target_n <= n0:
            continue
        h_interp = np.interp(
            np.linspace(0, 2, target_n, endpoint=False),
            np.linspace(0, 2, n0, endpoint=False),
            best_h
        )
        h_interp = project_onto_set(h_interp, target_n / 2.0)
        h_new, c5_new = projected_gradient_descent(h_interp, target_n, budget_s=min(300, time_left * 0.3))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        time_left = budget_s - (time.time() - start_time)
        if time_left > 60:
            h_new, c5_new = coordinate_descent(best_h, len(best_h), budget_s=min(60, time_left * 0.1))
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: Random restarts at current best resolution
    n_best = len(best_h)
    time_left = budget_s - (time.time() - start_time)
    attempt = 0
    while time_left > 150 and attempt < 15:
        attempt += 1
        noise_scale = 0.08 / (1.0 + attempt * 0.1)
        h_pert = best_h + rng.normal(0, noise_scale, n_best)
        h_pert = project_onto_set(h_pert, n_best / 2.0)
        h_new, c5_new = projected_gradient_descent(h_pert, n_best, budget_s=min(100, time_left * 0.2))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        time_left = budget_s - (time.time() - start_time)

    # Phase 6: Final refinement
    time_left = budget_s - (time.time() - start_time)
    if time_left > 60:
        h_new, c5_new = projected_gradient_descent(best_h, n_best, budget_s=min(120, time_left * 0.3))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    time_left = budget_s - (time.time() - start_time)
    if time_left > 30:
        h_new, c5_new = coordinate_descent(best_h, n_best, budget_s=min(60, time_left * 0.4))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    time_left = budget_s - (time.time() - start_time)
    if time_left > 20:
        h_new, c5_new = local_perturbation(best_h, n_best, rng, budget_s=min(30, time_left * 0.5))
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = project_onto_set(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
