            else:

                h[jdx] = np.clip(h[jdx] + max(diff, -h[jdx]), 0.0, 1.0)
        h = project_onto_simplex(h, n)
        c5_new = compute_c5(h)
        if c5_new < c5_curr - 1e-10:
            c5_curr = c5_new
            step = min(step * 1.1, 0.08)
        else:
            h[idx] = np.clip(h_old, 0.0, 1.0)
            h = project_onto_simplex(h, n)
            c5_curr = compute_c5(h)
            step = max(step * 0.8, 0.001)
    return h, c5_curr

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.95
    
    h_init = np.array(initial_h_values, dtype=float)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Strategies: (type, n, params, time_fraction)
    strategies = [
        ('parent', 180, {}, 0.20),
        ('parent', 360, {}, 0.20),
        ('sin', 180, {'freq': 1, 'phase': 0}, 0.12),
        ('sin', 180, {'freq': 2, 'phase': 0.5}, 0.12),
        ('sin', 360, {'freq': 1, 'phase': 0.3}, 0.12),
        ('rand', 180, {}, 0.10),
        ('rand', 360, {}, 0.10),
        ('polish', -1, {}, 0.04),
    ]
    
    for stype, n, params, tf in strategies:
        if time.time() > hard_deadline - 5:
            break
        alloc = min(budget_s * tf, hard_deadline - time.time() - 2)
        if alloc < 3:
            continue
        
        if stype == 'parent':
            h = upsample_linear(h_init, n)
        elif stype == 'sin':
            h = make_sinusoidal(n, freq=params['freq'], phase=params['phase'])
        elif stype == 'rand':
            h = make_random_h(n, rng)
        elif stype == 'polish':
            n = len(best_h)
            h = best_h.copy()
            h, c5_p = local_polish(h, n, rng, min(time.time() + alloc, hard_deadline))
            c5_p = compute_c5(h)
            if c5_p < best_c5:
                best_h, best_c5 = h.copy(), c5_p
            continue
        else:
            continue
        
        h_opt, c5_opt = optimize_projected_gradient(h, n, rng, min(time.time() + alloc, hard_deadline))
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
    
    # Final polish on best
    if time.time() < hard_deadline:
        n_f = len(best_h)
        best_h, best_c5 = local_polish(best_h, n_f, rng, hard_deadline)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_simplex(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
