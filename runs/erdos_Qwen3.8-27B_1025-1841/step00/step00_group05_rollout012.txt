from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # Bisection on lambda for sum(clip(h - lambda, 0, 1)) == target
    lo, hi = h.min() - 2, h.max() + 2
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

    # d/dh[j] = (1 - h[j-r+n-1]) - h[j+r-n+1] (with boundary handling)

    for j in range(n):

        idx2 = j + r - n + 1  # index in h for the (1-h) term
        g = 0.0
        if 0 <= idx1 < n:
            g += (1.0 - h[idx1])
        if 0 <= idx2 < n:
            g -= h[idx2]
        grad[j] = g
    return dx * grad

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.01
    for it in range(max_iter):
        g = grad_c5(h)
        # Project gradient onto constraint tangent space
        # The constraint is sum(h) = n/2, so tangent is {v : sum(v) = 0}
        g = g - np.mean(g)
        if np.linalg.norm(g) < tol:
            break
        g_norm = np.linalg.norm(g)
        g = g / g_norm
        # Line search
        best_alpha = step
        best_val = best_c5
        for alpha in [step, step*0.5, step*0.25, step*0.1, step*0.01]:
            h_trial = project_feasible(h - alpha * g, n)
            val = compute_c5(h_trial)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_val < best_c5:
            h = project_feasible(h - best_alpha * g, n)
            best_c5 = best_val
            best_h = h.copy()
        else:
            step *= 0.5
        if step < 1e-10:
            break
    return best_h, best_c5

    # For the second term: h shifted by (n-1-r)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    if np.allclose(h, 0) and target > 0:
        return np.full(n, target / n)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    return result

def make_feasible(n, rng):
    """Generate a random feasible vector."""
    h = rng.rand(n)
    return project_feasible(h, n)

    k = n // 2
    idx = rng.choice(n, k, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def slsqp_optimize(h0, n, maxiter=300):
    target = n / 2.0
    def obj(x):
        return compute_c5(x)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                         'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.success or res.fun < compute_c5(h0):
        return res.x, res.fun
    return h0, compute_c5(h0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def generate_start(n, rng, mode=0):
    if mode == 0:  # uniform random
        return project_feasible(rng.rand(n), n)
    elif mode == 1:  # binary
        k = n // 2
        idx = rng.permutation(n)[:k]
        h = np.zeros(n)
        h[idx] = 1.0
        return h
    elif mode == 2:  # sinusoidal
        f = rng.randint(1, 5)
        a = rng.uniform(0.1, 0.45)
        p = rng.uniform(0, 2*np.pi)
        x = np.arange(n) * 2.0 / n
        h = 0.5 + a * np.sin(np.pi * f * x + p)
        return project_feasible(h, n)
    elif mode == 3:  # block
        nblocks = rng.randint(2, 8)
        h = np.zeros(n)
        for _ in range(nblocks // 2):
            start = rng.randint(0, n)
            length = rng.randint(n // 10, n // 3)
            h[start:start+length] = 1.0
        return project_feasible(h, n)
    else:  # smooth random
        coeffs = rng.randn(10)
        x = np.arange(n) * 2.0 / n
        h = 0.5
        for i in range(1, 10):
            h += coeffs[i] / (i+1) * np.sin(np.pi * i * x + rng.uniform(0, 2*np.pi))
        return project_feasible(h, n)

def slsqp_optimize(h0, n, maxiter=300, ftol=1e-12):
    target = n / 2.0
    ones = np.ones(n)
    def obj(x):
        return compute_c5(x)
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return ones
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_out = np.asarray(res.x, dtype=float)
    h_out = project_feasible(h_out, n)
    c5_out = compute_c5(h_out)
    c5_in = compute_c5(h0)
    if c5_out <= c5_in:
        return h_out, c5_out
    return h0, c5_in

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def generate_start(n, rng, mode):
    if mode == 0:
        return project_feasible(rng.rand(n), n)
    elif mode == 1:
        k = n // 2
        idx = rng.permutation(n)[:k]
        h = np.zeros(n)
        h[idx] = 1.0
        return h
    elif mode == 2:
        f = rng.randint(1, 5)
        a = rng.uniform(0.1, 0.4)
        p = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) * 2.0 / n
        h = 0.5 + a * np.sin(np.pi * f * x + p)
        return project_feasible(h, n)
    elif mode == 3:
        h = np.zeros(n)
        nb = rng.randint(2, 7)
        for _ in range(nb):
            st = rng.randint(0, n)
            ln = rng.randint(max(1, n // 8), n // 2)
            h[st:st + ln] = 1.0
        return project_feasible(h, n)
    else:
        coeffs = rng.randn(8)
        x = np.arange(n) * 2.0 / n
        h = np.full(n, 0.5)
        for i in range(1, 8):
            h += coeffs[i] / (i + 1) * np.sin(np.pi * i * x + rng.uniform(0, 2 * np.pi))
        return project_feasible(h, n)

def slsqp_optimize(h0, n, maxiter=300):
    target = n / 2.0
    ones = np.ones(n)

    def obj(x):
        return compute_c5(x)

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return ones

    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    h_out = project_feasible(np.asarray(res.x, dtype=float), n)
    c5_out = compute_c5(h_out)
    if c5_out <= compute_c5(h0):
        return h_out, c5_out
    return h0, compute_c5(h0)

def fourier_optimize(K, n_grid, rng, maxiter=2000):
    n_params = 2 * K
    x = np.linspace(0.0, 2.0, n_grid, endpoint=False) + 1.0 / n_grid

    def h_from_params(params):
        h = np.full(n_grid, 0.5)
        for k in range(1, K + 1):
            h += params[2 * (k - 1)] * np.sin(k * np.pi * x)
            h += params[2 * (k - 1) + 1] * np.cos(k * np.pi * x)
        return h

    def obj(params):
        h = h_from_params(params)
        if h.min() < -0.15 or h.max() > 1.15:
            return 1.0 + (h.max() - 1.15) + (-0.15 - h.min())
        h = np.clip(h, 0.0, 1.0)
        return compute_c5(h)

    best_p = None
    best_f = 1e9
    for _ in range(5):
        p0 = rng.randn(n_params) * 0.15
        res = minimize(obj, p0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
        if res.fun < best_f:
            best_f = res.fun
            best_p = res.x.copy()
    if best_p is not None and best_f < 1.0:
        h = np.clip(h_from_params(best_p), 0.0, 1.0)
        h = project_feasible(h, n_grid)
        return h, compute_c5(h)
    return None, 1e9

def interpolate(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.concatenate([h, [h[0]]])
    h_new = np.interp(x_new, x_old, h_ext)
    return project_feasible(h_new, n_new)
