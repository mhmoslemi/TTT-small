```python
def compute_grad(h, n, temperature):
    """Compute gradient of soft_max(correlate(h, 1-h) * dx) w.r.t. h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp((corr - m) / temperature)
    weights = exps / exps.sum()  # softmax weights, shape (2n-1,)
    
    # grad[j] = sum_m weights[m] * d(corr[m]*dx)/dh[j]
    # = sum_m weights[m] * dx * d(corr[m])/dh[j]
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] < 1e-15:
            continue
        k = m - (n-1)
        w = weights[m] * dx
        # d(corr[m])/dh[j] for all j:
        # First term: I[max(0,k) ≤ j ≤ min(n-1, n-1+k)] * (1-h[j-k])
        j_start = max(0, k)
        j_end = min(n-1, n-1+k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end+1)
            grad[js] += w * (1 - h[js-k])
        # Second term: -I[max(0,-k) ≤ j ≤ min(n-1, n-1-k)] * h[j+k]
        j_start2 = max(0, -k)
        j_end2 = min(n-1, n-1-k)
        if j_start2 <= j_end2:
            js2 = np.arange(j_start2, j_end2+1)
            grad[js2] -= w * h[js2+k]
    
    return grad
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def fix_sum(h, n):
    """Project h onto the set {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h) - target
        if abs(s) < 1e-14:
            break
        if s > 0:
            mask = h > 1e-10
            if mask.sum() == 0:

            if mask.sum() == 0:
                break

        h = np.clip(h, 0, 1)
    return h

def upsample(h, n):
    """Upsample h from len(h) points to n points via linear interpolation"""
    if len(h) == n:
        return h.copy()
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    # Periodic extension for interpolation
    h_ext = np.concatenate([h, [h[0]]])
    h_out = np.interp(x_new, x_old, h_ext)
    return h_out
