import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def greedy_construct(n, seed=0):
    """Greedy construction of binary sequence with n/2 ones."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    available = list(range(n))
    rng.shuffle(available)  # Randomize tie-breaking order
    
    for step in range(n // 2):
        best_i = None
        best_val = float('inf')
        for i in available:
            h[i] = 1
            val = compute_c5(h)
            h[i] = 0
            if val < best_val - 1e-10:
                best_val = val
                best_i = i
            elif val == best_val and best_i is not None:
                # Keep first found (or random)
                pass
        h[best_i] = 1
        available.remove(best_i)
    
    return h, compute_c5(h)

def local_search_binary(h, dx, max_iter=100, seed=0):
    """Local search: try all single swaps, accept best improvement."""
    n = len(h)
    rng = np.random.RandomState(seed)
    current_c5 = compute_c5(h)
    
    for _ in range(max_iter):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 1e-10  # Must improve by at least this
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                h[i] = 1
                h[j] = 0
                delta = new_c5 - current_c5
                if delta < -best_delta:
                    best_delta = -delta
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            current_c5 -= best_delta
            improved = True
    
    return h, current_c5

def perturb_binary(h, k, rng):
    """Randomly swap k pairs in a binary sequence."""
    new_h = h.copy()
    ones = np.where(new_h == 1)[0]
    zeros = np.where(new_h == 0)[0]
    for _ in range(k):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        new_h[i] = 0
        new_h[j] = 1
    return new_h

def ils_binary(n, time_limit, seed=0):
    """Iterated Local Search for binary sequences."""
    rng = np.random.RandomState(seed)
    start = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Initial solution: greedy
    h, c5 = greedy_construct(n, seed)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    
    # ILS loop
    iteration = 0
    while time.time() - start < time_limit:
        # Local search
        h, c5 = local_search_binary(h, 2.0/n, max_iter=50, seed=seed + iteration)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Perturb
        k = rng.randint(2, 6)
        h = perturb_binary(h, k, rng)
        iteration += 1
        
        # Occasionally try a completely new starting point
        if iteration % 50 == 0:
            if time.time() - start < time_limit * 0.7:
                h, c5 = greedy_construct(n, seed + iteration * 100)
                h, c5 = local_search_binary(h, 2.0/n, max_iter=30, seed=seed + iteration * 100)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
    
    return best_h, best_c5

                    current_c5 = new_c5
                    if new_c5 < best_odd_c5:
