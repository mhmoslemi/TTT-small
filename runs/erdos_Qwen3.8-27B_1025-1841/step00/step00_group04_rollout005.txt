import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def obj_pnorm(h, p=40):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.sum(corr ** p) ** (1.0 / p)

def obj_pen(h, p=40, rho=5000.0):
    n = len(h)
    return obj_pnorm(h, p) + rho * (np.sum(h) - n / 2.0) ** 2

def project_onto_simplex(v, n):
    S = n / 2.0
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        if proj.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def binary_local_search(hb, max_rounds=300, swaps_per_round=800):
    n = len(hb)
    c5_cur, _ = compute_c5(hb)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        for _ in range(swaps_per_round):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            hb[i] = 0.0
            hb[j] = 1.0
            c5_new, _ = compute_c5(hb)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                improved = True
                ones = np.where(hb == 1)[0]
                zeros = np.where(hb == 0)[0]
            else:
                hb[i] = 1.0
                hb[j] = 0.0
        if not improved:
            break
    return hb, c5_cur

        h[(k * 2 + 1) % n] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5.0

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_parent = None

    if h_parent is not None and len(h_parent) > 0:
        n_par = len(h_parent)
    else:
        n_par = 100
        h_parent = np.ones(n_par) * 0.5

    best_h = h_parent.copy()
    best_c5, _ = compute_c5(best_h)
    best_n = n_par

    # Phase 1: continuous optimization at parent resolution
    if time.time() < t_end - 30:
        h_par = project_onto_simplex(h_parent, n_par)
        best_par, c5_par = opt_continuous(h_par, n_par, min(t0 + 300, t_end - 10))
        if c5_par < best_c5:
            best_c5 = c5_par
            best_h = best_par.copy()

    # Phase 2: binary local search from best continuous
    if time.time() < t_end - 20:
        hb = (best_h > 0.5).astype(float)
        n_ones = int(hb.sum())
        target_n = n_par // 2
        if n_ones > target_n:
            oi = np.where(hb == 1)[0]
            ord_i = np.argsort(best_h[oi])[::-1]
            for x in ord_i[:n_ones - target_n]:
                hb[oi[x]] = 0.0
        elif n_ones < target_n:
            zi = np.where(hb == 0)[0]
            ord_i = np.argsort(best_h[zi])[::-1]
            for x in ord_i[:target_n - n_ones]:
                hb[zi[x]] = 1.0
        hb, c5b = binary_local_search(hb, max_rounds=500, swaps_per_round=1000)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb.copy()

    # Phase 3: many random binary restarts
    n_restarts = 300
    r = 0
    while r < n_restarts and time.time() < t_end - 15:
        if time.time() < t0 + 200:
            hr = (best_h > 0.5).astype(float)
            # perturb
            nflip = max(1, n_par // 10)
            oi = np.random.choice(np.where(hr == 1)[0], size=min(nflip, int(hr.sum())), replace=False)
            zi = np.random.choice(np.where(hr == 0)[0], size=min(nflip, int((1-hr).sum())), replace=False)
            hr[oi] = 0.0
            hr[zi] = 1.0
        else:
            hr = make_random_binary(n_par)
        hr, c5r = binary_local_search(hr, max_rounds=150, swaps_per_round=500)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hr.copy()
        r += 1

    # Phase 4: try different resolution (double)
    if time.time() < t_end - 120:
        n2 = n_par * 2
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h, period=2.0)
        h2 = project_onto_simplex(h2, n2)
        h2, c5_2 = opt_continuous(h2, n2, min(time.time() + 120, t_end - 10))
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2
        # binary search at higher res
        if time.time() < t_end - 30:
            hb2 = (best_h > 0.5).astype(float)
            n_ones2 = int(hb2.sum())
            tgt2 = n2 // 2
            if n_ones2 > tgt2:
                oi2 = np.where(hb2 == 1)[0]
                ord2 = np.argsort(best_h[oi2])[::-1]
                for x in ord2[:n_ones2 - tgt2]:
                    hb2[oi2[x]] = 0.0
            elif n_ones2 < tgt2:
                zi2 = np.where(hb2 == 0)[0]
                ord2 = np.argsort(best_h[zi2])[::-1]
                for x in ord2[:tgt2 - n_ones2]:
                    hb2[zi2[x]] = 1.0
            hb2, c5_2b = binary_local_search(hb2, max_rounds=200, swaps_per_round=500)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = hb2.copy()
                best_n = n2

    final_c5, _ = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
