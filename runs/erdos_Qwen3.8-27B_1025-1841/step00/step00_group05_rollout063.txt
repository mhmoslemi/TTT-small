                e = eval_c5(h)
                h[ii], h[jj] = 1.0, 0.0
                if e < best:
                    best = e
                    best_pair = (ii, jj)
        if best_pair is None:
            break
        h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
        curr = best
    return h, curr

def sa_binary(h_init, n_iters, rng, t0=0.05, t1=0.0005):
    n = len(h_init)
    h = h_init.copy().astype(float)
    curr = eval_c5(h)
    best = curr
    best_h = h.copy()
    it = 0
    while it < n_iters:
        t = t0 * (t1 / t0) ** (it / max(n_iters - 1, 1))
        ii = rng.integers(n)
        jj = rng.integers(n)
        if h[ii] == h[jj]:
            it += 1
            continue
        h[ii], h[jj] = h[jj], h[ii]
        ne = eval_c5(h)
        d = ne - curr
        if d < 0 or rng.random() < np.exp(-d / max(t, 1e-15)):
            curr = ne
            if ne < best:
                best = ne
                best_h = h.copy()
        else:
            h[ii], h[jj] = h[jj], h[ii]
        it += 1
    return best_h, best

def make_binary(h, rng):
    n = len(h)
    hb = np.zeros(n)
    if n % 2 == 0:
        idx = np.argsort(-h)
        hb[idx[:n // 2]] = 1.0
    else:
        idx = np.argsort(-h)
        k = n // 2
        hb[idx[:k]] = 1.0
        tie = h[idx[k]]
        if tie > 0 and tie < 1:
            pass
    return hb

def interpolate_h(h, n_from, n_to):
    x_old = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    x_new = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    def remain():
        return budget_s - (time.time() - t_start)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)

    best_h = make_binary(h0, rng)
    best_c5 = eval_c5(best_h)

    h_ls, e_ls = local_search(best_h, max_passes=60)
    if e_ls < best_c5:
        best_c5 = e_ls
        best_h = h_ls

    for r in range(15):
        if remain() < 90:
            break
        if r == 0:
            h_curr = best_h.copy()
        else:
            h_curr = best_h.copy()
            npert = rng.integers(3, max(4, n // 4))
            for _ in range(npert):
                a = rng.integers(n)
                b = rng.integers(n)
                if h_curr[a] != h_curr[b]:
                    h_curr[a], h_curr[b] = h_curr[b], h_curr[a]
        n_it = max(30000, min(300000, int(remain() * 400)))
        h_sa, e_sa = sa_binary(h_curr, n_it, rng)
        h_ls2, e_ls2 = local_search(h_sa, max_passes=40)
        if e_ls2 < best_c5:
            best_c5 = e_ls2
            best_h = h_ls2
        elif e_sa < best_c5:
            best_c5 = e_sa
            best_h = h_sa

    for r in range(20):
        if remain() < 45:
            break
        h_r = np.zeros(n)
        h_r[rng.choice(n, n // 2, replace=False)] = 1.0
        h_ls, e_ls = local_search(h_r, max_passes=40)
        if remain() > 15:
            h_sa, e_sa = sa_binary(h_ls, 40000, rng, t0=0.04, t1=0.001)
            h_l2, e_l2 = local_search(h_sa, max_passes=30)
            e2 = min(e_sa, e_l2)
            h2 = h_l2 if e_l2 <= e_sa else h_sa
            if e2 < best_c5:
                best_c5 = e2
                best_h = h2
        if e_ls < best_c5:
            best_c5 = e_ls
            best_h = h_ls

    for mult in [2, 3]:
        n_hi = n * mult
        if remain() < 90:
            break
        h_hi = interpolate_h(best_h, n, n_hi)
        h_hi_bin = make_binary(h_hi, rng)
        e0 = eval_c5(h_hi_bin)
        n_it = max(20000, min(200000, int(remain() * 250)))
        h_sa, e_sa = sa_binary(h_hi_bin, n_it, rng)
        h_l2, e_l2 = local_search(h_sa, max_passes=25)
        if e_l2 < best_c5:
            best_c5 = e_l2
            best_h = h_l2
            n = n_hi
        elif e_sa < best_c5:
            best_c5 = e_sa
            best_h = h_sa
            n = n_hi

    if remain() > 10:
        h_f, e_f = local_search(best_h, max_passes=40)
        if e_f < best_c5:
            best_c5 = e_f
            best_h = h_f

    return (best_h, best_c5, len(best_h))
