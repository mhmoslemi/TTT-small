def smooth_objective_and_grad(h, n_points, beta=50):
    dx = 2.0 / n_points
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    x = dx * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    softmax = exp_x / exp_x.sum()
    obj = float(x_max + np.log(exp_x.sum()) / beta)
    
    n = n_points
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = softmax[k] * dx
        if w < 1e-15:
            continue
        # First term: (1-h[j-k+n-1]) for j in [max(0,k-n+1), min(n-1,k)]
        i_lo = max(0, k-n+1)
        i_hi = min(n-1, k)
        for j in range(i_lo, i_hi+1):
            vi = j - k + n - 1
            grad[j] += w * b[vi]
        # Second term: -h[j+k-n+1] for k >= n-1-j
        # j+k-n+1 in [0, n-1] means j in [max(0, n-1-k), min(n-1, 2n-2-k)]
        # Wait, I need: k >= n-1-j, i.e., j >= n-1-k
        # And j+k-n+1 <= n-1, i.e., j <= 2n-2-k
        # And j in [0, n-1]
        j_lo = max(0, n-1-k)
        j_hi = min(n-1, 2*n-2-k)
        for j in range(j_lo, j_hi+1):
            grad[j] -= w * h[j+k-n+1]
    
    return obj, grad

import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def project_feasible(h, n_points):
    target = n_points / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = h_p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, n_points, beta):
    dx = 2.0 / n_points
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    x = dx * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    s = exp_x.sum()
    softmax = exp_x / s
    obj = float(x_max + np.log(s) / beta)
    
    n = n_points
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = softmax[k] * dx
        if w < 1e-16:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        vi_lo = i_lo - k + n - 1
        vi_hi = i_hi - k + n - 1
        grad[i_lo:i_hi + 1] += w * b[vi_lo:vi_hi + 1]
        
        j_lo = max(0, n - 1 - k)
        j_hi = min(n - 1, 2 * n - 2 - k)
        if j_lo <= j_hi:
            hi_lo = j_lo + k - n + 1
            hi_hi = j_hi + k - n + 1
            grad[j_lo:j_hi + 1] -= w * h[hi_lo:hi_hi + 1]
    
    return obj, grad

def projected_grad_descent(h_init, n_points, budget_s, start_time):
    h = h_init.copy()
    best_h = h.copy()
    best_obj = compute_c5(h, n_points)
    
    # Anneal beta from high to low
    betas = [200, 100, 50, 25, 15, 10, 8, 6, 5, 4, 3, 2.5, 2]
    
    for beta in betas:
        if time.time() - start_time > budget_s * 0.9:
            break
        alpha = 0.01
        for it in range(500):
            if time.time() - start_time > budget_s * 0.9:
                break
            obj, grad = smooth_obj_grad(h, n_points, beta)
            # Backtracking line search
            for _ in range(30):
                h_new = project_feasible(h - alpha * grad, n_points)
                obj_new, _ = smooth_obj_grad(h_new, n_points, beta)
                if obj_new < obj:
                    h = h_new
                    obj = obj_new
                    break
                alpha *= 0.5
            # Check exact objective
            exact_obj = compute_c5(h, n_points)
            if exact_obj < best_obj:
                best_obj = exact_obj
                best_h = h.copy()
    
    return best_h, best_obj

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    n_init = 70
    if 'initial_h_values' in dir() or True:  # check if available
        try:
            h_init = np.array(initial_h_values, dtype=float)
            n_init = len(h_init)
        except:
            h_init = None
    
    if h_init is None:
        h_init = np.full(n_init, 0.5)
    
    # Project to feasible
    h_init = project_feasible(h_init, n_init)
    
    best_h = h_init.copy()
    best_obj = compute_c5(best_h, n_init)
    
    # Optimize at initial resolution
    best_h, best_obj = projected_grad_descent(h_init, n_init, budget_s, start_time)
    
    # Try different resolutions
    for n_try in [100, 140, 200, 300]:
        if time.time() - start_time > budget_s * 0.7:
            break
        # Upsample best solution
        h_new = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                          np.linspace(0, 2, n_init, endpoint=False),
                          best_h)
        h_new = project_feasible(h_new, n_try)
        
        # Also try random restart
        h_rand = project_feasible(np.full(n_try, 0.5) + 0.1 * rng.randn(n_try), n_try)
        
        for h_start in [h_new, h_rand]:
            if time.time() - start_time > budget_s * 0.9:
                break
            h_opt, obj = projected_grad_descent(h_start, n_try, budget_s, start_time)
            if obj < best_obj:
                best_obj = obj
                best_h = h_opt
                n_init = n_try
    
    # Final check
    h_final = project_feasible(best_h, len(best_h))
    c5_final = compute_c5(h_final, len(best_h))
    
    return (h_final, c5_final, len(h_final))
