import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def fix_sum(h, n):
    """Fix sum(h) to exactly n/2, keeping h in [0,1]."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase sum
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            # Need to decrease sum
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    val = m + np.log(e.sum()) / alpha
    return val * dx

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    w = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        n_lo = max(0, -d)
        n_hi = min(n - 1, n - 1 - d)
        wk = w[k]
        if wk < 1e-12:
            continue
        for nidx in range(n_lo, n_hi + 1):
            j1 = nidx + d
            j2 = nidx
            grad[j1] += wk * (1.0 - h[j2])
            grad[j2] += wk * (-h[j1])
    
    return dx * grad

def upsample(h, n_old, n_new):
    """Upsample h from n_old points to n_new points."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return h_new

    """Simple local search with random perturbations."""
    best = h.copy()
    best_c5 = c5_current
    rng = np.random.default_rng(seed_time % (2**31))
    
    for it in range(max_iters):
        if time.time() > seed_time:
            break
        h_trial = best + rng.normal(0, 0.02, size=n)
        h_trial = fix_sum(h_trial, n)
        c5_trial = compute_c5(h_trial, n)
        if c5_trial < best_c5 - 1e-10:
            best = h_trial
            best_c5 = c5_trial
    
    return best, best_c5

    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    best_n = n
    
    # Phase 1: Optimize at initial resolution with increasing alpha
    for alpha in [30, 60, 120, 250]:
        if time.time() > cutoff:

        try:

                jac=lambda x: smooth_gradient(x, n, alpha),
                bounds=[(0, 1)] * n,

                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 300, 'ftol': 1e-12}
            )
            h_new = fix_sum(res.x, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:

        except Exception:

        if c5_ls < best_c5:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    val = m + np.log(e.sum()) / alpha
    return val * dx

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    w = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        n_lo = max(0, -d)
        n_hi = min(n - 1, n - 1 - d)

            j2 = nidx

For output index m (0 to 2n-2): k = m - (n-1)
corr[m] = sum_n h[n + m - (n-1)] * (1-h[n]) for valid n

Let d = m - (n-1):
