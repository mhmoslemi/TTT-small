```python
def compute_c5_and_grad(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    
    # Compute all c_k values
    c_vals = np.zeros(2*n-1)
    for idx, k in enumerate(range(-(n-1), n)):
        if k >= 0:
            c_vals[idx] = np.sum(h[:n-k] * (1 - h[k:])) * dx
        else:
            s = -k
            c_vals[idx] = np.sum(h[s:] * (1 - h[:n-s])) * dx
    
    # Softmax weights
    max_c = np.max(c_vals)
    exp_vals = np.exp(beta * (c_vals - max_c))
    w = exp_vals / np.sum(exp_vals)
    
    # Smoothed objective
    obj = max_c + np.log(np.sum(exp_vals)) / beta
    
    # Compute gradient
    grad = np.zeros(n)
    for idx, k in enumerate(range(-(n-1), n)):
        wk = w[idx]
        if wk < 1e-12:
            continue
        if k >= 0:
            # Term A: (1-h[j+k])*dx for j in [0, n-1-k]
            grad[:n-k] += wk * (1 - h[k:]) * dx
            # Term B: -h[j-k]*dx for j in [k, n-1]
            grad[k:] -= wk * h[:n-k] * dx
        else:
            s = -k
            # Term A: (1-h[j-s])*dx for j in [s, n-1]
            grad[s:] += wk * (1 - h[:n-s]) * dx
            # Term B: -h[j+s]*dx for j in [0, n-s-1]
            grad[:n-s] -= wk * h[s:] * dx
    
    return obj, grad
```

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def compute_c5_and_grad(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    
    c_vals = np.zeros(2 * n - 1)
    for idx, k in enumerate(range(-(n-1), n)):
        if k >= 0:
            c_vals[idx] = np.dot(h[:n-k], 1.0 - h[k:]) * dx
        else:
            s = -k
            c_vals[idx] = np.dot(h[s:], 1.0 - h[:n-s]) * dx
    
    max_c = np.max(c_vals)
    exp_vals = np.exp(beta * (c_vals - max_c))
    w = exp_vals / np.sum(exp_vals)
    
    obj = max_c + np.log(np.sum(exp_vals)) / beta
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += wk * (1.0 - h[k:]) * dx
            grad[k:] -= wk * h[:n-k] * dx
        else:
            s = -k
            grad[s:] += wk * (1.0 - h[:n-s]) * dx
            grad[:n-s] -= wk * h[s:] * dx
    
    return obj, grad

def project_box_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Binary search for shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        sh = np.clip(h + mid, 0.0, 1.0)
        if sh.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def make_objective(n, beta=100.0, penalty_weight=50.0):
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        obj, grad = compute_c5_and_grad(x, beta)
        # Add penalty for sum constraint
        sum_dev = x.sum() - n / 2.0
        obj += penalty_weight * sum_dev ** 2
        grad += 2.0 * penalty_weight * sum_dev
        return obj, grad
    return obj_grad

    t0 = time.time()
    
    def elapsed():
        return time.time() - t0

                break
            obj_fn = make_objective(n_init, beta=beta)
            res = minimize(obj_fn, h_best, jac=True, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n_init,

            h_opt = project_box_sum(res.x, n_init)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < c5_best:

                h_best = h_opt.copy()

            penalty = 50.0 * (x.sum() - n / 2.0) ** 2
