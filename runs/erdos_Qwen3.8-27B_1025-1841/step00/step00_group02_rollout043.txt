def make_binary(h, n):
    """Convert h to binary with exactly n//2 ones."""
    h = np.array(h, dtype=np.float64)
    # Upscale if needed
    if len(h) != n:
        # Linear interpolation upsampling
        old_n = len(h)
        x_old = np.linspace(0, 2, old_n, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, h, period=2)
    
    # Threshold
    h_bin = (h > 0.5).astype(np.float64)
    
    # Adjust to have exactly n//2 ones
    target_ones = n // 2
    current_ones = int(np.sum(h_bin))
    
    if current_ones > target_ones:
        # Remove some ones (those with lowest original h values)
        one_positions = np.where(h_bin == 1)[0]
        # Sort by original h value, remove the smallest
        order = np.argsort(h[one_positions])

        order = np.argsort(-h[zero_positions])
        to_add = zero_positions[order[:target_ones - current_ones]]

    return h_bin

    if rng is None:
        rng = np.random.RandomState(42)

        j = zeros_idx[rng.randint(len(zeros_idx))]

def upscale_to_binary(h_src, n):
    """Upscale h_src to n points and convert to binary (n must be even)."""
    n_src = len(h_src)
    x_src = np.linspace(0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0, 2.0, n, endpoint=False)
    h_up = np.interp(x_new, x_src, h_src)
    
    # Threshold to binary
    h_bin = (h_up > 0.5).astype(np.float64)
    target = n // 2
    current = int(h_bin.sum())
    
    if current > target:
        # Remove excess ones (lowest h_up values among ones)
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_up[ones])
        h_bin[ones[order[:current - target]]] = 0
    elif current < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_up[zeros])

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def safe_interpolate(h_src, n):
    n_src = len(h_src)
    if n_src == n:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_src, h_src)

def make_binary(h_float, n):
    h_up = safe_interpolate(h_float, n) if len(h_float) != n else h_float.copy()
    h_bin = (h_up > 0.5).astype(np.float64)
    target = n // 2
    cur = int(h_bin.sum())
    if cur > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_up[ones])
        h_bin[ones[order[:cur - target]]] = 0.0
    elif cur < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_up[zeros])
        h_bin[zeros[order[:target - cur]]] = 1.0
    return h_bin

def thue_morse_binary(n, offset=0):
    seq = np.zeros(n, dtype=np.float64)
    for i in range(n):
        v = (i + offset)
        bits = 0
        while v > 0:
            bits ^= (v & 1)
            v >>= 1
        seq[i] = float(bits)
    # Ensure exactly n//2 ones
    cur = int(seq.sum())
    target = n // 2
    if cur > target:
        ones = np.where(seq == 1)[0]
        seq[ones[:cur - target]] = 0.0
    elif cur < target:
        zeros = np.where(seq == 0)[0]
        seq[zeros[:target - cur]] = 1.0
    return seq

def greedy_local_search(h, max_iters=200):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    current_max = float(np.max(corr))
    n_corr = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    
    for it in range(max_iters):
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        n1, n0 = len(ones_idx), len(zeros_idx)
        if n1 == 0 or n0 == 0:
            break
        
        p_idx = np.arange(n).reshape(-1, 1)
        lag_arr = lags.reshape(1, -1)
        pl = p_idx + lag_arr
        pm = p_idx - lag_arr
        pl_valid = (pl >= 0) & (pl < n)
        pm_valid = (pm >= 0) & (pm < n)
        pl_safe = np.clip(pl, 0, n - 1)
        pm_safe = np.clip(pm, 0, n - 1)
        h_pl = h[pl_safe]
        h_pm = h[pm_safe]
        
        total_rem = np.where(pl_valid, -(1.0 - h_pl), 0.0) + np.where(pm_valid, h_pm, 0.0)
        total_add = np.where(pl_valid, 1.0 - h_pl, 0.0) + np.where(pm_valid, -h_pm, 0.0)
        
        rem = total_rem[ones_idx]
        add = total_add[zeros_idx]
        
        best_val = current_max
        best_i, best_j = -1, -1
        max_elems = 80_000_000
        chunk = max(1, max_elems // (n0 * n_corr))
        
        for i0 in range(0, n1, chunk):
            i1 = min(i0 + chunk, n1)
            base_i = corr[None, :] + rem[i0:i1]
            diff = base_i[:, None, :] + add[None, :, :]
            new_max = np.max(diff, axis=2)
            lm = float(np.min(new_max))
            if lm < best_val - 1e-12:
                pos = np.unravel_index(np.argmin(new_max), new_max.shape)
                best_val = lm
                best_i = i0 + pos[0]
                best_j = pos[1]
        
        if best_i < 0:
            break
        h[ones_idx[best_i]] = 0.0
        h[zeros_idx[best_j]] = 1.0
        corr = np.correlate(h, 1.0 - h, mode="full")
        current_max = float(np.max(corr))
    
    return h, current_max * dx

def simulated_annealing(h, n_iters, T_start, T_end, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur = float(np.max(corr))
    best_h = h.copy()
    best = cur
    for it in range(n_iters):
        T = T_start * (T_end / T_start) ** (it / max(1, n_iters - 1))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        nc = np.correlate(h, 1.0 - h, mode="full")
        nm = float(np.max(nc))
        d = nm - cur
        if d < 0 or rng.rand() < np.exp(-d / max(T, 1e-10)):
            cur = nm
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
    return best_h, best * dx

    t0 = time.time()
