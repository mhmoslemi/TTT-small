```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(vals, t=100.0):
    m = np.max(vals)
    if t * (np.max(vals) - m) < -500:
        return m
    return m + np.log(np.sum(np.exp((vals - m) * t))) / t

def project_to_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to achieve target_sum
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum: add to elements not at 1
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            # Need to decrease sum: subtract from elements not at 0
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    global initial_h_values
    
    best_h = None
    best_c5 = float('inf')
    
    # Get initial solution
    if initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    else:
        n_init = 200
        h_init = np.zeros(n_init)
        h_init[:n_init//2] = 1.0
    
    def try_optimize(h_start, n, max_iter=300, penalty=1000.0):
        nonlocal best_h, best_c5
        target_sum = n / 2.0
        h = project_to_constraint(h_start.copy(), target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        dx = 2.0 / n
        
        def obj(x):
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1.0 - x, mode="full")
            sm = smooth_max(corr, t=100.0)
            penalty_term = penalty * (np.sum(x) - target_sum) ** 2
            return sm * dx + penalty_term
        
        # Try L-BFGS-B
        try:

                          bounds=[(0, 1)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
            h_opt = project_to_constraint(res.x, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:

                best_h = h_opt.copy()

    if best_h is not None:

        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, t=80.0):
    m = np.max(vals)
    shifted = (vals - m) * t
    shifted = np.clip(shifted, -600, 50)
    return m + np.log(np.sum(np.exp(shifted))) / t

def project_to_constraint(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, n, rng, max_rounds=5000):
    target_sum = n / 2.0
    h = project_to_constraint(h, target_sum)
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        n_pert = min(n, max(5, n // 20))
        idx = rng.choice(n, size=n_pert, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.uniform(-0.15, 0.15, size=n_pert)
        h_new = project_to_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    global initial_h_values

    best_h = None
    best_c5 = float('inf')

    if initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 200
        h_init = np.zeros(n_init)
        h_init[:n_init // 2] = 1.0

    target_init = n_init / 2.0
    h_init = project_to_constraint(h_init, target_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def update_best(h_cand):
        nonlocal best_h, best_c5
        hc = project_to_constraint(h_cand, len(h_cand) / 2.0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc.copy()

    def optimize_at_resolution(h_start, n, max_iter, t_smooth, penalty):
        target_sum = n / 2.0
        h0 = project_to_constraint(h_start.copy(), target_sum)
        dx = 2.0 / n

        def obj(x):
            xc = np.clip(x, 0.0, 1.0)
            corr = np.correlate(xc, 1.0 - xc, mode="full")
            sm = smooth_max(corr, t=t_smooth)
            pen = penalty * (np.sum(xc) - target_sum) ** 2
            return sm * dx + pen

        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11})
            update_best(res.x)
        except Exception:
            pass

    # Phase 1: optimize at initial resolution
    if n_init <= 100:
        optimize_at_resolution(h_init, n_init, 600, 80.0, 2000.0)

    # Phase 2: upscale to higher resolutions
    target_resolutions = [200, 350, 500]
    for n_res in target_resolutions:
        if time.time() - start_time > budget_s * 0.70:
            break
        if best_h is not None:
            old_n = len(best_h)
            if old_n == n_res:
                h_start = best_h.copy()
            else:
                x_old = np.linspace(0, 2, old_n, endpoint=False)
                x_new = np.linspace(0, 2, n_res, endpoint=False)
                h_start = np.interp(x_new, x_old, best_h)
        else:
            h_start = np.full(n_res, 0.5)

        optimize_at_resolution(h_start, n_res, 300, 100.0, 3000.0)

        if time.time() - start_time < budget_s * 0.78:
            h_pert = best_h.copy() if len(best_h) == n_res else h_start
            h_pert = h_pert + 0.12 * rng.randn(n_res)
            optimize_at_resolution(h_pert, n_res, 200, 60.0, 3000.0)

    # Phase 3: local search at best resolution
    n_best = len(best_h)
    if time.time() - start_time < budget_s * 0.88:
        h_ls, c5_ls = local_search(best_h, n_best, rng, max_rounds=8000)
        update_best(h_ls)

    # Phase 4: random restarts
    n_restart = len(best_h)
    max_restarts = 15
    for r in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        strategy = r % 5
        if strategy == 0:
            h_rand = np.clip(0.5 + 0.35 * rng.randn(n_restart), 0, 1)
        elif strategy == 1:
            h_rand = np.zeros(n_restart)
            pos = 0
            while pos < n_restart:
                blen = max(1, int(rng.exponential(n_restart / 8.0)))
                h_rand[pos:pos + blen] = rng.uniform(0.3, 1.0)
                pos += blen
        elif strategy == 2:
            h_rand = best_h + 0.25 * rng.randn(n_restart)
        elif strategy == 3:
            h_rand = (rng.rand(n_restart) < 0.5).astype(float)
        else:
            h_rand = np.clip(0.5 + 0.5 * rng.randn(n_restart), 0, 1)

        h_rand = project_to_constraint(h_rand, n_restart / 2.0)
        optimize_at_resolution(h_rand, n_restart, 150, 60.0, 3000.0)

    # Final local search refinement
    if time.time() - start_time < budget_s * 0.97:
        h_final, c5_final = local_search(best_h, len(best_h), rng, max_rounds=5000)
        update_best(h_final)

    best_h = project_to_constraint(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
