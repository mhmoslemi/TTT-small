def project_simplex(v, S):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = S}."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.clip(v - theta, 0, 1)

def project_to_constraint(h, target_sum):
    n = len(h)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = h.min() - 1.0
    hi = h.max() + 1.0

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:

        else:

def compute_c5(h, dx):

    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    grad = np.zeros(n)

            ks = np.arange(k2_lo, k2_hi + 1)
            idx = i - ks
            grad[i] -= np.dot(w[n-1+ks], h[idx])
    return grad
