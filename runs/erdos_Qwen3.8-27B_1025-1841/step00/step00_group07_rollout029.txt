    return float(np.max(corr) * dx)

    return evaluate_c5(h)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_obj(h):
    return evaluate_c5(h)

def _c5_obj_soft(h, alpha=40.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    vals = alpha * (corr - m)
    vals = np.minimum(vals, 0.0)
    return float(m + np.log(np.sum(np.exp(vals))) / alpha)

def _eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def _eq_jac(h):
    return np.ones(len(h))

def _project(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0.0, 1.0)
    return h

def _make_bounds(n):
    return [(0.0, 1.0)] * n

def _make_constraints(n):
    return ({'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},)

def _upscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new)

def _random_feasible(n, rng):
    h = rng.random(n)
    return _project(h, n)

def _binary_pattern(n, pattern_type, rng):
    h = np.zeros(n)
    half = n // 2
    if pattern_type == 0:
        idx = np.arange(0, n, 2)
        h[idx[:half]] = 1.0

            h[i:i+block] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return _project(h, n)

                      constraints=cons, bounds=bounds,
                      method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-10})

                      constraints=cons, bounds=bounds,
                      method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-10})
    if res.success:
        h_new = _project(res.x, n)
        return h_new
    return None
