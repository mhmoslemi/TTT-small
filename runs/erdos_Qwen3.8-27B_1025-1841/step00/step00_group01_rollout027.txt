def compute_c5(h, dx):
    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * dx

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def subgradient_step(h, alpha):
    n = len(h)
    c = np.correlate(h, 1-h, mode="full")
    k_max = np.argmax(c)
    s = k_max - (n-1)  # shift
    
    # Gradient of c[k_max] w.r.t. h
    g = np.zeros(n)
    for j in range(n):
        # h[j] appears in c[k_max] as:
        # Term 1: h[j] * (1-h[j+s]) if j+s in [0, n-1]
        # Term 2: h[j-s] * (1-h[j]) if j-s in [0, n-1], derivative w.r.t. h[j] is -h[j-s]
        val = 0.0
        if 0 <= j+s < n:
            val += (1 - h[j+s])
        if 0 <= j-s < n:
            val -= h[j-s]
        g[j] = val
    
    h_new = h - alpha * g
    h_new = project_sum(h_new, n/2.0)
    return h_new, k_max, c[k_max]

c[k] = sum_{valid i} h[i] * (1-h[i+s]) where s = k-(n-1)

∂c[k]/∂h[j] = (1-h[j+s]) * 1[j+s valid] - h[j-s] * 1[j-s valid]

def grad_c_at_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # h[j+s] valid for j = 0, ..., n-1-s
        g[:n-s] += (1 - h[s:])
        # h[j-s] valid for j = s, ..., n-1
        g[s:] -= h[:n-s]
    else:
        s_neg = -s
        # h[j+s] = h[j-s_neg] valid for j = s_neg, ..., n-1
        g[s_neg:] += (1 - h[:n-s_neg])
        # h[j-s] = h[j+s_neg] valid for j = 0, ..., n-1-s_neg
        g[:n-s_neg] -= h[s_neg:]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def grad_at_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    else:
        sp = -s
        if sp < n:
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:

        h_init = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h_init = np.ones(47) * 0.5

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, 2.0 / len(h_init))

    def slsqp_optimize(h0, max_iter=300, time_limit=60):
        nonlocal best_h, best_c5
        n = len(h0)
        target = n / 2.0
        dx = 2.0 / n

        def obj(x):
            c = np.correlate(x, 1 - x, mode="full")
            return np.max(c)

        def grad(x):
            c = np.correlate(x, 1 - x, mode="full")
            k_max = np.argmax(c)
            s = k_max - (n - 1)
            return grad_at_shift(x, s)

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                       'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n

        start = time.time()

                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-12})
            if time.time() - start > time_limit:
                return h0
            h_new = project_sum(res.x, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            return h_new
        except Exception:
            return h0

        nonlocal best_h, best_c5
        n = len(h0)
        target = n / 2.0
        dx = 2.0 / n
        h_cur = h0.copy()

        stall = 0

    if remaining > 5:
        slsqp_optimize(best_h, max_iter=200, time_limit=remaining)

    return (best_h, best_c5, len(best_h))
