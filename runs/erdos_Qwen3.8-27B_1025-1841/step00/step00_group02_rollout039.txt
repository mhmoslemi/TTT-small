import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr * dx).max()

def project_sum(h, target):
    """Project h to have sum = target, keeping 0 <= h <= 1."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(50):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
        diff = target - h.sum()
    return h

def solve_1d_minmax(A, B, lo, hi):
    """Solve min_{t in [lo,hi]} max_i (A[i] + B[i]*t)."""
    # Convex piecewise linear minimization
    # Find all kink points (intersections) in [lo, hi]
    n = len(A)
    best_val = max(np.max(A + B * lo), np.max(A + B * hi))
    best_t = lo if np.max(A + B * lo) <= np.max(A + B * hi) else hi

    # With 45 coordinates * 1000 sweeps = 45000 calls, that's 3.6 * 10^8
    # Too slow. Let me use a faster approach.

    # at least two lines are tied for the maximum.
    # We can use the fact that the function is convex and use bisection on the derivative.
    # The subgradient at t is the set of B[i] for i in argmax.
    # At the optimum, 0 is in the subgradient (if interior) or the function is decreasing/increasing at the boundary.
    
    # Use bisection: find t where the "right derivative" changes sign
    # Right derivative at t: max of B[i] for i in argmax at t (approximately)
    
    # Simpler: ternary search (works for convex functions)
    # But for piecewise linear, the minimum might be at a flat segment

    # 1. Find the argmax at lo and hi
    # 2. If same argmax, the function is linear on [lo,hi] and min is at a boundary
    # 3. If different, the minimum is at the intersection of those two lines (or nearby)
    # 4. Refine with a few more points

    idx_hi = np.argmax(A + B * hi)

But the ternary search is wasteful for a convex piecewise linear function. Let me use a better approach.

Actually, for the 1D subproblem (given all other h fixed, find optimal h[j]), the function f(t) = max_k (A_k + B_k * t) is convex piecewise linear. The minimum is at a "kink" point.

A kink occurs at the intersection of two lines that are both on the upper envelope. We can find the optimum by:
1. Finding which lines are on the upper envelope (the "active" ones)
2. Finding the intersection of the two active lines that bracket the minimum

But this is complex. A simpler approach: since f is convex, use a bisection on the subgradient.

The subgradient of f at t is: {B_k : k ∈ argmax(A_k + B_k * t)} (a set of values).
At the optimum (if interior), 0 must be in the subgradient, meaning there exist active lines with B_k ≤ 0 and B_k ≥ 0.

Bisection algorithm:
- At t: compute S_max = {B_k : k ∈ argmax} and S_min = {B_k : k ∈ argmax}
  (For a tie, the subgradient is the interval [min(B_k for k in argmax), max(B_k for k in argmax)])
- If max of subgradient < 0: f is decreasing at t, move left
- If min of subgradient > 0: f is increasing at t, move right
- If 0 is in [min_sub, max_sub]: t is optimal

But argmax might not be unique, making this tricky. Let me use a different approach.

Actually, the simplest robust approach for a convex piecewise linear function on [0,1]:
- The minimum is at one of the O(n) kink points (intersections of pairs of lines) or at the boundary
- There are O(n²) possible intersection points, but only O(n) of them are on the upper envelope
- For n=89 lines, the upper envelope has at most 89 vertices

For simplicity, let me just evaluate at O(n) strategically chosen points:
- The intersections of each line with the current "best" line

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:

        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def lse_objective(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = corr.max()
    lse = mc + np.log(np.sum(np.exp(corr - mc))) / alpha
    pen = mu * (h.sum() - n / 2.0) ** 2
    return lse + pen

def lse_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = corr.max()
    w = np.exp(alpha * (corr - mc))
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        lag = m - (n - 1)
        wm = w[m] * dx
        grad[:] += wm
        js = max(0, lag)
        je = min(n - 1, n - 1 + lag)
        if js <= je:
            idx = np.arange(js, je + 1) - lag
            grad[js:je+1] -= wm * h[idx]
        js2 = max(0, -lag)
        je2 = min(n - 1, n - 1 - lag)
        if js2 <= je2:
            idx2 = np.arange(js2, je2 + 1) + lag
            grad[js2:je2+1] -= wm * h[idx2]
