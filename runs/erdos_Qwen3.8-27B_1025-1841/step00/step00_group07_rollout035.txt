import numpy as np
from scipy.optimize import minimize
import time

_np_g = 80
_al_g = 30.0

def obj_fn(h):
    dx = 2.0 / _np_g
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    if m < 1e-15:
        return 0.0
    sh = _al_g * (corr - m)
    return m + float(np.log(np.sum(np.exp(sh)))) / _al_g

def c5_fn(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def sum_c(h):
    return float(np.sum(h)) - 0.5 * _np_g

def sum_j(h):
    return np.ones(_np_g)

def proj(h, n):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    d = 0.5 * n - float(np.sum(h))
    if d > 1e-14:
        msk = h < 1.0
        cnt = int(msk.sum())
        if cnt:
            h[msk] += d / cnt
    elif d < -1e-14:
        msk = h > 0.0
        cnt = int(msk.sum())
        if cnt:
            h[msk] += d / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample(h, ns, nd):
    if nd == ns:
        return h.copy()
    xs = np.linspace(0.0, 2.0, ns, endpoint=False)
    xd = np.linspace(0.0, 2.0, nd, endpoint=False)
    return proj(np.interp(xd, xs, h), nd)

def rand_f(n, rng):
    return proj(rng.uniform(0.0, 1.0, n), n)

def bin_f(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def do_opt(n, h0, alpha, miter):
    global _np_g, _al_g
    _np_g = n
    _al_g = alpha
    h0 = proj(h0, n)
    try:
        r = minimize(obj_fn, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[('eq', sum_c, sum_j)],
                     options={'maxiter': miter, 'ftol': 1e-9})
        ho = proj(r.x, n)
        return ho, c5_fn(ho, n)
    except Exception:
        return None, None
