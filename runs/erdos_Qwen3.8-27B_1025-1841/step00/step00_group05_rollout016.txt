        # Blend with best
        alpha = 0.3
        h_blend = alpha * h_rand + (1 - alpha) * best_h
        h_blend = project_feasible(h_blend, n_best)
        # Continuous opt
        if tleft() > 20:
            h_blend = do_optimize(h_blend, n_best, 0.002, 100)
        # Binary search
        h_b = make_binary_from_continuous(h_blend, n_best)
        h_b, c5b = binary_local_search(h_b, n_best, max_flips=15000, rstate=rng)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_b.copy()

    # Phase 5: Also try pure random binary starts
    for r in range(max(1, int(tleft() / 100))):
        if tleft() < 30:
            break
        h_rand = np.zeros(n_best)
        perm = rng.permutation(n_best)
        h_rand[perm[:n_best // 2]] = 1.0
        h_rand, c5r = binary_local_search(h_rand, n_best, max_flips=5000, rstate=rng)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_rand.copy()

    # Phase 6: Final continuous polish on best
    if tleft() > 20:
        best_h = do_optimize(best_h, n_best, 0.0005, 200)
        c5v = evaluate_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v

    # Verify and return
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n_final
