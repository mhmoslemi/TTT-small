def smooth_obj_and_grad(h, dx, alpha=100.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    # Softmax weights
    exp_corr = np.exp(alpha * (corr - m))
    w = exp_corr / np.sum(exp_corr)
    
    # Gradient: for each j, sum over lags
    # Lag index k in np.correlate corresponds to lag = k - (n-1)
    # dc_lag/ dh[j] = (1-h[j+lag])*I(0≤j+lag<n) - h[j-lag]*I(0≤j-lag<n)
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0  # sum of (1-h[j+l]) * w_l for valid l
        s2 = 0.0  # sum of h[j-l] * w_l for valid l
        for k in range(2*n - 1):
            lag = k - (n - 1)
            if 0 <= j + lag < n:
                s1 += w[k] * (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                s2 += w[k] * h[j - lag]
        grad[j] = dx * (s1 - s2)
    
    return val, grad

Second part: sum_k w_k * h[j+l_k] for valid j+l_k.
Let m = j + l_k, so l_k = m - j. k = m - j + n - 1.
Valid: 0 ≤ m < n, 0 ≤ k < 2n-1.
k = m - j + n - 1. For k ≥ 0: m ≥ j - n + 1. For k ≤ 2n-2: m ≤ j + n - 1 (always true).
So m ranges from max(0, j-n+1) to n-1.

Second part = sum_{m=max(0,j-n+1)}^{n-1} w_{m-j+n-1} * h[m]

So:
∂f/∂h[j] = dx * [sum_{m=max(0,j-n+1)}^{n-1} w_{j-m+n-1} * (1-h[m]) - sum_{m=max(0,j-n+1)}^{n-1} w_{m-j+n-1} * h[m]]

This is a convolution of w with (1-h) and w with h, evaluated at specific positions. Can be done with np.convolve or FFT.

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    val = float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)
    exp_c = np.exp(alpha * (corr - m))
    w = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            if 0 <= j - lag < n:
                s += w[k] * (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                s -= w[k] * h[j + lag]
        grad[j] = dx * s
    return val, grad
