        best_c5, best_h = c5, h

    # Phase 3: DE at current n
    if time.time() - t0 < budget_s * 0.3:
        h_de = _de_optimize(n, seed, 400, 35)
        c5_de = compute_c5(h_de)
        if c5_de < best_c5:
            best_c5, best_h = c5_de, h_de
        h = _nm_optimize(best_h, n, 5000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 4: Binary swap search at current n
    if time.time() - t0 < budget_s * 0.45:
        h_bin = _make_binary_init(n, seed + 100)
        h_bin, c5_bin = _binary_swap_search(h_bin, n, 200000, seed)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin
        # Also try from current best rounded
        h_bin2 = best_h.copy()
        h_bin2, c5_bin2 = _binary_swap_search(h_bin2, n, 100000, seed + 200)
        if c5_bin2 < best_c5:
            best_c5, best_h = c5_bin2, h_bin2

    # Phase 5: Coarse-to-fine at higher resolutions
    for n_new in [100, 140, 200, 280, 400]:
        if time.time() - t0 > budget_s * 0.88:
            break

        # Upsample best
        h_up = np.interp(
            np.linspace(0, 1, n_new),
            np.linspace(0, 1, len(best_h)),
            best_h
        )
        h_up = project_capped_simplex(h_up, n_new / 2.0)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h_up

        # NM from upsampled
        h = _nm_optimize(h_up, n_new, 3000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

        # DE at higher resolution (limited)
        if time.time() - t0 < budget_s * 0.82:
            max_it = max(50, min(200, int((budget_s * 0.82 - (time.time() - t0)) / max(0.02, 0.003 * n_new / 70))))
            h_de = _de_optimize(n_new, seed + n_new, max_it, 25)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_c5, best_h = c5_de, h_de
            h = _nm_optimize(best_h, n_new, 2000)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h

        # Golden ratio init at this resolution
        if time.time() - t0 < budget_s * 0.85:
            h_gold = _make_golden_init(n_new)
            h_gold, c5_gold = _binary_swap_search(h_gold, n_new, 50000, seed + n_new + 500)
            if c5_gold < best_c5:
                best_c5, best_h = c5_gold, h_gold
            h = _nm_optimize(h_gold, n_new, 2000)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h

    # Phase 6: Final polish
    if time.time() - t0 < budget_s * 0.95:
        h = _nm_optimize(best_h, len(best_h), 8000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    return best_h, best_c5, len(best_h)
