        if h is None or len(h) == 0:
            return
        h = make_feasible(h, len(h))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Load parent
    hp = None
    try:
        hp = np.asarray(initial_h_values, dtype=float).copy()
        hp = make_feasible(hp, len(hp))
        try_sol(hp)
    except Exception:
        pass

    # Phase 1: Optimize at parent resolution
    if hp is not None and time_left() > 60:
        n_p = len(hp)
        for trial in range(4):
            if time_left() < 40:
                break
            if trial == 0:
                h_s = hp.copy()
            elif trial == 1:
                h_s = hp + 0.1 * rng.standard_normal(n_p)
            else:
                h_s = hp + 0.2 * rng.standard_normal(n_p)
            h_opt = annealed_slsqp(make_feasible(h_s, n_p), t_start, deadline)
            try_sol(h_opt)

    # Phase 2: Higher resolutions
    for n in [90, 135, 180, 270]:
        if time_left() < 80:
            break
        starts = generate_starts(n, hp, best_h, rng)
        for h_s in starts[:6]:
            if time_left() < 40:
                break
            h_opt = annealed_slsqp(h_s, t_start, deadline)
            try_sol(h_opt)

    # Phase 3: Fine-tune best at its own resolution
    if best_h is not None and time_left() > 40:
        n_b = len(best_h)
        for trial in range(8):
            if time_left() < 20:
                break
            h_s = best_h + 0.03 * rng.standard_normal(n_b)
            h_opt = annealed_slsqp(make_feasible(h_s, n_b), t_start, deadline)
            try_sol(h_opt)

    # Phase 4: Coordinate polish on best
    if best_h is not None and time_left() > 15:
        h = best_h.copy()
        n = len(h)
        cur_c5 = compute_c5(h)
        for sweep in range(3):
            if time_left() < 5:
                break
            for i in range(n):
                if time_left() < 3:
                    break
                j = (i + 1) % n
                for d in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                    hn = h.copy()
                    hn[i] += d
                    hn[j] -= d
                    if 0.0 <= hn[i] <= 1.0 and 0.0 <= hn[j] <= 1.0:
                        c5_try = compute_c5(hn)
                        if c5_try < cur_c5 - 1e-14:
                            h = hn
                            cur_c5 = c5_try
                            break
        try_sol(h)

    # Final
    if best_h is None:
        best_h = np.full(45, 0.5)
        best_c5 = compute_c5(best_h)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
