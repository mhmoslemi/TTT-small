def compute_c5(h, n):

    return float(np.max(corr) * dx)

def smooth_max_fn(vals, T):

    m = np.max(v)
    return (m + np.log(np.sum(np.exp(v - m)))) / T

def project_sum(h, n):

    h += (n / 2.0 - np.sum(h)) / n
