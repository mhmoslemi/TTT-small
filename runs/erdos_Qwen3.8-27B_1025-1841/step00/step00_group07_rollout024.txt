import time

    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max()

def _objective_smooth(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (c - m))))

def _sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def _make_valid(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        order = idx[np.argsort(h[idx])]
        for i in order:
            adj = min(diff, 1.0 - h[i])
            h[i] += adj
            diff -= adj
            if diff <= 1e-12:
                break
    else:
        idx = np.where(h > 0.0)[0]
        order = idx[np.argsort(-h[idx])]
        for i in order:
            adj = min(-diff, h[i])
            h[i] -= adj
            diff += adj
            if diff >= -1e-12:
                break

    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)
