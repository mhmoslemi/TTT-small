def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = mask.sum()
        if c:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
        d2 = target - h.sum()
        if abs(d2) > 1e-13:
            if d2 > 0:
                j = int(np.argmax(1.0 - h))
            else:
                j = int(np.argmin(h))
            h[j] += d2
            h[j] = np.clip(h[j], 0.0, 1.0)
    return h

def do_move(h, n, rng, mode):
    """Return a perturbed copy of h."""
    h2 = h.copy()
    if mode == 0:
        # targeted: transfer at bottleneck
        corr = np.correlate(h, 1.0 - h, mode='full')
        mi = int(np.argmax(corr))
        k = mi - n + 1
        L = abs(k)
        if L == 0 or L >= n:
            i = int(rng.integers(n)); j = int(rng.integers(n))
            if i != j:
                d = (rng.random() - 0.5) * 0.1
                h2[i] += d; h2[j] -= d
        else:
            if k > 0:
                w = h[L:] * (1.0 - h[:n-L])
            else:
                w = h[:n-L] * (1.0 - h[L:])
            tw = w.sum()
            if tw > 1e-12:

            else:
                i = int(rng.integers(n - L))
            d = 0.01 + 0.09 * rng.random()
            if k > 0:
                h2[i] = min(1.0, h2[i] + d)
                h2[i+L] = max(0.0, h2[i+L] - d)
            else:
                h2[i+L] = min(1.0, h2[i+L] + d)
                h2[i] = max(0.0, h2[i] - d)
