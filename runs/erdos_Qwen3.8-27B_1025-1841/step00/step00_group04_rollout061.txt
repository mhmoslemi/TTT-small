Let me also think about a gradient-based approach. The soft-max of the correlations:
f_β(h) = (1/β) log(Σ_k exp(β * corr_k(h)))

∇f_β(h) = (1/β) * Σ_k exp(β*corr_k) * ∇corr_k / Σ_k exp(β*corr_k)

This is a weighted average of the gradients of individual correlations, weighted by temperature.

For a large β, this focuses on the worst correlation. As β → ∞, it becomes subgradient of the max.

            h[i] = max(0, h[i] - 2*eps)

For corr[j] = ∑_m h[m+p] * (1-h[m]) where p = j - (n-1):
∂corr[j]/∂h[i] = sum of:

So for a fixed i:
∂corr[j]/∂h[i] = [m = i-p is valid for this j] * (1-h[i-p]) - [m = i is valid for this j] * h[i+p]

Where "m is valid" means max(0,-p) ≤ m ≤ min(n-1, n-1-p).

This is doable in O(n) per (i, j) pair, giving O(n³) total, or can be vectorized.

def _corr_max(h):

def _project(h, target):

For the full correlation c[j] = ∑_m h[m+p_j] * (1-h[m]) where p_j = j - (n-1):

For a fixed j with p = p_j:

Where "valid" for j means: max(0, -p) ≤ m ≤ min(n-1, n-1-p)

- For contribution as h[m+p]: m = i-p, valid if max(0,-p) ≤ i-p ≤ min(n-1, n-1-p)
  i.e., max(p, 0) ≤ i ≤ min(n-1+p, n-1)
  i.e., max(p,0) ≤ i ≤ n-1 (if p ≥ 0) or 0 ≤ i ≤ n-1+p (if p < 0)
  Simplified: 0 ≤ i ≤ n-1 and 0 ≤ i-p ≤ n-1 → max(0,p) ≤ i ≤ min(n-1, n-1+p)
  
  Wait: i-p is valid (0 ≤ i-p ≤ n-1) AND m=i-p is in valid range for j: max(0,-p) ≤ i-p ≤ min(n-1, n-1-p)
  
  i-p ≥ max(0,-p) → i ≥ max(p, 0)
  i-p ≤ min(n-1, n-1-p) → i ≤ min(n-1+p, n-1)
  Also 0 ≤ i-p ≤ n-1 → p ≤ i ≤ n-1+p → max(0, p) ≤ i ≤ min(n-1, n-1+p)
  
  So valid i range: [max(0, p), min(n-1, n-1+p)]
  And the value is (1-h[i-p])

  And 0 ≤ i+p ≤ n-1 → -p ≤ i ≤ n-1-p → max(0, -p) ≤ i ≤ min(n-1, n-1-p)
  
  So valid i range: [max(0, -p), min(n-1, n-1-p)]
  And the value is -h[i+p]

So: ∂(weighted sum)/∂h[i] = Σ_j w_j * [(1-h[i-p_j]) * 1_{valid1} - h[i+p_j] * 1_{valid2}]

where valid1: max(0,p_j) ≤ i ≤ min(n-1, n-1+p_j)

def _compute_grad(h, weights):
    """weights[j] corresponds to corr[j], j=0..2n-2"""
    n = len(h)
    grad = np.zeros(n)
    for j in range(2*n - 1):
        wj = weights[j]
        if abs(wj) < 1e-15:
            continue
        p = j - (n - 1)
        # Term 1: (1-h[i-p]) for i in [max(0,p), min(n-1, n-1+p)]
        lo1 = max(0, p)
        hi1 = min(n-1, n-1+p)
        if lo1 <= hi1:
            indices = np.arange(lo1, hi1+1)
            grad[indices] += wj * (1 - h[indices - p])
        # Term 2: -h[i+p] for i in [max(0,-p), min(n-1, n-1-p)]
        lo2 = max(0, -p)
        hi2 = min(n-1, n-1-p)
        if lo2 <= hi2:
            indices = np.arange(lo2, hi2+1)
            grad[indices] -= wj * h[indices + p]
    return grad

import numpy as np
import time

def _project(h, target):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * (2.0 / n))

    n = len(h)
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if abs(wj) < 1e-16:
            continue
        p = j - (n - 1)
        lo1 = max(0, p)
        hi1 = min(n - 1, n - 1 + p)

            g[idx] += wj * (1.0 - h[idx - p])

        hi2 = min(n - 1, n - 1 - p)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            g[idx] -= wj * h[idx + p]
    return g

import numpy as np
import time

def _project(h, target):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h):
    return float(np.max(_corr(h)) * (2.0 / len(h)))

def _grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if abs(wj) < 1e-16:
            continue
        p = j - (n - 1)
        lo1, hi1 = max(0, p), min(n - 1, n - 1 + p)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += wj * (1.0 - h[lo1 - p:hi1 - p + 1])
        lo2, hi2 = max(0, -p), min(n - 1, n - 1 - p)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= wj * h[lo2 + p:hi2 + p + 1]
    return g

def _gd(h, tgt, beta, lr, n_steps, t_budget):
    n = len(h)
    h = h.copy()
    t0 = time.time()
    best = _c5(h)
    best_h = h.copy()
    for step in range(n_steps):
        if time.time() - t0 > t_budget:
            break
        c = _corr(h)
        xc = beta * (c - c.max())
        w = np.exp(xc)
        s = w.sum()
        if s < 1e-300:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= s
        g = _grad(h, w)
        h = _project(h - lr * g, tgt)
        if step % 80 == 0:
            cv = _c5(h)
            if cv < best:
                best = cv
                best_h = h.copy()
        if step % 1500 == 0 and step > 0:
            lr *= 0.8
    return best_h, best

def _local(h, tgt, t_budget, rng):
    n = len(h)
    h = h.copy()
    cur = _corr(h).max()
    dx = 2.0 / n
    t0 = time.time()
    stall = 0
    for _ in range(999999999):
        if time.time() - t0 > t_budget:
            break
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        s = h[j] - h[i]
        if abs(s) < 1e-9:
            continue
        d = s * rng.uniform(0.03, 0.95)
        hi, hj = h[i] + d, h[j] - d
        if hi < -1e-12 or hi > 1 + 1e-12 or hj < -1e-12 or hj > 1 + 1e-12:
            continue
        h[i], h[j] = hi, hj
        if _corr(h).max() < cur - 1e-15:
            cur = _corr(h).max()
            stall = 0
        else:
            h[i], h[j] = hi - d, hj + d
            stall += 1
            if stall > 5000:
                h = _project(h, tgt)
                stall = 0
    return h, cur * dx

def _local_targeted(h, tgt, t_budget, rng):
    n = len(h)
    h = h.copy()
    c = _corr(h)
    cur = c.max()
    dx = 2.0 / n
    t0 = time.time()
    for _ in range(999999999):
        if time.time() - t0 > t_budget:
            break
        worst_j = int(c.argmax())
        p = worst_j - (n - 1)
        lo = max(0, p)
        hi = min(n - 1, n - 1 - abs(p))
        lo_m = max(0, -p)
        hi_m = min(n - 1, n - 1 - p)
        if p >= 0:

        else:

        if nm < cur - 1e-15:

        else:
