```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    cc = np.correlate(h, 1 - h, mode="full")
    return float(np.max(cc) * dx)

def compute_c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    cc = np.correlate(h, 1 - h, mode="full")
    c5 = float(np.max(cc) * dx)
    k_max = int(np.argmax(cc))
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        il = j - lag
        if 0 <= il < n:
            g += (1.0 - h[il])
        ir = j + lag
        if 0 <= ir < n:
            g -= h[ir]
        grad[j] = g
    return c5, dx * grad

def upsample(h, n_out):
    n_in = len(h)
    x_in = np.arange(n_in)
    x_out = np.arange(n_out)
    h_out = np.interp(x_out, x_in, h)
```

```python
import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    cc = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cc) * dx)

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    cc = np.correlate(h, 1.0 - h, mode="full")
    c5_val = float(np.max(cc) * dx)
    k_max = int(np.argmax(cc))
    m = k_max - (n - 1)
    grad = np.zeros(n)
    idx = np.arange(n)
    jm = idx - m
    mask1 = (jm >= 0) & (jm < n)
    if np.any(mask1):
        grad[mask1] += (1.0 - h[jm[mask1]])
    jp = idx + m
    mask2 = (jp >= 0) & (jp < n)
    if np.any(mask2):
        grad[mask2] -= h[jp[mask2]]
    grad *= dx
    return c5_val, grad

def upsample(h, n_out):
    n_in = len(h)
    if n_in == n_out:
        return h.copy()
    return np.interp(np.linspace(0, n_in - 1, n_out), np.arange(n_in), h)

def enforce_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # Iterative projection
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def adam_optimize(h0, max_iter=2000, lr=0.01):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    
    for it in range(max_iter):
        t += 1
        c5, grad = compute_c5_grad(h)
        # Project gradient
        grad = grad - np.mean(grad)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def to_binary(h, n):
    h_bin = np.zeros(n)
    if n % 2 == 0:
        k = n // 2
        idx = np.argsort(-h)
        h_bin[idx[:k]] = 1.0
    else:
        k = n // 2
        idx = np.argsort(-h)
        h_bin[idx[:k]] = 1.0
        h_bin[idx[k]] = 1.0  # One extra at 0.5
        # Actually for odd n, sum = n/2 = (n//2) + 0.5
        # So k full ones and one at 0.5
        h_bin[idx[:k]] = 1.0
        h_bin[idx[k]] = 0.5
    return h_bin

def binary_anneal_search(h_bin, max_trials=30000, t_init=1.0, t_final=0.01):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for trial in range(max_trials):
        temp = t_init * (t_final / t_init) ** (trial / max_trials)
        
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / (temp * 0.01)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except:
        h_init = np.zeros(90)
        h_init[::2] = 1.0
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Strategy: multiple resolutions, multiple starts
    resolutions = [90, 180, 360, 600]
    
    for n_pts in resolutions:
        if time.time() - t_start > budget_s - 100:
            break
        
        # Multiple starting points
        starts = []
        
        # Start 1: upsampled initial
        starts.append(upsample(h_init, n_pts))
        
        # Start 2: periodic
        h_periodic = np.zeros(n_pts)
        period = max(3, n_pts // 20)
        for i in range(n_pts):
            h_periodic[i] = 1.0 if (i % period) < period // 2 else 0.0
        starts.append(h_periodic)
        
        # Start 3: golden ratio
        phi = (1 + 5**0.5) / 2
        h_gold = np.zeros(n_pts)
        for i in range(n_pts):
            h_gold[i] = 1.0 if (i * phi) % (n_pts / 2.0) < n_pts / 4.0 else 0.0
        starts.append(h_gold)
        
        # Start 4: sine wave
        h_sin = np.zeros(n_pts)
        for i in range(n_pts):
            h_sin[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i * 7 / n_pts)
        starts.append(h_sin)
        
        for h_start in starts:
            if time.time() - t_start > budget_s - 50:
                break
            
            h_start = np.clip(h_start, 0, 1)
            h_start = enforce_sum(h_start, n_pts)
            
            # Adam optimization
            h_opt, c5_opt = adam_optimize(h_start, max_iter=1000, lr=0.005)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            
            # Binary search from the optimized solution
            if time.time() - t_start > budget_s - 30:
                break
            h_bin = to_binary(h_opt, n_pts)
            h_bs, c5_bs = binary_anneal_search(h_bin, max_trials=10000)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()
        
        # Also try gradient from current best
        if time.time() - t_start > budget_s - 30:
            break
        h_from_best = upsample(best_h, n_pts)
        h_from_best = enforce_sum(h_from_best, n_pts)
        h_g, c5_g = adam_optimize(h_from_best, max_iter=500, lr=0.003)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
    
    # Final: ensure valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```
