```python
import numpy as np
from time import time

def evaluate_c5(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * (2.0/n)

def project(h, n):
    """Project h onto {h: sum h = n/2, 0 <= h <= 1}"""
    h = h.copy()
    h = np.clip(h, 0, 1)
    # Adjust sum
    diff = np.sum(h) - n/2
    if diff != 0:
        h -= diff / n
        h = np.clip(h, 0, 1)
        # Iterative water-filling
        for _ in range(100):
            diff = np.sum(h) - n/2
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                # Need to decrease; can only decrease where h > 0
                mask = h > 0
                if not np.any(mask):
                    break
                adj = diff / np.sum(mask)
                h[mask] = np.maximum(h[mask] - adj, 0)
            else:
                # Need to increase; can only increase where h < 1
                mask = h < 1
                if not np.any(mask):
                    break
                adj = -diff / np.sum(mask)
                h[mask] = np.minimum(h[mask] + adj, 1)
    return h
```

```python
    return np.correlate(h, 1-h, mode='full')

def compute_grad_Bk(h, k):
    """Compute gradient of B(k) with respect to h, for lag k in [-(n-1), n-1]"""
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1 - 2*h
    elif k > 0:
        # valid n: 0 to n-1-k
        # h[n+k]*(1-h[n])
        # ∂/∂h[j]: (1-h[j-k]) for j in [k, n-1] minus h[j+k] for j in [0, n-1-k]
        grad[k:] += (1 - h[:n-k])  # h[j-k] for j in [k, n-1]
        grad[:n-k] -= h[k:]        # h[j+k] for j in [0, n-1-k]
    else:
        m = -k
        # B(-m) = sum_{n=m}^{n-1} h[n-m]*(1-h[n])
        # ∂/∂h[j]: (1-h[j+m]) for j in [0, n-1-m] minus h[j-m] for j in [m, n-1]
        grad[:n-m] += (1 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def project_sum(h, target_sum):
    """Project h onto {h: sum h = target_sum, 0 <= h <= 1} using water-filling"""
    h = np.clip(h, 0, 1)
    for _ in range(1000):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        else:
            mask = h < 1
            if not np.any(mask):
                break
            h[mask] -= diff / np.sum(mask)  # diff is negative, so this increases
            h = np.clip(h, 0, 1)
    return h

def grad_descent(h, n, steps, lr, start_time, budget):
    """Projected gradient descent to minimize max B(k)"""
    best_h = h.copy()
    best_val = np.max(compute_B(h))
    for step in range(steps):
        if time() - start_time > budget * 0.85:
            break
        B = compute_B(h)
        max_idx = np.argmax(B)
        k = max_idx - (n-1)  # convert to lag
        grad = compute_grad_Bk(h, k)
        h_new = h - lr * grad
        h_new = project_sum(h_new, n/2)
        new_val = np.max(compute_B(h_new))
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        h = h_new
        # Adaptive learning rate
        if new_val < best_val * 0.99:
            lr = min(lr * 1.1, 0.5)
        else:
            lr = max(lr * 0.9, 0.001)
    return best_h, best_val

def upsample(h, new_n):
    """Upsample h to new_n points using linear interpolation"""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 2/(2*old_n)  # cell centers
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 2/(2*new_n)
    h_new = np.interp(new_x, old_x, h, left=h[0], right=h[-1])
    return project_sum(h_new, new_n/2)

def swap_local_search(h, n, time_budget, start_time, budget_frac=0.85):
    """Local search with swap moves"""
    best_h = h.copy()
    best_val = np.max(compute_B(h))
    n_swap = n // 2
    for it in range(100000):
        if time() - start_time > budget_frac * time_budget + (time() - start_time - (time() - start_time)):
            break
        if time() - start_time > budget_frac * time_budget:
            break
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # Swap h[i] and h[j]
        old_i, old_j = h[i], h[j]
        h[i], h[j] = old_j, old_i
        val = np.max(compute_B(h))
        if val < best_val:
            best_val = val
            best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
    return best_h, best_val
```
