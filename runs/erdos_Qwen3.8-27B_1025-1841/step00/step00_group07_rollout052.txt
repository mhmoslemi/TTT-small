```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) * alpha
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sw = np.sum(exp_vals)
    p = exp_vals / sw  # softmax weights
    f = m + np.log(sw) / alpha
    
    # Gradient: df/dh[j] = dx * sum_k p[k] * ((1-h[j+s_k]) - h[j-s_k])
    # where s_k = k - (n-1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            s = k - (n - 1)
            # (1-h[j+s]) if 0 <= j+s < n
            term1 = 0.0
            if 0 <= j + s < n:
                term1 = 1.0 - h[j + s]
            # -h[j-s] if 0 <= j-s < n
            term2 = 0.0
            if 0 <= j - s < n:
                term2 = -h[j - s]
            g += p[k] * (term1 + term2)
        grad[j] = dx * g
    return f, grad

def make_sum_constraint(n):
    def fun(h):
        return np.sum(h) - n / 2.0
    def jac(h):
        return np.ones(n)
    return fun, jac

def project_to_sum(h, n):
    """Adjust h to satisfy sum(h) = n/2 while keeping in [0,1]"""
    h = h.copy()
    diff = n / 2.0 - np.sum(h)
    if diff > 0:
        # Need to increase; add to elements < 1
        space = 1.0 - h
        total_space = np.sum(space)
        if total_space > 0:
            h += space * (diff / total_space)
    else:
        # Need to decrease; subtract from elements > 0
        total_avail = np.sum(h)
        if total_avail > 0:
            h += h * (diff / total_avail)
    return np.clip(h, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) * alpha
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    sw = np.sum(exp_vals)
    p = exp_vals / sw
    f = m + np.log(sw) / alpha
    grad = np.zeros(n)
    s_offsets = np.arange(2 * n - 1) - (n - 1)
    for j in range(n):
        idx1 = j + s_offsets
        idx2 = j - s_offsets
        term1 = np.where((idx1 >= 0) & (idx1 < n), 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
        term2 = np.where((idx2 >= 0) & (idx2 < n), -h[np.clip(idx2, 0, n - 1)], 0.0)
        grad[j] = dx * np.dot(p, term1 + term2)
    return f, grad

def sum_fun(h, n):
    return np.sum(h) - n / 2.0

def sum_jac(h, n):
    return np.ones(n)

def fix_sum(h, n, iters=20):
    for _ in range(iters):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    free = (h > 1e-6) & (h < 1.0 - 1e-6)
    if np.any(free) and abs(diff) > 1e-12:
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_up = np.interp(x_dst, x_src, h_src)
    return fix_sum(h_up, n_dst)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.asarray(initial_h_values, dtype=float)
        n_parent = len(parent)
        c5_parent = eval_c5(parent)
    else:
        parent = None
        n_parent = 0
        c5_parent = 0.5

    best_h = parent.copy() if parent is not None else np.full(80, 0.5)
    best_c5 = c5_parent if parent is not None else eval_c5(best_h)
    best_n = len(best_h)

    alphas = [30.0, 80.0, 200.0]
    resolutions = [80, 160, 320]

    for n_pts in resolutions:
        time_left = budget_s - (time.time() - t0)
        if time_left < 40:
            break

        starts = []
        if parent is not None:
            starts.append(fix_sum(upsample(parent, n_parent, n_pts), n_pts))
            starts.append(fix_sum(upsample(best_h, best_n, n_pts), n_pts))

        starts.append(fix_sum(np.random.uniform(0.2, 0.8, n_pts), n_pts))
        starts.append(fix_sum(np.random.uniform(0.0, 1.0, n_pts), n_pts))

        if n_pts <= 160:
            starts.append(fix_sum(np.random.uniform(0.0, 1.0, n_pts), n_pts))
            pattern = np.zeros(n_pts)
            period = max(2, n_pts // 8)
            for i in range(n_pts):
                pattern[i] = 1.0 if ((i * period * 7) % n_pts) < n_pts // 2 * 0.7 else 0.0
            starts.append(fix_sum(pattern + np.random.randn(n_pts) * 0.05, n_pts))

        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n_pts,)}]

        for alpha in alphas:
            time_left = budget_s - (time.time() - t0)
            if time_left < 30:
                break

            max_iter = 150 if n_pts <= 80 else 80 if n_pts <= 160 else 50

            for si in range(len(starts)):
                time_left = budget_s - (time.time() - t0)
                if time_left < 25:
                    break
                h0 = starts[si].copy()
                try:
                    res = minimize(
                        lambda x: smooth_max_and_grad(x, n_pts, alpha)[0],
                        h0,
                        jac=lambda x: smooth_max_and_grad(x, n_pts, alpha)[1],
                        method='SLSQP',
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': max_iter, 'ftol': 1e-8}
                    )
                    h_sol = res.x
                    h_sol = fix_sum(h_sol, n_pts)
                    c5_val = eval_c5(h_sol)
                    if c5_val < best_c5 - 1e-9:
                        best_c5 = c5_val
                        best_h = h_sol.copy()
                        best_n = n_pts
                        starts = [fix_sum(upsample(best_h, best_n, n_pts), n_pts)] + starts
                        starts = starts[:5]
                except Exception:
                    pass

        # Polish with large alpha
        time_left = budget_s - (time.time() - t0)
        if time_left > 20:
            try:
                res = minimize(
                    lambda x: smooth_max_and_grad(x, best_n, 500.0)[0],
                    best_h,
                    jac=lambda x: smooth_max_and_grad(x, best_n, 500.0)[1],
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * best_n,
                    constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (best_n,)}],
                    options={'maxiter': 100, 'ftol': 1e-10}
                )
                h_sol = fix_sum(res.x, best_n)
                c5_val = eval_c5(h_sol)
                if c5_val < best_c5 - 1e-9:
                    best_c5 = c5_val
                    best_h = h_sol.copy()
            except Exception:
                pass

    best_h = fix_sum(best_h, len(best_h))
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
```
