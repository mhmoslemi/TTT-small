```python
def project_feasible(h, n):
    target = n / 2.0
    # Bisection on tau: sum(clip(h - tau, 0, 1)) = target
    lo, hi = -2.0, 2.0  # Wide enough range
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0, 1)
```

import numpy as np
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def f_smooth(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    f_arr = corr * dx
    m = np.max(f_arr)
    return m + np.log(np.sum(np.exp(K * (f_arr - m)))) / K

def grad_f_smooth(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    f_arr = corr * dx
    m = np.max(f_arr)
    exps = np.exp(K * (f_arr - m))
    p = exps / np.sum(exps)
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    for idx in range(2*n-1):
        pk = p[idx]
        if pk < 1e-12:
            continue
        s = shifts[idx]
        j_min = max(0, -s)
        j_max = min(n, n - s)
        js = np.arange(j_min, j_max)
        grad[js + s] += pk * (1.0 - h[js])
        grad[js] += -pk * h[js + s]
    return grad * dx

def project_feasible(h, n):

    lo, hi = -5.0, 5.0

        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)
