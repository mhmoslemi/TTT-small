    h = h.copy()
    h = np.clip(h, 0, 1)
    # Adjust sum
    target = n / 2.0

            if total_room > 0:
                h[indices_down] += diff * room / total_room
    h = np.clip(h, 0, 1)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def subgradient_step(h, n, alpha, rng):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Find active shifts (those close to max)
    max_corr = np.max(corr)
    active = corr >= max_corr * 0.95
    
    if not np.any(active):
        return h
    
    # Compute subgradient for active shifts
    # corr[k] = sum_j h[j] * (1-h[j-m]) where m = n-1-k
    # d/dh[t] corr[k] = (1-h[t-m]) - h[t+m] (with bounds)
    
    g = np.zeros(n)
    for k in np.where(active)[0]:
        m = n - 1 - k  # shift
        for t in range(n):
            gm = 0.0
            if 0 <= t - m < n:
                gm += (1.0 - h[t - m])
            if 0 <= t + m < n:
                gm -= h[t + m]
            g[t] += gm * corr[k]  # weighted by correlation value
    
    # Normalize
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-12:
        return h
    
    # Project to zero-sum (preserve sum constraint)
    g -= np.mean(g)
    
    # Update
    h_new = h - alpha * g / g_norm
    # Clip and re-project
    h_new = np.clip(h_new, 0, 1)
    # Fix sum
    target = n / 2.0
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-10:

            if np.any(mask):

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_for_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    else:
        sp = -s
        if n - sp > 0:
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g

def project_sum(h, n):
    """Adjust h to have sum = n/2 while keeping in [0,1]."""
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 1.0
        if np.sum(mask) == 0:
            return np.clip(h, 0, 1)
        h[mask] += diff / np.sum(mask)
    else:
        mask = h > 0.0
        if np.sum(mask) == 0:
            return np.clip(h, 0, 1)
        h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def subgradient_descent(h, n, alpha_init, max_iter, rng, time_limit):
    """Subgradient descent with restarts on active set."""
    start = time.time()
    h = h.copy()
    h = project_sum(h, n)
    alpha = alpha_init
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    n_active = 5  # use top 5 correlations
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        current_c5 = max_corr * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find active shifts (top n_active by correlation value)
        top_idx = np.argsort(corr)[-n_active:]
        
        # Compute weighted gradient
        g = np.zeros(n)
        for k in top_idx:
            s = n - 1 - k
            w = corr[k]  # weight by correlation value
            g += w * grad_for_shift(h, n, s)
        
        # Project to zero-sum subspace
        g -= np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            alpha *= 0.9
            continue
        
        # Step
        h_new = h - alpha * g / g_norm
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum(h_new, n)
        
        new_c5 = compute_c5(h_new, n)
        if new_c5 < current_c5:
            h = h_new
            alpha = min(alpha * 1.01, 0.5)
        else:
            alpha *= 0.95
            # Small random perturbation to escape
            if rng.random() < 0.3:
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    delta = 0.01 * rng.randn()
                    h[i] += delta
                    h[j] -= delta
                    h = np.clip(h, 0, 1)
                    h = project_sum(h, n)
        
        # Decay alpha
        if it % 1000 == 999:
            alpha *= 0.8
    
    return best_h, best_c5

    corr = np.correlate(h, 1.0 - h, mode="full")

    if s >= 0:

            m = n - s
            g[:m] += (1.0 - h[s:])
            g[s:] -= h[:m]
    else:

            g[sp:] += (1.0 - h[:m])
            g[:m] -= h[sp:]

        hb = np.zeros(n0)
        idx = rng.choice(n0, n0 // 2, replace=False)
        hb[idx] = 1.0
        hs, cs = optimize_subgrad(hb, n0, 500000, max(t_each, 5), rng, 0.08, 5)
        if cs < best_c5:
            best_h, best_c5, best_n = hs, cs, n0

    # Phase 3: Coordinate descent polish (25%)
    t_left = budget_s - (time.time() - t0)
    t_alloc = min(t_left * 0.35, 300)
    h3, c5_3 = coord_descent(best_h, best_n, t_alloc, rng)
    if c5_3 < best_c5:
        best_h, best_c5, best_n = h3, c5_3, best_n

    # Phase 4: Try doubled resolution (remaining)
    t_left = budget_s - (time.time() - t0)
    if t_left > 120:
        n2 = 2 * best_n
        h2 = np.repeat(best_h, 2)
        # sum(h2) = 2*(n/2) = n = n2/2, so constraint is already satisfied
        h2 = project_box_sum(h2, n2)
        t_alloc = t_left * 0.7
        h4, c5_4 = optimize_subgrad(h2, n2, 500000, t_alloc, rng, 0.03, 5)
        if c5_4 < best_c5:
            best_h, best_c5, best_n = h4, c5_4, n2

    # Final subgradient refinement
    t_left = budget_s - (time.time() - t0)
    if t_left > 30:
        h5, c5_5 = optimize_subgrad(best_h, best_n, 200000, t_left * 0.8, rng, 0.03, 8)
        if c5_5 < best_c5:
            best_h, best_c5, best_n = h5, c5_5, best_n

    best_h = project_box_sum(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
