import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def project_feasible(x, target):
    """Project x to {h : 0<=h_i<=1, sum(h)=target} via water-filling"""
    if len(x) == 0:
        return x
    # Want sum(clamp(x - lam, 0, 1)) = target
    # sum(clamp(x - lam, 0, 1)) is decreasing in lam
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def optimize_local(h, target, n_iter=500, alpha=20.0):
    """Projected gradient descent on soft-max of correlations"""
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_val = compute_c5(h)
    
    lr = 1.0

        corr = compute_correlations(h)
        # Soft max weights
        c_max = np.max(corr)
        w = np.exp(alpha * (corr - c_max))
        w /= np.sum(w)
        
        # Gradient: weighted sum of individual correlation gradients
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-8:
                continue
            lag = k - (n - 1)
            j_lo = max(0, -lag)
            j_hi = min(n - 1, n - 1 - lag)
            for j in range(j_lo, j_hi + 1):
                if lag != 0:
                    grad[j + lag] += w[k] * (1.0 - h[j])
                    grad[j] -= w[k] * h[j + lag]
                else:
                    grad[j] += w[k] * (1.0 - 2.0 * h[j])
        
        # Step
        h_new = h - lr * grad * dx / n
        h_new = project_feasible(h_new, target)
        
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 5.0)
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_h, best_val

import numpy as np
import time

def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_feasible(x, target):
    n = len(x)
    if n == 0:
        return x.astype(float)
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def grad_corr_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    lag = k - (n - 1)
    L = abs(lag)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # c = sum_{j=0}^{n-1-L} h[j+L]*(1-h[j])
        js = np.arange(n - L)
        grad[js + L] += (1.0 - h[js])
        grad[js] -= h[js + L]
    else:
        # c = sum_{j=0}^{n-1-L} h[j]*(1-h[j+L])
        js = np.arange(n - L)
        grad[js] += (1.0 - h[js + L])
        grad[js + L] -= h[js]
    return grad

def optimize_local(h, target, max_iter=3000, alpha=25.0):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_val = compute_c5_and_corr(h)[0]

    lr = 0.3
    no_improve = 0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        w = np.exp(alpha * (corr - c_max))
        ws = np.sum(w)
        if ws < 1e-300:
            break
        w /= ws

        grad = np.zeros(n)
        active = w > 1e-8
        for k in np.where(active)[0]:
            grad += w[k] * grad_corr_k(h, k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        h_try = project_feasible(h - lr * grad * dx / n * n, target)
        val = compute_c5_and_corr(h_try)[0]

        if val < best_val - 1e-14:
            best_val = val
            best_h = h_try.copy()
            h = h_try
            no_improve = 0
            lr = min(lr * 1.02, 3.0)
        else:
            no_improve += 1
            lr *= 0.93
            if no_improve > 200 or lr < 1e-12:
                break

    return best_h, best_val

def binarize_step(h, target):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    # Adjust to meet sum constraint
    diff = target - h_bin.sum()
    if diff > 0:
        # Need to change some 0s to 1s
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) < int(diff) + 1:
            return project_feasible(h, target)
        idx = np.argsort(h[zeros])[::-1][:int(diff)]
        h_bin[zeros[idx]] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        if len(ones) < -int(diff) + 1:
            return project_feasible(h, target)
        idx = np.argsort(h[ones])[::-1][:-int(-diff)] if -int(diff) > 0 else []
        if len(idx) > 0:
            h_bin[ones[idx]] = 0.0
        else:
            h_bin[ones[:0]] = 0.0
    return project_feasible(h_bin, target)

def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h
