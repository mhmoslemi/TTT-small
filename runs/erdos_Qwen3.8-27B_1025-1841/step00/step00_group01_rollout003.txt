import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def smooth_max_value(corr, alpha):
    m = corr.max()
    scaled = alpha * (corr - m)
    # clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    return (m + np.log(np.sum(np.exp(scaled)))) / alpha

def gradient_smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    m = corr.max()
    scaled = alpha * (corr - m)
    scaled = np.clip(scaled, -50, 50)
    exps = np.exp(scaled)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if wk < 1e-10:
            continue
        # First part: (1-h[j-d]) for valid j
        j_lo = max(0, d)
        j_hi = min(n - 1, n - 1 + d)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += wk * (1.0 - h[j_lo - d:j_hi - d + 1])
        # Second part: -h[j+d] for valid j
        j_lo2 = max(0, -d)
        j_hi2 = min(n - 1, n - 1 - d)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= wk * h[j_lo2 + d:j_hi2 + d + 1]
    
    return grad * dx

def project_feasible(h, target_sum):
    # Project onto {x: 0<=x<=1, sum(x)=target_sum}
    n = len(h)
    h = np.array(h, dtype=float)
    # Bisection on water level
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix small error
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-10) & (proj < 1 - 1e-10)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0, 1)
    return proj

def binarize(h, target_sum):
    # Round h to binary, maintaining sum
    n = len(h)
    order = np.argsort(-h)  # descending
    nb = int(round(target_sum))
    hb = np.zeros(n)
    if nb >= 1:
        hb[order[:nb]] = 1.0
    # Adjust if needed (for non-integer target)
    frac = target_sum - nb
    if frac > 1e-10:
        # Set the (nb+1)-th element to frac
        if nb < n:
            hb[order[nb]] = frac
    return hb

def projected_gradient_step(h, alpha, lr, n_steps):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    
    for step in range(n_steps):
        grad = gradient_smooth_obj(h, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_f

def interpolate_to_resolution(h_parent, n_new):
    n_old = len(h_parent)
    dx_old = 2.0 / n_old
    x_old = np.arange(n_old) * dx_old
    x_new = np.linspace(0, 2 - 2.0/n_new, n_new)
    # Actually for the discrete problem, we want samples at the same points
    # The i-th sample represents the function on [i*dx, (i+1)*dx)
    x_old = (np.arange(n_old) + 0.5) * dx_old  # midpoints
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d_corrected(x_old, h_parent, x_new)
    # Enforce sum constraint
    target = n_new / 2.0
    f = project_feasible(f, target)
    return f

def interp1d_corrected(x_old, y_old, x_new):
    from scipy.interpolate import interp1d
    f = interp1d(x_old, y_old, kind='linear', fill_value=(y_old[0], y_old[-1]), bounds_error=False)
    return np.clip(f(x_new), 0, 1)

import numpy as np
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def gradient_smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    scaled = alpha * (corr - m)
    scaled = np.clip(scaled, -40, 40)
    exps = np.exp(scaled)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if wk < 1e-12:
            continue
        j_lo = max(0, d)
        j_hi = min(n - 1, n - 1 + d)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += wk * (1.0 - h[j_lo - d:j_hi - d + 1])
        j_lo2 = max(0, -d)
        j_hi2 = min(n - 1, n - 1 - d)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= wk * h[j_lo2 + d:j_hi2 + d + 1]
    return grad * dx

def project_feasible(h, target_sum):
    n = len(h)
    h = np.array(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def binarize(h, target_sum):
    n = len(h)
    order = np.argsort(-h)
    nb = int(round(target_sum))
    hb = np.zeros(n)
    if nb >= 1:
        hb[order[:nb]] = 1.0
    frac = target_sum - nb
    if frac > 1e-10 and nb < n:
        hb[order[nb]] = frac
    return hb

def interpolate_to_resolution(h_parent, n_new):
    n_old = len(h_parent)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h_parent, kind='cubic', fill_value=(h_parent[0], h_parent[-1]), bounds_error=False)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    target = n_new / 2.0
    h_new = project_feasible(h_new, target)
    return h_new

def pgd_optimize(h, target_sum, alpha_schedule, lr_schedule, time_limit, parent_start):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    
    for i, (alpha, lr) in enumerate(zip(alpha_schedule, lr_schedule)):
        step_deadline = min(parent_start + time_limit, 2.0 / (n * 0.02))
        
        while time.time() < step_deadline:
            grad = gradient_smooth_obj(h, alpha)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            f_new = compute_c5(h_new)
            if f_new < best_f:
                best_f = f_new
                best_h = h_new.copy()
            h = h_new
            if lr < 0.5e-5:
                break
            if f_new >= best_f:
                lr *= 0.7
    return best_h, best_f
