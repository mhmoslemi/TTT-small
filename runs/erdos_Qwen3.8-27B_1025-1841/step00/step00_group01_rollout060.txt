```python
def projected_grad_opt(h0, n, lr=0.01, T=50, maxiter=10000):
    h = h0.copy()
    tgt = n / 2.0
    for it in range(maxiter):
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Softmax weights
        max_c = np.max(corr)
        exp_c = np.exp(T * (corr - max_c))
        w = exp_c / np.sum(exp_c)
        
        # Compute gradient
        grad = np.zeros(n)
        for m in range(len(corr)):
            if w[m] < 1e-10:
                continue
            k = m - (n - 1)
            # First term: for j in S(k), add w[m]*(1-h[j-k])
            j_start = max(0, k)
            j_end = min(n-1, n-1+k)
            for j in range(j_start, j_end+1):
                grad[j] += w[m] * (1 - h[j-k])
            # Second term: for i in S(k), j = i-k, add w[m]*(-h[i])
            for i in range(j_start, j_end+1):
                j2 = i - k
                if 0 <= j2 < n:
                    grad[j2] += w[m] * (-h[i])
        
        # Gradient step (minimizing, so go opposite to gradient)
        h_new = h - lr * grad
        # Project
        h_new = np.clip(h_new, 0, 1)
        diff = np.sum(h_new) - tgt
        h_new -= diff / n
        h_new = np.clip(h_new, 0, 1)
        
        # Check if sum constraint is met (after projection)
        if not (0 <= np.all(h_new) and np.all(h_new <= 1)):
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    return h
```

```python
            # Second term
            idx2 = j_range  # i in S(k), j = i - k
            j2_range = idx2 - k
            grad[j2_range] += wm * (-h[idx2])
        
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h -= (np.sum(h) - tgt) / n
        h = np.clip(h, 0, 1)
    
    return h
```
