One important consideration: the interpolation. When we go from n to 2n samples, we need to think about what the step function looks like. If h has n samples over [0,2], each sample represents an interval of width 2/n. When we go to 2n samples, each new sample represents an interval of width 2/(2n) = 1/n.

For interpolation, the most natural approach is:
- The old sample h[i] represents the value on interval [i*2/n, (i+1)*2/n]
- The new samples at positions [j*2/(2n), (j+1)*2/(2n)] for j=0,...,2n-1
- For new sample j: if j is even, it's in the first half of old interval j/2, so value ≈ h[j/2]
- If j is odd, it's in the second half of old interval (j-1)/2, so value ≈ h[(j-1)/2]

So the simplest interpolation is just: new_h[2*i] = h[i], new_h[2*i+1] = h[i].

This preserves the integral: sum(new_h) * (2/(2n)) = sum(2*h) * (1/n) = 2*sum(h)/n = 2*(n/2)/n = 1. ✓

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    z = corr / temp
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    sum_exp = np.sum(exp_z)
    val = z_max + np.log(sum_exp)
    weights = exp_z / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k] / temp
        j1 = max(0, s)
        j2 = min(n, s + n)
        if j2 > j1:
            grad[j1:j2] += w * (1.0 - h[j1 - s:j2 - s])
        j3 = max(0, -s)
        j4 = min(n, n - s)
        if j4 > j3:
            grad[j3:j4] -= w * h[j3 + s:j4 + s]
    return val, grad

        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr) * dx)

    def constraint_eq(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_eq}]
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 200, 'ftol': 1e-8,

        c5_interp = compute_c5(h_interp)

        t_opt = min(remaining * 0.6, 500)
        if t_opt < 15:
            break

        h_opt, c5_opt = gradient_descent_optimize(h_interp, time_limit=t_opt, seed=seed + target_n)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            current_n = target_n
        else:
            # Keep interpolated as starting point for next level
            if c5_interp < best_c5:
                best_h = h_interp
                best_c5 = c5_interp
            current_n = target_n

        # Quick SLSQP polish if time allows
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 20
        if remaining > 60 and current_n <= 168:
            h_sls, c5_sls = slsqp_optimize(best_h, time_limit=min(remaining * 0.3, 120), seed=seed + 999)
            if c5_sls < best_c5:
                best_h = h_sls
                best_c5 = c5_sls
                current_n = len(best_h)

    # Phase 3: Random perturbations at best resolution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 10
    if remaining > 30:
        n = len(best_h)
        n_trials = min(8, int(remaining // 30))
        for i in range(n_trials):
            if time.time() - start_time > budget_s - 10:
                break
            noise_scale = 0.15 * (1.0 - i / max(n_trials, 1))
            noise = np.random.randn(n) * noise_scale
            h_trial = best_h + noise
            h_trial = project_constraint(h_trial, n / 2.0)
            t_trial = min(remaining * 0.15, 60)
            h_opt, c5_opt = gradient_descent_optimize(h_trial, time_limit=t_trial, seed=seed + 1000 + i)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Final check: make sure we return a feasible solution
    n_final = len(best_h)
    best_h = project_constraint(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
