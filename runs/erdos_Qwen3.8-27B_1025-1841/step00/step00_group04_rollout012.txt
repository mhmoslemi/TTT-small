```python
def c5_grad_smooth(h, alpha=30.0):
    """Compute smoothed gradient of max_k correlate(h,1-h,'full')[k]*dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max) / dx)  # scale by dx for consistent alpha
    # Actually, let me just use c directly
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def c5_grad(h, alpha=80.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(c)
    ec = np.exp(alpha * (c - cmax))
    w = ec / ec.sum()
    order = np.argsort(w)[::-1]
    M = min(20, 2 * n - 1)
    idx = order[:M]
    wi = w[idx].copy()
    wi /= wi.sum()
    grad = np.zeros(n)
    for weight, ii in zip(wi, idx):
        k = int(ii) - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:] += weight * (1.0 - h[:n - k])
                grad[:n - k] -= weight * h[k:]
        else:
            k2 = -k
            if n - k2 > 0:
                grad[:n - k2] += weight * (1.0 - h[k2:])
                grad[k2:] -= weight * h[:n - k2]
    grad *= dx
    return grad

def c5_obj(h):
    return compute_c5(h)

def project_feasible(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
        else:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
    return h

def upscale(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h.copy())
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new)

def try_binary_round(h):
    n = len(h)
    target = int(n / 2)
    sorted_h = np.sort(h)
    thresh = sorted_h[target - 1]
    h_bin = (h >= thresh).astype(float)
    s = int(h_bin.sum())
    if s > target:
        ones_idx = np.where(h_bin == 1)[0]
        h_bin[ones_idx[:s - target]] = 0
    elif s < target:
        zeros_idx = np.where(h_bin == 0)[0]
        h_bin[zeros_idx[:target - s]] = 1
    return h_bin

def local_search_binary(h, rng, max_iters=2000):
    n = len(h)
    h = try_binary_round(h)
    best_c5 = compute_c5(h)
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if h[i] == 0 and h[j] == 1:
            h[i] = 1
            h[j] = 0
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
            else:
                h[i] = 0
                h[j] = 1
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(42, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    n_parent = len(best_h)
    res_list = list(dict.fromkeys([n_parent, 2 * n_parent, 4 * n_parent,
                                   8 * n_parent, 16 * n_parent]))
    res_list = [r for r in res_list if r <= 4000]

    for target_n in res_list:
        if time.time() - t0 > budget_s * 0.92:
            break

        if target_n == n_parent:
            h_work = best_h.copy()
        else:
            h_work = upscale(best_h, target_n)

        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

        n_restarts = 4 if target_n <= 336 else 2
        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.92:
                break

            if r == 0:
                h_start = h_work.copy()
            elif r == 1:
                h_start = try_binary_round(h_work)
            else:
                noise = rng.uniform(-0.15, 0.15, target_n)
                h_start = project_feasible(h_work + noise)

            try:
                res = minimize(
                    c5_obj, h_start,
                    jac=c5_grad,
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * target_n,
                    constraints=[{
                        'type': 'eq',
                        'fun': lambda x: x.sum() - target_n / 2.0,
                        'jac': lambda x: np.ones(target_n)
                    }],
                    options={'maxiter': 400, 'ftol': 1e-15,
                             'gtol': 1e-10, 'maxls': 40, 'iprint': -1}
                )
                h_opt = project_feasible(res.x)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
                    h_work = h_opt.copy()
            except Exception:
                pass

        if time.time() - t0 > budget_s * 0.92:
            break

        # Binary local search at this resolution
        if target_n <= 672:
            h_bin, c5_bin = local_search_binary(h_work, rng, max_iters=3000)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
