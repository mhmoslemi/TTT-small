```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_h(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(corr, T=50.0):
    """Smooth max using log-sum-exp"""
    max_val = np.max(corr)
    shifted = T * (corr - max_val)
    log_sum = max_val + np.log(np.sum(np.exp(shifted))) / T
    return log_sum

def objective_smooth(h, n_points, T=50.0):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, T) * dx

def grad_objective_smooth(h, n_points, T=50.0):
    """Gradient of smooth max of correlation terms"""
    dx = 2.0 / n_points
    n = n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Weights
    max_c = np.max(corr)
    log_w = T * (corr - max_c) - np.log(np.sum(np.exp(T * (corr - max_c))))
    w = np.exp(log_w)  # softmax weights, sum to 1
    
    # Gradient: sum_k w_k * grad f_k
    # f_k(h) = dx * sum_i h[i + k - (n-1)] * (1 - h[i])
    # grad f_k / dh[j] = dx * sum over pairs involving j
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        # k corresponds to lag = k - (n-1)
        lag = k - (n - 1)
        # f_k = dx * sum_{valid i} h[i + lag] * (1 - h[i])
        # where valid means 0 <= i < n and 0 <= i+lag < n
        if lag >= 0:
            i_range = range(0, n - lag)
        else:
            i_range = range(-lag, n)
        
        for i in i_range:
            j = i + lag
            # ∂f_k/∂h[i] = -dx * h[j]
            # ∂f_k/∂h[j] = dx * (1-h[i])
            grad[i] += w[k] * (-dx * h[j])
            grad[j] += w[k] * (dx * (1 - h[i]))
    
    return grad
```

```python
                best_c5 = c5_new
                best_h = h_new.copy()
                # Polish with higher T
                T2 = min(T * 3.0, 300.0)
                if time_remaining() > 90:
                    try:
                        res2 = run_slsqp(h_new, n, T2, 1000)
                        h_p = np.clip(res2.x, 0.0, 1.0)
                        h_p = fix_sum(h_p, n)
                        c5_p = compute_c5(h_p, n)
                        if c5_p < best_c5:
                            best_c5 = c5_p
                            best_h = h_p
                    except Exception:
                        pass
        except Exception:
            pass

    # Phase 2: Higher resolutions
    for n_target in [160, 320, 512, 768]:
        if time_remaining() < 150:
            break

        h_work = resample_h(best_h, n_target)
        c5_work = compute_c5(h_work, n_target)

        n_runs = 4 if n_target <= 320 else 3
        if time_remaining() < 600 and n_target > 320:
            n_runs = 2
        if time_remaining() < 300 and n_target > 512:
            n_runs = 1

        for attempt in range(n_runs):
            if time_remaining() < 100:
                break

            if attempt == 0:
                h_start = h_work.copy()
            elif attempt == 1:
                h_start = h_work + rng.randn(n_target) * 0.02
                h_start = np.clip(h_start, 0.0, 1.0)
                h_start = fix_sum(h_start, n_target)
            elif attempt == 2:
                h_start = h_work + rng.randn(n_target) * 0.05
                h_start = np.clip(h_start, 0.0, 1.0)
                h_start = fix_sum(h_start, n_target)
            else:
                h_start = make_random_h(n_target, rng)

            T = 30.0 + 20.0 * attempt
            maxiter = 1500 if n_target <= 320 else 800
            try:
                res = run_slsqp(h_start, n_target, T, maxiter)
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new = fix_sum(h_new, n_target)
                c5_new = compute_c5(h_new, n_target)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_work = h_new.copy()
                    # Polish
                    T2 = min(T * 2.5, 250.0)
                    if time_remaining() > 80:
                        try:
                            res2 = run_slsqp(h_new, n_target, T2, 500)
                            h_p = np.clip(res2.x, 0.0, 1.0)
                            h_p = fix_sum(h_p, n_target)
                            c5_p = compute_c5(h_p, n_target)
                            if c5_p < best_c5:
                                best_c5 = c5_p
                                best_h = h_p
                                h_work = h_p.copy()
                        except Exception:
                            pass
            except Exception:
                pass

    # Final validation and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final)
    final_c5 = compute_c5(best_h, n_final)

    return (best_h, final_c5, n_final)
```
