    """Compute gradient of C[k_star] w.r.t. h"""

            grad[m] -= h[m + shift]

            grad[m] += (1.0 - h[m + s])
        for m in range(s, n):
            grad[m] -= h[m - s]

def project_simplex_bounded(v, target_sum):

import time

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_onto_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)
