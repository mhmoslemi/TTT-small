        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += subgradient_at(h, k, n)
        grad /= float(top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h = h - lr * grad / gn
        h = project_onto_constraint(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

def refine_h(h0, n, deadline, n_trials=500):
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h0.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()

    for trial in range(n_trials):
        if time.time() > deadline:
            break
        h_trial = best_h.copy()
        n_pert = max(1, n // 30)
        idx = np.random.choice(n, n_pert, replace=False)
        for i in idx:
            h_trial[i] += np.random.uniform(-0.4, 0.4)
        h_trial = project_onto_constraint(h_trial, target_sum)
        corr_t = np.correlate(h_trial, 1.0 - h_trial, mode='full')
        c5_t = np.max(corr_t) * dx
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_trial
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    np.random.seed(seed)

    best_h = None
    best_c5 = np.inf

    try:
        init = np.array(initial_h_values, dtype=float)
    except NameError:
        init = None

    if init is not None:
        n0 = len(init)
        init = project_onto_constraint(init, n0 / 2.0)
        c5_0, _ = eval_c5(init)
        best_c5 = c5_0
        best_h = init.copy()

        h_opt, c5_opt = optimize_h(init, n0, 4000, deadline - 90)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

        for n in [90, 100, 200]:
            if time.time() > deadline - 40:
                break
            h_interp = interpolate_to_n(best_h, n)
            h_opt, c5_opt = optimize_h(h_interp, n, 2500, deadline - 20)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt
    else:
        n = 100
        h_flat = np.zeros(n)
        h_flat[:n // 2] = 1.0
        best_h, best_c5 = optimize_h(h_flat, n, 2000, deadline - 60)

    # Random restarts at n=100
    n_r = 100
    restart_count = 0
    while time.time() < deadline - 30:
        h_rand = np.random.rand(n_r)
        per_restart_dl = min(deadline - 10, time.time() + 150)
        h_opt, c5_opt = optimize_h(h_rand, n_r, 2000, per_restart_dl)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt
        restart_count += 1
        if restart_count > 500:
            break

    # Refinement phase
    if best_h is not None and time.time() < deadline - 10:
        n_f = len(best_h)
        h_ref, c5_ref = refine_h(best_h, n_f, deadline - 2, 300)
        if c5_ref < best_c5:
            best_c5, best_h = c5_ref, h_ref

    # Also try a few random restarts at n=200 if time allows
    if time.time() < deadline - 60:
        n_r2 = 200
        h_interp_r = interpolate_to_n(best_h, n_r2)
        h_opt, c5_opt = optimize_h(h_interp_r, n_r2, 1500, deadline - 5)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

    # Final validation
    if best_h is None:
        n = 100
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
    n_final = len(best_h)
    best_h = project_onto_constraint(best_h, n_final / 2.0)
    best_c5, _ = eval_c5(best_h)

    return best_h, best_c5, n_final
