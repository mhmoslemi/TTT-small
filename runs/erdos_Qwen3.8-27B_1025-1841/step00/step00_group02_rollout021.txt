import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jpk = j + k
        if 0 <= jpk < n:
            grad[j] += (1.0 - h[jpk])
        jmk = j - k
        if 0 <= jmk < n:
            grad[j] -= h[jmk]
    return grad * dx

def fix_sum(h, target):
    h = h.copy()
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        i = idx[np.argmin(np.abs(h[idx] - 0.5))]
        step = np.clip(diff, -0.5, 0.5)
        h[i] = np.clip(h[i] + step, 0.0, 1.0)
    return h

def slsqp_optimize(h_start, n, max_iter=2000):
    def obj(h):
        return compute_c5(h)
    def obj_grad(h):
        return grad_c5(h)
    def con_fun(h):
        return np.sum(h) - n / 2.0
    def con_jac(h):
        return np.ones(n) / n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h_res = fix_sum(np.clip(res.x, 0, 1), n / 2.0)
        return h_res, compute_c5(h_res)
    except Exception:
        h_res = fix_sum(h_start, n / 2.0)
        return h_res, compute_c5(h_res)

def local_search(h, n, max_iters=5000):
