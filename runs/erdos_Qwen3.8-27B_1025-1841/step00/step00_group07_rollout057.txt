```python
import numpy as np
from scipy.optimize import minimize
import time


def _compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    threshold = 0.995 * max_val
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for idx in active:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            if n - 1 - lag >= 0:
                grad[0:n - lag] += (1.0 - h[lag:n]) * dx
                grad[lag:n] += (-h[0:n - lag]) * dx
        else:
            k = -lag
            if k <= n - 1:
                grad[k:n] += (1.0 - h[0:n - k]) * dx
                grad[0:n - k] += (-h[k:n]) * dx
    grad /= float(len(active))
    return grad


def _con_fun(h):
    return np.sum(h) - len(h) / 2.0


def _con_jac(h):
    return np.ones(len(h))


def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = n / 2.0 - s
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            adj = diff / cnt
            if adj > 1.0 - 1e-8:
                adj = 1.0 - 1e-8
            if adj < -(1e-8):
                adj = -1e-8
            h[mask] += adj
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _make_rng(seed):
    return np.random.default_rng(seed)


def _perturb(h, rng, scale=0.05):
    n = len(h)
    h_new = h + rng.standard_normal(n) * scale
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _fix_sum(h_new, n)
    return h_new


def _block_random(n, rng):
    h = np.zeros(n)
    placed = 0
    i = 0
    target = n // 2
    while placed < target and i < n:
        blk = max(1, int(rng.integers(2, max(3, n // 4))))
        end = min(i + blk, n)
        room = end - i
        take = min(room, target - placed)
        h[i:i + take] = 1.0
        placed += take
        i = end + max(1, int(rng.integers(0, max(2, n // 8))))
    h = _fix_sum(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = _make_rng(seed)

    def elapsed():
        return time.time() - t_start

    def time_left():
        return budget_s - elapsed()

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = 0.5 * np.ones(80)
    best_n = len(best_h)
    best_h = _fix_sum(best_h, best_n)
    best_c5 = _compute_c5(best_h, best_n)

    # Optimize at current resolution
    bounds = [(0.0, 1.0)] * best_n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    for attempt in range(30):
        if time_left() < 20:
            break
        if attempt == 0:
            h_start = best_h.copy()
        elif attempt == 1:
            h_start = _perturb(best_h, rng, 0.1)
        elif attempt == 2:
            h_start = _perturb(best_h, rng, 0.2)
        elif attempt == 3:
            h_start = _block_random(best_n, rng)
        else:
            scale = 0.03 + 0.1 * (attempt % 5) / 5.0
            h_start = _perturb(best_h, rng, scale)

        result = minimize(
            _obj, h_start, jac=_grad,
            bounds=bounds, constraints=cons,
            method='SLSQP',
            options={'maxiter': 800, 'ftol': 1e-10}
        )
        if result.x is not None:
            h_new = _fix_sum(result.x, best_n)
            c5_new = _compute_c5(h_new, best_n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Try higher resolution
    for mult in [2, 4]:
        n_new = best_n * mult
        if n_new > 640 or time_left() < 150:
            break
        # Upsample using linear interpolation
        x_old = np.arange(best_n)
        x_new = np.linspace(0, best_n - 1, n_new)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _fix_sum(h_up, n_new)

        bounds_new = [(0.0, 1.0)] * n_new
        cons_new = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

        for attempt in range(15):
            if time_left() < 40:
                break
            if attempt == 0:
                h_start = h_up.copy()
            elif attempt <= 3:
                h_start = _perturb(h_up, rng, 0.05 * attempt)
            else:
                h_start = _perturb(h_up, rng, 0.02 + 0.01 * (attempt % 4))

            maxiter = min(500, max(50, int(time_left() / 3)))
            result = minimize(
                _obj, h_start, jac=_grad,
                bounds=bounds_new, constraints=cons_new,
                method='SLSQP',
                options={'maxiter': maxiter, 'ftol': 1e-10}
            )
            if result.x is not None:
                h_new = _fix_sum(result.x, n_new)
                c5_new = _compute_c5(h_new, n_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    h_up = best_h.copy()
                    best_n = n_new

    return best_h, best_c5, len(best_h)
```
