                h_t3[i] += rng.normal(0, step3)
            h_t3 = np.clip(h_t3, 0, 1)
            d3 = target3 - np.sum(h_t3)
            if abs(d3) > 1e-12:
                h_t3 += d3 / n_new
                h_t3 = np.clip(h_t3, 0, 1)
                d3 = target3 - np.sum(h_t3)
                if abs(d3) > 1e-12:
                    m3 = (h_t3 < 1.0) if d3 > 0 else (h_t3 > 0.0)
                    c3 = int(np.sum(m3))
                    if c3 > 0:
                        h_t3[m3] += d3 / c3
                        h_t3 = np.clip(h_t3, 0, 1)
            c5_t3 = compute_c5(h_t3)
            del3 = c5_t3 - c5_cur3
            if del3 < 0 or rng.random() < np.exp(-del3 / temp3):
                h_cur3 = h_t3
                c5_cur3 = c5_t3
            temp3 = max(0.005, temp3 * 0.999)

        if c5_cur3 < best_c5:
            best_h, best_c5 = h_cur3.copy(), c5_cur3
            n0 = n_new

    # Phase 4: Final polish at best resolution
    deadline4 = min(t0 + budget_s * 0.98, time.time() + 60)
    n_f = len(best_h)
    target_f = n_f / 2.0
    h_cur4 = best_h.copy()
    c5_cur4 = best_c5
    temp4 = 0.015
    step4 = 0.02

    while time.time() < deadline4:
        h_t4 = h_cur4.copy()
        i4 = int(rng.integers(0, n_f))
        h_t4[i4] += rng.normal(0, step4)
        h_t4 = np.clip(h_t4, 0, 1)
        d4 = target_f - np.sum(h_t4)
        if abs(d4) > 1e-12:
            h_t4 += d4 / n_f
            h_t4 = np.clip(h_t4, 0, 1)
            d4 = target_f - np.sum(h_t4)
            if abs(d4) > 1e-12:
                m4 = (h_t4 < 1.0) if d4 > 0 else (h_t4 > 0.0)
                c4 = int(np.sum(m4))
                if c4 > 0:
                    h_t4[m4] += d4 / c4
                    h_t4 = np.clip(h_t4, 0, 1)
        c5_t4 = compute_c5(h_t4)
        if c5_t4 < c5_cur4:
            h_cur4 = h_t4
            c5_cur4 = c5_t4
            if c5_cur4 < best_c5:
                best_h = h_cur4.copy()
                best_c5 = c5_cur4
        temp4 = max(0.001, temp4 * 0.9999)

    # Ensure exact constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    nf = len(best_h)
    df = nf / 2.0 - np.sum(best_h)
    if abs(df) > 1e-15:
        best_h += df / nf
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, nf)
