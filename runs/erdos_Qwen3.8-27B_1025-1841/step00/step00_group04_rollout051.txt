    """Project h onto {0<=h<=1, sum(h)=n/2}"""

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_smooth(h, tau=0.03):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - np.max(corr)
    return tau * float(np.log(np.sum(np.exp(shifted / tau)))) + np.max(corr)

def project_sum(h, n):
    target = n / 2.0

        diff = target - np.sum(h)
        if abs(diff) < 1e-13:

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h
