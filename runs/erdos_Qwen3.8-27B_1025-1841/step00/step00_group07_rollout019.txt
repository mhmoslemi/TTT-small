    return np.correlate(h, 1.0 - h, mode='full')

def _eval_c5(h, n):
    dx = 2.0 / n
    corr = _compute_corr(h)
    return float(np.max(corr) * dx)

def _round_to_binary(h, n, rng):
    h = np.array(h, dtype=float).copy()
    n_ones = n // 2
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    cur = int(np.sum(h))
    if cur > n_ones:
        idx = np.where(h == 1.0)[0]
        to_flip = rng.choice(idx, size=cur - n_ones, replace=False)
        h[to_flip] = 0.0
    elif cur < n_ones:
        idx = np.where(h == 0.0)[0]
        to_flip = rng.choice(idx, size=n_ones - cur, replace=False)
        h[to_flip] = 1.0
    return h

def _make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _make_structured_binary(n, rng):
    """Create a structured random binary sequence."""
    h = np.zeros(n)
    n_ones = n // 2
    # Try different structured patterns
    pattern = rng.integers(3)
    if pattern == 0:
        # Alternating with random gaps
        positions = np.arange(n)
        # Pick every other position, shuffle within groups
        evens = positions[0::2]
        odds = positions[1::2]
        rng.shuffle(evens)
        rng.shuffle(odds)
        take_evens = evens[:n_ones // 2]
        take_odds = odds[:n_ones - len(take_evens)]
        idx = np.concatenate([take_evens, take_odds])[:n_ones]
    elif pattern == 1:
        # Block pattern
        block_sizes = np.array([2, 3, 5])
        positions = []
        pos = 0
        while pos < n and len(positions) < n_ones:
            bs = int(block_sizes[rng.integers(len(block_sizes))])
            for j in range(bs):
                if pos + j < n:
                    positions.append(pos + j)
            pos += bs * 2
        idx = np.array(positions[:n_ones])
    else:
        # Golden ratio spacing
        phi = (1 + np.sqrt(5)) / 2
        positions = [int((i * phi) % n) for i in range(n_ones)]
        idx = np.array(sorted(set(positions)))
        if len(idx) < n_ones:
            remaining = np.setdiff1d(np.arange(n), idx)
            extra = rng.choice(remaining, size=n_ones - len(idx), replace=False)
            idx = np.sort(np.concatenate([idx, extra]))
    h[idx[:n_ones]] = 1.0
    return h
