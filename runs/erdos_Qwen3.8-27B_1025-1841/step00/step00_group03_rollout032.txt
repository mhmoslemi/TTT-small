    trial_vals = np.linspace(0.0, 1.0, n_trials)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    sweep_count = 0

    while time.time() - start_time < time_limit:
        improved = False
        order = np.random.permutation(n)
        for j in order:
            old = h[j]
            best_t = old
            best_obj = best_c5
            for t in trial_vals:
                if abs(t - old) < 1e-12:
                    continue
                h[j] = t
                obj = compute_c5(h, dx)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            if best_t != old:
                h[j] = best_t
                improved = True
                if best_obj < best_c5:
                    best_c5 = best_obj
                    best_h = h.copy()
            else:
                h[j] = old

        h = project_feasible(h, n)
        sweep_count += 1

        # Occasional perturbation to escape local minima
        if sweep_count % restart_interval == 0 and time.time() - start_time < time_limit * 0.8:
            noise_level = 0.05
            h_perturbed = h + np.random.randn(n) * noise_level
            h_perturbed = project_feasible(h_perturbed, n)
            pert_c5 = compute_c5(h_perturbed, dx)
            if pert_c5 < best_c5 * 1.02:
                h = h_perturbed
            else:
                h = best_h.copy()

        post_c5 = compute_c5(h, dx)
        if post_c5 < best_c5:
            best_c5 = post_c5
            best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    # Get initial values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = project_feasible(h0, n0)

    global_best_h = h0.copy()
    global_best_c5 = compute_c5(h0, 2.0 / n0)
    global_best_n = n0

    # Phase 1: Optimize at base resolution
    t1 = min(60.0, budget_s * 0.08)
    if elapsed() < budget_s - 10:
        h, c5 = coord_descent(h0, n0, t1, start_time)
        if c5 < global_best_c5:
            global_best_c5 = c5
            global_best_h = h.copy()
            global_best_n = n0

    # Phase 2-4: Increase resolution geometrically
    current_h = h.copy()
    current_n = n0
    resolutions = [140, 280, 560]

    for target_n in resolutions:
        remaining = budget_s - elapsed() - 5
        if remaining < 60 or target_n > 600:
            break
        new_n = min(target_n, max(current_n * 2, 140))
        current_h = interpolate_h(current_h, current_n, new_n)
        current_h = project_feasible(current_h, new_n)
        current_n = new_n

        # Allocate time proportionally
        alloc = min(remaining * 0.45, 350.0)
        if alloc < 30:
            break

        h, c5 = coord_descent(current_h, current_n, alloc, start_time)
        if c5 < global_best_c5:
            global_best_c5 = c5
            global_best_h = h.copy()
            global_best_n = current_n
        current_h = h.copy()

    # Phase 5: Random restarts at best resolution
    remaining = budget_s - elapsed() - 5
    if remaining > 30 and global_best_n > 0:
        n_best = global_best_n
        max_restarts = int(remaining / 40)
        for r in range(max_restarts):
            if elapsed() > budget_s - 10:
                break
            # Try different random starts
            strategy = r % 3
            if strategy == 0:
                h_rand = np.random.rand(n_best)
            elif strategy == 1:
                # Smooth random
                k = np.random.randint(2, 6)
                h_rand = 0.5 + 0.3 * np.sin(np.pi * k * np.arange(n_best) / n_best * 2)
                h_rand = 0.5 + 0.2 * np.random.randn(n_best)
            else:
                # Blend of best and random
                frac = np.random.uniform(0.3, 0.7)
                h_rand = frac * global_best_h + (1 - frac) * np.random.rand(n_best)

            h_rand = project_feasible(h_rand, n_best)
            t_alloc = min(35.0, budget_s - elapsed() - 5)
            if t_alloc < 10:
                break
            h, c5 = coord_descent(h_rand, n_best, t_alloc, start_time)
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h.copy()
                global_best_n = n_best
            # Also refine from best with small perturbation
            if elapsed() < budget_s - 20:
                h_pert = global_best_h + np.random.randn(n_best) * 0.02
                h_pert = project_feasible(h_pert, n_best)
                t_alloc2 = min(20.0, budget_s - elapsed() - 5)
                if t_alloc2 > 10:
                    h2, c52 = coord_descent(h_pert, n_best, t_alloc2, start_time)
                    if c52 < global_best_c5:
                        global_best_c5 = c52
                        global_best_h = h2.copy()
                        global_best_n = n_best

    # Final: verify and return
    final_h = project_feasible(global_best_h, global_best_n)
    final_dx = 2.0 / global_best_n
    final_c5 = compute_c5(final_h, final_dx)

    return final_h, final_c5, global_best_n
