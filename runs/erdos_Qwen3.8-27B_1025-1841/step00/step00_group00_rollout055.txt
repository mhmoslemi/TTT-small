def c5_and_grad(h, alpha=100):
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1.0 - h
    c = np.correlate(h, ones_minus_h, mode='full')
    # Log-sum-exp for smooth max
    c_max = np.max(c)
    log_sum_exp = c_max + np.log(np.sum(np.exp(alpha * (c - c_max))))
    c5_smooth = (1.0 / alpha) * log_sum_exp * dx
    
    # Gradient
    exp_weighted = np.exp(alpha * (c - c_max))
    exp_weighted /= exp_weighted.sum()
    
    # For each lag, compute the gradient contribution
    # c[k] has lag = k - (n-1)
    # grad[m] for lag l: 1 - h[m+l] - h[m-l]
    # Total grad[m] = sum_l exp_w[l] * (1 - h[m+l] - h[m-l])
    
    grad = np.zeros(n)
    lags = np.arange(len(c)) - (n - 1)
    
    for idx, (lag, w) in enumerate(zip(lags, exp_weighted)):
        if w < 1e-10:
            continue
        g = np.ones(n)
        # Subtract h[m+lag]
        if lag >= 0:
            if lag > 0:
                g[:n-lag] -= h[lag:]
        else:
            if -lag > 0:
                g[-lag:] -= h[:n+lag]
        # Subtract h[m-lag]
        if lag >= 0:
            if lag > 0:
                g[lag:] -= h[:n-lag]
        else:
            if -lag > 0:
                g[:n+lag] -= h[-lag:]
        grad += w * g
    
    grad *= dx
    
    return c5_smooth, grad

def to_binary(h):
    n = len(h)
    target = n // 2
    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[idx[target:]] = 1.0  # top n/2 values become 1
    return h_bin

def upsample_h(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    return new_h

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    ew = np.exp(alpha * (c - c_max))
    ew_sum = ew.sum()
    ew_n = ew / ew_sum
    val = (1.0 / alpha) * (c_max + np.log(ew_sum)) * dx
    
    grad = np.zeros(n)
    for k in range(len(c)):
        w = ew_n[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        g = np.ones(n)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        if lo < hi:

        hi2 = min(n, n + lag)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2-lag:hi2-lag]
        grad += w * g
    grad *= dx
