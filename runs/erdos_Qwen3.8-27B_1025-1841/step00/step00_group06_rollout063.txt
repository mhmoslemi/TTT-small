def project_onto_feasible(x, S):
    """Project x onto {0<=h<=1, sum(h)=S}"""
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Fix rounding
    diff = S - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(x - (lo+hi)/2)
        # Add to elements not at bounds
        for i in idx:

                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif h[i] > 0 and diff < 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break

    g = np.zeros(n)
    if k >= 0:
        # term1: 1-h[m-k] for m in [k, n-1]
        g[k:] = 1.0 - h[:n-k]

        g[:n-kp] = 1.0 - h[kp:]
        # term2: -h[m-kp] for m in [kp, n-1]

def project_feasible(x, S):
    """Project x onto {h in [0,1]^n : sum(h) = S}"""
    n = len(x)
    lo, hi = -1.5, 2.5
    for _ in range(60):

        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        fm = (h > 1e-15) & (h < 1.0 - 1e-15)
        nf = fm.sum()
        if nf > 0:
            h[fm] += diff / nf
            np.clip(h, 0.0, 1.0, out=h)
    return h

def c5_value(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgrad(h, n):
    """Return (c5, subgradient)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(corr))
    g = np.zeros(n)
    n_active = 0

        if corr[m] >= max_val * 0.999:
            k = m - (n - 1)
            n_active += 1
            if k >= 0:
                g[k:] += 1.0 - h[:n-k]
                g[:n-k] -= h[k:]
            else:
                kp = -k
                g[:n-kp] += 1.0 - h[kp:]
                g[kp:] -= h[:n-kp]
    if n_active > 0:
        g /= n_active
    return max_val * dx, g * dx

import numpy as np
import time

def project_feasible(x, S):
    """Project onto {0<=h<=1, sum(h)=S} via bisection on water level."""
    lo, hi = -1.5, 2.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0, 1)
    d = S - h.sum()
    if abs(d) > 1e-12:
        fm = (h > 1e-15) & (h < 1 - 1e-15)
        if fm.any():
            h[fm] += d / fm.sum()
            np.clip(h, 0, 1, out=h)
    return h

def c5_eval(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def c5_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = float(np.max(corr))
    g = np.zeros(n)
    na = 0
    for m in range(len(corr)):
        if corr[m] >= mv - 1e-8:
            na += 1
            k = m - (n - 1)
            if k >= 0:
                g[:n-k] += 1.0 - h[k:]
                g[k:] -= h[:n-k]
            else:
                kp = -k
                g[kp:] += 1.0 - h[:n-kp]
                g[:n-kp] -= h[kp:]
    if na:
        g /= na
    return mv * dx, g * dx

def pgrad(h, n, max_iters, t0, t_end, alpha0=0.8):
    S = n / 2.0
    hc = h.copy()
    bh = hc.copy()
    bc = c5_eval(hc, n)
    alpha = alpha0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        c5v, g = c5_subgrad(hc, n)
        step_done = False
        a = alpha
        for _ in range(40):
            ht = project_feasible(hc - a * g, S)
            ct = c5_eval(ht, n)
            if ct < c5v - 1e-12:
                hc = ht
                c5c = ct
                step_done = True
                break

        if not step_done:

            bc = c5c
            bh = hc.copy()
        if c5c < c5v * 0.999:
            alpha = min(a * 1.3, 3.0)
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

_N = 0

def project_feasible(x, S):
    lo, hi = -1.5, 2.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0, 1)
    d = S - h.sum()
    if abs(d) > 1e-12:
        fm = (h > 1e-15) & (h < 1 - 1e-15)
        if fm.any():
            h[fm] += d / fm.sum()
            np.clip(h, 0, 1, out=h)
    return h

def c5_eval(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def c5_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = float(np.max(corr))
    g = np.zeros(n)
    na = 0
    for m in range(len(corr)):
        if corr[m] >= mv - 1e-8:
            na += 1
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += 1.0 - h[k:]
                g[k:] -= h[:n - k]
            else:
                kp = -k
                g[kp:] += 1.0 - h[:n - kp]
                g[:n - kp] -= h[kp:]
    if na:
        g /= na
    return mv * dx, g * dx

def _obj(x):
    return c5_eval(x, _N)

def _obj_jac(x):
    _, g = c5_subgrad(x, _N)
    return g

def _con_eq(x):
    return x.sum() - _N * 0.5

def _con_eq_jac(x):
    return np.ones(_N)

def pgrad(h, n, max_iters, t_end, alpha0=0.8):
    S = n * 0.5
    hc = h.copy()
    bh = hc.copy()
    bc = c5_eval(hc, n)
    alpha = alpha0
    stagnant = 0
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        c5v, g = c5_subgrad(hc, n)
        a = alpha
        found = False
        for _ in range(40):
            ht = project_feasible(hc - a * g, S)
            ct = c5_eval(ht, n)
            if ct < c5v - 1e-12:
                hc = ht
                found = True
                break
            a *= 0.6
            if a < 1e-10:
                break
        if not found:
            break
        if ct < bc:
            bc = ct
            bh = hc.copy()
            stagnant = 0
        else:
            stagnant += 1
        if ct < c5v * 0.999:
            alpha = min(a * 1.2, 3.0)
        elif stagnant > 100:
            alpha *= 0.7
            stagnant = 0
    return bh, bc

def slsqp_opt(h, n, maxiter=500):
    global _N
    _N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    try:
        res = minimize(_obj, h, jac=_obj_jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h_out = project_feasible(np.asarray(res.x), n * 0.5)
        return h_out
    except Exception:
        return h

def flip_segment(h, n, rng):
    S = n * 0.5
    i = rng.randint(0, max(1, n - 1))
    w = rng.randint(2, max(3, n // 3))
    j = min(i + w, n)
    h_new = h.copy()
    h_new[i:j] = 1.0 - h_new[i:j]
    return project_feasible(h_new, S)

def binarize(h, n):
    S = n * 0.5
    h_b = (h > 0.5).astype(float)
    cur_ones = int(h_b.sum())
    if cur_ones > int(S):
        order = np.argsort(h)[:cur_ones - int(S)]
        h_b[order] = 0.0
    elif cur_ones < int(S):
        order = np.argsort(h)[::-1][:int(S) - cur_ones]
        h_b[order] = 1.0
    return project_feasible(h_b, S)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 15
    rng = np.random.RandomState(seed)

    n = 95
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        h0 = 0.5 + 0.4 * np.sin(np.linspace(0, 8 * np.pi, n))

    S = n * 0.5
    h0 = project_feasible(h0, S)
    best_h = h0.copy()
    best_c5 = c5_eval(h0, n)
    best_n = n

    # Phase 1: Projected gradient
    h1, c5_1 = pgrad(h0, n, 600, t_end, 0.8)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n

    # Phase 2: SLSQP refinement
    if time.time() < t_end - 60:
        h2 = slsqp_opt(best_h, best_n, 800)
        c5_2 = c5_eval(h2, best_n)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        # Second SLSQP pass
        if time.time() < t_end - 40:
            h2 = slsqp_opt(best_h, best_n, 400)
            c5_2 = c5_eval(h2, best_n)
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2

    # Phase 3: Structured restarts
    for r in range(12):
        if time.time() > t_end - 30:
            break
        if r % 3 == 0:
            h_r = flip_segment(best_h, best_n, rng)
        elif r % 3 == 1:
            h_r = binarize(best_h + rng.randn(best_n) * 0.1)
        else:
            h_r = best_h + rng.randn(best_n) * (0.03 + 0.01 * r)
            h_r = project_feasible(h_r, best_n * 0.5)
        hr, cr = pgrad(h_r, best_n, 300, t_end, 0.6)
        if cr < best_c5:
            best_h, best_c5 = hr, cr
        if time.time() < t_end - 60 and cr < best_c5 + 0.01:
            hr = slsqp_opt(hr, best_n, 300)
            cr = c5_eval(hr, best_n)
            if cr < best_c5:
                best_h, best_c5 = hr, cr

    # Phase 4: Higher resolution
    if time.time() < t_end - 120:
        n2 = min(best_n * 2, 250)
        S2 = n2 * 0.5
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, n2)
        h2 = np.interp(xn, xo, best_h)
        h2 = project_feasible(h2, S2)
        h2, c5_2 = pgrad(h2, n2, 500, t_end, 0.5)
        if time.time() < t_end - 40:
            h2 = slsqp_opt(h2, n2, 500)
            c5_2 = c5_eval(h2, n2)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2

    # Phase 5: Final polish
    if time.time() < t_end - 10:
        h_f, c5_f = pgrad(best_h, best_n, 300, t_end, 0.3)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
        if time.time() < t_end - 5:
            h_f = slsqp_opt(best_h, best_n, 200)
            c5_f = c5_eval(h_f, best_n)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f

    best_h = project_feasible(best_h, best_n * 0.5)
    best_c5 = c5_eval(best_h, best_n)
    return best_h, best_c5, best_n
