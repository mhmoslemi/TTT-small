import numpy as np
from scipy.optimize import minimize
import time

def c5_objective(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def c5_obj_wrapper(x, n):
    return c5_objective(x, n)

def eq_constraint(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0

        cnt = np.sum(mask)
        if cnt == 0:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_max_corr(arr):

def binary_local_search(arr, rng, max_iter=2000):
    n = len(arr)
    best = arr.copy()
    best_val = binary_max_corr(best)
