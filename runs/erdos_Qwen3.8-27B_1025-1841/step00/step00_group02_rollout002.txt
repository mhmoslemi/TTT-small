def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(e)) / alpha

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 0.001) & (h < 0.999)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
        h = np.clip(h, 0, 1)
    return h

def slsqp_optimize(h, n, target, alpha, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smooth_obj(x, alpha), h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 500})
        if res.x is not None:
            return fix_sum(res.x, target)
    except Exception:
        pass
    return h

def pairwise_search(h, n, target, t0, deadline):
    current = h.copy()
    current_c5 = compute_c5(current)
    improved = True
    while improved and (time.time() - t0) < deadline:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                new_h = current.copy()
                new_h[i], new_h[j] = current[j], current[i]
                if abs(np.sum(new_h) - target) < 1e-10:
                    new_c5 = compute_c5(new_h)
                    if new_c5 < current_c5 - 1e-15:
                        current = new_h
                        current_c5 = new_c5
                        improved = True
                        break
            if improved:
                break
    return current, current_c5

def anneal_search(h, n, target, t0, deadline, rng):
    current = h.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    current_c5 = best_c5
    T = 0.003
    while (time.time() - t0) < deadline:
        i, j = rng.choice(n, 2, replace=False)
        r = rng.random()
        if r < 0.55:
            new_h = current.copy()
            new_h[i], new_h[j] = current[j], current[i]
        elif r < 0.85:
            amt = rng.uniform(0.01, 0.25)
            new_h = current.copy()
            new_h[i] = max(0.0, current[i] - amt)
            new_h[j] = min(1.0, current[j] + amt)
            new_h = fix_sum(new_h, target)
        else:
            delta = rng.uniform(-0.08, 0.08)
            new_h = current.copy()
            new_h[i] = np.clip(new_h[i] + delta, 0, 1)
            new_h = fix_sum(new_h, target)
        if abs(np.sum(new_h) - target) > 1e-8:
            continue
        new_c5 = compute_c5(new_h)
        d = new_c5 - current_c5
        if d < 0 or (T > 1e-9 and rng.random() < np.exp(-d / T)):
            current = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = new_h.copy()
        T = max(T * 0.9997, 1e-9)
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, TypeError):
        n_init = 45
        h0 = np.zeros(n_init)
        h0[ind::2 for ind in range(n_init//2)] = 1.0 if False else 1.0
        h0[:n_init//2] = 1.0
        if n_init % 2 == 1:
            h0[n_init//2] = 0.5

    n = len(h0)
    target = n / 2.0
    h0 = fix_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing alpha (refinement)
    h = h0.copy()
    alphas = [30, 60, 100, 200, 400]
    for alpha in alphas:
        if time.time() - t0 > budget_s * 0.18:
            break
        h_new = slsqp_optimize(h, n, target, alpha, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    # Phase 2: High-alpha SLSQP
    if time.time() - t0 < budget_s * 0.32:
        for alpha in [800, 1500]:
            h_new = slsqp_optimize(h, n, target, alpha, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new

    # Phase 3: Pairwise local search
    deadline3 = t0 + min(budget_s * 0.50, time.time() - t0 + 180)
    deadline3 = min(deadline3, t0 + budget_s * 0.50)
    if time.time() - t0 < budget_s * 0.50:
        h_ps, c5_ps = pairwise_search(best_h, n, target, t0, t0 + budget_s * 0.50)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 4: Simulated annealing
    remaining_for_anneal = budget_s * 0.72 - (time.time() - t0)
    if remaining_for_anneal > 30:
        h_an, c5_an = anneal_search(best_h, n, target, t0,
                                     t0 + remaining_for_anneal, rng)
        if c5_an < best_c5:
            best_c5 = c5_an
            best_h = h_an

    # Phase 5: Higher resolution attempts
    for n_hi in [90, 135, 180]:
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 80
        if remaining < 80:
            break
        n_lo = len(best_h)
        x_lo = np.linspace(0.0, 1.0, n_lo)
        x_hi = np.linspace(0.0, 1.0, n_hi)
        h_hi = np.interp(x_hi, x_lo, best_h)
        h_hi = fix_sum(h_hi, n_hi / 2.0)

        alloc = remaining * 0.5
        deadline_5 = t0 + alloc
        h_hi = slsqp_optimize(h_hi, n_hi, n_hi/2.0, 100, maxiter=400)
        c5_hi = compute_c5(h_hi)

        if (time.time() - t0) < deadline_5 + alloc * 0.6:
            h_hi = slsqp_optimize(h_hi, n_hi, n_hi/2.0, 500, maxiter=400)
            c5_hi = compute_c5(h_hi)

        h_hi, c5_hi = pairwise_search(h_hi, n_hi, n_hi/2.0, t0,
                                       min(time.time() - t0 + 30, budget_s - 80))
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi

    # Final: one more SLSQP sprint on best
    remaining_final = budget_s - (time.time() - t0) - 10
    if remaining_final > 20:
        nf = len(best_h)
        hf = best_h.copy()
        hf = slsqp_optimize(hf, nf, nf/2.0, 2000, maxiter=500)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    best_h = np.clip(best_h, 0.0, 1.0)
    nf = len(best_h)
    best_h = fix_sum(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, nf
