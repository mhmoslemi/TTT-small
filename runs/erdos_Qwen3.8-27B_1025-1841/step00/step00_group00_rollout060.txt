from scipy.optimize import minimize, linprog
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0, 1)
    # Adjust sum
    s = np.sum(h)
    target = n / 2.0
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    # Use bisection on the shift
    # h_proj = clip(h - shift, 0, 1), find shift so sum = target
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        h_test = np.clip(h - mid, 0, 1)
        s_test = np.sum(h_test)
        if s_test > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    return h_proj

def subgradient_obj(h, n):
    """Return (objective, subgradient) of max_k c[k]"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Find all k achieving max (for smoothed subgradient)
    max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    sg = np.zeros(n)
    for ki in max_indices:
        k = ki - (n - 1)  # actual shift
        for i in range(n):
            term1 = 0.0
            term2 = 0.0
            if 0 <= i - k < n:
                term1 = 1.0 - h[i - k]
            if 0 <= i + k < n:
                term2 = -h[i + k]
            sg[i] += term1 + term2
    sg /= len(max_indices)
    return max_val * dx, sg

def projected_gradient_descent(h, n, max_iter=10000, lr_init=0.01):
    h = h.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(max_iter):
        obj, sg = subgradient_obj(h, n)
        if obj < best_c5:

            best_h = h.copy()
        h_new = h - lr * sg
        h_new = project_h(h_new, n)

    if k >= 0:

            sg[k:] += (1.0 - h[:n-k])
            sg[:n-k] -= h[k:]
    else:

            sg[:n-k2] += (1.0 - h[k2:])
            sg[k2:] -= h[:n-k2]
    return sg

        h3, c5_3 = sgd_optimize(h2, n_r, 0.002, t_alloc * 0.18, top_k=3)

        # Phase 3b: Second SGD pass if time
        if time() - t_start < budget_s * 0.88:
            h4, c5_4 = sgd_optimize(h3, n_r, 0.001, t_alloc * 0.05, top_k=3)
            if c5_4 < c5_3:
                h3, c5_3 = h4, c5_4

        candidates = [(h1, c5_1), (h2, c5_2), (h3, c5_3)]
        best_local = min(candidates, key=lambda x: x[1])
        if best_local[1] < best_c5:
            best_c5 = best_local[1]
            best_h = best_local[0].copy()

    # Final polish at best resolution
    remain = budget_s * 0.95 - (time() - t_start)
    if remain > 5:
        h_f, c5_f = sgd_optimize(best_h, len(best_h), 0.0005, remain, top_k=3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    best_h = project_h(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
