import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def interpolate_to(h, new_n):
    old_n = len(h)
    old_x = np.arange(old_n, dtype=float)
    new_x = np.arange(new_n, dtype=float)
    new_h = np.interp(new_x, old_x, h)
    new_h = np.clip(new_h, 0.0, 1.0)
    target = new_n / 2.0
    diff = target - np.sum(new_h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = new_h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                new_h[mask] = np.minimum(new_h[mask] + diff / cnt, 1.0)
        else:
            mask = new_h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                new_h[mask] = np.maximum(new_h[mask] + diff / cnt, 0.0)
    return new_h

def to_binary(h, n):
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:n//2]] = 1.0
    return b

def local_search(h, n, time_limit, seed=42):

    zeros = list(np.where(h == 0)[0])
    if len(ones) != n // 2:
        h = to_binary(h, n)
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
    best = compute_c5(h)
    best_c5 = best
    start = time.time()
    T = 0.002
    T_min = 0.0001
    cool = 0.9995
    stuck = 0
    while time.time() - start < time_limit:
        improved = False
        n_tries = 200
        for _ in range(n_tries):
            i_idx = rng.randint(len(ones))
            j_idx = rng.randint(len(zeros))
            i = ones[i_idx]
            j = zeros[j_idx]
            h[i] = 0.0
            h[j] = 1.0
            c5 = compute_c5(h)
            delta = c5 - best
            if delta < -1e-14:
                best = c5
                if c5 < best_c5:
                    best_c5 = c5
                ones[i_idx] = j
                zeros[j_idx] = i
                improved = True
            else:
                if T > T_min and rng.random() < np.exp(-delta / T):
                    ones[i_idx] = j
                    zeros[j_idx] = i
                    improved = True
                else:
                    h[i] = 1.0

        if improved:
            stuck = 0
        else:

            if stuck > 30:

                    j_idx = rng.randint(len(zeros))
                    i = ones[i_idx]
                    j = zeros[j_idx]
                    h[i] = 0.0
                    h[j] = 1.0
                    ones[i_idx] = j
                    zeros[j_idx] = i
                best = compute_c5(h)
                T = max(T, 0.001)
                stuck = 0
    if best < best_c5:
