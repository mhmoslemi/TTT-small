import numpy as np
from scipy.optimize import minimize
import math
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old)
    x_new = np.arange(n_new)
    h_new = np.interp(x_new, x_old, h)
    diff = n_new / 2.0 - np.sum(h_new)
    h_new += diff / n_new
    return np.clip(h_new, 0.0, 1.0)

def compute_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[0:n-s] += (1.0 - h[s:n])
        grad[s:n] -= h[0:n-s]
    else:
        t = -s
        grad[t:n] += (1.0 - h[0:n-t])
        if 2 * t < n:
            grad[2*t:n] -= h[t : n-t + t]
    return dx * grad, max_idx

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):

        proj = np.clip(h - mid, 0.0, 1.0)

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    if target_sum is None:
        target_sum = n / 2.0
    h = h.copy()
    for step in range(n_steps):
        alpha = alpha0 / (1.0 + decay * step)
        grad, _ = compute_subgrad(h)
        h = h - alpha * grad
        h = project_box_sum(h, target_sum)
    return h

def smooth_objective(h, temperature=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Smooth max using log-sum-exp
    c = np.max(corr)
    log_terms = (corr - c) / temperature
    log_terms = np.clip(log_terms, -50, 50)
    return c + temperature * np.log(np.sum(np.exp(log_terms)))

def smooth_gradient(h, temperature=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c = np.max(corr)
    log_terms = (corr - c) / temperature
    log_terms = np.clip(log_terms, -50, 50)
    weights = np.exp(log_terms) / np.sum(np.exp(log_terms))
    # Weighted sum of subgradients
    grad_sum = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-8:
            continue
        s = k - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[0:n-s] += (1.0 - h[s:n])
            g[s:n] -= h[0:n-s]
        else:
            t = -s
            g[t:n] += (1.0 - h[0:n-t])
            if 2 * t < n:
                g[2*t:n] -= h[t : n-t + t]
        grad_sum += weights[k] * g
    return dx * grad_sum

def penalty_objective(h_flat, n, lam):
    h = np.clip(h_flat, 0.0, 1.0)
    c5 = compute_c5(h)
    pen = lam * (np.sum(h) - n / 2.0) ** 2
    return c5 + pen

def penalty_gradient(h_flat, n, lam, temperature=0.05):
    h = np.clip(h_flat, 0.0, 1.0)
    # Use smooth gradient for C5 part
    grad_c5 = smooth_gradient(h, temperature)

    # Project gradient to feasible directions (zero where h is at bound)
    mask = (h > 1e-6) & (h < 1.0 - 1e-6)

            params[2*(k-1)] -= lr * np.sum(grad_h * basis_c)
            params[2*(k-1)+1] -= lr * np.sum(grad_h * basis_s)
        # Clamp params to keep h in reasonable range
        params = np.clip(params, -0.49, 0.49)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Subgradient method at various resolutions
    for n in [90, 150, 200]:
        if time.time() - t0 > budget_s - 200:
            break
        h = upsample(h_init, n)
        # Multiple subgradient runs with different parameters
        for alpha0 in [0.05, 0.1, 0.2]:
            if time.time() - t0 > budget_s - 150:
                break
            h_opt = subgradient_method(h, 300, alpha0, 0.02)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
    
    # Phase 2: Fourier optimization
    for n in [100, 200, 300]:
        if time.time() - t0 > budget_s - 200:
            break
        for K in [5, 10, 15]:
            if time.time() - t0 > budget_s - 100:
                break
            h_f, c5_f = fourier_optimize(n, K, best_h, max_iters=2000, lr=0.02)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
    
    # Phase 3: Local refinement with L-BFGS-B on the smooth objective
    n_final = len(best_h)
    if time.time() - t0 < budget_s - 100:
        def obj_and_grad(h_flat):
            h, g = penalty_gradient(h_flat, n_final, lam=50.0, temperature=0.03)
            return penalty_objective(h_flat, n_final, 50.0), g
        try:
            res = minimize(obj_and_grad, best_h, method='L-BFGS-B',
                          bounds=[(0,1)]*n_final, options={'maxiter': 200, 'ftol': 1e-10})
            h_res = np.clip(res.x, 0, 1)
            h_res = project_box_sum(h_res, n_final/2.0)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
        except Exception:
            pass
    
    # Phase 4: Fine-tuning with smaller steps subgradient
    if time.time() - t0 < budget_s - 50:
        n_tune = max(len(best_h), 200)
        h_tune = upsample(best_h, n_tune)
        h_tune = subgradient_method(h_tune, 500, 0.02, 0.01)
        c5_tune = compute_c5(h_tune)
        if c5_tune < best_c5:
            best_c5 = c5_tune
            best_h = h_tune.copy()
        
        # Even finer
        if time.time() - t0 < budget_s - 30:
            h_tune2 = subgradient_method(best_h, 1000, 0.01, 0.005)
            c5_tune2 = compute_c5(h_tune2)
            if c5_tune2 < best_c5:
                best_c5 = c5_tune2
                best_h = h_tune2.copy()
    
    # Phase 5: Random perturbation + local search
    if time.time() - t0 < budget_s - 30:
        for trial in range(5):
            if time.time() - t0 > budget_s - 10:
                break
            h_pert = best_h + 0.05 * np.random.randn(len(best_h))
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = project_box_sum(h_pert, len(best_h)/2.0)
            h_pert = subgradient_method(h_pert, 300, 0.03, 0.01)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pert.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, len(best_h)
