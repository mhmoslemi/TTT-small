    dx = 2.0 / n

    best_val = curr_val
    best_h = h.copy()

        best_swap = None
        best_swap_val = curr_val

                h[i] = 0
                h[j] = 1
                val = binary_evaluate(h, n)

                    best_swap = (i, j)
                h[i] = 1
                h[j] = 0

            zeros = np.where(h == 0)[0]

import numpy as np
import time

def eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    h = h.astype(np.float64).copy()
    target = n // 2
    n_ones = int(np.sum(h >= 0.5))
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    if n_ones > target:
        # Remove excess ones (those closest to 0.5, i.e., most recently set)
        # Just remove arbitrary ones
        ones = np.where(h == 1)[0]
        to_remove = np.random.choice(ones, n_ones - target, replace=False)
        h[to_remove] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        to_add = np.random.choice(zeros, target - n_ones, replace=False)
        h[to_add] = 1
    return h

def local_search(h_init, n, time_limit):
    """Exhaustive best-improvement local search on binary solutions."""
    h = make_binary(h_init, n)
    curr_val = eval_c5(h, n)
    best_val = curr_val
    best_h = h.copy()
    dx = 2.0 / n
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    start = time.time()
    stall = 0
    
    while time.time() - start < time_limit:
        best_gain = -1e-15
        best_i = best_j = -1
        
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                corr = np.correlate(h, 1.0 - h, mode='full')
                val = float(np.max(corr)) * dx
                h[j] = 0
                gain = curr_val - val
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
            h[i] = 1
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            # Update lists
            oi = ones.index(best_i)
            oj = zeros.index(best_j)
            temp = ones[oi]
            ones[oi] = best_j
            zeros[oj] = temp
            curr_val = curr_val - best_gain
            stall = 0
            if curr_val < best_val:
                best_val = curr_val
                best_h = h.copy()
        else:
            stall += 1
            # Perturbation: random 3-swap
            if stall >= 3:

                    h[old_zero] = 1
                    ones[a] = old_zero
                    zeros[b] = old_one
                curr_val = eval_c5(h, n)
                stall = 0
    
    return best_h, best_val

        to_add = rng.choice(zeros, target - n_ones, replace=False)
        h[to_add] = 1
    return h

def upsample(h_small, n_small, n_large):
