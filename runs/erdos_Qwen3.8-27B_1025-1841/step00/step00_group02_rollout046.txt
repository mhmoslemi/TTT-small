            h_init = best_h.copy()
        else:
            xo = np.linspace(0.0, 1.0, len(best_h))
            xn = np.linspace(0.0, 1.0, n)
            h_init = np.interp(xn, xo, best_h)
        h_init = _project(h_init, n)
        c5i = _eval_c5(h_init)
        if c5i < best_c5:
            best_h = h_init.copy()
            best_c5 = c5i

        # Number of optimization attempts
        n_att = 10 if n <= 60 else (6 if n <= 120 else 4)
        for att in range(n_att):
            if rem() < 90:
                break

            if att == 0:
                hs = h_init.copy()
            elif att == 1:
                hs = _project(h_init + rng.normal(0, 0.015, n), n)
            elif att == 2:
                hs = _project(h_init + rng.normal(0, 0.04, n), n)
            elif att == 3:
                hs = _project(h_init + rng.uniform(-0.06, 0.06, n), n)
            elif att == 4:
                hs = _binary_start(n, rng)
            elif att == 5:
                # Smoothed binary
                hb = _binary_start(n, rng)
                hs = _project(0.3 * h_init + 0.7 * hb, n)
            elif att == 6:
                # Sine perturbation
                x = np.arange(n)
                hs = _project(h_init + 0.08 * np.sin(2 * np.pi * x / n * rng.integers(2, 6)), n)
            else:
                hs = _project(h_init + rng.normal(0, 0.08, n), n)

            tl = rem() - 70
            if tl < 5:
                break

            if n <= 120:
                ho = _slsqp_opt(hs, n, tl)
            elif n <= 300:
                ho = _lbfgs_opt(hs, n, tl)
            else:
                ho = _lbfgs_opt(hs, n, tl)

            if ho is not None:
                if rem() > 50:
                    ho = _subgrad_polish(ho, n, steps=40)
                    ho = _project(ho, n)
                c5 = _eval_c5(ho)
                if c5 < best_c5:
                    best_h = ho
                    best_c5 = c5
                    h_init = ho.copy()

        # Final SLSQP pass on best
        if rem() > 80 and n <= 120:
            hp = _slsqp_opt(best_h, n, min(rem() - 50, 180))
            if hp is not None:
                c5 = _eval_c5(hp)
                if c5 < best_c5:
                    best_h = hp
                    best_c5 = c5

        # Subgradient polish
        if rem() > 30:
            best_h = _subgrad_polish(best_h, len(best_h), steps=60)
            best_c5 = _eval_c5(best_h)

    # Final ensure
    best_h = _project(best_h, len(best_h))
    final_c5 = _eval_c5(best_h)
    return best_h, final_c5, len(best_h)
