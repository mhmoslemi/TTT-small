import numpy as np
from scipy.optimize import minimize_scalar

def project_to_simplex_box(g, target_sum):
    """Project g onto {h: sum(h)=target_sum, 0<=h<=1}"""
    n = len(g)
    def f(lam):
        return np.sum(np.clip(g - lam, 0, 1)) - target_sum
    # f is decreasing in lam
    # Find range
    lo, hi = g.min() - 1, g.max() + 1
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(g - lam, 0, 1)

    # sum(clip(g-lam, 0, 1)) is decreasing in lam
    # At lam = g.min() - 1: sum = n (all 1s)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, max_iters, time_limit, initial_lr=None):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    if initial_lr is None:
        lr = 0.5
    else:
        lr = initial_lr
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    best_stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 80:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-8:
                break

        h = h_new

    return best_h, best_obj


def optimize_subgrad(h, n, max_iters, time_limit, initial_lr=None):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if initial_lr is None:
        lr = 0.01
    else:
        lr = initial_lr
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(max_iters):

        corr = compute_corr(h)

            stall = 0
        else:

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:

        j0b = max(0, -sh)
        j1b = min(n, n - sh)

            v *= 0.5
            stall = 0
