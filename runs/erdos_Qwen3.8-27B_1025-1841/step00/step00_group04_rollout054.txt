    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-12)[0]
    
    grad = np.zeros(n)
    for p in active:
        lag = p - (n - 1)
        # dcorr/dh[j] = (1-h[j+lag])*[0<=j+lag<n] - h[j-lag]*[0<=j-lag<n]
        g = np.zeros(n)
        if lag >= 0:

            g[:n-lag] += (1 - h[lag:n])
            # -h[j-lag] for j in [lag, n-1]

            g[:n-m] -= h[m:n]
        grad += g
    
    grad *= dx / len(active)
    return c5, grad

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    c5 = float(max_val) * dx
    tol = 1e-14
    active = np.where(np.abs(corr - max_val) <= tol)[0]
    grad = np.zeros(n)

        g = np.zeros(n)

            # So first term: j in [k, n-1], contribution (1-h[j-k])
            # Second term: j+k < n means j < n-k. j >= 0. So j in [0, n-1-k]

import numpy as np
from scipy.optimize import minimize
import time

_optimize_n = 42

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    c5 = float(max_val) * dx
    active = np.where(corr >= max_val - 1e-10)[0]
    grad = np.zeros(n)
    for p in active:
        k = int(p - (n - 1))
        if k >= 0:
            if k < n:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
        else:
            m = -k
            if m < n:
                grad[:n-m] += (1.0 - h[m:])
                grad[m:] -= h[:n-m]
    grad *= dx / len(active)
    return c5, grad

def project_simplex(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_obj(x):
    return compute_c5_grad(x)[0]

def slsqp_grad(x):
    return compute_c5_grad(x)[1]

    return np.sum(x) - _optimize_n / 2.0

    while time.time() - t0 < time_limit:
        result = minimize(
            fun=slsqp_obj,
            x0=best_x,
            jac=slsqp_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}],
            options={'maxiter': 500, 'ftol': 1e-14}
        )
        new_x = project_simplex(result.x, n)
        new_c5 = compute_c5(new_x)
        if new_c5 < best_c5 - 1e-14:
            best_x = new_x
            best_c5 = new_c5
        else:
            break
    return best_x, best_c5

def optimize_pg(h_init, n, time_limit=30):

    best_h = h.copy()

    t0 = time.time()
    lr = 0.5
    stuck = 0

        c5, grad = compute_c5_grad(h)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-12:
            break
        h_new = project_simplex(h - lr * grad, n)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5 - 1e-14:
            h = h_new
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                stuck = 0
            lr = min(lr * 1.05, 2.0)
        else:
            lr *= 0.5
            stuck += 1

    return best_h, best_c5

def interpolate_fn(h, n_from, n_to):
    x_old = np.linspace(0.0, 1.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)

def make_structured_guess(n, kind=0):
