def _compute_c5(h, n_points):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] = np.clip(h[mask] + diff / np.sum(mask), 0.0, 1.0)
    return h

def _upsample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    return np.interp(x_new, x_old, h_old)

def _smooth_c5_grad(h, n_points, T):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = T * (corr - max_corr)
    shifted = np.minimum(shifted, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    c5 = dx * (max_corr + np.log(sum_exp) / T)
    weights = exp_s / sum_exp
    
    grad = np.zeros(n_points)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n_points - 1)
        i_min = max(0, -lag)
        i_max = min(n_points, n_points - lag)
        if i_min >= i_max:
            continue
        ii = np.arange(i_min, i_max)
        jj = ii + lag
        grad[ii] += w * (-h[jj])
        grad[jj] += w * (1.0 - h[ii])
    grad *= dx
    return c5, grad

            h = 0.5 * h + 0.5 * h_new  # Damped step

import numpy as np
import time

def _compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def _upsample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    return np.interp(x_new, x_old, h_old)

def _smooth_c5_grad(h, n_points, T):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = T * (corr - max_corr)
    shifted = np.minimum(shifted, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    c5 = dx * (max_corr + np.log(sum_exp) / T)
    weights = exp_s / sum_exp
    grad = np.zeros(n_points)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-9:
            continue
        lag = k - (n_points - 1)
        i_min = max(0, -lag)
        i_max = min(n_points, n_points - lag)
        if i_min >= i_max:
            continue
        ii = np.arange(i_min, i_max)
        jj = ii + lag
        grad[ii] += w * (-h[jj])
        grad[jj] += w * (1.0 - h[ii])
    grad *= dx
    return c5, grad

def _optimize(h_init, n_points, target_sum, time_limit, max_iter=3000):
    t0 = time.time()
    h = _project(h_init.copy(), target_sum)
    best_c5 = _compute_c5(h, n_points)
    best_h = h.copy()
    total_iters = max_iter
    for it in range(total_iters):
        if time.time() - t0 > time_limit:
            break
        frac = it / max(total_iters - 1, 1)
        T = 8.0 + 120.0 * frac
        c5, grad = _smooth_c5_grad(h, n_points, T)
        lr = 0.015 / (1.0 + 0.002 * it)
        h_new = h - lr * grad
        h_new = _project(h_new, target_sum)
        c5_new = _compute_c5(h_new, n_points)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= c5 + 1e-14:
            h = h_new
        else:
            h = _project(0.7 * h + 0.3 * h_new, target_sum)
    return best_h, best_c5

def _try_structured_init(n_points, target_sum):
    h = np.zeros(n_points)
    p = 4
    for i in range(n_points):
        if (i % p) < p // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    h = _project(h, target_sum)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t_start = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(100, 0.5)

    n_par = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _compute_c5(best_h, n_par)
    best_n = n_par

    if n_par % 2 != 0:
        best_h = _project(best_h, n_par / 2.0)
        best_c5 = _compute_c5(best_h, n_par)

    schedule = [
        (100, 0.08),
        (200, 0.12),
        (400, 0.20),
        (800, 0.28),
        (1600, 0.18),
        (3200, 0.14),
    ]
    total_frac = sum(f for _, f in schedule)

    current_h = h_parent
    for n_target, frac in schedule:
        remaining = budget_s - (time.time() - t_start)
        if remaining < 15:
            break
        alloc = (frac / total_frac) * budget_s * 0.88
        alloc = min(alloc, remaining - 10)
        if alloc < 5:
            break

        h_curr = _upsample(current_h, n_target)
        target_sum = n_target / 2.0
        h_curr = _project(h_curr, target_sum)

        h_opt, c5_opt = _optimize(h_curr, n_target, target_sum, alloc)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_target
            current_h = h_opt
        else:
            current_h = h_opt

    # Final polish
    remaining = budget_s - (time.time() - t_start)
    if remaining > 20:
        target_sum = len(best_h) / 2.0
        h_pol, c5_pol = _optimize(best_h, len(best_h), target_sum, min(remaining - 5, 100))
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Ensure constraints are met
    target_sum = len(best_h) / 2.0
    best_h = _project(best_h, target_sum)
    best_c5 = _compute_c5(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
