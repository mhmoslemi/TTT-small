                cur_c5 = c5_val(h)

            for it in range(5000):
                if elapsed() > budget_s * 0.91:
                    break
                grad = subgrad_exact(h)
                gn = np.linalg.norm(grad)
                if gn < 1e-13:
                    break
                step = 0.3 / (gn + 1e-10)
                improved = False
                for _ in range(20):
                    h_new = project_hyp(h - step * grad, n_pts)
                    c5_new = c5_val(h_new)
                    if c5_new < cur_c5 - 1e-13:
                        h = h_new
                        cur_c5 = c5_new
                        if c5_new < best_c5:
                            best_c5, best_h, best_n = c5_new, h.copy(), n_pts
                        improved = True
                        break
                    step *= 0.5
                if not improved:
                    h = h + rng.normal(0.0, 0.01, n_pts)
                    h = project_hyp(h, n_pts)
                    cur_c5 = c5_val(h)

        # Phase 3: Local coordinate search
        if elapsed() < budget_s * 0.93:
            for _ in range(int(40.0 * n_pts / 100)):
                if elapsed() > budget_s * 0.95:
                    break
                j = rng.randint(n_pts)
                for delta in (-0.04, -0.02, -0.01, 0.01, 0.02, 0.04):
                    h_try = h.copy()
                    h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                    h_try = project_hyp(h_try, n_pts)
                    c5_try = c5_val(h_try)
                    if c5_try < cur_c5 - 1e-13:
                        h = h_try
                        cur_c5 = c5_try
                        if c5_try < best_c5:
                            best_c5, best_h, best_n = c5_try, h.copy(), n_pts
                        break

    # Also try a fresh random start at a good resolution
    if elapsed() < budget_s * 0.60:
        n_try = max(200, 2 * n0)
        h_rand = project_hyp(rng.rand(n_try), n_try)
        cur_c5 = c5_val(h_rand)
        if cur_c5 < best_c5:
            best_c5, best_h, best_n = cur_c5, h_rand.copy(), n_try

        # Optimize the random start
        for alpha in [50, 200, 1000]:
            if elapsed() > budget_s * 0.95:
                break
            for it in range(3000):
                if elapsed() > budget_s * 0.95:
                    break
                _, grad = smooth_obj_and_grad(h_rand, alpha)
                gn = np.linalg.norm(grad)
                if gn < 1e-13:
                    break
                step = 0.3 / (gn + 1e-10)
                improved = False
                for _ in range(15):
                    h_new = project_hyp(h_rand - step * grad, n_try)
                    c5_new = c5_val(h_new)
                    if c5_new < cur_c5 - 1e-12:
                        h_rand = h_new
                        cur_c5 = c5_new
                        if c5_new < best_c5:
                            best_c5, best_h, best_n = c5_new, h_rand.copy(), n_try
                        improved = True
                        break
                    step *= 0.5
                if not improved:
                    h_rand = h_rand + rng.normal(0.0, 0.02, n_try)
                    h_rand = project_hyp(h_rand, n_try)
                    cur_c5 = c5_val(h_rand)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_hyp(best_h, best_n)
    best_c5 = c5_val(best_h)

    return best_h, best_c5, best_n
