```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_overlaps(h, n):
    """Compute overlap at all lags."""
    fh = 1.0 - h
    overlaps = np.zeros(2 * n - 1)
    for d in range(-(n-1), n):
        idx = d + n - 1
        i_start = max(0, -d)
        i_end = min(n, n - d)
        if i_end > i_start:
            overlaps[idx] = np.dot(h[i_start:i_end], fh[i_start+d:i_end+d])
    return overlaps

def evaluate_c5(h, n):
    """Evaluate C5 for given h."""
    dx = 2.0 / n
    overlaps = compute_overlaps(h, n)
    return np.max(overlaps) * dx

def project_h(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}."""
    h = np.clip(h, 0, 1)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

def gradient_at_lag(h, d, n):
    """Gradient of f_d with respect to h."""
    grad = np.zeros(n)
    i_start = max(0, -d)
    i_end = min(n, n - d)
    for i in range(i_start, i_end):
        grad[i] += (1.0 - h[i+d])
        grad[i+d] -= h[i]
    return grad

def subgradient_optimize(h, n, max_iters, alpha0=0.1, alpha_min=0.001):
    """Subgradient descent on max overlap."""
    h = h.copy().astype(float)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n)
    
    for it in range(max_iters):
        overlaps = compute_overlaps(h, n)
        max_idx = np.argmax(overlaps)
        d = max_idx - (n - 1)
        
        # Gradient of worst lag
        grad = gradient_at_lag(h, d, n)
        
        # Also include near-max for smoothing
        threshold = overlaps[max_idx] * 0.95
        near_max = np.where(overlaps >= threshold)[0]
        if len(near_max) > 1:
            grad = np.zeros(n)
            for idx in near_max:
                dd = idx - (n - 1)
                grad += gradient_at_lag(h, dd, n)
            grad /= len(near_max)
        
        # Adaptive step size
        alpha = alpha0 / (1 + it * alpha0 * 0.01)
        h = h - alpha * grad
        h = project_h(h, target)
        
        c5 = evaluate_c5(h, n)
```

```python
        if i > 0 and i % 300 == 0:
            a *= 0.96
        ov = _ov(h, n)
        pr = i / N
        ep = 2.0 * max(0.05, 1.0 - pr) ** 1.2 + 0.02
        shifted = ov / ep - np.max(ov) / ep
        w = np.exp(shifted)
        s = w.sum()
        if s > 1e-30:
            w /= s
        else:
            w = np.ones(nov) / nov
        g = np.zeros(n)
        th = 1e-3
        act = np.where(w > th)[0]
        for ix in act:
            g += w[ix] * _gd(h, ix - (n - 1), n)
        if np.max(np.abs(g)) > 1e-14:
            h = _pj(h - a * g, tgt)
        if i % 3000 == 2999 and pr < 0.7:
            h = _pj(h + rs.randn(n) * 0.02, tgt)
        c = _c5(h, n)
        if c < bc:
            bc, bh = c, h.copy()
    return bh, bc

def _local_refine(h, n, tl):
    h = _pj(h.copy(), n / 2.0)
    bc = _c5(h, n)
    bh = h.copy()
    t0 = time.time()
    improved = True
    while improved and time.time() - t0 < tl:
        improved = False
        for j in range(n):
            if time.time() - t0 > tl:
                break
            old = h[j]
            for val in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(val - old) < 1e-12:
                    continue
                h[j] = val
                h = _pj(h, n / 2.0)
                c = _c5(h, n)
                if c < bc - 1e-12:
                    bc = c
                    bh = h.copy()
                    improved = True
                else:
                    h[j] = old
                    h = _pj(h, n / 2.0)
    return bh, bc

def _make_inits(n, par, seed):
    rs = np.random.RandomState(seed)
    inits = []
    if par is not None:
        inits.append(_up(par, n))
    h_alt = np.zeros(n)
    h_alt[::2] = 1.0
    inits.append(_pj(h_alt, n / 2.0))
    h_rand = _pj(rs.rand(n) * 0.7 + 0.15, n / 2.0)
    inits.append(h_rand)
    if n >= 20:
        h_four = np.zeros(n)
        h_four += 0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n) / n)
        h_four += 0.1 * np.cos(4 * np.pi * np.arange(n) / n)
        inits.append(_pj(h_four, n / 2.0))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    ts = time.time()
    rs = np.random.RandomState(seed)

    par = None
    try:
        if initial_h_values is not None:
            par = np.array(initial_h_values, dtype=float)
    except Exception:
        par = None

    best_h = None
    best_c = float('inf')
    best_n = 45

    if par is not None:
        n_par = len(par)
        res_list = sorted(set([n_par, 90, 100, 150, 200, 300]))
    else:
        res_list = [45, 90, 100, 150, 200]

    for n in res_list:
        if time.time() - ts > budget_s - 70:
            break
        if n < 10 or n > 600:
            continue
        elapsed = time.time() - ts
        remaining = budget_s - elapsed - 50
        if remaining < 40:
            break
        n_remaining_res = len([r for r in res_list if r >= n])
        alloc = max(40, remaining / n_remaining_res)

        inits = _make_inits(n, par, seed)

        for ii, h_i in enumerate(inits):
            if time.time() - ts > budget_s - 50:
                break
            tl_i = max(20, alloc / len(inits) - 5)
            h_o, c_o = _opt(h_i, n, tl_i, seed + ii * 7919)
            if c_o < best_c:
                best_c = c_o
                best_h = h_o.copy()
                best_n = n

        if best_n != n:
            continue
        if time.time() - ts > budget_s - 50:
            break
        tl_loc = min(80, (budget_s - (time.time() - ts)) * 0.25)
        if tl_loc > 10:
            h_l, c_l = _local_refine(best_h, n, tl_loc)
            if c_l < best_c:
                best_c = c_l
                best_h = h_l.copy()

        if time.time() - ts > budget_s - 50:
            break
        tl2 = min(120, (budget_s - (time.time() - ts)) * 0.3)
        if tl2 > 20:
            h_o2, c_o2 = _opt(best_h, n, tl2, seed + 9999)
            if c_o2 < best_c:
                best_c = c_o2
                best_h = h_o2.copy()

    if best_h is None:
        n = 45
        best_h = _pj(np.ones(n) * 0.5, n / 2.0)
        best_c = _c5(best_h, n)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    target = best_n / 2.0
    if abs(s - target) > 1e-10:
        best_h = _pj(best_h, target)
    best_c = _c5(best_h, best_n)

    if 'evaluate_erdos_solution' in dir():
        try:
            eval_c5 = evaluate_erdos_solution(best_h.tolist(), best_n)
            if eval_c5 is not None:
                best_c = float(eval_c5)
        except Exception:
            pass

    return best_h, best_c, best_n
```
