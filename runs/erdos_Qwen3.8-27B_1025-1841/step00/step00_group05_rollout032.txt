import numpy as np
import time

def project_sum(h, target):
    for _ in range(1000):
        s = h.sum()
        if s < 1e-15:
            return np.full(len(h), target / len(h))
        h = h * (target / s)
        h = np.clip(h, 0, 1)
        if abs(h.sum() - target) < 1e-12:
            return h
    return h

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def compute_grad_for_lag(h, k):
    """Compute gradient of corr[k] with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] += (1.0 - h)
            grad[:] -= h
        else:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
    else:
        ka = -k
        if ka == 0:
            grad[:] += (1.0 - h)
            grad[:] -= h
        else:
            grad[:n-ka] += (1.0 - h[ka:])
            grad[ka:] -= h[:n-ka]
    return grad

import numpy as np
import time


def project_sum(h, target):
    for _ in range(500):
        s = h.sum()
        if s < 1e-15:
            return np.full(len(h), target / len(h))
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-12:
            return h
    return h


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr


def grad_lag(h, k):
    n = len(h)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if k > 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        ka = -k
        g[:n - ka] += 1.0 - h[ka:]
        g[ka:] -= h[:n - ka]
    return g


def local_search(h, dx, target, best_c5, best_h, t_start, time_limit, rng):

    no_improve = 0
