def compute_max_corr(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))


def compute_c5(h, n):
    return compute_max_corr(h) * (2.0 / n)


def fix_count_binary(h, n, rng):
    """Ensure h is binary with exactly n//2 ones."""
    h = (h > 0.5).astype(np.float64)
    cnt = int(np.sum(h))
    target = n // 2
    if cnt > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, size=cnt - target, replace=False)] = 0
    elif cnt < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, size=target - cnt, replace=False)] = 1
    return h
