import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def grad_at_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    shift = lag - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        s = shift
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s = -shift
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    return grad * dx


def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def upsample_to(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def gradient_descent(h, target_sum, max_iters, lr_init, deadline):
    n = len(h)
    momentum = np.zeros(n)
    beta = 0.8
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        c = np.correlate(h, 1.0 - h, mode="full")
        c5_val = float(np.max(c) * (2.0 / n))

        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()

        max_idx = np.argmax(c)
        grad = grad_at_lag(h, max_idx)

        c_temp = c.copy()
        c_temp[max_idx] = -np.inf
        second_idx = np.argmax(c_temp)
        grad += 0.3 * grad_at_lag(h, second_idx)
        grad *= 1.3

        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - lr * momentum
        h_new = project_sum(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new <= c5_val + 1e-10:
            h = h_new
            lr = min(lr * 1.01, 1.0)
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if lr < 1e-7 or stall > 300:
                break

    return best_h, best_c5

    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dets = [0.3, 0.15, 0.05, 0.02, -0.02, -0.05, -0.15, -0.3]

    for it in range(max_iters):
        if time.time() > deadline:
            break

        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)

        h_new = h.copy()
        if rng.random() < 0.7:
            d = dets[rng.randint(len(dets))]
            h_new[i] += d
            h_new[j] -= d
