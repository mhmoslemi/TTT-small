def _compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _compute_shifts(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")

    lo, hi = -1.0, 1.0

        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > S:

        else:

import numpy as np
import time


def _c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _shifts(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def _proj(h, S):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def _gshift(h, n, lag):
    g = np.zeros(n)
    if 0 <= lag < n:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    elif -n < lag < 0:
        l = -lag
        g[:n - l] -= h[l:]
        g[l:] += (1.0 - h[:n - l])
    return g


def _smooth_g(h, n, sh, k, beta):
    idx = np.argsort(sh)[-k:]
    s = sh[idx]
    w = np.exp(beta * (s - s.max()))
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        g += w[i] * _gshift(h, n, int(idx[i]) - (n - 1))
    return g


def _optimize(h, n, S, rng, t_end, step_init):
