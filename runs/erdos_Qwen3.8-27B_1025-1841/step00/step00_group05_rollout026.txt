from scipy.optimize import minimize
from scipy.optimize import differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def objective_lse(h, beta=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Smooth max via log-sum-exp
    max_c = np.max(corr)
    shifted = corr - max_c
    return max_c + np.log(np.sum(np.exp(beta * shifted))) / beta
