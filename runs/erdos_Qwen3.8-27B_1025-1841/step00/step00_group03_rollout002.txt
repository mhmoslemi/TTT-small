import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        idx = np.where(mask)[0]
        if len(idx) > 0:
            adj = diff / len(idx)
            x[idx] = np.clip(x[idx] + adj, 0.0, 1.0)
    return x

def nm_objective(x, n):
    return compute_c5_vals(project_feasible(x, n))

def do_nm(x0, n, maxiter, fatol, xatol):

        nm_objective, x0, args=(n,),
        method='Nelder-Mead',
        options={'maxiter': maxiter, 'xatol': xatol, 'fatol': fatol, 'adaptive': True}
    )
    return res.x
