import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    w = e / np.sum(e)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        lag = j - (n - 1)
        if lag == 0:
            grad += wj * (1 - 2 * h)
        elif lag > 0:
            grad[:n - lag] += wj * (1 - h[lag:])
            grad[lag:] -= wj * h[:n - lag]
        else:
            l = -lag
            grad[l:] += wj * (1 - h[:n - l])
            grad[:n - l] -= wj * h[l:]
    
    return grad

    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}"""

        if not np.any(mask):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    w = e / np.sum(e)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-13:
            continue
        lag = j - (n - 1)
        if lag == 0:
            grad += wj * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wj * (1.0 - h[lag:])
            grad[lag:] -= wj * h[:n - lag]
        else:
            l = -lag
            grad[l:] += wj * (1.0 - h[:n - l])
            grad[:n - l] -= wj * h[l:]
    return grad

def sum_con_fun(h):
    return np.sum(h) - len(h) * 0.5

def sum_con_jac(h):
    return np.full(len(h), 1.0 / len(h))

def project(h, target):
    h = np.array(h, dtype=float)
    for _ in range(500):
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.astype(float).sum()
        if cnt == 0:
            break
        h[mask] -= diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def make_initial(n, ref):
    if ref is not None and len(ref) > 0:
        x_old = (np.arange(len(ref)) + 0.5) / len(ref) * 2.0
        x_new = (np.arange(n) + 0.5) / n * 2.0
        h = np.interp(x_new, x_old, np.asarray(ref, dtype=float))
    else:
        h = np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    return project(h, n * 0.5)

def slide_optimize(h0, alpha, n, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [dict(type='eq', fun=sum_con_fun, jac=sum_con_jac)]
    res = minimize(smooth_obj, h0, args=(alpha,), jac=smooth_grad,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h = np.clip(res.x, 0.0, 1.0)
    return project(h, n * 0.5)

def local_search(h_best, c5_best, n, rng, max_iters, time_limit):
    t_end = time.time() + max(time_limit, 5)
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        h_try = h_best + rng.normal(0, 0.02, n)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project(h_try, n * 0.5)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best - 1e-12:
            return h_try, c5_try
    return h_best, c5_best

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    init_vals = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_vals = np.asarray(initial_h_values, dtype=float)

    configs = [
        (200, 15, 300),
        (300, 20, 250),
        (400, 25, 200),
        (250, 50, 200),
        (350, 80, 150),
    ]

    for n, alpha, maxiter in configs:
        if time.time() - t0 > budget_s - 40:
            break
        h0 = make_initial(n, init_vals)
        c5 = compute_c5(h0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()
            best_n = n

        h_opt = slide_optimize(h0, alpha, n, maxiter)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n

        for a2 in [int(alpha * 3), int(alpha * 6)]:
            if time.time() - t0 > budget_s - 40:
                break
            h_opt2 = slide_optimize(h_opt, a2, n, max(80, maxiter // 3))
            c5_2 = compute_c5(h_opt2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_opt2.copy()
                best_n = n

    # Try binary-like starts
    if time.time() - t0 < budget_s - 60:
        for n_try in [200, 400]:
            for _ in range(3):
                if time.time() - t0 > budget_s - 30:
                    break
                h_bin = np.zeros(n_try)
                h_bin[rng.choice(n_try, n_try // 2, replace=False)] = 1.0
                h_opt = slide_optimize(h_bin, 20, n_try, 150)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n_try

    # Local search refinement
    if best_h is not None and time.time() - t0 < budget_s - 10:
        n = len(best_h)
        residual = budget_s - 10 - (time.time() - t0)
        best_h, best_c5 = local_search(best_h, best_c5, n, rng, 2000, residual)

    if best_h is None:
        n = 200
        best_h = make_initial(n, init_vals)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, len(best_h)
