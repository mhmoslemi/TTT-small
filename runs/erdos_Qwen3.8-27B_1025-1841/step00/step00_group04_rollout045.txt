from scipy.optimize import minimize
import time


def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def constraint_eq(h):

def project_feasible(h, n):
    for _ in range(1000):

        diff = np.sum(hc) - n / 2.0
        if abs(diff) < 1e-14:
            return hc
        h = hc - diff / n
    return np.clip(h, 0.0, 1.0)
