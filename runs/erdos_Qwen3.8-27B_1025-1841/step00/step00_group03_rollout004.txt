        n = 100
        h0 = np.zeros(n)
        h0[::2] = 1.0
    else:
        n = len(h0)

    h0 = repair_h(h0, n)
    best_c5, _ = eval_c5(h0)
    best_h = h0.copy()

    def rem():
        return budget_s - (time.time() - t0)

    # Main SA at initial resolution
    h = best_h.copy()
    cur_c5, cur_c = eval_c5(h)
    T = 0.008
    min_T = 1e-9
    alpha = 0.99997
    step = 0.03
    acc_buf = []
    W = 300
    it = 0

    while rem() > 50:
        it += 1
        if it % W == 0 and len(acc_buf) >= W // 2:
            rate = np.mean(acc_buf)
            if rate < 0.08:
                step = min(0.12, step * 1.3)
            elif rate < 0.2:
                step = min(0.12, step * 1.1)
            elif rate > 0.8:
                step = max(0.003, step * 0.7)
            elif rate > 0.65:
                step = max(0.003, step * 0.85)
            acc_buf = []
        # Choose move type
        r = rng.random()
        if r < 0.55:
            # Random perturbation
            h_new = h.copy()
            k_p = rng.randint(1, max(2, n // 12))
            idx_p = rng.choice(n, k_p, replace=False)
            h_new[idx_p] += rng.normal(0, step, k_p)
            h_new = repair_h(h_new, n)
        elif r < 0.80:
            # Gradient step
            h_new = grad_step(h, n, cur_c, eta=step * 0.6)
        else:
            # Larger random perturbation
            h_new = h.copy()
            k_p = rng.randint(n // 6, n // 3)
            idx_p = rng.choice(n, k_p, replace=False)
            h_new[idx_p] += rng.normal(0, step * 0.4, k_p)
            h_new = repair_h(h_new, n)

        c5_new, c_new = eval_c5(h_new)
        delta = c5_new - cur_c5
        acc = False
        if delta < 0 or (T > min_T and rng.random() < np.exp(-delta / T)):
            h = h_new
            cur_c5 = c5_new
            cur_c = c_new
            acc = True
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        acc_buf.append(1.0 if acc else 0.0)
        T = max(min_T, T * alpha)
        if it % 8000 == 0 and T < 0.001:
            T = 0.003

    # Resolution refinement
    for mult in [2, 3, 4]:
        n2 = n * mult
        if rem() < 50:
            break
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, np.concatenate([best_h, best_h[:1]]))
        h2 = repair_h(h2, n2)
        c5_2, c2 = eval_c5(h2)

        h_cur = h2.copy()
        c5_cur = c5_2
        c_arr = c2
        T2 = 0.005
        step2 = 0.02

        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

        phase_budget = min(rem() - 20, 350)
        if phase_budget < 20:
            continue
        phase_end = time.time() + phase_budget
        it2 = 0
        acc_buf2 = []
        while time.time() < phase_end:
            it2 += 1
            if it2 % W == 0 and len(acc_buf2) >= W // 2:
                rate2 = np.mean(acc_buf2)
                if rate2 < 0.1:
                    step2 = min(0.1, step2 * 1.25)
                elif rate2 > 0.75:
                    step2 = max(0.002, step2 * 0.75)
                acc_buf2 = []
            r2 = rng.random()
            if r2 < 0.55:
                h_new = h_cur.copy()
                k_p = rng.randint(1, max(2, n2 // 12))
                idx_p = rng.choice(n2, k_p, replace=False)
                h_new[idx_p] += rng.normal(0, step2, k_p)
                h_new = repair_h(h_new, n2)
            elif r2 < 0.80:
                h_new = grad_step(h_cur, n2, c_arr, eta=step2 * 0.6)
            else:
                h_new = h_cur.copy()
                k_p = rng.randint(n2 // 6, n2 // 3)
                idx_p = rng.choice(n2, k_p, replace=False)
                h_new[idx_p] += rng.normal(0, step2 * 0.4, k_p)
                h_new = repair_h(h_new, n2)

            c5_new, c_new = eval_c5(h_new)
            delta = c5_new - c5_cur
            acc2 = False
            if delta < 0 or (T2 > min_T and rng.random() < np.exp(-delta / T2)):
                h_cur = h_new
                c5_cur = c5_new
                c_arr = c_new
                acc2 = True
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
            acc_buf2.append(1.0 if acc2 else 0.0)
            T2 = max(min_T, T2 * alpha)
            if it2 % 8000 == 0 and T2 < 0.001:
                T2 = 0.002

    return best_h, float(best_c5), len(best_h)
