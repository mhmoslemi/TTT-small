import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def smooth_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.sum(corr ** p) ** (1.0 / p)

def sum_eq_fun(h, target):
    return np.sum(h) - target

def sum_eq_jac(h, target):
    return np.ones(len(h))

def enforce_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            if not mask.any():
                break
            total_room = room[mask].sum()
            if total_room < 1e-15:
                break
            frac = min(1.0, diff / total_room)
            h[mask] += room[mask] * frac
        else:
            mask = h > 1e-10
            if not mask.any():
                break
            total_mass = h[mask].sum()
            if total_mass < 1e-15:
                break
            frac = min(1.0, -diff / total_mass)
            h[mask] -= h[mask] * frac
        h = np.clip(h, 0.0, 1.0)
    return h

def do_slsqp(h0, p, maxiter, ftol):
    n = len(h0)
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (target,)},)
    try:
        res = minimize(smooth_obj, h0, args=(p,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return enforce_feasible(res.x, n)
    except Exception:
        return h0

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])

def round_to_binary(h, n):
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    s = int(np.sum(h_bin))
    if s > target:
        excess = s - target
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h[ones])]
        for idx in order[:excess]:
            h_bin[idx] = 0.0
    elif s < target:
        deficit = target - s
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for idx in order[:-deficit]:
            h_bin[idx] = 1.0
    return h_bin

def local_perturb_search(h, n, n_iter, scale):
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.RandomState(42)
    for _ in range(n_iter):
        h_try = best.copy()
        for j in range(n):
            delta = rng.uniform(-scale, scale)
            h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
        h_try = enforce_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_val:
            best = h_try
            best_val = c5_try
    return best, best_val
