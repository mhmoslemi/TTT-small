import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_soft(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    logsum = m + np.log(np.sum(np.exp(corr - m)))
    return (logsum / len(corr)) * dx

def constraint_fn(x):
    return x.sum() - len(x) / 2

def make_feasible(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            step = diff / mask.sum()
            step = min(step, (1.0 - h[mask]).min())
        else:
            mask = h > 0.0
            if not mask.any():
                break
            step = diff / mask.sum()
            step = max(step, -h[mask].min())
        h[mask] += step
    return np.clip(h, 0.0, 1.0)

def interpolate_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0, 2, old_n)
    x_new = np.linspace(0, 2, new_n)
    return np.interp(x_new, x_old, h)

def do_optimize(h0, max_iter, beta):
    n = len(h0)
    constraints = [{'type': 'eq', 'fun': constraint_fn}]
    bounds = [(0.0, 1.0)] * n
    h0 = make_feasible(h0, n)
    try:
        res = minimize(obj_soft, h0, args=(beta,), method='SLSQP',
                       constraints=constraints, bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        h_opt = make_feasible(res.x, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return make_feasible(h0, n), compute_c5(make_feasible(h0, n))

def local_search(h, n, max_rounds=8, deltas=None):
    h = h.copy()
    current_c5 = compute_c5(h)
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, -0.002, -0.005, -0.01, -0.02]
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                if h[i] + d < -0.01 or h[i] + d > 1.01:
                    continue
                h_new = h.copy()
                h_new[i] += d
                h_new = make_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < current_c5 - 1e-12:
                    h = h_new
                    current_c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, current_c5

def generate_inits(n, rng, best_h):
    inits = []
    if best_h is not None:
        inits.append(make_feasible(interpolate_h(best_h, n), n))
        inits.append(make_feasible(interpolate_h(best_h, n) + 0.05 * rng.randn(n), n))
    for f in range(1, 10):
        phase = rng.rand() * 2 * np.pi
        amp = 0.3 + 0.1 * rng.rand()
        hs = 0.5 + amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
        inits.append(make_feasible(hs, n))
    hb = (rng.rand(n) > 0.5).astype(float)
    inits.append(make_feasible(hb, n))
    hr = 0.3 + 0.4 * rng.rand(n)
    inits.append(make_feasible(hr, n))
    hblk = np.zeros(n)
    bs = max(2, n // 8)
    for i in range(n):
        if (i // bs) % 2 == 0:
            hblk[i] = 1.0
    inits.append(make_feasible(hblk, n))
    # Triangular
    ht = 1.0 - 2.0 * np.abs(np.arange(n) / n - 0.5)
    inits.append(make_feasible(ht, n))
    return inits

    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')

    try:

    resolutions = [48, 64, 80, 100, 128, 160, 200, 256]

            break
        inits = generate_inits(n, rng, best_h)
        for h0 in inits:
            if time.time() - t0 > budget_s * 0.7:
                break
            h_opt, c5 = do_optimize(h0, 250, 80.0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                # Quick local search
                h_opt, c5 = local_search(h_opt, len(h_opt), max_rounds=3)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    # Multi-beta refinement
    if best_h is not None:
        for beta in [150.0, 300.0, 600.0, 1200.0]:
            if time.time() - t0 > budget_s * 0.85:
                break
            h_opt, c5 = do_optimize(best_h, 400, beta)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

        # Local search
        if time.time() - t0 < budget_s * 0.9:
            best_h, best_c5 = local_search(best_h, len(best_h), max_rounds=10)

        # Try higher resolution
        for n2 in [len(best_h) + 32, len(best_h) + 64]:
            if time.time() - t0 > budget_s * 0.93:
                break
            h_init = make_feasible(interpolate_h(best_h, n2), n2)
            h_opt, c5 = do_optimize(h_init, 300, 200.0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    if best_h is None:
        best_h = np.full(48, 0.5)
        best_c5 = compute_c5(best_h)

    n = len(best_h)
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
