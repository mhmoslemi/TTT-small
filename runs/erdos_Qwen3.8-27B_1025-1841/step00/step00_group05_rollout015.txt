        pert = rng.randn(N) * scale
        h_t = project_feasible(h + pert, target)
        c5_t = compute_c5(h_t, dx)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            h = h_t.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, N / 2.0)
    # Correct any tiny sum error
    err = N / 2.0 - np.sum(best_h)
    if abs(err) > 0:
        free = np.where((best_h > 0) & (best_h < 1))[0]
        if len(free) > 0:
            best_h[free] += err / len(free)
            best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, N
