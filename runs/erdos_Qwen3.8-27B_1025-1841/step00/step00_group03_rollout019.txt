def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h < 1.0 - 1e-10) if d > 0 else (h > 1e-10)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def evaluate_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)


def upsample(h, new_n):
