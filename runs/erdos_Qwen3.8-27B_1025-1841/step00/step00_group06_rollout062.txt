            if s >= 0:
                grad[:N - s] -= h[s:]
                grad[s:] += (1.0 - h[:N - s])
            else:
                tt = -s
                grad[:N - tt] += (1.0 - h[tt:])
                grad[tt:] -= h[:N - tt]
            grad *= dx
            h = h - lr * grad
            # Add small noise to escape plateaus
            if it % 2000 == 1999:
                h += rng.normal(0, 0.005, N)
            h = project_sum(h, target)
            if it % 200 == 199:
                c5n = compute_c5(h, N)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()

    # Phase 4: Local coordinate search
    if time.time() - t0 < budget_s * 0.97:
        h = best_h.copy()
        target = N / 2.0
        for it in range(40000):
            if time.time() - t0 > budget_s * 0.97:
                break
            i = int(rng.randint(N))
            j = int(rng.randint(N))
            if i == j:
                continue
            improved = False
            for delta in [0.08, 0.03, 0.01, -0.08, -0.03, -0.01]:
                trial = h.copy()
                trial[i] = min(1.0, max(0.0, trial[i] + delta))
                trial[j] = min(1.0, max(0.0, trial[j] - delta))
                trial = project_sum(trial, target)
                c5t = compute_c5(trial, N)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    best_h = trial.copy()
                    h = trial.copy()
                    improved = True
                    break
            if not improved and it % 500 == 499:
                h = best_h.copy()

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        hx, c5x = slsqp_refine(best_h, N, maxiter=200)
        hx = project_sum(hx, N / 2.0)
        c5x = compute_c5(hx, N)
        if c5x < best_c5:
            best_c5 = c5x
            best_h = hx.copy()

    best_h = project_sum(best_h, N / 2.0)
    best_c5 = compute_c5(best_h, N)
    return best_h, best_c5, N
