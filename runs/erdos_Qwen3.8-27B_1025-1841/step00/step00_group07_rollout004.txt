def local_search(h, max_iter=500, rng=None, time_limit=None, start_t=None):
    """Greedy local search on 0-1 vector with swap moves."""
    n = len(h)
    curr = h.copy()
    best_c5 = compute_c5(curr)
    
    if rng is None:
        rng = np.random.RandomState(0)

            break

        zeros = np.where(curr < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0
        best_swap = None
        
        # Try all swaps (or a random subset if too many)

            curr[i] = 1

            curr[i] = 0
            curr[j] = 1

    def update(h, c5, n):
        nonlocal best_c5, best_h, best_n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 1: Use parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
        pn = len(parent)
        pc5 = compute_c5(parent)
        update(parent, pc5, pn)

        # Binarize and local search
        h_01 = (parent > 0.5).astype(float)
        if abs(np.sum(h_01) - pn // 2) > 0.5:
            h_01 = np.zeros(pn)
            h_01[rng.choice(pn, pn // 2, replace=False)] = 1.0

        h_ls, c5_ls = local_search_01(h_01, rng, max_iter=500, time_limit=120, start_t=start_time)
        update(h_ls, c5_ls, pn)

        # SA from local search result
        h_sa, c5_sa = sa_search_01(h_ls, rng, max_iter=2000, T_init=0.4, T_min=0.005,
                                   time_limit=180, start_t=time.time())
        update(h_sa, c5_sa, pn)

        # More local search from SA result
        h_ls2, c5_ls2 = local_search_01(h_sa, rng, max_iter=500, time_limit=120, start_t=time.time())
        update(h_ls2, c5_ls2, pn)

        # SLSQP from best 0-1
        if elapsed() < budget_s * 0.5:
            h_opt, c5_opt = slsqp_optimize(best_h.astype(float), best_n, maxiter=800, ftol=1e-11)
            update(h_opt, c5_opt, best_n)

    # Phase 2: Try multiple random 0-1 initializations at various n
    n_list = [80, 120, 160, 240, 320, 480, 640]
    for n in n_list:
        if elapsed() > budget_s * 0.75:
            break
        n_trials = 4 if n <= 120 else 2
        for trial in range(n_trials):
            if elapsed() > budget_s * 0.8:
                break
            h_rand = np.zeros(n)
            h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            h_ls, c5_ls = local_search_01(h_rand, rng, max_iter=200,
                                          time_limit=min(60, budget_s * 0.05),
                                          start_t=time.time())
            update(h_ls, c5_ls, n)
            if elapsed() > budget_s * 0.7:
                break
            # SLSQP
            h_opt, c5_opt = slsqp_optimize(h_ls.astype(float), n, maxiter=300, ftol=1e-9)
            update(h_opt, c5_opt, n)

    # Phase 3: Final refinement of best solution
    if best_h is not None and elapsed() < budget_s * 0.9:
        h_opt, c5_opt = slsqp_optimize(best_h.astype(float), best_n, maxiter=2000, ftol=1e-12)
        update(h_opt, c5_opt, best_n)

    if best_h is None:
        n = 80
        best_h = np.zeros(n)
        best_h[rng.choice(n, n // 2, replace=False)] = 1.0
        best_n = n
        best_c5 = compute_c5(best_h)

    # Final constraint fix
    n = best_n
    diff = n / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-10:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
