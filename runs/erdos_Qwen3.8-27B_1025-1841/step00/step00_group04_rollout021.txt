    exp_shifted = np.exp(shifted)

    obj = log_sum * dx
    
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient computation
    # For each m (0 to 2n-2), lag = m - (n-1)
    # d(corr[m])/d(h[j]) = (1-h[j-lag])*1[0<=j-lag<n] - h[j+lag]*1[0<=j+lag<n]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_obj_and_grad(h, c=150.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = c * (corr - max_corr)
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    log_sum = max_corr + np.log(exp_s.sum()) / c
    obj = log_sum * dx
    weights = exp_s / exp_s.sum()
    grad = np.zeros(n)
    n_c = 2 * n - 1
    for m in range(n_c):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            k = n - lag
            grad[lag:] += w * (1.0 - h[:k])
            grad[:k] -= w * h[lag:]
        else:
            ll = -lag
            k = n - ll
            grad[:k] += w * (1.0 - h[ll:])
            grad[ll:] -= w * h[:k]
    grad *= dx
    return obj, grad

def fix_sum(h, target):

            if cnt == 0:
