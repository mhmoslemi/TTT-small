def project_sum(h, target_sum):
    # Adjust h to have the correct sum while keeping in [0,1]
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase some values
        # Add to positions with room
        mask = h < 1.0
        if not np.any(mask):
            return h
        num_good = np.sum(mask)
        add = diff / num_good
        h[mask] = np.minimum(1.0, h[mask] + add)
        # Re-normalize
        new_sum = np.sum(h)
        if new_sum > target_sum:
            excess = new_sum - target_sum
            still_mask = h > 0
            num_still = np.sum(still_mask)
            if num_still > 0:
                h[still_mask] -= excess / num_still
        h = np.clip(h, 0, 1)
    elif diff < 0:
        mask = h > 0.0
        if not np.any(mask):
            return h
        num_good = np.sum(mask)
        sub = -diff / num_good
        h[mask] = np.maximum(0.0, h[mask] - sub)
        new_sum = np.sum(h)

        h[idx] = 0

        h[idx] = 1

        ones = np.where(h == 1)[0]

def full_pass_local_search(h, n, dx, rng):
    """Do one full pass of best-improvement local search on binary h."""
    target = n // 2
    h = h.astype(float).copy()
    best_c5 = compute_c5(h, dx)
    improved = True
    while improved:

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

            h[i] = 0.0
            h[j] = 1.0

Since Δ(k) is non-zero only at a small number of lags, most of the max won't change. The max can only decrease at lags where Δ(k) < 0, and can only increase at lags where Δ(k) > 0.

Strategy: 
- Current max is M, achieved at lags L = {k : corr[k] = M}
- For a swap to improve, we need corr[k] + Δ(k) < M for all k in L, AND max of new corr < M
- If any k in L has Δ(k) ≥ 0, the max is still ≥ M (no improvement)
- If all k in L have Δ(k) < 0, we need to check the new max

This gives a fast rejection criterion!

def find_best_swap(h, corr, M, n, offset):
    """Find the best swap (i from 1→0, j from 0→1) that reduces the max.
    Returns (new_max, i, j) or (M, -1, -1) if no improvement."""
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    best_new_max = M
    best_ij = (-1, -1)
    
    for i in ones:
        for j in zeros:
            # Check if this swap can improve
            # Compute Δ at lags where corr is near max
            # First check: for all k where corr[k] == M, is Δ(k) < 0?
            
            # Affected lags: k where i+k valid or j+k valid or i-k valid or j-k valid
            # i+k valid: 0 ≤ i+k < n → -i ≤ k < n-i
            # j+k valid: 0 ≤ j+k < n → -j ≤ k < n-j
            # i-k valid: 0 ≤ i-k < n → i-n < k ≤ i
            # j-k valid: 0 ≤ j-k < n → j-n < k ≤ j
            
            # Union of these ranges
            k_lo = max(-i, -j, i-n+1, j-n+1, -(n-1))
            k_hi = min(n-i-1, n-j-1, i, j, n-1)
            
            can_improve = True
            new_max = 0.0
            for k in range(k_lo, k_hi + 1):
                dk = 0.0
                if 0 <= i+k < n:
                    dk += -(1.0 - h[i+k])
                if 0 <= j+k < n:
                    dk += (1.0 - h[j+k])
                if 0 <= i-k < n:
                    dk += h[i-k]
                if 0 <= j-k < n:
                    dk += -h[j-k]
                
                idx = k + offset
                if idx < 0 or idx >= 2*n-1:
                    continue
                new_val = corr[idx] + dk
                if new_val > best_new_max:
                    # This swap is worse
                    can_improve = False
                    break
                if new_val > new_max:
                    new_max = new_val
            
            if not can_improve:
                continue
            
            if new_max < best_new_max:
                best_new_max = new_max
                best_ij = (i, j)
    
    return best_new_max, best_ij

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(h, n, rng):
    """Convert to binary with exactly n//2 ones."""
    target = n // 2
    h = h.astype(float).copy()
    # Threshold
    h = (h > 0.5).astype(float)
    # Fix sum
    s = int(np.sum(h))
    while s > target:
        idx = rng.choice(np.where(h == 1.0)[0])
        h[idx] = 0.0
        s -= 1
    while s < target:
        idx = rng.choice(np.where(h == 0.0)[0])
        h[idx] = 1.0
        s += 1

            h[i] = 0.0

                    best_ij = (i, j)

            h[i] = 1.0

            break

        h[i] = 0.0
