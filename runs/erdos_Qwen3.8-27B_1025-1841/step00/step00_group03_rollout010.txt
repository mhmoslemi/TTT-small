        if blocks.sum() > tsum:
            blocks[blocks.sum() - tsum:] = 0
        elif blocks.sum() < tsum:
            idx0 = np.where(blocks == 0)[0]
            blocks[idx0[:tsum - blocks.sum()]] = 1
        starts.append(blocks.copy())

    phase_time = max(20, (budget_s * 0.15) // len(starts))
    for st in starts:
        if tl() < phase_time + 5:
            break
        hr, cr = binary_ls(st, phase_time, rng)
        if cr < best_c5:
            best_h, best_c5, best_n = hr, cr, n

    if tl() > 180:
        start = best_h if (np.max(best_h - best_h.round()) < 1e-6 and np.min(best_h - best_h.round()) > -1e-6) else hb
        start = np.round(start).clip(0, 1)
        hr, cr = anneal(start, min(180, tl() - 10), rng)
        if cr < best_c5:
            best_h, best_c5, best_n = hr, cr, n

    if tl() > 120:
        start = np.round(best_h).clip(0, 1)
        if abs(start.sum() - tsum) > 0.5:
            start = hb
        hr, cr = binary_ls(start, min(120, tl() - 10), rng)
        if cr < best_c5:
            best_h, best_c5, best_n = hr, cr, n

    while tl() > 50:
        hr = np.zeros(n)
        hr[rng.choice(n, tsum, replace=False)] = 1.0
        hr, cr = binary_ls(hr, min(35, tl() - 15), rng)
        if cr < best_c5:
            best_h, best_c5, best_n = hr, cr, n
        if tl() > 150:
            hr2, cr2 = anneal(hr, min(30, tl() - 20), rng)
            if cr2 < best_c5:
                best_h, best_c5, best_n = hr2, cr2, n

    if tl() > 100:
        n2, t2 = 140, 70
        h2 = np.zeros(n2)
        h2[rng.choice(n2, t2, replace=False)] = 1.0
        h2, c2 = binary_ls(h2, min(80, tl() - 15), rng)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    if tl() > 40:
        n3, t3 = 70, 35
        h3 = np.zeros(n3)
        h3[::2][:t3] = 1.0
        if h3.sum() != t3:
            h3 = np.zeros(n3)
            h3[rng.choice(n3, t3, replace=False)] = 1.0
        h3, c3 = binary_ls(h3, min(30, tl() - 5), rng)
        if c3 < best_c5:
            best_h, best_c5, best_n = h3, c3, n3

    if tl() > 20:
        b = np.round(best_h).clip(0, 1)
        if abs(b.sum() - best_n // 2) < 0.5:
            hr, cr = binary_ls(b, min(15, tl() - 3), rng)
            if cr < best_c5:
                best_h, best_c5 = hr, cr

    best_h = np.clip(best_h, 0, 1)
    fs = best_h.sum()
    if fs > 0:
        best_h = np.clip(best_h * (best_n / 2.0) / fs, 0, 1)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
