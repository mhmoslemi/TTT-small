from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    correl = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(correl) * dx)

def project_onto_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not free.any():
            # Can't project properly, just clip
            break
        h[free] -= diff / free.sum()
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = h.sum() - target
    if abs(diff) > 1e-10:
        free = (h > 0) & (h < 1)
        if free.any():
            h[free] -= diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def grad_active_lag(h, n, active_idx):
    """Gradient of c5 at the active lag"""
    dx = 2.0 / n
    lag = active_idx - (n - 1)
    grad = np.zeros(n)
    if lag > 0:
        for i in range(n - lag):
            j = i + lag
            grad[j] += (1.0 - h[i])
            grad[i] -= h[j]
    elif lag < 0:
        llag = -lag
        for i in range(n - llag):
            j = i + llag
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad * dx

def compute_c5(h):
