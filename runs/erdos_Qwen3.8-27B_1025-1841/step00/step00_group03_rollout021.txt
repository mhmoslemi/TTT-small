                    accepted = True
            if not accepted:
                h[i] -= delta
                h[j] += delta
                no_improve += 1

        if no_improve >= improve_patience:
            no_improve = 0
            n_kick = max(3, n // 6)
            idx = np.random.choice(n, n_kick, replace=False)
            noise = np.random.normal(0.0, 0.08, n_kick)
            h[idx] = np.clip(h[idx] + noise, 0.0, 1.0)
            h = repair_h(h)
            current_c5 = evaluate_c5(h)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    h = repair_h(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    n = len(h)

    # Phase 1: SA at initial resolution
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h1, c5_1 = sa_optimize(h, rem * 0.2, 0.018, 0.0001, 0.14, 0.008)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: 2x resolution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h2 = interpolate_h(best_h, 2 * n)
        c5_2_init = evaluate_c5(h2)
        if c5_2_init < best_c5:
            best_c5 = c5_2_init
            best_h = h2
        h2, c5_2 = sa_optimize(h2, rem * 0.2, 0.012, 0.00005, 0.09, 0.005)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: 4x resolution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        n4 = min(4 * n, 560)
        h4 = interpolate_h(best_h, n4)
        c5_4_init = evaluate_c5(h4)
        if c5_4_init < best_c5:
            best_c5 = c5_4_init
            best_h = h4
        h4, c5_4 = sa_optimize(h4, rem * 0.2, 0.007, 0.00002, 0.05, 0.003)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 4: Final fine-tuning at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        best_h, best_c5 = sa_optimize(best_h, rem * 0.95, 0.004, 0.000005, 0.025, 0.001)

    best_h = repair_h(best_h)
    final_c5 = evaluate_c5(best_h)

    return (best_h, final_c5, len(best_h))
