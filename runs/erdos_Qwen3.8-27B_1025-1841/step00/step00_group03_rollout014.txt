```python
def compute_gradients(corr_vec, h, n):
    """Compute gradients of each correlation w.r.t. each h coordinate.
    corr_vec has length 2n-1. Returns a 2D array of shape (2n-1, n)."""
    # This is expensive: (2n-1) * n entries
    # For n=70: 139*70 = 9730 entries
    # Let's compute analytically
    grads = np.zeros((2*n-1, n))
    for lag in range(-n+1, n):
        idx = lag + n - 1  # index in corr_vec
        # c_lag = sum_{valid i} h[i] * (1 - h[i+lag])
        # For lag >= 0: i ranges from 0 to n-1-lag
        # For lag < 0: let d = -lag, i ranges from d to n-1, term is h[i]*(1-h[i-d])
        if lag >= 0:
            for i in range(n - lag):
                grads[idx, i] += (1 - h[i+lag])       # derivative w.r.t. h[i]
                grads[idx, i+lag] += (-h[i])           # derivative w.r.t. h[i+lag]
        else:
            d = -lag
            for i in range(d, n):
                grads[idx, i] += (1 - h[i-d])         # derivative w.r.t. h[i]
                grads[idx, i-d] += (-h[i])            # derivative w.r.t. h[i-d]
    return grads
```

For n=70, this inner loop is O(n) per lag, O(n) lags, so O(n²) = 4900 total. Very fast.

        if abs(d) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += d / mask.sum()
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _active_gradients(h, n, corr, topm=5):
    """Compute sum of gradients over top-m active correlations."""
    threshold = corr.max() * (1.0 - 1e-4)
    active = np.where(corr >= threshold)[0]
    # Also include a few more for robustness
    sorted_idx = np.argsort(-corr)[:min(10, len(corr))]
    active = np.unique(np.concatenate([active, sorted_idx]))
    
    grad_sum = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)  # lag
        if k >= 0:
            # c = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[m]: (1-h[m-k]) if k<=m<=n-1, plus (-h[m+k]) if 0<=m<=n-1-k
            m_start1 = k
            m_end1 = n - 1
            if m_start1 <= m_end1:
                grad_sum[m_start1:m_end1+1] += (1.0 - h[0:m_end1+1-k])
            m_start2 = 0
            m_end2 = n - 1 - k
            if m_start2 <= m_end2:
                grad_sum[m_start2:m_end2+1] -= h[k:k+m_end2+1]
        else:
            d = -k
            # c = sum_{i=d}^{n-1} h[i-d]*(1-h[i])
            # d/dh[m]: (1-h[m+d]) if 0<=m<=n-1-d, plus (-h[m-d]) if d<=m<=n-1
            m_start1 = 0
            m_end1 = n - 1 - d
            if m_start1 <= m_end1:
                grad_sum[m_start1:m_end1+1] += (1.0 - h[d:d+m_end1+1])
            m_start2 = d
            m_end2 = n - 1
            if m_start2 <= m_end2:
                grad_sum[m_start2:m_end2+1] -= h[0:m_end2+1-d]
    return grad_sum
