import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_grad(h):
    """Subgradient of C5 w.r.t. h using the single maximizer."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    m = max_idx - (n - 1)
    
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        grad[m:n] += 1.0 - h[0:n-m]
        grad[0:n-m] += -h[m:n]
    else:
        mm = -m
        grad[0:n-mm] += 1.0 - h[mm:n]
        grad[mm:n] += -h[0:n-mm]
    
    grad *= dx
    return grad

    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

                    c5_cur = c5_new
                    lr = min(lr * 1.008, 0.015)
                    if c5_cur < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = c5_cur
                else:
                    lr = max(lr * 0.93, 0.0003)

            # Random restart local search
            for rs in range(300):
                if time_left() < 30:
                    break
                h_r = best_h.copy()
                if len(h_r) != n:
                    h_r = np.interp(np.linspace(0, 2, n, endpoint=False),
                                    np.linspace(0, 2, len(h_r), endpoint=False), h_r)
                h_r = project_h(h_r + np.random.randn(n) * 0.08, n)
                h_r_c5 = c5_value(h_r)
                lr2 = 0.002
                for it2 in range(300):
                    if time_left() < 30:
                        break
                    g2 = c5_subgrad(h_r)
                    h_r = project_h(h_r - lr2 * (g2 - g2.mean()) + np.random.randn(n) * 0.005, n)
                    c5_r = c5_value(h_r)
                    if c5_r < h_r_c5:
                        h_r_c5 = c5_r
                        lr2 = min(lr2 * 1.01, 0.01)
                        if c5_r < best_c5:
                            best_h = h_r.copy()
                            best_c5 = c5_r
                    else:
                        lr2 = max(lr2 * 0.92, 0.0002)

    n_final = len(best_h)
    best_h = project_h(best_h, n_final)
    best_c5 = c5_value(best_h)
    return (best_h, best_c5, n_final)
