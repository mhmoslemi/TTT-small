```python
def project_simplex_box(x, target_sum):
    """Project x onto {h : sum(h) = target_sum, 0 <= h <= 1}"""
    n = len(x)
    # Bisection on lambda
    # We want sum(clip(x + lambda, 0, 1)) = target_sum
    low = -x.max() - 1
    high = 1 - x.min() + 1
    
    for _ in range(100):
        mid = (low + high) / 2
        projected = np.clip(x + mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            high = mid
        else:
            low = mid
    
    return np.clip(x + (low + high) / 2, 0, 1)
```

```python
    N = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
```
