            improved = True
            while improved and not deadline():
                improved = False
                for j in range(0, n, max(1, n // 200)):
                    if deadline():
                        break
                    orig = best_h[j]
                    for val in [0.0, 0.5, 1.0]:
                        if abs(val - orig) < 1e-8:
                            continue
                        h_try = best_h.copy()
                        h_try[j] = val
                        h_try = repair(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-13:
                            best_h = h_try
                            best_c5 = c5_try
                            improved = True
                            break

    return best_h, float(best_c5), int(len(best_h))
