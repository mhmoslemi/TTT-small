This counts the number of "1 in h" at position i where the corresponding position j in (1-h) is also "1", i.e., h[j] = 0.

So C[k] counts "1s in h that are shifted to positions where h is 0" (or equivalently, "0s in h that are shifted to positions where h is 1" for the symmetric case).

To minimize the max, we want this count to be low for all shifts.

```python
        else:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_vals(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h, dx):
    return float(np.max(_c5_vals(h, dx)))

def _smax(x, T):
    if T < 1e-12:
        return float(np.max(x))
    xm = np.max(x)
    s = (x - xm) / T
    return float(xm + T * np.log(np.sum(np.exp(s))))

def _proj(h, ts):
    h = np.clip(h, 0.0, 1.0)
    d = ts - np.sum(h)
    if abs(d) > 1e-13:
        m = (h < 1.0 - 1e-9) if d > 0 else (h > 1e-9)
        if np.any(m):
            h[m] += d / np.sum(m)
            h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp(h0, n, dx, ts, T, mi=300):
    def f(x):
        return _smax(_c5_vals(x, dx), T)
    def g(x):
        return np.sum(x) - ts
    try:
        r = minimize(f, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': g}],
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _proj(np.clip(r.x, 0, 1), ts)
    except Exception:
        return h0

def _coord(h, n, dx, ts, passes=4):
    h = h.copy()
    for _ in range(passes):
        ch = False
        for i in range(n):
            ov = h[i]
            bv, bc = ov, _c5(h, dx)
            for v in [0.0, 1.0, 0.5, float(np.clip(ov - 0.15, 0, 1)), float(np.clip(ov + 0.15, 0, 1))]:
                if v == ov:
                    continue
                h[i] = v
                if abs(np.sum(h) - ts) > 0.5:
                    h[i] = ov
                    continue
                cv = _c5(h, dx)
                if cv < bc:
                    bc, bv = cv, v
            h[i] = bv
            if bv != ov:
                ch = True
        if not ch:
            break
    return _proj(h, ts)

def _bsa(h0, n, dx, ts, ni=12000, rs=0.15, re=0.002, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = (h0 > 0.5).astype(float)
    n1 = int(ts)
    if np.sum(h) != n1:
        h[:] = 0.0
        h[rng.permutation(n)[:n1]] = 1.0
    bh, bc, cc = h.copy(), _c5(h, dx), _c5(h, dx)
    for it in range(ni):
        T = rs * (re / rs) ** (it / float(ni))
        i, j = rng.randint(n), rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        nc = _c5(h, dx)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-15)):
            cc = nc
            if nc < bc:
                bc, bh = nc, h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return bh, bc

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h = np.ones(n) * 0.5

    dx = 2.0 / n
    ts = n / 2.0
    bh = h.copy()
    bc = _c5(bh, dx)

    # Phase 1: SLSQP annealing
    for T in [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - t0 > budget_s * 0.3:
            break
        hn = _slsqp(bh, n, dx, ts, T, mi=250)
        cn = _c5(hn, dx)
        if cn < bc:
            bh, bc = hn, cn

    # Phase 2: Coordinate descent
    if time.time() - t0 < budget_s * 0.45:
        bh = _coord(bh, n, dx, ts, passes=5)
        bc = _c5(bh, dx)

    # Phase 3: Binary SA
    if time.time() - t0 < budget_s * 0.65:
        hb, cb = _bsa(bh, n, dx, ts, ni=15000, rng=rng)
        if cb < bc:
            bh, bc = hb.copy(), cb
        hn = _slsqp(bh, n, dx, ts, 0.005, mi=200)
        cn = _c5(hn, dx)
        if cn < bc:
            bh, bc = hn, cn
        bh = _coord(bh, n, dx, ts, passes=3)
        bc = _c5(bh, dx)

    # Phase 4: Multi-start
    if time.time() - t0 < budget_s * 0.5:
        for ri in range(5):
            if time.time() - t0 > budget_s * 0.7:
                break
            hr = np.zeros(n)
            hr[rng.permutation(n)[:int(ts)]] = 1.0
            hr = _proj(hr, ts)
            hn = _slsqp(hr, n, dx, ts, 0.05, mi=200)
            cn = _c5(hn, dx)
            if cn < bc:
                bh, bc = hn, cn
                bh = _coord(bh, n, dx, ts, passes=3)
                bc = _c5(bh, dx)

    # Phase 5: Higher resolution
    if time.time() - t0 < budget_s * 0.35:
        n2 = min(2 * n, 280)
        h2 = np.zeros(n2)
        for i in range(n2):
            p = i * n / n2
            idx = int(p)
            fr = p - idx
            h2[i] = (1 - fr) * bh[idx] + fr * bh[min(idx + 1, n - 1)]
        dx2, ts2 = 2.0 / n2, n2 / 2.0
        h2 = _proj(h2, ts2)
        for T in [0.05, 0.02, 0.01]:
            if time.time() - t0 > budget_s * 0.85:
                break
            h2n = _slsqp(h2, n2, dx2, ts2, T, mi=200)
            c2 = _c5(h2n, dx2)

                bh, bc, h2 = h2n, c2, h2n
        if time.time() - t0 < budget_s * 0.9:
            h2 = _coord(h2, n2, dx2, ts2, passes=3)
            c2 = _c5(h2, dx2)
            if c2 < bc:
                bh, bc = h2, c2

    # Final validation
    fn = len(bh)
    fdx = 2.0 / fn
    bh = _proj(bh, fn / 2.0)
    fc = _c5(bh, fdx)
    return (bh, fc, fn)
```
