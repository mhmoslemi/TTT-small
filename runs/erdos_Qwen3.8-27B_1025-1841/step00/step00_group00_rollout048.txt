def compute_gradients(h):
    """Compute gradients for all lags. Returns array of shape (2*n-1, n)"""
    n = len(h)
    gr = np.zeros((2*n-1, n))
    for idx, lag in enumerate(range(-(n-1), n)):
        j = np.arange(n)
        valid1 = (j - lag >= 0) & (j - lag < n)
        gr[idx, valid1] += (1 - h[j[valid1] - lag])
        valid2 = (j + lag >= 0) & (j + lag < n)
        gr[idx, valid2] -= h[j[valid2] + lag]
    return gr

def solve_lp(h, active_lags, gradients, values):
    """Solve the LP to find optimal step"""
    n = len(h)
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        delta >= -h,
        delta <= 1 - h,
        cp.sum(delta) == 0
    ]
    
    # For active lags: values[k] + gradients[k] @ delta <= t
    for i in range(len(active_lags)):
        constraints.append(values[i] + gradients[i] @ delta <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if delta.value is not None:
                return delta.value
        except:
            continue
    
    # Try CBC as fallback
    try:
        prob.solve(solver=cp.CBC)
        if delta.value is not None:
            return delta.value
    except:
        pass
    
    return None
