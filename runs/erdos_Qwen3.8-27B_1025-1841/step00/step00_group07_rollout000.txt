def objective_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    _n_con = best_n
    h_start = fix_sum(best_h.copy(), best_n)
    try:
        res = minimize(compute_c5, h_start, method='SLSQP',
                       bounds=[(0, 1)] * best_n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
                       options={'maxiter': 500, 'ftol': 1e-12})
        ho = fix_sum(res.x, best_n)
        cv = compute_c5(ho)
        if cv < best_c5:
            best_c5, best_h, best_n = cv, ho.copy(), best_n
    except Exception:
        pass

    # Phase 2: multiple initializations at current n
    n = best_n
    inits = make_inits(n, seed)
    for ii in range(len(inits)):
        if elapsed() > budget_s - 180:
            break
        _n_con = n
        h_init = fix_sum(inits[ii], n)
        try:
            res = minimize(compute_c5, h_init, method='SLSQP',
                           bounds=[(0, 1)] * n,
                           constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
                           options={'maxiter': 400, 'ftol': 1e-12})
            ho = fix_sum(res.x, n)
            cv = compute_c5(ho)
            if cv < best_c5:
                best_c5, best_h, best_n = cv, ho.copy(), n
                print(f"  n={n} init{ii}: C5={best_c5:.6f} t={elapsed():.0f}s")
        except Exception:
            pass

    # Phase 3: perturb best and re-optimize
    for pp in range(6):
        if elapsed() > budget_s - 180:
            break
        _n_con = best_n
        pert = np.random.uniform(-0.15, 0.15, best_n)
        hp = np.clip(best_h + pert, 0, 1)
        hp = fix_sum(hp, best_n)
        try:
            res = minimize(compute_c5, hp, method='SLSQP',
                           bounds=[(0, 1)] * best_n,
                           constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
                           options={'maxiter': 300, 'ftol': 1e-12})
            ho = fix_sum(res.x, best_n)
            cv = compute_c5(ho)
            if cv < best_c5:
                best_c5, best_h, best_n = cv, ho.copy(), best_n
                print(f"  perturb{pp}: C5={best_c5:.6f} t={elapsed():.0f}s")
        except Exception:
            pass

    # Phase 4: higher resolution
    for n_hi in [120, 160, 200]:
        if elapsed() > budget_s - 150:
            break
        inits_hi = make_inits(n_hi, seed + n_hi)
        for ii in range(min(3, len(inits_hi))):
            if elapsed() > budget_s - 60:
                break
            _n_con = n_hi
            h_init = fix_sum(inits_hi[ii], n_hi)
            try:
                res = minimize(compute_c5, h_init, method='SLSQP',
                               bounds=[(0, 1)] * n_hi,
                               constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
                               options={'maxiter': 250, 'ftol': 1e-12})
                ho = fix_sum(res.x, n_hi)
                cv = compute_c5(ho)
                if cv < best_c5:
                    best_c5, best_h, best_n = cv, ho.copy(), n_hi
                    print(f"  n={n_hi} init{ii}: C5={best_c5:.6f} t={elapsed():.0f}s")
            except Exception:
                pass

    # Phase 5: final polish
    if elapsed() < budget_s - 20 and best_h is not None:
        _n_con = best_n
        try:
            res = minimize(compute_c5, best_h.copy(), method='SLSQP',
                           bounds=[(0, 1)] * best_n,
                           constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
                           options={'maxiter': 800, 'ftol': 1e-14})
            ho = fix_sum(res.x, best_n)
            cv = compute_c5(ho)
            if cv < best_c5:
                best_c5, best_h = cv, ho.copy()
        except Exception:
            pass

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    fn = len(best_h)
    best_h = fix_sum(best_h, fn)
    best_c5 = compute_c5(best_h)
    print(f"FINAL: n={fn}, C5={best_c5:.7f}, t={elapsed():.0f}s")
    return best_h, best_c5, fn
