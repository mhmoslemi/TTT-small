import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):

    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:

        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]

            t[i] = t[i // 2]

- If m+k = i, i.e., m = i-k: before h[i-k]*(1-h[i]) = h[i-k]*1 = h[i-k]. After: h[i-k]*(1-0) = h[i-k]. Change: 0. Wait no, after the swap h[i]=0, so (1-h[i])=1. Before, h[i]=1, so (1-h[i])=0. So the term h[i-k]*0 vs h[i-k]*1. Change: +h[i-k].
- If m = j: before h[j]*(1-h[j+k]) = 0. After: 1*(1-h[j+k]) = (1-h[j+k]). Change: +(1-h[j+k]).
- If m+k = j, i.e., m = j-k: before h[j-k]*(1-h[j]) = h[j-k]*1. After: h[j-k]*(1-1) = 0. Change: -h[j-k].

Plus similar changes for the negative lag (which is the same as swapping h and 1-h roles).

This is getting complex. For n=200, the full re-evaluation is only 20μs, so the overhead of the incremental computation in Python would likely negate the savings. Let's just do the full re-evaluation.

import numpy as np
import time


def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-8:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + diff / len(idx), 0.0, 1.0)
    return h


def make_thue_morse(n, flip=False):
    t = np.zeros(n)
    for i in range(1, n):
        if i % 2 == 0:
            t[i] = t[i // 2]
        else:
            t[i] = 1.0 - t[i // 2]
    if flip:
        t = 1.0 - t
    return repair_h(t, n)


def make_random_binary(n, rng):
    h = np.zeros(n)
    k = n // 2
    idx = rng.choice(n, k, replace=False)
    h[idx] = 1.0
    if n % 2 != 0:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros)] = 0.5
    return h


def make_golden(n, rng):
    phi_inv = 1.0 / ((1.0 + np.sqrt(5.0)) / 2.0)
    k = n // 2
    positions = np.floor(np.arange(k, dtype=float) * n * phi_inv).astype(int) % n
    positions = np.unique(positions)
    if len(positions) < k:
        remaining = np.setdiff1d(np.arange(n), positions)
        positions = np.concatenate([positions, rng.choice(remaining, k - len(positions), replace=False)])
    h = np.zeros(n)
    h[positions[:k]] = 1.0
    if n % 2 != 0:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros)] = 0.5
    return h


def make_balanced_blocks(n, rng):
    k = n // 2
    h = np.zeros(n)
    block_size = max(2, n // (2 * k))
    pos = 0
    placed = 0
    while placed < k:
        size = min(block_size, n - pos)
        h[pos:pos + size] = 1.0
        placed += size
        pos += 2 * size
        if pos >= n:
            break
    if n % 2 != 0:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[rng.choice(zeros)] = 0.5
    return h


def sa_optimize(h, n, rng, time_limit, t_start, iters=10**8):
    current_c5 = evaluate_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    T = 0.004
    no_improve = 0
    improve_window = 20000

    for it in range(iters):
        if time.time() - t_start > time_limit:
            break
        T = max(T * 0.9997, 1e-6)

        ones = np.where(h > 0.9)[0]
        zeros = np.where(h < 0.1)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        pi = ones[rng.integers(len(ones))]
        pj = zeros[rng.integers(len(zeros))]

        h[pi], h[pj] = h[pj], h[pi]
        new_c5 = evaluate_c5(h, n)
        delta = new_c5 - current_c5

        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if current_c5 < best_c5 - 1e-14:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[pi], h[pj] = h[pj], h[pi]
            no_improve += 1

        if no_improve > improve_window:
            no_improve = 0
            for _ in range(5):
                pi2 = ones[rng.integers(len(ones))]
                pj2 = zeros[rng.integers(len(zeros))]
                h[pi2], h[pj2] = h[pj2], h[pi2]
            h = repair_h(h, n)
            current_c5 = evaluate_c5(h, n)

    return best_h, best_c5


def deterministic_ls(h, n, time_limit, t_start):
    current_c5 = evaluate_c5(h, n)

    while True:
        if time.time() - t_start > time_limit:
            break
        improved = False
        ones = np.where(h > 0.9)[0]
        zeros = np.where(h < 0.1)[0]

        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                c5_try = evaluate_c5(h, n)
                if c5_try < current_c5 - 1e-13:
                    current_c5 = c5_try
                    improved = True
                    break
                h[i], h[j] = h[j], h[i]
            if improved:
                break
        if not improved:
            break

    return h, current_c5

    t_start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 100

        best_c5 = evaluate_c5(best_h, best_n)

    # Phase 1: Diverse exploration
    phase1_end = t_start + min(180, budget_s * 0.2)
    n_list = sorted(set([best_n, 96, 100, 128, 160, 200, 256]))

    for n_try in n_list:

                 make_golden(n_try, rng), make_random_binary(n_try, rng),
                 make_random_binary(n_try, rng), make_balanced_blocks(n_try, rng)]
        for hc in inits:
            if time.time() > phase1_end:
                break
            hc2, c5_2 = sa_optimize(hc, n_try, rng, time_limit=4, t_start=time.time(), iters=300000)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = hc2.copy()
                best_n = n_try

    # Phase 2: Focused SA on best
    remaining = budget_s - (time.time() - t_start) - 150
    if remaining > 20:
        best_h, best_c5 = sa_optimize(best_h, best_n, rng, time_limit=remaining * 0.6, t_start=time.time(), iters=10**8)

    # Phase 3: Deterministic local search
    remaining = budget_s - (time.time() - t_start) - 60
    if remaining > 5:
        best_h, best_c5 = deterministic_ls(best_h, best_n, time_limit=remaining, t_start=time.time())

    # Phase 4: Another SA + LS cycle
    remaining = budget_s - (time.time() - t_start) - 20
    if remaining > 15:
        best_h, best_c5 = sa_optimize(best_h, best_n, rng, time_limit=remaining * 0.5, t_start=time.time(), iters=10**8)
        remaining2 = budget_s - (time.time() - t_start) - 5
        if remaining2 > 3:
            best_h, best_c5 = deterministic_ls(best_h, best_n, time_limit=remaining2, t_start=time.time())

    # Final repair
    best_h = repair_h(best_h, best_n)
    final_c5 = evaluate_c5(best_h, best_n)

    return (best_h, final_c5, best_n)
