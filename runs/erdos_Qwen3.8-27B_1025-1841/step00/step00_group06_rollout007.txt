def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection to find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    # Correct small errors
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h_proj[mask] += diff / cnt
        else:
            mask = h_proj > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h_proj[mask] += diff / cnt
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

    print(f"Init pool: {len(inits)}, best: {best_obj:.6f}")

    # Subgradient optimization from best inits
    for idx, (h0, n, obj0) in enumerate(inits[:18]):
        if _remaining() < 80:
            break
        # Scale iterations by problem size
        est_cost = 4e-5 * n * n + 0.02
        max_it = max(40, min(800, int(40.0 / est_cost)))
        h_opt, o_opt = subgrad_optimize(h0, n, max_it, seed + idx * 17)
        _update(h_opt, o_opt, n)

    print(f"After subgrad: {best_obj:.6f} (n={best_n})")

    # Coordinate descent on best
    if _remaining() > 20 and best_h is not None:
        h_cd, o_cd = coord_descent(best_h, best_n, 4, seed + 777)
        _update(h_cd, o_cd, best_n)

    # More subgrad restarts around best
    if _remaining() > 30 and best_h is not None:
        for k in range(6):
            if _remaining() < 15:
                break
            h_p = best_h + 0.08 * rng_master.randn(best_n)
            h_p = project_feasible(h_p, best_n)
            h_s, o_s = subgrad_optimize(h_p, best_n, 150, seed + 600 + k)
            _update(h_s, o_s, best_n)

    # Final coordinate descent
    if _remaining() > 10 and best_h is not None:
        h_cd, o_cd = coord_descent(best_h, best_n, 3, seed + 999)
        _update(h_cd, o_cd, best_n)

    # Try a slightly different resolution for best
    if _remaining() > 60 and best_h is not None:
        for mul in [2, 3]:
            n_try = best_n * mul
            h_try = upscale_linear(best_h, n_try)
            h_try = project_feasible(h_try, n_try)
            h_opt, o_opt = subgrad_optimize(h_try, n_try, 60, seed + 2000)
            _update(h_opt, o_opt, n_try)
            if _remaining() > 10:
                h_cd, o_cd = coord_descent(h_opt, n_try, 2, seed + 2100)
                _update(h_cd, o_cd, n_try)

    # Safety: ensure we have a valid solution
    if best_h is None:
        n_fb = 95
        best_h = project_feasible(0.5 * np.ones(n_fb), n_fb)
        best_obj = evaluate_c5(best_h)
        best_n = n_fb

    elapsed = time.time() - t0
    print(f"Done in {elapsed:.1f}s. C5 = {best_obj:.6f}, n = {best_n}")
    return best_h, float(best_obj), int(best_n)
