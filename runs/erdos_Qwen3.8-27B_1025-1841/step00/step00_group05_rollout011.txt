def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    
    # Use top few active shifts
    num_full = 2 * n - 1
    k_top = min(10, num_full)
    top_idx = np.argsort(corr)[-k_top:][::-1]
    
    grad = np.zeros(n)
    for m in top_idx:
        offset = m - (n - 1)
        # d/dh[i] of sum_k h[k](1-h[k+offset])
        # = (1-h[i+offset]) for valid i + offset, minus h[i-offset] for valid i - offset
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1 + offset:hi1 + offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - offset:hi2 - offset]
    
    grad /= k_top
    return c5, grad

def project_feasible(h, n):
    """Project to 0<=h<=1, sum(h)=n/2 using bisection on lambda"""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    
    def try_lambda(lam):
        h_try = np.clip(h + lam, 0.0, 1.0)
        return h_try.sum()
    
    # Bisection: find lam such that try_lambda(lam) = target
    # try_lambda is non-decreasing in lam
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = try_lambda(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

- h[i] appears in term i with coefficient (1-h[i+j]) if i+j is in [0,n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    
    top_k = min(5, 2 * n - 1)
    top_idx = np.argsort(corr)[-top_k:][::-1]
    
    grad = np.zeros(n)
    for m in top_idx:
        offset = m - (n - 1)
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1 + offset:hi1 + offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - offset:hi2 - offset]
    
    grad /= top_k
    return c5, grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:

    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

    """Thue-Morse based initialization"""

        pos += block_len

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    max_corr = float(np.max(corr))
    
    # Use all correlations within 2% of max for a smoother subgradient
    threshold = max_corr - 0.02 * max(1.0, max_corr)
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    grad = np.zeros(n)
    for m in active:
        offset = int(m) - (n - 1)
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1 + offset:hi1 + offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - offset:hi2 - offset]
    
    grad /= float(len(active))
    return c5, grad

            lo = mid
        else:
            hi = mid

        else:

    beta = 0.85
    stall = 0

            batch = 100
        else:
