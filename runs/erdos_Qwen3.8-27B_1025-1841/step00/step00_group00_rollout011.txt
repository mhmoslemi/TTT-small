def compute_c5_and_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5, corr

def fix_sum(h, n, target=None):
    if target is None:
        target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    h = h.copy()
    # Uniform adjustment
    adj = diff / n
    h += adj
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    tol = 1e-12
    count = 0
    while abs(diff) > tol and count < 100 * n:
        if diff > 0:
            mask = h < 1.0 - tol
        else:
            mask = h > tol
        if not np.any(mask):
            # Force fix
            if diff > 0:
                idx = np.argmax(1.0 - h)
            else:
                idx = np.argmin(h)
            h[idx] += np.clip(diff, -(h[idx]), 1.0 - h[idx])
            break
        idxs = np.where(mask)[0]
        j = idxs[count % len(idxs)]
        if diff > 0:
            step = min(abs(diff), 1.0 - h[j])
            h[j] += step
        else:
            step = min(abs(diff), h[j])
            h[j] -= step
        diff = target - np.sum(h)
        count += 1
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    # Use piecewise constant (step function) upsample
    # h represents n_old bins, we want n_new bins
    # Each old bin covers n_new/n_new fraction
    ratio = n_new / n_old
    h_new = np.zeros(n_new)
    for i in range(n_old):
        start = int(i * ratio)
        end = int((i + 1) * ratio)
        if end > n_new:
            end = n_new
        if start >= n_new:
            start = n_new - 1
        h_new[start:end] = h[i]
    # Fix rounding
    h_new = fix_sum(h_new, n_new)
    return h_new

    """Simulated annealing with pair moves."""
    target = n / 2.0

    best_h = h.copy()
    best_c5 = current_c5
    t0 = time()
    temp = start_temp

        n_iter += 1
        # Generate a move: pick two indices and a delta
        i = rng.randint(n)
        j = (i + rng.randint(1, n)) % n
        # Step size decreases with temperature
        max_step = min(h[i], 1.0 - h[j]) if rng.random() < 0.5 else min(h[j], 1.0 - h[i])
        delta = rng.uniform(0.01, max_step * 10 + 0.01) * (0.5 if rng.random() < 0.5 else 1.0)
        
        delta = min(delta, 0.5)  # Cap
        if rng.random() < 0.5:
            hi = h[i] + delta
            hj = h[j] - delta
        else:
            hi = h[i] - delta
            hj = h[j] + delta
        
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:

            delta /= 2
            hi = h[i] + delta
            hj = h[j] - delta
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:

                continue

        new_h[i] = hi
        new_h[j] = hj
        new_c5, _ = compute_c5_and_lags(new_h)

                best_h = h.copy()

import numpy as np
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_array(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.maximum(0.0, np.minimum(1.0, h))
    diff = target - np.sum(h)
    for _ in range(10 * n):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        j = idx[np.random.randint(len(idx))]
        if diff > 0:
            h[j] += min(diff, 1.0 - h[j])
        else:
            h[j] += max(diff, -h[j])
        diff = target - np.sum(h)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    h_new = np.zeros(n_new)
    for i in range(n_old):
        lo = int(round(i * n_new / n_old))
        hi = int(round((i + 1) * n_new / n_old))
        if hi <= lo:
            hi = lo + 1
        if hi > n_new:
            hi = n_new
        h_new[lo:hi] = h[i]

    h_new = fix_sum(h_new, n_new)

                    c5 = compute_c5(new_h)
                    if c5 < best_c5 - 1e-15:
                        h = new_h
                        best_h = h.copy()
                        best_c5 = c5
                        found = True
                        break
            if found:
                break
        if not found:
            stall += 1
            step *= 0.95
            if stall > 200 or step < 1e-10:
                break
        else:
            stall = max(0, stall - 1)
    
    return best_h, best_c5

def sa_search(h, n, rng, time_limit):
    target = n / 2.0
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t0 = time()
    temp = 0.02
    floor = 0.002
    
    while time() - t0 < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-step_size(n), step_size(n))
        if rng.random() < 0.5:
            hi = h[i] + d
            hj = h[j] - d
        else:
            hi = h[i] - d
            hj = h[j] + d
        if not (0 <= hi <= 1 and 0 <= hj <= 1):
            d *= 0.3
            hi = h[i] + d
            hj = h[j] - d
            if not (0 <= hi <= 1 and 0 <= hj <= 1):
                temp = max(temp * 0.999, floor)
                continue
        new_h = h.copy()
        new_h[i] = hi
        new_h[j] = hj
        new_c5 = compute_c5(new_h)
        dc = new_c5 - cur_c5
        if dc < 0 or rng.random() < np.exp(-dc / max(temp, 1e-8)):
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        temp = max(temp * 0.9999, floor)
    
    return best_h, best_c5

def slsqp_optimize(h, n, time_limit):
    from scipy.optimize import minimize
    t0 = time()
    
    def objective(x):
        if time() - t0 > time_limit:
            return 999.0
        x = np.clip(x, 0, 1)
        return compute_c5(x)
    
    def objective_grad(x):
        # Numerical gradient (expensive for large n, but ok for small n)
        g = np.zeros(n)
        eps = 1e-6
        f0 = compute_c5(np.clip(x, 0, 1))
        for i in range(n):
            xp = x.copy()
            xp[i] += eps
            xp = np.clip(xp, 0, 1)
            xq = x.copy()
            xq[i] -= eps
            xq = np.clip(xq, 0, 1)
            g[i] = (compute_c5(xp) - compute_c5(xq)) / (2 * eps)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    x0 = h.copy().astype(float)
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 200, 'ftol': 1e-10})
    if res.success or res.fun < compute_c5(h):
        x = np.clip(res.x, 0, 1)
        x = fix_sum(x, n)
        return x, compute_c5(x)
    return h, compute_c5(h)
