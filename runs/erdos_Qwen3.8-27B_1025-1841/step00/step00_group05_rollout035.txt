import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:lag+m] += (1.0 - h[:m])
            g[:m] -= h[lag:lag+m]
    else:
        k = -lag
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:])
            g[k:] -= h[:m]
    return g
