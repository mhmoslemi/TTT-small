import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def obj_sm15(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(15.0 * (c - m)))) / 15.0) * dx

def obj_sm30(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(30.0 * (c - m)))) / 30.0) * dx

def obj_sm50(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(50.0 * (c - m)))) / 50.0) * dx

def obj_sm80(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(80.0 * (c - m)))) / 80.0) * dx

def obj_pnorm20(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    c_pos = np.maximum(c, 0.0)
    return (np.sum(c_pos ** 20) ** 0.05) * dx

def obj_pnorm50(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    c_pos = np.maximum(c, 0.0)
    return (np.sum(c_pos ** 50) ** 0.02) * dx

def upsample_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return np.clip(h_new, 0.0, 1.0)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(20):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_descent(h, n, max_iter=500):
    h = h.copy()
    dx = 2.0 / n
    best_val = evaluate_c5(h)
    for it in range(max_iter):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for new_val in [0.0, 1.0]:
                if h[i] == new_val:
                    continue
                h[i] = new_val
                h = fix_sum(h, n)
                val = evaluate_c5(h)
                if val < best_val - 1e-10:
                    best_val = val
                    improved = True
                    break
                else:
                    h[i] = h[i]  # revert (already changed)
                    h = fix_sum(h, n)
        if not improved:
            break
    return h, best_val

def try_binary_rounding(h, n):
    h_r = np.where(h > 0.5, 1.0, 0.0).copy()
    s = np.sum(h_r)
    if s != n / 2.0:
        # Adjust
        if s > n / 2.0:
            excess = int(s - n / 2.0)
            zeros = np.where(h_r == 0.0)[0]
            if len(zeros) >= excess:
                for i in range(excess):
                    h_r[zeros[i]] = 1.0
        else:
            deficit = int(n / 2.0 - s)
            ones = np.where(h_r == 1.0)[0]
            if len(ones) >= deficit:
                for i in range(deficit):
                    h_r[ones[i]] = 0.0
    h_r = fix_sum(h_r, n)
    return h_r
