def project_onto_simplex(h, target_sum):
    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = target_sum}"""
    # Use bisection on the threshold
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0, 1)
    # Adjust for floating point
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        # Distribute the difference
        n_pos = np.sum(projected < 1)
        n_neg = np.sum(projected > 0)
        if n_pos + n_neg > 0:
            projected += diff / max(1, n_pos + n_neg)
            projected = np.clip(projected, 0, 1)
    return projected

            g = np.zeros(n)
            g[:n-k] += (1.0 - h[k:])       # first factor contribution
            g[k:] += (-h[:n-k])            # second factor contribution
            grad += dx * g
        else:
            kp = -k
            # d/dh[j] of sum_{i=kp}^{n-1} h[i]*(1-h[i-kp])
            # = (1-h[j-kp]) for j in [kp, n-1]   [h[j] is first factor]
            #   + (-h[j+kp]) for j in [0, n-1-kp]  [h[j] is second factor]
            g = np.zeros(n)
            g[kp:] += (1.0 - h[:n-kp])
            g[:n-kp] += (-h[kp:])
            grad += dx * g
    
    na = len(active)
    if na > 1:
        grad /= na
    return grad, max_c * dx

def project_feasible(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - p.sum()
    if abs(diff) > 1e-13:
        mask = (p > 1e-15) & (p < 1.0 - 1e-15)
        if mask.any():
            p[mask] += diff / mask.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def subgradient_descent(h, target_sum, max_time=60.0, initial_lr=0.05):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = initial_lr
    t0 = time.time()
    
    while time.time() - t0 < max_time:
        grad, _ = get_subgradient(h)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_new = h - lr * grad / gn
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.8
        h = h_new
        if lr < 1e-8:
            break
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * corr.max()

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - 0.5*(lo + hi), 0.0, 1.0)
    diff = target_sum - p.sum()
    if abs(diff) > 1e-14:
        mask = (p > 1e-15) & (p < 1.0 - 1e-15)
        if mask.any():

def subgradient_descend(h, target_sum, max_time=60.0):
