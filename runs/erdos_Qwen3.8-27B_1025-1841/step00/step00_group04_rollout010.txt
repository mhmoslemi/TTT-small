    """Project h onto {0 <= h_i <= 1, sum(h) = target}"""
    def sum_clamped(lam):
        return np.sum(np.clip(h - lam, 0, 1))
    
    lo, hi = -2.0, 2.0
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_clamped(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def all_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    K = k_star - (n - 1)
    grad = np.zeros(n)
    if K == 0:

    elif K > 0:

        grad[0:n-d] += (1.0 - h[d:n])

    else:
        d = -K
        grad[d:n] += (1.0 - h[0:n-d])
        grad[0:n-d] -= h[d:n]
    return dx * grad

def smooth_max_grad(h, eps=0.05):
    """Gradient of smooth max (log-sum-exp) of corr*dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    corr_max = np.max(corr)
    weights = np.exp(eps * (corr - corr_max))
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        K = k - (n - 1)
        if K == 0:
            grad += weights[k] * (1.0 - 2.0 * h)
        elif K > 0:
            d = K
            grad[0:n-d] += weights[k] * (1.0 - h[d:n])
            grad[d:n] -= weights[k] * h[0:n-d]
        else:
            d = -K
            grad[d:n] += weights[k] * (1.0 - h[0:n-d])
            grad[0:n-d] -= weights[k] * h[d:n]
    return dx * grad, eps * (np.sum(weights * corr) - corr_max)

                if v < best_val:
                    best_val = v
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved = True
                    break
            else:
                continue
            break
        if not improved:
            break
    return h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(42)
        h_init[:21] = 1.0
    n_init = len(h_init)
    target = n_init / 2.0

    best_h = project_box_sum(h_init, target)
    best_c5 = evaluate(best_h)

    # Phase 1: Optimize at initial resolution
    if time_left() > 60:
        best_h, best_c5 = projected_optimize(best_h, max_iter=10000, budget_check=elapsed, budget_limit=budget_s - 60)

    # Phase 2: Random restarts at initial resolution
    for rs in range(8):
        if time_left() < 40:
            break
        h_rand = np.random.rand(n_init)
        h_rand = project_box_sum(h_rand, target)
        # Mix with best
        h_rand = 0.5 * h_rand + 0.5 * best_h
        h_rand = project_box_sum(h_rand, target)
        h_opt, c5_opt = projected_optimize(h_rand, max_iter=3000, budget_check=elapsed, budget_limit=budget_s - 40)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Binary optimization at initial resolution
    if time_left() > 60:
        h_bin = binary_project(best_h, int(target))
        h_b, c5_b = optimize_binary(200, h_bin)
        if c5_b < best_c5:
            best_c5 = c5_b
            # Use binary as starting point for continuous optimization
            h_cont = h_b.astype(float) + 0.02 * np.random.randn(n_init)
            h_cont = project_box_sum(h_cont, target)
            h_opt, c5_opt = projected_optimize(h_cont, max_iter=3000, budget_check=elapsed, budget_limit=budget_s - 60)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Multi-resolution upscaling
    for n_new in [210, 420, 840]:
        if time_left() < 80:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_box_sum(h_new, n_new / 2.0)
        h_opt, c5_opt = projected_optimize(h_new, max_iter=5000, budget_check=elapsed, budget_limit=budget_s - 80)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: Final refinement at best resolution
    if time_left() > 30:
        n = len(best_h)
        h_opt, c5_opt = projected_optimize(best_h, max_iter=10000, budget_check=elapsed, budget_limit=budget_s - 5)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    return best_h, best_c5, len(best_h)
