def to_binary(h, n_points, rng):
    """Convert to binary h with exactly n/2 ones."""
    n_ones = n_points // 2

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n_points, rng):
    """Round h to binary with exactly n/2 ones."""
    n_ones = n_points // 2
    scores = h.copy()
    # Add small random noise for tie-breaking
    scores += rng.random(n_points) * 1e-6
    idx = np.argsort(scores)[::-1][:n_ones]
    h_bin = np.zeros(n_points, dtype=float)
    h_bin[idx] = 1.0
    return h_bin

def weyl_sequence(n, alpha, n_ones):
    """Generate Weyl/Sturmian sequence."""
    h = np.zeros(n, dtype=float)
    positions = []
    for i in range(n):
        positions.append((np.floor((i+1)*alpha) - np.floor(i*alpha), i))
    # Actually, use the indicator of {i*alpha} < 1/2
    vals = np.array([ (i * alpha) % 1.0 for i in range(n) ])
    # Take the top n_ones
    idx = np.argsort(vals)[::-1][:n_ones]
    h[idx] = 1.0
    return h

def sa_optimize(h, n_points, t_start, t_end, n_iterations, rng):
    """Simulated annealing on binary arrangement."""
    h = h.copy()
    h_bin = make_binary(h, n_points, rng)
    n_ones = n_points // 2
    
    # Get positions of 1s and 0s
    one_pos = np.where(h_bin == 1)[0]
    zero_pos = np.where(h_bin == 0)[0]
    
    current_energy = evaluate_c5(h_bin, n_points)
    best_energy = current_energy
    best_h = h_bin.copy()
    
    T = t_start
    cooling = (t_end / t_start) ** (1.0 / n_iterations)
    
    for it in range(n_iterations):
        # Pick a random 1 and a random 0 to swap
        i = int(rng.integers(len(one_pos)))
        j = int(rng.integers(len(zero_pos)))
        pos_1 = one_pos[i]
        pos_0 = zero_pos[j]
        
        # Swap
        h_bin[pos_1] = 0
        h_bin[pos_0] = 1
        
        new_energy = evaluate_c5(h_bin, n_points)
        
        delta = new_energy - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            current_energy = new_energy
            # Update position arrays
            one_pos[i] = pos_0
            zero_pos[j] = pos_1
            if new_energy < best_energy:

                best_h = h_bin.copy()
        else:
            # Reject
            h_bin[pos_1] = 1
            h_bin[pos_0] = 0
        
        T *= cooling
    
    return best_h, best_energy

import numpy as np

def evaluate_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_binary(h, n, rng):
    n_ones = n // 2
    if h is None:
        h = np.zeros(n)
    scores = h.astype(np.float64).copy()
    scores += rng.random(n) * 0.1
    idx = np.argsort(scores)[::-1][:n_ones]
    out = np.zeros(n, dtype=np.float64)
    out[idx] = 1.0
    return out

def weyl_init(n, n_ones, alpha, rng):
    h = np.zeros(n, dtype=np.float64)

    h[idx] = 1.0
    return h

def upsample(h, n_from, n_to):
    x_from = np.linspace(0, 2.0, n_from, endpoint=False)
    x_to = np.linspace(0, 2.0, n_to, endpoint=False)

    h_interp = np.zeros(n_to)

        idx = min(int(x_to[j] / (2.0 / n_from)), n_from - 1)
        h_interp[j] = h[idx]
