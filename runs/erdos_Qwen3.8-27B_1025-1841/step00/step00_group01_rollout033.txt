def local_search_01(h0, n, rng, time_limit=60):
    """Local search for 0-1 binary sequence using pair swaps."""
    start = time.time()

    best_h = h.copy()

    # Simulated annealing parameters
    T = 0.02
    T_min = 0.002
    cooling = 0.9995
    
    no_improve = 0
    max_no_improve = 500
    
    while time.time() - start < time_limit and no_improve < max_no_improve:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        improved = False
        n_tries = 100

            j = zeros[rng.integers(len(zeros))]
            
            h[i] = 0
            h[j] = 1
            new_c5 = compute_c5(h)
            delta = new_c5 - best_c5
            
            if delta < 0:
                best_h = h.copy()
                best_c5 = new_c5
                improved = True
                no_improve = 0
            elif rng.random() < np.exp(-delta / max(T, 1e-15)):
                # Accept worse (simulated annealing)
                pass
            else:

                h[j] = 0

            no_improve += 1
            # Occasionally reset to best
            if no_improve % 100 == 0:

                T = 0.02
