            # df/dh[i] = 1_{i<nv} - 1_{i<nv}*h[i+m] - 1_{i-m>=0, i-m<nv}*h[i-m]
            grad[:nv] = 1.0 - h[m:n]  # from first two terms (i in [0,nv))
            # third term: i-m is valid first index => m <= i < nv+m, i.e., i in [m, m+nv)
            lo2 = max(m, 0)
            hi2 = min(m + nv, n)
            if lo2 < hi2:
                grad[lo2:hi2] -= h[lo2-m:hi2-m]
    else:
        mm = -m
        nv = n - mm
        if nv > 0:
            # valid j: 0 <= j < n, 0 <= j-mm < n => mm <= j < n
            grad[mm:n] = 1.0 - h[:nv]
            lo2 = 0
            hi2 = min(nv, n)
            if lo2 < hi2:
                grad[lo2:hi2] -= h[lo2+mm:hi2+mm]
    return grad

def sgsq_optimize(h0, n, max_iters=3000, time_limit=60):
    """Subgradient + SLSQP hybrid optimization."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_val = evaluate_c5(h, dx)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = int(np.argmax(c))
    
    step = 0.05
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        m = k_star - (n - 1)
        grad = compute_grad_shift(h, n, m)
        h_new = project_feasible(h - step * grad, target_sum)
        val = evaluate_c5(h_new, dx)
        if val < best_val - 1e-8:
            best_h = h_new.copy()
            best_val = val
            h = h_new
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            k_new = int(np.argmax(c))
            if k_new != k_star:
                k_star = k_new
            step = min(step * 1.05, 0.3)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return best_h, best_val

def slsqp_optimize(h0, n, time_limit=60):
    """SLSQP optimization with sum and box constraints."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0.copy(), 0.0, 1.0)
    
    def objective(h):
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0)] * n
    
    TMAX = time.time() + time_limit
    timeout = False
    def obj_wrapper(h):
        nonlocal timeout
        if time.time() > TMAX:
            timeout = True
            return np.max(np.correlate(h, 1.0-h, mode='full')) * dx
        return objective(h)
    
    result = minimize(obj_wrapper, h0, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-8})
    if result.success or result.fun < best_global:
        h_res = fix_sum_exact(np.clip(result.x, 0.0, 1.0), n)
        return h_res, evaluate_c5(h_res, dx)
    return None, None

best_global = 0.38085

def run(seed=42, budget_s=1300, **kwargs):
    global t_start, best_global
    t_start = time.time()
    remaining = budget_s
    np.random.seed(seed)
    
    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(80, 0.5)
    
    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = evaluate_c5(best_h, 2.0 / n_parent)
    best_n = n_parent
    best_global = best_c5
    
    def elapsed():
        return time.time() - t_start
    
    def remaining_time():
        return budget_s - elapsed()
    
    # Phase 1: Optimize at parent resolution with SG+SQ
    t_lim = min(120, remaining_time() * 0.25)
    h_opt, c5_opt = sgsq_optimize(h_parent, n_parent, max_iters=5000, time_limit=t_lim)
    if c5_opt < best_c5:
        best_h, best_c5, best_n = h_opt, c5_opt, n_parent
        best_global = best_c5
    
    # Phase 2: SLSQP at parent resolution
    if remaining_time() > 30:
        t_lim = min(180, remaining_time() * 0.25)
        res = slsqp_optimize(best_h, best_n, time_limit=t_lim)
        if res[0] is not None and res[1] < best_c5:
            best_h, best_c5, best_n = res[0], res[1], res[1]
            best_global = best_c5
        if res[0] is not None and res[1] < best_c5:
            best_c5 = res[1]
            best_global = best_c5
    
    # Phase 3: Multi-resolution search
    resolutions = [2*n_parent, 4*n_parent, 3*n_parent, 5*n_parent]
    resolutions = sorted(set([r for r in resolutions if r >= 64 and r <= 600]))
    # Ensure parent res is included
    if n_parent not in resolutions:
        resolutions = sorted(resolutions + [n_parent])
    
    for n_pts in resolutions:
        if remaining_time() < 45:
            break
        
        # Interpolate current best to new resolution
        h_init = interp_parent(best_h, n_pts)
        h_init = fix_sum_exact(h_init, n_pts)
        
        # SG+SQ optimization
        t_lim = min(200, remaining_time() * 0.35)
        h_opt, c5_opt = sgsq_optimize(h_init, n_pts, max_iters=3000, time_limit=t_lim)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_pts
            best_global = best_c5
        
        # SLSQP refinement
        if remaining_time() > 30 and best_n == n_pts:
            t_lim2 = min(120, remaining_time() * 0.3)
            res2 = slsqp_optimize(best_h, n_pts, time_limit=t_lim2)
            if res2[0] is not None and res2[1] < best_c5:
                best_c5 = res2[1]
                best_h = res2[0]
                best_global = best_c5
        
        # Random restart at this resolution
        if remaining_time() > 60:
            h_rand = np.random.rand(n_pts)
            h_rand = fix_sum_exact(h_rand, n_pts)
            t_lim3 = min(100, remaining_time() * 0.25)
            h_opt3, c5_opt3 = sgsq_optimize(h_rand, n_pts, max_iters=2000, time_limit=t_lim3)
            if c5_opt3 < best_c5:
                best_h, best_c5, best_n = h_opt3, c5_opt3, n_pts
                best_global = best_c5
    
    # Phase 4: Final SLSQP refinement at best resolution
    if remaining_time() > 20:
        t_lim = min(120, remaining_time() * 0.7)
        res_final = slsqp_optimize(best_h, best_n, time_limit=t_lim)
        if res_final[0] is not None and res_final[1] < best_c5:
            best_c5 = res_final[1]
            best_h = res_final[0]
            best_global = best_c5
    
    # Final SG+SQ polish
    if remaining_time() > 15:
        t_lim_f = max(10, remaining_time() - 5)
        h_pol, c5_pol = sgsq_optimize(best_h, best_n, max_iters=2000, time_limit=t_lim_f)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
    
    # Fix sum and evaluate
    best_h = fix_sum_exact(np.clip(best_h, 0.0, 1.0), best_n)
    final_c5 = evaluate_c5(best_h, 2.0 / best_n)
    
    return (best_h, final_c5, best_n)
