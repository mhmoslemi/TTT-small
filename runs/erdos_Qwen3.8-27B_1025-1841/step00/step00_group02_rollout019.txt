import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def c5_obj(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def c5_grad(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    L = max(0, n - 1 - idx)
    R = min(n - 1, 2 * (n - 1) - idx)
    grad = np.zeros(n)
    for t in range(L, R + 1):
        s = t + idx - n + 1
        grad[s] += (1.0 - h[t])
        grad[t] -= h[s]
    return grad * dx

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def project_set(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_to(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    return np.interp(x_new, x_old, h_old)

def make_starts(n, rng, h_ref, n_ref):
    pts = []
    h_up = upsample_to(h_ref, n_ref, n)
    pts.append(project_set(h_up, n))
    for freq in [1, 2, 3, 5]:
        x = np.arange(n) / float(n)
        h_s = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x)
        pts.append(project_set(h_s, n))
    for _ in range(2):
        h_r = 0.5 + 0.2 * rng.randn(n)
        k = np.array([1.0, 2.0, 4.0, 2.0, 1.0])
        k = k / k.sum()
        h_r = np.convolve(h_r, k, mode='same')
        pts.append(project_set(h_r, n))
    perm = rng.permutation(n)
    h_b = np.zeros(n)
    h_b[perm[: n // 2]] = 1.0
    pts.append(h_b)
    return pts
