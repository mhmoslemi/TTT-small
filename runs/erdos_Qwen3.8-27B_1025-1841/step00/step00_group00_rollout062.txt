import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_sum(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-12:
        return h
    # Bisection on water-level
    # We need: sum(min(1, max(0, h_i + lambda)) for i) = target_sum
    # f(lambda) is monotonically increasing in lambda
    lo, hi = -diff, diff  # generous bounds
    for _ in range(100):
        mid = (lo + hi) / 2
        h_trial = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    # Tiny correction for floating point
    h_out[np.argmin(np.abs(h_out - 0.5))] += target_sum - np.sum(h_out)
    return h_out

def subgradient(h, dx):
    """Compute subgradient of C5 at h."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    k_max = np.argmax(corr)
    # Active range: m such that 0 <= k_max - m <= n-1
    m_lo = max(0, k_max - (n - 1))
    m_hi = min(n - 1, k_max)
    g = np.zeros(n)
    for m in range(m_lo, m_hi + 1):
        g[m] = dx * (1 - 2 * h[k_max - m])
    return g

        c5_t = evaluate_c5(h_w)
        if c5_t < best_c5:
            best_h, best_c5 = h_w.copy(), c5_t
            stall = 0
        else:
            if np.random.random() < np.exp(-(c5_t - best_c5) / T):
                stall += 1
            else:
                h_w[i] -= d
                h_w[j] += d
                stall += 1
        T = max(1e-7, T * 0.99997)
        if stall > 1000:
            T = min(0.004, T * 1.5)
            stall = 0

    # Phase 3: Higher resolution optimization (~33% budget)
    p3_end = t0 + budget_s * 0.96
    for sc in [2, 3, 4]:
        tn = n * sc
        if time.time() > p3_end - 30:
            break
        xo = np.linspace(0, 2.0, best_n, endpoint=False)
        xn = np.linspace(0, 2.0, tn, endpoint=False)
        h_hi = project_box_sum(np.interp(xn, xo, best_h), tn / 2)
        if evaluate_c5(h_hi) > best_c5 * 1.02:
            continue
        T, alpha, stall = 0.002, 0.15, 0
        h_cur = h_hi.copy()
        while time.time() < p3_end:
            if stall > 2000:
                if time.time() > p3_end - 60:
                    break
                # restart from best
                h_cur = project_box_sum(
                    np.interp(xn, np.linspace(0, 2.0, best_n, endpoint=False), best_h), tn / 2)
                stall = 0
                continue
            r = np.random.random()
            if r < 0.6:
                i, j = np.random.randint(tn), np.random.randint(tn)
                if i == j:
                    continue
                d = np.random.uniform(-0.02, 0.02)
                if not (0.0 <= h_cur[i] + d <= 1.0 and 0.0 <= h_cur[j] - d <= 1.0):
                    continue
                h_t = h_cur.copy()
                h_t[i] += d
                h_t[j] -= d
            else:
                h_t = project_box_sum(h_cur - alpha * subgrad(h_cur, 2.0 / tn), tn / 2)
            c5_t = evaluate_c5(h_t)
            if c5_t < best_c5:
                best_h, best_c5, best_n = h_t.copy(), c5_t, tn
                h_cur = h_t
                stall = 0
            else:
                if np.random.random() < np.exp(-(c5_t - best_c5) / T):
                    h_cur = h_t
                stall += 1
            T = max(1e-7, T * 0.99998)
            alpha *= 0.99995

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_n
