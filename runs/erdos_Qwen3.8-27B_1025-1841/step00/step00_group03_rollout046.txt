import numpy as np
import time

def project_onto_constraint(v, S):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        tau = (lo + hi) * 0.5
        s = np.sum(np.clip(v - tau, 0.0, 1.0))
        if s > S:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) * 0.5
    return np.clip(v - tau, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_active(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        t = -s
        g[t:] += 1.0 - h[:n - t]
        g[:n - t] -= h[t:]
    return g

def optimize_level(h, n, time_budget, step_scale=0.1, rng=None):

    start_time = time.time()

    no_improve = 0
    patience = 2000

        elapsed = time.time() - start_time
        if elapsed >= time_budget:
            break

        c5_raw = float(np.max(corr))

            best_c5 = c5

            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:

            kick_scale = 0.1 * (no_improve / patience)

            no_improve = 0
