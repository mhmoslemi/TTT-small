    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def softmax_obj(h, tau=0.01):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Softmax approximation of max
    corr_scaled = corr / tau
    max_val = np.max(corr_scaled)
    return tau * (max_val + np.log(np.sum(np.exp(corr_scaled - max_val)))) * dx

    h = np.clip(h, 0, 1)
    target = n / 2.0
    current = np.sum(h)
    diff = target - current
    if abs(diff) > 1e-12:

        else:
            mask = h > 0
            count = np.sum(mask)
            if count > 0:
                h[mask] = np.clip(h[mask] + diff / count, 0, 1)
    # Final clip
    h = np.clip(h, 0, 1)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
