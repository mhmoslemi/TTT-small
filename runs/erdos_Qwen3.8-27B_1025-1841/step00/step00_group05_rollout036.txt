```python
import numpy as np
import time

def compute_prefix(h):
    """Compute prefix sums F where F[k] = sum(h[0:k])"""
    F = np.zeros(len(h) + 1)
    F[1:] = np.cumsum(h)
    return F

def compute_A_values(h):
    """Compute A(k) for k=1,...,n-1 using the prefix sum formula"""
    n = len(h)
    F = np.zeros(n + 1)
    F[1:] = np.cumsum(h)
    # A(k) = (n-k + n/2 - F[n-k] - F[k]) / 2 for k=1,...,n-1
    k = np.arange(1, n)
    A = (n - k + n/2 - F[n-k] - F[k]) / 2
    return A
```

```python
def c5_binary_fast(h):
    """Compute C5 for binary h using prefix sums"""
    n = len(h)
    F = np.zeros(n + 1)
    np.cumsum(h, out=F[1:])
    k = np.arange(1, n)
    A = (n - k + n / 2.0 - F[n - k] - F[k]) / 2.0
    return np.max(A) * (2.0 / n)

def local_search_binary(h, max_iters=100000, rng=None):
    """
    Local search: at each step, try all possible swaps and take the one
    that most reduces the max A(k).
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(h)
    h = h.astype(float).copy()
    best_cost = c5_binary_fast(h)
    
    for it in range(max_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try each swap (limited for speed)
        best_swap_cost = best_cost
        best_i, best_j = -1, -1
        
        # For efficiency, try a random subset of swaps
        n_tries = min(len(ones) * len(zeros), 500)
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            cost = c5_binary_fast(h)
            h[i], h[j] = 1.0, 0.0
            if cost < best_swap_cost:
                best_swap_cost = cost
                best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            best_cost = best_swap_cost
        else:
            # Stuck, try a random perturbation
            if rng.random() < 0.1:
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i], h[j] = 0.0, 1.0
                best_cost = c5_binary_fast(h)
            else:
                break
    
    return h, best_cost
```

```python
def project(h, target_sum):
    """Project h onto {x in [0,1]^n : sum x = target_sum}"""
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            # Need to increase sum: add to elements not at 1
            mask = h < 1
            if mask.sum() == 0:
                break
            add = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + add, 1)
        else:
            # Need to decrease sum: remove from elements not at 0
            mask = h > 0
            if mask.sum() == 0:
                break
            sub = (-diff) / mask.sum()
            h[mask] = np.maximum(h[mask] - sub, 0)
    return h
```
