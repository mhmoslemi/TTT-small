        x_old = (np.arange(n) + 0.5) * 2.0 / n
        x_new = (np.arange(n_try) + 0.5) * 2.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = _project(h_interp, n_try, ts_try)
        h_curr = _optimize(h_interp, n_try, dx_try, 40, ts_try, 500, 1e-12)
        c5_v = _eval_c5(h_curr, dx_try)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h_curr.copy()
            best_n = n_try
            best_dx = dx_try

    # Phase 4: Polish with high power mean
    if time.time() - t0 < budget_s - 30:
        for p in [60, 100, 200]:
            if time.time() - t0 > budget_s - 15:
                break
            h_curr = _optimize(best_h.copy(), best_n, best_dx, p, best_n / 2.0, 800, 1e-14)
            c5_v = _eval_c5(h_curr, best_dx)
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = h_curr.copy()

    # Phase 5: Local hill-climbing on individual coordinates
    if time.time() - t0 < budget_s - 10 and best_n <= 300:
        dx_f = 2.0 / best_n
        ts_f = best_n / 2.0
        improved = True
        while improved and time.time() - t0 < budget_s - 5:
            improved = False
            for i in range(best_n):
                for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                    h_trial = best_h.copy()
                    h_trial[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
                    h_trial = _project(h_trial, best_n, ts_f)
                    c5_trial = _eval_c5(h_trial, best_dx)
                    if c5_trial < best_c5 - 1e-12:
                        best_c5 = c5_trial
                        best_h = h_trial
                        improved = True
                        break

    return (best_h, best_c5, best_n)
