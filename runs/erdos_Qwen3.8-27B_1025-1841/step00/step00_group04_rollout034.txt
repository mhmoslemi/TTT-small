The solution is: z = -B^{-1} c + λ * B^{-1} * ones, where λ is chosen so that 1^T z = 0.

But B might not be invertible. Let me use the pseudoinverse or the null space approach.

Actually, the constraint 1^T z = 0 means z is in the (n-1)-dimensional subspace orthogonal to 1. I can parameterize z = P * y where P is an (n × (n-1)) matrix whose columns form an orthonormal basis for 1^⊥.

Then: minimize y^T (P^T B P) y + (P^T c)^T y + d
unconstrained in y.

Solution: y = -(P^T B P)^{-1} (P^T c)

This is O(n³) to solve, which is fine for n up to a few hundred.

def solve_quadratic_phase(h, weights, n):
    """Solve the quadratic phase of the optimization.
    
    h: current solution (length n)
    weights: weights for each shift
    n: length of h
    
    Returns: new h
    """
    s = n / 2.0
    base = np.ones(n) * 0.5
    
    # Compute B = sum_k w_k * S_k
    B = np.zeros((n, n))
    for k_shift, w in weights:
        if w < 1e-10:
            continue
        S = np.zeros((n, n))
        if k_shift >= 0:
            for i in range(n - k_shift):
                S[i, i + k_shift] = 0.5
                S[i + k_shift, i] = 0.5
        else:
            for i in range(-k_shift, n):
                S[i, i + k_shift] = 0.5
                S[i + k_shift, i] = 0.5
        B += w * S
    
    # c = 2 * sum_k w_k * S_k @ base = sum_k w_k * S_k @ ones
    c = np.zeros(n)
    for k_shift, w in weights:
        if w < 1e-10:
            continue
        S = np.zeros((n, n))  # wasteful but correct
        if k_shift >= 0:
            for i in range(n - k_shift):
                S[i, i + k_shift] = 0.5
                S[i + k_shift, i] = 0.5
        else:
            for i in range(-k_shift, n):
                S[i, i + k_shift] = 0.5
                S[i + k_shift, i] = 0.5
        c += w * 2 * (S @ base)
    
    # Project to subspace 1^⊥
    # P: (n x (n-1)) orthonormal basis for 1^⊥
    ones = np.ones(n)
    norm = np.linalg.norm(ones)
    u1 = ones / norm
    # Complete to orthonormal basis using QR
    # Actually, let's use a simpler approach:
    # z = y - (mean(y)) * ones (project y onto 1^⊥)
    # So the constraint is automatically satisfied if we work in 1^⊥.
    
    # The quadratic objective in z (with z ⊥ 1):
    # min z^T B z + c^T z + const
    # subject to z ⊥ 1
    
    # Parameterize: z = Q @ y where Q is (n x (n-1)) with orthonormal columns spanning 1^⊥
    # min y^T (Q^T B Q) y + (Q^T c)^T y
    
    # Use: Q^T B Q and Q^T c
    # To avoid explicitly forming Q, note that for any z ⊥ 1:
    # z^T B z = z^T (B - (1/n) * ones * ones^T * B - B * ones * ones^T / n + ...) 
    # This is getting complicated. Let me just form Q.
    
    # Form Q using Householder or QR
    A_mat = np.concatenate([ones.reshape(-1, 1), np.eye(n)], axis=1)
    Q_full, R = np.linalg.qr(A_mat)
    Q = Q_full[:, 1:]  # (n x (n-1)), first column is proportional to ones
    
    B_reduced = Q.T @ B @ Q  # (n-1 x n-1)
    c_reduced = Q.T @ c  # (n-1,)
    
    # Solve: 2 * B_reduced @ y + c_reduced = 0
    # y = -0.5 * B_reduced^{-1} @ c_reduced
    try:
        y = -0.5 * np.linalg.solve(B_reduced, c_reduced)
    except np.linalg.LinAlgError:
        y = -0.5 * np.linalg.lstsq(B_reduced, c_reduced, rcond=None)[0]
    
    z = Q @ y
    h_new = base + z
    h_new = project_box_hyperplane(h_new, s)
    return h_new

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_box_hyperplane(h, s):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > s:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_shift(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            grad[shift:] += (1.0 - h[:n - shift])
            grad[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
    return grad

def soft_max_gradient(h, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    # Numerical stability
    c_max = np.max(c)
    exp_c = np.exp(tau * (c - c_max))
    weights = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] > 1e-10:
            shift = k - (n - 1)
            grad += weights[k] * compute_grad_shift(h, shift)
    return grad, c_max * (2.0 / n)

def subgrad_optimize(h_init, max_iters, lr_init, tau, time_limit, start_time):
    h = h_init.copy()
    n = len(h)
    s = n / 2.0
    h = project_box_hyperplane(h, s)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        grad, c5 = soft_max_gradient(h, tau)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.002, 0.5)
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.95
            if lr < 0.0005:
                break

        h = h + v
        h = project_box_hyperplane(h, s)
    
    return best_h, best_c5

def quadratic_polish(h_init, time_limit, start_time):
    h = h_init.copy()
    n = len(h)
    s = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Compute correlation values
    c = np.correlate(h, 1.0 - h, mode="full")
    
    # Weight top shifts
    top_k = max(5, len(c) // 10)
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(top_k, len(c_sorted)-1)]
    
    # Solve quadratic subproblem
    base = np.full(n, 0.5)
    
    # Build reduced Hessian and gradient
    # Work in the subspace sum(z) = 0
    # z = h - base (approximately, but we'll project)
    
    # For efficiency, compute B @ v on the fly
    shifts = np.arange(-(n-1), n)
    weights = np.zeros(len(shifts))
    for k, sh in enumerate(shifts):
        if c[k] >= threshold:
            weights[k] = 1.0
    
    active_shifts = shifts[weights > 0]
    if len(active_shifts) == 0:
        active_shifts = [int(np.argmax(c)) - (n-1)]
    
    # Build B and c_vec
    # B = sum_k w_k * S_k (symmetric)
    # We'll compute B @ v for the QP
    
    # For now, use scipy to solve the QP
    # Actually, let's just use the null-space approach
    
    # Project onto 1^⊥
    ones = np.ones(n)
    # QR decomposition to get basis for 1^⊥
    M = np.column_stack([ones, np.eye(n)])
    Q, R = np.linalg.qr(M)
    Q_perp = Q[:, 1:]  # n x (n-1)
    
    # Compute B_reduced = Q_perp^T B Q_perp
    # and c_reduced = Q_perp^T (B @ base + grad_linear)
    
    # B @ v for v in span(Q_perp):
    # B @ (Q_perp @ y) = Q_perp @ (Q_perp^T B Q_perp) @ y
    
    # Compute B @ v for each column of Q_perp
    # B @ v = sum_k w_k * S_k @ v
    # S_k @ v[i] = 0.5*(v[i+k] if valid) + 0.5*(v[i-k] if valid)
    
    def B_mv(v):
        result = np.zeros(n)
        for sh in active_shifts:
            if sh >= 0:
                result[sh:] += 0.5 * v[:n-sh]
                result[:n-sh] += 0.5 * v[sh:]
            else:
                s2 = -sh
                result[:n-s2] += 0.5 * v[s2:]
                result[s2:] += 0.5 * v[:n-s2]
        return result
    
    # c_vec (linear part): from expanding (base+z)^T S_k (base+z)
    # = base^T S_k base + 2*z^T S_k base + z^T S_k z
    # Linear in z: 2 * S_k @ base (symmetric: S_k @ base)
    # Actually: d/dz (z^T S_k z + 2*base^T S_k z) = 2*S_k*z + 2*S_k*base
    # So the "gradient" of the quadratic w.r.t. z at z=0 is 2*S_k*base
    
    def linear_part(v_unused=None):
        result = np.zeros(n)
        for sh in active_shifts:
            if sh >= 0:
                result[sh:] += base[:n-sh]
                result[:n-sh] += base[sh:]
            else:
                s2 = -sh
                result[:n-s2] += base[s2:]
                result[s2:] += base[:n-s2]
        return result
    
    # B_reduced = Q_perp^T B Q_perp ((n-1) x (n-1))
    B_red = np.zeros((n-1, n-1))
    for j in range(n-1):
        B_red[:, j] = Q_perp.T @ B_mv(Q_perp[:, j])
    
    # c_red = 2 * Q_perp^T (S @ base) where S = sum S_k
    lin = np.zeros(n)
    for sh in active_shifts:
        if sh >= 0:
            lin[sh:] += base[:n-sh]
            lin[:n-sh] += base[sh:]
        else:
            s2 = -sh
            lin[:n-s2] += base[s2:]
            lin[s2:] += base[:n-s2]
    c_red = 2.0 * (Q_perp.T @ lin)
    
    # Solve: 2 * B_red @ y + c_red = 0
    # y = -0.5 * B_red^{-1} @ c_red
    try:
        y = -0.5 * np.linalg.solve(B_red + 1e-8 * np.eye(n-1), c_red)
    except:
        y = -0.5 * np.linalg.lstsq(B_red + 1e-8 * np.eye(n-1), c_red, rcond=None)[0]
    
    z = Q_perp @ y
    h_new = base + z
    h_new = project_box_hyperplane(h_new, s)
    c5_new = evaluate_c5(h_new)
    
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h_new.copy()
    
    return best_h, best_c5

def evaluate_c5(h):
