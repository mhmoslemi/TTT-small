import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    if m * alpha > 700:
        return m
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)
    grad = np.zeros(n)
    idx1 = np.arange(n) + s
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = np.arange(n) - s
    valid2 = (idx2 >= 0) & (idx2 < n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = float(corr.max())
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    return m + float(np.log(np.sum(np.exp(shifted)))) / alpha

def subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    s = k_star - (n - 1)
    grad = np.zeros(n)
    idx1 = np.arange(n) + s
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = np.arange(n) - s
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[idx2[valid2]]
    return dx * grad
