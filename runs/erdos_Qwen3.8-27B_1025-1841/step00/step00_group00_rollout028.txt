def upsample(h, factor=2):
    """Upsample h by repeating each value `factor` times, then smooth"""
    n = len(h)
    h_new = np.repeat(h, factor)
    # Project back to valid range
    target_sum = len(h_new) / 2.0
    h_new = np.clip(h_new, 0, 1)
    # Adjust sum
    diff = target_sum - np.sum(h_new)
    # Distribute the difference
    if abs(diff) > 1e-10:
        # Add/subtract from random positions
        n_new = len(h_new)
        for _ in range(1000):
            if diff > 0:
                idx = np.where(h_new < 1 - 1e-6)[0]
                if len(idx) == 0:
                    break
                i = np.random.choice(idx)
                amt = min(diff, 1 - h_new[i])
                h_new[i] += amt
                diff -= amt
            else:
                idx = np.where(h_new > 1e-6)[0]
                if len(idx) == 0:
                    break
                i = np.random.choice(idx)
                amt = min(-diff, h_new[i])
                h_new[i] -= amt
                diff += amt
    return h_new

def local_search_sa(h, target_sum, n_iter=50000, rng=None, T0=0.01, T_min=1e-6):

    if rng is None:
        rng = np.random.RandomState()
    n = len(h)
    h = h.copy()
    current_val = evaluate_c5(h)
    best_h = h.copy()
    best_val = current_val
    
    for it in range(n_iter):

        for i in valid_first:

        # Second: for i in valid_second, contribution to grad[i] += -h[i-m]
        for i in valid_second:
            grad[i] += -h[i - m] / len(active)

Let me implement the projection properly. The projection of a point y onto {0 ≤ x ≤ 1, sum(x) = S} is:
x[i] = clip(y[i] - τ, 0, 1)
where τ is chosen so that sum(x) = S.

Since sum(clip(y[i] - τ, 0, 1)) is continuous and monotonically decreasing in τ, we can find τ by bisection.

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_big_simplex(y, S):
    """Project y onto {0 <= x <= 1, sum(x) = S}"""
    n = len(y)
    # Bisection on tau
    lo = y.min() - S  # ensure sum(clip(y-tau,0,1)) >= S at lo
    hi = y.max() + 1  # ensure sum(clip(y-tau,0,1)) <= S at hi
    # Actually:
    # When tau = -inf, x = clip(y+inf, 0, 1) = ones, sum = n
    # When tau = +inf, x = clip(y-inf, 0, 1) = zeros, sum = 0
    # So sum is decreasing in tau
    # Want sum = S
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        tau = (lo + hi) / 2
        x = np.clip(y - tau, 0, 1)
        s = x.sum()
        if s > S:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    x = np.clip(y - tau, 0, 1)
    return x

def subgradient_F(h):
    """Compute a subgradient of F(h) = max_k c_k * dx at h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Find active lags
    tol = 1e-8 * max(1, max_val)
    active = np.where(np.abs(corr - max_val) <= tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)  # lag
        # corr[k] = sum over valid i of h[i] * (1 - h[i+m])
        # where i ranges such that both i and i+m are in [0, n)
        # i.e., i in [max(0, -m), min(n, n-m))
        
        # Gradient of corr[k] w.r.t. h[i]:
        # h[i] appears in the term where first index is i: (1 - h[i+m]) [if i+m valid]
        # h[i] appears in the term where second index is i: -h[i-m] [if i-m valid]
        
        # For first appearance: i valid, i+m valid => m <= i < n-m (if m>0) or -m <= i < n+m (if m<0)
        # Let's just compute directly
        if m >= 0:
            # i ranges from 0 to n-1-m
            i_range = range(0, n - m)
            # h[i] * (1-h[i+m]): grad w.r.t h[i] += (1-h[i+m]), grad w.r.t h[i+m] += -h[i]
            for i in i_range:
                grad[i] += (1 - h[i + m])
                grad[i + m] += -h[i]
        else:
            # m < 0, let p = -m > 0
            p = -m
            # i ranges from p to n-1
            i_range = range(p, n)
            # h[i] * (1-h[i-p]): grad w.r.t h[i] += (1-h[i-p]), grad w.r.t h[i-p] += -h[i]
            for i in i_range:
                grad[i] += (1 - h[i - p])
                grad[i - p] += -h[i]
    
    grad /= len(active)
    return grad * dx

def soft_max_gradient(h, tau):
    """Gradient of soft max of correlation values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Soft max weights
    corr_scaled = corr / tau
    corr_scaled = corr_scaled - np.max(corr_scaled)  # for numerical stability
    exps = np.exp(corr_scaled)
    weights = exps / np.sum(exps)
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_big_simplex(y, S):
    n = len(y)
    if S <= 0: return np.zeros(n)
    if S >= n: return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(200):
        tau = (lo + hi) / 2.0
        if np.clip(y - tau, 0.0, 1.0).sum() > S:
            lo = tau
        else:
            hi = tau
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def fista_solve(h, max_iter=10000, tau0=0.01, tau_min=0.001):
    """FISTA with soft-max objective"""
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    for it in range(max_iter):
        tau = tau0 * (tau_min / tau0) ** (it / max_iter)
        
        # Compute soft-max gradient
        corr = np.correlate(y, 1.0 - y, mode="full")
        dx = 2.0 / n
        scaled = (corr - corr.max()) / tau
        exps = np.exp(scaled)
        weights = exps / exps.sum()
        
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-8:
                continue
            lo1, hi1 = max(0, k), min(n-1, k+n-1)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += w * (1.0 - y[lo1-k:hi1-k+1])
            lo2, hi2 = max(0, -k), min(n-1, n-1-k)
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= w * y[lo2+k:hi2+k+1]
        grad *= dx
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        lr = 1.0 / (it + 3)  # or some other schedule
        # Actually, use lr = 1/L where L is Lipschitz constant
        # For now, use a small fixed lr
        lr = 0.1 / (1.0 + it / 1000)
        
        h = project_big_simplex(y - lr * grad, target_sum)
        
        y = h + ((t - 1.0) / t_new) * (h - y)
        t = t_new
        
        val = evaluate_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

def local_search(h, n_iter, rng):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j: continue
        delta = rng.choice([0.01, 0.02, 0.05, 0.1, 0.2]) * rng.choice([1, -1])
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        new_val = evaluate_c5(h_new)
        if new_val < best_val:
            h = h_new
            best_val = new_val
            best_h = h.copy()
    
    return best_h, best_val

def upsample(h, factor=2, rng=None):
    if rng is None: rng = np.random.RandomState(0)
    h_new = np.repeat(h, factor) + rng.uniform(-0.01, 0.01, factor * len(h))
    h_new = np.clip(h_new, 0, 1)
    return project_big_simplex(h_new, len(h_new) / 2.0)

def run(seed=42, budget_s=1300, **kwargs):

    # Optimize at current resolution
    h, v = fista_solve(h, max_iter=5000)
    if v < best_c5: best_c5, best_h = v, h.copy()
    h, v = local_search(h, 50000, rng)

    # Multi-resolution

        if v < best_c5: best_c5, best_h = v, h.copy()
    
    return best_h, evaluate_c5(best_h), len(best_h)
