def project_feasible(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Binary search for lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    return h_proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

    sum_pen = (np.sum(h) - target_sum) ** 2
    h = np.clip(h, 0, 1)
    c5 = compute_c5(h)
    return c5 + 100 * sum_pen

def compute_c5_and_weighted_grad(h, T=100.0):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full')

    c5_smooth = (C_max + logsum / T) * dx
    
    # Weights
    w = np.exp(T * (C - C_max))
    w /= np.sum(w)
    
    # Weighted gradient

            grad[:n-s] += w[m] * (1.0 - h[s:])
            grad[s:] -= w[m] * h[:n-s]
    
    return c5_smooth, grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def soft_max_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full')
    C_max = np.max(C)
    scaled = T * (C - C_max)
    # For numerical stability
    exp_scaled = np.exp(scaled - np.max(scaled))
    logsum = np.log(np.sum(exp_scaled)) + np.max(scaled)
    val = (C_max + logsum / T) * dx
    
    w = exp_scaled / np.sum(exp_scaled)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        d = m - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                grad[d:] += wm * (1.0 - h[:L])
                grad[:L] -= wm * h[d:]
        else:
            s = -d
            L = n - s
            if L > 0:
                grad[:L] += wm * (1.0 - h[s:])
                grad[s:] -= wm * h[:L]
    
    return val, grad * dx

    mu = 500.0

        h = np.clip(x, 0, 1)
        val, _ = soft_max_obj_grad(h, T)
        val += mu * (np.sum(h) - target) ** 2

import numpy as np
from scipy.optimize import minimize
import time
import math

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def soft_max_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full')
    C_max = np.max(C)
    scaled = T * (C - C_max)
    exp_scaled = np.exp(scaled - np.max(scaled))
    logsum = np.log(np.sum(exp_scaled)) + np.max(scaled)
    val = (C_max + logsum / T) * dx
    w = exp_scaled / np.sum(exp_scaled)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        d = m - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                grad[d:] += wm * (1.0 - h[:L])
                grad[:L] -= wm * h[d:]
        else:
            s = -d
            L = n - s
            if L > 0:
                grad[:L] += wm * (1.0 - h[s:])
                grad[s:] -= wm * h[:L]
    return val, grad * dx

def lbfgs_obj(x, T, mu, target):
    h = np.clip(x, 0.0, 1.0)
    val, _ = soft_max_obj_grad(h, T)
    val += mu * (np.sum(h) - target) ** 2
    return val

def lbfgs_grad(x, T, mu, target):
    h = np.clip(x, 0.0, 1.0)
    _, grad = soft_max_obj_grad(h, T)
    grad += 2.0 * mu * (np.sum(h) - target)
    return grad

def nm_obj(x, n):
    h = project_feasible(x, n)
    return compute_c5(h)

def gradient_descent(h, n, T, max_iters, t0, budget_s, margin):
    alpha = 0.02 / math.sqrt(T)
    c5_cur = compute_c5(h)
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > budget_s - margin:
            break
        _, grad = soft_max_obj_grad(h, T)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            alpha = min(alpha * 1.08, 0.1)
            stall = 0
        else:
            alpha *= 0.6
            stall += 1
            if stall > 30:
                break
    return h, c5_cur
