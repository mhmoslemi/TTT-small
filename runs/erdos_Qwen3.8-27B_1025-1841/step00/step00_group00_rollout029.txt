def project_onto_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} in L2."""
    # Binary search on lambda
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0.0, 1.0)

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    # Project onto {x: 0<=x<=1, sum(x)=target_sum}
    # Binary search on shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_active(h, k_max):
    n = len(h)
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n - lag]
    else:
        lag = -lag
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    return g

def upsample(h, target_n):
    n = len(h)
    if target_n == n:
        return h.copy()
    # Linear interpolation for upsampling
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, target_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def optimize_h(h, time_budget, seed=42):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Subgradient descent
    alpha = 0.1  # initial step size
    decay = 0.999
    patience = 500
    no_improve = 0
    
    start = time.time()
    for it in range(100000):
        if time.time() - start > time_budget:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        c5 = corr[k_max] * dx
        
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Restart with perturbation
            h = best_h + rng.randn(n) * 0.01
            h = project_feasible(h, target_sum)
            alpha = 0.1
            no_improve = 0
            continue
        
        g = compute_grad_active(h, k_max)
        h_new = h - alpha * g

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final adjustment for exact sum
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.any():
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_grad_active(h, k_max):
    n = len(h)
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] += 1.0 - h[lag:]
            g[lag:] -= h[:n - lag]
    else:
        lag = -lag
        if lag < n:
            g[lag:] += 1.0 - h[:n - lag]
            g[:n - lag] -= h[lag:]
    return g

def compute_weighted_grad(h, corr, top_m=3):
    n = len(h)
    top_indices = np.argsort(corr)[-top_m:]
    weights = np.zeros(len(corr))
    for idx in top_indices:
        weights[idx] = 1.0
    g = np.zeros(n)
    for idx in top_indices:
        lag = idx - (n - 1)
        if lag >= 0 and lag < n:
            g[:n - lag] += 1.0 - h[lag:]
            g[lag:] -= h[:n - lag]
        elif lag < 0 and -lag < n:
            lag = -lag
            g[lag:] += 1.0 - h[:n - lag]
            g[:n - lag] -= h[lag:]
    g /= top_m
    return g
