```python
import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair(h, n):
    """Ensure 0<=h[i]<=1 and sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if current_sum > 0:
        # Scale to fix the sum, but maintain bounds
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        # If clipping changed the sum, do it again
        for _ in range(10):
            current_sum = np.sum(h)
            if abs(current_sum - target_sum) < 1e-10:
                break
            # Adjust: add/subtract small amounts to elements not at boundary
            diff = target_sum - current_sum
            mask = (h > 0) & (h < 1)
            if np.sum(mask) == 0:
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def binary_repair(h, n):
    """Make h binary and ensure exactly n/2 ones"""
    threshold = 0.5
    h_bin = (h >= threshold).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Flip some 1s to 0s (those closest to 0.5)
        ones = np.where(h_bin == 1)[0]
        # Sort by value (ascending) - flip the smallest
        order = np.argsort(h[ones])
        for i in range(n_ones - target):
            h_bin[ones[order[i]]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])  # flip the largest 0s
        for i in range(target - n_ones):
            h_bin[zeros[order[i]]] = 1
    return h_bin
```

```python
                stall += 1
            else:
                h[i], h[j] = h[j], h[i]
                stall += 1
        temp *= sa_decay
        if stall > 3000:
            break
    return best_h, best_c5

def frac_perturb_search(h, rng, max_iters, step=0.1):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for it in range(max_iters):
        h_new = h + rng.standard_normal(n) * step
        h_new = repair_frac(h_new, n)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5:
            h = h_new
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        elif new_c5 < c5 + 1e-5:
            h = h_new
            c5 = new_c5
    return best_h, best_c5

def block_flip_search(h, rng, max_iters, block_size=3):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for it in range(max_iters):
        i = rng.randint(0, max(1, n - block_size))
        idx = np.arange(i, i + block_size)
        n_ones = int(np.sum(h[idx]))
        n_zeros = block_size - n_ones
        # Flip a random subset to try to balance
        if n_ones != n_zeros:
            target_ones = block_size // 2
            if n_ones > target_ones:
                # Need to flip some 1s to 0s
                to_flip = block_size - 2 * target_ones
                ones_in_block = idx[h[idx] == 1]
                if len(ones_in_block) >= to_flip:
                    chosen = rng.choice(ones_in_block, size=min(to_flip, len(ones_in_block)), replace=False)
                else:
                    chosen = ones_in_block
            else:
                to_flip = 2 * target_ones - block_size
                zeros_in_block = idx[h[idx] == 0]
                if len(zeros_in_block) >= to_flip:
                    chosen = rng.choice(zeros_in_block, size=min(to_flip, len(zeros_in_block)), replace=False)
                else:
                    chosen = zeros_in_block
            h[chosen] = 1.0 - h[chosen]
        new_c5 = compute_c5(h)
        if new_c5 > c5 + 0.01:
            # Revert
            h[chosen] = 1.0 - h[chosen]
            continue
        c5 = new_c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
        h_init = repair_frac(h_init, n_init)
    except Exception:
        n_init = 70
        h_init = np.full(n_init, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # First: binary local search at initial resolution
    h_bin = to_binary(h_init, n_init)
    c5_b = compute_c5(h_bin)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
        h_init = h_bin.copy()

    elapsed = time.time() - t0
    t1 = max(120, budget_s * 0.15)
    h_cur, c5_cur = binary_swap_search(h_init, rng, max_iters=200000, sa_temp=0.004, sa_decay=0.9998)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    # Multi-resolution: go up in resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining < 60:
        return best_h, best_c5, len(best_h)

    # Plan resolutions
    cur_n = n_init
    resolutions = []
    n = cur_n
    while n < 600 and remaining > 100:
        n = int(n * 1.5) + 1
        if n % 2 == 1:
            n += 1
        resolutions.append(n)
        remaining -= 200
    resolutions = [r for r in resolutions if r <= 500][:4]

    for target_n in resolutions:
        if len(resolutions) > 4:
            resolutions = resolutions[:4]
            break
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    time_per_res = max(60, remaining / (len(resolutions) + 1))

    cur_h = h_cur.copy()
    cur_n = n_init

    for target_n in resolutions:
        elapsed = time.time() - t0
        remaining = budget_s - elapsed
        if remaining < 80:
            break

        # Interpolate
        cur_h = interp_h(cur_h, cur_n, target_n)
        cur_n = target_n

        # Binary optimization
        h_b = to_binary(cur_h, target_n)
        c5_b = compute_c5(h_b)
        # Also try starting from rounded fractional
        h_round = (cur_h >= 0.5).astype(float)
        h_round = to_binary(h_round, target_n)
        c5_r = compute_c5(h_round)
        start_h = h_b if c5_b <= c5_r else h_round
        start_c5 = min(c5_b, c5_r)

        # Binary local search
        max_iter = int(150000 * time_per_res / 200)
        max_iter = max(50000, min(max_iter, 300000))
        h_opt, c5_opt = binary_swap_search(start_h, rng, max_iters=max_iter, sa_temp=0.003, sa_decay=0.9998)
        if c5_opt < start_c5:
            cur_h = h_opt
            start_c5 = c5_opt

        # Fractional refinement
        elapsed = time.time() - t0
        frac_time = min(120, max(30, (budget_s - elapsed) * 0.2))
        max_frac_iter = int(frac_time * 2000)
        max_frac_iter = max(5000, min(max_frac_iter, 50000))
        h_frac_opt, c5_frac = frac_perturb_search(cur_h, rng, max_iters=max_frac_iter, step=0.08)
        if c5_frac < start_c5:
            cur_h = h_frac_opt
            start_c5 = c5_frac

        # Update best
        final_c5 = compute_c5(cur_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = cur_h.copy()

    # Final polishing at best resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 60:
        n_best = len(best_h)
        # Binary polish
        h_b = to_binary(best_h, n_best)
        elapsed = time.time() - t0
        max_iter = int(remaining * 0.4 * 2000)
        max_iter = max(30000, min(max_iter, 200000))
        h_opt, c5_opt = binary_swap_search(h_b, rng, max_iters=max_iter, sa_temp=0.002, sa_decay=0.9999)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        # Fractional polish
        elapsed = time.time() - t0
        remaining2 = budget_s - elapsed
        if remaining2 > 30:
            max_frac = int(remaining2 * 0.5 * 1500)
            max_frac = max(3000, min(max_frac, 30000))
            h_frac, c5_frac = frac_perturb_search(best_h, rng, max_iters=max_frac, step=0.04)
            if c5_frac < best_c5:
                best_c5 = c5_frac
                best_h = h_frac.copy()

    # Validate and return
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
