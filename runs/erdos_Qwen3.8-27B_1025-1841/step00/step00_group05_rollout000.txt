from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_mid = np.clip(h - mid, 0.0, 1.0).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    np.clip(h - lam, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_res(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new, n_new)

class ObjWrapper:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        h = project_feasible(x, self.n)
        return compute_c5(h, self.n)

                        best_c5 = c5_t
                        best_h = h_t.copy()
            if best_h is not h:
                h = best_h
                improved = True
        if not improved:
            break
    return h
