def project(h, n):
    # Project onto {0 <= h <= 1, sum(h) = n/2}
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n / 2.0
    if s > target:
        # Need to reduce
        excess = s - target
        # Reduce from largest values
        order = np.argsort(h)[::-1]
        for i in order:
            red = min(excess, h[i])
            h[i] -= red
            excess -= red
            if excess <= 1e-10:
                break
    elif s < target:
        deficit = target - s
        order = np.argsort(h)
        for i in order:
            add = min(deficit, 1.0 - h[i])
            h[i] += add
            deficit -= add
            if deficit <= 1e-10:
                break
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    # Find all max indices
    tol = 1e-10
    max_idx = np.where(np.abs(corr - max_c) <= tol)[0]
    
    grad = np.zeros(n)
    for j in max_idx:
        lag = j - (n - 1)
        # Term 1: (1-h[m-lag]) for valid m
        ms = max(0, lag)
        me = min(n, n + lag)
        if ms < me:
            grad[ms:me] += (1.0 - h[ms - lag: me - lag])
        # Term 2: -h[m+lag] for valid m
        ms2 = max(0, -lag)
        me2 = min(n, n - lag)
        if ms2 < me2:
            grad[ms2:me2] -= h[ms2 + lag: me2 + lag]
    
    grad /= max(1, len(max_idx))
    grad *= dx
    return max_c * dx, grad

def obj_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    tol = 1e-10
    max_idx = np.where(np.abs(corr - max_c) <= tol)[0]
    grad = np.zeros(n)
    for j in max_idx:
        lag = j - (n - 1)
        ms = max(0, lag)
        me = min(n, n + lag)
        if ms < me:
            grad[ms:me] += (1.0 - h[ms - lag: me - lag])
        ms2 = max(0, -lag)
        me2 = min(n, n - lag)
        if ms2 < me2:
            grad[ms2:me2] -= h[ms2 + lag: me2 + lag]
    grad /= max(1, len(max_idx))
    grad *= dx
    return max_c * dx, grad

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def sum_constraint_jac(x):
    return np.ones(len(x))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] -= diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] -= diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def interp_res(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    if n == 1:
        return np.full(new_n, h[0])
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n - 1, new_n)
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h_init, n, max_iter):
    h_init = project_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    res = minimize(
        obj_and_grad, h_init, method='SLSQP',
        bounds=bounds, constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.full(42, 0.5)

    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    def time_ok(minutes_threshold):
        return (time.time() - t0) < (budget_s - minutes_threshold)

    # Optimize at initial resolution
    if time_ok(30):
        try:
            h_opt, c5 = slsqp_optimize(best_h, n_best, 300)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
        except Exception:
            pass

    # Multi-resolution path
    targets = [84, 168, 336, 504]
    targets = [t for t in targets if t > n_best]

    for n_t in targets:
        if not time_ok(45):
            break
        est = (n_t * n_t * 400) / 5.0e7
        max_it = int(min(500, max(30, (time_ok and (budget_s - (time.time() - t0)) * 0.5 * 5.0e7 / (n_t * n_t)) or 400)))
        max_it = max(30, min(max_it, 500))

        h_interp = interp_res(best_h, n_t)
        h_interp = project_feasible(h_interp, n_t)

        try:
            h_opt, c5 = slsqp_optimize(h_interp, n_t, max_it)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
                n_best = n_t
        except Exception:
            pass

    # Second pass: go back to a resolution slightly above current for fine-tuning
    if n_best < 336 and time_ok(60):
        n_fine = max(n_best * 2, 168)
        h_interp2 = interp_res(best_h, n_fine)
        h_interp2 = project_feasible(h_interp2, n_fine)
        try:
            h_opt, c5 = slsqp_optimize(h_interp2, n_fine, 300)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
                n_best = n_fine
        except Exception:
            pass

    # Perturbation and re-optimization at best resolution
    attempts = 0
    while (time.time() - t0) < (budget_s - 25) and attempts < 40:
        h_pert = best_h.copy()
        n_pert = max(1, n_best // 6)
        idx = rng.choice(n_best, size=n_pert, replace=False)
        scale = 0.15 + 0.1 * (attempts % 3)
        h_pert[idx] += rng.uniform(-scale, scale, size=n_pert)
        h_pert = project_feasible(h_pert, n_best)
        try:
            h_opt, c5 = slsqp_optimize(h_pert, n_best, 150)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
        except Exception:
            pass
        attempts += 1

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
