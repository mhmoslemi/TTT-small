    """Project h onto {0 ≤ x ≤ 1, sum(x) = n/2}."""
    target = n / 2.0
    
    def f(lam):
        x = np.clip(h - lam, 0, 1)
        return np.sum(x)
    
    # Bisection for lam
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if f(mid) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2} using bisection on shift."""
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def subgradient_at_lag(h, n, lag):
    """Compute subgradient of C(lag) = sum_{valid i} h[i]*(1-h[i+lag]) w.r.t. h."""
    grad = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            m = n - lag
            grad[:m] += (1.0 - h[lag:])
            grad[lag:] -= h[:m]
    else:
        r = -lag
        if r == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            m = n - r
            grad[r:] += (1.0 - h[:m])
            grad[:m] -= h[r:]
    return grad

    # For the tent: sum ≈ n * mean(h) = n * 0.5 = n/2. So it should be close.
    h = project_feasible(h, n)
    return h

def make_alternating(n, a_frac):
    """Create alternating pattern with values a and 1-a."""
    h = np.zeros(n)
    h[::2] = a_frac
    h[1::2] = 1.0 - a_frac
    h = project_feasible(h, n)
    return h

def upsample(h_src, n_out):
    n_src = len(h_src)
    x_src = np.arange(n_src) / n_src * 2.0
    x_out = np.arange(n_out) / n_out * 2.0
    return np.interp(x_out, x_src, h_src)

    resolutions = [200, 400, 800]
    inits_names = []

    candidates = []

    for n in resolutions:

    for n in resolutions:
        h_up = upsample(h_parent, n)
        h_up = project_feasible(h_up, n)
        candidates.append(('parent', h_up))
    
    # Alternating with various parameters
    for n in [400]:
        for afrac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            candidates.append((f'alt_{afrac}', make_alternating(n, afrac)))

    for n in [400]:
        for width in [0.3, 0.5, 0.7, 1.0]:
            x = np.linspace(0, 2, n, endpoint=False)
            h = np.exp(-((x - 1.0)**2) / (2 * width**2))
            h = h / np.sum(h) * (n / 2.0)  # Scale so sum = n/2
            h = project_feasible(h, n)
            candidates.append((f'gauss_{width}', h))

    for name, h_cand in candidates:

        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = len(h_cand)
    
    # Now optimize the best candidates
    time_per_opt = (budget_s * 0.85 - (time.time() - t0)) / max(len(candidates), 1)
    time_per_opt = max(time_per_opt, 5)  # At least 5s each
    
    for name, h_cand in candidates:
        if time.time() - t0 > budget_s * 0.9:
            break
        n_c = len(h_cand)
        t_budget = min(time_per_opt, budget_s * 0.9 - (time.time() - t0))

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_c
    
    # Final check
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
