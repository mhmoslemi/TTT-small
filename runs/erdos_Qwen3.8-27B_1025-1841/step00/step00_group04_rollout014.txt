```python
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, S):
    """Project onto {0<=h<=1, sum(h)=S} using bisection on offset"""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_correlates(h):
    """Return all lag values of correlate(h, 1-h, 'full')"""
    return np.correlate(h, 1 - h, mode='full')

def compute_gradient_subgradient(correlates, h, n, T=0.5):
    """Compute a smooth gradient using softmax weighting over active lags"""
    # correlates has length 2n-1, index n-1+k is lag k
    weights = np.exp(correlates / T - np.max(correlates) / T)
    weights /= weights.sum()
    
    # grad[j] = sum_k weights[k+n-1] * [(1-h[j-k]) - h[j+k]]
    # For lag k (index = n-1+k in correlates array):
    # grad[j] += w * [(1-h[j-k] if 0<=j-k<n else 0) - h[j+k] if 0<=j+k<n else 0]
    
    grad = np.zeros(n)
    for k_idx in range(len(correlates)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)  # actual lag
        # (1-h[j-k]) for valid j-k
        if k >= 0:

            # j-k < n means j < n+k (always true since j < n)
            grad[k:] += w * (1 - h[:n-k])
            # h[j+k] for valid j+k
            # j+k < n means j < n-k
            grad[:n-k] -= w * h[k:]
        else:
            kk = -k
            # j-k = j+kk >= 0 always
            # j+kk < n means j < n-kk
            grad[:n-kk] += w * (1 - h[kk:])
            # j+k = j-kk >= 0 means j >= kk
            # j-kk < n always
            grad[kk:] -= w * h[:n-kk]
    
    return grad

        c5 = np.max(corr) * dx

            best_h = h.copy()
            no_improve = 0
        else:

        grad = compute_gradient_subgradient(corr, h, n, T=T)

        h_new = h - lr * grad

            h[mask] += 0.1 * np.random.randn(mask.sum())
            h = project_to_constraint(h, S)
            lr = 0.1
            no_improve = 0
    
    return h
```

```python
def compute_grad(corr, h, n, top_k=5):
    """Compute gradient using top k lags"""
    n_corr = len(corr)  # 2n-1
    mid = n - 1
    # Find top k lags
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[top_indices >= 1 and top_indices <= n_corr-2]  # exclude boundaries
    
    grad = np.zeros(n)
    for idx in top_indices:
        k = idx - mid  # lag
        # grad[j] += (1-h[j-k]) if valid - h[j+k] if valid
        if k >= 0:
            # (1-h[j-k]) valid for j >= k
            grad[k:] += (1 - h[:n-k])
            # h[j+k] valid for j < n-k
            grad[:n-k] -= h[k:]
        else:
            kk = -k
            # (1-h[j+kk]) valid for j < n-kk
            grad[:n-kk] += (1 - h[kk:])
            # h[j-kk] valid for j >= kk
            grad[kk:] -= h[:n-kk]
    
    return grad
```

```python
import numpy as np
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_constraint(h, S):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(k, h, n):
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += 1.0 - h[:n - k]
            g[:n - k] -= h[k:]
    else:
        kk = -k
        if kk < n:
            g[:n - kk] += 1.0 - h[kk:]
            g[kk:] -= h[:n - kk]
    return g

def opt_cont(h0, dx, S, tb, seed):
    t0 = time.time()
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = project_to_constraint(h0.astype(float).copy(), S)
    bh = h.copy()
    bc5 = evaluate_c5(h, dx)
    lr = 0.004
    mom = 0.9
    v = np.zeros(n)
    stall = 0
    while time.time() - t0 < tb:
        corr = np.correlate(h, 1 - h, mode='full')
        c5n = float(corr.max()) * dx
        if c5n < bc5 - 1e-12:
            bc5 = c5n
            bh = h.copy()
            stall = 0
        else:
            stall += 1
        mid = n - 1
        idxs = np.argsort(corr)[-4:]
        g = np.zeros(n)
        for ix in idxs:
            k = int(ix) - mid
            if abs(k) < n:
                g += grad_lag(k, h, n)
        g /= max(1, len(idxs))
        v = mom * v - lr * g
        h = project_to_constraint(h + v, S)
        if stall > 80:
            h = bh + 0.02 * rng.randn(n)
            h = project_to_constraint(h, S)
            v[:] = 0.0
            lr = 0.004
            stall = 0

            lr = 0.004

            mom = 0.9
```
