    resolutions = [140, 200, 280, 400, 600]
    for n_new in resolutions:
        if remaining() < 80:
            break
        h_up = upsample(best_h, n_new)
        h_up_bin = enforce_binary(h_up, n_new, rng)

        # Short local search
        alloc_ls = min(60, int(remaining() * 0.2))
        h_ls, c_ls = swap_local_search(h_up_bin, n_new, alloc_ls, rng)
        if c_ls < best_c5:
            best_c5, best_h, best_n = c_ls, h_ls.copy(), n_new

        # Simulated annealing
        if remaining() > 60:
            alloc_sa = min(150, int(remaining() * 0.3))
            h_sa, c_sa = simulated_annealing(h_ls, n_new, alloc_sa, t_init=0.6, t_floor=0.1, rng=rng)
            h_sa, c_sa = swap_local_search(h_sa, n_new, min(30, int(remaining() * 0.1)), rng)
            if c_sa < best_c5:
                best_c5, best_h, best_n = c_sa, h_sa.copy(), n_new

    # Phase 4: Continuous refinement at best resolution
    if remaining() > 40 and best_n <= 300:
        h_cont, c_cont = continuous_refine(best_h, best_n, min(120, int(remaining() * 0.4)), rng)
        if c_cont < best_c5:
            best_c5, best_h = c_cont, h_cont

    # Phase 5: Final local search polish on best solution
    if remaining() > 20:
        h_bin_final = enforce_binary(best_h, best_n, rng)
        h_f, c_f = swap_local_search(h_bin_final, best_n, min(40, int(remaining() * 0.5)), rng)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
        # Also try continuous
        h_cf, c_cf = continuous_refine(best_h, best_n, min(20, max(0, int(remaining() * 0.3))), rng)
        if c_cf < best_c5:
            best_c5, best_h = c_cf, h_cf

    # Final validation
    h_final = np.clip(best_h, 0.0, 1.0)
    n_final = len(h_final)
    dx = 2.0 / n_final
    s = np.sum(h_final)
    if s > 0:
        h_final *= (n_final / 2.0) / s
        h_final = np.clip(h_final, 0.0, 1.0)
    final_c5 = compute_c5(h_final)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        h_final = best_h
        best_c5 = compute_c5(best_h)

    return h_final, best_c5, len(h_final)
