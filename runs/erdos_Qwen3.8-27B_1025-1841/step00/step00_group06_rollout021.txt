        # or distribute evenly
        h += diff / n
        h = np.clip(h, 0, 1)

            h[mask] += diff2 / mask.sum()
            h = np.clip(h, 0, 1)

        h = np.clip(h, 0, 1)

        if diff2 < -1e-10:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def get_worst_lags(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get indices of top_k largest values
    idx = np.argsort(corr)[-top_k:]
    return idx, corr[idx]

def project_sum(h, n):
    """Project h to satisfy sum(h) = n/2, keeping 0 <= h <= 1"""
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return np.clip(h, 0, 1)
    if diff > 0:
        # Need to increase: add to elements with room
        mask = h < 1.0
        if mask.sum() == 0:
            return np.clip(h, 0, 1)
        h[mask] += diff / mask.sum()
    else:
        # Need to decrease: remove from elements with mass
        mask = h > 0.0
        if mask.sum() == 0:
            return np.clip(h, 0, 1)
        h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)

def subgradient_step(h, n, lr=0.05, n_focus=3):
    """One subgradient step focusing on worst lags"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get worst lags
    idx = np.argsort(corr)[-n_focus:]
    
    grad = np.zeros(n)
    for m in idx:
        lag = m - (n - 1)
        # c[m] = sum_i h[i+lag] * (1-h[i]) for valid i
        # d(c[m])/d(h[j]) = (1-h[j-lag]) for j where j-lag valid and j in [0,n)
        #                   - h[j+lag] for j where j+lag valid and j in [0,n)
        for i in range(n):
            if 0 <= i + lag < n:
                grad[i + lag] += (1 - h[i])
                grad[i] -= h[i + lag]
    
    h_new = h - lr * grad
    h_new = project_sum(h_new, n)
    return h_new

    best = h.copy()

                trial[j] += amt

            d = rng.uniform(-0.15, 0.15)

            trial[j] = np.clip(trial[j] + d, 0, 1)

        val = compute_c5(trial, n)
        if val < best_val - 1e-10:

            best = trial.copy()

def simulated_annealing(h, n, T_start, T_end, n_iters, rng):
    """Simulated annealing"""
    best = h.copy()
    best_val = compute_c5(h, n)
    current = h.copy()
    current_val = best_val

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        if mask.sum() > 0:
            h = h.copy()
            h[mask] += diff / mask.sum()
    else:
        mask = h > 0.0
        if mask.sum() > 0:
            h = h.copy()
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def grad_at_lag(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        L = -lag
        grad[:n-L] += (1.0 - h[L:])
        grad[L:] -= h[:n-L]
    return grad

def subgradient_hurst(h, n, lr_schedule, n_focus=5, n_steps=5000):
    best = h.copy()
    best_val = compute_c5(h, n)
    dx = 2.0 / n
    
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        # Get worst lags
        idx = np.argpartition(corr, -n_focus)[-n_focus:]
        weights = corr[idx]  # Use weights proportional to how bad each lag is
        total_w = np.sum(weights)
        if total_w < 1e-10:
            break
        weights = weights / total_w
        
        grad = np.zeros(n)
        for w, m in zip(weights, idx):
            lag = m - (n - 1)
            grad += w * grad_at_lag(h, n, lag)
        
        lr = lr_schedule(step, n_steps)
        h_new = h - lr * grad
        h_new = project_sum(h_new, n)
        
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best = h_new.copy()
        h = h_new
    
    return best, best_val

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 1.0
        cnt = mask.sum()
    else:
        mask = h > 0.0
        cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def local_search(h, n, n_iters, rng, use_directed=True):
    cur = h.copy()
    cur_val = compute_c5(cur, n)
    dx = 2.0 / n
    target = n / 2.0

    for it in range(n_iters):
        if use_directed and it % 3 == 0:
            # Directed move targeting worst lag
            corr = np.correlate(cur, 1.0 - cur, mode='full') * dx
            worst_m = np.argmax(corr)
            lag = worst_m - (n - 1)
            if lag >= 0:
                contribs = cur[lag:] * (1.0 - cur[:n - lag])
                s = contribs.sum()
                if s > 1e-10:
                    idx = np.where(contribs > 1e-10)[0]
                    if len(idx) > 0:
                        probs = contribs[idx] / contribs[idx].sum()
                        j = idx[rng.choice(len(idx), p=probs)]
                        source = j + lag
                        dest = j
                        amt = min(cur[source], rng.uniform(0.02, 0.12))
                        if cur[dest] + amt <= 1.0 + 1e-9:
                            trial = cur.copy()
                            trial[source] -= amt
                            trial[dest] += amt
                            val = compute_c5(trial, n)
                            if val < cur_val:
                                cur_val = val
                                cur = trial
                            continue
            else:
                L = -lag

                if s > 1e-10:

                trial[i] += amt

def binarize(h, n):
    """Project to nearest binary-like solution (one fractional value)"""
    hb = (h > 0.5).astype(float)
    # Adjust to hit target
    target = n / 2.0
    diff = target - hb.sum()
    if abs(diff) < 1e-10:
        return hb
    # Fix by adjusting
    if diff > 0:
        # Need more mass: find a 0 to make 1 (or partial)
        zeros = np.where(hb == 0)[0]
        if len(zeros) > 0:
            j = zeros[0]
            hb[j] = min(1.0, diff)
            diff -= hb[j]
            if diff > 1e-10:
                zeros = np.where(hb < 1)[0]
                if len(zeros) > 0:
                    hb[zeros[0]] += diff
    else:
        diff = -diff

        if len(ones) > 0:
            j = ones[-1]
            hb[j] = max(0.0, 1.0 - diff)
            diff -= (1.0 - hb[j])
            if diff > 1e-10:
                ones = np.where(hb > 0)[0]
                if len(ones) > 0:
                    hb[ones[-1]] -= diff
    return np.clip(hb, 0, 1)
