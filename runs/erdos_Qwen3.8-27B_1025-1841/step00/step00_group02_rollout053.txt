            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(45, 0.5)
    
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_init)
    
    # Resolution to optimize at
    n_target = min(200, max(n_init, 120))
    
    # Upsample to target resolution
    if n_target != n_init:
        x_old = (np.arange(n_init) + 0.5) * (2.0/n_init)
        x_new = (np.arange(n_target) + 0.5) * (2.0/n_target)
        h = np.interp(x_new, x_old, best_h)
    else:
        h = best_h.copy()
    
    n = n_target
    h = project_simplex_hypercube(h, n)
    c5_current = compute_c5(h, n)
    
    if c5_current < best_c5:
        best_c5 = c5_current
        best_h = h.copy()
    
    # Optimized gradient descent with soft-max
    tau = 0.02  # temperature for soft max
    lr = 0.005
    momentum = 0.9
    vel = np.zeros(n)
    
    for it in range(3000):
        if time.time() - t0 > budget_s - 10:
            break
        
        # Compute all lag values
        c5_all = compute_c5_all(h, n)  # array of length 2n-1
        m = np.max(c5_all)
        
        # Soft max probabilities
        p = np.exp((c5_all - m) / tau)
        p /= np.sum(p)
        
        # Gradient of soft max
        # d/dh[j] of sum_s p_s * c5_all[s]
        # = sum_s p_s * dx * dC(s)/dh[j]
        # where c5_all[s] = dx * C(s), C(s) = sum_i h[i]*(1-h[i+s])
        # dC(s)/dh[j] = (1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)
        
        dx = 2.0 / n
        grad = np.zeros(n)
        
        # For each shift s (indexed as s - (n-1) in c5_all):
        # s ranges from -(n-1) to (n-1)
        for s in range(-(n-1), n):
            if p[s + n - 1] < 1e-6:
                continue
            # dC(s)/dh[j] for j = 0..n-1
            # Term 1: (1-h[j+s]) if 0 <= j+s < n
            # Term 2: -h[j-s] if 0 <= j-s < n
            
            # Vectorized:
            # j+s in [0, n-1] => j in [-s, n-1-s], intersect [0, n-1]
            j_start1 = max(0, -s)
            j_end1 = min(n-1, n-1-s)
            if j_start1 <= j_end1:
                js = np.arange(j_start1, j_end1+1)
                grad[js] += p[s+n-1] * dx * (1 - h[js + s])
            
            # j-s in [0, n-1] => j in [s, s+n-1], intersect [0, n-1]
            j_start2 = max(0, s)
            j_end2 = min(n-1, s+n-1)
            if j_start2 <= j_end2:
                js = np.arange(j_start2, j_end2+1)
                grad[js] -= p[s+n-1] * dx * h[js - s]
        
        # Momentum update
        vel = momentum * vel - lr * grad
        h_new = h + vel
        h_new = project_simplex_hypercube(h_new, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new < c5_current:
            h = h_new
            c5_current = c5_new
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
        else:
            lr *= 0.95
            vel *= 0.5
        
        # Adaptive tau
        if it % 500 == 0 and it > 0:
            tau *= 0.9
    
    return best_h, best_c5, len(best_h)
