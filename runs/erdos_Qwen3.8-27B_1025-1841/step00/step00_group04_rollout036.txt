import numpy as np
from scipy.optimize import minimize
import time

_N = 0

def project_feasible(h, n):
    target = n / 2.0
    lo = float(h.min()) - 1.0
    hi = float(h.max())
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())

def smoothed_max_val(corr, dx, tau):
    x = tau * corr * dx
    x_max = float(x.max())
    if x_max > 500:
        return float((corr * dx).max())
    return float((x_max + np.log(np.exp(x - x_max).sum())) / tau)

def corr_grad_single(x, k_idx, n):
    s = k_idx - (n - 1)
    m = abs(s)
    if m == 0:
        return 1.0 - 2.0 * x
    g = np.zeros(n)
    if s >= 0:
        g[m:] += (1.0 - x[:n - m])
        g[:n - m] -= x[m:]
    else:
        g[:n - m] += (1.0 - x[m:])
        g[m:] -= x[:n - m]
    return g

def opt_objective(x, tau, dx, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smoothed_max_val(corr, dx, tau)

def opt_gradient(x, tau, dx, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    x_scaled = tau * corr * dx
    x_max = float(x_scaled.max())
    if x_max > 500:
        k_star = int(np.argmax(corr))
        return dx * corr_grad_single(x, k_star, n)
    weights = np.exp(x_scaled - x_max)
    wsum = float(weights.sum())
    if wsum < 1e-300:
        k_star = int(np.argmax(corr))
        return dx * corr_grad_single(x, k_star, n)
    weights /= wsum
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if weights[k_idx] < 1e-14:
            continue
        g += weights[k_idx] * tau * dx * corr_grad_single(x, k_idx, n)
    return g

def sum_constraint(x):
    return np.sum(x) - _N / 2.0

def sum_constraint_jac(x):
    return np.ones(_N)

def do_slsqp(h, tau, dx, n, bounds, constraints, maxiter, ftol):
    result = minimize(opt_objective, h, jac=opt_gradient, args=(tau, dx, n),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
    return result.x

def run(seed=42, budget_s=1300, **kwargs):
    global _N
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.ones(42) / 2.0

    n = len(h)
    _N = n
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]

    # Main annealing
    taus = [3000, 1500, 700, 350, 150, 70, 30, 10, 4]
    for tau in taus:
        if time.time() - t0 > budget_s - 150:
            break
        h_new = do_slsqp(h, tau, dx, n, bounds, constraints, 250, 1e-14)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    # Random restarts
    attempts = 0
    while time.time() - t0 < budget_s - 120:
        attempts += 1
        scale = 0.015 + 0.02 * rng.random()
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n)
        for tau in [300, 80, 20]:

            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_pert = h_new

    # Try upscaling to 2n if time permits
    if time.time() - t0 < budget_s - 180:
        n2 = n * 2
        _N = n2
        dx2 = 2.0 / n2
        h2 = np.repeat(best_h, 2)
        h2 = project_feasible(h2, n2)

        best_c5_2 = compute_c5(h2, dx2)
