    c5_best = compute_c5(h_best, n_best)

    def elapsed():
        return time.time() - t_start

    # Phase 1: Local refinement on initial grid
    h_best = local_refine(h_best, n_best, budget_s * 0.3)
    c5_best = compute_c5(h_best, n_best)

    # Phase 2: DE for global exploration on same grid
    if elapsed() < budget_s * 0.5:
        h_de = de_search(h_best, n_best, seed, budget_s * 0.3)
        c5_de = compute_c5(h_de, n_best)
        if c5_de < c5_best:
            c5_best = c5_de
            h_best = h_de

    # Phase 3: Multi-start from random perturbations
    if elapsed() < budget_s * 0.7:
        for restart in range(8):
            if elapsed() > budget_s * 0.8:
                break
            noise = np.random.normal(0, 0.15, n_best)
            h_try = project_sum(h_best + noise, n_best)
            h_try = local_refine(h_try, n_best, max(5, budget_s * 0.05))
            c5_try = compute_c5(h_try, n_best)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try

    # Phase 4: Try higher resolution grids
    if elapsed() < budget_s * 0.8:
        for n_new in [140, 280, 560]:
            if elapsed() > budget_s * 0.92:
                break
            x_old = np.linspace(0.0, 2.0, n_best)
            x_new = np.linspace(0.0, 2.0, n_new)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_sum(h_interp, n_new)
            h_refined = local_refine(h_interp, n_new, max(5, budget_s * 0.06))
            c5_new = compute_c5(h_refined, n_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_refined
                n_best = n_new

    # Phase 5: Final polish
    if elapsed() < budget_s * 0.95:
        h_best = local_refine(h_best, n_best, max(5, budget_s * 0.05))
        c5_best = compute_c5(h_best, n_best)

    # Final projection and evaluation
    h_best = project_sum(h_best, n_best)
    c5_best = compute_c5(h_best, n_best)

    return (h_best, c5_best, n_best)
