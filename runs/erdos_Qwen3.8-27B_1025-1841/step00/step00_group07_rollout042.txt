    hi = v.max()

        if s > target_sum:

        else:

    return np.clip(v - lam, 0.0, 1.0)

def compute_subgradient(h, k_star):
    """Subgradient of max overlap at h, given argmax index k_star"""
    n = len(h)
    offset = k_star - (n - 1)
    indices = np.arange(n)
    grad = np.zeros(n)

    tgt = n // 2
    if s > tgt:
        o = np.where(h_bin == 1)[0]
        h_bin[rng.choice(o, size=s - tgt, replace=False)] = 0.0
    elif s < tgt:
        z = np.where(h_bin == 0)[0]
        h_bin[rng.choice(z, size=tgt - s, replace=False)] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = float('inf')

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = make_random_binary(80, rng)

    n0 = len(h0)
    h0 = project_onto_constraint(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Continuous optimization from initial
    h_r, c5_r = optimize_continuous(h0, n0 / 2.0, 12000, rng, start_time, budget_s)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r

    # Phase 2: Binary search from best
    h_bin = to_binary(best_h, rng)
    h_r, c5_r = binary_swap_search(h_bin, rng, 30000, start_time, budget_s)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r

    # Phase 3: Upsample and refine
    for new_n in [160, 320]:
        if time.time() - start_time > budget_s - 80:
            break
        h_up = upsample(best_h, new_n)
        h_up = project_onto_constraint(h_up, new_n / 2.0)
        h_r, c5_r = optimize_continuous(h_up, new_n / 2.0, 8000, rng, start_time, budget_s)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 4: Structured initializations + optimization
    for pat in range(5):
        if time.time() - start_time > budget_s - 40:
            break
        h_s = make_structured(n0, pat)
        h_r, c5_r = optimize_continuous(h_s, n0 / 2.0, 5000, rng, start_time, budget_s)
        h_bin = to_binary(h_r, rng)
        h_r2, c5_r2 = binary_swap_search(h_bin, rng, 8000, start_time, budget_s)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2

    # Phase 5: Random restarts
    while time.time() - start_time < budget_s - 15:
        h_rand = make_random_binary(n0, rng)
        h_r, c5_r = optimize_continuous(h_rand, n0 / 2.0, 4000, rng, start_time, budget_s)
        h_bin = to_binary(h_r, rng)
        h_r2, c5_r2 = binary_swap_search(h_bin, rng, 6000, start_time, budget_s)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2

    # Final: downsample if needed and ensure valid
    final_h = best_h
    n_final = len(final_h)
    # Ensure sum constraint
    final_h = project_onto_constraint(final_h, n_final / 2.0)
    final_c5 = compute_c5(final_h)

    return final_h, final_c5, n_final
