import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * 2.0 / n

def fix_feasibility(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        m = int(mask.sum())
        if m > 0:
            h[mask] += diff / m
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(new_x, old_x, h, left=h[0], right=h[-1])
    return fix_feasibility(h_new)
