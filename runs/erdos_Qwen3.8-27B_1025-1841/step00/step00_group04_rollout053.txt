    # Actually, lambda should be in range such that sum(clamp(h - lambda)) = target_sum

    # If lambda = +inf: project = 0 for all, sum = 0

        s = proj.sum()
        if s > target_sum:

        else:
            hi = mid

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def gradient_c5_smooth(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Numerical stability
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    w = exps / exps.sum()
    
    N = 2 * n - 1
    grad = np.zeros(n)
    
    for m_idx in range(N):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        s = m_idx - (n - 1)
        coeff = wm * dx
        
        # Term 1: (1-h[j-s]) for valid j
        # j-s in [0, n-1] => j in [s, s+n-1], intersect [0, n-1]
        jl = max(0, s)
        jh = min(n - 1, s + n - 1)
        if jl <= jh:
            il = jl - s
            ih = jh - s
            grad[jl:jh+1] += coeff * (1.0 - h[il:ih+1])
        
        # Term 2: -h[j+s] for valid j
        # j+s in [0, n-1] => j in [-s, n-1-s], intersect [0, n-1]
        jl2 = max(0, -s)
        jh2 = min(n - 1, n - 1 - s)
        if jl2 <= jh2:
            il2 = jl2 + s
            ih2 = jh2 + s
            grad[jl2:jh2+1] -= coeff * h[il2:ih2+1]
    
    return grad

    n = len(h)
    lo = -1e-3

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    t0 = time.time()
    target = n / 2.0
    h = project(h_init.copy(), target)
    best_h = h.copy()
    best_val = compute_c5(h)
    
    alpha = start_alpha
    lr = 0.002
    patience = 0
    max_patience = 50
    
    while time.time() - t0 < time_limit * 0.95:
        grad = gradient_c5_smooth(h, n, alpha)
        h_new = h - lr * grad
        h_new = project(h_new, target)
        
        val = compute_c5(h_new)
        
        if val < best_val - 1e-10:
            best_val = val
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        h = h_new

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def gradient_c5_smooth(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exps = np.exp(shifted - max_s)
    w = exps / exps.sum()
    N = 2 * n - 1
    grad = np.zeros(n)
    for m_idx in range(N):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        s = m_idx - (n - 1)
        coeff = wm * dx
        jl = max(0, s)
        jh = min(n - 1, s + n - 1)
        if jl <= jh:
            il = jl - s
            ih = jh - s
            grad[jl:jh + 1] += coeff * (1.0 - h[il:ih + 1])
        jl2 = max(0, -s)
        jh2 = min(n - 1, n - 1 - s)
        if jl2 <= jh2:
            il2 = jl2 + s
            ih2 = jh2 + s
            grad[jl2:jh2 + 1] -= coeff * h[il2:ih2 + 1]
    return grad

def project(h, target_sum):
    n = len(h)
    lo = -0.5
    hi = 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
