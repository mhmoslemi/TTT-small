def project_simplex(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            break
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):

    x_new = (np.arange(new_n) + 0.5) / new_n

    h_new = np.interp(x_new, x_ext, h_ext)
    h_new = project_simplex(h_new, new_n / 2.0)
    return h_new
