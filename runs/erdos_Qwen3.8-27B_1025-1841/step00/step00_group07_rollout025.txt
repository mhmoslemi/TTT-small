            h_r = _slsqp(h_r, n, maxiter=150, ftol=1e-9)
            c = _c5(h_r)
            if c < best_c5:
                best_c5 = c
                best_h = h_r.copy()
                best_n = n
            if budget_s - (time.time() - t0) > 15:
                h_rb = _make_binary(h_r, n, None)
                deadline = t0 + min(5, budget_s - (time.time() - t0) - 5)
                h_rb, c2 = _bin_search(h_rb, n, max_rounds=50, deadline=deadline)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_rb.copy()
                    best_n = n

    return (best_h, best_c5, best_n)
