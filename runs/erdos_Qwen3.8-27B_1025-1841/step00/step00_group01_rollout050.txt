import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _interp_to(h_parent, n_new):
    n_old = len(h_parent)
    if n_old == n_new:
        return h_parent.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_new - 1, n_new)
    return np.interp(x_new, x_old, h_parent)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if abs(s - target) > 1e-12:
        h = h + (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _make_binary(h_cont, n, rng):
    h = (h_cont >= 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - n_ones]] = 1
    return h

def _local_search(h, n, rng, max_iters, time_limit):
    best_c5 = _eval_c5(h)
    current = h.copy()
    t0 = time.time()
    for it in range(max_iters):
        if it % 200 == 0 and time.time() - t0 > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or current[i] == current[j]:
            continue
        current[i] = 1.0 - current[i]
        current[j] = 1.0 - current[j]
        c5 = _eval_c5(current)
        if c5 <= best_c5:
            best_c5 = c5
        else:
            current[i] = 1.0 - current[i]
            current[j] = 1.0 - current[j]
    return current, best_c5

def _simulated_annealing(h, n, rng, max_iters, t_start, t_end, time_limit):
    best_c5 = _eval_c5(h)
    current = h.copy()
    cur_c5 = best_c5
    t0 = time.time()
    for it in range(max_iters):
        if it % 200 == 0 and time.time() - t0 > time_limit:
            break
        temp = t_start * (t_end / t_start) ** (it / max_iters)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or current[i] == current[j]:
            continue
        current[i] = 1.0 - current[i]
        current[j] = 1.0 - current[j]
        new_c5 = _eval_c5(current)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            current[i] = 1.0 - current[i]
            current[j] = 1.0 - current[j]
    return current, best_c5

def _solsqp_obj(h_flat, n):
    h = np.clip(h_flat, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _solsqp_con(h_flat, n):
    return np.sum(h_flat) - n / 2.0

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
        try:
            res = minimize(_solsqp_obj, h0, args=(n,), method='SLSQP',
                          bounds=bounds, constraints=cons,

import numpy as np
from scipy.optimize import minimize
import time


def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

def _interp_to(h_parent, n_new):
    n_old = len(h_parent)
    if n_old == n_new:
        return h_parent.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, float(n_new - 1), n_new)
    return np.interp(x_new, x_old, h_parent)

    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0

        h = h + (target - s) / n

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        else:
            current[i] = 1.0 - current[i]
