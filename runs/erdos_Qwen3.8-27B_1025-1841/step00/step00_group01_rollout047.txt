    dx = 2.0 / n

    return np.max(corr) * dx

def _sum_fun(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones(x.shape[0])

def _obj(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * dx

def _obj_smooth(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = np.max(corr)
    shifted = beta * (corr - c_max)
    shifted = np.clip(shifted, -500, 500)
    return c_max + np.log(np.sum(np.exp(shifted))) / beta

def _fix_sum(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    diff2 = target - s2
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            adj = min(diff2, 1.0 - h[i]) if diff2 > 0 else max(diff2, -h[i])

                continue

            diff2 = target - np.sum(h)
            if abs(diff2) < 1e-12:
