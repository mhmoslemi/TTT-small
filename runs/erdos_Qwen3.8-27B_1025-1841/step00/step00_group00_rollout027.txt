For binary local search:
- State: binary array h of length n with exactly n/2 ones
- Objective: minimize max_k corr_full(k) (or equivalently, maximize min_k R(k) where R is the "auto-overlap")
- Move: pick a position i where h[i]=1 and a position j where h[j]=0, swap them
- Evaluation: check if the swap improves the objective

The challenge is efficiently evaluating the effect of a swap. When we flip h[i] from 1 to 0 and h[j] from 0 to 1:
- For each lag k, the correlation changes by at most 2 (one term removes, one adds)
- We can track the correlation for all lags and update incrementally

Let me think about the complexity:
- n lags to track
- Each swap O(n) to update
- Number of candidate swaps: O(n^2/4) (choose one 1 and one 0)
- Total per iteration: O(n^3) worst case

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum_binary(h, n):
    """Fix sum of binary h to be exactly n//2"""
    target = n // 2
    current = int(np.sum(h))
    if current > target:
        # Remove (current - target) ones
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        for i in range(current - target):
            h[ones[i]] = 0
    elif current < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        for i in range(target - current):
            h[zeros[i]] = 1
    return h

def project_h(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Add to elements with most room
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                if diff <= 0:
                    break
                add = min(diff, room[i])
                h[i] += add
                diff -= add
        else:
            # Remove from elements with most value
            idx = np.argsort(-h)
            for i in idx:
                if diff >= 0:
                    break
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
    return h
