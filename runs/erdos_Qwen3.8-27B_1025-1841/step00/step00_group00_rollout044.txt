    diff = target_s - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_s - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            idx = np.argsort(np.abs(h - 0.5))[:max(1, n // 5)]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h, max_iter=800):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    res = so.minimize(compute_c5, h, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
    c5_new = compute_c5(res.x) if hasattr(res, 'x') else 1e9
    c5_old = compute_c5(h)
    if c5_new < c5_old - 1e-15:
        return fix_feasibility(res.x), c5_new
    return h, c5_old


def pairwise_refine(h, rng, time_limit_s):
    n = len(h)
    best_c5 = compute_c5(h)
    start = time()
    no_imp = 0
    step_lo, step_hi = 0.002, 0.04
    while time() - start < time_limit_s:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] >= h[j]:
            max_delta = min(h[i], 1.0 - h[j])
        else:
            max_delta = min(h[j], 1.0 - h[i])
        if max_delta < 1e-10:
            continue
        delta = rng.uniform(step_lo, min(step_hi, max_delta))
        if h[i] >= h[j]:
            h[i] -= delta
            h[j] += delta
        else:
            h[i] += delta
            h[j] -= delta
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            no_imp = 0
            step_hi = min(0.05, step_hi * 1.02)
            step_lo = min(0.01, step_lo * 1.01)
        else:
            if h[i] <= h[j]:
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta
            no_imp += 1
            if no_imp > 800:
                step_hi = max(0.005, step_hi * 0.85)
                step_lo = max(0.001, step_lo * 0.9)
                no_imp = 0
    return h, best_c5


def coordinate_refine(h, rng, time_limit_s):
    n = len(h)
    best_c5 = compute_c5(h)
    start = time()
    no_imp = 0
    for _ in range(1000000):
        if time() - start >= time_limit_s:
            break
        i = rng.randint(n)
        j = (i + 1) % n
        if i == j:
            continue
        if h[i] >= h[j]:
            max_d = min(h[i], 1.0 - h[j])
        else:
            max_d = min(h[j], 1.0 - h[i])
        if max_d < 1e-12:
            continue
        delta = max_d * 0.5
        if h[i] >= h[j]:
            h[i] -= delta
            h[j] += delta
        else:
            h[i] += delta
            h[j] -= delta
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            no_imp = 0
        else:
            if h[i] <= h[j]:
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta
            no_imp += 1
            if no_imp > 5000:
                break
    return h, best_c5


def upscale(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_up = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return fix_feasibility(h_up)


def perturb(h, rng, strength=0.02):
    h_p = h + rng.randn(len(h)) * strength
    return fix_feasibility(h_p)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time()
    rng = np.random.RandomState(seed)

    h = initial_h_values.astype(float).copy()
    n0 = len(h)
    h = fix_feasibility(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time() - t0)

    # Phase 1: SLSQP at initial resolution
    if rem() > 90:
        h1, c5_1 = slsqp_optimize(h, max_iter=1000)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: Multi-start at initial resolution for diversity
    if rem() > 180:
        for trial in range(3):
            h_pert = perturb(best_h, rng, 0.05)
            hp, cp = slsqp_optimize(h_pert, max_iter=500)
            if cp < best_c5:
                best_h, best_c5 = hp, cp

    # Phase 3: Multi-resolution upscaling + SLSQP
    current_n = len(best_h)
    for factor in [2, 4, 8, 16]:
        target_n = n0 * factor
        if target_n <= current_n:
            continue
        if rem() < 200:
            break
        h_up = upscale(best_h, target_n)
        if rem() > 150:
            h_opt, c5_opt = slsqp_optimize(h_up, max_iter=600)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
                current_n = target_n
            else:
                current_n = target_n

    # Phase 4: Local refinement (pairwise)
    if rem() > 60:
        alloc = min(rem() * 0.5, 400)
        best_h, best_c5 = pairwise_refine(best_h.copy(), rng, alloc)

    # Phase 5: Another SLSQP pass
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, max_iter=500)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Final local refinement
    if rem() > 30:
        best_h, best_c5 = pairwise_refine(best_h.copy(), rng, rem() * 0.6)

    # Phase 7: Coordinate refinement
    if rem() > 20:
        best_h, best_c5 = coordinate_refine(best_h.copy(), rng, rem() * 0.7)

    # Final feasibility check
    best_h = fix_feasibility(best_h)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
