def soft_max_obj(h, dx, beta):

    c = corr * dx
    c_max = c[0]  # starting max, will update

    c_shifted = beta * (c - np.max(c))

The function h(x) on [0,2] with ∫h = 1. The overlap at shift k is ∫h(x)(1-h(x+k))dx.

This is minimized when ∫h^2 is maximized, i.e., when h is as "spread out" or "concentrated" as possible. With h∈[0,1], ∫h^2 is maximized when h is 0-1 (binary), giving ∫h^2 = 1 (since ∫h=1). So for k=0, the overlap = 1-1 = 0.

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import zoom
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def soft_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    cm = np.max(c)
    shifted = beta * (c - cm)
    shifted = np.clip(shifted, -80.0, 0.0)
    return cm + np.log(np.sum(np.exp(shifted)) / len(c)) / beta


def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            rs = np.sum(room)
            if rs > 1e-15:
                h += room * (diff / rs)
            else:
                h += diff / n
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h, n):
    if len(h) == n:
        return h.copy()
    factor = n / float(len(h))
    h_up = zoom(h, factor, order=3)
    if len(h_up) > n:
        h_up = h_up[:n]
    elif len(h_up) < n:
        h_up = np.pad(h_up, (0, n - len(h_up)), mode='edge')
    return h_up.astype(float)


def make_init(n, ri, rng):
    x = np.linspace(0.0, 2.0, n)
    if ri == 0:
        h = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    elif ri == 1:
        h = (x < 1.0).astype(float)
    elif ri == 2:
        h = np.exp(-((x - 0.5) ** 2) / 0.04)
    elif ri == 3:
        h = np.exp(-((x - 1.5) ** 2) / 0.04)
    elif ri == 4:
        h = 0.5 + 0.4 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi))
    elif ri == 5:
        c1, c2 = rng.uniform(0.1, 0.9, 2)
        w1, w2 = rng.uniform(0.1, 0.4, 2)
        h = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
    elif ri == 6:
        h = np.where(x < 0.5, 1.0, np.where(x < 1.5, 0.0, 1.0))
    elif ri == 7:
        h = 1.0 - np.abs(2 * x - 1.0)
    elif ri == 8:
        h = np.sin(x * np.pi / 2.0) ** 2
    elif ri == 9:
        h = np.zeros(n)
        blk = max(1, n // 8)
        h[0:blk] = 1.0
        h[3 * blk:4 * blk] = 1.0
        h[6 * blk:7 * blk] = 1.0
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(0.5, 3) * x)
    h = h + rng.standard_normal(n) * 0.02
    return repair(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h0 = np.ones(70) / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    resolutions = [70, 100, 140, 200, 280, 400]

    for n in resolutions:
        if elapsed() > budget_s - 180:
            break

        dx = 2.0 / n
        target = n / 2.0
        beta = 100.0

        def obj(x):
            return soft_max_obj(x, dx, beta)

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n

        # Start from upsampled best
        h_start = repair(upsample(best_h, n), n)

        # Main SLSQP
        try:
            res = minimize(obj, h_start, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 400, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = repair(res.x, n)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
        except Exception:
            pass

        n_restarts = 12
        for ri in range(n_restarts):
            if elapsed() > budget_s - 60:

            if ri == 0:
                h_try = repair(h_start + rng.standard_normal(n) * 0.04, n)
            else:
                h_try = make_init(n, ri - 1, rng)

                res = minimize(obj, h_try, method='SLSQP', bounds=bnds,
                               constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = repair(res.x, n)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
            except Exception:
                pass

        # Second pass with finer beta for polishing
        if elapsed() < budget_s - 120:
            beta2 = 200.0

            def obj2(x):
                return soft_max_obj(x, dx, beta2)

            h_polish = repair(upsample(best_h, n), n) if len(best_h) != n else best_h.copy()
            try:
                res = minimize(obj2, h_polish, method='SLSQP', bounds=bnds,

                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
            except Exception:
                pass

    # Final repair and evaluation
    n_final = len(best_h)
    best_h = repair(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
