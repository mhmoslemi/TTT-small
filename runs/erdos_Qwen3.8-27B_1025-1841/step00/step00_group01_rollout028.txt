import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, tau=50.0):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

def project_sum(h, target):
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        h = h - diff / len(h)
    return np.clip(h, 0.0, 1.0)

def interp_parent(parent, new_n):
    old_n = len(parent)
    if old_n == new_n:
        return parent.copy()
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, parent, left=parent[0], right=parent[-1])

def optimize_continuous(h, n, max_time, tau=40.0):

    dx = 2.0 / n
    h = project_sum(h.copy(), target)
    bounds = [(0.0, 1.0)] * n
    best_local = h.copy()
    best_val = compute_c5(h, n)
    
    t_start = time.time()
    
    # L-BFGS-B with penalty
    def obj_pen(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        sm = smooth_max(corr, tau)
        pen = 5.0 * (np.sum(x) - target) ** 2
        return sm + pen
    
    try:
        res = minimize(obj_pen, h.copy(), method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-8})
        h_new = project_sum(np.clip(res.x, 0, 1), target)
        v = compute_c5(h_new, n)
        if v < best_val:
            best_val = v
            best_local = h_new
    except:
        pass
    
    if time.time() - t_start > max_time * 0.6:
        return best_local, best_val
    
    # SLSQP for smaller n

        def obj_slsqp(x):
            corr = np.correlate(x, 1.0 - x, mode="full") * dx
            return smooth_max(corr, tau)

        try:

def local_combinatorial_search(h, n, max_time, n_iter=5000):
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h, n)
    dx = 2.0 / n
    t_start = time.time()
    rng = np.random.default_rng(12345)
    
    for it in range(n_iter):
        if time.time() - t_start > max_time:
            break
        # Pick two random positions and try to adjust
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        # Try: increase h[i] by d, decrease h[j] by d (maintaining sum)
        for d_sign in [1, -1]:
            for d_mag in [0.1, 0.25, 0.5, 0.9, 1.0]:
                d = d_sign * d_mag
                h[i] += d
                h[j] -= d
                if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:

                    h[j] += d
                    continue
                v = compute_c5(h, n)
                if v < best - 1e-12:
                    best = v
                else:
                    h[i] -= d
                    h[j] += d
    
    return h, best
