    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, np.ones(n) - h, mode="full")
    return corr * dx

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0, 1)
    # Fix integral
    target = n_new / 2.0
    h_new = h_new * (target / max(h_new.sum(), 1e-10))
    h_new = np.clip(h_new, 0, 1)
    # Final fix
    diff = target - h_new.sum()
    # spread correction
    mask = (h_new < 1) & (h_new > 0)
    if diff > 0:
        add_mask = (h_new < 1)
    else:
        add_mask = (h_new > 0)

        h_new = np.clip(h_new, 0, 1)
    return h_new

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() > 0:
            per = diff / mask.sum()
            h[mask] += per
            h = np.clip(h, 0.0, 1.0)
        # If still off (edge case), distribute to all
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)

- Entry m (0-indexed) corresponds to: sum over valid i of h[i] * (1-h)[i + m - (n-1)]
- The integral approximation: multiply by dx

So C₅ = max(corr) * dx where corr = np.correlate(h, 1-h, mode="full") and dx = 2/n.

Yes, that matches what I have.

Let me also double-check: for the "full" mode correlate of a (length n) and b (length n):
result[k] = sum_i a[i] * b[i + k - (n-1)]
where the sum is over all i such that 0 ≤ i < n and 0 ≤ i+k-(n-1) < n.

So result[n-1] = sum_i a[i] * b[i] (the "zero lag")
result[n-1+1] = sum_i a[i] * b[i+1] (lag +1, sum over i=0 to n-2)
result[n-1-1] = sum_i a[i] * b[i-1] (lag -1, sum over i=1 to n-1)

For a = h, b = 1-h:
result[n-1] = sum_i h[i] * (1-h[i]) = 0 for binary h ✓

Good. My evaluate_c5 is correct.

Now, one concern: the problem says the integral is over x from 0 to 2, but the step function is defined on [0,2]. The lag k in the continuous problem can be any real number, but in the discrete problem, the lags are restricted to multiples of dx = 2/n. As n increases, we sample more lags, which is good (we're more likely to find the true maximum).

But there's a subtlety: for very large |k| (close to 2), the overlap region is small, so the integral is small. The interesting lags are for |k| in some middle range.

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, np.ones(n) - h, mode="full")
    return float(np.max(corr) * dx)

def _upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.astype(float).copy()
    idx = (np.arange(n_new) * n_old + n_old // 2) // n_old
    h_new = h_old[idx].astype(float).copy()
    target = n_new / 2.0
    s = h_new.sum()
    if s > 1e-10:
        h_new *= target / s
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            m = h_new < 1.0
        else:
            m = h_new > 0.0
        if m.any():
            h_new[m] += diff / m.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _fix_sum(h, n):
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return
    if diff > 0:
        m = h < 1.0
    else:
        m = h > 0.0
    c = m.sum()
    if c > 0:
        h[m] = np.clip(h[m] + diff / c, 0.0, 1.0)
    d2 = target - h.sum()
    if abs(d2) > 1e-12:
        if d2 > 0:
            i = int(np.argmin(h))
        else:
            i = int(np.argmax(h))
        h[i] = np.clip(h[i] + d2, 0.0, 1.0)

def _local_search(h, n, budget, rng, T0=None):
    h = h.copy()
    cur = evaluate_c5(h)
    best = cur
    T = T0 if T0 is not None else cur * 0.04
    t0 = time.time()
    while time.time() - t0 < budget:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        while j == i:
            j = int(rng.integers(n))
        d = float(rng.uniform(-0.2, 0.2))
        h[i] += d
        h[j] -= d
        np.clip(h, 0.0, 1.0, out=h)
        _fix_sum(h, n)
        new_c = evaluate_c5(h)
        if new_c < cur or (T > 1e-12 and rng.random() < np.exp(-min((new_c - cur) / T, 60.0))):
            cur = new_c
            if cur < best:
                best = cur
        else:
            # Reject: restore
            h[i] -= d
            h[j] += d
            np.clip(h, 0.0, 1.0, out=h)
            _fix_sum(h, n)
        T = max(T * 0.9997, 1e-12)
    return h, best

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, np.ones(n) - h, mode="full")
    return float(np.max(corr) * dx)

def _upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.astype(float).copy()
    idx = (np.arange(n_new) * n_old + n_old // 2) // n_old
    h_new = h_old[idx].astype(float).copy()
    target = n_new / 2.0
    s = h_new.sum()
    if s > 1e-10:
        h_new *= target / s
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-9:
        m = (h_new < 1.0) if diff > 0 else (h_new > 0.0)
        if m.any():
            h_new[m] += diff / m.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _fix_sum(h, n):
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-11:
        return
    m = (h < 1.0) if diff > 0 else (h > 0.0)
    c = m.sum()
    if c > 0:
        h[m] += diff / c
        np.clip(h, 0.0, 1.0, out=h)
    d2 = target - h.sum()
    if abs(d2) > 1e-10:
        i = int(np.argmin(h)) if d2 > 0 else int(np.argmax(h))
        h[i] = np.clip(h[i] + d2, 0.0, 1.0)

def _local_search(h, n, budget, rng, T0=None):
    h = h.copy()
    h_old = h.copy()
    cur = evaluate_c5(h)
    best = cur
    T = T0 if T0 is not None else max(cur * 0.03, 0.003)
    t0 = time.time()
    while time.time() - t0 < budget:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        while j == i:
            j = int(rng.integers(n))
        d = float(rng.uniform(-0.2, 0.2))
        np.copyto(h_old, h)
        h[i] += d
        h[j] -= d
        np.clip(h, 0.0, 1.0, out=h)
        _fix_sum(h, n)
        new_c = evaluate_c5(h)
        if new_c < cur or (T > 1e-12 and float(rng.random()) < np.exp(-min((new_c - cur) / T, 60.0))):
            cur = new_c
            if cur < best:
                best = cur
        else:
            np.copyto(h, h_old)
        T = max(T * 0.9996, 1e-13)
    return h, best

def _block_move_search(h, n, budget, rng, T0=None):
    """Local search with block moves (shift contiguous blocks)."""
    h = h.copy()
    h_old = h.copy()
    cur = evaluate_c5(h)
    best = cur
    T = T0 if T0 is not None else max(cur * 0.03, 0.003)
    t0 = time.time()
    while time.time() - t0 < budget:
        # Block move: rotate a contiguous segment
        bstart = int(rng.integers(n))
        blen = int(rng.integers(1, min(10, n // 4 + 1)))
        bend = (bstart + blen) % n
        np.copyto(h_old, h)
        # Rotate: h[bstart:bend] -> h[bend: (bend+blen)%n]
        seg = h[bstart:bend].copy() if bstart + blen <= n else np.concatenate([h[bstart:], h[:bend]])
        if bstart + blen <= n:
            dst_start = bend
            h[dst_start:dst_start+blen] += seg - h[dst_start:dst_start+blen]
            h[bstart:bend] = h[dst_start:dst_start+blen]
            h[dst_start:dst_start+blen] = seg
        else:
            # Wrap-around: just do a simple two-point move instead
            i = bstart
            j = (bend + 1) % n
            d = float(rng.uniform(-0.2, 0.2))
            h[i] += d
            h[j] -= d
        np.clip(h, 0.0, 1.0, out=h)
        _fix_sum(h, n)
        new_c = evaluate_c5(h)
        if new_c < cur or (T > 1e-12 and float(rng.random()) < np.exp(-min((new_c - cur) / T, 60.0))):
            cur = new_c
            if cur < best:
                best = cur
        else:
            np.copyto(h, h_old)
        T = max(T * 0.9996, 1e-13)
    return h, best
