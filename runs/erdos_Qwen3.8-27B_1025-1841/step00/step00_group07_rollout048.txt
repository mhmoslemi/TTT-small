import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = np.sum(room)
        if tot > 1e-15:
            h += diff * room / tot
    else:
        room = h.copy()
        tot = np.sum(room)
        if tot > 1e-15:
            h += diff * room / tot
    return np.clip(h, 0.0, 1.0)

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return np.interp(x_new, x_old, h)

def do_slsqp(h_start, maxiter=2000):
    n = len(h_start)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return compute_c5(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = fix_sum(res.x, target)
        return h_out, compute_c5(h_out)
    except Exception:
        return fix_sum(h_start, target), compute_c5(fix_sum(h_start, target))
