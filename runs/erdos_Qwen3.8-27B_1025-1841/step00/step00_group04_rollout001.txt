import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return float(np.max(c) * dx)

def project_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0.0, 1.0)
    return h

def enforce_exact_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmax(h)) if diff < 0 else int(np.argmin(h))
        h[idx] -= diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(h, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    cmax = np.max(c)
    s = np.sum(np.exp(alpha * (c - cmax)))
    val = cmax + np.log(s) / alpha
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def optimize_lseb(h0, n, alpha, lam, maxiter, time_limit):

        return lse_objective(h, n, alpha, lam)
