import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += (d / tr) * room
        else:
            tm = h.sum()
            if tm < 1e-14:
                break
            h += (d / tm) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def coord_descent(h, n, dx, rng, time_limit):
    t0 = time.time()
    best_c5 = compute_c5(h, dx)
    no_improve = 0
    while time.time() - t0 < time_limit and no_improve < 4:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.time() - t0 > time_limit:
                break
            orig = h[i]
            best_obj = best_c5
            saved = h.copy()
            cands = [0.0, 1.0, 0.5]
            if 0.03 < orig < 0.97:
                cands.append(orig - 0.15)
                cands.append(orig + 0.15)
            for c in cands:
                c = max(0.0, min(1.0, c))
                if abs(c - orig) < 1e-5:
                    continue
                h[i] = c
                h = repair(h, n)
                c5 = compute_c5(h, dx)
                if c5 < best_obj - 1e-12:
                    best_obj = c5
                    improved = True
            if not improved:
                h = saved.copy()
        if improved:
            best_c5 = compute_c5(h, dx)
            no_improve = 0
        else:
            no_improve += 1
    return h, best_c5
