```python
import numpy as np
from scipy.optimize import minimize
import time

_beta = 80.0
_target = 0.0
_nvars = 0

def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    e = np.exp(_beta * (corr - m))
    return (m + np.log(np.sum(e)) / _beta) * dx

def smooth_max_grad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    e = np.exp(_beta * (corr - m))
    w = e / e.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        m1 = idx - s
        v1 = (m1 >= 0) & (m1 < n)
        m2 = idx + s
        v2 = (m2 >= 0) & (m2 < n)
        c = np.zeros(n)
        c[v1] = dx * (1.0 - x[m1[v1]])
        c[v2] = c[v2] - dx * x[m2[v2]]
        grad += wk * c
    return grad

def proj_set(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def con_fun(x):
    return np.sum(x) - _target

def con_jac(x):
    return np.ones(_nvars)

def run(seed=42, budget_s=1300, **kwargs):
    global _beta, _target, _nvars
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(42, 0.5)

    n = len(h0)
    tgt = n / 2.0
    _target = tgt
    _nvars = n
    h0 = proj_set(h0, tgt)

    best_h = h0.copy()
    best_c5 = compute_c5_exact(best_h)

    # SLSQP at parent resolution with increasing beta for sharpness
    for b in [30.0, 60.0, 120.0, 250.0]:
        if time.time() - t0 > budget_s * 0.3:
            break
        _beta = b
        _target = n / 2.0
        _nvars = n
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        bnd = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           method='SLSQP', bounds=bnd, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-15})
            h_new = proj_set(res.x, tgt)
            c5 = compute_c5_exact(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    # Random restarts at same resolution
    for _ in range(15):
        if time.time() - t0 > budget_s * 0.55:
            break
        _beta = 100.0
        _target = n / 2.0
        _nvars = n
        h_trial = best_h + rng.randn(n) * 0.04
        h_trial = proj_set(h_trial, tgt)
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        bnd = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_max_obj, h_trial, jac=smooth_max_grad,
                           method='SLSQP', bounds=bnd, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-13})
            h_new = proj_set(res.x, tgt)
            c5 = compute_c5_exact(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    # Higher resolutions via interpolation
    for factor in [2, 4, 6, 8]:
        res_n = n * factor
        if res_n > 1500:
            break
        if time.time() - t0 > budget_s * 0.82:
            break

        x_o = np.linspace(0, 2, len(best_h))
        x_n = np.linspace(0, 2, res_n)
        h_trib = np.interp(x_n, x_o, best_h)
        tgt_n = res_n / 2.0
        h_trib = proj_set(h_trib, tgt_n)

        _beta = 40.0
        _target = tgt_n
        _nvars = res_n
        cons_n = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        bnd_n = [(0.0, 1.0)] * res_n
        iters = 200 if res_n <= 100 else 100

        try:
            res = minimize(smooth_max_obj, h_trib, jac=smooth_max_grad,
                           method='SLSQP', bounds=bnd_n, constraints=cons_n,
                           options={'maxiter': iters, 'ftol': 1e-10})
            h_new = proj_set(res.x, tgt_n)
            c5 = compute_c5_exact(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

        # Second pass with higher beta at this resolution
        if time.time() - t0 < budget_s * 0.85:
            _beta = 100.0
            _target = tgt_n
            _nvars = res_n
            try:
                res = minimize(smooth_max_obj, best_h if c5 < best_c5 else h_new,
                               jac=smooth_max_grad,
                               method='SLSQP', bounds=bnd_n, constraints=cons_n,
                               options={'maxiter': iters, 'ftol': 1e-12})

                if c5_2 < best_c5:

                    best_c5 = c5_2
            except Exception:
                pass

    best_h = proj_set(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5_exact(best_h)
    return best_h, final_c5, len(best_h)
```
