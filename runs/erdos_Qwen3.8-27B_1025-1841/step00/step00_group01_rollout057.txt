    hi_lam = np.inf

    # We want sum(clip(v - t, 0, 1)) = target
    # f(t) = sum(clip(v-t, 0, 1)) is decreasing in t

    h_new = h - alpha * grad

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_grad(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    c5 = float(corr[idx] * dx)
    n = len(h)
    shift = idx - (n - 1)
    grad = np.zeros(n)
    j_lo1 = max(0, shift)
    j_hi1 = min(n, n + shift)
    if j_lo1 < j_hi1:
        idxs = np.arange(j_lo1, j_hi1)
        grad[idxs] += (1.0 - h[idxs - shift])
    j_lo2 = max(0, -shift)
    j_hi2 = min(n, n - shift)
    if j_lo2 < j_hi2:
        idxs = np.arange(j_lo2, j_hi2)
        grad[idxs] += -h[idxs + shift]
    return c5, grad * dx

def project_sum(v, target):
    n = len(v)
    t_lo = float(v.min()) - 1.5
    t_hi = float(v.max()) + 1.5
    for _ in range(60):
        t_mid = 0.5 * (t_lo + t_hi)
        s = float(np.clip(v - t_mid, 0.0, 1.0).sum())
        if s > target:
            t_lo = t_mid
        else:
            t_hi = t_mid
    t = 0.5 * (t_lo + t_hi)
    return np.clip(v - t, 0.0, 1.0)

def fix_sum(h, n):
    target = n / 2.0
    return project_sum(h.astype(float), target)

def interpolate_periodic(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)

    return np.interp(x_dst, x_src, h_src, period=2.0)
