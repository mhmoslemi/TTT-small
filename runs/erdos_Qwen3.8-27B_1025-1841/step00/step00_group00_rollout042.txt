def analytic_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)  # shift
        pk = p[k]
        if pk < 1e-15:
            continue
        if s >= 0:
            # First term: (1-h[j-s]) for j in [s, n-1]
            j_range = range(s, n)
            for j in j_range:
                grad[j] += pk * dx * (1.0 - h[j - s])
            # Second term: -h[j+s] for j in [0, n-1-s]
            j_range = range(0, n - s)
            for j in j_range:
                grad[j] -= pk * dx * h[j + s]
        else:
            t = -s
            # First term: (1-h[j-t]) for j in [0, n-1-t]
            j_range = range(0, n - t)
            for j in j_range:
                grad[j] += pk * dx * (1.0 - h[j + t])
            # Second term: -h[j-t] for j in [t, n-1]
            j_range = range(t, n)
            for j in j_range:
                grad[j] -= pk * dx * h[j - t]
    
    return grad

import numpy as np
from scipy.optimize import minimize
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)

        else:
            t = -s
            grad[:n-t] += pk * (1.0 - h[t:])
            grad[t:] -= pk * h[:n-t]
    return grad

def sum_con_fun(x, target):
    return np.sum(x) - target

def sum_con_jac(x, target):
    return np.ones(len(x))

def fix_sum(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
    else:
        room = h.copy()
    total = room.sum()
    if total < 1e-14:
        return h
    h = h + diff * room / total
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_up = np.interp(new_x, old_x, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    return fix_sum(h_up, new_n / 2.0)

def optimize_resolution(h_init, n_pts, alphas, maxiter, start_time, budget_s):
    best_h = h_init.copy()
    best_val = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n_pts
    target = n_pts / 2.0
    h_curr = h_init.copy()
    for alpha in alphas:
        if time() - start_time > budget_s - 5:
            break
        cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (target,)}]
        res = minimize(smooth_obj, h_curr, args=(alpha,), jac=smooth_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            cand = np.clip(res.x, 0.0, 1.0)
            cand = fix_sum(cand, target)
            val = compute_c5(cand)
            if val < best_val:
                best_val = val
                best_h = cand.copy()
            h_curr = cand.copy()
    return best_h, best_val

This is related to the problem of minimizing the autocorrelation side lobes!

For a binary sequence of length n with n/2 ones, we want to minimize max_k |autocorrelation at lag k| (for k ≠ 0). This is the classical "minimum peak sidelobe" problem in radar/communications.

Known optimal or near-optimal sequences include:

- M-sequences (maximum length sequences from LFSRs)
- Costas arrays

2. The "autocorrelation" is between h and (1-h), not h and h

correl[n-1+s] = number of i where h[i]=1 and h[i+s]=0

For a "balanced" arrangement, this would be approximately (n-s)/2 * 1/2 = (n-s)/4 for large n. But we want to minimize the maximum.

The maximum over s would be achieved near s=0 (where there are the most terms), giving approximately (n-1)/4.

C5 ≈ (n-1)/4 * dx = (n-1)/4 * 2/n ≈ 1/2 for large n.

But we can do better with structured arrangements. The key is to make the ones and zeros "interleave" as much as possible.

For odd shift s=1: sum of h[i] over valid i = (n-1)/2 (approximately n/2)
C5 at shift 1 ≈ (n/2) * 2/n = 1

import numpy as np
from scipy.optimize import minimize
from time import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)


def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        pk = p[k] * dx
        if pk < 1e-15:
            continue
        if s >= 0:
            grad[:n - s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n - s]
        else:
            t = -s
            grad[t:] += pk * (1.0 - h[:n - t])
            grad[:n - t] -= pk * h[t:]
    return grad


def sum_con_fun(x, target):
    return np.sum(x) - target


def sum_con_jac(x, target):
    return np.ones(len(x))


def fix_sum(h, target):
    h = h.astype(float).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
    else:
        room = h.copy()
    total = room.sum()
    if total < 1e-14:
        return h
    h = h + diff * room / total
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample_fn(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_up = np.interp(new_x, old_x, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    return fix_sum(h_up, new_n / 2.0)


def optimize_resolution(h_init, n_pts, alphas, maxiter_per_alpha, start_time, budget_s):
    best_h = h_init.copy()
    best_val = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n_pts
    target = n_pts / 2.0
    h_curr = h_init.copy()
    for alpha in alphas:
        if time() - start_time > budget_s - 3:
            break
        cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (target,)}]
        try:
            res = minimize(smooth_obj, h_curr, args=(alpha,), jac=smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
        except Exception:
            break
        if res.x is not None:
            cand = np.clip(res.x, 0.0, 1.0)
            cand = fix_sum(cand, target)
            val = compute_c5(cand)
            if val < best_val:
                best_val = val
                best_h = cand.copy()
            h_curr = cand.copy()
    return best_h, best_val


def coord_descent_polish(h, target, n_iter):
    """Simple coordinate descent polish maintaining sum constraint."""
    n = len(h)
    h = h.copy()
    for _ in range(n_iter):
        for j in range(n):
            # Try small perturbations of h[j], compensating on a random other index
            best_t = h[j]
            best_obj = compute_c5(h)
            for dt in [-0.02, 0.02, -0.01, 0.01, -0.05, 0.05]:
                t_new = h[j] + dt
                if t_new < 0.0 or t_new > 1.0:
                    continue
                j2 = (j + n // 2) % n
                if j2 == j:
                    j2 = (j + 1) % n
                t2_new = h[j2] - dt
                if t2_new < 0.0 or t2_new > 1.0:
                    continue
                h[j] = t_new
                h[j2] = t2_new
                obj = compute_c5(h)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_t = t_new
                    h[j2] = t2_new
                else:
                    h[j] = best_t
                    # restore j2
                    h[j2] = h[j2] + dt if dt != 0 else h[j2]
            h[j] = best_t
    return h
