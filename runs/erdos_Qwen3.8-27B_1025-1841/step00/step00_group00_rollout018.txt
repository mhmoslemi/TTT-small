    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -np.max(h) - 1.0, np.max(1.0 - h) + 1.0
    for _ in range(60):

        h_try = np.clip(h + mid, 0.0, 1.0)

        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:

            h = np.clip(h, 0.0, 1.0)
        else:
            # All at boundaries, adjust one
            if diff > 0:
                idx = np.argmax(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmin(h)
                h[idx] = max(0.0, h[idx] + diff)

    """
    Compute gradient of r[k] (the k-th entry of correlate(h, 1-h, 'full')) w.r.t. h.
    k ranges from 0 to 2n-2.
    The shift is s = k - (n-1).

    s = k - (n - 1)
    if s >= 0:

        upper = n - s

            grad[i] = (1.0 - h[i + s])
            # h[i+s] appears with coefficient -h[i]

    """Gradient of output[m] of correlate(h, 1-h, 'full') w.r.t. h."""
    s = m - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        # output[m] = sum_{n=0}^{n-1-s} h[n+s]*(1-h[n])
        # d/dh[j] = (1-h[j-s]) for j in [s, n-1]
        #          - h[j+s] for j in [0, n-1-s]
        if s < n:
            grad[s:] += (1.0 - h[:n-s])  # j from s to n-1, coefficient (1-h[j-s])
            grad[:n-s] -= h[s:]  # j from 0 to n-1-s, coefficient -h[j+s]
    else:
        t = -s
        # output[m] = sum_{j=0}^{n-1-t} h[j]*(1-h[j+t])
        # d/dh[j] = (1-h[j+t]) for j in [0, n-1-t]
        #          -h[j-t] for j in [t, n-1]
        if t < n:
            grad[:n-t] += (1.0 - h[t:])  # j from 0 to n-1-t
            grad[t:] -= h[:n-t]  # j from t to n-1
    return grad
