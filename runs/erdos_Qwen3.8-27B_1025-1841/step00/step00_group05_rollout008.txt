    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

    conv = np.correlate(h, 1.0 - h, mode="full")
    return int(np.argmax(conv)) - (n - 1)

            val += (1.0 - h[j - k])
        if 0 <= j + k < n:

    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n-k])
        g[:n-k] -= x[k:]
    else:

        g[:n-m] += (1.0 - x[m:])
        g[m:] -= x[:n-m]

    conv = np.correlate(x, 1.0 - x, mode="full")

    max_c = np.max(c)
    logits = -beta * (max_c - c)  # = beta*(c - max_c), ≤ 0

        k = int(idx) - (n - 1)
        g += p[idx] * dx * grad_single_k(x, k)

The max of linear functions of h[j] is a piecewise-linear convex function. We want to minimize the max over k, so it's a min of a max of linear functions = the lower envelope.

One more thought: the problem says C₅ = max(np.correlate(h, 1-h, mode="full") * dx). This means we take the max of the entire correlation vector, scaled by dx. The correlation vector has 2n-1 entries. For shifts far from 0, the overlap region is small, so the values are naturally smaller. The max will typically be at small shifts (k near n-1 in the full correlation indexing, which corresponds to shift 0).

Actually, shift 0 in the full correlation is at index n-1 (since index j corresponds to shift j-(n-1)). At shift 0:
c[n-1] = sum_i h[i]*(1-h[i]) = sum_i h[i] - sum_i h[i]²

Since sum_i h[i] = n/2 (fixed), we want to maximize sum_i h[i]² to minimize c[n-1].
This means we want h to be as "bimodal" as possible (values close to 0 or 1).

But other shifts also matter. At shift 1:
c[n] = sum_i h[i+1]*(1-h[i])

This is more complex to optimize.

The tension: we want h to be binary (to minimize shift-0 overlap) but also to have a structure that minimizes other shifts. This is exactly where the non-trivial optimization comes in.

import numpy as np
from scipy.optimize import minimize
import time

_opt_n = 70
_opt_beta = 100.0
_opt_penalty = 50000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_single_k(x, k):
    n = len(x)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n-k])
        g[:n-k] -= x[k:]
    else:
        m = -k
        g[:n-m] += (1.0 - x[m:])
        g[m:] -= x[:n-m]
    return g

def smoothed_obj_grad(x, n, beta, penalty):
    dx = 2.0 / n
    conv = np.correlate(x, 1.0 - x, mode="full")
    c = conv * dx
    max_c = np.max(c)
    logits = -beta * (max_c - c)
    exp_l = np.exp(np.clip(logits, -700.0, 0.0))
    p = exp_l / exp_l.sum()
    f = max_c + np.log(exp_l.sum()) / beta
    g = np.zeros(n)
    sig = p > 1e-12
    idxs = np.where(sig)[0]
    for idx in idxs:
        k = int(idx) - (n - 1)
        g += p[idx] * dx * grad_single_k(x, k)
    if penalty > 0.0:
        diff = x.sum() - n / 2.0
        f += penalty * diff * diff
        g += 2.0 * penalty * diff
    return f, g

def _obj_func(x):
    return smoothed_obj_grad(x, _opt_n, _opt_beta, _opt_penalty)[0]

def _jac_func(x):
    return smoothed_obj_grad(x, _opt_n, _opt_beta, _opt_penalty)[1]

def slsqp_minimize(h_init, n, beta, maxiter, penalty=50000.0):
    global _opt_n, _opt_beta, _opt_penalty
    _opt_n = n
    _opt_beta = beta
    _opt_penalty = penalty
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_func, h_init.astype(float), jac=_jac_func,
                       method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_sol = project_feasible(res.x, n)
        c5 = compute_c5(h_sol)
        if np.isfinite(c5):
            return h_sol, c5
    except Exception:
        pass
    return None

def coordinate_descent(h, n, n_rounds=20):
    best = h.copy()
    best_val = compute_c5(best)
    for r in range(n_rounds):
        improved = False
        for j in range(n):
            old = best[j]
            candidates = np.unique(np.concatenate([
                [old, 0.0, 1.0],
                [old + d for d in [-0.5, -0.25, -0.1, -0.05, 0.05, 0.1, 0.25, 0.5]
                 if 0.0 <= (old + d) <= 1.0]
            ]))
            best_candidate = old
            best_cand_val = best_val
            for cv in candidates:
                if abs(cv - old) < 1e-12:
                    continue
                test = best.copy()
                test[j] = cv
                test = project_feasible(test, n)
                v = compute_c5(test)
                if v < best_cand_val:
                    best_cand_val = v
                    best_candidate = cv
            if best_candidate != old:
                best[j] = best_candidate
                best = project_feasible(best, n)
                best_val = compute_c5(best)
                improved = True
        if not improved:
            break
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    best_h = project_feasible(h0, n)
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Primary SLSQP optimization
    if remaining() > 60:
        result = slsqp_minimize(best_h, n, beta=100.0, maxiter=500)
        if result is not None and result[1] < best_c5:
            best_h, best_c5 = result

    # Coordinate descent polish
    if remaining() > 30:
        best_h, best_c5 = coordinate_descent(best_h, n, n_rounds=15)

    # Phase 2: Multiple perturbations + SLSQP
    n_restarts = 40
    for i in range(n_restarts):
        if remaining() < 40:
            break
        scale = 0.15 * np.exp(-i / 15.0) + 0.02
        h_init = best_h + np.random.randn(n) * scale
        h_init = project_feasible(h_init, n)
        result = slsqp_minimize(h_init, n, beta=80.0, maxiter=300)
        if result is not None and result[1] < best_c5:
            best_h, best_c5 = result
            best_h, best_c5 = coordinate_descent(best_h, n, n_rounds=5)

    # Phase 3: Random initial conditions
    for i in range(15):
        if remaining() < 60:
            break
        h_init = np.random.uniform(0, 1, n)
        h_init = project_feasible(h_init, n)
        result = slsqp_minimize(h_init, n, beta=60.0, maxiter=400)
        if result is not None and result[1] < best_c5:
            best_h, best_c5 = result
            best_h, best_c5 = coordinate_descent(best_h, n, n_rounds=10)

    # Phase 4: Try higher resolution
    for mult in [2, 3]:
        if remaining() < 180:
            break
        n_new = n * mult
        if n_new > 600:
            break
        h_up = np.repeat(best_h, mult)
        h_up = project_feasible(h_up, n_new)
        result = slsqp_minimize(h_up, n_new, beta=50.0, maxiter=400, penalty=20000.0)
        if result is not None and result[1] < best_c5:
            best_h, best_c5, n = result[0], result[1], n_new
            best_h, best_c5 = coordinate_descent(best_h, n, n_rounds=10)

    # Final polish
    if remaining() > 20:
        best_h, best_c5 = coordinate_descent(best_h, n, n_rounds=10)
        if remaining() > 30:
            result = slsqp_minimize(best_h, n, beta=200.0, maxiter=200)
            if result is not None and result[1] < best_c5:
                best_h, best_c5 = result

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
