For the non-cyclic case, the boundary effects matter.

One approach: think of it as a scheduling problem. We have n/2 "1"s to place in n positions, and we want to minimize the maximum number of "conflicts" at any distance.

A "conflict" at distance k between position i and i+k occurs when h[i]=1 and h[i+k]=0.

We want to minimize the maximum number of conflicts over all distances k.

Total conflicts across all distances: sum_k c_k = sum_i h[i] * (number of zeros after i) = sum_i h[i] * (n-1-i - sum_{j=i+1}^{n-1} h[j])

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def time_left_check(t0, budget_s):
    return budget_s - 20 - (time.time() - t0)

def make_binary(h, n, rng):
    """Convert to binary with exactly n//2 ones."""
    h = np.array(h, dtype=np.float64)
    if n != len(h):
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h)), h)
    h = (h > 0.5).astype(np.float64)
    tgt = n // 2
    cnt = int(h.sum())
    while cnt > tgt:
        h[rng.choice(np.where(h == 1)[0])] = 0
        cnt -= 1
    while cnt < tgt:
        h[rng.choice(np.where(h == 0)[0])] = 1
        cnt += 1
    return h

def sa_optimize(h, n, rng, time_limit, T_init=0.03, cool=0.9998):
    """Simulated annealing with swap moves."""
    cur = compute_c5(h, n)
    best = cur
    best_h = h.copy()
    T = T_init
    stagnant = 0
    
    while time.time() < time_limit and T > 1e-7:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_val = compute_c5(h, n)
        delta = new_val - cur
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur = new_val
            if new_val < best:
                best = new_val
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            h[i], h[j] = h[i], h[j]
            stagnant += 1
        T *= cool
        if stagnant > 100000:
            break
    
    return best_h, best

def fractional_refine(h, n, rng, time_limit):
    """Local search for fractional h."""
    tgt = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    # Project to sum = tgt
    for _ in range(50):
        d = tgt - h.sum()
        if abs(d) < 1e-10:
            break
        if d > 0:
            adj = np.where(h < 0.999)[0]
        else:
            adj = np.where(h > 0.001)[0]
        if len(adj) == 0:
            break
        h[adj] += d / len(adj)
        np.clip(h, 0, 1, out=h)
    
    cur = compute_c5(h, n)
    best = cur
    best_h = h.copy()
    step = 0.1
    
    while time.time() < time_limit:
        i = rng.randint(n)
        d = rng.choice([-1, 1]) * rng.uniform(0.01, step)
        h_try = h.copy()
        h_try[i] = np.clip(h_try[i] + d, 0, 1)
        diff = tgt - h_try.sum()
        if abs(diff) > 1e-10:
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            if diff > 0:
                adj = np.where(mask & (h_try < 0.999))[0]
            else:
                adj = np.where(mask & (h_try > 0.001))[0]
            if len(adj) == 0:

            h_try[adj] += diff / len(adj)
            np.clip(h_try, 0, 1, out=h_try)
            if abs(h_try.sum() - tgt) > 0.5:
                continue
        val = compute_c5(h_try, n)
        if val < cur:
            cur = val
            h = h_try
            if val < best:
                best = val
                best_h = h.copy()
    
    return best_h, best
