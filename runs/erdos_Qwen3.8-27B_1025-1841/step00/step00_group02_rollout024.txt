def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase sum
        room = 1.0 - h
        order = np.argsort(-room)  # most room first
        for i in order:
            add = min(diff, room[i])
            h[i] += add
            room[i] -= add
            diff -= add

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def smooth_c5(h, beta=40.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(corr * (2.0/n) * beta) / beta)

def project(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = n/2.0 - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
        for i in idx:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
            if diff <= 0: break
    else:
        idx = np.argsort(h)
        for i in idx:
            a = min(-diff, h[i])
            h[i] -= a
            diff += a
            if diff >= 0: break
    return h

def local_search(h, n, max_rounds=2000, time_limit=120.0):
    t0 = time.time()
    h = h.copy()

def slsqp(h, n, t0, time_limit=300.0):
    beta = 40.0
