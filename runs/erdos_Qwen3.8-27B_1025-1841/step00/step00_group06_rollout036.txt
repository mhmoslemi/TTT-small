import numpy as np
import time

def compute_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Water filling
    if diff > 0:
        # Need to add mass, add to elements with most room
        order = np.argsort(h)[::-1]  # most room first (lowest h first)
        order = np.argsort(h)  # actually, add to smallest first
        remaining = diff
        per = remaining / n
        for i in order:
            add = min(per, 1.0 - h[i])
            if add <= 0:
                add = 1.0 - h[i]
            h[i] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        # Need to remove mass, remove from elements with most mass
        order = np.argsort(h)[::-1]  # largest first
        remaining = -diff
        per = remaining / n
        for i in order:
            rem = min(per, h[i])
            h[i] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def subgradient(h, n):
    """Compute subgradient of the objective at h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find active k (argmax)
    k_star = np.argmax(corr)
    # Translate to shift
    # corr[m] corresponds to shift k = m - (n-1)
    k = k_star - (n - 1)
    
    g = np.zeros(n)
    # ∂f_k / ∂h[j] = dx * [(1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)]
    if k >= 0:
        # j-k valid: j >= k
        # j+k valid: j < n-k
        g[k:] += (1.0 - h[:n-k])  # (1-h[j-k]) for j=k..n-1
        g[:n-k] -= h[k:]  # -h[j+k] for j=0..n-k-1
    else:
        # k < 0, let kk = -k
        kk = -k
        # j-k = j+kk valid: j < n-kk
        # j+k = j-kk valid: j >= kk
        g[:n-kk] += (1.0 - h[kk:])  # (1-h[j-k]) = (1-h[j+kk]) for j=0..n-kk-1
        g[kk:] -= h[:n-kk]  # -h[j+k] = -h[j-kk] for j=kk..n-1
    
    return g * dx, k

def optimize_subgradient(h, n, max_iter=5000, base_lr=0.5):
    """Projected subgradient descent"""
    best_h = h.copy()
    best_val = compute_objective(h)
    
    for it in range(max_iter):
        g, k = subgradient(h, n)
        # Adaptive learning rate
        lr = base_lr / (1.0 + it * 0.01)
        h_new = h - lr * g
        h_new = project_h(h_new, n)
        
        val = compute_objective(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_val

def coordinate_descent(h, n, max_sweeps=50):
    """Coordinate descent refinement"""
    best_val = compute_objective(h)
    dx = 2.0 / n
    
    for sweep in range(max_sweps):
        improved = False
        for j in range(n):
            for delta in [0.01, 0.005, 0.001, -0.001, -0.005, -0.01]:
                h_new = h.copy()
                h_new[j] = np.clip(h_new[j] + delta, 0, 1)
                # Adjust sum
                diff = n/2.0 - np.sum(h_new)
                if abs(diff) < 1e-10:
                    h_new = np.clip(h_new, 0, 1)
                    val = compute_objective(h_new)
                    if val < best_val - 1e-10:
                        h = h_new
                        best_val = val
                        improved = True
                        break
        if not improved:
            break
    
    return h, best_val
