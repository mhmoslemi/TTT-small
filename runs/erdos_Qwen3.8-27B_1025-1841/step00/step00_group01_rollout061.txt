        hb2 = (hc + np.random.randn(48) * 0.12 >= 0.5).astype(float)
        target = 24
        s = int(hb2.sum())
        if s > target:
            for idx in np.flatnonzero(hb2)[:s - target]:
                hb2[idx] = 0.0
        elif s < target:
            for idx in np.flatnonzero(1.0 - hb2)[:target - s]:
                hb2[idx] = 1.0
        hb3, c5_3 = binary_sa(hb2, 48, 150000, 0.015, 0.0005)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = hb3, c5_3, 48

    # Phase 5: Higher resolution SLSQP
    for nt in [n0 + 1, n0 * 2, 96, 128]:
        if rem() < 50 or nt > 200:
            continue
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, nt, endpoint=False)
        hi = fix_sum(np.interp(xn, xo, best_h), nt)
        h, c5 = slsqp(hi, nt, 80.0, 300)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, nt

    # Phase 6: Final SLSQP polish on best
    if rem() > 25:
        h, c5 = slsqp(best_h, best_n, 200.0, 600)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: If best is binary, try one more SA pass
    if best_n == 48 and rem() > 40:
        is_binary = np.all((best_h == 0) | (best_h == 1))
        if is_binary:
            hb4, c5_4 = binary_sa(best_h, 48, 300000, 0.01, 0.0002)
            if c5_4 < best_c5:
                best_h, best_c5 = hb4, c5_4

    return best_h, best_c5, best_n
