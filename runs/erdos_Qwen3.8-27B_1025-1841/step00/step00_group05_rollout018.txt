    # Restart phase
    if best_h is not None and budget_s - (time.time() - t0) > 30:
        n_best = best_n

        for r in range(20):

            # Mix with best
            h_r[:n_best//3] = best_h[:n_best//3]

            t_end = time.time() + 40
            h_s, c5_s = sa_swap(h_r, n_best, dx_best, rng, t_end)
            if c5_s < best_c5:
                best_h = h_s
                best_c5 = c5_s
                best_n = n_best
    
    return best_h, best_c5, best_n

import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def greedy_swap(h, n, dx, rng, deadline):
    h = h.copy()
    cur = compute_c5(h, dx)
    while time.time() < deadline:
        improved = False
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        best_val = cur
        best_pair = None
        for _ in range(150):
            i = ones_idx[rng.integers(len(ones_idx))]
            j = zeros_idx[rng.integers(len(zeros_idx))]
            h[i] = 0.0
            h[j] = 1.0
            v = compute_c5(h, dx)
            h[i] = 1.0
            h[j] = 0.0
            if v < best_val:
                best_val = v
                best_pair = (int(i), int(j))
        if best_pair is not None:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            cur = best_val
            improved = True
        if not improved:
            break
    return h, cur

def sa_swap(h, n, dx, rng, deadline):
    h = h.copy()
    cur = compute_c5(h, dx)
    best_h = h.copy()
    best_val = cur
    t_start = time.time()
    total_t = max(deadline - t_start, 1.0)
    t_high = 0.08
    t_low = 0.001
    while time.time() < deadline:
        frac = (time.time() - t_start) / total_t
        temp = t_high * (t_low / t_high) ** frac
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_c5(h, dx)
        delta = new_val - cur
        if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur = new_val
            if cur < best_val:
                best_val = cur
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_val

def make_random_binary(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def two_swap_search(h, n, dx, rng, deadline):
    h = h.copy()
    cur = compute_c5(h, dx)
    best_val = cur
    while time.time() < deadline:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) < 2 or len(zeros_idx) < 2:
            break
        best_delta = 0.0
        best_quartet = None
        for _ in range(60):
            i1 = ones_idx[rng.integers(len(ones_idx))]
            i2 = ones_idx[rng.integers(len(ones_idx))]
            j1 = zeros_idx[rng.integers(len(zeros_idx))]
            j2 = zeros_idx[rng.integers(len(zeros_idx))]
            if i1 == i2 or j1 == j2:
                continue
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
            v = compute_c5(h, dx)
            delta = v - cur
            h[j1] = 0.0; h[j2] = 0.0; h[i1] = 1.0; h[i2] = 1.0
            if delta < best_delta:
                best_delta = delta
                best_quartet = (int(i1), int(i2), int(j1), int(j2))
        if best_quartet is not None and best_delta < -1e-14:
            i1, i2, j1, j2 = best_quartet
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
            cur = cur + best_delta
            best_val = cur
        else:
            break
    return h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    parent = None
    if 'initial_h_values' in dir() and initial_h_values is not None:
        parent = np.asarray(initial_h_values, dtype=float)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if parent is not None:
        n0 = len(parent)
        dx0 = 2.0 / n0
        h0 = binarize(parent, n0)
        c5_0 = compute_c5(h0, dx0)
        best_h = h0.copy()
        best_c5 = c5_0
        best_n = n0

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Intensive optimization at parent resolution
    if parent is not None and time_left() > 60:
        n0 = len(parent)
        dx0 = 2.0 / n0
        h_work = binarize(parent, n0)

        # Greedy single swaps
        t_end = time.time() + min(100, time_left() * 0.15)
        h_work, c5_g = greedy_swap(h_work, n0, dx0, rng, t_end)
        if c5_g < best_c5:
            best_h = h_work.copy()
            best_c5 = c5_g
            best_n = n0

        # Two-swap search
        if time_left() > 40:
            t_end = time.time() + min(60, time_left() * 0.1)
            h_work, c5_2s = two_swap_search(h_work, n0, dx0, rng, t_end)
            if c5_2s < best_c5:
                best_h = h_work.copy()
                best_c5 = c5_2s
                best_n = n0

        # Simulated annealing (long run)
        if time_left() > 80:
            t_end = time.time() + min(400, time_left() * 0.35)
            h_work, c5_sa = sa_swap(h_work, n0, dx0, rng, t_end)
            if c5_sa < best_c5:
                best_h = h_work.copy()
                best_c5 = c5_sa
                best_n = n0

        # Another greedy pass after SA
        if time_left() > 30:
            t_end = time.time() + min(60, time_left() * 0.1)
            h_work, c5_g2 = greedy_swap(h_work, n0, dx0, rng, t_end)
            if c5_g2 < best_c5:
                best_h = h_work.copy()
                best_c5 = c5_g2
                best_n = n0

    # Phase 2: Try other resolutions
    n_candidates = [60, 80, 100, 120, 140]
    for n in n_candidates:
        if n % 2 != 0 or time_left() < 50:
            continue
        dx = 2.0 / n
        if parent is not None and n == len(parent):
            continue
        if parent is not None:
            pos = np.linspace(0, len(parent) - 1, n)
            h_init = np.interp(pos, np.arange(len(parent)), parent)
        else:
            h_init = make_random_binary(n, seed)
        h_bin = binarize(h_init, n)
        c5_init = compute_c5(h_bin, dx)
        if c5_init < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_init
            best_n = n

        t_greedy = time.time() + min(40, time_left() * 0.15)
        h_g, c5_g = greedy_swap(h_bin, n, dx, rng, t_greedy)
        if c5_g < best_c5:
            best_h = h_g.copy()
            best_c5 = c5_g
            best_n = n

        if time_left() > 40:
            t_sa = time.time() + min(100, time_left() * 0.25)
            h_s, c5_s = sa_swap(h_g, n, dx, rng, t_sa)
            if c5_s < best_c5:
                best_h = h_s.copy()
                best_c5 = c5_s
                best_n = n

    # Phase 3: Restarts on best resolution
    if best_h is not None and time_left() > 30:
        n_best = best_n
        dx_best = 2.0 / n_best
        for r in range(30):
            if budget_s - (time.time() - t0) < 15:
                break
            h_r = make_random_binary(n_best, seed + 5000 + r)
            # Hybrid: keep some structure from best
            n_keep = n_best // 3
            h_r[:n_keep] = best_h[:n_keep]
            h_r = binarize(h_r, n_best)
            t_end = time.time() + min(45, time_left() * 0.4)
            h_s, c5_s = sa_swap(h_r, n_best, dx_best, rng, t_end)
            if c5_s < best_c5:
                best_h = h_s.copy()
                best_c5 = c5_s
                best_n = n_best

    return best_h, best_c5, best_n
