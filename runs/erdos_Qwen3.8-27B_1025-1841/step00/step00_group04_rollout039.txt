        return h_start

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: optimize at initial resolution
    n = n_init
    for attempt in range(5):
        if time_left() < 30:
            break
        h_start = best_h.copy()
        if attempt > 0:
            h_start = h_start + np.random.randn(n) * 0.03 * (attempt * 0.5)
            h_start = project_feasible(h_start, n)
        h_new = optimize_slsqp(h_start, n, max_iter=400, ftol=1e-12)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: try higher resolutions
    resolutions = [56, 72, 84, 112, 144, 168, 224, 280, 336, 448, 560, 672]
    for new_n in resolutions:
        if new_n % 2 != 0:
            continue
        if time_left() < 45:
            break
        if new_n == n_init:
            continue

        h_start = upsample_h(best_h, new_n)

        if new_n <= 200:
            max_iter = min(250, max(40, int(time_left() / max(1, new_n * 0.002))))
            h_new = optimize_slsqp(h_start, new_n, max_iter=max_iter, ftol=1e-11)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        else:
            h_new = optimize_lbfgsb(h_start, new_n, alpha=50.0, max_iter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

        if time_left() > 40:
            for alpha in [100.0, 200.0]:
                if time_left() < 25:
                    break
                h_new2 = optimize_lbfgsb(h_start, new_n, alpha=alpha, max_iter=100)
                c5_new2 = compute_c5(h_new2)
                if c5_new2 < best_c5:
                    best_h = h_new2
                    best_c5 = c5_new2

    # Phase 3: random restarts at best resolution
    n_best = len(best_h)
    restarts = max(1, int(time_left() / max(1, n_best * 0.003)))
    restarts = min(restarts, 20)
    for _ in range(restarts):
        if time_left() < 20:
            break
        h_start = best_h + np.random.randn(n_best) * 0.02
        h_start = project_feasible(h_start, n_best)
        if n_best <= 200:
            h_new = optimize_slsqp(h_start, n_best, max_iter=200, ftol=1e-12)
        else:
            h_new = optimize_lbfgsb(h_start, n_best, alpha=80.0, max_iter=100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: final polish
    if time_left() > 15:
        h_new = optimize_slsqp(best_h, n_best, max_iter=500, ftol=1e-14)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
