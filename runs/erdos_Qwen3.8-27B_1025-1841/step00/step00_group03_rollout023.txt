    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def soft_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

def repair(h):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(20):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            # If all at bounds, spread uniformly
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def soft_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

def penalized_soft_max_obj(h, beta, n, penalty_coeff):
    base = soft_max_obj(h, beta)
    sum_h = np.sum(h)
    target = n / 2.0
    return base + penalty_coeff * (sum_h - target) ** 2

def repair(h):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_up(h, n_new):
    n_old = len(h)

        return repair(h.copy())

    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return repair(h_new)

def make_block(initialize, n):
    h = np.zeros(n)
    for (a, b) in initialize:
        ia = int(round(a * n / 2.0))
        ib = int(round(b * n / 2.0))
