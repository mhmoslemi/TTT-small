import numpy as np
import time

def project_to_constraint(h, target_sum):
    """Project h onto {h: 0<=h<=1, sum(h)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def get_corrs(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def eval_c5(h):
    return np.max(get_corrs(h, 2.0 / len(h)))

def resample_to(h_from, n_to):
    n_from = len(h_from)
    xf = (np.arange(n_from) + 0.5) * 2.0 / n_from
    xt = (np.arange(n_to) + 0.5) * 2.0 / n_to
    return np.interp(xt, xf, h_from)

def grad_smooth(h, dx, alpha):
    n = len(h)
    c = get_corrs(h, dx)
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    d = np.arange(-n + 1, n)
    j = np.arange(n)
    jp = j[:, None] + d[None, :]
    jm = j[:, None] - d[None, :]
    hp = np.where((jp >= 0) & (jp < n), h[np.clip(jp, 0, n - 1)], 0.0)
    hm = np.where((jm >= 0) & (jm < n), h[np.clip(jm, 0, n - 1)], 0.0)
    return dx * (w[None, :] * ((1.0 - hp) - hm)).sum(axis=1)

import numpy as np
import time

def project_to_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def get_corrs(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def eval_c5(h):
    return float(np.max(get_corrs(h, 2.0 / len(h))))

def resample_to(h_from, n_to):
    n_from = len(h_from)
    xf = (np.arange(n_from) + 0.5) * 2.0 / n_from
    xt = (np.arange(n_to) + 0.5) * 2.0 / n_to
    return np.interp(xt, xf, h_from)

def grad_smooth(h, dx, alpha):
    n = len(h)
    c = get_corrs(h, dx)
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    d = np.arange(-n + 1, n)
    j = np.arange(n)
    jp = j[:, None] + d[None, :]
    hm_arr = j[:, None] - d[None, :]
    hp = np.where((jp >= 0) & (jp < n), h[np.clip(jp, 0, n - 1)], 0.0)
    hm = np.where((hm_arr >= 0) & (hm_arr < n), h[np.clip(hm_arr, 0, n - 1)], 0.0)
    return dx * (w[None, :] * ((1.0 - hp) - hm)).sum(axis=1)

def pgd(h0, n, dx, target, alpha, deadline, iters=5000):
    h = project_to_constraint(h0.copy(), target)
    best = eval_c5(h)
    bh = h.copy()
    lr = 0.5
    cur_max = np.max(get_corrs(h, dx))
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = grad_smooth(h, dx, alpha)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        improved = False
        for _ in range(40):
            h2 = project_to_constraint(h - lr * g, target)
            m2 = np.max(get_corrs(h2, dx))
            if m2 <= cur_max + 1e-12:
                h = h2
                cur_max = m2
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr = 0.3
            idx = np.random.randint(n)
            h = project_to_constraint(h + 0.05 * np.random.randn(n), target)
            cur_max = np.max(get_corrs(h, dx))
        c = eval_c5(h)
        if c < best - 1e-12:
            best = c
            bh = h.copy()
    return bh, best

def bin_search(hb, n, dx, deadline, iters=50000, seed=0):
    rs = np.random.RandomState(seed)
    h = hb.astype(float).copy()
    best = eval_c5(h)
    bh = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    temp = 0.01
    for it in range(iters):
        if time.time() > deadline:
            break
        temp = max(0.001, 0.01 * (1.0 - it / float(iters)))
        i = ones[rs.randint(len(ones))]
        j = zeros[rs.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c = eval_c5(h)
        if c < best:

            bh = h.copy()
            ones = np.where(h > 0.5)[0]
