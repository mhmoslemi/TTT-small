        tl = min(per_start, remaining)
        h, c5 = sa_optimize(h_s, n1, T0=2.0, T_end=0.005, n_iter=80000, seed=seed + 1000 * s_idx, time_limit=tl)
        h2, c52 = local_search_binary(h, n1, max_sweeps=30, time_limit=min(20, remaining - 5), seed=seed + s_idx)
        if c52 < c5:
            h, c5 = h2, c52
        update(h, c5, n1)

    # Phase 2: Progressive upsampling and optimization
    current_h = best_h if best_h is not None else make_golden(n1)
    current_n = best_n if best_n > 0 else n1

    target_ns = [140, 280, 560, 1120]
    for tn in target_ns:
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 60
        if remaining < 20:
            break
        h_up = upsample_binary(current_h, current_n, tn, seed=seed + tn)
        sa_time = min(remaining * 0.5, 200)
        h, c5 = sa_optimize(h_up, tn, T0=1.0, T_end=0.002, n_iter=40000, seed=seed + 5000 + tn, time_limit=sa_time)
        ls_time = min(remaining * 0.15, 60)
        h2, c52 = local_search_binary(h, tn, max_sweeps=20, time_limit=ls_time, seed=seed + tn)
        if c52 < c5:
            h, c5 = h2, c52
        update(h, c5, tn)
        current_h = h
        current_n = tn

    # Phase 3: Final refinement at the best resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 5
    if remaining > 15 and best_h is not None:
        h, c5 = local_search_binary(best_h, best_n, max_sweeps=50, time_limit=remaining, seed=seed + 9999)
        update(h, c5, best_n)

    if best_h is None:
        best_h = make_golden(70)
        best_n = 70
        best_c5 = compute_c5(best_h, best_n)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
