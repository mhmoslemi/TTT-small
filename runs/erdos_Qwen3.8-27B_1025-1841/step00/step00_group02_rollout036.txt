    # Compute all correlations
    corr = np.correlate(h, 1-h, mode='full')  # length 2n-1

    tau_corr -= np.max(tau_corr)  # for numerical stability

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_and_corr(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5, corr

def project_to_constraint(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    if abs(np.sum(h) - target_sum) < 1e-12:
        return np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_grad_smooth_max(h, temperature):
    """Compute gradient of (1/T)*log(sum exp(T*C_s)) with respect to h"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')  # length 2n-1
    
    # Softmax weights (temperature scaling)
    tc = temperature * corr
    tc -= np.max(tc)
    weights = np.exp(tc)
    weights /= np.sum(weights)
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        s = m - (n - 1)
        if s >= 0:
            # dC/dh[j] = (1-h[j-s]) for j in [s, n-1]
            #           - h[j+s] for j in [0, n-1-s]
            if s < n:
                grad[s:] += w * (1 - h[:n-s])
                grad[:n-s] -= w * h[s:]
        else:
            t = -s
            if t < n:
                grad[:n-t] += w * (1 - h[t:])
                grad[t:] -= w * h[:n-t]
    
    return grad

def projected_gradient_step(h, grad, lr, target_sum):
    """Take a projected gradient step"""
    h_new = h - lr * grad
    h_new = project_to_constraint(h_new, target_sum)
    return h_new

def optimize_smooth_max(h, target_sum, n_steps=200, init_temp=10, final_temp=200, 
                         init_lr=0.01, final_lr=0.001):
    """Annealed projected gradient for smooth max"""
    n = len(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * (2.0/n)
    
    for step in range(n_steps):
        t_frac = step / max(n_steps - 1, 1)
        temp = init_temp + t_frac * (final_temp - init_temp)
        lr = init_lr * (1 - t_frac) + final_lr * t_frac
        
        grad = compute_grad_smooth_max(h, temp)
        h_new = projected_gradient_step(h, grad, lr, target_sum)
        
        c5_new = np.max(np.correlate(h_new, 1-h_new, mode='full')) * (2.0/n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def local_search(h, n_iterations=5000, n=45):
    """Simple local search: try changing individual values"""
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * (2.0/n)
    dx = 2.0 / n
    rng = np.random.RandomState(42)
    
    for it in range(n_iterations):
        i = rng.randint(n)
        old_val = h[i]
        if old_val == 0 or old_val == 1:
            # Try changing to a random value
            new_val = rng.random()
        else:
            # Try snapping to 0 or 1, or changing slightly
            choice = rng.random()
            if choice < 0.3:
                new_val = 0.0
            elif choice < 0.6:
                new_val = 1.0
            else:
                new_val = np.clip(old_val + rng.normal(0, 0.1), 0, 1)
        
        if abs(new_val - old_val) < 1e-10:
            continue
        
        # Try the change, adjusting another index to maintain sum
        delta = new_val - old_val
        # Find j to adjust
        j = (i + 1 + rng.randint(n-1)) % n
        if j == i:
            j = (j + 1) % n
        
        h_backup = h.copy()
        h[i] = new_val
        h[j] = h[j] - delta
        h[j] = np.clip(h[j], 0, 1)
        
        # Check if sum is still correct
        if abs(np.sum(h) - target_sum) > 1e-6:
            h = h_backup
            continue
        
        c5_new = np.max(np.correlate(h, 1-h, mode='full')) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            h = h_backup
    
    return best_h, best_c5

def compute_c5(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Binary search for shift
    lo, hi = -1.0, 1.0

        if s > target_sum:
            lo = mid
        else:
