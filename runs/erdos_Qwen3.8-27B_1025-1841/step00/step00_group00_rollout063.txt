    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    # Smooth max: (1/T) * log(sum(exp(T * corr)))
    # For numerical stability:
    max_c = np.max(corr)
    shifted = T * (corr - max_c)
    log_sum = max_c + np.log(np.sum(np.exp(shifted))) / T
    return log_sum

def smooth_gradient(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    max_c = np.max(corr)

    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)  # softmax

import numpy as np
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_box(x, target_sum):
    n = len(x)
    s = np.sum(x)
    if abs(s - target_sum) < 1e-14:
        return np.clip(x, 0.0, 1.0)
    lo = -3.0
    hi = 4.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s_mid = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s_mid > target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(x - t, 0.0, 1.0)

def upsample(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    return project_sum_box(new_h, new_n / 2.0)

def smooth_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    shifted = T * (corr - max_c)
    shifted = shifted - np.max(shifted)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_s / total
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag > 0:
            j = lag
            grad[0:n-j] += w * (1.0 - h[j:n])
            grad[j:n] -= w * h[0:n-j]
        elif lag < 0:
            j = -lag
            grad[j:n] += w * (1.0 - h[0:n-j])
            grad[0:n-j] -= w * h[j:n]
        else:
            grad += w * (1.0 - 2.0 * h)
    grad *= dx
    return grad

def subgradient_descent(h, n_iter, T, lr0, deadline):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        if time() > deadline:
            break
        g = smooth_gradient(h, T)
        lr = lr0 / (1.0 + 0.001 * it)
        h_new = h - lr * g
        h = project_sum_box(h_new, n / 2.0)
        if it % 10 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:

                best_h = h.copy()
    c5 = compute_c5(h)

        best_h = h.copy()
    return best_h, best_c5

def local_search(h, deadline):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time() < deadline:

        if i == j:

        old_i, old_j = h[i], h[j]

            else:
