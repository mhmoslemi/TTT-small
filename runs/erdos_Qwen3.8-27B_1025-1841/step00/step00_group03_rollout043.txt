```python
def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

def repair_h(h, n):
    """Repair h to satisfy 0<=h[i]<=1 and sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    diff = target_sum - current_sum
    # Distribute the difference
    if abs(diff) > 1e-10:
        # Find positions where we can add or subtract
        if diff > 0:
            # Need to increase some values
            mask = h < 1.0
        else:
            # Need to decrease some values
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
            # Second pass if needed
            current_sum = np.sum(h)
            diff = target_sum - current_sum
            if abs(diff) > 1e-10:
                if diff > 0:
                    mask = h < 1.0
                else:
                    mask = h > 0.0
                if np.sum(mask) > 0:
                    h[mask] += diff / np.sum(mask)
                    h = np.clip(h, 0, 1)
    return h

    eval_count = 0
    while time.time() - start_time < budget_s - 5:
        h_current = best_h.copy()
        current_c5 = best_c5
        
        while time.time() - start_time < budget_s - 5:
            # Generate perturbation

            j = np.random.randint(n)
            if i == j:

            delta = np.random.randn() * T

                if abs(still_off) > 1e-10:

            new_c5 = evaluate_c5(new_h)
            eval_count += 1
            
            if new_c5 < current_c5:
                h_current = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = current_c5
            else:

                if T > min_T:
```

One more optimization: instead of computing the full correlation each time in the local search, I can use the delta formula I derived earlier:
Δcv[k] = h[i+k] - h[j+k] + h[i-k] - h[j-k]

This lets me compute the new c5 in O(n) instead of O(n²) for the local search. Let me use this!

Given the current cv array and a swap (i, j):

```python
def evaluate_c5_fast(cv, dx):

def c5_from_correlation(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def swap_delta_cv(h, i, j, cv, n):
    """Compute delta in cv after swapping h[i]=1->0, h[j]=0->1"""
    # cv is indexed from 0 to 2n-2, corresponding to lags -(n-1) to (n-1)
    # cv[k] corresponds to lag k - (n-1)
    # For lag L: cv[n-1+L] = sum_m h[m+L]*(1-h[m])
    # Delta for changing h[i] from 1 to 0:
    #   +h[m+L] for m=i (h[i+L]*(1-0) - h[i+L]*(1-1) = h[i+L])
    #   -(1-h[m+L]) for m+L=i, i.e., m=i-L (h[i]*(1-h[i-L]) was there, now gone)
    # Delta for changing h[j] from 0 to 1:
    #   -h[m+L] for m=j (h[j+L]*(1-1) - h[j+L]*(1-0) = -h[j+L])
    #   +(1-h[m+L]) for m+L=j, i.e., m=j-L (new term h[j]*(1-h[j-L]) = 1-h[j-L])
    
    # So delta_cv[L] = h[i+L] - h[j+L] - (1-h[i-L]) + (1-h[j-L])
    #                 = h[i+L] - h[j+L] + h[i-L] - h[j-L]
    # (with out-of-bounds terms = 0)
    
    delta = np.zeros(2*n-1)
    for L in range(-(n-1), n):
        val = 0.0
        if 0 <= i+L < n:
            val += h[i+L]
        if 0 <= j+L < n:
            val -= h[j+L]
        if 0 <= i-L < n:
            val += h[i-L]
        if 0 <= j-L < n:
            val -= h[j-L]
        delta[n-1+L] = val
    
    return delta
```

```python
def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlation(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr, dx

def swap_delta_cv(h, i, j, n):
    L = np.arange(-(n-1), n)
    idx_ip = i + L
    h_ip = np.where((idx_ip >= 0) & (idx_ip < n), h[np.clip(idx_ip, 0, n-1)], 0.0)
    idx_jp = j + L
    h_jp = np.where((idx_jp >= 0) & (idx_jp < n), h[np.clip(idx_jp, 0, n-1)], 0.0)
    idx_im = i - L
    h_im = np.where((idx_im >= 0) & (idx_im < n), h[np.clip(idx_im, 0, n-1)], 0.0)
    idx_jm = j - L
    h_jm = np.where((idx_jm >= 0) & (idx_jm < n), h[np.clip(idx_jm, 0, n-1)], 0.0)
    return h_ip - h_jp + h_im - h_jm

def local_search(h_ls, cv, dx, time_limit):

    c5_ls = float(np.max(cv) * dx)

                if new_c5 < best_improve:
```

```python
def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb
```
