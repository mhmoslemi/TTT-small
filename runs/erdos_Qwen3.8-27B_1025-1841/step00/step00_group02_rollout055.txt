```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n):
    """Project h onto feasible set: 0<=h<=1, sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    diff = target - np.sum(h)
    if diff > 0:
        # Need to increase: add to elements that are < 1
        mask = h < 1.0
        if np.sum(mask) == 0:
            return h
        add = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + add, 0, 1)
    elif diff < 0:
        # Need to decrease: subtract from elements that are > 0
        mask = h > 0.0
        if np.sum(mask) == 0:
            return h
        sub = -diff / np.sum(mask)
        h[mask] = np.clip(h[mask] - sub, 0, 1)
    # Iterate for exactness
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if np.sum(mask) == 0:
            break
        adj = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + adj, 0, 1)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_init = np.full(45, 0.5)
    
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    elapsed = lambda: time.time() - start_time
    time_limit = budget_s - 30
    
    # Phase 1: Coordinate descent on parent resolution
    # For each coordinate, try to reduce C5 by adjusting h[i]
    step_size = 0.01
    for iteration in range(50000):
        if elapsed() > time_limit * 0.3:
            break
        i = rng.integers(n)
        old_val = best_h[i]
        # Try increasing and decreasing
        for delta in [step_size, -step_size, step_size*0.5, -step_size*0.5]:
            candidate = best_h.copy()
            candidate[i] = old_val + delta
            candidate = project_h(candidate, n)
            c5 = evaluate_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
                break
        step_size *= 0.999
    
    # Phase 2: Simulated annealing
    T = 0.01
    T_end = 0.001
    for iteration in range(200000):
        if elapsed() > time_limit:
            break
        # Block move
        block_len = rng.integers(1, min(8, n))
        start = rng.integers(n - block_len + 1)
        candidate = best_h.copy()
        candidate[start:start+block_len] += rng.normal(0, 0.05, block_len)
        candidate = project_h(candidate, n)
        c5 = evaluate_c5(candidate)
        delta = c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            best_c5 = c5
            best_h = candidate
        T = max(T_end, T * 0.9999)
    
    return best_h, best_c5, n
```

```python
def subgradient_step(h, n, lr):
    corr = np.correlate(h, 1-h, mode="full")
    worst = np.argmax(corr)
    m = worst - (n-1)  # lag
    
    # Compute subgradient of c[m] w.r.t. h
    grad = np.zeros(n)
    if m >= 0:
        # c[m] = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        # d/dh[j] = (1-h[j-m]) for j in [m, n-1]  (from h[j] appearing as h[i+m] at i=j-m)
        #          - h[j] for j in [0, n-1-m]  (from (1-h[j]) at i=j)
        for j in range(m, n):
            grad[j] += (1 - h[j-m])
        for j in range(0, n-m):
            grad[j] -= h[j]
    else:
        m_pos = -m
        # c[-m_pos] = sum_{i=m_pos}^{n-1} h[i]*(1-h[i-m_pos])
        for j in range(m_pos, n):
            grad[j] += (1 - h[j-m_pos])
        for j in range(0, n-m_pos):
            grad[j] -= h[j]
    
    # Descent step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new
```
