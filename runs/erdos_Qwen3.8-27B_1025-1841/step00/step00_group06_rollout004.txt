The total number of pairs at distance k is N-k. Among these:
- (1,1) pairs: a_k
- (1,0) pairs: c_k  
- (0,1) pairs: d_k
- (0,0) pairs: e_k
a_k + c_k + d_k + e_k = N-k

Also, sum of h over first N-k positions minus sum over last N-k positions relates to c_k - d_k.

For a balanced sequence (N/2 ones total), the expected c_k is roughly (N-k)/4.

To minimize max_k c_k, we want to distribute the (1,0) pairs as evenly as possible across lags.

import numpy as np
import math
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(n_points, rng):
    h = np.zeros(n_points, dtype=np.float64)
    positions = rng.permutation(n_points)[:n_points // 2]
    h[positions] = 1.0
    return h

def sa_search(h, n_points, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    current = h.copy()
    current_c5 = compute_c5(current)
    best = current.copy()
    best_c5 = current_c5
    start = time.time()
    t = 0.005
    t_min = 0.0005
    cooling = 0.9995
    stall = 0
    max_stall = 3000

    while time.time() - start < time_limit:
        ones = np.where(current == 1.0)[0]
        zeros = np.where(current == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        current[i] = 0.0
        current[j] = 1.0
        new_c5 = compute_c5(current)
        delta = new_c5 - current_c5
        if delta <= 0 or rng.random() < math.exp(-delta / t):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best = current.copy()
                best_c5 = new_c5
                stall = 0
            else:
                stall += 1
        else:
            current[i] = 1.0
            current[j] = 0.0
            stall += 1
        t = max(t * cooling, t_min)
        if stall > max_stall:
            current = best.copy()
            current_c5 = best_c5
            t = 0.005
            stall = 0
            max_stall = int(max_stall * 1.4)
            for _ in range(rng.randint(2, 6)):
                ones = np.where(current == 1.0)[0]
                zeros = np.where(current == 0.0)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                current[i] = 0.0
                current[j] = 1.0
            current_c5 = compute_c5(current)
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    global_best_h = None
    global_best_c5 = float('inf')
    global_best_n = 0
    rng_main = np.random.RandomState(seed)

    try:
        init = initial_h_values
        if init is not None and len(init) > 0:
            init_arr = np.array(init, dtype=np.float64)
            init_n = len(init_arr)
            h_init = np.where(init_arr > 0.5, 1.0, 0.0)
            n_ones = int(np.sum(h_init))
            target = init_n // 2
            if n_ones > target:
                idx = np.where(h_init == 1)[0]
                rng_main.shuffle(idx)
                h_init[idx[:n_ones - target]] = 0.0
            elif n_ones < target:
                idx = np.where(h_init == 0)[0]
                rng_main.shuffle(idx)
                h_init[idx[:target - n_ones]] = 1.0
            global_best_h = h_init.copy()
            global_best_c5 = compute_c5(global_best_h)
            global_best_n = init_n
    except (NameError, TypeError):
        pass

    resolutions = [80, 96, 100, 120, 128, 160, 200, 256, 320]
    for n_pts in resolutions:
        if n_pts % 2 != 0:
            continue
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 20
        if remaining < 12:
            break
        rel_cost = (n_pts / 100.0) ** 2
        alloc = min(remaining / 4.0, 100.0 / rel_cost)
        alloc = max(alloc, 12.0)
        n_restarts = max(2, int(alloc / 12.0))
        actual_alloc = alloc / n_restarts
        for r in range(n_restarts):
            h_start = make_binary(n_pts, rng_main)
            h_opt, c5_opt = sa_search(h_start, n_pts, actual_alloc, seed=rng_main.randint(10**7))
            if c5_opt < global_best_c5:
                global_best_c5 = c5_opt
                global_best_h = h_opt.copy()
                global_best_n = n_pts

    if global_best_h is not None:

        if c5_pol < global_best_c5:
            global_best_c5 = c5_pol
            global_best_h = h_pol.copy()

    if global_best_h is None:
        n_pts = 100

        global_best_n = n_pts

    return global_best_h, global_best_c5, global_best_n
