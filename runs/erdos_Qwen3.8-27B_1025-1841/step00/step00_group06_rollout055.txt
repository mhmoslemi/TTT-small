```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, n):
    """Project h onto {h in [0,1]^n : sum(h) = n/2}"""
    target = n / 2.0
    # Simple: scale and clip, iterate
    for _ in range(100):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Distribute the difference
        diff = target - s
        # Add to elements with room to grow, remove from elements that are > 0
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if mask.any():
                h[mask] += diff / mask.sum()
            else:
                h += diff / n
        else:
            mask = h > 1e-12
            if mask.any():
                h[mask] += diff / mask.sum()
            else:
                h += diff / n
    h = np.clip(h, 0, 1)
    # Final fix
    h += (target - h.sum()) / n
    h = np.clip(h, 0, 1)
    return h

def random_perturb(h, n, scale=0.1):
    """Randomly perturb h and project back to constraint"""
    h2 = h + np.random.randn(n) * scale
    return project_to_constraint(h2, n)

def coordinate_descent_step(h, n):
    """Try to improve h by changing one element at a time"""
    best = compute_c5(h)
    target = n / 2.0
    for i in range(n):
        # Try different values for h[i]
        candidates = [0.0, 1.0, max(0.0, h[i] - 0.1), min(1.0, h[i] + 0.1)]
        candidates = [c for c in candidates if c != h[i]]
        for new_val in candidates:
            h_trial = h.copy()
            h_trial[i] = new_val
            # Fix sum constraint: adjust element with most room
            diff = target - h_trial.sum()
            if abs(diff) < 1e-12:
                pass
            else:
                # Find best element to adjust
                if diff > 0:
                    # Need to add to some elements
                    rooms = 1.0 - h_trial
                    rooms[i] = -1  # don't use i
                    best_j = np.argmax(rooms)
                    if rooms[best_j] < 1e-9:
                        continue
                    h_trial[best_j] += diff
                else:
                    # Need to remove from some elements
                    rooms = h_trial.copy()
                    rooms[i] = -1
                    best_j = np.argmax(rooms)
                    if rooms[best_j] < 1e-9:
                        continue
                    h_trial[best_j] += diff
                h_trial = np.clip(h_trial, 0, 1)
                if abs(h_trial.sum() - target) > 1e-9:
                    continue
            c = compute_c5(h_trial)
            if c < best - 1e-12:
                h = h_trial
                best = c
    return h, best

def run(seed=42, budget_s=1300, **kwargs):

    start_time = time.time()
    
    # Get initial values
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:

        h = project_to_constraint(h, 95)
    
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Try coordinate descent
    deadline = start_time + budget_s * 0.3
    while time.time() < deadline:
        h, c5 = coordinate_descent_step(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try random perturbations
    deadline2 = start_time + budget_s * 0.6
    while time.time() < deadline2:
        h_new = random_perturb(best_h, n, scale=0.05)
        h_new, c5 = coordinate_descent_step(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    
    # Try different resolutions
    deadline3 = start_time + budget_s * 0.9
    for n_try in [50, 64, 80, 95, 128, 150, 200]:
        if time.time() > deadline3:
            break
        # Interpolate
        h_interp = np.interp(
            np.linspace(0, 1, n_try, endpoint=False),
            np.linspace(0, 1, n, endpoint=False),
            best_h
        )
        h_interp = project_to_constraint(h_interp, n_try)
        # A few CD steps
        for _ in range(20):
            if time.time() > deadline3:
                break
            h_interp, c5 = coordinate_descent_step(h_interp, n_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_interp.copy()
                n = n_try
    
    return best_h, best_c5, len(best_h)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    return h

def binary_swap_search(h, n, time_budget, rng):
    best = compute_c5(h)
    best_arr = h.copy()
    t_end = time.time() + time_budget
    ones_idx = np.where(h == 1)[0].tolist()
    zeros_idx = np.where(h == 0)[0].tolist()
    no_imp = 0
    stall_limit = 8000
    t_last_imp = time.time()
    while time.time() < t_end:
        if no_imp > stall_limit:
            break
        if time.time() - t_last_imp > 15:
            break
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        best_arr[i] = 0
        best_arr[j] = 1
        c = compute_c5(best_arr)
        if c < best - 1e-13:
            best = c
            oi = ones_idx.index(i)
            ones_idx[oi] = j
            zi = zeros_idx.index(j)
            zeros_idx[zi] = i
            no_imp = 0
            t_last_imp = time.time()
        else:
            best_arr[i] = 1
            best_arr[j] = 0
            no_imp += 1
    return best_arr, best

        trial = best_arr.copy()
        trial[i] = new_val
        trial = fix_sum(trial, target)
        c = compute_c5(trial)
        if c < best - 1e-13:
            best = c
            best_arr = trial
            step = max(0.003, step * 0.995)
            no_imp = 0
        else:
            step = min(0.4, step * 1.008)
```
