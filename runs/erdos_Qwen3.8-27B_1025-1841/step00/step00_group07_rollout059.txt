from scipy.optimize import minimize
import time

_N = 0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    return np.sum(h) - _N / 2.0

def con_jac(h):
    return np.ones(_N)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 0.9999
            s = np.sum(mask)
            if s > 0:
                h[mask] = np.minimum(1.0, h[mask] + diff / s)
        else:
            mask = h > 0.0001
            s = np.sum(mask)
            if s > 0:
                h[mask] = np.maximum(0.0, h[mask] + diff / s)

from scipy.optimize import minimize
import time

_N = 0
_BETA = 80.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m < -50.0 / _BETA:
        return m
    shifted = corr - m
    return m + np.log(np.sum(np.exp(_BETA * shifted))) / _BETA
