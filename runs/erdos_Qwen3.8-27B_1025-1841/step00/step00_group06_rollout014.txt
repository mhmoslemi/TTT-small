def project_onto_constraint(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: h_proj = clip(h - lambda, 0, 1), find lambda s.t. sum = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2

def grad_of_lag(h, n, lag):
    k = lag - n + 1
    idx1 = np.arange(n) - k
    idx2 = np.arange(n) + k
    valid1 = (idx1 >= 0) & (idx1 < n)

                break
    return best, best_c5

def slsqp_refine(h, N, elapsed_fn, time_limit):
    dx = 2.0 / N
    c5_init = get_c5(h, N)
    x0 = np.zeros(N + 1)
    x0[:N] = h
    x0[N] = c5_init + 0.05

    def obj(x):
        return x[N]

    def obj_grad(x):
        g = np.zeros(N + 1)
        g[N] = 1.0
        return g

    cons = []
    n_out = 2 * N - 1
    for m in range(n_out):
        def make_ineq(m_idx):
            def ineq(x):
                hv = x[:N]
                corr_m = np.dot(hv, np.roll(hv, -(m_idx - N + 1)))
                # Compute out[m] properly
                k = m_idx - N + 1
                # out[m] = sum_{valid n} h[n+k]*(1-h[n])
                out_m = 0.0
                for n in range(max(0, -k), min(N, N - k)):
                    out_m += hv[n + k] * (1.0 - hv[n])
                return x[N] - out_m * dx
            return ineq
        cons.append({'type': 'ineq', 'fun': make_ineq(m)})

    cons.append({'type': 'eq', 'fun': lambda x: np.sum(x[:N]) - N / 2.0,
                 'jac': lambda x: np.concatenate([np.ones(N), [-1.0]])})

    bounds = [(0.0, 1.0)] * N + [(0.0, 5.0)]

    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       constraints=cons, bounds=bounds,
                       options={'maxiter': 500, 'ftol': 1e-10, 'disp': False})
        if res.success and res.fun < c5_init:
            h_new = project_h(res.x[:N], N)
            c5_new = get_c5(h_new, N)
            if c5_new < c5_init:
                return h_new, c5_new
    except Exception:
        pass
    return h, get_c5(h, N)

def binary_push(h, N, elapsed_fn, time_limit):
    best = h.copy()
    best_c5 = get_c5(h, N)
    for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
        if elapsed_fn() > time_limit:
            break
        h_bin = (h > thresh).astype(float)
        h_bin = project_h(h_bin, N)
        h_opt, c5_opt = proj_gradient_descent(h_bin, N, elapsed_fn, time_limit)
        if c5_opt < best_c5:
            best = h_opt
            best_c5 = c5_opt
    return best, best_c5

def random_restart(h, N, n_restarts, elapsed_fn, time_limit):
    best = h.copy()
    best_c5 = get_c5(h, N)
    for r in range(n_restarts):
        if elapsed_fn() > time_limit:
            break
        noise = np.random.randn(N) * 0.1
        h_r = project_h(h + noise, N)
        h_opt, c5_opt = proj_gradient_descent(h_r, N, elapsed_fn, time_limit)
        if c5_opt < best_c5:
            best = h_opt
            best_c5 = c5_opt
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t0

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        N0 = len(h0)
    else:
        N0 = 100
        h0 = np.full(N0, 0.5)

    best_h = h0.copy()
    best_c5 = get_c5(best_h, N0)
    best_N = N0

    def update_best(h, c5, N):
        nonlocal best_h, best_c5, best_N
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_N = N

    # Phase 1: Projected gradient on parent resolution
    if elapsed() < budget_s - 200:
        h_pg, c5_pg = proj_gradient_descent(h0, N0, elapsed, min(budget_s - 300, 300))
        update_best(h_pg, c5_pg, N0)

    # Phase 2: SLSQP refinement
    if elapsed() < budget_s - 150:
        h_sl, c5_sl = slsqp_refine(best_h, best_N, elapsed, min(budget_s - 250, 120))
        update_best(h_sl, c5_sl, best_N)

    # Phase 3: Try other resolutions
    for N_try in [N0 + 1, N0 + 2, N0 + 4, N0 + 8, 100, 120, 160, 200]:
        if elapsed() > budget_s - 120:
            break
        if N_try == best_N:
            continue
        h_interp = np.interp(np.linspace(0.0, 1.0, N_try),
                             np.linspace(0.0, 1.0, best_N), best_h)
        h_interp = project_h(h_interp, N_try)
        h_opt, c5_opt = proj_gradient_descent(h_interp, N_try, elapsed,
                                               min(budget_s - 200, 150))
        update_best(h_opt, c5_opt, N_try)

    # Phase 4: Binary pushing on best
    if elapsed() < budget_s - 80:
        h_bin, c5_bin = binary_push(best_h, best_N, elapsed, min(budget_s - 150, 120))
        update_best(h_bin, c5_bin, best_N)

    # Phase 5: Random restarts
    if elapsed() < budget_s - 40:
        h_rr, c5_rr = random_restart(best_h, best_N, 5, elapsed, budget_s - 30)
        update_best(h_rr, c5_rr, best_N)

    # Phase 6: Final SLSQP polish
    if elapsed() < budget_s - 20:
        h_fin, c5_fin = slsqp_refine(best_h, best_N, elapsed, budget_s - 10)
        update_best(h_fin, c5_fin, best_N)

    # Final projected gradient sweep
    if elapsed() < budget_s - 10:
        h_fin2, c5_fin2 = proj_gradient_descent(best_h, best_N, elapsed, budget_s - 5)
        update_best(h_fin2, c5_fin2, best_N)

    # Validate
    best_h = project_h(best_h, best_N)
    best_c5 = get_c5(best_h, best_N)

    # Try evaluate_erdos_solution if available
    if 'evaluate_erdos_solution' in globals():
        try:
            val = evaluate_erdos_solution(best_h)
            if val is not None and float(val) < best_c5:
                best_c5 = float(val)
        except Exception:
            pass

    return best_h, best_c5, best_N
