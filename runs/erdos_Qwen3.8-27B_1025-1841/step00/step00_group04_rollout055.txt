    return np.clip(h + (lo + hi) / 2, 0, 1)

def pgd_optimize(h_init, n, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    h = project_simplex(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    start = time.time()
    dx = 2.0 / n
    lr = 2.0
    no_improve = 0
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        c = corr * dx
        max_idx = np.argmax(c)
        s = max_idx - (n - 1)
        
        grad = np.zeros(n)
        if s == 0:
            grad[:] = (1 - 2 * h) * dx
        elif s > 0:
            grad[s:] += (1 - h[:n-s]) * dx
            grad[:n-s] -= h[s:] * dx
        else:
            t = -s
            grad[:n-t] += (1 - h[t:]) * dx
            grad[t:] -= h[:n-t] * dx
        
        h_new = h - lr * grad
        h_new = project_simplex(h_new, n)
        c5_new = eval_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            no_improve = 0
        else:
            h = h_new  # Still move (momentum-like effect)
            no_improve += 1
            if no_improve > 200:
                # Random perturbation to escape local opt
                h = h + rng.randn(n) * 0.02
                h = project_simplex(h, n)
                no_improve = 0
            if no_improve > 100:
                lr *= 0.9
        
        lr = max(lr * 0.99995, 0.01)
    
    return best_h, best_c5

def upscale(h, n_old, n_new):
    """Upscale h from n_old to n_new points using linear interpolation"""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_simplex(h_new, n_new)

def random_binary(n, seed=42):

    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h
