        h_new = project_to_feasible(h_new, n)

def project_to_feasible(x, n):

    lo, hi = -1.0, 2.0  # range for λ
    for _ in range(60):

        clipped = np.clip(x - mid, 0, 1)
        s = clipped.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)
