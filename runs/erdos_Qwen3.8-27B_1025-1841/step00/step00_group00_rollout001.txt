import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def compute_c5_lags(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx, np.argmax(c)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    if new_n % old_n == 0:
        h_up = np.repeat(h, new_n // old_n).astype(float)
    else:
        x_old = np.linspace(0, 2, old_n)
        x_new = np.linspace(0, 2, new_n)
        h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0, 1)
    target = new_n / 2.0
    diff = target - np.sum(h_up)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_up < 1.0
        else:
            mask = h_up > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_up[mask] += diff / cnt
    return np.clip(h_up, 0, 1)

def fix_sum(h):
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0, 1)

def annealing_pass(h, t_deadline, rng, T_init, best_ref):
    n = len(h)
    current_c5 = compute_c5(h)
    T = T_init
    T_min = 0.00002
    
    while time.time() < t_deadline:
        r = rng.random()
        if r < 0.65:
            # Targeted move: reduce bottleneck lags
            c_lags, max_idx = compute_c5_lags(h)
            lag = max_idx - (n - 1)
            # Get top 3 lags
            top_idxs = np.argsort(c_lags)[-3:][::-1]

            if tl > 0:
                vi = np.arange(0, n - tl)
            elif tl < 0:
                vi = np.arange(-tl, n)
            else:
                vi = np.arange(n)
            if len(vi) == 0:
                vi = np.arange(n)
                tl = 1 if n > 1 else 0
            if tl >= 0:
                prods = h[vi] * (1.0 - h[vi + tl]) if vi[-1] + tl < n else h[vi[:n-tl]] * (1.0 - h[vi[:n-tl] + tl])
            else:
                vi2 = np.arange(-tl, n)
                prods = h[vi2] * (1.0 - h[vi2 + tl])
            if len(prods) > 0:
                top_k = min(5, len(prods))
                top_pos = np.argpartition(prods, -top_k)[-top_k:]

    improved = True
