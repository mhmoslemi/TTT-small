    h = h.copy()
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Compute subgradient
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = np.argmax(corr)  # Shift that achieves max
        # Gradient of f_{k_max} wrt h
        grad = np.zeros(n)
        # f_k(h) = sum_{i} h[i] * (1-h[i+k]) for valid i
        # shift k corresponds to corr index k_max
        # In np.correlate, corr[k] = sum_i a[i] * b[i+k-n+1]... 
        # Actually, for mode="full", the output has length 2n-1
        # corr[k] = sum_i a[i] * b[i + k - (n-1)]
        # So corr[k_max] = sum_i h[i] * (1-h[i + k_max - (n-1)])
        # Let shift = k_max - (n-1)
        shift = k_max - (n - 1)
        for i in range(n):
            # ∂/∂h[i] of h[i]*(1-h[i+shift]) = (1-h[i+shift]) if i+shift in range
            # ∂/∂h[i] of h[i-shift]*(1-h[i]) = -h[i-shift] if i-shift in range
            term1 = 0.0
            term2 = 0.0

            j2 = i - shift
            if 0 <= j2 < n:

        grad -= np.mean(grad)
