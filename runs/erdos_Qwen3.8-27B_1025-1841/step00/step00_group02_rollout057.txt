import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_obj(h, beta=300.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    shifted = corr - m
    return float(m + np.log(np.mean(np.exp(beta * shifted))) / beta)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        scale = target / s if s > 1e-12 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, target, max_iters=500, deadline=None):
    n = len(h)
    best_c5 = compute_c5(h)
    h = h.copy()
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            old_val = h[i]
            best_delta = 0.0
            best_c5_local = best_c5
            for s in step_sizes:
                for sign in (1, -1):
                    nv = old_val + sign * s
                    if nv < -0.01 or nv > 1.01:
                        continue
                    h[i] = np.clip(nv, 0.0, 1.0)
                    h = fix_sum(h, target)
                    c5_try = compute_c5(h)
                    if c5_try < best_c5_local - 1e-13:
                        best_c5_local = c5_try
                        best_delta = sign * s
            if best_delta != 0.0:
                h[i] = np.clip(old_val + best_delta, 0.0, 1.0)
                h = fix_sum(h, target)
                best_c5 = best_c5_local
