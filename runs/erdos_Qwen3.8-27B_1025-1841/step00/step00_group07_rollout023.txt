            if attempt == 0:
                h_start = h1opt
            else:
                h_start = h1opt + 0.1 * np.random.randn(n1)
                h_start = np.clip(h_start, 0, 1)
                h_start = h_start - np.mean(h_start) + n1 / 2.0 / n1
                h_start = np.clip(h_start, 0, 1)
            h_s = slsqp_optimize(h_start, p=50, max_iter=200)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_b = make_binary_from_continuous(h_s, n1)
            c5_b = compute_c5(h_b)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 3: Higher resolutions
    for n2 in [160, 320]:
        if remaining() < 60:
            break
        h2 = upsample(best_h, n2)
        h2b = make_binary_from_continuous(h2, n2)
        t_alloc = min(remaining() * 0.25, 200)
        h2opt, c5_2 = binary_local_search(h2b, n_random=200, max_time=t_alloc, start_time=start_time)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2opt.copy()
        if remaining() > 50 and n2 <= 160:
            h_s2 = slsqp_optimize(h2opt, p=50, max_iter=150)
            c5_s2 = compute_c5(h_s2)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2.copy()

    # Phase 4: Iterated local search with perturbations
    while remaining() > 20:
        n = len(best_h)
        if n <= 160:
            h_curr = best_h.copy()
            # Perturb: flip k pairs
            k = np.random.randint(2, max(3, n // 20))
            ones = np.where(h_curr == 1.0)[0] if np.max(h_curr) > 0.5 else np.where(h_curr > 0.5)[0]
            zeros = np.where(h_curr == 0.0)[0] if np.min(h_curr) < 0.5 else np.where(h_curr <= 0.5)[0]
            if len(ones) >= k and len(zeros) >= k:
                flip_ones = np.random.choice(ones, k, replace=False)
                flip_zeros = np.random.choice(zeros, k, replace=False)
                h_curr[flip_ones] = 0.0
                h_curr[flip_zeros] = 1.0
            t_alloc = min(remaining() * 0.15, 60)
            h_pert, c5_pert = binary_local_search(h_curr, n_random=150, max_time=t_alloc, start_time=start_time)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
        else:
            h_rand = random_binary(n)
            t_alloc = min(remaining() * 0.15, 40)
            h_opt, c5_opt = binary_local_search(h_rand, n_random=100, max_time=t_alloc, start_time=start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
