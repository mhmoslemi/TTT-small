c[n-1+m] = Σ_i a[i+m]*v[i] for m=0,...,n-1 (positive lag)
c[n-1-m] = Σ_i a[i]*v[i+m] for m=0,...,n-1 (negative lag)

c[n-1+m] corresponds to: Σ_i a[i+m]*v[i] = Σ_i h[i+m]*(1-h[i]) for m ≥ 0
c[n-1-m] corresponds to: Σ_i h[i]*(1-h[i+m]) for m ≥ 0

∫ h(x)(1-h(x+k)) dx ≈ dx * Σ_i h[i] * (1-h[i+k/dx]) for valid i

This corresponds to c[n-1-m] where m = k/dx (positive shift) or c[n-1+m] (negative shift).

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = float(n) / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    if s < target:
        deficit = target - s
        order = np.argsort(h)
        for idx in order:
            if deficit <= 1e-12:
                break
            add = min(1.0 - h[idx], deficit)
            h[idx] += add
            deficit -= add
    else:
        excess = s - target
        order = np.argsort(h)[::-1]
        for idx in order:
            if excess <= 1e-12:
                break
            sub = min(h[idx], excess)
            h[idx] -= sub
            excess -= sub

def compute_c5(h):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    if s < target:
        d = target - s
        for idx in np.argsort(h):
            if d <= 1e-12:
                break
            a = min(1.0 - h[idx], d)
            h[idx] += a
            d -= a
    else:
        e = s - target
        for idx in np.argsort(h)[::-1]:
            if e <= 1e-12:
                break
            a = min(h[idx], e)
            h[idx] -= a
            e -= a
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    h = np.round(h).astype(float)
    h = np.clip(h, 0, 1)
    t = n // 2
    while h.sum() > t:
        h[rng.choice(np.where(h == 1)[0])] = 0
    while h.sum() < t:
        h[rng.choice(np.where(h == 0)[0])] = 1
    return h

def local_search(h, n, tlimit, rng):
    h = to_binary(h, n, rng)
    best = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stall = 0
    while time.time() - start < tlimit and stall < 30:
        improved = False
        ones = np.where(h == 1)[0].copy()
        rng.shuffle(ones)
        for i in ones:
            if time.time() - start > tlimit:
                return best_h, best
            if h[i] != 1:
                continue
            h[i] = 0
            zeros = np.where(h == 0)[0].copy()
            rng.shuffle(zeros)
            bj = -1
            bv = best
            for j in zeros:
                if j == i:
                    continue
                h[j] = 1
                v = compute_c5(h)
                h[j] = 0
                if v < bv - 1e-12:
                    bv = v
                    bj = j
            if bj >= 0:
                h[bj] = 1
                best = bv
                best_h = h.copy()
                improved = True
            else:
                h[i] = 1
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best

def sa_search(h, n, tlimit, rng):
    h = to_binary(h, n, rng)
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    start = time.time()
    T = 0.015
    while time.time() - start < tlimit and T > 1e-9:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        pos_i = int(ones[rng.integers(len(ones))])
        pos_j = int(zeros[rng.integers(len(zeros))])
        h[pos_i] = 0.0
        h[pos_j] = 1.0
        nv = compute_c5(h)
        if nv < cur or rng.random() < np.exp(-(nv - cur) / T):
            cur = nv
            if nv < best:
                best = nv
                best_h = h.copy()
        else:
            h[pos_i] = 1.0
            h[pos_j] = 0.0
        T *= 0.9997
    return best_h, best

def upscale(h, n_old, n_new, rng):
    if n_new == n_old:
        return h.copy()
    ratio = n_new // n_old
    pad = n_new - ratio * n_old
    h_new = np.repeat(h, ratio)
    if pad > 0:
        h_new = np.concatenate([h_new, h[:pad]])
    h_new = h_new[:n_new]
    h_new = h_new + rng.uniform(-0.15, 0.15, n_new)
    return to_binary(project_feasible(h_new, n_new), n_new, rng)

def continuous_refine(h, n, tlimit, rng):
    h = project_feasible(h.astype(float).copy(), n)
    best = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.05
    while time.time() - start < tlimit:
        conv = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        k_s = int(np.argmax(conv))
        lag = k_s - (n - 1)
        grad = np.ones(n)
        for j in range(n):
            g = 1.0
            if 0 <= j + lag < n:
                g -= h[j + lag]
            if 0 <= j - lag < n:
                g -= h[j - lag]
            grad[j] = g * (2.0 / n)
        h = project_feasible(h - lr * grad, n)
        v = compute_c5(h)
        if v < best:
            best = v
            best_h = h.copy()
        lr *= 0.995
    return best_h, best

        if c5 < best_c5:

        if c5 < best_c5:
            best_c5, best_h = c5, h

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
