    h = x[:n]
    t = x[n]

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_f(h, n):

def smooth_max_val(f, alpha):
    m = np.max(f)
    return m + alpha * np.log(np.sum(np.exp((f - m) / alpha)))

def project_onto_simplex(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    threshold = (lo + hi) / 2.0
    result = np.clip(v - threshold, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.any():
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def grad_smooth_max(f, h, n, alpha):
    m = np.max(f)
    w = np.exp((f - m) / alpha)
    w /= np.sum(w)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = w[k_idx] * dx
        if wk < 1e-10:
            continue
        if k >= 0:
            if n - k > 0:
                # j ranges 0 to n-1-k: contributes (1 - h[j+k])
                j_range = np.arange(0, n - k)
                grad[j_range] += wk * (1.0 - h[j_range + k])
                # j ranges k to n-1: contributes -h[j-k]
                if k < n:
                    j_range2 = np.arange(k, n)
                    grad[j_range2] -= wk * h[j_range2 - k]
        else:

                j_range = np.arange(kp, n)
                grad[j_range] += wk * (1.0 - h[j_range - kp])
                # j ranges 0 to n-1-kp: contributes -h[j+kp]
                if n - kp > 0:
                    j_range2 = np.arange(0, n - kp)
                    grad[j_range2] -= wk * h[j_range2 + kp]
    return grad
