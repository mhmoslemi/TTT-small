                break
        return hh, c5

    def paper_grad(h0, nn, deadline):
        """Power method-like: iteratively minimize the active correlation."""
        tgt = nn / 2.0
        hh = h0.copy()
        c5 = compute_c5(hh)
        best_local_c5 = c5
        lr = 0.01
        step = 0
        while time.time() < deadline:
            corr = np.correlate(hh, 1.0 - hh, mode='full')
            k_active = int(np.argmax(corr))
            lag = k_active - (nn - 1)
            gr = np.zeros(nn)
            if lag >= 0:
                for m in range(nn - lag):
                    gr[m + lag] += 1.0 - hh[m]
                    gr[m] -= hh[m + lag]
            else:
                l2 = -lag
                for m in range(l2, nn):
                    gr[m] += 1.0 - hh[m - l2]
                    gr[m - l2] -= hh[m]
            hn = project(hh - lr * gr, tgt)
            c5n = compute_c5(hn)
            if c5n < c5 - 1e-14:
                hh = hn
                c5 = c5n
                if c5 < best_local_c5:
                    best_local_c5 = c5
                lr = min(lr * 1.02, 0.2)
            else:
                lr *= 0.7
            step += 1
            if step % 300 == 0 and lr < 1e-5:
                hp = project(hh + 0.03 * np.random.randn(nn), tgt)
                cp = compute_c5(hp)
                if cp < c5:
                    hh = hp
                    c5 = cp
                    lr = 0.01
                else:
                    break
            if step > 15000:
                break
        return hh, c5

    # Phase 1: SLSQP at initial resolution
    h1, c5_1 = slsqp_opt(best_h, n0, 400)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Projected gradient refinement at n0
    deadline = min(time.time() + 60, t_start + budget_s * 0.2)
    h2, c5_2 = proj_grad(best_h, n0, deadline)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: Try a few structured starting points at n0
    structured_starts = []
    # Alternating
    alt = np.zeros(n0)
    alt[::2] = 1.0
    alt[1::2] = max(0.0, 1.0 - 2.0 / (n0 // 2 + 1) if n0 % 2 == 0 else 1.0)
    alt = repair(alt)
    structured_starts.append(alt)
    # Sine
    xg = np.linspace(0, 2, n0)
    s1 = repair(0.5 + 0.4 * np.sin(np.pi * xg))
    structured_starts.append(s1)
    s2 = repair(0.5 + 0.3 * np.sin(2 * np.pi * xg))
    structured_starts.append(s2)

    for ss in structured_starts:
        if time.time() - t_start > budget_s * 0.35:
            break
        hs, cs = slsqp_opt(ss, n0, 200)
        if cs < best_c5:
            best_h, best_c5 = hs, cs
        dl = min(time.time() + 20, t_start + budget_s * 0.35)
        hg, cg = proj_grad(best_h, n0, dl)
        if cg < best_c5:
            best_h, best_c5 = hg, cg

    # Phase 4: Higher resolutions
    n_candidates = sorted(set([
        n0 + 20, n0 + 40, n0 + 70, n0 + 100, n0 + 150, n0 + 200, n0 + 300
    ]))
    n_candidates = [n for n in n_candidates if n <= 600]

    for nn in n_candidates:
        if time.time() - t_start > budget_s * 0.82:
            break
        deadline_sl = min(time.time() + 80, t_start + budget_s * 0.82)
        if deadline_sl - time.time() < 2:
            continue
        hi = repair(interp_to(best_h, nn))
        # Also try interpolation from best at n0
        hi = repair(interp_to(best_h, nn))
        hr, cr = slsqp_opt(hi, nn, 150)
        if cr < best_c5:
            best_h, best_c5 = hr, cr
        # Refine
        dl = min(time.time() + 40, t_start + budget_s * 0.82)
        if dl - time.time() > 1:
            hg, cg = paper_grad(best_h, nn, dl)
            if cg < best_c5:
                best_h, best_c5 = hg, cg

    # Phase 5: Final refinement at best resolution
    nf = len(best_h)
    if time.time() - t_start < budget_s - 5:
        dl = min(time.time() + 120, t_start + budget_s - 5)
        if dl - time.time() > 3:
            hf, cf = slsqp_opt(best_h, nf, 300)
            if cf < best_c5:
                best_h, best_c5 = hf, cf
        dl2 = t_start + budget_s - 3
        if time.time() < dl2:
            hg, cg = proj_grad(best_h, nf, dl2)
            if cg < best_c5:
                best_h, best_c5 = hg, cg

    best_h = repair(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
