    dx = 2.0 / n
    h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1.0-h, mode="full")
        k_star = np.argmax(c)
        m = k_star - (n - 1)
        
        # Gradient of c[k_star] w.r.t. h
        grad = np.zeros(n)
        if m >= 0:
            for i in range(n - m):
                grad[i] += (1.0 - h[i+m])
                grad[i+m] -= h[i]
        else:
            mp = -m
            for i in range(mp, n):
                grad[i] += (1.0 - h[i-mp])
                grad[i-mp] -= h[i]
        
        # Projected gradient step
        h_new = h - lr * grad * dx
        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        s = np.sum(h_new)
        h_new += (n/2 - s) / n
        h_new = np.clip(h_new, 0, 1)
        
        val_new = compute_c5(h_new)
        if val_new < best_val:
            h = h_new
            best_val = val_new
            # Slow down learning rate slightly
            lr = max(lr * 0.999, 0.001)
        else:
            lr = max(lr * 0.95, 0.0001)
    
    return h, best_val

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = 0.5 * n
    diff = target - s
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if diff > 1e-14:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff <= 1e-14:
                    break
        elif diff < -1e-14:
            order = np.argsort(-h)
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if diff >= -1e-14:
                    break
    return h

def grad_at_lag(h, m):
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:n] -= h[:n-m]
    else:
        mp = -m
        grad[mp:] += (1.0 - h[:n-mp])
        grad[:n-mp] -= h[mp:]
    return grad

def local_search(h, max_iter, time_limit, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    best_val = float(np.max(c)) * dx
    t0 = time.time()
    T = 0.008
    accept_count = 0

    for it in range(max_iter):
        if time.time() > deadline or time.time() - t0 > time_limit:
            break

        if rng.random() < 0.75:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            delta = float(rng.uniform(-0.06, 0.06))
        else:
            i = int(rng.integers(0, n - 1))
            j = i + 1
            delta = float(rng.uniform(-0.04, 0.04))

        hi_new = h[i] + delta
        hj_new = h[j] - delta
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue

        h[i] = hi_new
        h[j] = hj_new
        c_new = np.correlate(h, 1.0 - h, mode="full")
        val_new = float(np.max(c_new)) * dx

        if val_new < best_val:
            best_val = val_new
            c = c_new
            accept_count += 1
            T = max(T * 0.997, 0.0003)
        else:
            gap = val_new - best_val
            if rng.random() < np.exp(-gap / T):
                c = c_new
            else:
                h[i] = hi_new - delta
                h[j] = hj_new + delta
            T = max(T * 0.9993, 0.0003)

        if accept_count % 2000 == 0:
            T = max(T, 0.003)

    return h, best_val

def gradient_refine(h, n_iter, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_val = compute_c5(h)
    lr = 0.05
    stuck = 0

    for it in range(n_iter):
        if time.time() > deadline:
            break

        c = np.correlate(h, 1.0 - h, mode="full")
        top3 = np.argsort(c)[-3:]
        top3 = top3[::-1]

        grad = np.zeros(n)
        for k in top3:
            m = k - (n - 1)
            grad += grad_at_lag(h, m)
        grad /= len(top3)

        h_new = h - lr * grad * dx
        h_new = np.clip(h_new, 0.0, 1.0)
        s = np.sum(h_new)
        h_new += (0.5 * n - s) / n
        h_new = np.clip(h_new, 0.0, 1.0)

        val_new = compute_c5(h_new)
        if val_new < best_val:
            h = h_new
            best_val = val_new
            stuck = 0
            lr = min(lr * 1.01, 0.2)
        else:
            stuck += 1
            lr = max(lr * 0.95, 0.005)
            if stuck > 200:
                lr = 0.03
                stuck = 0

    return h, best_val

def upscale(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def dl(frac):
        return t0 + budget_s * frac

    try:

    # Phase 2: Gradient refinement
    h_cur, c = gradient_refine(h_best, 30000, dl(0.25))

        h_best = h_cur

    # Phase 3: Random restarts on same grid
    for trial in range(10):

        h_rand = np.where(rng.random(n_best) < 0.5, 1.0, 0.0)
        blend = 0.5 + 0.05 * trial

        h_rand = fix_sum(h_rand, n_best)
        h_cur, c = local_search(h_rand, 300000, 45, rng, dl(0.45))

    # Phase 4: Multi-scale refinement
    scales = sorted(set([n_best, n_best * 2, n_best * 4]))
    scales = [s for s in scales if s <= 1400 and s > 0]

    for target_n in scales:
        if time.time() > dl(0.82):

        else:

        alloc = min(70, max(10, dl(0.85) - time.time()))
