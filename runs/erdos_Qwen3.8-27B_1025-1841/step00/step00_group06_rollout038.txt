- Finding all "breakpoints" where the active set changes
- These occur at intersections of pairs of lines

The minimum of a convex piecewise-linear function on [0,1] occurs at a point where:
- All lines with positive slope are ≥ the value at that point
- All lines with negative slope are ≤ the value at that point

```python
def minimize_max_lines(a, b, t_min=0, t_max=1):
    """Minimize max_j (a[j]*t + b[j]) over t in [t_min, t_max].
    Returns the optimal t."""
    n_lines = len(a)
    # Find all intersection points in [t_min, t_max]
    # Also include t_min and t_max
    candidates = [t_min, t_max]
    for i in range(n_lines):
        for j in range(i+1, n_lines):
            if a[i] == a[j]:
                continue
            t = (b[j] - b[i]) / (a[i] - a[j])
            if t_min <= t <= t_max:
                candidates.append(t)
    # Evaluate at all candidates
    best_t = t_min
    best_val = np.inf
    for t in candidates:
        val = np.max(a * t + b)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t
```

```python
        for i in order:
            room = h[i]
            sub = min(room, remaining / (n - np.where(order==i)[0][0]))
            h[i] -= sub
            remaining -= sub
            if remaining <= 0:
                break
    # Final correction (small)
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute uniformly among non-bound variables
        free = (h > 0.01) & (h < 0.99)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h
```

Let me go back to the analytical approach for the 1D subproblem.

For a fixed m, and for each j:
- k = j - (n-1)
- The change in B_j when h[m] goes from old to new value t:

If k ≠ 0:
- B_j changes by: (t - h[m]) * (1-h[m-k]) * I[m-k in valid range for j] + (t - h[m]) * (-h[m+k]) * I[m in valid range for j]
  = (t - h[m]) * [(1-h[m-k])*I1 - h[m+k])*I2]
  = (t - h[m]) * a_j

where I1 = 1 if max(0,-k) ≤ m-k ≤ min(n-1, n-1-k), I2 = 1 if max(0,-k) ≤ m ≤ min(n-1, n-1-k)

If k = 0:
- B_j changes by: t*(1-t) - h[m]*(1-h[m]) = (t-h[m]) - (t²-h[m]²) = (t-h[m])(1-t-h[m])

So for k=0, the change is quadratic in t.

Now, the objective is max_j B_j(h) where B_j(h) = B_j_base + change_j(t).

For k=0: B_0(t) = B_0_base + (t-h[m])(1-t-h[m]) which is concave in t.
For k≠0: B_j(t) = B_j_base + (t-h[m])*a_j which is linear in t.

So the objective is max of (one concave quadratic + several linear functions). The minimum of this over t ∈ [0,1] can be found by:
1. Evaluating at t=0 and t=1
2. Finding critical points where the derivative is 0 or where the active set changes

For the linear functions, the critical points are at intersections. For the quadratic, the minimum of the quadratic alone is at t = (1-h[m])/2 (if in [0,1]).

A practical approach: evaluate the objective at a set of candidate points:
- t = 0, 1
- t = (1-h[m])/2 (minimizer of the quadratic term)
- Intersections of pairs of linear functions (that are in [0,1])
- Intersections of the quadratic with each linear function (that are in [0,1])

The number of candidates is O(n²) in the worst case, but in practice much less.

```python
def optimize_coordinate(h, m):
    """Find optimal h[m] given all other coordinates fixed."""
    n = len(h)
    one_mh = 1.0 - h
    B = np.correlate(h, one_mh, mode='full')  # Current B_j values
    
    # Compute a_j = coefficient of (t - h[m]) in B_j(t) - B_j for k ≠ 0
    a = np.zeros(2*n-1)
    for j in range(2*n-1):
        k = j - (n-1)
        if k == 0:
            continue  # Handle separately
        n_lo = max(0, -k)
        n_hi = min(n-1, n-1-k)
        # I1: m-k in [n_lo, n_hi]
        I1 = 1.0 if (n_lo <= m-k <= n_hi) else 0.0
        # I2: m in [n_lo, n_hi]
        I2 = 1.0 if (n_lo <= m <= n_hi) else 0.0
        a[j] = (1.0 - h[m-k]) * I1 - h[m+k] * I2  # Note: m-k and m+k must be valid indices
    
    # For j = n-1 (k=0): B_{n-1}(t) = B_{n-1} + (t-h[m])*(1-t-h[m])
    # = B_{n-1} + (t-h[m]) - (t^2 - h[m]^2)
    
    # Now find t in [0,1] that minimizes max_j B_j(t)
    # B_j(t) = B[j] + (t-h[m])*a[j] for j ≠ n-1
    # B_{n-1}(t) = B[n-1] + (t-h[m])*(1-t-h[m])
    
    old = h[m]
    # Evaluate at grid points
    ts = np.linspace(0, 1, 100)
    best_t = old
    best_obj = np.inf
    
    for t in ts:
        # Compute all B_j(t)
        Bt = B + (t - old) * a
        # Fix the k=0 term
        Bt[n-1] = B[n-1] + (t - old) * (1 - t - old)
        obj = np.max(Bt)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t
```

def smooth_obj_grad(h, alpha=80.0):
    n = len(h)
    dx = 2.0 / n
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = np.clip(alpha * (vals - m), -500, 0)
    exp_v = np.exp(shifted)
    w_sum = np.sum(exp_v)
    obj = m + np.log(w_sum) / alpha
    weights = exp_v / w_sum
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        k = j - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        if n_lo > n_hi:
            continue
        w = weights[j] * dx
        if w < 1e-12:
            continue
        idx = np.arange(n_lo, n_hi + 1)
        i1 = idx + k
        grad[i1] += w * (1.0 - h[idx])
        grad[idx] -= w * h[i1]
    
    return obj, grad

    n = len(h)

    for sweep in range(max_sweeps):
        if start_time is not None and time.time() - start_time > time_limit:
            break

            old = h[m]

            # For each j: B_j(t) = B[j] + (t-old)*a[j] (k≠0) or B[j] + (t-old)*(1-t-old) (k=0)
            a = np.zeros(2*n-1)
            for j in range(2*n-1):
                k = j - (n-1)
                if k == 0:
                    continue
                n_lo = max(0, -k)

                I1 = 1.0 if (0 <= m-k < n and n_lo <= m-k <= n_hi) else 0.0

                if I1:

                    a[j] -= h[m+k]
