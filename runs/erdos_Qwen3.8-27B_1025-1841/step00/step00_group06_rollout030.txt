def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n

    return np.max(corr) * dx

def compute_all_lags(h):
    """Compute values at all lags."""
    n = len(h)
    dx = 2.0 / n

def repair_h(h, n):

    target = n / 2.0

    # Scale to fix sum
    if current > 0:
        h = h * (target / current)
        h = np.clip(h, 0, 1)
    # If still not matching, adjust individual elements
    diff = target - np.sum(h)
    if diff > 0:
        # Need to increase some values
        mask = h < 1.0
        if np.any(mask):

            h[mask] = np.minimum(h[mask] + increments, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Using bisection

        if val > target:

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    dx = 2.0 / n

    if active_lags is None:

            for j in range(n):

Let me estimate: np.correlate for n=95 vectors should take about 1-2 microseconds (it's implemented in C in numpy). So we can do about 500,000-1,000,000 evaluations per second. That's plenty.

The bottleneck will be the subgradient computation (O(n * M) in Python) and the projection (O(n * 50) in Python due to bisection).

For n=95, M=5: subgradient is 95*5 = 475 operations.
Projection: 95 * 50 = 4750 operations.

So per iteration: ~5000 operations ≈ 5-10 microseconds. We can do 100,000+ iterations per second.

**Final algorithm design:**

1. **Phase 1** (60% of budget): Subgradient optimization with adaptive step size
   - Use top M=3 to 10 active lags
   - Adaptive alpha: increase on success, decrease on failure
   - Track best solution

2. **Phase 2** (20% of budget): Local search
   - Focus on worst lag
   - Try targeted modifications (flip, swap, nudge)
   - Accept only improvements

3. **Phase 3** (20% of budget): Multi-resolution
   - Try n=200, n=400
   - Interpolate from best, then optimize

4. **Return** best solution found

velocity = beta * velocity + (1-beta) * gradient
h_new = h + alpha * velocity

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def subgrad(h, n, active_idx):
    dx = 2.0 / n
    g = np.zeros(n)
    c = n - 1
    for idx in active_idx:
        k = idx - c
        if k == 0:
            g += 1.0 - 2.0 * h
        elif k > 0:
            g[k:] += 1.0 - h[:-k]
            g[:n-k] -= h[k:]
        else:
            m = -k
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:-m]
    return g * dx / len(active_idx)

            else:

                # Just pick random positions for simplicity
