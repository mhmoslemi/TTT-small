import numpy as np
from scipy.ndimage import zoom
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(h):
    n = len(h)
    k = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[order[:k]] = 1.0
    return h_bin

def local_search_binary(h, max_iter, rng, temp=0.0, deadline=None):
    n = len(h)
    dx = 2.0 / n
    h_work = np.array(h, dtype=np.float64)
    current = float(np.max(np.correlate(h_work, 1.0 - h_work, mode="full"))) * dx
    best_h = h_work.copy()
    best_val = current
    for it in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break
        ones = np.where(h_work == 1.0)[0]
        zeros = np.where(h_work == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h_work[i] = 0.0
        h_work[j] = 1.0
        new_val = float(np.max(np.correlate(h_work, 1.0 - h_work, mode="full"))) * dx
        delta = new_val - current
        if delta <= 1e-14 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            current = new_val
            if new_val < best_val - 1e-14:
                best_val = new_val
                best_h = h_work.copy()
        else:
            h_work[i] = 1.0
            h_work[j] = 0.0
    return best_h, best_val

def simulated_anneal_binary(h, max_iter, rng, deadline, t_start=0.5, t_end=0.01):
    n = len(h)
    dx = 2.0 / n
    h_work = np.array(h, dtype=np.float64)
    current = float(np.max(np.correlate(h_work, 1.0 - h_work, mode="full"))) * dx
    best_h = h_work.copy()
    best_val = current
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        frac = it / max(max_iter, 1)
        temp = t_start * (t_end / t_start) ** frac
        ones = np.where(h_work == 1.0)[0]
        zeros = np.where(h_work == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h_work[i] = 0.0
        h_work[j] = 1.0
        new_val = float(np.max(np.correlate(h_work, 1.0 - h_work, mode="full"))) * dx
        delta = new_val - current
        if delta <= 1e-14 or (rng.random() < np.exp(-delta / max(temp, 1e-10))):
            current = new_val
            if new_val < best_val - 1e-14:
                best_val = new_val
                best_h = h_work.copy()
        else:
            h_work[i] = 1.0
            h_work[j] = 0.0
    return best_h, best_val

def upsample(h, n_new):
    ratio = n_new / len(h)
    h_up = zoom(h, ratio, order=1)
    h_up = np.clip(h_up, 0.0, 1.0)
    return h_up

def multi_starts_binary(n, rng, n_starts, deadline):
    best_h = None
    best_val = 1e9
    for s in range(n_starts):
        if time.time() >= deadline:
            break
        phase = rng.uniform(0, 2 * np.pi)
        h_rand = np.zeros(n)
        h_rand[:] = 0.5 + 0.5 * np.sin(2 * np.pi * n / (2 * n) + phase) * rng.uniform(0.5, 1.5)
        h_rand = np.clip(h_rand, 0, 1)
        h_bin = make_binary(h_rand)
        h_res, v_res = local_search_binary(h_bin, 5000, rng, temp=0.0, deadline=deadline)
        if v_res < best_val:
            best_val = v_res
            best_h = h_res
    return best_h, best_val
