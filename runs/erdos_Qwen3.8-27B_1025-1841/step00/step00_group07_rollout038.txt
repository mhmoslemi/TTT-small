import time

    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def make_binary(h, n):
    """Project h to binary with exactly n//2 ones."""
    h = np.array(h)
    # Round
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Need to remove some ones
        # Remove the ones with lowest original values
        ones_idx = np.where(h_bin == 1)[0]
        ones_vals = h[ones_idx]
        order = np.argsort(ones_vals)
        for i in range(n_ones - target):
            h_bin[ones_idx[order[i]]] = 0
    elif n_ones < target:
        # Need to add some ones
        zeros_idx = np.where(h_bin == 0)[0]
        zeros_vals = h[zeros_idx]
        order = np.argsort(zeros_vals)[::-1]
        for i in range(target - n_ones):
            h_bin[zeros_idx[order[i]]] = 1
    return h_bin

    h = h.copy()
    c5 = compute_c5(h)
    
    T = T_start
    cooling_rate = (T_end / T_start) ** (1.0 / n_iters)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for iteration in range(n_iters):
        if time.time() > time_limit:
            break
        
        # Pick random 1 and random 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        
        # Update tracking
        ones[ones == i] = j
        zeros[zeros == j] = i
        
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
            ones[ones == j] = i
            zeros[zeros == i] = j

import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_binary(h):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx])[:n_ones - target]
        h_bin[idx[order]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx])[:target - n_ones]
        h_bin[idx[order]] = 1
    return h_bin

def sa_optimize(h, end_time, rng, T_start=0.05, T_end=1e-4):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    best_h = h.copy()
    best_c5 = c5
    est_time = max(1.0, end_time - time.time())
    iters_per_sec = 25000 * (80.0 / n) ** 2
    est_iters = max(1000, int(est_time * iters_per_sec))
    alpha = (T_end / T_start) ** (1.0 / est_iters)
    T = T_start
    while time.time() < end_time:
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        h[i] = 0.0
        h[j] = 1.0
        ones[ii] = j
        zeros[jj] = i
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta < 0 or (T > T_end * 1.01 and rng.rand() < np.exp(-delta / T)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones[ii] = i
            zeros[jj] = j
        T = max(T_end, T * alpha)
    return best_h, best_c5

def local_search(h, deadline, rng, max_rounds=200):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        best_delta = 1e-9
        best_i = -1
        best_j = -1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                delta = new_c5 - c5
                if delta < -best_delta:
                    best_delta = -delta
                    best_i = i
                    best_j = j
                h[i] = 1.0
                h[j] = 0.0
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5 -= best_delta
        else:
            break
    return h, c5

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h
