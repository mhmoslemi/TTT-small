import numpy as np

def compute_c5_and_lag(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    idx = np.argmax(corr)
    return corr[idx] * dx, idx

def fix_sum(h, n, rng):
    """Fix h to have sum = n/2, 0<=h<=1"""
    target = n / 2.0
    diff = np.sum(h) - target
    if abs(diff) < 1e-10:
        return h
    # Distribute the correction
    if diff > 0:
        # Need to decrease
        candidates = np.where(h > 0)[0]
    else:
        # Need to increase
        candidates = np.where(h < 1)[0]
    if len(candidates) == 0:
        return np.clip(h, 0, 1)
    # Spread correction
    per = diff / len(candidates)
    h[candidates] -= per
    h = np.clip(h, 0, 1)
    # If still off, do it greedily
    remaining = np.sum(h) - target
    if abs(remaining) > 1e-10:
        if remaining > 0:
            order = np.argsort(-h)
        else:
            order = np.argsort(h)
        for i in order:
            if remaining > 0:
                adj = min(h[i], remaining)
                h[i] -= adj
                remaining -= adj
            else:
                adj = min(1 - h[i], -remaining)
                h[i] += adj
                remaining += adj
            if abs(remaining) < 1e-10:
                break
    return h

    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    k_star = np.argmax(corr)  # index in full correlation array
    # Convert to actual lag
    # full corr index k corresponds to lag k - (n-1)

            j = i + l
            grad[j] += (1.0 - h[i])
