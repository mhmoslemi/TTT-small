So Δc[k] from h[i]: -1*(1-h[i-s]) + h[i+s] (for valid indices)

Similarly for h[j] going from 0 to 1:
Δc[k] from h[j]: +1*(1-h[j-s]) - h[j+s] (for valid indices)

Total: Δc[k] = -(1-h[i-s])*[valid] + h[i+s]*[valid] + (1-h[j-s])*[valid] - h[j+s]*[valid]

where s = k-(n-1) and [valid] means the index is in [0, n-1].

So I can compute the new correlation in O(n) per swap (just adding Δc to the current correlation).

This makes the full local search: O(n^2 * n) = O(n^3) per round (n^2 swaps * O(n) per swap). For n=80: 512,000 per round. Very fast.

import numpy as np
import time
from scipy.optimize import minimize

def c5_eval(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr * dx)

def adjust_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff < 1e-14:
                break
    else:
        order = np.argsort(-h)
        for idx in order:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
            if diff > -1e-14:
                break
    return h

def subgradient_optimize(h, n, max_iters=600, rng=None):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = adjust_sum(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = c5_eval(h, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    stall = 0
    for it in range(max_iters):
        c_scaled = corr * dx
        c5 = np.max(c_scaled)
        k_star = np.argmax(c_scaled)
        s = k_star - (n - 1)
        grad = np.zeros(n)
        js = np.arange(n)
        valid_p = (js - s >= 0) & (js - s < n)
        valid_m = (js + s >= 0) & (js + s < n)
        grad[valid_p] += (1.0 - h[js[valid_p] - s])
        grad[valid_m] -= h[js[valid_m] + s]
        alpha = 0.3 / (1.0 + 0.005 * it)
        h_new = adjust_sum(h - alpha * grad, target_sum)
        c5_new = c5_eval(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            corr = np.correlate(h, 1.0 - h, mode="full")
            stall = 0
        else:
            stall += 1
            if rng is not None and stall > 40:
                k_rand = rng.randint(max(0, k_star - 3), min(2 * n - 2, k_star + 3))
                s_r = k_rand - (n - 1)
                grad_r = np.zeros(n)
                vp = (js - s_r >= 0) & (js - s_r < n)
                vm = (js + s_r >= 0) & (js + s_r < n)
                grad_r[vp] += (1.0 - h[js[vp] - s_r])
                grad_r[vm] -= h[js[vm] + s_r]
                h_try = adjust_sum(h - alpha * grad_r, target_sum)
                if c5_eval(h_try, n) < best_c5:
                    h = h_try
                    best_c5 = c5_eval(h, n)
                    best_h = h.copy()
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    stall = 0
            elif stall > 200:
                break
    return best_h, best_c5

def local_search_01(h, n, max_rounds=200):
    dx = 2.0 / n
    target_ones = n // 2
    h = h.copy()
    ones_mask = h >= 0.5
    n_ones = int(ones_mask.sum())
    if n_ones > target_ones:
        idx = np.where(ones_mask)[0]
        for i in range(n_ones - target_ones):
            ones_mask[idx[i]] = False
    elif n_ones < target_ones:
        idx = np.where(~ones_mask)[0]
        for i in range(target_ones - n_ones):
            ones_mask[idx[i]] = True
    h_bin = ones_mask.astype(float)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode="full").astype(float)
    current_c = np.max(corr * dx)
    best_c = current_c
    best_h = h_bin.copy()
    for r in range(max_rounds):
        improved = False
        ones_list = np.where(h_bin == 1)[0]
        zeros_list = np.where(h_bin == 0)[0]
        for i in ones_list:
            for j in zeros_list:
                s_vals = np.arange(2 * n - 1) - (n - 1)
                js = np.arange(n)
                # delta for removing 1 at position i
                dp_i = np.zeros(2 * n - 1)
                for k_idx in range(2 * n - 1):
                    sk = k_idx - (n - 1)
                    cp = i - sk
                    cm = i + sk
                    val = 0.0
                    if 0 <= cp < n:
                        val += (1.0 - h_bin[cp])
                    if 0 <= cm < n:
                        val -= h_bin[cm]
                    dp_i[k_idx] = -val
                # delta for adding 1 at position j
                for k_idx in range(2 * n - 1):
                    sk = k_idx - (n - 1)
                    cp = j - sk
                    cm = j + sk
                    val = 0.0
                    if 0 <= cp < n:
                        val += (1.0 - h_bin[cp])
                    if 0 <= cm < n:
                        val -= h_bin[cm]
                    dp_i[k_idx] += val
                new_corr = corr + dp_i
                new_c = np.max(new_corr * dx)
                if new_c < current_c - 1e-14:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    corr = new_corr
                    current_c = new_c
                    if current_c < best_c:
                        best_c = current_c
                        best_h = h_bin.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)
    try:
        h_parent = np.array(initial_h_values, dtype=float)

    resolutions = [80, 160, 240, 320, 480]
    for n in resolutions:
        if time.time() - t_start > budget_s - 120:
            break
        target_sum = n / 2.0
        dx = 2.0 / n

        # Build initial guess

                h = h_parent.copy()
            else:
                xs_old = np.linspace(0, 2, len(h_parent), endpoint=False)

        h_opt, c5_opt = subgradient_optimize(h, n, max_iters=800, rng=rng)

        h_bin_ls, c5_ls = local_search_01(h_opt, n, max_rounds=300)

        n_restarts = 3 if n <= 160 else 2
        for _ in range(n_restarts):
            if time.time() - t_start > budget_s - 90:
                break
            h_rand = np.zeros(n)
            # Random block structure
            n_blocks = rng.randint(2, 6)
            remaining = target_sum
            pos = 0
            for b in range(n_blocks):
                block_len = rng.randint(n // (2 * n_blocks), n // n_blocks)
                block_len = min(block_len, n - pos)
                height = min(1.0, remaining / max(block_len, 1))
                h[pos:pos + block_len] = height
                remaining -= height * block_len
                pos += block_len + rng.randint(0, n // n_blocks)

            h_r_ls, c5_r_ls = local_search_01(h_r_opt, n, max_rounds=200)
            if c5_r_ls < best_c5:

        # Try SLSQP at small resolutions
        if n <= 160 and time.time() - t_start < budget_s - 180:

        best_c5 = final_c5

    return (best_h, best_c5, best_n)
