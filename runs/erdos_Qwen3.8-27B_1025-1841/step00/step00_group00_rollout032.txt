import numpy as np
from scipy.optimize import minimize
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def soft_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    if np.max(shifted) > 50:
        return m
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def project_onto_feasible(h, n):
    """Project h onto {sum=n/2, 0<=h<=1} using bisection"""
    target = n / 2
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_new = np.clip(h - mid, 0, 1)
        s = np.sum(h_new)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix remaining error
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:
        mask = (h_new > 0) & (h_new < 1)
        if np.sum(mask) > 0:
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0, 1)
    return h_new

def grad_soft_max_c5(h, alpha):
    """Gradient of soft_max(correlate(h, 1-h) * dx) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    if np.max(shifted) > 50:
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
    else:
        w = np.exp(shifted)
        w /= np.sum(w)
    
    # grad[j] = dx * sum_m w[m] * (d corr_m / d h[j])
    # corr_m = sum_i h[i] * (1-h[i+m-(n-1)])
    # d corr_m / d h[j] = (1-h[j+m-(n-1)]) if valid - h[j-m+(n-1)] if valid
    
    grad = np.zeros(n)
    for j in range(n):
        # First part: sum_m w[m] * (1-h[j+m-(n-1)]) where 0 <= j+m-(n-1) < n
        # j+m-(n-1) in [0,n-1] => m in [n-1-j, 2n-2-j]
        m_lo = max(0, n-1-j)
        m_hi = min(2*n-2, 2*n-2-j)
        if m_lo <= m_hi:
            indices = j + np.arange(m_lo, m_hi+1) - (n-1)
            grad[j] += np.sum(w[m_lo:m_hi+1] * (1 - h[indices]))
        
        # Second part: -sum_m w[m] * h[j-m+(n-1)] where 0 <= j-m+(n-1) < n
        # j-m+(n-1) in [0,n-1] => m in [j-(n-1), j]
        m_lo2 = max(0, j-(n-1))
        m_hi2 = min(2*n-2, j)
        if m_lo2 <= m_hi2:
            indices = j - np.arange(m_lo2, m_hi2+1) + (n-1)
            grad[j] -= np.sum(w[m_lo2:m_hi2+1] * h[indices])
    
    grad *= dx
    return grad

def projected_gradient(h, alpha, lr=0.01, max_iter=1000):
    """Projected gradient descent on soft max C5"""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for it in range(max_iter):
        g = grad_soft_max_c5(h_cur, alpha)
        h_new = h_cur - lr * g
        h_new = project_onto_feasible(h_new, n)
        
        # Adaptive step size
        c5_new = compute_c5(h_new)
        if c5_new > best_c5 * 1.01:
            lr *= 0.5
            h_new = h_cur - lr * g
            h_new = project_onto_feasible(h_new, n)
        else:
            h_cur = h_new
            c5_new = compute_c5(h_cur)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
    
    return best_h, best_c5

def slsqp_optimize(h, alpha, maxiter=500):
    n = len(h)
    
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1 - x, mode="full") * dx
        return soft_max(corr, alpha)
    
    def obj_grad(x):
        return grad_soft_max_c5(x, alpha)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    res = minimize(obj, h, method='SLSQP', jac=obj_grad, bounds=bounds,
                  constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

from scipy.optimize import minimize
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def soft_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    if np.max(shifted) > 50:
        return m
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def project_onto_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2.0, 0, 1)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:
        mask = (h_new > 1e-15) & (h_new < 1 - 1e-15)
        if np.any(mask):
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0, 1)
    return h_new

def grad_soft_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    if np.max(shifted) > 50:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    else:
        w = np.exp(shifted)
        w /= np.sum(w)
    
    grad = np.zeros(n)
    for j in range(n):
        # First part: (1-h[j+m-(n-1)]) for valid m
        m_lo = max(0, n - 1 - j)
        m_hi = min(2 * n - 2, 2 * n - 2 - j)

            idx = j + np.arange(m_lo, m_hi + 1) - (n - 1)
            grad[j] += np.dot(w[m_lo:m_hi + 1], 1 - h[idx])
        
        # Second part: -h[j-m+(n-1)] for valid m

            grad[j] -= np.dot(w[m_lo2:m_hi2 + 1], h[idx])
    
    grad *= dx

    def obj(x):

    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h, method='SLSQP', jac=obj_grad, bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")

def soft_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    if np.max(shifted) > 50:
        return m
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def project_onto_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2.0, 0, 1)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:
        mask = (h_new > 1e-15) & (h_new < 1.0 - 1e-15)
        if np.any(mask):
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0, 1)
    return h_new


def grad_soft_max_c5(h, alpha):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    if np.max(shifted) > 50:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    else:
        w = np.exp(shifted)
        w /= np.sum(w)

    for j in range(n):

        m_hi = min(2 * n - 2, 2 * n - 2 - j)
