import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(v, target_sum):
    n = len(v)
    if abs(v.sum() - target_sum) < 1e-10 and (v.min() >= -1e-10) and (v.max() <= 1.0 + 1e-10):
        return np.clip(v, 0, 1)
    lo = v.min() - target_sum - 2.0
    hi = v.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        p = np.clip(v - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s > 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    elif s < 0:
        t = -s
        g[:n-t] += 1.0 - h[t:]
        g[t:] -= h[:n-t]
    else:
        g[:] += 1.0 - 2.0 * h
    return g

def optimize(h_init, lr_init=0.01, max_iter=5000, time_limit=60.0, top_lags=5):
    n = len(h_init)
    target = n * 0.5
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c = eval_c5(h)
    t0 = time.time()
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if best_c <= 0.38:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Get top lags
        top_k = min(top_lags, 2*n-1)
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        
        g = np.zeros(n)
        for k in top_idx:
            s = int(k) - (n - 1)
            g += grad_shift(h, s)

            h = h_new
            # Slightly increase lr

                lr = lr_init

def upsample(h, new_n):
    n = len(h)
    if new_n <= n:
        return h[:new_n].copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)

def binary_round(h):
    """Round to binary, adjusting to meet sum constraint."""
    n = len(h)
    target = n * 0.5
    h_bin = (h > 0.5).astype(float)
    deficit = target - h_bin.sum()
    if deficit > 0:
        # Need 'deficit' more ones; set the largest fractional values to 1
        frac = h_bin < 1.0  # all zero
        # Actually, let's sort by h value descending and flip top 'deficit'
        order = np.argsort(-h)
        for i in range(int(np.ceil(deficit))):
            h_bin[order[i]] = 1.0
    elif deficit < 0:
        order = np.argsort(h)
        for i in range(int(np.ceil(-deficit))):
            h_bin[order[i]] = 0.0
    # For fractional target (odd n), set one value to 0.5
    if abs(target - h_bin.sum()) > 1e-10 and n % 2 == 1:
        # Find median value and set to 0.5
        h_bin[np.argsort(h)[n//2]] = 0.5
    return h_bin
