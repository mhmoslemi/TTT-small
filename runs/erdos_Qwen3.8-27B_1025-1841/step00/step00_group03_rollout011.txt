    corr = np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):

    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h

    else:
        h[:] = 0.5
    h = np.clip(h, 0.0, 1.0)

    if diff > 1e-12:
        idx = np.argsort(h)
        for i in idx:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
            if diff <= 1e-13:
