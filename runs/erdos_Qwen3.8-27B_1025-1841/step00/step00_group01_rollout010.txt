from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_values(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_max(v, beta=80.0):
    m = np.max(v)
    if m < -1000:
        return 0.0
    return float(m + np.log(np.sum(np.exp(beta * (v - m)))) / beta)

def enforce_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 0:
        h = h * (n / 2.0) / s
    h = np.clip(h, 0.0, 1.0)
    # If clipping changed the sum, try to fix
    s = np.sum(h)
    target = n / 2.0
    if abs(s - target) > 1e-10:
        # Add or remove from entries with room
        diff = target - s
        if diff > 0:
            # Need to increase some entries
            room = 1.0 - h
            order = np.argsort(-room)
            i = 0
            while diff > 1e-12 and i < len(h):
                add = min(diff, room[order[i]])
                h[order[i]] += add
                diff -= add
                i += 1
        else:
            # Need to decrease some entries
            room = h.copy()
            order = np.argsort(room)
            i = 0
            while diff < -1e-12 and i < len(h):
                rem = min(-diff, room[order[i]])
                h[order[i]] -= rem
                diff += rem
                i += 1
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def c5_vector(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max(v, beta=100.0):
    m = np.max(v)
    if not np.isfinite(m):
        return 0.0
    shifted = beta * (v - m)
    shifted = np.clip(shifted, -50, 50)
    return float(m + np.log(np.sum(np.exp(shifted))) / beta)

def normalize_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    target = float(n) / 2.0
    if s > 1e-15:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                if add > 0:
                    h[idx] += add
                    diff -= add
                    if abs(diff) < 1e-12:
                        break
        else:
            order = np.argsort(h)
            for idx in order:
                rem = min(-diff, h[idx])
                if rem > 0:
                    h[idx] -= rem
                    diff += rem
                    if abs(diff) < 1e-12:
                        break
    return h

def slsqp_optimize(h_init, n, beta=100.0, maxiter=400):
    dx = 2.0 / n
    def obj(hv):
        return smooth_max(c5_vector(hv, dx), beta)
    res = minimize(obj, h_init, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res

def animate_beta(h_init, n, betas=[30, 60, 100, 200], maxiter_each=150):
    h_cur = h_init.copy()
    for b in betas:
        res = slsqp_optimize(h_cur, n, beta=b, maxiter=maxiter_each)
        if res.x is not None:
            h_new = normalize_h(res.x, n)
            if compute_c5(h_new, 2.0/n) <= compute_c5(h_cur, 2.0/n) + 1e-6:
                h_cur = h_new
    return h_cur

def local_hillclimb(h, n, rng, n_rounds=300):
    dx = 2.0 / n
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    stagnation = 0
    for it in range(n_rounds):
        improved = False
        for attempt in range(5):
            i = rng.randint(n)
            old = h[i]
            deltas = rng.uniform(-0.15, 0.15, 5)
            for d in deltas:
                h[i] = old + d
                h_try = normalize_h(h, n)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-9:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            h[i] = old if not improved else h[i]
            if improved:
                break
        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 40:
                break
    return h, best_c5
