import numpy as np
import time

def c5_val(h):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def c5_corr(h):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr, dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        for i in np.argsort(1.0 - h):
            inc = min(d, 1.0 - h[i])
            if inc <= 0:
                break
            h[i] += inc
            d -= inc
            if d <= 1e-12:
                break
    else:
        for i in np.argsort(h)[::-1]:
            dec = min(-d, h[i])
            if dec <= 0:
                break
            h[i] -= dec
            d += dec
            if d >= -1e-12:
                break
    return h
