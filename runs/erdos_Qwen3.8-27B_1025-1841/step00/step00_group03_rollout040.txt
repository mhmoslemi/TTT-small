def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')

def repair_sum(h, n):
    target = n / 2.0

        total = np.sum(space)
        if total > 0:
            h += diff * space / total
    else:
        avail = np.clip(h, 0, None)
        total = np.sum(avail)
        if total > 0:
            h += diff * avail / total
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_old, n_new):

    x_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def subgradient_step(h, n, lr):
    """One subgradient step on the max overlap"""
    dx = 2.0 / n
    overlaps = compute_all_overlaps(h, n)
    # Find the active constraint(s)
    max_val = np.max(overlaps)
    # Use the top few for a smoother gradient
    threshold = max_val - 0.01
    active = np.where(overlaps >= threshold)[0]
    
    if len(active) == 0:
        return h
    
    # Compute subgradient as average over active constraints
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)  # actual shift
        for j in range(n):
            jm = j - m
            jp = j + m
            if 0 <= jm < n:
                grad[j] += (1.0 - h[jm])
            if 0 <= jp < n:
                grad[j] -= h[jp]
    
    grad /= len(active)
    
    h_new = h - lr * grad * dx
    h_new = np.clip(h_new, 0, 1)
    h_new = repair_sum(h_new, n)
    return h_new

def coordinate_descent(h, n, max_sweeps=100):
    """Sweep through coordinates, adjusting pairs"""
    c5 = compute_c5(h, n)
    for sweep in range(max_sweeps):
        improved = False

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def repair_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        space = np.clip(1.0 - h, 0, None)
        total = np.sum(space)
        if total > 0:
            h = h + diff * space / total
    else:
        avail = np.clip(h, 0, None)
        total = np.sum(avail)
        if total > 0:
            h = h + diff * avail / total
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def make_move(h, n, scale, rng):
    move_type = rng.random()
    if move_type < 0.55:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            return h
        delta = rng.uniform(-scale, scale)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            return h
        return h_new
    elif move_type < 0.80:
        blen = rng.randint(2, min(10, n // 4 + 2))
        start = rng.randint(0, max(1, n - blen))
        delta = rng.uniform(-scale * 0.5, scale * 0.5)
        h_new = h.copy()
        h_new[start:start + blen] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = repair_sum(h_new, n)
        return h_new
    else:
        freq = rng.randint(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        pert = (scale * 0.3) * np.sin(2 * np.pi * freq * x / n + phase)
        h_new = h + pert
        h_new = np.clip(h_new, 0, 1)
        h_new = repair_sum(h_new, n)
        return h_new

def guided_move(h, n, scale, rng):
    overlaps = compute_all_overlaps(h, n)
    k_max = int(np.argmax(overlaps))
    m = k_max - (n - 1)
    if m >= 0:
        valid = range(0, n - m)
    else:
        valid = range(-m, n)
    prods = np.array([h[i + m] * (1.0 - h[i]) for i in valid])
    top_count = min(8, len(valid))
    top_idx = np.argsort(prods)[::-1][:top_count]
    pick = top_idx[rng.randint(top_count)]
    i = valid[pick]
    im = i + m
    h_new = h.copy()
    if rng.random() < 0.5:
        h_new[im] -= scale
        j = rng.randint(n)
        attempts = 0
        while j in (im, i) and attempts < 20:
            j = rng.randint(n)
            attempts += 1
        h_new[j] += scale
    else:
        h_new[i] += scale
        j = rng.randint(n)
        attempts = 0
        while j in (im, i) and attempts < 20:
            j = rng.randint(n)
            attempts += 1
        h_new[j] -= scale
    if h_new[im] < 0 or h_new[im] > 1 or h_new[i] < 0 or h_new[i] > 1:
        return h
    if j < 0 or j >= n or h_new[j] < 0 or h_new[j] > 1:
        return h
    return h_new
