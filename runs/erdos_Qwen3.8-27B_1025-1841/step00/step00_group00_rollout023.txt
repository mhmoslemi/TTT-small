def upsample(h, new_n):
    """Upsample h to new_n points, preserving the integral."""
    old_n = len(h)
    # Use linear interpolation
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h, period=2)  # Wait, not periodic
    new_h = np.clip(new_h, 0, 1)
    # Fix the sum constraint
    target_sum = new_n / 2.0
    current_sum = np.sum(new_h)
    # Adjust
    if abs(current_sum - target_sum) > 1e-6:
        # Redistribute
        if current_sum < target_sum:
            # Add to randomly chosen positions
            diff = target_sum - current_sum
            # Distribute evenly or to positions with room
            room = 1.0 - new_h
            if np.sum(room) > diff:
                new_h += diff * room / np.sum(room)
            else:
                new_h = np.clip(new_h + diff / new_n, 0, 1)
        else:
            diff = current_sum - target_sum
            space = new_h
            if np.sum(space) > diff:
                new_h -= diff * space / np.sum(space)
            else:
                new_h = np.clip(new_h - diff / new_n, 0, 1)

def fix_sum(h, target_sum):

import numpy as np
from scipy.optimize import minimize
from time import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / n)

def fix_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h.copy()
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        total = np.sum(room)
        if total > 1e-12:
            h += diff * room / total
        else:
            h.fill(1.0)
    else:
        total = np.sum(h)
        if total > 1e-12:
            h += diff * h / total
        else:
            h.fill(0.0)
    return np.clip(h, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float) / old_n
    new_x = np.arange(new_n, dtype=float) / new_n
    h_new = np.interp(new_x, old_x, h)
    h_new = np.clip(h_new, 0, 1)
    return fix_sum(h_new, new_n / 2.0)

def slsqp_optimize(h, max_iter=300):
    n = len(h)
    target = n / 2.0
    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    bounds = [(0, 1)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, maxiter=max_iter,
                      options={'ftol': 1e-12})
        h_out = np.clip(res.x, 0, 1)
    except Exception:
        h_out = h.copy()
    return fix_sum(h_out, target)

def to_binary(h, rng):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[rng.choice(idx, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[rng.choice(idx, target - n_ones, replace=False)] = 1
    return h_bin

def exhaustive_ls(h_bin, max_rounds=200, time_limit=None):
    n = len(h_bin)
    h = h_bin.copy()
    best_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    t0 = time()
    for rnd in range(max_rounds):
        if time_limit and time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c = best_corr
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if c < best_c - 1e-12:
                    best_c = c
                    best_pair = (i, j)
                h[i] = 1
                h[j] = 0
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            best_corr = best_c
            improved = True
        if not improved:
            break
    return h, best_corr * (2.0 / n)

def random_ls(h_bin, n_iters=10000, rng=None):

def sa_binary(h, n_iters=20000, rng=None, t_start=2.0, t_end=0.005):

        rng = np.random.RandomState(42)
    n = len(h)

    cur_corr = np.max(np.correlate(h_cur, 1.0 - h_cur, mode="full"))
    best_corr = cur_corr
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / max(n_iters, 1))

        delta = new_corr - cur_corr

                best_h = h_cur.copy()
        else:
            h_cur[i] = 1
            h_cur[j] = 0
