    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    return np.sum(h) - len(h) / 2.0

    return [(0.0, 1.0)] * n

def _fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = float(np.sum(m))
        if c == 0:

        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    return np.sum(h) - len(h) / 2.0

def _bd(n):
    return [(0.0, 1.0)] * n

    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(50):
        d = T - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = float(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

    if len(h) == n:

    i = np.floor(x).astype(int)
    j = np.minimum(i + 1, len(h) - 1)
