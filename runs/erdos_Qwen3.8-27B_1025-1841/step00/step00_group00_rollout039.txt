def compute_c5(h, n):

    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def compute_grad(h, n):
    """Compute gradient of soft-max approximation of max_k c[k]"""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    tau = 20
    # Soft-max weights
    w = np.exp(tau * corr - np.max(corr * tau))  # numerically stable
    w = w / np.sum(w)
    # Gradient of sum_k w_k * c[k] with respect to h
    # c[k] = sum_i h[i+k_idx] * (1 - h[i]) where k_idx = k - (n-1)
    # dc[k]/dh[j] = (1-h[j-k]) * [0<=k<=j] + (-h[j+k]) * [0<=j+k<n]
    # where k is the lag (can be negative)
    
    grad = np.zeros(n)
    for k_idx in range(n):  # k_idx is the lag, 0 to n-1
        # c[k_idx] = sum_{i=0}^{n-1-k_idx} h[i+k_idx] * (1-h[i])
        weight = w[k_idx] * dx
        for i in range(n - k_idx):
            grad[i+k_idx] += weight * (1 - h[i])
            grad[i] += weight * (-h[i+k_idx])
        # For negative lag -k_idx (if k_idx > 0):
        # c[-k_idx] = sum_{i=k_idx}^{n-1} h[i-k_idx] * (1-h[i])
        if k_idx > 0:
            weight_neg = w[n-1-k_idx] * dx  # index n-1-k_idx in full corr corresponds to lag -k_idx
            for i in range(k_idx, n):
                grad[i-k_idx] += weight_neg * (1 - h[i])
                grad[i] += weight_neg * (-h[i-k_idx])
    return grad
```

            # 1D subproblem: min_{t in [0,1]} max_j (alpha[j]*t + beta[j])
            T_lo = np.max(np.where(alpha > 0, beta, np.where(alpha < 0, alpha+beta, beta)))
            T_hi = np.max(np.where(alpha > 0, alpha+beta, np.where(alpha < 0, beta, beta)))
            
            if T_hi - T_lo < 1e-12:
                continue
            
            # Binary search
            for _ in range(40):
                T_mid = 0.5*(T_lo + T_hi)
                lo_t = 0.0
                hi_t = 1.0
                pos = alpha > 0
                neg = alpha < 0
                if np.any(pos):
                    hi_t = min(hi_t, np.min((T_mid - beta[pos]) / alpha[pos]))
                if np.any(neg):
                    lo_t = max(lo_t, np.max((T_mid - beta[neg]) / alpha[neg]))
                if lo_t <= hi_t + 1e-14:
                    T_hi = T_mid
                else:
                    T_lo = T_mid
            
            # Find t
            lo_t = 0.0
            hi_t = 1.0
            pos = alpha > 0
            neg = alpha < 0
            if np.any(pos):
                hi_t = min(hi_t, np.min((T_hi - beta[pos]) / alpha[pos]))
            if np.any(neg):
                lo_t = max(lo_t, np.max((T_hi - beta[neg]) / alpha[neg]))
            t_new = max(0.0, min(1.0, 0.5*(lo_t+hi_t)))
            
            if abs(t_new - h[i]) > 1e-12:
                delta = t_new - h[i]
                h[i] = t_new
                corr += alpha * delta
                changed = True
        
        current_c5 = np.max(corr) * (2.0/n)
        if current_c5 < best_c5:
            best_c5 = current_c5
        if not changed:
            break
    
    return h, best_c5
```

import numpy as np
import time

def project_to_sum(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += (diff / tr) * room
        else:
            room = h.copy()
            tr = room.sum()
            if tr < 1e-14:
                break
            h += (-diff / tr) * room
        np.clip(h, 0.0, 1.0, out=h)
    return h

def c5_fast(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))
