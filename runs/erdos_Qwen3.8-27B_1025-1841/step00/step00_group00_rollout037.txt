    n = len(h)
    dx = 2.0 / n
    f_overlaps = np.correlate(h, 1-h, mode='full') * dx
    c5 = np.max(f_overlaps)
    start_time = time.time()
    
    while time.time() - start_time < time_budget:
        # Compute subgradient at active lags
        active = f_overlaps >= c5 - 1e-10
        n_active = np.sum(active)
        
        # Compute sg[i] = average over active lags of d f_l / d h[i]
        # This is: for each active lag l (shift s = l - (n-1)):
        #   contribution to h[i]: dx * [(1-h[i+s]) * [valid] - h[i-s] * [valid]]
        
        # More efficiently:
        # sg[i] = (1/n_active) * sum_{active l} dx * [(1-h[i+s_l]) * [valid] - h[i-s_l] * [valid]]
        
        # Let's compute this by iterating over active lags
        sg = np.zeros(n)
        active_lags = np.where(active)[0]
        for l in active_lags:
            s = l - (n - 1)  # shift
            # For each i, contribution is (1-h[i+s]) if 0<=i+s<n, minus h[i-s] if 0<=i-s<n
            # (1-h[i+s]) for i = 0 to n-1-s (if s>=0) or i = -s to n-1 (if s<0)
            if s >= 0:
                valid_i = range(0, n - s)
                sg[valid_i] += 1.0 - h[valid_i + s]
            else:
                valid_i = range(-s, n)
                sg[valid_i] += 1.0 - h[valid_i + s]
            
            if s >= 0:
                valid_i2 = range(s, n)
                sg[valid_i2] -= h[valid_i2 - s]
            else:
                valid_i2 = range(0, n + s)  # n+s = n-|s|
                sg[valid_i2] -= h[valid_i2 - s]
        
        sg /= n_active  # don't multiply by dx, it's a constant factor
        
        # Find best swap: increase h at most negative sg, decrease at most positive sg
        # But we need to maintain 0 <= h <= 1
        # j: position to increase (most negative sg, and h[j] < 1)
        # i: position to decrease (most positive sg, and h[i] > 0)
        
        # Mask valid positions
        can_increase = (sg < 0) & (h < 1 - 1e-10)
        can_decrease = (sg > 0) & (h > 1e-10)
        
        if not np.any(can_increase) or not np.any(can_decrease):
            # Try random perturbations
            break
        
        j = np.where(can_increase)[0][np.argmax(sg[can_increase] < 0) ... ]
        # Actually: j = argmin of sg among can_increase positions
        # i = argmax of sg among can_decrease positions
        
        candidates_j = np.where(can_increase)[0]
        candidates_i = np.where(can_decrease)[0]
        j = candidates_j[np.argmin(sg[candidates_j])]
        i = candidates_i[np.argmax(sg[candidates_i])]
        
        # Now find optimal delta for swap (i decreases, j increases)
        # h[i] -= delta, h[j] += delta, delta in [0, min(h[i], 1-h[j])]
        
        # Compute g = A_j - A_i (note: we're increasing j and decreasing i)
        # Change in f_l: delta * g_l where g_l = A_j[s_l] - A_i[s_l]
        
        # Actually, let me just compute the effect directly
        # For the 1D problem: minimize max_l (f_overlaps[l] + delta * g[l])
        # where g[l] is the change in lag l per unit delta
        
        shifts = active_lags - (n - 1)
        # Actually we need ALL lags, not just active ones
        all_shifts = np.arange(-(n-1), n)
        
        # g for position j (increased): g_j[s] = (1-h[j+s])*[valid] - h[j-s]*[valid]
        # g for position i (decreased): g_i[s] = (1-h[i+s])*[valid] - h[i-s]*[valid]
        # Total g = g_j - g_i (since we increase j and decrease i)
        
        g = np.zeros(2*n-1)
        for idx, s in enumerate(all_shifts):
            val = 0.0
            # j contribution
            if 0 <= j + s < n:
                val += 1.0 - h[j + s]
            if 0 <= j - s < n:
                val -= h[j - s]
            # i contribution (subtracted)
            if 0 <= i + s < n:
                val -= 1.0 - h[i + s]
            if 0 <= i - s < n:
                val += h[i - s]
            g[idx] = val
        
        # Minimize max_l (f_overlaps[l] + delta * g[l]) for delta in [0, min(h[i], 1-h[j])]
        a, b = 0.0, min(h[i], 1.0 - h[j])
        if b < 1e-15:
            continue
        
        # Bisection on subgradient
        lo, hi = a, b
        for _ in range(40):
            mid = (lo + hi) / 2
            val = f_overlaps + mid * g
            m = np.max(val)
            active_now = val >= m - 1e-12
            g_active = g[active_now]
            if np.min(g_active) > 1e-12:
                hi = mid
            elif np.max(g_active) < -1e-12:
                lo = mid
            else:
                break
        
        best_delta = (lo + hi) / 2
        new_c5 = np.max(f_overlaps + best_delta * g)
        
        if new_c5 < c5 - 1e-12:
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0, 1)
            f_overlaps = np.correlate(h, 1-h, mode='full') * dx
            c5 = np.max(f_overlaps)
        else:
            # Stuck, try random pairs
            # ...
            break
    
    return h, c5
