```python
def minimize_max_linear(a_vals, b_vals):
    """Minimize max_k (a_k + b_k * t) for t in [0,1]"""
    # Find candidate points: 0, 1, and all pairwise intersections in [0,1]
    candidates = [0.0, 1.0]
    M = len(a_vals)
    for i in range(M):
        for j in range(i+1, M):
            db = b_vals[i] - b_vals[j]
            if abs(db) < 1e-15:
                continue
            t = (a_vals[j] - a_vals[i]) / db
            if 0.0 <= t <= 1.0:
                candidates.append(t)
    
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        val = np.max(a_vals + b_vals * t)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t
```
