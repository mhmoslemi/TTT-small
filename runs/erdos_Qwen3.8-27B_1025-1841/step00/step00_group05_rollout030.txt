def enforce_sum(h, target):
    """Adjust h to have sum = target, keeping values in [0,1]"""
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to add diff total
        room = 1.0 - h  # how much we can add at each position
        order = np.argsort(-room, kind='stable')  # positions with most room first
        for i in order:
            add = min(room[i], diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        room = h  # how much we can subtract
        order = np.argsort(-room, kind='stable')  # positions with most to give first
        for i in order:
            sub = min(room[i], diff)
            h[i] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    return h

    c = np.correlate(h, 1-h, mode="full") * dx
    # Log-sum-exp
    m = np.max(c)
    shifted = c - m
    exp_c = np.exp(alpha * shifted)
    lse = m + np.log(np.sum(exp_c)) / alpha
    # Weights
    p = exp_c / np.sum(exp_c)
    # Gradient
    # p_k * dc_k/dh_i
    s_idx = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    # For each i, grad[i] = sum_k p[k] * dx * [(1-h[i-s_k])*valid1 - h[i+s_k]*valid2]
    # Vectorize: for each k, the contribution to all i is a shifted version
    # Let's just do it with loops over active k's (those with significant p[k])
    significant = p > 1e-6
    for k in np.where(significant)[0]:
        s = s_idx[k]
        pk = p[k] * dx
        # dc_k/dh_i = (1-h[i-s])*[0<=i-s<n] - h[i+s]*[0<=i+s<n]
        if s >= 0:
            # i ranges from s to n-1 for the first term
            # i ranges from 0 to n-1-s for the second term
            # First term: (1-h[i-s]) for i in [s, n-1]
            grad[s:] += pk * (1 - h[:n-s])
            # Second term: -h[i+s] for i in [0, n-1-s]
            grad[:n-s] -= pk * h[s:]
        else:
            s = -s
            # First term: (1-h[i+s]) for i in [0, n-1-s]
            grad[:n-s] += pk * (1 - h[s:])
            # Second term: -h[i-s] for i in [s, n-1]
            grad[s:] -= pk * h[:n-s]
    return lse, grad

```python
def evaluate_swap(h, c, i, j, n):
    """Evaluate the change in C₅ from swapping h[i]=1→0 and h[j]=0→1"""
    dx = 2.0/n
    # s_k = k - (n-1) for k = 0..2n-2
    s = np.arange(2*n-1) - (n-1)
    
    # Δc[k] = -(1-h[i-s_k]) + h[i+s_k] + (1-h[j-s_k]) - h[j+s_k]
    # with out-of-bounds = 0
    
    delta = np.zeros(2*n-1)
    
    for k_idx, s_k in enumerate(s):
        val = 0.0
        idx1 = i - s_k
        if 0 <= idx1 < n:
            val -= (1 - h[idx1])
        idx2 = i + s_k
        if 0 <= idx2 < n:
            val += h[idx2]
        idx3 = j - s_k
        if 0 <= idx3 < n:
            val += (1 - h[idx3])
        idx4 = j + s_k
        if 0 <= idx4 < n:
            val -= h[idx4]
        delta[k_idx] = val
    
    new_max = np.max(c + delta) * dx
    return new_max
```

def evaluate_swap_vec(h, c, i, j, n):

    s = np.arange(2*n-1) - (n-1)

    idx1 = i - s

    term2 = np.zeros(2*n-1)

def enforce_sum(h, target):
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room, kind='stable')
        for i in order:
            add = min(room[i], diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
