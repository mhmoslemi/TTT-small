def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):

            cnt = np.sum(mask)

                break
            h[mask] += diff / cnt

def local_search(h, n, max_iters, time_limit, t_start):
    """Local search via pairwise mass transfer."""

            trial[j] = hj

            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = trial

    """Project h onto {h: 0<=h[i]<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):

        diff = target - s

            break
        if diff > 0:

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    eps = 0.1

def legendre_sequence(p):
    """Generate Legendre sequence of length p-1 (mapped to {0,1})."""
    # Legendre symbol: (a|p) = 1 if a is QR mod p, -1 otherwise, 0 if a=0
    a = np.array([(i*i) % p for i in range(1, p)])
    qr_set = set(a.tolist())
    seq = np.array([1 if i in qr_set else 0 for i in range(1, p)])
    return seq
