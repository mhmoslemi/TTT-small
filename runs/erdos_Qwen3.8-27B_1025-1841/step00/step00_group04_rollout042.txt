def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def _subgrad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    ks = int(np.argmax(c))
    lag = ks - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        l = -lag
        g[:n - l] += 1.0 - h[l:]
        g[l:] -= h[:n - l]
    return g * dx, float(np.max(c) * dx)

def _proj(h, ts):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
