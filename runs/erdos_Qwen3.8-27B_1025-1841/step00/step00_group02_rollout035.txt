```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(c)

def project_onto_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    S = n / 2.0
    # Bisection on lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h_projected = np.clip(h - mid, 0, 1)
        s = h_projected.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    h_projected = np.clip(h - (lo + hi) / 2, 0, 1)
    return h_projected

def softmax_gradient(h, T):
    """Compute the gradient of the smoothed max objective"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    c_max = np.max(c)
    exp_c = np.exp(T * (c - c_max))
    weights = exp_c / exp_c.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        w = weights[m]
        if w < 1e-10:
            continue
        # dc_k/dh[j] = (1-h[j-k]) * I(0<=j-k<=n-1) - h[j+k] * I(0<=j+k<=n-1)
        # Vectorized:
        # (1-h[j-k]) for valid j-k: this is (1-h) shifted by k
        # -h[j+k] for valid j+k: this is -h shifted by -k
        
        # shift(1-h, k): result[j] = (1-h)[j-k] = 1-h[j-k] if 0<=j-k<n
        # This is (1-h) right-shifted by k (or left-shifted by -k)
        
        # For k > 0:
        #   (1-h[j-k]) is valid for j >= k, gives (1-h[0..n-1-k])
        #   (1-h[j-k]) = 0 for j < k
        #   h[j+k] is valid for j <= n-1-k, gives h[k..n-1]
        #   h[j+k] = 0 for j > n-1-k
        
        # So grad contribution = [0..0, (1-h[0..n-1-k])] - [h[k..n-1], 0..0]
        
        if k >= 0:
            # (1-h[j-k]) valid for j in [k, n-1], value = 1-h[j-k]
            grad[k:] += w * (1 - h[:n-k])
            # -h[j+k] valid for j in [0, n-1-k], value = -h[j+k]
            grad[:n-k] += w * (-h[k:])
        else:
            # k < 0, let k_neg = -k
            # (1-h[j-k]) = (1-h[j+k_neg]) valid for j in [0, n-1-k_neg], value = 1-h[j+k_neg]
            grad[:n+k] += w * (1 - h[-k:])
            # -h[j+k] = -h[j-k_neg] valid for j in [k_neg, n-1], value = -h[j-k_neg]
            grad[-k:] += w * (-h[:n+k])
    
    return grad

def projected_gradient_step(h, step_size, T):
    n = len(h)
    grad = softmax_gradient(h, T)
    h_new = h - step_size * grad
    h_new = project_onto_feasible(h_new, n)
    return h_new

def optimize_h(h0, n, T=50, max_iter=5000, base_lr=0.01):
    """Run projected gradient descent"""
    h = h0.copy().astype(float)
    h = project_onto_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        lr = base_lr / (1 + it * 0.001)
        h_new = projected_gradient_step(h, lr, T)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        # Increase T over time for sharper max
        T = min(T * 1.01, 200)
    
    return best_h, best_c5
```
