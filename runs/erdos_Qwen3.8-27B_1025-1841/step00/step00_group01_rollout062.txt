def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except (NameError, UnboundLocalError):
        parent_h = np.ones(50) * 0.5
        parent_n = 50

    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h, parent_n)
    best_n = parent_n

    def done(frac=0.85):
        return (time.time() - start_time) > budget_s * frac

    # Phase 1: SLSQP at multiple resolutions
    for n in [parent_n, 64, 96, 128]:
        if done(0.22):
            break
        x_cur = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n)
        h0 = np.interp(x_new, x_cur, best_h)
        h0 = repair_h(h0, n)
        cons = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}]
        bnds = [(0.0, 1.0)] * n
        mi = max(100, 40000 // n)
        try:
            res = minimize(lambda x: compute_c5(x, n), h0, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': mi, 'ftol': 1e-10, 'iprint': 0})
            h_opt = repair_h(res.x, n)
            c5 = compute_c5(h_opt, n)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt.copy(), n
        except Exception:
            pass

    # Phase 2: Local search at best resolution
    if not done(0.82):
        h_ls = best_h.copy()
        n_ls = best_n
        bh, bc = local_search(h_ls, n_ls, 0.86, start_time, budget_s, max_iter=400000)
        if bc < best_c5:
            best_c5, best_h, best_n = bc, bh, n_ls

    # Phase 3: Try a different resolution with local search
    if not done(0.9):
        n_alt = best_n + 8
        x_cur = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_alt)
        h_alt = np.interp(x_new, x_cur, best_h)
        h_alt = repair_h(h_alt, n_alt)
        bh, bc = local_search(h_alt, n_alt, 0.93, start_time, budget_s, max_iter=200000)
        if bc < best_c5:
            best_c5, best_h, best_n = bc, bh, n_alt

    # Phase 4: Final SLSQP polish
    if not done(0.96) and best_n <= 128:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - best_n / 2.0}]
        bnds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(lambda x: compute_c5(x, best_n), best_h, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-12, 'iprint': 0})
            h_opt = repair_h(res.x, best_n)
            c5 = compute_c5(h_opt, best_n)
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt.copy()
        except Exception:
            pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = repair_h(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return (best_h, best_c5, best_n)
