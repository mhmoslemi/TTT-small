    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    total = np.sum(h)
    if total > target:
        # need to reduce
        excess = total - target
        # reduce the larger values
