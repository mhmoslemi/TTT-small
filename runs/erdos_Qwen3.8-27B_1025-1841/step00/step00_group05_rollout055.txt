def _obj_and_grad(h, n, T=50.0):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    # Soft max
    max_corr = np.max(corr)
    diff = T * (corr - max_corr)
    exp_diff = np.exp(diff)
    sum_exp = np.sum(exp_diff)
    obj = (max_corr + np.log(sum_exp) / T) * dx
    
    # Gradient: sum_k p_k * grad_f_k
    p = exp_diff / sum_exp  # probabilities, shape (2n-1,)
    
    grad = np.zeros(n)
    # For efficiency, only use top few lags
    # But let's do all for now
    for k in range(len(corr)):
        if p[k] < 1e-8:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # f_k = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            upper = n - 1 - lag
            # d/dh[j] for j in [0, upper]: (1-h[j-lag]) - h[j+lag]
            # d/dh[j] for j in [upper+1, n-1]: (1-h[j-lag])
            if lag == 0:
                # f_k = sum h[i](1-h[i]), grad = 1-2h
                grad += p[k] * (1.0 - 2.0 * h)
            else:
                idx = np.arange(n)
                # j appears as i+lag when i = j-lag, valid for j-lag in [0, upper] i.e. j in [lag, n-1]
                mask1 = (idx >= lag)  # j in [lag, n-1]
                # j appears in (1-h[i]) when i = j, valid for j in [0, upper]
                mask2 = (idx <= upper)  # j in [0, n-1-lag]
                
                g = np.zeros(n)
                g[mask1] += (1.0 - h[idx[mask1] - lag])
                g[mask2] -= h[idx[mask2] + lag]
                grad += p[k] * g
        else:
            k2 = -lag

                grad += p[k] * (1.0 - 2.0 * h)
            else:
                idx = np.arange(n)
                # j in h[i]: i=j, valid for j in [0, upper]
                # j in h[i+k2]: i+k2=j, i=j-k2, valid for j-k2 in [0, upper], j in [k2, n-1]
                mask1 = (idx <= upper)
                mask2 = (idx >= k2)
                
                g = np.zeros(n)
                g[mask1] += (1.0 - h[idx[mask1] + k2])
                g[mask2] -= h[idx[mask2] - k2]
                grad += p[k] * g
    
    return obj, grad * dx
