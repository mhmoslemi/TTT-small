                    # g[k] = (1-h_bin[i+k-(n-1)]) - h_bin[i+(n-1)-k] - (1-h_bin[j+k-(n-1)]) + h_bin[j+(n-1)-k]
                    # with bounds
                    ia = i + offsets - (n-1)
                    ib = i + (n-1) - offsets
                    ic = j + offsets - (n-1)
                    id_ = j + (n-1) - offsets
                    
                    g = np.zeros(n_off)
                    ma = (ia >= 0) & (ia < n)
                    mb = (ib >= 0) & (ib < n)
                    mc = (ic >= 0) & (ic < n)
                    md = (id_ >= 0) & (id_ < n)
                    g[ma] += (1.0 - h_bin[ia[ma]])
                    g[mb] -= h_bin[ib[mb]]
                    g[mc] -= (1.0 - h_bin[ic[mc]])
                    g[md] += h_bin[id_[md]]
                    
                    new_max = float(np.max(corr + g))
                    if new_max < target_val - 1e-12:

                        h_bin[j] = 0
