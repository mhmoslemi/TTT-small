                n_steps = min(150, max(20, int((deadline - time.time()) * 0.3)))
                h_gd = proj_grad_desc(h_work, n, T, 0.015, n_steps, deadline)
                c5_gd = evaluate_c5(h_gd)
                if c5_gd < best_c5:
                    best_c5 = c5_gd
                    best_h = h_gd.copy()
                h_work = h_gd

    # Random restarts at best resolution
    n_best = len(best_h)
    for attempt in range(30):
        if time.time() > deadline - 20:
            break
        h_pert = best_h + np.random.randn(n_best) * 0.08
        h_pert = project(h_pert, n_best)
        h_new = slsqp_opt(h_pert, n_best, 100.0, maxiter=200, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Also try from a random starting point
    if time.time() < deadline - 30:
        for attempt in range(5):
            if time.time() > deadline:
                break
            h_rand = random_feasible(n_best, seed + 1000 + attempt)
            h_new = slsqp_opt(h_rand, n_best, 50.0, maxiter=300, deadline=deadline)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
