import numpy as np
import time
import math

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def make_feasible(h):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_steepest_descent(h, n, max_rounds, t_end, rng):
    h = (h > 0.5).astype(float).copy()
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = list(np.where(h == 1)[0])
        rng.shuffle(ones_idx)
        for idx in ones_idx[:n_ones - target]:
            h[idx] = 0
    elif n_ones < target:
        zeros_idx = list(np.where(h == 0)[0])
        rng.shuffle(zeros_idx)
        for idx in zeros_idx[:target - n_ones]:
            h[idx] = 1
    c5 = eval_c5(h)
    rounds = 0
    while rounds < max_rounds:
        if time.time() > t_end:
            break
        rounds += 1
        ones_idx = list(np.where(h == 1)[0])
        zeros_idx = list(np.where(h == 0)[0])
        best_c5 = c5
        best_pair = None
        for i in ones_idx:
            for j in zeros_idx:
                h[i], h[j] = 0, 1
                v = eval_c5(h)
                h[i], h[j] = 1, 0
                if v < best_c5 - 1e-12:
                    best_c5 = v
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = 0, 1
            c5 = best_c5
        else:
            break
    return h, c5

def binary_random_swaps(h, n, n_iters, t_end, rng):
    h = (h > 0.5).astype(float).copy()
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = list(np.where(h == 1)[0])
        rng.shuffle(ones_idx)
        for idx in ones_idx[:n_ones - target]:
            h[idx] = 0
    elif n_ones < target:
        zeros_idx = list(np.where(h == 0)[0])
        rng.shuffle(zeros_idx)
        for idx in zeros_idx[:target - n_ones]:
            h[idx] = 1
    c5 = eval_c5(h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i], h[j] = 0, 1
        new_c5 = eval_c5(h)
        if new_c5 < c5:
            c5 = new_c5
        else:
            h[i], h[j] = 1, 0
    return h, c5

def continuous_local_search(h, n, n_iters, t_end, rng):
    h = make_feasible(h).copy()
    c5 = eval_c5(h)
    best_c5 = c5
    best_h = h.copy()
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        d = 0.02 * (0.5 + rng.random())
        oi, oj = h[i], h[j]
        h[i] = max(0.0, min(1.0, oi + direction * d))
        h[j] = max(0.0, min(1.0, oj - direction * d))
        new_c5 = eval_c5(h)
        if new_c5 < c5:
            c5 = new_c5
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = oi, oj
    best_h = make_feasible(best_h)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5

def upscale(h, n_new):
    n_old = len(h)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        idx = min(int(i * n_old / n_new), n_old - 1)
        h_new[i] = h_old[idx]
    return make_feasible(h_new)

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    t_end = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(42, 0.5)

    best_n = len(best_h)
    best_h = make_feasible(best_h)
    best_c5 = eval_c5(best_h)
    best_global = best_h.copy()

    # Phase 1: Binary steepest descent at initial n
    h, c5 = binary_steepest_descent(best_h, best_n, 2000, t_end, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_global = h.copy()

    # Phase 2: Binary random swaps (escape local optima)
    h, c5 = binary_random_swaps(best_global, best_n, 100000, t_end, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_global = h.copy()

    # Phase 3: Binary steepest descent again
    h, c5 = binary_steepest_descent(best_global, best_n, 2000, t_end, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_global = h.copy()

    # Phase 4: Continuous local search
    h, c5 = continuous_local_search(best_global, best_n, 200000, t_end, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_global = h.copy()

    # Phase 5: Try random binary starts
    for trial in range(20):
        if time.time() > t_end:
            break
        h_rand = random_binary(best_n, rng)
        h, c5 = binary_steepest_descent(h_rand, best_n, 500, t_end, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_global = h.copy()

    # Phase 6: Multi-resolution upscale
    for n_new in [best_n * 2, best_n * 4]:
        if time.time() > t_end:
            break
        if n_new > 400:
            break
        h_up = upscale(best_global, n_new)
        h, c5 = binary_steepest_descent(h_up, n_new, 2000, t_end, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_global = h.copy()
        h, c5 = binary_random_swaps(best_global, n_new, 50000, t_end, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_global = h.copy()
        h, c5 = binary_steepest_descent(best_global, n_new, 1000, t_end, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_global = h.copy()
        h, c5 = continuous_local_search(best_global, n_new, 100000, t_end, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_global = h.copy()

    # Phase 7: Final refinement
    h, c5 = binary_steepest_descent(best_global, len(best_global), 2000, t_end, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_global = h.copy()
    h, c5 = continuous_local_search(best_global, len(best_global), 100000, t_end, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_global = h.copy()

    best_global = make_feasible(best_global)
    best_c5 = eval_c5(best_global)
    return (best_global, best_c5, len(best_global))
