from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(params, dx, alpha):
    h = params
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(corr - m))) / alpha

def sum_constraint(params, target_sum):
    return np.sum(params) - target_sum

def project_simplex(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.any(mask):
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        n = len(h)
        h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_at_resolution(h, n, max_iter, ftol):
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (target_sum,)}]
    for alpha in [50.0, 200.0, 1000.0]:
        result = minimize(
            smooth_max_obj, h,
            args=(dx, alpha),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': ftol}
        )
        h = project_simplex(result.x, target_sum)
    return h

        return time.time() - start

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_obj(params, dx, alpha):
    h = params
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m < -500.0:
        return m
    return m + float(np.log(np.sum(np.exp(corr - m)))) / alpha


def sum_constraint(params, target_sum):
    return np.sum(params) - target_sum


def project_simplex(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)

        h[mask] += diff / float(np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        n = len(h)
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
