import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def compute_c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(conv))
    offset = k_max - (n - 1)
    grad = np.zeros(n)
    # d/dh[j] of sum_i h[i]*(1-h[i+offset])
    # = sum_i [delta(i,j)*(1-h[i+offset]) + h[i]*(-delta(i+offset,j))]
    # For first sum: term is (1-h[j+offset]) if j+offset in [0,n)
    idx_pos = np.arange(n) + offset
    valid_pos = (idx_pos >= 0) & (idx_pos < n)
    grad[valid_pos] += (1.0 - h[idx_pos[valid_pos]])
    # For second sum: term is -h[j-offset] if j-offset in [0,n)
    idx_neg = np.arange(n) - offset
    valid_neg = (idx_neg >= 0) & (idx_neg < n)
    grad[valid_neg] -= h[idx_neg[valid_neg]]
    return grad * dx

def compute_c5_smoothed(h, temperature=0.01):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    # Smoothed max: temperature * log(sum(exp(conv_i / temperature)))
    # For temperature -> 0, this approaches max
    shifted = conv - conv.max()
    smooth_max = conv.max() + temperature * np.log(np.sum(np.exp(shifted / temperature)))
    return smooth_max * dx

def compute_c5_smoothed_gradient(h, temperature=0.01):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    shifted = conv - conv.max()
    weights = np.exp(shifted / temperature) / np.sum(np.exp(shifted / temperature))
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-10:
            continue
        offset = k - (n - 1)
        idx_pos = np.arange(n) + offset
        valid_pos = (idx_pos >= 0) & (idx_pos < n)

        idx_neg = np.arange(n) - offset
        valid_neg = (idx_neg >= 0) & (idx_neg < n)
        grad[valid_neg] -= weights[k] * h[idx_neg[valid_neg]]

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def compute_c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(conv))
    offset = k_max - (n - 1)
    grad = np.zeros(n)
    idx_pos = np.arange(n) + offset
    valid_pos = (idx_pos >= 0) & (idx_pos < n)
    grad[valid_pos] += (1.0 - h[idx_pos[valid_pos]])
    idx_neg = np.arange(n) - offset
    valid_neg = (idx_neg >= 0) & (idx_neg < n)
    grad[valid_neg] -= h[idx_neg[valid_neg]]
    return grad * dx


def compute_c5_smoothed_gradient(h, temp=0.005):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    shifted = conv - conv.max()
    weights = np.exp(shifted / temp)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-12:
            continue
        offset = k - (n - 1)
        idx_pos = np.arange(n) + offset
        valid_pos = (idx_pos >= 0) & (idx_pos < n)
        grad[valid_pos] += weights[k] * (1.0 - h[idx_pos[valid_pos]])
        idx_neg = np.arange(n) - offset
        valid_neg = (idx_neg >= 0) & (idx_neg < n)
        grad[valid_neg] -= weights[k] * h[idx_neg[valid_neg]]
    return grad * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='linear', fill_value=(h[0], h[-1]), bounds_error=False)
    h_new = f(x_new)
    return project_feasible(h_new, n_new)


def projected_gradient_opt(h0, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    step = 0.05
    t0 = time.time()
    it = 0
    no_improve = 0
    while time.time() - t0 < time_limit:
        it += 1
        if it < 200:
            grad = compute_c5_gradient(best)
        else:
            grad = compute_c5_smoothed_gradient(best)
        x_new = best - step * grad
        h_new = project_feasible(x_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best = h_new
            best_c5 = c5_new
            step = min(step * 1.05, 0.5)
            no_improve = 0
        else:
            step *= 0.8
            no_improve += 1
        if no_improve > 200 and it % 50 == 0:
            noise_scale = max(0.005, 0.05 * np.exp(-it / 500.0))
            for _ in range(5):
                noise = rng.randn(n) * noise_scale
                pert = project_feasible(best + noise, n)
                c5_p = compute_c5(pert)
                if c5_p < best_c5:
                    best = pert
                    best_c5 = c5_p
                    no_improve = 0
                    break
            if no_improve > 500:
                break
    return best, best_c5


def slsqp_refine(h0, n, time_limit, seed):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)

    def obj(x):
        return compute_c5(x)

    def obj_grad(x):
        return compute_c5_gradient(x)

    def cons_fun(x):
        return x.sum() - n / 2.0

    def cons_jac(x):
        return ones

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    result = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': 500, 'ftol': 1e-14})
    h_out = np.clip(result.x, 0.0, 1.0)
    h_out = project_feasible(h_out, n)
    return h_out, compute_c5(h_out)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = project_feasible(h0, len(h0))

    remaining = budget_s - elapsed
    alloc = min(remaining * 0.2, 120.0)
    if alloc > 10:

        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: SLSQP refinement at initial resolution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30 and n_init <= 280:
        alloc = min(remaining * 0.15, 100.0)
        h_opt, c5_opt = slsqp_refine(best_h, n_init, alloc, seed + 1)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: multi-resolution upsampling and optimization
    resolutions = [140, 280, 560, 1120, 2240]
    for n_new in resolutions:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 90:
            break
        h_init = upsample(best_h, len(best_h), n_new)
        c5_init = compute_c5(h_init)
        alloc = min(remaining * 0.35, 300.0)
        h_opt, c5_opt = projected_gradient_opt(h_init, n_new, alloc, seed + n_new)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        else:
            best_h = h_init

        # SLSQP refinement if resolution is small enough
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 30 and n_new <= 280:
            alloc2 = min(remaining * 0.2, 80.0)
            h_opt2, c5_opt2 = slsqp_refine(best_h, n_new, alloc2, seed + n_new + 7)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

    # Phase 4: random restarts at best resolution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    n_best = len(best_h)
    rng = np.random.RandomState(seed + 42)
    restarts = min(10, max(1, int(remaining / 30)))
    for r in range(restarts):
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 15:
            break
        noise = rng.randn(n_best) * 0.08
        h_restart = project_feasible(best_h + noise, n_best)
        alloc = min(remaining * 0.5, 150.0)

        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: final fine-tuning
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 10:
        alloc = min(remaining - 2, 60.0)
        best_h, best_c5 = projected_gradient_opt(best_h, n_best, alloc, seed + 777)

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
