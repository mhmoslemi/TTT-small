What about h=1 on [0, 1/2] ∪ [1, 3/2]? ∫h = 1/2 + 1/2 = 1. 
This is h(x) = 1_{[0,1/2]} + 1_{[1,3/2]}.
For k=0: ∫h(1-h) = 0 (since h is binary)
For k=1/2: h(x) and 1-h(x+1/2)...
h(x+1/2) = 1 on [−1/2, 0] ∪ [1/2, 1] (shifted), so 1-h(x+1/2) = 1 on [0, 1/2] ∪ [1, ∞) (within [0,2]).
h(x) = 1 on [0,1/2] ∪ [1, 3/2].
Product: 1 on [0,1/2] ∪ [1, 3/2] ∩ [0,1/2] ∪ [1,∞) = 1 on [0,1/2] ∪ [1,3/2].
Measure = 1/2 + 1/2 = 1. So C₅ ≥ 1 here too. Bad.

What about a more spread-out h? If h(x) = 1/2 + a*cos(2πx) for some a < 1/2, then ∫h = 1 (on [0,2]).
h(x)(1-h(x+k)) = (1/2 + a cos(2πx))(1/2 - a cos(2π(x+k)))
= 1/4 - a/2 cos(2π(x+k)) + a/2 cos(2πx) - a² cos(2πx)cos(2π(x+k))
= 1/4 + a/2 [cos(2πx) - cos(2π(x+k))] - a² cos(2πx)cos(2π(x+k))

Integrating over x in [0, 2-|k|] (the overlap region):
∫cos(2πx) dx over any full period = 0
∫cos(2πx)cos(2π(x+k)) dx = 1/2 ∫[cos(4πx+2πk) + cos(2πk)] dx = (2-|k|)/2 * cos(2πk) (for integer k, this is (2-|k|)/2)
For non-integer k: = (2-|k|)/2 * cos(2πk) + oscillatory term that's small for large (2-|k|)

Let me code the gradient analytically:

corr_k = sum_{i: 0<=i<n, 0<=i+offset_k<n} h[i] * (1-h[i+offset_k])
where offset_k = k - (n-1) for k in [0, 2n-2]

d(corr_k)/d(h[j]) = 
  (1 - h[j+offset_k]) * 1_{0<=j+offset_k<n} - h[j-offset_k] * 1_{0<=j-offset_k<n}

The smooth max: F(h) = (1/β) log Σ_k exp(β * corr_k)
∇F(h) = Σ_k softmax_weight_k * ∇corr_k
where softmax_weight_k = exp(β*corr_k) / Σ_j exp(β*corr_j)

This is computable in O(n²) time.

import time

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)

def corr_and_gradient(h, beta):
    """Compute smooth max of correlations and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Smooth max
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    # For numerical stability
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    softmax_w = exp_shifted / sum_exp
    F = cmax + (1.0/beta) * np.log(sum_exp)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = softmax_w[k]
        if w < 1e-10:
            continue
        offset = k - (n - 1)
        # d(corr_k)/d(h[j]) = (1-h[j+offset])*1_{0<=j+offset<n} - h[j-offset]*1_{0<=j-offset<n}
        for j in range(n):
            g = 0.0
            if 0 <= j + offset < n:
                g += 1.0 - h[j + offset]
            if 0 <= j - offset < n:
                g -= h[j - offset]
            grad[j] += w * g
    return F, grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        h[:] = 0.5
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_corr(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))


def max_corr_gradient(h, beta=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if beta is None:
        idx = np.argmax(corr)
        weights = np.zeros_like(corr)
        weights[idx] = 1.0
    else:
        cmax = np.max(corr)
        exp_shifted = np.exp(beta * (corr - cmax))
        weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-8:
            continue
        offset = k - (n - 1)
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if hi1 > lo1:
            j = np.arange(lo1, hi1)
            grad[j] += w * (1.0 - h[j + offset])
        if hi2 > lo2:
            j = np.arange(lo2, hi2)
            grad[j] -= w * h[j - offset]
    return grad


def local_search(h, n, start_time, deadline, max_iter=2000):
    best_c5 = compute_c5(h, n)
    h = h.copy()
    step = 0.03
    beta_soft = 30.0
    stall = 0
    rng_local = np.random.RandomState(0)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c5_now = compute_c5(h, n)
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            stall = 0
        else:
            stall += 1
            if stall > 100:
                h = h + rng_local.randn(n) * 0.05
                h = repair_h(h, n)
                step = 0.03
                stall = 0
        if stall > 500:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = np.argsort(corr)[-5:]
        g = np.zeros(n)
        for kk in top_k:
            g += max_corr_gradient(h, None) * 0.2
        g = max_corr_gradient(h, beta_soft)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_trial = h - step * frac * g / gn
            h_trial = repair_h(h_trial, n)
            c5_trial = compute_c5(h_trial, n)
            if c5_trial < best_c5:
                h = h_trial
                best_c5 = c5_trial
                break
        if stall > 20:
            step *= 0.9
        elif stall == 0:
            step = min(step * 1.05, 0.08)
    return h, best_c5

    best_c5 = compute_c5(h0, n)

    # Phase 1: SLSQP
    beta_slsqp = 25.0

    def obj_slsqp(h):
        return smooth_max_corr(h, beta_slsqp)

    eq_con = {
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target_sum,
        'jac': lambda h: np.ones(n),

    if result.x is not None:

        c5_cand = compute_c5(h_cand, n)

        if c5_res < best_c5:
            best_h = h_res
            best_c5 = c5_res

    # Phase 4: Fine-tune with smaller beta
    t_rem = deadline - time.time()
    if t_rem > 10:

        for _ in range(min(500, int(t_rem * 0.5))):
