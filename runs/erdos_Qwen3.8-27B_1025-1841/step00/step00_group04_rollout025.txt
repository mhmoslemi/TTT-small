import numpy as np
from scipy.optimize import minimize
import time

_opt_n = 42

def _objective_smooth(h):
    n = _opt_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    tau = 500.0
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

def _constraint(h):
    return np.sum(h) - _opt_n / 2.0

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    d = n * 0.5 - np.sum(h)
    if abs(d) > 1e-13:
        if d > 0:
            s = np.maximum(1.0 - h, 1e-8)
            h += d * s / np.sum(s)
        else:
            s = np.maximum(h, 1e-8)
            h += d * s / np.sum(s)
    return np.clip(h, 0.0, 1.0)

def _interp(h, n2):
    n1 = len(h)
    if n1 == n2:
        return h.copy()
    x1 = np.arange(n1, dtype=float) * (2.0 / n1)
    x2 = np.arange(n2, dtype=float) * (2.0 / n2)
    return _proj(np.interp(x2, x1, h), n2)

        return h0
