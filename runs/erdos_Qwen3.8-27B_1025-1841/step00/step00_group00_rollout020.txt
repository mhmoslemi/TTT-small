```python
def evaluate_and_grad(h, tau=50.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c_scaled = c * dx
    
    # Soft max
    max_c = np.max(c_scaled)
    exp_vals = np.exp(tau * (c_scaled - max_c))
    weights = exp_vals / np.sum(exp_vals)
    f_soft = max_c + np.log(np.sum(exp_vals)) / tau
    
    # Gradient
    grad = np.zeros(n)
    for lag_idx in range(len(c_scaled)):
        if weights[lag_idx] < 1e-10:
            continue
        lag = lag_idx - (n - 1)
        # Forward: (1-h[j-lag]) for valid j
        if lag >= 0:
            fwd = np.zeros(n)
            fwd[lag:] = 1.0 - h[:n-lag]
            bwd = np.zeros(n)
            bwd[:n-lag] = -h[lag:]
        else:
            k = -lag
            fwd = np.zeros(n)
            fwd[:n-k] = 1.0 - h[k:]
            bwd = np.zeros(n)
            bwd[k:] = -h[:n-k]
        grad += weights[lag_idx] * (fwd + bwd)
    
    grad *= dx
    return f_soft, max_c, grad
```

```python
import copy

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def soft_max_eval_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c_scaled = c * dx
    max_c = float(np.max(c_scaled))
    exp_vals = np.exp(tau * (c_scaled - max_c))
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    f_soft = max_c + np.log(sum_exp) / tau
    grad = np.zeros(n)
    for lag_idx in range(len(c_scaled)):
        w = weights[lag_idx]
        if w < 1e-10:
            continue
        lag = lag_idx - (n - 1)
        if lag >= 0:
            fwd = np.zeros(n)
            if n - lag > 0:
                fwd[lag:] = 1.0 - h[:n - lag]
            bwd = np.zeros(n)
            if n - lag > 0:
                bwd[:n - lag] = -h[lag:]
        else:
            k = -lag
            fwd = np.zeros(n)
            if n - k > 0:
                fwd[:n - k] = 1.0 - h[k:]
            bwd = np.zeros(n)
            if n - k > 0:
                bwd[k:] = -h[:n - k]
        grad += w * (fwd + bwd)
    grad *= dx
    return f_soft, max_c, grad

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_to_feasible(h_new, n_new / 2.0)
    return h_new

    n_res = len(resolutions)

        time_remaining = time_limit - (time.time() - t_start)

        time_budget = max(time_budget, 5.0)
        
        if n == base_n:
            h = best_h.copy()
        else:
            h = upsample(best_h, n)
        
        c5_cur = evaluate(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        target_sum = n / 2.0
        alpha = 0.05
        tau = 30.0
        
        t_phase_start = time.time()
        max_iter_phase = int(time_budget * 0.7)
        
        for it in range(max_iter_phase):
            if time.time() - t_start > time_limit:
                break
            if it % 50 == 0 and it > 0:
                elapsed = time.time() - t_phase_start
                if elapsed > time_budget * 0.7:
                    break
                tau = max(5.0, tau * 0.7)
                alpha = max(0.01, alpha * 0.85)
            
            f_soft, c5_now, grad = soft_max_eval_and_grad(h, tau)
            c5_now = max(c5_now, evaluate(h))
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            
            grad_centered = grad - np.mean(grad)
            h_new = h - alpha * grad_centered
            h_new = project_to_feasible(h_new, target_sum)
            
            if np.sum(np.abs(h_new - h)) < 1e-14:
                break
            h = h_new
        
        # Random restarts / perturbations
        t_phase2_start = time.time()
        max_phase2 = int(time_budget * 0.3)

            if time.time() - t_start > time_limit:
                break
            if time.time() - t_phase2_start > time_budget * 0.3:
                break
            
            perturbation_scale = 0.1 * (1.0 - it / max(1, max_phase2))
            h_pert = h + np.random.randn(n) * perturbation_scale
            h_pert = project_to_feasible(h_pert, target_sum)
            
            c5_pert = evaluate(h_pert)
            if c5_pert < c5_cur:

                c5_cur = c5_pert

            # Short gradient refinement from perturbed point

                h_try = h - 0.03 * grad_c
                h_try = project_to_feasible(h_try, target_sum)
                c5_try = evaluate(h_try)
                if c5_try <= c5_cur:
                    h = h_try
                    c5_cur = c5_try
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h.copy()
                else:
                    break
    
    # Final check
    final_c5 = evaluate(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
```
