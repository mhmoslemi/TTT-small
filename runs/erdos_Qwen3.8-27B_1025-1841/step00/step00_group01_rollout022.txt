        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': scon}],
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        return project_to_feasible(res.x, n)
    except Exception:
        return h0

def local_refine(h0, n, rng, n_iter, time_budget=60):
    h = h0.copy()
    c5 = evaluate_c5(h)
    step = 0.08
    for _ in range(n_iter):
        if time_left_global < time_budget:
            break
        h_p = h + rng.normal(0, step, n)
        h_p = project_to_feasible(h_p, n)
        c5_p = evaluate_c5(h_p)
        if c5_p < c5:
            h = h_p
            c5 = c5_p
            step = min(step * 1.05, 0.25)
        else:
            step = max(step * 0.97, 0.02)
    return h, c5

time_left_global = 1300.0

def run(seed=42, budget_s=1300, **kwargs):
    global time_left_global
    start_time = time.time()
    time_left_global = budget_s
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    best_h = None
    best_c5 = float('inf')

    n = 48
    half_n = n // 2

    candidates = []

    if 'initial_h_values' in globals():
        try:
            hp = np.array(initial_h_values, dtype=float)
            if len(hp) >= 2:
                x_old = np.linspace(0, 1, len(hp))
                x_new = np.linspace(0, 1, n)
                candidates.append(np.interp(x_new, x_old, hp))
        except (NameError, Exception):
            pass

    tm = thue_morse(n)
    candidates.append(adjust_to_half(tm, n, rng))

    candidates.append(np.tile([1.0, 0.0], half_n).copy())

    # Thue-Morse variant: shifted
    tm2 = np.roll(thue_morse(n), n // 4)
    candidates.append(adjust_to_half(tm2, n, rng))

    # More structured patterns
    h_1100 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
    candidates.append(adjust_to_half(h_1100, n, rng))

    h_111000 = np.tile([1.0, 1.0, 1.0, 0.0, 0.0, 0.0], n // 6)
    if len(h_111000) < n:
        h_111000 = np.concatenate([h_111000, np.zeros(n - len(h_111000))])
    candidates.append(adjust_to_half(h_111000, n, rng))

    for _ in range(8):
        h_r = np.zeros(n)
        h_r[rng.choice(n, half_n, replace=False)] = 1.0
        candidates.append(h_r)

    # Evaluate all initial candidates
    for c in candidates:
        h_bin = adjust_to_half(c, n, rng)
        c5 = evaluate_c5(h_bin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()

    # Phase 2: Binary simulated annealing
    if time_left() > 180:
        time_left_global = time_left()
        h_sa, c5_sa = binary_sa(best_h, n, rng, None, time_budget=120)
        time_left_global = budget_s
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: SLSQP continuous refinement
    if time_left() > 80 and best_h is not None:
        h0 = best_h.astype(float)
        h_ref = slsqp_refine(h0, n, rng, None, max_iter=500)
        c5_ref = evaluate_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # Phase 4: Local search
    if time_left() > 30 and best_h is not None:
        h_loc, c5_loc = local_refine(best_h, n, rng, 3000, time_budget=10)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc

    # Phase 5: Try another SA round from different start
    if time_left() > 60:
        time_left_global = time_left()
        h_r2 = np.zeros(n)
        h_r2[rng.choice(n, half_n, replace=False)] = 1.0
        h_sa2, c5_sa2 = binary_sa(h_r2, n, rng, None, time_budget=40)
        time_left_global = budget_s
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        if time_left() > 30 and best_h is not None:
            h_ref2 = slsqp_refine(best_h.astype(float), n, rng, None, max_iter=300)
            c5_ref2 = evaluate_c5(h_ref2)
            if c5_ref2 < best_c5:
                best_c5 = c5_ref2
                best_h = h_ref2

    # Phase 6: Final local polish
    if time_left() > 10 and best_h is not None:
        h_final, c5_final = local_refine(best_h, n, rng, 2000, time_budget=5)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    if best_h is None:
        best_h = np.zeros(n)
        best_h[:half_n] = 1.0
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n
