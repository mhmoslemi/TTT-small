import numpy as np
import time

def evaluate_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Positive lags: indices n-1 to 2n-2
    vals = corr[n-1:] * dx
    return np.max(vals)

def fix_sum(h, target_sum):
    """Adjust h to have the correct sum while maintaining [0,1] bounds."""
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        # Need to increase sum
        candidates = np.where(h < 1.0)[0]
    else:
        candidates = np.where(h > 0.0)[0]
    if len(candidates) == 0:
        return h
    # Distribute the adjustment
    per_elem = diff / len(candidates)
    for i in candidates:
        if diff > 0:
            h[i] = min(1.0, h[i] + per_elem)
        else:
            h[i] = max(0.0, h[i] + per_elem)
    # If still not exact, do it iteratively
    for _ in range(10):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            candidates = np.where(h < 1.0 - 1e-15)[0]
        else:
            candidates = np.where(h > 1e-15)[0]
        if len(candidates) == 0:
            break
        idx = candidates[0]
        adj = np.sign(diff) * min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
        h[idx] += adj
    return h

def binary_local_search(h, dx, n, time_budget, rng):

    best_c5 = current_c5

    # Temperature schedule
    T = current_c5 * 0.1
    T_min = 1e-6
    cooling = 0.999

        j = rng.integers(0, n)

                best = current.copy()
        else:
            current[i], current[j] = current[j], current[i]

def solve_1d_convex(a, b, n):
    """Find t in [0,1] minimizing max_m (a[m]*t + b[m]).
    a, b are arrays of length n.
    Uses the walking algorithm on the upper envelope."""
    # Find the line with max b (active at t=0)
    # Actually at t=0, g(0) = max(b)
    # The active lines are those with b[m] = max(b)
    
    # Walk algorithm:
    # Start at t=0, find the line with the largest value
    # Then find the next intersection point
    # Continue until slope >= 0 or t >= 1
    
    t = 0.0
    # Find initial active line
    vals = b.copy()
    active = np.argmax(vals)
    
    for _ in range(n + 2):
        if t >= 1.0:
            break
        slope = a[active]
        if slope >= 0:
            break  # At minimum or t=0 is optimal
        # Find next intersection: a[active]*t + b[active] = a[m]*t + b[m]
        # t = (b[m] - b[active]) / (a[active] - a[m])
        diffs = b - b[active]
        adiffs = a[active] - a
        # Next intersection is at t' = diffs[m] / adiffs[m] for the m that gives smallest valid t' > t
        valid = (adiffs != 0) & (diffs / adiffs > t) & (diffs / adiffs <= 1.0)
        if not valid.any():
            t = 1.0
            break
        ratios = np.where(valid, diffs / adiffs, np.inf)
        next_t = ratios.min()
        # Find which line becomes active at next_t
        vals_at_next = a * next_t + b
        active = np.argmax(vals_at_next)
        t = next_t
    
    # Clamp to [0,1]
    t = np.clip(t, 0.0, 1.0)
    return t

def make_valid(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Distribute correction
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h
    per = diff / cnt
    h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    # Iterative fix for any remaining error
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idxs = np.where(h < 1.0 - 1e-15)[0]
        else:
            idxs = np.where(h > 1e-15)[0]
        if len(idxs) == 0:
            break
        for i in idxs:
            room = (1.0 - h[i]) if diff > 0 else h[i]
            adj = np.sign(diff) * min(abs(diff), room)
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-12:
                break
    return h

    a = np.zeros(n)
