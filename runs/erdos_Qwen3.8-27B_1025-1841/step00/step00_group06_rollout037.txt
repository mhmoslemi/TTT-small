This is indeed a 1D convex optimization (minimize max of linear functions).

The minimum value is the smallest v such that there exists t ∈ [0,1] with C_j + E_j + (D_j - E_j)*t ≤ v for all j.

For each j, the constraint is:
(D_j - E_j)*t ≤ v - C_j - E_j

If D_j - E_j > 0: t ≤ (v - C_j - E_j) / (D_j - E_j)
If D_j - E_j < 0: t ≥ (v - C_j - E_j) / (D_j - E_j)
If D_j - E_j = 0: need v ≥ C_j + E_j

The feasible t must be in [0,1] and satisfy all these constraints. So:
- lower bound: max(0, max over j with D_j-E_j < 0 of (v - C_j - E_j)/(D_j - E_j))
- upper bound: min(1, min over j with D_j-E_j > 0 of (v - C_j - E_j)/(D_j - E_j))
- Feasible if lower ≤ upper and v ≥ max over j with D_j-E_j=0 of (C_j + E_j)

We can binary search on v to find the minimum feasible v.

This gives us an exact coordinate descent method! Let me implement this.

For each coordinate m:
1. Compute C_j + E_j and D_j - E_j for all j (189 values)
2. Binary search on v to find min feasible v
3. Set h[m] to the feasible t (e.g., the midpoint of the feasible interval)

Actually, for step 3, we should set h[m] to a value that minimizes the max, which could be at the boundary of the feasible interval or in the interior. For simplicity, we can set it to the midpoint of the feasible interval, or better, to the value that equalizes the most constraining terms.

            hi = mid

    lower = 0.0
    upper = 1.0
    for j in range(len(slopes)):

        elif slopes[j] < -1e-12:

    t_new = max(0, min(1, t_new))
    
    h_new = h.copy()
    h_new[m] = t_new
    return h_new, v_min * dx

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def optimize_coordinate(h, n, m):
    """Find optimal h[m] to minimize max correlation (ignoring sum constraint)."""
    k_shifts = np.arange(-(n-1), n)
    corr = np.correlate(h, 1 - h, mode='full')
    
    idx_mk = m + k_shifts
    idx_mkmk = m - k_shifts
    valid_mk = (idx_mk >= 0) & (idx_mk < n)
    valid_mkmk = (idx_mkmk >= 0) & (idx_mkmk < n)
    
    D = np.zeros(2*n-1)
    E = np.zeros(2*n-1)
    D[valid_mk] = 1.0 - h[idx_mk[valid_mk]]
    E[valid_mkmk] = h[idx_mkmk[valid_mkmk]]
    
    slopes = D - E
    intercepts = corr - slopes * h[m]
    
    # Binary search for min v
    lo_vals = np.where(slopes > 0, intercepts, intercepts + slopes)
    hi_vals = np.where(slopes > 0, intercepts + slopes, intercepts)
    lo = lo_vals.min()
    hi = hi_vals.max()
    
    if lo >= hi - 1e-15:
        return float(h[m])
    
    for _ in range(50):
        mid = (lo + hi) / 2
        pos = slopes > 1e-14
        neg = slopes < -1e-14
        zro = ~(pos | neg)
        
        upper = 1.0
        lower = 0.0
        
        if pos.any():
            ub = (mid - intercepts[pos]) / slopes[pos]
            upper = min(upper, float(ub.min()))
        
        if neg.any():
            lb = (mid - intercepts[neg]) / slopes[neg]
            lower = max(lower, float(lb.max()))

            lo = mid
            continue
        
        if lower > upper + 1e-14:
            lo = mid
        else:
            hi = mid

    pos = slopes > 1e-14
    neg = slopes < -1e-14
    upper = 1.0
    lower = 0.0
    if pos.any():

    if neg.any():
        lower = max(lower, float(((v_opt - intercepts[neg]) / slopes[neg]).max()))
    
    t = (max(0.0, lower) + min(1.0, upper)) / 2.0
    return max(0.0, min(1.0, t))

def coordinate_descent(h, n, n_passes=200, time_limit=None, start_time=None):
    """Run coordinate descent with projection."""
    target = n / 2.0
    best_h = h.copy()
    best_obj = compute_c5(h)
    h = h.copy()
    
    for p in range(n_passes):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        for m in range(n):
            new_val = optimize_coordinate(h, n, m)
            h[m] = new_val
        
        # Project to feasible set
        h = project_feasible(h, n)
        
        obj = compute_c5(h)
        if obj < best_obj - 1e-7:
            best_obj = obj
            best_h = h.copy()
    
    return best_h, best_obj

    rng = np.random.RandomState(seed)
    h = rng.random(n)
    return project_feasible(h, n)

    x = np.arange(n)
    h = 0.5 + amplitude * np.cos(2 * np.pi * freq * x / n)
    return project_feasible(h, n)

- If i = m: term changes from h[m]*(1-h[m+k]) to (h[m]+delta)*(1-h[m+k]), change = delta*(1-h[m+k])

def _find_optimal_t(slopes, intercepts):
    """Find t in [0,1] minimizing max_j (intercept_j + slope_j * t)."""
    pos = slopes > 1e-14
    neg = slopes < -1e-14
    zro = ~(pos | neg)
    
    # Feasible v range
    lo = 0.0
    hi = 0.0
    for j in range(len(slopes)):
        lo = max(lo, min(intercepts[j], intercepts[j] + slopes[j]))
        hi = max(hi, max(intercepts[j], intercepts[j] + slopes[j]))
    
    # Binary search
    for _ in range(50):
        mid = (lo + hi) / 2
        
        # Check if exists t in [0,1] with all intercept_j + slope_j * t <= mid
        upper = 1.0
        lower = 0.0
        
        if pos.any():
            ub = (mid - intercepts[pos]) / slopes[pos]
            upper = min(upper, float(np.min(ub)))

            lb = (mid - intercepts[neg]) / slopes[neg]

                continue

    lower = 0.0

        upper = min(upper, float(np.min((v_opt - intercepts[pos]) / slopes[pos])))

    t = (max(0.0, lower) + min(1.0, upper)) / 2.0

    h = 0.5 + amplitude * np.cos(2.0 * np.pi * freq * x / n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=float)
        n_parent = len(parent)
    except NameError:
        parent = None
        n_parent = 95

    if parent is not None:
        parent = project_feasible(parent, n_parent)

    best_h = None
    best_obj = 1.0
    best_n = 0

    if parent is not None:
        obj = compute_c5(parent)
        if obj < best_obj:
            best_obj = obj
            best_h = parent.copy()
            best_n = n_parent

    # Phase 1: optimize parent (40% budget)
    t1 = start_time + budget_s * 0.40
    if parent is not None and time.time() < t1:
        rem = t1 - time.time()
        h, obj = coord_descent(parent, n_parent, max_passes=5000,
                               time_limit=rem, start_time=start_time, rng=rng)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
            best_n = n_parent

    # Phase 2: try other resolutions and inits (30% budget)
    t2 = start_time + budget_s * 0.70
    n_options = sorted(set([n_parent, n_parent + 10, n_parent + 20, max(100, n_parent * 2)]))
    if n_parent != 95:
        n_options.append(95)
        n_options = sorted(set(n_options))

    for n in n_options:
        if time.time() > t2:
            break
        rem_phase = t2 - time.time()
        if rem_phase < 10:
            break
        per_attempt = min(40, rem_phase / 4.0)

        # Upsampled parent
        if parent is not None:
            if n != n_parent:
                h_init = upsample(parent, n_parent, n)
            else:
                h_init = parent.copy()
            if time.time() < t2:
                h, obj = coord_descent(h_init, n, max_passes=1000,
                                       time_limit=min(per_attempt, t2 - time.time()),
                                       start_time=start_time, rng=rng)
                if obj < best_obj:
                    best_obj = obj
                    best_h = h.copy()
                    best_n = n

        # Random init
        if time.time() < t2:
            h_init = random_init(n, seed + n)
            h, obj = coord_descent(h_init, n, max_passes=500,
                                   time_limit=min(20, t2 - time.time()),
                                   start_time=start_time, rng=rng)
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
                best_n = n

        # Periodic inits
        for freq in [3, 5, 7]:
            if time.time() > t2:
                break
            h_init = periodic_init(n, freq, 0.4)
            h, obj = coord_descent(h_init, n, max_passes=300,
                                   time_limit=min(15, t2 - time.time()),
                                   start_time=start_time, rng=rng)
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
                best_n = n

    # Phase 3: final refinement (25% budget)
    t3 = start_time + budget_s * 0.95
    if best_h is not None and time.time() < t3:
        rem = t3 - time.time()
        h, obj = coord_descent(best_h, best_n, max_passes=10000,
                               time_limit=rem, start_time=start_time, rng=rng)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
            best_n = best_n

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n)
    final_obj = compute_c5(best_h)

    return best_h, final_obj, best_n
