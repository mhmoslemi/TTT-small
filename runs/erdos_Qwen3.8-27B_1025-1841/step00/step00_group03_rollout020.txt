import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1-h, mode="full") * dx))

def make_valid(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(np.sum(h))
    if s > 1e-12:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - float(np.sum(h))
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, dx, n, rng, max_time, best_c5_ref, best_h_ref):
    """Local search with targeted and random moves. Returns (h, c5, best_h, best_c5)."""
    t_end = time.time() + max_time
    curr_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = curr_c5
    temp = 0.01
    min_temp = 1e-7
    no_improve = 0
    it = 0

    while time.time() < t_end:
        it += 1
        r = rng.random()

        if r < 0.35:
            # Targeted: reduce bottleneck correlation
            corr = np.correlate(h, 1-h, mode="full") * dx
            k_star = int(np.argmax(corr))
            s = k_star - (n - 1)
            best_contrib = -1.0
            best_i = best_j = -1
            for i in range(n):
                j = i + s
                if 0 <= j < n:
                    c = h[i] * (1.0 - h[j])
                    if c > best_contrib:
                        best_contrib = c
                        best_i, best_j = i, j
            if best_i >= 0 and best_contrib > 1e-8:
                max_t = min(h[best_i], 1.0 - h[best_j])
                if max_t > 1e-9:
                    improved = False
                    for frac in (0.5, 0.25, 0.1, 0.05):
                        t = max_t * frac
                        h[best_i] -= t
                        h[best_j] += t
                        new_c5 = compute_c5(h, dx)
                        if new_c5 <= curr_c5 + 1e-14:
                            curr_c5 = new_c5
                            improved = True
                            break
                        h[best_i] += t
                        h[best_j] -= t
                    if improved:
                        no_improve = 0
                        temp = max(temp * 0.999, min_temp)
                        if curr_c5 < best_c5:
                            best_c5 = curr_c5
                            best_h = h.copy()
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1

        elif r < 0.75:
            # Random pairwise transfer
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            step = rng.uniform(0.001, 0.06)
            d = step if rng.random() < 0.5 else -step
            nh_i = h[i] + d
            nh_j = h[j] - d
            if nh_i < 0 or nh_i > 1 or nh_j < 0 or nh_j > 1:
                continue
            h[i] = nh_i
            h[j] = nh_j
            new_c5 = compute_c5(h, dx)
            delta_obj = new_c5 - curr_c5
            if delta_obj <= 0:
                curr_c5 = new_c5
                no_improve = 0

                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                if delta_obj < temp and rng.random() < np.exp(-delta_obj / temp):
                    curr_c5 = new_c5
                else:
                    h[i] -= d
                    h[j] += d
                no_improve += 1
        else:
            # Multi-perturb
            np_ = int(rng.randint(2, 6))
            idx = rng.choice(n, size=np_, replace=False)
            old = h[idx].copy()
            h[idx] += rng.normal(0, 0.06, np_)
            h = make_valid(h, n)
            new_c5 = compute_c5(h, dx)
            if new_c5 <= curr_c5 + 1e-14:
                curr_c5 = new_c5

                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:

            h = make_valid(h, n)
            curr_c5 = compute_c5(h, dx)
            no_improve = 0
            temp = 0.01

    return h, curr_c5, best_h, best_c5
