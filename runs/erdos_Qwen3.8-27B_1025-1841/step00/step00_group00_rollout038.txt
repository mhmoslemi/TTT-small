import numpy as np
from time import time

def eval_score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, 'full')
    return float(c.max() * (2.0 / n))

def upsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    old_t = (np.arange(n) + 0.5) / n
    new_t = (np.arange(new_n) + 0.5) / new_n
    out = np.interp(new_t, old_t, h)
    out += (new_n * 0.5 - out.sum()) / new_n
    np.clip(out, 0.0, 1.0, out=out)
    return out

def fix_sum(h, n):
    target = n * 0.5
    h += (target - h.sum()) / n
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def try_binarize(h, n):
    if n % 2 != 0:
        return None
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[n // 2:]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time()

    h_parent = initial_h_values.copy()
    best_h = h_parent.copy()
    best_score = eval_score(h_parent)

    schedule = [
        (90, 0.00, 0.03),
        (180, 0.03, 0.13),
        (360, 0.13, 0.42),
        (720, 0.42, 0.78),
        (1440, 0.78, 1.00),
    ]

    for target_n, f_start, f_end in schedule:
        t_end = t0 + budget_s * f_end - 3.0
        t_begin = max(t0 + budget_s * f_start, time())
        if time() >= t_end:
            break
        phase_dur = t_end - t_begin
        if phase_dur < 5.0:
            break

        cands = []
        cands.append(upsample(best_h, target_n))
        if len(h_parent) != target_n:
            cands.append(upsample(h_parent, target_n))

        if target_n % 2 == 0:
            hb = np.zeros(target_n)
            hb[rng.choice(target_n, target_n // 2, replace=False)] = 1.0
            cands.append(hb)

        hs = np.ones(target_n)
        hs[1::2] = 0.0
        cands.append(fix_sum(hs.copy(), target_n))

        hg = np.zeros(target_n)
        bit = 1
        for k in range(target_n):
            hg[k] = 1.0 if (k & bit) else 0.0
            if (k & (k + 1)) == 0:
                bit += 1
        cands.append(fix_sum(hg.copy(), target_n))

        # Spatial-frequency candidate: cosine
        tc = (np.arange(target_n) + 0.5) / target_n
        hc = 0.5 + 0.4 * np.cos(2.0 * np.pi * tc)
        cands.append(fix_sum(hc.copy(), target_n))

        best_init = None
        best_init_sc = 1e9
        for c in cands:
            c = fix_sum(c.copy(), target_n)
            sc = eval_score(c)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = c

        n = target_n
        h = best_init.copy()
        cur = best_init_sc
        loc_best = h.copy()
        loc_sc = cur

        T0 = 0.005
        Tbot = 5e-9
        s0 = 0.05
        sbot = 0.0001
        last_improve = time()
        staleness = 25.0

        while time() < t_end:
            elapsed = time() - t_begin
            frac = min(1.0, elapsed / phase_dur)
            T = T0 * (Tbot / T0) ** frac
            step = s0 * (sbot / s0) ** frac

            if time() - last_improve > staleness:
                T = max(T, cur * 0.003)
                step = max(step, 0.03)
                staleness = max(5.0, staleness * 0.85)
                last_improve = time()

            i = rng.randint(n)
            j = rng.randint(n)
            if j == i:
                j = (j + 1) % n

            delta = (rng.random() * 2.0 - 1.0) * step
            ni = h[i] + delta
            nj = h[j] - delta

            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue

            h[i] = ni
            h[j] = nj
            ns = eval_score(h)

            d = ns - cur
            if d < 0.0 or rng.random() < np.exp(-d / T):
                cur = ns
                if ns < loc_sc:
                    loc_sc = ns
                    loc_best = h.copy()
                    last_improve = time()
            else:
                h[i] = ni - delta
                h[j] = nj + delta

        # Try binarization
        h_bin = try_binarize(loc_best, n)
        if h_bin is not None:
            sc_bin = eval_score(h_bin)
            if sc_bin < loc_sc:
                loc_sc = sc_bin
                loc_best = h_bin

        if loc_sc < best_score:
            best_score = loc_sc
            best_h = loc_best

    return (best_h, eval_score(best_h), len(best_h))
