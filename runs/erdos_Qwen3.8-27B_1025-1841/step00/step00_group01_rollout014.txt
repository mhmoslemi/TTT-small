import numpy as np
import time

def project_to_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    nf = np.sum(free)
    if nf == 0:
        return h
    h[free] += diff / nf
    h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-10:
        free2 = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf2 = np.sum(free2)
        if nf2 > 0:
            h[free2] += diff2 / nf2
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def optimize_projected(h0, n, t_start, time_limit, alpha_init=0.03):
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    c5, _ = compute_c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    alpha = alpha_init
    vel = np.zeros(n)
    momentum = 0.9
    stall = 0

    while time.time() - t_start < time_limit:
        c5, corr = compute_c5(h, n)
        m_star = int(np.argmax(corr))
        shift = m_star - (n - 1)

        grad = np.ones(n)
        jp = shift
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
        else:
            grad[-shift:] -= h[:n + shift]
        jm = -shift
        if jm >= 0:
            grad[:n - jm] -= h[jm:]
        else:
            grad[-jm:] -= h[:n + jm]
        grad *= dx

        vel = momentum * vel - alpha * grad
        h_new = project_to_feasible(h + vel, target_sum)
        c5_new, _ = compute_c5(h_new, n)

        if c5_new <= c5 + 1e-14:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            alpha = min(alpha * 1.01, 0.15)
            if stall > 500:
                vel *= 0.5
                alpha *= 0.8
                stall = 0
        else:
            alpha *= 0.6
            vel *= 0.5
            if alpha < 1e-9:
                break
    return best_h, best_c5

def local_search(h0, n, t_start, time_limit):

        if c5_try < best_c5:

            best_h = h_try.copy()
            h = h_try.copy()
    return best_h, best_c5
