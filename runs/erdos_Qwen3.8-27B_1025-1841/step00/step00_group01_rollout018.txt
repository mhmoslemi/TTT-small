Let me also consider: for the binary case, the problem has a nice structure. The correlation at lag k for a binary sequence is:
corr[k] = #{i : h[i+k]=1, h[i]=0}

And corr[-k] = #{i : h[i]=1, h[i+k]=0} = #{i : h[i+k]=1, h[i]=0}... wait, no.

For binary: this is the count of i where h[i+k]=1 and h[i]=0.

For binary: this is the count of i where h[i]=1 and h[i+k]=0.

Let me just denote: for binary h, in the overlap of h[0:n-k] and h[k:n]:
- a = #{both 1}
- b = #{h[i]=1, h[i+k]=0} = corr[-k]
- c = #{h[i]=0, h[i+k]=1} = corr[k]  
- d = #{both 0}

a+b+c+d = n-k
a+b = #{1s in h[0:n-k]} = S1
a+c = #{1s in h[k:n]} = S2
b = S1 - a
c = S2 - a
c - b = S2 - S1

So the max(corr[k], corr[-k]) = max(S2-a, S1-a) and the min = min(S2-a, S1-a).
And corr[k] + corr[-k] = S1 + S2 - 2a.

To minimize the max, we want a to be as large as possible (maximize overlap of 1s) AND |S1-S2| as small as possible.

                break

    best_h = np.clip(best_h, 0, 1)
    return best_h, eval_c5(best_h, n)

def continuous_opt(h, n, max_iter=300):
    dx = 2.0 / n
    target = n / 2.0
    beta = 80.0

    def obj_fun(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(c)
        val = m + np.log(np.sum(np.exp(beta * (c - m)))) / beta
        val += 500.0 * (np.sum(x) - target) ** 2
        return val

    def obj_jac(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(c)
        w = np.exp(beta * (c - m))
        sw = np.sum(w)
        g = np.zeros(n)
        for k in range(len(c)):
            lag = k - (n - 1)
            if lag >= 0:
                lo = 0
                hi = n - lag
                g[lo+lag:hi+lag] += w[k] / sw * dx * (1.0 - x[lo:hi])
                g[lo:hi] -= w[k] / sw * dx * x[lo+lag:hi+lag]
            else:
                ll = -lag
                lo = 0
                hi = n - ll
                g[lo:hi] += w[k] / sw * dx * (1.0 - x[lo+ll:hi+ll])
                g[lo+ll:hi+ll] -= w[k] / sw * dx * x[lo:hi]
        g += 1000.0 * (np.sum(x) - target)
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj_fun, h, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_opt = h.copy()

    diff = target - np.sum(h_opt)
    if abs(diff) > 1e-10:
        order = np.argsort(h_opt) if diff > 0 else np.argsort(h_opt)[::-1]
        for i in order:
            if diff > 0:
                step = min(1.0 - h_opt[i], diff)
                h_opt[i] += step
                diff -= step
            else:
                step = min(h_opt[i], -diff)
                h_opt[i] -= step
                diff += step
            if abs(diff) < 1e-12:
                break
    h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, eval_c5(h_opt, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def update(h, n):
        nonlocal best_h, best_c5, best_n
        h = np.clip(h, 0.0, 1.0)
        c5 = eval_c5(h, n)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n
        return c5

    # Parent
    parent_h = None
    parent_n = 0
    try:
        parent_h = initial_h_values.astype(np.float64).copy()
        parent_n = len(parent_h)
        if parent_n % 2 == 0:
            parent_h = make_binary_balanced(parent_h, parent_n, rng)
        update(parent_h, parent_n)
    except Exception:
        pass

    n_options = [48, 64, 96, 128, 192, 256]

    # Phase 1: Evaluate structured inits quickly
    inits = []
    for n in n_options:
        tm = make_binary_balanced(thue_morse(n), n, rng)
        inits.append((tm, n))
        hr = np.zeros(n)
        hr[rng.choice(n, size=n // 2, replace=False)] = 1.0
        inits.append((hr, n))
    if parent_h is not None:
        for n in [48, 64, 96, 128]:
            if n == parent_n:
                continue
            x_old = np.linspace(0, 2, parent_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            hr = np.interp(x_new, x_old, parent_h, period=2.0)
            hr = make_binary_balanced(hr, n, rng)
            inits.append((hr, n))

    for h_i, n_i in inits:
        update(h_i, n_i)

    # Phase 2: Local search on best init per n value
    best_per_n = {}
    for h_i, n_i in inits:
        c5_i = eval_c5(h_i, n_i)
        if n_i not in best_per_n or c5_i < best_per_n[n_i][1]:
            best_per_n[n_i] = (h_i, c5_i)
    if parent_h is not None:
        pn = parent_n
        pc5 = eval_c5(parent_h, pn)
        if pn not in best_per_n or pc5 < best_per_n[pn][1]:
            best_per_n[pn] = (parent_h, pc5)

    ls_time_each = budget_s * 0.25 / max(len(n_options), 1)
    for n in sorted(best_per_n.keys(), key=lambda x: best_per_n[x][1]):
        if time.time() - t0 > budget_s * 0.5:
            break
        tl = min(ls_time_each, budget_s * 0.5 - (time.time() - t0))
        if tl < 3:
            break
        h_init = best_per_n[n][0]
        h_ls, c5_ls = local_search(h_init, n, tl, rng)
        update(h_ls, n)

    # Phase 3: Multi-start on most promising n
    if time.time() - t0 < budget_s * 0.8:
        n_total = budget_s * 0.8 - (time.time() - t0)
        n_starts = max(2, int(n_total / 25))
        n_starts = min(n_starts, 12)
        per_start = n_total / n_starts

        for s in range(n_starts):
            if time.time() - t0 > budget_s * 0.8:
                break
            tl = min(per_start * 0.85, budget_s * 0.8 - (time.time() - t0))
            if tl < 3:
                break
            if best_h is not None:
                h_init = best_h.copy()
                np_ = best_n
                nf = max(3, np_ // 12)
                flips = rng.choice(np_, size=nf, replace=False)
                h_init[flips] = 1.0 - h_init[flips]
                h_init = make_binary_balanced(h_init, np_, rng)
            else:
                np_ = 64
                h_init = np.zeros(np_)
                h_init[rng.choice(np_, size=np_ // 2, replace=False)] = 1.0
            h_ls, c5_ls = local_search(h_init, np_, tl, rng, anneal=(s % 2 == 1))
            update(h_ls, np_)

    # Phase 4: Continuous refinement
    if best_h is not None and time.time() - t0 < budget_s * 0.92:
        tl = budget_s * 0.92 - (time.time() - t0)
        if tl > 5:
            h_c, c5_c = continuous_opt(best_h, best_n, max_iter=min(200, int(tl * 5)))
            update(h_c, best_n)

    if parent_h is not None and time.time() - t0 < budget_s * 0.96:
        tl = budget_s * 0.96 - (time.time() - t0)
        if tl > 5:
            h_c2, c5_c2 = continuous_opt(parent_h, parent_n, max_iter=min(150, int(tl * 5)))
            update(h_c2, parent_n)

    # Final: one more round of local search from best
    if best_h is not None and time.time() - t0 < budget_s * 0.98:
        tl = budget_s * 0.98 - (time.time() - t0)
        if tl > 5:
            h_ls, c5_ls = local_search(best_h, best_n, tl, rng)
            update(h_ls, best_n)

    # Enforce exact constraint
    final_h = best_h.astype(np.float64).copy()
    final_n = best_n
    target = final_n / 2.0
    final_h = np.clip(final_h, 0.0, 1.0)
    diff = target - np.sum(final_h)
    if abs(diff) > 1e-12:
        order = np.argsort(final_h) if diff > 0 else np.argsort(final_h)[::-1]
        for i in order:
            if diff > 0:
                step = min(1.0 - final_h[i], diff)
                final_h[i] += step
                diff -= step
            else:
                step = min(final_h[i], -diff)
                final_h[i] -= step
                diff += step
            if abs(diff) < 1e-12:
                break
    final_h = np.clip(final_h, 0.0, 1.0)
    final_c5 = eval_c5(final_h, final_n)

    return final_h, final_c5, final_n
