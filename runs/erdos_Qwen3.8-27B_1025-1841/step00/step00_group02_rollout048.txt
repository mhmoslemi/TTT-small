    n = len(h)

    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), corr

def project_onto_simplex_with_bounds(g, target_sum):
    """Project g onto {0 <= h <= 1, sum(h) = target_sum}"""
    n = len(g)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2
        h = np.clip(g - lam, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(g - (lo + hi) / 2, 0, 1)
