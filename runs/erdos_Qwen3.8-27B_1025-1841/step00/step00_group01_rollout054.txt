                h0 = np.full(n_pts, 0.5)

            if si == 1:
                h0 = np.clip(h0 + rng.uniform(-0.12, 0.12, n_pts), 0, 1)
            elif si == 2:
                h0 = (interpolated if False else h0.copy())
                h0 = (h0 > 0.5).astype(float)
            elif si == 3:
                h0 = np.clip(h0 + rng.uniform(-0.25, 0.25, n_pts), 0, 1)

            h0 = fix_sum(h0, n_pts)

            # SLSQP
            h_opt = slsqp_optimize(h0, n_pts)
            h_opt = fix_sum(h_opt, n_pts)
            record(h_opt, n_pts)

            # Quick local search after each SLSQP
            if time.time() < t_opt_end:
                t_ls_end = min(t_opt_end, time.time() + 30)
                h_cur = h_opt.copy()
                h_cur, c5_ls = local_search(h_cur, n_pts, rng, 2000, t_ls_end)
                if c5_ls < best_c5:
                    record(h_cur, n_pts)

    # Extended local refinement on best
    if best_h is not None and time.time() < t_refine_end:
        h_fin = best_h.copy()
        n_fin = best_n
        h_fin, c5_fin = local_search(h_fin, n_fin, rng, 20000, t_refine_end)
        if c5_fin < best_c5:
            record(h_fin, n_fin)
        else:
            h_fin = best_h.copy()
            h_fin, c5_fin = local_search(h_fin, n_fin, rng, 10000, t_refine_end)
            if c5_fin < best_c5:
                record(h_fin, n_fin)

    # Final fine-tuning
    if best_h is not None and time.time() < t0 + budget_s * 0.95:
        h_fin = best_h.copy()
        n_fin = best_n
        cur_c5 = best_c5
        for _ in range(5000):
            if time.time() > t0 + budget_s * 0.96:
                break
            i = int(rng.integers(n_fin))
            j = int(rng.integers(n_fin))
            if i == j:
                continue
            for d in [0.005, -0.005, 0.002, -0.002]:
                hi = h_fin[i] + d
                hj = h_fin[j] - d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h_fin[i] = hi
                    h_fin[j] = hj
                    c5_tr = compute_c5(h_fin)
                    if c5_tr < cur_c5 - 1e-12:
                        cur_c5 = c5_tr
                        break
                    else:
                        h_fin[i] -= d
                        h_fin[j] += d
        if cur_c5 < best_c5:
            record(h_fin, n_fin)

    if best_h is None:
        best_h = np.full(48, 0.5)
        best_n = 48
        best_c5 = compute_c5(best_h)

    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), int(best_n)
