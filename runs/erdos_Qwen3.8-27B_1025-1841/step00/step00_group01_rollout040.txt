```python
import time

def compute_c5(h):
    """Compute C5 = max of full correlation of h with 1-h, times dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def compute_c5_gradient(h):
    """Compute C5 and its subgradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m_star = np.argmax(corr)
    s = m_star - (n - 1)  # shift
    
    # Compute gradient of g_{m_star}
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        # First part: (1-h[j-s]) if 0 <= j-s < n
        if 0 <= j - s < n:
            g += (1 - h[j - s])
        # Second part: -h[j+s] if 0 <= j+s < n
        if 0 <= j + s < n:
            g -= h[j + s]
        grad[j] = g * dx
    
    return corr[m_star], grad
```

```python
def smooth_obj_and_grad(h, alpha=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    corr_max = np.max(corr)
    shifted = alpha * (corr - corr_max)
    exps = np.exp(shifted - np.max(shifted))  # numerical stability
    weights = exps / np.sum(exps)
    val = corr_max + np.log(np.sum(exps)) / alpha
    
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-8:
            continue
        s = m - (n-1)
        i_min = max(0, s)
        i_max = min(n, n+s)
        if i_max > i_min:
            grad[i_min:i_max] += w * (1.0 - h[i_min-s:i_max-s])
        j_min = max(0, i_min - s)
        j_max = min(n, i_max - s)
        if j_max > j_min:
            grad[j_min:j_max] -= w * h[j_min+s:j_max+s]
    grad *= dx
    return val, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def c5_val_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m_star = int(np.argmax(corr))
    s = m_star - (n - 1)
    i_min = max(0, s)
    i_max = min(n, n + s)
    grad = np.zeros(n)
    if i_max > i_min:
        grad[i_min:i_max] += (1.0 - h[i_min - s:i_max - s])
    j_min = max(0, i_min - s)
    j_max = min(n, i_max - s)
    if j_max > j_min:
        grad[j_min:j_max] -= h[j_min + s:j_max + s]
    grad *= dx
    return float(corr[m_star]), grad

def smooth_obj_grad(h, alpha=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    wsum = np.sum(exps)
    weights = exps / wsum
    val = cmax + np.log(wsum) / alpha
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-10:
            continue
        s = m - (n - 1)
        i_min = max(0, s)
        i_max = min(n, n + s)
        if i_max > i_min:
            grad[i_min:i_max] += w * (1.0 - h[i_min - s:i_max - s])
        j_min = max(0, i_min - s)
        j_max = min(n, i_max - s)
        if j_max > j_min:
            grad[j_min:j_max] -= w * h[j_min + s:j_max + s]
    grad *= dx
    return val, grad

def project_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = 1.0 - g.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(g + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(g + (lo + hi) * 0.5, 0.0, 1.0)

def polish(h, n, max_steps=3000, time_limit=None, t0=None):
    best = h.copy()
    best_val = compute_c5(h)
    for step in range(max_steps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        val, grad = c5_val_and_grad(h)
        lr = 0.08 / (1.0 + 0.005 * step)
        g = h - lr * grad
        h_new = project_feasible(g, n)
        val_new = compute_c5(h_new)
        if val_new < best_val:
            best = h_new.copy()
            best_val = val_new
        if val_new < val:
            h = h_new
        else:
            lr *= 0.5
            g = h - lr * grad
            h_new = project_feasible(g, n)
            if compute_c5(h_new) <= val:
                h = h_new
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = np.ones(100) * 0.5
    
    best_h = None
    best_c5 = float('inf')
    
    def update_best(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    alpha = 80.0
    
    for n in [150, 250]:
        if time.time() - t0 > budget_s * 0.75:
            break
        
        x_old = np.linspace(0, 2, len(h_init))
        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_old, h_init)
        h = project_feasible(h, n)
        c5_val = compute_c5(h)
        update_best(h, c5_val)
        
        # SLSQP on smooth objective
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - n / 2.0,
                 'jac': lambda hv: np.ones(n)}]
        
        time_limit_1 = t0 + budget_s * 0.45
        try:
            res = minimize(smooth_obj_grad, h, args=(alpha,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12,
                                   'disp': False})
            if res.x is not None:
                h_res = project_feasible(res.x, n)
                update_best(h_res, compute_c5(h_res))
        except Exception:
            pass
        
        # Polish with subgradient
        time_limit_2 = min(t0 + budget_s * 0.65, time_limit_1 + budget_s * 0.2)
        h_pol, c5_pol = polish(h_res if 'h_res' in dir() else h, n,
                               max_steps=5000, time_limit=time_limit_2, t0=t0)
        update_best(h_pol, c5_pol)
        
        # Random restart
        if time.time() - t0 < budget_s * 0.7:
            h_rand = np.clip(np.random.randn(n) * 0.15 + 0.5, 0.05, 0.95)
            h_rand = project_feasible(h_rand, n)
            try:
                res2 = minimize(smooth_obj_grad, h_rand, args=(alpha,), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-12})
                if res2.x is not None:
                    h_r2 = project_feasible(res2.x, n)
                    update_best(h_r2, compute_c5(h_r2))
            except Exception:
                pass
        
        # Another restart: periodic pattern
        if time.time() - t0 < budget_s * 0.75:
            h_per = 0.5 + 0.4 * np.cos(2.0 * np.pi * x_new)
            h_per = project_feasible(h_per, n)
            h_per2, c5_per2 = polish(h_per, n, max_steps=3000,
                                     time_limit=t0 + budget_s * 0.8, t0=t0)
            update_best(h_per2, c5_per2)
    
    # Final polish at best resolution
    if best_h is not None:
        n_best = len(best_h)
        time_limit_final = t0 + budget_s * 0.95
        h_fin, c5_fin = polish(best_h, n_best, max_steps=10000,
                               time_limit=time_limit_final, t0=t0)
        update_best(h_fin, c5_fin)
    
    if best_h is None:
        best_h = h_init
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```
