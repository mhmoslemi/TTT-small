import numpy as np
import time

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def upsample_to_n(h, n):
    """Upsample h to length n using linear interpolation."""
    n_old = len(h)
    if n_old == n:
        return h.copy()
    # Create coordinate grids
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # Adjust to meet sum constraint
    target_sum = n / 2.0
    current_sum = h_new.sum()
    # Distribute the difference
    diff = target_sum - current_sum
    h_new += diff / n
    # Clip to [0,1] and redistribute
    h_new = np.clip(h_new, 0, 1)
    # Final adjustment
    diff = target_sum - h_new.sum()
    # Add to positions with most room
    if diff > 0:
        idx = np.argsort(h_new)  # lowest first
        for i in idx:
            add = min(diff, 1.0 - h_new[i])
            h_new[i] += add
            diff -= add
            if diff <= 1e-10:
                break
    else:
        idx = np.argsort(-h_new)  # highest first
        for i in idx:
            sub = min(-diff, h_new[i])
            h_new[i] -= sub
            diff += sub
            if diff >= -1e-10:
                break
    return h_new

def local_search_binary(h, n, time_limit, rng):
    """Local search on binary (or near-binary) h using flip moves."""
    h = h.copy()
    # Round to binary
    # Use a threshold approach
    h_binary = (h > 0.5).astype(float)
    # Adjust to meet sum constraint
    ones = np.sum(h_binary)
    target = n // 2
    if ones > target:
        # Need to flip some 1s to 0s
        # Flip those at positions where h is lowest
        one_positions = np.where(h_binary == 1)[0]
        order = np.argsort(h[one_positions])
        for idx in order[:int(ones - target)]:
            h_binary[one_positions[idx]] = 0
    elif ones < target:
        zero_positions = np.where(h_binary == 0)[0]
        order = np.argsort(-h[zero_positions])
        for idx in order[:int(target - ones)]:
            h_binary[zero_positions[idx]] = 1
    
    best_c5 = compute_c5(h_binary)
    start = time.time()
    
    # Precompute positions of 0s and 1s
    zeros = np.where(h_binary == 0)[0]
    ones_pos = np.where(h_binary == 1)[0]
    
    no_improve_count = 0
    step_temp = 1.0  # for simulated annealing
    
    while time.time() - start < time_limit:
        if len(zeros) == 0 or len(ones_pos) == 0:
            break
        
        # Pick random zero and one
        i = zeros[rng.integers(len(zeros))]
        j = ones_pos[rng.integers(len(ones_pos))]
        
        # Flip: h[i] 0->1, h[j] 1->0
        h_binary[i] = 1
        h_binary[j] = 0
        c5 = compute_c5(h_binary)
        
        if c5 <= best_c5:
            best_c5 = c5
            # Update sets
            zeros = np.delete(zeros, np.where(zeros == i))
            zeros = np.insert(zeros, np.where(zeros <= j)[0][-1] if len(zeros)>0 else 0, j)
            ones_pos = np.delete(ones_pos, np.where(ones_pos == j))
            ones_pos = np.insert(ones_pos, np.where(ones_pos <= i)[0][-1] if len(ones_pos)>0 else 0, i)
            no_improve_count = 0
        else:
            # Revert (with probability for simulated annealing)
            if rng.random() < np.exp(-(c5 - best_c5) / step_temp):
                # Accept anyway
                zeros = np.delete(zeros, np.where(zeros == i))
                zeros = np.insert(zeros, np.where(zeros <= j)[0][-1] if len(zeros)>0 else 0, j)

                best_c5 = c5  # update tracking
            else:
                h_binary[i] = 0
                h_binary[j] = 1
            no_improve_count += 1
        
        # Cool down
        if no_improve_count > 1000:
            step_temp *= 0.95

                zero_arr = np.append(zero_arr, j)
                one_arr = np.delete(one_arr, pj)
                one_arr = np.append(one_arr, i)
            if compute_c5(h) > best_c5 + 0.05:
                h = best_h.copy()
                zero_arr = np.where(h == 0)[0].copy()
                one_arr = np.where(h == 1)[0].copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(100) * 0.5
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: medium resolutions (n=180, 360) with multiple starts
    phase1_deadline = start_time + budget_s * 0.35
    for n in [180, 360]:
        if time.time() > phase1_deadline - 5:
            break
        t_avail = phase1_deadline - time.time() - 3
        if t_avail < 5:
            break
        starts = [upsample(h_init, n), make_thue_morse(n, seed), rng.random(n)]
        t_per = t_avail / len(starts)
        for s in starts:
            if time.time() > phase1_deadline - 2:
                break
            h_opt, c5 = sa_search(s, n, t_per, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    # Phase 2: large resolution n=720
    phase2_deadline = start_time + budget_s * 0.75
    if time.time() < phase2_deadline - 10:
        n2 = 720
        t_avail = phase2_deadline - time.time() - 3
        if t_avail > 5:
            h_up = upsample(best_h, n2)
            h_opt, c5 = sa_search(h_up, n2, t_avail, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    # Phase 3: final refinement
    remaining = budget_s - (time.time() - start_time) - 3
    if remaining > 10:
        n_best = len(best_h)
        if n_best % 2 != 0:
            n_best += 1
            best_h = upsample(best_h, n_best)
        h_opt, c5 = sa_search(best_h, n_best, remaining, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    return best_h, best_c5, len(best_h)
