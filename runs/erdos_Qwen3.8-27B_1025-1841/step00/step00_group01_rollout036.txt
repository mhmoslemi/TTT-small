def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def compute_gradient(h, beta):
    """Compute gradient of smooth_max(correlation) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    w = smooth_max_weights(corr, beta)
    
    # Only use significant weights
    threshold = 1e-6
    active = np.where(w > threshold)[0]
    
    grad = np.zeros(n)
    for j in active:
        s = j - (n - 1)

- If k=0: ∫₀² h(x)(1-h(x)) dx
- If k>0: ∫₀^{2-k} h(x)(1-h(x+k)) dx (x ranges from 0 to 2-k)
- If k<0: ∫_{-k}^{2} h(x)(1-h(x+k)) dx (x ranges from -k to 2)

In the discrete correlation:
- k=n-1 (s=0): sum h[i](1-h[i]) = the zero-lag term
- k=n-1+s (s>0): sum_{i=0}^{n-1-s} h[i](1-h[i+s]) = positive shift s
- k=n-1-s (s>0): sum_{i=s}^{n-1} h[i](1-h[i-s]) = negative shift s (i.e., h(x) vs h(x-s))

For negative shift s (i.e., x+k = x-s, k=-s):
∫_{s}^{2} h(x)(1-h(x-s)) dx ≈ dx * sum_{i=s}^{n-1} h[i](1-h[i-s])

This is corr_full[n-1-s] in my notation. Good.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    w = smooth_max_weights(corr, beta)
    threshold = 1e-10
    active = np.where(w > threshold)[0]
    grad = np.zeros(n)
    for j in active:
        s = j - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += w[j] * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w[j] * h[lo2 - s:hi2 - s]
    return dx * grad

def fix_constraints(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        total = np.sum(room)
        if total < 1e-15:
            break
        h += (diff / total) * room
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    x_ext = np.concatenate([x_old, [2.0]])
    h_ext = np.concatenate([h, [h[0]]])
    return np.interp(x_new, x_ext, h_ext)

def run_slsqp(h0, n, beta, max_iter, ftol):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
        return smooth_max_val(corr, beta)

    def grad(x):
        return compute_gradient(x, beta)

    def obj_grad(x):
        return obj(x), grad(x)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj_grad, h0, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': max_iter, 'ftol': ftol})
    return result

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def time_left():
        return budget_s - elapsed()

    parent = np.array(initial_h_values, dtype=float)
    best_h = parent.copy()
    best_c5 = evaluate_c5(parent)
    best_n = len(parent)

    beta = 500
    resolutions = [48, 96, 192, 384]

    for n in resolutions:
        if time_left() < 150:
            break

        h0 = interpolate_to(best_h, n)
        h0 = fix_constraints(h0, n)

        c5_0 = evaluate_c5(h0)
        if c5_0 < best_c5 - 1e-10:
            best_c5 = c5_0
            best_h = h0.copy()
            best_n = n

        max_iter = 300 if n <= 96 else (200 if n <= 192 else 100)
        result = run_slsqp(h0, n, beta, max_iter, 1e-12)

        if result.fun < c5_0:
            h_opt = fix_constraints(result.x, n)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5 - 1e-10:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

        # Local search at this resolution
        h_search = best_h.copy() if best_n == n else fix_constraints(interpolate_to(best_h, n), n)
        max_iterations = 300 if n <= 96 else 150

        for it in range(max_iterations):
            if time_left() < 100:
                break

            scale = 0.03 * max(0.1, 1.0 - it / max_iterations)
            noise = np.random.uniform(-scale, scale, n)
            h_try = h_search + noise
            h_try = fix_constraints(h_try, n)
            c5_try = evaluate_c5(h_try)

            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n
                h_search = h_try.copy()
            elif c5_try < evaluate_c5(h_search) + 1e-8:
                if np.random.random() < 0.15:
                    h_search = h_try.copy()

            if it % 80 == 79 and time_left() > 200:
                result2 = run_slsqp(h_search, n, beta, 100, 1e-12)
                h_ref = fix_constraints(result2.x, n)
                c5_ref = evaluate_c5(h_ref)
                if c5_ref < best_c5 - 1e-10:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                    best_n = n
                h_search = h_ref.copy()

    # Final: try binary-like projection on best solution
    if time_left() > 60:
        h_bin = interpolate_to(best_h, best_n)
        h_bin = (h_bin > 0.5).astype(float)
        h_bin = fix_constraints(h_bin, best_n)
        c5_bin = evaluate_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        # Try thresholding at different levels
        for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
            h_t = (interpolate_to(best_h, best_n) > thresh).astype(float)
            h_t = fix_constraints(h_t, best_n)
            c5_t = evaluate_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Final SLSQP polish if time permits
    if time_left() > 120:
        n_final = best_n
        h_final = fix_constraints(best_h, n_final)
        max_iter_final = 200 if n_final <= 96 else 100
        result_final = run_slsqp(h_final, n_final, beta, max_iter_final, 1e-14)
        if result_final.fun < best_c5:
            h_pol = fix_constraints(result_final.x, n_final)
            c5_pol = evaluate_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
                best_n = n_final

    return best_h, best_c5, best_n
