```python
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-11:
        active = (h > 1e-9) & (h < 1.0 - 1e-9)
        if active.any():
            h[active] += diff / active.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_gradient(h, beta=50.0):
    """Gradient of smooth max approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    one_minus = 1.0 - h
    corr = np.correlate(h, one_minus, mode="full") * dx  # scaled
    
    # Softmax weights
    max_val = corr.max()
    exp_corr = np.exp(beta * (corr - max_val))
    weights = exp_corr / exp_corr.sum()
    
    # Compute weighted sum of gradients
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        shift = k - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j - shift < n:
                val += (1.0 - h[j - shift])
            if 0 <= j + shift < n:
                val -= h[j + shift]
            g[j] += weights[k] * val
    return g * dx
```

```python
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        active = (h > 1e-8) & (h < 1.0 - 1e-8)
        if active.sum() > 0:
            h[active] += diff / active.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_values(h):
    """Return all correlation values (unshifted by dx)."""
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_grad(h, beta):
    """Smooth gradient of C5 using softmax weighting."""
    n = len(h)
    dx = 2.0 / n
    corr = corr_values(h)  # raw, length 2n-1
    scaled = corr * dx     # C5 contributions per shift
    
    max_s = scaled.max()
    exp_s = np.exp(beta * (scaled - max_s))
    w = exp_s / exp_s.sum()
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = (n - 1) - k
        # Part 1: (1-h[j+s]) for j+s in [0,n)
        a = max(0, -s)
        b = min(n, n - s)
        if b > a:
            g[a:b] += wk * (1.0 - h[a+s:b+s])
        # Part 2: -h[j-s] for j-s in [0,n)
        c = max(0, s)
        d = min(n, n + s)
        if d > c:
            g[c:d] += wk * (-h[c-s:d-s])
    return g

def optimize_at_resolution(h, n, rng, time_limit, beta_start=30, beta_end=300):
    """Optimize h at fixed resolution n using smooth gradient + SA."""
    target_sum = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    current = h.copy()
    start_t = time()
    t = 0
    no_improve = 0
    lr = 0.1  # initial learning rate
    
    while time() - start_t < time_limit:
        # Adaptive beta
        progress = min(1.0, (time() - start_t) / time_limit)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute gradient
        g = smooth_grad(current, beta)
        
        # Adaptive step size (decrease over time)
        lr_t = lr * (1.0 / (1.0 + 0.01 * t))
        
        # Gradient step
        cand = current - lr_t * g
        cand = project_feasible(cand, target_sum)
        cand_c5 = compute_c5(cand)
        
        if cand_c5 < best_c5:
            best = cand.copy()
            best_c5 = cand_c5
            current = cand
            no_improve = 0
        else:
            no_improve += 1
            # SA acceptance for exploration
            temp = best_c5 * 0.05 * np.exp(-2.0 * progress)
            if rng.random() < np.exp(-(cand_c5 - current_c5) / max(temp, 1e-10)):
                current = cand
            
            # Periodic random perturbation to escape local optima
            if no_improve > 200:
                sigma = 0.05 * (1.0 + 0.5 * rng.random())
                current = project_feasible(current + sigma * rng.randn(n), target_sum)
                no_improve = 0
        
        # Occasional swap move (preserves sum exactly)
        if t % 50 == 0 and rng.random() < 0.3:
            i, j = rng.choice(n, 2, replace=False)
            current[i], current[j] = current[j], current[i]
            current_c5_check = compute_c5(current)
            if current_c5_check < best_c5:
                best = current.copy()
                best_c5 = current_c5_check
        
        t += 1
    
    return best, best_c5
```
