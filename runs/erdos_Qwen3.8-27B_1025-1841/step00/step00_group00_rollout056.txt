```python
def compute_gradient(h, p, n):
    """
    h: array of length n
    p: softmax weights, array of length 2n-1 (indexed by m=0,...,2n-2)
    Returns gradient of smooth max of correlations
    """
    grad = np.zeros(n)
    for m in range(2*n - 1):
        pm = p[m]
        if pm < 1e-15:
            continue
        k = m - (n - 1)  # lag
        # Term 1: (1-h[i-k]) for i where 0 <= i-k < n, i.e., k <= i <= n-1+k
        i_lo = max(0, k)
        i_hi = min(n-1, n-1+k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi+1)
            grad[idx] += pm * (1.0 - h[idx - k])
        # Term 2: -h[i+k] for i where 0 <= i+k < n, i.e., -k <= i <= n-1-k
        i_lo2 = max(0, -k)
        i_hi2 = min(n-1, n-1-k)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2+1)
            grad[idx2] -= pm * h[idx2 + k]
    return grad
```
