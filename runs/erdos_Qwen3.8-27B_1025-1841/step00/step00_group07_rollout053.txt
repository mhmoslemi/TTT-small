import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import zoom
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    m = len(corr)
    exps = np.exp((corr - cmax) * alpha)
    return float(cmax + np.log(np.sum(exps)) / (alpha * m))

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)
    nmask = np.sum(mask)
    if nmask > 0:
        h[mask] += diff / nmask
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h
