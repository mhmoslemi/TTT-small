import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return float(np.max(corr) * dx)

def make_bivalent(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.round(h)
    h = np.clip(h, 0, 1)
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        ones = np.where(h == 1)[0]
        idx = np.random.choice(len(ones), num_ones - target, replace=False)
        h[ones[idx]] = 0
    elif num_ones < target:
        zeros = np.where(h == 0)[0]
        idx = np.random.choice(len(zeros), target - num_ones, replace=False)
        h[zeros[idx]] = 1
    return h.astype(float)

def bivalent_local_search(h, dx, max_time=100, seed=42):

    rng = np.random.RandomState(seed)
    n = len(h)
    h = make_bivalent(h, n)
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    start = time.time()
    while time.time() - start < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap = None
        best_val = current_c5
        
        # Try all swaps
        for a in ones:
            for b in zeros:
                h[a], h[b] = 0, 1
                val = compute_c5(h, dx)
                h[a], h[b] = 1, 0
                if val < best_val - 1e-15:

                    best_swap = (a, b)

            # No improvement, try random restart
            break
        
        h[best_swap[0]], h[best_swap[1]] = 0, 1
        current_c5 = best_val
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return float(np.max(corr) * dx)

def make_bivalent(h, n, rng):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(np.round(h), 0.0, 1.0)
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        ones_idx = np.where(h == 1.0)[0]
        sel = rng.choice(len(ones_idx), num_ones - target, replace=False)
        h[ones_idx[sel]] = 0.0
    elif num_ones < target:
        zero_idx = np.where(h == 0.0)[0]
        sel = rng.choice(len(zero_idx), target - num_ones, replace=False)
        h[zero_idx[sel]] = 1.0
    return h

def random_bivalent(n, rng):
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
