def log_sum_exp_objective(h, alpha, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    scaled = alpha * corr * dx
    # numerical stability
    max_scaled = np.max(scaled)
    if max_scaled > 700:
        return max_scaled / alpha  # overflow protection
    log_sum = np.log(np.sum(np.exp(scaled - max_scaled))) + max_scaled
    return log_sum / alpha

def project_onto_feasible(y, target_sum):
    """Project y onto {h: 0 <= h[i] <= 1, sum(h) = target_sum}"""
    # Bisection on lambda
    lo, hi = y.min() - target_sum, y.max()  # lambda range
    # Actually, we need: sum(clip(y - lambda, 0, 1)) = target_sum
    # As lambda increases, each component decreases, so sum decreases
    # We want the lambda where sum = target_sum
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0))
    
    # Find range
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

    # Renormalize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(y, target_sum):
    lo = y.min() - target_sum / len(y) - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h_old, new_n):
    old_n = len(h_old)
    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h_old)
    target_sum = new_n / 2.0
    return project_onto_feasible(h_new, target_sum)

def softmax_weights(corr, alpha, dx):
    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    return exp_s / np.sum(exp_s)

def compute_grad(h, p, n):

    dx = 2.0 / n
    # p[k] corresponds to shift tau = k - (n-1)

                result[t:] += pk * h[:n - t]
            # h[j+t] valid for j in [0, n-1-t]
            if t < n:
                result[:n - t] += pk * h[:n - t]
    return dx * (1.0 - result)

        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def upsample_linear(h_old, new_n):
    old_n = len(h_old)
    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h_old)
    return project_onto_feasible(h_new, new_n / 2.0)

def compute_gradient(h, p, n):
    dx = 2.0 / n
    result = np.zeros(n)
    for k in range(2 * n - 1):
        tau = k - (n - 1)
        pk = p[k]
        if pk < 1e-14:
            continue
        if tau == 0:
            result[:] += 2.0 * pk * h[:]
        elif tau > 0:
            m = n - tau
            if m > 0:
                result[:m] += pk * h[tau:]
                result[tau:] += pk * h[:m]
        else:
            t = -tau
            m = n - t
            if m > 0:
                result[t:] += pk * h[:m]
                result[:m] += pk * h[t:]
    return dx * (1.0 - result)

def optimize_gradient(h_init, n, time_limit, seed):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 40.0
    lr = 0.04
    stall_count = 0

    for it in range(5000):
        if time.time() - t_start > time_limit:
            break
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 200:
            delta = rng.randn(n) * 0.03
            delta -= np.mean(delta)
            h_try = h + delta
            h_try = project_onto_feasible(h_try, target_sum)
            if compute_c5(h_try) < best_c5:
                h = h_try
                # keep best_h as is
            stall_count = 0
            lr = 0.04

        corr = np.correlate(h, 1.0 - h, mode="full")
        scaled = alpha * corr * (2.0 / n)
        max_s = float(np.max(scaled))
        exp_s = np.exp(scaled - max_s)
        p = exp_s / np.sum(exp_s)
        grad = compute_gradient(h, p, n)

        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_new = h - lr_try * grad
            h_new = project_onto_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                improved = True
                break

        if not improved:
            lr *= 0.6
            if lr < 0.003:
                lr = 0.04
                alpha = min(alpha * 1.4, 300.0)

    return best_h, best_c5

def random_perturb_search(h_init, n, time_limit, seed):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    sigma = 0.02

    while time.time() - t_start < time_limit:
        batch_size = 20
        for _ in range(batch_size):
            if time.time() - t_start > time_limit:
                break
            delta = rng.randn(n) * sigma
            delta -= np.mean(delta)
            h_new = h + delta
            h_new = project_onto_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                sigma = min(sigma * 1.08, 0.12)
            else:
                sigma = max(sigma * 0.97, 0.005)

    return best_h, best_c5

def local_search_01(h_init, n, time_limit, seed):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    target_sum = int(round(n / 2.0))
    if target_sum * 2 != n:
        return h_init, compute_c5(h_init)

    h = (h_init > 0.5).astype(float)
    num_ones = int(np.sum(h))
    if num_ones != target_sum:
        idx_ones = np.where(h == 1)[0]
        idx_zeros = np.where(h == 0)[0]
        if num_ones > target_sum:
            remove = rng.choice(idx_ones, num_ones - target_sum, replace=False)
            h[remove] = 0
        else:
            add = rng.choice(idx_zeros, target_sum - num_ones, replace=False)
            h[add] = 1

    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_iters = 0

    while time.time() - t_start < time_limit:
        n_iters += 1
        if n_iters % 200 == 0:
            pass
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_global_start = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    h0 = project_onto_feasible(h0, len(h0) / 2.0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    n0 = len(h0)

    # Phase 1: optimize at parent resolution
    if budget_s - (time.time() - t_global_start) > 100:
        t_lim = min(100, budget_s - (time.time() - t_global_start) - 80)
        h, c5 = optimize_gradient(h0, n0, t_lim, seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 2: upsample and optimize at higher resolutions
    for n in [2 * n0, 4 * n0, 8 * n0]:
        remaining = budget_s - (time.time() - t_global_start)
        if remaining < 80:
            break
        t_lim = min(300, remaining - 60)
        if t_lim < 40:
            break
        h_init = upsample_linear(best_h, n)
        h, c5 = optimize_gradient(h_init, n, t_lim, seed + n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 3: random perturbation polish
    remaining = budget_s - (time.time() - t_global_start)
    if remaining > 60:
        t_lim = min(remaining - 20, 400)
        if t_lim > 30:
            h, c5 = random_perturb_search(best_h, len(best_h), t_lim, seed + 9999)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    # Phase 4: 0-1 local search
    remaining = budget_s - (time.time() - t_global_start)
    if remaining > 40:
        n_curr = len(best_h)
        if n_curr % 2 == 0:
            t_lim = remaining - 15
            if t_lim > 20:
                h01, c5 = local_search_01(best_h, n_curr, t_lim, seed + 7777)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h01

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_onto_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
