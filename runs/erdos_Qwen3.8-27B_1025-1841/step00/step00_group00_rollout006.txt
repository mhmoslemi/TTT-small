import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def subgradient(h, n):
    """Compute subgradient of max_k correlate(h, 1-h, 'full')[k]"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(c)
    # Find all indices achieving the max (or near max for numerical stability)
    mask = c >= max_val - 1e-10
    max_indices = np.where(mask)[0]
    
    grad = np.zeros(n)
    for k in max_indices:
        lag = k - (n - 1)
        if lag >= 0:
            # c[k] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
            for i in range(n - lag):
                grad[i] += -h[i+lag]
                grad[i+lag] += (1-h[i])
        else:
            # c[k] = sum_{i=-lag}^{n-1} h[i+lag] * (1-h[i])
            for i in range(-lag, n):
                grad[i] += -h[i+lag]
                grad[i+lag] += (1-h[i])
    
    return grad * dx  # Scale by dx since we multiply by dx in the objective

def project(h):
    """Project onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    h = np.clip(h, 0, 1)
    target = len(h) / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Adjust: if diff > 0, increase small values; if diff < 0, decrease large values
    if diff > 0:
        # Increase values that are below 1, prioritizing smallest
        order = np.argsort(h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-10:
                break
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            sub = min(remaining, h[idx])
            h[idx] -= sub
            remaining -= sub
            if remaining <= 1e-10:
                break
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(45, 0.5)
    
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Subgradient descent
    lr = 0.01
    for iteration in range(50000):
        if time.time() - start > budget_s * 0.7:
            break
        grad = subgradient(best_h, n)
        best_h = project(best_h - lr * grad)
        c5 = evaluate_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            lr *= 0.999

def subgrad_step(h, max_idx, n):
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[:m] -= h[lag:]
        grad[lag:] += 1.0 - h[:m]
    else:
        m = n + lag
        grad[-lag:] -= h[:m]
        grad[:m] += 1.0 - h[-lag:]
    return grad

def project_feasible(h):
    h = np.clip(h, 0.0, 1.0)
    target = float(len(h)) / 2.0
    diff = target - h.sum()
    if diff > 1e-14:
        idx = np.argsort(h)
        rem = diff
        for i in idx:
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
            if rem < 1e-14:
                break
    elif diff < -1e-14:
        idx = np.argsort(-h)
        rem = -diff
        for i in idx:
            sub = min(rem, h[i])
            h[i] -= sub
            rem -= sub
            if rem < 1e-14:
                break
