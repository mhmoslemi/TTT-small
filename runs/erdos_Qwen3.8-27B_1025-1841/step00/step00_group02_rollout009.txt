```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def upsample(h, N_new):
    """Upsample h to length N_new using linear interpolation"""
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = np.linspace(0, 1, N_old)
    x_new = np.linspace(0, 1, N_new)
    return np.interp(x_new, x_old, h)

def project(h, target_sum):
    """Project onto {sum(t)=target_sum, 0<=t<=1}"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            if diff <= 0:
                break
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
    else:
        idx = np.argsort(-h)
        for i in idx:
            if diff >= 0:
                break
            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
    return h

def objective_and_grad(h_vals, N, t_sm):
    """Compute soft-max objective and gradient for h reshaped to length N"""
    h = h_vals.reshape(N)
    # Compute all correlations
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")  # shape (2N-1,)
    
    # Soft max
    tc = t_sm * corr
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    sum_exp = np.sum(exp_tc)
    f = (tc_max + np.log(sum_exp)) / t_sm
    
    # Gradient of soft max w.r.t. corr: weights = exp(tc - tc_max) / sum_exp = softmax(tc)
    weights = exp_tc / sum_exp  # shape (2N-1,)
    
    # Now compute gradient w.r.t. h
    # corr[m] where m = 0,...,2N-2 corresponds to shift s = m - (N-1)
    grad = np.zeros(N)
    
    for m in range(2 * N - 1):
        s = m - (N - 1)  # shift
        w = weights[m]
        if w < 1e-15:
            continue
        if s >= 0:
            # corr = sum_{i=0}^{N-1-s} h[i+s] * (1-h[i])
            # d/dh[j]: (1-h[j-s]) for j in [s, N-1] (from h[j] in first factor)
            #          -h[j+s] for j in [0, N-1-s] (from h[j] in second factor)
            for j in range(N):
                contrib = 0.0
                if s <= j <= N - 1:
                    contrib += (1 - h[j - s])
                if 0 <= j <= N - 1 - s:
                    contrib -= h[j + s]
                grad[j] += w * contrib
        else:
            # s < 0, let r = -s
            # corr = sum_{i=r}^{N-1} h[i-r] * (1-h[i])  where r=-s
            r = -s
            # Same as positive shift r but with h and (1-h) swapped
            # Actually: corr_{-r} = sum_{i=r}^{N-1} h[i-r]*(1-h[i])
            # Let j be the index: h[j]*(1-h[j+r]) for j in [0, N-1-r]
            for j in range(N):
                contrib = 0.0
                if 0 <= j <= N - 1 - r:
                    contrib += (1 - h[j + r])
                if r <= j <= N - 1:
                    contrib -= h[j - r]
                grad[j] += w * contrib
    
    return f, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def upsample(h, target_len):
    N = len(h)
    if N == target_len:
        return h.copy()
    if N == 1:
        return np.full(target_len, h[0])
    x_old = np.arange(N, dtype=float) / (N - 1)
    x_new = np.arange(target_len, dtype=float) / (target_len - 1)
    return np.interp(x_new, x_old, h)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        for i in np.argsort(h):
            if diff <= 1e-14:
                break
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
    else:
        for i in np.argsort(-h):
            if diff >= -1e-14:
                break
            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
    return h

def soft_max_val(corr, t):
    xc = t * corr
    xm = np.max(xc)
    return (xm + np.log(np.sum(np.exp(xc - xm)))) / t

def soft_max_grad(corr, t):
    xc = t * corr
    xm = np.max(xc)
    e = np.exp(xc - xm)
    return e / np.sum(e)

def obj_grad(params, N, t_sm, penalty):
    h = params
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = soft_max_val(corr, t_sm)
    w = soft_max_grad(corr, t_sm)
    
    # Gradient of soft_max w.r.t. h
    grad = np.zeros(N)
    for m in range(2 * N - 1):
        s = m - (N - 1)
        wi = w[m]
        if wi < 1e-14:
            continue
        if s >= 0:
            grad[s:] += wi * (1.0 - h[:N-s])
            grad[:N-s] -= wi * h[s:]
        else:
            r = -s
            grad[:N-r] += wi * (1.0 - h[r:])
            grad[r:] -= wi * h[:N-r]
    
    # Add penalty gradient
    diff = np.sum(h) - N / 2.0
    grad += 2.0 * penalty * diff
    
    return val + penalty * diff * diff, grad

def optimize_continuous(h_init, N, max_time):
    h = h_init.copy()
    target_sum = N / 2.0
    h = project(h, target_sum)
    
    t_sm = 50.0  # soft-max parameter
    penalty = 100.0
    start = time.time()
    best = h.copy()
    best_val = compute_c5(h)
    
    # Use L-BFGS-B
    for it in range(50):  # restart with increasing t
        if time.time() - start > max_time:
            break
        t_sm = 50.0 + it * 20  # increase sharpness
        
        def obj(p):
            v, g = obj_grad(p, N, t_sm, penalty)
            return v, g
        
        res = minimize(obj, h, method="L-BFGS-B",
                       bounds=[(0, 1)] * N,
                       options={"maxiter": 200, "gtol": 1e-4,
                                "maxfun": 500})
        
        h_new = project(res.x, target_sum)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best = h_new.copy()
            h = h_new.copy()
        else:
            h = best.copy()
            # Perturb
            h += 0.01 * np.random.randn(N)
            h = project(h, target_sum)
    
    return best, best_val
```
