            h_init = project_feasible(h_init, n)
            candidates.append((h_init, n))

    # Additional diverse initializations
    for n in [180, 360, 720]:
        x = np.linspace(0, 2, n, endpoint=False)

        # Periodic binary patterns with different periods
        for period_frac in [0.25, 0.333, 0.4, 0.5]:
            period = 2.0 * period_frac
            h_pat = np.zeros(n)
            for i in range(n):
                pos_in_period = x[i] % period
                if pos_in_period < period * 0.5:
                    h_pat[i] = 1.0
            h_pat = project_feasible(h_pat, n)
            candidates.append((h_pat, n))

        # Random binary
        h_bin = (rng.rand(n) > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        candidates.append((h_bin, n))

        # Smooth sine
        h_sine = 0.5 + 0.45 * np.sin(2 * np.pi * x / 0.5)
        h_sine = np.clip(h_sine, 0, 1)
        h_sine = project_feasible(h_sine, n)
        candidates.append((h_sine, n))

    # Time management: allocate time per candidate
    n_cand = len(candidates)
    total_opt_time = max(60, t_global_end - start_time - 120)
    time_per = total_opt_time / n_cand

    # Sort by expected quality (upscaled parent first)
    for ci, (h_init, n) in enumerate(candidates):
        if time.time() >= t_global_end - 60:
            break
        t_end = min(time.time() + time_per * 1.8, t_global_end - 60)
        if t_end <= time.time() + 2:
            t_end = time.time() + 10

        h_opt, c5_opt = optimize_projected(h_init, n, t_end, K=5, lr0=0.04, momentum=0.9)

        # Try binary rounding
        h_bin, c5_bin = try_binary_round(h_opt, n)
        if c5_bin < c5_opt:
            h_opt, c5_opt = h_bin, c5_bin

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n

    # Final refinement with more iterations on best
    if time.time() < t_global_end - 30:
        t_end = min(time.time() + 60, t_global_end)
        h_opt, c5_opt = optimize_projected(best_h, best_n, t_end, K=10, lr0=0.02, momentum=0.95)
        h_bin, c5_bin = try_binary_round(h_opt, best_n)
        if c5_bin < c5_opt:
            h_opt, c5_opt = h_bin, c5_bin
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Random restarts from best
    for _ in range(5):
        if time.time() >= t_global_end:
            break
        h_rand = best_h + 0.08 * rng.randn(best_n)
        h_rand = project_feasible(h_rand, best_n)
        t_end = min(time.time() + 20, t_global_end)
        h_opt, c5_opt = optimize_projected(h_rand, best_n, t_end, K=10, lr0=0.03, momentum=0.9)
        h_bin, c5_bin = try_binary_round(h_opt, best_n)
        if c5_bin < c5_opt:
            h_opt, c5_opt = h_bin, c5_bin
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = best_n

    # Try higher resolution from best
    if time.time() < t_global_end - 60 and best_n < 720:
        n_hi = min(best_n * 2, 1440)
        xp = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(xn, xp, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        t_end = min(time.time() + 60, t_global_end)
        h_opt, c5_opt = optimize_projected(h_hi, n_hi, t_end, K=10, lr0=0.02, momentum=0.9)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_hi

    # Final validation
    dx = 2.0 / best_n
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    return best_h, final_c5, best_n
