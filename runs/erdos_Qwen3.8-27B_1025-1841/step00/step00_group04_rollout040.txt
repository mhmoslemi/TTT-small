import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, n_points):
    """Project h onto {x: 0≤x≤1, sum(x)=n/2}"""
    target = n_points / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        # Need to adjust
        diff = target - s
        if diff > 0:
            # Need to increase sum; increase elements that are < 1
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            # Need to decrease sum; decrease elements that are > 0
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0, 1)
    # Final adjustment for numerical precision
    s = np.sum(h)
    if abs(s - target) > 0:
        # Distribute the error
        diff = target - s
        mask = (h > 0) & (h < 1)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def subgradient_of_max(h, n_points):
    """Compute subgradient of C5(h) at point h"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    i_star = np.argmax(corr)
    lag = i_star - (n_points - 1)
    grad = np.zeros(n_points)

        j1 = j - lag
        j2 = j + lag

        if 0 <= j2 < n_points:
            g -= h[j2]
        grad[j] = dx * g
    return grad, i_star, corr[i_star] * dx

def suav_max(corr, dx, temperature=0.01):
    """Smoothed max of corr*dx using log-sum-exp"""
    vals = corr * dx
    vals = vals / temperature
    max_val = np.max(vals)
    log_sum = max_val + np.log(np.sum(np.exp(vals - max_val)))
    return log_sum * temperature

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_constraint(h, n_points):
    target = n_points / 2.0
    h = h.astype(float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h0, n_points, maxiter=300):
    dx = 2.0 / n_points
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}]
    bounds = [(0.0, 1.0)] * n_points
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        h_opt = project_constraint(res.x, n_points)
        c5 = compute_c5(h_opt)
        if c5 < compute_c5(h0) + 1e-15:
            return h_opt, c5
    except Exception:
        pass
    return h0.copy(), compute_c5(h0)

def gradient_optimize(h0, n_points, max_iters=15000, beta_start=15, beta_end=400):
    dx = 2.0 / n_points
    h = h0.copy()
    best_h = h0.copy()

    step = 1.0 / (2.0 * n_points)
    j_arr = np.arange(n_points)
    stagnant = 0

    for it in range(max_iters):
        frac = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        corr = np.correlate(h, 1 - h, mode='full')
        vals = corr * dx
        mv = np.max(vals)

        grad = np.zeros(n_points)

        if len(active) == 0:
            active = np.array([np.argmax(p)])
        for i in active:
            lag = i - (n_points - 1)
            j1 = j_arr - lag
            j2 = j_arr + lag
            v1 = (j1 >= 0) & (j1 < n_points)
            v2 = (j2 >= 0) & (j2 < n_points)
