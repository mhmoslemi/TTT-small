def project_onto_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    # Binary search for lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

            lo = mid
        else:

    """Smooth objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    c = np.correlate(h, omh, mode="full") * dx
    mx = c.max()
    e = np.exp(beta * (c - mx))
    s = e.sum()
    w = e / s
    obj = mx + np.log(s) / beta
    g = np.empty(n)
    for i in range(n):
        g[i] = dx * (-np.dot(h, w[n-1-i:2*n-1-i]) + np.dot(omh, w[i:i+n][::-1]))
    return obj, g

    return project_feasible(np.interp(np.linspace(0,2,nn,endpoint=False),

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=np.float64)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    c = np.correlate(h, omh, mode="full") * dx
    mx = c.max()
    e = np.exp(beta * (c - mx))
    s = e.sum()
    w = e / s
    grad = np.empty(n)
    for i in range(n):
        grad[i] = dx * (-np.dot(h, w[n-1-i:2*n-1-i]) + np.dot(omh, w[i:i+n][::-1]))
