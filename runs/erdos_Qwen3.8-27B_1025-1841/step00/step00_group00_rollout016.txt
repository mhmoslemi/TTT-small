def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def ensure_binary_balanced(h, rng):
    """Make h binary with sum = len(h)/2"""
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0)
    target = n // 2
    ones = int(h.sum())
    if ones > target:
        idx = np.where(h == 1)[0]
        pick = rng.choice(len(idx), size=ones - target, replace=False)
        h[idx[pick]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        pick = rng.choice(len(idx), size=target - ones, replace=False)
        h[idx[pick]] = 1
    return h

def sa_optimize(h, rng, time_limit, T_init=None):
    """Simulated annealing with swap moves"""
    n = len(h)
    h = h.copy()
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    best_h = h.copy()
    
    if T_init is None:
        T_init = max(0.05, curr_c5 * 0.1)
    T = T_init
    T_min = 1e-10
    cool = 0.99999  # per iteration

    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    t0 = time.time()
    no_improve = 0
    while time.time() - t0 < time_limit:
        # Pick random 1 and 0 to swap
        li = rng.randint(len(ones_idx))
        lz = rng.randint(len(zeros_idx))
        i = ones_idx[li]
        j = zeros_idx[lz]
        
        # Swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        if new_c5 <= curr_c5 or (T > T_min and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            # Update indices
            ones_idx[li] = j
            zeros_idx[lz] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:

            h[i] = 1
            h[j] = 0
            no_improve += 1

                b = rng.randint(len(zeros_idx))

                ones_idx[a] = oz
                zeros_idx[b] = oa
            curr_c5 = compute_c5(h)
            T = max(T, 0.03)
            no_improve = 0

    h_new = np.zeros(n_new)
    for i in range(n):

                h_new[a], h_new[b] = h_new[b], h_new[a]
