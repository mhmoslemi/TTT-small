import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    m = np.max(arr)

    h_start = fix_constraint(h_start, n)
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)

    # Phase 1: Beta annealing SLSQP from initial
    h_curr = best_h.copy()
    for beta in [5, 20, 60, 200, 800, 3000]:
        if tl() < 40:
            break
        h_opt, c5 = _slsqp_solve(h_curr, n, beta, 400)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            h_curr = h_opt.copy()

    # Phase 2: Random restarts
    n_restarts = 40
    for r in range(n_restarts):
        if tl() < 40:
            break
        scale = 0.12 + 0.08 * (r / max(1, n_restarts))
        h_r = best_h + scale * np.random.randn(n)
        h_r = fix_constraint(h_r, n)
        h_opt, c5 = _slsqp_solve(h_r, n, 200, 250)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Phase 3: Binary-like restarts
    if tl() > 60:
        for r in range(10):
            if tl() < 30:
                break
            idx = np.random.choice(n, n // 2, replace=False)
            h_bin = np.zeros(n)
            h_bin[idx] = 1.0
            # Add small noise to help escape
            h_bin += 0.05 * np.random.randn(n)
            h_bin = fix_constraint(h_bin, n)
            h_opt, c5 = _slsqp_solve(h_bin, n, 50, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 4: Pair descent polish on best_n
    if tl() > 30:
        best_h, best_c5 = _pair_descent(best_h.copy(), n, max_sweeps=15)

    # Phase 5: Try higher resolution (2x)
    if tl() > 250:
        n2 = n * 2
        _glob_n = n2
        h_fine = np.zeros(n2)
        h_fine[0::2] = best_h
        h_fine[1::2] = best_h
        h_fine = fix_constraint(h_fine, n2)
        c5_fine = compute_c5(h_fine)

        if c5_fine <= best_c5 * 1.05:
            h_curr2 = h_fine.copy()
            for beta in [30, 150, 600]:
                if tl() < 60:
                    break
                h_opt2, c5_2 = _slsqp_solve(h_curr2, n2, beta, 300)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h_opt2
                    n = n2
                    h_curr2 = h_opt2.copy()
            if tl() > 30:
                best_h, best_c5 = _pair_descent(best_h.copy(), n, max_sweeps=8)

    # Final cleanup
    best_h = fix_constraint(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
