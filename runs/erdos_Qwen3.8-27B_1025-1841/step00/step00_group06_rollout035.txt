def repair_h(h, n):

    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n / 2.0
    if s == 0:

        h[:int(target)] = 1.0

        diff = target - np.sum(h)
        if abs(diff) > 1e-10:

                    diff -= gain

                    diff += loss

```python
def steepest_descent_step(h, n):
    """Find the best single swap that reduces the max overlap"""
    c = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(c)
    max_val = c[max_idx]
    s = max_idx - (n - 1)
    
    # Find all (i, j) where h[i]=1, h[j]=0, and j = i+s (or i = j+s)
    # Swapping i and j changes the overlap at shift s
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_improvement = 0
    best_swap = None
    
    # For each 1 at position i, check if position i+s is a 0
    # Swapping would reduce g_s by 1 (the pair (i, i+s) no longer contributes)
    # But it might increase other g_k values
    
    # Simpler: just try all swaps that are "promising" (reduce the current max)
    candidates = []
    for i in ones:
        j = i + s
        if 0 <= j < n and h[j] == 0:
            candidates.append((i, j))
    
    if not candidates:
        # Try the opposite direction
        for j in zeros:
            i = j + s
            if 0 <= i < n and h[i] == 1:
                candidates.append((i, j))
    
    if not candidates:
        return None, max_val
    
    # Evaluate top candidates
    rng = np.random.RandomState(42)
    rng.shuffle(candidates)
    
    for i, j in candidates[:50]:
        h[i], h[j] = 0, 1
        new_c = np.max(np.correlate(h, 1-h, mode='full'))
        h[i], h[j] = 1, 0
        improvement = max_val - new_c
        if improvement > best_improvement:
            best_improvement = improvement
            best_swap = (i, j)
    
    if best_swap:
        i, j = best_swap
        h[i], h[j] = 0, 1
        return best_swap, max_val - best_improvement
    return None, max_val
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def repair_binary(h, n, rng):
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2

        ones = np.where(h == 1)[0].copy()

        zeros = np.where(h == 0)[0].copy()
        rng.shuffle(zeros)
        for i in range(target - n_ones):
            h[zeros[i]] = 1
    return h
