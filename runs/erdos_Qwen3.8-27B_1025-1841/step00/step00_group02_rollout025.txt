def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    thresh = max_c - 1e-8
    active = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for k in active:
        off = k - (n - 1)
        lo1 = max(0, off)
        hi1 = min(n-1, n-1+off)
        if lo1 <= hi1:

            grad[idx] += 1.0 - h[idx - off]

            grad[idx2] -= h[idx2 + off]
