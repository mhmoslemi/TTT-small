            g[:n-d] += (1 - h[d:n])  # h[j+d] for j in [0, n-1-d]
            # Second factor: h[i+d] at i+d=j, so i=j-d, for j-d in [0, n-1-d], j in [d, n-1]
            g[d:n] += (-h[:n-d])  # -h[j-d] for j in [d, n-1]
            g *= dx
        else:  # d < 0, let d' = -d > 0
            d_neg = -d
            # i ranges from d_neg to n-1 (shifted by d)
            # f_k = dx * sum_{i=d_neg}^{n-1} h[i]*(1-h[i-d_neg])
            # = dx * sum_{i=d_neg}^{n-1} h[i]*(1-h[i-d_neg])
            # Let j = i-d_neg, then i = j+d_neg, j from 0 to n-1-d_neg
            # f_k = dx * sum_{j=0}^{n-1-d_neg} h[j+d_neg]*(1-h[j])
            g = np.zeros(n)
            # First factor: h[j+d_neg] at position j+d_neg
            # ∂/∂h[j+d_neg] = (1-h[j]) for j in [0, n-1-d_neg]
            g[d_neg:n] += (1 - h[:n-d_neg])
            # Second factor: h[j] at position j
            # ∂/∂h[j] of -h[j+d_neg]*h[j] = -h[j+d_neg]
            g[:n-d_neg] += (-h[d_neg:n])
            g *= dx
        grad += weights[k] * g
    
    return smooth_max, grad

def project_to_feasible(h, n_points):
    """Project h onto {0≤h≤1, sum(h)=n/2}."""
    target = n_points / 2.0
    h = np.clip(h, 0, 1)
    # Adjust sum
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Add diff/n where possible
        h_new = h + diff / n_points
        if np.all(h_new >= 0) and np.all(h_new <= 1):
            h = h_new
        else:
            # More careful adjustment
            if diff > 0:
                # Add to elements with room
                room = 1 - h
                mask = room > 1e-10
                if not mask.any():
                    break
                add = diff / mask.sum()
                h[mask] = np.minimum(h[mask] + add, 1)
            else:
                # Subtract from elements with room
                room = h
                mask = room > 1e-10
                if not mask.any():
                    break
                sub = -diff / mask.sum()
                h[mask] = np.maximum(h[mask] - sub, 0)
    h = np.clip(h, 0, 1)
    return h

def c5_smooth_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    
    # Smooth max
    mx = vals.max()
    exp_v = np.exp(alpha * (vals - mx))
    Z = exp_v.sum()
    smooth = mx + np.log(Z) / alpha
    w = exp_v / Z  # weights
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        d = k - (n-1)
        g = np.zeros(n)
        if d == 0:
            g[:] = dx * (1 - 2*h)
        elif d > 0:
            # sum_{i=0}^{n-1-d} h[i](1-h[i+d])
            # d/dh[j]: (1-h[j+d]) for j in [0, n-1-d] + (-h[j-d]) for j in [d, n-1]

            g[:n-p] += (-h[p:])
            g *= dx

                        h_p = proj_grad_step(h_p, n, alpha, lr, deadline)
                    if lr > 0.001:
                        lr *= 0.998
                    if lr < 1e-9:
                        break
            c5_p = compute_c5(h_p, n)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
            if c5_p < compute_c5(h0, n):
                h0 = h_p.copy()

        # Update parent for next interpolation
        parent = best_h if len(best_h) != pn else h0
        if len(parent) != pn:
            xp2 = np.arange(pn) * (len(parent) / pn)
            parent = np.interp(np.arange(pn) * (len(parent) / pn), np.arange(len(parent)), parent)
            parent = project(parent, pn)

    # SLSQP polish on best solution if time allows
    n_best = len(best_h)
    if not time_up() and n_best <= 300:
        def obj(h):
            return compute_c5(h, n_best)

        cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n_best / 2.0,
                 'jac': lambda h: np.ones(n_best)}]
        bounds = [(0.0, 1.0)] * n_best
        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_res = project(res.x, n_best)
            c5_res = compute_c5(h_res, n_best)
            if c5_res < best_c5:
                best_h = h_res
                best_c5 = c5_res
        except Exception:
            pass

    return (best_h, best_c5, len(best_h))
