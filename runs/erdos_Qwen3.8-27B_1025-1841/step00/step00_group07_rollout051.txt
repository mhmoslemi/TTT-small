```python
def c5_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = n - 1 - max_idx  # lag
    
    # Gradient: for each m, d_corr/d_h[m] = (1-h[m+k]) - h[m-k] (with bounds)
    grad = np.zeros(n)
    for m in range(n):
        # Term 1: (1-h[m+k]) if 0 <= m+k < n
        if 0 <= m+k < n:
            grad[m] += (1.0 - h[m+k])
        # Term 2: -h[m-k] if 0 <= m-k < n
        if 0 <= m-k < n:
            grad[m] -= h[m-k]
    return dx * grad
```

```python
import numpy as np
```

```python
def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def c5_objective(h):
    n = len(h)
    return compute_c5(h, n)


def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = corr >= max_val - 1e-8
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if not active[idx]:
            continue
        k = n - 1 - idx
        for m in range(n):
            if 0 <= m + k < n:
                grad[m] += (1.0 - h[m + k])
            if 0 <= m - k < n:
                grad[m] -= h[m - k]
    n_active = np.sum(active)
    if n_active > 0:
        grad /= n_active
    return dx * grad
```
