import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def enforce_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, alpha):
    m = np.max(vals)
    if alpha * (np.max(vals) - m) < 700:
        return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha
    return m

def optimize_smooth(h_init, n, deadline):
    dx = 2.0 / n
    h_init = enforce_constraint(h_init, n)
    c5 = compute_c5(h_init)
    
    def make_obj(alpha):
        def obj(x):
            corr = np.correlate(x, 1 - x, mode="full")
            vals = corr * dx
            return smooth_max(vals, alpha) + 50.0 * (np.sum(x) - n / 2.0) ** 2
        return obj
    
    for alpha in [20, 50, 100, 200, 500]:
        if time.time() > deadline:
            break
        try:
            res = minimize(make_obj(alpha), h_init, method='L-BFGS-B',
                          bounds=[(0, 1)] * n,
                          options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = enforce_constraint(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-10:
                h_init = h_new
                c5 = c5_new
        except Exception:
            pass
    return h_init, c5

def optimize_slsqp(h_init, n, deadline):
    dx = 2.0 / n
    h_init = enforce_constraint(h_init, n)
    c5 = compute_c5(h_init)
    
    for _ in range(3):
        if time.time() > deadline:
            break
        corr = np.correlate(h_init, 1 - h_init, mode="full") * dx
        top_k = min(15, len(corr))
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        def objective(x):
            return x[-1]
        
        constraints = []
        for idx in top_idx:
            def constr(x, i=idx):
                h = x[:-1]
                t = x[-1]
                cv = np.correlate(h, 1 - h, mode="full")[i] * dx
                return t - cv
            constraints.append({'type': 'ineq', 'fun': constr})
        constraints.append({'type': 'eq', 'fun': lambda x: np.sum(x[:-1]) - n / 2.0})
        
        bounds = [(0, 1)] * n + [(0, None)]
        x0 = np.append(h_init, c5 + 1e-4)
        
        try:
            res = minimize(objective, x0, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 1000, 'ftol': 1e-10})
            h_new = enforce_constraint(res.x[:-1], n)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-10:
                h_init = h_new
                c5 = c5_new
        except Exception:
            pass
    return h_init, c5

def make_sin_start(n, freq):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    h = (1.0 + np.sin(freq * np.pi * x)) / 2.0
    return enforce_constraint(h, n)

def make_periodic_start(n, period):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    h = np.zeros(n)
    for i, xi in enumerate(x):
        frac = (xi / period) % 1.0
        if frac < 0.5:
            h[i] = 1.0
    return enforce_constraint(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = make_sin_start(47, 1)
    
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: Optimize at original resolution
    deadline1 = start_time + budget_s * 0.25
    h_opt, c5_opt = optimize_smooth(h0, n0, deadline1)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt
    h_opt, c5_opt = optimize_slsqp(best_h, n0, deadline1)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt
    
    # Phase 2: Try higher resolutions
    resolutions = sorted(set([n0 * 2, n0 * 3, 94, 128, 147, 180]))
    for n in resolutions:
        if time.time() > start_time + budget_s * 0.75:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        h_interp = np.interp(x_new, x_old, best_h)
        
        deadline2 = time.time() + (start_time + budget_s * 0.75 - time.time()) * 0.5
        h_opt, c5_opt = optimize_smooth(h_interp, n, deadline2)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt
            # SLSQP refinement at this resolution
            deadline3 = time.time() + 60
            h_opt, c5_opt = optimize_slsqp(h_opt, n, deadline3)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt
    
    # Phase 3: Random restarts with different initial patterns
    if time.time() < start_time + budget_s * 0.85:
        n_best = len(best_h)
        for restart in range(8):
            if time.time() > start_time + budget_s * 0.9:
                break
            # Mix of strategies
            if restart % 3 == 0:
                h_start = make_sin_start(n_best, np.random.choice([1, 2, 3, 4, 5]))
            elif restart % 3 == 1:
                h_start = make_periodic_start(n_best, 2.0 / np.random.choice([2, 3, 4, 5, 6]))
            else:
                h_start = best_h + np.random.randn(n_best) * 0.3
                h_start = enforce_constraint(h_start, n_best)
            
            deadline_r = min(time.time() + 60, start_time + budget_s * 0.9)
            h_opt, c5_opt = optimize_smooth(h_start, n_best, deadline_r)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt
                h_opt, c5_opt = optimize_slsqp(h_opt, n_best, min(time.time() + 30, start_time + budget_s * 0.92))
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt
    
    # Phase 4: Fine local search
    if time.time() < start_time + budget_s - 10:
        n_best = len(best_h)
        for it in range(30):
            if time.time() > start_time + budget_s - 5:
                break
            eps = 0.02 / (1 + it * 0.5)
            h_pert = best_h + np.random.randn(n_best) * eps
            h_pert = enforce_constraint(h_pert, n_best)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                # Quick refinement
                deadline_f = min(time.time() + 20, start_time + budget_s - 5)
                h_opt, c5_opt = optimize_smooth(h_pert, n_best, deadline_f)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt
    
    return best_h, best_c5, len(best_h)
