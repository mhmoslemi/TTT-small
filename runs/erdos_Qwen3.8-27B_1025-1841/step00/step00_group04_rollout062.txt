import time

            return h_opt, compute_c5(h_opt, n)

        pass

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_gradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = int(np.argmax(corr))
    grad = np.zeros(n)
    i_lo = max(0, n - 1 - m)
    i_hi = min(n - 1, 2 * n - 2 - m)
    for i in range(i_lo, i_hi + 1):
        j2 = i + m - n + 1
        grad[i] += (1.0 - h[j2])
        grad[j2] += (-h[i])
    return dx * grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = int(np.sum(mask))
        if count == 0:
            break
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(h, n):
    return compute_c5(h, n)

def jac_func(h, n):
    return c5_gradient(h, n)

def eq_fun(h, n):
    return np.sum(h) - n / 2.0

    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj_func, h_init, args=(n,), jac=jac_func,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_feasible(res.x, n)
        c5 = compute_c5(h_out, n)
        return h_out, c5
    except Exception:
        return h_init, compute_c5(h_init, n)

    h_up = np.interp(x2, x1, h)
    return project_feasible(h_up, n_to)

    h = project_feasible(h, n)
    for s in range(steps):

        # Project gradient to satisfy sum constraint
        g = g - np.mean(g)  # ensure sum of step is 0 (approximately)

            h_new = h - lr * g
            h_new = project_feasible(h_new, n)
        h = h_new

    return h

def run(seed=42, budget_s=1300, **kwargs):

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def c5_gradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = int(np.argmax(corr))
    grad = np.zeros(n)
    i_lo = max(0, n - 1 - m)
    i_hi = min(n - 1, 2 * n - 2 - m)
    for i in range(i_lo, i_hi + 1):
        j2 = i + m - n + 1
        grad[i] += (1.0 - h[j2])
        grad[j2] += (-h[i])
    return dx * grad


def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_func(h, n):
    return compute_c5(h, n)


def jac_func(h, n):
    return c5_gradient(h, n)


def eq_fun(h, n):
    return np.sum(h) - n / 2.0


def obj_pen(h, n, pen):
    c5 = compute_c5(h, n)
    s = np.sum(h) - n / 2.0
    return c5 + pen * s * s


def do_slsqp(h_init, n, maxiter=2000):
    h_init = project_feasible(h_init, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'args': (n,)}]
    try:
        res = minimize(obj_func, h_init, args=(n,), jac=jac_func,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            h_out = project_feasible(res.x, n)
            return h_out, compute_c5(h_out, n)
    except Exception:
        pass
    return h_init, compute_c5(h_init, n)


def do_lbfgsb(h_init, n, pen=500.0, maxiter=2000):
    h_init = project_feasible(h_init, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h_init, args=(n, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        if res.x is not None:
            h_out = project_feasible(res.x, n)
            return h_out, compute_c5(h_out, n)
    except Exception:
        pass
    return h_init, compute_c5(h_init, n)


def interp_up(h, n_from, n_to):
    x1 = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    x2 = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    h_up = np.interp(x2, x1, h)
    return project_feasible(h_up, n_to)


def gradient_step(h, n, lr, steps=300):
    h = project_feasible(h, n)
    best_local = compute_c5(h, n)
    for s in range(steps):
        g = c5_gradient(h, n)
        g = g - np.mean(g)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_local:
            best_local = c5_new
            h = h_new
        else:
            lr *= 0.7
            h_new = h - lr * g
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_local:
                best_local = c5_new
                h = h_new
    return h, best_local


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(42) * 0.5

    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c5 = compute_c5(best_h, n0)

    # Phase 1: SLSQP at initial resolution
    h_opt, c5v = do_slsqp(best_h, n0, maxiter=3000)
    if c5v < best_c5:
        best_c5, best_h = c5v, h_opt

    # Phase 2: L-BFGS-B with penalty
    if time.time() - t0 < budget_s * 0.2:
        h_opt, c5v = do_lbfgsb(best_h, n0, pen=500.0, maxiter=3000)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_opt

    # Phase 3: Gradient descent + SLSQP refinement
    if time.time() - t0 < budget_s * 0.3:
        h_gd, c5_gd = gradient_step(best_h, n0, lr=0.003, steps=1000)
        if c5_gd < best_c5:
            best_c5, best_h = c5_gd, h_gd
        h_opt, c5v = do_slsqp(best_h, n0, maxiter=2000)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_opt

    # Phase 4: Multiple random restarts at n0
    if time.time() - t0 < budget_s * 0.55:
        for i in range(12):
            if time.time() - t0 > budget_s * 0.55:
                break
            if i % 3 == 0:
                h_rand = np.random.uniform(0.0, 1.0, n0)
            elif i % 3 == 1:
                x = np.linspace(0, 2, n0)
                h_rand = 0.5 + 0.4 * np.sin(np.pi * x * (1 + np.random.randint(1, 5)) + np.random.uniform(0, 2 * np.pi))
            else:
                h_rand = best_h.copy()
                h_rand += np.random.normal(0, 0.1, n0)
            h_rand = project_feasible(h_rand, n0)
            h_opt, c5v = do_slsqp(h_rand, n0, maxiter=1000)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_opt

    # Phase 5: Higher resolutions
    res_list = [84, 126, 168, 252, 336]
    cur_n = len(best_h)
    for tn in res_list:
        if time.time() - t0 > budget_s * 0.85:
            break
        remaining = budget_s - (time.time() - t0)
        if tn > 200 and remaining < 120:
            break
        h_interp = interp_up(best_h, cur_n, tn)
        h_opt, c5v = do_slsqp(h_interp, tn, maxiter=1000)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_opt
            cur_n = tn
        elif time.time() - t0 > budget_s * 0.7:
            break

    # Phase 6: Final polish
    if time.time() - t0 < budget_s * 0.92:
        nb = len(best_h)
        h_opt, c5v = do_slsqp(best_h, nb, maxiter=3000)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_opt
        h_opt, c5v = do_lbfgsb(best_h, len(best_h), pen=1000.0, maxiter=2000)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_opt

    # Ensure exact feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
