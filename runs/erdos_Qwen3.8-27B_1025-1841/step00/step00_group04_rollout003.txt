def compute_c5_exact(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_simplex(z, S):
    lo = float(z.min()) - 1.5
    hi = float(z.max()) + 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s_val = float(np.clip(z - mid, 0.0, 1.0).sum())
        if s_val > S:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)

def grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s > 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    elif s < 0:
        sp = -s
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    else:
        g[:] = 1.0 - 2.0 * h
    return g
