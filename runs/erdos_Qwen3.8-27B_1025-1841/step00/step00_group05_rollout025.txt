def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def thue_morse_fast(n):
    i = np.arange(n, dtype=np.int64)

    h = h.copy()
    n = len(h)
    while np.sum(h) > target_ones:
        idx = rng.choice(np.where(h == 1.0)[0])
        h[idx] = 0.0

        idx = rng.choice(np.where(h == 0.0)[0])

    n_old = len(h_old)

    h_bin = (h_interp > 0.5).astype(float)

    best_c5 = current_c5

        current[j] = 1.0
