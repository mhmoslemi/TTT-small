import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, N):
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_lag(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        if lag == 0:
            g[:] = (1.0 - h) - h
        else:
            g[lag:] += (1.0 - h[:N - lag])
            g[:N - lag] -= h[lag:]
    else:
        L = -lag
        if L == 0:
            g[:] = (1.0 - h) - h
        else:
            g[:N - L] += (1.0 - h[L:])
            g[L:] -= h[:N - L]
    return g
