    dx = 2.0 / n

    for iteration in range(max_iters):
        best_improvement = 0
        best_idx = -1
        best_corr = None
        
        for p in range(n):
            if h[p] < 0.5:  # flip 0 → 1
                flip_val = 1.0
            else:  # flip 1 → 0
                flip_val = 0.0

            delta = np.zeros(2*n-1)
            old_val = h[p]
            factor = 1 - 2*old_val  # -1 if old=1, +1 if old=0
            
            # Case 1: first factor at position p
            # k range: 0 <= p+k-n+1 < n → n-1-p <= k < 2n-1-p

            k_max1 = min(2*n-2, 2*n-2-p)  # p+k-n+1 <= n-1 → k <= 2n-2-p

            k_max2 = min(2*n-2, p+n-1)  # i = p-k+n-1 < n → k > p-1

def smooth_obj(h, tau, pw):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = np.max(corr)
    s = mx + np.log(np.sum(np.exp(tau * (corr - mx)))) / tau
    tgt = n / 2.0
    s += pw * (np.sum(h) - tgt) ** 2
    return s
