```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_and_grad(h):
    """Compute all correlation values and the gradient matrix."""
    n = len(h)
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    # corr has length 2n-1
    # Positive lags: indices n-1 to 2n-2, lag = 0 to n-1
    # Negative lags: indices 0 to n-2, lag = -1 to -(n-1) (reversed)
    # grad[m, j] = d corr[m] / d h[j]
    
    grad = np.zeros((2*n-1, n))
    # Positive lags
    for k in range(n):
        idx = n - 1 + k
        if k == 0:
            grad[idx, :] = 1.0 - 2.0*h
        else:
            grad[idx, k:] += (1.0 - h[:n-k])
            grad[idx, :n-k] -= h[k:]
    # Negative lags
    for k in range(1, n):
        idx = n - 1 - k
        grad[idx, :n-k] += (1.0 - h[k:])
        grad[idx, k:] -= h[:n-k]
    
    return corr, grad
```
