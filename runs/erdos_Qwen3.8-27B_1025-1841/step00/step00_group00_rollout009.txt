            best_local = h.copy()
            patience = 0
        else:
            patience += 1
        
        if c5 < current_c5 or rng.random() < math.exp(-(c5 - current_c5) / max(T, 1e-15)):
            current_c5 = c5
        else:
            h[j], h[l] = old_j, old_l
        
        if patience > max_patience:
            patience = 0
            h = best_local.copy()
            current_c5 = best_local_c5
            T = max(T, T_start * 0.3)
    
    return best_local, best_local_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time()
    rng = np.random.default_rng(seed)
    
    h = initial_h_values.copy().astype(float)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n
    
    # Phase 1: SA at parent resolution
    t1 = min(budget_s * 0.30, time() - t0 + 300)
    h_opt, c5_opt = sa_optimize(h, n, rng, 0.015, 0.00005, t1 - t0)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
        best_n = n
    
    # Phase 2: Gradient polish at parent resolution
    if time() - t0 < budget_s * 0.35:
        h_g = best_h.copy()
        for it in range(5000):
            if time() - t0 > budget_s * 0.38:
                break
            lr = 0.01 * (1.0 - it / 5000) + 0.001
            h_new = gradient_step(h_g, best_n, lr)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h_g):
                h_g = h_new
            else:
                h_g = gradient_step(h_g, best_n, lr * 0.3)
                if compute_c5(h_g) > compute_c5(best_h):
                    h_g = best_h.copy()
        if compute_c5(h_g) < best_c5:
            best_c5 = compute_c5(h_g)
            best_h = h_g.copy()
            best_n = best_n
    
    # Phase 3: Multi-resolution SA
    resolutions = [n * 2, n * 4, n * 8]
    for ti, target_n in enumerate(resolutions):
        if time() - t0 > budget_s * 0.82:
            break
        time_avail = min(budget_s * 0.15, budget_s * 0.82 - (time() - t0))
        if time_avail < 5:
            break
        
        h_hi = upsample_linear(best_h, target_n)
        h_hi = fix_sum(h_hi, target_n)
        c5_hi = compute_c5(h_hi)
        
        h_hi_opt, c5_hi_opt = sa_optimize(h_hi, target_n, rng, 0.008, 0.00003, time_avail)
        
        # Gradient polish
        h_hi_opt = fix_sum(h_hi_opt, target_n)
        for it in range(2000):
            if time() - t0 > budget_s * 0.85:
                break
            lr = 0.005 * (1.0 - it / 2000) + 0.0005
            h_try = gradient_step(h_hi_opt, target_n, lr)
            if compute_c5(h_try) < compute_c5(h_hi_opt):
                h_hi_opt = h_try
        
        c5_final = compute_c5(h_hi_opt)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_hi_opt.copy()
            best_n = target_n
    
    # Phase 4: Final SA polish on best solution
    if time() - t0 < budget_s * 0.92:
        time_left = budget_s * 0.92 - (time() - t0)
        if time_left > 10:
            h_pol = best_h.copy()
            n_pol = best_n
            h_pol, c5_pol = sa_optimize(h_pol, n_pol, rng, 0.005, 0.00002, time_left)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_pol
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
