    current_sum = np.sum(h_clamped)
    if abs(current_sum - target_sum) < 1e-10:
        return h_clamped
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h_clamped - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h_clamped - tau, 0, 1)

def compute_gradient_smooth(h, w, n):
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n-1)
        if lag >= 0:
            if lag < n:
                grad[:n-lag] += wk * (1.0 - h[lag:])
                grad[lag:] -= wk * h[:n-lag]
        else:
            if -lag < n:
                grad[-lag:] += wk * (1.0 - h[:n+lag])
                grad[:n+lag] -= wk * h[-lag:]
    return grad

def lse_value_and_grad(h, dx, temp):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode="full")
    cs = c / temp
    mx = np.max(cs)
    exps = np.exp(cs - mx)
    s = np.sum(exps)
    value = temp * (mx + np.log(s)) * dx
    w = exps / s
    grad = compute_gradient_smooth(h, w, n) * dx
    return value, grad

def slsqp_run(h0, n, dx, temp, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target, 
             'jac': lambda h: np.ones(n)}]
    
    def fun(h):
        return lse_value_and_grad(h, dx, temp)[0]
    def jac(h):
        return lse_value_and_grad(h, dx, temp)[1]
    
    res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-7})
    h_opt = project_onto_feasible(res.x, target)
    return h_opt, compute_c5(h_opt, dx)

        w = e / float(np.sum(e))
        grad = grad_smooth_max(h, w, n)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = project_feasible(h - update, target)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            # Slowly decrease temperature
            temp = max(0.01, temp * (1 - 1e-6))
        else:
            stall += 1
            if stall > 1000:
                lr *= 0.5
                # Reset moments
                m = np.zeros(n)
                v = np.zeros(n)
                h = best_h.copy()
                stall = 0
                if lr < 1e-8:
                    break
        
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def project_feasible(h, target):
    hc = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(hc)) - target) < 1e-12:
        return hc
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(hc - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(hc - 0.5 * (lo + hi), 0.0, 1.0)


def grad_smooth_max(h, w, n):
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h[lag:])
            g[lag:] -= wk * h[:n - lag]
        else:
            g[-lag:] += wk * (1.0 - h[:n + lag])
            g[:n + lag] -= wk * h[-lag:]
    return g


def smooth_max_fg(h, dx, temp):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    cs = c / temp
    mx = float(np.max(cs))
    e = np.exp(cs - mx)
    s = float(np.sum(e))
    val = temp * (mx + np.log(s)) * dx
    w = e / s
    g = grad_smooth_max(h, w, n) * dx
    return val, g


def obj_fun(h, dx, temp):
    return smooth_max_fg(h, dx, temp)[0]


def obj_jac(h, dx, temp):
    return smooth_max_fg(h, dx, temp)[1]


def con_eq_fun(h):
    return float(np.sum(h)) - len(h) * 0.5


def con_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, dx, temp, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq_fun, 'jac': con_eq_jac}]
    res = minimize(obj_fun, h0, args=(dx, temp), jac=obj_jac,
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-9})
    h_out = project_feasible(res.x, n * 0.5)
    return h_out, compute_c5(h_out, dx)


def proj_grad_optimize(h0, dx, time_limit, temp_init=0.05):
    start = time.time()
    n = len(h0)
    target = n * 0.5
    h = project_feasible(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.03
    temp = temp_init
    stall = 0
    for _ in range(10000000):
        if time.time() - start > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        cs = c / temp
        mx = float(np.max(cs))
        e = np.exp(cs - mx)
        w = e / float(np.sum(e))
        grad = grad_smooth_max(h, w, n)
        momentum = beta * momentum + (1.0 - beta) * (-lr * grad)
        h_new = project_feasible(h + momentum, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            temp = max(0.008, temp * 0.9998)
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                momentum = np.zeros(n)
                h = best_h.copy()
                stall = 0
                if lr < 1e-9:
                    break
        h = h_new
    return best_h, best_c5


def upsample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)


def perturbed_h(h, sigma, rng):
    return project_feasible(h + sigma * rng.randn(len(h)), float(np.sum(h)))
