import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(v, target):
    lo, hi = -np.max(v) - 1.0, 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    e = np.exp(alpha * (scaled - m))
    Z = e.sum()
    obj = m + np.log(Z) / alpha
    w = dx * (e / Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        nlo = max(0, -s)
        nhi = min(n - 1, n - 1 - s)
        j1lo = max(nlo + s, 0)
        j1hi = min(nhi + s, n - 1)
        if j1lo <= j1hi:
            idx = np.arange(j1lo, j1hi + 1)
            grad[idx] += wk * (1.0 - h[idx - s])
        j2lo = max(nlo, -s)
        j2hi = min(nhi, n - 1 - s)
        if j2lo <= j2hi:
            idx = np.arange(j2lo, j2hi + 1)
            grad[idx] -= wk * h[idx + s]
    return obj, grad

def upsample_h(h, n_new):
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def optimize_pgd(h_init, n, time_limit, t0, budget_s):
    target = n / 2.0
    h = project_sum(h_init.copy(), target)
    v = np.zeros(n)
    beta = 0.85
    alpha = 25.0
    best_h = h.copy()
    best_c = c5_value(h)
    t_start = time.time()
    it = 0
    while it < 20000:
        if time.time() - t_start > time_limit:
            break
        if time.time() - t0 > budget_s - 50:
            break
        obj, g = obj_grad(h, alpha)
        lr = 0.05
        improved = False
        for _ in range(8):
            v = beta * v - lr * g
            h_try = project_sum(h + v, target)
            obj_try, _ = obj_grad(h_try, alpha)
            if obj_try < obj:
                improved = True
                break
            lr *= 0.5
            v = beta * v - lr * g
        if improved:
            c_try = c5_value(h_try)
            h = h_try
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
        else:
            pass
        if it % 400 == 399:
            alpha = min(alpha * 1.4, 300.0)
            beta = min(beta * 0.99, 0.92)
            lr = lr * 0.7
        it += 1
    # Coordinate descent refinement
    c_best = best_c

                h[i] += eps; h[j] -= eps
            else:

        else:
            d = np.random.uniform(-step, step)
            h[i] += d; h[j] -= d
        h = project_sum(h, target)
        cn = c5_value(h)
        if cn < c_best:
            c_best = cn
            best_h = h.copy()
    return best_h, c_best
