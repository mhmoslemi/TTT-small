    # Let's just return the objective and let SLSQP use FD
    return val

```python
def obj_smooth(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = corr - max_c  # for numerical stability
    return max_c + (1.0/beta) * np.log(np.sum(np.exp(beta * shifted)))

def grad_smooth(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_c = np.max(corr)
    shifted = corr - max_c
    weights = np.exp(beta * shifted)
    weights /= np.sum(weights)
    
    # corr[m] = sum_n h[n+k_m] * (1-h[n]) where k_m = m - (n-1)
    # ∂corr[m]/∂h[j] = (1-h[j-k_m]) * [valid] - h[j+k_m] * [valid]
    
    grad = np.zeros(n)
    k_values = np.arange(2*n-1) - (n-1)  # k_m for m=0,...,2n-2
    
    for m in range(2*n-1):
        k = k_values[m]
        w = weights[m] * dx
        # Term 1: j = n_idx + k, contributing (1-h[n_idx])
        # n_idx valid: 0 <= n_idx < n, 0 <= n_idx+k < n
        # j = n_idx + k, so 0 <= j < n, 0 <= j-k < n
        # j ranges from max(0,k) to min(n-1, n-1+k)
        j_start = max(0, k)
        j_end = min(n-1, n-1+k)
        if j_start <= j_end:
            n_indices = np.arange(j_start - k, j_end - k + 1)
            grad[j_start:j_end+1] += w * (1.0 - h[n_indices])
        
        # Term 2: j = n_idx, contributing -h[j+k]
        # n_idx valid: 0 <= n_idx < n, 0 <= n_idx+k < n
        # j = n_idx, so 0 <= j < n, 0 <= j+k < n
        j_start2 = max(0, -k)
        j_end2 = min(n-1, n-1-k)
        if j_start2 <= j_end2:
            grad[j_start2:j_end2+1] += w * (-h[j_start2+k:j_end2+k+1])
    
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, beta=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    s = np.exp(beta * (corr - mc))
    return mc + np.log(s.sum()) / beta

def grad_smooth(h, beta=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        wm = w[m] * dx
        j_lo = max(0, -k)
        j_hi = min(n - 1, n - 1 - k)
        if j_lo > j_hi:
            continue
        a_lo = max(0, j_lo + k)
        a_hi = min(n - 1, j_hi + k)
        if a_lo <= a_hi:
            grad[a_lo:a_hi + 1] += wm * (1.0 - h[a_lo - k:a_hi - k + 1])
        grad[j_lo:j_hi + 1] += wm * (-h[j_lo + k:j_hi + k + 1])
    return grad

def project_feasible(h, n_pts):
    h = np.clip(h, 0.0, 1.0)
    target = n_pts / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_random_h(n_pts, rng):
    h = rng.dirichlet(np.ones(n_pts))  # sums to 1
    h = h * n_pts  # now sums to n_pts, scale to n_pts/2
    h = h * 0.5
    # Actually let me just do uniform and adjust
    h = rng.uniform(0.2, 0.8, n_pts)
    h = project_feasible(h, n_pts)
    return h

def make_binary_h(n_pts, rng):
    idx = rng.permutation(n_pts)
    h = np.zeros(n_pts)
    h[idx[:n_pts // 2]] = 1.0
    return h

    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    return np.interp(x_new, x_old, h)

    return h.sum() - len(h) / 2.0

def sum_jac(h):
    return np.ones(len(h))

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    s = np.exp(beta * (corr - mc))
    return mc + np.log(np.sum(s)) / beta

def grad_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= np.sum(w)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        if n_lo > n_hi:
            continue
        wm = w[m] * dx
        a_lo = max(0, n_lo + k)
        a_hi = min(n - 1, n_hi + k)
        if a_lo <= a_hi:
            grad[a_lo:a_hi + 1] += wm * (1.0 - h[a_lo - k:a_hi - k + 1])
        grad[n_lo:n_hi + 1] += wm * (-h[n_lo + k:n_hi + k + 1])
    return grad

def project_feasible(h, n_pts):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n_pts / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_random_h(n_pts, rng):
    h = rng.uniform(0.1, 0.9, n_pts)
    return project_feasible(h, n_pts)

def make_binary_h(n_pts, rng):
    idx = rng.permutation(n_pts)
    h = np.zeros(n_pts)
    h[idx[:n_pts // 2]] = 1.0
    return h

def make_block_h(n_pts, rng):
    h = np.zeros(n_pts)
    n_blocks = rng.randint(2, 5)
    block_len = n_pts * 0.5 // n_blocks
    positions = rng.choice(n_pts, size=n_blocks, replace=False)
    for p in positions:
        h[p:min(p + block_len, n_pts)] = 1.0
    return project_feasible(h, n_pts)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def sum_fun(h):
    return h.sum() - len(h) / 2.0

def sum_jac(h):
    return np.ones(len(h))

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_best = project_feasible(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    except NameError:
        h_best = np.full(80, 0.5)

    n_best = len(h_best)
    c5_best = evaluate_c5(h_best)
    beta = 80.0
    base_n = n_best

    resolutions = sorted(set([base_n, 2 * base_n, 4 * base_n, 8 * base_n]))
    resolutions = [r for r in resolutions if r <= 1600]

    for n_pts in resolutions:
        if time.time() - t0 > budget_s * 0.75:
            break

        starts = []
        starts.append(upsample(h_best, n_pts))
        h_noisy = upsample(h_best, n_pts) + rng.normal(0, 0.08, n_pts)
        starts.append(project_feasible(h_noisy, n_pts))
        starts.append(make_random_h(n_pts, rng))
        starts.append(make_binary_h(n_pts, rng))
        starts.append(make_block_h(n_pts, rng))
        if n_pts <= 320:
            starts.append(make_random_h(n_pts, rng))
            h_s = upsample(h_best, n_pts) + rng.normal(0, 0.15, n_pts)
            starts.append(project_feasible(h_s, n_pts))

        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

        remaining = budget_s * 0.75 - (time.time() - t0)
        per_opt = remaining / max(1, len(starts))

        if n_pts <= 128:
            maxiter = min(800, max(200, int(per_opt * 4)))
        elif n_pts <= 256:
            maxiter = min(500, max(100, int(per_opt * 3)))
        elif n_pts <= 512:
            maxiter = min(300, max(60, int(per_opt * 2)))
        else:
            maxiter = min(150, max(30, int(per_opt)))

        for si in range(len(starts)):
            if time.time() - t0 > budget_s * 0.8:
                break

            h_start = starts[si].copy()
            try:
                res = minimize(
                    lambda x: obj_smooth(x, beta),
                    h_start,
                    jac=lambda x: grad_smooth(x, beta),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-12}
                )
                h_opt = project_feasible(res.x, n_pts)
                c5_opt = evaluate_c5(h_opt)
                if c5_opt < c5_best:
                    h_best = h_opt
                    c5_best = c5_opt
                    n_best = n_pts
            except Exception:
                pass

        # L-BFGS-B with penalty
        if time.time() - t0 < budget_s * 0.65 and n_pts <= 640:
            h_lb = upsample(h_best, n_pts)
            try:
                def obj_pen(x):
                    return obj_smooth(x, beta) + 50.0 * (x.sum() - n_pts / 2.0) ** 2
                def grad_pen(x):
                    pen_term = 100.0 * (x.sum() - n_pts / 2.0)
                    return grad_smooth(x, beta) + pen_term * np.ones(n_pts)
                res2 = minimize(obj_pen, h_lb, jac=grad_pen,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-14})
                h_opt2 = project_feasible(res2.x, n_pts)
                c5_opt2 = evaluate_c5(h_opt2)
                if c5_opt2 < c5_best:
                    h_best = h_opt2
                    c5_best = c5_opt2
                    n_best = n_pts
            except Exception:
                pass

    # Post: binary projections and threshold sweep
    n_final = n_best
    for thresh in [0.25, 0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]:
        h_b = (h_best > thresh).astype(float)
        h_b = project_feasible(h_b, n_final)
        c5_b = evaluate_c5(h_b)
        if c5_b < c5_best:
            h_best = h_b
            c5_best = c5_b

    # Local random search

        for _ in range(200):
            if time.time() - t0 > budget_s * 0.9:
                break
            h_try = h_best.copy()
            idx = rng.randint(0, len(h_try))
            h_try[idx] = np.clip(h_try[idx] + rng.uniform(-0.2, 0.2), 0, 1)
            h_try = project_feasible(h_try, n_final)
            c5_try = evaluate_c5(h_try)
            if c5_try < c5_best - 1e-10:
                h_best = h_try
                c5_best = c5_try

    h_best = project_feasible(h_best, n_final)
    c5_final = evaluate_c5(h_best)
    return h_best, c5_final, n_final
