import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def soft_max_obj(h, tau=0.01):
    c = compute_c5_vec(h)
    c = c - np.max(c)  # numerical stability
    return float(np.max(c) + tau * np.log(np.sum(np.exp(c / tau))))

def soft_max_grad(h, tau=0.01):
    c = compute_c5_vec(h)
    c_max = np.max(c)
    c_shift = c - c_max
    weights = np.exp(c_shift / tau)
    weights = weights / np.sum(weights)
    # Gradient of sum_k w_k * c_k(h) where w_k depends on h
    # For simplicity, use the weighted average of individual gradients
    # d/dh [w_k * c_k] ≈ w_k * dc_k/dh (ignoring dw_k/dh for approximation)
    grad = np.zeros(len(h))
    n = len(h)

        if k >= 0:
            valid = np.arange(n - k)
            grad[valid + k] += weights[idx] * (1.0 - h[valid])
            grad[valid] -= weights[idx] * h[valid + k]
        else:
            valid = np.arange(-k, n)
            grad[valid + k] += weights[idx] * (1.0 - h[valid])
            grad[valid] -= weights[idx] * h[valid + k]
