    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_soft_subgrad(h, n, eps):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    w = np.exp((vals - m) / eps)
    w /= np.sum(w)
    subgrad = np.zeros(n)
    nm1 = n - 1
    for li in range(2 * n - 1):
        wi = w[li]
        if wi < 1e-10:
            continue
        k = li - nm1
        if k == 0:
            subgrad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            subgrad[k:] += wi * (1.0 - h[:n - k])
            subgrad[:n - k] -= wi * h[k:]
        else:
            kk = -k
            subgrad[:n - kk] += wi * (1.0 - h[kk:])
            subgrad[kk:] -= wi * h[:n - kk]
    subgrad *= dx
    return subgrad

def try_slsqp(start, n, bounds, cons, obj_func, maxiter, ftol):
    try:
        res = minimize(obj_func, start, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return start

def make_starts(h_base, n, rng):
    starts = [h_base.copy()]
    for scale in [0.02, 0.05, 0.1, 0.2, 0.35]:
        s = h_base + rng.randn(n) * scale
        starts.append(project_feasible(s, n))
    for _ in range(4):
        s = (rng.rand(n) < 0.5).astype(float)
        n_ones = int(np.sum(s))
        target = n // 2
        if n_ones < target:
            idx = np.where(s == 0)[0]
            rng.shuffle(idx)
            for j in range(target - n_ones):
                s[idx[j]] = 1.0
        elif n_ones > target:
            idx = np.where(s == 1)[0]
            rng.shuffle(idx)
            for j in range(n_ones - target):
                s[idx[j]] = 0.0
        starts.append(s.copy())
    for block in [2, 3, 4, 6]:
        s = np.zeros(n)
        for i in range(0, n, block):
            s[i:i + block // 2] = 1.0
        starts.append(project_feasible(0.3 + 0.4 * s, n))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values, dtype=float)
    else:
        h_base = np.ones(80) * 0.5

    n = len(h_base)
    best_h = h_base.copy()
    best_c5 = compute_c5(best_h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP with soft objective from diverse starts
    starts = make_starts(h_base, n, rng)
    eps_soft = 0.02

    def obj_soft(x):
        return soft_c5(x, n, eps_soft)

    for i, s in enumerate(starts):
        if time.time() - t0 > budget_s * 0.35:
            break
        rh = try_slsqp(s, n, bounds, cons, obj_soft, 600, 1e-10)
        c5 = compute_c5(rh, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = rh.copy()

    # Phase 2: Projected subgradient descent
    if time.time() - t0 < budget_s * 0.75:
        h_curr = best_h.copy()
        lr = 0.003
        no_improve = 0
        eps_sg = 0.15
        for it in range(30000):
            if time.time() - t0 > budget_s * 0.8:
                break
            sg = compute_soft_subgrad(h_curr, n, eps_sg)
            h_new = project_feasible(h_curr - lr * sg, n)
            c5_new = compute_c5(h_new, n)
            if c5_new <= compute_c5(h_curr, n) + 1e-12:
                h_curr = h_new
                no_improve = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                no_improve += 1
                lr *= 0.97
                if lr < 1e-6:
                    break
            if no_improve > 300:
                h_curr = project_feasible(h_curr + rng.randn(n) * 0.03, n)
                lr = 0.003
                no_improve = 0

    # Phase 3: SLSQP with hard objective (polish)
    if time.time() - t0 < budget_s * 0.9:
        def obj_hard(x):
            return compute_c5(x, n)
        rh = try_slsqp(best_h, n, bounds, cons, obj_hard, 800, 1e-12)
        c5 = compute_c5(rh, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = rh.copy()

    # Phase 4: Try coarser resolution n2=40
    if time.time() - t0 < budget_s * 0.93 and best_c5 > 0.38:
        n2 = 40
        h2 = np.zeros(n2)
        for i in range(n2):
            h2[i] = np.mean(h_base[i * n // n2:(i + 1) * n // n2])
        h2 = project_feasible(h2, n2)
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n2 / 2.0,
                  'jac': lambda x: np.ones(n2)}]
        starts2 = make_starts(h2, n2, rng)
        for s in starts2[:6]:
            if time.time() - t0 > budget_s * 0.95:
                break
            rh2 = try_slsqp(s, n2, bounds2, cons2,
                            lambda x: soft_c5(x, n2, 0.02), 400, 1e-10)
            c5_2 = compute_c5(rh2, n2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = rh2.copy()
                n = n2

    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
