            if 0 <= m + lag < n:

            if 0 <= m + l < n:  # m-lag = m+l

            if 0 <= m - l < n:  # m+lag = m-l
                g -= h[m - l]

    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1.0 - h[idx1[mask1]])
    # Second term: -h[m+lag]*I(0<=m+lag<n)

    mask2 = (idx2 >= 0) & (idx2 < n)

    # Project onto polytope

    return h_new, float(np.max(c) * dx)
