import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, np.argmax(corr)

def project_to_feasible(v, target_sum):
    """Project v onto {h: 0 <= h[i] <= 1, sum(h) = target_sum}"""
    n = len(v)
    # Binary search for lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def subgradient_step(h, step_size, alpha=1.0):
    """One subgradient step with smoothing over top lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Use top few lags with softmax weighting
    top_k = min(5, len(corr))
    max_corr = np.max(corr)
    # Softmax over the corr values (scaled)
    scaled = (corr - max_corr) / (0.01 * dx * n)  # smoothing parameter
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    # Compute weighted subgradient
    grad = np.zeros(n)
    for lag_idx, w in enumerate(weights):
        if w < 1e-10:
            continue
        s = lag_idx - (n - 1)  # actual shift
        for j in range(n):
            if j - s >= 0:
                grad[j] += w * (1.0 - h[j - s])
            if j + s < n:
                grad[j] -= w * h[j + s]
    
    return grad

        grad = subgradient_step(h, step_size)
        h_new = project_to_feasible(h - step_size * grad, target_sum)
        c_new = compute_c5(h_new)[0]

            h = h_new

def get_subgrad(h, top_k=3, smooth=0.01):

def project_feasible(v, target_sum):
    v = np.clip(v, -3.0, 4.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def get_smoothed_grad(h, top_k=5, temp=1.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = float(np.max(corr))
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(corr)[-k:]
    # Scale for softmax: we want the weights to be meaningful
    # corr values are O(n), so temp should be O(n)
    w = np.exp((corr[top_idx] - max_c) / temp)

        w = np.zeros(k)
        w[-1] = 1.0

        w /= w_sum

        if wt < 1e-10:
            continue
        shift = int(idx) - (n - 1)

                grad[shift:] += wt * (1.0 - h[:n - shift])
                grad[:n - shift] -= wt * h[shift:]
        else:
            s = -shift
            if s < n:
                grad[:n - s] += wt * (1.0 - h[s:])
                grad[s:] -= wt * h[:n - s]
    return grad
