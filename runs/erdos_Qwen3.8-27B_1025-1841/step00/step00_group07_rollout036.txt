import numpy as np
from scipy.optimize import minimize
import time

_g_n = 80
_g_T = 0.01

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h):
    n = _g_n
    T = _g_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + T * np.log(np.sum(np.exp((corr - m) / T)) / len(corr))

def con_fun(h):
    return np.sum(h) - _g_n / 2.0

def con_fun_jac(h):
    return np.ones(_g_n)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(h)
            remaining = diff
            for i in order:
                add = min(remaining, 1.0 - h[i])
                h[i] += add
                remaining -= add
                if remaining < 1e-12:
                    break
        else:
            order = np.argsort(h)[::-1]
            remaining = -diff
            for i in order:
                rem = min(remaining, h[i])
                h[i] -= rem
                remaining -= rem
                if remaining < 1e-12:
                    break
    return h

def optimize_single(h0, n, T, maxiter=200):
    global _g_n, _g_T
    _g_n = n
    _g_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_fun_jac}]
    try:
        res = minimize(smooth_obj, h0.copy(), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-10})
        return make_feasible(res.x, n)
    except Exception:
        return h0
