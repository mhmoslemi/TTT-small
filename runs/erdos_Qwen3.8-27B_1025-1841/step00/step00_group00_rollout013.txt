def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)

        # Distribute proportionally

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def make_binary(h, n):
    """Convert to a binary-like sequence with sum = n/2."""
    target = n // 2
    h_sorted_indices = np.argsort(-h)
    h_new = np.zeros(n)
    h_new[h_sorted_indices[:target]] = 1.0
    if n % 2 == 1:
        h_new[h_sorted_indices[target]] = 0.5
    return h_new

def sa_optimize(h, n, time_limit, rng):
    """Simulated annealing for binary-ish sequence."""
    h = h.astype(float).copy()
    
    # Make it binary (or close to binary)
    if n % 2 == 0:
        target = n // 2
        h = make_binary(h, n)
    else:
        target = n // 2
        h = make_binary(h, n)  # This gives (n-1)/2 ones, one 0.5

    best_val = compute_c5(h, n)
    cur_val = best_val
