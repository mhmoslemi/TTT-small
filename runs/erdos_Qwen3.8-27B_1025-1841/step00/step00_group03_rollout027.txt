- "i+l valid" (j = i+l is valid): 0 ≤ i+l < n and 0 ≤ i < n
  Simplified: max(-l, 0) ≤ i ≤ min(n-1-l, n-1)

```python
def compute_subgradient(h, l):
    """Compute subgradient of corr at lag l with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        # First term: (1-h[i-l]) if i is valid as j
        if i - l >= 0 and i - l < n:
            val += (1.0 - h[i - l])
        # Second term: -h[i+l] if i+l is valid as j
        if i + l >= 0 and i + l < n:
            val -= h[i + l]
        grad[i] = val
    return grad
```

```python
    n = len(h)
    h = (h >= 0.5).astype(float)
    n_ones = int(h.sum())
    target_ones = n // 2
    
    # Fix if needed
    if n_ones != target_ones:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if n_ones > target_ones:
            for idx in ones_idx[(target_ones):]:
                h[idx] = 0
        else:
            for idx in zeros_idx[:(target_ones - n_ones)]:
                h[idx] = 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)
    best_c5 = current_max * dx
    start_t = time.time()
    
    while time.time() - start_t < time_limit_s:
        M = np.max(corr)
        active = np.where(corr == M)[0]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_swap = None
        best_new_max = M
        
        for i in ones:
            for j in zeros:
                # Compute Δcorr[m] for m in active
                improved = True
                new_max_candidate = M
                for m in active:
                    l = m - (n - 1)
                    delta = 0.0
                    # h[i]: 1→0
                    if 0 <= i - l < n:
                        delta -= (1.0 - h[i - l])
                    if 0 <= i + l < n:
                        delta += h[i + l]
                    # h[j]: 0→1
                    if 0 <= j - l < n:
                        delta += (1.0 - h[j - l])
                    if 0 <= j + l < n:
                        delta -= h[j + l]
                    if corr[m] + delta >= M and delta >= 0:
                        improved = False
                        break
                    new_max_candidate = max(new_max_candidate, corr[m] + delta)
                
                if improved:
                    # Check full max (approximate: just check active set gives decrease)
                    if new_max_candidate < best_new_max:
                        best_new_max = new_max_candidate
                        best_swap = (i, j)
        
        if best_swap is None:
            # Try any swap that reduces the max (relax condition)
            for i in ones:
                for j in zeros:
                    h[i], h[j] = h[j], h[i]
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
```

```python
    target = n / 2.0
    lo, hi = -2.0, 2.0

        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
```
