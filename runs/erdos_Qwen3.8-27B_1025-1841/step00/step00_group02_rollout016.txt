def compute_c5(h):

def upsample_h(h, n_out):

    if n_in == n_out:
        return h.astype(float).copy()
    x_in = np.arange(n_in, dtype=float) * (2.0 / n_in)
    x_out = np.arange(n_out, dtype=float) * (2.0 / n_out)
    return np.interp(x_out, x_in, h)

def project_feasible(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

        if abs(diff) < 1e-14:

            h = np.clip(h, 0.0, 1.0)

def c5_and_grad(h, n, temp):
    """Compute smooth max of correlations and its gradient w.r.t. h."""
    h_p = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h_p, 1.0 - h_p, mode="full")
    corr_s = corr * dx
    v = corr_s / temp
    m = np.max(v)
    ev = np.exp(v - m)
    se = np.sum(ev)
    obj = temp * (m + np.log(se))
    
    gc = np.zeros(n)
    for s in range(-(n-1), n):
        k = s + n - 1
        w = ev[k]
        if w < 1e-18:
            continue

            gc[:n-t] -= w * h_p[t:]
        else:
            gc += w * (1.0 - 2.0 * h_p)
    
    grad = gc * (dx / se)
    return obj, grad

    best_h = project_feasible(h0, target)
    best_obj = compute_c5(best_h)
    
    # Decreasing temperature schedule
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]

            c, g = c5_and_grad(h_c, n, temp)

            return np.sum(x) - target

            return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upsample_h(h, n_out):
    n_in = len(h)
    if n_in == n_out:
        return h.astype(float).copy()
    x_in = np.arange(n_in, dtype=float) * (2.0 / n_in)
    x_out = np.arange(n_out, dtype=float) * (2.0 / n_out)
    return np.interp(x_out, x_in, h)

def project_feasible(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_data(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max_val(corr_s, temp):
    v = corr_s / temp
    m = np.max(v)
    ev = np.exp(v - m)
    se = np.sum(ev)
    return temp * (m + np.log(se)), ev, se

def grad_smooth_max(h_p, n, ev, se, dx):
    gc = np.zeros(n)
    for s in range(-(n - 1), n):
        k = s + n - 1
        w = ev[k]
        if w < 1e-18:
            continue
        if s > 0:
            gc[:n - s] += w * (1.0 - h_p[s:])
            gc[s:] -= w * h_p[:n - s]
        elif s < 0:
            t = -s
            gc[t:] += w * (1.0 - h_p[:n - t])
            gc[:n - t] -= w * h_p[t:]
        else:
            gc += w * (1.0 - 2.0 * h_p)
    return gc * (dx / se)

def obj_grad(x, n, temp, target, pen_c):
    h_p = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr_s = corr_data(h_p, n)
    val, ev, se = smooth_max_val(corr_s, temp)
    g = grad_smooth_max(h_p, n, ev, se, dx)
    if pen_c > 0:
        s_diff = np.sum(h_p) - target
        val += pen_c * s_diff * s_diff
        g += 2.0 * pen_c * s_diff
    return val, g

def local_refine(h, n, target, max_iter=2000):
    """Local coordinate descent refinement."""
    h = project_feasible(h, target)
    dx = 2.0 / n
    c5 = compute_c5(h)
    step = 0.02
    for it in range(max_iter):
        improved = False
        i = it % n
        for delta in [step, -step, step * 0.5, -step * 0.5]:
            h_try = h.copy()
            h_try[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5 - 1e-12:
                h = h_try
                c5 = c5_try
                improved = True
                break
        if not improved:
            step *= 0.9
            if step < 1e-5:
                break
    return h, c5

def optimize_single(h0, n, target, time_limit=25.0, t_start=None):
    if t_start is None:
        t_start = time.time()
    deadline = t_start + time_limit
    h_curr = project_feasible(h0.copy(), target)
    best_h = h_curr.copy()
    best_c5 = compute_c5(best_h)
    temps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    for temp in temps:
        if time.time() > deadline:
            break
        def obj_fn(x):
            v, _ = obj_grad(x, n, temp, target, 50.0)
            return v
        def jac_fn(x):
            _, g = obj_grad(x, n, temp, target, 50.0)
            return g
        try:
            res = minimize(obj_fn, h_curr, jac=jac_fn, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 300, 'maxfun': 1000, 'ftol': 1e-12})
            h_new = project_feasible(np.clip(res.x, 0, 1), target)
        except Exception:
            h_new = h_curr.copy()
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new.copy()
    h_ref, c5_ref = local_refine(best_h, n, target, max_iter=500)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref
    return best_h, best_c5
