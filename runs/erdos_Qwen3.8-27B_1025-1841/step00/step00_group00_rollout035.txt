from time import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def c5_func_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    
    # Compute gradient
    grad = np.zeros(n)
    # For m in range(n):
    # grad[m] = (1-h[m-shift]) * I[0<=m-shift<n] - h[m+shift] * I[0<=m+shift<n]
    
    mask1 = (np.arange(n) - shift) >= 0
    mask1 &= (np.arange(n) - shift) < n
    mask2 = (np.arange(n) + shift) >= 0
    mask2 &= (np.arange(n) + shift) < n
    
    idx1 = np.arange(n)[mask1]
    grad[idx1] += (1 - h[idx1 - shift])
    
    idx2 = np.arange(n)[mask2]
    grad[idx2] -= h[idx2 + shift]
    
    return np.max(corr) * dx, grad * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_objective(h):
    return compute_c5(h)

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    
    m = np.arange(n)
    mask1 = (m - shift >= 0) & (m - shift < n)
    mask2 = (m + shift >= 0) & (m + shift < n)
    
    grad = np.zeros(n)
    grad[mask1] += (1 - h[m[mask1] - shift])
    grad[mask2] -= h[m[mask2] + shift]
    
    return grad * dx

def fix_sum(h, n):
    """Fix the sum constraint: sum(h) = n/2, with 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 0:
                h += diff * room / total
        else:
            total = h.sum()
            if total > 0:
                h += diff * h / total
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Add/subtract from a single element
        if diff > 0:
            idx = np.argmax(1 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def upsample_h(h, new_n):

    old_n = len(h)
    old_x = np.arange(old_n) / old_n * 2.0
    new_x = np.arange(new_n) / new_n * 2.0
    new_h = np.interp(new_x, old_x, h)
    return fix_sum(new_h, new_n)

    grad[mask2] -= h[m[mask2] + shift]
    return grad * dx

def c5_smooth_func(h, tau=30.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    val = mx + (1.0 / tau) * np.log(np.sum(np.exp(tau * (corr - mx))))
    return val * dx

def c5_smooth_grad(h, tau=30.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    exponents = tau * (corr - mx)
    exponents -= np.max(exponents)
    weights = np.exp(exponents)
    weights /= weights.sum()
    m = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        shift = k_idx - (n - 1)
        mask1 = (m - shift >= 0) & (m - shift < n)
        mask2 = (m + shift >= 0) & (m + shift < n)
        grad[mask1] += w * (1.0 - h[m[mask1] - shift])
        grad[mask2] -= w * h[m[mask2] + shift]
    return grad * dx

def fix_sum(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t > 1e-10:
                h += (diff / t) * room
        else:
            t = h.sum()
            if t > 1e-10:
                h += (diff / t) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.arange(old_n) * 2.0 / old_n
    new_x = np.arange(new_n) * 2.0 / new_n
    new_h = np.interp(new_x, old_x, h)
    return fix_sum(new_h, new_n)

def optimize_slsqp(h_init, n, max_iter=500, ftol=1e-10, use_smooth=False):
    global _g_n
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    if use_smooth:
        def func(h):
            return c5_smooth_func(h)
        def jac(h):
            return c5_smooth_grad(h)
    else:
        def func(h):
            return compute_c5(h)
        def jac(h):
            return c5_gradient(h)
    res = minimize(
        func, h_init, jac=jac, method='SLSQP',
        bounds=bounds, constraints=[{'type': 'eq', 'fun': _constraint_func}],
        options={'maxiter': max_iter, 'ftol': ftol}
    )
    h_opt = fix_sum(res.x, n)
    return h_opt, compute_c5(h_opt)

def pair_local_search(h, n, n_pairs=200, max_steps=50, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    c5_cur = compute_c5(h)
    for _ in range(max_steps):
        improved = False
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            best_delta = 0.0
            best_val = c5_cur
            for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                if h[j] - delta < 0 or h[j] - delta > 1:
                    continue
                h[i] += delta
                h[j] -= delta
                val = compute_c5(h)
                h[i] -= delta
                h[j] += delta
                if val < best_val:
                    best_val = val
                    best_delta = delta
            if best_delta != 0.0:
                h[i] += best_delta
                h[j] -= best_delta
                h = np.clip(h, 0.0, 1.0)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time()
    rng = np.random.RandomState(seed)

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def remaining():
        return budget_s - (time() - t0)

    h_work = h_best.copy()

    # Optimization at increasing resolutions
    for n in [n_best, n_best * 2, n_best * 3, n_best * 4]:
        if remaining() < 45:
            break
        h_work = upsample_h(h_work, n)

        # Main SLSQP run
        if remaining() > 30:
            try:
                h_opt, val = optimize_slsqp(h_work, n, max_iter=400, ftol=1e-10)
                if val < c5_best:
                    h_best = h_opt.copy()
                    c5_best = val
                    h_work = h_opt.copy()
            except Exception:
                pass

        # Smooth max SLSQP
        if remaining() > 30:
            try:
                h_opt, val = optimize_slsqp(h_work, n, max_iter=300, ftol=1e-10, use_smooth=True)
                if val < c5_best:
                    h_best = h_opt.copy()
                    c5_best = val
                    h_work = h_opt.copy()
            except Exception:
                pass

        # Random restarts
        n_restarts = 4 if n <= 100 else 3 if n <= 200 else 2
        for r in range(n_restarts):
            if remaining() < 30:
                break
            noise = 0.2 / (r + 1)
            h_start = fix_sum(h_work + rng.randn(n) * noise, n)
            try:
                h_opt, val = optimize_slsqp(h_start, n, max_iter=250, ftol=1e-10)
                if val < c5_best:
                    h_best = h_opt.copy()
                    c5_best = val
                    h_work = h_opt.copy()
            except Exception:
                pass

        # Pair local search
        if remaining() > 60 and n <= 200:
            h_ps, val = pair_local_search(h_best.copy(), len(h_best), n_pairs=300, max_steps=30, rng=rng)
            if val < c5_best:
                h_best = h_ps
                c5_best = val
                h_work = h_ps.copy()

    # Final polish at best resolution
    if remaining() > 60:
        n_pol = len(h_best)
        try:
            h_opt, val = optimize_slsqp(h_best, n_pol, max_iter=800, ftol=1e-12)
            if val < c5_best:
                h_best = h_opt
                c5_best = val
        except Exception:
            pass
        if remaining() > 40 and n_pol <= 200:
            h_ps, val = pair_local_search(h_best.copy(), n_pol, n_pairs=500, max_steps=50, rng=rng)
            if val < c5_best:
                h_best = h_ps
                c5_best = val

    c5_best = compute_c5(h_best)
    return (h_best, c5_best, len(h_best))
