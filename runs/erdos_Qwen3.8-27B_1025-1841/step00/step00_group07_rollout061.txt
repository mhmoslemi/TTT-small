from scipy.optimize import minimize
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    obj = (max_corr + np.log(sum_exp) / alpha) * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        w = weights[k]
        if l >= 0:
            if l < n:
                grad[l:] += w * (1.0 - h[:n-l])
            if l + 1 <= n:
                grad[:n-l] -= w * h[l:]
        else:
            lp = -l
            if lp < n:
                grad[:n-lp] += w * (1.0 - h[lp:])
            if lp + 1 <= n:
                grad[lp:] -= w * h[:n-lp]
    
    grad *= dx
    return obj, grad

def make_feasible(h, n):
    """Project h to be feasible: 0<=h<=1, sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the correction evenly
        h += diff / n
        h = np.clip(h, 0, 1)
        # If clipping changed the sum, do a second pass
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-10:
            # Only adjust non-saturated elements
            mask = (h > 1e-6) & (h < 1 - 1e-6)
            if np.any(mask):
                h[mask] += diff2 / np.sum(mask)
                h = np.clip(h, 0, 1)
    # Final enforcement
    h = np.clip(h, 0, 1)
    # Exact sum fix (tiny adjustment)
    h[-1] += (target - np.sum(h))
    h = np.clip(h, 0, 1)

    h_dst = np.interp(x_dst, x_src, h_src)

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            mask = (h > 1e-9) & (h < 1.0 - 1e-9)
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff2 / cnt
            else:
                h[0] += diff2
            h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 0:
        idx = int(np.argmin(h)) if residual > 0 else int(np.argmax(h))
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

    n_src = len(h_src)
    if n_src == n_dst:
        return make_feasible(h_src, n_dst)
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    hd = np.interp(xd, xs, h_src)
    return make_feasible(hd, n_dst)

    return float(np.sum(h) - len(h) / 2.0)

def _eq_jac(h):
    return np.ones(len(h))

    h_start = make_feasible(h_start, n)
    fun = partial(erf_soft_obj, alpha=alpha)
    jac = partial(erf_soft_grad, alpha=alpha)
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h_start.copy(), jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = make_feasible(res.x, n)
        return h_opt, erf_c5(h_opt)
    except Exception:
        return h_start, erf_c5(h_start)

def binary_c5(h_bin, n):
    """Compute C5 for binary h."""
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx)

def binary_local_search(h_bin, n, max_swaps=2000):
    """Local search for binary h: swap 1s and 0s."""
    h_bin = h_bin.copy()
    c5 = binary_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    improved = True
    swap_count = 0
    while improved and swap_count < max_swaps:
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones[:min(50, len(ones))]:
            for j in zeros[:min(50, len(zeros))]:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                new_c5 = binary_c5(h_bin, n)
                if new_c5 < c5 - 1e-12:
                    c5 = new_c5
                    improved = True
                    ones = np.where(h_bin == 1)[0]
                    zeros = np.where(h_bin == 0)[0]
                    break
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
                swap_count += 1
            if improved:
                break
    return h_bin, c5

def erf_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def erf_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(erf_corr(h)) * dx)

def erf_soft_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = erf_corr(h)
    mc = float(np.max(corr))
    sh = alpha * (corr - mc)
    es = np.exp(sh)
    se = float(np.sum(es))
    return (mc + np.log(se) / alpha) * dx

def erf_soft_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = erf_corr(h)
    mc = float(np.max(corr))
    sh = alpha * (corr - mc)
    es = np.exp(sh)
    se = float(np.sum(es))
    w = es / se
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        wm = w[m]
        if wm < 1e-14:
