        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = exact_project(h_interp, target_n)
        c5_interp = compute_c5(h_interp)

        # DE at new resolution
        time_remaining = budget_s * 0.85 - elapsed()
        if time_remaining > 20:
            bounds_new = [(0.0, 1.0)] * target_n
            mi = min(250, max(40, int(time_remaining / max(1.0, target_n / 70.0))))
            ps = max(15, min(30, int(1500.0 / target_n) + 15))
            try:
                res = differential_evolution(
                    de_fit, bounds=bounds_new, args=(target_n,),
                    maxiter=mi, popsize=ps,
                    seed=seed + target_n, tol=1e-12,
                    mutation=(0.3, 0.7), recombination=0.6,
                    polish=False, init='latinhypercube'
                )
                h_cand = exact_project(res.x, target_n)
                c5_cand = compute_c5(h_cand)
                if c5_cand < c5_interp:
                    h_interp = h_cand
                    c5_interp = c5_cand
            except Exception:
                pass

        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = target_n

        # Subgradient refinement
        if elapsed() < budget_s * 0.82:
            best_h, best_c5 = subgrad_refine(best_h, best_n, 80, 0.005)

        # Smoothed L-BFGS-B local optimization
        if elapsed() > budget_s * 0.87:
            break
        n_local = best_n
        if n_local > 1000:
            continue
        try:
            h_free = best_h[:n_local - 1].copy()
            T_smooth = max(0.002, best_c5 * 0.08)
            _params['n'] = n_local
            _params['T'] = T_smooth
            res_loc = minimize(
                smooth_obj, h_free, method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * (n_local - 1),
                options={'maxiter': 60, 'ftol': 1e-13}
            )
            h_loc = np.empty(n_local)
            h_loc[:n_local - 1] = res_loc.x
            h_loc[n_local - 1] = n_local / 2.0 - np.sum(res_loc.x)
            h_loc = exact_project(h_loc, n_local)
            c5_loc = compute_c5(h_loc)
            if c5_loc < best_c5:
                best_c5 = c5_loc
                best_h = h_loc.copy()
        except Exception:
            pass

    # Final: one more subgradient pass if time allows
    if elapsed() < budget_s * 0.95:
        best_h, best_c5 = subgrad_refine(best_h, best_n, 50, 0.002)

    # Final exact projection
    best_h = exact_project(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
