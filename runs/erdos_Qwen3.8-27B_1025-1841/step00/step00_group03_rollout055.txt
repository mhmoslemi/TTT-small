import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    """Project h onto [0,1]^n with sum = n/2"""
    h = np.clip(h, 0, 1)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to add diff total mass
        # Add to entries with room, proportional to room
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 1e-15:
            h += diff * room / total_room
        h = np.clip(h, 0, 1)
    else:
        # Need to remove -diff total mass
        total_mass = np.sum(h)
        if total_mass > 1e-15:
            h += diff * h / total_mass
        h = np.clip(h, 0, 1)
    # Final fix
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute among entries with room
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
    h = np.clip(h, 0, 1)
    return h

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    current_c5 = best_c5
    t_start = time.time()
    T = 0.5
    T_min = 0.001
    decay = 0.9995
    step_size = 0.1
    no_improve = 0
    max_no_improve = 500
    
    while time.time() - t_start < time_limit:
        # Choose perturbation type
        if rng.random() < 0.7:
            # Transfer perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if h[i] > h[j]:
                delta = rng.random() * min(h[i], 1 - h[j]) * step_size
            elif h[j] > h[i]:
                delta = -rng.random() * min(h[j], 1 - h[i]) * step_size
            else:
                continue
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            h_new = np.clip(h_new, 0, 1)
        else:

            i = rng.integers(n)
            j = rng.integers(n)

                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        c5_new = evaluate_c5(h_new)
        delta_c5 = c5_new - current_c5
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-10)):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T = max(T * decay, T_min)
        if no_improve > max_no_improve:
            no_improve = 0
            T = max(T * 1.5, 0.1)  # Restart temperature
    
    return best_h, best_c5

def project(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    # Binary search for lambda
    lo = -1.0
    hi = 2.0 + target_sum / len(h)  # upper bound for lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - mid, 0, 1)
    # Small correction
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute among non-saturated entries
        mask = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

        else:

        if cnt > 0:
            hp[mask] += diff / cnt
    return np.clip(hp, 0.0, 1.0)

def grad_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    w = np.exp(beta * (corr - cmax))
    w /= w.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        ilo = max(0, s)
        ihi = min(n, s + n)
        if ihi > ilo:
            jlo = ilo - s
            grad[ilo:ihi] += wk * omh[jlo:jlo + ihi - ilo]
        ilo2 = max(0, -s)
        ihi2 = min(n, n - s)
        if ihi2 > ilo2:
            jlo2 = ilo2 + s
            grad[ilo2:ihi2] -= wk * h[jlo2:jlo2 + ihi2 - ilo2]
    return dx * grad

def upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    return np.interp(x_new, x_old, h)

def local_search(h, t_limit, rng):
    n = len(h)
    best = h.copy()
    best_c5 = evaluate_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    T = 5e-4
    T_min = 5e-7
    t0 = time.time()
    sf = 0.12
    while time.time() - t0 < t_limit:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        oi = cur[i]
        oj = cur[j]
        if oi >= oj:
            d = rng.random() * oi * sf
        else:
            d = -rng.random() * (1.0 - oj) * sf
        ni = oi + d
        nj = oj - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        cur[i] = ni
        cur[j] = nj
        nc5 = evaluate_c5(cur)
        accept = nc5 <= cur_c5 or (T > T_min and rng.random() < np.exp(-(nc5 - cur_c5) / T))
        if accept:
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best = cur.copy()
        else:
            cur[i] = oi
            cur[j] = oj
        T = max(T * 0.9997, T_min)
    return best, best_c5

def grad_opt(h, t_limit, rng):
    n = len(h)
    tgt = n / 2.0
    best = h.copy()
    best_c5 = evaluate_c5(best)
    cur = h.copy()
    lr = 0.02
    beta = 30.0
    t0 = time.time()
    stall = 0
    while time.time() - t0 < t_limit:
        g = grad_obj(cur, beta)
        cur = project(cur - lr * g, tgt)
        c5 = evaluate_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best = cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                cur = best.copy()
                lr *= 0.5
                beta = min(beta * 1.15, 400.0)
                stall = 0
                if rng.random() < 0.4:
                    idx = rng.choice(n, size=min(5, n), replace=False)
                    cur[idx] += (rng.random(idx.size) - 0.5) * 0.06
                    cur = project(cur, tgt)
        if lr < 5e-7:
            break
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)

    n0 = len(h0)
    h0 = project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    print(f"Start: C5={best_c5:.6f} n={n0}", flush=True)

    # Phase 1: local search at initial resolution
    if time.time() < deadline - 400:
        t1 = min(80, (deadline - time.time()) * 0.06)
        best_h, best_c5 = local_search(best_h, t1, rng)
        print(f"LS1: C5={best_c5:.6f} n={len(best_h)}", flush=True)

    # Phase 2: upsample x3, gradient + local search
    n2 = min(n0 * 3, 300)
    if time.time() < deadline - 350:
        h2 = upsample(best_h, n2)
        h2 = project(h2, n2 / 2.0)
        tg = min(180, (deadline - time.time()) * 0.35)
        h2, c2 = grad_opt(h2, tg, rng)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
        if time.time() < deadline - 150:
            h2, c2 = local_search(h2, min(100, (deadline - time.time()) * 0.3), rng)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2
        print(f"P2: C5={best_c5:.6f} n={len(best_h)}", flush=True)

    # Phase 3: extended local search at current resolution
    if time.time() < deadline - 100:
        t3 = min(250, (deadline - time.time()) * 0.55)
        best_h, best_c5 = local_search(best_h, t3, rng)
        print(f"LS3: C5={best_c5:.6f} n={len(best_h)}", flush=True)

    # Phase 4: try 2x higher resolution
    n4 = len(best_h) * 2
    if time.time() < deadline - 150 and n4 <= 2000:
        h4 = upsample(best_h, n4)
        h4 = project(h4, n4 / 2.0)
        tg4 = min(120, (deadline - time.time()) * 0.45)
        h4, c4 = grad_opt(h4, tg4, rng)
        if time.time() < deadline - 40:
            h4, c4 = local_search(h4, min(60, (deadline - time.time()) * 0.3), rng)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4
        print(f"P4: C5={best_c5:.6f} n={len(best_h)}", flush=True)

    # Phase 5: final polish local search
    if time.time() < deadline - 10:
        tp = min(60, (deadline - time.time()) * 0.9)
        best_h, best_c5 = local_search(best_h, tp, rng)

    best_h = project(best_h, len(best_h) / 2.0)
    nf = len(best_h)
    cf = evaluate_c5(best_h)
    print(f"Final: C5={cf:.6f} n={nf}", flush=True)
    return best_h, cf, nf
