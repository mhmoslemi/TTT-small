import time

    """Project onto {0 <= h <= 1, sum(h) = N/2}"""
    target = N / 2.0
    # Bisection for lambda
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    """Compute C5 value and subgradient"""
    dx = 2.0 / N
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c_val = c_raw * dx
    c5 = np.max(c_val)
    
    # Find argmax
    k_max = np.argmax(c_raw)
    s = k_max - (N - 1)
    
    # Gradient of c_raw[k_max] w.r.t. h
    grad = np.zeros(N)

        # Second term: -h[j-m] for j in [m, N-1]
        idx2 = np.arange(m, N)
        grad[idx2] -= h[idx2 - m]
    
    return c5, grad * dx  # gradient of c5 w.r.t. h

def subgradient_descent(h_init, N, n_iters, lr_init=0.01, lr_decay=0.999):
    """Run projected subgradient descent"""
    h = h_init.copy()
    h = project_feasible(h, N)

```python
def soft_gradient(h, N, tau=0.005):
    """Compute gradient of soft-max C5"""
    dx = 2.0 / N
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c_val = c_raw * dx
    
    # Softmax weights (shifted for numerical stability)
    max_c = np.max(c_val)
    logits = (c_val - max_c) / tau
    p = np.exp(logits)
    p /= np.sum(p)
    
    # Compute gradient: dx * sum_k p[k] * grad_c_k
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        if p[k] < 1e-12:
            continue
        s = k - (N - 1)
        w = p[k]
        if s >= 0:
            idx = np.arange(s, N)
            grad[idx] += w * (1.0 - h[idx - s])
            idx2 = np.arange(0, N - s)
            grad[idx2] -= w * h[idx2 + s]
        else:
            m = -s
            idx = np.arange(0, N - m)
            grad[idx] += w * (1.0 - h[idx + m])
            idx2 = np.arange(m, N)
            grad[idx2] -= w * h[idx2 - m]
    
    return dx * grad, np.max(c_val)
```

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, N):
    target = N / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def c5_value(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def upscale_linear(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0.0, 2.0, N_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def grad_top_k(h, N, k=5):
    dx = 2.0 / N
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c_val = c_raw * dx
    c5 = float(np.max(c_val))
    top_idx = np.argsort(c_val)[-k:]
    w = 1.0 / k
    grad = np.zeros(N)
    one_minus_h = 1.0 - h
    for ki in top_idx:
        s = ki - (N - 1)
        if s >= 0:
            if s < N:
                idx = np.arange(s, N)
                grad[idx] += w * one_minus_h[idx - s]
            if N - s > 0:
                idx2 = np.arange(0, N - s)
                grad[idx2] -= w * h[idx2 + s]
        else:
            m = -s
            if N - m > 0:
                idx = np.arange(0, N - m)
                grad[idx] += w * one_minus_h[idx + m]
            idx2 = np.arange(m, N)
            grad[idx2] -= w * h[idx2 - m]
    return c5, grad * dx

def subgradient_descent(h_init, N, n_iters, lr_init=0.3, top_k=5, time_limit=30):
    h = project_feasible(h_init.copy(), N)
    best_h = h.copy()
    best_c5 = c5_value(h)
    t_start = time.time()
    gnorm = 1.0
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        c5, grad = grad_top_k(h, N, top_k)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        gnorm = 0.9 * gnorm + 0.1 * gn
        alpha = lr_init / (1.0 + 0.005 * it) * 0.1
        h = h - alpha * grad / gnorm * gnorm
        h = project_feasible(h, N)
    return best_h, best_c5

def softmax_obj_grad(h, N, tau=0.005):
    dx = 2.0 / N
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c_val = c_raw * dx
    max_c = np.max(c_val)
    logits = (c_val - max_c) / tau
    p = np.exp(logits)
    psum = np.sum(p)
    obj = tau * np.log(psum) + max_c
    p = p / psum
    grad = np.zeros(N)
    one_minus_h = 1.0 - h
    for k in range(2 * N - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        s = k - (N - 1)
        if s >= 0:
            if s < N:
                idx = np.arange(s, N)
                grad[idx] += pk * one_minus_h[idx - s]
            if N - s > 0:
                idx2 = np.arange(0, N - s)
                grad[idx2] -= pk * h[idx2 + s]
        else:
            m = -s
            if N - m > 0:
                idx = np.arange(0, N - m)
                grad[idx] += pk * one_minus_h[idx + m]
            idx2 = np.arange(m, N)
            grad[idx2] -= pk * h[idx2 - m]
    return obj, dx * grad

def lbfgsb_polish(h_init, N, tau=0.005, time_limit=30):
    t_start = time.time()
    h = project_feasible(h_init.copy(), N)
    
    def objective(x):
        if time.time() - t_start > time_limit:
            return 1e6
        h2 = np.clip(x, 0.0, 1.0)
        val, _ = softmax_obj_grad(h2, N, tau)
        return val
    
    def gradient(x):
        if time.time() - t_start > time_limit:
            return np.zeros(N)
        h2 = np.clip(x, 0.0, 1.0)
        _, g = softmax_obj_grad(h2, N, tau)
        g[np.clip(x, 0.0, 1.0) != x] = 0.0
        return g
    
    bounds = [(0.0, 1.0)] * N
    res = minimize(objective, h, method='L-BFGS-B', jac=gradient,
                   bounds=bounds, options={'maxiter': 500, 'maxfun': 10000,
                                           'time_limit': time_limit * 0.9})
    h_out = project_feasible(np.clip(res.x, 0.0, 1.0), N)
    c5_out = c5_value(h_out)
    return h_out, c5_out
