    diverse_inits.append(proj(saw, n))
    tri = 1.0 - np.abs(np.linspace(-1.0, 1.0, n))
    diverse_inits.append(proj(tri, n))

    for di in diverse_inits:
        if time.time() - t0 > budget_s - 80:
            break
        c = _Cons(n)
        cons = [{'type': 'eq', 'fun': c.fun, 'jac': c.jac}]
        bds = [(0.0, 1.0)] * n
        try:
            r = minimize(smooth_obj, di, args=(n, 200.0),
                         method='SLSQP', constraints=cons, bounds=bds,
                         options={'maxiter': 300, 'ftol': 1e-12, 'iprint': 0})
            hc = proj(r.x, n)
            cv = compute_c5(hc, n)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc.copy()
                best_n = n
        except Exception:
            pass

    # Random restarts at best resolution
    for _ in range(50):
        if time.time() - t0 > budget_s - 40:
            break
        sc = 0.03 + 0.35 * rng.random()
        hp = proj(best_h + rng.randn(n) * sc, n)
        c = _Cons(n)
        cons = [{'type': 'eq', 'fun': c.fun, 'jac': c.jac}]
        bds = [(0.0, 1.0)] * n
        try:
            r = minimize(smooth_obj, hp, args=(n, 300.0),
                         method='SLSQP', constraints=cons, bounds=bds,
                         options={'maxiter': 250, 'ftol': 1e-12, 'iprint': 0})
            hc = proj(r.x, n)
            cv = compute_c5(hc, n)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc.copy()
                best_n = n
        except Exception:
            pass

    # Final polish
    if time.time() - t0 < budget_s - 30 and best_n >= 84:
        c = _Cons(best_n)
        cons = [{'type': 'eq', 'fun': c.fun, 'jac': c.jac}]
        bds = [(0.0, 1.0)] * best_n
        try:
            r = minimize(smooth_obj, best_h.copy(), args=(best_n, 500.0),
                         method='SLSQP', constraints=cons, bounds=bds,
                         options={'maxiter': 500, 'ftol': 1e-14, 'iprint': 0})
            hc = proj(r.x, best_n)
            cv = compute_c5(hc, best_n)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc.copy()
        except Exception:
            pass

    return best_h, best_c5, best_n
