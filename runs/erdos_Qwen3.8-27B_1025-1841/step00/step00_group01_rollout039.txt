  contribution = -h[j+shift]

import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    h = h.copy()
    np.clip(h, 0, 1, out=h)
    for _ in range(1000):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1 - 1e-14
            c = m.sum()
            if c == 0:
                break
            h[m] = np.minimum(h[m] + d / c, 1.0)
        else:
            m = h > 1e-14
            c = m.sum()
            if c == 0:
                break
            h[m] = np.maximum(h[m] + d / c, 0.0)
    return h

def c5_subgrad(h, tol=1e-6):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    c5 = float(np.max(corr))
    active = np.where(corr >= c5 - tol)[0]

        if shift >= 0:
            if n - shift > 0:
                grad[shift:n] += 1.0 - h[0:n-shift]
                grad[0:n-shift] -= h[shift:n]
        else:
            s = -shift
            if n - s > 0:
                grad[0:n-s] += 1.0 - h[s:n]
                grad[s:n] -= h[0:n-s]
    grad /= len(active)
    grad *= dx
    return c5, grad

def project(h, n):

        else:

            c = m.sum()
            if c == 0:

            def obj(hv):
                return float(np.max(np.correlate(hv, 1 - hv, mode="full") * dx))
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda hv: hv.sum() - n / 2.0,
                     'jac': lambda hv: np.ones(n)}]
            try:
                res = minimize(obj, best_h.copy(), method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
                h2 = project(res.x, n)
                c2 = compute_c5(h2)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2.copy()
            except Exception:
                pass

    # Phase 4: Final gradient polish
    if rem() > 20:
        n = len(best_h)
        h, c = gradient_descent(best_h.copy(), n, max_iter=15000, lr=0.01,
                               deadline=time.time() + max(10, rem() - 10))
        if c < best_c5:
            best_c5, best_h = c, h

    # Verify constraint
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
