def _correlate_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _upscale(src, n_src, n_dst):
    if n_src == n_dst:
        return src.copy()
    x0 = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x1, x0, src)
