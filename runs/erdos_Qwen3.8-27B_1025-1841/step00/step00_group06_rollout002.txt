def project_simplex(h, target_sum, upper=1.0):
    h = h.copy()
    h = np.clip(h, 0, upper)
    for _ in range(100):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum: increase elements below upper
            mask = h < upper
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, upper)

            mask = h > 0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, upper)

    h = np.clip(h, 0.0, upper)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < upper - 1e-14
        else:
            free = h > 1e-14
        if not np.any(free):
            break
        h[free] += diff / np.count_nonzero(free)
        h = np.clip(h, 0.0, upper)
    return h

def gradient_descent(h, n_pts, rng, max_iters=5000, timeout_check=None):
    T_list = [2.0, 5.0, 10.0, 20.0, 50.0]
    c5 = compute_c5(h)
    for T in T_list:
        if timeout_check and timeout_check():
            break
        lr = 0.05
        consec_fail = 0
        for step in range(max_iters):
            if timeout_check and timeout_check():
                break
            obj, grad = objective_and_gradient(h, T)
            h_new = h - lr * grad
            h_new = project_to_feasible(h_new, n_pts / 2.0)
            new_c5 = compute_c5(h_new)
            if new_c5 < c5 - 1e-13:
                h = h_new
                c5 = new_c5
                consec_fail = 0
                lr = min(lr * 1.01, 0.5)
            else:
                consec_fail += 1
                lr *= 0.93
                if lr < 1e-7 or consec_fail > 150:
                    break
        # Perturbation
        if timeout_check and timeout_check():
            break
        scale = 0.03 + 0.02 * len(T_list)
        h_p = h + rng.randn(n_pts) * scale
        h_p = project_to_feasible(h_p, n_pts / 2.0)
        cp = compute_c5(h_p)
        if cp < c5:
            h = h_p
            c5 = cp
    return h, c5

def binary_local_search(h_bin, n_pts, rng, max_iters=8000, timeout_check=None):
    c5 = compute_c5(h_bin)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    anneal_temp = 0.001
    for it in range(max_iters):
        if timeout_check and timeout_check():
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        delta = c5_new - c5
        if delta < 0 or rng.rand() < np.exp(-delta / anneal_temp):
            c5 = c5_new
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            anneal_temp *= 0.999
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return h_bin, c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def time_up():
        return (time.time() - t0) > (budget_s - 15)

    init = None
    if 'initial_h_values' in globals():
        try:
            init = np.array(initial_h_values, dtype=float)
        except Exception:
            pass

    resolutions = [100, 96, 128, 120, 150, 200, 95]

    for n_pts in resolutions:
        if time_up():
            break
        n_starts = 4 if n_pts <= 130 else 2

        for si in range(n_starts):
            if time_up():
                break
            if si == 0 and init is not None:
                if len(init) == n_pts:
                    h = init.copy()
                else:
                    h = np.interp(np.linspace(0, 1, n_pts),
                                  np.linspace(0, 1, len(init)), init)
            elif si == 1:
                if n_pts % 2 == 0:
                    h = np.zeros(n_pts)
                    h[rng.choice(n_pts, n_pts // 2, replace=False)] = 1.0
                else:
                    h = rng.uniform(0, 1, n_pts)
            elif si == 2 and init is not None:
                h = np.interp(np.linspace(0, 1, n_pts),
                              np.linspace(0, 1, len(init)), init)
                h = h + rng.randn(n_pts) * 0.1
            else:
                h = np.zeros(n_pts)
                n1 = n_pts // 2
                h[rng.choice(n_pts, n1, replace=False)] = 1.0
                if n_pts % 2 == 1:
                    h[rng.randint(n_pts)] = 0.5

            h = project_to_feasible(h, n_pts / 2.0)
            c5 = compute_c5(h)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n_pts

            # Gradient descent phase
            h, c5 = gradient_descent(h, n_pts, rng,
                                     max_iters=4000, timeout_check=time_up)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n_pts

            # Second gradient descent with different T emphasis
            if not time_up():
                h2, c5_2 = gradient_descent(h + rng.randn(n_pts) * 0.02,
                                            n_pts, rng,
                                            max_iters=3000, timeout_check=time_up)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()
                    best_n = n_pts

            # Binary refinement (only for even n)
            if n_pts % 2 == 0 and not time_up():
                h_bin = np.zeros(n_pts)
                order = np.argsort(h_h) if (h_h := h2) is not None else np.argsort(h)[::-1]
                h_bin[order[:n_pts // 2]] = 1.0
                c5_b = compute_c5(h_bin)
                if c5_b < c5:
                    h = h_bin.copy()
                    c5 = c5_b
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n_pts
                # Run binary local search from this binary
                h_bl, c5_bl = binary_local_search(h.copy(), n_pts, rng,
                                                   max_iters=5000, timeout_check=time_up)
                if c5_bl < best_c5:
                    best_c5 = c5_bl
                    best_h = h_bl.copy()
                    best_n = n_pts

    # Final: try to improve best_h with another round
    if best_h is not None and not time_up():
        n_final = best_n
        h_ref = best_h.copy()
        h_ref, c5_ref = gradient_descent(h_ref, n_final, rng,
                                         max_iters=3000, timeout_check=time_up)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    if best_h is None:
        best_n = 100
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
