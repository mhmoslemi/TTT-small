def targeted_local_search(h, n, n_iter, rng):
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(n_iter):
        # Compute correlation to find worst lag
        c_full = np.correlate(best, 1.0 - best, mode='full')
        k_star = int(np.argmax(c_full))
        shift = k_star - (n - 1)  # actual shift in position units
        
        # With 50% probability, target the worst lag
        if rng.random() < 0.5 and shift != 0:
            # Pick i that contributes to c[k_star]
            valid_starts = range(max(0, -shift), min(n, n - shift))
            i = int(rng.choice(list(valid_starts)))
            j = i + shift
            # Move mass from i to j (reduces best[i]*(1-best[j]))
            # or from j to i (reduces best[j]*(1-best[i]))
            if best[i] * (1 - best[j]) > best[j] * (1 - best[i]):
                delta = min(best[i], 1 - best[j]) * rng.random() * 0.3
                h_new = best.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                delta = min(best[j], 1 - best[i]) * rng.random() * 0.3
                h_new = best.copy()
                h_new[i] += delta
                h_new[j] -= delta
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            delta = (best[i] - best[j]) * rng.random() * 0.3
            if abs(delta) < 1e-10:
                continue
            h_new = best.copy()
            h_new[i] -= delta
            h_new[j] += delta
        
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        new_c = compute_c5(h_new)
        if new_c < best_c:
            best = h_new
            best_c = new_c
    return best, best_c

import numpy as np
from scipy.optimize import minimize
from time import time


def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))


def upsample(h, new_n):
    n = len(h)
    idx = np.round(np.linspace(0, n - 1, new_n)).astype(int)
    h_new = h[idx].astype(float)
    target = new_n / 2.0
    diff = target - np.sum(h_new)
    h_new += diff / new_n
    h_new = np.clip(h_new, 0, 1)
    h_new[0] = np.clip(h_new[0] + (target - np.sum(h_new)), 0, 1)
    return h_new


def local_search(h, n, n_iter, rng, target_worst=True):
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(n_iter):
        h_new = None
        if target_worst and rng.random() < 0.6:
            c_full = np.correlate(best, 1.0 - best, mode='full')
            k_star = int(np.argmax(c_full))
            shift = k_star - (n - 1)
            if shift != 0 and abs(shift) < n:
                lo = max(0, -shift)
                hi = min(n, n - shift)
                if hi > lo:
                    i = int(rng.integers(lo, hi))
                    j = i + shift
                    if best[i] * (1.0 - best[j]) >= best[j] * (1.0 - best[i]):
                        delta = min(best[i], 1.0 - best[j]) * rng.random() * 0.4
                    else:
                        delta = -min(best[j], 1.0 - best[i]) * rng.random() * 0.4
                    h_new = best.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
        if h_new is None:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            delta = (best[i] - best[j]) * rng.random() * 0.4
            if abs(delta) < 1e-12:
                continue
            h_new = best.copy()
            h_new[i] -= delta
            h_new[j] += delta
        if h_new[i] < -1e-12 or h_new[i] > 1.0 + 1e-12 or h_new[j] < -1e-12 or h_new[j] > 1.0 + 1e-12:
            continue
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        new_c = compute_c5(h_new)
        if new_c < best_c:
            best = h_new
            best_c = new_c
    return best, best_c
