```python
    k_star = np.argmax(corr)

    grad = np.zeros(n_points)
    
    # S = {j : 0 <= j < n, 0 <= j+s < n}
    # For h[m] as the j-index: valid if 0 <= m < n and 0 <= m+s < n
    # For h[m] as the j+s-index: valid if 0 <= m-s < n and 0 <= m < n
    
    for m in range(n_points):
        # Contribution 1: m as j
        if 0 <= m + s < n_points:
            grad[m] += (1.0 - h[m + s])
        # Contribution 2: m as j+s (j = m-s)
        if 0 <= m - s < n_points:
            grad[m] -= h[m - s]
    
    return val, grad * dx
```

```python
def c5_gradient(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
```

```python
    # Contribution 1: for m where 0 ≤ m+s < n, grad[m] += (1 - h[m+s])
    # m ranges from max(0, 0-s) to min(n-1, n-1-s)
    lo1 = max(0, -s)
    hi1 = min(n_points, n_points - s)  # exclusive
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
    
    # Contribution 2: for m where 0 ≤ m-s < n, grad[m] -= h[m-s]
    # m ranges from max(0, s) to min(n-1, n-1+s)
    lo2 = max(0, s)
    hi2 = min(n_points, n_points + s)  # exclusive
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2-s:hi2-s]
    
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_value(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr[np.argmax(corr)] * dx)

def c5_gradient(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s = k_star - (n_points - 1)
    grad = np.zeros(n_points)
    lo1 = max(0, -s)
    hi1 = min(n_points, n_points - s)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
    lo2 = max(0, s)
    hi2 = min(n_points, n_points + s)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2-s:hi2-s]
    return grad * dx

def eq_fun(h, n_points):
    return np.sum(h) - n_points / 2.0

def eq_jac(h, n_points):
    return np.ones(n_points)

def proj_feasible(h, n_points):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n_points / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
    else:
        mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def interp_upscale(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    x_src = np.arange(n_src, dtype=float) / n_src
    x_tgt = np.arange(n_target, dtype=float) / n_target
    h_up = np.interp(x_tgt, x_src, h_src)
    return proj_feasible(h_up, n_target)

def run_slsqp(h_start, n_pts, maxiter):
    bounds = [(0.0, 1.0)] * n_pts
    cons = ({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},)
    res = minimize(c5_value, h_start, args=(n_pts,),
                   jac=c5_gradient, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    return proj_feasible(res.x, n_pts)

    np.random.seed(seed)
    t0 = time.time()
    
    try:
        h0_raw = np.array(initial_h_values, dtype=float)

        h0_raw = np.resize(h0_raw, n0 + 1)
        n0 = len(h0_raw)
    h0 = proj_feasible(h0_raw, n0)
    
    best_c5 = compute_c5(h0, n0)
    best_h = h0.copy()
    best_n = n0
    
    # Plan: (n_points, n_restarts, maxiter_per_restart)
    plan = [(n0, 10, 500), (n0*2, 6, 300), (n0*4, 4, 200), (n0*8, 3, 100)]
    
    for idx, (n_pts, n_restarts, maxiter) in enumerate(plan):
        elapsed = time.time() - t0
        if budget_s - elapsed < 60:
            break
        
        # Starting point for this resolution
        if idx == 0:
            h_start = h0.copy()
        else:
            h_start = interp_upscale(best_h, n_pts)
        
        cur_best = compute_c5(h_start, n_pts)
        
        for r in range(n_restarts):
            elapsed = time.time() - t0
            if budget_s - elapsed < 30:
                break
            
            # Different starting strategies
            if r == 0:
                h_try = h_start.copy()
            elif r == 1:
                h_try = proj_feasible(h_start + 0.2 * np.random.randn(n_pts), n_pts)
            elif r == 2:
                h_try = proj_feasible(h_start + 0.05 * np.random.randn(n_pts), n_pts)
            elif r == 3:
                sc = np.random.uniform(0.3, 1.7)
                h_try = proj_feasible(h_start * sc, n_pts)
            elif r == 4:
                h_try = proj_feasible(0.5 + 0.3 * np.random.randn(n_pts), n_pts)
            elif r == 5:
                h_try = proj_feasible(np.random.uniform(0, 1, n_pts), n_pts)
            elif r == 6:
                # Binary-like starting point
                h_try = proj_feasible(np.round(np.random.uniform(0, 1, n_pts)), n_pts)
            else:
                h_try = proj_feasible(h_start + 0.1 * np.random.randn(n_pts), n_pts)
            
            try:
                h_opt = run_slsqp(h_try, n_pts, maxiter)
            except Exception:
                continue
            
            c5_val = compute_c5(h_opt, n_pts)
            
            if c5_val < cur_best:
                cur_best = c5_val
                h_start = h_opt.copy()
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()

                    best_h = h_pol.copy()

    best_h = proj_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return (best_h, final_c5, best_n)
```
