import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            return h
        h -= diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample_to(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, new_n)

def sa_phase(h, c0, t_end, rng, n):

    c = c0
    best = h.copy()
    best_c = c0
    T = max(c0 * 0.15, 0.01)
    T_floor = 1e-7
    cool = 0.9999
    while time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] > 1e-10 and h[j] < 1.0 - 1e-10:
            ep = min(h[i], 1.0 - h[j]) * rng.uniform(0.05, 0.5)

            h[j] += ep
            nc = compute_c5(h)
            d = nc - c

                c = nc
                if nc < best_c:
                    best_c = nc
                    best = h.copy()
            else:
                h[i] += ep
                h[j] -= ep
        elif h[j] > 1e-10 and h[i] < 1.0 - 1e-10:
