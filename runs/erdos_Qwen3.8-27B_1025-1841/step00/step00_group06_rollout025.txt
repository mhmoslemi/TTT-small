def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_max(v, beta):
    m = np.max(v)
    if m < -500.0:
        return float(m)
    return float(m + np.log(np.sum(np.exp((v - m) * beta))) / beta)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = float(np.sum(h))
    if s > 1e-10 and abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    d = target - s
    if abs(d) > 1e-10:
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(hf):
    n = _obj_n
    beta = _obj_beta
    h = np.clip(hf, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    val = smooth_max(corr, beta)
    val += abs(float(np.sum(h)) * dx - 1.0) * 50.0
    return val

def local_search(h, n, rng, max_iters, t0, budget):
    best_c5 = compute_c5(h)
    best = h.copy()
    cur = h.copy()
    cur_c5 = best_c5
    temp = 0.003
    for _ in range(max_iters):
        if time.time() - t0 > budget * 0.93:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = compute_c5(cur)
        delta = new_c5 - cur_c5
        accept = delta < 0
        if not accept and temp > 1e-12:
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
        temp *= 0.997
    return best, best_c5

def make_binary(h, n, rng):
    """Project h to a binary vector with exactly n/2 ones."""
    h_bin = np.zeros(n)
    # Rank by value, top n/2 become 1
    indices = np.argsort(-h)[:n // 2]
    h_bin[indices] = 1.0
    return h_bin
