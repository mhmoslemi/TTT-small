            cnt = mask.sum()
            if cnt > 0:

            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)

    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # More aggressive correction
        h = h + diff / n
        h = np.clip(h, 0, 1)
    return h

    weights = np.maximum(0, c_vals[top_indices] - (max_c - 0.01))
    if weights.sum() < 1e-10:
        weights = np.ones(len(top_indices))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for idx, k in enumerate(top_indices):

        hi = min(n - 1, int(k))
        m = np.arange(lo, hi + 1)

        grad[m] += weights[idx] * (1 - h[m2])
        if int(k) != n - 1:
            np.add.at(grad, m2, -weights[idx] * h[m])
        else:
            grad[m] += -weights[idx] * h[m]
    
    return h - learning_rate * grad, max_c
