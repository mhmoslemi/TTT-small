import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _sample_to_n(h_src, n):
    n_src = len(h_src)
    if n_src == n:
        return h_src.copy()
    x_src = np.arange(n_src) / n_src * 2.0
    x_new = np.arange(n) / n * 2.0
    h_new = np.interp(x_new, x_src, np.concatenate([h_src, [h_src[0]]]), period=2.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        diff2 = target - np.sum(h_new)
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                mask = h_new < 1.0
            else:
                mask = h_new > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h_new[mask] += diff2 / cnt
                h_new = np.clip(h_new, 0.0, 1.0)
    return h_new


def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _slsqp_optimize(h0, n, maxiter=300, ftol=1e-9):
    def obj(x):
        return x[-1]

    def obj_jac(x):
        j = np.zeros(n + 1)
        j[-1] = 1.0
        return j

    def sum_con(x):
        return np.sum(x[:-1]) - n / 2.0

    def sum_jac(x):
        return np.concatenate([np.ones(n), [0.0]])

    def make_c5_con(k):
        def con(x):
            h = x[:-1]
            t = x[-1]
            dx = 2.0 / n
            corr = np.correlate(h, 1.0 - h, mode='full')
            return t - corr[k] * dx
        return con

    # Only use top correlations as active constraints for speed
    dx = 2.0 / n
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    c5_initial = np.max(corr0) * dx
    # Pick top 30% of lags as constraints (plus the max)
    n_lags = len(corr0)
    n_cons = min(n_lags, max(30, n_lags // 3))
    top_idx = np.argsort(corr0)[::-1][:n_cons]

    constraints = [
        {'type': 'eq', 'fun': sum_con, 'jac': sum_jac}
    ]
    for ki in sorted(top_idx):
        c = {
            'type': 'ineq',
            'fun': make_c5_con(int(ki))
        }
        constraints.append(c)

    x0 = np.concatenate([h0.copy(), [c5_initial + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c5_initial + 0.5)]

    try:
        res = minimize(
            obj, x0, method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
    except Exception:
        return h0, compute_c5(h0, n)

    h_opt = np.clip(res.x[:-1], 0.0, 1.0)
    h_opt = _fix_sum(h_opt, n)
    c5_opt = compute_c5(h_opt, n)

    if c5_opt < compute_c5(h0, n):
        return h_opt, c5_opt
    return h0, compute_c5(h0, n)


def _direct_slsqp(h0, n, maxiter=200):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr) * dx)

    def con(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    try:
        res = minimize(
            obj, h0.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
    except Exception:
        return h0, compute_c5(h0, n)

    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = _fix_sum(h_opt, n)
    c5_opt = compute_c5(h_opt, n)
    if c5_opt < compute_c5(h0, n):
        return h_opt, c5_opt
    return h0, compute_c5(h0, n)


def _coordinate_descent(h, n, max_rounds=50, tol=1e-10):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    dx = 2.0 / n
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                delta = 0.01
                for sign in [1, -1]:
                    h_trial = best.copy()
                    dv = sign * delta
                    if h_trial[i] + dv < 0 or h_trial[i] + dv > 1:
                        continue
                    if h_trial[j] - dv < 0 or h_trial[j] - dv > 1:
                        continue
                    h_trial[i] += dv
                    h_trial[j] -= dv
                    c5_t = compute_c5(h_trial, n)
                    if c5_t < best_c5 - tol:
                        best = h_trial
                        best_c5 = c5_t
                        improved = True
                        break
                else:
                    continue
                break
        if not improved:
            # Try smaller step
            delta = 0.001
            for i in range(n):
                for j in range(n):
                    if i == j:
                        continue
                    for sign in [1, -1]:
                        dv = sign * delta
                        if 0 <= best[i] + dv <= 1 and 0 <= best[j] - dv <= 1:
                            h_trial = best.copy()
                            h_trial[i] += dv
                            h_trial[j] -= dv
                            c5_t = compute_c5(h_trial, n)
                            if c5_t < best_c5 - tol:
                                best = h_trial
                                best_c5 = c5_t
                                improved = True
                                break
                    else:
                        continue
                    break
            if not improved:
                break
    return best, best_c5
