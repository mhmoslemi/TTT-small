    while time.time() < end_time:
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        i = int(zeros[rng.randint(len(zeros))])
        j = int(ones[rng.randint(len(ones))])

        cur[i] = 1.0
        cur[j] = 0.0
        new_c5 = float(np.max(np.correlate(cur, 1.0 - cur, mode="full"))) * dx_local
        delta = new_c5 - cur_c5

        if delta < 0 or (T > 1e-5 and rng.random() < np.exp(-min(50.0, -delta / T))):
            cur_c5 = new_c5
            if new_c5 < best_local_c5:
                best_local_c5 = new_c5
                best_local = cur.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            cur[i] = 0.0
            cur[j] = 1.0
            stuck += 1

        T *= 0.99997
        if stuck > 2000:
            for _ in range(4):
                ones = np.where(cur == 1.0)[0]
                zeros = np.where(cur == 0.0)[0]
                ii = int(zeros[rng.randint(len(zeros))])
                jj = int(ones[rng.randint(len(ones))])
                cur[ii] = 1.0
                cur[jj] = 0.0
            cur_c5 = float(np.max(np.correlate(cur, 1.0 - cur, mode="full"))) * dx_local
            T = 0.025
            stuck = 0

    return best_local, best_local_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h = None

    if h is None or len(h) == 0:
        n = 70
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        n = len(h)

    h = _make_binary(h, n, rng)
    best_h = h.copy()
    best_c5 = _eval_c5(h)

    # Phase 1: Main SA at parent resolution
    rem = deadline - time.time()
    if rem > 20:
        d1 = rem * 0.55
        best_h, best_c5 = _sa_optimize(best_h, n, d1, rng, deadline)

    # Phase 2: Upsample to 2n and refine
    rem = deadline - time.time()
    if rem > 40:
        n2 = 2 * n
        h2 = np.repeat(best_h, 2).astype(float)
        h2 = _make_binary(h2, n2, rng)
        d2 = rem * 0.45
        h2_best, c5_2 = _sa_optimize(h2, n2, d2, rng, deadline)
        if c5_2 < best_c5:
            best_h = h2_best
            best_c5 = c5_2
            n = n2

    # Phase 3: Multi-start with structured and random inits at best resolution
    n_trial = len(best_h)
    n_half = n_trial // 2

    for trial in range(10):
        rem = deadline - time.time()
        if rem < 12:
            break

        if trial == 0 and n_trial == n:
            # Golden-ratio structured start
            phi = (1.0 + 5.0 ** 0.5) / 2.0
            h_try = np.zeros(n_trial)
            seen = set()
            for k in range(n_half * 3):
                pos = int(int(k * n_trial / (n_half * phi)) % n_trial)
                if pos not in seen and len(seen) < n_half:
                    seen.add(pos)
                    h_try[pos] = 1.0
            while len(seen) < n_half:
                p = int(rng.randint(n_trial))
                if p not in seen:
                    seen.add(p)
                    h_try[p] = 1.0
        elif trial == 1:
            # Block-alternating start
            h_try = np.zeros(n_trial)
            block = n_trial // (2 * n_half)
            for k in range(n_half):
                start = int(k * n_trial / n_half)
                end = min(start + max(block, 1), n_trial)
                h_try[start:end] = 1.0
            h_try = _make_binary(h_try, n_trial, rng)
        else:
            h_try = np.zeros(n_trial)
            h_try[rng.choice(n_trial, n_half, replace=False)] = 1.0

        d3 = min(rem * 0.35, 25.0)
        ht, ct = _sa_optimize(h_try, n_trial, d3, rng, deadline)
        if ct < best_c5:
            best_h = ht
            best_c5 = ct

    # Final refinement on best solution
    rem = deadline - time.time()
    if rem > 10:
        best_h, best_c5 = _sa_optimize(best_h, len(best_h), rem - 1, rng, deadline)

    final_c5 = _eval_c5(best_h)
    return (best_h, final_c5, len(best_h))
