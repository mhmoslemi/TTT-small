def compute_gradient_max_lag(h, n, dx, top_k=5):
    """Compute gradient of the top-k max correlations w.r.t. h"""
    corr = np.correlate(h, 1-h, mode='full')  # length 2n-1
    # Sort by value, get top_k indices
    top_idx = np.argsort(corr)[-top_k:]
    # Weight by softmax
    weights = np.exp(corr[top_idx] - np.max(corr[top_idx]))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        lag = idx - (n - 1)
        # Compute gradient for this lag

            j_plus = j + lag
            j_minus = j - lag
            term = 0
            if 0 <= j_plus < n:
                term += (1 - h[j_plus])
            if 0 <= j_minus < n:
                term -= h[j_minus]
            grad[j] += w * term * dx
    return grad

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

    grad = np.zeros(n)
    if lag == 0:
        grad = 1 - 2*h
    elif lag > 0:
        grad[:n-lag] += (1 - h[lag:])
        grad[lag:] -= h[:n-lag]
    else:
        l = -lag
        grad[l:] += (1 - h[:n-l])
        grad[:n-l] -= h[l:]
    return grad

    if budget_s - (time.time() - start_time) > 200:
        N2 = N * 2
        dx2 = 2.0 / N2
        target_sum2 = N2 / 2.0
        x_old = np.linspace(0, 1, N, endpoint=False)
        x_new = np.linspace(0, 1, N2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, target_sum2)

        bounds2 = [(0.0, 1.0)] * N2
        cons2 = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac,
                  'args': (target_sum2, N2)}]

        res = minimize(_obj, h_up, args=(N2,), method='SLSQP',
                       bounds=bounds2, constraints=cons2,
                       options={'maxiter': 1000, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.fun):
            h_new = project_feasible(res.x, target_sum2)
            c5_new = compute_c5(h_new, N2)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                N = N2

        # Subgradient at new resolution
        if best_h.size == N2:
            alpha2 = 0.003
            mom2 = np.zeros(N2)
            for it2 in range(50000):
                if budget_s - (time.time() - start_time) < 30:
                    break
                h = best_h
                corr2 = np.correlate(h, 1 - h, mode='full')
                c5_val2 = float(np.max(corr2) * dx2)
                top_idx2 = np.argsort(corr2)[-top_k:]
                top_vals2 = corr2[top_idx2]
                w2 = np.exp(top_vals2 - top_vals2.max())
                w2 /= w2.sum()
                grad2 = np.zeros(N2)
                for i in range(top_k):
                    lag2 = int(top_idx2[i] - (N2 - 1))
                    grad2 += w2[i] * compute_grad_for_k(h, N2, lag2)
                grad2 *= dx2
                mom2 = mu * mom2 - alpha2 * grad2
                h2 = h + mom2
                h2 = project_feasible(h2, target_sum2)
                c5_new2 = compute_c5(h2, N2)
                if c5_new2 < c5_val2 - 1e-15:
                    best_h = h2
                    best_c5 = c5_new2
                    alpha2 = min(alpha2 * 1.02, 0.05)
                else:
                    alpha2 = max(alpha2 * 0.95, 0.0001)

    # Phase 5: Final SLSQP polish
    if budget_s - (time.time() - start_time) > 15:
        final_N = len(best_h)
        final_ts = final_N / 2.0
        final_bounds = [(0.0, 1.0)] * final_N
        final_cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac,
                       'args': (final_ts, final_N)}]
        res = minimize(_obj, best_h, args=(final_N,), method='SLSQP',
                       bounds=final_bounds, constraints=final_cons,
                       options={'maxiter': 1000, 'ftol': 1e-15})
        if res.x is not None and np.isfinite(res.fun):
            h_new = project_feasible(res.x, final_ts)
            c5_new = compute_c5(h_new, final_N)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Ensure final feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
