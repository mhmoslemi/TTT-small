        return h_old.copy()

    indices = np.clip(indices, 0, n_old - 1)
    h_new = h_old[indices].astype(float).copy()

    current_ones = int(np.sum(h_new == 1.0))

        rng.shuffle(ones_idx)
        h_new[ones_idx[:excess]] = 0.0

        zeros_idx = np.where(h_new == 0.0)[0]
        rng.shuffle(zeros_idx)
        h_new[zeros_idx[:deficit]] = 1.0

import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    overlap = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(overlap)) * dx

def sa_optimize(h, n, time_limit, rng, T_init=0.02, T_min=0.0001, cooling=0.99999):
    """Simulated annealing with binary swap moves."""
    current = h.copy()
    current_c5 = compute_c5(current, n)
    best = current.copy()
    best_c5 = current_c5
    
    T = T_init
    start = time.time()
    
    while time.time() - start < time_limit:
        ones = np.where(current == 1.0)[0]
        zeros = np.where(current == 0.0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        
        current[i] = 0.0
        current[j] = 1.0
        
        new_c5 = compute_c5(current, n)
        delta = new_c5 - current_c5
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best = current.copy()
                best_c5 = new_c5
        else:
            current[i] = 1.0
            current[j] = 0.0
        
        T *= cooling
        if T < T_min:
            T = T_init  # restart temperature
    
    return best, best_c5

def upsample_binary(h_old, n_old, n_new, rng):
    """Upsample a binary array, preserving the fraction of ones."""
    if n_old == n_new:
        return h_old.copy()
    # Map new indices to old
    ratios = np.linspace(0, 1, n_new, endpoint=False)
    indices = (ratios * n_old).astype(int)
    indices = np.clip(indices, 0, n_old - 1)
    h_new = h_old[indices].astype(float).copy()
    
    # Fix the number of ones
    target = n_new // 2
    n_ones = int(np.sum(h_new == 1.0))
    if n_ones > target:
        excess = n_ones - target
        ones_idx = np.where(h_new == 1.0)[0]
        perm = rng.permutation(len(ones_idx))
        for idx in perm[:excess]:
            h_new[ones_idx[idx]] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zeros_idx = np.where(h_new == 0.0)[0]
        perm = rng.permutation(len(zeros_idx))
        for idx in perm[:deficit]:
            h_new[zeros_idx[idx]] = 1.0
    
    return h_new

        return time.time() - t_start

        return budget_s - 3 - elapsed()
    
    # Get and clean initial solution

    n0 = len(h0)
    h0 = (h0 > 0.5).astype(float)
    target0 = n0 // 2
    n_ones = int(np.sum(h0))
    if n_ones > target0:
        ones = np.where(h0 == 1.0)[0]

        for k in perm[:n_ones - target0]:
            h0[ones[k]] = 0.0

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    overlap = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(overlap)) * dx

def sa_optimize(h, n, t_limit, rng, T_init=0.03, T_min=0.0001):
    current = h.copy()

    best = current.copy()

        j = int(zeros[rng.integers(len(zeros))])

        current[j] = 1.0
        new_c5 = compute_c5(current, n)
