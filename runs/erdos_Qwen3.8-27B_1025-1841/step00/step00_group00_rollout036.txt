import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx


def get_active_grad(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    best_idx = np.argmax(c)
    shift = best_idx - (n - 1)
    n_lo = max(0, -shift)
    n_hi = min(n - 1, n - 1 - shift)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        n1 = i - shift
        if n_lo <= n1 <= n_hi:
            g += 1.0 - h[n1]
        if n_lo <= i <= n_hi and 0 <= i + shift < n:
            g -= h[i + shift]
        grad[i] = g
    return grad


def project_constrained(y, target_sum):
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
