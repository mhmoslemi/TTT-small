from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_fn(h, n_pts):
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sample_resize(h_src, n_dst):
    """Resize h to n_dst points preserving integral."""
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    if n_dst > n_src:
        # Upsample: each source point becomes n_dst//n_src points (approximately)
        ratio = n_dst / n_src
        h_dst = np.zeros(n_dst)
        for i in range(n_src):
            start = int(i * ratio)
            end = int((i + 1) * ratio)
            if end > start:
                h_dst[start:end] = h_src[i]
        # Fill any gaps
        # Actually this might not fill perfectly. Let's use a different approach.
        # Use linear interpolation on the center of each cell
        x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
        x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
        # For step function upsampling, nearest is better
        src_idx = np.clip(((x_dst / 2.0) * n_src).astype(int), 0, n_src - 1)
        h_dst = h_src[src_idx]
    else:
        # Downsample: average groups
        ratio = n_src / n_dst
        h_dst = np.zeros(n_dst)
        for i in range(n_dst):
            start = int(i * ratio)
            end = int((i + 1) * ratio)
            if end > start:
                h_dst[i] = np.mean(h_src[start:end])
    # Fix sum
    target = n_dst / 2.0
    current = np.sum(h_dst)
    if abs(current - target) > 1e-12:
        h_dst += (target - current) / n_dst
    h_dst = np.clip(h_dst, 0.0, 1.0)
    return h_dst

def project_polytope(v, n_pts):
    """Project v onto {0<=h<=1, sum(h)=n_pts/2}."""
    target = n_pts / 2.0
    # Bisection on tau: find tau such that sum(clip(v-tau, 0, 1)) = target
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        projected = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(v - tau, 0.0, 1.0)

def subgradient_c5(h, n_pts):
    """Compute subgradient of max_k corr(k) * dx."""
    dx = 2.0 / n_pts
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n_pts - 1)
    grad = np.zeros(n_pts)
    # d/dh[i] corr[idx] = (1-h[i+shift])*I(valid) - h[i-shift]*I(valid)
    # Term 1: (1-h[i+shift]) for i+shift in [0, n)
    lo1 = max(0, -shift)
    hi1 = min(n_pts, n_pts - shift)
    if hi1 > lo1:
        indices = np.arange(lo1, hi1)
        grad[indices] += (1.0 - h[indices + shift])
    # Term 2: -h[i-shift] for i-shift in [0, n)
    lo2 = max(0, shift)
    hi2 = min(n_pts, n_pts + shift)
    if hi2 > lo2:
        indices = np.arange(lo2, hi2)
        grad[indices] -= h[indices - shift]
    return grad * dx

def pgd_step(h, n_pts, lr, time_limit):
    """Projected gradient descent step."""
    t = time.time()
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)

            best_c5 = c5
        h_curr = h_next
        # Adaptive learning rate

    return h_start, obj(h_start)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_polytope(v, n_pts):
    target = n_pts / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(v - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(v - tau, 0.0, 1.0)


def subgradient_c5(h, n_pts):
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n_pts - 1)
    grad = np.zeros(n_pts)
    lo1 = max(0, -shift)
    hi1 = min(n_pts, n_pts - shift)
    if hi1 > lo1:
        idxs = np.arange(lo1, hi1)
        grad[idxs] += (1.0 - h[idxs + shift])
    lo2 = max(0, shift)
    hi2 = min(n_pts, n_pts + shift)
    if hi2 > lo2:
        idxs = np.arange(lo2, hi2)
        grad[idxs] -= h[idxs - shift]
    return grad * dx


def resize_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    if n_dst > n_src:
        x_dst = (np.arange(n_dst) + 0.5) / n_dst
        src_idx = np.clip((x_dst * n_src).astype(int), 0, n_src - 1)
        h_dst = h_src[src_idx].astype(float)
    else:
        ratio = n_src / n_dst
        h_dst = np.zeros(n_dst)
        for i in range(n_dst):
            s = int(i * ratio)
            e = int((i + 1) * ratio)
            e = min(e, n_src)
            if e > s:
                h_dst[i] = np.mean(h_src[s:e])
    h_dst = project_polytope(h_dst, n_dst)
    return h_dst


def pgd_optimize(h_init, n_pts, lr, n_iter, time_limit):
    t0 = time.time()
    h_curr = h_init.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        grad = subgradient_c5(h_curr, n_pts)
        h_next = h_curr - lr * grad
        h_next = project_polytope(h_next, n_pts)
        c5 = compute_c5(h_next)
        if c5 < best_c5:
            best_h = h_next.copy()
            best_c5 = c5
        h_curr = h_next
        if it > 0 and it % 200 == 0:
            lr *= 0.9
    return best_h, best_c5


def slsqp_optimize(h_start, n_pts, maxiter=200, time_limit=60):
    t0 = time.time()

    def obj(h):
        return compute_c5(h)

    def obj_grad(h):
        return subgradient_c5(h, n_pts)

    def eq_fun(h):
        return np.sum(h) - n_pts / 2.0

    def eq_jac(h):
        return np.ones(n_pts)

    bounds = [(0.0, 1.0)] * n_pts
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-9})
        if time.time() - t0 > time_limit:
            return h_start, compute_c5(h_start)
        if res.fun < compute_c5(h_start):
            return np.clip(res.x, 0.0, 1.0), float(compute_c5(res.x))
    except Exception:
        pass
    return h_start, compute_c5(h_start)


def nelder_mead_optimize(h_start, n_pts, time_limit=120):
    t0 = time.time()

    def obj(h):
        if time.time() - t0 > time_limit:
            return 1e10
        return compute_c5(h)

    try:
        res = minimize(obj, h_start, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-7,
                                'adaptive': True})
        h_res = project_polytope(res.x, n_pts)
        return h_res, compute_c5(h_res)
    except Exception:
        pass
    return h_start, compute_c5(h_start)
