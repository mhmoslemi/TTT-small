    scaled = tau * corr * dx
    m = np.max(scaled)
    if m > 500.0:
        return m / tau
    return (m + np.log(np.sum(np.exp(scaled - m)))) / tau

def constraint_eq(x, n_points):
    return np.sum(x) - n_points / 2.0

def constraint_eq_jac(x, n_points):
    return np.ones(n_points)

def enforce_constraints(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n_points
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            mask = (h > 0.0) & (h < 1.0)
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
    return h

def resize_solution(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def single_optimization(h_start, n, tau, maxiter, t_start, budget):
    t_limit = t_start + budget
    if time.time() > t_limit:
        return None
    res = minimize(
        smooth_max_obj, h_start,
        args=(n, tau),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac, 'args': (n,)}],
        options={'maxiter': maxiter, 'ftol': 1e-9}
    )
    return res

def projected_gradient_step(h, n, tau, lr, n_iter, t_start, budget):
    t_limit = t_start + budget
    h = enforce_constraints(h, n)
    for i in range(n_iter):
        if time.time() > t_limit:
            break
        h_old = h.copy()
        eps = 1e-5
        grad = np.zeros(n)
        for j in range(n):
            h_p = h.copy()
            h_p[j] += eps
            h_p = enforce_constraints(h_p, n)
            h_m = h.copy()
            h_m[j] -= eps
            h_m = enforce_constraints(h_m, n)
            grad[j] = (smooth_max_obj(h_p, n, tau) - smooth_max_obj(h_m, n, tau)) / (2.0 * eps)
        h = h - lr * grad
        h = enforce_constraints(h, n)
        if np.max(np.abs(h - h_old)) < 1e-10:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = initial_h_values.copy()
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init)

    resolutions = [80, 120, 160, 200, 240, 320, 400, 480]

    for n in resolutions:
        if time.time() - t0 > budget_s * 0.7:
            break

        if n == len(best_h):
            h_base = best_h.copy()
        else:
            h_base = resize_solution(best_h, n)
            h_base = enforce_constraints(h_base, n)

        n_attempts = 4 if n <= 200 else 3

        for attempt in range(n_attempts):
            if time.time() - t0 > budget_s * 0.85:
                break

            if attempt == 0:
                h_try = h_base.copy()
            elif attempt == 1:
                h_try = h_base + rng.normal(0, 0.01, n)
            elif attempt == 2:
                h_try = h_base + rng.normal(0, 0.03, n)
            else:
                h_try = h_base * 0.6 + 0.4 * rng.uniform(0.2, 0.8, n)

            h_try = enforce_constraints(h_try, n)

            for tau in [15.0, 30.0, 60.0]:
                if time.time() - t0 > budget_s * 0.85:
                    break

                maxiter = 400 if n <= 200 else 250
                res = single_optimization(h_try, n, tau, maxiter, t0, budget_s * 0.85 - (time.time() - t0))

                if res is not None and res.x is not None:
                    if np.all(res.x >= -0.05) and np.all(res.x <= 1.05):
                        h_res = enforce_constraints(res.x, n)
                        c5 = evaluate_erdos_solution(h_res)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h_res.copy()
                            h_base = h_res.copy()

        # Update base for next resolution
        if len(best_h) != n:
            h_base = resize_solution(best_h, n)
            h_base = enforce_constraints(h_base, n)
        else:
            h_base = best_h.copy()

    # Final polish at best resolution
    n_final = len(best_h)
    h_final = best_h.copy()

    for tau in [50.0, 100.0, 200.0]:
        if time.time() - t0 > budget_s * 0.92:
            break
        res = single_optimization(h_final, n_final, tau, 300, t0, budget_s * 0.95 - (time.time() - t0))
        if res is not None and res.x is not None:
            if np.all(res.x >= -0.05) and np.all(res.x <= 1.05):
                h_res = enforce_constraints(res.x, n_final)
                c5 = evaluate_erdos_solution(h_res)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_res.copy()
                    h_final = h_res.copy()

    # Projected gradient polish
    if time.time() - t0 < budget_s * 0.95:
        h_pg = projected_gradient_step(h_final, n_final, 30.0, 0.01, 50, t0, budget_s * 0.98 - (time.time() - t0))
        c5_pg = evaluate_erdos_solution(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    return best_h, best_c5, len(best_h)
