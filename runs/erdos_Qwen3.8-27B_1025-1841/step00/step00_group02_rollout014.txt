from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_correlation(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_power(h, p=8):
    corr = get_correlation(h)
    if np.max(corr) < 1e-15:
        return 0.0
    return float(np.mean(corr**p)**(1.0/p))

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # Fix residual by adjusting non-saturated entries
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def local_refine(h, n, n_iter=100, lr=0.01):
    """Local search: reduce the peak correlation lag."""
    best = h.copy()
    best_val = compute_c5(best)
    for it in range(n_iter):
        corr = get_correlation(best)
        peak_idx = np.argmax(corr)
        peak_val = corr[peak_idx]
        if peak_val < 1e-10:
            break
        # Subgradient direction: increase h where it reduces the peak lag
        n_pts = len(best)
        tau = peak_idx - (n_pts - 1)
        grad = np.zeros(n_pts)
        for i in range(n_pts):
            # dR/dh[i] at this lag
            j = i - tau
            if 0 <= j < n_pts:
                grad[i] += (1.0 - best[j])
            j2 = i + tau
            if 0 <= j2 < n_pts:
                grad[i] -= best[j2]
        # We want to DECREASE the peak, so go against gradient
        # But we need to maintain sum constraint
        grad -= np.mean(grad)  # project out mean (preserves sum)
        best_new = best - lr * grad
        best_new = make_feasible(best_new, n_pts)
        new_val = compute_c5(best_new)
        if new_val < best_val:
            best = best_new
            best_val = new_val
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break
    return best, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_power(h, p=8.0):
    corr = get_corr(h)
    mx = np.max(corr)
    if mx < 1e-15:
        return 0.0
    return float(np.mean(corr**p)**(1.0/p))

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_c5_power(h, p=8.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    N = 2 * n - 1
    mx = np.max(corr)
    if mx < 1e-15:
        return np.zeros(n)
    Mp = np.mean(corr**p)**(1.0/p)
    w = corr**(p-1)
    grad = np.zeros(n)
    prefix = Mp**(1-p) / N
    for j in range(n):
        t_lo = max(-(n-1), j - (n-1))
        t_hi = min(n-1, j)
        idx_lo = t_lo + (n-1)
        idx_hi = t_hi + (n-1)
        taus1 = np.arange(idx_lo, idx_hi+1) - (n-1)
        part1 = np.sum(w[idx_lo:idx_hi+1] * (1.0 - h[j - taus1]))
        t_lo2 = max(-(n-1), -j)
        t_hi2 = min(n-1, n-1-j)
        idx_lo2 = t_lo2 + (n-1)
        idx_hi2 = t_hi2 + (n-1)
        taus2 = np.arange(idx_lo2, idx_hi2+1) - (n-1)
        part2 = np.sum(w[idx_lo2:idx_hi2+1] * h[j + taus2])
        grad[j] = prefix * (part1 - part2)
    return grad

def subgrad_c5(h, p=8.0):
    n = len(h)
    corr = get_corr(h)
    peak_idx = np.argmax(corr)
    tau = peak_idx - (n-1)
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j - tau < n:
            grad[j] += (1.0 - h[j - tau])
        if 0 <= j + tau < n:
            grad[j] -= h[j + tau]
    grad -= np.mean(grad)
    return grad

def interp_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    idx = np.linspace(0, old_n - 1, new_n)
    i0 = np.floor(idx).astype(int)
    i1 = np.minimum(i0 + 1, old_n - 1)
    frac = idx - i0
    return h[i0] * (1.0 - frac) + h[i1] * frac

def local_search(h, n, max_iter=200, lr_init=0.02):
    best = h.copy()
    best_val = compute_c5(best)
    lr = lr_init
    for _ in range(max_iter):
        corr = get_corr(best)
        peak_idx = np.argmax(corr)
        tau = peak_idx - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j - tau < n:
                grad[j] += (1.0 - best[j - tau])
            if 0 <= j + tau < n:
                grad[j] -= best[j + tau]
        grad -= np.mean(grad)
        new_h = make_feasible(best - lr * grad, n)
        new_val = compute_c5(new_h)
        if new_val < best_val - 1e-12:
            best = new_h
            best_val = new_val
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

            best_n = len(best_h)
    except Exception:
        pass

        best_c5 = compute_c5(best_h)
        best_n = n0

        # Phase A: Nelder-Mead (good for non-smooth, no gradient)

                                        'xatol': 1e-10, 'fatol': 1e-10})
                h_a = make_feasible(res.x, n)
                c5_a = compute_c5(h_a)
                if c5_a < best_c5:
                    best_c5 = c5_a
                    best_h = h_a.copy()
                    best_n = n

            except Exception:
                pass

        # Phase B: L-BFGS-B with smooth objective and gradient

                return g * (n/2.0) / s

                h_b = make_feasible(res.x, n)

                h0 = h_b
            except Exception:

        if remaining() > 30:

            if c5_c < best_c5:

                    res = minimize(obj_r, h_r, method='Nelder-Mead',

                        best_n = n

    # Final polish at best resolution

        # Multiple local searches from best
