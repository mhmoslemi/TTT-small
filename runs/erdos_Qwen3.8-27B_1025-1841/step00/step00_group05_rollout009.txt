import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def make_binary(h, n, rng):
    b = (h >= 0.5).astype(float)
    n_ones = int(b.sum())
    half = n // 2
    if n_ones > half:
        idx = np.where(b == 1)[0]
        to_flip = rng.choice(idx, size=n_ones - half, replace=False)
        b[to_flip] = 0.0
    elif n_ones < half:
        idx = np.where(b == 0)[0]
        to_flip = rng.choice(idx, size=half - n_ones, replace=False)
        b[to_flip] = 1.0
    return b

def local_search_binary(h, n, rng, n_iters, temp0, cool, deadline):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best_c5 = cur_c5
    best = cur.copy()
    temp = temp0
    n_half = n // 2
