- This is equivalent to finding h' that minimizes ||h' - h||^2 subject to sum(h') = n/2, 0 ≤ h' ≤ 1
- This is a convex QP that can be solved by bisection on the Lagrange multiplier

The projection algorithm:
- Find λ such that sum(clip(h - λ, 0, 1)) = n/2
- Use bisection on λ (since the sum is monotonically decreasing in λ)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_set(h, n):
    """Project h onto {sum(h)=n/2, 0<=h<=1}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient(h):
    """Compute a subgradient of the max objective at h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    k_max = np.argmax(corr)
    m = k_max - (n - 1)  # lag
    
    grad = np.zeros(n)
    if m >= 0:
        for j in range(n - m):
            grad[j] += (1 - h[j + m])
        for j in range(m, n):
            grad[j] -= h[j - m]
    else:
        p = -m
        for j in range(p, n):
            grad[j] += (1 - h[j - p])
        for j in range(n - p):
            grad[j] -= h[j + p]
    
    return grad * dx

def projected_gradient_step(h, lr, n):
    """Take one projected gradient step."""
    g = subgradient(h)
    h_new = h - lr * g
    return project_onto_set(h_new, n)

def line_search(h, n, max_steps=50, initial_lr=0.1):
    """Backtracking line search."""
    f0 = compute_c5(h)
    lr = initial_lr
    for _ in range(max_steps):
        h_new = projected_gradient_step(h, lr, n)
        f_new = compute_c5(h_new)
        # Check for sufficient decrease (since we're using subgradient, just check decrease)
        if f_new < f0:
            return h_new, lr
        lr *= 0.5
    return h, lr

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp((corr - m) * alpha))) / alpha

    grad = np.zeros(n)

            for j in range(p, n):
                grad[j] += w * (1 - h[j - p])
            for j in range(n - p):

For our problem (size n/2 in a set of size n), the relevant concept is a "half-set" with balanced differences. This is related to "discrete Fourier transform" analysis.

If h is the indicator of a set S, then corr[m] = |S ∩ (S+m)^c| = |S| - |S ∩ (S+m)| = n/2 - |S ∩ (S+m)|.

So minimizing max corr[m] is equivalent to maximizing min_m |S ∩ (S+m)|, i.e., maximizing the minimum autocorrelation.

For a "random" set S of size n/2, |S ∩ (S+m)| ≈ n/4 for each m, with fluctuations of order √(n/16) = √n/4.

So a random set gives corr[m] ≈ n/4 ± √n/4, and max corr ≈ n/4 + c√n for some constant c.
C₅ ≈ 1/2 + c/(2√n).

This converges to 1/2 as n → ∞, which matches the uniform h(x) = 1/2 case.

import numpy as np
from scipy.optimize import minimize
import time

_target_n = 0

def project_onto_set(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0, 1)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp((corr - m) * alpha))) / alpha

def smooth_c5_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m_max = np.max(corr)
    exp_vals = np.exp((corr - m_max) * alpha)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            for j in range(n - m):
                grad[j] += w * (1.0 - h[j + m])
            for j in range(m, n):
                grad[j] -= w * h[j - m]
        else:
            p = -m
            for j in range(p, n):
                grad[j] += w * (1.0 - h[j - p])
            for j in range(n - p):
                grad[j] -= w * h[j + p]
    return grad

def subgradient_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    k_max = np.argmax(corr)
    m = k_max - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        for j in range(n - m):
            grad[j] += (1.0 - h[j + m])
        for j in range(m, n):
            grad[j] -= h[j - m]
    else:
        p = -m
        for j in range(p, n):
            grad[j] += (1.0 - h[j - p])
        for j in range(n - p):
            grad[j] -= h[j + p]
    return grad * dx

def constraint_fun(x):
    return np.sum(x) - _target_n / 2.0

def constraint_jac(x):
    return np.ones(_target_n)

def projected_gradient_descent(h, n, max_steps=2000, time_limit_fn=None):
    global _target_n
    _target_n = n
    f0 = compute_c5(h)
    best_h = h.copy()
    best_f = f0
    lr = 1.0 / n
    for step in range(max_steps):
        if time_limit_fn is not None and time_limit_fn():
            break
        g = subgradient_max(h)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        h_new = h - lr * g
        h_new = project_onto_set(h_new, n)
        f_new = compute_c5(h_new)
        if f_new < f0:
            f0 = f_new
            h = h_new
            if f_new < best_f:
                best_f = f_new
                best_h = h_new.copy()
            lr = min(lr * 1.05, 2.0 / n)
        else:
            lr *= 0.7
    return best_h, best_f

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1300, **kwargs):
    global _target_n
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_init = 42
        h0 = np.full(n_init, 0.5)
        h0[:n_init // 2] = 1.0
        h0[n_init // 2:] = 0.0

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def has_time(frac):
        return elapsed() < budget_s * frac

    resolutions = sorted(set([r for r in [n0, 2 * n0, 4 * n0, 64, 128, 256, 512] if r >= 42]))

    current_h = h0.copy()
    current_n = n0

    for target_n in resolutions:
        if not has_time(0.80):
            break
        if target_n > 1024:
            continue

        if target_n == current_n:
            h_work = current_h.copy()
        else:
            h_work = interpolate_h(current_h, current_n, target_n)
        h_work = np.clip(h_work, 0, 1)
        h_work = project_onto_set(h_work, target_n)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

        # SLSQP with smoothed objective
        if has_time(0.70) and target_n <= 512:
            _target_n = target_n
            for alpha in [20, 60, 150, 400]:
                if not has_time(0.60):
                    break
                try:
                    res = minimize(
                        smooth_c5,
                        h_work,
                        args=(alpha,),
                        jac=lambda x, a=alpha: smooth_c5_grad(x, a),
                        method='SLSQP',
                        bounds=[(0.0, 1.0)] * target_n,
                        constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                        options={'maxiter': 200, 'ftol': 1e-12}
                    )
                    h_cand = np.clip(res.x, 0, 1)
                    h_cand = project_onto_set(h_cand, target_n)
                    c5_cand = compute_c5(h_cand)
                    if c5_cand < c5_work:
                        h_work = h_cand
                        c5_work = c5_cand
                        if c5_work < best_c5:
                            best_c5 = c5_work
                            best_h = h_work.copy()
                except Exception:
                    pass

        # Projected gradient with exact max
        if has_time(0.45):
            h_work, c5_work = projected_gradient_descent(
                h_work, target_n, max_steps=3000,
                time_limit_fn=lambda: not has_time(0.40)
            )
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()

        # Final SLSQP polish
        if has_time(0.30) and target_n <= 256:
            _target_n = target_n
            try:
                res = minimize(
                    smooth_c5,
                    h_work,
                    args=(500,),
                    jac=lambda x, a=500: smooth_c5_grad(x, a),
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * target_n,
                    constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                    options={'maxiter': 300, 'ftol': 1e-14}
                )
                h_cand = np.clip(res.x, 0, 1)
                h_cand = project_onto_set(h_cand, target_n)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
            except Exception:
                pass

        current_h = h_work
        current_n = target_n

    # Try a few random perturbations of best solution
    if has_time(0.15):
        n_best = len(best_h)
        for attempt in range(5):
            if not has_time(0.10):
                break
            h_pert = best_h + np.random.randn(n_best) * 0.05
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = project_onto_set(h_pert, n_best)
            _target_n = n_best
            h_pert, _ = projected_gradient_descent(h_pert, n_best, max_steps=500,
                                                    time_limit_fn=lambda: not has_time(0.08))
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    return best_h, best_c5, len(best_h)
