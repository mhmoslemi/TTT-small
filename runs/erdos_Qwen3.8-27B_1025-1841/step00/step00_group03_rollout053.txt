def binary_initial(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def block_initial(n):
    h = np.zeros(n)
    block = n // 4
    h[0:block] = 1.0
    h[n//2:n//2 + block] = 1.0
    return repair(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _current_n
    t0 = time.time()

    def elapsed_frac():
        return (time.time() - t0) / budget_s

    n_init = 70
    h_init = np.array(initial_h_values, dtype=float).copy()
    h_init = repair(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5_exact(h_init)
    best_n = n_init

    # Phase 1: SLSQP at n=70 from initial
    if elapsed_frac() < 0.05:
        dl = t0 + budget_s * 0.12
        h_opt, c5_opt = slsqp_optimize(h_init, n_init, alpha=30, max_iter=300, deadline=dl)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_init

    # Phase 1b: Multi-start SLSQP at n=70
    starts = [random_initial(n_init, seed + 1), binary_initial(n_init, seed + 2), block_initial(n_init)]
    for si in range(len(starts)):
        if elapsed_frac() > 0.22:
            break
        dl = t0 + budget_s * (0.22 + 0.03 * si)
        h_opt, c5_opt = slsqp_optimize(starts[si], n_init, alpha=25, max_iter=150, deadline=dl)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_init

    # Phase 2: SLSQP polish at n=70
    if elapsed_frac() < 0.28:
        dl = t0 + budget_s * 0.32
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, alpha=80, max_iter=500, deadline=dl)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_init

    # Phase 3: Simulated annealing at n=70
    if elapsed_frac() < 0.50:
        sa_start = time.time()
        sa_budget = budget_s * 0.25 - (sa_start - t0)
        if sa_budget > 20:
            h_sa, c5_sa = sim_anneal(best_h, best_n, sa_budget, seed + 500)
            if c5_sa < best_c5:
                best_h, best_c5, best_n = h_sa.copy(), c5_sa, best_n

    # SLSQP polish after SA
    if elapsed_frac() < 0.60:
        dl = t0 + budget_s * 0.63
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, alpha=80, max_iter=300, deadline=dl)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, best_n

    # Phase 4: Higher resolution n=140
    if elapsed_frac() < 0.70:
        n_high = 140
        h_inter = interpolate_resolve(best_h, best_n, n_high)
        dl = t0 + budget_s * 0.80
        h_opt, c5_opt = slsqp_optimize(h_inter, n_high, alpha=20, max_iter=250, deadline=dl)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_high

    # Final SA polish at best resolution
    if elapsed_frac() < 0.88:
        remaining = budget_s * 0.97 - (time.time() - t0)
        if remaining > 15:
            h_sa, c5_sa = sim_anneal(best_h, best_n, remaining, seed + 999)
            if c5_sa < best_c5:
                best_h, best_c5, best_n = h_sa.copy(), c5_sa, best_n

    # Final polish
    if elapsed_frac() < 0.97:
        dl = t0 + budget_s * 0.99
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, alpha=100, max_iter=200, deadline=dl)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, best_n

    # Final validation
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5_exact(best_h)

    return (best_h, best_c5, best_n)
