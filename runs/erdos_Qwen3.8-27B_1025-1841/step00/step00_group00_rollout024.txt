def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) < 1e-12:
        return h
    diff = target - np.sum(h)
    if diff > 0:
        # Need to add diff total
        # Add to elements not at 1, but weighted by how much room they have
        # Simple: add uniformly to non-saturated elements
        mask = h < 1.0
        count = mask.sum()
        if count == 0:
            return h
        # We need to add diff total across count elements, each can take at most (1-h[i])
        # Binary search for the right add amount
        # If diff <= count * min_room, just add diff/count to each
        rooms = 1.0 - h[mask]
        if diff <= count * rooms.min():
            h[mask] += diff / count
        else:
            # Water filling
            h[mask] += diff / count  # some will hit 1
            h = np.clip(h, 0, 1)
            # Handle remainder
            rem = target - h.sum()
            mask2 = h < 1.0
            if rem > 0 and mask2.sum() > 0:
                h[mask2] += rem / mask2.sum()
            h = np.clip(h, 0, 1)
    else:
        mask = h > 0.0
        count = mask.sum()
        if count == 0:
            return h
        floors = h[mask]
        if -diff <= count * floors.min():
            h[mask] += diff / count  # diff is negative, so subtract
        else:
            h[mask] += diff / count
            h = np.clip(h, 0, 1)
            rem = target - h.sum()
            mask2 = h > 0.0
            if rem < 0 and mask2.sum() > 0:
                h[mask2] += rem / mask2.sum()
            h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.signal import correlate as scipy_correlate
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = scipy_correlate(h, 1.0 - h, mode='full', method='fft')
    return float(np.max(corr) * dx)

def fast_corr(h):
    return scipy_correlate(h, 1.0 - h, mode='full', method='fft')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
        h = np.clip(h, 0, 1)
        rem = target - h.sum()
        if abs(rem) > 1e-13:
            m2 = h < 1.0
            if m2.sum() > 0:
                h[m2] += rem / m2.sum()
                h = np.clip(h, 0, 1)
    else:
        mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
        h = np.clip(h, 0, 1)
        rem = target - h.sum()
        if abs(rem) > 1e-13:
            m2 = h > 0.0
            if m2.sum() > 0:
                h[m2] += rem / m2.sum()
                h = np.clip(h, 0, 1)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def subgrad(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        m = -s
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

        c5s = (xm + np.log(w_sum)) / beta

        if c5s < best_local - 1e-10:
            best_local = c5s
