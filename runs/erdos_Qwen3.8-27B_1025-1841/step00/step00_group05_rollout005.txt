import numpy as np
from scipy import optimize
import time

_OPT_N = 70
_OPT_TS = 35.0

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def _project(h, ts):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(200):
        s = np.sum(h)
        d = ts - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = np.sum(m)
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def _set_opt(n, ts):
    global _OPT_N, _OPT_TS
    _OPT_N = n
    _OPT_TS = ts

def _obj_fn(h_raw):
    hp = _project(h_raw, _OPT_TS)
    return _eval_c5(hp)

def _grad_smoothed(h, temp):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    cs = conv / temp
    mx = np.max(cs)
    w = np.exp(cs - mx)
    w /= np.sum(w)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kp = -k
            if kp < n:
                grad[:n - kp] += wi * (1.0 - h[kp:])
                grad[kp:] -= wi * h[:n - kp]
    return dx * grad

def _proj_grad_method(h0, n, ts, max_iter=5000):
    h = _project(h0, ts).copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    dx = 2.0 / n
    alpha = 0.01
    temp = 1.0
    stall = 0
    for it in range(max_iter):
        temp = max(0.02, 1.0 * np.exp(-it / 500.0))
        g = _grad_smoothed(h, temp)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        step = alpha / max(gnorm, 1e-8)
        h_new = _project(h - step * g, ts)
        c5_new = _eval_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            alpha = min(alpha * 1.05, 0.5)
        else:
            h = h_new
            stall += 1
            alpha *= 0.9
            if stall > 100:
                alpha = 0.01
                stall = 0
    return best_h, best_c5

def _powell_opt(h0, n, ts, max_iter=2000):
    _set_opt(n, ts)
    try:
        res = optimize.minimize(_obj_fn, h0, method='Powell',
                                options={'maxiter': max_iter, 'xtol': 1e-9, 'ftol': 1e-12})
        ho = _project(res.x, ts)
        return ho, _eval_c5(ho)
    except Exception:
        return h0, _eval_c5(h0)

def _nm_opt(h0, n, ts, max_iter=3000):
    _set_opt(n, ts)
    try:
        res = optimize.minimize(_obj_fn, h0, method='Nelder-Mead',
                                options={'maxiter': max_iter, 'xatol': 1e-10, 'fatol': 1e-12})
        ho = _project(res.x, ts)
        return ho, _eval_c5(ho)
    except Exception:
        return h0, _eval_c5(h0)

def _make_struct(n, kind, rng):
    idx = np.arange(n)
    if kind == 0:
        h = np.full(n, 0.5)
    elif kind == 1:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif kind == 2:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif kind == 3:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * idx / n)
    elif kind == 4:
        h = 0.5 + 0.3 * np.sin(4 * np.pi * idx / n)
    elif kind == 5:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * idx / n) + 0.2 * np.cos(4 * np.pi * idx / n)
    elif kind == 6:
        h = rng.uniform(0, 1, n)
    elif kind == 7:
        h = np.zeros(n)
        for f in range(1, 6):
            h += rng.uniform(-0.3, 0.3) * np.sin(2 * np.pi * f * idx / n + rng.uniform(0, 6.28))
        h += 0.5
    else:
        h = 0.5 + 0.3 * np.sin(np.pi * idx / n)
    return _project(h, n / 2.0)

def _coord_descent(h, n, ts, max_rounds=200):
    h = _project(h, ts).copy()
    best_c5 = _eval_c5(h)
    improvement = True
    rounds = 0
    while improvement and rounds < max_rounds:
        improvement = False
        rounds += 1
        for i in range(n):
            for trial_val in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(trial_val - h[i]) < 1e-9:
                    continue
                h_copy = h.copy()
                h_copy[i] = trial_val
                h_copy = _project(h_copy, ts)
                c5 = _eval_c5(h_copy)
                if c5 < best_c5 - 1e-12:
                    h = h_copy
                    best_c5 = c5
                    improvement = True
                    break
    return h, best_c5
