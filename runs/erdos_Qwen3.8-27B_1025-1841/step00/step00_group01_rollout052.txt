            h_bin[idx[order[k]]] = 1.0
    c5_bin = _eval(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin

    best_h = _fix(best_h, best_n / 2.0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
