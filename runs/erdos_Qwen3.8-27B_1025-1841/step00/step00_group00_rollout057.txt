import scipy.optimize as so

def compute_obj(h, n):
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_box_sum(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = 0.5 * n
    # Bisection for projection
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fix any tiny sum error
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        free = (proj > 1e-7) & (proj < 1.0 - 1e-7)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def grad_c(h, k, n):
    """Gradient of c[k] = np.correlate(h,1-h,'full')[k] w.r.t. h"""
    offset = k - (n - 1)
    j_lo = max(0, -offset)
    j_hi = min(n-1, n-1-offset)
    g = np.zeros(n)
    if j_lo > j_hi:
        return g
    if offset == 0:
        g[j_lo:j_hi+1] = 1.0 - 2.0*h[j_lo:j_hi+1]
    else:
        js = np.arange(j_lo, j_hi+1)
        g[js] += 1.0 - h[js + offset]
        g[js + offset] -= h[js]
    return g

def subgrad_optimize(h, n, time_limit, rng):
    """Subgradient descent for minimize max_k c[k]*dx"""
    start = time()
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(h, 1.0-h, mode="full")
    best_val = corr.max() * dx
    alpha = 0.05
    min_alpha = 1e-7
    
    while time() - start < time_limit:
        k_star = np.argmax(corr)
        if corr[k_star] * dx < best_val:
            best_val = corr[k_star] * dx
            best_h = h.copy()
        
        g = grad_c(h, k_star, n)
        
        # Try a few step sizes
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25]:
            trial = h - a * g
            trial = project_box_sum(trial, n)
            new_corr = np.correlate(trial, 1.0-trial, mode="full")
            if new_corr.max() * dx < corr[k_star] * dx:
                h = trial
                corr = new_corr
                alpha = min(a * 1.5, 0.2)
                improved = True
                break
        
        if not improved:
            alpha *= 0.7
            if alpha < min_alpha:
                break
        
        # Occasionally do a random perturbation to escape local min
        if rng.random() < 0.05:
            h = h + rng.normal(0, 0.01, n)
            h = project_box_sum(h, n)
            corr = np.correlate(h, 1.0-h, mode="full")
    
    return best_h, best_val

def sa_optimize(h, n, time_limit, rng):
    """Simulated annealing"""
    start = time()
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_obj(h, n)
    cur_h = h.copy()
    cur_val = best_val
    temp = 0.05
    cool = 0.999
    
    while time() - start < time_limit:
        # Generate neighbor: change a few random coordinates
        idx = rng.integers(0, n)
        delta = rng.uniform(-0.1, 0.1)
        trial = cur_h.copy()
        trial[idx] = np.clip(trial[idx] + delta, 0, 1)
        trial = project_box_sum(trial, n)
        new_val = compute_obj(trial, n)
        
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-8)):
            cur_h = trial
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
        
        temp *= cool
    
    return best_h, best_val

def upsample_to(h, old_n, new_n):
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0, 1)
    h_new = project_box_sum(h_new, new_n)
    return h_new

def polish_scipy(h, n, time_limit, rng):
    """Use scipy SLSQP for polishing"""
    start = time()
    dx = 2.0 / n
    
    def obj(x):
        return np.max(np.correlate(x, 1.0-x, mode="full")) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5*n}]
    bounds = [(0, 1)] * n
    
    elapsed = time() - start
    max_iter = max(50, int((time_limit - elapsed) / max(0.5, (time() - start) / max(1, 0))))
    
    try:
        res = so.minimize(obj, h, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.success or res.fun < obj(h):
            return np.clip(res.x, 0, 1), obj(res.x)
    except:
        pass
    return h, obj(h)
