        x_new = np.linspace(0, 1, n_try, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_try / 2.0)
        
        buf = min(120, remaining() * 0.2)
        h7, c5_7 = subgradient_optimize(h_up, n_try, buf, margin=2.0)
        if c5_7 < best_c5:
            best_h, best_c5 = h7, c5_7
        
        if remaining() < 40:
            break
        h8, c5_8 = local_search(best_h, len(best_h), min(50, remaining() * 0.1))
        if c5_8 < best_c5:
            best_h, best_c5 = h8, c5_8
    
    # Phase 6: Final polish with more aggressive local search
    if remaining() > 30:
        h_fin, c5_fin = local_search(best_h, len(best_h), min(100, remaining() * 0.5), n_random=800)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = eval_c5(best_h)
    
    return best_h, best_c5, n_final
