Wait, I need to be more careful. c_k = sum_n h[n+k] * (1-h[n]) where n ranges over valid indices.

For a fixed j:
- h[j] appears in h[n+k] when n+k = j, i.e., n = j-k. The derivative contribution is (1-h[j-k]) * I[j-k valid].
- h[j] appears in (1-h[n]) when n = j. The derivative contribution is -h[j+k] * I[j+k valid].

So: ∂c_k/∂h[j] = (1-h[j-k]) * I[0≤j-k<n] - h[j+k] * I[0≤j+k<n]

This is O(n) to compute for a given k. So the full gradient is O(n) per component, O(n²) total.

```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def project_feasible(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1}"""
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def c5_gradient(h, n):
    """Compute gradient of C5 = max_k dx * c_k at h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)  # 0-indexed in the full output
    # k_max corresponds to lag k = k_max - (n-1)
    lag = k_max - (n - 1)
    
    # grad[j] = (1 - h[j-lag]) * I[0<=j-lag<n] - h[j+lag] * I[0<=j+lag<n]
    grad = np.zeros(n)
    # First term: (1 - h[j-lag]) for j where 0 <= j-lag < n
    # j-lag >= 0 → j >= lag
    # j-lag < n → j < n + lag
    j_start1 = max(0, lag)
    j_end1 = min(n, n + lag)
    if j_start1 < j_end1:
        grad[j_start1:j_end1] += (1.0 - h[j_start1-lag:j_end1-lag])
    
    # Second term: -h[j+lag] for j where 0 <= j+lag < n
    # j+lag >= 0 → j >= -lag
    # j+lag < n → j < n - lag
    j_start2 = max(0, -lag)
    j_end2 = min(n, n - lag)
    if j_start2 < j_end2:
        grad[j_start2:j_end2] -= h[j_start2+lag:j_end2+lag]
    
    return dx * grad

def projected_gradient(h, n, lr=0.01, max_iter=2000, tol=1e-10):
    """Projected gradient descent to minimize C5"""
    target = n / 2.0
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_c5 = c5_objective(h)
    
    for it in range(max_iter):
        c5 = c5_objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        grad = c5_gradient(h, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        if np.max(np.abs(h_new - h)) < tol:
            break
        h = h_new
    
    return best_h, best_c5
```

Also, the gradient I computed is for the k* that achieves the maximum at the current point. If the maximizer changes, the gradient can jump discontinuously. This is a known issue with optimizing the max of functions.

A smoother alternative: use the log-sum-exp:
f_smooth(h) = (1/α) * log(sum_k exp(α * dx * c_k))

This approximates max_k for large α, and is smooth. The gradient is:
∇f_smooth = sum_k (exp(α*dx*c_k) / sum_j exp(α*dx*c_j)) * dx * ∇c_k

This is a weighted average of the gradients, which is smoother and might converge better.

```python
def c5_logsumexp(h, n, alpha=100.0):
    """Smooth approximation of C5 using log-sum-exp"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = alpha * dx * corr
    # Numerical stability
    vals -= np.max(vals)
    return (1.0 / alpha) * (np.max(alpha * dx * corr) + np.log(np.sum(np.exp(vals))))

def c5_logsumexp_grad(h, n, alpha=100.0):
    """Gradient of log-sum-exp C5"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = alpha * dx * corr
    vals_shifted = vals - np.max(vals)
    weights = np.exp(vals_shifted) / np.sum(np.exp(vals_shifted))
    
    # Gradient is weighted average of gradient of each c_k
    grad = np.zeros(n)
    for k_idx, w in enumerate(weights):
        if w < 1e-10:
            continue
        lag = k_idx - (n - 1)
        # ∂c_k/∂h[j] = (1-h[j-lag]) * I[0<=j-lag<n] - h[j+lag] * I[0<=j+lag<n]
        j_start1 = max(0, lag)
        j_end1 = min(n, n + lag)
        if j_start1 < j_end1:
            grad[j_start1:j_end1] += w * (1.0 - h[j_start1-lag:j_end1-lag])
        j_start2 = max(0, -lag)
        j_end2 = min(n, n - lag)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= w * h[j_start2+lag:j_end2+lag]
    
    return dx * grad
```
