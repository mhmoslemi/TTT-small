from scipy.interpolate import interp1d
import time

    return float(np.max(corr) * dx)

    h = h.flatten()
    corr = np.correlate(h, 1-h, mode="full")

    m = np.max(scaled)

    """Project to {0<=h<=1, sum(h)=n/2}"""
    h = np.array(h, dtype=np.float64).copy()
    target = n / 2.0
    # Water-filling projection
    for _ in range(100):
        h = np.clip(h, 0, 1)
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        # Adjust: increase where h<1, decrease where h>0
        if diff > 0:  # need to decrease
            mask = h > 0
            h[mask] -= diff / mask.sum()
        else:  # need to increase
            mask = h < 1
            h[mask] -= diff / mask.sum()
    h = np.clip(h, 0, 1)
    return h

    """Interpolate from n_parent points to n_new points."""
    n_old = len(h_parent)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    # Actually for step functions, the points represent midpoints of intervals
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    # Use nearest-neighbor interpolation for step functions
    h_new = np.interp(x_new, x_old, h_parent)
    # Project to feasible
    h_new = project_to_feasible(h_new, n_new)
    return h_new

        remaining = time_limit - (time.time() - t_start)
        if remaining < 5:
            break
        _G_ALPHA = alpha
        maxiter = min(500, max(50, int(remaining * 50)))
        try:
            res = minimize(_obj_gs, best_h, jac=_jac_gs, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                v_new = compute_c5(h_new, dx)
                if v_new < best_val - 1e-12:
                    best_val = v_new
                    best_h = h_new
                    alpha = min(alpha * 1.3, alpha_max)
                elif v_new < best_val + 1e-8:
                    best_h = h_new
                    alpha = min(alpha * 1.5, alpha_max)
        except Exception:
            pass
        if time.time() - t_start > time_limit * 0.75:
            alpha = min(alpha * 2.0, alpha_max)
    return best_h, best_val

def run_pg(h_init, n, dx, time_limit, alpha_start=60.0):
    t_start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    alpha = alpha_start
    lr = 0.05
    
    while time.time() - t_start < time_limit:
        g = softmax_grad(h, dx, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        obj_cur = softmax_obj_val(h, dx, alpha)
        step_ok = False
        for _ in range(40):
            h_try = project_feasible(h - lr * g, n)
            obj_try = softmax_obj_val(h_try, dx, alpha)
            if obj_try < obj_cur - 1e-12:
                h = h_try
                step_ok = True
                break
            lr *= 0.5
        if not step_ok:
            alpha = min(alpha * 1.05, 400.0)
        lr = min(lr * 1.15, 0.15)
        v = compute_c5(h, dx)
        if v < best_val - 1e-10:
            best_val = v
            best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    try:
        h_parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        h_parent = np.full(48, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    
    def check_update(h, n):
        nonlocal best_h, best_c5
        h = np.clip(h, 0.0, 1.0)
        c5 = compute_c5(h, 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 1: n=48
    n1 = 48
    h1 = interpolate_to_n(h_parent, n1)
    t_limit = min((budget_s - (time.time() - t0)) * 0.35, 420.0)
    if t_limit > 8:
        h1, v1 = run_slsqp(h1, n1, 2.0 / n1, t_limit, alpha_start=40.0, alpha_max=200.0)
        check_update(h1, n1)
    
    # Phase 2: n=96
    n2 = 96
    h2 = interpolate_to_n(h1, n2)
    remaining = budget_s - (time.time() - t0)
    t_limit = min(remaining * 0.30, 300.0)
    if t_limit > 8:
        h2, v2 = run_slsqp(h2, n2, 2.0 / n2, t_limit, alpha_start=60.0, alpha_max=250.0)
        check_update(h2, n2)
    
    # Phase 3: n=192
    n3 = 192
    h3 = interpolate_to_n(h2, n3)
    remaining = budget_s - (time.time() - t0)
    t_limit = min(remaining * 0.30, 250.0)
    if t_limit > 8:
        h3, v3 = run_pg(h3, n3, 2.0 / n3, t_limit, alpha_start=80.0)
        check_update(h3, n3)
    
    # Final polish at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        nb = len(best_h)
        h_pol, v_pol = run_slsqp(best_h, nb, 2.0 / nb, remaining * 0.85,
                                  alpha_start=150.0, alpha_max=400.0)
        check_update(h_pol, nb)
    
    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    nb = len(best_h)
    diff = best_h.sum() - nb / 2.0
    if abs(diff) > 1e-8:
        best_h = project_feasible(best_h, nb)
    best_c5 = compute_c5(best_h, 2.0 / nb)
    
    return (best_h, best_c5, nb)
