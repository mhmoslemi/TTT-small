import math

def project_to_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    n = len(h)
    # Bisection on lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix any numerical issues
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-10:
        # Add/subtract from elements with room
        if diff > 0:
            room = 1 - proj
            order = np.argsort(-room)
            for i in order:
                add = min(diff, room[i])
                proj[i] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            room = proj.copy()
            order = np.argsort(-room)
            for i in order:
                sub = min(-diff, room[i])

Now, I'm also thinking: maybe I should use a more sophisticated approach. The problem is:
minimize max_k c_k(h)

This can be rewritten as:
minimize t

This is a non-convex problem (c_k is bilinear in h), but for a fixed active set it's a QCQP.

Alternative: Use the CVXPY approach with successive linearization. At each iteration:
1. Evaluate c_k(h_current) for all k
2. Linearize each c_k at h_current: c_k(h) ≈ c_k(h_cur) + ∇c_k(h_cur)^T (h - h_cur)
3. Solve: minimize t s.t. linearized constraints, box, sum constraint
4. Update h, repeat

This is SLP (Successive Linear Programming). Each subproblem is an LP.

For the linearized constraint:
c_k(h_cur) + ∑_j ∂c_k/∂h[j] * (h[j] - h_cur[j]) <= t

Let d_k = ∂c_k/∂h (the gradient vector for correlation k)
Then: c_k(h_cur) + d_k^T (h - h_cur) <= t
→ d_k^T h <= t - c_k(h_cur) + d_k^T h_cur

This is linear in (h, t).

The LP is:
minimize t
subject to: d_k^T h <= t - c_k(h_cur) + d_k^T h_cur, for all k
0 <= h <= 1
sum(h) = n/2

This can be solved with CVXPY or even a custom LP solver. With n up to 540 and 2n-1 ~ 1079 constraints, this is a moderate LP.

from scipy.optimize import linprog

def slp_step(h_cur, n):
    """One SLP iteration: linearize and solve LP."""
    target_sum = n / 2.0
    dx = 2.0 / n
    
    # Compute all correlations
    one_minus_h = 1 - h_cur
    corr = np.correlate(h_cur, one_minus_h, mode='full')
    max_corr = np.max(corr)
    
    n_corr = len(corr)  # 2n-1
    
    # For each correlation k, compute gradient d_k with respect to h
    # And the linearized constraint: d_k^T h <= rhs_k
    # where rhs_k = max_corr - c_k(h_cur) + d_k^T h_cur ... 
    # Actually: we want c_k(h) <= max_corr (or some t)
    # Linearized: c_k(h_cur) + d_k^T (h - h_cur) <= t
    # d_k^T h - t <= c_k(h_cur) - d_k^T h_cur
    
    # Let's set t = max_corr initially and minimize t
    # Variables: h[0..n-1], t
    # Total n+1 variables
    
    # Constraints:
    # For each k: d_k^T h - t <= c_k(h_cur) - d_k^T h_cur
    # 0 <= h <= 1 (box)
    # sum(h) = n/2
    
    # Compute all gradients
    A_rows = []
    b_vals = []
    
    for k in range(n_corr):
        shift = k - (n - 1)
        d_k = np.zeros(n)
        if shift == 0:
            d_k = 1 - 2 * h_cur
        elif shift > 0:
            d_k[:-shift] = (1 - h_cur[shift:])
            d_k[shift:] -= h_cur[:-shift]
        else:
            s = -shift
            d_k[s:] = (1 - h_cur[:-s])
            d_k[:-s] -= h_cur[s:]
        
        rhs = corr[k] - d_k @ h_cur
        # d_k^T h - t <= rhs
        row = np.append(d_k, -1.0)
        A_rows.append(row)
        b_vals.append(rhs)
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    
    # Box constraints on h: -1 <= -h[i] <= 0, already handled by bounds
    # sum(h) = n/2: equality constraint
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.array([target_sum])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        t_new = res.x[-1]
        return h_new, t_new * dx
    return h_cur, max_corr * dx

                    best_obj = eval_c5(h)
                    best_h = h.copy()

        # Convergence
        if it > 20 and best_obj > 0:
            improvement = (max_corr * dx - best_obj) / best_obj
            if improvement < 1e-9:
                break

    return best_h, best_obj

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    # Initialize
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except:
        h_init = np.full(45, 0.5)

    n_init = len(h_init)
    h_init = project_onto_box_and_sum(h_init, n_init / 2.0)

    best_h = h_init.copy()
    best_c5 = eval_c5(best_h)

    # Phase 1: SLP on initial grid
    tlim1 = min(120.0, budget_s * 0.12)
    h1, c5_1 = slp_optimize(h_init, n_init, max_iters=400, time_limit=tlim1)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Sequential upscaling and optimization
    elapsed = time.time() - t_start
    remaining_frac = (budget_s * 0.85 - elapsed) / budget_s
    target_grids = [max(n_init, 90), 180, 360, 720, 1440]
    for tg in target_grids:
        if time.time() - t_start > budget_s * 0.82:
            break
        t_avail = min(budget_s * 0.12, budget_s * 0.82 - (time.time() - t_start))
        if t_avail < 10:
            break
        h_up = upsample(best_h, tg)
        h_up = project_onto_box_and_sum(h_up, tg / 2.0)
        h_opt, c5_opt = slp_optimize(h_up, tg, max_iters=500, time_limit=t_avail)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Random restarts on best grid
    n_curr = len(best_h)
    t_restart_end = budget_s * 0.92
    t0r = time.time()
    n_restarts = 0
    while time.time() < t_restart_end and n_restarts < 30:
        n_restarts += 1
        scale = np.random.uniform(0.05, 0.2)
        h_r = project_onto_box_and_sum(best_h + scale * np.random.randn(n_curr), n_curr / 2.0)
        t_rem = t_restart_end - time.time()
        if t_rem < 3:
            break
        h_opt, c5_opt = slp_optimize(h_r, n_curr, max_iters=200, time_limit=min(30, t_rem / (30 - n_restarts + 1)))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Final fine-tuning
    t_final_end = min(time.time() + (budget_s * 0.07), budget_s * 0.98)
    t_rem = t_final_end - time.time()
    if t_rem > 5:
        n_fin = len(best_h)
        h_f, c5_f = slp_optimize(best_h, n_fin, max_iters=500, time_limit=t_rem)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Ensure exact feasibility
    best_h = project_onto_box_and_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return (best_h, best_c5, len(best_h))
