import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def project_to_feasible(x, N):
    """Project x onto {0 <= h <= 1, sum(h) = N/2}"""
    target = N / 2.0
    # Use bisection on lambda
    lo, hi = -max(x) - 1, -min(x) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(x + mid, 0, 1)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2, 0, 1)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.logaddexp.reduce([np.log(1.0) + beta * (xi - m) for xi in x]) / beta

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Start from parent or create initial
    if 'initial_h_values' in globals() and initial_h_values is not None:

        N = len(h)
    else:

        h = np.zeros(N)
        h[:N//2] = 1.0
    
    # Project to feasible
    h = project_to_feasible(h, N)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for beta in [5, 20, 50, 100, 200, 500]:
        if time.time() - start_time > budget_s - 30:

            c = np.correlate(x, 1 - x, mode="full") * dx
            return smooth_max(c, beta)

            c = np.correlate(x, 1 - x, mode="full") * dx

            m = np.max(c)

            # Gradient of c[i] w.r.t. x

    hi = 1.0

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(x, N):
    target = N / 2.0
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_val(c, beta):
    m = np.max(c)
    return m + float(np.log(np.sum(np.exp(beta * (c - m)))) / beta)


def sum_constraint(x, N):
    return float(np.sum(x) - N / 2.0)


def obj_grad(x, N, beta):
    dx = 2.0 / N
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    f = smooth_max_val(c, beta)
    shifted = beta * (c - c.max())
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        w = weights[i]
        if w < 1e-15:
            continue
        m = i - (N - 1)
        p_lo1 = max(0, m)
        p_hi1 = min(N - 1, m + N - 1)
        if p_lo1 <= p_hi1:
            grad[p_lo1:p_hi1 + 1] += w * dx * (1.0 - x[p_lo1 - m:p_hi1 - m + 1])
        p_lo2 = max(0, -m)
        p_hi2 = min(N - 1, N - 1 - m)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2 + 1] -= w * dx * x[p_lo2 + m:p_hi2 + m + 1]
    return f, grad


def binary_local_search(h, N, rng, time_limit):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, N)
    c5_cur = compute_c5(h_bin)
    best_bin = h_bin.copy()
    best_c = c5_cur
    T = 0.004
    while time.time() < time_limit and T > 1e-7:
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c5_new = compute_c5(h_bin)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            if c5_new < best_c:
                best_c = c5_new
                best_bin = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        T *= 0.9999
    return best_bin, best_c
