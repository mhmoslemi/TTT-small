import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def make_binary_from_parent(parent, n, rng):
    h = np.zeros(n)
    if parent is not None and len(parent) > 1:
        x_parent = np.linspace(0, 2, len(parent), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_parent, parent)
        h = (h_interp > 0.5).astype(np.float64)
    else:
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    target = n // 2
    while h.sum() > target:
        h[int(rng.choice(np.where(h == 1.0)[0]))] = 0.0
    while h.sum() < target:
        h[int(rng.choice(np.where(h == 0.0)[0]))] = 1.0
    return h


def local_search_binary(h, n, rng, deadline):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    current_max = int(np.max(corr))
    best_max = current_max
    best_h = h.copy()

    temp = 1.0
    patience = 0
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    while time.time() < deadline:
        if patience > 30000:
            temp *= 0.92
            patience = 0
            if temp < 0.005:
                break

        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:
            continue

        h[i], h[j] = 0.0, 1.0
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = int(np.max(new_corr))

        delta = new_max - current_max
        if delta < 0 or (temp > 0.005 and rng.random() < np.exp(-float(delta) / temp)):
            current_max = new_max
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                patience = 0
        else:
            h[i], h[j] = 1.0, 0.0
            patience += 1

    return best_h, best_max * dx


def targeted_local_search(h, n, rng, deadline):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    current_max = int(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    patience = 0
    temp = 0.5

    while time.time() < deadline:
        if patience > 20000:
            if temp > 0.003:
                temp *= 0.9
                patience = 0
            else:
                break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)

        if rng.random() < 0.6:
            # Targeted: find a 1 that contributes to max lag
            if 0 <= lag < n:
                contrib = [m for m in range(max(0, -lag), min(n - lag, n)) if h[m] == 0 and m + lag < n and h[m + lag] == 1]
                if contrib:
                    m = contrib[int(rng.randint(len(contrib)))]
                    i = m + lag
                    j = int(rng.choice(zeros))
                    if i == j:
                        continue
                else:
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    if i == j:
                        continue
            else:
                k_abs = -lag if lag < 0 else lag
                if k_abs < n:
                    contrib = [m for m in range(max(0, k_abs), n) if h[m] == 0 and m - k_abs >= 0 and h[m - k_abs] == 1]

                        m = contrib[int(rng.randint(len(contrib)))]

                        j = int(rng.choice(zeros))

                        j = int(zeros[rng.randint(len(zeros))])

                    i = int(ones[rng.randint(len(ones))])

                    if i == j:

            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])

def perturb(h, n, rng, num_swaps):
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(num_swaps):
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    return h


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 10

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)

    if parent is not None:
        best_h = parent.copy()
        best_n = len(parent)
        best_c5 = compute_c5(parent, best_n)

    # Candidate n values (even for binary)
    n_list = []
    if parent is not None and len(parent) % 2 == 0:
        n_list.append(len(parent))
    for n in [96, 100, 112, 120, 128, 144, 160, 176, 192, 200, 224, 256]:
        if n not in n_list:
            n_list.append(n)

    # Allocate time budget
    total_time = max(60, deadline - time.time())
    # Give more time to promising sizes
    weights = [3.0 if n == 96 else 2.5 if n <= 128 else 2.0 if n <= 192 else 1.0 for n in n_list]
    w_sum = sum(weights)
    time_alloc = [total_time * w / w_sum for w in weights]

    for idx, n in enumerate(n_list):
        if time.time() > deadline - 15:
            break
        alloc = time_alloc[idx]
        n_deadline = min(deadline - 5, time.time() + alloc)

        # Multiple restarts
        num_restarts = 3
        per_restart = (n_deadline - time.time()) / num_restarts

        for r in range(num_restarts):
            if time.time() > deadline - 10:
                break
            r_deadline = min(n_deadline - 2, time.time() + per_restart)

            if r == 0:
                h = make_binary_from_parent(parent, n, rng)
            else:
                h = make_binary_from_parent(parent, n, rng)
                h = perturb(h, n, rng, 20 + r * 15)

            # Phase 1: basic local search
            h, c5 = local_search_binary(h, n, rng, min(r_deadline, time.time() + 0.5 * per_restart))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

            # Phase 2: targeted local search
            if time.time() < r_deadline - 1:
                h, c5 = targeted_local_search(h, n, rng, r_deadline)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n

    # Final polish on best solution
    if best_h is not None and best_n % 2 == 0 and time.time() < deadline - 15:
        h = (best_h > 0.5).astype(np.float64)
        while h.sum() > best_n // 2:
            h[int(rng.choice(np.where(h == 1.0)[0]))] = 0.0
        while h.sum() < best_n // 2:
            h[int(rng.choice(np.where(h == 0.0)[0]))] = 1.0
        h, c5 = local_search_binary(h, best_n, rng, deadline - 5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Fallback
    if best_h is None:
        best_n = 96
        best_h = np.zeros(best_n)
        best_h[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        best_c5 = compute_c5(best_h, best_n)

    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, best_n)

    return (best_h, final_c5, best_n)
