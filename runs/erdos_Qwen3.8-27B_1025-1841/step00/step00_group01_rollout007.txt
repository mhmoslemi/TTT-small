        while h.sum() < n/2:
            zeros = np.where(h == 0)[0]
            h[rng_idx] = 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    def remaining():
        return budget_s - (time.time() - t0)
    
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Evaluate parent
    try:
        h_p = np.array(initial_h_values, dtype=float)
        n_p = len(h_p)
        c5_p = compute_c5(h_p, n_p)
        best_h = h_p.copy()
        best_c5 = c5_p
        best_n = n_p
    except Exception:
        pass
    
    # Phase 1: Local search on parent (if available)
    if best_h is not None and remaining() > 300:
        n = best_n
        h = best_h.copy()
        c5 = best_c5
        # Multiple passes of hill climbing
        for _ in range(50):
            if remaining() < 280:
                break
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    if h[i] == h[j]:
                        continue
                    h[i], h[j] = h[j], h[i]
                    nc = compute_c5(h, n)
                    if nc < c5 - 1e-15:
                        c5 = nc
                        improved = True
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                            best_n = n
                    else:
                        h[i], h[j] = h[j], h[i]
            if not improved:
                break
    
    # Phase 2: SA at various sizes
    # Try sizes from small to large
    sizes = [48, 64, 96, 128, 192, 256, 384, 512]
    
    for n in sizes:
        if remaining() < 100:
            break
        
        # Reuse best if same size
        if n == best_n and best_h is not None and len(best_h) == n:
            h = best_h.copy()
        else:
            # Initialize with random binary
            h = np.zeros(n)
            if n % 2 == 0:
                idx = rng.choice(n, n // 2, replace=False)
                h[idx] = 1.0
            else:
                idx = rng.choice(n, n // 2, replace=False)
                h[idx] = 1.0
                zeros = np.where(h == 0)[0]
                h[rng.choice(zeros)] = 0.5
        
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        
        # SA
        # Time allocation: more time for mid-range n
        if n <= 128:
            sa_time = 60
        elif n <= 256:
            sa_time = 120
        else:
            sa_time = 90
        
        sa_time = min(sa_time, remaining() - 80)
        if sa_time <= 0:
            continue
        
        t_sa = time.time()
        T = 0.02
        T_min = 0.00005
        
        # Estimate iterations
        # For n=128, correlate ~5μs → ~10M iterations in 60s
        # For n=512, correlate ~50μs → ~1.2M iterations in 60s
        
        while T > T_min and (time.time() - t_sa) < sa_time and remaining() > 80:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                T *= 0.9999
                continue
            
            h[i], h[j] = h[j], h[i]
            nc = compute_c5(h, n)
            d = nc - c5
            
            if d < 0:
                c5 = nc
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n
            elif rng.random() < np.exp(-d / T):
                c5 = nc
            else:
                h[i], h[j] = h[j], h[i]
            
            T *= 0.9999
    
    # Phase 3: Fractional refinement on best
    if remaining() > 40 and best_h is not None:
        n = best_n
        h = best_h.copy()
        c5 = best_c5
        target = n / 2.0
        
        t_ref = time.time()
        for _ in range(200000):
            if remaining() < 30:
                break
            if (time.time() - t_ref) > remaining() - 25:
                break
            
            i = rng.randint(n)
            old_i = h[i]
            
            if old_i <= 0.01:
                h[i] = rng.uniform(0.01, 0.2)
            elif old_i >= 0.99:
                h[i] = rng.uniform(0.8, 0.99)
            else:
                h[i] = np.clip(old_i + rng.uniform(-0.1, 0.1), 0.0, 1.0)
            
            diff = target - h.sum()
            j = rng.randint(n)
            if j == i:
                j = (j + 1) % n
            old_j = h[j]
            new_j = h[j] + diff
            if new_j < 0.0 or new_j > 1.0:
                h[i] = old_i
                continue
            h[j] = new_j
