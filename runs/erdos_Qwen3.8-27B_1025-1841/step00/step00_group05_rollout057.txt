import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * dx)

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = h.sum()
    if s > target:
        # Need to reduce
        order = np.argsort(h)[::-1]  # largest first
        for i in order:
            if s <= target:
                break
            reduce = min(h[i], s - target)
            h[i] -= reduce
            s -= reduce
    elif s < target:
        order = np.argsort(h)  # smallest first
        for i in order:
            if s >= target:
                break
            add = min(1.0 - h[i], target - s)
            h[i] += add
            s += add
    return h

def binary_local_search(h, n, max_iter=10000):
    """Local search on binary vectors with exactly n/2 ones"""
    h = (h > 0.5).astype(float)
    if h.sum() != n // 2:
        # Fix
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > n // 2:
            for i in ones[len(ones)//2:]:
                h[i] = 0
        elif len(ones) < n // 2:
            for i in zeros[:n//2 - len(ones)]:
                h[i] = 1
    
    current_c5 = compute_c5(h, n)
    
    for it in range(max_iter):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_val = current_c5
        best_swap = None
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                val = compute_c5(h, n)
                if val < best_val - 1e-12:
                    best_val = val
                    best_swap = (i, j)
                h[i], h[j] = h[j], h[i]
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            current_c5 = best_val
            improved = True
        
        if not improved:
            break
    
    return h, current_c5

    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * dx)

    h = (h > 0.5).astype(float)
    target = n // 2
    while h.sum() > target:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx)] = 0
    while h.sum() < target:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx)] = 1
    return h

    h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * dx)

def fix_binary(h, n, rng):
    h = (h > 0.5).astype(float)
    target = n // 2
    while h.sum() > target:
        idx = np.where(h == 1)[0]
        if len(idx) == 0:
            break
        h[rng.choice(idx)] = 0
    while h.sum() < target:
        idx = np.where(h == 0)[0]
        if len(idx) == 0:
            break
        h[rng.choice(idx)] = 1
    return h

def local_search_binary(h, n, max_time, rng):
    h = h.copy()
    current = compute_c5(h, n)
    t0 = time.time()
    stuck = 0
    best_ever = current
    
    while time.time() - t0 < max_time:
        best_val = current
        best_swap = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                val = compute_c5(h, n)
                if val < best_val - 1e-15:
                    best_val = val
                    best_swap = (i, j)
                h[i], h[j] = h[j], h[i]
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            current = best_val
            stuck = 0
            if current < best_ever:
                best_ever = current
        else:
            stuck += 1
            if stuck > 3:
                k = min(stuck, 5)
                for _ in range(k):
                    o = np.where(h == 1)[0]
                    z = np.where(h == 0)[0]
                    a, b = rng.choice(o), rng.choice(z)
                    h[a], h[b] = h[b], h[a]
                stuck = 0
                if current > best_ever:
                    current = best_ever
    
    return h, current

def greedy_construct(n, rng):
    h = np.zeros(n)
    target = n // 2
    remaining = set(range(n))

            h[pos] = 0.0
            if val < best_val:
                best_val = val
                best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1.0
            remaining.discard(best_pos)
    return h

    h_interp = np.interp(xs_new, xs_old + 1.0/n_old, h_old)
    return h_interp
