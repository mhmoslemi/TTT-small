import numpy as np
import time

def c5_compute(h, dx):
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial solution from parent
    if 'initial_h_values' in globals():
        h_init = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h_init = None

    if h_init is not None:
        n = len(h_init)
        dx = 2.0 / n
        h = np.clip(h_init, 0.0, 1.0).copy()
        target = n / 2.0
        s = np.sum(h)
        if s > 1e-10:
            h *= target / s
        h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            h += diff / n
            h = np.clip(h, 0, 1)
        c = c5_compute(h, dx)
    else:
        n = 70
        dx = 2.0 / n
        h = np.full(n, 0.5)
        c = c5_compute(h, dx)

    best_h = h.copy()
    best_c5 = c
    best_n = n

    # Simulated Annealing
    T = 0.005
    alpha = 0.99999
    no_improve = 0
    reheat_T = 0.001

    while time_left() > budget_s * 0.15:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:

            delta = 0.3

        else:
            amt = min(delta, h[j], 1.0 - h[i])
            i, j = j, i

        if amt < 1e-12:
            continue

        h[j] += amt

                best_c5 = c
                best_h = h.copy()
                best_n = n

                no_improve += 1

            h[j] -= amt

            no_improve = 0

    # Phase 2: try finer grids
    if time_left() > 50:

            h_curr = h2.copy()
            c_curr = c2

            alpha2 = 0.99997
