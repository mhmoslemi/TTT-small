def projected_subgradient(h, n, iterations=1000, step_size=0.01):
    """Simple subgradient method for the max objective"""
    dx = 2.0 / n
    for it in range(iterations):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = np.argmax(corr)
        # Subgradient of O(k_max) w.r.t. h
        h_sub = np.zeros(n)
        # O(k) = sum_{valid i} h[i] * (1-h[i+k]) for the appropriate indexing
        # In numpy correlate(h, g, 'full'):
        # result[m] = sum_i h[i] * g[i+k] where the indexing depends on m
        # For m = k + n - 1 (k >= 0): sum_i h[i] * g[i+k]
        # For m = n - 1 - k (k < 0, i.e., m < n-1): sum_i h[i] * g[i-m+n-1]
        
        # Actually let me just use the autocorrelation formulation
        # A(k) = sum_i h[i] * h[i+k] for the valid range
        # dA(k)/dh[j] = h[j+k] + h[j-k] (with bounds)
        
        # For the correlate indexing:
        # np.correlate(h, 1-h, 'full')[m] where m ranges from 0 to 2n-2
        # m = n-1 corresponds to lag 0
        # m = n-1+k corresponds to lag k (right shift)
        # m = n-1-k corresponds to lag -k (left shift)
        
        lag = k_max - (n - 1)
        # d/dh[j] of O(lag) = d/dh[j] of sum_i h[i](1-h[i+lag])
        # = (1 - h[j+lag]) - h[j-lag] (with appropriate bounds checking)
        for j in range(n):
            dl1 = 0.0
            dl2 = 0.0
            if 0 <= j + lag < n:
                dl1 = 1.0 - h[j + lag]
            if 0 <= j - lag < n:
                dl2 = h[j - lag]
            h_sub[j] = dl1 - dl2
        
        # Also account for the fact that we want to minimize, so we subtract gradient
        h_new = h - step_size * h_sub
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n, rng)
        
        # Adaptive step size
        if evaluate_obj(h_new) < evaluate_obj(h):
            h = h_new
        else:
            step_size *= 0.95
    
    return h

import numpy as np

def evaluate_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def upsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def fix_sum(h, n, rng):
    target = n / 2.0

            break
        if diff > 0:
            idxs = np.where(h > 1e-9)[0]
            if len(idxs) == 0:
                break
            idx = idxs[rng.randint(len(idxs))]
            h[idx] = max(0.0, h[idx] - diff)
        else:
            idxs = np.where(h < 1.0 - 1e-9)[0]
            if len(idxs) == 0:
                break
            idx = idxs[rng.randint(len(idxs))]
            h[idx] = min(1.0, h[idx] - diff)

def slsqp_optimize(h0, n, rng, maxiter=2000):

        return x.sum() - n / 2.0
