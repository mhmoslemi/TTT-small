import numpy as np
from scipy.optimize import minimize
import time

def compute_conv(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h):
    return np.max(compute_conv(h)) * (2.0 / len(h))

def project_to_simplex(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h - mid, 0, 1)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def grad_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, -s)
    hi1 = min(n, n - s)
    if lo1 < hi1:
        grad[lo1:hi1] += (1 - h[lo1+s:hi1+s])
    lo2 = max(0, s)
    hi2 = min(n, n + s)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2-s:hi2-s]
    return grad

def pgd_optimize(h, max_iter=30000, top_k=5, time_limit=None, start_time=None):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2
    h = project_to_simplex(h.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 2.0 / n
    stall = 0
    
    for it in range(max_iter):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        conv = compute_conv(h)
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        grad = np.zeros(n)
        for m in top_indices:
            s = m - (n - 1)
            grad += grad_shift(h, s)
        grad *= dx / top_k
        
        accepted = False
        for _ in range(12):
            h_new = project_to_simplex(h - step * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new <= best_c5 + 1e-12:
                h = h_new
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step *= 0.5
            if step < 1e-12:
                break
            continue
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            step = min(step * 1.05, 4.0 / n)
        else:
            stall += 1
            if stall > 2000:
                break
            if stall % 200 == 0:
                step *= 0.9
    
    return best_h, best_c5

    n = len(h01)
    h = h01.copy()
    c5 = compute_c5(h)
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    improved = True
    sweeps = 0
    
    while improved and sweeps < max_sweeps:
        improved = False
        sweeps += 1
        for a in list(ones):
            for b in list(zeros):
                h[a], h[b] = 0.0, 1.0
                c5_new = compute_c5(h)
                if c5_new < c5 - 1e-12:
                    c5 = c5_new
                    ones.remove(a)
                    ones.add(b)
                    zeros.remove(b)
                    zeros.add(a)
                    improved = True
                else:
                    h[a], h[b] = 1.0, 0.0
    
    return h, c5
