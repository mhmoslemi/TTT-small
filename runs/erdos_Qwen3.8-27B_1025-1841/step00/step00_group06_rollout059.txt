                h_r = proj_box_sum(h_r, n)

                for _ in range(800):
                    if time.time() - t0 > budget_s * 0.8:
                        break
                    grad_r, _ = compute_smoothed_grad(h_r, n, 35.0)
                    h_r = proj_box_sum(h_r - 0.06 * grad_r, n)

                c5_r = c5_val(h_r, n)
                if c5_r < best_c5:
                    best_c5, best_h, best_n = c5_r, h_r.copy(), n
                    h = h_r.copy()

    # Final validation and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_box_sum(best_h, best_n)
    best_c5 = c5_val(best_h, best_n)

    return (best_h, best_c5, best_n)
