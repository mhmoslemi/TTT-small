    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    # corr[idx] corresponds to lag = idx - (n-1)
    # We want the gradient of O(k) for the worst lags
    # O(k) = sum_{i: 0<=i<n, 0<=i+k<n} h[i]*(1-h[i+k])
    # dO/dh[j] = (1-h[j+k])*1[j+k<n] - h[j-k]*1[j>=k]  for k>0

    # which is O(-k) in our notation... let me be careful.

    # = sum_i h[i+k] - h[i+k]*h[i]

def adjust_sum(h, target_sum):
    n = len(h)
    h = h.copy()
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:

            mask = room > 1e-10

            else:

            mask = room > 1e-10
