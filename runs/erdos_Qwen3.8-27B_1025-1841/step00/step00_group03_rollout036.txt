    h = np.asarray(h, dtype=float)
    if len(h) != n:
        old_n = len(h)
        idx = np.clip((np.arange(n) * old_n) // n, 0, old_n - 1)
        h = h[idx]

    # Otherwise, pick top n/2
    if abs(np.sum(h) - target) < n * 0.01:

        out = (h >= 0.5).astype(float)

        cur = int(np.sum(out))

            idx_1 = np.where(out == 1)[0]

            vals = h[idx_1]
            flip_order = idx_1[np.argsort(vals)]
