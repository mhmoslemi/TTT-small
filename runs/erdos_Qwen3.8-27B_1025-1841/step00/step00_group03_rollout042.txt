def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair(h):
    h = np.clip(h, 0, 1)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    s = np.sum(h)
    if s > 0:
        h = h * (target / s)
        h = np.clip(h, 0, 1)

        for _ in range(10):

            if diff > 0:
                mask = h < 1
            else:
                mask = h > 0
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)

                    best_i, best_j = i, j
                h[j] = 0.0
            h[i] = 1.0
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        c5 += best_delta
    return h, c5

def sa_loop(current, best_h, best_c5, target, rng, t0, budget, T_init, patience):
    n = len(current)
    current_c5 = evaluate_c5(current)
    minimum_c5 = best_c5
    T = T_init
    T_min = 0.0003
    decay = 0.99998
    no_improve = 0
    dead_zone = 4.0 * T

    while time.time() - t0 < budget - 5:
        ones_idx = np.flatnonzero(current == 1)
        zeros_idx = np.flatnonzero(current == 0)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        current[i] = 0.0
        current[j] = 1.0
        new_c5 = evaluate_c5(current)
        if new_c5 <= current_c5:
            current_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current.copy()
            dead_zone = 4.0 * T
        else:
            no_improve += 1
            delta = new_c5 - current_c5
            if delta < dead_zone and T > T_min:
                if rng.random() < np.exp(-delta / T):
                    current_c5 = new_c5
                else:
                    current[i] = 1.0
                    current[j] = 0.0
            else:
                current[i] = 1.0
                current[j] = 0.0
        T = max(T_min, T * decay)
        if no_improve > patience:
            current = best_h.copy()
            current_c5 = best_c5
            no_improve = 0
            T = max(T_min, T_init * 0.5)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    n = 70
    target = n // 2

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = None

    if h_init is not None and len(h_init) == n:
        h = to_binary(h_init, target, rng)
    elif h_init is not None and len(h_init) > 0:
        h_resized = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h_init)), h_init)
        h = to_binary(h_resized, target, rng)
    else:
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, size=target, replace=False)] = 1.0

    # Quick greedy improvement
    h, c5_greedy = greedy_pass(h, rng, max_steps=50)
    best_h = h.copy()
    best_c5 = c5_greedy

    # Main SA with multiple temperature restarts
    T_starts = [0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    patience = 80000

    for T_s in T_starts:
        if time.time() - t0 > budget_s - 30:
            break
        current = best_h.copy()
        best_h, best_c5 = sa_loop(current, best_h, best_c5, target, rng, t0, budget_s, T_s, patience)

    # Try a different n value if time allows
    if time.time() - t0 < budget_s - 60:
        n2 = 80
        target2 = n2 // 2
        h_up = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, n), best_h)
        h2 = to_binary(h_up, target2, rng)
        h2, c5_2 = greedy_pass(h2, rng, max_steps=20)
        current2 = h2.copy()
        h2_b, c5_2b = sa_loop(current2, h2.copy(), c5_2, target2, rng, t0, budget_s, 0.03, 50000)
        if c5_2b < best_c5:
            best_h = h2_b
            best_c5 = c5_2b
            n = n2

    return best_h, best_c5, len(best_h)
