import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def to_binary(h, n, rng):
    h = np.clip(h, 0.0, 1.0)
    h = (h >= 0.5).astype(float)
    target = n // 2
    cnt = int(np.sum(h))
    if cnt > target:
        ones = np.where(h == 1)[0]
        sel = rng.choice(cnt, size=cnt - target, replace=False)
        h[ones[sel]] = 0.0
    elif cnt < target:
        zeros = np.where(h == 0)[0]
        sel = rng.choice(n - cnt, size=target - cnt, replace=False)
        h[zeros[sel]] = 1.0
    return h

def interp_up(h, n_from, n_to):
    if n_to == n_from:
        return h.copy()
    x_f = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    x_t = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    return np.interp(x_t, x_f, h)

def sa_binary(h, n, deadline, rng, T_max=0.06, T_min=0.001):
    dx = 2.0 / n
    T = T_max
    cur = h.copy()
    cur_c5 = compute_c5(cur, dx)
    best = cur.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        c5 = compute_c5(cur, dx)
        d = c5 - cur_c5
        if d < 1e-12 or (T > T_min and rng.random() < np.exp(-d / T)):
            cur_c5 = c5
            T = max(T * 0.99999, T_min)
            if c5 < best_c5:
                best_c5 = c5
                best = cur.copy()
                stall = 0
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
        if stall >= 40000:
            nk = rng.randint(3, 9)
            cur = best.copy()
            cur_c5 = best_c5
            for _ in range(nk):
                a, b = rng.choice(n, 2, replace=False)
                if cur[a] != cur[b]:
                    cur[a], cur[b] = cur[b], cur[a]
            T = T_max
            stall = 0
    return best, best_c5

def fractional_polish(h, n, deadline, rng):
    dx = 2.0 / n
    cur = h.astype(float).copy()
    cur_c5 = compute_c5(cur, dx)
    best = cur.copy()
    best_c5 = cur_c5
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        eps = rng.uniform(0.005, 0.08)
        if cur[i] < eps or cur[j] > 1.0 - eps:
            continue
        cur[i] -= eps
        cur[j] += eps
        c5 = compute_c5(cur, dx)
        if c5 < cur_c5 - 1e-12:
            cur_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best = cur.copy()
        else:
            cur[i] += eps
            cur[j] -= eps
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = np.array([1.0, 0.0] * 35)
        rng.shuffle(h_init)

    n0 = len(h_init)
    h = to_binary(h_init, n0, rng)

    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / n0)
    best_n = n0

    resolutions = [n0, 2 * n0, 4 * n0]
    time_splits = [0.35, 0.35, 0.30]

    for idx, n in enumerate(resolutions):
        if time.time() > deadline - 60:
            break
        if idx == 0:
            h_cur = h.copy()
        else:
            h_cur = interp_up(best_h, best_n, n)
            h_cur = to_binary(h_cur, n, rng)

        d = min(t0 + (deadline - t0) * (time_splits[idx] + time_splits[idx-1] if idx > 0 else 0),
                deadline - 30)
        if idx == 0:
            d = t0 + (deadline - t0) * 0.35
        elif idx == 1:
            d = t0 + (deadline - t0) * 0.68
        else:
            d = deadline - 30

        h_cur, c5 = sa_binary(h_cur, n, d, rng)
        if c5 < best_c5:
            best_h, best_c5, best_n = h_cur.copy(), c5, n

    if time.time() < deadline - 10:
        best_h, best_c5 = fractional_polish(best_h, best_n, deadline, rng)

    n_final = len(best_h)
    c5_final = compute_c5(best_h, 2.0 / n_final)
    return best_h, c5_final, n_final
