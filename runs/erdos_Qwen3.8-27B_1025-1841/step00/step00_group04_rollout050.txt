    return h

def run_slsqp(h0, n, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': eq_con, 'args': (n,)}
    res = minimize(
        fun=obj_fn, x0=h0, args=(n,),
        method='SLSQP', bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return res

def run_lbfgsb(h0, n, maxiter=2000, mu=50.0):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        fun=penalty_obj, x0=h0, args=(n, mu),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return res

def perturb(h, n, alpha=0.1):
    noise = np.random.randn(n) * alpha
    hp = np.clip(h + noise, 0.0, 1.0)
    hp = fix_feasibility(hp, n)
    return hp

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float)
        n_cur = len(h_cur)
    else:
        n_cur = 42
        h_cur = np.full(n_cur, 0.5)

    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution
    elapsed = time.time() - t0
    if budget_s - elapsed > 20:
        try:
            res = run_slsqp(h_cur, n_cur, maxiter=200)
            h_new = fix_feasibility(res.x, n_cur)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new.copy()
        except Exception:
            pass
        # L-BFGS-B polish
        elapsed = time.time() - t0
        if budget_s - elapsed > 15:
            try:
                res = run_lbfgsb(h_cur, n_cur, maxiter=1000, mu=20.0)
                h_new = fix_feasibility(res.x, n_cur)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
            except Exception:
                pass

    # Phase 2: Multi-resolution optimization
    resolutions = [84, 126, 168, 210, 252, 336, 420]
    
    for n_next in resolutions:
        if n_next <= n_cur:
            continue
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 15
        if remaining < 25:
            break

        h_try = interp_h(h_cur, n_next)
        h_trial = h_try.copy()
        improved = False

        # SLSQP
        try:
            res = run_slsqp(h_trial, n_next, maxiter=250)
            h_new = fix_feasibility(res.x, n_next)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_trial = h_new.copy()
                improved = True
        except Exception:
            pass

        # L-BFGS-B
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 15
        if remaining > 20:
            try:
                res = run_lbfgsb(h_trial, n_next, maxiter=1500, mu=30.0)
                h_new = fix_feasibility(res.x, n_next)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_trial = h_new.copy()
                    improved = True
            except Exception:
                pass

        # SLSQP again from improved point
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 15
        if remaining > 20 and improved:
            try:
                res = run_slsqp(h_trial, n_next, maxiter=200)
                h_new = fix_feasibility(res.x, n_next)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_trial = h_new.copy()
            except Exception:
                pass

        h_cur = h_trial.copy()
        n_cur = n_next

    # Phase 3: Multi-start at best resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 10
    n_best = len(best_h)
    
    if remaining > 40 and n_best >= 8:
        for attempt in range(5):
            elapsed = time.time() - t0
            if budget_s - elapsed < 25:
                break
            h_pert = perturb(best_h, n_best, alpha=0.05 * (attempt + 1))
            try:
                res = run_slsqp(h_pert, n_best, maxiter=150)
                h_new = fix_feasibility(res.x, n_best)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 4: Final fine-tune
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 3
    if remaining > 15:
        try:
            res = run_lbfgsb(best_h, n_best, maxiter=1000, mu=10.0)
            h_new = fix_feasibility(res.x, n_best)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
        try:
            res = run_slsqp(best_h, n_best, maxiter=100)
            h_new = fix_feasibility(res.x, n_best)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Ensure exact feasibility
    n_final = len(best_h)
    best_h = fix_feasibility(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
