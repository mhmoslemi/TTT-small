    n = len(h)
    dx = 2.0 / n
    
    # Compute all correlation values for lags 0 to n-1
    lags = np.arange(n)
    corr_vals = np.zeros(n)
    for lag in range(n):
        if lag == 0:
            corr_vals[lag] = np.sum(h * (1 - h))
        else:
            corr_vals[lag] = np.sum(h[lag:] * (1 - h[:-lag]))
    
    obj = np.max(corr_vals) * dx

    shifted = p * (corr_vals - np.max(corr_vals))

def project_onto_constraint(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1 - 1e-15)
        if not np.any(free):
            # Distribute equally
            h += diff / n
            continue
        h[free] += diff / np.sum(free)
    return np.clip(h, 0, 1)

def smooth_max_obj_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation values for lags 0 to n-1
    corr_vals = np.empty(n)
    corr_vals[0] = np.sum(h * (1 - h))
    for lag in range(1, n):
        corr_vals[lag] = np.sum(h[lag:] * (1 - h[:-lag]))
    
    # Smooth max
    m = np.max(corr_vals)
    shifted = p * (corr_vals - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    
    obj = (m + np.log(sum_exp) / p) * dx
    
    # Gradient
    grad = np.zeros(n)
    for lag in range(n):
        w = weights[lag] * dx
        if w < 1e-15:
            continue
        if lag == 0:
            grad[:] += w * (1 - 2 * h)
        else:
            grad[lag:] += w * (1 - h[:n-lag])
            grad[:n-lag] += -w * h[lag:]
    
    return obj, grad

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    
    m = np.max(corr)
    shifted = p * (corr - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    obj = (m + np.log(sum_exp) / p) * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-16:
