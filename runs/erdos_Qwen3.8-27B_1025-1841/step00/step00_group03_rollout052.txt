import numpy as np
import time
from scipy.optimize import minimize

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_for_shift(h, k):
    n = len(h)
    m = abs(k)
    if m == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if k > 0:
        if m < n:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
    else:
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def subgradient_optimize(h, n, time_limit=60.0, rng=None):
    t_start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    lr = 0.08
    vel = np.zeros(n)
    mom = 0.85
    while time.time() - t_start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        thresh = max_val * 0.9 - 1e-12
        active = np.where(corr >= thresh)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += grad_for_shift(h, k)
        grad /= (len(active) + 1e-10)
        grad *= dx
        vel = mom * vel - lr * grad
        h_new = h + vel
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.9998
    return best_h, best_c5

def nelder_mead_fine(h, n, time_limit=30.0):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        c = np.max(np.correlate(hc, 1.0 - hc, mode='full')) * dx
        pen = 50.0 * (np.sum(hc) - target) ** 2
        return c + pen
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() - t_start < time_limit:
        res = minimize(obj, best_h, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-14, 'adaptive': True})
        h_new = repair(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        else:
            break
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

        alloc_r = max(5.0, min(15.0, remaining() * 0.15))

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
