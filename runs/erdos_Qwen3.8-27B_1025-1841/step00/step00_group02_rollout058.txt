import numpy as np
from scipy.optimize import minimize
import math
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def shift_array(h, s):
    n = len(h)
    result = np.zeros(n)
    if s >= 0:
        if s < n:
            result[s:] = h[:n - s]
    else:
        sn = -s
        if sn < n:
            result[:n - sn] = h[sn:]
    return result

def upsample_linear(h, n_out):
    n_in = len(h)
    if n_in == n_out:
        return h.copy()
    x_in = np.arange(n_in, dtype=float)
    x_out = np.linspace(0, n_in - 1, n_out)
    return np.interp(x_out, x_in, h)

def topk_gradient(h, dx, k_top=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_indices = np.argsort(corr)[-k_top:]
    max_c = corr[k_indices[0]]
    weights = np.exp(5.0 * (corr[k_indices] - max_c))
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for idx, w in zip(k_indices, weights):
        s = int(idx) - (n - 1)
        grad -= w * (shift_array(h, s) + shift_array(h, -s))
    return grad * dx, corr.max() * dx, k_indices

def optimize_projected(h0, time_budget, rng):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_onto_constraint(h0.copy(), target_sum)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    v = np.zeros(n)
    mu = 0.8
    step = 0.005
    start = time.time()
    no_improve = 0
    phase = 0

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        if frac > 0.8 and phase == 0:
            phase = 1
            step = 0.001
            mu = 0.5
        elif frac > 0.95 and phase == 1:
            phase = 2
            step = 0.0003
            mu = 0.3

        grad, _, _ = topk_gradient(h, dx, k_top=3 if phase == 0 else 1)
        v = mu * v - step * grad
        h_new = project_onto_constraint(h + v, target_sum)
        if np.any(np.isnan(h_new)):
            h = best_h.copy()
            v = np.zeros(n)
            step *= 0.5
            continue
        h = h_new
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                step *= 0.85
                v *= 0.5
            if no_improve > 1500:

                v = np.zeros(n)
                step = 0.003 if phase == 0 else 0.001
                no_improve = 0

    return best_h, best_c5
