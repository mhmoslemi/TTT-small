def project_to_feasible(x, target_sum):
    """Project x onto {h: 0<=h_i<=1, sum(h_i)=target_sum}"""
    n = len(x)
    # Bisection to find lambda such that sum(clip(x - lambda, 0, 1)) = target_sum
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Fix small numerical issues
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Find elements not at boundary
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        nmask = mask.sum()
        if nmask > 0:
            adj = diff / nmask
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_val = corr.max()

    tol = max(1e-8, 1e-6 * max_val)

            grad[offset:] -= h[:n - offset]
        else:
            grad[-offset:] += one_minus_h[:n + offset]
            grad[:n + offset] -= h[-offset:]
    grad /= len(active)
    return grad * dx

    smooth_max = m + np.log(np.sum(np.exp((corr - m) * T)) / len(corr)) / T

    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for m in active:
        off = m - n + 1
        if off >= 0:
            grad[:n - off] += omh[off:]
            grad[off:] -= h[:n - off]
        else:
            grad[-off:] += omh[:n + off]
            grad[:n + off] -= h[-off:]
    grad /= len(active)
    return grad * dx

def grad_smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full") * dx
    m = corr.max()
    exp_c = np.exp((corr - m) * T)
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        off = k - n + 1
        wk = w[k]
        if off >= 0:
            grad[:n - off] += wk * omh[off:]
            grad[off:] -= wk * h[:n - off]
        else:
            grad[-off:] += wk * omh[:n + off]
            grad[:n + off] -= wk * h[-off:]
    return grad

def optimize_h(h, max_iter, lr_init, decay, momentum, smooth_T, t_deadline):
    n = len(h)
    target = n / 2.0
    h = project_to_feasible(h.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        if smooth_T is not None and smooth_T > 0:
            g = grad_smooth_c5(h, smooth_T)
        else:
            g = grad_c5(h)
        g = g - g.mean()
        vel = momentum * vel - lr * g
        h_new = project_to_feasible(h + vel, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if it % 200 == 199:
            lr *= decay
            if no_improve > 1000:
                lr *= 0.5
                vel *= 0.5
                no_improve = 0
    return best_h, best_c5

def upsample(h, factor):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = project_to_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Phase 1: Smooth optimization at parent resolution
    for T in [5.0, 10.0, 20.0, 50.0]:
        if time.time() > deadline:
            break
        h_opt, c5_opt = optimize_h(best_h, 15000, 0.01, 0.998, 0.7, T, deadline)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Phase 2: Subgradient (hard max) at parent resolution
    if time.time() < deadline:
        h_opt, c5_opt = optimize_h(best_h, 20000, 0.005, 0.998, 0.75, None, deadline)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Phase 3: Upsample and optimize
    for factor in [2, 3, 4]:
        if time.time() > deadline:
            break
        h_up = upsample(best_h, factor)
        new_n = len(h_up)
        new_target = new_n / 2.0
        h_up = project_to_feasible(h_up, new_target)
        for T in [10.0, 30.0]:
            if time.time() > deadline:
                break
            h_opt, c5_opt = optimize_h(h_up, 15000, 0.005, 0.9985, 0.8, T, deadline)
            if c5_opt < best_c5:
                h_up = h_opt
                best_h, best_c5 = h_opt, c5_opt
        if time.time() < deadline:
            h_opt, c5_opt = optimize_h(h_up, 15000, 0.003, 0.999, 0.8, None, deadline)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
    
    # Phase 4: Random restarts at best resolution
    n_best = len(best_h)
    target_best = n_best / 2.0
    attempts = 0
    while time.time() < deadline - 20 and attempts < 30:
        attempts += 1
        scale = 0.05 + 0.15 * np.random.rand()
        h_r = best_h + scale * np.random.randn(n_best)
        h_r = project_to_feasible(h_r, target_best)
        h_opt, c5_opt = optimize_h(h_r, 5000, 0.008, 0.999, 0.7, 20.0, deadline)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        h_opt, c5_opt = optimize_h(h_r, 3000, 0.005, 0.999, 0.75, None, deadline)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Phase 5: Final polish
    if time.time() < deadline - 5:
        h_opt, c5_opt = optimize_h(best_h, 10000, 0.001, 0.9995, 0.7, None, deadline)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    return best_h, best_c5, len(best_h)
