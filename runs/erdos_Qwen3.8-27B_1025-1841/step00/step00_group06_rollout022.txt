    dx = 2.0 / n_points
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, n_points):
    target = n_points / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_lag(h, m, n_points):
    grad = np.ones(n_points)
    if m >= 0:
        if m < n_points:
            grad[:n_points - m] -= h[m:]
            grad[m:] -= h[:n_points - m]
    else:
        p = -m
        if p < n_points:
            grad[p:] -= h[:n_points - p]
            grad[:n_points - p] -= h[p:]
    return grad

def smooth_gradient_descent(h, n_points, time_limit, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    alpha = 0.01
    no_improve = 0
    n_iter = 0
    
    while time.time() < time_limit:
        # Anneal temperature
        t = 5.0 + 45.0 * min(1.0, n_iter / 300.0)
        
        dx = 2.0 / n_points
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full') * dx
        
        # Softmax weights
        max_c = np.max(corr)
        exp_t = np.exp(t * (corr - max_c))
        total_exp = np.sum(exp_t)
        if total_exp == 0:
            break
        weights = exp_t / total_exp
        
        # Weighted gradient
        grad = np.zeros(n_points)
        n_offsets = len(corr)
        for k in range(n_offsets):
            if weights[k] < 1e-12:
                continue
            m = k - (n_points - 1)
            grad += weights[k] * gradient_lag(h, m, n_points)
        grad *= dx
        
        # Line search
        improved = False
        for _ in range(10):

            h_new = project_box_sum(h_new, n_points)
            c5_new = compute_c5(h_new, n_points)

                h = h_new
                improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                break

            alpha = max(alpha * 0.5, 1e-8)
            no_improve += 1

            h = np.clip(h, 0, 1)
            h = project_box_sum(h, n_points)
            alpha = 0.02
            no_improve = 0
        
        n_iter += 1
    
    return best_h, best_c5

            else:

            for idx_i in idx:
                h[idx_i] += rng.randn() * 0.05
            h = np.clip(h, 0, 1)
            h = project_box_sum(h, n_points)
