import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return logsumexp(alpha * corr * dx) / alpha  # This gives smooth max of corr*dx

def smooth_grad(h, n, alpha=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    weights = np.exp(alpha * corr * dx - logsumexp(alpha * corr * dx))
    # weights[k] = exp(alpha*dx*corr[k]) / sum(exp(alpha*dx*corr[j]))
    
    grad = np.zeros(n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def smooth_obj_and_grad(h, alpha=100.0, penalty=1000.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    Z = np.sum(exp_s)
    obj = (max_s + np.log(Z)) / alpha
    
    # Penalty for sum constraint
    s = np.sum(h)
    target = n / 2.0
    penalty_val = penalty * (s - target) ** 2
    obj += penalty_val
    
    # Gradient of smooth max
    p = exp_s / Z
    grad = np.zeros(n)
    
    for j in range(n):
        m = np.arange(-(n-1), n)  # m_k for k=0..2n-2
