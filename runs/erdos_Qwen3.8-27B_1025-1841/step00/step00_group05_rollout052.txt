import numpy as np
from scipy.optimize import differential_evolution
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room, kind='stable')
        rem = diff
        for i in order:
            d = min(room[i], rem)
            h[i] += d
            rem -= d
            if rem <= 1e-15:
                break
    else:
        order = np.argsort(-h, kind='stable')
        rem = -diff
        for i in order:
            d = min(h[i], rem)
            h[i] -= d
            rem -= d
            if rem <= 1e-15:
                break
    return h

def sa_move(h, n, rng, mode):
    if mode == 0:
        i = rng.randint(n)
        h[i] = np.clip(h[i] + rng.normal(0, 0.1), 0, 1)
    elif mode == 1:
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            j = (j + 1) % n
        d = rng.normal(0, 0.1)
        h[i] = np.clip(h[i] + d, 0, 1)
        h[j] = np.clip(h[j] - d, 0, 1)
    elif mode == 2:
        w = rng.randint(2, min(8, n // 4 + 1))
        s = rng.randint(0, max(1, n - w + 1))
        h[s:s+w] = np.clip(h[s:s+w] + rng.normal(0, 0.08, w), 0, 1)
    else:
        k = max(1, n // 10)
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = np.clip(h[idx] + rng.uniform(-0.3, 0.3, k), 0, 1)
    return h

def sa_refine(h, n, t0, deadline, rng, T0=0.03, n_iter=100000):
    h = h.copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    cur_c5 = best_c5
    T = T0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        mode = rng.randint(4)
        h2 = h.copy()
        sa_move(h2, n, rng, mode)
        h2 = project(h2, n)
        c5_new = eval_c5(h2)
        if c5_new <= cur_c5 or (T > 1e-7 and rng.random() < np.exp(-(c5_new - cur_c5) / T)):
            h = h2
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
        if it % 2000 == 1999:
            T = max(0.0003, T * 0.95)
        if it % 10000 == 9999 and T < 0.001:
            T = T0 * 0.5
            h = best + rng.normal(0, 0.05, n)
            h = project(h, n)
            cur_c5 = eval_c5(h)
    return best, best_c5

def make_binary_pattern(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return project(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 30

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)
    n0 = len(h0)
    h0 = project(h0, n0)
    best_h = h0.copy()
    best_c5 = eval_c5(h0)

    # Phase 1: SA on initial resolution with the given starting point
    n_iter1 = max(50000, int((budget_s * 0.25) / max(1.0, n0 * 1e-4)))
    h_r, c5_r = sa_refine(h0, n0, t0, deadline, rng, T0=0.04, n_iter=n_iter1)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r.copy()

    for _ in range(4):
        if time.time() > deadline - 300:
            break
        h_rand = make_binary_pattern(n0, rng)
        h_r2, c5_r2 = sa_refine(h_rand, n0, t0, deadline, rng, T0=0.08, n_iter=30000)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2.copy()

    # Phase 3: Progressive grid refinement
    resolutions = sorted(set([n0 * 2, n0 * 4, n0 * 8, min(n0 * 16, 1400)]))
    for n_fine in resolutions:
        if time.time() > deadline - 120:
            break

        x_new = np.linspace(0, 2, n_fine, endpoint=False)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project(h_fine, n_fine)
        c5_fine = eval_c5(h_fine)

        n_iter_fine = max(5000, int(min(40000, (deadline - time.time() - 60) / max(0.05, n_fine * 1e-4))))

        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
        elif c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

        n_final = len(best_h)

        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project(best_h, len(best_h))
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, len(best_h))
