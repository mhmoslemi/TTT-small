import numpy as np
from scipy.optimize import minimize
import time

_TARGET_SUM = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-15
        if np.any(mask):
            slack = 1.0 - h[mask]
            total = slack.sum()
            if total > 1e-15:
                h[mask] += diff * slack / total
    else:
        mask = h > 1e-15
        if np.any(mask):
            vals = h[mask]
            total = vals.sum()
            if total > 1e-15:
                h[mask] += diff * vals / total
    return np.clip(h, 0.0, 1.0)

def _c5_obj(x):
    return compute_c5(x)

def _sum_constraint(x):
    return np.sum(x) - _TARGET_SUM

def _sum_jacobian(x):
    return np.ones(len(x))

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def elapsed_time(start):
    return time.time() - start

def make_h_init(best_h, n, target, rng, scale):
    if len(best_h) == n:
        base = best_h
    else:
        base = upsample_h(best_h, n)
    h_init = base + rng.normal(0, scale, n)
    return project_feasible(h_init, target)
