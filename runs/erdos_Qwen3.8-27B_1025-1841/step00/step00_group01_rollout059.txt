```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return corr.max() * dx

def resample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0, 1)
    # Fix sum
    target_sum = n_new / 2.0
    current_sum = h_new.sum()
    # Adjust
    if abs(current_sum - target_sum) > 1e-10:
        h_new += (target_sum - current_sum) / n_new
        h_new = np.clip(h_new, 0, 1)
        # Fine adjustment
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-10:
            # Distribute among non-bound elements
            mask = (h_new > 0.01) & (h_new < 0.99)
            if mask.sum() > 0:
                h_new[mask] += diff / mask.sum()
                h_new = np.clip(h_new, 0, 1)
    return h_new

def project_onto_simplex(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}"""
    n = len(h)
    lo, hi = h.min() - target_sum/n - 1, h.max() - target_sum/n + 1
    # More standard: find mu such that sum(clamp(h - mu, 0, 1)) = target_sum
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mu = (lo + hi) / 2
        proj = np.clip(h - mu, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    return np.clip(h - (lo + hi) / 2, 0, 1)

def slsqp_optimize(h0, n, time_limit=60):
    """Optimize using SLSQP"""
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def objective(vars_):
        t = vars_[n]
        h = vars_[:n]
        return t
    
    def constraints(vars_):
        h = vars_[:n]
        t = vars_[n]
        one_m = 1.0 - h
        corr = np.correlate(h, one_m, mode='full')
        # corr[k] * dx <= t
        cons = t - corr * dx
        return list(cons)
    
    def eq_constraint(vars_):
        h = vars_[:n]
        return h.sum() - target_sum
    
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq_constraint}] + 
                   [{'type': 'ineq', 'fun': lambda v, k=k: v[n] - np.correlate(v[:n], 1-v[:n], mode='full')[k]*dx} for k in range(2*n-1)],
                   maxiter=200,
                   options={'disp': False, 'maxiter': 200})
    
    if res.success:
        h_opt = np.clip(res.x[:n], 0, 1)
        # Fix sum
        h_opt = project_onto_simplex(h_opt, target_sum)
        return h_opt
    return h0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_sum(h, target):
    n = len(h)
    if n == 0:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mu = (lo + hi) / 2.0
        s = np.clip(h - mu, 0.0, 1.0).sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_new = np.linspace(0, n_old - 1, n_new)
    idx = np.floor(x_new).astype(int)
    idx_hi = np.minimum(idx + 1, n_old - 1)
    frac = x_new - idx
    h_new = h_old[idx] * (1.0 - frac) + h_old[idx_hi] * frac
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_sum(h_new, n_new / 2.0)

def slsqp_optimize(h0, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(v):
        return v[n]
    
    def eq_con(v):
        return v[:n].sum() - target
    
    def ineq_con(v):
        h = v[:n]
        t = v[n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr * dx
    
    x0 = np.concatenate([h0, [compute_c5(h0) + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq_con},
                                {'type': 'ineq', 'fun': ineq_con}],
                   options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
    
    if res.fun < 100:  # check it didn't blow up
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_sum(h_opt, target)
        return h_opt
    return h0

def lbfgsb_optimize(h0, n, time_limit=60.0, tau=0.05):
    dx = 2.0 / n
    target = n / 2.0
    start = time.time()
    
    def obj_grad(params):
        h = np.clip(params, 1e-6, 1.0 - 1e-6)
        one_m = 1.0 - h
        corr = np.correlate(h, one_m, mode='full')
        cmax = corr.max()
        exp_c = np.exp((corr - cmax) / tau)
        S = exp_c.sum()
        w = exp_c / S
        f = (cmax + tau * np.log(S)) * dx
        # Penalty
        f += 50.0 * (h.sum() - target) ** 2
        
        # Gradient
        grad = np.zeros(n)
        for m in range(len(corr)):
            wm = w[m]
            if wm < 1e-14:
                continue
            k = m - (n - 1)
            if k >= 0:
                # (1-h[j-k]) for j in [k, n-1]
                j_lo, j_hi = k, n - 1
                grad[j_lo:j_hi+1] += wm * (1.0 - h[j_lo-k:j_hi+1-k])
                # -h[j+k] for j in [0, n-1-k]
                j_lo2, j_hi2 = 0, n - 1 - k
                grad[j_lo2:j_hi2+1] -= wm * h[j_lo2+k:j_hi2+1+k]
            else:
                kp = -k
                # (1-h[j+kp]) for j in [0, n-1-kp]
                j_lo, j_hi = 0, n - 1 - kp
                grad[j_lo:j_hi+1] += wm * (1.0 - h[j_lo+kp:j_hi+1+kp])
                # -h[j-kp] for j in [kp, n-1]
                j_lo2, j_hi2 = kp, n - 1
                grad[j_lo2:j_hi2+1] -= wm * h[j_lo2-kp:j_hi2+1-kp]
        
        grad *= dx
        # Penalty gradient
        grad += 100.0 * (h.sum() - target)  # d/dh[j] of 50*(sum-target)^2 = 100*(sum-target)
        
        return f, grad
    
    h_curr = h0.copy()
    best_h = h0.copy()
    best_f = float('inf')
    
    for restart in range(5):
        if time.time() - start > time_limit:
            break
        res = minimize(obj_grad, h_curr, method='L-BFGS-B',
                       jac=True, bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 200, 'maxfun': 2000, 'disp': False})
        h_curr = np.clip(res.x, 0.0, 1.0)
        h_curr = project_sum(h_curr, target)
        f = compute_c5(h_curr)
        if f < best_f:
            best_f = f
            best_h = h_curr.copy()
        # Perturb for next restart
        if time.time() - start > time_limit:
            break
        h_curr = best_h + 0.05 * np.random.randn(n)
        h_curr = np.clip(h_curr, 0.0, 1.0)
        h_curr = project_sum(h_curr, target)
    
    return best_h

def run(seed=42, budget_s=1300, **kwargs):

    start_time = time.time()
    
    # Get initial
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        try:
            h_init = np.array(initial_h_values, dtype=float)
        except:
            h_init = np.full(48, 0.5)
    else:

    best_c5 = compute_c5(h_init)
    best_n = len(h_init)

        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n_parent

    resolutions = sorted(set([n_parent, 48, 64, 96, 128, 192, 256, 384, 512, 768]))

        if elapsed > budget_s * 0.8:

        remaining = budget_s - elapsed
        time_per_res = max(10, remaining / len(resolutions))

        else:

        c5 = compute_c5(h_opt)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_opt
            best_n = n

        c5 = compute_c5(h_ref)

            best_h = h_ref

    best_h = project_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, best_n
```

```python
import numpy as np
from scipy.optimize import minimize
import time

# Global state for optimizer callbacks
_G_N = 0
_G_DX = 0.0
_G_TARGET = 0.0
_G_TAU = 0.05

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _project_sum(h, target):
    n = len(h)
    if n == 0:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mu = (lo + hi) / 2.0
        s = np.clip(h - mu, 0.0, 1.0).sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return np.clip(h_old, 0.0, 1.0).copy()
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    idx_lo = np.floor(x_new).astype(int)
    idx_hi = np.minimum(idx_lo + 1, n_old - 1)
    frac = x_new - idx_lo
    h_new = h_old[idx_lo] * (1.0 - frac) + h_old[idx_hi] * frac
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project_sum(h_new, n_new / 2.0)

def _obj_grad(params):
    n = _G_N
    h = np.clip(params, 1e-8, 1.0 - 1e-8)
    one_m = 1.0 - h
    corr = np.correlate(h, one_m, mode='full')
    cmax = corr.max()
    exp_c = np.exp((corr - cmax) / _G_TAU)
    S = exp_c.sum()
    w = exp_c / S
    f = (cmax + _G_TAU * np.log(S)) * _G_DX
    f += 50.0 * (h.sum() - _G_TARGET) ** 2
    grad = np.zeros(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            j_lo, j_hi = k, n - 1
            grad[j_lo:j_hi + 1] += wm * (1.0 - h[j_lo - k:j_hi + 1 - k])
            j_lo2, j_hi2 = 0, n - 1 - k
            grad[j_lo2:j_hi2 + 1] -= wm * h[j_lo2 + k:j_hi2 + 1 + k]
        else:
            kp = -k
            j_lo, j_hi = 0, n - 1 - kp
            grad[j_lo:j_hi + 1] += wm * (1.0 - h[j_lo + kp:j_hi + 1 + kp])
            j_lo2, j_hi2 = kp, n - 1
            grad[j_lo2:j_hi2 + 1] -= wm * h[j_lo2 - kp:j_hi2 + 1 - kp]
    grad *= _G_DX / _G_TAU
    grad += 100.0 * (h.sum() - _G_TARGET)
    return f, grad

def _lbfgsb_opt(h0, n, time_limit=60.0):
    global _G_N, _G_DX, _G_TARGET, _G_TAU
    _G_N = n
    _G_DX = 2.0 / n
    _G_TARGET = n / 2.0
    _G_TAU = 0.05
    start = time.time()
    best_h = h0.copy()
    best_f = _compute_c5(h0)
    h_curr = h0.copy()
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        if r > 0:
            scale = 0.03 * (1.0 + 0.1 * r)
            h_curr = best_h + scale * np.random.randn(n)
            h_curr = np.clip(h_curr, 0.0, 1.0)
            h_curr = _project_sum(h_curr, n / 2.0)
        _G_TAU = max(0.02, 0.1 * (0.85 ** r))
        res = minimize(_obj_grad, h_curr, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 300, 'maxfun': 3000, 'disp': False})
        h_res = np.clip(res.x, 0.0, 1.0)
        h_res = _project_sum(h_res, n / 2.0)
        f = _compute_c5(h_res)
        if f < best_f - 1e-12:
            best_f = f
            best_h = h_res.copy()
    return best_h

def _slsqp_obj(v):
    return v[_G_N]

def _slsqp_eq(v):
    return v[:_G_N].sum() - _G_TARGET

def _slsqp_ineq(v):
    h = v[:_G_N]
    t = v[_G_N]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr * _G_DX

def _slsqp_opt(h0, maxiter=100):
    global _G_N, _G_DX, _G_TARGET
    n = len(h0)
    _G_N = n
    _G_DX = 2.0 / n
    _G_TARGET = n / 2.0
    t0 = _compute_c5(h0) + 0.01
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'eq', 'fun': _slsqp_eq}, {'type': 'ineq', 'fun': _slsqp_ineq}]
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = _project_sum(h_opt, n / 2.0)
            return h_opt
    except Exception:
        pass
    return h0.copy()

def _random_binary_init(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    if n % 2 == 1:
        for _ in range(n_ones):
            idx = rng.randint(n)
            h[idx] = 1.0
        h[rng.randint(n)] = 0.5
    else:
        idxs = rng.permutation(n)
        h[idxs[:n_ones]] = 1.0
        h[idxs[n_ones:]] = 0.0
    return h

def _sawtooth_init(n, rng):
    h = np.zeros(n)
    n_periods = rng.randint(2, max(3, n // 10))
    width = n / n_periods
    for p in range(n_periods):
        start = int(p * width)
        end = int(min((p + 1) * width / 2, n))
        h[start:end] = 1.0
    h = _project_sum(h, n / 2.0)
    return h
```
