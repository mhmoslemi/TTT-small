So: max(0, k-m+1) ≤ j ≤ min(m-1, k)

result[k] = sum_{j=max(0,k-m+1)}^{min(m-1,k)} a[j] * v[j-k+m-1]

For a=h, v=1-h, m=n:
c[k] = sum_{j=L}^{R} h[j] * (1 - h[j-k+n-1])
where L = max(0, k-n+1), R = min(n-1, k)

Let me verify: for k=0, L=0, R=0 (since min(n-1,0)=0). c[0] = h[0]*(1-h[n-1]).
For k=n-1, L=0, R=n-1. c[n-1] = sum_{j=0}^{n-1} h[j]*(1-h[j]) = sum h[j](1-h[j]).
For k=n, L=1, R=n-1. c[n] = sum_{j=1}^{n-1} h[j]*(1-h[j-1]).
For k=2n-2, L=n-1, R=n-1. c[2n-2] = h[n-1]*(1-h[n-2]).

So c[n-1] = sum h[j](1-h[j]) is the "self" correlation (k=n-1 corresponds to shift 0 in the continuous sense? Or is it the full-length overlap?)

    max_c = np.max(c)

    """Project h to nearest vertex of {0<=x<=1, sum(x)=n/2}."""
    if n % 2 == 0:
        # Vertices have exactly n/2 ones and n/2 zeros
        target_ones = n // 2
        # Keep the top target_ones values as 1
        threshold = np.sort(h)[::-1][target_ones - 1]
        h_v = np.where(h >= threshold, 1.0, 0.0)
        # Fix if we have more than target_ones ones (ties at threshold)
        n_ones = np.sum(h_v)
        if n_ones != target_ones:
            # Sort by h value, pick top target_ones
            order = np.argsort(h)[::-1]
            h_v = np.zeros(n)
            h_v[order[:target_ones]] = 1.0
        return h_v
    else:

        target_ones = (n - 1) // 2

        h_v = np.zeros(n)

        h_v[order[target_ones]] = 0.5

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n)), c


def project_feasible(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def compute_subgrad(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(c)
    active = np.where(c >= max_c - 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        L = max(0, k - n + 1)
        R = min(n - 1, k)
        idx1 = np.arange(L, R + 1)
        j2 = idx1 - k + n - 1
        grad[idx1] += (1.0 - h[j2])
        j_lo = max(0, L - k + n - 1)
        j_hi = min(n - 1, R - k + n + 1)
        if j_lo <= j_hi:
            idx2 = np.arange(j_lo, j_hi + 1)
            j3 = idx2 + k - n + 1
            grad[idx2] -= h[j3]
    grad *= (2.0 / n)
    return grad, c
