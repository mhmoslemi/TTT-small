    """Compute all correlation values * dx."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n_points):
    return np.max(compute_c_values(h, n_points))

def project_feasible(h, target_sum):
    """Project onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):

        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time

def compute_c_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return np.max(compute_c_values(h, n))

def project_feasible(h, target_sum):
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def grad_for_shift(h, s, n, dx):
    """Gradient of c[k]*dx for shift s (where k = n-1+s for positive, n-1+s for negative).
    s can be 0, positive, or negative."""
    if s == 0:
        return dx * (1.0 - 2.0 * h)
    elif s > 0:
        grad = np.zeros(n)
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
        return dx * grad
    else:
        s = -s
        grad = np.zeros(n)
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
        return dx * grad

def softmax_gradient(h, n, temperature, top_m=10):
    """Compute gradient of softmax approximation to max(c_values)."""
    dx = 2.0 / n
    c_vals = compute_c_values(h, n)
    
    # Compute softmax weights (focus on top values)
    c_max = c_vals.max()
    log_w = (c_vals - c_max) / temperature
    log_w = np.clip(log_w, -50, 0)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c_vals))
        w[np.argmax(c_vals)] = 1.0
    else:
        w /= w_sum
    
    # Get top indices
    top_idx = np.argsort(c_vals)[-top_m:]
    w_top = w[top_idx]
    w_top_sum = w_top.sum()
    if w_top_sum < 1e-30:
        w_top = np.zeros(top_m)
        w_top[np.argmax(c_vals[top_idx])] = 1.0
    else:
        w_top /= w_top_sum
    
    # Compute gradient
    center = n - 1
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        s = idx - center
        grad += w_top[i] * grad_for_shift(h, s, n, dx)
    
    return grad

    return np.max(compute_c_values(h, n))

def project_feasible(h, target_sum):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_for_shift(h, s, n, dx):
    if s == 0:
        return dx * (1.0 - 2.0 * h)
    elif s > 0:
        grad = np.zeros(n)
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
        return dx * grad
    else:
        s = -s
        grad = np.zeros(n)
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
        return dx * grad

def softmax_gradient(h, n, temperature, top_m=20):
    dx = 2.0 / n
    c_vals = compute_c_values(h, n)
    c_max = c_vals.max()
    log_w = (c_vals - c_max) / temperature
    log_w = np.clip(log_w, -60.0, 0.0)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        top_idx = np.array([np.argmax(c_vals)])
        w_top = np.array([1.0])
    else:
        w /= w_sum
        top_m = min(top_m, len(c_vals))
        top_idx = np.argsort(c_vals)[-top_m:]
        w_top = w[top_idx]
        ws = w_top.sum()
        if ws < 1e-30:
            w_top = np.zeros(top_m)
            w_top[np.argmax(c_vals[top_idx])] = 1.0
        else:
            w_top /= ws
    center = n - 1
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        s = int(top_idx[i]) - center
        grad += w_top[i] * grad_for_shift(h, s, n, dx)
    return grad

def optimize_at_n(h, n, time_limit, lr_init=0.005, temp_init=0.02, max_iters=50000):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = lr_init
    temp = temp_init
    stall = 0
    for it in range(max_iters):
        if time.time() - time_limit > 0:
            break
        if it % 5000 == 0 and it > 0 and time.time() > time_limit:
            break
        grad = softmax_gradient(h, n, temp, top_m=min(30, 2 * n - 1))
        h_new = project_feasible(h - lr * grad, n // 2)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stall = 0
            lr = min(lr * 1.003, 0.03)
        else:
            h = h_new
            stall += 1
            lr *= 0.995
            if lr < 0.001:
                h = project_feasible(h + np.random.randn(n) * 0.04, n // 2)
                lr = lr_init
                stall = 0
        temp = max(temp * 0.99995, 0.002)
    return best_h, best_c5, h

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(42, 0.5)

    n = len(h)
    h = project_feasible(h, n // 2)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    def remaining():
        return budget_s - (time.time() - start)

    # Phase 1: Optimize at initial resolution
    if remaining() > 100:
        t1 = time.time() + min(remaining() - 100, 300)
        bh, bc5, h = optimize_at_n(h, n, t1, lr_init=0.005, temp_init=0.02)
        if bc5 < best_c5:
            best_h, best_c5 = bh.copy(), bc5

    # Phase 2: Try SLSQP at current resolution
    if remaining() > 200:
        def obj_slsqp(hv):
            return compute_c5(hv, n)
        def con_slsqp(hv):
            return np.sum(hv) - n / 2.0
        try:
            res = minimize(obj_slsqp, h, method='SLSQP',
                          bounds=[(0, 1)] * n,
                          constraints=[{'type': 'eq', 'fun': con_slsqp}],
                          options={'maxiter': 300, 'ftol': 1e-10})
            if res.x is not None:
                h_res = project_feasible(res.x, n / 2)
                c5_res = compute_c5(h_res, n)
                if c5_res < best_c5:
                    best_h, best_c5 = h_res.copy(), c5_res
                    h = h_res
        except Exception:
            pass

    # Phase 3: Multi-resolution upsample and optimize
    for target_n in [84, 168, 336, 504]:
        if remaining() < 150:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        h_fine = np.interp(x_new, x_old, h, period=2)
        h_fine = project_feasible(h_fine, target_n // 2)
        n = target_n
        h = h_fine

        alloc = min(remaining() - 120, 400)
        t_end = time.time() + alloc
        bh, bc5, h = optimize_at_n(h, n, t_end, lr_init=0.003, temp_init=0.01)
        if bc5 < best_c5:
            best_h, best_c5 = bh.copy(), bc5

    # Phase 4: Final SLSQP at best resolution if time permits
    if remaining() > 180:
        n_final = len(best_h)
        def obj_final(hv):
            return compute_c5(hv, n_final)
        def con_final(hv):
            return np.sum(hv) - n_final / 2.0
        try:
            res = minimize(obj_final, best_h, method='SLSQP',
                          bounds=[(0, 1)] * n_final,
                          constraints=[{'type': 'eq', 'fun': con_final}],
                          options={'maxiter': 500, 'ftol': 1e-12})
            if res.x is not None:
                h_res = project_feasible(res.x, n_final / 2)
                c5_res = compute_c5(h_res, n_final)
                if c5_res < best_c5:
                    best_h, best_c5 = h_res.copy(), c5_res
        except Exception:
            pass

    # Ensure exact feasibility
    best_h = project_feasible(best_h, len(best_h) // 2)
    # Snap to exact sum if there's a tiny residual
    residual = (len(best_h) / 2.0) - best_h.sum()
    if abs(residual) > 1e-10:
        best_h += residual / len(best_h)
        best_h = np.clip(best_h, 0, 1)
        best_h = project_feasible(best_h, len(best_h) / 2.0)

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
