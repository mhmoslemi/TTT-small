    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Log-sum-exp
    max_corr = np.max(corr)
    logits = beta * (corr - max_corr)
    weights = np.exp(logits)
    weights /= np.sum(weights)
    value = dx * (max_corr + np.log(np.sum(np.exp(logits))) / beta)

    for k in range(len(corr)):
        if weights[k] < 1e-8:
            continue
        grad += weights[k] * sg_component(h, k, n)
    grad *= dx
    
    return value, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def sg_component(h, k_idx, n):
    s = k_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        length = n - s
        for idx in range(length):
            i = idx + s
            j = idx
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]
    else:
        offset = -s
        length = n - offset
        for idx in range(length):
            i = idx
            j = idx + offset
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]
    return grad

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = 0.5 * np.ones(70)
    
    n = len(h)
    target_sum = n / 2.0
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Subgradient with log-sum-exp
    momentum = np.zeros(n)
    beta = 10.0
    lr = 0.02
    
    for it in range(10000):
        if elapsed() > budget_s * 0.35:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_c = np.max(corr)
        logits = beta * (corr - max_c)
        exp_logits = np.exp(logits)
        weights = exp_logits / np.sum(exp_logits)
        
        # Gradient
        grad = np.zeros(n)
        for k in range(len(weights)):
            w = weights[k]
            if w > 1e-6:
                grad += w * sg_component(h, k, n)
        
        dx = 2.0 / n
        grad *= dx
        
        # Momentum
        momentum = 0.9 * momentum - lr * grad
        h = h + momentum
        h = project_feasible(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Schedule
        if it % 200 == 0:
            lr = max(0.002, lr * 0.97)
            beta = min(50.0, beta * 1.1)
    
    # Phase 2: Fine-tune with pure subgradient (beta -> inf)
    lr = 0.005
    for it in range(3000):
        if elapsed() > budget_s * 0.5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(corr))
        g = sg_component(h, k_max, n) * (2.0 / n)
        h = h - lr * g
        h = project_feasible(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 3: Random perturbation + restart
    for restart in range(5):
        if elapsed() > budget_s * 0.7:
            break
        h = best_h.copy()
        # Add random perturbation
        h += rng.randn(n) * 0.05
        h = project_feasible(h, target_sum)
        
        lr = 0.01
        for it in range(2000):
            if elapsed() > budget_s * 0.8:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_max = int(np.argmax(corr))
            g = sg_component(h, k_max, n) * (2.0 / n)
            h = h - lr * g
            h = project_feasible(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Phase 4: Binary local search
    if elapsed() < budget_s * 0.85:
        h_local = best_h.copy()
        for it in range(5000):
            if elapsed() > budget_s * 0.9:
                break
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            # Swap h[i] and h[j]
            tmp = h_local[i]
            h_local[i] = h_local[j]
            h_local[j] = tmp
            c5_new = compute_c5(h_local)
            if c5_new > best_c5 + 1e-9:
                h_local[j], h_local[i] = h_local[i], h_local[j]
            else:
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_local.copy()
    
    # Final
    final_n = len(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, final_n

Let me think about this differently. For the FINITE (non-periodic) case with no wraparound:

We have n positions, n/2 ones, n/2 zeros.
For lag k (1 ≤ k ≤ n-1):
output[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])  [right side of center]
output[n-1-k] = sum_{i=k}^{n-1} h[i]*(1-h[i-k])  [left side of center]

For the right side (shift s = k):
output[n-1+s] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])

This counts the number of "1 followed by (s positions later) 0" or equivalently "0 followed by (s positions later) 1."

For a random sequence: E[output[n-1+s]] = (n-s)/4.
Max at s=1: ≈ (n-1)/4. C5 ≈ (2/n)(n-1)/4 ≈ 1/2.

To beat 1/2, we need a sequence where the cross-correlation is MORE uniform than random.

The minimum possible max is at least (sum of all outputs) / (number of outputs) = average.
Sum of all output[n-1+s] for s=1 to n-1 (right side only):
= sum_s sum_i h[i+s](1-h[i])
= sum_i (1-h[i]) * sum_{s=1}^{n-1-i} h[i+s]
= sum_i (1-h[i]) * (number of ones to the right of i)

For a random binary sequence, this is ≈ (n/2) * (n/4) = n²/8.
Average per s: (n²/8) / (n-1) ≈ n/8.
So C5 ≥ (2/n)(n/8) = 1/4.

But we also have the left side. The total number of non-center outputs is 2(n-1), and the sum of all of them is:
sum of right side + sum of left side = 2 * (n²/8) = n²/4 (approximately)
Average per output: (n²/4) / (2(n-1)) ≈ n/8.
C5 ≥ (2/n)(n/8) = 1/4.
