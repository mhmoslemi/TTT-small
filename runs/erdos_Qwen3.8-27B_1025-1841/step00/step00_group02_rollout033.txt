from scipy.optimize import minimize
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection."""
    if h.sum() == target:
        return np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    return np.interp(x_new, x_old, h)

def soft_max_obj(x, n, beta, penalty_w):
    x = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = corr * dx
    m = vals.max()
    soft = m + np.log(np.sum(np.exp(vals - m))) / beta
    pen = penalty_w * (x.sum() - n / 2.0) ** 2
    return soft + pen

def binary_refine(h, rng, max_time=60):
    """Local search on binary h using random swaps."""
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    # Fix sum
    target = n // 2
    for _ in range(100):
        if h.sum() > target:
            ones = np.where(h == 1)[0]
            if len(ones) == 0: break
            h[ones[rng.randint(len(ones))]] = 0
        elif h.sum() < target:
            zeros = np.where(h == 0)[0]
            if len(zeros) == 0: break
            h[zeros[rng.randint(len(zeros))]] = 1
        else:
            break
    
    t_start = time.time()
    current_c5 = compute_C5(h)
    
    while time.time() - t_start < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_C5(h)
        if new_c5 <= current_c5:
            current_c5 = new_c5
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return h, current_c5

def run(seed=42, budget_s=1300, **kwargs):

    for n in target_ns:

            best_c5 = c5
