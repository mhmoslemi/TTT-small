  For m < n_points - 1: lag k = m - (n_points-1) < 0
  result[m] = sum_n h[n] * (1-h[n+k]) for valid n (where n+k >= 0)

from scipy.optimize import minimize

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return h

def make_binary_even(h_cont, n):
    h_bin = (h_cont >= 0.5).astype(float)
    ones = int(np.sum(h_bin))
    target = n // 2
    if ones > target:
        order = np.argsort(h_cont)
        cnt = 0
        for i in order:
            if cnt >= ones - target:

    while time.time() < deadline:

            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        old_i, old_j = h[i], h[j]
        h[i] = 1.0 - old_i
        h[j] = 1.0 - old_j
        new = evaluate_c5(h, n)
        
        accept = False
        if new < curr:
            accept = True
        elif use_saar and new - curr < temperature:
            if rng.random() < np.exp(-(new - curr) / max(temperature, 1e-10)):
                accept = True

            curr = new
            no_improve = 0
            ones_idx = np.where(h >= 0.5)[0]
            zeros_idx = np.where(h < 0.5)[0]
        else:
            h[i] = old_i
            h[j] = old_j
            no_improve += 1
        
        if no_improve > 3000:
            temperature *= 0.5
            no_improve = 0
            if temperature < 1e-6:

def golden_ratio_binary(n):
