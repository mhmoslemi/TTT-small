                    break
                # Random perturbation
                n_perturb = max(1, n_points // 10)
                idx_perturb = rng.choice(n_points, n_perturb, replace=False)
                h[idx_perturb] += rng.uniform(-0.1, 0.1, n_perturb)
                h = _project_to_slice(h, S)
                overlaps = compute_overlaps(h)
                stall_count = 0
        else:
            stall_count = 0
        
        # Periodic exact recomputation
        if time.time() - start < time_limit - 5:
            overlaps = compute_overlaps(h)
    
    h = _project_to_slice(h, S)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    
    return best_h, best_c5

def _projected_grad(h, n_points, time_limit):
    S = n_points / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = 0.02
    beta = 5.0
    dx = 2.0 / n_points
    n_ov = 2 * n_points - 1
    
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        
        overlaps = compute_overlaps(h)
        max_ov = overlaps.max()
        scaled = beta * (overlaps - max_ov)
        exp_ov = np.where(scaled > -50, np.exp(scaled), 0.0)
        total = exp_ov.sum()
        if total < 1e-30:
            beta = max(1.0, beta * 0.5)
            continue
        softmax = exp_ov / total
        
        grad = np.zeros(n_points)
        one_minus_h = 1.0 - h
        for k_idx in range(n_ov):
            w = softmax[k_idx]
            if w < 1e-10:
                continue
            m = k_idx - (n_points - 1)
            grad += w * (_shift_zero(one_minus_h, m, n_points) - _shift_zero(h, -m, n_points))
        grad *= dx
        
        h_new = h - lr * grad
        h_new = _project_to_slice(h_new, S)
        c5_new = compute_c5(h_new)
        h = h_new
        
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h.copy()
        
        if it % 2000 == 1999:
            lr *= 0.7
            beta = min(30.0, beta * 1.3)
    
    return best_h, best_c5

def _try_binary(h, n_points):
    """Create a binary version of h."""
    if n_points % 2 != 0:
        return None
    S = n_points / 2
    n_half = int(S)
    h_bin = np.zeros(n_points)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:n_half]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    except:
        n_points = 70
        h_init = np.full(n_points, 0.5)
    
    S = n_points / 2.0
    h_init = _project_to_slice(h_init, S)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Phase 1: Projected gradient (short)
    t1 = min(100.0, max(10.0, budget_s * 0.1))
    elapsed = time.time() - start_time
    h, c5 = _projected_grad(h_init, n_points, min(t1, budget_s - elapsed))
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Phase 2: Pair descent (main optimization)
    t2 = remaining * 0.6
    h, c5 = _pair_descent(best_h, n_points, t2, rng)
