The objective is max_k ∫ h(x)(1-h(x+k)) dx.

Note that ∫ h(x)(1-h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k) dx (over the domain where both are defined).

For shift k: the domain of integration is [0,2] ∩ [-k, 2-k] (for the second term).

If k=0: ∫₀² h(x) dx - ∫₀² h(x)² dx = 1 - ∫₀² h² dx.

So for k=0, we want to maximize ∫ h² dx (to minimize 1 - ∫h²). This means we want h to be as "spiky" as possible (close to 0 or 1).

For k≠0, we want ∫ h(x)h(x+k) dx to be large (to minimize the overlap), which means h should be "anti-correlated" with its shifts.

So there's a tension: for k=0, we want h to be spiky (0 or 1), but for k≠0, we want h to be spread out.

This suggests the optimal solution might be a mixture: mostly 0 or 1, with some transition regions.

        f, g = smooth_max_f(h, n, dx, T), smooth_max_g(h, n, dx, T)
        step = lr
        for _ in range(40):
            h_new = project_feasible(h - step * g, n)
            f_new = smooth_max_f(h_new, n, dx, T)
            if f_new < f - 1e-15:
                break
            step *= 0.5
        h = h_new
        c5 = exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        if abs(f - f_prev) < 1e-12:
            break
        f_prev = f
        lr = step
    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_init = np.asarray(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = exact_c5(h_init)
    best_n = len(h_init)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP at initial resolution
    n = best_n
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constr_eq, 'jac': constr_eq_jac, 'args': (n,)}]
    for T in [5.0, 12.0]:
        if rem() < 20:
            break
        try:
            res = minimize(slsqp_obj, h_init.copy(), jac=slsqp_grad,
                           args=(n, dx, T), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-12})
            c5 = exact_c5(res.x)
            if c5 < best_c5:
                best_h, best_c5, best_n = res.x.copy(), c5, n
        except Exception:
            pass

    # Phase 2: Multi-resolution
    cur_h = best_h.copy()
    cur_n = best_n
    for n_t in [100, 200, 400, 800, 1600]:
        if rem() < 50:
            break
        if n_t <= cur_n:
            continue
        cur_h = upsample(cur_h, n_t)
        n = n_t
        dx = 2.0 / n

        if n <= 200:
            bounds_n = [(0.0, 1.0)] * n
            cons_n = [{'type': 'eq', 'fun': constr_eq, 'jac': constr_eq_jac, 'args': (n,)}]
            for T in [8.0, 20.0]:
                if rem() < 25:
                    break
                try:
                    res = minimize(slsqp_obj, cur_h.copy(), jac=slsqp_grad,
                                   args=(n, dx, T), method='SLSQP',
                                   bounds=bounds_n, constraints=cons_n,
                                   options={'maxiter': 400, 'ftol': 1e-12})
                    c5 = exact_c5(res.x)
                    if c5 < best_c5:
                        best_h, best_c5, best_n = res.x.copy(), c5, n
                    cur_h = project_feasible(res.x, n)
                except Exception:
                    pass

        if rem() > 20:
            T_pg = 10.0
            h_pg, c5_pg = pg_optimize(cur_h, n, dx, T_pg, 800, time.time() + rem() - 10)
            if c5_pg < best_c5:
                best_h, best_c5, best_n = h_pg.copy(), c5_pg, n
            cur_h = h_pg.copy()

    # Phase 3: Polish at best resolution
    if rem() > 15:
        n = best_n
        dx = 2.0 / n
        if n <= 200:
            bounds_n = [(0.0, 1.0)] * n
            cons_n = [{'type': 'eq', 'fun': constr_eq, 'jac': constr_eq_jac, 'args': (n,)}]
            for T in [10.0, 30.0, 60.0]:
                if rem() < 10:
                    break
                try:
                    res = minimize(slsqp_obj, best_h.copy(), jac=slsqp_grad,
                                   args=(n, dx, T), method='SLSQP',
                                   bounds=bounds_n, constraints=cons_n,
                                   options={'maxiter': 300, 'ftol': 1e-14})
                    c5 = exact_c5(res.x)
                    if c5 < best_c5:
                        best_h, best_c5, best_n = res.x.copy(), c5, n
                except Exception:
                    pass
        if rem() > 8:
            h_f, c5_f = pg_optimize(best_h, best_n, 2.0 / best_n, 25.0, 600,
                                    time.time() + rem() - 5)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f

    # Phase 4: Random restarts at moderate n if time permits
    if rem() > 60:
        for _ in range(5):
            if rem() < 20:
                break
            n_r = 200
            dx_r = 2.0 / n_r
            h_r = np.random.rand(n_r)
            h_r = project_feasible(h_r, n_r)
            # Blend with best
            h_r = 0.3 * h_r + 0.7 * upsample(best_h, n_r)
            h_r = project_feasible(h_r, n_r)
            bounds_r = [(0.0, 1.0)] * n_r
            cons_r = [{'type': 'eq', 'fun': constr_eq, 'jac': constr_eq_jac, 'args': (n_r,)}]
            try:
                res = minimize(slsqp_obj, h_r, jac=slsqp_grad,
                               args=(n_r, dx_r, 10.0), method='SLSQP',
                               bounds=bounds_r, constraints=cons_r,
                               options={'maxiter': 200, 'ftol': 1e-12})
                c5 = exact_c5(res.x)
                if c5 < best_c5:
                    best_h, best_c5, best_n = res.x.copy(), c5, n_r
            except Exception:
                pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
