    """Compute the change in correlation array when swapping h[a] 1→0 and h[b] 0→1."""
    delta = np.zeros(2*n - 1)
    # lag l corresponds to index l + (n-1) in the delta array
    
    # Term 1: -(1-h[a+l]) for -a <= l < n-a
    l_min1 = max(-a, -(n-1))
    l_max1 = min(n-1-a, n-1)  # l < n-a means l <= n-a-1
    if l_min1 <= l_max1:
        idx = np.arange(l_min1, l_max1+1) + (n-1)
        vals = l_min1 + np.arange(l_max1 - l_min1 + 1) + a  # a + l values
        delta[idx] -= (1.0 - h[vals])
    
    # Term 2: h[a-l] for a-n < l <= a, i.e., a-n+1 <= l <= a
    l_min2 = max(a-n+1, -(n-1))
    l_max2 = min(a, n-1)

        idx = np.arange(l_min2, l_max2+1) + (n-1)
        vals = a - (l_min2 + np.arange(l_max2 - l_min2 + 1))  # a - l values
        delta[idx] += h[vals]

    l_min3 = max(-b, -(n-1))

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def delta_for_swap(h, a, b, n):
    """Compute change in correlation when h[a]: 1->0, h[b]: 0->1."""
    # Lag l ranges from -(n-1) to (n-1), stored at index l+(n-1)
    # delta[l] = -(1-h[a+l]) + h[a-l] + (1-h[b+l]) - h[b-l]
    #          (terms = 0 if index out of [0,n-1])
    
    # Vectorized computation
    lags = np.arange(-(n-1), n)  # all lags
    
    # Term 1: -(1-h[a+l]) where 0 <= a+l < n
    idx1 = (a + lags >= 0) & (a + lags < n)
    term1 = np.zeros(2*n-1)
    term1[idx1] = -(1.0 - h[a + lags[idx1]])
    
    # Term 2: h[a-l] where 0 <= a-l < n
    idx2 = (a - lags >= 0) & (a - lags < n)
    term2 = np.zeros(2*n-1)
    term2[idx2] = h[a - lags[idx2]]
    
    # Term 3: (1-h[b+l]) where 0 <= b+l < n
    idx3 = (b + lags >= 0) & (b + lags < n)
    term3 = np.zeros(2*n-1)
    term3[idx3] = (1.0 - h[b + lags[idx3]])
    
    # Term 4: -h[b-l] where 0 <= b-l < n
    idx4 = (b - lags >= 0) & (b - lags < n)
    term4 = np.zeros(2*n-1)
    term4[idx4] = -h[b - lags[idx4]]
    
    return term1 + term2 + term3 + term4

def sa_binary(h_init, time_limit=30.0, seed=None):
    """Simulated annealing on binary (or near-binary) vectors."""
    if seed is not None:
        rng = np.random.RandomState(seed)
    else:
        rng = np.random.RandomState()
    
    n = len(h_init)
    h = h_init.copy()
    corr = compute_corr(h)
    c5 = float(np.max(corr) * (2.0/n))
    
    best_h = h.copy()
    best_c5 = c5
    
    # Identify ones and zeros (and halves for odd n)
    T = c5 * 0.1  # initial temperature
    T_min = 1e-6
    cooling = 0.9995
    
    start = time.time()
    it = 0
    
    while time.time() - start < time_limit:
        # Find indices with value > 0.5 and < 0.5
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        delta = delta_for_swap(h, a, b, n)
        new_corr = corr + delta
        
        new_c5 = float(np.max(new_corr) * (2.0/n))
        delta_c5 = new_c5 - c5
        
        if delta_c5 < 0 or rng.random() < np.exp(-max(delta_c5, 0) / max(T, T_min)):
            # Accept
            h[a], h[b] = h[b], h[a]
            corr = new_corr
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            pass  # reject, no change
        
        T *= cooling
        it += 1
    
    return best_h, best_c5
