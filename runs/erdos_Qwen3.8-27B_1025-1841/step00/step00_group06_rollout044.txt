import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def snap_to_binary(h, n):
    if n % 2 == 0:
        order = np.argsort(h)[::-1]
        hb = np.zeros(n)
        hb[order[:n // 2]] = 1.0
        return hb
    else:
        order = np.argsort(h)[::-1]
        hb = np.zeros(n)
        hb[order[:n // 2]] = 1.0
        hb[order[n // 2]] = 0.5
        return hb


def fast_eval_swap(h, i, j, n):
    if h[i] != 1.0 or h[j] != 0.0:
        return None
    h[i] = 0.0
    h[j] = 1.0
    val = compute_c5(h, n)
    h[i] = 1.0
    h[j] = 0.0
    return val


def simulated_annealing(h0, n, t_end):
    h = h0.astype(float).copy()
    c5 = compute_c5(h, n)
    ones = np.where(h >= 1.0)[0].tolist()
    zeros = np.where(h <= 0.0)[0].tolist()
    if not ones or not zeros:
        return h, c5

    T = 0.004
    T_min = 0.0003
    cool = 0.9997
    rng = np.random

    while time.time() < t_end:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[i] = 0.0
        h[j] = 1.0
        nc = compute_c5(h, n)
        d = nc - c5

        if d < 0 or rng.random() < np.exp(-min(d / T, 60.0)):
            c5 = nc
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0

        T = max(T_min, T * cool)

    return h, c5


def greedy_polish(h0, n, t_end):
    h = h0.astype(float).copy()
    c5 = compute_c5(h, n)
    ones = np.where(h >= 1.0)[0].tolist()
    zeros = np.where(h <= 0.0)[0].tolist()
    if not ones or not zeros:
        return h, c5

    improved = True
    while improved and time.time() < t_end:
        improved = False
        best_d = 0.0
        best_i = -1
        best_j = -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nc = compute_c5(h, n)
                h[i] = 1.0
                h[j] = 0.0
                d = nc - c5
                if d < best_d - 1e-12:
                    best_d = d
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5 += best_d
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            improved = True

    return h, c5
