        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    
    lo = v.min() - 1
    hi = v.max() + 1

    grad = np.zeros(n)
    i_lo = max(0, -k)
    i_hi = min(n - 1, n - 1 - k)
