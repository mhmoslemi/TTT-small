def project_h(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        n_mask = np.sum(mask)
        if n_mask == 0:
            break
        step = diff / n_mask
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_out):
    """Upsample h to n_out points"""
    n_in = len(h)
    x_out = np.linspace(0, n_in - 1, n_out)
    h_out = np.interp(x_out, np.arange(n_in), h)
    return project_h(h_out, n_out)
