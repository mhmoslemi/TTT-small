    corr = np.correlate(h, 1-h, mode="full")
    
    # Compute weights for smooth max
    m = np.max(corr)
    exp_shifted = np.exp(corr - m)
    weights = exp_shifted / np.sum(exp_shifted) * alpha
    
    grad = np.zeros(n)

        w = weights[m_idx]
        if w < 1e-12:
            continue

        # j-k >= 0 => j >= k (if k > 0) or always (if k <= 0)
        # j-k < n => j < n+k (if k > 0, this is j < n which is always; if k < 0, j < n+k)
        # (j-k)+k = j, which is always in [0, n-1]
        # So: j >= max(0, k) and j < min(n, n+k)

        j_end = min(n, n + k)  # exclusive
        if j_end > j_start:

        # j >= 0 (always)
        # j < n (always)
        # j+k >= 0 => j >= -k (if k < 0) or always (if k >= 0)
        # j+k < n => j < n-k (if k > 0) or always (if k <= 0)
