import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Adjust for exact sum
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        nf = free.sum()
        if nf > 0:
            proj[free] += diff / nf
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def subgradient_step(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    s = k_max - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        sa = -s
        grad[:n - sa] += (1.0 - h[sa:])
        grad[sa:] -= h[:n - sa]
    return grad * dx

def optimize_h(h_init, n, budget_s, seed=42):
    """Subgradient optimization with momentum"""
    rng = np.random.RandomState(seed)
    target_sum = float(n) / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_val = compute_c5(h)
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.05
    t0 = time.time()
    
    for it in range(50000):
        if time.time() - t0 > budget_s:
            break
        grad = subgradient_step(h, n)
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - lr * momentum
        h = project_feasible(h_new, target_sum)
        val = compute_c5(h)
        if val < best_val - 1e-8:
            best_val = val
            best_h = h.copy()
        # Adaptive learning rate
        if it % 500 == 0:
            lr *= 0.99
    return best_h, best_val

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = h.min() - target_sum - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-13:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    exps = np.exp(np.clip(m * (corr - mc), -50, 50))
    return float(mc + np.log(exps.sum()) / m)

def sm_gradient(h, m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    exps = np.exp(np.clip(m * (corr - mc), -50, 50))
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        if w < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            sa = -s
            grad[:n - sa] += w * (1.0 - h[sa:])
