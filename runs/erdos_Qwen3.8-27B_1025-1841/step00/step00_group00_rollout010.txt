    np.random.seed(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except (NameError, TypeError):
        h_parent = None

    if h_parent is None or len(h_parent) == 0:
        h_parent = np.zeros(45)
        h_parent[::2] = 1.0
        h_parent = project_feasible(h_parent, 45)

    n_p = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    resolutions = sorted(set([n_p, 90, 150, 200, 300, 400, 600]))
    resolutions = [r for r in resolutions if r >= n_p and r != n_p]

    for n_t in resolutions:
        elapsed = time.time() - wall_start
        if elapsed > budget_s - 40:
            break

        x_p = np.linspace(0, 2, n_p)
        x_t = np.linspace(0, 2, n_t)
        h_t = np.interp(x_t, x_p, h_parent)
        h_t = project_feasible(h_t, n_t)

        c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

        rem = budget_s - 40 - elapsed
        t_opt = min(rem * 0.35, 100)

        if t_opt > 5:
            h_opt, c5_opt = pgd_optimize(h_t, t_opt)
            n_opt = len(h_opt)
            if n_opt % 2 == 0:
                hb = try_binary_project(h_opt, n_opt)
                c5_b = compute_c5(hb)
                if c5_b < c5_opt:
                    c5_opt = c5_b
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

        for rs in range(3):
            rem = budget_s - 40 - (time.time() - wall_start)
            if rem < 15:
                break
            h_r = np.random.uniform(0, 1, n_t)
            h_r = project_feasible(h_r, n_t)
            t_r = min(60, rem * 0.25)
            if t_r < 5:
                break
            h_opt, c5_opt = pgd_optimize(h_r, t_r)
            n_opt = len(h_opt)
            if n_opt % 2 == 0:
                hb = try_binary_project(h_opt, n_opt)
                c5_b = compute_c5(hb)
                if c5_b < c5_opt:
                    c5_opt = c5_b
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    rem = budget_s - 5 - (time.time() - wall_start)
    if rem > 5:
        h_fin, c5_fin = pgd_optimize(best_h, rem)
        n_fin = len(h_fin)
        if n_fin % 2 == 0:
            hb = try_binary_project(h_fin, n_fin)
            c5_b = compute_c5(hb)
            if c5_b < c5_fin:
                c5_fin = c5_b
                h_fin = hb
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    n_final = len(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
