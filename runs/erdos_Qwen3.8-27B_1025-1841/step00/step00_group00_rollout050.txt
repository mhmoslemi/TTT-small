import numpy as np
from scipy.optimize import minimize
from time import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)

def enforce_sum(h, n):
    h = h.copy()
    for _ in range(2000):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        c = np.sum(mask)
        if c == 0:
            break
        h[mask] += diff / c
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, beta):
    c = np.maximum(corr, 0.0)
    m = c.max()
    if m < 1e-15:
        return 0.0
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def interpolate_func(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h_init, n, beta, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max(corr, beta)
    def obj_grad(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        c = np.maximum(corr, 0.0)
        m = c.max()
        if m < 1e-15:
            return 0.0, np.zeros(n)
        w = np.exp(beta * (c - m))
        sw = w.sum()
        val = m + np.log(sw) / beta
        grad = np.zeros(n)
        for pos in range(2 * n - 1):
            if w[pos] < 1e-15 * sw:
                continue
            k = pos - (n - 1)
            wpos = w[pos] / sw * (1.0 + beta * max(c[pos] - m, 0)) / (beta * sw)
            # Simplified: gradient of log-sum-exp of c at pos is w[pos]/sw
            wnorm = w[pos] / sw
            if k >= 0:
                lo = 0
                hi = n - 1 - k
            else:
                lo = -k
                hi = n - 1
            for idx in range(lo, hi + 1):
                # c_pos = sum over valid: h[n+k]*(1-h[n]) where the valid range gives us
                # ∂c/∂h[i]: h[i] appears as first factor (if i = n+k, i.e., i-k >= 0 and i < n)
                #           and as second factor (if i = n, i.e., i+k >= 0 and i-k < n... wait)
                # Let me just use: c_k = sum_j h[j+k]*(1-h[j]) for valid j
                # ∂c_k/∂h[i] = (1-h[i-k]) if i-k in [0,n-1] (h[i] in first factor)
                #             - h[i+k] if i+k in [0,n-1] (h[i] in second factor)
                pass
        return val, grad
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_out = enforce_sum(np.clip(res.x, 0, 1), n)
            return h_out, eval_c5(h_out)
    except Exception:
        pass
    return None, None

def local_search(h_init, n, t0, budget_s, rng, max_iters=20000):
    h_curr = h_init.copy()
    best_c5 = eval_c5(h_curr)
    for it in range(max_iters):
        if time() - t0 > budget_s * 0.97:
            break
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(-0.03, 0.03)
        h_try = h_curr.copy()
        h_try[i] += delta
        h_try[j] -= delta
        if h_try[i] < -1e-10 or h_try[i] > 1 + 1e-10 or h_try[j] < -1e-10 or h_try[j] > 1 + 1e-10:
            continue
        h_try[i] = np.clip(h_try[i], 0, 1)
        h_try[j] = np.clip(h_try[j], 0, 1)
        c5_try = eval_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            h_curr = h_try
            best_c5 = c5_try
    return h_curr, best_c5
