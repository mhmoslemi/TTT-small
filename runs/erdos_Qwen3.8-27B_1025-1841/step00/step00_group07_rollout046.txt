    else:

        x_new = np.linspace(0, 2, n_target, endpoint=False)

    h = np.clip(h, 0, 1)
    h += (target - np.sum(h)) / n_target
    h = np.clip(h, 0, 1)

        rng = np.random.default_rng(0)

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _obj(h):
    return _compute_c5(h)

def _cons_fun(h):
    return np.sum(h) - len(h) / 2.0

def _normalize(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def _interp_to(base, n_target):
    n_src = len(base)
    if n_src == n_target:
        return base.copy()
    x_old = np.arange(n_src) / (2.0 * n_src) * 2.0
    x_new = np.arange(n_target) / (2.0 * n_target) * 2.0
    h = np.interp(x_new, x_old, base)
    return _normalize(h, n_target)

def _random_binary(n, rng):
    n_ones = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return h

def _binary_local_search(h_bin, max_rounds=3000, swaps_per_round=80, rng=None):
    n = len(h_bin)
    if rng is None:
        rng = np.random.default_rng(123)
    best = h_bin.astype(float).copy()
    best_max = float(np.max(np.correlate(best, 1.0 - best, mode='full')))
    
    for _ in range(max_rounds):
        improved = False
        ones_pos = np.where(best > 0.5)[0]
        zeros_pos = np.where(best < 0.5)[0]
        n_o = len(ones_pos)
        n_z = len(zeros_pos)
        if n_o == 0 or n_z == 0:
            break
        for _ in range(swaps_per_round):
            i = ones_pos[rng.integers(n_o)]
            j = zeros_pos[rng.integers(n_z)]
            if i == j:
                continue
            best[i] = 0.0
            best[j] = 1.0
            new_max = float(np.max(np.correlate(best, 1.0 - best, mode='full')))
            if new_max < best_max:
                best_max = new_max
                improved = True
                ones_pos = np.where(best > 0.5)[0]
                zeros_pos = np.where(best < 0.5)[0]
                n_o = len(ones_pos)
                n_z = len(zeros_pos)
            else:
                best[i] = 1.0
                best[j] = 0.0
        if not improved:
            break
    return best, best_max * (2.0 / n)

def _slsqp_optimize(h_start, n, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    try:
        res = minimize(_obj, h_start, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = _normalize(res.x, n)
        return h_out
    except Exception:
        return h_start
