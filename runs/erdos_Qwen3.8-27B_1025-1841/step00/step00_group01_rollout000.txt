import numpy as np
from scipy.optimize import minimize
import time

def _evaluate(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    h_new = np.interp(x_new * 2.0, x_old * 2.0, h_old)
    return _feasible(h_new, n_new)

def _obj_func(x):
    n = len(x)
    return _evaluate(x, n)

def _sum_eq(x):
    return x.sum() - 0.5 * len(x)

def _slsqp_jac(x):
    n = len(x)
    return np.zeros(n)

def _run_slsqp(h0, n, maxiter=300, ftol=1e-10):
    h0 = _feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq}]
    res = minimize(_obj_func, h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_new = _feasible(res.x, n)
    return h_new, _evaluate(h_new, n)

def _local_search(h, n, time_limit, rng):
    h = h.copy()
    c5 = _evaluate(h, n)
    t0 = time.time()
    delta = 0.08
    iters = 0
    while time.time() - t0 < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = delta * rng.choice([1, -1])
        if 0.0 <= h[i] + d <= 1.0 and 0.0 <= h[j] - d <= 1.0:
            h[i] += d
            h[j] -= d
            c5_new = _evaluate(h, n)
            if c5_new < c5:
                c5 = c5_new
            else:
                h[i] -= d
                h[j] += d
        iters += 1
        if iters % 500 == 0:
            delta *= 0.92
            delta = max(delta, 0.002)
    return h, c5
