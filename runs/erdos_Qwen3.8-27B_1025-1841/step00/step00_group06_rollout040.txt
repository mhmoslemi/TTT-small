def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-6)[0]
    
    grad = np.zeros(n)

        d = k_idx - (n - 1)  # actual lag
        # For each j, dc/dh[j] = (1-h[j+d]) if 0<=j+d<n - h[j-d] if 0<=j-d<n
        # Vectorize:

        # j+d in [0,n) => j in [-d, n-d) => j in [max(0,-d), min(n, n-d))
        if d >= 0:
            # j+d valid when j < n-d
            j_max = n - d
            grad[:j_max] += (1 - h[d:j_max + d])  # h[j+d] for j in [0, j_max)
            # j-d valid when j >= d

            grad[m:] += (1 - h[:n-m])  # 1-h[j+d] = 1-h[j-m] for j in [m,n)

    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")

            best_h = h_test
            h_bin = h_test
        elif it > 10000:
            temp = max(0.0, 1.0 - it / 50000.0)
            if c5_test < best_c5 + 0.01 * temp and np.random.random() < temp * 0.1:
                h_bin = h_test
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline_total = t0 + budget_s * 0.94

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    h = repair_h(h_init, target_init)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    def remaining():
        return deadline_total - time.time()

    # Phase 1: Gradient + local search at initial n
    h1, c5_1 = gradient_descent(h, target_init, 8000, deadline_total)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1
    h1, c5_1 = local_search(best_h, target_init, 30000, deadline_total, noise=0.07)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Try multiple resolutions
    for mult in [2, 4, 8, 16]:
        if remaining() < 15:
            break
        new_n = n_init * mult
        target_new = new_n / 2.0
        h_hi = upsample_h(best_h, new_n)

        # Gradient at this resolution
        phase_deadline = min(deadline_total, t0 + budget_s * (0.3 + 0.12 * mult ** 0.5))
        h_g, c5_g = gradient_descent(h_hi, target_new, 5000, phase_deadline)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g

        if remaining() < 15:
            continue
        # Local search
        h_l, c5_l = local_search(best_h, target_new, 15000, deadline_total, noise=0.04)
        if c5_l < best_c5:
            best_h, best_c5 = h_l, c5_l

    # Phase 3: Binary optimization at best resolution
    if remaining() > 30:
        bin_deadline = min(deadline_total, t0 + budget_s * 0.85)
        h_b, c5_b = binary_optimize(best_h, bin_deadline)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b

    # Phase 4: Final fractional polish
    if remaining() > 10:
        n_f = len(best_h)
        target_f = n_f / 2.0
        h_f, c5_f = local_search(best_h, target_f, 20000, deadline_total, noise=0.03)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
        h_f, c5_f = gradient_descent(best_h, target_f, 3000, deadline_total)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Final
    best_c5 = compute_c5(best_h)
    return list(best_h), best_c5, len(best_h)
