import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr) * dx)

def compute_corrs(h):
    return np.correlate(h, 1-h, mode='full')

def to_binary(h, n):
    """Convert h to binary with exactly n//2 ones"""
    n = len(h)
    target = n // 2
    # Sort by value, keep top target as 1s
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, max_iters=10000, time_limit=None, time_start=None):
    """Steepest descent local search on binary solutions"""
    target = n // 2
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    it = 0
    while it < max_iters:
        if time_limit and time.time() - time_start > time_limit:
            break
        
        best_move = None
        best_c5_move = best_c5
        eps = 1e-10
        
        # Try all swaps
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h)
                h[i] = 1
                h[j] = 0
                if c5 < best_c5_move - eps:
                    best_c5_move = c5
                    best_move = (i, j)
        
        if best_move is None:
            break  # Local minimum
        
        i, j = best_move

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h):
    n = len(h)
    target = n // 2
    h = np.asarray(h, dtype=float)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -1.0, 1.0
    for _ in range(200):
        lam = (lo + hi) / 2.0
        proj = np.clip(h - lam, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return proj

def interpolate_to_resolution(h, n_new):
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return h_new

def steepest_descent_binary(h, n, max_time, time_start, seed=42):
    h = h.copy()
    corrs = np.correlate(h, 1-h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corrs)) * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    no_improve = 0
    while time.time() - time_start < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        current_c5 = float(np.max(corrs)) * dx
        
        best_move = None
        best_move_c5 = current_c5
        
        # Try all swaps
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_corrs = np.correlate(h, 1-h, mode='full')
                new_c5 = float(np.max(new_corrs)) * dx
                h[i] = 1
                h[j] = 0
                if new_c5 < best_move_c5 - 1e-14:
                    best_move_c5 = new_c5
                    best_move = (i, j)
        
        if best_move is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        i, j = best_move
        h[i] = 0
        h[j] = 1
        corrs = np.correlate(h, 1-h, mode='full')
        no_improve = 0
        
        if best_move_c5 < best_c5:
            best_c5 = best_move_c5
            best_h = h.copy()
    
    return best_h, best_c5

def simulated_annealing_binary(h, n, max_time, time_start, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corrs = np.correlate(h, 1-h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corrs)) * dx
    best_h = h.copy()
    current_c5 = best_c5
    temp = 0.005
    min_temp = 1e-6
    
    while time.time() - time_start < max_time and temp > min_temp:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_corrs = np.correlate(h, 1-h, mode='full')
        new_c5 = float(np.max(new_corrs)) * dx
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / temp):
            current_c5 = new_c5
            corrs = new_corrs
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
            corrs = np.correlate(h, 1-h, mode='full')
        
        temp *= 0.9995
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h):
    n = len(h)
    target = n // 2
    h = np.asarray(h, dtype=float)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
