    corr = np.correlate(h, 1.0 - h, mode='full')

    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

def compute_subgradient(h, n_points, top_k=5):
    n = n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    dx = 2.0 / n
    
    for m in top_idx:
        k = m - (n - 1)  # actual lag
        g = np.zeros(n)
        if k >= 0:
            # c_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[j] = (1-h[j-k])*I[k<=j<=n-1] - h[j+k]*I[0<=j<=n-1-k]
            g[k:] += (1.0 - h[:n-k])    # (1-h[i]) for i=0..n-1-k, placed at j=i+k
            g[:n-k] -= h[k:]           # -h[i+k] for i=0..n-1-k, placed at j=i
        else:
            kp = -k
            # c_{-kp} = sum_{m=0}^{n-1-kp} h[m]*(1-h[m+kp])
            # d/dh[j] = (1-h[j+kp])*I[0<=j<=n-1-kp] - h[j-kp]*I[kp<=j<=n-1]
            g[:n-kp] += (1.0 - h[kp:])   # (1-h[m+kp]) for m=0..n-1-kp
            g[kp:] -= h[:n-kp]           # -h[m] for m=0..n-kp, placed at j=m+kp
        grad += g
    
    grad /= top_k
    return grad, np.max(corr) * dx

import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-14:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        nm = np.sum(mask)
        if nm > 0:
            h_proj[mask] += diff / nm
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_subgradient(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            kp = -k
            g[:n - kp] += (1.0 - h[kp:])
            g[kp:] -= h[:n - kp]
        grad += g
    grad /= top_k
    return grad, np.max(corr) * (2.0 / n)

def subgradient_descent(h, n, n_iters, lr0, target_sum, t_end, cool_rate=0.002):
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    cur = best_obj
    plateaus = 0
    for it in range(n_iters):
        if time.time() > t_end:
            break
        grad, obj = compute_subgradient(h, n, top_k=5)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            plateaus = 0
        else:
            plateaus += 1
        lr = lr0 * max(0.01, 1.0 - cool_rate * it)
        h_new = project_feasible(h - lr * grad, target_sum)
        new_obj = compute_c5(h_new, n)
        if new_obj < obj + 1e-12:
            h = h_new
        else:
            h_new = project_feasible(h - lr * 0.3 * grad, target_sum)
            if compute_c5(h_new, n) < obj:
                h = h_new
        if plateaus > 50000:
            break
    return best_h, best_obj

def perturb_h(h, n, target_sum, strength):
    h_p = h + strength * np.random.normal(0, 1, n)
    return project_feasible(h_p, target_sum)

def swap_perturb(h, n, target_sum, n_swaps):
    h_p = h.copy()
    for _ in range(n_swaps):
        i, j = np.random.choice(n, 2, replace=False)
        h_p[i], h_p[j] = h_p[j], h_p[i]
    return project_feasible(h_p, target_sum)

def random_feasible(n, target_sum):
    h = np.random.uniform(0, 1, n)
    h = project_feasible(h, target_sum)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) > 0:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h_init = None

    if h_init is not None:
        n = len(h_init)
    else:
        n = 95

    target = n / 2.0
    if h_init is not None:
        h_init = project_feasible(h_init, target)
    else:
        h_init = random_feasible(n, target)

    best_h = h_init.copy()
    best_obj = compute_c5(best_h, n)

    # Phase 1: short exploration from multiple starts
    phase1_end = t_start + 0.12 * budget_s
    starts = [best_h.copy()]
    starts.append(project_feasible(1.0 - best_h, target))
    for _ in range(3):
        starts.append(random_feasible(n, target))

    for s in starts:
        if time.time() > phase1_end:

        h_opt, obj = subgradient_descent(s.copy(), n, 150000, 0.02, target, phase1_end)
        if obj < best_obj:
            best_obj = obj
            best_h = h_opt.copy()

    # Phase 2: main optimization with restarts
    phase2_end = t_start + 0.88 * budget_s
    batch_size = 80000
    lr0 = 0.025
    restarts = 0
    h_cur = best_h.copy()
    last_best = best_obj

    while time.time() < phase2_end:

        if obj < best_obj:
            best_obj = obj
            best_h = h_opt.copy()
            last_best = best_obj
            h_cur = best_h.copy()
            restarts = 0
            lr0 = max(0.003, lr0 * 0.85)

                else:
                    h_cur = swap_perturb(best_h, n, target, np.random.randint(1, 6))
                lr0 = max(0.005, lr0 * 0.6)
            else:
                h_cur = best_h.copy()

    # Phase 3: fine-tuning
    t_end3 = min(deadline, t_start + 0.97 * budget_s)

        best_h = h_opt.copy()

    final_obj = compute_c5(best_h, n)
    best_obj = min(best_obj, final_obj)
    best_h = project_feasible(best_h, target)

    return (best_h, best_obj, n)
