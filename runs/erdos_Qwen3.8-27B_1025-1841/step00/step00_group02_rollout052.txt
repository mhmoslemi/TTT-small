import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_objective_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = np.max(corr) * dx
    k_star = np.argmax(corr)
    m = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + m < n:
            grad[j] += (1.0 - h[j + m])
        if 0 <= j - m < n:
            grad[j] -= h[j - m]
    return val, dx * grad

def project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to decrease: reduce largest elements
            h -= diff / len(h)
        else:
            h += (-diff) / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = np.sum(h) - target_sum
    if diff > 0:
        idx = np.argmax(h)
        h[idx] = np.maximum(h[idx] - diff, 0.0)
    elif diff < 0:
        idx = np.argmin(h)
        h[idx] = np.minimum(h[idx] - diff, 1.0)
    return h

    n = len(h)
    # Bisection on threshold
    lo, hi = -max(h) - 1, max(h) + 1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_gradesim(h, tau=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(corr) * dx)
    c = corr - np.max(corr)
    w = np.exp(c / tau)
    w /= w.sum()
    m_vals = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for j in range(n):
        m_lo1, m_hi1 = -j, n - 1 - j
        i1, i2 = m_lo1 + (n - 1), m_hi1 + (n - 1)
        if i1 <= i2:
            grad[j] += np.dot(w[i1:i2 + 1], 1.0 - h[j:m_hi1 + 1])
        m_lo2, m_hi2 = j - (n - 1), j
        i2a, i2b = m_lo2 + (n - 1), m_hi2 + (n - 1)
        if i2a <= i2b:
            grad[j] -= np.dot(w[i2a:i2b + 1], h[0:m_hi2 + 1])
    return dx * grad, val

def project_nthsum(h, target_sum):
    y = h.copy()
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():

            x = np.clip(x, 0.0, 1.0)
