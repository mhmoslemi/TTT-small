import numpy as np
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        h[:] = 0.5
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def fix_binary(h, n, rng):
    h = (h >= 0.5).astype(np.float64)
    target = n // 2
    cnt = int(h.sum())
    if cnt > target:
        idx1 = np.where(h == 1)[0]
        to_zero = rng.choice(idx1, cnt - target, replace=False)
        h[to_zero] = 0
    elif cnt < target:
        idx0 = np.where(h == 0)[0]
        to_one = rng.choice(idx0, target - cnt, replace=False)
        h[to_one] = 1
    return h

def sa_binary(h, n, dx, t_budget, rng, T0=0.5):
    t0 = time.time()
    h_c = fix_binary(h + rng.normal(0, 0.05, n), n, rng)
    best_h = h_c.copy()
    best_c5 = eval_c5(h_c, dx)
    c5_cur = best_c5
    T = T0
    no_imp = 0

    while time.time() - t0 < t_budget:
        ones = np.where(h_c == 1)[0]
        zeros = np.where(h_c == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_c[i] = 0.0
        h_c[j] = 1.0
        c5_new = eval_c5(h_c, dx)

        if c5_new <= c5_cur:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_h = h_c.copy()
                best_c5 = c5_new
                no_imp = 0
            else:
                no_imp += 1
        else:
            d = c5_new - c5_cur
            if T > 1e-12 and rng.random() < np.exp(-d / T):
                c5_cur = c5_new
            else:
                h_c[i] = 1.0
                h_c[j] = 0.0
            no_imp += 1

        if no_imp > 4000:
            n_kick = int(rng.integers(4, 12))
            for _ in range(n_kick):
                o = np.where(h_c == 1)[0]
                z = np.where(h_c == 0)[0]
                if len(o) == 0 or len(z) == 0:
                    break
                h_c[o[rng.integers(len(o))]] = 0.0
                h_c[z[rng.integers(len(z))]] = 1.0
            c5_cur = eval_c5(h_c, dx)
            T = T0 * 0.6
            no_imp = 0
        else:
            T *= 0.9995

    return best_h, best_c5

def cont_ls(h, n, dx, t_budget, rng):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)

    while time.time() - t0 < t_budget:
        new_h = best_h.copy()
        r = rng.random()
        if r < 0.5:
            npert = int(rng.integers(1, 4))
            idx = rng.integers(0, n, npert)
            new_h[idx] += rng.normal(0, 0.07, npert)
        elif r < 0.8:
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                tot = new_h[i] + new_h[j]
                lo, hi = max(0.0, tot - 1.0), min(1.0, tot)
                new_h[i] = rng.uniform(lo, hi)
                new_h[j] = tot - new_h[i]
        else:
            npert = int(rng.integers(2, 5))
            idx = rng.integers(0, n, npert)
            new_h[idx] = rng.integers(0, 2, npert).astype(float)
        new_h = repair(new_h, n)
        c5 = eval_c5(new_h, dx)
        if c5 < best_c5:
            best_h = new_h
            best_c5 = c5
    return best_h, best_c5
