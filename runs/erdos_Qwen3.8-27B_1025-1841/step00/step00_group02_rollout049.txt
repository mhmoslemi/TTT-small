import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max(x, beta=50.0):
    mx = np.max(x)
    if mx < 1e-15:
        return 0.0
    s = np.clip((x - mx) * beta, -60, 60)
    return mx + np.log(np.sum(np.exp(s))) / beta

def upsample(h, n_out):
    ox = np.arange(len(h), dtype=float)
    nx = np.linspace(0, len(h), n_out, endpoint=False)
    return np.interp(nx, ox, h)

def fix_sum(h, target):
    h = h.copy().astype(float)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def binarize(h, n):
    target = n / 2.0
    b = (h >= 0.5).astype(float)
    if b.sum() > target:
        order = np.argsort(h)[::-1]
        for idx in order:
            if b.sum() == target:
                break
            b[idx] = 0.0
    elif b.sum() < target:
        order = np.argsort(h)
        for idx in order:
            if b.sum() == target:
                break
            b[idx] = 1.0
    return b

def binary_local_search(h_bin, max_iters, seed=42, temperature=0.0):
    n = len(h_bin)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    best_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = temperature
    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        io = rng.randint(len(ones))
        iz = rng.randint(len(zeros))
        oi, zi = ones[io], zeros[iz]
        if oi == zi:
            continue
        h[oi] = 0.0
        h[zi] = 1.0
        corr_max = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        delta = corr_max - best_c5
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            best_c5 = min(best_c5, corr_max)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[oi] = 1.0
            h[zi] = 0.0
        if T > 0:
            T *= 0.9995
    return h, best_c5

def slsqp_opt(h0, beta=50, max_iter=200, include_sum_con=True):
    n = len(h0)
    target = n / 2.0

    def obj(u):
        h = np.clip(u, 0, 1)
        return smooth_max(corr_vec(h), beta)

    cons = []
    if include_sum_con:
        cons.append({'type': 'eq', 'fun': lambda u: np.sum(u) - target,
                     'jac': lambda u: np.ones(n)})
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-9})
    h_opt = np.clip(res.x, 0, 1)
    h_opt = fix_sum(h_opt, target)
    return h_opt

def continuous_local_search(h, max_iters, step=0.05, seed=42):
    n = len(h)
    target = n / 2.0
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-1, 1]) * step * rng.uniform(0.2, 1.0)
        hn = h.copy()
        hn[i] += d
        hn[j] -= d
        hn = np.clip(hn, 0, 1)
        if abs(hn.sum() - target) > 1e-9:
            continue
        c5n = compute_c5(hn)
        if c5n < best_c5:
            h, best_c5 = hn, c5n
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def tleft():
        return budget_s - elapsed()

    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    c5_parent = compute_c5(h_parent)

    best_h = h_parent.copy()
    best_c5 = c5_parent
    best_n = n_parent

    def update(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 1: Binarize parent and do binary local search
    if tleft() > 60:
        for n_try in [n_parent if n_parent % 2 == 0 else n_parent + 1]:
            if tleft() < 30:
                break
            h_up = upsample(h_parent, n_try) if n_try != n_parent else h_parent.copy()
            h_bin = binarize(h_up, n_try)
            c5b = compute_c5(h_bin)
            update(h_bin, c5b, n_try)
            for trial in range(8):
                if tleft() < 10:
                    break
                h_bs, c5_bs = binary_local_search(h_bin, max_iters=5000, seed=seed + trial * 7, temperature=0.02)
                update(h_bs, c5_bs, n_try)
                h_bin = h_bs

    # Phase 2: SLSQP on continuous
    if tleft() > 80:
        for n_try in [n_parent, n_parent + 5, 60, 90]:
            if tleft() < 60:
                break
            h_init = upsample(best_h, n_try) if n_try != len(best_h) else best_h.copy()
            h_init = fix_sum(h_init, n_try / 2.0)
            for beta in [30, 80]:
                if tleft() < 40:
                    break
                try:
                    h_opt = slsqp_opt(h_init, beta=beta, max_iter=200)
                    c5 = compute_c5(h_opt)
                    update(h_opt, c5, n_try)
                    h_init = h_opt
                except Exception:
                    pass

    # Phase 3: Binary local search at multiple even resolutions
    for n_try in [44, 50, 56, 64, 72, 80, 90, 100, 120]:
        if tleft() < 50:
            break
        h_up = upsample(best_h, n_try)
        h_bin = binarize(h_up, n_try)
        c5b = compute_c5(h_bin)
        update(h_bin, c5b, n_try)
        for trial in range(4):
            if tleft() < 15:
                break
            h_bs, c5_bs = binary_local_search(h_bin, max_iters=8000, seed=seed + 1000 + trial * 13, temperature=0.015)
            update(h_bs, c5_bs, n_try)
            h_bin = h_bs

    # Phase 4: Continuous refinement on best
    if tleft() > 30:
        h_ref = best_h.copy()
        h_ref = fix_sum(h_ref, best_n / 2.0)
        h_cs, c5_cs = continuous_local_search(h_ref, max_iters=5000, step=0.03, seed=seed + 2000)
        update(h_cs, c5_cs, best_n)

    # Phase 5: More SLSQP attempts if time remaining
    if tleft() > 50:
        n_s = best_n
        h_init = best_h.copy()
        for beta in [20, 50, 100, 200]:
            if tleft() < 30:
                break
            try:
                h_opt = slsqp_opt(h_init, beta=beta, max_iter=300)
                c5 = compute_c5(h_opt)
                update(h_opt, c5, n_s)
                h_init = h_opt
            except Exception:
                pass

    # Phase 6: Final binary search at best n or nearby
    if tleft() > 40:
        n_final = best_n if best_n % 2 == 0 else best_n + 1
        h_bin = binarize(best_h, n_final)
        for trial in range(6):
            if tleft() < 15:
                break
            h_bs, c5_bs = binary_local_search(h_bin, max_iters=10000, seed=seed + 3000 + trial * 17, temperature=0.01)
            update(h_bs, c5_bs, n_final)
            h_bin = h_bs

    # Final validation
    final_h = fix_sum(best_h, best_n / 2.0)
    final_h = np.clip(final_h, 0, 1)
    final_c5 = compute_c5(final_h)
    # Ensure sum is exact
    if abs(final_h.sum() - best_n / 2.0) > 1e-10:
        final_h = fix_sum(final_h, best_n / 2.0)
        final_c5 = compute_c5(final_h)

    return final_h, final_c5, best_n
