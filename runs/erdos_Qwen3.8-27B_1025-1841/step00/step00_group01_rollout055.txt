    return project_feasible(np.interp(xn, xo, h_src), n_new)

def smooth_obj(x, n, eps=0.03):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(c)
    if m < 1e-10:
        return 0.0
    s = (c - m) / eps
    s = np.where(s < 0, s, 0.0)
    s = np.maximum(s, -600.0)
    return float(m + eps * np.log(np.exp(s).sum()))

def slsqp_opt(h0, n, maxiter=200, eps=0.03):
    target = n / 2.0
    def obj(x):
        return smooth_obj(x, n, eps)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-9})
    return project_feasible(res.x, n), evaluate(project_feasible(res.x, n), n)

def swap_search(h, n, n_iter, tb):
    t0 = time.time()
    best = project_feasible(h, n).copy()
    bv = evaluate(best, n)
    
    # Identify "on" and "off" positions
    for it in range(n_iter):
        if time.time() - t0 > tb:
            break
        # Find positions to swap
        on_mask = best > 0.99
        off_mask = best < 0.01
        if not np.any(on_mask) or not np.any(off_mask):
            break
        on_idx = np.where(on_mask)[0]
        off_idx = np.where(off_mask)[0]
        i = on_idx[np.random.randint(len(on_idx))]
        j = off_idx[np.random.randint(len(off_idx))]
        
        # Swap
        best[i], best[j] = best[j], best[i]
        val = evaluate(best, n)
        if val < bv:
            bv = val
        else:
            best[i], best[j] = best[j], best[i]
    
    return best, bv

def perturb_search(h, n, n_iter, step_scale, tb):
    t0 = time.time()
    best = project_feasible(h, n).copy()
    bv = evaluate(best, n)
    target = n / 2.0
    
    for it in range(n_iter):
        if time.time() - t0 > tb:
            break
        h_new = best.copy()
        # Perturb 1 position
        i = np.random.randint(n)
        h_new[i] = np.clip(h_new[i] + np.random.uniform(-step_scale, step_scale), 0, 1)
        # Compensate
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-12:
            # Adjust another position
            j = np.random.randint(n)
            h_new[j] = np.clip(h_new[j] + diff, 0, 1)
            # Second correction
            diff2 = target - np.sum(h_new)
            if abs(diff2) > 1e-12:
                # Spread to all non-bound positions
                if diff2 > 0:
                    mask = h_new < 1.0
                else:
                    mask = h_new > 0.0
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h_new[mask] += diff2 / cnt
                    h_new = np.clip(h_new, 0, 1)
        
        if abs(np.sum(h_new) - target) < n * 1e-10:
            val = evaluate(h_new, n)
            if val < bv:
                best = h_new
                bv = val
    
    return project_feasible(best, n), bv

def make_binary(h, n):
    """Convert to pure binary (or near-binary for odd n)."""
    target = n / 2.0
    h_bin = np.zeros(n)
    if n % 2 == 0:
        h_bin[:n//2] = 1.0
    else:
        h_bin[:(n-1)//2] = 1.0
        h_bin[(n-1)//2] = 0.5
    return project_feasible(h_bin, n)

def random_binary(n, rng):
    """Generate random binary sequence."""
    h = np.zeros(n)
    if n % 2 == 0:
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
    else:
        ones = rng.choice(n, (n-1)//2, replace=False)
        h[ones] = 1.0
        h[~np.isin(np.arange(n), ones)] # remaining are 0
        # Need one 0.5: pick one of the zeros and set to 0.5
        zeros = np.where(h == 0.0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = 0.5
    return project_feasible(h, n)
