A better approach for SLSQP: use a smooth approximation to the max. The log-sum-exp is a smooth upper bound on the max:
smooth_max(x, t) = t * log(sum(exp(x_i/t)))

As t → 0, this approaches max(x_i). For small t, it's a good approximation.

Let me use this:

```python
def smooth_max_obj(x, n, dx, temp):
    hx = np.clip(x, 0, 1)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    # log-sum-exp
    m = np.max(corr)
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx
```

d/dx = (1/t) * sum_i exp((corr_i - m)/t) / sum_j exp((corr_j - m)/t) * d(corr_i)/dx * dx

Which is a weighted average of the gradients of the individual correlation components.

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-14):
            h = h_new
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-13:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)

def smooth_max_obj(x, dx, temp):
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    m = np.max(corr)

        m = n - lo

        grad[:m] -= h[lo:]
    return grad
