    n = len(h_init)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)

    # Phase 1: Optimize at n from initial point with increasing alpha
    best_h, best_c5 = do_optimize(
        h_init, n, [20, 50, 100, 200, 400, 800, 1500], 1000, 50000,
        t0, budget_s, 0.12, best_h, best_c5)

    # Hill climb from best
    if time.time() - t0 < budget_s * 0.22:
        h_hc, c5_hc = hill_climb(best_h, n, 5000, 0.05, t0, budget_s, 0.22)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc

    # Phase 2: Random and structured restarts at same n
    x_grid = np.linspace(0, 2.0, n, endpoint=False)
    structured = [
        project_feasible(0.5 + 0.5 * np.sin(2 * np.pi * x_grid), n),
        project_feasible(0.5 + 0.5 * np.cos(4 * np.pi * x_grid), n),
        project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * x_grid) + 0.2 * np.cos(6 * np.pi * x_grid), n),
    ]
    restarts = structured + [project_feasible(np.random.uniform(0, 1, n), n) for _ in range(20)]
    for r in range(len(restarts)):
        if time.time() - t0 >= budget_s * 0.5:
            break
        h_rand = restarts[r]
        if not (np.all(h_rand >= 0) and np.all(h_rand <= 1) and abs(np.sum(h_rand) - n/2.0) < 0.1):
            continue
        best_h, best_c5 = do_optimize(
            h_rand, n, [30, 100, 300, 600], 500, 20000,
            t0, budget_s, 0.5, best_h, best_c5)

    # Hill climb after restarts
    if time.time() - t0 < budget_s * 0.55:
        h_hc, c5_hc = hill_climb(best_h, n, 3000, 0.03, t0, budget_s, 0.55)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc

    # Phase 3: Multi-resolution (upscale and optimize)
    for factor in [2, 4]:
        if time.time() - t0 >= budget_s * 0.78:
            break
        n_new = n * factor
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_new)
        best_h, best_c5 = do_optimize(
            h_up, n_new, [50, 150, 400, 800], 500, 30000,
            t0, budget_s, 0.78, best_h, best_c5)
        if time.time() - t0 < budget_s * 0.8:
            h_hc, c5_hc = hill_climb(best_h, n_new, 2000, 0.03, t0, budget_s, 0.8)
            if c5_hc < best_c5:
                best_c5 = c5_hc
                best_h = h_hc

    # Phase 4: Final refinement at best resolution
    if time.time() - t0 < budget_s * 0.95:
        n_final = len(best_h)
        best_h, best_c5 = do_optimize(
            best_h, n_final, [1500, 3000, 5000], 500, 30000,
            t0, budget_s, 0.95, best_h, best_c5)

    # Final hill climb
    if time.time() - t0 < budget_s * 0.98:
        h_hc, c5_hc = hill_climb(best_h, len(best_h), 2000, 0.02, t0, budget_s, 0.98)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
