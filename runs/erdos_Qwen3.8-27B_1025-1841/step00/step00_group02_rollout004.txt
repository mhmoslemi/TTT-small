import numpy as np
from scipy import optimize
import time

_curr_n = 45
_curr_t = 50.0
_curr_pen = 800.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    tv = t * vals
    m = np.max(tv)
    return float(m / t + np.log(np.sum(np.exp(tv - m))) / t)

def _penalized_obj(h):
    diff = np.sum(h) - _curr_n * 0.5
    return smooth_obj(h, _curr_t) + _curr_pen * diff * diff

def project_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = n * 0.5 - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_resample(h, n_tgt):
    n_src = len(h)
    if n_src == n_tgt:
        return h.copy()
    x = np.linspace(0.0, 1.0, n_src)
    y = np.linspace(0.0, 1.0, n_tgt)
    return np.interp(y, x, h)

def slsqp_run(h0, n, t, max_iter, rng):
    global _curr_n, _curr_t, _curr_pen
    _curr_n = n
    _curr_t = t
    _curr_pen = 800.0
    bounds = [(0.0, 1.0)] * n
    x0 = project_sum(h0 + rng.normal(0, 0.02, n), n)
    try:
        res = optimize.minimize(_penalized_obj, x0, method='SLSQP',
                                bounds=bounds,
                                options={'maxiter': max_iter, 'ftol': 1e-10, 'iprint': 0})
        if res.x is not None:
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0.copy()

def lfbfgs_run(h0, n, t, max_iter, rng):
    global _curr_n, _curr_t, _curr_pen
    _curr_n = n
    _curr_t = t
    _curr_pen = 1000.0
    bounds = [(0.0, 1.0)] * n
    x0 = project_sum(h0 + rng.normal(0, 0.02, n), n)
    try:
        res = optimize.minimize(_penalized_obj, x0, method='L-BFGS-B',
                                bounds=bounds,
                                options={'maxiter': max_iter, 'ftol': 1e-10, 'gtol': 1e-6, 'disp': False})
        if res.x is not None:
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0.copy()

def local_refine(h, n, max_passes, rng):
    best = h.copy()
    best_val = compute_c5(best)
    deltas = [0.01, -0.01, 0.03, -0.03, 0.05, -0.05, 0.1, -0.1, 0.15, -0.15]
    window = min(25, n // 2)
    for _ in range(max_passes):
        improved = False
        idxs = rng.permutation(n)
        for i in idxs:
            for di in range(1, window + 1):
                j = (i + di) % n
                for d in deltas:
                    h_try = best.copy()
                    h_try[i] += d
                    h_try[j] -= d
                    if 0.0 <= h_try[i] <= 1.0 and 0.0 <= h_try[j] <= 1.0:
                        v = compute_c5(h_try)
                        if v < best_val - 1e-9:
                            best = h_try
                            best_val = v
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_val

    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()

        best_h[:22] = 1.0
        best_h[22] = 0.5

    n0 = len(best_h)
    best_h = project_sum(best_h, n0)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Optimize at parent resolution
    for t in [25, 40, 60, 100, 150]:
        if tl() < 20:
            break

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:

    for nres in res_list:

        h_res = project_sum(interp_resample(best_h, nres), nres)

        # SLSQP for smaller, L-BFGS-B for larger

            if c5d < best_c5:

    # Phase 4: Random restarts at parent resolution

                break

    return best_h, best_c5, len(best_h)

def run(seed=42, budget_s=1300, **kwargs):
    return generic_run(seed=seed, budget_s=budget_s, **kwargs)
