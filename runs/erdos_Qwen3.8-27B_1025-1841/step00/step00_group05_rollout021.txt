                       options={'maxiter': maxiter, 'fatol': 1e-12, 'xatol': 1e-10})
        h_opt = project_to_feasible(res.x, n)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    except Exception:
        return h_start, compute_c5(h_start)

def random_binary_try(h_cont, n, rng, num_tries=20):
    target = n / 2.0
    best_h = h_cont.copy()
    best_c5 = compute_c5(h_cont)
    for _ in range(num_tries):
        # Threshold with random shifts
        t = rng.uniform(0.3, 0.7)
        h_bin = (h_cont + rng.uniform(-0.05, 0.05) > t).astype(float)
        # Adjust sum
        s = np.sum(h_bin)
        if s > target:
            # Remove some 1s
            idx_ones = np.where(h_bin == 1)[0]
            extra = int(s - target)
            if extra > 0 and len(idx_ones) >= extra:
                to_remove = rng.choice(idx_ones, size=min(extra, len(idx_ones)), replace=False)
                h_bin[to_remove] = 0.0
        elif s < target:
            idx_zeros = np.where(h_bin == 0)[0]
            need = int(target - s)
            if need > 0 and len(idx_zeros) >= need:
                to_add = rng.choice(idx_zeros, size=min(need, len(idx_zeros)), replace=False)
                h_bin[to_add] = 1.0
        # Final projection
        h_bin = project_to_feasible(h_bin, n)
        c5 = compute_c5(h_bin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()
    return best_h, best_c5

def coordinate_descent(h_start, n, max_iter=2000, lr=0.05):
    h = h_start.copy()
    target = n / 2.0
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        k_star = np.argmax(corr)
        # Subgradient of f_{k_star} w.r.t. h
        # f_k(h) = sum_i h[i]*(1-h[i+k-(n-1)]) for valid i
        grad = np.zeros(n)
        for i in range(n):
            j1 = i  # index in first array
            j2 = i + k_star - (n - 1)  # index in second array (1-h)
            # df/dh[i] from first factor: (1 - h[j2]) if valid
            if 0 <= j2 < n:
                grad[i] += (1.0 - h[j2])
            # df/dh[i] from second factor: -h[j1'] if i = j1' + k_star - (n-1)
            j1_prime = i - k_star + (n - 1)
            if 0 <= j1_prime < n:
                grad[i] -= h[j1_prime]
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_to_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - start_time
    
    # Get initial solution
    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Phase 1: SLSQP at initial resolution
    h_opt, c5_opt = slsqp_optimize(best_h, n_init, maxiter=500)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt
    
    # Phase 2: Coordinate descent refinement
    h_cd, c5_cd = coordinate_descent(best_h, n_init, max_iter=3000, lr=0.02)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd
    
    # Phase 3: SLSQP again after CD
    h_opt, c5_opt = slsqp_optimize(best_h, n_init, maxiter=500)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt
    
    # Phase 4: Random restarts at initial resolution
    num_restarts = 100
    for i in range(num_restarts):
        if elapsed() > budget_s * 0.45:
            break
        sigma = 0.1 + 0.15 * (i / num_restarts)
        h_pert = best_h + rng.normal(0, sigma, size=n_init)
        h_pert = project_to_feasible(h_pert, n_init)
        c5_pert = compute_c5(h_pert)
        
        if c5_pert < best_c5 + 0.02:
            h_opt, c5_opt = slsqp_optimize(h_pert, n_init, maxiter=150)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Phase 5: Binary solutions
    if elapsed() < budget_s * 0.55:
        h_bin, c5_bin = random_binary_try(best_h, n_init, rng, num_tries=50)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
            # Refine binary with SLSQP
            h_opt, c5_opt = slsqp_optimize(best_h, n_init, maxiter=300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Phase 6: Increase resolution
    for factor in [2, 3, 4]:
        if elapsed() > budget_s * 0.75:
            break
        n_new = n_init * factor
        old_x = np.linspace(0, 1, n_init)
        new_x = np.linspace(0, 1, n_new)
        h_interp = np.interp(new_x, old_x, best_h)
        h_interp = project_to_feasible(h_interp, n_new)
        
        h_opt, c5_opt = slsqp_optimize(h_interp, n_new, maxiter=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        
        # Refine with CD at new resolution
        h_cd, c5_cd = coordinate_descent(best_h, n_new, max_iter=2000, lr=0.01)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    # Phase 7: Final restarts at best resolution
    n_curr = len(best_h)
    count = 0
    while elapsed() < budget_s - 15:
        count += 1
        sigma = 0.05 + 0.1 * (count % 10) / 10.0
        h_pert = best_h + rng.normal(0, sigma, size=n_curr)
        h_pert = project_to_feasible(h_pert, n_curr)
        c5_pert = compute_c5(h_pert)
        
        if c5_pert < best_c5 + 0.01:
            h_opt, c5_opt = slsqp_optimize(h_pert, n_curr, maxiter=100)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Final check with evaluate_erdos_solution if available
    try:
        if 'evaluate_erdos_solution' in globals():
            final_c5 = evaluate_erdos_solution(best_h)
            if final_c5 is not None:
                best_c5 = float(final_c5)
    except Exception:
        pass
    
    return (best_h, best_c5, len(best_h))
