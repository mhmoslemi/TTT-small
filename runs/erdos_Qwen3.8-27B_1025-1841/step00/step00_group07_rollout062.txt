import numpy as np
from scipy.optimize import minimize
import time

_g_n = 80
_g_alpha = 30.0

def _obj(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    a = _g_alpha
    shifted = a * (corr - m)
    return m + (1.0 / a) * np.log(np.sum(np.exp(shifted)))

def _sum_con(h):
    return np.sum(h) - _g_n / 2.0

def _sum_jac(h):
    return np.ones(_g_n)

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _enforce(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    for _ in range(500):
        d = t - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        nm = int(m.sum())
        if nm == 0:
            break
        h[m] += d / nm
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _interp(h, n1, n2):
    if n1 == n2:
        return h.copy()
    x1 = (np.arange(n1) + 0.5) / n1
    x2 = (np.arange(n2) + 0.5) / n2
    return np.interp(x2, x1, h)

def _binarize(h, n, rng):
    h_b = (h > 0.5).astype(float)
    bs = int(h_b.sum())
    target = n // 2
    if bs > target:
        idx = np.where(h_b == 1.0)[0]
        rng.shuffle(idx)
        h_b[idx[:bs - target]] = 0.0
    elif bs < target:
        idx = np.where(h_b == 0.0)[0]
        rng.shuffle(idx)
        h_b[idx[:target - bs]] = 1.0
    return h_b

def _swap_search(h, n, rng, iters=5000):
    h = h.copy()
    best = _c5(h, n)
    for _ in range(iters):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h[i], h[j] = 0.0, 1.0
        c = _c5(h, n)
        if c < best - 1e-12:
            best = c
        else:
            h[i], h[j] = 1.0, 0.0
    return h, best

def _swap_search_greedy(h, n, max_passes=30):
    h = h.copy()
    best = _c5(h, n)
    for _ in range(max_passes):
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = _c5(h, n)

                    break
                else:
                    h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, best

    # Try improving at initial resolution with local search

    resolutions = sorted(set([n_best, 80, 100, 120, 160, 200, 240]))

        h_curr = _interp(h_best, n_best, n_pts)

        _g_alpha = 50.0

    c5_best = _c5(h_best, n_best)
    return h_best, float(c5_best), int(n_best)
