    corr = np.correlate(h, 1.0 - h, mode="full")

    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    
    grad = np.zeros(n)

            continue

        #   Contribution: -h[j+k]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-13:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.arange(n_new, dtype=float) * (n_old / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def smooth_max(corr, temp):
    m = np.max(corr)
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def obj_smooth(x, dx, temp):
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    return smooth_max(corr, temp) * dx

def grad_smooth(h, n, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-16:
            continue
        k = r - (n - 1)
        j_lo = max(0, k)
        j_hi = min(n, n + k)
        if j_lo < j_hi:
            i_idx = np.arange(j_lo, j_hi) - k
            grad[j_lo:j_hi] += w[r] * (1.0 - h[i_idx])
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:
            i2_idx = np.arange(j_lo2, j_hi2) + k
            grad[j_lo2:j_hi2] += w[r] * (-h[i2_idx])
    return grad

def projected_gradient(h, n, temps, time_limit, t_start, rng):
    dx = 2.0 / n
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_f = compute_c5(h_curr, dx)
    v = np.zeros(n)
    momentum = 0.9
    
    for temp in temps:
        alpha = 4.0 / n
        n_iter = 2000 if n <= 200 else 800
        for it in range(n_iter):
            if time.time() - t_start > time_limit:
                return best_h, best_f
            g = grad_smooth(h_curr, n, temp)
            h_proposed = h_curr - alpha * (g + momentum * v)
            h_new = project_target(h_proposed, n)
            v = momentum * v + (h_new - h_curr)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_f - 1e-15:
                best_f = c5_new
                best_h = h_new.copy()
            alpha *= 0.999
            if it % 200 == 199:
                alpha = 4.0 / n
        h_curr = best_h.copy()
    return best_h, best_f

def binary_local_search(h_bin, n, max_rounds=500, rng=None, time_limit=None, t_start=None):
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr)
    zero_idx = np.where(h == 0)[0]
    one_idx = np.where(h == 1)[0]
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        improved = False
        n_candidates = min(300, len(zero_idx) * len(one_idx))
        for _ in range(n_candidates):
            j = zero_idx[rng.randint(len(zero_idx))]
            k = one_idx[rng.randint(len(one_idx))]
            h[j], h[k] = 1, 0
            new_corr = np.correlate(h, 1.0 - h, mode="full")
            new_c5 = np.max(new_corr)
            if new_c5 < c5:
                c5 = new_c5
                zero_idx = np.where(h == 0)[0]
                one_idx = np.where(h == 1)[0]
                improved = True
                break
            else:
                h[j], h[k] = 0, 1
        if not improved:
            break
    return h, c5 * dx

def slsqp_solve(h_init, n, temp, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0
    def obj_fn(x):
        return obj_smooth(x, dx, temp)
    def cons_fun(x):
        return np.sum(x) - target
    slsqp_constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h_init, method='SLSQP', bounds=bnds,
                       constraints=slsqp_constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if np.isfinite(res.fun):
            h_new = project_target(res.x, n)
            return h_new, compute_c5(h_new, dx)
    except Exception:
        pass
    return h_init, compute_c5(h_init, dx)
