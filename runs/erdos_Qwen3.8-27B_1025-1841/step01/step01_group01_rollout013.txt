                grad[s:] += (1.0 - h[:n-s])
            # -h[j+s] for j=0..n-s-1
            if s < n:

                # (1-h[j+s]) for j=0..n-s-1
                grad[:n-s] += (1.0 - h[s:])
                # -h[j-s] for j=s..n-1
                grad[s:] += -h[:n-s]
    grad /= len(active_indices)
    return grad
