    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_c5 = compute_c5(h0, dx0)
    best_h = h0.copy()
    best_n = n0

    deadline = t0 + budget_s * 0.93

    res_list = [n0]
    if 2 * n0 <= 1120:
        res_list.append(2 * n0)

    for n in res_list:
        if time.time() > deadline:
            break

        if n == n0:
            h_curr = h0.copy()
        else:
            h_curr = upsample_h(best_h, n)

        dx = 2.0 / n
        target = n / 2.0
        c5_curr = compute_c5(h_curr, dx)

        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

        # Phase 1: SLSQP with LSE smoothing
        if n <= 560:
            if n <= 200:
                temps = [0.06, 0.02, 0.007]
                maxit_lse = 180
            else:
                temps = [0.04, 0.012]
                maxit_lse = 150
            for temp in temps:
                if time.time() > deadline:
                    break
                bnds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian,
                         'args': (target,)}]
                try:
                    res = minimize(lse_objective, h_curr, args=(n, dx, temp),
                                   method='SLSQP', bounds=bnds, constraints=cons,
                                   options={'maxiter': maxit_lse, 'ftol': 1e-14})
                    if np.isfinite(res.fun):
                        h_new = project_target(res.x, n)
                        c5_new = compute_c5(h_new, dx)
                        if c5_new < c5_curr:
                            h_curr = h_new
                            c5_curr = c5_new
                except Exception:
                    pass

        # Phase 2: Multi-active subgradient with momentum
        n_iter = 12000 if n <= 200 else 8000
        K = min(7, n)
        alpha = 2.5 / n
        v = np.zeros(n)

        for it in range(n_iter):
            if time.time() > deadline:
                break
            corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")

            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            mx = np.max(top_vals)

            w = np.exp(-(mx - top_vals) / (0.04 * mx + 1e-12))
            w /= np.sum(w)

            grad = np.zeros(n)
            for wi, ki in zip(w, top_idx):
                s = ki - (n - 1)
                grad += wi * grad_shift(h_curr, n, s)

            v = 0.82 * v - alpha * grad
            h_new = project_target(h_curr + v, n)
            c5_new = compute_c5(h_new, dx)

            if c5_new < c5_curr - 1e-14:
                h_curr = h_new
                c5_curr = c5_new
            elif it % 400 == 399:
                alpha *= 0.88
                v *= 0.3

        # Phase 3: Nelder-Mead local refinement
        if n <= 280 and time.time() < deadline:
            for restart in range(5):
                if time.time() > deadline:
                    break
                h_pert = project_target(h_curr + rng.randn(n) * 0.025, n)
                try:
                    res = minimize(exact_c5_objective, h_pert, args=(dx,),
                                   method='Nelder-Mead',
                                   options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-13})
                    h_test = project_target(res.x, n)
                    c5_test = compute_c5(h_test, dx)
                    if c5_test < c5_curr:
                        h_curr = h_test
                        c5_curr = c5_test
                except Exception:
                    pass

        # Phase 4: Second SLSQP pass with very low temperature
        if n <= 280 and time.time() < deadline:
            bnds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian,
                     'args': (target,)}]
            try:
                res = minimize(lse_objective, h_curr, args=(n, dx, 0.003),
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-15})
                if np.isfinite(res.fun):
                    h_new = project_target(res.x, n)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < c5_curr:
                        h_curr = h_new
                        c5_curr = c5_new
            except Exception:
                pass

        # Phase 5: Coordinate refinement on strided coordinates
        if time.time() < deadline:
            for _pass in range(4):
                if time.time() > deadline:
                    break
                for i in range(0, n, 3):
                    old = h_curr[i]
                    for delta in [-0.04, -0.015, 0.015, 0.04]:
                        h_test = h_curr.copy()
                        h_test[i] = np.clip(old + delta, 0.0, 1.0)
                        h_test = project_target(h_test, n)
                        c5_test = compute_c5(h_test, dx)
                        if c5_test < c5_curr - 1e-14:
                            h_curr = h_test
                            c5_curr = c5_test
                            break

        # Update global best
        h_curr = project_target(h_curr, n)
        c5_curr = compute_c5(h_curr, dx)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_target(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
