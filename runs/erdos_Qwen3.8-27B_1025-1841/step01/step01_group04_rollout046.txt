import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T)

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = T * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()
    j_i = np.arange(n)[:, None]
    l_i = np.arange(n)[None, :]
    idx_a = j_i - l_i + (n - 1)
    mask_a = (idx_a >= 0) & (idx_a < 2 * n - 1)
    idx_b = l_i - j_i + (n - 1)
    mask_b = (idx_b >= 0) & (idx_b < 2 * n - 1)
    A = np.where(mask_a, w[idx_a], 0.0).dot(1.0 - h)
    B = np.where(mask_b, w[idx_b], 0.0).dot(h)
    return (dx / T) * (A - B)

def project_to_simplex_clip(h):
    target = len(h) / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def backtrack_line_search(h, grad, T, alpha0, max_iter=60):
    n = len(h)
    f0 = softmax_obj(h, T)
    c = 1e-4
    alpha = alpha0
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_to_simplex_clip(h_new)
        f_new = softmax_obj(h_new, T)
        if f_new <= f0 + c * alpha * np.dot(grad, h_new - h):
            return h_new, alpha
        alpha *= 0.5
    return h, alpha

def fancy_minimize(h0, T, max_iter, deadline, alpha0=0.01):
    h = h0.astype(float).copy()
    alpha = alpha0
    for i in range(max_iter):
        if time.time() > deadline:
            break
        grad = compute_grad(h, T)
        h_new, alpha = backtrack_line_search(h, grad, T, alpha)
        if np.array_equal(h_new, h):
            break
        h = h_new
    return h

def fancy_minimize_nesterov(h0, T, max_iter, deadline, alpha0=0.005):
    h = h0.astype(float).copy()
    y = h.copy()
    t_k = 1.0
    alpha = alpha0
    for i in range(max_iter):
        if time.time() > deadline:
            break
        grad = compute_grad(y, T)
        y_new = project_to_simplex_clip(y - alpha * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        h_new = y_new + beta * (y_new - h)
        h_new = project_to_simplex_clip(h_new)
        h, y = h_new, y_new
        t_k = t_new
        alpha *= 0.999
    return h

def slsqp_opt(h0, T, maxiter=300, deadline=None):
    n = len(h0)
    target = n / 2.0
    try:
        res = minimize(softmax_obj, h0, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'ftol': 1e-12, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_simplex_clip(res.x)
    except Exception:
        pass
    return h0
