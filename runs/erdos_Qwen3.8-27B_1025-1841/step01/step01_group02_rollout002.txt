def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:

        else:
            mask = h > 0.0
        cnt = np.sum(mask)

            h[mask] += diff / cnt

    h_new = np.interp(new_x, old_x, h, period=2.0) if False else np.interp(new_x, old_x, h)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T_global * dx * corr
    mc = np.max(scaled)
    val = np.max(corr) * dx + np.log(np.sum(np.exp(scaled - mc))) / T_global
    return val + lambda_global * (np.sum(h) - n/2.0)**2

def soft_max_grad(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T_global * dx * corr
    mc = np.max(scaled)
    exp_s = np.exp(scaled - mc)
    p = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        w = p[k] * dx
        # (1-h[j+m]) for valid j: 0<=j<n, 0<=j+m<n
        js = max(0, -m)
