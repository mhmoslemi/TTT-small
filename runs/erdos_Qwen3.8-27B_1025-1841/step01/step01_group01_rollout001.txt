import numpy as np
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def do_swap(h, i, j, max_delta, rng):
    max_d = min(h[i], 1.0 - h[j], max_delta)
    min_d = max(-h[j], h[i] - 1.0, -max_delta)
    if min_d >= max_d - 1e-14:
        return None
    delta = float(rng.uniform(min_d, max_d))
    h[i] += delta
    h[j] -= delta
    return delta

def undo_swap(h, i, j, delta):
    h[i] -= delta
    h[j] += delta

def fix_sum(h, target):
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(-h)
        for i in idx:
            a = min(1.0 - h[i], diff)
            h[i] += a
            diff -= a
            if abs(diff) < 1e-13:
                break
    else:
        idx = np.argsort(h)

            a = min(h[i], -diff)
