import numpy as np
from scipy.optimize import minimize, nelder_mead
import time

def evaluate_c5(h_vec):
    n = len(h_vec)
    dx = 2.0 / n
    corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
    return (corr * dx).max()

def project_feasible(h_vec, n):
    """Project h_vec onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Binary search on lambda
    lo, hi = -1.0, 1.0  # lambda range
    # h_proj[i] = clip(h[i] - lambda, 0, 1)
    # sum decreases with lambda
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h_vec - mid, 0.0, 1.0)
        s = h_try.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h_vec - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-boundary elements
        free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def interp_to_resolution(h_old, n_old, n_new):
    """Interpolate h from n_old points to n_new points"""
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_old_ext = np.concatenate([x_old, [2.0]])
    h_old_ext = np.concatenate([h_old, [h_old[-1]]])
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old_ext, h_old_ext)
    # Scale to satisfy integral constraint
    h_new = np.clip(h_new, 0.0, 1.0)
    s = h_new.sum()
    if s > 0:
        h_new *= (n_new / 2.0) / s
        h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new)

def projected_gradient_step(h, n, learning_rate, max_iter=5000):
    """Optimize using projected gradient on the active constraint"""
    best_h = h.copy()
    best_f = evaluate_c5(h)
    n_actives = np.zeros(2*n-1)
    
    for it in range(max_iter):
        f_val = evaluate_c5(h)
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        m_active = np.argmax(corr)
        n_actives[m_active] += 1
        
        # Compute gradient of active term
        dx = 2.0 / n
        grad = np.zeros(n)
        j_min = max(0, m_active - n + 1)
        j_max = min(n - 1, m_active - 1)
        for j in range(j_min, j_max + 1):
            i = j - m_active + n - 1
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]
        grad *= dx
        
        # Projected gradient step
        h_new = h - learning_rate * grad
        h_new = project_feasible(h_new, n)
        f_new = evaluate_c5(h_new)
        
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        
        if f_new <= f_val:
            h = h_new
        else:
            # Reduce step size
            learning_rate *= 0.5
            if learning_rate < 1e-10:
                learning_rate = 0.01
                h = h_new  # Accept anyway to make progress
                # Or perturb
                break
        
        if it > 0 and it % 500 == 0:
            # Periodically reset step size
            learning_rate = min(learning_rate, 0.1)
    
    return best_h, best_f
