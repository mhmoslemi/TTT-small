        h2 = project_to_feasible(h2, n2)
        tau_schedule = [(40.0, 0.22, 2.0), (200.0, 0.22, 1.5), (800.0, 0.20, 1.0)]
        for tau, frac, lr0 in tau_schedule:
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            tl = min(rem * frac, 250.0)
            h2 = smooth_max_opt(h2, n2, tau, 15000, tl, lr0, t0)
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            h2, c5_2 = subgrad_opt(h2, n2, 25000, rem * 0.30, 0.003, t0)
            c5_2 = compute_c5(h2)
        else:
            c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: random perturbation restart at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 70:
        n_best = len(best_h)
        h_p = best_h + np.random.randn(n_best) * 0.04
        h_p = project_to_feasible(h_p, n_best)
        h_p = smooth_max_opt(h_p, n_best, 200.0, 8000, rem * 0.25, 1.5, t0)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 4: final subgradient polish
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        h_f, c5_f = subgrad_opt(best_h, len(best_h), 20000, rem * 0.45, 0.002, t0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
