import numpy as np
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1 - h.astype(float), mode='full')
    return np.max(corr) * dx

def local_search(h, n, time_limit):
    """Improve h via local search with simulated annealing."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n)
    
    start_time = time.time()
    n_ones = n // 2
    ones_positions = np.where(h == 1)[0]
    zeros_positions = np.where(h == 0)[0]
    
    temp = 0.05
    cool_factor = 0.999
    
    while time.time() - start_time < time_limit:
        # Random swap
        i_idx = np.random.randint(len(ones_positions))
        j_idx = np.random.randint(len(zeros_positions))
        i = ones_positions[i_idx]
        j = zeros_positions[j_idx]
        
        h[i] = 0
        h[j] = 1
        new_c5 = evaluate_c5(h, n)
        
        delta = new_c5 - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            # Accept the move
            ones_positions[i_idx] = j
            zeros_positions[j_idx] = i
        else:
            # Reject: undo
            h[i] = 1
            h[j] = 0
        
        temp *= cool_factor
    
    return best_h, best_c5
