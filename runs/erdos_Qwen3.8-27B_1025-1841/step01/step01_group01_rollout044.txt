    if s < 1e-10:
        h[:] = 0.5
        return h
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    # Distribute adjustment to non-saturated coordinates
    mask = (h > 0.05) & (h < 0.95)
    nmask = np.sum(mask)
    if nmask > 0:
        h[mask] += diff / nmask
        h = np.clip(h, 0, 1)
        # Second pass
        s2 = np.sum(h)
        diff2 = target - s2
        if abs(diff2) > 1e-8:
            h[:] = 0.5 + (h - 0.5) * (target / max(s2, 1e-10))
            h = np.clip(h, 0, 1)
    else:
        h[:] = target / n
    return h

def smooth_max(vals, T=50.0):
    m = np.max(vals)
    s = np.sum(np.exp((vals - m) * T))
    return m + np.log(s) / T

def obj_smooth(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, T) * dx

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.ones(45) * 0.5
    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def do_slsqp(h_start, n, max_iter=2000, ftol=1e-12):
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]

        def obj(x):
            return compute_c5(x)

        res = minimize(obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        return res.x, res.fun

    def do_smooth_slsqp(h_start, n, T_list=None, max_iter=500):
        """Iteratively optimize with decreasing temperature."""
        if T_list is None:
            T_list = [200, 100, 50, 25, 10, 5]
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        h_cur = h_start.copy()
        for T in T_list:
            def obj(x):
                return obj_smooth(x, T)
            res = minimize(obj, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12})
            h_cur = res.x
        return h_cur, compute_c5(h_cur)

    def coordinate_polish(h, n, n_rounds=50):
        """Coordinate descent polish."""
        target = n / 2.0
        dx = 2.0 / n
        best = compute_c5(h)
        for r in range(n_rounds):
            improved = False
            for i in range(n):
                old = h[i]
                # Minimize over h[i] in [0,1]
                # Project constraint: if we change h[i], adjust others proportionally
                def obj1d(x):
                    h_temp = h.copy()
                    h_temp[i] = x
                    # Simple projection: scale others to maintain sum
                    oth_sum = target - x
                    if oth_sum <= 0 or oth_sum >= n - 1:
                        return 999.0
                    others = np.delete(h_temp, i)
                    s_o = np.sum(others)
                    if s_o < 1e-10:
                        return 999.0
                    h_temp2 = np.empty(n)
                    h_temp2[i] = x
                    h_temp2[:i] = others[:i] * (oth_sum / s_o)
                    h_temp2[i+1:] = others[i:] * (oth_sum / s_o)
                    h_temp2 = np.clip(h_temp2, 0, 1)
                    return compute_c5(h_temp2)

                res = minimize_scalar(obj1d, bounds=(0, 1), method='bounded',
                                     options={'xatol': 1e-9})
                if res.success:
                    new_val = res.x
                    h[i] = new_val
                    # Re-project
                    h = make_feasible(h, n)
                    c5_now = compute_c5(h)
                    if c5_now < best - 1e-12:
                        best = c5_now
                        improved = True
            if not improved:
                break
        return h, best

    # Phase 1: SLSQP from initial
    if remaining() > 120:
        try:
            h_o, c5_o = do_slsqp(h_init, n_init, max_iter=2000)
            h_o = make_feasible(h_o, n_init)
            c5_a = compute_c5(h_o)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_o.copy()
        except Exception:
            pass

    # Phase 2: Smooth SLSQP from best
    if remaining() > 120:
        try:
            h_o, c5_o = do_smooth_slsqp(best_h, n_init, T_list=[100, 50, 20, 10])
            h_o = make_feasible(h_o, n_init)
            c5_a = compute_c5(h_o)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_o.copy()
        except Exception:
            pass

    # Phase 3: Multiple restarts
    n_restarts = 15
    for i in range(n_restarts):
        if remaining() < 40:
            break
        # Generate diverse starting points
        n = n_init
        if i == 0:
            h_r = np.ones(n) * 0.5
        elif i == 1:
            # Block pattern
            h_r = np.zeros(n)
            h_r[::2] = 1.0
        elif i == 2:
            # 3-block
            h_r = np.zeros(n)
            h_r[:n//3] = 1.0
            h_r[2*n//3:] = 1.0
        elif i == 3:
            # Sine structure
            h_r = 0.5 + 0.4*np.sin(np.linspace(0, 2*np.pi, n))
        elif i == 4:
            # Two-frequency
            x_l = np.linspace(0, 2, n)
            h_r = 0.5 + 0.45*np.sin(2*np.pi*x_l) + 0.3*np.sin(4*np.pi*x_l)
        elif i == 5:
            # Random binary-ish
            h_r = (rng.rand(n) > 0.5).astype(float)
        elif i == 6:
            # Smooth random
            from scipy.interpolate import interp1d
            n_pts = 8
            pts = rng.rand(n_pts)
            vals = rng.rand(n_pts)
            h_r = interp1d(np.sort(pts), vals[np.argsort(pts)])(np.linspace(0, 1, n))
        elif i == 7:
            # Perturb best
            h_r = best_h + 0.2 * rng.randn(n)
        elif i == 8:
            # Complementary blocks
            h_r = np.zeros(n)
            for b in range(0, n, max(1, n//6)):
                if (b // max(1, n//6)) % 3 != 0:
                    h_r[b:b+max(1, n//6)] = 1.0
        elif i == 9:
            # Ramp
            h_r = np.linspace(0, 1, n)
            h_r = np.clip(h_r + 0.2, 0, 1)
        else:
            # Random smooth perturbation of best
            h_r = best_h + 0.1 * np.sin(np.linspace(0, np.pi*4, n) + rng.rand()*2*np.pi)

        h_r = make_feasible(h_r, n)

        # Quick SLSQP
        try:
            h_o, c5_o = do_slsqp(h_r, n, max_iter=500)
            h_o = make_feasible(h_o, n)
            c5_a = compute_c5(h_o)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_o.copy()
        except Exception:
            pass

    # Phase 4: Coordinate polish
    if remaining() > 60:
        best_h, best_c5 = coordinate_polish(best_h, n_init, n_rounds=30)

    # Phase 5: Try higher resolution
    if remaining() > 180:
        n2 = n_init * 2  # 90
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n2)
        h_hi = interp1d(x_old, best_h, kind='linear')(x_new)
        h_hi = make_feasible(h_hi, n2)
        try:
            h_o, c5_o = do_slsqp(h_hi, n2, max_iter=2000)
            h_o = make_feasible(h_o, n2)
            c5_a = compute_c5(h_o)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_o.copy()
        except Exception:
            pass
        if remaining() > 90:
            try:
                h_o, c5_o = do_smooth_slsqp(best_h, n2, T_list=[50, 20, 10])
                h_o = make_feasible(h_o, n2)
                c5_a = compute_c5(h_o)
                if c5_a < best_c5:
                    best_c5 = c5_a
                    best_h = h_o.copy()
            except Exception:
                pass
        if remaining() > 30:
            best_h, best_c5 = coordinate_polish(best_h, len(best_h), n_rounds=20)

    # Final: go back to lower resolution if that's better
    final_h = best_h
    final_n = len(best_h)
    final_c5 = compute_c5(final_h)

    # Verify
    final_h = np.clip(final_h, 0, 1)
    target = final_n / 2.0
    s = np.sum(final_h)
    if abs(s - target) > 1e-4:
        final_h = make_feasible(final_h, final_n)
        final_c5 = compute_c5(final_h)

    return final_h, final_c5, final_n
