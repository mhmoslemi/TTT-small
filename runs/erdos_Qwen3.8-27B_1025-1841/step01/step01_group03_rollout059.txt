import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # If still not feasible, adjust
    if np.sum(h) > target:
        # Reduce largest values
        idx = np.argsort(h)[::-1]
        excess = np.sum(h) - target
        for i in idx:
            adj = min(h[i], excess)
            h[i] -= adj
            excess -= adj
            if excess < 1e-10:
                break
    elif np.sum(h) < target:
        # Increase smallest values
        idx = np.argsort(h)
        deficit = target - np.sum(h)
        for i in idx:
            adj = min(1.0 - h[i], deficit)
            h[i] += adj
            deficit -= adj
            if deficit < 1e-10:
                break
    return h

def compute_objective(h, dx, alpha):
    """Smooth max of correlations"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m))) / alpha

def compute_gradient(h, dx, alpha):
    """Gradient of smooth objective"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    exp_vals = np.exp(vals - m)
    total = np.sum(exp_vals)
    probs = exp_vals / total
    
    grad = np.zeros(n)
    for idx in range(len(probs)):
        p = probs[idx]
        if p < 1e-12:
            continue
        k = idx - (n - 1)  # actual shift
        
        # ∂c_k/∂h[j] = (1-h[j-k])*I(valid) - h[j+k]*I(valid)
        # Term 1: (1-h[j-k]) where 0<=j-k<n
        # Term 2: -h[j+k] where 0<=j+k<n
        
        if k >= 0:
            # j-k in [0,n) means j in [k, n-1]
            grad[k:] += p * dx * (1.0 - h[:n-k])
            # j+k in [0,n) means j in [0, n-1-k]
            grad[:n-k] -= p * dx * h[k:]
        else:
            # k < 0, |k| = -k
            m_abs = -k
            # j-k = j+m_abs in [0,n) means j in [0, n-1-m_abs]
            grad[:n-m_abs] += p * dx * (1.0 - h[m_abs:])
            # j+k = j-m_abs in [0,n) means j in [m_abs, n-1]
            grad[m_abs:] -= p * dx * h[:n-m_abs]
    
    return grad / alpha

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            adj = min(1.0 - h[i], diff)
            h[i] += adj
            diff -= adj
            if diff < 1e-12:
                break
    elif diff < 0:
        idx = np.argsort(h)[::-1]
        for i in idx:
            adj = min(h[i], -diff)
            h[i] -= adj
            diff += adj
            if diff > -1e-12:
                break
    return h

def obj_and_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = vals - m
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    f = m + np.log(total) / alpha
    
    probs = exp_shifted / total
    grad = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        p = probs[idx]
        if p < 1e-14:
            continue
        k = idx - (n - 1)
        coeff = p * dx
        if k >= 0:
            if n - k > 0:
                grad[k:] += coeff * (1.0 - h[:n-k])
                grad[:n-k] -= coeff * h[k:]
        else:
            mA = -k
            if n - mA > 0:
                grad[:n-mA] += coeff * (1.0 - h[mA:])
                grad[mA:] -= coeff * h[:n-mA]
    
    return f, grad / alpha

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if diff > 1e-12:
        idx = np.argsort(h)
        for i in idx:
            adj = min(1.0 - h[i], diff)
            h[i] += adj
            diff -= adj
            if diff < 1e-12:
                break
    elif diff < -1e-12:
        idx = np.argsort(h)[::-1]
        for i in idx:
            adj = min(h[i], -diff)
            h[i] -= adj
            diff += adj
            if diff > -1e-12:
                break
    return h


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


class Problem:
    def __init__(self, n, dx, alpha):
        self.n = n
        self.dx = dx
        self.alpha = alpha

    def objective(self, h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = corr * self.dx
        m = np.max(vals)
        return m + np.log(np.sum(np.exp(vals - m))) / self.alpha

    def gradient(self, h):
        h = np.clip(h, 0.0, 1.0)
        n = self.n
        dx = self.dx
        alpha = self.alpha
        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = corr * dx
        m = np.max(vals)
        exp_sh = np.exp(vals - m)
        tot = np.sum(exp_sh)
        probs = exp_sh / tot
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            p = probs[idx]
            if p < 5e-14:
                continue
            k = idx - (n - 1)
            coeff = p * dx
            if k >= 0:
                if n > k:
                    grad[k:] += coeff * (1.0 - h[:n - k])
                    grad[:n - k] -= coeff * h[k:]
            else:
                mA = -k
                if n > mA:
                    grad[:n - mA] += coeff * (1.0 - h[mA:])
                    grad[mA:] -= coeff * h[:n - mA]
        return grad / alpha

    def constraint(self, h):
        return np.sum(h) - self.n / 2.0


def make_constraint(n):
    def con_func(h):
        return np.sum(h) - n / 2.0
    return con_func


def make_jac(state):
    def jac_func(h):
        return state.gradient(h)
    return jac_func


def make_obj(state):
    def obj_func(h):
        return state.objective(h)
    return obj_func


def optimize_h(h0, n, dx, max_time, alphas):
    t_start = time.time()
    h = np.clip(h0.copy(), 0.0, 1.0)
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - t_start > max_time:
            break
        state = Problem(n, dx, alpha)
        con = make_constraint(n)
        res = minimize(
            make_obj(state),
            h.copy(),
            method='SLSQP',
            jac=make_jac(state),
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con}],
            options={'maxiter': 300, 'ftol': 1e-12, 'iprint': 0}
        )
        if res.x is not None:
            h = np.clip(res.x, 0.0, 1.0)
            h = project_feasible(h, n)

    return h

        return time.time() - t0
