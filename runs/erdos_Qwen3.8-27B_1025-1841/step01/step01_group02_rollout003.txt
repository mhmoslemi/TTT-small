import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, N):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = N / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_linear(h, new_N):
    old_N = len(h)
    if new_N == old_N:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_N)
    new_x = np.linspace(0.0, 2.0, new_N)
    return make_feasible(np.interp(new_x, old_x, h), new_N)

def soft_obj_and_grad(h, T):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    gmax = np.max(g)
    exp_g = np.exp(T * (g - gmax))
    s = np.sum(exp_g)
    p = exp_g / s
    obj = gmax + np.log(s) / T
    idx = np.arange(N)
    diff = idx[None, :].astype(np.int64) - idx[:, None].astype(np.int64)
    P1 = p[diff + N - 1]
    P2 = p[-diff + N - 1]
    grad = (P1 @ (1.0 - h) - P2 @ h) * dx
    return obj, grad

        if res.x is not None and np.isfinite(res.x).all():

    return h_start

            best_local = best_c5

                if cand < -0.01 or cand > 1.01:
