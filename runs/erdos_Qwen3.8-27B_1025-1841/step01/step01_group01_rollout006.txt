        diff = target - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            break
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h

def opt_1d_minmax(a, b):
    n_lag = len(a)
    if n_lag == 0:
        return 0.5
    if np.all(a >= -1e-14):
        return 0.0
    if np.all(a <= 1e-14):
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = np.max(a * m1 + b)
        v2 = np.max(a * m2 + b)
        if v1 <= v2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coordinate_sweep(h):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    for i in range(n):
        old = h[i]
        a = np.zeros(2 * n - 1)
        for l in range(-(n - 1), n):
            coeff = 0.0
            if 0 <= i + l < n:
                coeff += 1.0 - h[i + l]
            if 0 <= i - l < n:
                coeff += -h[i - l]
            a[l + n - 1] = coeff
        b = corr - a * old
        new_val = opt_1d_minmax(a, b)
        h[i] = np.clip(new_val, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
    h = repair_sum(h, target)
    c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    return h, c5

def interpolate_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(45, 0.5)

    candidates = []
    for n in [128, 256, 512]:
        h_interp = interpolate_to(h_parent, n)
        h_interp = repair_sum(h_interp, n / 2)
        candidates.append(('parent', h_interp, n))

    candidates.append(('uniform', np.full(256, 0.5), 256))
    for ri in range(3):
        hr = rng.rand(256)
        hr = repair_sum(hr, 128.0)
        candidates.append(('rand', hr, 256))
    hb = np.zeros(256)
    hb[:128] = 1.0
    candidates.append(('binary', hb, 256))

    # Sorted by expected quality: parent high-res first
    candidates.sort(key=lambda x: (x[0] != 'parent', x[2]))

    for name, h0, n in candidates:
        if time.time() - t0 > budget_s - 10:
            break
        time_left = budget_s - (time.time() - t0) - 3
        if time_left < 20:
            break

        h_cur = h0.copy()
        c5_cur = compute_c5(h_cur)
        stall = 0
        best_local = c5_cur

        max_sweeps = int(time_left * 8)
        for sweep in range(max_sweeps):
            if time.time() - t0 > budget_s - 3:
                break
            h_cur, c5_cur = coordinate_sweep(h_cur)
            if c5_cur < best_local - 1e-10:
                best_local = c5_cur
                stall = 0
            else:
                stall += 1
            if stall > 150:
                idx = rng.randint(n)
                h_cur[idx] += rng.uniform(-0.15, 0.15) * n / 512.0
                h_cur = np.clip(h_cur, 0.0, 1.0)
                h_cur = repair_sum(h_cur, n / 2)
                stall = 0
            if c5_cur < best_c5 - 1e-9:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n

    if best_h is None:
        best_h = h_parent.copy()
        best_n = len(h_parent)
        best_c5 = compute_c5(best_h)

    best_h = repair_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    try:
        if 'evaluate_erdos_solution' in globals() and evaluate_erdos_solution is not None:
            _ = evaluate_erdos_solution(best_h)
    except Exception:
        pass

    return best_h, best_c5, best_n
