    grad_corr = np.zeros_like(h)

```python
def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    g = corr * dx
    log_max = np.max(beta * g)
    exp_bg = np.exp(beta * g - log_max)
    weights = exp_bg / np.sum(exp_bg)
    obj = (log_max + np.log(np.sum(exp_bg))) / beta  # = logsumexp(beta*g)/beta
    
    grad = np.zeros(n)
    j_range = np.arange(-(n-1), n)
    for k in range(2*n - 1):
        j = j_range[k]
        wk = weights[k] * dx
        if wk < 1e-15:
            continue
        # First term: (1-h[m+j]) for valid m
        m_lo1 = max(0, -j)
        m_hi1 = min(n-1, n-1-j)
        if m_lo1 <= m_hi1:
            grad[m_lo1:m_hi1+1] += wk * (1 - h[m_lo1+j:m_hi1+j+1])
        # Second term: -h[m-j] for valid m
        m_lo2 = max(0, j)
        m_hi2 = min(n-1, n-1+j)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2+1] -= wk * h[m_lo2-j:m_hi2-j+1]
    
    return obj, grad
```

This is O(n * (2n-1)) = O(n²) worst case but with the early skip (wk < 1e-15), in practice only a subset of k will contribute (those with significant weight). For small beta, all k contribute; for large beta, only a few do.

import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Binary search on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix any rounding
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute to elements not at boundary
        mask = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def smooth_obj_grad(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    g = corr * dx
    max_bg = np.max(beta * g)
    exp_bg = np.exp(beta * g - max_bg)
    sum_exp = np.sum(exp_bg)
    weights = exp_bg / sum_exp
    obj = (max_bg + np.log(sum_exp)) / beta
    
    grad = np.zeros(n)
    j_range = np.arange(-(n-1), n)
    for k in range(2 * n - 1):
        j = j_range[k]
        wk = weights[k] * dx
        if abs(wk) < 1e-15:
            continue

        m_hi1 = min(n - 1, n - 1 - j)

            grad[m_lo1:m_hi1+1] += wk * (1.0 - h[m_lo1+j:m_hi1+j+1])
        m_lo2 = max(0, j)

        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2+1] -= wk * h[m_lo2-j:m_hi2-j+1]

    best_c5 = compute_c5(h)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0, 1)

        return h
