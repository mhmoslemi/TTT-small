import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def softmax_obj_and_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    sum_exp = np.sum(exp_scaled)
    obj = (m + np.log(sum_exp)) / T * dx
    w = exp_scaled / sum_exp
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        wk = w[k]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            j_idx = idx + n - 1 - k
            grad[idx] += wk * (1.0 - h[j_idx])
        lo2 = max(0, n - 1 - k)
        hi2 = min(n - 1, n + k - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            j_idx = idx + k - n + 1
            grad[idx] -= wk * h[j_idx]
    grad *= dx
    return obj, grad

def nesterov_pg(h0, n, T, deadline, lr_init=0.008, mu=0.85, max_steps=8000, warmup=200):
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    obj, g = softmax_obj_and_grad(h, n, T)
    patience = 3000
    last_improve = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        h_k = h - mu * v
        h_k = np.clip(h_k, 0.0, 1.0)
        _, g_k = softmax_obj_and_grad(h_k, n, T)
        lr_k = lr
        if step < warmup:
            lr_k = lr * (step + 1) / warmup
        v = mu * v + lr_k * g_k
        h = h - v
        h = project(h, n)
        if step % 200 == 0:
            c5v = evaluate_c5(h)
            if c5v < best_c5 - 1e-8:
                best_c5 = c5v
                best_h = h.copy()
                last_improve = step
        if step - last_improve > patience:
            break
    c5f = evaluate_c5(h)
    if c5f < best_c5:
        best_c5 = c5f
        best_h = h.copy()
    return best_h, best_c5

def slsqp_polish(h0, n, T=300, maxiter=200):
    target = n / 2.0
    def f(x):
        o, _ = softmax_obj_and_grad(x, n, T)
        return o
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def block_perturb(h, n, seed_rng, block_size=16):

    end = min(n, start + block_size)
    h_new[start:end] += seed_rng.rand(end - start) * 0.1 - 0.05
    return project(h_new, n)

            best_h = h_new.copy()

    # Phase 2: SLSQP polish
    if time.time() < deadline - 200:
        h_pol = slsqp_polish(best_h, n0, T=400, maxiter=150)
        c5_pol = evaluate_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: Upsample to 2x and optimize
    n1 = n0 * 2
    if time.time() < deadline - 150:
        h_up = interp_to(best_h, n0, n1)
        h_up = project(h_up, n1)
        c5_up = evaluate_c5(h_up)
        h_work = h_up
        for T in [100.0, 300.0, 600.0]:
            if time.time() > deadline - 40:
                break
            rem = deadline - 40 - time.time()
            steps = int(min(5000, max(300, rem * 2500)))
            h_work, c5_w = nesterov_pg(h_work, n1, T, deadline - 40,
                                        lr_init=0.004, mu=0.88, max_steps=steps)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_work.copy()
        if time.time() < deadline - 20:
            h_pol1 = slsqp_polish(h_work, n1, T=400, maxiter=80)
            c5_p1 = evaluate_c5(h_pol1)
            if c5_p1 < best_c5:
                best_c5 = c5_p1
                best_h = h_pol1.copy()

    # Phase 4: Diverse restarts at best resolution
    nb = len(best_h)
    for attempt in range(40):
        if time.time() > deadline - 15:
            break
        r = rng.rand()
        if r < 0.3:
            h_pert = best_h + rng.randn(nb) * 0.06
            h_pert = project(h_pert, nb)
        elif r < 0.6:
            h_pert = block_perturb(best_h, nb, rng, block_size=12)
        else:
            h_pert = best_h + rng.randn(nb) * 0.12
            h_pert = project(h_pert, nb)
        T_r = 200.0 + 100.0 * rng.rand()
        rem = deadline - 15 - time.time()
        steps = int(min(2000, max(200, rem * 3000)))
        h_new, c5_new = nesterov_pg(h_pert, nb, T_r, deadline - 15,
                                     lr_init=0.005, mu=0.85, max_steps=steps)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Final SLSQP polish on best
    if time.time() < deadline - 5:
        h_final = slsqp_polish(best_h, len(best_h), T=500, maxiter=100)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
