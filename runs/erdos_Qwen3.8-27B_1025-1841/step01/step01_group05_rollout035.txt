        else:

def compute_c5_smooth(h, beta=50):
    """Smooth approximation of C5 using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft max: (1/beta) * log(sum(exp(beta * corr)))
    max_c = np.max(corr)
    return max_c + np.log(np.sum(np.exp(beta * (corr - max_c)))) / beta

def compute_c5_smooth_grad(h, beta=50):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_c = np.max(corr)
    weights = np.exp(beta * (corr - max_c))
    weights /= np.sum(weights)
    
    # Gradient is weighted average of individual correlation gradients
    grad = np.zeros(n)
    for k in range(len(corr_raw)):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        h_pos = np.zeros(n)

            h_pos[:] = h
            h_neg[:] = h

def compute_c5_smooth(h, beta=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    return float(max_c + np.log(np.sum(np.exp(beta * (corr - max_c)))) / beta)

def compute_c5_smooth_grad(h, beta=20):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    weights = exps / np.sum(exps)
    
    # Gradient: weighted average of per-lag gradients
    grad = np.zeros(n)
    for k in range(len(corr_raw)):
        w = weights[k]

        t_lim = min(150, max(40, remaining() * 0.2))
        h_o, c5_o = sgdm_optimize(h_cur, time_limit=t_lim, lr=0.02)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
        if remaining() > 120 and target_n <= 200:
            h_o, c5_o = slsqp_optimize(best_h, maxiter=50, beta=25.0)
            if c5_o < best_c5:
                best_h, best_c5 = h_o, c5_o

    # Phase 4: Random restarts
    n_best = len(best_h)
    while remaining() > 80:
        h_pert = project_feasible(best_h + np.random.randn(n_best) * 0.08)
        t_lim = min(25, remaining() - 40)
        if t_lim < 10:
            break
        h_o, c5_o = sgdm_optimize(h_pert, time_limit=t_lim, lr=0.03)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Final SLSQP polish
    if remaining() > 40 and len(best_h) <= 400:
        h_o, c5_o = slsqp_optimize(best_h, maxiter=100, beta=40.0)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    return best_h, best_c5, len(best_h)
