def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final correction
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff2 / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def obj_grad_pen(h, T, pen):
    n = len(h)
    n_full = 2 * n - 1
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    sum_exp = np.sum(exp_s)
    log_sum = np.log(sum_exp)
    f_val = (mc + log_sum / T) * dx
    sum_h = np.sum(h)
    f_val += pen * (sum_h - n / 2.0) ** 2
    w = exp_s / sum_exp
    grad_h = np.zeros(n)
    for k in range(n_full):
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        j_lo = max(0, -shift)
        j_hi = min(n - 1, n - 1 - shift)
        if j_lo <= j_hi:
            grad_h[j_lo:j_hi + 1] += wk * (1.0 - h[j_lo + shift:j_hi + shift + 1])
        j_lo2 = max(0, shift)
        j_hi2 = min(n - 1, n - 1 + shift)
        if j_lo2 <= j_hi2:
            grad_h[j_lo2:j_hi2 + 1] -= wk * h[j_lo2 - shift:j_hi2 - shift + 1]
    grad_h *= dx
    grad_h += 2.0 * pen * (sum_h - n / 2.0)
    return f_val, grad_h

def _lbfgs_objective(x):
    return _OBJ_ARGS['fn'](x)

def _lbfgs_gradient(x):
    return _OBJ_ARGS['grad'](x)

_OBJ_ARGS = {}

def _obj_fn(x):
    return obj_grad_pen(x, _OBJ_ARGS['T'], _OBJ_ARGS['pen'])[0]

def _obj_grad(x):
    return obj_grad_pen(x, _OBJ_ARGS['T'], _OBJ_ARGS['pen'])[0]

def _obj_grad2(x):
    return obj_grad_pen(x, _OBJ_ARGS['T'], _OBJ_ARGS['pen'])[1]

                    except Exception:

    n_pts = len(best_h)
    improved = True
    polish_iter = 0
    while improved and polish_iter < 40:
        improved = False
        polish_iter += 1
        if time.time() - start_time > budget_s - 15:
            break
        for j in range(n_pts):
            if time.time() - start_time > budget_s - 15:

            cands = [0.0, 1.0, 0.5,

                     old_val + 0.01, old_val - 0.01,

    final_c5 = compute_c5(best_h)
