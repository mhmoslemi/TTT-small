To minimize c_0, we want to maximize Σ h[j]², which means h should be close to 0 or 1 (polarized).

For small s: c_s = dx * Σ_j h[j](1-h[j+s])
This measures the "misalignment" between h and its shift.

There's a tension: polarizing h (to reduce c_0) tends to increase c_s for small s (because adjacent values are more different), while making h smooth (to reduce c_s) increases c_0.

The optimal balance is somewhere in between, which is why the optimal h takes values in (0,1) rather than just {0,1}.

    return best_h, best_c5


def optimize_curriculum(h, n, tau_schedule, time_limit, lr_base=0.2):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for i, (tau, frac) in enumerate(tau_schedule):
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        tl = min(frac * time_limit, time_limit - elapsed)
        if tl < 10:
            break
        lr = lr_base * (1.0 + 0.3 * (i + 1))
        h_new, c5_new = fista_stage(h, n, tau, tl, lr_init=lr)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def subgrad_polish(h, n, max_time, lr_init=0.01):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.8
    stal
    while time.time() - start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0, j1 = max(0, sh), min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
        j0b, j1b = max(0, -sh), min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + sh:j1b + sh]
        grad *= (2.0 / n)
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 80:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-13:
                break
    return best_h, best_c5


def make_cosine_init(n):
    j = np.arange(n)
    h = 0.5 + 0.5 * np.cos(2.0 * np.pi * j / n)
    return project_to_feasible(h, n)


def make_plateau_init(n, center, width):
    j = np.arange(n)
    x = 2.0 * j / n
    h = 1.0
    too_far = np.abs(x - center) > width
    h[too_far] = 0.0
    return project_to_feasible(h, n)


def make_random_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = rng.uniform(0, 1, n)
    return project_to_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    n_work = 3000

    # Build candidate initializations
    candidates = []

    # Candidate 1: parent interpolated
    h1 = interp_periodic(h_init, n_work)
    h1 = project_to_feasible(h1, n_work)
    candidates.append(h1)

    # Candidate 2: cosine
    h2 = make_cosine_init(n_work)
    candidates.append(h2)

    # Candidate 3: plateau
    h3 = make_plateau_init(n_work, 1.0, 0.5)
    candidates.append(h3)

    # Candidate 4: random
    h4 = make_random_init(n_work, seed)
    candidates.append(h4)

    # Optimize each candidate
    tau_sched = [(60, 0.25), (30, 0.25), (15, 0.2), (8, 0.15), (4, 0.15)]
    n_cand = len(candidates)
    time_per_cand = (budget_s * 0.45) / n_cand

    for i, h_cand in enumerate(candidates):
        if remaining() < 60:
            break
        h_opt, c5_opt = optimize_curriculum(h_cand, n_work, tau_sched, time_per_cand, lr_base=0.15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: refine best at higher resolution
    if remaining() > 120:
        n_hi = 6000
        h_hi = interp_periodic(best_h, n_hi)
        h_hi = project_to_feasible(h_hi, n_hi)
        tau_sched_hi = [(30, 0.4), (15, 0.3), (8, 0.3)]
        tl_hi = remaining() * 0.3
        h_hi, c5_hi = optimize_curriculum(h_hi, n_hi, tau_sched_hi, tl_hi, lr_base=0.1)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()

    # Phase 3: subgradient polish
    if remaining() > 30:
        n_pol = len(best_h)
        h_pol = best_h.copy()
        tl_pol = remaining() * 0.3
        h_pol, c5_pol = subgrad_polish(h_pol, n_pol, tl_pol, lr_init=0.008)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: try subgrad at a different resolution
    if remaining() > 20:
        n_alt = 4500
        h_alt = interp_periodic(best_h, n_alt)
        h_alt = project_to_feasible(h_alt, n_alt)
        tl_alt = remaining() * 0.4
        h_alt, c5_alt = subgrad_polish(h_alt, n_alt, tl_alt, lr_init=0.01)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
