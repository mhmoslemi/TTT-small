import numpy as np
import time

def project_to_feasible(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2} using bisection on threshold."""
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def fft_correlation(h):
    """Compute full correlation of h with (1-h) using FFT."""
    n = len(h)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[:2*n-1]
    return corr

def compute_gradient_fft(h, corr, tau):
