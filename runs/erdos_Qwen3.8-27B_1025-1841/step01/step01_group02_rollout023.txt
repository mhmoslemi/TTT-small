import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # Second pass for exact feasibility
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(h)[::-1]
        else:
            idx = np.argsort(h)
        for i in idx:
            room = (1.0 - h[i]) if diff > 0 else h[i]
            step = min(room, abs(diff))
            h[i] += np.sign(diff) * step
            diff -= np.sign(diff) * step
            if abs(diff) < 1e-10:
                break
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_max_obj_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_shifted = np.exp(T * shifted)
    p = exp_shifted / np.sum(exp_shifted)
    
    obj = (max_corr + np.log(np.sum(exp_shifted)) / T) * dx
    obj += penalty * (np.sum(h) - n / 2.0) ** 2
    
    # Gradient using FFT-based correlation
    P = p
    F_full = np.correlate(P, 1.0 - h, mode='full')
    G_full = np.correlate(P[::-1], h, mode='full')
    grad = dx * (F_full[2*n-2:n-2:-1] - G_full[2*n-2:n-2:-1])
    grad += 2.0 * penalty * (np.sum(h) - n / 2.0)
