def compute_grad_fast(h, p, n):
    """Compute gradient of smooth max objective using FFT.
    h: array of length n
    p: array of length 2n-1 (softmax weights over correlations)
    Returns: grad of length n
    """
    L = 4 * n  # FFT size (zero-padding)
    
    # A[m] = (p ⋆ (1-h))[m] = cross-correlation of p and (1-h) at lag m
    # Using FFT: ifft(FFT(p) * conj(FFT(1-h)))
    P = np.fft.fft(p, L)
    
    one_minus_h = 1.0 - h
    O = np.fft.fft(one_minus_h, L)
    
    A_full = np.real(np.fft.ifft(P * np.conj(O)))
    A = A_full[:n]  # We need A[0], ..., A[n-1]
    
    # B[m] = (p ⋆ h_rev)[n-1-m]
    h_rev = h[::-1]
    H = np.fft.fft(h_rev, L)
    B_full = np.real(np.fft.ifft(P * np.conj(H)))
    B = B_full[:n][::-1]  # B[m] = B_full[n-1-m], so B = B_full[0:n] reversed
    
    return A - B

    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:

        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad_fft(h, tau):
    """Compute smooth max objective and gradient using FFT.
    Returns (objective_value, gradient)
    """
    n = len(h)
    dx = 2.0 / n
    c = compute_corr(h)
    max_c = c.max()
    
    shifted = tau * (c - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    s_exp = exp_c.sum()
    p = exp_c / s_exp
    obj = dx * (max_c + np.log(s_exp) / tau)
    
    # Compute gradient using FFT
    L = 1
    while L < 3 * (2*n - 1) + n:
        L *= 2
    # Actually, for cross-correlation of p (length 2n-1) and h (length n),
    # we need L >= (2n-1) + n - 1 = 3n - 2
    L = 1
    while L < 3 * n:
        L *= 2
    
    P = np.fft.rfft(p, L)
    
    one_minus_h = 1.0 - h
    O = np.fft.rfft(one_minus_h, L)
    A = np.real(np.fft.irfft(P * np.conj(O), L))[:n]
    
    h_rev = h[::-1]
    H = np.fft.rfft(h_rev, L)
    B_full = np.real(np.fft.irfft(P * np.conj(H), L))
    B = B_full[n-1::-1][:n]  # B[m] = B_full[n-1-m]
    
    grad = dx * (A - B)
    
    return obj, grad
