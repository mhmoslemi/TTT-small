```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def next_pow2(n):
    return 1 << (n - 1).bit_length()

def smooth_max_obj_grad_fft(h, tau):
    """Compute smooth max objective and gradient using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s  # softmax weights, length 2n-1
    obj = dx * (max_c + np.log(s) / tau)
    
    # Compute gradient using FFT
    # p is indexed for shifts k = 0 to 2n-2 (corresponding to actual shifts -(n-1) to n-1)
    # p_array[k] = p[k] where k=0 corresponds to shift -(n-1)
    P = p  # length 2n-1
    P_flipped = P[::-1]
    
    n_fft = next_pow2(3 * n)
    H = fft(h, n_fft)
    OH = fft(1.0 - h, n_fft)
    PF = fft(P, n_fft)
    PFF = fft(P_flipped, n_fft)
    
    T1 = np.real(ifft(OH * PF))
    T2 = np.real(ifft(H * PFF))
    
    grad = (T1[n-1:2*n-1] - T2[n-1:2*n-1]) * dx
    return obj, grad

def optimize_fft_adam(h, n, tau, time_limit, initial_lr=0.1):
    """Adam optimizer with smooth max objective."""
    start = time.time()
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_max_obj_grad_fft(h, tau)
    best_h = h.copy()
    
    lr = initial_lr
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    stall = 0
    
    while time.time() - start < time_limit:
        t += 1
        obj, grad = smooth_max_obj_grad_fft(h, tau)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h_new, n)
        
        if stall > 100:
            lr *= 0.7
            stall = 0
            if lr < 1e-10:
                break
        
        h = h_new
    
    return best_h, best_obj

def optimize_subgrad_fft(h, n, time_limit, initial_lr=0.01):
    """Subgradient method with FFT-based evaluation."""
    start = time.time()
    h = project_to_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = initial_lr
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    
    while time.time() - start < time_limit:
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= h[idx + sh]
        grad *= dx
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > 200:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        
        h = h_new
    
    return best_h, best_c5
```

```python
def project_to_feasible(g, n):
    target = n / 2.0
```
