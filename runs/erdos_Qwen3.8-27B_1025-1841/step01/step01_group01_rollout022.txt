    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    """Generate Thue-Morse-like binary string of even length n with exactly n/2 ones."""
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    n_ones = int(np.sum(tm))

        ones_idx = np.where(tm == 1)[0]
        zeros_idx = np.where(tm == 0)[0]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def thue_morse_even(n, rng):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    n_ones = int(np.sum(tm))
    target = n // 2
    if n_ones != target:
        ones_idx = np.where(tm == 1)[0]
        zeros_idx = np.where(tm == 0)[0]
        if n_ones > target:
            flip = rng.choice(ones_idx, n_ones - target, replace=False)
            tm[flip] = 0
        else:
            flip = rng.choice(zeros_idx, target - n_ones, replace=False)
            tm[flip] = 1
    return tm

def sa_optimize(h, n, rng, T_init, T_min, steps, end_time):
    c5 = compute_c5(h)
    T = T_init
    best = c5
    best_h = h.copy()
    for _ in range(steps):
        if time.time() > end_time:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best:
                best = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T = max(T * 0.99999, T_min)
    return best_h, best

def fractional_refine(h, n, rng, T_init, steps, end_time):
    c5 = compute_c5(h)
    T = T_init
    best = c5
    best_h = h.copy()
    for _ in range(steps):
        if time.time() > end_time:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.03, 0.03)
        h[i] += d
        h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d
            h[j] += d
            continue
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best:
                best = c5
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
        T = max(T * 0.9999, 0.00005)
    return best_h, best

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + budget_s * 0.95
    end_time = start_time + budget_s * 0.90

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Use initial_h_values if available
    try:
        h0 = initial_h_values.astype(float).copy()
        n0 = len(h0)
        c5_0 = compute_c5(h0)
        best_c5 = c5_0
        best_h = h0.copy()
        best_n = n0
    except:
        pass

    # Binary optimization at even resolutions
    resolutions = [40, 44, 48, 50, 52, 56, 60, 64, 72, 80, 96, 100, 120, 128, 160, 200]

    for idx, n in enumerate(resolutions):
        if time.time() > end_time:
            break
        remaining = end_time - time.time()
        n_left = len(resolutions) - idx
        time_per_res = remaining / n_left
        # Allocate 60% to Thue-Morse, 40% to random restarts
        tm_time = time_per_res * 0.6
        restart_time = time_per_res * 0.4
        n_restarts = max(3, min(25, int(restart_time / 0.05)))

        # Thue-Morse start

        h, c5 = sa_optimize(tm.copy(), n, rng, 0.005, 0.0001, int(tm_time * 12000), end_time)
        if c5 < best_c5:

            h, c5 = sa_optimize(h_r, n, rng, 0.006, 0.0002,
