import numpy as np
from scipy.optimize import minimize

def project_onto_feasible(x, n):
    """Project x onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = x.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        # Distribute diff among non-clipped elements
        free = (h > 0) & (h < 1)
        if not free.any():
            break
        step = diff / free.sum()
        h[free] += step
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-12:
            break
    return h

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
