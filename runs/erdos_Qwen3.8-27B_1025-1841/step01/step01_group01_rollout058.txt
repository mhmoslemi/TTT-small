from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_onto_set(y, n):
    """Project y onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    # Bisection
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(y - t, 0.0, 1.0)

def smooth_c5_and_grad(h, n, eps):
    """Compute smoothed C5 objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    c_shifted = (corr - c_max) / eps
    exp_vals = np.exp(c_shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    obj = (c_max + eps * np.log(total)) * dx
    
    # Gradient
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = weights[s + n - 1]
        if w < 1e-12:
            continue
        # (1-h[j-s]) for valid j-s
        j_lo = max(0, s)
        j_hi = min(n, s + n)
        if j_hi > j_lo:

        # -h[j+s] for valid j+s
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 + s:j_hi2 + s]
    
    return obj, grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_onto_set(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(y - t, 0.0, 1.0)

def smooth_obj_and_grad(h, n, eps):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = (corr - c_max) / eps
    exp_v = np.exp(shifted)
    total = np.sum(exp_v)
    weights = exp_v / total
    obj = (c_max + eps * np.log(total)) * dx
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = weights[s + n - 1]
        if w < 1e-12:
            continue
        j1_lo, j1_hi = max(0, s), min(n, s + n)
        if j1_hi > j1_lo:
            grad[j1_lo:j1_hi] += w * (1.0 - h[j1_lo - s:j1_hi - s])
        j2_lo, j2_hi = max(0, -s), min(n, n - s)
        if j2_hi > j2_lo:
            grad[j2_lo:j2_hi] -= w * h[j2_lo + s:j2_hi + s]
    return obj, grad * dx

def solve_slsqp(h0, n, eps, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lambda x: smooth_obj_and_grad(x, n, eps)[0],
        h0, jac=lambda x: smooth_obj_and_grad(x, n, eps)[1],
        method='SLSQP', bounds=bounds, constraints=cons,
        maxiter=maxiter, ftol=1e-12)
    return res.x

def subgrad_polish(h0, n, n_iters, step_init):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        s_star = int(np.argmax(corr)) - (n - 1)
        sg = np.zeros(n)
        for j in range(n):
            g = 0.0
            js = j - s_star
            if 0 <= js < n:
                g += 1.0 - h[js]
            jp = j + s_star
            if 0 <= jp < n:
                g -= h[jp]
            sg[j] = g
        sg *= 2.0 / n
        step = step_init / (1.0 + it * 0.005)
        h = project_onto_set(h - step * sg, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def optimize_resolution(h, n, time_limit, seed):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    remaining = time_limit - (time.time() - t0)

        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg
    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
