For the 1D problem: minimize max_m (a_m * x + b_m) for x ∈ [0,1].

This is a convex piecewise linear function. The minimum occurs where the subgradient contains 0, or at a boundary.

A more efficient approach:

2. Find all crossing points of pairs of lines (a_i * x + b_i = a_j * x + b_j)
3. Evaluate the objective at each crossing point
4. Return the best value

But finding all crossing points is O(n²) which is 90K for n=300, and evaluating at each is O(n), so total O(n³) = 27M per coordinate. That's worse!

Alternative: use a smarter 1D optimization. Since the function is convex and piecewise linear, I can use a "bisection on the subgradient" approach:
- The subgradient at x is {sum of a_m for m where a_m*x + b_m = max, plus 0 for others}
- The function is decreasing when the subgradient is negative, increasing when positive
- Binary search for where the subgradient changes sign

This is O(log(precision) * n) per coordinate. Much better!

```python
def coord_best_val_bisect(a, b, lo=0.0, hi=1.0, tol=1e-10):
    # Minimize f(x) = max_m (a_m * x + b_m) for x in [lo, hi]
    # f is convex piecewise linear
    # Find x where 0 is in the subgradient
    
    def subgradient(x):
        vals = a * x + b
        max_val = np.max(vals)
        active = vals >= max_val - 1e-15
        return float(np.sum(a[active]))
    
    # Check boundaries
    f_lo = np.max(a * lo + b)
    f_hi = np.max(a * hi + b)
    
    # If subgradient at lo is >= 0, minimum is at lo
    sg_lo = subgradient(lo)
    if sg_lo >= 0:
        return lo
    # If subgradient at hi is <= 0, minimum is at hi
    sg_hi = subgradient(hi)
    if sg_hi <= 0:
        return hi
    
    # Binary search
    for _ in range(60):
        mid = (lo + hi) / 2
        sg_mid = subgradient(mid)
        if sg_mid > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < tol:
            break
    return (lo + hi) / 2
```

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_func(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def con_func(h):
    return np.sum(h) - len(h) / 2.0

def con_jac(h):
    return np.ones(len(h))

def fix_sum(h, target):
    s = float(np.sum(h))
    if abs(s - target) < 1e-10:
        return h.copy()
    h = h + (target - s) / len(h)
    return np.clip(h, 0, 1)

def coord_best_val(a, b):
    n_lines = len(a)
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        sg = float(np.sum(a[active]))
        if sg > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-12:
            break
    x = (lo + hi) / 2.0
    return float(np.clip(x, 0.0, 1.0))

def coordinate_descent(h, max_iters, start_time, budget_s):
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    h_best = h.copy()
    for it in range(max_iters):
        if budget_s - (time.time() - start_time) < 5:
            break
        c = correlate(h, 1 - h, mode='full')
        for i in range(n):
            if budget_s - (time.time() - start_time) < 2:
                break
            a = np.zeros(2 * n - 1)
            for m in range(-(n - 1), n):
                idx = m + n - 1
                im = i - m
                ip = i + m
                am = 0.0
                if 0 <= im < n:
                    am += (1.0 - h[im])
                if 0 <= ip < n:
                    am -= h[ip]
                a[idx] = am
            b = c - h[i] * a
            new_val = coord_best_val(a, b)
            if abs(new_val - h[i]) > 1e-9:
                h[i] = new_val
        s = float(np.sum(h))
        if abs(s - target) > 1e-8:
            diff = target - s
            if diff > 0:
                room = 1.0 - h
                total = room.sum()
                if total > 1e-10:
                    h = h + diff * room / total
                    h = np.clip(h, 0, 1)
            else:
                room = h.copy()
                total = room.sum()
                if total > 1e-10:
                    h = h + diff * room / total
                    h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()
    return h_best

def try_slsqp(h0, start_time, budget_s, max_iter=500):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_new = np.clip(res.x, 0, 1)
        s = float(np.sum(h_new))
        if abs(s - target) > 1e-6:
            h_new = fix_sum(h_new, target)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h0, compute_c5(h0)
