            grad[kp:] += dx * (1.0 - h[:n-kp])
            grad[:n-kp] -= dx * h[kp:]

    return grad, len(active)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    corr = np.correlate(h, 1.0 - h, mode='full')

def subgradient(h, n, c5_vec):
    """Compute subgradient of max(c5_vec) with respect to h"""
    dx = 2.0 / n
    max_val = np.max(c5_vec)
    # Use a small tolerance to include near-max lags for smoother gradient
    tol = max_val - 1e-9
    active = np.where(c5_vec >= tol)[0]
    
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        if k == 0:
            grad += (1.0 - 2.0 * h)
        elif k > 0:

            grad[k:] -= h[:n-k]
        else:

            grad[kp:] += (1.0 - h[:n-kp])

    dx = 2.0 / n
    max_val = np.max(c5_vec)
    tol = max_val - 1e-9
    active = np.where(c5_vec >= tol)[0]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        if k == 0:
            grad += (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
        else:
            kp = -k
            grad[kp:] += (1.0 - h[:n-kp])
            grad[:n-kp] -= h[kp:]
    grad *= dx / len(active)
    return grad

def smooth_gradient(h, n, beta=15.0):
    dx = 2.0 / n
    c5v = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = c5v - np.max(c5v)
    weights = np.exp(beta * shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for m in range(len(c5v)):
        w = weights[m]
        if w < 1e-8:
            continue
        k = m - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:n-kp])
            grad[:n-kp] -= w * h[kp:]
    grad *= dx
    return grad

def project_feasible(h, n):
    target = n * 0.5
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        free = (p > 1e-9) & (p < 1.0 - 1e-9)
        nf = free.sum()
        if nf > 0:
            p[free] += diff / nf
            np.clip(p, 0.0, 1.0, out=p)
    return p

def grad_descent(h, n, max_iter=5000, lr=0.005, momentum=0.85, time_limit=None):
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    t0 = time.time()
    for it in range(max_iter):
        if time_limit and (time.time() - t0) > time_limit:
            break
        c5v = compute_c5_vec(h, n)
        c5 = np.max(c5v)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 10 == 0:
            g = smooth_gradient(h, n, beta=15.0)
        else:
            g = subgradient(h, n, c5v)
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        if np.max(np.abs(h_new - h)) < 1e-14:
            break
        h = h_new
        if it % 800 < 800:
            lr *= 0.9995
    return best_h, best_c5

def binary_local_search(h, n, max_iters=8000, time_limit=None, seed=42):
    h = h.astype(float).copy()
    h = (h > 0.5).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng_s = np.random.RandomState(seed)
        rng_s.shuffle(idx)
        h[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng_s = np.random.RandomState(seed)
        rng_s.shuffle(idx)
        h[idx[:target - n_ones]] = 1
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed + 7)
    for it in range(max_iters):
        if time_limit and (time.time() - t0) > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        i = ones[i]
        j = zeros[j]
        h[i] = 0
        h[j] = 1
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5

def upscale(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    positions = rng.choice(n, n // 2, replace=False)
    h[positions] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        h0 = random_binary(42, seed)
        n0 = 42

    best_h = h0.copy()
    best_c5 = compute_c5(h0, n0)
    best_n = n0

    def elapsed():
        return time.time() - t_start

    def time_left():
        return budget_s - elapsed() - 5

    # Phase 1: Gradient descent from initial at original resolution
    if time_left() > 30:
        tl = time_left() * 0.25
        h1, c5_1 = grad_descent(h0, n0, max_iter=10000, lr=0.004, momentum=0.85, time_limit=tl)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n0

    # Phase 2: Binary local search at original resolution
    if time_left() > 30:
        tl = time_left() * 0.15
        h2, c5_2 = binary_local_search(h0, n0, max_iters=15000, time_limit=tl, seed=seed)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n0

    # Phase 3: Higher resolutions from best solution
    scales = [2, 3, 4, 5, 6, 8, 10, 14, 20]
    for scale in scales:
        if time_left() < 25:
            break
        n_new = n0 * scale
        if n_new < n0 or n_new > 2000:
            continue
        frac = min(0.2, time_left() / max(budget_s * 0.5, 1))
        tl = time_left() * frac
        h_new = upscale(best_h, best_n, n_new)
        h3, c5_3 = grad_descent(h_new, n_new, max_iter=5000, lr=0.003, momentum=0.85, time_limit=tl)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n_new

    # Phase 4: Random restarts at various resolutions
    n_options = [42, 56, 64, 84, 96, 120, 140, 168, 200, 252]
    for i in range(len(n_options)):
        if time_left() < 20:
            break
        n_try = n_options[i]
        tl = time_left() * 0.2
        h_rand = random_binary(n_try, seed + 2000 + i * 137)
        h4, c5_4 = grad_descent(h_rand, n_try, max_iter=4000, lr=0.004, momentum=0.85, time_limit=tl)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
            best_n = n_try

    # Phase 5: Binary search on best solution's resolution and nearby
    if time_left() > 20:
        tl = time_left() * 0.2
        h5, c5_5 = binary_local_search(best_h, best_n, max_iters=10000, time_limit=tl, seed=seed + 999)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Final polishing with very small lr
    if time_left() > 15:
        tl = time_left() * 0.5
        h6, c5_6 = grad_descent(best_h, best_n, max_iter=5000, lr=0.0005, momentum=0.7, time_limit=tl)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
