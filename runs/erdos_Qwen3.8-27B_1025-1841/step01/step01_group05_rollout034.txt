        return np.sum(np.clip(p - lam, 0, 1))

        s = sum_clipped(mid)
        if s > target:

        else:

    return np.clip(p - lam, 0, 1)

import time

def project_feasible(p, n_points):
    target = n_points / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(p - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(p - lam, 0, 1)

    c = np.correlate(h, 1 - h, mode='full')

    c_max = c.max()

            if n - d > 0:

            best_h = h_new.copy()

    dx = 2.0 / n

    c_max = c.max()
    # Softmax weights
    z = (c - c_max) / mu

            hi = mid

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_obj(x, n, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    return float(c_max + mu * np.log(np.sum(np.exp((c - c_max) / mu))))

def optimize_slsqp(h, n, time_limit, start_time):
    """Optimize using SLSQP with smooth max objective."""
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    
    for mu in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if time.time() - start_time > time_limit:
            break
        res = minimize(smooth_obj, h, args=(n, mu), method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 300, 'ftol': 1e-10})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5 = compute_c5(h_new, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
            h = h_new
    
    return best_h, best_c5

def lag_grad(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:] += (1.0 - h[:n - lag])
        if n - lag > 0:
            g[:n - lag] -= h[lag:]
    else:
        d = -lag
        if n - d > 0:
            g[:n - d] += (1.0 - h[d:])
        if d < n:
            g[d:] -= h[:n - d]
    return g

def optimize_pg(h, n, time_limit, start_time):
    """Projected gradient with smooth max."""
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    
    mus = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    for mu in mus:
        if time.time() - start_time > time_limit:
            break
        alpha = 1.0 / n
        for _ in range(10000):
            if time.time() - start_time > time_limit:
                break
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            c_max = c.max()
            z = (c - c_max) / mu
            e = np.exp(z - z.max())
            w = e / e.sum()
            
            grad = np.zeros(n)
            sig = np.where(w > 1e-10)[0]
            for k in sig:
                lag = k - (n - 1)
                grad += w[k] * lag_grad(h, n, lag)
            grad *= dx
            
            h = project_feasible(h - alpha * grad, n)
            c5 = compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5
