def make_feasible(h, n):

    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n

If λ is very positive, h - λ is very negative, so clip gives all 0s, sum = 0.

def smart_binary_search(h, max_passes=50):
    n = len(h)
    h = (h >= 0.5).astype(float)
    h = project_feasible(h)
    best_c5 = compute_c5(h)
    
    for _ in range(max_passes):
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        active_k = np.argmax(corr)
        offset = active_k - (n - 1)
        
        # Find swaps that reduce corr[active_k]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_swap = None
        best_delta = 0
        
        for i in ones:
            for j in zeros:
                # Compute change in corr[active_k]
                delta = 0
                # Term: -(1-h[i+offset]) if valid
                if 0 <= i + offset < n:
                    delta -= (1 - h[i + offset])
                # Term: +h[j+offset] if valid
                if 0 <= j + offset < n:
                    delta += h[j + offset]
                # Term: +h[i-offset] if valid
                if 0 <= i - offset < n:
                    delta += h[i - offset]
                # Term: -(h[j-offset]) if valid
                if 0 <= j - offset < n:
                    delta -= h[j - offset]
                
                if delta < -best_delta:
                    best_delta = -delta
                    best_swap = (i, j)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    return m + np.logaddexp.reduce(T * shifted) / T  # More numerically stable

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = T * (corr - m)
    exps = np.exp(shifted)
    S = np.sum(exps)
    weights = exps / S
    
    m_idx = n - 1
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - m_idx
        
        j_lo = max(0, -offset)
        j_hi = min(n - 1, n - 1 - offset)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * dx * (1.0 - h[j_lo + offset:j_hi + offset + 1])
        
        j_lo2 = max(0, offset)
        j_hi2 = min(n - 1, n - 1 + offset)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= w * dx * h[j_lo2 - offset:j_hi2 - offset + 1]
    
    return grad

def projected_gradient(h, T, max_iters, lr, time_limit):
    n = len(h)
    h = h.copy()
    f_prev = smooth_max_obj(h, T)
    
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        grad = smooth_max_grad(h, T)
        h_trial = project_feasible(h - lr * grad)
        f_trial = smooth_max_obj(h_trial, T)
        
        if f_trial < f_prev - 1e-13:
            h = h_trial
            f_prev = f_trial
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
            h_trial = project_feasible(h - lr * grad)
            f_trial = smooth_max_obj(h_trial, T)
            if f_trial < f_prev - 1e-13:
                h = h_trial
                f_prev = f_trial
            else:
                lr *= 0.5
    
    return h

def binary_local_search(h, max_time):
    n = len(h)
    h = (h >= 0.5).astype(float)
    h = project_feasible(h)
    best_c5 = compute_c5(h)
    
    dx = 2.0 / n
    m_idx = n - 1
    
    for _ in range(100):
        if time.time() > max_time:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        active_k = np.argmax(corr)
        offset = active_k - m_idx
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_delta = 0
        best_swap = None
        
        for i in ones:
            # Pre-compute contributions for this i
            term_i1 = 0
            if 0 <= i + offset < n:
                term_i1 = (1 - h[i + offset])
            term_i2 = 0
            if 0 <= i - offset < n:
                term_i2 = h[i - offset]
            
            for j in zeros:
                term_j1 = 0
                if 0 <= j + offset < n:
                    term_j1 = h[j + offset]
                term_j2 = 0
                if 0 <= j - offset < n:
                    term_j2 = h[j - offset]
                
                delta = -term_i1 + term_j1 + term_i2 - term_j2
                if delta < -best_delta:
                    best_delta = -delta
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
        else:
            h[i] = 1.0
            h[j] = 0.0
            break
    
    return h, best_c5
