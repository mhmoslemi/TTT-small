    dx = 2.0 / n

    mc = np.max(corr)
    exp_vals = np.exp(T * (corr - mc))
    se = np.sum(exp_vals)
    return (mc + np.log(se) / T) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_grad(h, T, lam=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(T * shifted)
    se = np.sum(exp_vals)
    val = (mc + np.log(se) / T) * dx
    # Penalty for sum constraint
    s = np.sum(h) - n / 2.0
    val += lam * s * s
    w = exp_vals / se
    conv1 = np.convolve(1.0 - h, w, 'full')
    conv2 = np.convolve(h, w[::-1], 'full')
    term1 = conv1[n - 1:2 * n - 1]
    term2 = conv2[n - 1:2 * n - 1]
    grad = dx * (term1 - term2)
    grad += 2.0 * lam * s
    return val, grad

def random_binary_h(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    n_ones = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n_ones:
        to_flip = rng.choice(ones, len(ones) - n_ones, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < n_ones:
        to_flip = rng.choice(zeros, n_ones - len(ones), replace=False)
        h[to_flip] = 1.0
    return h

    best_c5 = float('inf')

    best_c5 = compute_c5(best_h)

    resolutions = [len(h_init), 480, 640, 800]

        if target_n == len(best_h):

                v, _ = smooth_obj_grad(x, T)
                return v

                _, g = smooth_obj_grad(x, T)
                return g

            bounds = [(0.0, 1.0)] * target_n
