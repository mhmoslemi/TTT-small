    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        adj = diff / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-15):
            h = h_new
            break
        h = h_new

    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-13:
        idx = np.where((h < 1.0 - 1e-12) if rem > 0 else (h > 1e-12))[0]

            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)

    return grad

def lse_obj_and_grad(h, n, temp, pen_w, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_v = np.exp((corr - m) / temp)
    p = exp_v / np.sum(exp_v)
    lse_val = m + temp * np.log(np.sum(exp_v))
    f = lse_val * dx + pen_w * (np.sum(h) - target) ** 2
    n_c = len(corr)
    grad = np.zeros(n)
    for k in range(n_c):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            mm = n - s
            grad[s:s+mm] += pk * (1.0 - h[:mm])
            grad[:mm] -= pk * h[s:s+mm]
        else:
            lo = -s
            mm = n - lo
            grad[lo:lo+mm] += pk * (1.0 - h[lo:])
            grad[:mm] -= pk * h[lo:]
    grad *= dx
    grad += 2.0 * pen_w * (np.sum(h) - target)
    return f, grad

def coord_cd(h, n, n_grid=51, passes=2):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_c = len(corr)
    s_arr = np.arange(n_c) - (n - 1)
    t_coarse = np.linspace(0, 1, n_grid)
    for _ in range(passes):
        for i in range(n):
            a_k = np.zeros(n_c)
            for k in range(n_c):
                s = s_arr[k]
                c = 0.0
                j1 = i + s
                if 0 <= j1 < n:
                    c += (1.0 - h[j1])
                j2 = i - s
                if 0 <= j2 < n:
                    c += (-h[j2])
                a_k[k] = c
            base_k = corr - a_k * h[i]
            c5_v = np.max(np.outer(t_coarse, a_k) + base_k, axis=1)
            bj = int(np.argmin(c5_v))
            tb = t_coarse[bj]
            if 0 < bj < n_grid - 1:
                tl, th = t_coarse[bj-1], t_coarse[bj+1]
                tf = np.linspace(tl, th, 21)
                c5_f = np.max(np.outer(tf, a_k) + base_k, axis=1)
                tb = tf[int(np.argmin(c5_f))]
            if abs(tb - h[i]) > 1e-12:
                h[i] = tb
                corr = base_k + a_k * tb
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
    return h

def subgrad_loop(h, n, max_iters, t_deadline):
    dx = 2.0 / n
    c5_curr = compute_c5(h, n)
    alpha = 1.5 / np.sqrt(n)
    alpha_max = 3.0 / np.sqrt(n)
    pat = 0
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = get_subgrad(h, n, int(np.argmax(corr)))
        h_new = project_feasible(h - alpha * g, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < c5_curr - 1e-13:
            h, c5_curr = h_new, c5_new
            pat = 0
            alpha = min(alpha * 1.015, alpha_max)
        else:
            pat += 1
            if pat % 80 == 0:
                alpha *= 0.92
            if pat > 500:
                h = project_feasible(h + np.random.randn(n) * 0.02, n)
                c5_curr = compute_c5(h, n)
                alpha, pat = 1.5 / np.sqrt(n), 0
    return h, c5_curr

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s * 0.90

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_c5 = compute_c5(h0, n0)
    best_h = h0.copy()
    best_n = n0

    n_list = sorted(set([n0, n0*2, n0*3, n0*4]))
    n_list = [n for n in n_list if n <= 600]

    for n in n_list:
        if time.time() > deadline:
            break
        dx = 2.0 / n
        target = n / 2.0

        h_curr = h0.copy() if n == n0 else upsample(best_h, n)
        h_curr = project_feasible(h_curr, n)
        c5_curr = compute_c5(h_curr, n)

        # L-BFGS-B with analytical LSE gradient
        for temp in [0.05, 0.01, 0.002]:
            if time.time() > deadline:
                break
            for pen_w in [100.0, 500.0]:
                if time.time() > deadline:
                    break
                def obj(x, t=temp, p=pen_w):
                    return lse_obj_and_grad(x, n, t, p, target)[0]
                def grad(x, t=temp, p=pen_w):
                    return lse_obj_and_grad(x, n, t, p, target)[1]
                try:
                    res = minimize(obj, h_curr, jac=grad, method='L-BFGS-B',
                                   bounds=[(0.0, 1.0)]*n,
                                   options={'maxiter': 300, 'ftol': 1e-15,
                                            'gtol': 1e-10, 'maxfun': 8000})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new, n)
                    if c5_new < c5_curr:
                        h_curr, c5_curr = h_new, c5_new
                except Exception:
                    pass

        # Subgradient descent
        n_it = 3000 if n <= 200 else 2000
        h_curr, c5_curr = subgrad_loop(h_curr, n, n_it, deadline)

        # Coordinate descent refinement
        if time.time() < deadline:
            for _ in range(3):
                if time.time() > deadline:
                    break
                h_curr = coord_cd(h_curr, n, passes=2)
                c5_curr = compute_c5(h_curr, n)

        # One more short L-BFGS polish
        if time.time() < deadline - 10:
            try:
                def obj2(x, t=0.001, p=1000.0):
                    return lse_obj_and_grad(x, n, t, p, target)[0]
                def grad2(x, t=0.001, p=1000.0):
                    return lse_obj_and_grad(x, n, t, p, target)[1]
                res = minimize(obj2, h_curr, jac=grad2, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)]*n,
                               options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-11})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new, n)
                if c5_new < c5_curr:
                    h_curr, c5_curr = h_new, c5_new
            except Exception:
                pass

        # Final coord pass
        if time.time() < deadline - 5:
            h_curr = coord_cd(h_curr, n, passes=1)
            c5_curr = compute_c5(h_curr, n)

        h_curr = project_feasible(h_curr, n)
        c5_curr = compute_c5(h_curr, n)
        if c5_curr < best_c5:
            best_c5, best_h, best_n = c5_curr, h_curr.copy(), n

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
