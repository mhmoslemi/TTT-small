    for _ in range(50):
        h = h * (target / max(s, 1e-12))

        if diff > 0:
            room = 1.0 - h
            idx = np.argmax(room)

            idx = np.argmax(room)
            h[idx] = max(0.0, h[idx] - (-diff))

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add

                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break

def compute_gradient_smooth(h, n, M=50):
    """Compute gradient of log-sum-exp approximation of max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Compute weights
    mc = M * corr
    mc -= mc.max()  # for numerical stability
    w = np.exp(mc)
    w /= w.sum()
    
    # Compute gradient as weighted sum of subgradients
    n_corr = len(corr)  # 2n-1
    grad = np.zeros(n)
    for k in range(n_corr):
        if w[k] < 1e-10:
            continue
        shift = k - n + 1
        lo = max(0, shift)
        hi = min(n, n + shift)
        grad[lo:hi] += w[k] * (1.0 - h[lo-shift:hi-shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        grad[lo2:hi2] -= w[k] * h[lo2+shift:hi2+shift]
    
    return grad * dx  # Include the dx factor
