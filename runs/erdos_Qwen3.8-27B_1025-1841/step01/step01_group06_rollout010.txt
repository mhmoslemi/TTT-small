```python
def smooth_obj_grad(h, dx, temp):
    """Compute value and gradient of smooth max objective."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temp)
    weights = exp_vals / np.sum(exp_vals)
    val = (m + temp * np.log(np.sum(exp_vals))) * dx
    
    # Gradient: dx * sum_k weights[k] * grad_k
    grad = np.zeros(len(h))
    n = len(h)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        s = k_idx - (n - 1)  # shift
        if s >= 0:
            m_len = n - s
            grad[:m_len] += weights[k_idx] * (1.0 - h[s:s+m_len])
            grad[s:s+m_len] -= weights[k_idx] * h[:m_len]
        else:
            lo = -s
            m_len = n - lo
            grad[lo:lo+m_len] += weights[k_idx] * (1.0 - h[lo:])
            grad[:m_len] -= weights[k_idx] * h[lo:]
    grad *= dx
    return val, grad
```

def _correlate_and_smooth(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temp)
    sum_exp = np.sum(exp_vals)
    val = (m + temp * np.log(sum_exp)) * dx
    weights = exp_vals / sum_exp
