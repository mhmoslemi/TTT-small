import numpy as np
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return np.max(compute_corr(h))

def project_feasible(h):
    n = len(h)
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_corr(h, m):
    n = len(h)
    idx = np.arange(n)
    g = np.zeros(n)
    jm = idx + m
    v = (jm >= 0) & (jm < n)
    g[v] += 1.0 - h[jm[v]]
    jm2 = idx - m
    v2 = (jm2 >= 0) & (jm2 < n)
    g[jm2[v2]] -= h[idx[v2]]
    return g

def upsample(h_old, n_new):
    n_old = len(h_old)
    idx = np.minimum(((np.arange(n_new) + 0.5) * n_old / n_new).astype(int), n_old - 1)
    return h_old[idx].astype(float).copy()

def opt_continuous(h_init, deadline, rng):
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy())
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 10.0
    patience = 0
    max_patience = 200

    while time.time() < deadline:
        corr = compute_corr(h)
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                h = project_feasible(h + 0.04 * (rng.random(n) - 0.5))
                alpha = min(alpha * 1.5, 15.0)
                patience = 0
                continue

        k = min(5, len(corr))
        top = np.argsort(corr)[-k:]
        w = corr[top].copy()
        ws = w.sum()
        if ws <= 1e-15:
            break
        w /= ws

        g = np.zeros(n)
        for wi, ti in zip(w, top):
            m = int(ti) - (n - 1)
            g += wi * grad_corr(h, m)
        g *= dx

        ok = False
        for _ in range(25):
            hn = project_feasible(h - alpha * g)
            c5n = compute_c5(hn)
            if c5n <= c5:
                h = hn
                ok = True
                break
            alpha *= 0.5
        if not ok:
            alpha *= 0.1
            if alpha < 1e-12:
                break

    return best_h, best_c5

def opt_01(h_cont, deadline, rng):
    n = len(h_cont)
    dx = 2.0 / n
    h = (h_cont > 0.5).astype(float)
    diff = 0.5 * n - h.sum()
    if diff > 0:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:int(diff)]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:int(-diff)]] = 0.0

    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_cur = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5_cur
    T = 0.08

    while time.time() < deadline:
        T = max(T * 0.9995, 0.0005)
        a = rng.integers(n)
        b = rng.integers(n)
        while a == b:
            b = rng.integers(n)
        if h[a] == h[b]:
            continue
        # swap: a goes 1->0 (or 0->1), b goes opposite
        h[a], h[b] = h[b], h[a]
        corr_new = np.correlate(h, 1.0 - h, mode="full")
        c5_new = np.max(corr_new) * dx

        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-12)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[a], h[b] = h[b], h[a]

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = project_feasible(rng.random(45))

    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(budget_s * 0.92 - elapsed(), 1.0)

    configs = [
        (180, "cont", 0.30),
        (180, "bin", 0.25),
        (360, "cont", 0.25),
        (360, "bin", 0.15),
        (90, "cont", 0.05),
    ]

    for n, mode, frac in configs:
        t_alloc = budget_s * frac
        deadline = start_time + t_alloc
        if remaining() < 10:
            break

        h_up = upsample(h_parent, n)
        h_up = project_feasible(h_up)
        h_pert = project_feasible(h_up + 0.08 * (rng.random(n) - 0.5))

        if mode == "cont":
            for h_init in [h_up, h_pert]:
                if remaining() < 5:
                    break
                t_sub = min(deadline - time.time(), t_alloc / 2 - 1)
                if t_sub < 2:
                    break
                h_opt, c5_opt = opt_continuous(h_init, start_time + t_sub, rng)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        else:
            for h_init in [h_up, h_pert]:
                if remaining() < 5:
                    break
                t_sub = min(deadline - time.time(), t_alloc / 2 - 1)
                if t_sub < 2:
                    break
                h_opt, c5_opt = opt_01(h_init, start_time + t_sub, rng)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                # Also try continuous refinement from the 0-1 solution
                if c5_opt < best_c5 + 0.02 and remaining() > 15:
                    h_ref = h_opt + 0.01 * (rng.random(n) - 0.5)
                    t_sub2 = min(remaining(), budget_s * 0.05)
                    h_ref, c5_ref = opt_continuous(h_ref, start_time + t_sub2, rng)
                    if c5_ref < best_c5:
                        best_c5 = c5_ref
                        best_h = h_ref.copy()

    # Final polish on the best solution found
    if remaining() > 20:
        n_best = len(best_h)
        t_polish = min(remaining(), budget_s * 0.1)
        h_pol = best_h + 0.02 * (rng.random(n_best) - 0.5)
        h_pol, c5_pol = opt_continuous(h_pol, start_time + t_polish, rng)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Verify
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
