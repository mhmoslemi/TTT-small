def compute_corr(h):
    """Compute full correlation of h with 1-h using FFT."""
    n = len(h)
    n_fft = 2 * n * 4  # pad to avoid circular effects
    H = np.fft.rfft(h, n_fft)
    OH = np.fft.rfft(1.0 - h, n_fft)
    corr = np.fft.irfft(H * np.conj(OH), n_fft)

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft-max value
    s = corr * T
    m = np.max(s)
    val = float((m + np.log(np.sum(np.exp(s - m)))) / T) * dx
    # Soft-max weights
    p = np.exp(s - m) / np.sum(np.exp(s - m))
    # p has length 2n-1
    # grad[j] = dx * (conv(p, 1-h)[n-1+j] - conv(p, h)[n-1+j])
    n_fft = 1

    H = np.fft.rfft(h, n_fft)

    conv2 = np.fft.irfft(P * H, n_fft)
    grad = dx * (conv1[n-1:2*n-1] - conv2[n-1:2*n-1])

def project(h, target):
    """Project h onto {x : 0≤x≤1, sum(x)=target}."""
    n = len(h)
    lo, hi = -2.0, 2.0  # τ range
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val(corr, T, dx):
    s = corr * T
    m = np.max(s)
    return (m + np.log(np.sum(np.exp(s - m)))) / T * dx

def compute_grad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * T
    m = np.max(s)
    p = np.exp(s - m) / np.sum(np.exp(s - m))

        n_fft *= 2

    conv1 = np.fft.irfft(P * np.fft.rfft(1.0 - h, n_fft), n_fft)
    conv2 = np.fft.irfft(P * np.fft.rfft(h[::-1], n_fft), n_fft)
    S1 = conv1[n-1 : 2*n-1]
    S2 = conv2[n-1 : 2*n-1][::-1]
    return dx * (S1 - S2)

def project_box_hyperplane(h, target):

        return np.ones(n)

def projected_grad_descent(h0, n, T, lr, n_steps, deadline):

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val(corr, T, dx):
    s = corr * T
    m = np.max(s)
    return float((m + np.log(np.sum(np.exp(s - m)))) / T) * dx

def compute_grad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * T
    m = np.max(s)
    p = np.exp(s - m) / np.sum(np.exp(s - m))
    n_fft = 1
    while n_fft < 3 * n - 1:
        n_fft *= 2
    P = np.fft.rfft(p, n_fft)
    conv1 = np.fft.irfft(P * np.fft.rfft(1.0 - h, n_fft), n_fft)
    conv2 = np.fft.irfft(P * np.fft.rfft(h[::-1], n_fft), n_fft)
    S1 = conv1[n - 1:2 * n - 1]
    S2 = conv2[n - 1:2 * n - 1][::-1]
    return dx * (S1 - S2)

def project_box_hyperplane(h, target):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_opt(h0, n, T, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0

    def func(x):
        return soft_max_val(np.correlate(x, 1.0 - x, mode='full'), T, dx)

    def grad(x):
        return compute_grad(x, T, dx)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(func, h0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_box_hyperplane(res.x, target)
    except Exception:
        pass
    return project_box_hyperplane(h0, target)

def proj_grad(h0, n, T, n_steps, deadline, lr_init=0.05):
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    lr = lr_init

        if c5 < best_c5:

            best_h = h_new.copy()
        h = h_new

    return best_h

def interp_h(h, n_from, n_to):

        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
