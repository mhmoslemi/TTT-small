    return np.correlate(h, 1-h, mode='full')

def compute_c5(h, n):

    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr) * dx)

def flip_bit_best(h, corr, n):
    """Try all single bit flips, return the best one (index, new_corr, improvement)"""
    best_idx = -1
    best_max = np.max(corr)
    best_new_corr = None
    
    for i in range(n):
        delta = 1 if h[i] == 0 else -1
        # Compute new corr
        new_corr = corr.copy()
        for m in range(2*n-1):
            d = 0
            # Part 1: h[i] in a-position
            j1 = i + n - 1 - m
            if 0 <= j1 < n and m != n-1:
                d += delta * (1 - h[j1])
            # Part 2: (1-h)[i] in v-position  
            j2 = i + m - (n-1)
            if 0 <= j2 < n and m != n-1:
                d -= delta * h[j2]
            # Special case m = n-1: for binary, change is 0
            new_corr[m] += d
        
        new_max = np.max(new_corr)
        if new_max < best_max:
            best_max = new_max
            best_idx = i
            best_new_corr = new_corr
    
    return best_idx, best_new_corr, best_max

    """Hill climbing with random restarts on kick"""
    start = time.time()
    dx = 2.0 / n
    corr = compute_corr(h)

    for it in range(max_iter):

            break
        
        idx, new_corr, new_max = flip_bit_best(h, corr, n)
        
        if idx >= 0:
            h[idx] = 1 - h[idx]
            corr = new_corr
            new_c5 = new_max * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:

            i, j = random.sample(range(n), 2)
            h[i] = 1 - h[i]
            h[j] = 1 - h[j]
            corr = compute_corr(h)
            # Don't update best_h here, just continue searching
    
    return best_h, best_c5

Individual change for flip i: d_i[m] = delta_i * (1-h[j1_i[m]]) - delta_i * h[j2_i[m]]
Individual change for flip j: d_j[m] = delta_j * (1-h[j1_j[m]]) - delta_j * h[j2_j[m]]

- h[i] is in the "a" position and (1-h)[j] is in the "v" position (or vice versa)
- h[i] is in the "a" position and h[j] is also in the "a" position (but that's not possible in a single term)

import numpy as np

def compute_corr(h):
    hf = h.astype(np.float64)
    return np.correlate(hf, 1.0 - hf, mode='full')

def flip1(h, i, n):
    """Return delta corr array for flipping bit i"""
    d_i = 1.0 if h[i] == 0 else -1.0
    lo = max(0, n - 1 - i)
    hi = min(2*n - 2, n - 1 + i)
    d = np.zeros(2*n - 1)

    d[lo:hi+1] = d_i * (1.0 - h[i + n - 1 - m]) - d_i * h[i + m - (n-1)]
    return d

def flip2(h, i, j, n):
    """Return delta corr array for flipping bits i and j"""
    d = flip1(h, i, n) + flip1(h, j, n)
    d_i = 1.0 if h[i] == 0 else -1.0
    d_j = 1.0 if h[j] == 0 else -1.0

    if 0 <= m1 < 2*n - 1:
        d[m1] -= d_i * d_j

    return d
