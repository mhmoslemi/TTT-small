import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_exactly(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] - (-diff) / cnt, 0.0)
    return h

def grad_corr_k(h, n, k):
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:s+m])
        grad[s:] -= h[:m]
    else:
        t = -s
        m = n - t
        grad[t:] += (1.0 - h[:m])
        grad[:m] -= h[t:]
    return grad

    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)

    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_simplex(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_exactly(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex(h, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_corr_k(h, n, k):
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:s + m])
        grad[s:] -= h[:m]
    else:
        t = -s
        m = n - t
        grad[t:] += (1.0 - h[:m])
        grad[:m] -= h[t:]
    return grad

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_simplex(h_new, n_new)

def slsqp_polish(h, n, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0

    def obj_and_grad(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode="full") * dx
        m_val = np.max(corr)
        temp = 0.005
        exps = np.exp((corr - m_val) / temp)
        w = exps / np.sum(exps)
        val = m_val + temp * np.log(np.sum(exps))
        grad = np.zeros(n)
        top_n = min(40, 2 * n - 1)
        idx = np.argpartition(w, -top_n)[-top_n:]
        for k in idx:
            if w[k] > 1e-7:
                grad += w[k] * grad_corr_k(hx, n, k)
        return val, grad

    def obj(x):
        v, _ = obj_and_grad(x)
        return v

    def grad(x):
        _, g = obj_and_grad(x)
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        h_new = project_simplex(res.x, n)
        return h_new
    except Exception:
        return h
