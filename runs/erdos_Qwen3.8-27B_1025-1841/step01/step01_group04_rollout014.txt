def slsqp_opt(h0, n, T, maxiter=300, deadline=None):
    dx = 2.0 / n
    target = n / 2.0
    
    res = minimize(
        lambda x: soft_max_obj(x, T, dx),
        h0,
        jac=lambda x: soft_max_obj_grad(x, T, dx)[1],
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                      'jac': lambda x: np.ones(n)}],
        options={'ftol': 1e-12, 'maxiter': maxiter}
    )
    
    if deadline is not None and time.time() > deadline:
        return h0
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project(res.x, n)
    return h0

def lbgfb_opt(h0, n, T, maxiter=500, deadline=None):
    dx = 2.0 / n
    target = n / 2.0
    
    # Use L-BFGS-B with penalty for the equality constraint
    mu = 10.0  # penalty parameter
    
    def penalty_obj(x):
        base = soft_max_obj(x, T, dx)
        pen = mu * (np.sum(x) - target) ** 2
        return base + pen

        _, g = soft_max_obj_grad(x, T, dx)
        g = g + 2 * mu * (np.sum(x) - target) * np.ones(n)
        return g
    
    res = minimize(
        penalty_obj,
        h0,
        jac=penalty_grad,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'ftol': 1e-12, 'maxiter': maxiter, 'maxfun': maxiter*10}
    )
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project(res.x, n)
    return h0
