import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def smooth_obj_grad(h, tau, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    thresh = 1e-12
    active = np.where(p > thresh)[0]
    for k in active:
        pk = p[k]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            xs = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[xs - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            xs = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[xs + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h, n, tau_start, tau_end, max_iters, time_limit, initial_lr=0.3, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr
    v = np.zeros(n)
    beta = momentum
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        frac = (it + 1) / max_iters
        tau = tau_start * (tau_end / tau_start) ** frac
        obj, grad = smooth_obj_grad(h, tau, n)
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if (it + 1) % 30 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        if stall > 20:
            lr = max(lr * 0.6, 1e-10)
            v *= 0.3
            stall = 0
    return best_h, best_c5


def optimize_subgrad_topk(h, n, topk, max_iters, time_limit, initial_lr=0.01, momentum=0.88):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr
    v = np.zeros(n)
    beta = momentum
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        topk_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                xs = np.arange(j0, j1)
                grad[j0:j1] += (1.0 - h[xs - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                xs = np.arange(j0b, j1b)
                grad[j0b:j1b] -= h[xs + sh]
        grad *= (2.0 / n) / topk

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

            lr = max(lr * 0.4, 1e-13)
            v *= 0.3
            stall = 0
            if lr < 1e-13:
