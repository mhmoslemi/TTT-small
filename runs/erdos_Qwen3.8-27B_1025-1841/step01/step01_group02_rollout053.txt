    # Wait: (1/T) * log(sum exp(T * corr[k] * dx))

    return smooth_val

def smooth_gradient(h, T=40.0):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    max_c = np.max(corr)
    log_weights = T * dx * (corr - max_c)
    p = np.exp(log_weights)
    p /= p.sum()
    # grad[m] = dx * sum_j p_j * ((1-h[m+j])*1[m+j<n] - h[m-j]*1[m-j>=0])
    grad = np.zeros(n)
    omh = 1.0 - h

        if pj < 1e-10:

            grad[:n-j] += pj * omh[j:]
            grad[j:] -= pj * h[:n-j]
        else:
            k = -j
            grad[k:] += pj * omh[:n-k]
            grad[:n-k] -= pj * h[k:]
    return grad * dx

def projected_gradient_step(h, grad, lr):

    return make_feasible(h_new, len(h))

One more idea: **alternating optimization**. Fix all h[i] except h[j], and optimize h[j] to minimize the max. Since the max is piecewise linear in h[j] (for fixed other values), the minimum is at a breakpoint or boundary. We can find it analytically!

For fixed h[0..j-1, j+1..n-1], the function f(h[j]) = max_k g_k(h[j]) where:
g_k(h[j]) = h[j]*(1-h[j+k]) + (rest for lag k)  [for lag k where j+k < n]
           + h[j-k]*1 + ... [for lag -k where j-k >= 0, the h[j] appears as the "second" term]

More precisely, for the lag j (let me call the index variable m to avoid confusion):
For lag ℓ:
- g_ℓ contains h[m]*(1-h[m+ℓ]) if m+ℓ < n (h[m] is the "first" factor)
- g_ℓ contains h[m-ℓ]*(1-h[m]) if m-ℓ >= 0 (h[m] is the "second" factor, as (1-h[m]))

So g_ℓ as a function of t = h[m]:
g_ℓ(t) = A_ℓ * t + B_ℓ * (1-t) + C_ℓ
where:
- A_ℓ = (1-h[m+ℓ]) if m+ℓ < n, else 0
- B_ℓ = h[m-ℓ] if m-ℓ >= 0, else 0
- C_ℓ = sum of all other terms in g_ℓ (independent of t)

So g_ℓ(t) = (A_ℓ - B_ℓ) * t + (B_ℓ + C_ℓ) = slope_ℓ * t + intercept_ℓ

And f(t) = max_ℓ (slope_ℓ * t + intercept_ℓ) is a piecewise linear (convex) function of t!

The minimum of a piecewise linear convex function on [0,1] is either at t=0, t=1, or at a breakpoint (intersection of two lines).

This gives us an EXACT coordinate descent! For each coordinate, we find the exact optimal value in O(n²) time (compute all slopes and intercepts, then find the minimum of the upper envelope).

Even better: the minimum of the upper envelope of lines on [0,1] can be found by:
1. Evaluate f at t=0 and t=1
2. The minimum is at the point where the maximum switches from one line to another (or at boundary)

Actually, for a convex piecewise linear function, the minimum is at a unique point (or a flat region). We can find it by:
- Computing all the lines (slope_ℓ, intercept_ℓ)
- Finding the point where the upper envelope has its minimum

This is equivalent to: minimize max_ℓ (slope_ℓ * t + intercept_ℓ) over t ∈ [0,1].

This can be solved by bisection on t (since the function is convex) or by finding the intersection points.

For bisection: evaluate the convex function and its "derivative" (which lag is active) and bisect. O(log(1/eps) * n) per coordinate.

            break
        improved = False
        for m in range(n):
            if time_limit and time.time() - start_time > time_limit:
                break

            slopes = np.zeros(2 * n - 1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_polytope(v, target_sum):
    n = len(v)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-13:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(T * dx * (corr - max_c))
    obj = (max_c + np.log(exp_c.sum()) / T) * dx
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for j in range(-(n-1), n):
        pj = p[j + n - 1]
        if pj < 1e-10:
            continue
        if j >= 0:
            grad[:n-j] += pj * omh[j:]
            grad[j:] -= pj * h[:n-j]
        else:
            k = -j
            grad[k:] += pj * omh[:n-k]
            grad[:n-k] -= pj * h[k:]
    grad *= dx
    return obj, grad

def pgd_optimize(h, T, n_iter, lr_init, time_limit, start_time):
    n = len(h)
    target = n / 2.0
    f_cur, g = smooth_obj_and_grad(h, T)
    for it in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        lr = lr_init * (0.99 ** it)
        h_new = project_polytope(h - lr * g, target)
        f_new, g_new = smooth_obj_and_grad(h_new, T)
        if f_new > f_cur:
            # backtracking
            for bt in range(10):
                lr *= 0.5
                h_new = project_polytope(h - lr * g, target)
                f_new, g_new = smooth_obj_and_grad(h_new, T)
                if f_new <= f_cur:
                    break
            else:
                break
        h, f_cur, g = h_new, f_new, g_new
    return h

                t_new = 1.0
            else:
                # Interior minimum via ternary search
                a, b = 0.0, 1.0
                for _ in range(40):
                    c1 = a + (b - a) / 3.0
                    c2 = b - (b - a) / 3.0
                    if np.max(slopes * c1 + intercepts) < np.max(slopes * c2 + intercepts):
                        b = c2
                    else:
                        a = c1
                t_new = (a + b) / 2.0
                f_check = np.max(slopes * t_new + intercepts)
                if f_check > f_old - 1e-10:
                    continue  # no improvement
            
            if abs(t_new - t_old) > 1e-12:
                h[m] = t_new
                corr += slopes * (t_new - t_old)
                new_max = np.max(corr)
                if new_max * dx < c5:
                    c5 = float(new_max * dx)
                improved = True
        
        if not improved:
            break
    
    return h, c5

    final_c5 = compute_c5(best_h)

        best_c5 = final_c5
