def smooth_max_gradient_analytical(h, beta):
    n = len(h)
    dx = 2.0 / n
    K = 2*n - 1  # number of correlation lags
    # Compute correlations
    corr = np.correlate(h, 1-h, mode='full')  # length 2n-1
    scaled = beta * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    weights = exp_scaled / np.sum(exp_scaled)  # length 2n-1
    
    # Now compute gradient
    grad = np.zeros(n)
    # For each lag k (indexed 0 to 2n-2 in corr array), corresponding to shift k-(n-1)
    # c_k = sum_i h[i+k_shift] * (1-h[i]) where k_shift = k_idx - (n-1)
    # ∂c_k/∂h[j] = (1-h[j-k_shift])*[0<=j-k_shift<n] - h[j+k_shift]*[0<=j+k_shift<n]
    
    for k_idx in range(K):
        k_shift = k_idx - (n-1)
        w = weights[k_idx] * dx
        if w < 1e-15:
            continue
        # Term 1: (1-h[j-k_shift]) for j where 0 <= j-k_shift < n => k_shift <= j < n+k_shift
        # Term 2: -h[j+k_shift] for j where 0 <= j+k_shift < n => -k_shift <= j < n-k_shift
        
        # For term 1: j ranges from max(0, k_shift) to min(n-1, n-1+k_shift) = n-1
        j1_start = max(0, k_shift)
        j1_end = min(n, n + k_shift)  # exclusive
        if j1_start < j1_end:
            grad[j1_start:j1_end] += w * (1 - h[j1_start - k_shift : j1_end - k_shift])
        
        # For term 2: j ranges from max(0, -k_shift) to min(n-1, n-1-k_shift)
        j2_start = max(0, -k_shift)
        j2_end = min(n, n - k_shift)  # exclusive
        if j2_start < j2_end:
            grad[j2_start:j2_end] -= w * h[j2_start + k_shift : j2_end + k_shift]
    
    return grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n

    s = beta * corr * dx
    m = s.max()
    return m / beta + np.log(np.sum(np.exp(s - m))) / beta

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = beta * corr * dx
    m = s.max()

    w = exp_s / exp_s.sum()  # normalized weights, sum to 1
    # w[k_idx] corresponds to lag k_idx - (n-1)
    grad = np.zeros(n)

        if wk < 1e-14:

            else:
                no_improve += 1
        if no_improve > 80:
            beta = min(beta * 1.4, 120.0)
            alpha = 0.05
            h = project_feasible(h + 0.04 * rng.randn(n), n)
            no_improve = 0
            if beta > 60:
                h = project_feasible(h_init + 0.1 * rng.randn(n), n)
                beta = 5.0
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 0
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        if parent_h.ndim != 1:
            parent_h = None
    except Exception:
        parent_h = None
    if parent_h is not None and len(parent_h) > 0:
        h = project_feasible(parent_h.copy(), len(parent_h))
        best_h = h.copy()
        best_c5 = compute_c5(h)
        best_n = len(h)
    else:
        n0 = 42
        best_h = project_feasible(np.ones(n0) / 2.0, n0)
        best_c5 = compute_c5(best_h)
        best_n = n0
    base_n = best_n
    if base_n < 42:
        base_n = 42
    grids = sorted(set([base_n, base_n * 2, base_n * 4, base_n * 8, min(base_n * 14, 700)]))
    grids = [g for g in grids if g <= 1000]
    n_grids = len(grids)
    time_per = (budget_s - 20) / n_grids
    prev_h = best_h
    for gi, n in enumerate(grids):
        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break
        tb = min(time_per, rem - 5)
        if gi == 0 and parent_h is not None:
            h_init = project_feasible(prev_h.copy(), n)
        else:
            h_init = interp_to_grid(prev_h, n)
            h_init = project_feasible(h_init, n)
        h_opt, c5_opt = optimize_h(h_init, n, tb, seed=seed + gi * 7)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
        prev_h = best_h
    return best_h, best_c5, best_n
