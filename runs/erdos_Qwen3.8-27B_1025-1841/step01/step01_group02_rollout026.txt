import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    shifted = np.clip(shifted, -60.0 / T, 0.0)
    exp_s = np.exp(T * shifted)
    sum_exp = np.sum(exp_s)
    obj = (mc + np.log(sum_exp) / T) * dx
    weights = exp_s / sum_exp

    grad = np.zeros(n)
    for m in range(-(n - 1), n):
        w = weights[n - 1 + m]
        if w < 1e-14:
            continue
        if m >= 0:
            h1 = h[:n - m]
            grad[m:] += w * (1.0 - h1)
            grad[:n - m] -= w * h[m:]
        else:
            mp = -m
            h1 = h[mp:]
            grad[:n - mp] += w * (1.0 - h1)
            grad[mp:] -= w * h[:n - mp]
    grad *= dx
    return obj, grad

def run_lbfgs(h0, T, maxiter, tol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj_wrapper(x):
        v, _ = smooth_obj_grad(x, T)
        return v

    def grad_wrapper(x):
        _, g = smooth_obj_grad(x, T)
        return g

    res = minimize(obj_wrapper, h0, jac=grad_wrapper, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8})
    return res

def coordinate_polish(h, n, max_rounds=10, time_limit=None, start_time=None):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        improved = False
        rnd += 1
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        for j in range(n):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            old = h[j]
            for cand in [0.0, 1.0, 0.5, old + 0.05, old - 0.05, old + 0.02, old - 0.02]:
                if cand < 0 or cand > 1:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = trial
                    improved = True
    return h, best_c5

    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    best_c5 = compute_c5(h0)

    def time_left():

    N1 = 640

            c = compute_c5(h_full)
            if c < best_c5:

                best_h = h_full.copy()

            h_start = upsample_h(best_h, N3)

                    best_h = hx.copy()

                    c = compute_c5(hx)
                    if c < best_c5:

                        best_h = hx.copy()

    n_cur = len(best_h)
    if time_left() > 60:
        for T in [20.0, 10.0]:
            if time_left() < 40:
                break
            def slsqp_obj(x):
                v, _ = smooth_obj_grad(x, T)
                return v
            def slsqp_grad(x):
                _, g = smooth_obj_grad(x, T)
                return g
            def slsqp_con(x):
                return np.sum(x) - n_cur / 2.0
            def slsqp_con_jac(x):
                return np.ones(n_cur)

            cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
            bounds = [(0.0, 1.0)] * n_cur
            try:
                res = minimize(slsqp_obj, best_h.copy(), jac=slsqp_grad,
