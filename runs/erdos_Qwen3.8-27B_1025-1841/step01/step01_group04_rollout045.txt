def corr_grad_one(h, k):
    n = len(h)
    v = 1.0 - h
    shift = k - (n - 1)
    grad = np.zeros(n)
    lo_A = max(0, -shift)
    hi_A = min(n, n - shift)

    lo_B = max(0, shift)

    if lo_B < hi_B:

    return grad

                h_try[k] += eps
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
            # Move mass from k to j
            if h[k] > eps and h[j] < 1.0 - eps:
                h_try = h.copy()
                h_try[k] -= eps
                h_try[j] += eps
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 25
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(42, 0.5)
    h_best = project_h(h_best, len(h_best))
    n_best = len(h_best)
    best_c5 = evaluate_c5(h_best)

    # Phase 1: Gradient optimization at increasing T on current resolution
    T_list = [20, 50, 150, 400, 1000]
    steps_list = [300, 400, 500, 600, 600]
    for i, (T, nsteps) in enumerate(zip(T_list, steps_list)):
        if time.time() > deadline - 200:
            break
        alpha_init = 2.0 / (T * 4.0 / n_best + 1e-8)
        alpha_init = min(alpha_init, 5.0)
        h_new, c5_new = projected_grad_ls(h_best, n_best, T, alpha_init, nsteps, deadline, check_every=80)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new

    # Phase 2: Upsample and optimize at 2x resolution
    if time.time() < deadline - 300:
        n_hi = n_best * 2
        h_hi = interp_h(h_best, n_best, n_hi)
        h_hi = project_h(h_hi, n_hi)
        for T in [50, 200, 600]:
            if time.time() > deadline - 100:
                break
            alpha_init = 2.0 / (T * 4.0 / n_hi + 1e-8)
            alpha_init = min(alpha_init, 5.0)
            h_hi, _ = projected_grad_ls(h_hi, n_hi, T, alpha_init, 300, deadline, check_every=60)
        # Downsample and check
        h_ds = interp_h(h_hi, n_hi, n_best)
        h_ds = project_h(h_ds, n_best)
        c5_ds = evaluate_c5(h_ds)
        if c5_ds < best_c5:
            best_c5 = c5_ds
            h_best = h_ds
        # Also keep the high-res if it's better
        c5_hi = evaluate_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            h_best = h_hi
            n_best = n_hi

    # Phase 3: Fine polishing with very high T
    if time.time() < deadline - 80:
        T = 2000
        alpha_init = 1.0 / (T * 4.0 / n_best + 1e-8)
        alpha_init = min(alpha_init, 1.0)
        h_new, c5_new = projected_grad_ls(h_best, n_best, T, alpha_init, 1500, deadline, check_every=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new

    # Phase 4: Coordinate-level polish (pair swaps)
    if time.time() < deadline - 30:
        h_pol, c5_pol = coordinate_polish(h_best, n_best, deadline, max_rounds=30)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            h_best = h_pol

    # Phase 5: Random perturbations + re-optimize
    for attempt in range(15):
        if time.time() > deadline - 15:
            break
        noise_scale = 0.03 + 0.02 * attempt
        h_pert = h_best + np.random.randn(n_best) * noise_scale
        h_pert = project_h(h_pert, n_best)
        T = 300
        alpha_init = 2.0 / (T * 4.0 / n_best + 1e-8)
        alpha_init = min(alpha_init, 3.0)
        h_new, c5_new = projected_grad_ls(h_pert, n_best, T, alpha_init, 200, deadline, check_every=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new

    # Final: ensure feasibility and evaluate
    h_best = project_h(h_best, n_best)
    best_c5 = evaluate_c5(h_best)
    return (h_best, best_c5, n_best)
