```python
def _smooth_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)  # lag
        w = weights[m]
        if w < 1e-15:
            continue
        # For each j, add w * [(1-h[j-k]) if valid1 - h[j+k] if valid2]
        # valid1: 0 <= j-k < n and 0 <= j < n
        # valid2: 0 <= j < n and 0 <= j+k < n
        
        if k >= 0:
            # j-k >= 0 means j >= k; j < n means j < n
            # So j ranges from k to n-1
            # valid1: j in [k, n-1]
            # valid2: j+k < n means j < n-k, so j in [0, n-k-1]
            j_start1 = k
            j_end1 = n
            grad[j_start1:j_end1] += w * (1.0 - h[0:n-k])  # (1-h[j-k]) for j=k..n-1 → h[0..n-k-1]
            j_start2 = 0
            j_end2 = n - k
            grad[j_start2:j_end2] -= w * h[k:n]  # h[j+k] for j=0..n-k-1 → h[k..n-1]
        else:
            # k < 0, let d = -k > 0
            d = -k
            # j-k = j+d >= 0 always (j>=0), j+d < n means j < n-d
            # valid1: j in [0, n-d-1]
            # valid2: j+k = j-d >= 0 means j >= d, j-d < n always (j<n)
            # So valid2: j in [d, n-1]
            grad[0:n-d] += w * (1.0 - h[d:n])  # (1-h[j+d]) for j=0..n-d-1 → h[d..n-1]
            grad[d:n] -= w * h[0:n-d]  # h[j-d] for j=d..n-1 → h[0..n-d-1]
    
    return grad
```
