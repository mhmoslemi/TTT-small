import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = np.exp((corr - m) * beta)
    return (m + np.log(e.sum() / len(corr)) / beta) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            return np.clip(h * (target / max(h.sum(), 1e-15)), 0, 1)
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_golden_init(n, seed=0):
    rng = np.random.RandomState(seed)
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    h[np.floor(np.arange(n) * (1.0 / golden)).astype(int) % n < n // 2 + (1 if n % 2 else 0)] = 1.0
    h[np.floor(np.arange(n) * (1.0 / (2.0 * golden))).astype(int) % n < n // 3] += 0.3
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_binary_init(n, seed=0):
