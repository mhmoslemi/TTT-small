import time
from scipy.optimize import minimize

def compute_c5(h, n, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target, iters=100):
    h = np.array(h, dtype=float).copy()
    for _ in range(iters):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        active = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(active):
            h[active] += diff / np.sum(active)
        else:
            h += diff / len(h)
        np.clip(h, 0, 1, out=h)
    return h

def obj_penalty(x, n, dx, w=1e6):
    h = np.clip(x, 0, 1)
    return compute_c5(h, n, 2.0/n) + w * (np.sum(h) - n/2.0)**2

    best_h = h.copy()

def fix_sum(h, target, iters=100):

        if abs(diff) < 1e-13:
            break
        active = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(active):
            h[active] += diff / np.sum(active)
        else:
            h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_subgradient(h, n, dx, corr, tol=0.3):
    """Compute subgradient of max corr[k]*dx"""
    max_corr = np.max(corr)
    active = corr >= (max_corr - tol)
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        lag = k_idx - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                g[:n-lag] += (1.0 - h[lag:])
                g[lag:] += -h[:n-lag]
        else:
            m = -lag
            if n - m > 0:
                g[m:] += (1.0 - h[:n-m])
                g[:n-m] += -h[m:]
        grad += g
    grad *= dx / max(1, np.sum(active))
    return grad

from scipy.optimize import minimize

def compute_c5(h, n, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target, iters=200):
    h = np.array(h, dtype=float).copy()
    for _ in range(iters):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        active = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(active):
            h[active] += diff / np.sum(active)
        else:
            h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_pen(x, n, dx, w):
    h = np.clip(x, 0, 1)
    return compute_c5(h, n, dx) + w * (np.sum(h) - n / 2.0) ** 2

def grad_active(h, n, dx, corr, tol=0.5):
    max_c = np.max(corr)
    active = np.where(corr >= max_c - tol)[0]

                g[:sz] += (1.0 - h[lag:])
                g[lag:] += -h[:sz]
        else:
            m = -lag
            sz = n - m
            if sz > 0:
                g[m:] += (1.0 - h[:sz])
                g[:sz] += -h[m:]
        grad += g
    grad *= dx / max(1, len(active))
    return grad

            h = h_new

            break
