    h = x

    k_max = int(np.argmax(corr))
    p = k_max - (n - 1)
    grad = np.zeros(n)
    # ∂C[k_max]/∂h[a] = (1-h[a-p])*I[0≤a-p<n] - h[a+p]*I[0≤a+p<n]
    a_idx = np.arange(n)

    grad[mask1] += (1.0 - h[a_idx[mask1] - p])
    # Second term: -h[a+p] where 0≤a+p<n

    grad[mask2] -= h[a_idx[mask2] + p]
    return grad * dx

    return np.sum(x) - n / 2.0

def _sum_constraint_jac(x, n):
    return np.ones(n)

    """Project h onto {0≤h≤1, sum(h)=target} approximately."""
    h = np.clip(h, 0.0, 1.0)

def _make_ineq_constraint(k, n, dx):

    def fun(params):
        h = params[:n]
        t = params[n]
        p = k - (n - 1)
        # Compute corr_k = sum_m h[m+p] * (1-h[m]) for valid m
        # m ranges so that 0≤m<n and 0≤m+p<n

        if m_min > m_max:

        else:
            c = np.sum(h[m_min + p:m_max + p] * (1.0 - h[m_min:m_max + 1]))

        h = params[:n]
        g = np.zeros(n + 1)
        g[n] = 1.0  # dt/dt = 1
        p = k - (n - 1)
        # d(corr_k * dx)/dh[a] = dx * [(1-h[a-p])*I - h[a+p]*I]

            if 0 <= a + p <= n - 1:
                val -= h[a + p]
            g[a] = -dx * val  # negative because constraint is t - corr*dx >= 0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def _project_sum(h, target):
    h = h.copy()
