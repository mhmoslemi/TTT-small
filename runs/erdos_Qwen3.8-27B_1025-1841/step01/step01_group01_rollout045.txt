import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            inc = diff / cnt
            h[mask] = np.minimum(h[mask] + inc, 1.0)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            dec = -diff / cnt
            h[mask] = np.maximum(h[mask] - dec, 0.0)
    return h

def smooth_max_obj(x, n, dx, eps):
    h = repair(x, n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
