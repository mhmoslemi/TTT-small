        for ss in step_sizes:
            for sgn in [1.0, -1.0]:
                h[i] = np.clip(old + sgn * ss, 0.0, 1.0)
                diff = n / 2.0 - h.sum()
                if abs(diff) > 1e-14:
                    h = h + diff / n
                    h = np.clip(h, 0.0, 1.0)
                v = compute_c5(h)
                if v < best_local[1] - 1e-14:
                    best_local = (h[i], v)
        if best_local[1] < current - 1e-14:
            h[i] = best_local[0]
            diff = n / 2.0 - h.sum()
            if abs(diff) > 1e-14:
                h = h + diff / n
                h = np.clip(h, 0.0, 1.0)
            current = best_local[1]
        else:
            h[i] = old
    return h, current

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def rem():
        return budget_s - (time.time() - t0)

    has_parent = ('initial_h_values' in globals() and initial_h_values is not None)

    tiers = [(80, 8, 2000, 1e-13), (160, 6, 1500, 1e-12), (320, 4, 1000, 1e-11)]

    for n, n_starts, maxiter, ftol in tiers:
        if rem() < 120:
            break
        for restart in range(n_starts):
            if rem() < 60:
                break
            if restart == 0 and has_parent:
                parent = np.asarray(initial_h_values, dtype=float)
                h0 = parent.copy() if len(parent) == n else interpolate_to_n(parent, n)
            elif restart == 1:
                h0 = make_binary_init(n, rng)
            elif restart == 2:
                h0 = make_golden_init(n)
            else:
                h0 = make_random_init(n, rng)
                h0 += rng.normal(0, 0.1, n)
            h0 = project_feasible(h0, n)
            try:
                res = slsqp_solve(h0, n, maxiter, ftol)
                if res.x is not None:
                    h_opt = project_feasible(res.x, n)
                    val = compute_c5(h_opt)
                    if val < best_c5:
                        best_c5, best_h, best_n = val, h_opt.copy(), n
            except Exception:
                pass
        if best_n == n and best_h is not None and rem() > 120:
            h0 = best_h + rng.normal(0, 0.02, n)
            try:
                res = slsqp_solve(h0, n, maxiter, ftol * 0.01)
                if res.x is not None:
                    h_opt = project_feasible(res.x, n)
                    val = compute_c5(h_opt)
                    if val < best_c5:
                        best_c5, best_h, best_n = val, h_opt.copy(), n
            except Exception:
                pass
            if best_n == n and rem() > 60:
                h_ref, c5_ref = local_refine(best_h, n, rng, 500)
                if c5_ref < best_c5:
                    best_c5, best_h = c5_ref, h_ref

    if best_h is not None and rem() > 200:
        for ups_n in [160, 320, 640]:
            if best_n < ups_n and rem() > 100:
                h_int = project_feasible(interpolate_to_n(best_h, ups_n), ups_n)
                try:
                    res = slsqp_solve(h_int, ups_n, 1200, 1e-12)
                    if res.x is not None:
                        h_opt = project_feasible(res.x, ups_n)
                        val = compute_c5(h_opt)
                        if val < best_c5:
                            best_c5, best_h, best_n = val, h_opt.copy(), ups_n
                except Exception:
                    pass
            elif rem() < 100:
                break

    if best_h is not None and rem() > 60:
        h_ref, c5_ref = local_refine(best_h, best_n, rng, 1000)
        if c5_ref < best_c5:
            best_c5, best_h = c5_ref, h_ref

    if best_h is None:
        best_n = 80
        best_h = project_feasible(make_binary_init(best_n, rng), best_n)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
