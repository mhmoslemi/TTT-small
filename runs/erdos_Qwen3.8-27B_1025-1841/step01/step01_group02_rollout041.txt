**Alternative: Use scipy's L-BFGS-B with equality constraint**

Actually, scipy's L-BFGS-B doesn't support equality constraints well. SLSQP does but it's slow for large n.

**Alternative: Use the augmented Lagrangian method**

Minimize f(h) + λ/2 * (sum(h) - n/2)² + penalty for box constraints.

This is equivalent to: minimize f(h) + μ/2 * ||clip(h,0,1) - h||² + λ/2*(sum(h)-n/2)²

Where f(h) = soft_max_obj(h, T).

This is a smooth unconstrained problem (if we use soft constraints) that can be solved with L-BFGS-B.

Actually, the standard approach is:
- Use box constraints [0,1] (handled by L-BFGS-B)
- Handle equality constraint via augmented Lagrangian:
  L(h, λ) = f(h) + λ*(sum(h) - n/2) + μ/2*(sum(h) - n/2)²
  Minimize over h ∈ [0,1]^n, then update λ ← λ + μ*(sum(h*) - n/2)

This should work well! L-BFGS-B is very efficient for box-constrained problems.

def optimize_augmented_lagrangian(h0, n, T, time_budget):
    start = time.time()

    target = n / 2.0
    
    lam = 0.0
    mu = 1.0
    
    h = h0.copy()

        if time.time() - start > time_budget * 0.95:
            break
        
        # Define objective with augmented Lagrangian
        def obj(h):
            c5_soft = soft_max_obj(h, T)
            constraint_violation = np.sum(h) - target
            return c5_soft + lam * constraint_violation + 0.5 * mu * constraint_violation**2
        
        def grad_obj(h):
            g = compute_gradient(h, T)
            constraint_grad = np.ones(n)
            constraint_violation = np.sum(h) - target
            return g + (lam + mu * constraint_violation) * constraint_grad
        
        # Use L-BFGS-B
        res = minimize(obj, h, jac=grad_obj, method='L-BFGS-B',
                       bounds=[(0,1)]*n, options={'maxiter': 100, 'ftol': 1e-12})
        
        h = res.x
        constraint_violation = np.sum(h) - target

        if abs(constraint_violation) > 1e-10:
            lam += mu * constraint_violation
            if abs(constraint_violation) > 1e-6:
                mu *= 2.0
        else:
            mu *= 0.5  # reduce penalty when constraint satisfied
            lam = 0.0  # reset multiplier
    
    h = make_feasible(h, n/2.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_c5(h, T):
    """Smooth approximation of C5"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = corr * dx
    m = val.max()
    if m == 0:
        return 0.0
    tc = T * (val - m)
    return float(m + np.log(np.sum(np.exp(tc))) / T)

def grad_soft_c5(h, T):
    """Gradient of smooth C5 approximation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = corr * dx
    m = val.max()
    tc = T * (val - m)
    w = np.exp(tc)
    w /= w.sum()
    
    grad = np.zeros(n)
    grad += w[n-1] * (1.0 - 2.0 * h)
    
    for k in range(n, 2*n - 1):
        d = k - (n - 1)
        wk = w[k]
        if wk < 1e-16:
            continue
        grad[d:] += wk * (1.0 - h[:n-d])
        grad[:n-d] -= wk * h[d:]
    
    for k in range(0, n - 1):
        dp = (n - 1) - k
        wk = w[k]
        if wk < 1e-16:
            continue
        grad[:n-dp] += wk * (1.0 - h[dp:])
        grad[dp:] -= wk * h[:n-dp]
    
    return dx * grad

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n / 2.0)

def random_binary(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        h[rng.choice(ones, len(ones) - target, replace=False)] = 0.0
    elif len(ones) < target:
        h[rng.choice(zeros, target - len(ones), replace=False)] = 1.0
    return h

def optimize_al(h0, n, T, time_limit):
    """Augmented Lagrangian + L-BFGS-B"""
    start = time.time()
    target = n / 2.0
    lam = 0.0
    mu = 1.0
    h = h0.copy()
    
    for outer in range(50):
        if time.time() - start > time_limit:
            break
        
        cv0 = np.sum(h) - target
        
        def _f(hx):
            return soft_c5(hx, T) + lam * (np.sum(hx) - target) + 0.5 * mu * (np.sum(hx) - target)**2
        
        def _g(hx):
            g = grad_soft_c5(hx, T)
            cv = np.sum(hx) - target
            return g + (lam + mu * cv)
        
        try:
            res = minimize(_f, h, jac=_g, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10})
            h = np.clip(res.x, 0.0, 1.0)
        except:
            break
        
        cv = np.sum(h) - target
        if abs(cv) < 1e-10:
            mu = max(mu * 0.5, 0.1)
            lam = 0.0
        else:
            lam += mu * cv
            if abs(cv) > 1e-5:
                mu = min(mu * 2.0, 1000.0)
    
    h = make_feasible(h, n / 2.0)
    return h
