∑_k p_k f_k(h) = ∑_k p_k ∑_i h[i+k](1-h[i]) * dx

Q(h) = h^T M h + c^T h

∑_{i,k} p_k h[i+k]h[i] = ∑_i h[i] ∑_k p_k h[i+k]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constrained(h, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(h - tau, 0.0, 1.0)

def upsample_h(h, new_n):

    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_constrained(h_new, new_n)

def pgd_step(h, T, lr):
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode='full')
    f_shifted = f - f.max()
    p = np.exp(T * f_shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(f)
        p[np.argmax(f)] = 1.0
    else:
        p /= p_sum
    one_minus_h = 1.0 - h
    idx = np.arange(n) + n - 1
    corr1 = np.convolve(one_minus_h, p, 'full')[idx]
    corr2 = np.convolve(h, p[::-1], 'full')[idx]
    grad = dx * (corr1 - corr2)
    h_new = h - lr * grad
    h_new = project_constrained(h_new, n)
    f_new = np.correlate(h_new, 1.0 - h_new, mode='full')
