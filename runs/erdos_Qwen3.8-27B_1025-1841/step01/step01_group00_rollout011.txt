import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def next_pow2(m):
    p = 1
    while p < m:
        p *= 2
    return p

def fast_smooth_grad(h, n, tau):
    """Compute smooth max objective and gradient using FFT for gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s_exp = exp_c.sum()
    p = exp_c / s_exp  # length 2n-1
    
    # Gradient via FFT convolution
    # w = p, indexed as w[s] for s = -(n-1), ..., n-1, stored in p[0], ..., p[2n-2]
    # w_arr[m] = p[m] where m = s + n - 1, so s = m - (n-1)
    # w_rev_arr[m] = p[2n-2-m] (reversed)
    
    w = p.copy()  # length 2n-1
    w_rev = p[::-1].copy()  # reversed, length 2n-1
    
    N = next_pow2(3 * n - 2)
    Fw = np.fft.rfft(w, N)
    Fwr = np.fft.rfft(w_rev, N)
    F1h = np.fft.rfft(1.0 - h, N)
    Fh = np.fft.rfft(h, N)
    
    # (w * (1-h))[i] = ifft(fft(w) * fft(1-h))[i]
    conv1 = np.fft.irfft(Fw * F1h, N)[:n]
    # (w_rev * h)[i] = ifft(fft(w_rev) * fft(h))[i]
    conv2 = np.fft.irfft(Fwr * Fh, N)[:n]
    
    grad = dx * (conv1 - conv2)
    obj = dx * (max_c + np.log(s_exp) / tau)
    return obj, grad

def nesterov_loop(h, n, stages, time_budget, v0=None):
    """Run Nesterov-optimized smooth max for given tau stages."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if v0 is None:
        v = np.zeros(n)
    else:
        v = v0.copy()
    start = time.time()
    
    for tau, n_iters, lr, beta in stages:
        for it in range(n_iters):
            elapsed = time.time() - start
            if elapsed > time_budget:
                return best_h, best_c5, v
            obj, grad = fast_smooth_grad(h, n, tau)
            v = beta * v - lr * grad
            h_new = project_to_feasible(h + v, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        v *= 0.3  # dampen between stages
    
    return best_h, best_c5, v
