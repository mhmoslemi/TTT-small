import numpy as np
from scipy.optimize import minimize
import time

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_gradient(h, n, shifts):
    grad = np.zeros(n)
    for s in shifts:
        if s >= 0 and s < n:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
        elif s < 0 and (-s) < n:
            t = -s
            grad[:n-t] += (1.0 - h[t:])
            grad[t:] -= h[:n-t]
    return grad

def project_feasible(x, n):
    target = n * 0.5
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.clip(x - mid, 0.0, 1.0)
        if val.sum() > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    h = np.clip(x - tau, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += res / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
