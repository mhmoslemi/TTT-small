import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

_G_N = 0
_G_TARGET = 0.0
_G_DX = 0.0
_G_TAU = 0.0


def _c5_exact(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def _smooth_obj(h, dx, tau):
    c = np.correlate(h, 1.0 - h, mode="full")
    fc = c / tau
    m = np.max(fc)
    return float(tau * (m + np.log(np.sum(np.exp(fc - m)))) * dx)


def _smooth_grad(h, dx, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    fc = c / tau
    m = np.max(fc)
    w = np.exp(fc - m)
    w /= w.sum()
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = int(lags[k])
        j0 = max(0, lag)
        j1 = min(n, n + lag)
        if j1 > j0:
            grad[j0:j1] += wk * (1.0 - h[j0 - lag:j1 - lag])
        j0b = max(0, -lag)
        j1b = min(n, n - lag)
        if j1b > j0b:
            grad[j0b:j1b] -= wk * h[j0b + lag:j1b + lag]
    grad *= dx
    return grad


def _obj_fn(h):
    return _smooth_obj(h, _G_DX, _G_TAU)


def _obj_grad(h):
    return _smooth_grad(h, _G_DX, _G_TAU)


def _cons_eq(h):
    return h.sum() - _G_TARGET


def _cons_eq_jac(h):
    return np.ones(_G_N)


def _make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        if not m.any():
            break
        h[m] += diff / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _upsample(h_in, n_out):
    n_in = len(h_in)
    if n_in == n_out:
        return h_in.copy()
    xi = np.linspace(0.0, 2.0, n_in, endpoint=False)
    xo = np.linspace(0.0, 2.0, n_out, endpoint=False)
    return np.interp(xo, xi, h_in)


def _slsqp_optimize(h0, n, tau, max_iter):
    global _G_N, _G_TARGET, _G_DX, _G_TAU
    _G_N = n
    _G_TARGET = n / 2.0
    _G_DX = 2.0 / n
    _G_TAU = tau
    h0 = _make_feasible(h0, n)
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_fn, h0, jac=_obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _make_feasible(res.x, n)
    except Exception:
        pass
    return h0


def _try_trust_constr(h0, n, tau, max_iter):
    global _G_N, _G_TARGET, _G_DX, _G_TAU
    _G_N = n
    _G_TARGET = n / 2.0
    _G_DX = 2.0 / n
    _G_TAU = tau
    h0 = _make_feasible(h0, n)
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_fn, h0, jac=_obj_grad, method='trust-constr',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': max_iter, 'gtol': 1e-8})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _make_feasible(res.x, n)
    except Exception:
        pass
    return h0
