def compute_gradient(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    p = exp_scaled / np.sum(exp_scaled)
    
    # p has length 2n-1
    # For element j, shift = j - (n-1)
    # grad[m] = dx * sum_j p[j] * ((1-h[m+shift_j])*valid - h[m-shift_j]*valid)
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)  # shifts[j] = j - (n-1)
    
    for m in range(n):
        # First term: (1-h[m+shift]) for valid m+shift
        # m+shift in [0,n-1] => shift in [-m, n-1-m]
        # j such that shifts[j] in [-m, n-1-m] => j in [j0, j1]
        j0 = max(0, -(-m) + (n-1))  # j where shifts[j] >= -m => j >= (n-1) - m... 
        
        # Actually: shifts[j] = j - (n-1), so shifts[j] >= -m means j >= (n-1) - m
        # shifts[j] <= n-1-m means j <= 2n-2-m
        # j in [max(0, n-1-m), min(2n-2, 2n-2-m)]
        
        lo1 = max(0, (n-1) - m)
        hi1 = min(2*n-2, 2*n-2 - m)
        # For j in [lo1, hi1], m + shifts[j] is in [0, n-1]
        # (1-h[m+shifts[j]])
        idx1 = np.arange(lo1, hi1+1)
        m_plus_shift = m + shifts[idx1]
        term1 = np.sum(p[idx1] * (1.0 - h[m_plus_shift]))
        
        # Second term: h[m-shift] for valid m-shift
        # m-shifts[j] in [0,n-1] => shifts[j] in [m-(n-1), m]
        # j-(n-1) in [m-(n-1), m] => j in [m, m+n-1]
        lo2 = max(0, m)
        hi2 = min(2*n-2, m + n - 1)
        idx2 = np.arange(lo2, hi2+1)
        m_minus_shift = m - shifts[idx2]
        term2 = np.sum(p[idx2] * h[m_minus_shift])
        
        grad[m] = dx * (term1 - term2)
    
    return grad, p, corr

def adam_optimize(h0, n, T, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, max_steps=5000):
    h = h0.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    step = 0
    
    while step < max_steps and time.time() < deadline:
        grad, p, corr = compute_gradient(h, n, T)
        step += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**step)
        v_hat = v_v / (1 - beta2**step)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project(h_new)
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14 * n:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14 * n:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def _compute_grad(h, n, p):
    dx = 2.0 / n
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        v1 = (m + shifts >= 0) & (m + shifts < n)
        if np.any(v1):
            grad[m] += dx * np.dot(p[v1], 1.0 - h[m + shifts[v1]])
        v2 = (m - shifts >= 0) & (m - shifts < n)
        if np.any(v2):
            grad[m] -= dx * np.dot(p[v2], h[m - shifts[v2]])
    return grad

def grad_softmax(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    mx = np.max(scaled)
    exp_s = np.exp(scaled - mx)
    p = exp_s / np.sum(exp_s)
    return _compute_grad(h, n, p), p

def grad_max(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    mask = corr >= mx - 1e-12
    p = np.zeros(2 * n - 1)
    p[mask] = 1.0 / np.sum(mask)
    return _compute_grad(h, n, p)

def coord_descent_step(h, n, deadline):
    dx = 2.0 / n
    c5_old = evaluate_c5(h)
    for m in range(n):
        if time.time() > deadline - 2:
            break
        old = h[m]
        best_t = old
        best_val = c5_old
        for t in [0.0, 0.25, 0.5, 0.75, 1.0, max(0, min(1, old - 0.02)), max(0, min(1, old + 0.02))]:
            h[m] = t
            h = project(h)
            c5_t = evaluate_c5(h)
            if c5_t < best_val - 1e-15:
                best_val = c5_t
                best_t = t
        h[m] = best_t
        h = project(h)
    return h, best_val

def optimize_adam(h0, n, deadline, T_list, lr=0.1, check_interval=100):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for T in T_list:
        if time.time() > deadline - 3:
            break
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        step = 0
        local_best = best_c5
        stall = 0
        while time.time() < deadline - 3:
            step += 1
            if step % 2000 == 0:
                break
            grad, p = grad_softmax(h, n, T)
            m_v = beta1 * m_v + (1 - beta1) * grad
            v_v = beta2 * v_v + (1 - beta2) * grad**2
            m_hat = m_v / (1 - beta1**step)
            v_hat = v_v / (1 - beta2**step)
            cur_lr = lr / (1.0 + 0.0003 * step)
            h = project(h - cur_lr * m_hat / (np.sqrt(v_hat) + eps))
            if step % check_interval == 0:
                c5 = evaluate_c5(h)
                if c5 < local_best - 1e-12:
                    local_best = c5
                    stall = 0
                else:
                    stall += 1
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                if stall > 5:
                    break
    c5 = evaluate_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def optimize_subgrad(h0, n, deadline, alpha0=0.05):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    for step in range(50000):
        if time.time() > deadline - 2:
            break
        grad = grad_max(h, n)
        a = alpha0 / (1.0 + 0.002 * step)
        h = project(h - a * grad)
        if step % 300 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = evaluate_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 8

    if 'initial_h_values' in globals() and initial_h_values is not None:

        T_list=[200, 500, 1000, 2000, 4000],
        lr=0.03, check_interval=50
    )

    # Phase 4: Coordinate descent fine-tuning (10% budget)
    d4 = t0 + 0.90 * budget_s
    h_cd, c5_cd = coord_descent_step(h_best, n, d4)
