def project_to_feasible(g, n):
    """Project g onto {h : 0 <= h <= 1, sum(h) = n/2} using bisection."""

    # Binary search for the threshold
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):

def project_to_feasible(g, n):

    lo = g.min() - 2.0
    hi = g.max() + 2.0

        s = np.sum(np.clip(g - mid, 0.0, 1.0))

        else:
            hi = mid

    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_obj_grad(h, tau, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        corr_active = corr[idx]
    else:
        idx = np.arange(len(corr))
        corr_active = corr
    
    max_c = corr_active.max()
    shifted = np.clip(tau * (corr_active - max_c), -50, 50)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.zeros(n)
    for ii in range(len(idx)):
        pk = p[ii]
        if pk < 1e-12:
            continue
        k = idx[ii]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            js = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[js - sh])

        if j0b < j1b:

            grad[j0b:j1b] -= pk * h[jb + sh]

    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_obj_grad(h, tau, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    
    if top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        idx = np.arange(len(corr))
    
    corr_a = corr[idx]
    max_c = corr_a.max()
    shifted = np.clip(tau * (corr_a - max_c), -50, 50)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.zeros(n)
    for ii in range(len(idx)):
        pk = p[iii]
        if pk < 1e-12:
            continue
        k = idx[ii]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            js = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[js - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            jb = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[jb + sh]
    
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr

    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)

    stall = 0

            break
        frac = min(1.0, it / max(1, max_iters * 0.7))
        tau = tau_start + (tau_end - tau_start) * frac
        tk = max(3, int(tk_start + (tk_end - tk_start) * frac))
