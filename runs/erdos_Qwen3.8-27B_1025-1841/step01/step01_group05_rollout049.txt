import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = dx * np.max(corr)
    # Subgradient
    j_star = np.argmax(corr)
    lag = j_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        # First term: (1 - h[i-lag]) if 0 <= i-lag < n
        if 0 <= i - lag < n:
            g += (1.0 - h[i - lag])
        # Second term: -h[i+lag] if 0 <= i+lag < n
        if 0 <= i + lag < n:
            g -= h[i + lag]
        grad[i] = g
    return c5, dx * grad

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 0:
        h = h * (n / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def c5_objective(h):
    return compute_c5(h)

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = dx * np.max(corr)
    j_star = np.argmax(corr)
    lag = j_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if 0 <= i - lag < n:
            g += 1.0 - h[i - lag]
        if 0 <= i + lag < n:
            g -= h[i + lag]
        grad[i] = g
    return c5, dx * grad

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-10:
        h *= (n / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_up(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def c5_smooth(h, epsilon=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    m = np.max(corr)
    if m > 0:
        val = m + epsilon * np.log(np.sum(np.exp((corr - m) / epsilon)))
    else:
        val = epsilon * np.log(np.sum(np.exp(corr / epsilon)))
    # Gradient
    weights = np.exp((corr - m) / epsilon) / np.sum(np.exp((corr - m) / epsilon))
    # d(val)/d(h[j]) for each j
    grad = np.zeros(n)
    for k in range(len(corr)):
        lag = k - (n - 1)
        dk_dh = np.zeros(n)
        for i in range(n):
            if 0 <= i - lag < n:
                dk_dh[i] += 1.0 - h[i - lag]
            if 0 <= i + lag < n:
                dk_dh[i] -= h[i + lag]
        grad += weights[k] * dk_dh
    grad *= dx
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (n / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def corr_grad_for_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:n] += 1.0 - h[:n - lag]
            g[:n - lag] -= h[lag:n]
    else:
        d = -lag
        if d < n:
            g[:n - d] += 1.0 - h[d:n]
            g[d:n] -= h[:n - d]
    return g

def c5_smooth_obj(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    c = raw_corr * dx
    m = np.max(c)
    shifted = (c - m) / epsilon
    exp_shifted = np.exp(shifted - np.max(shifted))
    weights = exp_shifted / np.sum(exp_shifted)
    val = m + epsilon * (np.max(shifted) + np.log(np.sum(exp_shifted)))
    return val, weights, raw_corr

def c5_smooth_grad(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    val, weights, raw_corr = c5_smooth_obj(h, epsilon)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        grad += weights[k] * corr_grad_for_lag(h, lag)
    grad *= dx
    return val, grad

def c5_hard_grad(h):
    n = len(h)
    dx = 2.0 / n
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    val = dx * np.max(raw_corr)
    j_star = np.argmax(raw_corr)
    lag = j_star - (n - 1)
    grad = dx * corr_grad_for_lag(h, lag)
    return val, grad

def interpolate_up(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_local_search(h, n, max_iters, t_max):
    h_bin = np.zeros(n)
    idx = np.argsort(-h)
    h_bin[idx[:n // 2]] = 1.0
    current_c5 = compute_c5(h_bin)
    current = h_bin.copy()
    T = 0.02
    best_c5 = current_c5
    best = current.copy()
    ones_idx = np.where(current == 1.0)[0]
    zeros_idx = np.where(current == 0.0)[0]
    for it in range(max_iters):
        if time.time() > t_max:
            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i1 = ones_idx[np.random.randint(len(ones_idx))]
        i0 = zeros_idx[np.random.randint(len(zeros_idx))]
        current[i1] = 0.0
        current[i0] = 1.0
        c_new = compute_c5(current)
        if c_new < current_c5 or (c_new - current_c5) / max(T, 1e-10) < np.log(np.random.rand() + 1e-30):
            current_c5 = c_new
            ones_idx = np.where(current == 1.0)[0]
            zeros_idx = np.where(current == 0.0)[0]
            if c_new < best_c5:
                best_c5 = c_new
                best = current.copy()
        else:
            current[i1] = 1.0
            current[i0] = 0.0
        T *= 0.99995
    return best, best_c5

def run_slsqp(h0, n, epsilon, maxiter):
    bounds = [(0.0, 1.0)] * n
    
    def obj_func(h):
        return c5_smooth_grad(h, epsilon)[0]
    
    def grad_func(h):
        return c5_smooth_grad(h, epsilon)[1]
    
    constraint = {'type': 'eq', 'fun': sum_constraint}
    try:
        res = minimize(obj_func, h0, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=[constraint],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n))
    except Exception:
        return h0, compute_c5(h0)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        initial_h_values = globals()['initial_h_values']
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 42
        h0 = np.zeros(n0)
        h0[::2] = 1.0

    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at n0 with SLSQP (smooth objective)
    epsilon = 0.005
    h_try, c_try = run_slsqp(h0, n0, epsilon, 500)
    if c_try < best_c5:
        best_c5 = c_try
        best_h = h_try

    # Multiple restarts at n0
    for r in range(80):
        if time.time() - t0 > budget_s * 0.15:
            break
        h_r = best_h + np.random.randn(n0) * 0.05
        h_r = project_feasible(h_r, n0)
        h_try, c_try = run_slsqp(h_r, n0, epsilon, 200)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Binary local search at n0
    if time.time() - t0 < budget_s * 0.35:
        h_bin, c_bin = binary_local_search(best_h, n0, 200000, t0 + budget_s * 0.35)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin

    # Final SLSQP polish at n0
    if time.time() - t0 < budget_s * 0.4:
        h_try, c_try = run_slsqp(best_h, n0, 0.002, 300)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 2: Upscale and optimize at higher resolutions
    for target_n in [84, 168, 336, 672, 1344]:
        if time.time() - t0 > budget_s * 0.85:
            break
        h_up = interpolate_up(best_h, target_n)
        eps_up = 0.003
        maxit_up = max(50, min(200, int((budget_s * 0.1) / (target_n * 0.01))))
        h_try, c_try = run_slsqp(h_up, target_n, eps_up, maxit_up)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
            # Second pass with smaller epsilon
            if time.time() - t0 < budget_s * 0.85:
                h_try2, c_try2 = run_slsqp(h_try, target_n, 0.001, 100)
                if c_try2 < best_c5:
                    best_c5 = c_try2
                    best_h = h_try2

    return (best_h, best_c5, len(best_h))
