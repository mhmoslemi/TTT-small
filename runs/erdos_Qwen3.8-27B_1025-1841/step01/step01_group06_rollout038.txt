def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-14):
            h = h_new
            break
        h = h_new

    rem = target - s

            end = int((i + 1) * ratio)

Let me also think about the structure of the problem more. The objective max_k corr_k * dx where corr_k = sum_i h[i](1-h[i+k-(n-1)]).

For the "self" term (k = n-1, i.e., zero shift):
corr_{n-1} = sum_i h[i](1-h[i]) = sum_i h[i] - sum_i h[i]²

Given sum(h) = n/2, this is n/2 - sum_i h[i]².
To minimize this, we want to MAXIMIZE sum_i h[i]².
This is achieved by making h as "spread out" as possible between 0 and 1 (i.e., as many 0s and 1s as possible).
The minimum of n/2 - sum h[i]² subject to sum h = n/2, 0≤h≤1 is:
- If n is even: set n/2 values to 1 and n/2 to 0. Then sum h² = n/2, so corr_{n-1} = 0.
- But wait, for binary h, corr_k = (number of 1s at positions where the shifted value is 0). For the self term (k=n-1), this is the number of 1s where h[i]=1 and h[i]=0, which is 0. So corr_{n-1} = 0 for binary h.

For other shifts, the situation is different. For binary h:
corr_k = number of i such that h[i]=1 and h[i+k-(n-1)]=0 (with valid indices)

For a "good" binary pattern, we want this to be small for all k. This is related to the concept of "low autocorrelation" sequences or "m-sequences".

So the problem for binary h is: find a binary sequence of length n with n/2 ones that minimizes the maximum "cross-correlation" with its complement over all cyclic shifts.

This is related to the "minimum overlap" problem studied in combinatorics.

For the continuous case (h ∈ [0,1]), we have more freedom, so the optimum should be at least as good as the best binary solution.

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        deficit = target - s
        if deficit > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        adj = deficit / np.sum(mask)
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_linear(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return np.clip(h_new, 0.0, 1.0)

def solve_1d(A, B, lam):
    A_eff = A + lam
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        vals = A_eff * mid + B
        max_v = vals.max()
        active = vals >= max_v - 1e-10
        if not active.any():
            active = vals >= max_v - 1e-8
        slopes = A_eff[active]
        if slopes.min() > 1e-12:
            lo = mid
        elif slopes.max() < -1e-12:
            hi = mid
        else:
            break
    return (lo + hi) / 2.0

def coord_desc_sweep(h, corr, n, lam):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n, dtype=np.int64)
    for p in range(n):

        A = np.zeros(n_corr)

        t = solve_1d(A, B, lam)
        delta = t - old
        if abs(delta) > 1e-14:
            corr += delta * A
            h[p] = t
    return h, corr
