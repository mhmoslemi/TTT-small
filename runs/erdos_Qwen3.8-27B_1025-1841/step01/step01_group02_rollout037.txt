from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    
    # Stabilized log-sum-exp
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_vals = np.exp(T * dx * shifted)
    log_sum_exp = np.log(np.sum(exp_vals)) + T * dx * cmax
    obj = log_sum_exp / T
    
    p = exp_vals / np.sum(exp_vals)

    best_c5 = compute_c5(best_h)
    deltas = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01]
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 10:
                break
            old_val = best_h[j]
            found = False
            for d in deltas:
                nv = old_val + d
                if nv < 0.0 or nv > 1.0:
                    continue
                trial = best_h.copy()
                trial[j] = nv
                trial = project_h(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = trial
                    improved = True
                    found = True
                    break
            if found:
                continue
            for nv in [0.0, 1.0]:
                if abs(nv - old_val) < 1e-8:
                    continue
                trial = best_h.copy()
                trial[j] = nv
                trial = project_h(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = trial
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def random_binary_h(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = random_binary_h(320, rng)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: n=640 with FISTA annealing
    n1 = 640
    h1 = upsample_h(best_h, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    temps_640 = [(20, 6000), (35, 5000), (60, 4000), (100, 3000), (160, 2000)]
    for T, niter in temps_640:
        if time.time() - start_time > budget_s - 80:
            break
        h1, _ = fista_optimize(h1, T, niter, start_time, budget_s)
        c5_now = compute_c5(h1)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h1.copy()

    # SLSQP polish at n=640
    if time.time() - start_time < budget_s - 150:
        h1 = slsqp_polish(h1, n1, T=80, max_iter=300)
        c5_s = compute_c5(h1)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h1.copy()

    # Phase 2: n=1280
    n2 = 1280
    if time.time() - start_time < budget_s - 200:
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        temps_1280 = [(30, 4000), (60, 3000), (100, 2000)]
        for T, niter in temps_1280:
            if time.time() - start_time > budget_s - 60:
                break
            h2, _ = fista_optimize(h2, T, niter, start_time, budget_s)
            c5_now = compute_c5(h2)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h2.copy()
        if time.time() - start_time < budget_s - 80:
            h2 = slsqp_polish(h2, n2, T=100, max_iter=200)
            c5_s = compute_c5(h2)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h2.copy()

    # Phase 3: Diversification - try random starts at n=640
    if time.time() - start_time < budget_s - 100:
        for trial in range(3):
            if time.time() - start_time > budget_s - 50:
                break
            h_r = random_binary_h(n1, rng)
            h_r, _ = fista_optimize(h_r, 40, 3000, start_time, budget_s)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 4: Coordinate descent polish
    n_final = len(best_h)
    max_cd_rounds = 30 if n_final <= 640 else 15
    best_h, best_c5 = coord_descent(best_h, n_final, start_time, budget_s, max_rounds=max_cd_rounds)

    # Final validation
    best_h = project_h(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
