def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def compute_c5_grad(h, dx):
    """Compute value and gradient of C5 objective."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    k_star = np.argmax(corr)
    offset = k_star - (n - 1)
    val = np.max(corr) * dx
    
    grad = np.zeros(n)
    for m in range(n):
        j1 = m - offset  # from (1-h[m-offset]) term
        j2 = m + offset  # from h[m+offset] term
        if 0 <= j1 < n:
            grad[m] += (1 - h[j1])
        if 0 <= j2 < n:
            grad[m] += -h[j2]
    
    return val, dx * grad

def repair_sum(h, target):
    """Repair h to have sum = target, keeping 0 <= h <= 1."""
    h = np.clip(h, 0, 1)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_and_grad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    k_star = int(np.argmax(corr))
    offset = k_star - (n - 1)
    val = np.max(corr) * dx
    grad = np.zeros(n)
    for m in range(n):
        j1 = m - offset
        j2 = m + offset
        if 0 <= j1 < n:
            grad[m] += (1.0 - h[j1])
        if 0 <= j2 < n:
            grad[m] -= h[j2]
    return val, dx * grad

def repair_sum(h, target):
    h = np.clip(h.astype(float), 0, 1)
    for _ in range(300):
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        if s < 1e-15:
            h = np.full(len(h), target / len(h))
            break
        h = h * (target / s)
        h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-11:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0, 1)
    return h

def thue_morse(n):
    seq = [0]
    while len(seq) < n:
        seq = seq + [1 - x for x in seq]
    return np.array(seq[:n], dtype=float)

def optimize_candidate(h_init, n, dx, target, maxiter=400):
    def obj(x):
        return compute_c5_and_grad(x, dx)
    def cfun(x):
        return np.sum(x) - target
    def cjac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        return result
    except Exception:
        return None

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_score = float('inf')
    best_h = None
    best_n = 0

    candidates = []

    if 'initial_h_values' in dir() and initial_h_values is not None and len(initial_h_values) > 0:
        candidates.append(np.array(initial_h_values, dtype=float).copy())

    for n_tm in [64, 128, 256]:
        candidates.append(thue_morse(n_tm))

    for n_r in [64, 96, 128]:
        candidates.append((rng.rand(n_r) < 0.5).astype(float))

    candidates.append(np.ones(64) * 0.5)
    candidates.append(np.concatenate([np.ones(32), np.zeros(32)]))

    for h_init in candidates:
        if time.time() - t0 > budget_s - 40:
            break
        n = len(h_init)
        dx = 2.0 / n
        target = n / 2.0
        h_init = repair_sum(h_init, target)

        score_init = compute_c5(h_init, dx)
        if score_init < best_score:
            best_score = score_init
            best_h = h_init.copy()
            best_n = n

        time_left = budget_s - (time.time() - t0) - 30
        if time_left < 10:
            break

        maxiter = 200 if n <= 64 else (400 if n <= 128 else 600)
        result = optimize_candidate(h_init, n, dx, target, maxiter=maxiter)
        if result is not None and result.fun < best_score:
            best_score = result.fun
            best_h = repair_sum(result.x, target)
            best_n = n

    # Refine best solution with perturbation
    if best_h is not None and time.time() - t0 < budget_s - 60:
        n = best_n
        dx = 2.0 / n

        for _attempt in range(4):
            if time.time() - t0 > budget_s - 30:

        best_score = compute_c5(best_h, 2.0 / best_n)
