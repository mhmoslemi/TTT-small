    zeros_list = list(np.where(cur == 0)[0])

def swap_delta(h, a, b, n):
    """Compute the change in the full correlation vector when h[a]:1->0, h[b]:0->1."""
    # p = m - (n-1), so m = p + (n-1)
    # Delta for each p:
    # = -(1-h[a-p]) + h[a+p] + (1-h[b-p]) - h[b+p]
    # where h[x] = 0 for x outside [0, n-1]
    
    delta = np.zeros(2*n - 1)
    for p in range(-(n-1), n):
        d = 0.0
        # -(1 - h[a-p])
        ap = a - p
        if 0 <= ap < n:
            d -= (1.0 - h[ap])
        else:
            d -= 1.0  # h[a-p] = 0, so 1 - 0 = 1
        # + h[a+p]
        aq = a + p
        if 0 <= aq < n:
            d += h[aq]
        # + (1 - h[b-p])
        bp = b - p
        if 0 <= bp < n:
            d += (1.0 - h[bp])
        else:
            d += 1.0
        # - h[b+p]
        bq = b + p
        if 0 <= bq < n:
            d -= h[bq]
        
        m = p + n - 1
        delta[m] = d
    
    return delta

def resample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.clip(np.interp(x_new, x_old, h), 0, 1)

def binarize(h, n, rng):
    target = n // 2
    h_noisy = h + rng.random(n) * 1e-10
    idx = np.argsort(h_noisy)[::-1][:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)[:n//2]
    h[idx] = 1.0
    return h

def sa_optimize(h, n, time_limit, rng):
    start = time.time()
    cur = h.copy()
    c = np.correlate(cur, 1.0 - cur, mode="full")
    cur_max = float(np.max(c))
    best = cur.copy()
    best_max = cur_max
    
    p = np.arange(-(n-1), n)
    ones_list = np.where(cur == 1)[0].tolist()
    zeros_list = np.where(cur == 0)[0].tolist()
    
    T0 = 5.0
    T1 = 0.01
    half = 2.0 / n
    
    while time.time() - start < time_limit:
        frac = min(1.0, (time.time() - start) / time_limit)
        T = T0 * (T1 / T0) ** frac
        
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        
        a, b = i, j
        am = a - p
        ap = a + p
        bm = b - p
        bp = b + p
        
        delta = np.zeros(2*n-1)
        m1 = (am >= 0) & (am < n)
        m2 = (ap >= 0) & (ap < n)
        m3 = (bm >= 0) & (bm < n)
        m4 = (bp >= 0) & (bp < n)
        delta[m1] += cur[am[m1]]
        delta[m2] += cur[ap[m2]]
        delta[m3] -= cur[bm[m3]]
        delta[m4] -= cur[bp[m4]]
        
        c += delta
        new_max = float(np.max(c))
        
        if new_max < cur_max or (T > 1e-10 and rng.random() < np.exp(-(new_max - cur_max) / T)):
            cur[i] = 0
            cur[j] = 1
            cur_max = new_max
            oi = ones_list.index(i)
            oz = zeros_list.index(j)
            ones_list[oi] = j
            zeros_list[oz] = i
            if new_max < best_max:
                best_max = new_max
                best = cur.copy()
        else:
            c -= delta
    
    return best, best_max * half

    resolutions = [n if n % 2 == 0 else n + 1 for n in resolutions]
    resolutions = sorted(set(resolutions))

    total_weight = sum(n for n in resolutions)
    
    for n in resolutions:
        if tleft() < 30:
            break
        
        n_time = tleft() * (n / total_weight) * 0.9  # 90% of allocated
        n_time = max(10.0, min(n_time, tleft() * 0.5))
        
        # Start from resampled initial
        h_cont = resample(h_init, n)
        h_bin = binarize(h_cont, n, rng)
        
        c5_init = compute_c5(h_bin)
        if c5_init < best_c5:
            best_c5 = c5_init

                best_c5 = c5_sa2
                best_h = h_sa2.copy()
        
        total_weight -= n  # adjust for remaining
    
    return (best_h, best_c5, len(best_h))

pos_in_zeros[i] = oz  # i is now in zeros at position oz
pos_in_ones[i] = -1
pos_in_zeros[j] = -1
