    n = len(h)

    # Softmax weights

        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

def interp_up(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    shifted = np.clip(tau * (corr - cmax), -500.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad *= dx
    obj = dx * (cmax + np.log(s) / tau)
    return obj, grad


def adam_optimize(h, n, tau, lr, max_iters, deadline):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 200:
            lr *= 0.5
            m *= 0.5
            stall_count = 0
            if lr < 1e-6:
                break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_to_feasible(h, n)
    return best_h, best_obj


def subgrad_polish(h, n, max_iters, deadline, lr_init=0.005):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = int(np.argmax(corr))
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 + sh:j1 + sh])
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b - sh:j1b - sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 400:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
    return best_h, best_c5
