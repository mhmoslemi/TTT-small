Let me also think about the tau schedule more carefully. The smooth objective is:
f(h) = dx * (max_c + log(sum(exp(tau*(c - max_c))))/tau)

As tau → ∞, f(h) → dx * max_c = C5(h). So higher tau gives a better approximation to the true objective.

But higher tau also makes the gradient concentrate on the argmax shift, which can cause issues (the gradient becomes very sparse). A good schedule is to start moderate and increase.

I'll use: tau(t) = tau0 * (1 + t/T) where T is a time constant. Or exponentially: tau(t) = tau0 * exp(α*t).

Let me use: tau = tau_start * (tau_end/tau_start)^(progress) = tau_start * exp(progress * ln(tau_end/tau_start))

with progress = min(1, elapsed_time / total_planned_time).

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    p_rev = p[::-1]
    T1 = np.correlate(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(h, p_rev, mode='full')[n-1:2*n-1]
    grad = dx * (T1 - T2)
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def adam_phase(h, n, start_time, time_limit, tau_start, tau_end, lr_init):
    dx = 2.0 / n
    beta1, beta2 = 0.75, 0.99
    eps = 1e-8
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    best_c5_check = best_c5
    check_interval = 20

    while time.time() - start_time < time_limit:
        t += 1
        progress = min(1.0, (time.time() - start_time) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = lr_init * (1.0 - 0.7 * progress)

        obj, grad = smooth_obj_and_grad_fast(h, tau, n)

        if t % check_interval == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-13:
                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    lr *= 0.5
                    stall = 0
                if stall > 80:
                    break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def smooth_obj_and_grad_fast(h, tau, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    p_rev = p[::-1]
    T1 = np.correlate(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(h, p_rev, mode='full')[n-1:2*n-1]
    grad = dx * (T1 - T2)
    return 0.0, grad


def subgrad_phase(h, n, start_time, time_limit):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    lr = 0.004

    while time.time() - start_time < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = int(np.argmax(corr))
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1) - sh
            grad[j0:j1] += (1.0 - h[idx])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b) + sh
            grad[j0b:j1b] -= h[idx]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

        if stall > 200:
            lr *= 0.6
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def coord_descent_polish(h, n, start_time, time_limit):

    h = project_to_feasible(h, n).copy()

        j0 = max(0, sh)

        if j0 >= j1:

        idx = np.arange(j0, j1) - sh
        overlap_idx = idx

        improved = False
        for i in range(j0, j1):
