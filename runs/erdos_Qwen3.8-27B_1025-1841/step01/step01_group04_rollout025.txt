```python
def softmax_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_term = np.exp(scaled - m)
    w = exp_term / np.sum(exp_term)  # length 2n-1
    
    lags = np.arange(-(n-1), n)  # length 2n-1, index k corresponds to lag k-(n-1)
    
    # A[j] = sum_k w[k] * x[j-lags[k]] (x=0 outside [0,n-1])
    # B[j] = sum_k w[k] * x[j+lags[k]] (x=0 outside [0,n-1])
    j_idx = np.arange(n)[:, None]  # n x 1
    A_idx = j_idx - lags[None, :]  # n x (2n-1)
    B_idx = j_idx + lags[None, :]
    
    A_valid = (A_idx >= 0) & (A_idx < n)
    B_valid = (B_idx >= 0) & (B_idx < n)
    
    A_idx_c = np.clip(A_idx, 0, n-1)
    B_idx_c = np.clip(B_idx, 0, n-1)
    
    A = (x[A_idx_c] * A_valid.astype(float)) @ w
    B = (x[B_idx_c] * B_valid.astype(float)) @ w
    
    # Prefix sum term: for each j, sum of w[k] where 0 <= j-lags[k] <= n-1
    # i.e., j-(n-1) <= lags[k] <= j
    # lags[k] = k-(n-1), so j-(n-1) <= k-(n-1) <= j => j <= k <= j+n-1
    # k ranges from 0 to 2n-2, so k from max(0,j) to min(2n-2, j+n-1)
    # = sum_{k=max(0,j)}^{min(2n-2,j+n-1)} w[k]
    
    # Precompute prefix sums of w
    w_prefix = np.concatenate([[0], np.cumsum(w)])  # length 2n
    k_lo = np.maximum(0, np.arange(n))  # max(0, j)
    k_hi = np.minimum(2*n-2, np.arange(n) + n - 1)  # min(2n-2, j+n-1)
    prefix_sum = w_prefix[k_hi+1] - w_prefix[k_lo]
    
    grad = dx * (prefix_sum - A - B)
    return grad
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)

    # d(corr[k])/dh[j] = I[j-k+(n-1) in [0,n-1]] - x[j-k+(n-1)] - x[j+k-(n-1)]

        if 0 <= idx1 <= n-1:
            val -= h[idx1]
        if 0 <= idx2 <= n-1:
            val -= h[idx2]
        sg[j] = val
    
    return sg * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_term = np.exp(scaled - m)
    w = exp_term / np.sum(exp_term)
    
    lags = np.arange(-(n - 1), n)
    j_idx = np.arange(n)[:, None]
    A_idx = j_idx - lags[None, :]
    B_idx = j_idx + lags[None, :]
    
    A_valid = (A_idx >= 0) & (A_idx < n)
    B_valid = (B_idx >= 0) & (B_idx < n)
    
    A_idx_c = np.clip(A_idx, 0, n - 1)
    B_idx_c = np.clip(B_idx, 0, n - 1)
    
    A = (x[A_idx_c].astype(np.float64) * A_valid) @ w
    B = (x[B_idx_c].astype(np.float64) * B_valid) @ w
    
    w_prefix = np.concatenate([[0.0], np.cumsum(w)])
    j_arr = np.arange(n)
    k_lo = np.maximum(0, j_arr)
    k_hi = np.minimum(2 * n - 2, j_arr + n - 1)
    prefix_sum = w_prefix[k_hi + 1] - w_prefix[k_lo]
    
    return dx * (prefix_sum - A - B)

def project_h(h, n):
    target = n / 2.0
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def proj_gd_anneal(h0, n, T_list, deadline, max_iter_per_T=2000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for T in T_list:
        if time.time() > deadline:
            break
        
        lr = 1.0
        f_curr = softmax_obj(h, n, T)
        
        for it in range(max_iter_per_T):
            if time.time() > deadline - 5:
                break
            
            g = softmax_grad(h, n, T)
            
            # Backtracking line search
            step = lr
            improved = False
            for _ in range(30):
                h_trial = h - step * g
                h_trial = project_h(h_trial, n)
                f_trial = softmax_obj(h_trial, n, T)
                if f_trial < f_curr - 1e-15:
                    h = h_trial
                    f_curr = f_trial
                    improved = True
                    break
                step *= 0.5
            
            if not improved:
                break
            
            c5_trial = evaluate_c5(h)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
            
            # Adaptive learning rate
            if improved:
                lr = min(lr * 1.05, 5.0)
            else:
                lr = max(lr * 0.5, 0.01)
    
    return best_h, best_c5

def subgrad_polish(h, n, n_steps, deadline, lr_init=0.05):
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        
        j = np.arange(n)
        idx1 = j - lag
        idx2 = j + lag
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        
        h_at_1 = np.zeros(n)
        h_at_2 = np.zeros(n)
        h_at_1[valid1] = h[idx1[valid1]]
        h_at_2[valid2] = h[idx2[valid2]]
        
        sg = valid1.astype(np.float64) - h_at_1 - h_at_2
        sg *= dx
        
        # Decay learning rate
        lr = lr_init / (1.0 + 0.05 * step)
        
        h_new = h - lr * sg
        h_new = project_h(h_new, n)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 20
    
    # Get starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h_start = np.full(100, 0.5)
    
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Try multiple resolutions
    n_start = len(h_start)
    resolutions = sorted(set([n_start, 504, 672]))
    
    h_work = h_start.copy()
    
    for n in resolutions:
        if time.time() > deadline - 60:
            break
        
        if n == len(h_work):
            h_n = h_work.copy()
        else:
            h_n = interp_to(h_work, len(h_work), n)
            h_n = project_h(h_n, n)
        
        # Phase 1: Annealed projected gradient
        T_list_early = [20.0, 50.0, 100.0]
        T_list_late = [200.0, 500.0]
        
        h_n, c5_n = proj_gd_anneal(h_n, n, T_list_early, deadline, max_iter_per_T=1500)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n.copy()
            h_work = h_n.copy()
        
        if time.time() < deadline - 120:
            h_n, c5_n = proj_gd_anneal(h_n, n, T_list_late, deadline, max_iter_per_T=2000)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()
                h_work = h_n.copy()
        
        # Phase 2: Subgradient polishing
        if time.time() < deadline - 30:
            h_n, c5_n = subgrad_polish(h_n, n, n_steps=5000, deadline=deadline, lr_init=0.08)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()
                h_work = h_n.copy()
    
    # Random restarts from best solution
    n_best = len(best_h)
    rng = np.random.RandomState(seed)
    
    for attempt in range(20):
        if time.time() > deadline - 30:
            break
        
        # Structured perturbation
        scale = 0.05 + 0.1 * (attempt % 3) / 3.0
        h_pert = best_h + rng.randn(n_best) * scale
        h_pert = project_h(h_pert, n_best)
        
        h_new, c5_new = proj_gd_anneal(h_pert, n_best, [50.0, 200.0], deadline, max_iter_per_T=800)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final subgradient polish on best
    if time.time() < deadline - 10:
        best_h, best_c5 = subgrad_polish(best_h, len(best_h), n_steps=3000, deadline=deadline, lr_init=0.03)
    
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
