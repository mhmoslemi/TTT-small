import time

def project_onto_feasible(h, target_sum):
    """Project h onto {0<=h[i]<=1, sum(h)=target_sum}"""
    n = len(h)
    # Binary search for tau
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_clipped = np.clip(h - mid, 0, 1)
        s = h_clipped.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_clipped = np.clip(h - (lo+hi)/2, 0, 1)
    return h_clipped

    """Compute C5 = max_k corr_k * dx"""
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return (corr * dx).max(), corr * dx

    n = len(h)

def project(h, target_sum):
    lo, hi = -2.0, 2.0

def grad_all_lags(h, lags):
    """Compute sum of gradients for given lags"""
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        k = abs(lag)
        if k == 0:
            grad += 1.0 - 2.0 * h
        elif lag > 0:
            idx = np.arange(n - k)
            grad[idx] += (1.0 - h[idx + k])
            grad[idx + k] += (-h[idx])
        else:
            idx = np.arange(n - k)
            grad[idx + k] += (1.0 - h[idx])
            grad[idx] += (-h[idx + k])
    return grad
