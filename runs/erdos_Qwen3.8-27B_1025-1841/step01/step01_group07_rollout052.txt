import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def local_search_binary(h, dx, max_iter=50000, time_limit=None, start_time=None):
    n = len(h)
    best = compute_c5(h, dx)
    current = h.copy()
    current_c5 = best
    
    for it in range(max_iter):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        # Pick random positions to swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or current[i] == current[j]:
            continue
        # Swap
        current[i], current[j] = current[j], current[i]
        new_c5 = compute_c5(current, dx)
        if new_c5 < current_c5 or np.random.random() < np.exp(-max(0, new_c5 - current_c5) / 0.01):
            current_c5 = new_c5
            if new_c5 < best:
                best = new_c5
                best_h = current.copy()
        else:
            # Undo
            current[i], current[j] = current[j], current[i]
    
    return best_h if 'best_h' in dir() else current, best

    n = len(h)
    current = h.copy().astype(float)
    current_val = compute_c5(current, dx)
    best = current.copy()
    best_val = current_val
    T = 0.015

        if time.time() > time_limit:
