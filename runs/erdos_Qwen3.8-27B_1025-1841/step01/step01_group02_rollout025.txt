            return g

    mc = np.max(top_corr)
    ts = t * (top_corr - mc)
    ts -= np.max(ts)
    w = np.exp(ts)
    w_sum = np.sum(w)
    w /= w_sum
    val = (mc + np.log(w_sum) / t) * dx
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi < 1e-14:
            continue
        m = idx - (n - 1)
        if m >= 0:
            grad[:n - m] += wi * (1.0 - h[m:])
            grad[m:] += -wi * h[:n - m]
        else:
            p = -m
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] += -wi * h[p:]
    return val, grad * dx

def aug_lag_optimize(h0, n, t_schedule, rho=8.0, time_limit=120):
    start = time.time()
    lam = 0.0
    h = h0.copy()
    max_act = min(200, 2 * n - 1)
    for t_val in t_schedule:
        if time.time() - start > time_limit:
            break
        def _obj(x):
            v, _ = smooth_obj_grad(x, n, t_val, max_act)
            s = np.sum(x) - n / 2.0
            return v + lam * s + 0.5 * rho * s * s
        def _grad(x):
            _, g = smooth_obj_grad(x, n, t_val, max_act)
            s = np.sum(x) - n / 2.0
            return g + (lam + rho * s)
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'gtol': 1e-9, 'maxfun': 800})
            h = res.x
        except Exception:
            pass
        s = np.sum(h) - n / 2.0
        lam += rho * s
        if abs(s) < 1e-6:
            lam = 0.0
    return make_feasible(h, n)

def pair_cd(h, n, time_limit=100, rng=None, sa_temp=0.0):
    if rng is None:
        rng = np.random.RandomState(42)
    start = time.time()
    best = compute_c5(h)
    h_best = h.copy()
    deltas = [0.02, 0.05, 0.1, -0.02, -0.05, -0.1, 0.01, -0.01, 0.15, -0.15]
    while time.time() - start < time_limit:
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        for d in deltas:
            t_j = h[j] + d
            t_k = h[k] - d
            if t_j < 0 or t_j > 1 or t_k < 0 or t_k > 1:
                continue
            trial = h.copy()
            trial[j] = t_j
            trial[k] = t_k
            c = compute_c5(trial)
            if c < best - 1e-12:
                best = c
                h = trial
                h_best = trial.copy()
                break
            elif sa_temp > 0 and c < best + sa_temp and rng.rand() < np.exp(-(c - best) / sa_temp):
                h = trial
                h_best = trial.copy()
                break
    return h_best, best

def snap_cd(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(99)
    start = time.time()
    best = compute_c5(h)
    h_best = h.copy()
    while time.time() - start < time_limit:
        j = rng.randint(n)
        for target in [0.0, 1.0, 0.5]:
            if abs(h[j] - target) < 0.01:
                continue
            trial = h.copy()
            trial[j] = target
            trial = make_feasible(trial, n)
            if np.sum(np.abs(trial - h)) < 1e-10:
                continue
            c = compute_c5(trial)
            if c < best - 1e-12:
                best = c
                h = trial
                h_best = trial.copy()
                break
    return h_best, best

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None:
        h0 = make_feasible(rng.rand(320), 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Optimize at original resolution with annealed L-BFGS-B
    n0 = len(h0)
    t_sched1 = [60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0]
    h1 = aug_lag_optimize(h0, n0, t_sched1, rho=8.0, time_limit=150)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Pair CD at original resolution
    h2, c2 = pair_cd(best_h, n0, time_limit=80, rng=rng)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2

    # Phase 3: Snap CD
    h3, c3 = snap_cd(best_h, n0, time_limit=40, rng=rng)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3

    # Phase 4: Higher resolutions
    for n_target in [480, 640, 960]:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 40
        if remaining < 180:
            break

        h_up = upsample_h(best_h, n_target)
        t_sched = [50.0, 30.0, 18.0, 12.0, 8.0, 5.0, 4.0]
        opt_time = min(int(remaining * 0.4), 250)
        h_opt = aug_lag_optimize(h_up, n_target, t_sched, rho=8.0, time_limit=opt_time)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

        # Pair CD at this resolution
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 25
        if remaining > 60:
            cd_time = min(int(remaining * 0.5), 200)
            h_cd, c_cd = pair_cd(best_h, len(best_h), time_limit=cd_time, rng=rng, sa_temp=0.005)
            if c_cd < best_c5:
                best_c5 = c_cd
                best_h = h_cd.copy()

        # Snap CD
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 15
        if remaining > 30:
            h_sn, c_sn = snap_cd(best_h, len(best_h), time_limit=min(remaining, 60), rng=rng)
            if c_sn < best_c5:
                best_c5 = c_sn
                best_h = h_sn.copy()

    # Phase 5: Final intensive pair CD with simulated annealing
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 10
    if remaining > 40:
        h_final, c_final = pair_cd(best_h, len(best_h), time_limit=min(remaining, 200),
                                    rng=rng, sa_temp=0.002)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
