    return np.max(np.correlate(h_flat, 1-h_flat, mode='full')) * dx

def make_feasible(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum"""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = target_sum - np.sum(h)
    if diff > 0:
        # Need to increase some values
        mask = h < 1
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    elif diff < 0:
        mask = h > 0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def interpolate_to_n(parent_h, new_n):
    """Interpolate parent_h (n_parent points) to new_n points"""
    n_parent = len(parent_h)
    x_old = np.linspace(0, 2, n_parent, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    f = interp1d(x_old, parent_h, kind='linear', fill_value='extrapolate')
    new_h = f(x_new)
    new_h = np.clip(new_h, 0, 1)
    return make_feasible(new_h, new_n / 2)

    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return make_feasible(h, n / 2)

∫₀² -ε²f(x)f(x+k) dx = -ε² * autocorr_f(k)

So the overlap ≈ 1/2 - ε² * autocorr_f(k)

To make this < 1/2, we need autocorr_f(k) > 0 for all k. That means f should be "positively self-correlating" at all lags!

But wait, for a mean-zero function, the integral of f(x)f(x+k) over all k gives ∫f²(x) (by Parseval, it's the sum of squares of Fourier coefficients). And the integral at k=0 is ∫f². So the autocorrelation is a positive definite function. Its integral over all k is ∫f² > 0. But individual values can be negative!

Actually, for a positive definite function, it can be negative at some lags. E.g., sin has negative autocorrelation at some lags.

So to get C₅ < 1/2, we need -ε² * autocorr_f(k) < 0 for the worst k, meaning autocorr_f(k) > 0 for that k. But we need this for ALL k (since we're taking the max).

Hmm, but we also have the boundary effects (x+k might go outside [0,2]).
