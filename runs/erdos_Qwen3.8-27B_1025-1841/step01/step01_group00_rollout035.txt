def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

            break
        corr = compute_corr(h)
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if topM > 0 and topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p_top = exp_c / s
            grad = compute_gradient_topM(h, p_top, top_idx, n, dx)
        else:
            max_g = g.max()
            shifted = np.clip(tau * (g - max_g), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p = exp_c / s
            grad = compute_gradient_full(h, p, n, dx)

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        h = h_new

    return best_h, best_obj


def subgrad_optimize(h, n, max_iters, time_limit, lr_init=0.005, topK=3):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    patience = 400

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK

        cur_lr = lr / (1.0 + it * 0.005)
        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        h = h_new

    return best_h, best_c5


def averaged_subgrad(h, n, max_iters, time_limit, lr_init=0.01, topK=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    h_sum = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    count = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK

        cur_lr = lr_init / (1.0 + it * 0.02)
        h_new = project_to_feasible(h - cur_lr * grad, n)
        h_sum += h_new
        count += 1

        if count % 200 == 0 and count > 200:
            h_avg = project_to_feasible(h_sum / count, n)
            c5_avg = compute_c5(h_avg)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = h_avg.copy()
        h = h_new

    if count > 0:
        h_avg = project_to_feasible(h_sum / count, n)
        c5_avg = compute_c5(h_avg)
        if c5_avg < best_c5:
            best_c5 = c5_avg
            best_h = h_avg.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase 1: Optimize at initial resolution with increasing tau
    n1 = n_init
    if remaining() > 100:
        for tau in [8.0, 20.0, 40.0]:
            if remaining() < 30:
                break
            tl = min(remaining() * 0.22, 150.0)
            h1, _ = project_grad_optimize(best_h, n1, tau, max_iters=15000,
                                           time_limit=tl, lr_init=0.15, topM=0)
            c5_1 = compute_c5(h1)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()

    # Phase 2: Double resolution
    n2 = n_init * 2
    if remaining() > 100:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        for tau in [15.0, 40.0, 80.0]:
            if remaining() < 30:
                break
            tl = min(remaining() * 0.22, 180.0)
            h2, _ = project_grad_optimize(h2, n2, tau, max_iters=10000,
                                           time_limit=tl, lr_init=0.12, topM=200)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 3: Quadruple resolution
    n3 = n_init * 4
    if remaining() > 80:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        for tau in [20.0, 60.0, 100.0]:
            if remaining() < 25:
                break
            tl = min(remaining() * 0.25, 200.0)
            h3, _ = project_grad_optimize(h3, n3, tau, max_iters=6000,
                                           time_limit=tl, lr_init=0.1, topM=100)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()

    # Phase 4: Subgradient polish at best resolution
    if remaining() > 40:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = subgrad_optimize(h_f, n_f, max_iters=80000,
                                      time_limit=remaining() * 0.3, lr_init=0.003, topK=3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Averaged subgradient
    if remaining() > 30:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = averaged_subgrad(h_f, n_f, max_iters=80000,
                                      time_limit=remaining() * 0.3, lr_init=0.008, topK=5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 6: Random perturbations to escape local minima
    if remaining() > 40:
        for attempt in range(5):
            if remaining() < 20:
                break
            n_p = len(best_h)
            noise_scale = 0.02 * (attempt + 1)
            h_p = best_h + noise_scale * np.random.randn(n_p)
            h_p = project_to_feasible(h_p, n_p)
            tl = min(remaining() * 0.15, 60.0)
            h_p, _ = project_grad_optimize(h_p, n_p, tau=40.0,
                                            max_iters=3000, time_limit=tl,
                                            lr_init=0.1,
                                            topM=min(100, 2 * n_p - 1))
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 7: Final subgradient polish
    if remaining() > 15:
        n_final = len(best_h)
        h_final = best_h.copy()
        h_final, c5_final = subgrad_optimize(h_final, n_final, max_iters=50000,
                                              time_limit=remaining() * 0.4, lr_init=0.001, topK=3)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
