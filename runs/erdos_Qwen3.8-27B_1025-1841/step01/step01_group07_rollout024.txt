from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def _proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _resample(h, n1, n2):
    x1 = (np.arange(n1) + 0.5) * 2.0 / n1
    x2 = (np.arange(n2) + 0.5) * 2.0 / n2
    h2 = np.interp(x2, x1, h)
    return _proj(h2, n2 * 0.5)

def _obj(h, dx):
    return _c5(h, dx)

def _con(h, tsum):
    return h.sum() - tsum

def _con_jac(h, tsum):
    return np.ones_like(h)

def _optimize(h0, n, dx, tsum, time_limit):
    """Optimize from h0 using SLSQP. Returns (h, c5) or None."""
    t0 = time.time()

    cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac, 'args': (tsum,)}]
    
    for ftol in [1e-8, 1e-10]:
        try:
            res = minimize(_obj, h0.copy(), args=(dx,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': ftol})
            h = _proj(res.x, tsum)
            c5 = _c5(h, dx)
            return h, c5

            pass
    return None

                    h_pol, c5_pol = _local_polish(h_opt, n, dx, tsum)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol
                        best_n = n
                    else:
                        best_c5 = c5_opt
                        best_h = h_opt
                        best_n = n
            except Exception:
                pass

    # Final polish on best
    if best_h is not None and time.time() - start_time < budget_s - 10:
        dx_best = 2.0 / best_n
        h_pol, c5_pol = _local_polish(best_h, best_n, dx_best, best_n * 0.5, n_rounds=10)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    return best_h, best_c5, best_n
