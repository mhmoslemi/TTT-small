    """Compute soft-max objective and gradient with sum penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = T * dx * corr
    m = np.max(s)
    exp_s = np.exp(s - m)
    Z = np.sum(exp_s)
    p = exp_s / Z  # softmax weights, shape (2n-1,)
    val = (m + np.log(Z)) / T
    
    # Add penalty for sum constraint
    penalty_val = penalty * (np.sum(h) - n/2.0)**2
    penalty_grad = 2 * penalty * (np.sum(h) - n/2.0)
    
    # Compute gradient of soft-max part
    # ∂val/∂h[m] = dx * Σ_k p[k] * ∂corr[k]/∂h[m]
    # ∂corr[k]/∂h[m] = (1-h[m+shift[k]]) * 1[0≤m+shift[k]<n] - h[m-shift[k]] * 1[0≤m-shift[k]<n]
    # where shift[k] = k-(n-1)
    
    # Use vectorized computation
    shift = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    m_idx = np.arange(n)  # shape (n,)
    
    # idx1[m,k] = m + shift[k], valid if 0<=idx1<n
    idx1 = m_idx[:, None] + shift[None, :]  # (n, 2n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx1_safe = np.clip(idx1, 0, n-1)
    M1 = np.where(valid1, 1.0 - h[idx1_safe], 0.0)
    
    # idx2[m,k] = m - shift[k], valid if 0<=idx2<n
    idx2 = m_idx[:, None] - shift[None, :]  # (n, 2n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx2_safe = np.clip(idx2, 0, n-1)
    M2 = np.where(valid2, h[idx2_safe], 0.0)
    
    # grad_softmax = dx * (M1 - M2) @ p
    grad = dx * ((M1 - M2).dot(p)) + penalty_grad
    val += penalty_val
    
    return val, grad

import numpy as np

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_softobj_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = T * dx * corr
    m = np.max(s)
    es = np.exp(s - m)
    Z = np.sum(es)
    p = es / Z
    obj = (m + np.log(Z)) / T + mu * (np.sum(h) - n / 2.0) ** 2
    
    shift = np.arange(2 * n - 1) - (n - 1)
    mi = np.arange(n)
    idx1 = mi[:, None] + shift[None, :]
    v1 = (idx1 >= 0) & (idx1 < n)
    i1s = np.clip(idx1, 0, n - 1)
    M1 = np.where(v1, 1.0 - h[i1s], 0.0)
    
    idx2 = mi[:, None] - shift[None, :]
    v2 = (idx2 >= 0) & (idx2 < n)
    i2s = np.clip(idx2, 0, n - 1)
    M2 = np.where(v2, h[i2s], 0.0)
    
    g = dx * (M1 - M2).dot(p) + 2.0 * mu * (np.sum(h) - n / 2.0)
    return obj, g

def lfbgsh_opt(h0, n, T, mu, maxiter, deadline):
    def fun(x):
        return compute_softobj_grad(x, T, mu)[0]
    def jac(x):
        return compute_softobj_grad(x, T, mu)[1]
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8, 'maxfun': maxiter * 3})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_h(res.x, n)
    except Exception:
        pass
    return h0
