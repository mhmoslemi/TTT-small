        alpha = 6.0 / np.sqrt(n)
        no_improve = 0
        for it in range(n_sg):
            if time.time() > deadline - 60:
                break
            corr = compute_corr(h_curr)
            max_val = np.max(corr)
            # Use multiple active lags
            thresh = max_val - 0.05 * max_val
            active = np.where(corr >= thresh)[0]
            grad = np.zeros(n)
            for k in active:
                grad += single_lag_grad(h_curr, n, k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            grad /= gnorm
            h_new = project_target(h_curr - alpha * grad, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5_curr - 1e-13:
                h_curr = h_new
                c5_curr = c5_new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 100:
                    alpha *= 0.85
                    no_improve = 0
            if it % 500 == 499:
                alpha *= 0.92

        # Phase 3: Random restarts with SLSQP (only for smaller n)
        if n <= 280 and time.time() < t0 + budget_s * 0.55:
            for ri in range(5):
                if time.time() > deadline - 120:
                    break
                scale = 0.04 * (1 + ri * 0.5)
                h_pert = project_target(h_curr + rng.randn(n) * scale, n)
                temp = 0.01
                bnds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': lambda x, t=target_sum: np.sum(x) - t,
                         'jac': lambda x: np.ones(n)}]
                try:
                    res = minimize(slsqp_objective, h_pert, method='SLSQP',
                                   bounds=bnds, constraints=cons,
                                   jac=lambda x: slsqp_gradient(x, dx, temp),
                                   args=(dx, temp),
                                   options={'maxiter': 200, 'ftol': 1e-13, 'iprint': 0})
                    if np.isfinite(res.fun):
                        h_new = project_target(res.x, n)
                        c5_new = compute_c5(h_new, dx)
                        if c5_new < c5_curr:
                            h_curr = h_new
                            c5_curr = c5_new
                except Exception:
                    pass

        # Phase 4: Fine-tuning with very low temperature SLSQP
        if time.time() < t0 + budget_s * 0.7:
            for temp in [0.001, 0.0005, 0.0002]:
                if time.time() > deadline - 60:
                    break
                bnds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': lambda x, t=target_sum: np.sum(x) - t,
                         'jac': lambda x: np.ones(n)}]
                try:
                    res = minimize(slsqp_objective, h_curr, method='SLSQP',
                                   bounds=bnds, constraints=cons,
                                   jac=lambda x: slsqp_gradient(x, dx, temp),
                                   args=(dx, temp),
                                   options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
                    if np.isfinite(res.fun):
                        h_new = project_target(res.x, n)
                        c5_new = compute_c5(h_new, dx)
                        if c5_new < c5_curr - 1e-14:
                            h_curr = h_new
                            c5_curr = c5_new
                except Exception:
                    pass

        # Update global best
        h_curr = project_target(h_curr, n)
        c5_curr = compute_c5(h_curr, dx)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

    # Also try optimizing at a lower resolution for shape, then upsample
    if time.time() < t0 + budget_s * 0.7:
        n_low = max(70, n0 // 2)
        if n_low < n0:
            h_down = upsample(best_h, n_low)
            dx_low = 2.0 / n_low
            c5_low = compute_c5(h_down, dx_low)
            bnds = [(0.0, 1.0)] * n_low
            target_low = n_low / 2.0
            cons = [{'type': 'eq', 'fun': lambda x, t=target_low: np.sum(x) - t,
                     'jac': lambda x: np.ones(n_low)}]
            for temp in [0.05, 0.01, 0.005]:
                if time.time() > deadline - 90:
                    break
                try:
                    res = minimize(slsqp_objective, h_down, method='SLSQP',
                                   bounds=bnds, constraints=cons,
                                   jac=lambda x: slsqp_gradient(x, dx_low, temp),
                                   args=(dx_low, temp),
                                   options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
                    if np.isfinite(res.fun):
                        h_down = project_target(res.x, n_low)
                except Exception:
                    pass
            # Upsample back and check
            for n_up in [n0, 2 * n0]:
                if time.time() > deadline - 60:
                    break
                h_up = upsample(h_down, n_up)
                dx_up = 2.0 / n_up
                c5_up = compute_c5(h_up, dx_up)
                if c5_up < best_c5:
                    # Refine at this resolution
                    bnds = [(0.0, 1.0)] * n_up
                    target_u = n_up / 2.0
                    cons = [{'type': 'eq', 'fun': lambda x, t=target_u: np.sum(x) - t,
                             'jac': lambda x: np.ones(n_up)}]
                    for temp in [0.02, 0.005, 0.001]:
                        if time.time() > deadline - 30:
                            break
                        try:
                            res = minimize(slsqp_objective, h_up, method='SLSQP',
                                           bounds=bnds, constraints=cons,
                                           jac=lambda x: slsqp_gradient(x, dx_up, temp),
                                           args=(dx_up, temp),
                                           options={'maxiter': 150, 'ftol': 1e-14, 'iprint': 0})
                            if np.isfinite(res.fun):
                                h_up = project_target(res.x, n_up)
                        except Exception:
                            pass
                    c5_final = compute_c5(h_up, dx_up)
                    if c5_final < best_c5:
                        best_c5 = c5_final
                        best_h = h_up.copy()
                        best_n = n_up

    # Final verification
    best_h = project_target(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
