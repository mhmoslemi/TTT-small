def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    return np.correlate(h, 1.0 - h, mode='full')

    fa = np.fft.fft(a)
    fa_rev = np.fft.fft(a_rev)
    fb = np.fft.fft(b)
    fh = np.fft.fft(hp)
    T1 = np.fft.ifft(fa * fb).real[:n]
    T2 = np.fft.ifft(fa_rev * fh).real[:n]
    grad = dx * (T1 - T2)
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr


def optimize_smooth(h, n, tau, time_limit, lr_init=0.4):
    h = project_to_feasible(h, n).copy()
    obj, grad, _ = obj_grad_fft(h, tau)
    best_obj = obj
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    eval_count = 0
    while time.time() - start < time_limit:
        obj, grad, _ = obj_grad_fft(h, tau)
        eval_count += 1
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        obj_new, _, _ = obj_grad_fft(h_new, tau)
        eval_count += 1
        if obj_new > best_obj + 2e-3:
            lr *= 0.6
            v *= 0.4
        else:
            h = h_new
        if stall > 150:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-11:
                break
    return best_h


def subgrad_opt(h, n, time_limit, lr_init=0.008, diversify=False):
    h = project_to_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    rng = np.random.RandomState(12345)
    while time.time() - start < time_limit:
        corr = compute_corr(h)
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if diversify and stall % 50 == 0:
            top_k = np.argsort(corr)[-20:]
            k_star = top_k[rng.randint(len(top_k))]
        else:
            k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += 1.0 - h[np.arange(j0, j1) - sh]
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[np.arange(j0b, j1b) + sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 400:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-13:
                break
    return best_h, best_c5


def perturb_and_reopt(base_h, n, tau, time_limit, lr_init, seed=77):
    rng = np.random.RandomState(seed)
    scale = 0.05
    h_p = base_h + rng.randn(n) * scale
    h_p = project_to_feasible(h_p, n)
    return optimize_smooth(h_p, n, tau, time_limit, lr_init)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    n1 = 1800
    h1 = interp_periodic(best_h, n1)
    h1 = project_to_feasible(h1, n1)

    if rem() > 150:
        tl = min(280.0, rem() * 0.32)
        h1 = optimize_smooth(h1, n1, tau=10.0, time_limit=tl, lr_init=0.4)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    if rem() > 120:
        tl = min(250.0, rem() * 0.3)
        h2 = optimize_smooth(best_h, n1, tau=60.0, time_limit=tl, lr_init=0.12)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    if rem() > 100:
        tl = min(180.0, rem() * 0.28)
        h3 = optimize_smooth(best_h, n1, tau=250.0, time_limit=tl, lr_init=0.04)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    if rem() > 120:
        tl = min(150.0, rem() * 0.2)
        h4 = perturb_and_reopt(best_h, n1, tau=80.0, time_limit=tl, lr_init=0.15, seed=77)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    if rem() > 80:
        tl = min(180.0, rem() * 0.3)
        h5, c5_5 = subgrad_opt(best_h, n1, time_limit=tl, lr_init=0.006, diversify=True)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    if rem() > 120:
        n2 = 2700
        h6 = interp_periodic(best_h, n2)
        h6 = project_to_feasible(h6, n2)
        tl = min(180.0, rem() * 0.35)
        h6 = optimize_smooth(h6, n2, tau=60.0, time_limit=tl, lr_init=0.12)
        c5_6 = compute_c5(h6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        nr = len(best_h)
        if rem() > 50:
            h7, c5_7 = subgrad_opt(best_h, nr, time_limit=min(80.0, rem() * 0.3), lr_init=0.004, diversify=True)
            if c5_7 < best_c5:
                best_c5 = c5_7
                best_h = h7.copy()

    if rem() > 30:
        nr = len(best_h)
        h8, c5_8 = subgrad_opt(best_h, nr, time_limit=min(60.0, rem() * 0.5), lr_init=0.002, diversify=False)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
