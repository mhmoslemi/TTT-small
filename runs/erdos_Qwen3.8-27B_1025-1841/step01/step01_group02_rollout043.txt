def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(4):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-8
        else:
            mask = h > 1e-8
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = T * dx
    mc = np.max(corr)
    exp_shifted = np.exp(s * (corr - mc))
    log_sum = s * mc + np.log(exp_shifted.sum())
    p = exp_shifted / exp_shifted.sum()
    obj = (mc + np.log(exp_shifted.sum()) / s) * dx
    k = np.arange(-(n-1), n)
    ii = np.arange(n)[:, None]
    j1 = ii + k[None, :]
    v1 = (j1 >= 0) & (j1 < n)
    j1c = np.clip(j1, 0, n-1)
    first = (p[None, :] * (1.0 - h[j1c]) * v1).sum(axis=1)
    j2 = ii - k[None, :]
    v2 = (j2 >= 0) & (j2 < n)
    j2c = np.clip(j2, 0, n-1)
    second = (p[None, :] * h[j2c] * v2).sum(axis=1)
    grad = dx * dx * (first - second)
    return obj, grad

            h_pert, _ = pgd_solve(h_pert, 60, time_limit=min(remaining()-60, 80.0), init_step=0.012)
            h_pert = make_feasible(h_pert, n0)
            h_pert, c5_p = fast_coord_descent(h_pert, max_rounds=10, time_limit=min(remaining()-40, 40.0))
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pert.copy()

    # Phase 4: Binary local search
    if remaining() > 50:
        h_bin, c5_bin = binary_local_search(best_h, time_limit=min(remaining()-30, 60.0))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 5: Try higher resolution
    if remaining() > 120:
        n_high = 640
        h_high = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, n0), best_h)
        h_high = make_feasible(h_high, n_high)
        h_high, _ = pgd_solve(h_high, 50, time_limit=min(remaining()-80, 100.0), init_step=0.01)
        h_high = make_feasible(h_high, n_high)
        h_high, c5_high = fast_coord_descent(h_high, max_rounds=15, time_limit=min(remaining()-60, 60.0))
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high

    # Final: coordinate descent on best
    if remaining() > 20:
        best_h, best_c5 = fast_coord_descent(best_h, max_rounds=10, time_limit=remaining()-10)

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
