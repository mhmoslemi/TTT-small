import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_grad(h, n, T):
    dx = 2.0 / n
    oh = 1.0 - h
    corr = np.correlate(h, oh, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    shifted = np.where(shifted < -50.0, -50.0, shifted)
    exp_vals = np.exp(T * shifted)
    exp_sum = np.sum(exp_vals)
    obj = (mc + np.log(exp_sum) / T) * dx
    weights = exp_vals / exp_sum

    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-13:
            continue
        off = k - (n - 1)
        i_lo = max(0, -off)
        i_hi = min(n - 1, n - 1 - off)
        i_range = np.arange(i_lo, i_hi + 1)
        idx = i_range + off
        grad[i_range] += w * (1.0 - h[idx])
        grad[idx] += -w * h[i_range]
    grad *= dx
    return obj, grad

def adam_optimize(h0, n, T, max_steps, lr, budget_s, start_time, reserve=60):
    h = h0.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    obj_curr = compute_c5(h)
    b1, b2, eps = 0.9, 0.999, 1e-8
    for it in range(max_steps):
        if time.time() - start_time > budget_s - reserve:
            break
        T_curr = T * (1.0 + 0.3 * np.sin(it * 0.005))
        obj, grad = smooth_obj_grad(h, n, T_curr)
        t += 1
        lr_t = lr / (1.0 + 0.0005 * t)
        m_arr = b1 * m_arr + (1 - b1) * grad
        v_arr = b2 * v_arr + (1 - b2) * grad * grad
        m_hat = m_arr / (1 - b1**t)
        v_hat = v_arr / (1 - b2**t)
        h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = make_feasible(h, n)
    return h

def binary_local_search(h, n, budget_s, start_time, reserve=60, max_rounds=5):
    best = h.copy()
    best_c5 = compute_c5(best)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - reserve:
            break
        improved = False
        idx_order = np.arange(n)
        np.random.RandomState(42 + rnd).shuffle(idx_order)
        for j in idx_order:
            if time.time() - start_time > budget_s - reserve:
                break
            for target_val in [0.0, 1.0]:
                if abs(best[j] - target_val) < 1e-9:
                    continue
                trial = best.copy()
                trial[j] = target_val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best = trial
                    best_c5 = c
                    improved = True
        if not improved:
            break
    return best, best_c5

def coord_descent_polish(h, n, budget_s, start_time, reserve=30, max_rounds=20):
    best = h.copy()
    best_c5 = compute_c5(best)
    steps = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - reserve:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - reserve:
                break
            ov = best[j]
            for s in steps:
                cand = ov + s
                if cand < 0 or cand > 1:
                    continue
                trial = best.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best = trial
                    best_c5 = c
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

            return smooth_obj_grad(hv, n2, 120.0)[1]

            h_lb = make_feasible(res.x, n2)
