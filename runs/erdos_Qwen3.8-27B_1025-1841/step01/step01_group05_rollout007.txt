import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, N):
    return np.max(compute_corr(h)) * (2.0 / N)

def vectorized_grad(h, N, k):
    grad = np.zeros(N)
    if k >= 0:
        grad[k:] += (1.0 - h[:N - k])
        grad[:N - k] -= h[k:]
    else:
        ka = -k
        grad[:N - ka] += (1.0 - h[ka:])
        grad[ka:] -= h[:N - ka]
    return grad

def project_feasible(h, N):
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.maximum(0.0, h[mask] + step)
    return h

def optimize_subgradient(h, N, time_limit, rng, max_iter=300):
    dx = 2.0 / N
    h = project_feasible(h.copy(), N)
    best_h = h.copy()
    best_c5 = np.max(compute_corr(h)) * dx
    stall = 0
    lr_scale = 1.0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = compute_corr(h)
        max_val = np.max(corr)
        c5 = max_val * dx
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            lr_scale = max(0.1, 1.0 - stall * 0.01)
        threshold = max_val - 0.05 * max_val if max_val > 1e-8 else 0
        active = np.where(corr >= max(0, threshold))[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(N)
        for idx in active:
            k = idx - (N - 1)
            grad += vectorized_grad(h, N, k)
        grad /= len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        lr = 0.3 * lr_scale / gnorm
        h_new = project_feasible(h - lr * grad, N)
        h = h_new
    return best_h, best_c5

def local_binary_search(h, N, time_limit, rng, max_rounds=50):
    dx = 2.0 / N
    h = h.copy()
    best_c5 = np.max(compute_corr(h)) * dx
    for rnd in range(max_rounds):

            break

        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        trial_count = min(len(ones) * len(zeros), 200)
        for _ in range(trial_count):
            if time.time() > time_limit:
                break
            i1 = ones[rng.integers(len(ones))]
            i2 = zeros[rng.integers(len(zeros))]
            h[i1] = 0
            h[i2] = 1

                improved = True

                zeros = np.where(h == 0)[0]
            else:
                h[i1] = 1
                h[i2] = 0
        if not improved:
            break
    best_c5 = min(best_c5, np.max(compute_corr(h)) * dx)
    return h, best_c5
