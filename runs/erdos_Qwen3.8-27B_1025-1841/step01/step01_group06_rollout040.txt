from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-14:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_target(h.copy(), n_new)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def corr_gradient(h, n, lag):
    """Gradient of correlation at given lag with respect to h"""
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        # h[i]*(1-h[i+lag]) for i in [0, m-1]
        grad[:m] += (1.0 - h[lag:lag+m])
        grad[lag:lag+m] -= h[:m]
    else:
        p = -lag
        m = n - p
        # h[i+p]*(1-h[i]) for i in [0, m-1]
        grad[p:p+m] += (1.0 - h[:m])
        grad[:m] -= h[p:p+m]
    return grad

def smooth_grad(h, n, temp):
    """Smoothed objective gradient using log-sum-exp"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Avoid overflow
    shifted = (corr - m) / temp
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-12:
            continue
        lag = k_idx - (n - 1)
        grad += w[k_idx] * corr_gradient(h, n, lag)
    
    obj = (m + temp * np.log(w_sum)) 
    return obj, grad

where ∇f is the gradient of the smoothed objective and proj is the projection onto our feasible set.

def fista_smooth(h_init, n, dx, max_iters, t0, deadline, temp=0.02):
    x = h_init.copy()
    z = h_init.copy()
    t = 1.0
    # Step size: L is the Lipschitz constant of the gradient
    # For the smoothed objective, L ≈ 2*(n/dx) * (1/temp) ... this is hard to compute
    # Let me use a backtracking line search instead
    
    step = 1.0 / (0.5 * n)  # Initial guess
    best_c5 = compute_c5(x, dx)
    best_h = x.copy()
    
    old_c5 = best_c5
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # FISTA momentum
        t_old = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t_old - 1.0) / t) * (x - z)
        y = np.clip(y, 0.0, 1.0)  # Keep in bounds for gradient eval
        
        # Compute gradient at y
        obj_y, grad_y = smooth_obj_grad(y, n, temp)
        
        # Take step
        x_new = project_target(y - step * grad_y, n)
        c5_new = compute_c5(x_new, dx)
        
        if c5_new > obj_y * dx + 1.0:  # Too large step
            step *= 0.5
            x_new = project_target(y - step * grad_y, n)
            c5_new = compute_c5(x_new, dx)
        else:
            step = min(step * 1.5, 1.0 / (0.1 * n))
        
        z = x
        x = x_new
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decrease temperature
        if it % 500 == 499:
            temp *= 0.8
            step *= 0.7
    
    return best_h, best_c5
