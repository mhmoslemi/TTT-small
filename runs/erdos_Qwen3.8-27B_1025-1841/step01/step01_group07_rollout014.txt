def smooth_max_grad(h, T, n):
    dx = 2.0/n
    lag = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # For each lag l and index k:
    # d c_{l+(n-1)} / d h[k] = (1-h[k-l])*[0<=k-l<n] - h[k+l]*[0<=k+l<n]
    
    k = np.arange(n)  # shape (n,)
    # k - lag: shape (2n-1, n) with lag in rows, k in cols
    kl = k[None, :] - lag[:, None]  # k - lag, shape (2n-1, n)
    klp = k[None, :] + lag[:, None]  # k + lag, shape (2n-1, n)
    
    valid1 = (kl >= 0) & (kl < n)  # 0 <= k-lag < n
    valid2 = (klp >= 0) & (klp < n)  # 0 <= k+lag < n
    
    # (1-h[k-lag]) where valid
    h_kl = np.zeros_like(kl, dtype=float)
    for idx in np.where(valid1):
        pass  # This is slow, let me vectorize
    
    # Actually: h_kl[j,k] = h[k - lag[j]] if valid, 0 otherwise
    # We can use: create an array where h_kl[j,k] = h[k-lag[j]]
    # k-lag[j] = k[None,:] - lag[:,None] = kl
    # So h_kl[j,k] = h[kl[j,k]] when kl[j,k] is in [0,n)
    h_idx = np.clip(kl.astype(int), 0, n-1)
    h_kl = h[h_idx] * valid1  # shape (2n-1, n)
    
    h_idx_p = np.clip(klp.astype(int), 0, n-1)
    h_klp = h[h_idx_p] * valid2  # shape (2n-1, n)
    
    dcdh = (1.0 - h_kl) * valid1 - h_klp * valid2  # shape (2n-1, n)
    
    # Now compute soft weights
    corr = np.correlate(h, 1.0-h, mode='full')  # shape (2n-1,)
    vals = corr * dx
    mx = vals.max()
    exp_vals = np.exp(T * (vals - mx))
    w = exp_vals / exp_vals.sum()  # shape (2n-1,)
    
    # Gradient: df/dh[k] = dx * sum_j w[j] * dcdh[j,k]
    grad = dx * (w[:, None] * dcdh).sum(axis=0)  # shape (n,)
    
    # Add penalty gradient
    target = n/2.0
    pen_coeff = 1e3
    grad += 2 * pen_coeff * (h.sum() - target)
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    d = target - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        room = 1.0 - h
        idx = np.argsort(-room)
        rem = d
        for i in idx:
            a = min(rem, room[i])
            h[i] += a
            rem -= a
            if rem < 1e-14:
                break
    else:
        rem = -d
        idx = np.argsort(h)
        for i in idx:
            a = min(rem, h[i])
            h[i] -= a
            rem -= a
            if rem < 1e-14:
                break
    return h

def _slsqp_opt(h0, n, iters=80):
    target = n / 2.0
    dx = 2.0 / n
    def f(h):
        return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    def c_eq(h):
        return h.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_eq}]
    res = minimize(f, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': iters, 'ftol': 1e-12})
    return _project(res.x, target)

def _smooth_grad(h, T, n, pen_coeff=1e3):
    dx = 2.0 / n
    lag = np.arange(2 * n - 1) - (n - 1)
    k = np.arange(n)
    kl = k[None, :] - lag[:, None]
    klp = k[None, :] + lag[:, None]
    valid1 = (kl >= 0) & (kl < n)
    valid2 = (klp >= 0) & (klp < n)
    h_idx = np.clip(kl.astype(np.int64), 0, n - 1)
    h_kl = h[h_idx] * valid1
    h_idx_p = np.clip(klp.astype(np.int64), 0, n - 1)
    h_klp = h[h_idx_p] * valid2
    dcdh = (1.0 - h_kl) * valid1 - h_klp * valid2
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    mx = vals.max()
    exp_v = np.exp(T * (vals - mx))
    s = exp_v.sum()
    if s < 1e-300:
        w = np.zeros(2 * n - 1)
        w[mx == vals] = 1.0 / np.sum(mx == vals)
    else:
        w = exp_v / s
    grad = dx * (w[:, None] * dcdh).sum(axis=0)
    target = n / 2.0
    grad += 2.0 * pen_coeff * (h.sum() - target)
    return grad

def _lbfgsb_opt(h0, n, T=60.0, iters=200):
    target = n / 2.0
    pen_coeff = 1e3
    dx = 2.0 / n
    def f(h):
        hc = np.clip(h, 0, 1)
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        vals = corr * dx
        mx = vals.max()
        ev = np.exp(T * (vals - mx))
        sm = (1.0 / T) * np.log(ev.sum()) + mx
        return sm + pen_coeff * (hc.sum() - target) ** 2
    def df(h):
        hc = np.clip(h, 0, 1)
        return _smooth_grad(hc, T, n, pen_coeff)
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h0, method='L-BFGS-B', bounds=bounds, jac=df,
                   options={'maxiter': iters, 'maxls': 80, 'ftol': 1e-15})
    return _project(res.x, target)

def _binary_local_search(h0, n, max_sweeps=40, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    h = h0.copy()
    idx_s = np.sort(h)
    thresh = idx_s[n // 2]
    h[h >= thresh] = 1.0
    h[h < thresh] = 0.0
    ones = int(np.sum(h == 1))
    if ones > n // 2:
        oi = np.where(h == 1)[0]
        rng.shuffle(oi)
        h[oi[:ones - n // 2]] = 0.0
    elif ones < n // 2:
        zi = np.where(h == 0)[0]
        rng.shuffle(zi)
        h[zi[:n // 2 - ones]] = 1.0
    cur = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    for _ in range(max_sweeps):
        improved = False
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        rng.shuffle(oi)
        rng.shuffle(zi)
        for i in oi:
            for j in zi:
                h[i] = 0.0
                h[j] = 1.0
                nm = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                if nm < cur - 1e-10:
                    cur = nm
                    improved = True
                    zi = np.where(h == 0)[0]
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    return h

def _resample(h_parent, n_new):
    m = len(h_parent)
    x_old = np.arange(m, dtype=float)
    x_new = np.linspace(0, m - 1, n_new)
    h_int = np.interp(x_new, x_old, h_parent)
    return _project(h_int, n_new / 2.0)

def _generate_inits(n, rng, parent_h):
    target = n / 2.0
    inits = []
    if parent_h is not None:
        inits.append(_resample(parent_h, n))
        for ph in [0.3, 0.7]:
            x = np.linspace(ph, len(parent_h) - 1 + ph, n)
            h_ph = np.interp(x % len(parent_h), np.arange(len(parent_h), dtype=float), parent_h)
            inits.append(_project(h_ph, target))
    for _ in range(2):
        inits.append(_project(rng.random(n), target))
    for _ in range(2):
        idx = rng.permutation(n)
        hb = np.zeros(n)
        hb[idx[:n // 2]] = 1.0
        inits.append(hb)
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    inits.append(_project(tm, target))
    for bf in [0.25, 0.4]:
        b = int(n * bf)
        hb2 = np.zeros(n)
        hb2[:b] = 1.0
        hb2[n // 2:n // 2 + b] = 1.0
        inits.append(_project(hb2, target))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError, TypeError):
        parent_h = None

    if parent_h is not None and len(parent_h) > 0:
        best_h = parent_h.copy()
        best_c5 = _compute_c5(best_h)
        best_n = len(best_h)

    def improve(h):
        nonlocal best_h, best_c5, best_n
        c = _compute_c5(h)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
            best_n = len(h)

    configs = [
        (48, ['slsqp', 'lbfgsb', 'binary']),
        (64, ['slsqp', 'lbfgsb', 'binary']),
        (96, ['slsqp', 'lbfgsb', 'binary']),
        (128, ['lbfgsb', 'binary'], 2),
        (160, ['lbfgsb', 'binary'], 2),
        (192, ['lbfgsb', 'binary'], 2),
        (256, ['lbfgsb', 'binary'], 2),
    ]

    for cfg in configs:
        n = cfg[0]
        methods = cfg[1]
        max_inits = cfg[2] if len(cfg) > 2 else 4
        if remaining() < 50:
            break
        inits = _generate_inits(n, rng, parent_h)
        for hi in inits[:max_inits + 3]:
            if remaining() < 25:
                break
            for method in methods:
                if remaining() < 20:
                    break
                try:
                    if method == 'slsqp':
                        it = max(20, min(80, int(remaining() // 4)))
                        h_opt = _slsqp_opt(hi, n, iters=it)
                    elif method == 'lbfgsb':
                        h_opt = _lbfgsb_opt(hi, n, T=60.0, iters=200)
                    elif method == 'binary':
                        h_opt = _binary_local_search(hi, n, max_sweeps=30, rng=rng)
                    else:
                        continue
                    improve(h_opt)
                except Exception:
                    pass

    # Re-sample best to higher resolution and refine
    for n_up in [128, 192, 256]:
        if remaining() < 60:
            break
        if best_n >= n_up:
            continue
        h_res = _resample(best_h, n_up)
        for _ in range(2):
            if remaining() < 30:
                break
            try:
                h_opt = _lbfgsb_opt(h_res, n_up, T=80.0, iters=300)
                improve(h_opt)
                h_res = h_opt
            except Exception:
                pass
            try:
                h_opt = _slsqp_opt(h_res, n_up, iters=40)
                improve(h_opt)
            except Exception:
                pass

    # Final polish
    if best_h is not None and remaining() > 20:
        try:
            nf = best_n
            h_opt = _slsqp_opt(best_h, nf, iters=50)
            improve(h_opt)
        except Exception:
            pass

    if best_h is None:
        n = 64
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = _compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
