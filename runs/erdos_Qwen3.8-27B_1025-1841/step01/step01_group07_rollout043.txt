def project_onto_constraints(h, n_half):
    """Project h onto {0 <= h <= 1, sum(h) = n_half}"""
    # Water-filling / bisection approach
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_projected = np.clip(h - mid, 0, 1)
        s = h_projected.sum()
        if s > n_half:
            lo = mid
        else:
            hi = mid
    h_projected = np.clip(h - (lo+hi)/2, 0, 1)
    return h_projected

def compute_c5(h, dx):
    """Compute C5 = max of correlation * dx"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def soft_max_c5(h, dx, tau):
    """Smooth approximation of C5 using log-sum-exp"""
    corr = np.correlate(h, 1-h, mode="full") * dx
    # Shift to avoid overflow
    m = np.max(corr)
    return (1.0/tau) * m + (1.0/tau) * np.log(np.sum(np.exp(tau * (corr - m))))

        s = h_proj.sum()
        if s > target_sum:

        else:
            hi = mid

    n = len(h)

def project_onto_constraints(h, target_sum):

    return float(np.max(corr) * dx)
