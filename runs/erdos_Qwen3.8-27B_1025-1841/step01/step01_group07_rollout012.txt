def interp_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)

def rand_h(n, seed):
    rng = np.random.RandomState(seed)
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for _ in range(rng.randint(1, 4)):

        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.1, 0.35)
        h += amp * np.sin(2.0 * np.pi * freq * t + phase)
    h = np.clip(h, 0.0, 1.0)
    return fix_sum(h, n)

    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return compute_c5(x, dx)

    def cons_fun(x):
        return np.sum(x) - target

    res = minimize(obj, h0, method='SLSQP', bounds=bounds,

                   options={'maxiter': maxiter, 'ftol': 1e-10})
