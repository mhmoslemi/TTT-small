```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - float(np.sum(h))
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def project(g, n):
    target = n / 2.0
    lo = float(np.min(g)) - 1.0
    hi = float(np.max(g)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(g - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(g - lam, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tdx = T * dx
    scaled = Tdx * corr
    max_s = float(np.max(scaled))
    return (max_s + float(np.log(np.sum(np.exp(scaled - max_s))))) / T

def smooth_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tdx = T * dx
    scaled = Tdx * corr
    max_s = float(np.max(scaled))
    exp_s = np.exp(scaled - max_s)
    p = exp_s / float(np.sum(exp_s))
    N = n
    grad = np.zeros(N)
    for d in range(-(N-1), N):
        w = p[d + N - 1]
        if w < 1e-14:
            continue
        i_lo = max(0, d)
        i_hi = min(N-1, N-1+d)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi+1)
            jdx = idx - d
            grad[idx] += w * (1.0 - h[jdx])
        i_lo2 = max(0, -d)
        i_hi2 = min(N-1, N-1-d)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2+1)
            jdx2 = idx2 + d
            grad[idx2] -= w * h[jdx2]
    grad *= dx
    return grad

def fista(h_init, n, T, max_iter, time_limit, start_t):
    h = h_init.copy()
    y = h.copy()
    t_k = 1.0
    step = 1.0
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        grad = smooth_grad(y, n, T)
        z = y - step * grad
        x_new = project(z, n)
        if np.max(np.abs(x_new - h)) < 1e-12:
            h = x_new
            break
        h = x_new
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = h + ((t_k - 1.0) / t_new) * (h - x_new + step * grad)
        t_k = t_new
    return h

def fista_bt(h_init, n, T, max_iter, time_limit, start_t):
    h = h_init.copy()
    y = h.copy()
    t_k = 1.0
    step = 2.0
    obj_cur = smooth_obj(h, n, T)
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        grad = smooth_grad(y, n, T)
        z = y - step * grad
        x_new = project(z, n)
        obj_new = smooth_obj(x_new, n, T)
        model = obj_cur + float(np.dot(grad, x_new - y)) + 0.5 * step * float(np.dot(x_new - y, x_new - y))
        while obj_new > model + 1e-8:
            step *= 0.5
            z = y - step * grad
            x_new = project(z, n)
            obj_new = smooth_obj(x_new, n, T)
            model = obj_cur + float(np.dot(grad, x_new - y)) + 0.5 * step * float(np.dot(x_new - y, x_new - y))
        if obj_new < obj_cur:
            step = min(step * 1.5, 5.0)
        obj_cur = obj_new
        old_h = h.copy()
        h = x_new
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = h + ((t_k - 1.0) / t_new) * (h - old_h)
        t_k = t_new
        if np.max(np.abs(h - old_h)) < 1e-11:
            break
    return h

def slsqp_refine(h_init, n, T, time_limit, start_t):
    dx = 2.0 / n
    def obj_fn(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        Tdx = T * dx
        scaled = Tdx * corr
        max_s = float(np.max(scaled))
        return (max_s + float(np.log(np.sum(np.exp(scaled - max_s))))) / T
    def con_fun(h):
        return float(np.sum(h)) - n / 2.0
    def con_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
        if res.success or res.x is not None:
            return res.x
    except Exception:
        pass
    return h_init.copy()

def coord_descent(h, best_c5, time_limit, start_t):
    n = len(h)
    improved = True
    max_rounds = 20
    for rnd in range(max_rounds):
        if not improved:
            break
        if time.time() - start_t > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_t > time_limit:
                break
            old = h[j]
            best_j = old
            best_c = best_c5
            for cand in [0.0, 1.0, 0.25, 0.5, 0.75,
                         old + 0.02, old - 0.02,
                         old + 0.005, old - 0.005]:
                if cand < 0.0 or cand > 1.0:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-12:
                    best_c = c
                    best_j = cand
                    best_h = trial.copy()
            if best_j != old:
                h = best_h
                best_c5 = best_c
                improved = True
    return h, best_c5

def binary_swap_search(h, best_c5, time_limit, start_t):
    n = len(h)
    thresh = 0.4
    h_bin = np.where(h > thresh, 1.0, 0.0)
    h_bin = make_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin > best_c5 + 0.05:

        for i in range(n):

            if h_bin[i] > 0.5:

                    if h_bin[j] < 0.5:

                        h_bin[j] = 1.0

                        else:

            else:

                        else:
                            h_bin[i] = 0.0

    if best_c5 < compute_c5(h):
```
