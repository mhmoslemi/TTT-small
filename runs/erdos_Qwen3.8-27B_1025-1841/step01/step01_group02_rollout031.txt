    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # Softmax weights
    mc = np.max(corr)
    shifted = t * (corr - mc)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Objective value
    obj = (mc/t + np.log(np.sum(exp_shifted))/t) * dx
    
    # Gradient
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def precompute_grad_data(n):
    s_vals = np.arange(2 * n - 1) - n + 1
    i_vals = np.arange(n)
    idx1 = i_vals[:, None] - s_vals[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2 = i_vals[:, None] + s_vals[None, :]
    mask2 = (idx2 >= 0) & (idx2 < n)
    idx2c = np.clip(idx2, 0, n - 1)
    return idx1c, mask1, idx2c, mask2

def smooth_grad_values(h, t, grad_data):
    n = len(h)
    dx = 2.0 / n
    idx1c, mask1, idx2c, mask2 = grad_data
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = t * (corr - mc)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    term1 = np.where(mask1, 1.0 - h[idx1c], 0.0)
    term2 = np.where(mask2, h[idx2c], 0.0)
    return (term1 @ w - term2 @ w) * dx

def smooth_obj_val(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(t * shifted))) / t) * dx

def run_slsqp(h0, t, max_iter, grad_data):
    n = len(h0)
    
    def obj(x):
        return smooth_obj_val(x, t)
    
    def jac(x):
        return smooth_grad_values(x, t, grad_data)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', jac=jac,
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return res.x

def run_fista(h0, t, max_iter, grad_data, step_init=None):
    n = len(h0)
    dx = 2.0 / n
    if step_init is None:
        step_init = 0.1 / n  # rough Lipschitz estimate
    
    h = h0.copy()
    z = h0.copy()
    alpha = 1.0
    step = step_init
    
    for it in range(max_iter):
        g = smooth_grad_values(z, t, grad_data)
        y = z - step * g
        h_new = np.clip(y, 0.0, 1.0)
        # Project to satisfy sum constraint
        h_new = make_feasible(h_new, n)
        
        # Backtracking
        obj_new = smooth_obj_val(h_new, t)
        obj_z = smooth_obj_val(z, t)
        if obj_new > obj_z + 0.5 * step * np.dot(g, h_new - z):
            step *= 0.5
            y = z - step * g
            h_new = make_feasible(np.clip(y, 0.0, 1.0), n)
            obj_new = smooth_obj_val(h_new, t)
        
        alpha_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * alpha * alpha))
        z = h_new + ((alpha - 1.0) / alpha_new) * (h_new - z)
        z = np.clip(z, 0.0, 1.0)
        alpha = alpha_new
        h = h_new
    
    return make_feasible(h, n)

def binary_local_search(h, max_evals, rng):
    n = len(h)
    h_b = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_b == 1))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_b == 1)[0]
        to_flip = rng.choice(ones, n_ones - target, replace=False)
        h_b[to_flip] = 0.0
    elif n_ones < target:
        zeros = np.where(h_b == 0)[0]
        to_flip = rng.choice(zeros, target - n_ones, replace=False)
        h_b[to_flip] = 1.0
    
    best_c5 = compute_c5(h_b)
    best_h = h_b.copy()
    no_improve = 0
    evals = 0
    
    while evals < max_evals and no_improve < 5000:
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        trial = h_b.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        evals += 1
        c5 = compute_c5(trial)
        if c5 < best_c5 - 1e-14:
            h_b = trial
            best_c5 = c5
            best_h = trial.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def coord_descent(h, max_rounds, start_time, budget_s):
    n = len(h)
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 20:
            break
        improved = False
        for i in range(n):
            if time.time() - start_time > budget_s - 20:
                return h, best_c5
            old_val = h[i]
            best_local = old_val
            best_local_c5 = best_c5
            # Candidates
            cands = [0.0, 1.0, 0.5,
                     max(0.0, old_val - 0.15), min(1.0, old_val + 0.15),
                     max(0.0, old_val - 0.07), min(1.0, old_val + 0.07),
                     max(0.0, old_val - 0.03), min(1.0, old_val + 0.03)]
            for c in cands:
                trial = h.copy()
                trial[i] = c
                trial = make_feasible(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_local_c5:
                    best_local_c5 = c5
                    best_local = c
            if best_local != old_val:
                h[i] = best_local
                h = make_feasible(h, n)
                best_c5 = best_local_c5
                improved = True
        if not improved:
            break
    return h, best_c5
