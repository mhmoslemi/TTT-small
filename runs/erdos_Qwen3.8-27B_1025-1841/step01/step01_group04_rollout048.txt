    g = corr * dx
    scaled = g * T
    m = np.max(scaled)

            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def obj_softmax(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    s = c * T
    m = s.max()
    return float((m + np.log(np.sum(np.exp(s - m)))) / T)

def grad_softmax(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    s = c * T
    m = s.max()
    e = np.exp(s - m)
    w = e / e.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-18:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            if lag < n:
                g[lag:] += wi * (1.0 - h[:n - lag])
                g[:n - lag] -= wi * h[lag:]
        else:
            L = -lag
            g[:n - L] += wi * (1.0 - h[L:])
            g[L:] -= wi * h[:n - L]
    return g

def slsqp_opt(h0, T, maxiter=400):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return obj_softmax(x, T)

    def jac(x):
        return grad_softmax(x, T)

    def con_eq(x):
        return np.sum(x) - target

    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'ftol': 1e-14, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h0, n)

def proj_grad_anneal(h0, T, lr_init, n_steps, deadline):
    n = len(h0)
    h = h0.copy()
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = grad_softmax(h, T)
        f = obj_softmax(h, T)
        alpha = lr
        for _ in range(40):
            h_try = project_feasible(h - alpha * g, n)
            f_try = obj_softmax(h_try, T)
            if f_try < f - 1e-10:
                h = h_try
                f = f_try
                break
            alpha *= 0.5
        if step % 30 == 0 and step > 0:
            lr *= 0.85
        if step % 5 == 0:
            lr = max(lr, 1e-5)
    return h

def active_lag_opt(h0, n_active=5, maxiter=200):
    """Minimize top-n_active correlation terms directly."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-n_active:]
        return np.sum(c[idx]) / n_active

    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-n_active:]
        g = np.zeros(n)
        for idx_i in idx:
            lag = idx_i - (n - 1)
            if lag >= 0:
                if lag < n:
                    g[lag:] += dx / n_active * (1.0 - x[:n - lag])
                    g[:n - lag] -= dx / n_active * x[lag:]
            else:
                L = -lag
                g[:n - L] += dx / n_active * (1.0 - x[L:])
                g[L:] -= dx / n_active * x[:n - L]
        return g

    def con_eq(x):
        return np.sum(x) - target

    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'ftol': 1e-13, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h0, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n_start = len(h_start)

    # Phase 1: Multi-resolution SLSQP with temperature annealing
    n_list = sorted(set([n_start, 504, 672]))
    n_list = [n for n in n_list if n >= n_start]

    for n in n_list:
        if time.time() > deadline - 80:
            break
        if n == len(best_h):
            h_work = best_h.copy()
        else:
            h_work = project_feasible(resample(best_h, len(best_h), n), n)

        T_schedule = [30.0, 100.0, 400.0, 1500.0]
        for T in T_schedule:
            if time.time() > deadline - 40:
                break
            h_new = slsqp_opt(h_work, T, maxiter=500)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_work = h_new

        # Projected gradient refinement
        if time.time() < deadline - 60:
            n_steps = min(400, max(50, int((deadline - time.time() - 30) * 1.5)))
            h_pg = proj_grad_anneal(h_work, T=500.0, lr_init=0.003,
                                     n_steps=n_steps, deadline=deadline - 20)
            c5_pg = evaluate_c5(h_pg)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
            h_work = h_pg

        # Active lag optimization
        if time.time() < deadline - 50:
            h_al = active_lag_opt(h_work, n_active=8, maxiter=300)
            c5_al = evaluate_c5(h_al)
            if c5_al < best_c5:
                best_c5 = c5_al
                best_h = h_al.copy()
            h_work = h_al

    # Phase 2: Random restarts and perturbations at best resolution
    n_best = len(best_h)
    for attempt in range(80):
        if time.time() > deadline - 20:
            break
        scale = 0.03 + 0.04 * (attempt % 5)
        h_pert = project_feasible(best_h + rng.randn(n_best) * scale, n_best)
        T = 200.0 + 400.0 * (attempt % 4)
        h_new = slsqp_opt(h_pert, T, maxiter=300)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if attempt % 10 == 9 and time.time() < deadline - 30:
            h_al = active_lag_opt(h_new, n_active=5, maxiter=200)
            c5_al = evaluate_c5(h_al)
            if c5_al < best_c5:
                best_c5 = c5_al
                best_h = h_al.copy()

    # Phase 3: Structured random starts
    if time.time() < deadline - 40:
        for attempt in range(8):
            if time.time() > deadline - 20:
                break
            h_rand = project_feasible(0.5 + 0.3 * rng.randn(n_best), n_best)
            h_new = slsqp_opt(h_rand, T=100.0, maxiter=400)
            h_new2 = slsqp_opt(h_new, T=800.0, maxiter=300)
            c5_new = evaluate_c5(h_new2)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new2.copy()

    # Phase 4: Final high-T polish
    if time.time() < deadline - 15:
        h_final = slsqp_opt(best_h, T=3000.0, maxiter=500)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        h_pol = active_lag_opt(best_h, n_active=10, maxiter=300)
        c5_pol = evaluate_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
