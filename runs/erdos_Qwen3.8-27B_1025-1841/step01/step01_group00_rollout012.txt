For l >= 0: C(l) = sum_{i=0}^{n-1-l} h[i+l] * (1-h[i])
For l < 0: C(l) = sum_{i=-l}^{n-1} h[i+l] * (1-h[i])

  Contribution: -h[j+l]

    max_c = corr.max()

    active = np.where(p > thresh)[0]
    if len(active) > top_m:

        active = active[np.argsort(p[active])[-top_m:]]

    grad *= dx

    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, tau, top_m=60):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    thresh = max(1e-10, p.max() * 1e-5)
    active = np.where(p > thresh)[0]
    if len(active) > top_m:
        active = active[np.argsort(p[active])[-top_m:]]
    grad = np.zeros(n)
    for k in active:
        pk = p[int(k)]
        sh = int(k) - n + 1
        a0 = max(0, sh)
        a1 = min(n, n + sh)
        if a0 < a1:
            js = np.arange(a0, a1)
            grad[js] += pk * (1.0 - h[js - sh])
        b0 = max(0, -sh)
        b1 = min(n, n - sh)
        if b0 < b1:
            js = np.arange(b0, b1)
            grad[js] -= pk * h[js + sh]
    grad *= dx
    return obj, grad


def optimize_nesterov(h, n, tau_start, tau_end, top_m, max_iters, time_limit, lr0=0.3, beta=0.91):
    h = np.array(h, dtype=float)
    best_h = h.copy()
    best_obj = float('inf')
    start = time.time()
    v = np.zeros(n)
    lr = lr0
    stall = 0
    n_tau = 4
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        obj, grad = smooth_obj_grad(h, tau, top_m)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 60:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        elif stall == 0 and lr < lr0:
            lr = min(lr * 1.03, lr0)
    obj, _ = smooth_obj_grad(best_h, tau_end, top_m)
    if obj < best_obj:
        best_obj = obj
    return best_h, best_obj


def subgrad_polish(h, n, max_iters, time_limit, lr0=0.002):
    h = np.array(h, dtype=float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    v = np.zeros(n)
    beta = 0.82
    lr = lr0
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        corr_all = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr_all)[-3:]
        grad = np.zeros(n)
        for k in top_idx:
            sh = int(k) - n + 1
            a0 = max(0, sh)
            a1 = min(n, n + sh)
            if a0 < a1:
                js = np.arange(a0, a1)
                grad[js] += (1.0 - h[js - sh])
            b0 = max(0, -sh)
            b1 = min(n, n - sh)
            if b0 < b1:
                js = np.arange(b0, b1)
                grad[js] -= h[js + sh]
        grad *= (2.0 / n) / len(top_idx)
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 300:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break
    return best_h, best_c5


def multi_smooth_pass(h, n, time_limit, top_m=40, lr0=0.25):
    h = np.array(h, dtype=float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    tau_seq = [(40, 15), (25, 10), (15, 6)]
    for ts, te in tau_seq:
        tl_each = (time_limit - (time.time() - start)) / (len(tau_seq) + 1)
        if tl_each < 10:
            break
        h, _ = optimize_nesterov(h, n, ts, te, top_m, max_iters=5000, time_limit=tl_each, lr0=lr0)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    def rem():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase A: n=2000, multi-pass smooth
    if rem() > 250:
        nA = 2000
        hA = project_to_feasible(interp_periodic(best_h, nA), nA)
        hA, c5A = multi_smooth_pass(hA, nA, time_limit=rem() * 0.22, top_m=50, lr0=0.3)
        if c5A < best_c5:
            best_c5 = c5A
            best_h = hA.copy()

    # Phase B: n=3600, multi-pass smooth
    if rem() > 250:
        nB = 3600
        hB = project_to_feasible(interp_periodic(best_h, nB), nB)
        hB, c5B = multi_smooth_pass(hB, nB, time_limit=rem() * 0.22, top_m=40, lr0=0.2)
        if c5B < best_c5:
            best_c5 = c5B
            best_h = hB.copy()

    # Phase C: n=1000, fine-tune (faster per-iter, more iterations)
    if rem() > 150:
        nC = 1000
        hC = project_to_feasible(interp_periodic(best_h, nC), nC)
        hC, c5C = multi_smooth_pass(hC, nC, time_limit=rem() * 0.15, top_m=50, lr0=0.4)
        if c5C < best_c5:
            best_c5 = c5C
            best_h = hC.copy()

    # Phase D: Multi-restart at best resolution
    if rem() > 120:
        nD = len(best_h)
        for r in range(4):
            if rem() < 50:
                break
            hR = best_h + np.random.randn(nD) * 0.015
            hR = project_to_feasible(hR, nD)
            tl = (rem() - 50) / (4 - r)
            hR, _ = optimize_nesterov(hR, nD, 25, 8, top_m=30,
                                       max_iters=4000, time_limit=tl * 0.6, lr0=0.2)
            c5R = compute_c5(hR)
            if c5R < best_c5:
                best_c5 = c5R
                best_h = hR.copy()

    # Phase E: Subgradient polish (top-3 lags)
    if rem() > 40:
        nE = len(best_h)
        hE, c5E = subgrad_polish(best_h, nE, max_iters=50000,
                                  time_limit=rem() * 0.2, lr0=0.0015)
        if c5E < best_c5:
            best_c5 = c5E
            best_h = hE.copy()

    # Phase F: High-res final (n=4000) quick pass
    if rem() > 60:
        nF = 4000
        hF = project_to_feasible(interp_periodic(best_h, nF), nF)
        hF, _ = optimize_nesterov(hF, nF, 15, 6, top_m=30,
                                   max_iters=3000, time_limit=rem() * 0.25, lr0=0.15)
        c5F = compute_c5(hF)
        if c5F < best_c5:
            best_c5 = c5F
            best_h = hF.copy()
        if rem() > 25:
            hF, c5F2 = subgrad_polish(hF, nF, max_iters=20000,
                                       time_limit=rem() * 0.3, lr0=0.001)
            if c5F2 < best_c5:
                best_c5 = c5F2
                best_h = hF.copy()

    # Phase G: Final subgradient at best resolution
    if rem() > 20:
        nG = len(best_h)
        hG, c5G = subgrad_polish(best_h, nG, max_iters=40000,
                                  time_limit=rem() * 0.5, lr0=0.001)
        if c5G < best_c5:
            best_c5 = c5G
            best_h = hG.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
