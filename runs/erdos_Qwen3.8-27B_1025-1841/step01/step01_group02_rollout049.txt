        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):

    if new_n == old_n:
        return h.copy()

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n

    return make_feasible(h_new, new_n)

def softmax_obj_grad(h, T, penalty=1e8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_s = np.exp(T * shifted)
    log_sum = np.log(np.sum(exp_s))
    smooth_val = max_c + log_sum / T
    s = np.sum(h) - n / 2.0
    obj = smooth_val * dx + penalty * s * s
    
    # Gradient
    weights = exp_s / np.sum(exp_s)
    w_rev = weights[::-1]
    conv1 = np.convolve(weights, h, mode='full')
    conv2 = np.convolve(w_rev, h, mode='full')
    
    # Term A: for each j, sum of w[k] for k in [j, j+n-1] clipped to [0, 2n-2]
    # This is a sliding window sum
    term_a = np.zeros(n)
    cumsum_w = np.concatenate([[0.0], np.cumsum(weights)])

    grad = dx * grad_corr + 2.0 * penalty * s * np.ones(n)

def _single_lag_grad(h, m, n):
    grad = np.zeros(n)
    if m >= 0:
        idx1 = np.arange(n) - m
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[valid1] += (1.0 - h[idx1[valid1]])
        idx2 = np.arange(n) + m
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid2] -= h[idx2[valid2]]
    else:
        neg_m = -m
        idx2 = np.arange(n) + neg_m
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid2] -= h[idx2[valid2]]
        idx1 = np.arange(n) + neg_m
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[valid1] += (1.0 - h[idx1[valid1]])
