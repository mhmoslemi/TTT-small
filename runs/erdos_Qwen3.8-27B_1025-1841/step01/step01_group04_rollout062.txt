def compute_smooth_max_grad(h, n, T):
    dx = 2.0 / n
    # Compute corr
    v = 1.0 - h
    corr = np.correlate(h, v, 'full')  # length 2n-1
    # corr[k] corresponds to shift s = k - (n-1)
    
    # Softmax weights
    scaled = T * corr
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)
    p = exp_scaled / np.sum(exp_scaled)  # length 2n-1
    
    # grad[j] = dx * sum_s p[s] * ((1-h[j-s]) * [valid] - h[j+s] * [valid])
    # Let's compute directly
    grad = np.zeros(n)
    for s_idx in range(2*n-1):
        s = s_idx - (n-1)  # actual shift
        p_s = p[s_idx]
        # Term 1: (1-h[j-s]) for j-s in [0,n)
        # j ranges from max(0,s) to min(n-1, n-1+s)
        # j-s ranges from max(0,s)-s to min(n-1,n-1+s)-s
        lo1 = max(0, s)
        hi1 = min(n-1, n-1+s)
        if lo1 <= hi1:
            j_range = np.arange(lo1, hi1+1)
            grad[j_range] += p_s * (1.0 - h[j_range - s])
        
        # Term 2: -h[j+s] for j+s in [0,n)
        # j ranges from max(0,-s) to min(n-1, n-1-s)
        lo2 = max(0, -s)
        hi2 = min(n-1, n-1-s)
        if lo2 <= hi2:
            j_range = np.arange(lo2, hi2+1)
            grad[j_range] -= p_s * h[j_range + s]
    
    return grad * dx

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            break
        # Distribute evenly among free variables, but prioritize those
        # closest to the boundary in the right direction
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(scaled)

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = np.count_nonzero(m)
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = T * corr
    mx = np.max(sc)
    es = np.exp(sc - mx)
    p = es / es.sum()
    grad = np.zeros(n)
    for si in range(2 * n - 1):
        ps = p[si]
        if ps < 1e-13:
            continue
        s = si - (n - 1)
        lo = max(0, s)
        hi = min(n - 1, n - 1 + s)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            grad[j] += ps * (1.0 - h[j - s])
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            grad[j] -= ps * h[j + s]
    return grad * dx
