import time
import math

def evaluate_objective(h, n_points):
    dx = 2.0 / n_points
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0)

    s = np.sum(h)

    else:

        h = np.clip(h, 0.0, 1.0)
        # If clipping changed the sum, do a second pass
        s2 = np.sum(h)
        if abs(s2 - target) > 1e-6 and s2 > 0:
            h = h * (target / s2)
            h = np.clip(h, 0.0, 1.0)
    return h

    grad = np.zeros(n_points)
    for m in active_lags:

        g = np.zeros(n_points)
        if shift >= 0:
            g[:n_points - shift] += (1.0 - h[shift:])
            g[shift:] += (-h[:n_points - shift])

            s = -shift
            g[s:] += (1.0 - h[:n_points - s])
            g[:n_points - s] += (-h[s:])
        grad += g
    grad /= len(active_lags)
    return grad

    indices = (np.arange(n_new) * n_old / n_new).astype(int)

    patience = 0
    max_patience = 200

            patience = 0

            room = 1.0 - h
            tr = room.sum()
            if tr > 0:
                h = h + diff * room / tr
        else:
            th = h.sum()
            if th > 0:
                h = h + diff * h / th
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, n_points, active_lags):
    grad = np.zeros(n_points)
    for m in active_lags:
        shift = m - (n_points - 1)
        if shift >= 0:
            grad[:n_points - shift] += (1.0 - h[shift:])
            grad[shift:] += (-h[:n_points - shift])
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n_points - s])
            grad[:n_points - s] += (-h[s:])
    grad /= len(active_lags)
    return grad

def get_active_and_value(h, n_points, tol_frac):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = float(np.max(corr))
    active = np.where(corr >= mx - tol_frac * mx)[0]
    return active, mx

def scale_up(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.astype(float).copy()
    idx = np.round(np.arange(n_new) * n_old / float(n_new)).astype(int)
    idx = np.clip(idx, 0, n_old - 1)
    return h_old[idx].astype(float)

def optimize(h_init, n_points, rng, deadline, lr_init):
    h = project_to_feasible(h_init.astype(float).copy(), n_points)
    best_h = h.copy()
    best_val = evaluate_objective(h, n_points)
    lr = lr_init
    tol_frac = 0.02
    no_improve = 0
    while time.time() < deadline:
        active, cur_val = get_active_and_value(h, n_points, tol_frac)
        if cur_val < best_val - 1e-8:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 600:
            break
        grad = compute_gradient(h, n_points, active)
        noise_scale = max(0.0005, 0.008 * (1.0 - no_improve / 600.0))
        accepted = False
        for trial in range(3):
            ss = lr * (0.5 ** trial)
            nz = noise_scale * (0.5 ** trial) * rng.standard_normal(n_points)
            h_new = project_to_feasible(h - ss * grad + nz, n_points)
            new_val = evaluate_objective(h_new, n_points)
            if new_val < cur_val:
                h = h_new
                accepted = True
                break
            if trial == 2:
                delta = new_val - cur_val
                ann_t = max(0.0005, 0.005 * (1.0 - no_improve / 600.0))
                if delta > 0 and rng.random() < math.exp(-delta / ann_t):
                    h = h_new
                    accepted = True
        if no_improve % 25 == 0:
            lr *= 0.93
        if no_improve > 250:
            tol_frac = max(0.003, tol_frac * 0.96)
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def _rem():
        return budget_s - (time.time() - t0)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except (NameError, TypeError):
        n = 200
        h = np.full(n, 0.5)

    h = project_to_feasible(h, n)
    best_h = h.copy()
    best_c5 = evaluate_objective(h, n)

    # Phase 1: Optimize at base resolution
    if _rem() > 80:
        dl = t0 + min(100, _rem() * 0.12)
        h_opt, c5 = optimize(h, n, rng, dl, 0.015)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt

    # Phase 2: Scale up progressively
    n_try = n * 2
    while n_try <= 600 and _rem() > 150:
        h_scaled = project_to_feasible(scale_up(best_h, n_try), n_try)
        dl = t0 + min(_rem() - 60, 180)
        if dl - time.time() < 25:
            break
        h_opt, c5 = optimize(h_scaled, n_try, rng, dl, 0.01)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt
        n_try *= 2

    # Phase 3: Try different resolutions from parent
    for mult in [3, 4]:
        n_r = n * mult
        if n_r > 600 or _rem() < 80:
            break
        h_r = project_to_feasible(scale_up(best_h, n_r), n_r)
        dl = t0 + min(_rem() - 40, 120)
        if dl - time.time() < 20:
            break
        h_opt, c5 = optimize(h_r, n_r, rng, dl, 0.012)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt

    # Phase 4: Random restarts
    restart_seed = seed + 1
    while _rem() > 60:
        r_rng = np.random.default_rng(restart_seed)
        restart_seed += 1
        n_rand = int(r_rng.choice([n, n * 2, min(n * 3, 500), min(n * 4, 500)]))
        h_rand = project_to_feasible(0.5 + 0.2 * r_rng.standard_normal(n_rand), n_rand)
        dl = min(t0 + budget_s - 15, time.time() + min(40, _rem() - 25))
        if dl - time.time() < 15:
            break
        h_opt, c5 = optimize(h_rand, n_rand, rng, dl, 0.025)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt

    # Phase 5: Structured initializations
    if _rem() > 50:
        for freq in [1.0, 2.0, 3.0, 0.5, 4.0]:
            n_s = min(n * 2, 400)
            x = np.linspace(0, 2, n_s, endpoint=False)
            h_s = 0.5 + 0.25 * np.sin(freq * math.pi * x) + 0.1 * np.cos(2 * freq * math.pi * x)
            h_s = project_to_feasible(h_s, n_s)
            dl = min(t0 + budget_s - 10, time.time() + min(30, _rem() - 15))
            if dl - time.time() < 10:
                break
            h_opt, c5 = optimize(h_s, n_s, rng, dl, 0.018)
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt

    # Final polish at best resolution
    if _rem() > 20:
        dl = t0 + budget_s - 2
        h_opt, c5 = optimize(best_h, len(best_h), rng, dl, 0.003)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt

    best_c5 = evaluate_objective(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
