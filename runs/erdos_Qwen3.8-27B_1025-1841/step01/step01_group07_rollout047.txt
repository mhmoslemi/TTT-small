import numpy as np
from scipy.optimize import minimize
import time

_n_g = 0

def _smooth_max(x, t):
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp((x - x_max) * t))) / t

def _c5_smooth(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _smooth_max(corr, t) * dx

def _c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _sum_fun(h):
    return np.sum(h) - _n_g / 2.0

def _sum_jac(h):
    return np.ones(_n_g)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff2 = n / 2.0 - np.sum(h)
    if abs(diff2) > 1e-10:
        step = diff2 / n
        h += step
        h = np.clip(h, 0.0, 1.0)
    return h

def _local_refine(h, n, max_iter=200):
    h = _project(h, n)
    best_c = _c5_raw(h)
    dx = 2.0 / n
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_k = int(np.argmax(corr))
        m = worst_k - (n - 1)
        improved = False
        for i in range(n):
            j2 = i + m
            if j2 < 0 or j2 >= n:
                continue
            if h[i] <= 0:
                continue
            delta = -h[i] * 0.1
            h[i] += delta
            h[j2] -= delta
            h = _project(h, n)
            c_new = _c5_raw(h)
            if c_new < best_c:
                best_c = c_new
                improved = True
            else:
                h[i] -= delta
                h[j2] += delta
                h = _project(h, n)
        if not improved:
            break
    return h, best_c

def _thue_morse(n):
    h = np.zeros(n)
    for j in range(n):
        c = bin(j).count('1')
        h[j] = float(c % 2)
    return h

    h = rng.random(n)
    if np.sum(h) > 0:
        h *= (n / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    return _project(h, n)

def _binary_random(rng, n):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def _interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 1, n_src, endpoint=False)
    x_dst = np.linspace(0, 1, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return _project(h_dst, n_dst)

def _make_obj(n, temp):
    def obj(h):
        return _c5_smooth(h, temp)
    return obj

    except Exception:
        ih = None

    n_candidates = []
    if ih is not None:
        n_candidates.append(len(ih))
    n_candidates.extend([47, 64, 96, 128, 192, 256, 384, 512])
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time.time() > deadline - 60:
            break
        _n_g = n
        remaining = deadline - time.time()
        # Budget per n: larger n gets more time
        alloc = min(remaining, max(60, 1300.0 / len(n_candidates) * 1.5))

        constraints = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
        bounds = [(0.0, 1.0)] * n

        # Number of starts
        n_starts = max(3, int(alloc / 8.0))
        n_starts = min(n_starts, 40)

        for si in range(n_starts):
            if time.time() > deadline - 15:

            # Build starting point
            if si == 0 and ih is not None:
                h0 = _interpolate_to_n(ih, n)
            elif si == 1 and n >= 2:
                h0 = _thue_morse(n)
                h0 = _project(h0, n)
            elif si == 2 and n >= 2:
                h0 = _binary_random(rng, n)
            elif si == 3:

                h0 = np.zeros(n)
                period = max(2, n // 6)
                for j in range(n):
                    h0[j] = 1.0 if (j % period) < period // 2 else 0.0
                h0 = _project(h0, n)

                if c_check < best_c5:

    best_c5 = _c5_raw(best_h)
