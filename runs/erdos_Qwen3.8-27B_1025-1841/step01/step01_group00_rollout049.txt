                break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return max(0.0, budget_s - elapsed())

    h_parent = initial_h_values.copy()
    n_parent = len(h_parent)
    best_c5, _ = compute_c5(h_parent)
    best_h = h_parent.copy()

    # Phase 1: Quick polish at parent resolution (n=900)
    n1 = n_parent
    h1 = h_parent.copy()
    tl = remaining() * 0.08
    h1, c5_1 = opt_smooth(h1, n1, tau=25, max_iters=5000, time_limit=tl, lr0=0.25)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Main optimization at n=1800
    n2 = 2 * n_parent
    h2 = interp_periodic(best_h, n2)
    h2 = project_to_feasible(h2, n2)
    tl = remaining() * 0.35
    h2, c5_2a = opt_smooth(h2, n2, tau=15, max_iters=10000, time_limit=tl * 0.5, lr0=0.2)
    if remaining() > 10:
        h2, c5_2b = opt_smooth(h2, n2, tau=40, max_iters=8000, time_limit=tl * 0.5, lr0=0.1)
        if c5_2b < c5_2a:
            c5_2a = c5_2b
    if c5_2a < best_c5:
        best_c5 = c5_2a
        best_h = h2.copy()

    # Phase 3: Higher resolution n=3600
    n3 = 4 * n_parent
    if remaining() > 120:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        tl = remaining() * 0.25
        h3, c5_3a = opt_smooth(h3, n3, tau=20, max_iters=6000, time_limit=tl * 0.55, lr0=0.12)
        if remaining() > 10:
            h3, c5_3b = opt_smooth(h3, n3, tau=60, max_iters=4000, time_limit=tl * 0.45, lr0=0.06)
            if c5_3b < c5_3a:
                c5_3a = c5_3b
        if c5_3a < best_c5:
            best_c5 = c5_3a
            best_h = h3.copy()

    # Phase 4: Subgradient polish at best resolution
    n_f = len(best_h)
    h_f = best_h.copy()
    if remaining() > 30:
        tl = remaining() * 0.15
        h_f, c5_f = opt_subgrad(h_f, n_f, max_iters=30000, time_limit=tl, lr0=0.004, top_k=3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Perturbation restart
    if remaining() > 60:
        n_p = len(best_h)
        h_p = best_h.copy()
        # Add small random perturbation
        noise = np.random.randn(n_p) * 0.02
        h_p = project_to_feasible(h_p + noise, n_p)
        tl = remaining() * 0.25
        h_p, c5_p = opt_smooth(h_p, n_p, tau=30, max_iters=5000, time_limit=tl * 0.5, lr0=0.15)
        if remaining() > 10:
            h_p, c5_p2 = opt_subgrad(h_p, n_p, max_iters=10000, time_limit=tl * 0.5, lr0=0.003, top_k=3)
            if c5_p2 < c5_p:
                c5_p = c5_p2
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 6: Final subgradient at the very end
    if remaining() > 15:
        n_fin = len(best_h)
        h_fin = best_h.copy()
        tl = remaining() * 0.3
        h_fin, c5_fin = opt_subgrad(h_fin, n_fin, max_iters=20000, time_limit=tl, lr0=0.002, top_k=2)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5, _ = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
