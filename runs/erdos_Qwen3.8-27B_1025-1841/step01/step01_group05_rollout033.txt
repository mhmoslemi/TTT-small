The issue with the smooth max approach is that for high beta, the gradient computation involves 2n-1 lags, each with O(n) work, giving O(n²) per gradient evaluation. For n=672, that's about 450000 per gradient. SLSQP might do 100-200 iterations, so total is about 45 million ops. That should be fine within the time budget.

But actually, I realize there might be a much more efficient way to compute the gradient. Let me think...

corr[k] * dx = sum_{valid i} h[i] * (1 - h[i-k]) * dx

The gradient w.r.t. h[j] is:
For each k where j is "valid":
- j contributes to h[i] term: (1 - h[j-k]) * dx if j is in valid range for i
- j contributes to h[i-k] term: -h[j+k] * dx if j+k is in valid range for i

This can be written as:
d(corr[k])/dh[j] * dx = dx * [(1 - h[j-k]) * 1[j in range for i at lag k] - h[j+k] * 1[j+k in range for i at lag k]]

For lag k >= 0: valid i are [k, n-1]
  - j in [k, n-1]: contributes (1-h[j-k])
  - j+k in [k, n-1], i.e., j in [0, n-1-k]: contributes -h[j+k]

For lag k < 0 (let l = -k > 0): valid i are [0, n-1-l]
  - j in [0, n-1-l]: contributes (1-h[j+l])
  - j-l in [0, n-1-l], i.e., j in [l, n-1]: contributes -h[j-l]

So the total gradient is:
grad[j] = sum over all k of weights[k] * dx * (contribution)

This is O(n * (2n-1)) = O(n²) which is what we have. For n up to 1000, that's 2*10^6 per gradient, which is fine.

Let me also think about whether we can do better byexploiting the banded structure. Actually, for the final answer, what matters is the max, not all lags. In the limit beta → ∞, only the active lag matters, and the gradient is just the gradient of that single correlation value, which is O(n). But with finite beta, we need all lags.

For a practical approach, let me use a moderate beta (like 50-100) which gives a good approximation of the max while being smooth enough for optimization.

1. Find the active lag k* = argmax corr
2. Minimize corr[k*] with constraints (this is a QP!)
3. Repeat

This is like a block coordinate / active set method and should converge quickly.

corr[k] = sum_{valid i} h[i] - sum_{valid i} h[i]*h[i-k]

The second term sum h[i]*h[i-k] is not convex in general (it's an indefinite quadratic form). So minimizing corr[k] is minimizing a non-convex quadratic. Hmm.

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return np.zeros(len(corr))
    return e / s

def smooth_max_val(corr, beta):
    weights = soft_weights(corr, beta)
    if np.sum(weights) < 1e-30:
        return np.max(corr)
    return np.sum(weights * corr)  # This is actually the softmax-weighted average, which approximates max

    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    s = np.sum(np.exp(shifted))
    return m + np.log(s) / beta

def gradient_corr(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    num_lags = 2 * n - 1
    for lag_idx in range(num_lags):
        w = weights[lag_idx]
        if w < 1e-12:
            continue
        lag = lag_idx - (n - 1)
        if lag >= 0:
            # corr[lag] = sum_{i=lag}^{n-1} h[i] * (1-h[i-lag])
            lo1, hi1 = lag, n - 1
            grad[lo1:hi1+1] += w * dx * (1.0 - h[0:n-lag])
            lo2, hi2 = 0, n - 1 - lag
            grad[lo2:hi2+1] -= w * dx * h[lag:n]
        else:

            lo1, hi1 = 0, n - 1 - l
            grad[lo1:hi1+1] += w * dx * (1.0 - h[l:n])
            lo2, hi2 = l, n - 1
            grad[lo2:hi2+1] -= w * dx * h[0:n-l]
    return grad

def project_feasible(h, n):
    h = np.array(h, dtype=float)
    target = n / 2.0

            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return h

        h_new = f(x_new)

    return np.clip(h_new, 0, 1)

    else:
        h_curr = np.full(42, 0.5)

    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)
    best_n = len(h_curr)

    def time_left():
        return budget_s - (time.time() - t0)

    base_n = best_n
    resolutions = sorted(set([base_n]))
    for mult in [2, 3, 4, 6, 8, 12, 16, 24, 32]:
        n = base_n * mult
        if n <= 4096:
            resolutions.append(n)

    for n_target in resolutions:
        if time_left() < 40:
            break
        if n_target <= best_n:
            continue

        h_init = interpolate_h(best_h, n_target)

        for beta in [15, 40, 100, 300]:
            if time_left() < 25:
                break

            est_iter_time = max(0.01, 3.0 * n_target * n_target * 1e-8)
            maxiter = max(40, min(300, int(15.0 / est_iter_time)))

            try:
                def _obj(h):
                    return smooth_max_val(compute_corr(h), beta)

                def _objg(h):
                    c = compute_corr(h)
                    w = soft_weights(c, beta)
                    return smooth_max_val(c, beta), gradient_corr(h, w)

                cons = [{'type': 'eq', 'fun': lambda h, nt=n_target: np.sum(h) - nt / 2.0,
                         'jac': lambda h, nt=n_target: np.ones(nt)}]
                bounds = [(0.0, 1.0)] * n_target

                res = minimize(_objg, h_init, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-12})

                h_new = project_feasible(res.x, n_target)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_target
                    h_init = h_new.copy()
            except Exception:
                pass

        # Active-set refinement
        if time_left() > 25:
            h_work = best_h.copy()
            n = best_n
            for _ in range(20):
                if time_left() < 10:
                    break
                c = compute_corr(h_work)
                k_star = int(np.argmax(c))

                def _obj2(h):
                    return compute_corr(h)[k_star]

                def _obj2g(h):
                    w = np.zeros(2 * n - 1)
                    w[k_star] = 1.0
                    return compute_corr(h)[k_star], gradient_corr(h, w)

                cons = [{'type': 'eq', 'fun': lambda h, nt=n: np.sum(h) - nt / 2.0,
                         'jac': lambda h, nt=n: np.ones(nt)}]
                bounds = [(0.0, 1.0)] * n

                try:
                    res = minimize(_obj2g, h_work, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 150, 'ftol': 1e-14})
                    h_trial = project_feasible(res.x, n)
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        h_work = h_trial.copy()
                    else:
                        break
                except Exception:
                    break

    # Try random restarts at best resolution if time allows
    if time_left() > 60:
        n = best_n
        for _ in range(5):
            if time_left() < 30:
                break
            h_rand = np.clip(best_h + 0.3 * np.random.randn(n), 0, 1)
            h_rand = project_feasible(h_rand, n)
            for beta in [50, 150]:
                if time_left() < 20:
                    break
                try:
                    def _obj(h):
                        return smooth_max_val(compute_corr(h), beta)

                    def _objg(h):
                        c = compute_corr(h)
                        w = soft_weights(c, beta)
                        return smooth_max_val(c, beta), gradient_corr(h, w)

                    cons = [{'type': 'eq', 'fun': lambda h, nt=n: np.sum(h) - nt / 2.0,
                             'jac': lambda h, nt=n: np.ones(nt)}]
                    bounds = [(0.0, 1.0)] * n
                    res = minimize(_objg, h_rand, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 100, 'ftol': 1e-12})
                    h_new = project_feasible(res.x, n)
                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                except Exception:
                    pass

    # Final quantization attempts
    if time_left() > 5:
        for levels in [2, 3, 4, 8]:
            h_q = np.round(best_h * levels) / float(levels)
            h_q = project_feasible(h_q, best_n)
            c5_q = evaluate_c5(h_q)
            if c5_q < best_c5:
                best_c5 = c5_q
                best_h = h_q.copy()

    return best_h, best_c5, best_n
