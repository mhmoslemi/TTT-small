import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-15):
            h = h_new
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-13:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, dx, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exp_vals = np.exp(shifted / temp)
    w_sum = np.sum(exp_vals)
    weights = exp_vals / w_sum
    value = (m + temp * np.log(w_sum)) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m_s = n - s
            grad[:m_s] += wk * (1.0 - h[s:s+m_s])
            grad[s:] -= wk * h[:m_s]
        else:
            r = -s
            m_r = n - r
            grad[r:] += wk * (1.0 - h[:m_r])
            grad[:m_r] -= wk * h[r:r+m_r]
    return value, grad * dx

def fista_smooth(h, dx, temp, max_iters=3000, deadline=None):
    n = len(h)
    x = h.copy()
    y = h.copy()
    t_idx = 1.0
    f0, _ = smooth_obj_and_grad(y, dx, temp)
    # Calibrate step size
    alpha = temp / (2.0 * n)
    for _ in range(5):
        _, g = smooth_obj_and_grad(y, dx, temp)
        xn = project_target(y - alpha * g, n)
        fn, _ = smooth_obj_and_grad(xn, dx, temp)
        if fn < f0:
            alpha = min(alpha * 1.5, temp)
        else:
            alpha *= 0.6
    for _ in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        f_val, g = smooth_obj_and_grad(y, dx, temp)
        x_new = project_target(y - alpha * g, n)
        f_new = smooth_obj_and_grad(x_new, dx, temp)[0]
        if f_new > f_val + 0.5 * alpha * np.dot(g, x_new - y):
            alpha *= 0.7
            x_new = project_target(y - alpha * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_idx * t_idx)) / 2.0
        mu = (t_idx - 1.0) / (t_new + 1.0)
        y = x_new + mu * (x_new - x)
        t_idx = t_new
        x = x_new
    return project_target(x, n)

def coord_1d_obj(h, j, n, dx):
    h[j] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode="full").copy()
    h[j] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode="full").copy()
    # For k != 0: c_k(t) = c0[k] + t*(c1[k]-c0[k])
    # For k = 0: c_0(t) = c0[k0] + t*(c1[k0]-c0[k0]) - t*(1-t) (quadratic correction)
    k0 = n - 1
    return c0, c1, k0

def coord_descent(h, dx, max_sweeps=300, deadline=None):
    n = len(h)
    best = compute_c5(h, dx)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        any_improve = False
        for j in range(n):
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full").copy()
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full").copy()
            h[j] = 0.0
            k0 = n - 1
            diff = c1 - c0
            # Objective at t: max(c0 + t*diff - t*(1-t)*[k==k0])
            # For k != k0: linear in t
            # For k == k0: c0[k0] + t*diff[k0] - t + t^2
            #   = c0[k0] - t + t*diff[k0] + t^2  ... wait
            # Actually c0[k0] is the value at t=0, c1[k0] at t=1
            # c_k(t) = c0[k] + t*(c1[k]-c0[k]) for k != k0
            # c_{k0}(t) = c0[k0] + t*(c1[k0]-c0[k0]) + t*(t-1)  [since h[j]*(1-h[j]) = t-t^2, and at t=0 it's 0, at t=1 it's 0, but we added t*(c1-c0) which at the quadratic part...]
            # Let me just do: c_{k0}(t) = c0[k0] + t*(c1[k0]-c0[k0]) + t*(t-1)
            # Check: at t=0: c0[k0] + 0 + 0 = c0[k0] ✓
            # at t=1: c0[k0] + (c1[k0]-c0[k0]) + 0 = c1[k0] ✓
            # The extra term t*(t-1) = t^2 - t accounts for h[j]*(1-h[j]) being t-t^2 not t
            
            # Evaluate at candidate points
            # Find argmax at t=0 and t=1
            best_t = h[j] if h[j] > 0 else 0.0
            # Try current, 0, 1, and midpoint of top-2 lags
            candidates = [0.0, 1.0, min(max(h[j], 0.0), 1.0)]
            # Also try to minimize the max: the max at t is approximately
            # max(c0 + t*diff) [ignoring quadratic for now]
            # The minimax of lines is at the intersection of the two highest lines
            idx_max0 = np.argmax(c0)
            idx_max1 = np.argmax(c1)
            if idx_max0 != idx_max1 and diff[idx_max1] != diff[idx_max0]:
                t_cross = (c0[idx_max0] - c0[idx_max1]) / (diff[idx_max1] - diff[idx_max0])
                if 0.05 < t_cross < 0.95:
                    candidates.append(t_cross)
                    candidates.append(max(0.0, t_cross - 0.05))
                    candidates.append(min(1.0, t_cross + 0.05))
            candidates.append(0.5)
            
            best_t = 0.0
            best_val = np.inf
            for t in candidates:
                t = max(0.0, min(1.0, t))
                h[j] = t
                # Fast evaluation using precomputed values
                corr_tmp = c0 + t * diff
                corr_tmp[k0] += t * (t - 1.0)
                val = np.max(corr_tmp)
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Refine with golden section near best_t
            lo = max(0.0, best_t - 0.1)
            hi = min(1.0, best_t + 0.1)
            if hi - lo > 1e-6:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c_pt = hi - gr * (hi - lo)
                d_pt = lo + gr * (hi - lo)
                h[j] = c_pt
                corr_tc = c0 + c_pt * diff
                corr_tc[k0] += c_pt * (c_pt - 1.0)
                fc = np.max(corr_tc)
                h[j] = d_pt
                corr_td = c0 + d_pt * diff
                corr_td[k0] += d_pt * (d_pt - 1.0)
                fd = np.max(corr_td)
                for _ in range(40):
                    if fc < fd:
                        hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = hi - gr * (hi - lo)
                        h[j] = c_pt
                        corr_tc = c0 + c_pt * diff
                        corr_tc[k0] += c_pt * (c_pt - 1.0)
                        fc = np.max(corr_tc)
                    else:
                        lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = lo + gr * (hi - lo)
                        h[j] = d_pt
                        corr_td = c0 + d_pt * diff
                        corr_td[k0] += d_pt * (d_pt - 1.0)
                        fd = np.max(corr_td)
                mid = (lo + hi) / 2.0
                h[j] = mid
                corr_tm = c0 + mid * diff
                corr_tm[k0] += mid * (mid - 1.0)
                fm = np.max(corr_tm)
                if fm < best_val:
                    best_val = fm
                    best_t = mid
            
            h[j] = best_t
            h = project_target(h, n)
        
        current = compute_c5(h, dx)
        if current < best - 1e-12:
            best = current
            any_improve = True
        elif not any_improve and sweep > 3:
            break

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def subgradient_polish(h, dx, n, n_iters=20000, deadline=None):
    alpha = 1.0 / (n ** 0.7)
    best = compute_c5(h, dx)
    no_improve = 0
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            m_s = n - s
            grad[:m_s] += (1.0 - h[s:s+m_s])
            grad[s:] -= h[:m_s]
        else:
            r = -s
            m_r = n - r
            grad[r:] += (1.0 - h[:m_r])
            grad[:m_r] -= h[r:r+m_r]
        h_new = project_target(h - alpha * grad, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best - 1e-13:
            h = h_new
            best = c5_new
            no_improve = 0
            alpha = min(alpha * 1.02, 5.0)
        else:
            no_improve += 1
            if no_improve > 1000:
                alpha *= 0.85
                no_improve = 0
        if it % 5000 == 4999:
            alpha *= 0.92
    return h, best
