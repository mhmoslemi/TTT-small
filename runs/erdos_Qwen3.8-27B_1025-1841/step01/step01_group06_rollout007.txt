Actually, one more thought: the projection onto {0≤h≤1, sum(h)=n/2} is not a simple operation (it's not just clip + rescale). The correct projection solves:
min ||h - z||² subject to 0≤h_i≤1, Σh_i = n/2

This is a quadratic program. The solution involves finding a threshold λ such that:
h_i = clip(z_i - λ, 0, 1) and Σ h_i = n/2

This can be solved by bisection on λ.

def project_box_sum(z, n):
    target = n / 2.0
    # Find lambda such that sum(clip(z - lambda, 0, 1)) = target
    lo, hi = -1.0, 2.0  # ample range
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(z - mid, 0, 1)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Adjust the free variables
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_box_sum(z, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_box_sum(h_new, n_new)

def soft_max_val(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx

def soft_max_grad(h, dx, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temp)
    wsum = exp_vals.sum()
    weights = (exp_vals / wsum) * dx

    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        k = s + n - 1
        w = weights[k]
        if w < 1e-16:
            continue
        i1s = max(0, s)
        i1e = min(n, n + s)
        if i1s < i1e:
            grad[i1s:i1e] += w * (1.0 - h[i1s - s:i1e - s])
        i2s = max(0, -s)
        i2e = min(n, n - s)
        if i2s < i2e:
            grad[i2s:i2e] -= w * h[i2s + s:i2e + s]
    return grad

def fista_optimize(h, dx, n, max_iter, t0, deadline):
    h_curr = h.copy()
    h_prev = h.copy()
    t_curr = 1.0
    alpha = 1.0 / (2.0 * n)
    best_c5 = compute_c5(h_curr, dx)[0]
    best_h = h_curr.copy()
    tau = 0.01
    n_iter = 0
    for it in range(max_iter):
        if time.time() > deadline - 60:
            break
        g = soft_max_grad(h_curr, dx, tau)
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr))
        y = h_curr + ((t_curr - 1.0) / t_next) * (h_curr - h_prev)
        h_new = project_box_sum(y - alpha * g, n)
        h_prev = h_curr
        h_curr = h_new
        t_curr = t_next
        n_iter += 1
        if it % 500 == 499:
            alpha *= 0.7
            tau *= 0.5
            if tau < 0.002:
                tau = 0.002
        c5_val = compute_c5(h_curr, dx)[0]
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_curr.copy()
    return best_h, best_c5, n_iter

def slsqp_optimize(h, dx, n, t0, deadline, maxiter=300, ftol=1e-12):
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)[0]
    for temp in [0.005, 0.002, 0.001]:
        if time.time() > deadline - 30:
            break
        try:
            res = minimize(
                lambda x: soft_max_val(np.clip(x, 0, 1), dx, temp),
                best_h,
                jac=lambda x: soft_max_grad(np.clip(x, 0, 1), dx, temp),
                method='SLSQP', bounds=bnds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_box_sum(res.x, n)
            c5_new = compute_c5(h_new, dx)[0]
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5

    best_c5, corr = compute_c5(h, dx)
