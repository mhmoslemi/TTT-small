import numpy as np
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n_points, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = mask.sum()
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmax(h < 1.0) if diff > 0 else np.argmax(h > 0.0)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def to_binary(h, n_points, rng):
    """Convert to binary with exactly n/2 ones"""
    # Use the values as priorities
    n_ones = n_points // 2
    # Add small random perturbation to break ties
    perturbed = h + rng.random(n_points) * 0.001
    indices = np.argsort(perturbed)[::-1][:n_ones]
    b = np.zeros(n_points)
    b[indices] = 1.0
    return b

def local_search_binary(h, n_points, max_iters, rng, time_limit, start_time):
    """Local search for binary h: try swaps"""
    dx = 2.0 / n_points
    current_c5 = evaluate_c5(h, n_points)
    best_h = h.copy()
    best_c5 = current_c5
    n_ones = int(np.sum(h))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # Try a random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        # Update sets
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        new_c5 = evaluate_c5(h, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Revert
            h[j] = 0.0
            h[i] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
    
    return best_h, best_c5

Actually, one more important consideration: the `np.correlate` function. Let me make sure I understand what it computes correctly.

For arrays a (length Na) and v (length Nv), `np.correlate(a, v, mode='full')` returns an array of length Na+Nv-1.

The k-th element (0-indexed) is:
c[k] = sum_n a[n+k-(Nv-1)] * v[n]

where the sum is over all n such that 0 ≤ n < Nv and 0 ≤ n+k-(Nv-1) < Na.

For a = h, v = 1-h, both of length n:
c[k] = sum_n h[n+k-(n-1)] * (1-h)[n]

Let me define lag = k - (n-1). Then:
c[k] = sum_n h[n+lag] * (1-h[n])

where 0 ≤ n < n and 0 ≤ n+lag < n.

So lag ranges from -(n-1) to (n-1), and k = lag + (n-1) ranges from 0 to 2n-2.

For lag = 0 (k = n-1): c = sum_n h[n] * (1-h[n]) = sum h - sum h²
For lag = 1 (k = n): c = sum_{n=0}^{n-2} h[n+1] * (1-h[n])
For lag = -1 (k = n-2): c = sum_{n=1}^{n-1} h[n-1] * (1-h[n])

This makes sense. The "overlap integral" for shift k (in the original problem) corresponds to lag = k/dx in the discrete setting.

So C₅ = max_k c[k] * dx, which is what we're computing.

Now, for binary h, c[n-1] (lag=0) = 0 since h[n](1-h[n]) = 0. Good.

def _subgrad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    g = np.zeros(n)
    # g[j] = (1-h[j-lag]) if 0<=j-lag<n + (-h[j+lag]) if 0<=j+lag<n
    # First term: j-lag in [0,n-1] → j in [lag, lag+n-1], intersect [0,n-1]
    lo1 = max(0, lag)
    hi1 = min(n, lag + n)
    for j in range(lo1, hi1):
        g[j] += (1.0 - h[j - lag])
    # Second term: j+lag in [0,n-1] → j in [-lag, n-1-lag], intersect [0,n-1]
    lo2 = max(0, -lag)
    hi2 = min(n, n - lag)
    for j in range(lo2, hi2):
        g[j] += (-h[j + lag])
    return g

        cur_h, c5 = _sa(cur_h, cur_n, rng, t2)

            best_val = c5
            best_h = cur_h.copy()

        # Phase 3: Binary local search (30% of allocated time)
        t3 = ta * 0.3
        hb, c5b = _bs(cur_h, cur_n, rng, t3)
        if c5b < best_val:
            best_val = c5b
            best_h = hb.copy()
            cur_h = hb.copy()

    # Multi-start at the best resolution found
    if remaining() > 60:
        top_n = cur_n
        t_remaining = remaining() * 0.8
        n_starts = min(int(t_remaining / 30), 8)
        per_start = t_remaining / max(n_starts, 1)

        # Include parent at top resolution
        starts = [_interp(h_init, n0, top_n)]
        # Add diverse starts
        for s in _multi_start(top_n, rng):
            if len(starts) < n_starts:
                starts.append(s)

        for si, sh in enumerate(starts):
            if remaining() < 15:
                break
            per_t = max(per_start * 0.5, remaining() * 0.3)
            sh = _proj(sh, top_n)
            # Quick GD
            sh, c5 = _gd(sh, top_n, rng, per_t * 0.5)
            if c5 < best_val:
                best_val = c5
                best_h = sh.copy()
            # Quick SA
            sh, c5 = _sa(sh, top_n, rng, per_t * 0.5)
            if c5 < best_val:
                best_val = c5
                best_h = sh.copy()

    # Final evaluation
    final_val = _eval(best_h, len(best_h))
    if final_val < best_val:
        best_val = final_val

    return best_h, best_val, len(best_h)
