def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def projected_gradient_descent(h, T, max_iter, alpha_init, time_limit):
    """PGD with backtracking line search."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    f_best = soft_max_value(compute_corr(h), T) * dx
    alpha = alpha_init
    
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        g = grad_smooth(h, T)
        # Remove mean component (tangent to constraint)
        g -= np.mean(g)
        
        # Backtracking line search
        for _ in range(30):
            h_new = project_feasible(h - alpha * g, n)
            f_new = soft_max_value(compute_corr(h_new), T) * dx
            if f_new < f_best - 1e-14:
                h = h_new
                f_best = f_new
                break
            alpha *= 0.5
        
        # Adaptive alpha: increase if we converged quickly
        # (handled by the backtracking starting from previous alpha)
    
    return h, f_best

def coordinate_descent(h, max_rounds, time_limit):
    """CD polish: try small changes to each coordinate."""
    n = len(h)
    h = h.copy()
    c5_best = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        perm = np.random.permutation(n)
        for j in perm:
            if time.time() > time_limit:
                break
            old = h[j]
            candidates = [0.0, 1.0, 0.5, old-0.1, old-0.05, old-0.01, old+0.01, old+0.05, old+0.1]
            for c in candidates:
                if c < 0 or c > 1:
                    continue
                if abs(c - old) < 1e-6:
                    continue
                trial = h.copy()
                trial[j] = c
                trial = project_feasible(trial, n)
                c_trial = compute_c5(trial)
                if c_trial < c5_best - 1e-12:
                    c5_best = c_trial
                    h = trial
                    improved = True
                    break
        if not improved:
            break
    
    return h, c5_best

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    time_limit = start_time + budget_s - 5
    rng = np.random.RandomState(seed)

import numpy as np
import time

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(corr, T):

    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T)
