def smooth_max_grad(h, dx, eps):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / eps)

    w = exp_vals / sum_exp

    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = (n - 1) - k
        # Term 1: (1-h[j+lag]) for 0 <= j+lag < n → j in [max(0,-lag), min(n-1, n-1-lag)]
        t1_start = max(0, -lag)
        t1_end = min(n, n - lag)  # exclusive
        if t1_end > t1_start:
            grad[t1_start:t1_end] += wk * (1.0 - h[t1_start + lag:t1_end + lag])
        # Term 2: -h[j-lag] for 0 <= j-lag < n → j in [max(0,lag), min(n-1, n-1+lag)]
        t2_start = max(0, lag)
        t2_end = min(n, n + lag)  # exclusive
        if t2_end > t2_start:
            grad[t2_start:t2_end] -= wk * h[t2_start - lag:t2_end - lag]
    
    grad *= dx
    return F, grad
