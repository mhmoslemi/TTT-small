import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_c5_from_binary(h):
    """h is 0/1 array. Compute c5."""
    n = len(h)
    dx = 2.0 / n
    # corr[k] = sum_i h[i+s] * (1-h[i]) for s = k-(n-1)
    # For binary h, this is just the count
    corr = np.correlate(h.astype(float), (1-h).astype(float), mode='full')
    return float(np.max(corr)) * dx

def project_binary_feasible(h, n):
    """Ensure h has exactly n/2 ones and n/2 zeros."""
    n_ones = h.sum()
    target = n // 2
    if n_ones == target:
        return h.copy()
    if n_ones > target:
        # Need to change (n_ones - target) ones to zeros
        ones_idx = np.where(h == 1)[0]
        np.random.shuffle(ones_idx)
        for idx in ones_idx[:int(n_ones - target)]:
            h[idx] = 0
    else:
        # Need to change (target - n_ones) zeros to ones
        zeros_idx = np.where(h == 0)[0]
        np.random.shuffle(zeros_idx)
        for idx in zeros_idx[:int(target - n_ones)]:
            h[idx] = 1
    return h

def local_search_binary(h, n, max_sweeps=5000, time_limit=None, t_start=None):
    """Local search on binary sequences with n/2 ones."""
    best = h.copy()
    best_c5 = compute_c5_from_binary(h, n)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        improved = False
        best_swap_improve = 0
        best_swap = None
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try all swaps (move a 1 to a 0 position)
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = compute_c5_from_binary(h, n)
                improve = best_c5 - c5
                if improve > best_swap_improve:
                    best_swap_improve = improve
                    best_swap = (i, j)
                h[i] = 1
                h[j] = 0

            h[i] = 0
            h[j] = 1
            best_c5 = compute_c5_from_binary(h, n)
            best = h.copy()
            improved = True
        
        if not improved:

            for _ in range(10):
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                h[i] = 0
                h[j] = 1

                    best = h.copy()
                    break
                h[i] = 1
                h[j] = 0
            # Reset to best

            break
    
    return best, best_c5
