            x0 = np.arange(best_n) * 2.0 / best_n
            h_int = np.interp(xs, x0, best_h)
        else:
            h_int = np.full(n, 0.5)
        h_int = project(h_int, n)

        # Random binary start
        h_bin = np.zeros(n)
        h_bin[np.random.choice(n, size=n // 2, replace=False)] = 1.0
        h_bin = project(h_bin + np.random.randn(n) * 0.04, n)

        c5_int = compute_c5(h_int, n)
        c5_bin = compute_c5(h_bin, n)
        h_start = h_int if c5_int < c5_bin else h_bin
        c5_start = min(c5_int, c5_bin)

        t_alloc = min(140, rem * 0.22)
        if t_alloc < 30:
            continue

        h2, c5_2 = try_resolution(h_start, n, seed + 1000 + n, t_alloc)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n

    # Phase 3: Final polish on best
    if best_h is not None:
        rem = budget_s - (time.time() - t_start)
        if rem > 40:
            t_polish = min(100, rem - 20)
            h3, c5_3 = sa_optimize(best_h, best_n, seed + 54321, t_polish,
                                   step0=0.04, T0=0.002, max_pert=3)
            if c5_3 < best_c5:
                best_h, best_c5 = h3, c5_3

    if best_h is None:
        best_h = np.full(45, 0.5)
        best_n = 45
        best_c5 = compute_c5(best_h, best_n)

    # Enforce exact constraint
    target = best_n / 2.0
    d = target - best_h.sum()
    if abs(d) > 1e-10:
        m = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        if m.sum() > 0:
            best_h[m] += d / m.sum()
        else:
            idx = np.argmin(np.abs(best_h - 0.5))
            best_h[idx] = np.clip(best_h[idx] + d, 0, 1)
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
