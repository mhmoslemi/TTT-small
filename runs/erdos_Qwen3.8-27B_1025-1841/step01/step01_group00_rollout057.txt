import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400.0, 400.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    # Gradient via correlate
    term1 = np.correlate(p, 1.0 - h, mode='full')
    term2 = np.correlate(p, h, mode='full')
    grad = dx * (term1[n-1:2*n-1] - term2[n-1:2*n-1])
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def optimize_smooth(h, n, tau, max_iters, time_limit, initial_lr=None, momentum=0.9):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr is not None else 0.5
    v = np.zeros(n)
    beta = momentum
    stall = 0
    best_stall_count = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 100:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break

        h = h_new

    return best_h, best_obj

def optimize_subgrad(h, n, max_iters, time_limit, initial_lr=None):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr is not None else 0.01
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = np.argmax(corr)
        s = k_star - n + 1
        # grad[i] = dx * [(1-h[i+s]) - h[i-s]] for valid indices
        grad = np.zeros(n)
        # (1-h[i+s]) for valid i: i+s in [0,n-1], i in [0,n-1] → i in [max(0,-s), min(n-1, n-1-s)]

        h = h_new

    return best_h, best_c5

def perturb(h, n, scale, rng):

    h_p += scale * rng.standard_normal(n)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400.0, 400.0)
    exp_c = np.exp(shifted)
    s_val = exp_c.sum()
    p = exp_c / s_val
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-14:
            continue
        sh = m - n + 1
        i_lo = max(0, sh)
        i_hi = min(n - 1, sh + n - 1)
        if i_lo <= i_hi:
            T1[i_lo:i_hi + 1] += pm * (1.0 - h[i_lo - sh:i_hi - sh + 1])
        i_lo2 = max(0, -sh)
        i_hi2 = min(n - 1, n - 1 - sh)
        if i_lo2 <= i_hi2:
            T2[i_lo2:i_hi2 + 1] += pm * h[i_lo2 + sh:i_hi2 + sh + 1]
    grad = dx * (T1 - T2)
    obj = dx * (max_c + np.log(s_val) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, max_iters, time_limit, initial_lr=None, momentum=0.9):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr is not None else 0.5
    v = np.zeros(n)
    beta = momentum
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > 80:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break
        h = h_new
    return best_h, best_obj


def optimize_topk_subgrad(h, n, max_iters, time_limit, initial_lr=None, top_k=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr is not None else 0.008
    v = np.zeros(n)
    beta = 0.82
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[::-1][:top_k]
        top_idx = top_idx[corr[top_idx] > corr.max() - 1e-6]
        grad = np.zeros(n)
        for k_star in top_idx:
            sh = k_star - n + 1
            i_lo = max(0, sh)
            i_hi = min(n - 1, sh + n - 1)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += (1.0 - h[i_lo - sh:i_hi - sh + 1])
            i_lo2 = max(0, -sh)
            i_hi2 = min(n - 1, n - 1 - sh)
            if i_lo2 <= i_hi2:
                grad[i_lo2:i_hi2 + 1] -= h[i_lo2 + sh:i_hi2 + sh + 1]
        grad *= dx / len(top_idx)

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 150:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    return best_h, best_c5


def perturb_h(h, n, scale, rng):
    h_p = h.copy()
    h_p += scale * rng.standard_normal(n)
    return project_to_feasible(h_p, n)


def multi_tau_smooth(h, n, taus, time_limit, rng):
    h_cur = h.copy()
    total_tl = time_limit
    n_passes = len(taus)
    per_pass = total_tl / n_passes
    for tau in taus:
        if time.time() > 0:
            elapsed = 0
        elif True:
            pass
        elapsed_frac = (sum(per_pass) ) / total_tl
        remaining_frac = 1.0 - elapsed_frac
        actual_tl = min(per_pass * remaining_frac * n_passes, per_pass * 1.5)
        if actual_tl < 5:

        h_cur, _ = optimize_smooth(h_cur, n, tau=tau, max_iters=200000,
                                   time_limit=actual_tl,
                                   initial_lr=0.4 / np.sqrt(max(tau, 2.0)),
                                   momentum=0.9)
    return h_cur
