        maxiters = [250, 250, 200, 150]

        for idx in range(n_starts):
            if time.time() - t_start > budget_s * 0.93:
                break

            h0 = inits[idx].copy()

            for phase in range(len(Ts)):
                if time.time() - t_start > budget_s * 0.93:
                    break

                T = Ts[phase]
                cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
                bounds = [(0.0, 1.0)] * n

                try:
                    res = minimize(
                        soft_max_obj, h0,
                        args=(n, T),
                        jac=soft_max_grad,
                        bounds=bounds,
                        constraints=cons,
                        method='SLSQP',
                        options={'maxiter': maxiters[phase], 'ftol': 1e-12,
                                 'iprint': 0}
                    )
                    h0 = project_onto_feasible(res.x, target_sum, n)
                except Exception:
                    h0 = project_onto_feasible(h0, target_sum, n)

            c5 = compute_c5(h0, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
                best_n = n

    # Local refinement: coordinate descent on best solution
    if best_h is not None:
        n = best_n
        target_sum = n / 2.0
        improved = True
        max_cd_iters = 50
        for _ in range(max_cd_iters):
            if time.time() - t_start > budget_s * 0.97:
                break
            improved = False
            curr_c5 = compute_c5(best_h, n)
            for i in range(n - 1, -1, -1):
                if time.time() - t_start > budget_s * 0.97:
                    break
                old_val = best_h[i]
                if old_val <= 1e-9 or old_val >= 1.0 - 1e-9:
                    continue
                # Try adjusting h[i] and compensating h[j] for some j
                for j in range(0, n, max(1, n // 64)):
                    if j == i:
                        continue
                    if best_h[j] <= 1e-9 or best_h[j] >= 1.0 - 1e-9:
                        continue
                    # Move delta from i to j
                    delta = 0.1
                    new_i_val = old_val + delta
                    new_j_val = best_h[j] - delta
                    if new_i_val < 0 or new_i_val > 1 or new_j_val < 0 or new_j_val > 1:
                        delta = -0.1
                        new_i_val = old_val + delta
                        new_j_val = best_h[j] - delta
                        if new_i_val < 0 or new_i_val > 1 or new_j_val < 0 or new_j_val > 1:
                            continue
                    best_h[i] = new_i_val
                    best_h[j] = new_j_val
                    new_c5 = compute_c5(best_h, n)
                    if new_c5 < curr_c5:
                        curr_c5 = new_c5
                        improved = True
                    else:
                        best_h[i] = old_val
                        best_h[j] = best_h[j] + 2 * delta

                        # Try opposite direction
                        delta = -0.1
                        new_i_val = old_val + delta
                        new_j_val = best_h[j] - delta
                        if new_i_val < 0 or new_i_val > 1 or new_j_val < 0 or new_j_val > 1:
                            continue
                        best_h[i] = new_i_val
                        best_h[j] = new_j_val
                        new_c5 = compute_c5(best_h, n)
                        if new_c5 < curr_c5:
                            curr_c5 = new_c5
                            improved = True
                        else:
                            best_h[i] = old_val
                            best_h[j] += delta
            if not improved:
                break

        best_h = project_onto_feasible(best_h, target_sum, n)
        final_c5 = compute_c5(best_h, n)
        if final_c5 < best_c5:
            best_c5 = final_c5

    if best_h is None:
        n = 128
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, n)
        best_n = n

    return best_h, best_c5, best_n
