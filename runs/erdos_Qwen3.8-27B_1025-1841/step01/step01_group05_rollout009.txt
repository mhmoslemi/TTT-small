def objective(h):

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            adj = diff * h[mask] / np.sum(h[mask])
            h[mask] -= adj
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h < 1
            if not np.any(mask):
                break
            adj = (-diff) * (1 - h[mask]) / np.sum(1 - h[mask])
            h[mask] += adj
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, n, beta=100):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    max_c = np.max(corr)

    w = np.exp(beta * (corr - max_c))

    if ws < 1e-300:

    mx = c.max()
    w = np.exp(beta * (c - mx))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += wi * om[:n - k]
            g[:n - k] -= wi * h[k:]
        else:
            st = -k
            g[:n - st] += wi * om[st:]
            g[st:] -= wi * h[:n - st]
    return g * dx

def _interp(h, n1, n2):
    return np.interp(np.linspace(0, 2, n2), np.linspace(0, 2, n1), h)

def _binary_ls(h, n, rs, max_iter, timeout):
    tgt = n // 2
    idx = np.argsort(h)[::-1][:tgt]
    hb = np.zeros(n)
    hb[idx] = 1.0
    hb = _proj(hb, n)
    bc5 = _obj(hb)
    bh = hb.copy()
    t0 = time.time()
    for _ in range(max_iter):
        if time.time() - t0 > timeout:
            break
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        o = ones[rs.randint(len(ones))]
        z = zeros[rs.randint(len(zeros))]
        hb[o] = 0.0
        hb[z] = 1.0
        c5 = _obj(hb)
        if c5 < bc5:
            bc5 = c5
            bh = hb.copy()
        else:
            hb[o] = 1.0
            hb[z] = 0.0
    return bh, bc5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rs = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(42, 0.5)
    n_cur = len(h)
    h = _proj(h, n_cur)

    best_h = h.copy()
    best_c5 = _obj(h)

    resolutions = [n_cur]
    while resolutions[-1] * 2 <= 448:
        resolutions.append(resolutions[-1] * 2)

    time_alloc = {}
    total_alloc = budget_s - 80
    fracs = np.array([1.0, 1.5, 2.0, 2.5, 3.0])[:len(resolutions)]
    fracs = fracs / fracs.sum()
    for i, nr in enumerate(resolutions):
        time_alloc[nr] = fracs[i] * total_alloc

    for ri, nr in enumerate(resolutions):
        if time.time() - t_start > budget_s - 40:
            break
        t_stage = time.time()
        t_budget = time_alloc[nr]

        if nr != len(h):
            h = _interp(h, len(h), nr)
            h = _proj(h, nr)
        ln = nr
        lb_c5 = _obj(h)
        lb_h = h.copy()

        # Phase 1: Subgradient descent with momentum
        lr = 0.08
        mom = np.zeros(ln)
        beta_sg = 60
        no_improve = 0
        max_no_improve = 300
        while time.time() - t_stage < t_budget * 0.55:
            if no_improve > max_no_improve and lr < 0.005:
                break
            g = _grad(h, ln, beta=beta_sg)
            mom = 0.85 * mom + 0.15 * g
            h2 = _proj(h - lr * mom, ln)
            c5 = _obj(h2)
            if c5 < lb_c5:
                lb_c5 = c5
                lb_h = h2.copy()
                no_improve = 0
            else:
                no_improve += 1
            h = h2
            lr *= 0.9997
            beta_sg = min(beta_sg * 1.001, 200)
        
        # Phase 1b: Binary local search at this resolution
        if t_budget > 30:
            bh_lt, bc5_lt = _binary_ls(lb_h, ln, rs, max_iter=5000, timeout=min(t_budget * 0.2, 120))
            if bc5_lt < lb_c5:
                lb_c5 = bc5_lt
                lb_h = bh_lt
                # Re-continuous from binary
                h = _proj(lb_h + rs.normal(0, 0.02, ln), ln)
                for _ in range(2000):
                    if time.time() - t_stage > t_budget * 0.75:
                        break
                    g = _grad(h, ln, beta=80)
                    h = _proj(h - 0.03 * g, ln)
                    c5 = _obj(h)
                    if c5 < lb_c5:
                        lb_c5 = c5
                        lb_h = h.copy()

        # Phase 2: Simulated annealing
        T = 0.04
        hi = lb_h.copy()
        restart_count = 0
        while time.time() - t_stage < t_budget:
            ht = _proj(hi + rs.normal(0, T, ln), ln)
            ct = _obj(ht)
            if ct < lb_c5 or rs.rand() < np.exp(min(0.0, (lb_c5 - ct) / max(T, 1e-10))):
                hi = ht
                if ct < lb_c5:
                    lb_c5 = ct
                    lb_h = ht.copy()
            T *= 0.9996
            if T < 0.004:
                restart_count += 1
                T = 0.03
                hi = _proj(lb_h + rs.normal(0, 0.04, ln), ln)
                if restart_count > 30:
                    break

        if lb_c5 < best_c5:
            best_c5 = lb_c5
            best_h = lb_h.copy()
        h = lb_h.copy()

    # Final refinement at best resolution
    nf = len(best_h)
    if time.time() - t_start < budget_s - 15:
        hf = best_h.copy()
        lr = 0.02
        mom = np.zeros(nf)
        t_fin = time.time()
        fin_budget = budget_s - 20 - t_fin
        for _ in range(100000):
            if time.time() - t_fin > fin_budget:
                break
            g = _grad(hf, nf, beta=100)
            mom = 0.9 * mom + 0.1 * g
            h2 = _proj(hf - lr * mom, nf)
            c5 = _obj(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
            hf = h2
            lr *= 0.9999

    best_h = _proj(best_h, len(best_h))
    best_c5 = _obj(best_h)
    return best_h, best_c5, len(best_h)
