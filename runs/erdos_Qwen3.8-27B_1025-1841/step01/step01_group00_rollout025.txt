Actually, I just realized something important. Let me re-examine the correlation computation.

np.correlate(h, 1-h, mode='full') computes:
c[k] = sum_{j} h[j] * (1-h)[j + k - (n-1)]

for k = 0, 1, ..., 2n-2.

The shift is sh = k - (n-1), ranging from -(n-1) to (n-1).

For sh = 0: c[n-1] = sum_j h[j] * (1-h[j]) = sum h - sum h² = n/2 - sum h² (in discrete units, times dx gives the continuous value).

For the maximum to be small, we need ALL of these to be small, including sh=0.

At sh=0: sum h[j](1-h[j]) = sum h[j] - sum h[j]² = n/2 - sum h[j]².
To minimize this, we want to maximize sum h[j]², which means h should be as "peaky" as possible (close to 0-1 valued).

But if h is 0-1 valued, then at other shifts, the overlap can be large.

So the trade-off is between making h peaky (good for sh=0) and making h "spread out" (good for other shifts).

This confirms that the optimum is non-trivial and requires careful optimization.

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, time_limit, lr=None, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.4 / tau
    v = np.zeros(n)
    beta = momentum
    stall = 0
    c5_ema = best_c5

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        c5_now = compute_c5(h)
        c5_ema = 0.95 * c5_ema + 0.05 * c5_now
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 1500:
            lr *= 0.6
            v *= 0.2
            stall = 0
            if lr < 1e-12:
                break

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def optimize_subgrad(h, n, time_limit, lr=None):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.003 * dx * n
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 8000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + sh:j1b + sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def optimize_multishift_subgrad(h, n, time_limit, lr=None, top_k=5):
    """Subgradient using top-k maximizing shifts for better exploration."""
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.002
    v = np.zeros(n)
    beta = 0.8
    stall = 0

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 6000:
            lr *= 0.7
            v *= 0.4
            stall = 0
            if lr < 1e-14:
                break

        # Use top-k shifts
        idx_sorted = np.argsort(corr)[::-1]
        top_idx = idx_sorted[:top_k]
        # Weight by softmax over top-k
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b + sh:j1b + sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5
