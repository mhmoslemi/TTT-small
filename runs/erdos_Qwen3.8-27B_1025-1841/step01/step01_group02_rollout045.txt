    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, new_n)

def smooth_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    obj = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    s = np.sum(h) - n / 2.0
    obj += lam * s * s
    return obj

def smooth_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exp_t = np.exp(T * (corr - mc))
    sum_exp = np.sum(exp_t)
    w = exp_t / sum_exp
    obj = (mc + np.log(sum_exp) / T) * dx
    s = np.sum(h) - n / 2.0
    obj += lam * s * s
    
    grad = np.zeros(n)
    n2 = len(w)
    for k in range(n2):
        wk = w[k]
        if wk < 1e-14:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            idx_h = np.arange(j_lo, j_hi + 1) - k + n - 1
            grad[j_lo:j_hi+1] += wk * (1.0 - h[idx_h])
        j_lo2 = max(0, n - 1 - k)
        j_hi2 = min(n - 1, 2*n - 2 - k)
        if j_hi2 >= j_lo2:
            idx_h2 = np.arange(j_lo2, j_hi2 + 1) + k - n + 1
            grad[j_lo2:j_hi2+1] -= wk * h[idx_h2]
    grad *= dx
    grad += 2.0 * lam * s
    return obj, grad
