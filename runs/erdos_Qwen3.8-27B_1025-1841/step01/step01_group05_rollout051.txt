import time

def get_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(dx * np.max(corr))

def get_subgrad(h, n):
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    k_am = int(np.argmax(corr))
    lag = k_am - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n-lag] -= h[lag:]
        grad[lag:] += 1.0 - h[:n-lag]
    else:
        grad[-lag:] -= h[:n+lag]
        grad[:n+lag] += 1.0 - h[-lag:]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = np.sum(h)
    if s < 1e-10:
        h[:] = 0.5
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # add to positions with room
            order = np.argsort(1.0 - h)  # most room first
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                diff -= add
                if diff < 1e-12: break
        else:
            order = np.argsort(h)  # least h first (most room to reduce)
            for idx in order:
                sub = min(-diff, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                diff += sub
                if diff > -1e-12: break
    return h

def scale_up(h, n_old, n_new):
    if n_new == n_old:
        return h.copy()
    # Linear interpolation
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = h.copy()

    alpha = 0.05
    vel = np.zeros(n)

    stall_count = 0
    max_stall = 1000
    
    for it in range(1_000_000):
        if time.time() - t0 > time_limit:
            break
        grad = get_subgrad(h, n)
        vel = mu * vel - alpha * grad
        h_candidate = h + vel
        h_candidate = project(h_candidate, n)
        c5 = get_c5(h_candidate, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            stall_count = 0
            alpha = min(alpha * 1.02, 0.2)
        else:
            alpha *= 0.98
            stall_count += 1
        h = h_candidate
        if stall_count > max_stall:
            # Random perturbation
            h = best_h + 0.1 * rng.randn(n)
            h = project(h, n)
            alpha = 0.03
            stall_count = 0
    
    return best_h, best_c5

import numpy as np
import time

def get_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def get_subgrad(h, n):
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    k_am = int(np.argmax(corr))
    lag = k_am - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += 1.0 - h[:n - lag]
    else:
        grad[-lag:] -= h[:n + lag]
        grad[:n + lag] += 1.0 - h[-lag:]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = np.sum(h)
    if s < 1e-12:
        h[:] = 0.5
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                if add <= 1e-15:
                    break
                h[idx] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-diff, h[idx])
                if sub <= 1e-15:
                    break
                h[idx] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return h

def scale_up(h, n_old, n_new):
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def optimize_subgrad(h, n, time_limit, seed=0):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    alpha = 0.04
    vel = np.zeros(n)
    mu = 0.75
    stall = 0
    max_stall = 800
    it = 0
    while time.time() - t0 < time_limit:
        it += 1
        grad = get_subgrad(h, n)
        vel = mu * vel - alpha * grad
        h_cand = project(h + vel, n)
        c5 = get_c5(h_cand, n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_cand.copy()
            stall = 0
            alpha = min(alpha * 1.03, 0.15)
        else:
            alpha *= 0.97
            stall += 1
        h = h_cand
        if stall > max_stall:
            h = best_h + 0.08 * rng.randn(n)
            h = project(h, n)
            alpha = 0.02
            vel[:] = 0.0
            stall = 0
    return best_h, best_c5, best_c5

def local_refine(h, n, time_limit, seed=0):
    rng = np.random.RandomState(seed + 777)
    t0 = time.time()
