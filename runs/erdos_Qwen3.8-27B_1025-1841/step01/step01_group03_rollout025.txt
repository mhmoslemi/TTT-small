import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _single_lag_grad(h, n, lag):
    """Gradient of C_lag = sum_i h[i]*(1-h[i+lag]) w.r.t. h (unscaled by dx)."""
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += (-h[:n-lag])
    else:
        m = -lag
        if n - m > 0:
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] += (-h[m:])
    return grad

def smooth_max_obj_grad(h, n, dx, beta):
    """Smooth max objective and gradient using log-sum-exp."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exps = np.exp(shifted)
    sum_exps = np.sum(exps)
    obj = max_c + np.log(sum_exps) / beta
    weights = exps / sum_exps
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        grad += w * _single_lag_grad(h, n, lag)
    grad *= dx
    return obj, grad

def exact_max_obj_grad(h, n, dx):
    """Exact max objective and gradient (subgradient)."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    max_idx = np.argmax(corr)
    obj = corr[max_idx]
    lag = max_idx - (n - 1)
    grad = _single_lag_grad(h, n, lag) * dx
    return obj, grad

def project_to_feasible(h, n):
    """Project h to {x : 0<=x<=1, sum(x)=n/2} approximately."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Add/subtract from non-saturated entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(mask):
            h = np.clip(h + diff / n, 0, 1)
        else:
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += (-h[:n-lag])

            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] += (-h[m:])

    grad = np.zeros(n)

        if n - lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += (-h[:n-lag])
    else:

                          bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-20, 'gtol': 1e-12})
            bh = np.clip(res.x, 0.0, 1.0)
            bh = project_feasible(bh, n)
            v = compute_c5(bh)
            if v < bv:
                bv = v
        except Exception:
            pass

    if time.time() - t0 < tlimit * 0.95:
        for _ in range(500):
            if time.time() - t0 > tlimit:
                break
            corr = np.correlate(bh, 1.0 - bh, mode='full') * dx
            mi = np.argmax(corr)
            lag = mi - (n - 1)
            g = _single_lag_grad(bh, n, lag) * dx
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            imp = False
            for a in [10.0/n, 5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.1/n, 20.0/n]:
                hn = project_feasible(bh - a * g, n)
                vn = compute_c5(hn)
                if vn < bv - 1e-16:
                    bh, bv, imp = hn, vn, True
                    break
            if not imp:
                break

    bh = project_feasible(bh, n)
    bv = compute_c5(bh)
    return bh, bv

def run(seed=42, budget_s=1300, **kwargs):
    gs = time.time()
    rng = np.random.default_rng(seed)
    bh, bv, bn = None, float('inf'), 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except:
        h0 = np.ones(80) * 0.5

    h1, v1 = optimize_res(h0, budget_s * 0.2, seed)
    bh, bv, bn = h1, v1, len(h1)
    rem = budget_s - (time.time() - gs)

    for nu in [160, 320, 480]:
        if rem < 100:
            break
        xl = np.linspace(0, 2, len(bh), endpoint=False)
        xu = np.linspace(0, 2, nu, endpoint=False)
        hu = np.interp(xu, xl, bh)
        hu = project_feasible(hu, nu)
        tl = min(rem * 0.3, 250)
        h2, v2 = optimize_res(hu, tl, seed + nu)
        if v2 < bv:
            bh, bv, bn = h2, v2, nu
        rem = budget_s - (time.time() - gs)

    attempts = 0
    while rem > 100 and attempts < 30:
        nt = bn
        rt = rng.integers(0, 5)
        if rt == 0:
            ht = project_feasible(bh + rng.normal(0, 0.08, nt), nt)
        elif rt == 1:
            ht = np.zeros(nt)
            idx = rng.choice(nt, nt // 2, replace=False)
            ht[idx] = 1.0
        elif rt == 2:
            x = np.linspace(0, 2, nt, endpoint=False)
            f1 = rng.uniform(1, 5)
            f2 = rng.uniform(5, 15)
            p1 = rng.uniform(0, 6.28)
            ht = 0.5 + 0.35 * np.sin(f1 * np.pi * x + p1) + 0.15 * np.sin(f2 * np.pi * x)
            ht = project_feasible(np.clip(ht, 0, 1), nt)
        elif rt == 3:
            ht = bh.copy()
            mask = rng.random(nt) < 0.4
            ht[mask] = rng.random(np.sum(mask))
            ht = project_feasible(ht, nt)
        else:
            ht = np.zeros(nt)
            pos = 0
            rem2 = nt // 2
            while pos < nt and rem2 > 0:
                bl = int(rng.integers(1, max(2, nt // 5)))
                bl = min(bl, nt - pos, rem2)
                ht[pos:pos+bl] = 1.0
                pos += bl
                rem2 -= bl
                gp = int(rng.integers(1, max(2, nt // 5)))
                gp = min(gp, nt - pos - rem2)
                pos += gp
            ht = project_feasible(ht, nt)

        tl = min(rem * 0.2, 150)
        hr, vr = optimize_res(ht, tl, seed + attempts * 13)
        if vr < bv:
            bh, bv, bn = hr, vr, nt
        attempts += 1
        rem = budget_s - (time.time() - gs)

    bh = project_feasible(bh, bn)
    bv = compute_c5(bh)
    return bh, bv, bn
