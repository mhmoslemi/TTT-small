```python
def compute_gradient(h, T):
    n = len(h)
    c = np.correlate(h, 1-h, 'full')  # length 2n-1
    # softmax weights
    max_c = np.max(c)
    exp_c = np.exp(T * (c - max_c))
    w = exp_c / np.sum(exp_c)
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        for m in range(2*n-1):
            l = m - (n-1)
            # Contribution from (1-h[j-l])
            if 0 <= j - l < n:
                grad[j] += w[m] * (1 - h[j-l])
            # Contribution from -h[j+l]
            if 0 <= j + l < n:
                grad[j] -= w[m] * h[j+l]
    return grad * (2.0 / n)  # multiply by dx = 2/n
```

```python
def softmax_gradient(h, T):
    dim = len(h)
    dx = 2.0 / dim
    c = np.correlate(h, 1.0-h, 'full')  # length 2*dim-1
    max_c = np.max(c)
    exp_c = np.exp(T * (c - max_c))
    w = exp_c / np.sum(exp_c)
    
    # Lags: L[i] = i - (dim-1) for i in range(2*dim-1)
    lags = np.arange(2*dim-1) - (dim-1)  # shape (2*dim-1,)
    
    # For each j in [0, dim-1]:
    # grad[j] = dx * sum_i w[i] * ((1-h[j-L_i])*[0≤j-L_i<dim] - h[j+L_i]*[0≤j+L_i<dim])
    
    j = np.arange(dim)  # (dim,)
    # j - L_i: shape (dim, 2*dim-1)
    j_minus_L = j[:, None] - lags[None, :]  # (dim, 2*dim-1)
    j_plus_L = j[:, None] + lags[None, :]  # (dim, 2*dim-1)
    
    valid1 = (j_minus_L >= 0) & (j_minus_L < dim)
    valid2 = (j_plus_L >= 0) & (j_plus_L < dim)
    
    # (1-h[j-L_i]) for valid positions
    val1 = np.where(valid1, 1.0 - h[j_minus_L.clip(0, dim-1)], 0.0)
    val2 = np.where(valid2, h[j_plus_L.clip(0, dim-1)], 0.0)
    
    grad = (val1 * w[None, :] - val2 * w[None, :]).sum(axis=1) * dx
    return grad
```

```python
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_onto_feasible(h, n):
    """Project h onto {x: 0≤x≤1, sum(x)=n/2}"""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:

        if d > 0:

        else:

        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_and_grad(h, T):
    """Compute softmax objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c)
    scaled = T * (c - max_c)
    exp_c = np.exp(scaled)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    
    # Objective value
    obj = ((max_c + np.log(sum_exp)) / T) * dx
    
    # Gradient
    lags = np.arange(2*n-1) - (n-1)
    j = np.arange(n)
    j_minus_L = j[:, None] - lags[None, :]
    j_plus_L = j[:, None] + lags[None, :]
    
    valid1 = (j_minus_L >= 0) & (j_minus_L < n)
    valid2 = (j_plus_L >= 0) & (j_plus_L < n)

    val2 = np.where(valid2, h[np.clip(j_plus_L, 0, n-1)], 0.0)
    
    grad = (val1 * w[None, :] - val2 * w[None, :]).sum(axis=1) * dx
    
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c)
    scaled = T * (c - max_c)
    return float((max_c + np.log(np.sum(np.exp(scaled)))) / T * dx)

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c)
    scaled = T * (c - max_c)
    exp_c = np.exp(scaled)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    
    lags = np.arange(2 * n - 1) - (n - 1)
    j = np.arange(n)
    jm = j[:, None] - lags[None, :]
    jp = j[:, None] + lags[None, :]
    
    v1 = (jm >= 0) & (jm < n)
    v2 = (jp >= 0) & (jp < n)
    
    val1 = np.where(v1, 1.0 - h[np.clip(jm, 0, n-1)], 0.0)
    val2 = np.where(v2, h[np.clip(jp, 0, n-1)], 0.0)
    
    return (val1 * w[None, :] - val2 * w[None, :]).sum(axis=1) * dx

    n = len(h0)
    target = n / 2.0
    
    def fun(x):
        return soft_max_obj(x, T)
    
    def jac(x):
        return soft_max_grad(x, T)
    
    def con(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})

        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_onto_feasible(res.x, n)
    except Exception:
        pass
    return h0.copy()

    xf = np.linspace(0, 2, n_from, endpoint=False)
```
