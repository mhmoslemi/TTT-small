    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        if abs(s - target) < 1e-14:
            break

    # shift = s - (n-1)
    k = s - (n - 1)

        m = n - k
        for i in range(m):

        for i in range(k, n):

        for i in range(kp, n):

        for i in range(n - kp):
            grad[i] -= h[i + kp]

        # Constraints:
        #   g_k^T h - t <= b_k for each cut (len(cuts) constraints)
        #   -h <= 0 (n constraints)
        #   h <= 1 (n constraints)
        #   sum(h) = n/2 (1 constraint)

        h_var = cp.Variable(n)

        objective = cp.Minimize(t_var)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]
        for g, b in zip(cuts_A, cuts_b):
            constraints.append(g @ h_var - t_var <= b)
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, tolerance=1e-9)
        
        if prob.status == 'optimal':
            h_new = h_var.value
            t_new = t_var.value
        else:
            break
