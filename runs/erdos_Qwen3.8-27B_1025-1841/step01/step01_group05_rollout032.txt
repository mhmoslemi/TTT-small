def gradient_at_lag(h, k):
    """Compute gradient of correlation at lag k w.r.t. h."""
    N = len(h)
    # k is the lag, ranges from -(N-1) to N-1
    grad = np.zeros(N)
    for i in range(N):
        # Contribution 1: i is the h[n+k] term, so n = i-k
        if 0 <= i - k < N:
            grad[i] += (1 - h[i-k])
        # Contribution 2: i is the h[n] term in (1-h[n]), so n = i
        if 0 <= i + k < N:
            grad[i] += (-h[i+k])
    return grad
