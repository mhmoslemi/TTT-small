import numpy as np
from scipy.optimize import minimize
import time

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    return float(m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def soft_max_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    p = exp_scaled / np.sum(exp_scaled)
    
    grad = np.zeros(n)
    # corr[k] corresponds to lag = k - (n-1), k in 0..2n-2
    for k in range(2*n - 1):
        if p[k] < 1e-16:
            continue
        coeff = p[k] * dx
        lag = k - (n - 1)
        if lag == 0:
            grad += coeff * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:] += coeff * (1.0 - x[:-lag])
            grad[:n-lag] -= coeff * x[lag:]
        else:
            m2 = -lag
            grad[:n-m2] += coeff * (1.0 - x[m2:])
            grad[m2:] -= coeff * x[:n-m2]
    return grad

    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def lbfgsb_opt(h0, n, T, deadline, maxiter=200):

        return soft_max_obj(x, n, T)
    def g(x):
        return soft_max_grad(x, n, T)
    try:

                       bounds=[(0.0, 1.0)] * n,
                       options={'ftol': 1e-15, 'maxiter': maxiter, 'gtol': 1e-8, 'maxls': 20})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):

    return h0

def slsqp_opt(h0, n, T, deadline, maxiter=300):

    def f(x):
        return soft_max_obj(x, n, T)
    def g(x):

                       options={'ftol': 1e-14, 'maxiter': maxiter})

import numpy as np
from scipy.optimize import minimize
import time

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    return float(m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def soft_max_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    p = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-16:
            continue
        coeff = p[k] * dx
        lag = k - (n - 1)
        if lag == 0:
            grad += coeff * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:] += coeff * (1.0 - x[:n - lag])
            grad[:n - lag] -= coeff * x[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += coeff * (1.0 - x[m2:])
            grad[m2:] -= coeff * x[:n - m2]
    return grad

    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)

    return soft_max_obj(x, n, T) + mu * (np.sum(x) - n / 2.0) ** 2

def penalty_grad(x, n, T, mu):
    return soft_max_grad(x, n, T) + 2.0 * mu * (np.sum(x) - n / 2.0) * np.ones(n)

def lbfgsb_pen(h0, n, T, mu, deadline, maxiter=200):

        return penalty_obj(x, n, T, mu)
    def g(x):
        return penalty_grad(x, n, T, mu)
    try:
        res = minimize(f, h0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'ftol': 1e-15, 'maxiter': maxiter, 'gtol': 1e-7, 'maxls': 30})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def slsqp_opt(h0, n, T, deadline, maxiter=200):
    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T)
    def g(x):
        return soft_max_grad(x, n, T)

        return np.sum(x) - target
    def c_eq_jac(x):
        return np.ones(n)

        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                       options={'ftol': 1e-13, 'maxiter': maxiter})
        if time.time() > deadline:
            return h0

        pass
