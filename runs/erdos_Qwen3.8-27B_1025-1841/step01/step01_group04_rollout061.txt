import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    obj = (m + np.log(np.sum(exp_s))) / T * dx

    j_idx = np.arange(n)[:, None]
    jp = j_idx + lags[None, :]
    jm = j_idx - lags[None, :]
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    hp = np.where(valid_p, h[np.clip(jp, 0, n-1)], 0.0)
    hm = np.where(valid_m, h[np.clip(jm, 0, n-1)], 0.0)
    grad = dx * (1.0 - (w[None, :] * hp).sum(axis=1) - (w[None, :] * hm).sum(axis=1))

    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    out = np.interp(xt, xf, h)
    return project(out, n_to)

        obj, grad = softmax_obj_grad(h, T)

        obj, grad = softmax_obj_grad(x, T)
        return obj, grad

        s = np.sum(x)

    except Exception:
        pass

    """Project to [0,1]^n with sum = n/2, maintaining symmetry"""
    h_full = np.array(h_full, dtype=float).copy()
    h_full = np.clip(h_full, 0.0, 1.0)

    h_full = (h_full + h_full[::-1]) / 2.0

        h_full = np.clip(h_full, 0.0, 1.0)
        h_full = (h_full + h_full[::-1]) / 2.0

def softmax_obj_grad_sym(h, T):
    """Compute softmax objective and gradient for symmetric problem"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    obj = (m + np.log(np.sum(exp_s))) / T * dx
    
    lags = np.arange(-(n-1), n)
    j_idx = np.arange(n)[:, None]
    jp = j_idx + lags[None, :]
    jm = j_idx - lags[None, :]
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    hp = np.where(valid_p, h[np.clip(jp, 0, n-1)], 0.0)
    hm = np.where(valid_m, h[np.clip(jm, 0, n-1)], 0.0)
    grad = dx * (1.0 - (w[None, :] * hp).sum(axis=1) - (w[None, :] * hm).sum(axis=1))
    
    # Enforce symmetry in gradient
    grad = (grad + grad[::-1]) / 2.0
    return obj, grad

    h_full = np.zeros(n)
    h_full[:n_sym] = h_sym
    h_full[n_sym:] = h_sym[::-1]

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.ascontiguousarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        h[m] += diff / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_grad_soft(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = corr * T
    mx = sc.max()
    e = np.exp(sc - mx)
    w = e / e.sum()
    obj = (mx + np.log(e.sum())) / T * dx
    lags = np.arange(-(n - 1), n)
    jp = np.arange(n)[:, None] + lags[None, :]
    jm = np.arange(n)[:, None] - lags[None, :]
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    hp = np.where(vp, h[jp.clip(0, n - 1)], 0.0)
    hm = np.where(vm, h[jm.clip(0, n - 1)], 0.0)
    g = dx * (1.0 - (w * hp).sum(1) - (w * hm).sum(1))
    return obj, g

def obj_grad_max(h):
    """Subgradient of max_s corr_s * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    mx = vals.max()
    # Subgradient: gradient of the max active constraint(s)
    active = np.zeros(len(corr))
    tol = 1e-6
    active[vals >= mx - tol] = 1.0
    w = active / active.sum()
    lags = np.arange(-(n - 1), n)
    jp = np.arange(n)[:, None] + lags[None, :]
    jm = np.arange(n)[:, None] - lags[None, :]
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    hp = np.where(vp, h[jp.clip(0, n - 1)], 0.0)
    hm = np.where(vm, h[jm.clip(0, n - 1)], 0.0)
    g = dx * (1.0 - (w * hp).sum(1) - (w * hm).sum(1))
    return mx, g

def interp_fn(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def pgd_loop(h0, n, T, n_steps, deadline, lr=0.02):
    h = h0.copy()
    for i in range(n_steps):
        if time.time() > deadline:
            break
        _, g = obj_grad_soft(h, T)
        h = project(h - lr * g, n)
        if i % 200 == 0:
            lr *= 0.95
    return h

def max_pgd_loop(h0, n, n_steps, deadline, lr=0.01):
    """Use true max subgradient"""
    h = h0.copy()
    for i in range(n_steps):
        if time.time() > deadline:
            break
        _, g = obj_grad_max(h)
        h = project(h - lr * g, n)
        if i % 100 == 0:
            lr *= 0.97
    return h

def lbfgs_run(h0, n, T, deadline, maxiter=150):
    def f(x):
        xc = np.clip(x, 0.0, 1.0)
        o, g = obj_grad_soft(xc, T)
        target = n / 2.0
        pen = 20.0
        s = xc.sum()
        return o + pen * (s - target) ** 2, g + 2.0 * pen * (s - target) / n
    try:
        r = minimize(f, h0, jac=True, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        if r.x is not None and np.all(np.isfinite(r.x)):
            return project(r.x, n)
    except:
        pass
    return h0
