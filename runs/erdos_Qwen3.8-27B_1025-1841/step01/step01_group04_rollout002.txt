import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def dca_one_step(h_t, n, method='highs'):

    num_lags = 2 * n - 1  # lags from -(n-1) to n-1
    
    # Build A_ub, b_ub for the LP
    # For each lag s = -(n-1), ..., 0, ..., n-1:
    #   c[s] ≈ sum_ell coeff[ell] * h[ell] + const
    #   where coeff[ell] = (1-h_t[ell-s])*I(0≤ell-s<n) - h_t[ell+s]*I(0≤ell+s<n)
    #   and const = sum_j h_t[j]*h_t[j-s] (over valid j)
    #   Constraint: (sum coeff[ell]*h[ell] + const)*dx ≤ t
    #   i.e., sum (coeff[ell]*dx)*h[ell] - t ≤ -const*dx
    
    # h_t extended with zeros
    # For index i, h_t_ext[i] = h_t[i] if 0<=i<n else 0
    
    A_ub_rows = []
    b_ub_vals = []

        coeff = np.zeros(n)
        const = 0.0

        # j-s ≥ 0 → j ≥ s (if s > 0)
        # j-s < n → j < n+s (always true since j < n and s can be negative)
        # j ≥ 0, j < n (always)

            j_max = n - 1

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(168, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n0 = len(h_start)

    # Phase 1: DCA at parent resolution
    h_d, c5_d = dca_optimize(best_h, n0, max_iters=35, deadline=deadline)
    if c5_d < best_c5:
        best_c5 = c5_d
        best_h = h_d.copy()

    # Phase 2: SLSQP polish
    if time.time() < deadline - 80:
        h_p = slsqp_polish(best_h, n0, deadline)
        c5_p = evaluate_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 3: DCA again (refined)
    if time.time() < deadline - 100:
        h_d2, c5_d2 = dca_optimize(best_h, n0, max_iters=20, deadline=deadline)
        if c5_d2 < best_c5:
            best_c5 = c5_d2
            best_h = h_d2.copy()
        h_p2 = slsqp_polish(best_h, n0, deadline)
        c5_p2 = evaluate_c5(h_p2)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()

    # Phase 4: Higher resolution DCA
    for mult in [2, 3]:
        n_hi = n0 * mult
        if time.time() > deadline - 120:
            break
        h_hi = interp_to(best_h, len(best_h), n_hi)
        h_hi = project(h_hi, n_hi)
        h_dh, c5_dh = dca_optimize(h_hi, n_hi, max_iters=20, deadline=deadline)
        if c5_dh < best_c5:
            best_c5 = c5_dh
            best_h = h_dh.copy()
        if time.time() < deadline - 60:
            h_ph = slsqp_polish(best_h, len(best_h), deadline)
            c5_ph = evaluate_c5(h_ph)
            if c5_ph < best_c5:
                best_c5 = c5_ph
                best_h = h_ph.copy()

    # Phase 5: Random restarts with DCA
    nb = len(best_h)
    rng = np.random.RandomState(seed)
    for attempt in range(40):
        if time.time() > deadline - 15:
            break
        sigma = 0.04 if attempt < 10 else 0.02
        h_pert = best_h + rng.randn(nb) * sigma
        h_pert = project(h_pert, nb)
        h_r, c5_r = dca_optimize(h_pert, nb, max_iters=12, deadline=deadline)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Alternating DCA and SLSQP on best
    for _ in range(3):
        if time.time() > deadline - 15:
            break
        h_a, c5_a = dca_optimize(best_h, nb, max_iters=10, deadline=deadline)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
        if time.time() > deadline - 10:
            break
        h_b = slsqp_polish(best_h, nb, deadline)
        c5_b = evaluate_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
