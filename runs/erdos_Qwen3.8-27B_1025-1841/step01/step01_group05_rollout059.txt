```python
    # Bisection to find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
```

```python
def compute_subgradient(h, n, k_star):
    """Compute subgradient of out[k_star] with respect to h."""
    grad = np.zeros(n)
    # out[k] = sum_{i in S_k} h[i+k-n+1] * (1-h[i])
    # S_k = {i : max(0, n-1-k) <= i <= min(n-1, 2n-2-k)}
    i_lo = max(0, n-1-k_star)
    i_hi = min(n-1, 2*n-2-k_star)
    for i in range(i_lo, i_hi+1):
        j = i + k_star - n + 1
        # h[j] * (1-h[i])
        # d/dh[j]: (1-h[i])
        grad[j] += (1 - h[i])
        # d/dh[i]: -h[j]
        grad[i] -= h[j]
    return grad
```

```python
    grad = np.zeros(n)
    i_lo = max(0, n - 1 - k)
    i_hi = min(n - 1, 2 * n - 2 - k)
    if i_lo > i_hi:
        return grad

    j = i + k - n + 1
    np.add.at(grad, j, 1.0 - h[i])
    np.add.at(grad, i, -h[j])

    alpha = alpha_init
```

```python
def project_feasible(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, n, k):
    """Gradient of correlate(h,1-h,'full')[k] w.r.t. h"""
    grad = np.zeros(n)
    i_lo = max(0, n - 1 - k)
    i_hi = min(n - 1, 2 * n - 2 - k)
    if i_lo > i_hi:
        return grad
    i = np.arange(i_lo, i_hi + 1)
    j = i + k - n + 1
    np.add.at(grad, j, 1.0 - h[i])
    np.add.at(grad, i, -h[j])
    return grad

def interp_resample(h, n_from, n_to):
    x_old = np.linspace(0.0, 2.0, n_from)
    x_new = np.linspace(0.0, 2.0, n_to)
    return np.interp(x_new, x_old, h)

def projected_subgrad(h, n, n_iters, t_limit):

    c5 = compute_c5(h)
    dx = 2.0 / n
    alpha = 1.0
    stall = 0

        c5_now = np.max(corr) * dx
        if c5_now < c5:
            c5 = c5_now
            stall = 0
        else:
            stall += 1

        grad = grad_lag(h, n, k_star) * dx

                break

            h = project_feasible(h + 0.1 * np.random.randn(n), n)
            stall = 0

def smooth_slsqp(h0, n, t_limit, alpha_sm=100.0):
    """SLSQP with smooth max approximation."""
    dx = 2.0 / n
    
    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = corr * dx
        m = np.max(vals)
        return m + np.log(np.sum(np.exp(alpha_sm * (vals - m)))) / alpha_sm
    
    def obj_grad(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = corr * dx
        m = np.max(vals)
        w = np.exp(alpha_sm * (vals - m))
        w /= np.sum(w)
        grad = np.zeros(n)

                    np.add.at(g, j, 1.0 - h[i])
                    np.add.at(g, i, -h[j])
                    grad += w[k] * g
        grad *= dx
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

                                bounds=bounds, constraints=cons,

        h_out = np.clip(res.x, 0, 1)
        h_out = project_feasible(h_out, n)
        return h_out
    except:
```

```python
import numpy as np
from scipy import optimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def grad_lag(h, n, k):
    grad = np.zeros(n)
    i_lo = max(0, n - 1 - k)
    i_hi = min(n - 1, 2 * n - 2 - k)
    if i_lo > i_hi:
        return grad
    i = np.arange(i_lo, i_hi + 1)
    j = i + k - n + 1
    np.add.at(grad, j, 1.0 - h[i])
    np.add.at(grad, i, -h[j])
    return grad


def interp_resample(h, n_from, n_to):
    x_old = np.linspace(0.0, 2.0, n_from)
    x_new = np.linspace(0.0, 2.0, n_to)
    return np.interp(x_new, x_old, h)


def projected_subgrad(h, n, n_iters, t_limit, seed_rng):
    h = project_feasible(h.copy(), n)
    c5 = compute_c5(h)
    dx = 2.0 / n
    alpha = 0.5
    best_local = h.copy()
    best_c5_local = c5
    stall = 0

    for it in range(n_iters):
        if time.time() > t_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = np.max(corr) * dx
        if c5_now < best_c5_local - 1e-12:
            best_c5_local = c5_now
            best_local = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = np.argmax(corr)
        grad = grad_lag(h, n, k_star) * dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new <= c5 + 1e-12:
            h = h_new
            c5 = c5_new
            alpha = min(alpha * 1.01, 5.0)
        else:
            alpha *= 0.5
            if alpha < 1e-12:
                break

        if stall > 500 and it % 100 == 0:
            h = project_feasible(best_local + 0.1 * seed_rng.randn(n), n)
            c5 = compute_c5(h)
            stall = 0
            alpha = 0.5

    return best_local, best_c5_local


def smooth_slsqp(h0, n, t_limit, alpha_sm, maxiter):
    dx = 2.0 / n

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = corr * dx
        m = np.max(vals)
        return m + np.log(np.sum(np.exp(alpha_sm * (vals - m)))) / alpha_sm

    def obj_grad(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = corr * dx
        m = np.max(vals)
        w = np.exp(alpha_sm * (vals - m))
        w /= np.sum(w)
        grad = np.zeros(n)
        nz = np.where(w > 1e-10)[0]
        for k in nz:
            i_lo = max(0, n - 1 - k)
            i_hi = min(n - 1, 2 * n - 2 - k)
            if i_lo <= i_hi:
                i = np.arange(i_lo, i_hi + 1)
                j = i + k - n + 1
                g = np.zeros(n)
                np.add.at(g, j, 1.0 - h[i])
                np.add.at(g, i, -h[j])
                grad += w[k] * g
        return grad * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    x0 = project_feasible(h0.copy(), n)

    try:
        res = optimize.minimize(obj, x0, method='SLSQP', jac=obj_grad,
                                bounds=bounds, constraints=cons,
                                options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return x0

    t0 = time.time()

    res_list = []

            res_list.append(n_cand)

    t_multi_end = t0 + budget_s * 0.82
    if len(res_list) > 0:

    assert abs(np.sum(best_h) - len(best_h) / 2.0) < 1e-6

    return best_h, best_c5, len(best_h)
```
