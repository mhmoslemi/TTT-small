def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def fft_correlate(h):
    n = len(h)
    size = 2 * n
    o = 1.0 - h
    fa = np.fft.rfft(h, size)
    fo = np.fft.rfft(o[::-1], size)
    result = np.fft.irfft(fa * fo, size)
    return result[:2 * n - 1]

def compute_c5(h):
    n = len(h)
    corr = fft_correlate(h)
    return float(corr.max() * (2.0 / n))

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def smooth_obj_grad(h, tau, max_k_terms=50):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h)
    max_c = corr.max()
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = float(dx * (max_c + np.log(s) / tau))
    
    # Get top-k lags by probability
    if max_k_terms < len(p):
        top_k = np.argpartition(p, -max_k_terms)[-max_k_terms:]
    else:
        top_k = np.arange(len(p))
    
    grad = np.zeros(n)
    for k in top_k:
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - n + 1
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        if i0 < i1:
            grad[i0:i1] += pk * (1.0 - h[i0 + lag:i1 + lag])
        i0b = max(0, lag)
        i1b = min(n, n + lag)
        if i0b < i1b:
            grad[i0b:i1b] -= pk * h[i0b - lag:i1b - lag]
    grad *= dx
    return obj, grad

def adam_optimize(h, tau, max_iters, time_limit, lr=0.1, max_k_terms=50):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    best_obj = smooth_obj_grad(h, tau, max_k_terms)[0]
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    stall = 0
    
    for it in range(1, max_iters + 1):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau, max_k_terms)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        # Adam step
        h_raw = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_raw, n)
        
        if stall > 200:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break
    
    return best_h, best_obj

def subgrad_optimize(h, max_iters, time_limit, lr=0.005):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = fft_correlate(h)
        c5_cur = float(corr.max() * dx)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        k_star = int(np.argmax(corr))
        lag = k_star - n + 1
        grad = np.zeros(n)
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        if i0 < i1:
            grad[i0:i1] = 1.0 - h[i0 + lag:i1 + lag]
        i0b = max(0, lag)
        i1b = min(n, n + lag)
        if i0b < i1b:
            grad[i0b:i1b] -= h[i0b - lag:i1b - lag]
        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        
        if stall > 300:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-15:
                break
    
    return best_h, best_c5

- h has n elements, representing h at x = 0, dx, 2dx, ..., (n-1)dx
- (1-h) has n elements
- np.correlate(h, 1-h, 'full') has 2n-1 elements
- Element k (0-indexed) corresponds to the cross-correlation at lag (k-(n-1))*dx
- The integral at lag L is sum_i h[i] * (1-h[i+L/dx]) * dx (approximately)

My FFT computation:
- fa = rfft(h, 2n), fo = rfft((1-h)[::-1], 2n)
- result = irfft(fa * fo, 2n)[:2n-1]
- This gives convolve(h, (1-h)[::-1], 'full')[:2n-1]
- Which equals np.correlate(h, 1-h, 'full')

Yes, this is correct.

Now, one subtlety: the FFT-based convolution can have tiny numerical errors (on the order of 1e-15). This shouldn't matter for our purposes.

import numpy as np
import time
import math


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def fft_correlate(h):
    n = len(h)
    size = 2 * n
    o = 1.0 - h
    fa = np.fft.rfft(h, size)
    fo = np.fft.rfft(o[::-1], size)
    result = np.fft.irfft(fa * fo, size)
    return result[:2 * n - 1]


def compute_c5(h):
    n = len(h)
    corr = fft_correlate(h)
    return float(corr.max() * (2.0 / n))


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def smooth_obj_grad(h, tau, max_k_terms=40):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h)
    max_c = corr.max()
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = float(dx * (max_c + np.log(s) / tau))
    if max_k_terms < len(p):
        top_k = np.argpartition(p, -max_k_terms)[-max_k_terms:]
    else:
        top_k = np.arange(len(p))
    grad = np.zeros(n)
    for k in top_k:
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - n + 1
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        if i0 < i1:
            grad[i0:i1] += pk * (1.0 - h[i0 + lag:i1 + lag])
        i0b = max(0, lag)
        i1b = min(n, n + lag)
        if i0b < i1b:
            grad[i0b:i1b] -= pk * h[i0b - lag:i1b - lag]
    grad *= dx
    return obj, grad


def adam_optimize(h, tau, max_iters, time_limit, lr=0.15, max_k_terms=40):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    best_obj = smooth_obj_grad(h, tau, max_k_terms)[0]
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    stall = 0
    for it in range(1, max_iters + 1):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau, max_k_terms)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        h_raw = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_raw, n)
        if stall > 150:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break
    return best_h, best_obj


def multi_subgrad(h, n_lags, max_iters, time_limit, lr=0.004):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = fft_correlate(h)
        c5_cur = float(corr.max() * dx)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        thresh = corr.max() - 0.005
        top = np.where(corr >= thresh)[0]
        if len(top) == 0:
            top = np.array([np.argmax(corr)])
        if len(top) > n_lags:
            top = np.argpartition(corr, -n_lags)[-n_lags:]
        grad = np.zeros(n)
        for k in top:
            lag = k - n + 1
            i0 = max(0, -lag)
            i1 = min(n, n - lag)
            if i0 < i1:
                grad[i0:i1] += 1.0 - h[i0 + lag:i1 + lag]
            i0b = max(0, lag)
            i1b = min(n, n + lag)
            if i0b < i1b:
                grad[i0b:i1b] -= h[i0b - lag:i1b - lag]
        grad *= dx / len(top)
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 200:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-15:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: smooth optim at initial resolution with moderate tau
    if remaining() > 100:
        tl = min(remaining() * 0.2, 250)
        h1 = adam_optimize(h_init.copy(), tau=25, max_iters=8000, time_limit=tl, lr=0.2, max_k_terms=30)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: increase resolution to 2000
    if remaining() > 80:
        n2 = 2000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(remaining() * 0.2, 200)
        h2 = adam_optimize(h2, tau=35, max_iters=6000, time_limit=tl, lr=0.12, max_k_terms=30)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: high resolution 4000
    if remaining() > 80:
        n3 = 4000
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        tl = min(remaining() * 0.2, 180)
        h3 = adam_optimize(h3, tau=50, max_iters=4000, time_limit=tl, lr=0.08, max_k_terms=20)
        c5_3 = compute_c5(h3)

        h4 = best_h.copy()
        h4, c5_4 = multi_subgrad(h4, n_lags=5, max_iters=200000, time_limit=remaining() * 0.2, lr=0.003)

    # Phase 5: try higher tau refinement at 2000
    if remaining() > 30:
        n5 = 2000
        h5 = interp_periodic(best_h, n5)
        h5 = project_to_feasible(h5, n5)
        h5 = adam_optimize(h5, tau=100, max_iters=3000, time_limit=remaining() * 0.15, lr=0.05, max_k_terms=10)
        c5_5 = compute_c5(h5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: perturbation restart at current res
    if remaining() > 40:
        n6 = len(best_h)
        noise = np.random.randn(n6) * 0.02
        h6 = project_to_feasible(best_h + noise, n6)
        h6 = adam_optimize(h6, tau=60, max_iters=4000, time_limit=remaining() * 0.2, lr=0.1, max_k_terms=20)
        h6, c5_6 = multi_subgrad(h6, n_lags=4, max_iters=100000, time_limit=remaining() * 0.1, lr=0.002)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: final subgrad polish
    if remaining() > 15:
        h_f = best_h.copy()
        h_f, c5_f = multi_subgrad(h_f, n_lags=3, max_iters=200000, time_limit=remaining() * 0.35, lr=0.001)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
