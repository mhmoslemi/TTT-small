            if res2.x is not None:
                h_s = project_feasible(res2.x[:n_target].copy(), n_target)
                c5_s = compute_c5(h_s, dx_t)
                if c5_s < c5_start:
                    h_start, c5_start = h_s, c5_s

        # Gradient optimization
        t_end = t0 + budget_s * 0.88 - (len(resolutions) - resolutions.index(n_target)) * budget_s * 0.05
        t_end = max(t_end, time.time() + 10)
        h_res, c5_res = gradient_optimize(h_start, n_target, dx_t, t_end, top_k=3, max_iter=60000)
        if c5_res < best_c5:
            best_h, best_c5, best_n = h_res, c5_res, n_target

    # Final: verify and return
    final_dx = 2.0 / best_n
    final_c5 = compute_c5(best_h, final_dx)
    if not (np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = fix_sum(best_h, best_n / 2.0)
        final_c5 = compute_c5(best_h, final_dx)

    return (best_h, final_c5, best_n)
