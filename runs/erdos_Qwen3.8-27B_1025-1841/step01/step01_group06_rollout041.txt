import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-15):
            h = h_new
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-12:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + rem / len(idx), 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_target(h.copy(), n_new)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def smooth_max_grad(h, temp, top_k=40):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_shift = corr - np.max(corr)
    exp_c = np.exp(temp * c_shift)
    weights = exp_c / exp_c.sum()
    if top_k < len(corr):
        idx = np.argsort(weights)[-top_k:]
        w = np.zeros_like(weights)
        w[idx] = weights[idx]
        w /= w.sum()
        weights = w
    n = len(h)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k]
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += w * (1.0 - h[lag:lag+m])
            grad[lag:] -= w * h[:m]
        else:
            lo = -lag
            m = n - lo
            grad[lo:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[lo:]
    val = np.max(corr) + np.log(exp_c.sum()) / temp
    return val, grad

def fista_optimize(h, n, temps, iters_per_temp, deadline_cb):
    h_curr = h.copy()
    for temp in temps:
        L = 4.0 * np.sqrt(n)
        t_mom = 1.0
        x_prev = h_curr.copy()
        y = h_curr.copy()
        for it in range(iters_per_temp):
            if deadline_cb():
                return h_curr
            _, grad_y = smooth_max_grad(y, temp, top_k=40)
            x_new = y - (1.0 / L) * grad_y
            x_new = project_target(x_new, n)
            if it == 0:
                y = x_new
            else:
                t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
                y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x_prev)
                y = np.clip(y, 0.0, 1.0)
                t_mom = t_new
            x_prev = h_curr
            h_curr = x_new
    return h_curr

def grad_corr_one(h, i, n):
    b = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag >= 0:
            if i < n - lag:
                b[k] += (1.0 - h[i + lag])
            if i >= lag:
                b[k] -= h[i - lag]
        else:
            lo = -lag
            m = n - lo
            if i >= lo:
                b[k] += (1.0 - h[i - lo])
            if i < m:
                b[k] -= h[i + lo]
    return b

def coord_descent(h, dx, n_sweeps, deadline_cb):
    n = len(h)
    h_curr = h.copy()
    corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
    for sweep in range(n_sweeps):
        if deadline_cb():
            break
        for i in range(n):
            old_x = h_curr[i]
            b = grad_corr_one(h_curr, i, n)
            a = corr - b * old_x
            k0 = np.argmax(a)
            k1 = np.argmax(a + b)
            candidates = [0.0, 1.0, 0.5]
            if k0 != k1 and abs(b[k0] - b[k1]) > 1e-12:
                x_cross = (a[k1] - a[k0]) / (b[k0] - b[k1])
                if 0.0 <= x_cross <= 1.0:
                    candidates.append(x_cross)
            best_x = old_x
            best_val = np.max(a + b * old_x)
            for xc in candidates:
                xc = float(np.clip(xc, 0.0, 1.0))
                v = np.max(a + b * xc)
                if v < best_val - 1e-12:
                    best_val = v
                    best_x = xc
            if abs(best_x - old_x) > 1e-14:
                delta = best_x - old_x
                corr += b * delta
                h_curr[i] = best_x
        h_curr = project_target(h_curr, n)
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
    return h_curr

def block_perturb(h, n, rng, block_size, deadline_cb):
    n_tries = 50
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr, 2.0 / n)
    dx = 2.0 / n
    for _ in range(n_tries):
        if deadline_cb():
            break
        start = rng.randint(0, max(1, n - block_size))
        h_try = h_curr.copy()
        h_try[start:start+block_size] = 1.0 - h_try[start:start+block_size]
        h_try = project_target(h_try, n)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5:
            h_curr = h_try
            best_c5 = c5_try
    return h_curr, best_c5
