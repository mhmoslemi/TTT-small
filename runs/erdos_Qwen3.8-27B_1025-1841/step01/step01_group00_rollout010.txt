def project_to_feasible(g, n):
    target = n / 2.0
    lo = max(0.0, g.min()) - 1.0
    hi = min(1.0, g.max()) + 1.0
    for _ in range(50):

        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0

    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    while N < 2*n - 1:
        N <<= 1
    f_h = np.fft.rfft(h, N)

    h = project_to_feasible(h.copy(), n_pts)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n_pts)
    beta = 0.85
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        T = T0 * (T1 / T0) ** (it / max(1, max_iters))
        corr = fast_corr(h)
        c5_cur = corr.max() * dx
        k_star = np.argmax(corr)
        s = k_star - (n_pts - 1)
        grad = np.zeros(n_pts)
        i0 = max(0, -s)
        i1 = min(n_pts, n_pts - s)
        if i0 < i1:
            idx = np.arange(i0, i1)
            grad[idx] += (1.0 - h[idx + s])
        i0b = max(0, s)
        i1b = min(n_pts, n_pts + s)
        if i0b < i1b:
            idxb = np.arange(i0b, i1b)
            grad[idxb] -= h[idxb - s]
        grad *= dx
        noise = T * np.random.randn(n_pts)
        v = beta * v - 0.005 * grad
        h_new = project_to_feasible(h + v + noise, n_pts)
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_cur
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = h.copy()
    return best_h, best_c5


def multi_tau_smooth(h, n, time_limit):
    taus = [8.0, 15.0, 30.0, 60.0, 120.0]
    best_h = h.copy()
    for tau in taus:
        if time.time() > time_limit:
            break
        elapsed_frac = (time.time() - time_limit + time_limit) / 1.0
        tl = max(5.0, (time_limit - time.time()) * 0.45)
        if tl < 5:
            break
        best_h = smooth_optimize(best_h, n, tau, tau * 2, max_iters=5000,
                                 time_limit=tl, initial_lr=max(0.05, 0.5 / tau))
    return best_h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase 1: Refine at native resolution with multi-tau smooth
    if remaining() > 100:
        h1 = multi_tau_smooth(best_h, n_init, start_time + 120)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Interpolate to 2x resolution, smooth optimize
    n2 = max(1800, 2 * n_init)
    if remaining() > 120:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl_remaining = remaining() - 10
        h2 = smooth_optimize(h2, n2, 10.0, 80.0, max_iters=8000,
                             time_limit=tl_remaining * 0.4, initial_lr=0.25)
        if remaining() > 30:
            h2 = smooth_optimize(h2, n2, 40.0, 200.0, max_iters=4000,
                                 time_limit=tl_remaining * 0.3, initial_lr=0.1)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Subgradient at n2
    if remaining() > 60:
        h3 = best_h.copy()
        n3 = len(h3)
        h3, c5_3 = subgradient_optimize(h3, n3, max_iters=200000,
                                        time_limit=remaining() * 0.35,
                                        initial_lr=0.008)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Interpolate to 4x resolution
    n4 = max(3600, 4 * n_init)
    if remaining() > 100:
        h4 = interp_periodic(best_h, n4)
        h4 = project_to_feasible(h4, n4)
        tl4 = remaining() * 0.3
        h4 = smooth_optimize(h4, n4, 10.0, 50.0, max_iters=4000,
                             time_limit=tl4 * 0.5, initial_lr=0.15)
        if remaining() > 20:
            h4 = smooth_optimize(h4, n4, 30.0, 150.0, max_iters=3000,
                                 time_limit=tl4 * 0.4, initial_lr=0.08)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Subgradient at highest resolution
    if remaining() > 50:
        n5 = len(best_h)
        h5 = best_h.copy()
        lr_init = 0.003 if n5 > 2000 else 0.006
        h5, c5_5 = subgradient_optimize(h5, n5, max_iters=500000,
                                        time_limit=remaining() * 0.35,
                                        initial_lr=lr_init)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: SA at best resolution
    if remaining() > 40:
        n6 = len(best_h)
        h6 = best_h.copy()
        T0 = 0.01 if n6 > 2000 else 0.02
        h6, c5_6 = sa_optimize(h6, n6, max_iters=100000,
                               time_limit=remaining() * 0.35, T0=T0, T1=0.0003)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Final subgradient polish
    if remaining() > 20:
        n7 = len(best_h)
        h7 = best_h.copy()
        lr_final = 0.002 if n7 > 2000 else 0.004
        h7, c5_7 = subgradient_optimize(h7, n7, max_iters=500000,
                                        time_limit=remaining() * 0.5,
                                        initial_lr=lr_final)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Try random restart from perturbed best
    if remaining() > 40:
        n8 = len(best_h)
        h8 = project_to_feasible(best_h + 0.05 * np.random.randn(n8), n8)
        h8, c5_8 = subgradient_optimize(h8, n8, max_iters=200000,
                                        time_limit=remaining() * 0.4,
                                        initial_lr=0.005)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Ensure feasibility and final check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
