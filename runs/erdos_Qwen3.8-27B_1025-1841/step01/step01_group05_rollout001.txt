The output index in the array corresponds to k + (n-1) (since k ranges from -(n-1) to n-1, and array index ranges from 0 to 2n-2).

So output[i] = c_{i-(n-1)} where c_k = sum_n h[n+k] * (1-h[n]).

Now, c_k = sum_{n: max(0,-k)≤n≤min(n-1, n-1-k)} h[n+k] * (1-h[n])

dc_k/dh[j] = sum_{n: max(0,-k)≤n≤min(n-1, n-1-k)} [1_{n+k=j}(1-h[n]) + h[n+k](-1_{n=j})]

= (1-h[j-k]) * 1_{max(0,-k)≤j-k≤min(n-1,n-1-k)} * 1_{0≤j<n}
  - h[j+k] * 1_{max(0,-k)≤j≤min(n-1,n-1-k)} * 1_{0≤j+k<n}

Simplifying the conditions:
First term: need 0≤j-k<n (n is valid) and 0≤j<n. Since j≥0 always in our range, and j-k≥0 means j≥k, and j-k≤n-1 means j≤n-1+k. Also need 0≤j<n.
So: 0≤j<n and max(0,k)≤j≤min(n-1, n-1+k)

Second term: need 0≤j+k<n and 0≤j<n. So j≥max(0,-k) and j+k≤n-1 means j≤n-1-k. Also 0≤j<n.
So: max(0,-k)≤j≤min(n-1, n-1-k)

        jm = j - k
        if 0 <= jm < n:
            grad[j] += (1.0 - h[jm])
        # Second term: -h[j+k] if j+k is in [0, n-1]
        jp = j + k
        if 0 <= jp < n:
            grad[j] -= h[jp]
    grad *= dx
    return c5, grad

        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5

def random_01(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    if n % 2 != 0:
        extra = rng.randint(n)
        if h[extra] < 0.5:
            h[extra] = 1.0 / 2.0
    return normalize_h(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_base = np.array(initial_h_values, dtype=float)
    except:
        h_base = None

    if h_base is not None:
        best_h = h_base.copy()
        best_c5 = compute_c5(best_h)
    else:
        best_h = random_01(100, rng)
        best_c5 = compute_c5(best_h)

    n_base = len(best_h)
    resolutions = sorted(set([n_base] + [100, 200, 400, 800]))

    for n in resolutions:
        if budget_s - (time.time() - t0) < 80:
            break

        h_cur = normalize_h(upsample(best_h, n), n)
        c5_cur = compute_c5(h_cur)

        if n % 2 == 0:
            h_bin = try_binary(h_cur, n)
            c5_bin = compute_c5(h_bin)
            if c5_bin < c5_cur:
                h_cur = h_bin
                c5_cur = c5_bin

        h_rand = random_01(n, rng)
        c5_rand = compute_c5(h_rand)
        if c5_rand < c5_cur:
            h_cur = h_rand
            c5_cur = c5_rand

        if c5_cur < best_c5:
            best_h = h_cur.copy()
            best_c5 = c5_cur

        # SLSQP optimization
        if n <= 800 and budget_s - (time.time() - t0) > 40:
            cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
            bnds = [(0.0, 1.0)] * n
            maxiter = 1500 if n <= 200 else (800 if n <= 500 else 300)
            n_attempts = 4 if n <= 200 else (3 if n <= 500 else 2)

            for att in range(n_attempts):
                if budget_s - (time.time() - t0) < 35:
                    break
                if att == 0:
                    h_start = h_cur.copy()
                elif att == 1:
                    h_start = try_binary(h_cur, n).copy()
                else:
                    h_start = h_cur + rng.normal(0, 0.08, n)
                    h_start = normalize_h(h_start, n)

                try:
                    res = minimize(compute_c5, h_start, method='SLSQP',
                                  jac=c5_gradient,
                                  constraints=cons, bounds=bnds,
                                  options={'maxiter': maxiter, 'ftol': 1e-12})
                    h_opt = normalize_h(res.x, n)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < c5_cur:
                        h_cur = h_opt
                        c5_cur = c5_opt
                    if c5_cur < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = c5_cur
                except:
                    pass

        # Local search (coordinate perturbation)
        if budget_s - (time.time() - t0) > 20:
            remaining = budget_s - (time.time() - t0)
            steps = int(min(10000, remaining * 10))
            h_cur, c5_cur = local_search(h_cur, n, rng, budget_steps=steps)
            if c5_cur < best_c5:
                best_h = h_cur.copy()
                best_c5 = c5_cur

        # Simulated annealing
        if budget_s - (time.time() - t0) > 50 and n <= 1000:
            h_sa = h_cur.copy()
            c5_sa = compute_c5(h_sa)
            T = 0.025
            remaining = budget_s - (time.time() - t0) - 30
            n_steps = int(min(remaining * 5, 50000))
            n_steps = max(n_steps, 1000)
            for s in range(n_steps):
                if budget_s - (time.time() - t0) < 30:
                    break
                h_new = h_sa.copy()
                npert = rng.randint(1, 4)
                for _ in range(npert):
                    idx = rng.randint(n)
                    h_new[idx] += rng.uniform(-0.4, 0.4)
                h_new = normalize_h(h_new, n)
                c5_new = compute_c5(h_new)
                d = c5_new - c5_sa
                if d < 0 or rng.random() < np.exp(-d / T):
                    h_sa = h_new
                    c5_sa = c5_new
                T *= 0.9997
                if c5_sa < best_c5:
                    best_h = h_sa.copy()
                    best_c5 = c5_sa

            if budget_s - (time.time() - t0) > 20:
                h_cur, c5_cur = local_search(h_sa, n, rng, budget_steps=3000)
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
