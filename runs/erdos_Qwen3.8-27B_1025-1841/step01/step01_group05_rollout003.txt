def fix_sum(h, n, tol=1e-10):
    """Adjust h to have exactly sum = n/2 while staying in [0,1]"""
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < tol:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def upsample(h_old, n_new):
    """Upsample h from n_old points to n_new points, preserving integral"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 2/(2*n_old)  # cell centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 2/(2*n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0, 1)
    h_new = fix_sum(h_new, n_new)
    return h_new

def binary_local_search(h, n, max_iter=1000):
    """Local search: try all single-position changes"""
    h = h.copy()
    c5_curr = c5_value(h, n)
    for it in range(max_iter):
        improved = False
        best_delta = 0
        best_j = -1
        for j in range(n):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_gradient(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m_star = int(np.argmax(corr))
    k = m_star - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * x)
    elif k > 0:
        idx1 = np.arange(k, n)
        grad[idx1] += dx * (1.0 - x[idx1 - k])
        idx2 = np.arange(0, n - k)
        grad[idx2] += dx * (-x[idx2 + k])
    else:
        p = -k
        idx1 = np.arange(0, n - p)
        grad[idx1] += dx * (1.0 - x[idx1 + p])
        idx2 = np.arange(p, n)
        grad[idx2] += dx * (-x[idx2 - p])

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_gradient(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    ms = int(np.argmax(corr))
    k = ms - (n - 1)
    g = np.zeros(n)
    if k == 0:
        g[:] = dx * (1.0 - 2.0 * x)
    elif k > 0:
        g[k:] += dx * (1.0 - x[:n-k])
        g[:n-k] += dx * (-x[k:])
    else:
        p = -k
        g[:n-p] += dx * (1.0 - x[p:])
        g[p:] += dx * (-x[:n-p])
    return g

def fix_sum(h, n):
    tgt = n * 0.5
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        if not m.any():
            break
        h[m] += d / m.sum()

    n_old = len(h)
    xo = (np.arange(n_old) + 0.5) * 2.0 / n_old
    xn = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return fix_sum(np.clip(np.interp(xn, xo, h), 0, 1), n_new)

import numpy as np
from scipy.optimize import minimize
import time


def c5_value(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def c5_gradient(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    ms = int(np.argmax(corr))
    k = ms - (n - 1)
    g = np.zeros(n)
    if k == 0:
        g[:] = dx * (1.0 - 2.0 * x)
    elif k > 0:
        g[k:] += dx * (1.0 - x[:n - k])
        g[:n - k] += -dx * x[k:]
    else:
        p = -k
        g[:n - p] += dx * (1.0 - x[p:])
        g[p:] += -dx * x[:n - p]
    return g


def fix_sum(h, n):
    tgt = n * 0.5
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    xo = (np.arange(n_old) + 0.5) * 2.0 / n_old
    xn = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(xn, xo, h_old)
    return fix_sum(np.clip(h_new, 0.0, 1.0), n_new)


def loc_search(h, n, max_it=500, t_lim=None):
    h = h.astype(float).copy()
    cv = c5_value(h, n)
    for _ in range(max_it):
        if t_lim is not None and time.time() > t_lim:
            break
        bc, bj = cv, -1
        for j in range(n):
            old = h[j]
            h[j] = 1.0 - old
            c = c5_value(h, n)
            h[j] = old

        if bj < 0:
            break
        h[bj] = 1.0 - h[bj]
        cv = bc
    return h, c5_value(h, n)


def slsqp_opt(h0, n, mi=500):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(c5_value, h0, args=(n,), method='SLSQP',
                     jac=lambda x: c5_gradient(x, n),
                     bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-12, 'disp': False})
        c0 = c5_value(h0, n)
        if r.fun < c0 - 1e-14:
            ho = fix_sum(np.clip(r.x, 0.0, 1.0), n)
            c = c5_value(ho, n)
            if c < c0:
                return ho, c
    except Exception:
        pass
    return h0.copy(), c5_value(h0, n)


def nelder_mead_opt(h0, n, max_it=500):
    def penalized(x):
        h = np.clip(x, 0.0, 1.0)
        pen = 100.0 * (h.sum() - n * 0.5) ** 2
        return c5_value(h, n) + pen
    try:
        r = minimize(penalized, h0, method='Nelder-Mead',
                     options={'maxiter': max_it, 'xatol': 1e-10, 'fatol': 1e-12})
        ho = fix_sum(np.clip(r.x, 0.0, 1.0), n)
        c = c5_value(ho, n)
        c0 = c5_value(h0, n)
        if c < c0:
            return ho, c
    except Exception:
        pass
    return h0.copy(), c5_value(h0, n)

    hb = np.zeros(n)
    hb[hs[:n // 2]] = 1.0
    return hb
