def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        if sh == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif sh > 0:
            grad[sh:] += pk * (1.0 - h[:-sh])
            grad[:-sh] -= pk * h[sh:]
        else:
            s2 = -sh
            grad[:-s2] += pk * (1.0 - h[s2:])
            grad[s2:] -= pk * h[:-s2]
    grad *= dx
    return obj, grad
