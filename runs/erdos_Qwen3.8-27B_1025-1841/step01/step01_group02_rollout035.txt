def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = np.clip(h + diff / n, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h = np.clip(h + diff / cnt * mask, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def softmax_obj_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    max_f = np.max(f)
    exp_f = np.exp(T * (f - max_f))
    S_exp = np.sum(exp_f)
    obj = max_f + np.log(S_exp) / T
    p = exp_f / S_exp
    
    # W1 via cumulative sum
    P = np.zeros(2 * n)
    P[1:] = np.cumsum(p)
    W1 = np.array([P[2*n-1-j] - P[n-1-j] for j in range(n)])
    
    # A and B via loop
    A = np.zeros(n)
    B = np.zeros(n)
    for s in range(-(n-1), n):
        pv = p[s + n - 1]
        if pv == 0.0:
            continue
        if s < 0:
            A[-s:] += pv * h[:n+s]
            B[:n+s] += pv * h[-s:]
        elif s == 0:
            A += pv * h
            B += pv * h
        else:
            A[:n-s] += pv * h[s:]
            B[s:] += pv * h[:n-s]
    
    grad = dx * (W1 - A - B)
    return obj, grad
