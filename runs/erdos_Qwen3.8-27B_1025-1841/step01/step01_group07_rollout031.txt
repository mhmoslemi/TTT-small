def thue_morse(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def eval_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def local_search(h, rng, time_limit, best_c5_tracker):

    cur = h.copy()
    cur_c5 = eval_c5(cur)
    best_local = cur.copy()
    best_local_c5 = cur_c5
    best_global_c5, best_global_h = best_c5_tracker

        best_global_c5 = cur_c5
        best_global_h = cur.copy()

    max_no_improve = 5000

        zeros = np.where(cur == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        cur[j] = 1

                best_local = cur.copy()
            if new_c5 < best_global_c5:
                best_global_c5 = new_c5
                best_global_h = cur.copy()
        else:

                cur[j] = 0

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def thue_morse_init(n, rng):
    """Create Thue-Morse based initialization."""
    h = np.zeros(n)
    for i in range(n):
        # Count bits
        x = i
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        h[i] = cnt % 2
    # Adjust to have exactly n/2 ones
    target = n // 2
    cur = int(h.sum())
    if cur < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:target-cur]] = 1
    elif cur > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:cur-target]] = 0
    return h
