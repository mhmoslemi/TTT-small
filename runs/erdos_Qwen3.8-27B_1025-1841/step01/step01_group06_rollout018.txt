from scipy.optimize import minimize
import time

_OPT = {}

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_h(h, n_new)
    h_ext = np.concatenate([h, [h[0]]])
    x_old = np.linspace(0, 2, n_old + 1)
    x_new = np.linspace(0, 2, n_new + 1)
    h_new = np.interp(x_new, x_old, h_ext)[:-1]
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_h(h_new, n_new)

def optimize_smooth(h, dx, n, temps, mu, maxiter_per_temp):
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr, dx)
    for temp in temps:
        _OPT['dx'] = dx; _OPT['temp'] = temp; _OPT['n'] = n; _OPT['mu'] = mu

def subgradient_refine(h, dx, n, n_iter):
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr, dx)
    alpha = 3.0 / n
    no_improve = 0

                    decreased = True

            if step_size < 1e-6:
                break
    return h_curr, c5_curr
