import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_box_sum(h, target):
    h = h.astype(float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    s = np.sum(proj)
    rem = target - s
    if abs(rem) > 1e-12:
        if rem > 0:
            idx = np.where(proj < 1.0 - 1e-14)[0]
        else:
            idx = np.where(proj > 1e-14)[0]
        if len(idx) > 0:
            proj[idx] += rem / len(idx)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def gradient_corr(h, n, k_full):
    s = k_full - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[s:s+m] += (1.0 - h[:m])
        grad[:m] -= h[s:s+m]
    else:
        lo = -s
        m = n - lo
        grad[lo:lo+m] += (1.0 - h[lo:])
        grad[:m] -= h[lo:]
    return grad

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return np.clip(h_new, 0.0, 1.0)

def penalized_obj(x, dx_val, target_val):
    hx = np.clip(x, 0.0, 1.0)
    penalty = 1e4 * (np.sum(hx) - target_val) ** 2
    corr = get_corr(hx)
    return np.max(corr) * dx_val + penalty

def smooth_max_val(corr, temp):
    m = np.max(corr)
    val = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    return val

def adam_optimize(h, n, dx, target, max_iter, t_deadline):
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.025
    temp = 0.5
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t_adam = 0
    best_local = compute_c5(h, dx)
    best_h = h.copy()
    stall = 0

    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        corr = get_corr(h)
        c5_val = np.max(corr) * dx
        if c5_val < best_local - 1e-10:
            best_local = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_k = np.argsort(corr)[-6:]
        sm = np.max(corr[top_k])
        wts = np.exp((corr[top_k] - sm) / temp)
        wts /= np.sum(wts)

        grad = np.zeros(n)
        for ik, wk in zip(top_k, wts):
            if wk > 1e-8:
                grad += wk * gradient_corr(h, n, ik)

        t_adam += 1
        m_acc = beta1 * m_acc + (1.0 - beta1) * grad
        v_acc = beta2 * v_acc + (1.0 - beta2) * (grad * grad)
        m_hat = m_acc / (1.0 - beta1 ** t_adam)
        v_hat = v_acc / (1.0 - beta2 ** t_adam)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_box_sum(h, target)

        if it % 400 == 399:
            lr *= 0.96
            temp *= 0.95
            if temp < 0.03:
                temp = 0.03
        if stall > 1500:
            lr *= 0.7
            temp = max(temp * 0.7, 0.02)
            stall = 0
            m_acc[:] = 0.0
            v_acc[:] = 0.0
            t_adam = 0

    return best_h, best_local
