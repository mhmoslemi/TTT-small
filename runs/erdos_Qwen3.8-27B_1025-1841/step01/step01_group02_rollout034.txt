def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_simplex(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Binary search for lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    lambda_val = (lo + hi) / 2
    result = np.clip(h - lambda_val, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_soft_obj_and_grad(h, n, T):
    """Compute soft-max objective and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_vals = np.exp(T * shifted)
    Z = np.sum(exp_vals)
    rho = exp_vals / Z
    soft_obj = max_corr + np.log(Z) / T
    
    # Compute gradient
    h_ext = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    rho_flip = rho[::-1].copy()
    
    conv_A = np.convolve(h_ext, rho_flip, mode='full')

    B = conv_B[offset:offset+n]

    grad = grad - np.sum(grad) / n

    h_new = np.interp(new_x, old_x, h)
    return project_onto_simplex(h_new, new_n / 2.0)
