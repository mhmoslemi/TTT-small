            i0 = max(0, sh)
            i1 = min(n, sh + n)
            grad[i0:i1] += pk * (1.0 - h[i0 - sh:i1 - sh])
            grad[i0 - sh:i1 - sh] -= pk * h[i0:i1]
        grad *= dx
    return obj, grad


def optimize_adam(h, tau, time_limit, lr_init=0.4, top_k=None):
    n = len(h)
    h = project_to_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    stall = 0
    stall_best = 0

    while time.time() - start < time_limit:
        obj, grad = smooth_max_gradient(h, tau, top_k)
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h - step, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            if stall_best > 5:
                lr = min(lr * 1.08, 2.0)
            stall_best = 0
        else:
            stall += 1
            stall_best += 1
            if stall > 40:
                lr *= 0.6
                stall = 0
            if lr < 1e-10:
                break
        h = h_new
    return best_h, best_c5


def optimize_subgrad_multi(h, time_limit, lr_init=0.01, n_top=5):
    n = len(h)
    h = project_to_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.88
    stall = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for k in top_idx:
            sh = k - (n - 1)
            i0 = max(0, sh)
            i1 = min(n, sh + n)
            grad[i0:i1] += (1.0 - h[i0 - sh:i1 - sh])
            grad[i0 - sh:i1 - sh] -= h[i0:i1]
        grad *= (2.0 / n) / n_top

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 300:
            lr *= 0.85
            v *= 0.4
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: n=3000, smooth max Adam
    n1 = 3000
    if rem() > 80:
        h1 = interp_periodic(best_h, n1)
        h1 = project_to_feasible(h1, n1)
        tl = rem() * 0.22
        h1, c5_1 = optimize_adam(h1, tau=80.0, time_limit=tl * 0.6, lr_init=0.5, top_k=30)
        if rem() > 20:
            h1, c5_1 = optimize_adam(h1, tau=150.0, time_limit=rem() * 0.15, lr_init=0.3, top_k=20)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: n=6000, fine resolution
    n2 = 6000
    if rem() > 80:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        h2, c5_2 = optimize_adam(h2, tau=100.0, time_limit=rem() * 0.22, lr_init=0.3, top_k=25)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: multi-shift subgradient refinement
    if rem() > 50:
        nf = len(best_h)
        hf = best_h.copy()
        hf, c5_f = optimize_subgrad_multi(hf, time_limit=rem() * 0.25, lr_init=0.01, n_top=5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()

    # Phase 4: random perturbation restart
    if rem() > 60:
        np.random.seed(seed + 7)
        hp = best_h + np.random.randn(len(best_h)) * 0.04
        hp = project_to_feasible(hp, len(best_h))
        hp, c5_p = optimize_adam(hp, tau=120.0, time_limit=rem() * 0.2, lr_init=0.25, top_k=25)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = hp.copy()

    # Phase 5: try a different random init
    if rem() > 60:
        np.random.seed(seed + 13)
        h_rand = np.clip(np.random.randn(len(best_h)) * 0.3, 0, 1)
        h_rand = project_to_feasible(h_rand, len(best_h))
        h_rand, c5_r = optimize_adam(h_rand, tau=80.0, time_limit=rem() * 0.2, lr_init=0.8, top_k=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_rand.copy()

    # Phase 6: final subgradient polish
    if rem() > 15:
        nf = len(best_h)
        hf = best_h.copy()
        hf, c5_f = optimize_subgrad_multi(hf, time_limit=rem() * 0.4, lr_init=0.005, n_top=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()

    # Final ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
