def optimize_one_coord(h, i, n, T, c, dx):
    """Find optimal value of h[i] given all other coords fixed."""
    # Precompute a[k] for each lag k
    # a[k] = (1-h[i+k])[0<=i+k<n] - h[i-k][0<=i-k<n]
    a = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        val = 0.0
        if 0 <= i+k < n:
            val += (1.0 - h[i+k])
        if 0 <= i-k < n:
            val -= h[i-k]
        a[k + (n-1)] = val
    
    old_t = h[i]
    
    def f_of_t(t):
        # c[k](t) = c[k](old) + (t - old_t) * dx * a[k]
        c_new = c + (t - old_t) * dx * a
        scaled = c_new * T
        m = np.max(scaled)
        return m + np.log(np.sum(np.exp(scaled - m))) / T
    
    # f is convex in t, minimize over [0,1]
    # Use golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_bound, b_bound = 0.0, 1.0
    c_pt = b_bound - gr * (b_bound - a_bound)
    d_pt = a_bound + gr * (b_bound - a_bound)
    fc = f_of_t(c_pt)
    fd = f_of_t(d_pt)
    for _ in range(60):
        if fc < fd:
            b_bound = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b_bound - gr * (b_bound - a_bound)
            fc = f_of_t(c_pt)
        else:
            a_bound = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_bound + gr * (b_bound - a_bound)
            fd = f_of_t(d_pt)
    
    t_opt = (a_bound + b_bound) / 2
    return t_opt

def soft_max_val_and_w(h, n, T):
    dx = 2.0 / n

    scaled = c * T
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    obj = m + np.log(np.sum(exp_s)) / T
    return obj, w, c

def soft_max_grad_from_w(h, n, T, w):

    grad = np.zeros(n)

    """Find the value of h[i] in [0,1] that minimizes max(c_new)."""
    # c_new[k] = c[k] + (t - h[i]) * dx * a_raw[k]
    # where a_raw[k] = (1-h[i+k])[valid] - h[i-k][valid]

    a_raw = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        val = 0.0
        j1 = i + k
        if 0 <= j1 < n:
            val += (1.0 - h[j1])
        j2 = i - k
        if 0 <= j2 < n:
            val -= h[j2]
        a_raw[k + (n-1)] = val
    
    old_t = h[i]
    # c_new[k] = c[k] + (t - old_t) * dx * a_raw[k]
    # Let b[k] = dx * a_raw[k], alpha[k] = c[k] - old_t * b[k]
    # c_new[k] = alpha[k] + t * b[k]
    b = dx * a_raw
    alpha = c - old_t * b
    
    def g(t):
        return np.max(alpha + t * b)
    
    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if g(m1) < g(m2):
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2.0
    return t_opt, g(t_opt)

def compute_a_raw(h, i, n):
    """Compute a_raw[k] for all lags k, for coordinate i."""
    a_raw = np.zeros(2*n-1)
    m = n - 1  # max absolute lag
    k_vals = np.arange(-m, m+1)
    # First term: (1-h[i+k]) where 0 <= i+k < n
    idx1 = i + k_vals  # potential indices into h
    valid1 = (idx1 >= 0) & (idx1 < n)
    a_raw[valid1] += (1.0 - h[idx1[valid1]])
    # Second term: -h[i-k] where 0 <= i-k < n
    idx2 = i - k_vals
    valid2 = (idx2 >= 0) & (idx2 < n)
    a_raw[valid2] -= h[idx2[valid2]]
    return a_raw

def soft_max_grad(h, n, T):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = c * T
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(-(n-1), n):
        wj = w[k + (n-1)]
        if wj < 1e-12:
            continue
        i1s = max(0, -k)
        i1e = min(n, n - k)
        if i1s < i1e:
            grad[i1s:i1e] += wj * omh[i1s+k:i1e+k]

        i2e = min(n, n + k)
        if i2s < i2e:
            grad[i2s:i2e] -= wj * h[i2s-k:i2e-k]
    grad *= dx
    return grad

def slsqp_step(h0, n, T, maxiter=200):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        scaled = c * T
        m = np.max(scaled)
        return m + np.log(np.sum(np.exp(scaled - m))) / T
    def grad(x):
        return soft_max_grad(x, n, T)
    def con_eq(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_onto_constraint(res.x, n)
    except Exception:
        pass
    return h0

def exact_coord_descent(h0, n, deadline, max_rounds=1000, seed=0):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    best = float(np.max(c))
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            k_vals = np.arange(-(n-1), n)
            idx1 = i + k_vals
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i - k_vals
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            a_raw = np.zeros(2*n-1)
            a_raw[valid1] += (1.0 - h[idx1[valid1]])
            a_raw[valid2] -= h[idx2[valid2]]
            
            b = dx * a_raw
            old_t = h[i]
            alpha = c - old_t * b
            
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(alpha + m1 * b) < np.max(alpha + m2 * b):
                    hi = m2
                else:
                    lo = m1
            
            t_opt = (lo + hi) / 2.0
            delta = (t_opt - old_t)
            if abs(delta) < 1e-15:
                continue
            new_max = float(np.max(alpha + t_opt * b))
            if new_max < best - 1e-12:
                h[i] = t_opt
                c += delta * b
                best = new_max
                improved = True
        
        if not improved:
            break
    
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    best = float(np.max(c))
    return h, best
