    target = n / 2
    # Binary search for t such that sum(clip(p_i - t, 0, 1)) = target
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(p - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(p - (lo + hi) / 2, 0, 1)
    # Fine-tune: adjust the closest-to-boundary element
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Find element closest to boundary to adjust
        # Elements strictly inside (0,1) can be adjusted
        interior = (h > 1e-8) & (h < 1 - 1e-8)
        if np.any(interior):
            idx = np.where(interior)[0][0]
            h[idx] += diff
            h[idx] = np.clip(h[idx], 0, 1)
    return h

        if len(idx) > 0:
            grad[idx] += (1.0 - h[idx - s2])
            grad[idx - s2] -= h[idx]
    return grad

def project_feasible(p, n):
    target = n / 2.0
    lo = float(p.min()) - 1.5
    hi = float(p.max()) + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(p - t, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(interior) > 0:
            h[interior[0]] = np.clip(h[interior[0]] + diff, 0.0, 1.0)
    return h

def interp_to(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

def smooth_obj(h, n, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx / eps
    m = np.max(scaled)
    return eps * (m + np.log(np.sum(np.exp(scaled - m))))

def smooth_grad(h, n, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx / eps
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            idx = np.arange(n - s)
            if len(idx) > 0:
                grad[idx] += w[k] * (1.0 - h[idx + s])
                grad[idx + s] -= w[k] * h[idx]
        else:
            s2 = -s
            idx = np.arange(s2, n)
            if len(idx) > 0:
                grad[idx] += w[k] * (1.0 - h[idx - s2])
                grad[idx - s2] -= w[k] * h[idx]
    return grad * dx / eps

def cons_eq(h, n):
    return np.sum(h) - n / 2.0

def cons_eq_jac(h, n):
    return np.ones(n)

def try_sqp(h_init, n, dx, eps_seq, max_iter_each):
    h = h_init.copy()
    for eps in eps_seq:
        try:
            res = minimize(
                smooth_obj, h, args=(n, dx, eps),
                jac=smooth_grad, method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac, 'args': (n,)}],
                options={'maxiter': max_iter_each, 'ftol': 1e-10}
            )
            h = np.clip(res.x, 0.0, 1.0)
            h = project_feasible(h, n)
        except Exception:
            pass
    return h

def project_subgrad(h_init, n, dx, time_limit):
    h = h_init.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.1
    mom = 0.8
    v = np.zeros(n)
    stagnant = 0
    t_start = time.time()
    while time.time() - t_start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        k_max = int(np.argmax(corr))
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 800:
            h = project_feasible(h + np.random.randn(n) * 0.08, n)
            v *= 0.0
            lr = 0.1
            stagnant = 0
            continue
        s = k_max - (n - 1)
        grad = gradient_at_shift(h, n, s)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        v = mom * v - lr * grad
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new <= c5_val + 1e-10:
            h = h_new
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.9
            if lr < 0.01:
                lr = 0.1
                v *= 0.0
    return best_h, best_c5

def local_search(h_init, n, dx, time_limit):
    h = h_init.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    t_start = time.time()
    step = 0.05
    while time.time() - t_start < time_limit:
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        delta = np.random.uniform(-step, step)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        elif np.random.random() < 0.1:
            h = h_new
        if (time.time() - t_start) % 5.0 > 5.0:
            step *= 0.95
    return best_h, best_c5

def binary_round(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        n_parent = 42
        h_parent = np.zeros(n_parent)
        h_parent[:n_parent // 2] = 1.0

    best_h = project_feasible(h_parent, n_parent)
    dx_p = 2.0 / n_parent
    best_c5 = compute_c5(best_h, dx_p)
    best_n = n_parent

    def elapsed_ok(frac):
        return (time.time() - t0) < budget_s * frac + 5.0

    def time_left():
        return max(budget_s - 5.0 - (time.time() - t0), 0.0)

    # Phase 1: SLSQP at parent resolution
    if time_left() > 30:
        h_sqp = try_sqp(best_h, n_parent, dx_p, [0.3, 0.15, 0.08, 0.04, 0.02], 400)
        c5_s = compute_c5(h_sqp, dx_p)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_sqp

    # Phase 2: Projected subgradient at parent
    if time_left() > 20:
        t_alloc = min(time_left() * 0.4, 400.0)
        h_pg, c5_pg = project_subgrad(best_h, n_parent, dx_p, t_alloc)
        if c5_pg < best_c5:
            best_c5, best_h = c5_pg, h_pg

    # Phase 3: Local search at parent
    if time_left() > 15:
        t_alloc = min(time_left() * 0.3, 100.0)
        hs, cs = local_search(best_h, n_parent, dx_p, t_alloc)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Binary rounding check
    h_bin = binary_round(best_h, n_parent)
    c5_bin = compute_c5(h_bin, dx_p)
    if c5_bin < best_c5:
        best_c5, best_h, best_n = c5_bin, h_bin, n_parent

    # Phase 4: Finer resolutions
    for mult in [2, 3, 4]:
        if time_left() < 15:
            break
        n_new = n_parent * mult
        if n_new > 5000:
            break
        h_i = interp_to(best_h, best_n, n_new)
        h_i = project_feasible(h_i, n_new)
        dx_i = 2.0 / n_new
        c5_i = compute_c5(h_i, dx_i)
        if c5_i < best_c5:
            best_c5, best_h, best_n = c5_i, h_i, n_new
        if time_left() > 25 and n_new <= 200:
            h_s = try_sqp(h_i, n_new, dx_i, [0.15, 0.08, 0.04], 200)
            c5_s = compute_c5(h_s, dx_i)
            if c5_s < best_c5:
                best_c5, best_h, best_n = c5_s, h_s, n_new
        if time_left() > 15:
            t_alloc = min(time_left() * 0.5, 150.0)
            h_pg2, c5_pg2 = project_subgrad(best_h, best_n, 2.0 / best_n, t_alloc)
            if c5_pg2 < best_c5:
                best_c5, best_h = c5_pg2, h_pg2

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
