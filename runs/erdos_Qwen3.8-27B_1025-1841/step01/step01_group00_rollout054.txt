    ```python
    def fast_smooth_grad(h, tau, n):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -400, 400)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s  # length 2n-1
        
        # FFT size
        N = 1
        while N < 3*n:  # need at least len(p) + len(h) - 1 = 3n-2
            N *= 2
        
        p_pad = np.zeros(N)
        p_pad[:2*n-1] = p
        h_pad = np.zeros(N)
        h_pad[:n] = h
        p_rev_pad = np.zeros(N)
        p_rev_pad[:2*n-1] = p[::-1]
        
        fp = np.fft.rfft(p_pad)
        fh = np.fft.rfft(h_pad)
        fp_rev = np.fft.rfft(p_rev_pad)
        
        # C[j] = conv(p, h)[n-1+j]
        conv1 = np.fft.irfft(fp * fh, N)
        C = conv1[n-1 : n-1+n]  # C[j] for j=0..n-1
        
        # B[j] = conv(p_rev, h)[2n-2+j]
        conv2 = np.fft.irfft(fp_rev * fh, N)
        B = conv2[2*n-2 : 2*n-2+n]  # B[j] for j=0..n-1
        
        # A[j] = prefix[n+j] - prefix[n-1-j]
        prefix = np.concatenate([[0.0], np.cumsum(p)])
        j_arr = np.arange(n)
        A = prefix[n + j_arr] - prefix[n - 1 - j_arr]
        
        grad = dx * (A - B - C)
        obj = dx * (max_c + np.log(s) / tau)
        return obj, grad
    ```
