def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    # Use np.correlate as specified
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_all_correlations(h):
    """Compute all correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def upsample(h, new_n):
    """Upsample h to new_n points using linear interpolation, maintain sum constraint."""
    n = len(h)
    old_x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(new_x, old_x, h)
    h_new = np.clip(h_new, 0, 1)
    # Adjust to satisfy sum constraint
    target_sum = new_n / 2.0
    current_sum = np.sum(h_new)
    if current_sum > 0:
        h_new *= target_sum / current_sum
        h_new = np.clip(h_new, 0, 1)
    # Fine-tune: add/remove small amounts
    diff = target_sum - np.sum(h_new)
    if abs(diff) > 1e-10:
        # Distribute the difference
        step = np.sign(diff) * min(abs(diff) / new_n, 0.01)
        h_new += step
        h_new = np.clip(h_new, 0, 1)
        # Final adjustment on one element

def softmax_objective(h, beta=50):
    """Softmax approximation of max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx

    max_c = np.max(corr)

    shifted = corr - max_c
    log_sum = np.log(np.sum(np.exp(beta * shifted)) / len(corr)) / beta
    return max_c + log_sum

        h_minus[j] -= eps

Key insight: the `np.correlate` call is the bottleneck. For n=400, it's a correlation of two length-400 vectors, producing a length-799 output. NumPy implements this via FFT, so it's O(n log n) ≈ 400 * 9 ≈ 3600 ops. Very fast.

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def upsample(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0, 2, n + 1)
    h_old = np.concatenate([h, [h[-1]]])  # endpoint
    # Actually, h is defined at points 0, 2/n, 4/n, ..., 2(n-1)/n
    # Let's use midpoint interpretation
    x_old = (np.arange(n) + 0.5) / n * 2
    x_new = (np.arange(new_n) + 0.5) / new_n * 2
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0, 1)
    target = new_n / 2.0
    s = np.sum(h_new)
    if s > 1e-10:
        h_new *= target / s
        h_new = np.clip(h_new, 0, 1)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:
        candidates = np.where((h_new > 1e-9) & (h_new < 1-1e-9))[0]
        if len(candidates) > 0:
            h_new[candidates] += diff / len(candidates)
            h_new = np.clip(h_new, 0, 1)
        resid = target - np.sum(h_new)
        if abs(resid) > 1e-14:
            if resid > 0:
                idx = np.argmin(h_new)
                h_new[idx] = min(1.0, h_new[idx] + resid)
            else:
                idx = np.argmax(h_new)
                h_new[idx] = max(0.0, h_new[idx] + resid)
    return h_new

def smooth_max_obj(h, beta=200.0, lam=500.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = np.max(corr)
    # For numerical stability
    shifted = beta * (corr - c_max)
    log_sum = np.logaddexp.reduce(shifted) - np.log(len(corr))
    obj = c_max + log_sum / beta
    obj += lam * (np.sum(h) - n/2.0)**2
    return obj

def binary_sa(h_init, time_limit=60, T_start=0.02, cooling=0.99995):
    n = len(h_init)
    target = n // 2
    b = (h_init > 0.5).astype(np.float64)
    n_ones = int(np.sum(b))
    if n_ones > target:
        idx = np.where(b == 1)[0]
        np.random.shuffle(idx)
        b[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(b == 0)[0]
        np.random.shuffle(idx)
        b[idx[:target - n_ones]] = 1
    
    cur_c5 = compute_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    T = T_start
    t_start = time.time()
    
    while time.time() - t_start < time_limit:
        T *= cooling
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        b[i] = 0; b[j] = 1
        new_c5 = compute_c5(b)
        delta = new_c5 - cur_c5
        if delta <= 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1; b[j] = 0
    
    return best_b, best_c5

    n = len(h_init)

    best = cur.copy()

    T = T_start

        T *= cooling
        i = np.random.randint(n)

            continue

        made = False
        if cur[i] >= step and cur[j] <= 1.0 - step:
            new = cur.copy()
            new[i] -= step; new[j] += step
            made = True
        elif cur[j] >= step and cur[i] <= 1.0 - step:
            new = cur.copy()
            new[j] -= step; new[i] += step
            made = True
        
        if made:
            new_c5 = compute_c5(new)

                cur = new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best = cur.copy()
    
    return best, best_c5
