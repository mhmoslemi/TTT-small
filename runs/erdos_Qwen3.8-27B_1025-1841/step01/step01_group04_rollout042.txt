```python
    dx = 2.0 / n
    y = 1.0 - x
    corr = np.correlate(x, y, mode='full')  # length 2n-1
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    w = exp_vals / np.sum(exp_vals)  # weights, length 2n-1
    
    grad = np.zeros(n)
    for j in range(n):
        for k in range(2*n-1):
            l = k - (n-1)
            # Term 1: (1-x[j+l]) if 0 <= j+l < n
            if 0 <= j + l < n:
                grad[j] += w[k] * (1.0 - x[j+l])
            # Term 2: -x[j-l] if 0 <= j-l < n
            if 0 <= j - l < n:
                grad[j] -= w[k] * x[j-l]
    grad *= dx
    return grad
```

```python
def compute_gradient(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0-x, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    w = exp_vals / np.sum(exp_vals)
    
    # w[k] corresponds to lag l = k - (n-1)
    # grad[j] = dx * sum_l w_{l} * [(1-x[j+l])*1[0<=j+l<n] - x[j-l]*1[0<=j-l<n]]
    
    # Create lag array: lags = np.arange(-(n-1), n)
    lags = np.arange(-(n-1), n)  # length 2n-1
    # w is already indexed the same way: w[k] for k=0..2n-2 corresponds to lags[k]
    
    # For each j, compute:
    # term1[j] = sum_l w[k] * (1-x[j+l]) * 1[0<=j+l<n]
    # term2[j] = sum_l w[k] * x[j-l] * 1[0<=j-l<n]
    
    # Vectorize: create matrix M where M[j,k] = (1-x[j+lags[k]]) if valid, 0 otherwise
    # This is O(n * 2n) = O(n²) memory, fine for n=672.
    
    # j ranges 0..n-1, lags[k] ranges -(n-1)..n-1
    # j + lags[k] ranges -(n-1)..2n-2, valid when 0 <= j+lags[k] < n
    
    j_idx = np.arange(n)[:, None]  # n x 1
    l = lags[None, :]  # 1 x (2n-1)
    
    # For term 1: valid when 0 <= j+l < n
    valid1 = (j_idx + l >= 0) & (j_idx + l < n)
    # (1-x[j+l]) where valid
    idx1 = np.clip(j_idx + l, 0, n-1)
    term1 = np.sum(w[None, :] * (1.0 - x[idx1]) * valid1, axis=1)
    
    # For term 2: valid when 0 <= j-l < n
    valid2 = (j_idx - l >= 0) & (j_idx - l < n)
    idx2 = np.clip(j_idx - l, 0, n-1)
    term2 = np.sum(w[None, :] * x[idx2] * valid2, axis=1)
    
    grad = dx * (term1 - term2)
    return grad
```

```python
def top_k_directed_refine(x, n, K=5, n_steps=5000, lr0=0.05, deadline=None):
    x = x.copy()
    dx = 2.0 / n
    lr = lr0
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        # Top K lags
        top_k = np.argsort(corr)[-K:]
        c_max = np.max(corr)
        
        # Weighted gradient (weighted by how close each lag is to the max)
        weights = np.zeros(2*n-1)
        for k in top_k:
            weights[k] = np.exp(50 * (corr[k] - c_max))  # sharp weights around max
        
        # Normalize
        weights /= np.sum(weights)
        
        # Compute weighted gradient
        j = np.arange(n)
        grad = np.zeros(n)
        for k in top_k:
            if weights[k] < 1e-10:
                continue
            l = k - (n-1)
            valid1 = (j + l >= 0) & (j + l < n)
            valid2 = (j - l >= 0) & (j - l < n)
            idx1 = np.clip(j + l, 0, n-1)
            idx2 = np.clip(j - l, 0, n-1)
            grad += weights[k] * dx * ((1.0 - x[idx1]) * valid1 - x[idx2] * valid2)
        
        x_new = x - lr * grad
        x_new = project(x_new, n)
        
        corr_new = np.correlate(x_new, 1.0 - x_new, mode='full') * dx
        c_new = np.max(corr_new)

            x = x_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return x
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def softmax_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    w = exp_vals / np.sum(exp_vals)
    
    lags = np.arange(-(n-1), n)
    j_idx = np.arange(n)[:, None]
    l = lags[None, :]
    
    valid1 = (j_idx + l >= 0) & (j_idx + l < n)
    idx1 = np.clip(j_idx + l, 0, n - 1)
    term1 = np.sum(w[None, :] * (1.0 - x[idx1]) * valid1, axis=1)
    
    valid2 = (j_idx - l >= 0) & (j_idx - l < n)
    idx2 = np.clip(j_idx - l, 0, n - 1)
    term2 = np.sum(w[None, :] * x[idx2] * valid2, axis=1)
    
    return dx * (term1 - term2)

def slsqp_opt(x0, n, T, maxiter=200, deadline=None):
    target = n / 2.0
    try:
        res = minimize(softmax_obj, x0, method='SLSQP', jac=lambda x: softmax_grad(x, n, T),
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        if deadline is not None and time.time() > deadline:
            return x0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return x0

def max_directed_refine(x, n, n_steps=5000, lr0=0.05, deadline=None):
    x = x.copy()
    dx = 2.0 / n
    lr = lr0
    best_c = evaluate_c5(x)
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        k_max = np.argmax(corr)
        c_max = corr[k_max]
        l = k_max - (n - 1)
        
        j = np.arange(n)
        valid1 = (j + l >= 0) & (j + l < n)
        valid2 = (j - l >= 0) & (j - l < n)
        idx1 = np.clip(j + l, 0, n - 1)
        idx2 = np.clip(j - l, 0, n - 1)
        
        grad = dx * ((1.0 - x[idx1]) * valid1 - x[idx2] * valid2)
        
        x_new = x - lr * grad
        x_new = project(x_new, n)
        
        c_new = evaluate_c5(x_new)
        
        if c_new <= c_max + 1e-12:
            x = x_new
            if c_new < best_c:
                best_c = c_new
            lr = min(lr * 1.01, 0.2)
        else:
            lr *= 0.85
            if lr < 1e-12:
                break
    
    return x
```
