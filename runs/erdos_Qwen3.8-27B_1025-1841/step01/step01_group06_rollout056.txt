                    g += w[k] * grad_corr_k(hx, n_local, int(k))
                g *= dx_local
                g += 2.0 * pen_coef * (np.sum(hx) - target_sum)
                return g

            try:
                res = minimize(lbgfgs_obj, h, jac=lbgfgs_grad, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 2000, 'ftol': 1e-15,
                                       'gtol': 1e-10, 'maxls': 40})
                if np.isfinite(res.fun):
                    h_ref = project_h(res.x, n)
                    c5_ref = compute_c5(h_ref, dx)
                    if c5_ref < c5:
                        h = h_ref
                        c5 = c5_ref
                        best_c5 = c5
                        best_h = h.copy()
                        best_n = n
            except Exception:
                pass

    # Phase 3: Fine-tuning with small projected gradient steps
    if time.time() < t0 + budget_s * 0.88:
        alpha = 0.3
        for it in range(6000):
            if time.time() > deadline:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = int(np.argmax(corr))
            g = grad_corr_k(h, n, k_star)
            h_new = project_h(h - alpha * g, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5 - 1e-13:
                h = h_new
                c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

    # Phase 4: Multistart perturbation
    if time.time() < t0 + budget_s * 0.90:
        for restart in range(5):
            if time.time() > deadline:
                break
            h_pert = project_h(best_h + 0.015 * rng.randn(best_n), best_n)
            c5_pert = compute_c5(h_pert, 2.0 / best_n)
            if c5_pert < best_c5:
                # Quick local refinement
                alpha = 0.4
                for _ in range(500):
                    if time.time() > deadline:
                        break
                    dx_r = 2.0 / best_n
                    nr = best_n
                    corr_r = np.correlate(h_pert, 1.0 - h_pert, mode="full")
                    ks = int(np.argmax(corr_r))
                    gr = grad_corr_k(h_pert, nr, ks)
                    h_pert = project_h(h_pert - alpha * gr, nr)
                    c5_pert = compute_c5(h_pert, dx_r)
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_pert.copy()

    # Phase 5: Try parent resolution n0
    h_down = project_h(
        np.interp(np.linspace(0, 2, n0, endpoint=False),
                  np.linspace(0, 2, best_n, endpoint=False), best_h), n0)
    c5_down = compute_c5(h_down, 2.0 / n0)
    if c5_down < best_c5:
        best_c5 = c5_down
        best_h = h_down
        best_n = n0

    # Final verification
    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
