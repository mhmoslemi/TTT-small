import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Water-filling projection
    for _ in range(50):
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        new_diff = target - np.sum(h)
        if abs(new_diff) < 1e-14:
            break
        # Only adjust free variables
        if new_diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += new_diff / cnt
        h = np.clip(h, 0.0, 1.0)
        diff = new_diff
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, np.concatenate([h, [h[0]]]))
    return make_feasible(h_new, new_n)
