import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def smooth_max_grad(h, dx, temp, n):
    """Returns (objective, gradient) of smoothed max of correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    log_c = (corr - m) / temp
    log_c = np.minimum(log_c, 0.0)
    log_c = np.maximum(log_c, -40.0)
    w = np.exp(log_c)
    w_sum = w.sum()
    w /= w_sum
    obj = (m + temp * np.log(w_sum)) * dx
    
    grad = np.zeros(n)
    lags = np.arange(len(corr)) - (n - 1)
    for k_idx in range(len(corr)):
        wk = w[k_idx]
        if wk < 1e-9:
            continue
        l = int(lags[k_idx])
        if l >= 0:
            grad[l:] += wk * (1.0 - h[:n - l])
            grad[:n - l] += wk * (-h[l:])
        else:
            lp = -l
            grad[:n - lp] += wk * (1.0 - h[lp:])
            grad[lp:] += wk * (-h[:n - lp])
    
    return obj, grad

def optimize_lbfgs(h, n, dx, budget_frac):
    """Optimize using L-BFGS-B with penalized objective."""
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for temp in [0.05, 0.02, 0.01, 0.005]:
        mu = 50.0 / (dx * dx)
        
        def obj(x):
            hx = np.clip(x, 0.0, 1.0)
            sm, _ = smooth_max_grad(hx, dx, temp, n)
            penalty = mu * (np.sum(hx) - target) ** 2
            return sm + penalty
        
        def grad(x):
            hx = np.clip(x, 0.0, 1.0)
            _, g = smooth_max_grad(hx, dx, temp, n)
            g += 2 * mu * (np.sum(hx) - target)
            return g

def projected_gradient(h, n, dx, n_iter, t0, deadline):

    best_c5 = compute_c5(h, dx)
    y = h.copy()
    prev_y = h.copy()
    target = n / 2.0

        y = h_new

For a step function with mean 1/2 (since ∫h = 1 over [0,2] means average = 1/2), the minimum overlap would be achieved when h and its shifts are maximally anti-correlated. For a single shift, the minimum of ∫h(x)(1-h(x+k)) would be 0 (when h(x)=1 wherever h(x+k)=0 and vice versa). But we need this for ALL k simultaneously, which is impossible.

The best we can do is minimize the maximum over k. For a "checkerboard" pattern (alternating 0 and 1), the odd shifts give 0 overlap and even shifts give maximum overlap. So the max would be ∫h(1-h) = ∫(1/2)(1/2) = 1/4... wait no.

If h alternates 1,0,1,0,... (equal blocks), then ∫h = 1 (half the domain is 1), and for shift k=1 (half a period), h(x) and h(x+1) are complementary, so ∫h(x)(1-h(x+1)) = 0. But for k=0, ∫h(1-h) = 0 (since h is 0 or 1). For k=2 (full period), same as k=0.

Wait, if h is a step function taking values 0 or 1, then h(1-h) = 0 always! So ∫h(x)(1-h(x+k)) would be... hmm, no. ∫h(x)(1-h(x+k)) means we integrate the product of h(x) and (1-h(x+k)). If both are step functions:
- Where h(x)=1 and h(x+k)=0: contribution is 1*1=1
- Where h(x)=1 and h(x+k)=1: contribution is 1*0=0
- Where h(x)=0: contribution is 0

So ∫h(x)(1-h(x+k)) = measure of {x : h(x)=1 and h(x+k)=0}

For a checkerboard with period 2p (p blocks of h=1, p blocks of h=0), for shift 2p (one full period): ∫ = measure of h=1 = 1. For shift p (half period): ∫ = 0 (complementary).

So the max would be 1 (for shift 0 or full period). That's bad!

Wait, shift 0: ∫h(x)(1-h(x)) = 0 for a step function. OK so for a pure step function, the k=0 term is always 0.

For the checkerboard: the problematic shifts are those that align the 1-blocks of h(x) with the 0-blocks of h(x+k), and vice versa. The maximum overlap occurs at shifts where the 1-blocks of h(x) overlap with the 1-blocks of h(x+k) the least (i.e., h(x+k) is 0 where h(x) is 1 the most).

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h *= target / max(s, 1e-30)
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_target(h_new, n_new)

def compute_grad_correlations(h, n, k_indices):
    """Compute gradient of sum of correlations at specified lag indices."""
    grad = np.zeros(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    for k_idx in k_indices:
        l = int(lags[k_idx])
        if l >= 0:
            grad[l:] += (1.0 - h[:n - l])
            grad[:n - l] += (-h[l:])
        else:
            lp = -l
            grad[:n - lp] += (1.0 - h[lp:])
            grad[lp:] += (-h[:n - lp])
    return grad / len(k_indices)
