    scale = fnorm ** (p-1) * dx

    # First term: sum_k w[k] * (1-h)[j+k-(n-1)]
    # Second term: sum_k w[k] * h[j-k+(n-1)]
    
    # For efficiency, use matrix operations:
    # For j in 0..n-1, k in 0..2n-2:

    # idx2 = j - k + (n-1), valid if 0 <= idx2 < n

    K = np.arange(2*n-1)[None, :]  # (1, 2n-1)

    idx2 = J - K + (n-1)  # (n, 2n-1)

    idx1_c = np.clip(idx1, 0, n-1)
    idx2_c = np.clip(idx2, 0, n-1)

    term2 = valid2 * h[idx2_c] * w           # (n, 2n-1)
    
    grad = scale * (term1.sum(axis=1) - term2.sum(axis=1))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def project_h(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    return (m + np.log(np.sum(exps)) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    s = np.sum(exps)
    if s < 1e-300:
        return np.zeros(n)
    weights = exps / s
    J = np.arange(n)[:, None]
    K = np.arange(2 * n - 1)[None, :]
    idx1 = J + K - (n - 1)
    idx2 = J - K + (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2c = np.clip(idx2, 0, n - 1)
    th = 1.0 - h
    term1 = (th[idx1c] * weights)[valid1]
    term2 = (h[idx2c] * weights)[valid2]
    grad = dx * (term1.sum(axis=1) - term2.sum(axis=1))
    return grad

def proj_grad_desc(h, n, T, max_iter=400, init_step=None):
    h = h.copy()
    f_best = softmax_obj(h, T)
    step = init_step if init_step else 0.05
    prev_grad = None
    momentum = 0.9
    for it in range(max_iter):
        g = softmax_grad(h, T)
        if prev_grad is not None:
            g_net = g + momentum * prev_grad
        else:
            g_net = g
        prev_grad = g
        gn = np.linalg.norm(g_net)
        if gn < 1e-13:
            break
        step_it = step
        improved = False
        for _ in range(40):
            h_try = project_h(h - step_it * g_net, n)
            f_try = softmax_obj(h_try, T)
            if f_try < f_best - 1e-14:
                h = h_try
                f_best = f_try
                improved = True
                break
            step_it *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
        else:
            step = min(step * 1.15, 0.5)
    return h, f_best

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def slsqp_polish(h, n, T, max_iter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(corr)
        sh = corr - mx
        return (mx + np.log(np.sum(np.exp(T * sh))) / T) * dx
    def obj_jac(x):
        return softmax_grad(x, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success or (res.fun is not None and res.fun < softmax_obj(h, T)):
            h_new = project_h(res.x, n)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h, compute_c5(h)

def local_search(h, n, max_rounds=40):
    improved = True
    rnd = 0
    c5 = compute_c5(h)
    while improved and rnd < max_rounds:
        improved = False
        rnd += 1
        step = max(0.02, 0.15 / (rnd + 1))
        for j in range(n):
            old = h[j]
            best_trial = None
            best_c = c5
            for delta in [step, -step, 2*step, -2*step]:
                cand = np.clip(old + delta, 0.0, 1.0)
                if cand == old:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-12:
                    best_c = c
                    best_trial = trial.copy()
            if old < 0.3:
                trial = h.copy()
                trial[j] = 1.0
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-12:
                    best_c = c
                    best_trial = trial.copy()
            elif old > 0.7:
                trial = h.copy()
                trial[j] = 0.0
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-12:
                    best_c = c
                    best_trial = trial.copy()
            if best_trial is not None:
                h = best_trial
                c5 = best_c
                improved = True
    return h, c5
