def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_to(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    # Clip for numerical stability
    shifted = np.clip(T * (corr - mx), -500, 500)
    return (mx + np.log(np.sum(np.exp(shifted))) / T) * dx

def lse_grad(h, n, T):

    shifted = np.clip(T * (corr - mx), -500, 500)

    w /= w.sum()

    A = np.zeros(n)
    A[0] = w[:n].sum()  # sum of w[0] to w[n-1]

    B_full = np.convolve(h, w, mode='full')
    B = B_full[n-1 : 2*n-1]
    
    grad = dx * (A - 2.0 * B)

    """SLSQP optimization with log-sum-exp objective."""
    def obj(h_):
        return lse_value(h_, n, T)
    
    def obj_grad(h_):
        return lse_grad(h_, n, T)
    
    def cons_fun(h_):
        return h_.sum() - n / 2.0
    
    def cons_jac(h_):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds=20, time_limit=None, start_time=None):
    """Coordinate descent: for each coordinate, find the best value that minimizes C5."""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if start_time is not None and time.time() - start_time > time_limit - 15:
            break
        improved = False
        for j in range(n):
            if start_time is not None and time.time() - start_time > time_limit - 15:
                return h, best_c5
            old = h[j]
            # Try a few values
            candidates = [0.0, 1.0, 0.5, old - 0.1, old + 0.1, old - 0.05, old + 0.05]
            candidates = [max(0.0, min(1.0, c)) for c in candidates if 0.0 <= c <= 1.0]

                trial[j] = c
                trial = make_feasible(trial, n)

                    best_val = c
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5
