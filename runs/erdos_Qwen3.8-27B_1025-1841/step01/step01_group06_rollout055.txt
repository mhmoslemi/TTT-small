        # Phase 4: Nelder-Mead (small n only)
        if n <= 160 and tl() > 70:
            for rst in range(4):
                if tl() < 25:
                    break
                hp = project_feasible(h_c + rng.randn(n) * 0.015, n)

                def nmo(x):
                    x = project_feasible(x, n)
                    return compute_c5(x, dx)[0]

                try:
                    r = minimize(nmo, hp, method='Nelder-Mead',
                                 options={'maxiter': 5000, 'xatol': 1e-11,
                                          'fatol': 1e-13, 'adaptive': True})
                    ht = project_feasible(r.x, n)
                    ct, _ = compute_c5(ht, dx)
                    if ct < best_c5:
                        best_c5, best_h, best_n = ct, ht.copy(), n
                        h_c = ht.copy()
                except Exception:
                    pass

    # Final polish: a few subgradient steps on best
    if tl() > 5:
        n_f = best_n
        dx_f = 2.0 / n_f
        h_f = best_h.copy()
        step_f = 0.1
        for _ in range(2000):
            if tl() < 2:
                break
            corr = np.correlate(h_f, 1.0 - h_f, mode="full")
            k_s = int(np.argmax(corr))
            g = grad_corr(h_f, n_f, k_s) * dx_f
            h_f = project_feasible(h_f - step_f * g, n_f)
            c5_f, _ = compute_c5(h_f, dx_f)
            if c5_f < best_c5:
                best_c5, best_h = c5_f, h_f.copy()
            step_f *= 0.999

    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
