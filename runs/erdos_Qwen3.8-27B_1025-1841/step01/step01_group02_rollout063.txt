```python
def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    p = np.exp(T * shifted)
    p /= np.sum(p)
    
    lags = np.arange(-(n-1), n)  # length 2n-1
    
    grad = np.zeros(n)
    for i in range(n):
        # Term A: sum p[k] where 0 <= i + lags[k] < n
        valid_a = (i + lags >= 0) & (i + lags < n)
        A = np.sum(p[valid_a])
        
        # Term B: sum p[k]*h[i+lags[k]] where 0 <= i+lags[k] < n
        idx_b = i + lags[valid_a]
        B = np.sum(p[valid_a] * h[idx_b])
        
        # Term C: sum p[k]*h[i-lags[k]] where 0 <= i-lags[k] < n
        valid_c = (i - lags >= 0) & (i - lags < n)
        idx_c = i - lags[valid_c]
        C = np.sum(p[valid_c] * h[idx_c])
        
        grad[i] = dx * (A - B - C)
    
    return grad
```

This is O(n * n) = O(n²) per gradient evaluation. For n=320, that's ~100,000 operations. For n=640, ~400,000. With L-BFGS-B doing maybe 100-200 iterations, that's manageable.

```python
def _compute_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(T * shifted)
    sum_exp = np.sum(exp_vals)
    obj = (mc + np.log(sum_exp) / T) * dx
    
    p = exp_vals / sum_exp

    i_idx = np.arange(n)[:, None]

    A = valid @ p

    h_pad[n:2*n] = h
```

```python
def _compute_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(T * shifted)
    sum_exp = np.sum(exp_vals)
    obj = (mc + np.log(sum_exp) / T) * dx
    
    w = exp_vals / sum_exp  # weights, length 2n-1
    lags = np.arange(-(n-1), n)  # lags[k] = k-(n-1), length 2n-1
    
    i_idx = np.arange(n)[:, None]  # (n, 1)
    
    # A[i] = sum_k w[k] * [0 <= i - lags[k] < n]
    # i - lags[k] = i - (k - (n-1)) = i + n - 1 - k
    # valid when 0 <= i+n-1-k < n, i.e., i <= k <= i+n-1
    idx_B = i_idx + n - 1 - np.arange(2*n-1)[None, :]  # i + n - 1 - k, shape (n, 2n-1)
    valid_B = (idx_B >= 0) & (idx_B < n)  # (n, 2n-1)
    A = valid_B.astype(float) @ w  # (n,)
    
    # B[i] = sum_k w[k] * h[i - lags[k]] * valid = sum_k w[k] * h[idx_B[i,k]] * valid_B[i,k]
    h_pad = np.zeros(3 * n)
    h_pad[n:2*n] = h
    # h[idx_B[i,k]] when valid = h_pad[idx_B[i,k] + n]
    idx_B_pad = np.clip(idx_B + n, 0, 3*n - 1)
    h_vals_B = h_pad[idx_B_pad]  # (n, 2n-1)
    B = (w[None, :] * h_vals_B * valid_B).sum(axis=1)  # (n,)
    
    # C[i] = sum_k w[k] * h[i + lags[k]] * [0 <= i + lags[k] < n]
    # i + lags[k] = i + k - (n-1)
    idx_C = i_idx + np.arange(2*n-1)[None, :] - (n - 1)  # i + k - (n-1), shape (n, 2n-1)
    valid_C = (idx_C >= 0) & (idx_C < n)
    idx_C_pad = np.clip(idx_C + n, 0, 3*n - 1)
    h_vals_C = h_pad[idx_C_pad]
    C = (w[None, :] * h_vals_C * valid_C).sum(axis=1)
    
    grad = dx * (A - B - C)
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_GLOBAL_T = 40.0
_GLOBAL_MU = 1000.0

def _compute_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(T * shifted)
    sum_exp = np.sum(exp_vals)
    obj = (mc + np.log(sum_exp) / T) * dx
    w = exp_vals / sum_exp
    lags = np.arange(-(n - 1), n)
    k_arr = np.arange(2 * n - 1)[None, :]
    i_idx = np.arange(n)[:, None]
    h_pad = np.zeros(3 * n)
    h_pad[n:2 * n] = h
    idx_B = i_idx + n - 1 - k_arr
    valid_B = (idx_B >= 0) & (idx_B < n)
    A = valid_B.astype(float) @ w
    idx_B_pad = np.clip(idx_B + n, 0, 3 * n - 1)
    h_vals_B = h_pad[idx_B_pad]
    B = (w[None, :] * h_vals_B * valid_B.astype(float)).sum(axis=1)
    idx_C = i_idx + k_arr - (n - 1)
    valid_C = (idx_C >= 0) & (idx_C < n)
    idx_C_pad = np.clip(idx_C + n, 0, 3 * n - 1)
    h_vals_C = h_pad[idx_C_pad]
    C = (w[None, :] * h_vals_C * valid_C.astype(float)).sum(axis=1)
    grad = dx * (A - B - C)
    return obj, grad

def _obj_func(h):
    obj, _ = _compute_obj_grad(h, _GLOBAL_T)
    s = np.sum(h) - len(h) / 2.0
    return obj + _GLOBAL_MU * s * s

def _grad_func(h):
    _, g = _compute_obj_grad(h, _GLOBAL_T)
    s = np.sum(h) - len(h) / 2.0
    return g + 2.0 * _GLOBAL_MU * s

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def random_binary_h(n, rng):
    h = np.zeros(n)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1.0
    return h

def periodic_h(n, rng, period):
    h = np.zeros(n)
    for i in range(n):
        if i % (2 * period) < period:
            h[i] = 1.0
    return make_feasible(h, n)
```
