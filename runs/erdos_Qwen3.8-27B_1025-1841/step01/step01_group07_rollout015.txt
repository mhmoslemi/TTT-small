import numpy as np
from scipy.optimize import minimize
import time

def project_to_box_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Binary search for lambda
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        lam = (target_sum - s) / len(h)
        h_new = np.clip(h + lam, 0, 1)
        if np.allclose(h, h_new):
            break
        h = h_new
    # Fine-tune
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute among free variables
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    n = len(h)
    dx = 2.0 / n

        n1 = j - lag
        if 0 <= n1 < n:
            g += (1.0 - h[n1])
        if 0 <= j + lag < n:
            g -= h[j + lag]
        grad[j] = g

import numpy as np
from scipy.optimize import minimize
import time

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        lo, hi = -1.0, 1.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            if np.clip(h + mid, 0, 1).sum() < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + 0.5 * (lo + hi), 0, 1)
    d = target - h.sum()
    if abs(d) > 1e-10:
        f = (h > 1e-12) & (h < 1.0 - 1e-12)
        if f.any():
            h[f] += d / f.sum()
            h = np.clip(h, 0, 1)
    return h

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _soft_obj_grad(h, beta=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = c.max()
    e = np.exp(beta * (c - m))
    s = e.sum()
    w = e / s
    obj = (m + np.log(s)) / beta
    grad = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for k in active:
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] -= wk * h[lag:n]
        else:
            l = -lag
            grad[0:n - l] += wk * (1.0 - h[l:n])
            grad[l:n] -= wk * h[0:n - l]
    grad *= dx
    return obj, grad

def _optimize_start(h0, target, n, deadline, seed_rng):
    h = h0.copy()
    best_local = _c5(h)
    h_best = h.copy()
    beta = 60.0

    def fun(x):
        o, _ = _soft_obj_grad(x, beta)
        return o + 80.0 * (x.sum() - target) ** 2

    def fun_g(x):
        _, g = _soft_obj_grad(x, beta)
        g += 160.0 * (x.sum() - target)
        return g

    rem = max(1.0, deadline - time.time())
    res = minimize(fun, h, jac=fun_g, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 4000, 'maxfun': 80000,
                            'ftol': 1e-16, 'gtol': 1e-9,
                            'max_time': rem})
    h_opt = _project(res.x, target)
    c5_opt = _c5(h_opt)
    if c5_opt < best_local:
        best_local = c5_opt
        h_best = h_opt.copy()

    # SLSQP polish
    rem2 = max(1.0, deadline - time.time())
    if n <= 256 and rem2 > 5:
        beta2 = 200.0

        def fun2(x):
            o, _ = _soft_obj_grad(x, beta2)
            return o + 500.0 * (x.sum() - target) ** 2

        def fun2_g(x):
            _, g = _soft_obj_grad(x, beta2)
            g += 1000.0 * (x.sum() - target)
            return g

        try:
            res2 = minimize(fun2, h_best, jac=fun2_g, method='SLSQP',
                            bounds=[(0.0, 1.0)] * n,
                            options={'maxiter': 300, 'ftol': 1e-14,
                                     'max_time': rem2})
            h_s = _project(res2.x, target)
            c5_s = _c5(h_s)
            if c5_s < best_local:
                best_local = c5_s
                h_best = h_s.copy()
        except Exception:
            pass

    # Projected gradient refinement
    rem3 = max(1.0, deadline - time.time())
    step = 0.01
    for it in range(3000):
        if time.time() > deadline:
            break
        _, g = _soft_obj_grad(h_best, 150.0)
        h_try = _project(h_best - step * g, target)
        c5_try = _c5(h_try)
        if c5_try < best_local:
            h_best = h_try
            best_local = c5_try
            if step < 0.05:
                step *= 1.05
        else:
            step *= 0.7
            if step < 1e-6:
                break

    return h_best, best_local

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        hp = np.array(initial_h_values, dtype=float).copy()
    else:
        hp = np.zeros(47)
        hp[:23] = 1.0
        hp[23] = 0.5

    best_h = hp.copy()
    best_c5 = _c5(hp)

    def remain():
        return max(0.0, budget_s - (time.time() - t0))

    # Resolutions to try
    n_list = [len(hp), 94, 128, 256, 512, 1024]
    # Deduplicate and sort
    n_list = sorted(set(n_list))

    total_sternels = sum(4 for _ in n_list)
    per_n_budget = budget_s * 0.85 / len(n_list)

    for n in n_list:
        if remain() < 15:
            break
        target = n / 2.0
        alloc = min(per_n_budget, remain() * 0.9)
        per_start = max(2.0, alloc / 4)
        deadline_n = time.time() + alloc

        # Generate starts
        starts = []
        # 1) Interp from parent
        xp = np.linspace(0, 2, len(hp), endpoint=False)
        xf = np.linspace(0, 2, n, endpoint=False)
        starts.append(_project(np.interp(xf, xp, hp), target))
        # 2) Random
        starts.append(_project(rng.random(n), target))
        # 3) Blocky random
        nb = rng.integers(3, 8)
        block_size = n // nb
        blocks = rng.random(nb)
        h_blk = np.repeat(blocks, block_size)[:n]
        starts.append(_project(h_blk, target))
        # 4) Symmetric
        h_sym = rng.random(n // 2 + 1)
        h_full = np.concatenate([h_sym, h_sym[:n // 2][::-1]])[:n]
        starts.append(_project(h_full, target))

        for si, h0 in enumerate(starts):
            if time.time() > deadline_n or remain() < 10:
                break
            dl = min(deadline_n, time.time() + per_start)
            h_opt, c5_opt = _optimize_start(h0, target, n, dl, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Final: try to improve best_h at its own resolution with more time
    n_best = len(best_h)
    if remain() > 30:
        target = n_best / 2.0
        dl = time.time() + remain() * 0.9
        h0 = best_h + rng.normal(0, 0.02, n_best)
        h0 = _project(h0, target)
        h_opt, c5_opt = _optimize_start(h0, target, n_best, dl, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final exact evaluation
    final_c5 = _c5(best_h)
    return best_h.astype(float), final_c5, len(best_h)
