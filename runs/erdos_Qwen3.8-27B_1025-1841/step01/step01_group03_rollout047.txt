    """Project h onto the feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = n / 2.0 - np.sum(h)

        mask = h < 1.0
        if np.sum(mask) > 0:

        if np.sum(mask) > 0:

import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
        if np.any(mask):
            frac = min(1.0, diff / max(1e-10, np.sum(1.0 - h[mask])))
            h[mask] += frac * (1.0 - h[mask])
    else:
        mask = h > 0.0
        if np.any(mask):
            frac = min(1.0, -diff / max(1e-10, np.sum(h[mask])))
            h[mask] -= frac * h[mask]
    return np.clip(h, 0.0, 1.0)

import time
from scipy.optimize import minimize

_target_sum = 0.0

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _objective(h):
    return evaluate_c5(h)

def _eq_fun(h):
    return np.sum(h) - _target_sum

def _eq_jac(h):
    return np.ones(len(h))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 1e-12:
        mask = h < 1.0
        cap = np.sum(1.0 - h[mask]) if np.any(mask) else 1.0
        h[mask] = np.clip(h[mask] + diff / cap, 0, 1)
    elif diff < -1e-12:
        mask = h > 0.0
        cap = np.sum(h[mask]) if np.any(mask) else 1.0
        h[mask] = np.clip(h[mask] - (-diff) / cap, 0, 1)
    return np.clip(h, 0.0, 1.0)

        return h_src.copy()
    x_old = np.arange(n_src) * (2.0 / n_src)

    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
