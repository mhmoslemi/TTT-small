import numpy as np
import time

def project_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def corr_fft(h):
    n = len(h)
    M = 1
    while M < 2 * n - 1:
        M <<= 1
    H = np.fft.rfft(h, M)
    O = np.fft.rfft(1.0 - h, M)
    c = np.fft.irfft(H * np.conj(O), M)
    return np.concatenate([c[M-n+1:], c[:n]])

def c5_val(h):
    n = len(h)
    return float(corr_fft(h).max() * (2.0 / n))

def interp_to(h, target_n):
