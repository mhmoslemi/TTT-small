    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    sum_exp = np.sum(exp_scaled)
    f = (m + np.log(sum_exp)) / T * dx
    w = exp_scaled * dx / sum_exp
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n-1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] -= wk * h[s:]
        else:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:n-t]

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

    def f(x):
        val, _ = softmax_obj_and_grad(x, T)
        return val
    
    def grad(x):
        _, g = softmax_obj_and_grad(x, T)
        return g
    
    def c_eq(x):
        return np.sum(x) - target
    
    def c_eq_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(f, h0, method='SLSQP', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                       options={'ftol': 1e-12, 'maxiter': maxiter, 'disp': False})

        pass
    return project(h0, n)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    sum_exp = np.sum(exp_scaled)
    f = (m + np.log(sum_exp)) / T * dx
    w = exp_scaled * dx / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        else:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:n - t]
    return f, grad

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def slsqp_opt(h0, n, T, maxiter=200, deadline=None):
    target = n / 2.0
    def f(x):
        val, _ = softmax_obj_and_grad(x, T)
        return val
    def grad(x):
        _, g = softmax_obj_and_grad(x, T)
        return g
    def c_eq(x):
        return np.sum(x) - target
    def c_eq_jac(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, method='SLSQP', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                       options={'ftol': 1e-13, 'maxiter': maxiter, 'disp': False})
        if deadline is not None and time.time() > deadline:
            return project(h0, n)
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return project(h0, n)

def lbfgs_pen(h0, n, T, mu, maxiter=500, deadline=None):
    target = n / 2.0
    def fp_val(x):
        val, g = softmax_obj_and_grad(x, T)
        s = np.sum(x) - target
        return val + mu * s * s
    def fp_grad(x):
        val, g = softmax_obj_and_grad(x, T)
        s = np.sum(x) - target
        return g + 2.0 * mu * s * np.ones(n)
    try:
        res = minimize(fp_val, h0, jac=fp_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        if deadline is not None and time.time() > deadline:
            return project(h0, n)
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return project(h0, n)

def local_search(h, n, deadline, n_rounds=5):
    h = project(h, n)
    best_c5 = evaluate_c5(h)
    rng = np.random.RandomState(12345)
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        idx_order = rng.permutation(n)
        improved = False
        for i in idx_order:
            if time.time() > deadline:
                break
            old = h[i]
            for trial in [0.0, 1.0, 0.5, old + 0.05, old - 0.05, old + 0.1, old - 0.1]:
                if trial < 0.0 or trial > 1.0:
                    continue
                h[i] = trial
                h = project(h, n)
                c5_new = evaluate_c5(h)
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    improved = True
                    break
            else:
                h[i] = old
                h = project(h, n)
        if not improved:
            break
    return h, best_c5

def random_feasible(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = rng.rand(n)
    h = project(h, n)
    return h

def structured_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    n_blocks = max(4, n // 42)
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        if b % 3 == 0:
            h[start:end] = 0.9 + 0.1 * rng.rand(end - start)
        elif b % 3 == 1:
            h[start:end] = 0.1 * rng.rand(end - start)
        else:
            h[start:end] = 0.3 + 0.4 * rng.rand(end - start)
    return project(h, n)

def run(seed=42, budget_s=1300, **kwargs):

    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n0 = len(h_start)

    resolutions = sorted(set([n0, 336, 512, 768]))
    resolutions = [r for r in resolutions if r >= n0]
    if not resolutions:

    T_sweep = [40.0, 100.0, 250.0, 600.0]

    for n in resolutions:
        if time.time() > deadline - 60:
            break
        if n == len(best_h):
            h_work = best_h.copy()
        else:

            h_work = project(h_work, n)

        for T in T_sweep:
            if time.time() > deadline - 40:

            maxit = 200 if n <= 512 else 120
            h_new = slsqp_opt(h_work, n, T, maxiter=maxit, deadline=deadline)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-9:

            h_work = h_new

                    if time.time() > deadline - 40:
                        break
                    h_lb = lbfgs_pen(h_work, n, T, mu, maxiter=400, deadline=deadline)
                    c5_lb = evaluate_c5(h_lb)
                    if c5_lb < best_c5 - 1e-9:

                        best_h = h_lb.copy()
                    h_work = h_lb

    # Local search polishing

        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

            if time.time() > deadline - 15:
