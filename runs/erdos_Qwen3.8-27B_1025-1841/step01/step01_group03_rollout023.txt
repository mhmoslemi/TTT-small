    n = len(h)
    c = h.copy()
    c5 = c5_value(c)
    for _ in range(n_pairs):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        t_lo = max(0.0, s - 1.0)
        t_hi = min(1.0, s)

            c[j] = s - t

        c[j] = s - best_t

def project_onto_set(h, target_sum):
