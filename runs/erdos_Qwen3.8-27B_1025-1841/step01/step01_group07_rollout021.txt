    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    # Map old positions to new
    # For each new point, find which old point it corresponds to

import numpy as np
import time

def compute_c5_fast(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def resample_nearest(h, n_new):
    n_old = len(h)
    result = np.zeros(n_new)
    for i in range(n_new):
        t = (i + 0.5) * n_old / n_new
        j = min(int(t), n_old - 1)
        result[i] = h[j]
    return result

def to_binary_even(n, h_init):
    h = np.zeros(n)
    vals = h_init.copy().astype(float)
    vals += np.random.randn(n) * 1e-9
    top = np.argsort(vals)[::-1][:n // 2]
    h[top] = 1.0
    return h

def to_binary_odd(n, h_init):
    h = np.zeros(n)
    vals = h_init.copy().astype(float)
    vals += np.random.randn(n) * 1e-9
    top = np.argsort(vals)[::-1]
    h[top[:n // 2]] = 1.0
    h[top[n // 2]] = 0.5
    return h

def sa_search(h, n, max_iters, T0=0.004, T1=0.00005, seed=42, time_limit=30):
    rng = np.random.RandomState(seed)
    h = h.copy()
    current_c5 = compute_c5_fast(h)
    best_h = h.copy()
    best_c5 = current_c5
    start_t = time.time()
    odd = (n % 2 != 0)

    if odd:
        half_pos = np.where(h == 0.5)[0]
        half_idx = half_pos[0] if len(half_pos) > 0 else -1

    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        T = T0 * (T1 / T0) ** (it / max_iters)

        if not odd:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        else:
            r = rng.random()
            if r < 0.8:
                i = rng.randint(n)
                j = rng.randint(n)
                if h[i] == h[j]:
                    continue
                if h[i] == 0.5 or h[j] == 0.5:
                    h[i], h[j] = h[j], h[i]
                else:
                    h[i], h[j] = h[j], h[i]
            else:
                if half_idx >= 0:
                    j = rng.randint(n)
                    if h[j] == 0.5 or j == half_idx:
                        continue
                    old_h = h[half_idx]
                    old_j = h[j]
                    h[half_idx] = old_j
                    h[j] = old_h
                    half_idx = j

        new_c5 = compute_c5_fast(h)
        if new_c5 <= current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if not odd:
                h[i], h[j] = h[j], h[i]
            else:
                if r < 0.8:
                    h[i], h[j] = h[j], h[i]
                else:
                    if half_idx >= 0:
                        h[j], h[half_idx] = h[half_idx], h[j]
                        # restore
                        h[half_idx], h[j] = h[j], h[half_idx]

    return best_h, best_c5

def hill_climb(h, n, time_limit=20):
    h = h.copy()
    current_c5 = compute_c5_fast(h)
    best_h = h.copy()
    best_c5 = current_c5
    start_t = time.time()

    if n % 2 != 0:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    else:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

    improved = True
    while improved and time.time() - start_t < time_limit:
        improved = False
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in ones.tolist():
            if time.time() - start_t > time_limit:
                break
            for j in zeros.tolist():
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5_fast(h)
                if new_c5 < current_c5 - 1e-15:
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                else:
                    h[i] = 1.0
                    h[j] = 0.0

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_c5 = np.inf
    best_n = None

    init_h = None
    try:
        if initial_h_values is not None:
            init_h = np.array(initial_h_values, dtype=float)
            best_h = init_h.copy()
            best_c5 = compute_c5_fast(init_h)
            best_n = len(init_h)
    except Exception:
        pass

    if init_h is None:
        n0 = 48
        h0 = np.zeros(n0)
        h0[::2] = 1.0
        init_h = h0
        best_h = h0.copy()
        best_c5 = compute_c5_fast(h0)
        best_n = n0

    n_candidates = [46, 47, 48, 49, 50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72, 74, 76, 80, 84]

    for n in n_candidates:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 30
        if remaining < 25:
            break

        n_left = len([x for x in n_candidates if x >= n])
        time_per_n = max(15.0, min(remaining / max(n_left, 1), 100.0))

        h_resampled = resample_nearest(init_h, n)

        # Multiple initial solutions
        inits = []
        inits.append(to_binary_even(n, h_resampled) if n % 2 == 0 else to_binary_odd(n, h_resampled))

        n_random = max(1, int(time_per_n / 35))
        for _ in range(n_random):
            h_r = np.zeros(n)
            if n % 2 == 0:
                pos = np.random.choice(n, n // 2, replace=False)
                h_r[pos] = 1.0
            else:
                pos1 = np.random.choice(n, n // 2, replace=False)
                h_r[pos1] = 1.0
                remaining_pos = np.setdiff1d(np.arange(n), pos1)
                half_pos = np.random.choice(remaining_pos, 1)
                h_r[half_pos] = 0.5
            inits.append(h_r)

        time_per_init = time_per_n / len(inits)
        max_iters = max(5000, int(time_per_init * 5000))

        for h_i in inits:
            if time.time() - start_time > budget_s - 30:
                break
            h_opt, c5_opt = sa_search(h_i, n, max_iters=max_iters, T0=0.004, T1=0.00005,
                                       seed=seed + np.random.randint(10000), time_limit=time_per_init)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n

    # Final hill climb polish
    if time.time() - start_time < budget_s - 10 and best_h is not None:
        hc_time = min(60, budget_s - (time.time() - start_time) - 5)
        if hc_time > 5:
            h_polished, c5_polished = hill_climb(best_h, best_n, time_limit=hc_time)
            if c5_polished < best_c5:
                best_c5 = c5_polished
                best_h = h_polished

    return best_h, best_c5, best_n
