    """Smooth approximation of max"""
    if temp < 1e-10:
        return np.max(corr)
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

        return w
    m = np.max(corr)
    w = np.exp((corr - m) / temp)

    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = n2

    # Phase 3: Nelder-Mead local refinement at n=280
    t_phase3_end = t0 + budget_s * 0.78
    h3, c5_3 = nm_refine(best_h.copy(), best_n, 2.0 / best_n, rng, 6, t_phase3_end)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3
        best_n = best_n

    # Phase 4: Subgradient polish (sharp, temp=0)
    t_phase4_end = t0 + budget_s * 0.88
    h4, c5_4 = subgrad_polish(best_h.copy(), best_n, 2.0 / best_n, 30000, t_phase4_end)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4
        best_n = best_n

    # Phase 5: Try a second upsample to n=420 if time permits
    if time.time() < t0 + budget_s * 0.86:
        n3 = 420
        h3b = upsample(best_h, n3)
        dx3b = 2.0 / n3
        c5_3b = compute_c5(h3b, dx3b)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()
            best_n = n3
        t_phase5_end = t0 + budget_s * 0.95
        h3b, c5_3b = smooth_optimize(h3b, n3, dx3b, 8000, t_phase5_end,
                                     temp_init=0.01, alpha_init=3.0 / n3,
                                     temp_decay=0.92, alpha_decay=0.94)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()
            best_n = n3

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_target(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
