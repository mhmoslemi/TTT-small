```python
import numpy as np
from scipy.optimize import minimize, l_bfgs_b
import time

def compute_corr_dx(h):
    """Compute correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def soft_max_and_p(corr_dx, T):
    """Compute soft-max value and probability weights"""
    m = np.max(corr_dx)
    exps = np.exp(T * (corr_dx - m))
    s = np.sum(exps)
    val = (m + np.log(s) / T)
    p = exps / s
    return val, p

def objective_and_grad(x, n, T, penalty):
    """Objective and gradient with penalty for sum constraint"""
    dx = 2.0 / n
    h = x  # assuming 0 ≤ h ≤ 1 already (L-BFGS-B handles bounds)
    corr_dx = np.correlate(h, 1.0 - h, 'full') * dx
    
    val_sm, p = soft_max_and_p(corr_dx, T)
    
    # Gradient of soft-max
    # p has length 2n-1, p[k] is weight for corr index k
    # shift j = n-1-k, so p_shift[j] = p[n-1-j]
    
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    omh = 1.0 - h
    
    for j in range(-(n-1), n):
        w = p[n-1-j]
        if w < 1e-18:
            continue
        if j >= 0:
            term1[:n-j] += w * omh[j:]
            term2[j:] += w * h[:n-j]
        else:
            jj = -j
            term1[jj:] += w * omh[:n-jj]
            term2[:n-jj] += w * h[jj:]
    
    grad_sm = dx * (term1 - term2)
    
    # Penalty gradient
    s = np.sum(h)
    grad_pen = 2.0 * penalty * (s - n/2.0)
    
    val = val_sm + penalty * (s - n/2.0)**2
    grad = grad_sm + grad_pen
    
    return val, grad

def project(h, n):
    """Project onto {h: 0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(corr) * dx)

def optimize_at_scale(h0, n, T_list, deadline, max_iter_per_T=500):
    """Run L-BFGS-B at given scale with multiple T values"""
    h = h0.copy().astype(float)
    h = project(h, n)
    
    for T in T_list:
        if time.time() > deadline - 30:
            break
        
        def fun(x):
            val, _ = objective_and_grad(x, n, T, 1e6)
            return val
        
        def gfun(x):
            _, g = objective_and_grad(x, n, T, 1e6)
            return g
        
        try:
            res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_iter_per_T, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxls': 20})
            if time.time() > deadline:
                break
            h_new = project(res.x, n)
            h = h_new
        except Exception:
            # Fallback: projected gradient
            h = projected_gradient(h, n, T, deadline, max_steps=200)
    
    return h

def projected_gradient(h, n, T, deadline, max_steps=200):
    """Simple projected gradient as fallback"""
    h = h.copy()
    lr = 0.5 / n
    for step in range(max_steps):
        if time.time() > deadline:
            break
        _, grad = objective_and_grad(h, n, T, 1e6)
        h_new = h - lr * grad
        h_new = project(h_new, n)
        # Simple line search
        val_new = objective_and_grad(h_new, n, T, 1e6)[0]
        val_old = objective_and_grad(h, n, T, 1e6)[0]
        if val_new <= val_old:
            h = h_new
        else:
            lr *= 0.5
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
```

```python
def slsqp_optimize(h0, n, T, deadline, maxiter=300):
    """SLSQP optimization with analytical gradient"""
    target = n / 2.0
    dx = 2.0 / n
    
    def fun(x):
        corr_dx = np.correlate(x, 1.0-x, 'full') * dx
        m = np.max(corr_dx)
        return float(m + np.log(np.sum(np.exp(T*(corr_dx - m)))) / T)
    
    def gfun(x):
        corr_dx = np.correlate(x, 1.0-x, 'full') * dx
        m = np.max(corr_dx)
        exps = np.exp(T * (corr_dx - m))
        p = exps / exps.sum()
        
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        omh = 1.0 - x
        for j in range(-(n-1), n):
            w = p[n-1-j]
            if w < 1e-18:
                continue
            if j >= 0:
                term1[:n-j] += w * omh[j:]
                term2[j:] += w * x[:n-j]
            else:
                jj = -j
                term1[jj:] += w * omh[:n-jj]
                term2[:n-jj] += w * x[jj:]
        return dx * (term1 - term2)
    
    def ceq(x):
        return np.sum(x) - target
    
    try:
        res = minimize(fun, h0, jac=gfun, method='SLSQP',
                      bounds=[(0.0, 1.0)]*n,
                      constraints=[{'type': 'eq', 'fun': ceq}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0
```

```python
    _, p = _soft_max_val_p(corr_dx, T)
    t1 = np.zeros(n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def _soft_max_val_p(corr_dx, T):
    m = np.max(corr_dx)
    e = np.exp(T * (corr_dx - m))
    s = e.sum()
    return m + np.log(s) / T, e / s

def _grad_soft_max(h, T):
    n = len(h)
    dx = 2.0 / n
    corr_dx = _corr_dx(h)
    _, p = _soft_max_val_p(corr_dx, T)
    t1 = np.zeros(n)
    t2 = np.zeros(n)
    omh = 1.0 - h
    for j in range(-(n - 1), n):
        w = p[n - 1 - j]
        if w < 1e-18:
            continue
        if j >= 0:
            t1[:n - j] += w * omh[j:]
            t2[j:] += w * h[:n - j]
        else:
            jj = -j
            t1[jj:] += w * omh[:n - jj]
            t2[:n - jj] += w * h[jj:]
    return dx * (t1 - t2)

def _project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = h < 1.0 if d > 0 else h > 0.0
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _interp(h, nf, nt):
    if nf == nt:
        return h.copy()
    xf = np.linspace(0, 2, nf, endpoint=False)
    xt = np.linspace(0, 2, nt, endpoint=False)
    return np.interp(xt, xf, h)

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _grad_exact(h, j):
    dx = 2.0 / len(h)
    n = len(h)
    g = np.zeros(n)
    if j >= 0:
        g[:n - j] += (1.0 - h[j:])
        g[j:] -= h[:n - j]
    else:
        jj = -j
        g[jj:] += (1.0 - h[:n - jj])
        g[:n - jj] -= h[jj:]
    return g * dx

def _slsqp(h0, n, T, deadline, maxit):
    target = n * 0.5
    def fun(x):
        v, _ = _soft_max_val_p(_corr_dx(x), T)
        return v
    def gfun(x):
        return _grad_soft_max(x, T)
    def ceq(x):
        return x.sum() - target
    try:
        r = minimize(fun, h0, jac=gfun, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': ceq}],
                     options={'maxiter': maxit, 'ftol': 1e-15})
        if time.time() > deadline:
            return h0
        if r.x is not None and np.all(np.isfinite(r.x)):
            return _project(r.x, n)
    except Exception:
        pass
    return h0

def _lgbfgs(h0, n, T, deadline, maxit):
    pen = 1e6
    target = n * 0.5
    def fun_g(x):
        cd = _corr_dx(x)
        v, _ = _soft_max_val_p(cd, T)
        s = x.sum() - target
        val = v + pen * s * s
        g = _grad_soft_max(x, T) + 2.0 * pen * s
        return val, g
    def fun(x):
        return fun_g(x)[0]
    def gfun(x):
        return fun_g(x)[1]
    try:
        r = minimize(fun, h0, jac=gfun, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 30})
        if time.time() > deadline:
            return h0
        if r.x is not None and np.all(np.isfinite(r.x)):
            return _project(r.x, n)
    except Exception:
        pass
    return h0

def _polish_max(h, n, deadline, steps=200):
    h = h.copy()
    lr = 1.0 / (n * 0.3)
    for _ in range(steps):
        if time.time() > deadline:
            break
        cd = _corr_dx(h)
        kmax = int(np.argmax(cd))
        j = n - 1 - kmax
        grad = _grad_exact(h, j)
        h_try = _project(h - lr * grad, n)
        cd_try = _corr_dx(h_try)
        if cd_try.max() <= cd.max() + 1e-14:
            h = h_try
            lr = min(lr * 1.1, 2.0 / n)
        else:
            lr *= 0.6
        if lr < 1e-10:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Phase 4: Try higher resolution

    # Phase 5: Random restarts at best resolution
```
