```python
    dx = 2.0 / n_points
    n = n_points
    best_c5 = compute_c5(h, n_points)
    best_h = h.copy()
    
    for it in range(max_iters):
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        c5 = corr[k_max] * dx
        
        # Compute gradient of the active constraint
        offset = k_max - (n - 1)
        grad = np.zeros(n)
        for i in range(n):

            if 0 <= j < n:
                grad[i] += (1 - h[j])
                grad[j] -= h[i]
        grad *= dx
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project to constraint: 0 <= h <= 1, sum(h) = n/2
        h_new = np.clip(h_new, 0, 1)
        # Adjust to satisfy sum constraint
        diff = n/2 - h_new.sum()
        if abs(diff) > 1e-10:
            # Add/subtract uniformly from elements that can accommodate
            if diff > 0:
                # Need to increase sum; add to elements < 1
                mask = h_new < 1
                if mask.sum() > 0:
                    h_new[mask] += diff / mask.sum()
                    h_new = np.clip(h_new, 0, 1)
            else:

                mask = h_new > 0
                if mask.sum() > 0:
                    h_new[mask] += diff / mask.sum()
                    h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        c5 = compute_c5(h, n_points)
        if c5 < best_c5:

            best_h = h.copy()
```

```python
            grad[valid_i] += weights[idx] * (1 - h[valid_j])
            grad[valid_j] -= weights[idx] * h[valid_i]
        else:
            valid_i = slice(-offset, n)
            valid_j = slice(0, n + offset)
            grad[valid_i] += weights[idx] * (1 - h[valid_j])
            grad[valid_j] -= weights[idx] * h[valid_i]

    # So the gradient of f = (1/tau)*logsumexp * dx is dx * sum(w * grad_corr)
    return grad * dx, corr.max() * dx
```

This is linear in h[j]! (The coefficients depend on other h values but not on h[j] itself.)

So for coordinate descent, at each step:
1. Find the active constraint k* (the one with max f_k)
2. The coefficient of h[j*] in f_{k*} is: c = 1 - h[j*+s_{k*}] - h[j*-s_{k*}] (with appropriate boundary handling)
3. If c > 0, decreasing h[j*] decreases f_{k*} → set h[j*] as low as possible (0)
4. If c < 0, increasing h[j*] decreases f_{k*} → set h[j*] as high as possible (1)
5. If c = 0, h[j*] doesn't affect f_{k*}

But we also need to maintain the constraint sum(h) = n/2! So we can't just set h[j*] to 0 or 1.

Hmm, this complicates things. We'd need to adjust another coordinate to maintain the sum.

Actually, for coordinate descent with the sum constraint, we could:
1. Pick coordinate j to update
2. Determine the best value of h[j] (which might be 0 or 1 or somewhere in between)
3. Adjust another coordinate to maintain the sum

This is getting complicated. Let me stick with the projected gradient approach but make it more robust.

Alternative idea: use scipy.optimize.minimize with the 'L-BFGS-B' method, treating the sum constraint as a penalty. The penalty approach:

minimize C₅(h) + λ * (sum(h) - n/2)²

subject to 0 ≤ h[i] ≤ 1

For large enough λ, the constraint is approximately satisfied. We can increase λ over time.

Let me try this:

The issue with L-BFGS-B is that it needs gradients, and computing the gradient of C₅ (the max of many functions) is non-smooth. We'd need to use the smooth max approximation for the gradient.

Actually, L-BFGS-B can work without explicit gradients (it uses finite differences), but that's O(n) function evaluations per gradient, each O(n) for the correlation. So O(n²) per gradient step, which is too slow for n=320.

Let me provide an explicit gradient using the smooth max:

```python
def c5_smooth_objective(h, n, tau):
    """Smooth approximation of C5 for use with gradient-based methods."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    shifted = corr * tau
    max_shifted = shifted.max()
    logsum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return (logsum / tau) * dx

def c5_smooth_gradient(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    shifted = corr * tau
    max_shifted = shifted.max()
    weights = np.exp(shifted - max_shifted)
    weights /= weights.sum()
    
    significant = np.where(weights > 1e-4)[0]
    grad = np.zeros(n)
    for idx in significant:
        offset = idx - (n - 1)
        if offset >= 0:
            vi = slice(0, n - offset)
            vj = slice(offset, n)
            w = weights[idx]
            grad[vi] += w * (1.0 - h[vj])
            grad[vj] -= w * h[vi]
        else:
            vi = slice(-offset, n)
            vj = slice(0, n + offset)
            w = weights[idx]
            grad[vi] += w * (1.0 - h[vj])
            grad[vj] -= w * h[vi]
    return grad * dx
```

Now, for the full objective with penalty:
∇[C₅_smooth + λ(sum(h) - n/2)²] = ∇C₅_smooth + 2λ(sum(h) - n/2) * 1

Let me put this all together into a clean solution.

Actually, I think the most effective approach given the time budget is:
1. Start from the parent's solution
2. Use projected gradient (which handles the constraint exactly via projection)
3. Use adaptive step size and smooth max
4. Try a few random restarts
5. Upsample if time allows

Let me write the final clean version:

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, n):
    """Project onto {0<=x<=1, sum(x)=n/2}"""
    target = 0.5 * n
    lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def smooth_grad(h, n, tau):
    """Gradient of smooth-max C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    s = corr * tau
    s_max = s.max()
    w = np.exp(s - s_max)
    w /= w.sum()
    mask = w > 1e-4
    grad = np.zeros(n)
    for idx in np.where(mask)[0]:
        off = idx - (n - 1)
        if off >= 0:
            i_sl = slice(0, n - off)
            j_sl = slice(off, n)
            grad[i_sl] += w[idx] * (1.0 - h[j_sl])
            grad[j_sl] -= w[idx] * h[i_sl]
        else:
            i_sl = slice(-off, n)
            j_sl = slice(0, n + off)
            grad[i_sl] += w[idx] * (1.0 - h[j_sl])
            grad[j_sl] -= w[idx] * h[i_sl]
    return grad * dx

def pg_optimize(h0, n, t_budget):
    """Projected gradient optimization with smooth max."""
    t0 = time.time()
    h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.5
    tau = 10.0
    stall = 0
    
    while time.time() - t0 < t_budget:
        g = smooth_grad(h, n, tau)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new, n)
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h_new.copy()
            stall = 0
            tau = min(tau * 1.01, 80.0)
            lr = min(lr * 1.01, 2.0)
        else:
            stall += 1
            lr *= 0.85
            if stall > 300:
                break
        h = h_new
    
    return best_h, best_c5

def upsample(h, no, nn):
    xo = np.linspace(0, 2, no, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    xe = np.concatenate([xo - 2, xo, xo + 2])
    he = np.concatenate([h, h, h])
    return np.interp(xn, xe, he)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    # Get parent solution
    try:
        hp = np.array(initial_h_values, dtype=float)
        np_ = len(hp)
    except NameError:
        np_ = 80
        hp = np.full(np_, 0.5)
    
    hp = project_feasible(hp, np_)
    best_h = hp.copy()
    best_c5 = compute_c5(best_h, np_)
    
    # Phase 1: optimize at parent resolution
    phase1_time = min(budget_s * 0.25, 200)
    h1, c1 = pg_optimize(hp, np_, phase1_time)
    if c1 < best_c5:
        best_c5, best_h = c1, h1
    
    # Phase 2: upsample and optimize at 2x
    remaining = budget_s - (time.time() - t_start) - 30
    if remaining > 60:
        n2 = np_ * 2
        h2_up = upsample(h1, np_, n2)
        h2_up = project_feasible(h2_up, n2)
        h2, c2 = pg_optimize(h2_up, n2, remaining * 0.5)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
    
    # Phase 3: upsample and optimize at 4x
    remaining = budget_s - (time.time() - t_start) - 30
    if remaining > 60:
        n3 = np_ * 4
        h3_up = upsample(h2, n2, n3)
        h3_up = project_feasible(h3_up, n3)
        h3, c3 = pg_optimize(h3_up, n3, remaining)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
    
    # Phase 4: random restarts at best resolution
    remaining = budget_s - (time.time() - t_start) - 10
    if remaining > 30:
        n_best = len(best_h)
        n_restarts = max(1, int(remaining / 30))
        for r in range(n_restarts):
            if time.time() - t_start > budget_s - 10:
                break
            t_r = (time.time() - t_start)
            rem_r = budget_s - t_r - 5
            if rem_r < 10:
                break
            h_r = best_h + np.random.randn(n_best) * 0.03
            h_r = project_feasible(h_r, n_best)
            hr, cr = pg_optimize(h_r, n_best, rem_r * 0.4)
            if cr < best_c5:
                best_c5, best_h = cr, hr
    
    # Final sanitize
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
```
