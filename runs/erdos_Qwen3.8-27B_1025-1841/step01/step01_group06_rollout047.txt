def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-14):
            h = h_new
            break

    s = np.sum(h)
    rem = target - s

        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)

- Projected gradient
- Coordinate descent (since it's bilinear, each coordinate update is linear)

Plus possibly h[j-s] if j-s ≥ 0 (for the symmetric part)

Actually Q_s is the matrix where (Q_s)_{i,j} = 1 if j = i+s or i = j+s (to make it symmetric).
So (Q_s h)[j] = h[j+s] (if valid) + h[j-s] (if valid).

Max-QP coordinate update:
Maximize 2*(Q_s h)[j]*h[j] + (constant)
Since (Q_s h)[j] doesn't depend on h[j] (no diagonal in Q_s for s≠0):
If (Q_s h)[j] > 0: set h[j] = 1
If (Q_s h)[j] < 0: set h[j] = 0

Then adjust for the constraint sum(h) = n/2 by adjusting the least "urgent" coordinates.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(3000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-12:
        if rem > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if rem > 0:
                add = min(rem, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                rem -= add
            else:
                sub = min(-rem, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                rem += sub
            if abs(rem) < 1e-14:
                break
    return h

def upsample_func(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_h(h.copy(), n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new)

def smooth_obj_grad(h, n, dx, temp):
    hx = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    shifted = (scaled - m) / temp
    shifted -= np.max(shifted)
    expv = np.exp(shifted)
    Z = np.sum(expv)
    obj = m + temp * np.log(Z)
    w = expv / Z * dx
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):

            continue
        s = m_idx - (n - 1)
        wv = w[m_idx]

            lo = -s
            nz = n - lo
            if nz > 0:
                grad[lo:] += wv * (1.0 - hx[:nz])
                grad[:nz] -= wv * hx[lo:]
    return obj, grad
