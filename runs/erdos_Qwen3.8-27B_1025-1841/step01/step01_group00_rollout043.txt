def quick_project(g, n):
    target = n / 2.0
    h = np.clip(g, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Adjust: add (target - s) / (n - n_clipped) to non-clipped elements
    n_clipped = (h == 0.0).sum() + (h == 1.0).sum()
    n_free = n - n_clipped
    if n_free == 0:
        return h  # Degenerate case
    adj = (target - s) / n_free
    h = np.clip(h, 0.0, 1.0)
    # Apply adjustment only to non-clipped
    mask = (h > 0.0) & (h < 1.0)
    h[mask] += adj
    h = np.clip(h, 0.0, 1.0)
    # One more pass if needed
    s = h.sum()
    if abs(s - target) > 1e-10:
        n_clipped = (h == 0.0).sum() + (h == 1.0).sum()
        n_free = n - n_clipped
        if n_free > 0:
            adj = (target - s) / n_free
            mask = (h > 0.0) & (h < 1.0)
            h[mask] += adj
            h = np.clip(h, 0.0, 1.0)
    return h

    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))

            lo = mid
        else:

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def fft_correlate(a, v):
    n = len(a) + len(v) - 1
    Np = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, Np)
    fv = np.fft.rfft(v[::-1], Np)
    return np.fft.irfft(fa * fv, Np)[:n]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    return corr.max() * dx


def subgrad_vec(h, sp):
    n = len(h)
    grad = np.zeros(n)
    if sp == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        mid = n - 2 * sp
        grad[sp:sp + mid] = (1.0 - h[:mid]) - h[mid + sp:n]
        grad[:sp] = -h[sp:sp + sp]
        grad[n - sp:] = 1.0 - h[n - 2 * sp:n - sp]
    return grad


def optimize_subgrad(h, n, max_iters, time_limit, lr0, mu, topk):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr0
    v = np.zeros(n)
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = fft_correlate(h, 1.0 - h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if topk <= 1:
            ki = int(np.argmax(corr))
            sp = abs(ki - (n - 1))
            grad = subgrad_vec(h, sp)
        else:
            top_idx = np.argsort(corr)[-topk:][::-1]
            grad = np.zeros(n)
            for ki in top_idx:
                sp = abs(int(ki) - (n - 1))
                grad += subgrad_vec(h, sp)
            grad /= topk
        grad *= dx
        v = mu * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 80:
            lr *= 0.75
            v *= 0.4
            stall = 0
            if lr < 1e-11:
                break
    return best_h, best_c5


def optimize_smooth(h, n, tau, max_iters, time_limit, lr0, M):

    h = project_to_feasible(h, n).copy()
