import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def next_power_of_2(m):
    s = 1
    while s < m:
        s *= 2
    return s

def fft_correlation(h):
    """Compute all linear cross-correlation values of h with 1-h using FFT."""
    n = len(h)
    size = next_power_of_2(2 * n - 1)
    o = 1.0 - h
    H = np.fft.rfft(h, size)
    O = np.fft.rfft(o, size)
    corr = np.fft.irfft(H * np.conj(O), size)
    # Extract the 2n-1 linear correlation values
    # corr[0:n] gives lags 0 to n-1
    # corr[size-n+1:size] gives lags -(n-1) to -1
    # Reorder to match np.correlate(h, o, mode='full') ordering
    # np.correlate: c[0] = h[n-1]*o[0], c[n-1] = sum h[i]*o[i], c[2n-2] = h[0]*o[n-1]
    # Let me just return the full array and take max
    return corr

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlation(h)
    return corr.max() * dx

def smooth_obj_grad_fft(h, tau):
    """Compute smooth max objective and gradient using FFT."""
    n = len(h)
    dx = 2.0 / n
    o = 1.0 - h
    
    # Compute correlation
    size = next_power_of_2(2 * n - 1)
    H = np.fft.rfft(h, size)
    O = np.fft.rfft(o, size)
    corr = np.fft.irfft(H * np.conj(O), size)
    
    # Extract 2n-1 values
    c_vals = np.concatenate([corr[size-(n-1):size], corr[0:n]])
    
    # Smooth max
    max_c = c_vals.max()
    shifted = tau * (c_vals - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    # Gradient via FFT
    # A = p (length 2n-1), A_rev = p[::-1]
    A_rev = p[::-1]
    size2 = next_power_of_2(3 * n - 1)  # need at least (2n-1) + n - 1
    AR = np.fft.rfft(A_rev, size2)
    OF = np.fft.rfft(o, size2)
    HF = np.fft.rfft(h, size2)
    T1 = np.fft.irfft(AR * OF, size2)[:n]  # convolution of A_rev and o
    T2 = np.fft.irfft(AR * HF, size2)[:n]  # convolution of A_rev and h
    grad = (T1 - T2) * dx
    
    return obj, grad
