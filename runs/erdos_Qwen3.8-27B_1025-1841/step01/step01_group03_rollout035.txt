import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(h, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    s = tau * (corr - max_c)
    s = np.clip(s, -500, 500)
    return max_c + np.log(np.sum(np.exp(s))) / tau

def penalized_obj(h, tau, dx, target_sum, mu):
    base = smooth_max_obj(h, tau, dx)
    penalty = mu * (np.sum(h) - target_sum) ** 2
    return base + penalty

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def sum_constraint_jac(h):
    return np.ones(len(h))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room < 1e-15:
                break
            h += diff * room / total_room
        else:
            total = np.sum(h)
            if total < 1e-15:
                break
            h += diff * h / total
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return np.interp(x_new, x_old, h_old)

def try_solve_slsqp(h_start, n, tau, maxiter, state):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    try:
        res = minimize(smooth_max_obj, h_start, args=(tau, dx),
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_feasible(np.clip(res.x, 0, 1), n)
        c5 = compute_c5(h_opt)
        if c5 < state['best_c5']:
            state['best_c5'] = c5
            state['best_h'] = h_opt.copy()
            state['best_n'] = n
    except Exception:
        h_opt = project_feasible(h_start, n)
        c5 = compute_c5(h_opt)
        if c5 < state['best_c5']:
            state['best_c5'] = c5
            state['best_h'] = h_opt.copy()
            state['best_n'] = n

    dx = 2.0 / n
    target_sum = n / 2.0
    mu = 500.0

        res = minimize(penalized_obj, h_start, args=(tau, dx, target_sum, mu),
                     method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_opt = project_feasible(np.clip(res.x, 0, 1), n)

def make_binary_start(n, rng, pattern='random'):
    h = np.zeros(n)
    if pattern == 'random':
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
    elif pattern == 'alternating':
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
        if np.sum(h) > n // 2:
            h[rng.choice(np.where(h == 1)[0], size=np.sum(h) - n // 2, replace=False)] = 0
        elif np.sum(h) < n // 2:
            h[rng.choice(np.where(h == 0)[0], size=n // 2 - np.sum(h), replace=False)] = 1
    elif pattern == 'blocks':
        block_size = max(2, n // 10)
        for i in range(0, n, block_size * 2):
            h[i:i + block_size] = 1.0
        excess = np.sum(h) - n // 2
        if excess > 0:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, size=excess, replace=False)] = 0
        elif excess < 0:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, size=-excess, replace=False)] = 1
    return project_feasible(h, n)
