import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    for _ in range(5):
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj(h, n, t, pen_lambda=500.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    shifted = np.minimum(shifted, 0.0)
    exponents = np.clip(t * shifted, -500, 0)
    soft = mc + np.log(np.sum(np.exp(exponents))) / t
    penalty = pen_lambda * (np.sum(h) - n / 2.0) ** 2
    return soft * dx + penalty

def smooth_grad(h, n, t, pen_lambda=500.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    shifted = np.minimum(shifted, 0.0)
    exponents = np.clip(t * shifted, -500, 0)
    weights = np.exp(exponents)
    sw = weights.sum()
    weights /= sw
    grad = np.zeros(n)
    ncorr = len(corr)
    for m in range(ncorr):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            g = np.zeros(n)
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
            grad += w * g
        else:
            k2 = -k
            g = np.zeros(n)
            g[k2:] += (1.0 - h[:n - k2])
            g[:n - k2] -= h[k2:]
            grad += w * g
    grad *= dx
    sum_diff = np.sum(h) - n / 2.0
    grad += 2.0 * pen_lambda * sum_diff * np.ones(n)
    return grad

def coordinate_descent(h, n, max_iter=30, time_limit=None):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for it in range(max_iter):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            if time_limit is not None and time.time() > time_limit:
                break
            old_val = h[j]
            candidates = [0.0, 1.0]
            if old_val > 0.05:
                candidates.append(old_val - 0.05)
            if old_val < 0.95:
                candidates.append(old_val + 0.05)
            if old_val > 0.02:
                candidates.append(old_val - 0.02)
            if old_val < 0.98:
                candidates.append(old_val + 0.02)
            for c in candidates:
                if c < 0.0 or c > 1.0:
                    continue
                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                c_trial = compute_c5(trial)
                if c_trial < best_c5 - 1e-12:
                    best_c5 = c_trial
                    h = trial
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def dither_improve(h, n, rng, n_trials=50, time_limit=None):
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        if time_limit is not None and time.time() > time_limit:

        trial = h + rng.randn(n) * 0.03
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c5:
            best_c5 = c
            h = trial
    return h, best_c5
