    """Compute weighted gradient using top K lags with softmax weights."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_K = np.argsort(corr)[-K:]
    weights = corr[top_K] * T
    mw = np.max(weights)
    exps = np.exp(weights - mw)
    sw = np.sum(exps)
    probs = exps / sw
    
    g = np.zeros(n)
    for idx, p in zip(top_K, probs):
        lag = idx - (n - 1)
        g += p * gradient_at_lag(h, n, lag)
    
    return g * dx

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = d / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_at_lag(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        N = n - lag
        if N > 0:
            g[:N] += 1.0 - h[lag:lag+N]
            g[lag:lag+N] += -h[:N]
    else:
        L = -lag
        N = n - L
        if N > 0:
            g[L:] += 1.0 - h[:N]
            g[:N] += -h[L:]
    return g

def subgrad_descent(h0, n, n_steps, lr, K, T, deadline):
    h = h0.copy()
    dx = 2.0 / n
    for t in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Top K lags with softmax weighting
        top_K = np.argsort(corr)[-K:]
        w = corr[top_K] * T
        mw = np.max(w)
        exps = np.exp(w - mw)
        sw = np.sum(exps)
        probs = exps / sw
        
        g = np.zeros(n)
        for idx, p in zip(top_K, probs):
            lag = idx - (n - 1)
            g += p * gradient_at_lag(h, n, lag)
        
        h -= lr * g * dx
        h = project(h, n)
    return h

def binary_local_search(h0, n, n_passes, deadline):
    """Local search on near-binary vectors: flip pairs to reduce C5."""
    h = h0.copy()
    h = project(h, n)
    # Round to binary
    idx_sorted = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx_sorted[:n//2]] = 1.0
    c5 = evaluate_c5(h_bin)
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(h_bin == 1.0)[0])
        zeros = list(np.where(h_bin == 0.0)[0])
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c5_new = evaluate_c5(h_bin)
                if c5_new < c5 - 1e-10:
                    c5 = c5_new
                    improved = True
                    ones.remove(i)
                    zeros.remove(j)
                    ones.append(j)
                    zeros.append(i)
                    break
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            break
    return h_bin

def slsqp_opt(h0, n, T, maxiter, deadline):
    target = n / 2.0
    dx = 2.0 / n
    
    def f(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        scaled = corr * T
        m = np.max(scaled)
        return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)
    
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'ftol': 1e-12, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return project(h0.copy(), n)

def interp_to(h, n_from, n_to):

        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)
