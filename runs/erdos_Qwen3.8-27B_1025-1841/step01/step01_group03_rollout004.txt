            best_h = h4.copy()

    # Phase 5: Try n=320
    if remaining() > 150:
        n4 = 320
        target4 = n4 / 2.0
        h4 = upsample(best_h, n4)
        h4 = project_feasible(h4, target4)
        alloc = min(remaining() * 0.4, 450)
        h5, c5_5 = projected_subgrad_momentum(h4, target4, max_iters=8000,
                                              budget_s=alloc, alpha_init=0.012, momentum=0.92)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Final polish
    if remaining() > 40:
        n_f = len(best_h)
        target_f = n_f / 2.0
        alloc = remaining() * 0.85
        h6, c5_6 = projected_subgrad_momentum(best_h, target_f, max_iters=20000,
                                              budget_s=alloc, alpha_init=0.005, momentum=0.93)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
