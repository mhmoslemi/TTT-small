- Sort h, find the threshold τ such that sum(clip(h-τ, 0, 1)) = n/2
- Actually, it's: find τ such that sum(clip(h - λ, 0, 1)) = n/2 where λ is the Lagrange multiplier

The projection onto {x : 0 ≤ x ≤ 1, 1^T x = s} can be done as:
1. Find λ such that sum(clip(h - λ, 0, 1)) = s
2. Project: x = clip(h - λ, 0, 1)

To find λ, sort h and use the fact that clip(h-λ, 0, 1) = 
- 0 if h_i ≤ λ
- h_i - λ if λ < h_i < λ+1
- 1 if h_i ≥ λ+1

So sum(clip(h-λ, 0, 1)) = sum_{h_i > λ, h_i < λ+1} (h_i - λ) + count_{h_i ≥ λ+1}
= (sum of h_i in (λ, λ+1)) - λ * count_in_(λ,λ+1) + count_{h_i ≥ λ+1}

This is a piecewise linear function of λ, so we can binary search on it.

```python
def project_feasible(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target}."""
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    # Use binary search on lambda
    lo = h.min() - 1.0
    hi = h.max()
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid  # need to increase lambda to decrease sum
        else:
            hi = mid
        if abs(s - target) < 1e-15:
            break
    
    lam = (lo + hi) / 2
    x = np.clip(h - lam, 0.0, 1.0)
    # Final adjustment for numerical issues
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        x[mask] += diff / mask.sum()
        x = np.clip(x, 0.0, 1.0)
    return x
```

```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    val = m + np.log(np.sum(np.exp(scaled - m)))
    return val / T * dx

def soft_max_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue

            term[:n-s] -= x[s:]
            term[s:] = 1.0 - x[:n-s]
        grad += wk * term

    lr = lr0
    best_h = h.copy()

    for it in range(max_iter):

            break
        g = soft_max_grad(h, n, T)
        # Backtracking line search

        f0 = soft_max_obj(h, n, T)

            f_new = soft_max_obj(h_new, n, T)

        if f_new >= f0:
            lr = max(1e-6, lr * 0.8)
        else:
            lr = min(0.1, lr * 1.05)
```

```python
        #  we need to be careful. Actually if i-k==j, then the change at n=j is affected.)
        
        # So Δ(sum h[n]*h[n+k]) = -h[i+k] + h[j+k] - h[i-k] + h[j-k]
        # (with h[m]=0 if m not in [0,n-1])
        
        # Δcorr[k] = -Δ(sum h[n]*h[n+k]) = h[i+k] - h[j+k] + h[i-k] - h[j-k]
        
        # To compute the new max, we need the full new corr.
        # But we can just update the max efficiently if we track which k is the max.
        
        # For simplicity, let's just recompute corr each time (it's fast for n=336)
        h[i] = 0
        h[j] = 1
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = new_corr
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5
```

    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def slsqp_opt(h0, n, T, deadline, maxiter=300):
    target = n / 2.0
    try:
        res = minimize(soft_max_obj, h0,
                       jac=lambda x: soft_max_grad(x, n, T),
                       args=(n, T),
                       method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        if time.time() < deadline:
            if res.x is not None and np.all(np.isfinite(res.x)):
                return project_feasible(res.x, target)
    except Exception:
        pass
    return h0

def proj_grad_desc(h0, n, T, deadline, lr0=0.01, max_iter=2000):
    h = h0.copy()
    target = n / 2.0
    lr = lr0
    f0 = soft_max_obj(h, n, T)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = soft_max_grad(h, n, T)
        gnorm = np.dot(g, g)
        if gnorm < 1e-20:
            break
        alpha = lr
        accepted = False
        for bt in range(25):
            h_new = project_feasible(h - alpha * g, target)
            f_new = soft_max_obj(h_new, n, T)
            if f_new < f0 - 1e-4 * alpha * np.dot(g, h_new - h):
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            break
        h = h_new
        f0 = f_new
        if alpha >= lr:
            lr = min(0.05, lr * 1.05)
        else:
            lr = max(1e-7, alpha * 0.9)
    return h

def interp_upscale(h, n_to):
    n_from = len(h)
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)
