def corr_grad_at_shift(h, s):
    """Gradient of C(s) = correlate(h,1-h,'full')[s+(n-1)] w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        # Forward: j in [0, n-1-s]
        if n - s > 0:
            grad[:n-s] += (1.0 - h[s:s+n-s])  # (1-h[j+s])
        # Backward: j in [s, n-1]
        if n - s > 0:
            grad[s:] -= h[:n-s]  # -h[j-s]

        if n - ss > 0:

        # Backward: j in [0, n-1-ss]
        if n - ss > 0:
            grad[:n-ss] -= h[ss:]  # -h[j-s] = -h[j+ss]

def smooth_obj(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum_exp = T * mc + np.log(np.sum(np.exp(T * shifted)))
    soft = log_sum_exp / T
    penalty = mu * (np.sum(h) - n/2.0)**2
    return soft * dx + penalty

def smooth_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_shifted = np.exp(T * shifted)
    total = np.sum(exp_shifted)
    w = exp_shifted / total
    
    grad = np.zeros(n)
    # Use top shifts for efficiency
    top_k = np.argsort(w)[::-1][:min(30, len(w))]
    for k in top_k:
        if w[k] < 1e-12:
            break
        s = int(k) - (n - 1)
        grad += w[k] * corr_grad_at_shift(h, s)
    
    grad *= dx
    grad += 2.0 * mu * (np.sum(h) - n/2.0)
    return grad

For the coordinate descent, given that we fix all coordinates except h[j], the overlap at shift s becomes:
C_s = h[j](1-h[j+s]) + rest  (if j+s valid)
   + (1-h[j])h[j-s] + rest2  (contribution from the other positions)
   
Wait, more precisely:
C_s = sum_i h[i](1-h[i+s])
The terms involving h[j]:
- i=j: h[j](1-h[j+s]) [if j+s valid]
- i=j-s: h[j-s](1-h[j]) [if j-s valid and j-s ≥ 0, j valid]

So C_s as a function of h[j] (call it x for the free variable):
= x(1-h[j+s]) + h[j-s](1-x) + const
= x - x*h[j+s] + h[j-s] - x*h[j-s] + const
= x(1 - h[j+s] - h[j-s]) + h[j-s] + const

This is LINEAR in x! So the optimal x is either 0 or 1 (at the boundary), unless the coefficient is zero.

For x ∈ [0,1]:
- If (1 - h[j+s] - h[j-s]) > 0: minimize by setting x = 0
- If (1 - h[j+s] - h[j-s]) < 0: minimize by setting x = 1
- If = 0: any x works

But we're minimizing the MAX over shifts, not a single shift. So it's not simply optimal to set x to 0 or 1 for the worst shift.

However, for the specific case where the max is achieved at shift s* and we're doing a "greedy" step on the worst shift, the linear structure tells us that x should go to a boundary.

This means the coordinate descent should primarily try x=0 and x=1! The intermediate values only matter when multiple shifts are tied for maximum.

Let me refine the CD to focus on binary swaps (setting to 0 or 1), with a penalty for violating the sum constraint.

Actually, a simpler and very effective approach: since C_s is linear in each h[j] for a fixed set of other values, the optimal solution (for minimizing max) will have each h[j] at a boundary (0 or 1) in the limit. The continuous relaxation helps with the optimization landscape but the true optimum is likely binary or near-binary.

So a good strategy is:
1. Use the continuous optimization to get close to the optimum
2. Then do local search: flip individual bits (0→1 or 1→0) while maintaining the sum constraint (flip one 1→0 and one 0→1 simultaneously)
3. Evaluate the exact max after each flip

This "pair-swap" local search is O(n²) per pair (to compute the new max), but we can be smart about which pairs to try.

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def corr_grad_at_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        n_valid = n - s
        if n_valid > 0:
            grad[:n_valid] += (1.0 - h[s:s+n_valid])
            grad[s:] -= h[:n_valid]
    else:
        ss = -s
        n_valid = n - ss
        if n_valid > 0:
            grad[ss:] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[ss:]
    return grad

def _smooth_obj(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # log-sum-exp
    max_shifted = 0.0  # shifted max is 0
    log_sum = T * mc + np.log(np.sum(np.exp(T * shifted)))
    soft = log_sum / T
    penalty = mu * (np.sum(h) - n / 2.0) ** 2
    return soft * dx + penalty

def _smooth_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_shifted = np.exp(T * shifted)
    total = np.sum(exp_shifted)
    w = exp_shifted / total
    
    grad = np.zeros(n)
    top_idx = np.argsort(w)[::-1][:min(30, 2*n-1)]
    for k in top_idx:
        wk = w[k]
        if wk < 1e-14:
            break
        s = int(k) - (n - 1)
        grad += wk * corr_grad_at_shift(h, s)
    
    grad *= dx
    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    return grad

def binary_round(h, n, rng):
    """Round h to binary with exactly n/2 ones."""
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def pair_swap_local_search(h, n, rng, max_rounds=100, evals_per_round=50):
    """Local search via pairwise swaps on binary near-binary h."""
    best_c5 = compute_c5(h)
    h = h.copy()
    # Ensure exactly n/2 ones (for binary) or sum = n/2 (for continuous)
    h = make_feasible(h, n)

        improved = False
        # Get indices likely to be 0 or 1
        ones = np.where(h > 0.7)[0]
        zeros = np.where(h < 0.3)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for _ in range(evals_per_round):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            trial = h.copy()

            trial[j] = 1.0

                zeros = np.where(h < 0.3)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
        if not improved:
            break
    return h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def corr_grad_at_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        nv = n - s

            grad[:nv] += (1.0 - h[s:s + nv])
            grad[s:] -= h[:nv]
    else:
        ss = -s
        nv = n - ss
        if nv > 0:
            grad[ss:] += (1.0 - h[:nv])

    return grad
