        if abs(diff) < 1e-12:

        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9

    if len(h_old) == n_new:
        return h_old.copy()
    x_old = np.arange(len(h_old))
    x_new = np.linspace(0, len(h_old) - 1, n_new)
    return np.interp(x_new, x_old, h_old)

def coord_descent(h, n, dx, target, max_rounds=200, budget_deadline=None):

            if best_c5 < cur - 1e-10:
