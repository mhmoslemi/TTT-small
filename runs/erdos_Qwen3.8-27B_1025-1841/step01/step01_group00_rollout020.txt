def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx

    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def adam_anneal(h, n, time_limit, tau_start=30.0, tau_end=5.0, lr=0.05):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_count = 0
    stall = 0
    stall_limit = 80

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start + (tau_end - tau_start) * frac

        obj, grad = smooth_obj_grad(h, tau)
        t_count += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)

        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            break

        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)

    return best_h, best_c5


def subgrad_polish(h, n, time_limit, lr_init=0.003):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    v = np.zeros(n)
    beta = 0.8
    lr = lr_init
    stall = 0

    while time.time() - start < time_limit:
        corr = compute_corr(h)
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        dx = 2.0 / n
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + sh:j1b + sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def multi_start_polish(best_h, n, time_limit, n_restarts=3):
    rng = np.random.default_rng(123)
    best_c5 = compute_c5(best_h)
    best = best_h.copy()
    per_restart = time_limit / n_restarts

    for r in range(n_restarts):
        if time_limit - (time.time() - time_limit) < 5:
            break
        h_r = best + rng.uniform(-0.05, 0.05, n)
        h_r = project_to_feasible(h_r, n)
        h_r, c5_r = adam_anneal(h_r, n, time_limit=per_restart,
                                tau_start=15.0, tau_end=5.0, lr=0.03)
        if c5_r < best_c5:
            best_c5 = c5_r
            best = h_r.copy()

    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: n=1800, main Adam optimization
    n1 = 1800
    h1 = interp_to_n(best_h, n1)
    h1 = project_to_feasible(h1, n1)
    tl1 = remaining() * 0.35
    if tl1 > 30:
        h1, c5_1 = adam_anneal(h1, n1, time_limit=tl1,
                               tau_start=40.0, tau_end=6.0, lr=0.05)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: n=3600, higher resolution
    n2 = 3600
    if remaining() > 120:
        h2 = interp_to_n(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl2 = remaining() * 0.3
        h2, c5_2 = adam_anneal(h2, n2, time_limit=tl2,
                               tau_start=40.0, tau_end=6.0, lr=0.03)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: multi-start at best resolution
    if remaining() > 60:
        nf = len(best_h)
        tl3 = remaining() * 0.3
        best_h, best_c5 = multi_start_polish(best_h, nf, time_limit=tl3,
                                             n_restarts=3)

    # Phase 4: subgradient polish
    if remaining() > 30:
        nf = len(best_h)
        tl4 = remaining() * 0.35
        h4, c5_4 = subgrad_polish(best_h, nf, time_limit=tl4, lr_init=0.002)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: final subgrad with even smaller lr
    if remaining() > 20:
        nf = len(best_h)
        tl5 = remaining() * 0.5
        h5, c5_5 = subgrad_polish(best_h, nf, time_limit=tl5, lr_init=0.0005)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
