import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    th = (lo + hi) / 2.0
    return np.clip(g - th, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    ec = np.exp(shifted)
    s = ec.sum()
    p = ec / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (mc + np.log(s) / tau)
    return obj, grad

def subgrad_step(h, lr):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)

    grad = np.zeros(n)

    j1 = min(n, sh + n)

        idx = np.arange(j0, j1)
        grad[idx] += (1.0 - h[idx - sh])

    j1b = min(n, n - sh)

        grad[idx] -= h[idx + sh]
    grad *= dx
    h_new = project_to_feasible(h - lr * grad, n)
