def c5_subgradient(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr))
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] += (-h[i])
    else:
        m = -lag
        for i in range(n - m):
            grad[i + m] += (1.0 - h[i])
            grad[i] += (-h[i + m])
    return grad * dx

def project_feasible(h, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hc = np.clip(h - mid, 0.0, 1.0)
        if hc.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def upsample_to(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    scaled = np.arange(new_n) * old_n / new_n
    idx_lo = np.floor(scaled).astype(int) % old_n
    idx_hi = (idx_lo + 1) % old_n
    frac = scaled - np.floor(scaled)
    h_new = h[idx_lo] * (1.0 - frac) + h[idx_hi] * frac
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, new_n)

def optimize_pgd(h, n, time_limit, start_time, rng):
    best = h.copy()
    best_val = c5_value(h)
    step_size = 0.08
    no_improve = 0
    kicks = 0
    kick_threshold = 80
    beta_mom = 0.7
    vel = np.zeros(n)

    while time.time() - start_time < time_limit:
        grad = c5_subgradient(h)

        if no_improve > kick_threshold and kicks < 40:
            kick_size = 0.02 + 0.015 * kicks
            h = h + rng.normal(0, kick_size, n)
            h = project_feasible(h, n)
            vel = np.zeros(n)
            kicks += 1
            no_improve = 0
            kick_threshold = min(kick_threshold + 60, 400)
            continue

        vel = beta_mom * vel - step_size * grad
        h_new = project_feasible(h + vel, n)
        val = c5_value(h_new)

        if val < best_val - 1e-13:
            best = h_new.copy()
            best_val = val
            no_improve = 0
            step_size = min(step_size * 1.015, 0.4)
        else:
            no_improve += 1
            step_size = max(step_size * 0.97, 0.003)

        h = h_new

    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_start = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_start = np.ones(80) * 0.5
    n0 = len(h_start)
    h_start = project_feasible(h_start, n0)

    best_h = h_start.copy()
    best_c5 = c5_value(best_h)

    # Phase 1: PGD at original resolution
    t1 = min(budget_s * 0.25, 300)
    h1, v1 = optimize_pgd(h_start, n0, t1, t0, rng)
    if v1 < best_c5:
        best_c5 = v1
        best_h = h1

    # Phase 2: Second PGD run with perturbation at same resolution
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.55:
        h_pert = best_h + rng.normal(0, 0.04, n0)
        h_pert = project_feasible(h_pert, n0)
        t2 = min(budget_s * 0.2, 250)
        h2, v2 = optimize_pgd(h_pert, n0, t2, time.time(), rng)
        if v2 < best_c5:
            best_c5 = v2
            best_h = h2

    # Phase 3: Upsample to 2x and optimize
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.7:
        n2 = min(2 * n0, 200)
        h_u = upsample_to(best_h, n2)
        t3 = min(budget_s * 0.25, 350)
        h3, v3 = optimize_pgd(h_u, n2, t3, time.time(), rng)
        if v3 < best_c5:
            best_c5 = v3
            best_h = h3

    # Phase 4: Upsample to next level
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.85 and len(best_h) < 240:
        n4 = min(len(best_h) * 2, 240)
        h_u4 = upsample_to(best_h, n4)
        t4 = min(budget_s * 0.15, 200)
        h4, v4 = optimize_pgd(h_u4, n4, t4, time.time(), rng)
        if v4 < best_c5:
            best_c5 = v4
            best_h = h4

    # Phase 5: Random restarts if time remains
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.9:
        n_r = len(best_h)
        remaining = budget_s * 0.9 - elapsed
        n_restarts = int(remaining // 30)
        n_restarts = max(1, min(n_restarts, 5))
        for _ in range(n_restarts):
            if time.time() - t0 > budget_s * 0.92:
                break
            h_r = best_h + rng.normal(0, 0.08, n_r)
            h_r = project_feasible(h_r, n_r)
            t_r = min(budget_s * 0.05, 60)
            h_r2, v_r = optimize_pgd(h_r, n_r, t_r, time.time(), rng)
            if v_r < best_c5:
                best_c5 = v_r
                best_h = h_r2

    # Final: ensure feasibility and compute exact value
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = c5_value(best_h)

    return best_h, final_c5, len(best_h)
