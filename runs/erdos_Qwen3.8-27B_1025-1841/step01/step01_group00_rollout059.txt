Change: (1 - h_new[b-sh])

Change: 1
Using formula with old: (1 - h_old[b-sh]) = (1 - h_old[a]) = (1-1) = 0 ✗!

The correct formula:
Change at i=b: (1 - h_new[b-sh])
- If b-sh ≠ a and b-sh ≠ b: h_new[b-sh] = h_old[b-sh], so change = (1 - h[b-sh])

- If b-sh = b: h_new[b] = 1, so change = (1 - 1) = 0 (but this is shift 0, self-term)

And for the term at i where i-sh = a (i.e., i = a+sh):
Old: h_old[i] * (1 - h_old[a]) = h[i] * (1-1) = 0
New: h_new[i] * (1 - h_new[a]) = h[i] * (1-0) = h[i]
Change: h[i]

And for the term at i where i-sh = b (i.e., i = b+sh):
Old: h_old[i] * (1 - h_old[b]) = h[i] * (1-0) = h[i]
New: h_new[i] * (1 - h_new[b]) = h[i] * (1-1) = 0
Change: -h[i]

Sum of all terms that change:

   Wait, that's the change in the VERTEX value, but the CHANGE in corr is new - old = 0 - (1-h_old[a-sh])

2. Term at i=b: 0 → 1*(1-h_new[b-sh])

3. Term at i=a+sh (if 0≤a+sh<n and a+sh≠a and a+sh≠b): 
   h[a+sh]*(1-1) → h[a+sh]*(1-0)
   Change: h[a+sh]

4. Term at i=b+sh (if 0≤b+sh<n and b+sh≠a and b+sh≠b):

import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    target = n // 2
    idx = np.argpartition(h, -target)[-target:]

    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()

    best_h = h.copy()

    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0

            break

        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -400, 400)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        obj = dx * (max_c + np.log(s) / tau)

            best_h = h.copy()

            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)

            v *= 0.5
            stall = 0
            if lr < 1e-9:

        h = h_new
    
    return best_h, compute_c5(best_h)

def binary_local_search(h_bin, time_limit, max_iters=50000, rng=None):
    """Local search on binary sequence using simulated annealing swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h_bin)
    corr = compute_corr(h_bin)
    best_c5 = corr.max() * (2.0 / n)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()
    T = 0.02  # initial temperature
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random swap
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        if a == b:
            continue
        
        # Try the swap
        h_bin[a] = 0
        h_bin[b] = 1
        new_corr = compute_corr(h_bin)
        new_c5 = new_corr.max() * (2.0 / n)
        
        # Accept/reject (simulated annealing)
        if new_c5 <= curr_c5 or rng.random() < np.exp((curr_c5 - new_c5) / max(T, 1e-10)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            # Reject: undo
            h_bin[a] = 1
            h_bin[b] = 0
        
        T *= 0.9995
        if T < 1e-6:
            T = 1e-6
    
    return best_h, best_c5
