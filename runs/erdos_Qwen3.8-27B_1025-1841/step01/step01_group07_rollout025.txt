def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_grad(h):
    """Compute c5 and its (sub)gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5val = np.max(corr) * dx
    k_idx = np.argmax(corr)
    k = k_idx - (n - 1)  # actual shift
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        # I(0<=j-k<n AND 0<=j<n) * (1-h[j-k])
        if 0 <= j - k < n and 0 <= j < n:
            val += (1.0 - h[j - k])
        # - I(0<=j<n AND 0<=j+k<n) * h[j+k]
        if 0 <= j < n and 0 <= j + k < n:
            val -= h[j + k]
        grad[j] = val
    grad *= dx
    return c5val, grad

def smooth_c5(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # log-sum-exp trick
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / beta

    weights = np.exp(corr - m)
    weights /= np.sum(weights)
    val = m + np.log(np.sum(weights)) / beta
    # Gradient is weighted average of gradients of each c_k
    grad = np.zeros(n)

    if diff > 0:

        order = np.argsort(1 - h)  # most room first

    corr = np.correlate(h, 1.0 - h, mode='full')
    c5val = float(np.max(corr) * dx)
    k_idx = int(np.argmax(corr))

    j_end2 = min(n, n - k)
    if j_start2 < j_end2:
        grad[j_start2:j_end2] -= h[j_start2+k:j_end2+k]
    
    grad *= dx
    return c5val, grad

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_grad_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5val = float(np.max(corr) * dx)
    k_idx = int(np.argmax(corr))
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    j_start = max(0, k)
    j_end = min(n, n + k)
    if j_start < j_end:
        grad[j_start:j_end] += (1.0 - h[j_start - k:j_end - k])
    j_start2 = max(0, -k)
    j_end2 = min(n, n - k)
    if j_start2 < j_end2:
        grad[j_start2:j_end2] -= h[j_start2 + k:j_end2 + k]
    grad *= dx
    return c5val, grad

def smooth_c5_grad(h, beta):

        w = weights[i]

        j_start = max(0, k)
        j_end = min(n, n + k)
        if j_start < j_end:

            cnt = mask.sum()
            if cnt == 0:
