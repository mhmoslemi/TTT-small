    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    scaled = T * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    p = exp_s / np.sum(exp_s)
    
    # Gradient: df/dh[j] = dx * [sum_{k=0}^{j} p[k]*(1-h[j-k]) - sum_{k=0}^{n-1-j} p[k]*h[j+k]]
    grad = np.zeros(n)

        t1 = np.dot(p[0:j+1], 1.0 - h[j::-1])
        # Second term: sum_{k=0}^{n-1-j} p[k] * h[j+k]
        t2 = np.dot(p[0:n-j], h[j:])
        grad[j] = dx * (t1 - t2)
    return grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def soft_max_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    p = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for j in range(n):
        t1 = np.dot(p[0:j+1], 1.0 - x[j::-1])
        t2 = np.dot(p[0:n-j], x[j:])
        grad[j] = dx * (t1 - t2)
    return grad

def project(h, target):
    # Project onto {0<=h<=1, sum(h)=target}
    h = np.array(h, dtype=float).copy()
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def slsqp_with_grad(h0, n, T, maxiter=500):
    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T)
    def g(x):
        return soft_max_grad(x, n, T)
    def c_eq(x):
        return np.sum(x) - target
    def dc_eq(x):
        return np.ones(n)
    
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': dc_eq}],
                       options={'ftol': 1e-16, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except:
        pass
    return project(h0, target)

def proj_grad_desc(h0, n, T, lr=0.01, n_steps=1000, deadline=None):
    target = n / 2.0
    h = project(h0.copy(), target)
    prev_f = soft_max_obj(h, n, T)
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        g = soft_max_grad(h, n, T)
        # Backtracking line search
        alpha = lr
        for _ in range(20):
            h_new = project(h - alpha * g, target)
            f_new = soft_max_obj(h_new, n, T)
            if f_new < prev_f:
                break
            alpha *= 0.5
        else:
            break
        if f_new >= prev_f:
            break
        h = h_new
        prev_f = f_new
    
    return h

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def soft_max_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    p = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for j in range(n):
        t1 = np.dot(p[j:n+j][::-1], 1.0 - x)
        t2 = np.dot(p[n-1-j:2*n-1-j], x)
        grad[j] = dx * (t1 - t2)
    return grad

def project(h, target):
    h = np.array(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def slsqp_opt(h0, n, T, maxiter=500, deadline=None):
    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T)
    def g(x):
        return soft_max_grad(x, n, T)
    def c_eq(x):
        return np.sum(x) - target
    
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-15, 'maxiter': maxiter})
        if deadline is not None and time.time() > deadline:
            return project(h0, target)
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return project(h0, target)

def proj_grad_desc(h0, n, T, lr, n_steps, deadline):
    target = n / 2.0
    h = project(h0.copy(), target)
    prev_f = soft_max_obj(h, n, T)
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break

        alpha = lr

            f_new = soft_max_obj(h_new, n, T)
            if f_new < prev_f:
                break
            alpha *= 0.5
        if f_new >= prev_f - 1e-16:
            break
        h = h_new
        prev_f = f_new
    
    return h
