import numpy as np
import time

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    for k in range(2 * n - 1):
        wk = softmax[k]
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        if m >= 0:
            if n - m > 0:
                grad[0:n-m] += wk * (1.0 - h[m:n])
                grad[m:n] -= wk * h[0:n-m]
        else:
            p = -m
            if n - p > 0:
                grad[p:n] += wk * (1.0 - h[0:n-p])
                grad[0:n-p] -= wk * h[p:n]
    grad *= dx
    return soft_val, grad

def adam_optimize(h_start, n, T_schedule, max_steps, time_limit, start_time, budget_s):
    """Run Adam optimization with temperature schedule."""
    h = h_start.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_c5 = compute_c5(h)
    
    for step in range(max_steps):
        if time.time() - start_time > budget_s - 30:
            break
        
        T = T_schedule[step % len(T_schedule)]
        val, grad = soft_max_obj_grad(h, T)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = make_feasible(h_new, n)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            h = h_new
        else:
            # Rejection sampling: don't accept if worse
            h = h_new  # Still update h to keep momentum going
        
        # Decay learning rate
        if step % 1000 == 0 and step > 0:
            lr *= 0.8
    
    return h, best_c5

def coord_descent_polish(h, n, time_limit, start_time, budget_s):
    """Coordinate descent with binary search for optimal value of each coordinate."""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for iteration in range(20):
        if time.time() - start_time > budget_s - 20:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 20:
                break
            # For coordinate j, find optimal value via ternary search
            # The objective is convex in h[j] (max of linear functions)
            lo = 0.0
            hi = 1.0
            # Try a few candidate values
            best_val = h[j]
            best_c = best_c5
            for _ in range(8):  # 8 iterations of ternary search
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                for cand in [m1, m2]:
                    trial = h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c:
                        best_c = c
                        best_val = cand
                # Ternary search step
                t1 = h.copy()
                t1[j] = m1
                t1 = make_feasible(t1, n)
                c1 = compute_c5(t1)
                t2 = h.copy()
                t2[j] = m2
                t2 = make_feasible(t2, n)
                c2 = compute_c5(t2)
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            
            if best_c < best_c5 - 1e-12:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    
    return h, best_c5

import numpy as np
import time

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c)
    shifted = c - max_c
    exp_shifted = np.exp(T * shifted)
    sum_exp = np.sum(exp_shifted)
    soft_val = (max_c + np.log(sum_exp) / T) * dx
    softmax = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = softmax[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            L = n - m
            if L > 0:
                grad[0:L] += wk * (1.0 - h[m:n])
                grad[m:n] -= wk * h[0:L]
        else:
            p = -m
            L = n - p
            if L > 0:
                grad[p:n] += wk * (1.0 - h[0:L])
                grad[0:L] -= wk * h[p:n]
    grad *= dx
    return soft_val, grad

def adam_opt(h0, T_list, budget_fn):

    h = h0.copy()
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.005
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while not budget_fn():
        T = T_list[t % len(T_list)]
        val, grad = soft_max_obj_grad(h, T)
        t += 1
        m_acc = b1 * m_acc + (1 - b1) * grad
        v_acc = b2 * v_acc + (1 - b2) * (grad * grad)
        m_hat = m_acc / (1.0 - b1 ** t)
        v_hat = v_acc / (1.0 - b2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = make_feasible(h, n)
        if t % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr *= 0.95
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def coord_polish(h, max_passes, budget_fn):
    n = len(h)
    best_c5 = compute_c5(h)

        if budget_fn():
