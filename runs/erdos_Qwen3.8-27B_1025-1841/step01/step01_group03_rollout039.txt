import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate, convolve
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(f, T):
    f_max = np.max(f)
    if f_max > 10 * T:
        shifted = f - f_max
        shifted = np.clip(shifted / T, -100, 100)
        return f_max + T * np.log(np.sum(np.exp(shifted)))
    else:
        f_clipped = np.clip(f / T, -100, 100)
        return T * np.log(np.sum(np.exp(f_clipped)))

def weighted_sum_exp(corr, T):
    f_max = np.max(corr)
    if f_max > 10 * T:
        shifted = np.clip((corr - f_max) / T, -100, 100)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
    else:
        clamped = np.clip(corr / T, -100, 100)
        weights = np.exp(clamped)
        weights /= np.sum(weights)
    return weights

def obj_and_grad(h, n, T):
    h = np.clip(h, 0, 1)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    obj = smooth_max(corr, T)
    w = weighted_sum_exp(corr, T)
    
    # Gradient
    C1 = correlate(w, h, mode='full')[:n]
    C2 = convolve(w, h, mode='full')[n-1:n-1+n]
    grad = dx * (1.0 - C1 - C2)
    
    return obj, grad

def project(h, n):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase sum by diff
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            add = min(remaining, room[i])
            h[i] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to decrease sum by -diff
        space = h.copy()
        order = np.argsort(-space)
        remaining = -diff
        for i in order:
            sub = min(remaining, space[i])
            h[i] -= sub
            remaining -= sub
            if remaining <= 1e-12:
                break
    return h

def interpolate(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Circular interpolation
    h_ext = np.concatenate([h_old, h_old[:n_new if n_new < n_old else 0]])
    # Simpler: just use linear interpolation
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]))
    h_new = project(h_new, n_new)
    return h_new

    h = h_init.copy()

            bounds=[(0, 1)] * n,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-8, 'iprint': 0}

        h_new = project(result.x, n)
