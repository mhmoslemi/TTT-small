def project_feasible(h, N):
    """Project h onto {0<=h<=1, sum(h)=N/2}"""

    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def smooth_objective(h, N, temp):
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    # Smooth max
    scaled = temp * corr
    max_scaled = np.max(scaled)
    logsum = max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))
    return (dx / temp) * logsum

def smooth_gradient(h, N, temp):

    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = temp * corr

    weights = np.exp(scaled - max_scaled)
    weights /= weights.sum()
    # grad of corr[p] w.r.t. h[j]

    gradient = np.zeros(N)

                c5_2 = c5_of(h2, N)
                if c5_2 < best_c5 - 1e-10:
                    h = h2
                    best_c5 = c5_2
                    improvement = True
                    break
        if time.time() - t0 > t_limit:
            break
        noise_scale *= 0.7
        if noise_scale < 0.001:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    N0 = 42
    try:
        h0 = np.array(initial_h_values, dtype=float)
        N0 = len(h0)
    except Exception:
        h0 = np.ones(N0) * 0.5

    best_h = h0.copy()
    best_c5 = c5_of(h0, N0)
    best_N = N0

    t_left = budget_s - (time.time() - t_start)
    t_alloc = min(250, t_left * 0.25)
    h1, c5_1 = pg_optimize(h0, best_N, t_alloc, rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    t_left = budget_s - (time.time() - t_start)
    h2, c5_2 = local_refine(best_h, best_N, min(60, t_left * 0.08), rng)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    n_restarts = 5
    t_left = budget_s - (time.time() - t_start) - 300
    if t_left > 30:
        t_per = max(15, t_left / n_restarts - 5)
        for i in range(n_restarts):
            if time.time() - t_start > budget_s - 250:
                break
            if i % 2 == 0:
                h_r = project_to_feasible(rng.dirichlet(np.ones(best_N) * 0.4), best_N)
            else:
                h_r = project_to_feasible(best_h + 0.15 * rng.randn(best_N), best_N)
            h_r, c5_r = pg_optimize(h_r, best_N, t_per, rng)
            h_r, c5_r = local_refine(h_r, best_N, min(20, t_per * 0.3), rng)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r

    t_left = budget_s - (time.time() - t_start) - 60
    for mult in [2, 3]:
        if t_left < 50:
            break
        N = best_N * mult
        if N > 300:
            break
        idx = np.round(np.linspace(0, best_N - 1, N)).astype(int)
        h_up = best_h[idx]
        h_up = project_to_feasible(h_up, N)
        t_alloc = min(100, t_left * 0.6)
        h_up, c5_up = pg_optimize(h_up, N, t_alloc, rng)
        t_left2 = budget_s - (time.time() - t_start) - 40
        if t_left2 > 20:
            h_up, c5_up = local_refine(h_up, N, min(30, t_left2 * 0.4), rng)
        if c5_up < best_c5:
            best_c5, best_h, best_N = c5_up, h_up, N
        t_left = budget_s - (time.time() - t_start) - 60

    t_left = budget_s - (time.time() - t_start)
    if t_left > 30:
        h_f, c5_f = local_refine(best_h, best_N, min(80, t_left * 0.7), rng)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    Nf = len(best_h)
    best_h = project_to_feasible(best_h, Nf)
    best_c5 = c5_of(best_h, Nf)

    return best_h, best_c5, Nf
