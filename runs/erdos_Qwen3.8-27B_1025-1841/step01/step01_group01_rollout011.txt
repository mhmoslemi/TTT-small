    corr = np.correlate(h, 1 - h, mode="full") * dx

    if t > 1e6:
        return cmax
    shifted = t * (corr - cmax)

def project_constraint(h, n):

import numpy as np
import time
from scipy.optimize import minimize

_g_n = 45
_g_t = 10.0
_g_dx = 2.0 / 45.0

def _objective(h):
    dx = _g_dx
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    t = _g_t
    if t < 10.0:
        return float(cmax)
    exp_arg = np.exp(np.clip(t * (corr - cmax), -400.0, 0.0))
    return float(cmax + np.log(np.mean(exp_arg)) / t)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def interp_h(old_h, old_n, new_n):
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def make_sine(n, freq, amp):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x)
    return project(h, n)

def make_two_sine(n, f1, f2, a1, a2):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + a1 * np.sin(np.pi * f1 * x) + a2 * np.sin(np.pi * f2 * x)
    return project(h, n)

def optimize_local(h_init, n, time_limit):
    global _g_n, _g_t, _g_dx
    t0 = time.time()
    _g_n = n
    _g_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    best = project(h_init, n)
    best_c5 = compute_c5(best, n)

    t_schedule = [3.0, 8.0, 20.0, 50.0, 100.0, 300.0, 800.0, 2000.0, 5000.0]
    for t_val in t_schedule:
        if time.time() - t0 > time_limit - 2:
            break
        _g_t = t_val
        maxfun = max(80, min(2000, int((time_limit * 0.7) / max(0.01, n / 40000.0) / 15)))
        maxiter = max(15, min(300, maxfun // 8))
        try:
            res = minimize(_objective, best.copy(), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'maxfun': maxfun,
                                    'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40})
            h_new = project(res.x, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Final pass with very large t
    if time.time() - t0 < time_limit - 1:
        _g_t = 1e6
        try:
            res = minimize(_objective, best.copy(), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 100, 'maxfun': 400,
                                    'ftol': 1e-14, 'gtol': 1e-10})
            h_new = project(res.x, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass

    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    parent = None
    try:
        if initial_h_values is not None:
            parent = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    candidates = []

    if parent is not None:
        pn = len(parent)
        h_p = project(parent.copy(), pn)
        candidates.append(('parent', h_p, pn))
        for mult in [2, 3, 4]:
            nn = pn * mult
            if nn <= 1000:
                h_up = project(interp_h(parent, pn, nn), nn)
                candidates.append(('p_x%d' % mult, h_up, nn))

    # Sine and two-sine initializations
    for n_s in [45, 60, 90, 120, 180, 240]:
        for f in [1, 2, 3, 4, 5, 7, 9]:
            h_s = make_sine(n_s, f, 0.35)
            candidates.append(('s%d_n%d' % (f, n_s), h_s, n_s))
        for f1 in [2, 3, 4]:
            for f2 in [5, 7, 11]:
                if f1 != f2:
                    h2 = make_two_sine(n_s, f1, f2, 0.25, 0.15)
                    candidates.append(('ts_%d_%d_n%d' % (f1, f2, n_s), h2, n_s))

    # Constant
    for n_c in [60, 120, 240]:
        candidates.append(('c_n%d' % n_c, np.full(n_c, 0.5), n_c))

    # Score and sort
    scored = []
    for name, h, n in candidates:
        c5 = compute_c5(h, n)
        scored.append((c5, name, h, n))
    scored.sort(key=lambda x: x[0])

    # Deduplicate: keep best per n
    best_per_n = {}
    for c5, name, h, n in scored:
        if n not in best_per_n or c5 < best_per_n[n][0]:
            best_per_n[n] = (c5, name, h, n)
    unique = sorted(best_per_n.values(), key=lambda x: x[0])

    # Time allocation
    reserve = 80
    n_try = min(len(unique), 25)
    available = budget_s - (time.time() - t_start) - reserve
    time_per_cand = max(15.0, available / max(1, n_try))

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    for i in range(n_try):
        if time.time() - t_start > budget_s - reserve:
            break
        c5_i, name, h_init, n = unique[i]

        if best_h is not None:
            h_fb = project(interp_h(best_h, best_n, n), n)
            c5_fb = compute_c5(h_fb, n)
            if c5_fb < c5_i:
                h_init = h_fb
                c5_i = c5_fb

        t_left = min(time_per_cand, budget_s - reserve - (time.time() - t_start) - 5.0)
        if t_left < 10:
            break

        h_opt, c5_opt = optimize_local(h_init, n, t_left)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n

    # Final refinement rounds
    if best_h is not None:
        elapsed = time.time() - t_start
        rem = budget_s - elapsed - 10
        if rem > 40:
            h_r, c5_r = optimize_local(best_h, best_n, rem)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
                best_n = best_n
        elapsed = time.time() - t_start
        rem = budget_s - elapsed - 10
        if rem > 30 and best_n * 2 <= 1200:
            h_up = project(interp_h(best_h, best_n, best_n * 2), best_n * 2)
            h_up, c5_up = optimize_local(h_up, best_n * 2, rem)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up
                best_n = best_n * 2

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100

    best_h = project(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return (best_h, final_c5, best_n)
