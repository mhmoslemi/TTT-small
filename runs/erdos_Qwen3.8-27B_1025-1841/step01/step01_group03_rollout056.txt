import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h + mid, 0.0, 1.0)
        if h_try.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


def obj_slsqp_fn(h, n, dx):
    target = n / 2.0

    corr = np.correlate(h_c, 1.0 - h_c, mode='full') * dx

def make_soft_obj_factory(n, dx, beta, pw):

class SoftObj:
    def __init__(self, n, dx, beta, pw):
        self.n = n
        self.dx = dx
        self.beta = beta
        self.pw = pw
        self.target = n / 2.0

    def __call__(self, h):
        h_c = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h_c, 1.0 - h_c, mode='full') * self.dx
        mx = np.max(corr)
        lse = mx + np.log(np.sum(np.exp((corr - mx) * self.beta))) / self.beta
        pen = self.pw * (h_c.sum() - self.target) ** 2
        return lse + pen
