import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    d = max_idx - (n - 1)
    
    grad = np.zeros(n)
    if d >= 0:
        # f = dx * sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
        # ∂f/∂h[j] = dx * [(1-h[j-d]) * 1_{d≤j} - h[j+d] * 1_{j≤n-1-d}]
        if d == 0:
            grad += dx * (1 - 2*h)
        else:
            grad[d:] += dx * (1 - h[:n-d])  # (1-h[j-d]) for j=d,...,n-1
            grad[:n-d] -= dx * h[d:]  # -h[j+d] for j=0,...,n-1-d
    else:
        dp = -d
        # f = dx * sum_{i=0}^{n-1-dp} h[i] * (1-h[i+dp])
        if dp == 0:
            grad += dx * (1 - 2*h)
        else:
            grad[:n-dp] += dx * (1 - h[dp:])  # (1-h[j+dp]) for j=0,...,n-1-dp
            grad[dp:] -= dx * h[:n-dp]  # -h[j-dp] for j=dp,...,n-1
    
    return grad

def project_feasible(h, target_sum):
    n = len(h)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    # sum(clip(h-tau,0,1)) is decreasing in tau
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0, 1)

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    old_x = np.linspace(0, 2, n_old, endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(new_x, old_x, h_old)
    # Ensure sum constraint
    h_new = project_feasible(h_new, n_new / 2)
    return h_new

def optimize_gradient(h, n, max_iters=5000, time_limit=60):
    start = time.time()
    h = project_feasible(h.copy(), n/2)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    lr = 0.05  # initial learning rate
    patience = 0
    max_patience = 200
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        c5, grad = evaluate_c5(h), c5_gradient(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n/2)
        
        # Check if improvement
        c5_new = evaluate_c5(h_new)
        if c5_new < c5:
            h = h_new
            # Adaptive lr
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n/2)
            h = h_new
    
    return best_h, best_c5

def simulated_annealing(h, n, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    start = time.time()
    target_sum = n / 2
    h = project_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    T = 0.1
    T_min = 0.001
    cool_rate = 0.999
    
    while time.time() - start < time_limit and T > T_min:
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Generate neighbor: random perturbation
        # Choose a few random positions to modify
        n_change = max(1, n // 50)
        idx = rng.choice(n, size=n_change, replace=False)
        delta = rng.uniform(-0.3, 0.3, size=n_change)
        h_new = h.copy()
        h_new[idx] += delta
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = evaluate_c5(h_new)
        delta_c5 = c5_new - c5
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            h = h_new
        
        T *= cool_rate
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Resolution and time allocation
    resolutions = [200, 400, 800]
    time_alloc = [0.3, 0.4, 0.3]  # fractions of budget
    
    for res_idx, n in enumerate(resolutions):
        if time.time() - start_time > budget_s * 0.95:
            break
        
        time_left = budget_s - (time.time() - start_time)
        alloc_time = time_left * time_alloc[res_idx] / (sum(time_alloc[res_idx:]) )
        if alloc_time < 10:
            alloc_time = time_left * 0.5
        
        # Create initial guess
        if h_init is not None:
            h = upsample(h_init, n)
        else:
            h = np.full(n, 0.5)
        
        # Phase 1: Gradient optimization
        t1 = time.time()
        h, c5 = optimize_gradient(h, n, time_limit=alloc_time * 0.4)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n
        
        # Phase 2: Simulated annealing
        if time.time() - start_time < budget_s * 0.85:
            t2_alloc = min(alloc_time * 0.3, budget_s * 0.85 - (time.time() - start_time))
            if t2_alloc > 5:
                h2, c5_2 = simulated_annealing(h, n, time_limit=t2_alloc, seed=seed+1)
                if c5_2 < best_c5:
                    best_c5, best_h, best_n = c5_2, h2.copy(), n
                # Follow up with gradient
                if time.time() - start_time < budget_s * 0.9:
                    h3, c5_3 = optimize_gradient(h2, n, time_limit=alloc_time * 0.3)
                    if c5_3 < best_c5:
                        best_c5, best_h, best_n = c5_3, h3.copy(), n
        
        # Phase 3: Local refinement with SLSQP
        if time.time() - start_time < budget_s * 0.9:
            def slsqp_obj(hv):
                return evaluate_c5(hv)
            
            def slsqp_grad(hv):
                return c5_gradient(hv)
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2, 
                     'jac': lambda x: np.ones(n)}]
            bounds = [(0, 1)] * n
            
            try:
                res = minimize(slsqp_obj, h, method='SLSQP', jac=slsqp_grad,
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 100, 'ftol': 1e-8})
                if res.success or res.fun < best_c5:
                    h_slq = project_feasible(res.x, n/2)
                    c5_slq = evaluate_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5, best_h, best_n = c5_slq, h_slq.copy(), n
            except:
                pass
    
    # Final check
    c5_final = evaluate_c5(best_h)
    if c5_final > best_c5:
        best_c5 = c5_final
    
    return best_h, best_c5, best_n

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    d = max_idx - (n - 1)

    if d >= 0:
        if d == 0:
            grad[:] = dx * (1.0 - 2.0 * h)

    else:

        return evaluate_c5(hv)

        return c5_gradient(hv)

        h_new = project_feasible(h_new, ts)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / T):
            h = h_new
        T *= cool
    return best_h, best_c5

def _sum_con(x):
    return np.sum(x) - len(x) / 2.0

def _sum_con_jac(x):
    return np.ones(len(x))

def slsqp_refine(h, n, maxiter=150):
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(evaluate_c5, h, method='SLSQP', jac=c5_gradient,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h_out = project_feasible(res.x, n / 2.0)
        return h_out, evaluate_c5(h_out)
    except Exception:
        return h, evaluate_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h_init = None
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_init = np.array(globals()['initial_h_values'], dtype=float)

    best_h = None
    best_c5 = float('inf')

    if h_init is not None:
        cur_h = h_init.copy()
        cur_n = len(h_init)
    else:
        cur_n = 42
        cur_h = np.full(cur_n, 0.5)

    # Optimize at base resolution
    t1 = min(45.0, budget_s * 0.08)
    h1, c5_1 = gradient_descent(cur_h, cur_n, time_limit=t1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Build resolution list
    res_list = []
    for target in [84, 126, 200, 350, 550, 800]:
        if target > cur_n:
            res_list.append(target)
    res_list = res_list[:4]

    for n in res_list:
        if time.time() - start_time > budget_s * 0.92:
            break
        elapsed = time.time() - start_time
        remaining = budget_s * 0.92 - elapsed
        if remaining < 10:
            break

        h_cur = upsample(h1, n)

        # Gradient phase
        t_gd = min(60.0, remaining * 0.35)
        h_gd, c5_gd = gradient_descent(h_cur, n, time_limit=t_gd)
        if c5_gd < best_c5:
            best_c5, best_h = c5_gd, h_gd.copy()

        # Annealing phase
        if time.time() - start_time < budget_s * 0.75:
            t_sa = min(30.0, (budget_s * 0.75 - (time.time() - start_time)) * 0.5)
            if t_sa > 5:
                h_sa, c5_sa = anneal_refine(h_gd, n, time_limit=t_sa, seed=seed + n)
                if c5_sa < best_c5:
                    best_c5, best_h = c5_sa, h_sa.copy()
                # Gradient follow-up after SA
                if time.time() - start_time < budget_s * 0.85:
                    h_f, c5_f = gradient_descent(h_sa, n, time_limit=min(25.0, budget_s * 0.85 - (time.time() - start_time)))
                    if c5_f < best_c5:
                        best_c5, best_h = c5_f, h_f.copy()
                h_gd = h_f if time.time() - start_time < budget_s * 0.85 else h_gd

        # SLSQP polish
        if time.time() - start_time < budget_s * 0.85:
            h_sl, c5_sl = slsqp_refine(h_gd, n, maxiter=100)
            if c5_sl < best_c5:
                best_c5, best_h = c5_sl, h_sl.copy()

    # Random restarts at best resolution
    if time.time() - start_time < budget_s * 0.82:
        n_best = len(best_h)
        rng = np.random.RandomState(seed + 777)
        for trial in range(5):
            if time.time() - start_time > budget_s * 0.88:
                break
            h_r = rng.uniform(0.0, 1.0, size=n_best)
            h_r = project_feasible(h_r, n_best / 2.0)
            h_r2, c5_r2 = gradient_descent(h_r, n_best, time_limit=min(15.0, budget_s * 0.03))
            if c5_r2 < best_c5:
                best_c5, best_h = c5_r2, h_r2.copy()

    # Final SLSQP on best
    if time.time() - start_time < budget_s * 0.93:
        h_fin, c5_fin = slsqp_refine(best_h, len(best_h), maxiter=200)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin.copy()

    # Verify
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
