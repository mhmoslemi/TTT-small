import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj(h, T):
    """Log-sum-exp smoothed max of correlation*dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if T < 1e6:  # avoid overflow
        return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T)
    return float(m)

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, T, maxiter=300):
    target = n / 2.0
    def obj(x):

    n = len(h0)
    target = n / 2.0
    def obj(x):
        return lse_obj(x, T)
    def eq(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq}],
                       options={'ftol': 1e-15, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0.copy()

def to_binary(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin[idx] = 1.0
    return h_bin

def make_binary_pattern(n, pattern_type, seed=0):
    h = np.zeros(n)
    if pattern_type == 'rand':
        rng = np.random.RandomState(seed)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    elif pattern_type == 'interleaved':
        h[:n // 2] = np.arange(n // 2) % 2
        h[n // 2:] = 1 - h[:n // 2]
    elif pattern_type == 'block':
        h[:n // 2] = 1.0
    elif pattern_type == 'quasi':
        for i in range(n):
            h[i] = 1.0 if ((i * 168) % n) < n // 2 else 0.0
        if np.sum(h) != n // 2:
            idx = np.argsort(h)[::-1][:n // 2]
            h = np.zeros(n)
            h[idx] = 1.0
    return h

def sa_binary(h_bin, deadline, seed=42, T_start=0.08, T_end=0.002):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    cur_c5 = evaluate_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    it = 0
    no_improve = 0
    while time.time() < deadline:
        it += 1
        T = T_start * (T_end / T_start) ** (it / 200000.0)
        T = max(T, 1e-5)
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        oi = ones_idx[rng.randint(len(ones_idx))]
        zj = zeros_idx[rng.randint(len(zeros_idx))]
        h[oi] = 0.0
        h[zj] = 1.0
        new_c5 = evaluate_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 100000 and T > T_end * 2:
                T_start = T
                no_improve = 0
        else:
            h[oi] = 1.0
            h[zj] = 0.0
    return best_h, best_c5

def sweep_binary(h_bin, deadline, seed=0):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    cur_c5 = evaluate_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    while time.time() < deadline:
        improved = False
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        n_try = 200
        for _ in range(n_try):
            oi = ones_idx[rng.randint(len(ones_idx))]
            zj = zeros_idx[rng.randint(len(zeros_idx))]
            h[oi] = 0.0
            h[zj] = 1.0
            new_c5 = evaluate_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                cur_c5 = new_c5
                improved = True
            else:
                h[oi] = 1.0
                h[zj] = 0.0
        if not improved:
            break
    return best_h, best_c5

def interp_to(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(336, 0.5)
    n = len(h)
    h = project_feasible(h, n)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: SLSQP continuous optimization
    h_work = best_h.copy()
    for T in [30, 100, 400, 1500]:
        if time.time() > deadline - 500:
            break
        h_new = slsqp_opt(h_work, T, maxiter=200)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_work = h_new

    # Phase 2: Binary optimization via SA
    sa_budget = deadline - 250
    h_bin_from_cont = to_binary(best_h, n)

    # SA from continuous projection
    h_b1, c5_b1 = sa_binary(h_bin_from_cont, sa_budget, seed=seed)
    if c5_b1 < best_c5:
        best_c5 = c5_b1
        best_h = h_b1.copy()

    # SA from random
    if time.time() < deadline - 200:
        h_rand = make_binary_pattern(n, 'rand', seed=seed + 100)
        h_b2, c5_b2 = sa_binary(h_rand, deadline - 200, seed=seed + 200)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 3: Sweep / steepest descent on best binary
    if time.time() < deadline - 100:
        h_bin_best = to_binary(best_h, n)
        h_sw, c5_sw = sweep_binary(h_bin_best, deadline - 100, seed=seed + 300)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()

    # Phase 4: Continuous re-optimization from best binary
    if time.time() < deadline - 50:
        h_work = best_h.copy()
        # Add small noise to escape exact binary
        h_work += np.random.randn(n) * 0.01
        h_work = project_feasible(h_work, n)
        for T in [500, 3000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_opt(h_work, T, maxiter=200)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_work = h_new

    # Phase 5: Try one more SA round with different temp from best
    if time.time() < deadline - 15:
        h_bin_final = to_binary(best_h, n)
        h_b3, c5_b3 = sa_binary(h_bin_final, deadline, seed=seed + 500,
                                T_start=0.03, T_end=0.001)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
