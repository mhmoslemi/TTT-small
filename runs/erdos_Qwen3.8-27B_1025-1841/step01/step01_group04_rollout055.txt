import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        step = d / cnt
        # Don't overshoot
        if d > 0:
            max_step = 1.0 - h[mask].max()
            h[mask] += min(step, max_step)
        else:
            min_step = h[mask].min()
            h[mask] += max(step, min_step)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_grad(h, n, alpha):
    """Compute smoothed max objective value and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr  # not multiplying by dx yet
    m = scaled.max()
    exps = np.exp(scaled - m)
    sum_exps = exps.sum()
    f = dx * (m + np.log(sum_exps)) / alpha
    
    weights = exps / sum_exps  # softmax weights over lags
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-14:
            continue
        s = m_idx - (n - 1)
        wd = w * dx
        if s >= 0:
            if s < n:
                grad[:n-s] += wd * (1.0 - h[s:])
                grad[s:] -= wd * h[:n-s]
        else:
            t = -s
            if t < n:
                grad[t:] += wd * (1.0 - h[:n-t])
                grad[:n-t] -= wd * h[t:]
    
    return f, grad

def active_lag_grad(h, n, topk=3):
    """Gradient using top-k active lags (hard version)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top-k lags
    top_idx = np.argsort(corr)[-topk:]
    # Check if they're truly at the max (within tolerance)
    max_val = corr.max()
    active = top_idx[corr[top_idx] > max_val - 1e-10]
    
    grad = np.zeros(n)
    for m_idx in active:
        s = m_idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[:n-s] += (1.0 - h[s:])
                grad[s:] -= h[:n-s]
        else:
            t = -s
            if t < n:
                grad[t:] += (1.0 - h[:n-t])
                grad[:n-t] -= h[t:]
    grad /= len(active)
    grad *= dx
    return grad

def grad_descent(h0, n, alpha, lr, n_steps, deadline):
    """Projected gradient descent with backtracking line search."""
    h = h0.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        f, g = softmax_grad(h, n, alpha)
        
        # Backtracking line search
        lr_cur = lr
        for bt in range(20):
            h_try = h - lr_cur * g
            h_try = proj_feasible(h_try, n)
            f_try, _ = softmax_grad(h_try, n, alpha)
            if f_try < f - 1e-12:
                h = h_try
                f = f_try
                break
            lr_cur *= 0.5
        
        if f < best_f:
            best_f = f
            best_h = h.copy()
    
    return best_h

    xt = np.linspace(0, 2, n_to, endpoint=False)
    h_interp = np.interp(xt, xf, h)
    return proj_feasible(h_interp, n_to)
