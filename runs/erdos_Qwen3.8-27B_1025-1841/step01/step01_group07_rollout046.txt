import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_func(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx)

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def sum_constraint_jac(x):
    return np.ones(len(x))

def random_feasible(n, rng):
    h = rng.random(n)
    # Scale to meet sum constraint
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h * target / s
    h = np.clip(h, 0, 1)
    # Fix sum
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = (h < 1.0)
        else:
            mask = (h > 0.0)
        if mask.sum() == 0:
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0, 1)
    return h

def block_initial(n):
    h = np.zeros(n)
    h[n//4:3*n//4] = 1.0
    return h

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = (h < 1.0)
        else:
            mask = (h > 0.0)
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def interpolate_to_resolution(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    h_new = project_to_feasible(h_new, target)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Get initial values if available
    try:
        init_vals = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        init_vals = None
    
    if init_vals is not None:
        c5_init = compute_c5(init_vals)
        best_h = init_vals.copy()
        best_c5 = c5_init
    
    rng = np.random.RandomState(seed)
    
    # Progressive resolution optimization
    resolutions = [64, 128, 256]
    
    for n in resolutions:
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
        
        # Number of restarts based on budget
        elapsed = time.time() - t0
        remaining = budget_s - elapsed
        if remaining < 60:
            break
        
        n_restarts = max(5, int(remaining * 0.4 / (n / 64)))
        n_restarts = min(n_restarts, 30)
        
        for r in range(n_restarts):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.92:
                break
            
            # Choose initial point
            if r == 0 and best_h is not None:
                h0 = interpolate_to_resolution(best_h, n)
            elif r == 1 and best_h is not None:
                # Perturbed version
                h0 = interpolate_to_resolution(best_h, n)
                noise = 0.05 * rng.randn(n)
                h0 = project_to_feasible(h0 + noise, target)
            else:
                h0 = random_feasible(n, rng)
            
            # Run SLSQP
            try:
                maxiter = 300 if n <= 128 else 150
                res = minimize(
                    obj_func, h0,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
                )
                h_opt = project_to_feasible(res.x, target)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
            except Exception:
                pass
    
    if best_h is None:
        best_h = block_initial(64)
        best_h = project_to_feasible(best_h, 32.0)
        best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, len(best_h))

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def obj_func(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx)


def sum_constraint(x):
    return x.sum() - len(x) / 2.0


def sum_constraint_jac(x):
    return np.ones(len(x))


def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = (h < 1.0 - 1e-15)
        else:
            mask = (h > 1e-15)
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible(n, rng):
    h = rng.random(n)
    target = n / 2.0
    s = h.sum()
    if s > 1e-10:
        h = h * target / s
    return project_to_feasible(h, target)


def block_initial(n):
    h = np.zeros(n)
    h[n // 4:3 * n // 4] = 1.0
    return h


def alternating_initial(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return project_to_feasible(h, n / 2.0)


def fringe_initial(n, rng, block_n):
    h = block_initial(n).copy()
    noise = 0.15 * rng.randn(n)
    return project_to_feasible(h + noise, n / 2.0)


def interpolate_to_resolution(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_to_feasible(h_new, n_new / 2.0)


def try_optimize(h0, n, maxiter=200, ftol=1e-9):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    try:
        res = minimize(
            obj_func, h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
        )
        h_opt = project_to_feasible(res.x, target)
        return h_opt
    except Exception:
        return h0
