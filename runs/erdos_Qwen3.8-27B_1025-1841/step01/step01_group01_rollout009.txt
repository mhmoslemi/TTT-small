                        base_o += h[i]  # h[i] * (1 - 0) part
                        coeff_o += -1.0 * h[i]  # coefficient of t: -h[i]
                    else:

            elif idx == len(ts) - 1:

                ts_fine = np.linspace(max(0, ts[idx-1]), min(1, ts[idx+1]), 101)

                idx_fine = np.argmin(vals_fine)
                t_opt = ts_fine[idx_fine]
            
            h[j] = t_opt
            max_change = max(max_change, abs(t_opt - old_val))
        
        # After each sweep, repair the sum constraint
        h = repair_h(h, n)
        
        if max_change < tol:
            break
    
    return h
