import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def optimize_smooth(h, n, tau, max_iters, time_limit, initial_lr=None):

    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr is not None else 0.5
    v = np.zeros(n)
    beta = 0.9
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stall = 0

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > 80:

            v *= 0.5
            stall = 0
            if lr < 1e-8:

        h = h_new
    return best_h, best_obj

def interp_periodic(h, target_n):

    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

        # Coord descent on lower res
        h_lo, c5_lo = coord_descent_pass(h_lo, n_lo, m_search=20)
        if remaining() > 30:
            h_lo, c5_lo = coord_descent_pass(h_lo, n_lo, m_search=20)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            # Upsample back
            best_h = interp_periodic(h_lo, n_cd)
            best_h = project_to_feasible(best_h, n_cd)
            best_c5 = compute_c5(best_h)

    # Phase 6: Final subgradient + coord descent polish
    if remaining() > 20:
        h_f = best_h.copy()
        hl = len(h_f)
        h_f, c5_f = optimize_subgrad(h_f, hl, max_iters=30000, time_limit=min(80.0, remaining() * 0.4), initial_lr=0.003)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if remaining() > 15:
        h_f = best_h.copy()
        _, c5_f = coord_descent_pass(h_f, len(h_f), m_search=15)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
