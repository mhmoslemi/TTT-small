    n = len(h)
    h = h.astype(float).copy()
    for _ in range(50):

        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break

        if free.sum() > 0:

        else:

                idx = np.argsort(h)[::-1]
                for i in idx:
                    amt = min(h[i], diff)
                    h[i] -= amt
                    diff -= amt
                    if diff < 1e-14:
                        break
            else:
                # Need to increase sum; increase the smallest
                idx = np.argsort(h)
                for i in idx:
                    amt = min(1.0 - h[i], -diff)
                    h[i] += amt
                    diff += amt

    k_star = int(np.argmax(corr))

    grad = np.zeros(n)
    # First term: (1-h[j+m]) for j where 0 <= j+m < n
    j1 = np.arange(n)
