```python
def smooth_max_gradient(h, n, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -60, 0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    # corr has 2n-1 entries, index k corresponds to lag k - (n-1)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)  # signed lag
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            # First factor: k <= j < n, contribution (1-h[j-k])
            grad[k:] += w * (1.0 - h[:n-k])
            # Second factor: 0 <= j < n-k, contribution -h[j+k]
            grad[:n-k] += w * (-h[k:])
        else:
            kp = -k
            # First factor: 0 <= j < n-kp, contribution (1-h[j+kp])
            grad[:n-kp] += w * (1.0 - h[kp:])
            # Second factor: kp <= j < n, contribution -h[j-kp]
            grad[kp:] += w * (-h[:n-kp])
    
    return grad * dx
```

```python
        h = np.clip(h, 0.0, 1.0)
```

```python
                candidates.append(old + 0.08)
            if old > 0.03:
                candidates.append(old - 0.03)
            if old < 0.97:
                candidates.append(old + 0.03)
            best_val = old
            curr_c5 = compute_c5(h, dx)
            for c in candidates:
                c_clipped = max(0.0, min(1.0, c))
                h[i] = c_clipped
                c5_test = compute_c5(h, dx)
                if c5_test < curr_c5 - 1e-13:
                    curr_c5 = c5_test
                    best_val = c_clipped
            h[i] = best_val
            if best_val != old:
                any_change = True
        h = project_to_feasible(h, n)
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if not any_change and p > 2:
            break
    return best_h, best_c5

def multi_stage_optimize(h, n, deadline):
    dx = 2.0 / n
    c5 = compute_c5(h, dx)

    # Stage 1: Adam with high temp
    frac = (deadline - time.time())
    iters1 = max(500, int(frac * 0.35))
    dl1 = time.time() + frac * 0.30
    h, c5 = adam_optimize(h, n, iters1, dx, dl1)

    # Stage 2: Coordinate descent
    dl2 = time.time() + (deadline - time.time()) * 0.25
    h, c5 = coord_descent(h, n, 80, dl2)

    # Stage 3: Adam with lower temp for refinement
    if time.time() < deadline - 30:
        frac3 = deadline - time.time()
        iters3 = max(300, int(frac3 * 0.4))
        dl3 = time.time() + frac3 * 0.45
        h, c5_new = adam_optimize(h, n, iters3, dx, dl3)
        if c5_new < c5:
            c5 = c5_new

    # Stage 4: Final coordinate descent polish
    h, c5 = coord_descent(h, n, 60, deadline)

    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s * 0.93

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_c5 = compute_c5(h0, dx0)
    best_h = h0.copy()
    best_n = n0

    # Resolution plan: start at n0, go to 2*n0, maybe 4*n0
    n_list = [n0]
    if 2 * n0 <= 560:
        n_list.append(2 * n0)
    if 4 * n0 <= 560:
        n_list.append(4 * n0)

    # Allocate time: 30% for first, 50% for second, 20% for third
    time_fracs = []
    if len(n_list) == 1:
        time_fracs = [0.9]
    elif len(n_list) == 2:
        time_fracs = [0.3, 0.65]
    else:
        time_fracs = [0.2, 0.5, 0.45]

    for idx, n in enumerate(n_list):
        frac_end = sum(time_fracs[:idx + 1])
        stage_deadline = t0 + (budget_s * 0.93) * min(frac_end, 0.95)

        if n == n0:
            h_curr = h0.copy()
        else:
            h_curr = upsample_periodic(best_h, n)

        h_curr, c5_curr = multi_stage_optimize(h_curr, n, stage_deadline)

        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

    # Random restarts at best resolution if time remains
    remaining = deadline - time.time()
    if remaining > 45:
        n_restart = best_n
        dx_r = 2.0 / n_restart
        n_restarts = min(8, int(remaining // 40))
        for r in range(n_restarts):
            rl = time.time() + min(35, (deadline - time.time()) * 0.6)
            scale = 0.02 + 0.03 * r
            h_pert = project_to_feasible(best_h + rng.randn(n_restart) * scale, n_restart)
            dx_r = 2.0 / n_restart
            h_pert, c5_pert = adam_optimize(h_pert, n_restart, 800, dx_r, rl)
            h_pert, c5_pert = coord_descent(h_pert, n_restart, 30, deadline)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                best_n = n_restart

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
```
