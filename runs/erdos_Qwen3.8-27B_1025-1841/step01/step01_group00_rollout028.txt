            # Random perturbation to escape local optima
            if tau > 100 and np.random.random() < 0.3:
                h = project_to_feasible(h + 0.05 * np.random.randn(n), n)

    # Phase 2: Subgradient polish with decreasing step size
    h_polish = best_h.copy()
    c5_pol = compute_c5(h_polish)
    lr_sg = 0.02
    v_sg = np.zeros(n)
    beta_sg = 0.9
    top_k_sg = 5
    iter_count = 0

    while remaining() > 20:
        iter_count += 1
        grad, c5_cur = subgrad_step(h_polish, top_k_sg)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_polish.copy()

        v_sg = beta_sg * v_sg - lr_sg * grad
        h_polish = project_to_feasible(h_polish + v_sg, n)

        lr_sg *= 0.9995
        if lr_sg < 1e-9:
            break

    # Phase 3: Fine-tune with very small steps on the true max
    h_final = best_h.copy()
    c5_final = compute_c5(h_final)
    lr_fine = 0.005
    v_fine = np.zeros(n)
    for _ in range(5000):
        if remaining() < 10:
            break
        corr = fft_full_correlate(h_final, 1.0 - h_final)
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h_final.copy()
        k_star = np.argmax(corr)
        sh = k_star - (n - 1)
        grad_f = np.zeros(n)
        a0 = max(0, sh)
        a1 = min(n, n + sh)
        if a0 < a1:
            grad_f[a0:a1] = (1.0 - h_final[a0 - sh:a1 - sh])
        b0 = max(0, -sh)
        b1 = min(n, n - sh)
        if b0 < b1:
            grad_f[b0:b1] -= h_final[b0 + sh:b1 + sh]
        grad_f *= (2.0 / n)
        v_fine = 0.9 * v_fine - lr_fine * grad_f
        h_final = project_to_feasible(h_final + v_fine, n)
        lr_fine *= 0.999

    # Final verification with exact computation
    final_c5 = exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    best_h = project_to_feasible(best_h, n)
    final_c5 = exact_c5(best_h)

    return best_h, final_c5, n
