import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            h[mask] += d / np.sum(mask)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            h[mask] += d / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, n, T):
    dx = 2.0 / n
    h_c = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    cvals = corr * dx
    cmax = np.max(cvals)
    e = np.exp((cvals - cmax) / T)
    f = cmax + T * np.log(np.sum(e))
    p = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 
1):
        pk = p[k]
        if pk < 1e-18:
            continue
        s = k - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                grad[s:] += pk * (1.0 - h_c[:L])
                grad[:L] += pk * (-h_c[s:])

            L = n - t
            if L > 0:
                grad[:L] += pk * (1.0 - h_c[t:])
                grad[t:] += pk * (-h_c[:L])
    grad *= dx
    return f, grad
