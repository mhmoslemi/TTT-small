    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr has length 2N-1, indexed 0 to 2N-2
    # corr[k] = sum_n h[n + k - (N-1)] * (1-h[n]) for valid n
    
    # Soft-max weights
    scaled = T * corr * dx
    scaled -= np.max(scaled)  # for numerical stability
    w = np.exp(scaled)
    w /= np.sum(w)

    # dcorr[k]/dh[j] = (1-h[j-tau])*1[valid] - h[j+tau]*1[valid]

    grad = np.zeros(N)
    for k in range(2*N - 1):

        j_start = max(0, tau)

        if j_end > j_start:

            grad[j_start:j_end] += w[k] * (1.0 - h[indices - tau])

        j_start2 = max(0, -tau)

            grad[j_start2:j_end2] -= w[k] * h[indices2 + tau]
    
    return grad * dx, (1.0/T) * np.log(np.sum(np.exp(T * corr * dx)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, N):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = N / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / N
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_N):
    old_N = len(h)
    if new_N == old_N:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_N, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_N, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_N)

def softmax_obj_grad(h, T, M=50000.0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    obj = (max_s + np.log(sum_exp)) / T
    sum_h = np.sum(h)
    penalty = M * (sum_h - N / 2.0) ** 2
    total_obj = obj + penalty
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        if w[k] < 1e-15:
            continue
        tau = k - (N - 1)
        j_start = max(0, tau)
        j_end = min(N, N + tau)
        if j_end > j_start:
            idx = np.arange(j_start, j_end)
            grad[j_start:j_end] += w[k] * (1.0 - h[idx - tau])
        j_start2 = max(0, -tau)
        j_end2 = min(N, N - tau)
        if j_end2 > j_start2:
            idx2 = np.arange(j_start2, j_end2)
            grad[j_start2:j_end2] -= w[k] * h[idx2 + tau]
    grad *= dx
    grad += 2.0 * M * (sum_h - N / 2.0)
    return total_obj, grad

def optimize_lbfgsb(h_start, T_val, M_val, max_iter, time_limit):
    N = len(h_start)
    t0 = time.time()
    def obj_wrapper(x):
        return softmax_obj_grad(x, T_val, M_val)[0]
    def grad_wrapper(x):
        return softmax_obj_grad(x, T_val, M_val)[1]
    bounds = [(0.0, 1.0)] * N
    res = minimize(softmax_obj_grad, h_start, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14,
                   'gtol': 1e-8, 'maxls': 50, 'maxfun': max_iter * 3})
    if time.time() - t0 > time_limit:
        return h_start, compute_c5(h_start)
    x_opt = make_feasible(res.x, N)
    return x_opt, compute_c5(x_opt)

    current_c5 = compute_c5(current_h)

        if time.time() - t0 > time_limit:

        max_lag = np.argmax(corr)
        target = corr[max_lag] * dx - 0.001
        T = 80.0
        M = 50000.0
        def cp_obj(x):

            excess = max(0.0, c[max_lag] * dx - target)
            pen2 = 100.0 * excess ** 2
            return sm + pen + pen2
        def cp_grad(x):

            max_s = np.max(scaled)

def coordinate_polish(h, N, max_rounds=10, time_limit=60.0):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for rnd in range(max_rounds):

        improved = False
        for j in range(N):
            if time.time() - t0 > time_limit:

            old = best_h[j]

                    best_h = trial
                    improved = True
