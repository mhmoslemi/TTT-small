def project_to_sum(h, target):
    """Project h to have exact sum = target, keeping in [0,1]"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        # Distribute adjustment evenly, but respect bounds
        mask_up = h < 1.0 - 1e-12
        mask_dn = h > 1e-12
        if diff > 0:
            n_up = mask_up.sum()
            if n_up == 0:
                break
            h[mask_up] += diff / n_up
        else:
            n_dn = mask_dn.sum()
            if n_dn == 0:
                break
            h[mask_dn] += diff / n_dn
        h = np.clip(h, 0.0, 1.0)

    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-12:

            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)

    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)

    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)

    """Local search on binary h via pair swaps"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]

        delta = c5_new - best_c5

            # Update indices

            zeros_idx = np.where(h == 0.0)[0]
        else:
            # Undo
            h[i] = 0.0
            h[j] = 1.0
        
        T *= cooling
    
    return h, best_c5

    if temp < 1e-10:
        return m
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def grad_smooth_max(h, dx, temp):

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

            for j in range(n - s_abs):
                grad[j] += -h[j + s_abs]  # h[j+s_abs] term

                for j in range(m_valid):
                    g[j] -= h[j + s_abs]

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        n_m = mask.sum()
        if n_m == 0:
            break
        h[mask] += diff / n_m
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-10:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_to_sum(h_new, n_new / 2.0)

def to_binary(h, n):
    target = n // 2
    if n % 2 == 0:
        sorted_idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[sorted_idx[:target]] = 1.0
        return h_bin
    else:
        # Odd n: use n//2 ones
        sorted_idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[sorted_idx[:target]] = 1.0
        return h_bin

def sa_binary(h, n, n_iter, seed, T_start, T_end, time_limit):
    """Simulated annealing on binary h"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    ones_idx = list(np.where(h == 1.0)[0])
    zeros_idx = list(np.where(h == 0.0)[0])
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr)) * dx
    curr_c5 = best_c5
    t0 = time.time()
    
    T = T_start
    n_total = n_iter
    cool_rate = (T_end / T_start) ** (1.0 / max(n_total, 1))
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Pick random swap
        pi = rng.randint(len(zeros_idx))
        pj = rng.randint(len(ones_idx))
        i = zeros_idx[pi]
        j = ones_idx[pj]
        
        # Perform swap
        h[i] = 1.0
        h[j] = 0.0
        zeros_idx[pi] = j
        ones_idx[pj] = i
        
        corr_new = np.correlate(h, 1.0 - h, mode="full")
        c5_new = float(np.max(corr_new)) * dx
        
        delta = c5_new - curr_c5
        if delta < 0 or (T > 1e-8 and rng.rand() < np.exp(max(-delta / T, -50.0))):
            curr_c5 = c5_new
            corr = corr_new
            if c5_new < best_c5:
                best_c5 = c5_new
        else:
            # Undo
            h[i] = 0.0
            h[j] = 1.0
            zeros_idx[pi] = i
            ones_idx[pj] = j
        
        T *= cool_rate
    
    return h, best_c5

def continuous_refine(h, n, time_limit, seed):
    """Refine using projected gradient on smooth max"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n / 2.0
    h = project_to_sum(h.copy(), target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr)) * dx
    t0 = time.time()
    
    # Projected gradient with adaptive step
    lr = 0.01
    for it in range(5000):
        if time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        
        # Compute subgradient
        grad = np.zeros(n)
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
        else:
            sa = -s
            grad[sa:] += (1.0 - h[:n-sa])
            grad[:n-sa] -= h[sa:]
        grad *= dx
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.9
            if lr < 1e-6:
                break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        nm = mask.sum()
        if nm == 0:
            break
        h[mask] += diff / nm
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-10:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_to_sum(h_new, n_new / 2.0)

def to_binary(h, n):
    target = n // 2
    if n % 2 != 0:
        target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def sa_binary(h, n, n_iter, seed, T_start, T_end, time_limit):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    ones_idx = list(np.where(h == 1.0)[0])
    zeros_idx = list(np.where(h == 0.0)[0])
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr)) * dx
    curr_c5 = best_c5
    t0 = time.time()
    cool_rate = (max(T_end, 1e-8) / T_start) ** (1.0 / max(n_iter, 10))
    T = T_start
    stall = 0
    best_stall_reset = 0

    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        pi = rng.randint(len(zeros_idx))
        pj = rng.randint(len(ones_idx))
        i = zeros_idx[pi]
        j = ones_idx[pj]

        h[i] = 1.0
        h[j] = 0.0
        zeros_idx[pi] = j
        ones_idx[pj] = i

        corr_new = np.correlate(h, 1.0 - h, mode="full")
        c5_new = float(np.max(corr_new)) * dx

        delta = c5_new - curr_c5
        if delta < -1e-14 or (T > 1e-7 and rng.rand() < np.exp(max(-delta / max(T, 1e-10), -60.0))):
            curr_c5 = c5_new
            corr = corr_new
            if c5_new < best_c5:
                best_c5 = c5_new
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 0.0
            h[j] = 1.0
            zeros_idx[pi] = i
            ones_idx[pj] = j
            stall += 1

        T *= cool_rate
        if stall > 5000 and T > 0.005:
            T = min(T * 1.5, 0.05)
            stall = 0

    return h, best_c5

def projected_grad_descent(h, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n / 2.0
    h = project_to_sum(h.copy(), target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr)) * dx
    t0 = time.time()
    lr = 0.05
    patience = 0

    for it in range(20000):
        if time.time() - t0 > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            sa = -s
            grad[sa:] += (1.0 - h[:n - sa])
            grad[:n - sa] -= h[sa:]
        grad *= dx

        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new
            patience = 0
            lr = min(lr * 1.02, 0.2)
        else:
            patience += 1
            lr *= 0.95
            if patience > 500:
                break

    # Push toward binary
    h_binary = to_binary(h, n)
    c5_bin = compute_c5(h_binary, dx)
    if c5_bin < best_c5:
        return h_binary, c5_bin
    return h, best_c5

def slsqp_smooth(h, n, temp, maxiter):
    dx = 2.0 / n
    target = n / 2.0

    def obj_fn(x):
        hx = np.clip(x, 0.0, 1.0)
        cr = np.correlate(hx, 1.0 - hx, mode="full")
        m = np.max(cr)
        ls = np.log(np.sum(np.exp((cr - m) / temp)))
        return (m + temp * ls) * dx

    def con_fun(x):
        return np.sum(x) - target

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_fn, h, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9})
    if np.isfinite(res.fun):
        return project_to_sum(res.x, target)
    return h
