def slsqp_optimize(h_init, n, max_iter, t0, deadline):
    def obj(x):
        if time.time() > deadline:
            return 1.0
        return compute_c5(x)
    def con(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h_init, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        hx = project_feasible(res.x, n)
        return hx, compute_c5(hx)
    except Exception:
        return h_init.copy(), compute_c5(h_init)

def try_binary(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[order[:target]] = 1.0
    return hb, compute_c5(hb)
