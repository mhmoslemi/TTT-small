    """Compute F matrix: F[p, s_idx] where s_idx = s + (n-1)
    F[p, s] = h[p-s]*I(0≤p-s<n) + (h[p+s]-1)*I(0≤p+s<n)"""
    F = np.zeros((n, 2*n - 1))
    for p in range(n):
        # First term: h[p-s] for s where 0<=p-s<n → p-(n-1)<=s<=p
        s_vals = np.arange(max(-(n-1), -p), min(n-1, n-1-p) + 1)
        # Wait, 0<=p-s<n means p-n+1<=s<=p, intersected with [-(n-1), n-1]
        s_lo = max(-(n-1), p - (n-1))
        s_hi = min(n-1, p)
        if s_lo <= s_hi:
            s_arr = np.arange(s_lo, s_hi + 1)
            F[p, s_arr + (n-1)] += h[p - s_arr]
        
        # Second term: (h[p+s]-1) for s where 0<=p+s<n → -p<=s<=n-1-p
        s_lo2 = max(-(n-1), -p)
        s_hi2 = min(n-1, n-1-p)
        if s_lo2 <= s_hi2:
            s_arr2 = np.arange(s_lo2, s_hi2 + 1)
            F[p, s_arr2 + (n-1)] += h[p + s_arr2] - 1.0
    
    return F

def update_F_entry(F, p, s_idx, delta):
    """Update F[p, s_idx] by adding delta."""
    F[p, s_idx] += delta

def ada_optimize(h, n, n_iter, T_start, T_end, lr, rng):
    """Adam optimization with smooth max and temperature annealing."""
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for i in range(n_iter):
        t += 1
        frac = i / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # Compute smooth max gradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_max = np.max(corr)
        shifted = (corr - m_max) / T
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        weights_norm = weights / wsum
        
        grad = np.zeros(n)
        for k in range(2*n - 1):
            s = k - (n - 1)
            w = weights_norm[k]
            if w < 1e-12:
                continue
            # d(corr_s)/d(h[j]) = (1-h[j+s])*I(0≤j+s<n) - h[j-s]*I(0≤j-s<n)
            lo1 = max(0, -s)
            hi1 = min(n, n - s)
            if lo1 < hi1:
                grad[lo1:hi1] += w * (1.0 - h[lo1+s:hi1+s])
            lo2 = max(0, s)
            hi2 = min(n, n + s)
            if lo2 < hi2:
                grad[lo2:hi2] -= w * h[lo2-s:hi2-s]
        
        # Adam step
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad**2
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_target(h_new, n)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def binary_sa(h, n, max_iter, T_sa, rng):
    """Simulated annealing on binary configurations."""
    dx = 2.0 / n
    # Round to binary
    h_b = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_b))
    target_ones = n // 2
    if n_ones > target_ones:
        ones_idx = np.where(h_b == 1)[0]
        for idx in ones_idx[:n_ones - target_ones]:
            h_b[idx] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_b == 0)[0]
        for idx in zeros_idx[:target_ones - n_ones]:
            h_b[idx] = 1
    
    corr = get_corr(h_b)
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    F = compute_F(h_b, n)
    
    for it in range(max_iter):
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute new max
        delta = F[i] - F[j]
        new_corr = corr + delta
        new_max = float(np.max(new_corr) * dx)
        
        # Accept?
        if new_max < best_c5 or (it < max_iter // 2 and rng.random() < np.exp(-(new_max - best_c5) / T_sa)):
            # Accept
            h_b[i] = 0
            h_b[j] = 1
            corr = new_corr
            
            # Update F
            old_hi, old_hj = 1.0, 0.0
            # F changes for positions affected by i going 1→0 and j going 0→1
            # For position p: F[p, s] = h[p-s]*I(...) + (h[p+s]-1)*I(...)
            # h[i] changed from 1 to 0: affects F[p, s] where p-s=i or p+s=i
            #   p-s=i → s=p-i: F[p, p-i] -= 1 (term h[p-s] changes)
            #   p+s=i → s=i-p: F[p, i-p] -= 1 (term h[p+s]-1 changes: was 0, now -1)
            # Wait: (h[p+s]-1) when p+s=i: was (1-1)=0, now (0-1)=-1. Change: -1. ✓
            # h[j] changed from 0 to 1: affects F[p, s] where p-s=j or p+s=j
            #   p-s=j → s=p-j: F[p, p-j] += 1 (term h[p-s] changes from 0 to 1)
            #   p+s=j → s=j-p: F[p, j-p] += 1 (term h[p+s]-1 changes from -1 to 0)
            
            for p in range(n):
                # Effect of h[i]: 1→0
                s1 = p - i  # for term h[p-s]
                if -(n-1) <= s1 <= n-1:
                    F[p, s1 + (n-1)] -= 1.0
                s2 = i - p  # for term h[p+s]-1
                if -(n-1) <= s2 <= n-1:
                    F[p, s2 + (n-1)] -= 1.0
                # Effect of h[j]: 0→1
                s3 = p - j
                if -(n-1) <= s3 <= n-1:
                    F[p, s3 + (n-1)] += 1.0
                s4 = j - p
                if -(n-1) <= s4 <= n-1:
                    F[p, s4 + (n-1)] += 1.0
            
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h_b.copy()
    
    return best_h, best_c5
