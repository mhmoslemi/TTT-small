def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)

    n = len(h)
    dx = 2.0 / n
    hc = np.clip(h, 0.0, 1.0)

    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(_G_T * (corr - mc)))) / _G_T) * dx

    mc = np.max(corr)

            jb = np.arange(s, n)
            grad[jb] -= wk * hc[jb - s]
        else:
            ja = np.arange(-s, n)
            grad[ja] += wk * g[ja + s]
            jb = np.arange(0, n + s)
            grad[jb] -= wk * hc[jb - s]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

_G_T = 40.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _smoobj(h):
    n = len(h)
    dx = 2.0 / n
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(_G_T * (corr - mc)))) / _G_T) * dx

def _smograd(h):
    n = len(h)
    dx = 2.0 / n
    hc = np.clip(h, 0.0, 1.0)
    g = 1.0 - hc
    corr = np.correlate(hc, g, mode='full')
    mc = np.max(corr)
    exp_c = np.exp(_G_T * (corr - mc))
    w = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-13:
            continue
        s = idx - (n - 1)
        if s >= 0:
            ja = np.arange(0, n - s)
            grad[ja] += wk * g[ja + s]
            jb = np.arange(s, n)
            grad[jb] -= wk * hc[jb - s]
        else:
            ja = np.arange(-s, n)
            grad[ja] += wk * g[ja + s]
            jb = np.arange(0, n + s)
            grad[jb] -= wk * hc[jb - s]
    return grad * dx

def _penobj(h):
    return _smoobj(h) + 500.0 * (np.sum(h) - len(h) / 2.0) ** 2

def _penngrad(h):
    sg = _smograd(h)
    pen = 1000.0 * (np.sum(h) - len(h) / 2.0)
    return sg + pen * np.ones(len(h))

def random_binary_h(n, rng):
    indices = np.arange(n)
    rng.shuffle(indices)
    h = np.zeros(n)
    h[indices[:n // 2]] = 1.0
    return h

def _run_lbfgs(h_start, n, maxiter, maxfun):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_penobj, h_start, jac=_penngrad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                           'gtol': 1e-8, 'maxfun': maxfun})
    hx = np.clip(res.x, 0.0, 1.0)
    hx = make_feasible(hx, n)
    return hx

def _coord_polish(h, n, max_passes, max_time):
    best_c = compute_c5(h)
    start = time.time()
    for _pass in range(max_passes):
        if time.time() - start > max_time:
            break
        improved = False
        for j in range(n):
            if time.time() - start > max_time:
                break
            if time.time() > max_time:
                break
            ov = h[j]
            candidates = [0.0, 1.0]
            if 0 < ov < 1:
                candidates += [ov + 0.02, ov - 0.02, ov + 0.05, ov - 0.05,
                               ov + 0.01, ov - 0.01]
            for cand in candidates:
                if cand < -1e-9 or cand > 1 + 1e-9:
                    continue
                trial = h.copy()
                trial[j] = np.clip(cand, 0.0, 1.0)
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-11:
                    best_c = c
                    h = trial
                    improved = True
                    break
    return h, best_c
