    corr = np.correlate(h, 1.0 - h, mode='full') * dx

            h += diff / n
            break
        h[free] += diff / float(free.sum())
    h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible(n, rng):
    h = rng.uniform(0, 1, n)
    freq = rng.integers(1, 7)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    h = 0.4 * rng.uniform(0, 1, n) + 0.6 * h
    return project_feasible(h, n)

def binary_like_feasible(n, rng):
    h = (rng.random(n) < 0.5).astype(float)
    h += rng.normal(0, 0.15, n)
    return project_feasible(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def soft_max_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    if T < 1e7:
        shifted = T * (corr - mc)
        shifted = shifted - np.max(shifted)
        return mc + float(np.log(np.sum(np.exp(shifted))) / T)
    return mc

def soft_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    if T < 1e7:
        shifted = T * (corr - mc)
        shifted = shifted - np.max(shifted)
        exp_s = np.exp(shifted)
        weights = exp_s / np.sum(exp_s)
    else:
        weights = np.zeros(2 * n - 1)
        active = corr >= mc - 1e-10
        weights[active] = 1.0 / np.sum(active)

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        q = abs(s)
        if s >= 0:
            if q == 0:
                grad += w * dx * (1.0 - 2.0 * h)
            else:
                m1 = np.arange(s, n)
                grad[m1] += w * dx * (1.0 - h[m1 - s])
                m2 = np.arange(0, n - q)
                grad[m2] -= w * dx * h[m2 + s]
        else:
            m1 = np.arange(0, n - q)
            grad[m1] += w * dx * (1.0 - h[m1 + q])
            m2 = np.arange(q, n)
            grad[m2] -= w * dx * h[m2 - q]
    return grad

def _obj_wrapper(h):
    return soft_max_obj(h, _n_global, _T_global)

def _grad_wrapper(h):
    return soft_max_grad(h, _n_global, _T_global)

_T_global = 50.0

def optimize_single(h0, n, T_schedule, max_iter_per_stage):
    global _n_global, _T_global
    _n_global = n
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_f = compute_c5(h, n)

    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}

    for T in T_schedule:
        _T_global = T
        try:
            res = minimize(
                _obj_wrapper, h,
                jac=_grad_wrapper,
                method='SLSQP',
                bounds=bounds,
                constraints=[constraint],
                options={'maxiter': max_iter_per_stage, 'ftol': 1e-12, 'disp': False}
            )
        except Exception:
            break
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            f_new = compute_c5(h_new, n)
            if f_new < best_f:
                best_f = f_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_f

def upsample(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = 0

    try:
        h_init = initial_h_values.copy()
        n_init = len(h_init)
    except NameError:
        h_init = None
        n_init = 0

    if h_init is None:
        n_init = 80
        h_init = random_feasible(n_init, rng)

    resolutions = sorted(set([n_init] + [80, 120, 160, 200, 240, 320]))
    resolutions = [r for r in resolutions if r % 2 == 0 and r <= 480]

    T_sched_full = [20, 50, 100, 250, 500, 1000]
    T_sched_short = [50, 150, 400, 1000]

    for n in resolutions:
        if time.time() - t0 > budget_s * 0.82:
            break
        remaining = budget_s * 0.82 - (time.time() - t0)
        n_res_remaining = len([r for r in resolutions if r >= n])
        time_per_start = max(15, min(120, remaining / max(1, n_res_remaining) / 2.5))

        n_starts = 3 if n <= 160 else 2
        for si in range(n_starts):
            if time.time() - t0 > budget_s * 0.85:
                break
            if time.time() - t0 > t0 + budget_s - 30:
                break

            if si == 0 and n == n_init:
                h0 = h_init.copy()
            elif si == 1:
                if n == n_init:
                    h0 = h_init + rng.normal(0, 0.08, n)
                    h0 = project_feasible(h0, n)
                else:
                    h0 = upsample(h_init, n_init, n) + rng.normal(0, 0.05, n)
                    h0 = project_feasible(h0, n)
            else:
                if si == 2 and n <= 200:
                    h0 = binary_like_feasible(n, rng)
                else:
                    h0 = random_feasible(n, rng)

            sched = T_sched_full if n <= 160 else T_sched_short
            h_opt, c5_val = optimize_single(h0, n, sched, 120)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

    # Final high-res polish
    if best_h is not None and time.time() - t0 < budget_s * 0.7:
        for factor in [2, 3]:
            n_h = best_n * factor
            if n_h > 600:
                break
            if time.time() - t0 > budget_s * 0.78:
                break
            h_h = upsample(best_h, best_n, n_h)
            h_h = project_feasible(h_h, n_h)
            h_opt, c5_val = optimize_single(h_h, n_h, T_sched_short, 80)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n_h

    if best_h is None:
        n = n_init
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, n)
        best_n = n

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
