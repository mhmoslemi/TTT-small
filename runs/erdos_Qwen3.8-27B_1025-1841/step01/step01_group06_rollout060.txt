```python
def project_onto_simplex_box(x, n, target):
    """Project x onto {h: 0<=h_i<=1, sum(h)=target}"""
    h = np.clip(x, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    
    # Need to find lambda: sum(clip(h_i - lambda, 0, 1)) = target
    # If s > target: need to decrease, so lambda > 0
    # If s < target: need to increase, so lambda < 0
    
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_test = np.clip(h - mid, 0.0, 1.0)
        s_test = np.sum(h_test)
        if s_test > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    
    # Fix any small numerical issues
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
    corr = np.correlate(h, 1.0 - h, mode="full")

def smooth_max_val(corr, tau):
    m = np.max(corr)
    return m + tau * np.log(np.sum(np.exp((corr - m) / tau)))

def compute_sm_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = (corr - np.max(corr)) / tau
    w = np.exp(shifted)
    w /= np.sum(w)
    # Gradient via convolutions
    term1 = np.convolve(1.0 - h, w, mode='full')[n-1:2*n-1]
    term2 = np.correlate(h, w, mode='full')[n-1::-1]
    return term1 - term2, corr

def project_h(x, target):

    s = np.sum(h)
```
