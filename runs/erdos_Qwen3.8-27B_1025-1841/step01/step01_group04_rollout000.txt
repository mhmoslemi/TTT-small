def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import zoom
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def soft_max_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    w = np.exp(scaled - m) / np.sum(np.exp(scaled - m))
    N_fft = 1
    total_len = n + 2 * n - 2
    while N_fft < total_len:
        N_fft *= 2
    W = np.fft.rfft(w, N_fft)
    X = np.fft.rfft(x, N_fft)
    OX = np.fft.rfft(1.0 - x, N_fft)
    conv1 = np.fft.irfft(OX * W, N_fft)
    conv2 = np.fft.irfft(X * W, N_fft)
    s1 = conv1[n - 1:n - 1 + n]
    s2 = conv2[n - 1:n - 1 + n]
    return dx * (s1 - s2)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    return zoom(h, n_to / n_from, order=3)

def slsqp_opt(h0, n, T, maxiter, deadline, ftol=1e-14):
    target = n / 2.0

    def f(x):
        return soft_max_obj(x, n, T)

    def fgrad(x):
        return soft_max_grad(x, n, T)

    def c_eq(x):
        return np.sum(x) - target

    def c_eq_grad(x):
        return np.ones(n)

    try:
        res = minimize(f, h0, jac=fgrad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_grad}],
                       options={'ftol': ftol, 'maxiter': maxiter, 'disp': False})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def proj_momentum(h0, n, T, lr, n_steps, momentum, deadline):
    h = h0.copy()
    v = np.zeros(n)
    target = n / 2.0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = soft_max_grad(h, n, T)
        g = g - np.mean(g)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project(h_new, n)
        obj_new = soft_max_obj(h_new, n, T)
        if not np.isfinite(obj_new):
            h = project(h - lr * g, n)
        else:
            h = h_new
    return h

        h_new = slsqp_opt(best_h, n_best, T, maxiter=800, deadline=deadline, ftol=1e-15)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    rng = np.random.RandomState(seed + 7)
    for attempt in range(40):
        if time.time() > deadline - 30:
            break
        scale = 0.03 + 0.04 * (attempt % 6) / 6.0
        h_pert = best_h + rng.randn(n_best) * scale
        h_pert = project(h_pert, n_best)
        T_use = 500.0 if attempt % 3 == 0 else 1500.0
        h_new = slsqp_opt(h_pert, n_best, T_use, maxiter=250, deadline=deadline, ftol=1e-14)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    for attempt in range(10):
        if time.time() > deadline - 20:
            break
        x = np.linspace(0, 2, n_best, endpoint=False)
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(1, 5)
        h_struct = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x + phase)
        h_struct = project(h_struct, n_best)
        h_new = slsqp_opt(h_struct, n_best, 300.0, maxiter=300, deadline=deadline, ftol=1e-14)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time.time() < deadline - 30:
        h_final = slsqp_opt(best_h, n_best, 8000.0, maxiter=1000, deadline=deadline, ftol=1e-15)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
