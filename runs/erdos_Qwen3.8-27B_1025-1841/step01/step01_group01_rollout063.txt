    return dx * np.max(corr), corr

def smooth_max_val_and_weights(vals, tau):
    vs = vals / tau
    vmax = np.max(vs)
    ev = np.exp(vs - vmax)
    se = np.sum(ev)
    obj = tau * (vmax + np.log(se))
    w = ev / se
    return obj, w

    sm, w = smooth_max_val_and_weights(vals, tau)

    s = np.sum(h)
    pen = penalty * (s - target_sum) ** 2
