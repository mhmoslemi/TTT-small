import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode='full')
    return (c * dx).max()

def project_feasible(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase sum; add to entries with room
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        add = diff / mask.sum()
        h[mask] = np.clip(h[mask] + add, 0, 1)
    else:
        # Need to decrease sum
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        sub = (-diff) / mask.sum()
        h[mask] = np.clip(h[mask] - sub, 0, 1)
    # Fix remaining
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Fine adjustment
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            for i in idx:
                amt = min(diff, 1.0 - h[i])
                h[i] += amt
                diff -= amt
                if abs(diff) < 1e-12:
                    break
        else:
            idx = np.where(h > 0.0)[0]
            for i in idx:
                amt = min(-diff, h[i])
                h[i] -= amt
                diff += amt
                if abs(diff) < 1e-12:
                    break
    return h

def logsumexp_c5_and_grad(h, n_points, eps):
    dx = 2.0 / n_points
    c_all = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # Compute log-sum-exp of c_all
    max_c = c_all.max()
    shifted = (c_all - max_c) / eps
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    weights = exp_shifted / total  # softmax weights
    
    # Objective: dx * (max_c + eps * log(total))
    obj = dx * (max_c + eps * np.log(total))
    
    # Gradient
    # For shift s (0 to n-1):
    # w_s = weights[n-1+s] + weights[n-1-s] (for s>0), or weights[n-1] for s=0
    n_half = n_points - 1

        if s > 0:
            w_s += weights[n_half - s]
        if w_s < 1e-15:
            continue
        grad[s:] += w_s * (1.0 - h[:n_points - s])
        grad[:n_points - s] -= w_s * h[s:]

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode='full')
    return float((c * dx).max())

def project_feasible(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        amt = diff / mask.sum()
        h[mask] = np.clip(h[mask] + amt, 0.0, 1.0)
    return h

def lse_grad(h, n_points, eps):
    dx = 2.0 / n_points
    c_all = np.correlate(h, 1.0 - h, mode='full')
    max_c = c_all.max()
    shifted = (c_all - max_c) / eps
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    w = exp_vals / total
    obj = dx * (max_c + eps * np.log(total))
    nh = n_points - 1
    grad = np.zeros(n_points)
    for s in range(n_points):
        ws = w[nh + s]
        if s > 0:
            ws += w[nh - s]
        if ws < 1e-14:
            continue
        m = n_points - s
        grad[s:] += ws * (1.0 - h[:m])
        grad[:m] -= ws * h[s:]
    grad *= dx / eps
    return obj, grad

    np.clip(idx, 0, n1 - 1, out=idx)
    return h[idx].copy()
