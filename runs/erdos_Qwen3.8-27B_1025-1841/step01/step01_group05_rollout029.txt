import numpy as np
import time

def evaluate_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def make_binary(h, n):
    """Round h to binary with exactly n/2 ones"""
    h_b = (h > 0.5).astype(float)
    ones = int(np.sum(h_b))
    target = n // 2
    if ones > target:
        idx = np.where(h_b == 1)[0]
        # Remove the ones with smallest original h values
        order = np.argsort(h[idx])[:ones - target]
        h_b[idx[order]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])[:target - ones]
        h_b[idx[order]] = 1
    return h_b

def scale_up(h, n_old, n_new):
    """Scale h from n_old to n_new by nearest-neighbor interpolation"""
    # Each old point maps to n_new/n_new old positions
    ratio = n_new / n_old
    indices = (np.arange(n_new) * ratio).astype(int)
    indices = np.clip(indices, 0, n_old - 1)
    h_new = h[indices].copy()
    # Round to binary and fix count
    return make_binary(h_new, n_new)

def coordinate_descent(h, n, dx, rng, time_limit, max_passes=100000, sa_temp=0.0, sa_cool=0.999):
    """Coordinate descent with optional simulated annealing"""
    h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    current_c5 = best_c5
    temp = sa_temp
    n_accepted = 0
    n_rejected = 0
    n_improved = 0
    
    for iteration in range(max_passes):
        if time.time() > time_limit:
            break
        
        # Pick random coordinate
        i = rng.randint(n)
        
        # Try the other binary value
        new_val = 1.0 - h[i]  # flip
        if h[i] == new_val:
            continue
            
        old_val = h[i]
        h[i] = new_val
        
        # Check constraint: we need exactly n/2 ones
        # Flipping changes the count by ±1
        # We need to flip another coordinate to compensate
        # This is a "swap" move: change h[i] and h[j]
        
        # Actually, let's do a swap: pick i and j, swap their values
        # If h[i]=1, h[j]=0: after swap h[i]=0, h[j]=1
        # This preserves the count
        
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            h[i] = old_val
            continue
        
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        
        new_c5 = evaluate_c5(h, dx)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            n_improved += 1
            if new_c5 < best_c5:
                best_c5 = new_c5
            # Accept
        else:
            # Simulated annealing: accept with probability exp(-delta/temp)
            if temp > 1e-10:
                delta = (new_c5 - current_c5) / dx  # in "unit" scale
                if rng.random() < np.exp(-delta / temp):
                    current_c5 = new_c5
                    n_accepted += 1
                else:
                    # Reject: undo
                    h[i] = 1.0 - h[i]
                    h[j] = 1.0 - h[j]
                    n_rejected += 1
            else:
                # Reject: undo
                h[i] = 1.0 - h[i]
                h[j] = 1.0 - h[j]
        
        temp *= sa_cool
    
    return h, best_c5
