def grad_correlation(h, n, lag):
    """Gradient of C[lag] with respect to h.
    C[lag] = sum_{valid} h[i] * (1-h[i-lag])
    where lag = k - (n-1) for the k-th output element.
    """
    grad = np.zeros(n)
    if lag >= 0:
        # C[lag] = sum_{j=0}^{n-1-lag} h[j+lag] * (1-h[j])

    n = len(h)
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14

        if cnt == 0:
            break
        step = d / cnt
        # Don't overshoot bounds

def proj(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(n + 10):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:

        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14

            break
        step = d / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    # Exact fix

    if abs(d) > 1e-12:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if d > 0:

            else:
                a = max(d, -h[i])
            h[i] += a

        h = np.clip(h, 0.0, 1.0)

def proj(h, target):

        else:

            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    d = target - h.sum()

        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)

            h[i] += a
            d -= a
            if abs(d) < 1e-14:
