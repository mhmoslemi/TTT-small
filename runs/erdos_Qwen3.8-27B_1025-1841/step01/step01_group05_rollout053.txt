```python
import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    complement = 1 - h
    corr = np.correlate(h, complement, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    complement = 1 - h
    corr = np.correlate(h, complement, mode='full')
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        # h[i] appears as first factor with partner h[i+shift]
        j1 = i + shift
        if 0 <= j1 < n:
            grad[i] += (1 - h[j1]) - h[i]
        # h[i] appears as second factor (1-h[i]) where h[i-shift] is first
        # contributing -h[i-shift] to d/dh[i]
        j2 = i - shift
        if 0 <= j2 < n:
            grad[i] += -h[j2]
    return dx * grad, k_star

def project(h, n):
    """Project to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        err = s - target
        if abs(err) < 1e-12:
            break
        if err > 0:
            # Need to decrease sum by err
            mask = h > 0  # can decrease
            if not np.any(mask):
                break
            h[mask] -= err / np.sum(mask)
        else:
            # Need to increase sum by -err
            mask = h < 1  # can increase
            if not np.any(mask):
                break
            h[mask] += (-err) / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h
```

```python
def project_to_constraint(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Binary search for lambda
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h - mid, 0, 1)
        s_test = np.sum(h_test)
        if s_test > target:
            lo = mid
        else:
            hi = mid
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(np.sum(h))
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        ht = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(ht)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def interp_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(xn, xo, h_old)

def topk_grad(h, n, t, k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = t * corr
    ms = np.max(scaled)
    exps = np.exp(scaled - ms)
    w = exps / np.sum(exps)
    
    if k >= len(w):
        idxs = np.arange(len(w))
        wt = w.copy()
    else:
        idxs = np.argsort(w)[::-1][:k]
        wt = w[idxs].copy()
        wt /= wt.sum()
    
    i_arr = np.arange(n)
    grad = np.zeros(n)
    for j in range(len(idxs)):
        s = int(idxs[j]) - (n - 1)
        v1 = i_arr + s
        v2 = i_arr - s
        m1 = (v1 >= 0) & (v1 < n)
        m2 = (v2 >= 0) & (v2 < n)
        if np.any(m1):
            grad[m1] += wt[j] * (1.0 - h[v1[m1]])
        if np.any(m2):
            grad[m2] -= wt[j] * h[v2[m2]]
    return dx * grad

def optimize_level(h0, n, max_time, rng):
    t0 = time.time()
    h = project_constraint(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    step = 0.1
    t_sm = 3.0
    no_improve = 0
    k_top = 30
    
    while time.time() - t0 < max_time:
        grad = topk_grad(h, n, t_sm, k_top)
        sum_err = float(np.sum(h)) - n / 2.0
        grad += 5.0 * sum_err
        
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        
        h_new = h - step * grad
        h_new = project_constraint(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.02, 1.0)
            no_improve = 0
        else:
            step *= 0.92
            no_improve += 1
        
        if no_improve > 150:
            scale = min(0.1, 0.02 * (1 + 0.5 * no_improve / 150))
            h = h + rng.normal(0, scale, n)
            h = project_constraint(h, n)
            step = 0.1
            t_sm = min(t_sm * 1.5, 50.0)
            no_improve = 0
            k_top = min(k_top + 10, 80)
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 42
    
    if h_parent is None:
        h_parent = np.ones(n_parent) * 0.5
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    
    # Build resolution levels
    levels = [n_parent]
    cur = n_parent
    max_n = 2000
    while cur < max_n:
        nxt = min(cur * 2, max_n)
        if nxt == cur:
            break
        levels.append(nxt)
        cur = nxt
    
    # Time allocation proportional to sqrt(n)
    usable = budget_s - 30
    weights = [np.sqrt(float(n)) for n in levels]
    tw = sum(weights)
    allocs = [usable * w / tw for w in weights]
    
    current = best_h.copy()
    for i, n in enumerate(levels):
        elapsed = time.time() - t_start
        remaining = budget_s - 30 - elapsed
        if remaining < 15:
            break
        t_alloc = min(allocs[i], remaining)
        
        h_init = interp_to(current, n)
        h_init = project_constraint(h_init, n)
        
        h_opt, c5_val = optimize_level(h_init, n, t_alloc, rng)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
        
        current = h_opt.copy()
    
    n_best = len(best_h)
    best_h = project_constraint(best_h, n_best)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_best
```
