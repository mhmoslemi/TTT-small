from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(2000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) * T
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp == 0:
        return m, np.zeros(n)
    weights = exp_shifted / sum_exp
    value = m + np.log(sum_exp) / T
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        l = k - (n - 1)
        if l >= 0:
            if n - l > 0:
                grad[:n-l] += w * (1.0 - h[l:])
                grad[l:] -= w * h[:n-l]
        else:
            m2 = -l
            if n - m2 > 0:
                grad[m2:] += w * (1.0 - h[:n-m2])
                grad[:n-m2] -= w * h[m2:]
    
    grad *= dx
    return value, grad

def pgd_solve(h0, n, T_schedule, max_iter, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T_schedule[min(it, len(T_schedule)-1)] if T_schedule else 100.0
        if isinstance(T_schedule, float):
            T = T_schedule
        
        val, grad = compute_smooth_max_grad(h, n, T)
        
        # Backtracking line search
        lr = 1.0
        for _ in range(50):
            h_new = h - lr * grad
            h_new = project(h_new, n)
            val_new = evaluate_c5(h_new)
            if val_new < val or it == 0:
                break
            lr *= 0.5
            if lr < 1e-8:
                h_new = h
                val_new = val
                break
        
        h = h_new
        if val_new < best_val:
            best_val = val_new
            best_h = h.copy()
    
    return best_h, best_val

def perturb(h, n, sigma, rng):
    h_new = h + rng.randn(n) * sigma
    return project(h_new, n)

def block_swap(h, n, rng):
    """Swap two blocks of h."""
    block_len = rng.randint(1, n // 4 + 1)
    i = rng.randint(0, n - block_len)
    j = rng.randint(0, n - block_len)
    if abs(i - j) < block_len:
        j = (i + block_len + rng.randint(0, n - 2*block_len)) % max(1, n - block_len)
    h_new = h.copy()
    h_new[i:i+block_len], h_new[j:j+block_len] = h_new[j:j+block_len].copy(), h_new[i:i+block_len].copy()
    return project(h_new, n)

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

For a pair (j, j') with h[j] + h[j'] = c (fixed), I want to minimize max_k corr[k]. Each corr[k] is affine in (h[j], h[j']), and since h[j'] = c - h[j], each corr[k] becomes affine in h[j]. So I'm minimizing a piecewise-linear function of h[j] over the range [max(0, c-1), min(1, c)].

def pair_optimize(h, n, j, jp):

    hi = min(1.0, c)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    x = (corr - m) * T
    x = np.clip(x, -500, 0)
    ex = np.exp(x)
    se = ex.sum()
    if se < 1e-300:
        return m, np.zeros(n)
    w = ex / se
    val = m + np.log(se) / T
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-11:
            continue
        l = k - (n - 1)
        if l >= 0:
            r = n - l
            if r > 0:
                grad[:r] += wk * (1.0 - h[l:])
                grad[l:] -= wk * h[:r]
        else:
            m2 = -l
            r = n - m2
            if r > 0:
                grad[m2:] += wk * (1.0 - h[:r])
                grad[:r] -= wk * h[m2:]
    grad *= dx
    return val, grad

def lbfgs_solve(h0, n, T, deadline, maxiter=800):
    target = n / 2.0
    pen = 500.0
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        val, grad = smooth_max_grad(x, n, T)
        viol = np.sum(x) - target
        val += 0.5 * pen * viol * viol
        grad = grad + pen * viol
        return val, grad
    try:
        res = minimize(obj_and_grad, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def pgd_refine(h0, n, T, deadline, maxiter=200):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for it in range(maxiter):
        if time.time() > deadline:
            break
        val, grad = smooth_max_grad(h, n, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        lr = 0.5 / (gnorm + 1e-10)
        for _ in range(30):
            h_new = project(h - lr * grad, n)
            val_new, _ = smooth_max_grad(h_new, n, T)
            if val_new < val:
                h = h_new
                val = val_new
                break
            lr *= 0.5
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def make_perturbation(h, n, rng, mode):
    if mode == 0:
        sigma = 0.05 + 0.2 * rng.rand()
        return project(h + rng.randn(n) * sigma, n)
    elif mode == 1:
        h2 = h.copy()
        blen = rng.randint(2, max(3, n // 6))
        i = rng.randint(0, n - blen)
        j = rng.randint(0, n - blen)
        h2[i:i+blen], h2[j:j+blen] = h2[j:j+blen].copy(), h2[i:i+blen].copy()
        return project(h2, n)
    elif mode == 2:
        return project(0.3 * h + 0.7 * rng.rand(n), n)
    elif mode == 3:
        h2 = h.copy()
        n_swap = n // 5
        idx = rng.choice(n, size=2 * n_swap, replace=False)
        for i in range(0, 2 * n_swap, 2):
            h2[idx[i]], h2[idx[i+1]] = h2[idx[i+1]], h2[idx[i]]
        return project(h2, n)
    else:
        return project(h + rng.randn(n) * 0.15, n)
