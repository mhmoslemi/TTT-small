import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def smooth_grad_tophk(h, tau, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    idx = np.argsort(corr)[::-1][:top_k]
    c_sub = corr[idx] - max_c
    shifted = np.clip(tau * c_sub, -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for i_k, k in enumerate(idx):
        pk = p[i_k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            a = np.arange(j0, j1)
            grad[a] += pk * (1.0 - h[a - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            a = np.arange(j0b, j1b)
            grad[a] -= pk * h[a + sh]
    grad *= dx
    return max_c * dx, grad


def optimize_smooth(h, n, tau, time_limit, top_k, lr_init):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.75, 0.999, 1e-8
    t = 0
    stall = 0
    while time.time() - start < time_limit:
        t += 1
        _, grad = smooth_grad_tophk(h, tau, top_k)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h - step, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 300:
            lr *= 0.55
            stall = 0
            if lr < 1e-14:
                break
    return best_h, best_c5


def subgrad_refine(h, n, time_limit, lr_init):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    stall = 0
    vel = np.zeros(n)
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = int(np.argmax(corr))
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            a = np.arange(j0, j1)
            grad[a] += (1.0 - h[a - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            a = np.arange(j0b, j1b)
            grad[a] -= h[a + sh]
        grad *= dx

            vel *= 0.4
            stall = 0
            if lr < 1e-15:
                break
    return best_h, best_c5
