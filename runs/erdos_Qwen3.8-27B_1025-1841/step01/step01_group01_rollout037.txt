def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_AB(j, h, lags_array, dx):
    n = len(h)
    c_vals = np.correlate(h, 1 - h, mode="full") * dx
    n_lags = 2 * n - 1
    B = np.zeros(n_lags)
    for idx, l in enumerate(lags_array):
        i_min = max(0, -l)
        i_max = min(n - 1, n - 1 - l)
        if i_min <= j <= i_max:
            B[idx] += -dx * h[j + l]
        m = j - l
        if 0 <= m < n and i_min <= m <= i_max:
            B[idx] += dx * (1.0 - h[m])
    A = c_vals - B * h[j]
    return A, B

def minimize_1d_convex(A, B):

def make_interpolated(h_parent, n_new):
    n_old = len(h_parent)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_parent)
    return project_onto_constraint(h_new, n_new / 2.0)

def make_periodic_pattern(n, period):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % period) < period // 2 else 0.0
    h = project_onto_constraint(h, n / 2.0)
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i // 2] ^ (i % 2)
    h[tm == 1] = 1.0
    h = project_onto_constraint(h, n / 2.0)
    return h
