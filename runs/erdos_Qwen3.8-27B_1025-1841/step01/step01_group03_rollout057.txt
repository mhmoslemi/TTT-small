    c_max = float(np.max(c))
    c_shifted = c - c_max
    return c_max + (1.0 / T) * float(np.log(np.sum(np.exp(T * c_shifted))))

def obj_func(x):
    return smooth_c5_val(x, _obj_n, _obj_T)

def con_func(x):
    return float(np.sum(x) - _obj_target)

def fix_sum(h, target):
    h = h.copy()
    diff = target - float(np.sum(h))
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = int(np.sum(mask))
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    diff2 = target - float(np.sum(h))
    if abs(diff2) > 1e-10:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff2
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _obj_n, _obj_T, _obj_target
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 80

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None:
        c5_parent = compute_c5(parent, len(parent))
        best_c5 = min(best_c5, c5_parent)
        best_h = parent.copy()
        best_n = len(parent)

    resolutions = [80, 120, 160, 200]

    for n in resolutions:
        elapsed = time.time() - t0
        if budget_s - elapsed < 40:
            break
        if n >= 120 and budget_s - elapsed < 90:
            break
        if n >= 160 and budget_s - elapsed < 180:
            break
        if n >= 200 and budget_s - elapsed < 300:
            break

        target = n / 2.0
        _obj_n = n
        _obj_target = target
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': con_func}]

        inits = []

        if parent is not None:
            if n == len(parent):
                inits.append(parent.copy())
            else:
                x_old = np.linspace(0.0, 2.0, len(parent), endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h_i = np.interp(x_new, x_old, parent)
                np.clip(h_i, 0.0, 1.0, out=h_i)
                h_i = fix_sum(h_i, target)
                inits.append(h_i)

        n_rand = 3 if (budget_s - (time.time() - t0)) > 400 else 2
        for _ in range(n_rand):
            h_r = 0.5 + 0.35 * rng.randn(n)
            np.clip(h_r, 0.0, 1.0, out=h_r)
            h_r = fix_sum(h_r, target)
            inits.append(h_r)

        if (budget_s - (time.time() - t0)) > 200:
            for frac in [0.25, 0.3, 0.35, 0.4]:
                h_blk = np.zeros(n)
                blen = max(2, int(frac * n))
                for s in range(0, n, 2 * blen):
                    h_blk[s:s + blen] = 1.0
                h_blk = fix_sum(h_blk, target)
                inits.append(h_blk)

        for h0 in inits:
            if budget_s - (time.time() - t0) < 15:
                break
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = fix_sum(h0, target)

            for T in [50.0, 200.0]:
                if budget_s - (time.time() - t0) < 10:
                    break
                _obj_T = T
                maxiter = 400 if n <= 100 else 200
                try:
                    res = minimize(obj_func, h0, method='SLSQP',
                                  bounds=bounds, constraints=constraints,
                                  options={'maxiter': maxiter, 'ftol': 1e-12})
                    if res.x is not None:
                        h_opt = np.clip(res.x, 0.0, 1.0)
                        h_opt = fix_sum(h_opt, target)
                        c5 = compute_c5(h_opt, n)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h_opt.copy()
                            best_n = n
                            h0 = h_opt.copy()
                except Exception:
                    pass

    if best_h is not None and (budget_s - (time.time() - t0)) > 20:
        n = best_n
        target = n / 2.0
        _obj_n = n
        _obj_target = target
        _obj_T = 300.0
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': con_func}]

        for trial in range(12):
            if budget_s - (time.time() - t0) < 5:
                break
            noise = 0.01 * rng.randn(n)
            h_pert = best_h + noise
            np.clip(h_pert, 0.0, 1.0, out=h_pert)
            h_pert = fix_sum(h_pert, target)
            try:
                res = minimize(obj_func, h_pert, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 100, 'ftol': 1e-12})
                if res.x is not None:
                    hp = np.clip(res.x, 0.0, 1.0)
                    hp = fix_sum(hp, target)
                    c5p = compute_c5(hp, n)
                    if c5p < best_c5:
                        best_c5 = c5p
                        best_h = hp.copy()
                        best_n = n
            except Exception:
                pass

    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, n)
        best_n = n

    return (best_h, best_c5, best_n)
