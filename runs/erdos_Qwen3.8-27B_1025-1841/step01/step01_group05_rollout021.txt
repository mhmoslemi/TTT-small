        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 3: Smooth L-BFGS-B (log-sum-exp)
    for beta in [30, 100, 300]:
        if time_left() < 30:
            break
        h_cand, c5_cand = smooth_lbfqsb(best_h, n, max_iter=300, beta=beta)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 4: Local search
    if time_left() > 30:
        h_cand, c5_cand = local_search(best_h, n, rng, n_trials=min(10000, int(time_left() * 20)))
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 5: More subgradient with smaller steps (polish)
    if time_left() > 30:
        h_cand, c5_cand = subgradient_descent(best_h, n, iters=5000, lr0=0.002, lr_decay=0.9999, tol_frac=0.05)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 6: Try higher resolution
    for _ in range(2):
        if time_left() < 180:
            break
        n_cur = len(best_h)
        n_new = n_cur * 2
        if n_new > 800:
            break
        h_up = upsample(best_h, 2)
        h_up = project_feasible(h_up, n_new)

        h_cand, c5_cand = subgradient_descent(h_up, n_new, iters=2000, lr0=0.01, lr_decay=0.9995)
        if time_left() > 60:
            h_cand, c5_cand = lbfqsb_refine(h_cand, n_new, max_iter=200)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Final polish
    if time_left() > 10:
        h_cand, c5_cand = lbfqsb_refine(best_h, len(best_h), max_iter=200)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
