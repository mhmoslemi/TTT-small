import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def objective_func(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def constraint_func(h, n):
    return np.sum(h) - n / 2.0


def smooth_objective(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float((1.0 / beta) * np.log(np.sum(np.exp(beta * c)) - (len(c) - 1) * np.exp(-beta * 10)))


def do_optimize(h0, n, maxiter, ftol, deadline, smooth=False):
    if time.time() > deadline - 4:
        return None, float('inf')
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    diff = n / 2.0 - np.sum(h0)
    if abs(diff) > 1e-6:
        h0 = h0 + diff / n
        h0 = np.clip(h0, 0.0, 1.0)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]

    if smooth:
        obj_wrapper = smooth_objective
        args = (n, 50.0)
    else:
        obj_wrapper = objective_func
        args = (n,)

    try:
        res = minimize(obj_wrapper, h0, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = np.clip(res.x, 0.0, 1.0)
        diff2 = n / 2.0 - np.sum(h_opt)
        if abs(diff2) > 1e-4:
            h_opt = h_opt + diff2 / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        c5_val = objective_func(h_opt, n)
        return h_opt, c5_val
    except Exception:
        return None, float('inf')
