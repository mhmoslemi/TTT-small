        else:

        if len(idx) > 0:

            h = np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_constraints(h_new, n_new)

def w_matvec(w, h, n):
    """Compute W @ h where W = sum_k w[k] M_k"""
    # w has length 2n-1 (corresponding to shifts -(n-1) to (n-1))
    result = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            # M_k has (M_k)_{j, j+k} = (M_k)_{j+k, j} = 1/2
            m = n - shift
            result[:m] += w[k] * 0.5 * h[shift:shift+m]
            result[shift:] += w[k] * 0.5 * h[:m]
        else:
            s = -shift
            m = n - s
            result[s:] += w[k] * 0.5 * h[:m]
            result[:m] += w[k] * 0.5 * h[s:]
    return result

def reweighting_iteration(h, n, dx, T, max_inner=50):
    """One outer iteration of the iterative reweighting method"""
    for inner in range(max_inner):
        # Compute current correlations
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        # Compute weights
        m = np.max(corr)
        w = np.exp((corr - m) / T)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(2*n-1)
            w[n-1] = 1.0
        else:
            w /= w_sum
        
        # Gradient ascent on h^T W h (which is what we want to maximize)
        # grad = 2 * W @ h
        grad = 2.0 * w_matvec(w, h, n)
        
        # Set step size
        # Use a conservative step
        alpha = 0.5 / (np.max(np.abs(grad)) + 1e-10)
        
        h_new = project_onto_constraints(h + alpha * grad, n)
        
        # Check improvement
        new_corr = np.correlate(h_new, 1.0 - h_new, mode="full")
        old_max = np.max(corr)
        new_max = np.max(new_corr)
        
        if new_max < old_max:
            h = h_new
        else:
            # Reduce step size and try again
            alpha *= 0.5
            if alpha < 1e-8:
                break
            h_new = project_onto_constraints(h + alpha * grad, n)
            new_corr = np.correlate(h_new, 1.0 - h_new, mode="full")
            if np.max(new_corr) < old_max:
                h = h_new
            else:
                break
    
    return h

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraints(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-15):
            h = h_new
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-14:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_constraints(h_new, n_new)

def weighted_grad(h, w, n):
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ws = w[s + n - 1]
        if abs(ws) < 1e-15:
            continue
        if s >= 0:
            m = n - s
            grad[:m] += ws * (1.0 - h[s:s+m])
            grad[s:] += -ws * h[:m]
        else:
            t = -s
            m = n - t
            grad[t:] += ws * (1.0 - h[:m])
            grad[:m] += -ws * h[t:]
    return grad

def smooth_corr_obj(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + T * np.log(np.sum(np.exp((corr - m) / T)))) * dx

def smooth_corr_grad(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / T)
    w_sum = np.sum(w)
    w /= w_sum
    grad = weighted_grad(h, w, len(h))
    return grad * dx

        for T in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]:

                            c5 = c5_new

            best_h = h.copy()
            best_n = n

    best_h = project_onto_constraints(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
