1. It's slow for large n (O(n³) per iteration due to the constraint handling)
2. It might not converge well

Alternative: use L-BFGS-B with a penalty for the sum constraint.

Objective: smooth_max(corr) * dx + mu * (sum(h) - target)²

L-BFGS-B handles box constraints [0,1] natively. The sum constraint is handled by the penalty. For large mu, the constraint is approximately satisfied.

This is much faster than SLSQP for n > 50. Let me use this.

            if np.isfinite(res.fun):
                h_new = project_target(res.x, n)
                if compute_c5(h_new, dx) < compute_c5(h, dx):
                    h = h_new
        except Exception:
            pass
    return h

def lbfgs_obj_fun(x, n, dx, target, mu, temp):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0-h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / temp
    logsum = np.log(np.sum(np.exp(shifted)))
    f = (m + temp * logsum) * dx + mu * (np.sum(h) - target)**2
    # Gradient
    grad = smooth_max_grad(h, n, dx, temp) + 2 * mu * (np.sum(h) - target) * np.ones(n)
    return f, grad

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        adj = diff / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.max(np.abs(h_new - h)) < 1e-15:
            h = h_new
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-12:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def max_linear_obj(t, A, B):
    return float(np.max(A + t * B))

def coord_descent(h, n, dx, target, n_sweeps, time_limit):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    for sweep in range(n_sweeps):
        if time.time() > time_limit:
            break
        for j in range(n):
            if time.time() > time_limit:
                break
            t_old = h[j]
            m = np.arange(-(n - 1), n)
            v1 = (m >= -j) & (m < n - j)
            v2 = (m > j - n) & (m <= j)
            B = np.zeros(2 * n - 1)
            if np.any(v1):
                B[v1] = 1.0 - h[j + m[v1]]
            if np.any(v2):
                B[v2] -= h[j - m[v2]]
            A = corr - t_old * B
            res = minimize_scalar(max_linear_obj, bounds=(0.0, 1.0),
                                  method='bounded', args=(A, B),
                                  options={'xatol': 1e-8})
            t_new = res.x
            corr += (t_new - t_old) * B
            h[j] = t_new
        h = project_target(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
    return h

def smooth_max_grad(h, n, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / temp
    p = np.exp(shifted)
    p /= np.sum(p)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        s = k - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += p[k] * (1.0 - h[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= p[k] * h[j_lo2 - s:j_hi2 - s]
    return grad * dx

def lbfgs_smooth_obj(x, n, dx, target, mu, temp):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / temp
    logsum = np.log(np.sum(np.exp(shifted)))
    f = (m + temp * logsum) * dx + mu * (np.sum(h) - target) ** 2
    g = smooth_max_grad(h, n, dx, temp) + 2.0 * mu * (np.sum(h) - target)
    return f, g

def lbfgs_optimize(h, n, dx, target, temps, mu, time_limit, maxiter):
    for temp in temps:
        if time.time() > time_limit:
            break
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(lbfgs_smooth_obj, h, method='L-BFGS-B',
                           bounds=bnds, args=(n, dx, target, mu, temp),
                           options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-7,
                                    'maxls': 30})
            if np.isfinite(res.fun):
                h_new = project_target(res.x, n)
                if compute_c5(h_new, dx) < compute_c5(h, dx):
                    h = h_new
        except Exception:
            pass
    return h

def subgradient_refine(h, n, dx, target, n_iter, time_limit):
    alpha = 4.0 / (n * 0.1)
    best = compute_c5(h, dx)
    no_improve = 0
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:n])
            grad[s:] -= h[:n - s]
        else:
            t = -s
            grad[t:] += (1.0 - h[:n - t])
            grad[:n - t] -= h[t:]
        h_new = project_target(h - alpha * grad, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best - 1e-12:
            h = h_new
            best = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                alpha *= 0.85
                no_improve = 0
        if it % 300 == 299:
            alpha *= 0.92
    return h

def binarize(h, n):
    b = (h >= 0.5).astype(float)
    s = b.sum()
    target = n / 2
    if s > target:
        ones = np.where(b == 1)[0]
        remove_count = int(np.ceil(s - target))
        if remove_count > 0:
            idx_remove = ones[np.argsort(h[ones])[:remove_count]]
            b[idx_remove] = 0
    elif s < target:
        zeros = np.where(b == 0)[0]
        add_count = int(np.ceil(target - s))
        if add_count > 0:
            idx_add = zeros[np.argsort(-h[zeros])[:add_count]]
            b[idx_add] = 1
    return b

def make_binary_start(n, rng):
    b = rng.randint(0, 2, n).astype(float)

        elif s < target:
            idx = np.where(b == 0)[0]
            b[idx[rng.randint(len(idx))]] = 1
        else:
            break
    return b
