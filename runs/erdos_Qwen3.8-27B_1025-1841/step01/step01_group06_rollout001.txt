import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-10
        else:
            m = h > 1e-10
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def subsgrad(h, n, ks):
    s = ks - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    if hi < lo:
        return g
    idx = np.arange(lo, hi + 1)
    js = idx + s
    g[idx] += 1.0 - h[js]
    g[js] -= h[idx]
    return g

def smooth_obj(x, dx, T):
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    mx = c.max()
    cs = np.clip((c - mx) / T, -50.0, 0.0)
    return (mx + T * np.log(np.sum(np.exp(cs)))) * dx

def upsample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new)

def sg_optimize(h_c, n, dx_n, iters, rem_fn, min_rem=25):
    alpha = 6.0 / n
    mom = 0.93
    v = np.zeros(n)
    best_c5 = compute_c5(h_c, dx_n)
    best_h = h_c.copy()
    stale = 0
    for it in range(iters):
        if rem_fn() < min_rem:
            break
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        ks = int(np.argmax(corr))
        g = subsgrad(h_c, n, ks)
        v = mom * v - alpha * g
        h_t = project_h(h_c + v, n)
        c5_t = compute_c5(h_t, dx_n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            stale = 0
        else:
            stale += 1
        h_c = h_t
        if stale > 250:
            alpha *= 0.87
            v *= 0.4
            stale = 0
        if it % 3000 == 2999:
            alpha *= 0.82
    return best_h, best_c5

def coord_descent(h_c, n, dx_n, max_sweeps, rem_fn, min_rem=15):
    c5_c = compute_c5(h_c, dx_n)
    for sw in range(max_sweeps):
        if rem_fn() < min_rem:
            break
        any_imp = False
        for i in range(n):
            cands = [0.0, 1.0]
            if 0.02 < h_c[i] < 0.98:
                cands.append(h_c[i] - 0.04)
                cands.append(h_c[i] + 0.04)
            for val in cands:
                val = max(0.0, min(1.0, val))
                if abs(val - h_c[i]) < 1e-8:
                    continue
                h_t = h_c.copy()
                h_t[i] = val
                h_t = project_h(h_t, n)
                c5_t = compute_c5(h_t, dx_n)
                if c5_t < c5_c - 1e-12:
                    h_c = h_t
                    c5_c = c5_t
                    any_imp = True
                    break
        if not any_imp:
            break
    return h_c, c5_c
