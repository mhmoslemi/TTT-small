import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):

    return compute_corr(h).max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:

    obj = dx * (max_c + np.log(s) / tau)

where f is smooth and C is convex. The update is:

Each H_k has entries bounded by 1 (since ∂²corr_k/∂h[i]∂h[j] ∈ {-1, 0, 1}), so ||H_k||₂ ≤ 2 (roughly).

        threshold = c5_cur - 0.01  # include shifts within 0.01 of max

        if not active.any():

        weights = c5_vals[active] ** 10  # emphasize the max
        weights /= weights.sum()

            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[j0:j1] += w * (1.0 - h[idx - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[j0b:j1b] -= w * h[idx + sh]

        h = project_to_feasible(h - lr * grad, n)
        
        lr *= 0.9995  # slow decay
        if stall > 500:
            lr *= 0.5
            stall = 0
            if lr < 1e-12:
                break

∂f_k/∂h[i] = (1 - h[i+k']) - h[i-k'] (where k' is the effective shift with wrapping)

2. Use convolutions

Σ_k p_k * shifted(1-h, k)[i] = 1 - Σ_k p_k * shifted(h, k)[i]
= 1 - (convolution of p with h)

Let me think more carefully. The correlation is:
corr[k] = Σ_j h[j] * (1-h[j+k]) for k = -(n-1), ..., (n-1)

where j ranges over valid indices (0 ≤ j < n and 0 ≤ j+k < n).

The gradient of corr[k] with respect to h[i]:
∂corr[k]/∂h[i] = (1-h[i+k]) * δ_{i+k valid} - h[i-k] * δ_{i-k valid}

So the total gradient is:
∇f[i] = dx * Σ_k p_k * [(1-h[i+k])_valid - h[i-k]_valid]

def smooth_grad_fast(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    # Fast gradient using convolution
    T2 = np.convolve(h, p[::-1], mode='full')[n-1:n-1+n]
    T3 = np.convolve(h, p, mode='full')[n-1:n-1+n]
    grad = dx * (1.0 - T2 - T3)
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad
