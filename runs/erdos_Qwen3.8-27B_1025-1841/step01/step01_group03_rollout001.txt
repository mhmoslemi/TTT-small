    return f, g

def _proj(h, target):
    h = h.copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        np.clip(h, 0, 1, out=h)
    return h

def _interp_h(src, n_src, n_dst):
    x_src = np.linspace(0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, src)

def _optimize_phase(h, beta, pen, max_iter, n):
    bounds = [(0.0, 1.0)] * n
    try:
        res = optimize.minimize(
            lambda x: _og(x, beta, pen)[0],
            h,
            jac=lambda x: _og(x, beta, pen)[1],
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-13, 'gtol': 1e-9, 'maxls': 40}
        )
        return _proj(np.clip(res.x, 0, 1), n / 2.0)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 999.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
    else:
        ph, pn = None, 0

    resolutions = []
    if pn > 0:
        resolutions.append(pn)
    for nt in [80, 160, 320, 640]:
        if nt not in resolutions:
            resolutions.append(nt)

    time_alloc = {}
    total_rem = budget_s - (time.time() - t0)
    n_res = len(resolutions)
    for i, n in enumerate(resolutions):
        if i == 0 and pn > 0:
            time_alloc[n] = total_rem * 0.40
        elif n <= 160:
            time_alloc[n] = total_rem * 0.25
        else:
            time_alloc[n] = total_rem * 0.15

    for n in resolutions:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        alloc = min(time_alloc.get(n, 200), rem - 30)
        if alloc < 30:
            break

        n_end = t0 + budget_s - 30
        if ph is not None and n == pn:
            h = np.array(ph, dtype=float)
        elif ph is not None:
            h = _interp_h(ph, pn, n)
        else:
            h = np.full(n, 0.5)
            x = np.linspace(0, 2, n, endpoint=False)
            h = 0.5 + 0.05 * np.sin(np.pi * 2 * x)

        h = _proj(np.clip(h, 0, 1), n / 2.0)
        cur_c5 = _c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        betas = [10, 25, 60, 150, 350, 700, 1200, 2000]
        pens = [300.0, 200.0, 200.0, 100.0, 100.0, 50.0, 50.0, 50.0]

        for pi, beta in enumerate(betas):
            rem = budget_s - (time.time() - t0)
            if rem < 20:
                break
            phase_budget = alloc / len(betas)
            iters = max(5, int(min(60, phase_budget / 1.5)))
            pen = pens[pi] if pi < len(pens) else 50.0

            h_new = _optimize_phase(h, beta, pen, iters, n)
            c5_new = _c5(h_new)
            if c5_new < _c5(h):
                h = h_new
            c5_cur = _c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()

        # Cross-refine: second pass with high beta
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            for bt in [500, 1000, 2000]:
                hd = _optimize_phase(h, bt, 100.0, 30, n)
                cd = _c5(hd)
                if cd < _c5(h):
                    h = hd
                if _c5(h) < best_c5:
                    best_c5 = _c5(h)
                    best_h = h.copy()

        # Local perturbation refinement
        rem = budget_s - (time.time() - t0)
        if rem > 40:
            npert = int(min(40, rem / 5))
            for _ in range(npert):
                sigma = 0.005
                hp = h + rng.normal(0, sigma, n)
                hp = _proj(np.clip(hp, 0, 1), n / 2.0)
                cp = _c5(hp)
                if cp < _c5(h):
                    h = hp
                    if cp < best_c5:
                        best_c5 = cp
                        best_h = h.copy()
                if rng.random() < 0.1:
                    sigma *= 0.5

    if best_h is None:
        best_h = np.full(80, 0.5)
        best_c5 = _c5(best_h)

    return best_h, float(_c5(best_h)), len(best_h)
