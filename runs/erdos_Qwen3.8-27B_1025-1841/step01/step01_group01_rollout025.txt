    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_feasible(y, target_sum):
    """Project y onto {0≤h≤1, sum(h)=target_sum}"""
    n = len(y)
    # Bisection on lambda
    lo, hi = -1, 1
    for _ in range(50):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        if np.sum(h) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0, 1)

def subgradient_descent(h0, n_iters=10000):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Learning rate schedule
    for it in range(n_iters):
        # Compute all g_k values
        corr = np.correlate(h, 1-h, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(n):
            val = 0.0
            # Term from j = i: -h[i+lag] if i+lag in range
            if 0 <= i + lag < n:
                val -= h[i + lag]
            # Term from j+lag = i, i.e., j = i-lag: (1-h[i-lag]) if i-lag in range
            if 0 <= i - lag < n:
                val += (1 - h[i - lag])
            sg[i] = dx * val

        alpha = 1.0 / (1 + 0.01 * it)
        
        # Update

        h = project_feasible(y, target_sum)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
