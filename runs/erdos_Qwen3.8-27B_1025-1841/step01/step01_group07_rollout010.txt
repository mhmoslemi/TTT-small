    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    return np.max(corr) * dx

def sum_constraint(x, target):
    return np.sum(x) - target

def fix_sum(h, target_sum):
    """Adjust h to exactly satisfy sum(h) = target_sum while keeping 0<=h<=1."""
    h = h.copy()
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            count = np.sum(mask)
            if count == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / count)
        else:
            mask = h > 1e-15
            count = np.sum(mask)
            if count == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / count)
    return h

    """Local search for binary pattern minimization."""
    # Convert to binary: sort values, set top n/2 to 1, rest to 0
    # Actually for the parent, let's just try flipping
    best = evaluate_c5(h)
    for it in range(max_iters):
        improved = False
        for i in range(n):
            # Try flipping h[i] (or adjusting)
            for new_val in [0.0, 1.0]:
                if new_val == h[i]:
                    continue
                h[i] = new_val
                h_adj = fix_sum(h, target_sum)
                c = evaluate_c5(h_adj)
                if c < best - 1e-12:
                    best = c
                    h[i] = new_val
                    h = h_adj
                    improved = True
                    break

                break

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def obj_data(x, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * dx

def sum_constraint(x, target):
    return np.sum(x) - target

def fix_sum(h, target_sum):
    h = h.copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
