import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] = np.clip(h[mask] + diff / np.sum(mask), 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.arange(n_new, dtype=float) * (n_old / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def gradient_active_lag(h, n, k_star):
    s = k_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:s+m])
        grad[s:] -= h[:m]
    else:
        t = -s
        m = n - t
        grad[t:] += (1.0 - h[:m])
        grad[:m] -= h[t:t+m]
    return grad

def frank_wolfe_step(h, n):
    """One Frank-Wolfe step: return the vertex z that minimizes g·z over the hypersimplex."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    g = gradient_active_lag(h, n, k_star)
    # Linear minimization: set z[i]=1 for the n/2 smallest g[i]
    m = n // 2
    idx = np.argpartition(g, m)[:m]
    z = np.zeros(n)
    z[idx] = 1.0
    return z, k_star

    best_h = h0.copy()
    best_n = n0

    resolutions = []
    n = n0
    while n <= 4000:
        resolutions.append(n)
        n *= 2
    
    for n in resolutions:
        if time.time() > deadline:
            break
        dx = 2.0 / n
        target = n / 2.0
        
        if n == n0:

        else:

        h_curr = project_target(h_curr, n)
        c5_curr = compute_c5(h_curr, dx)
        
        # Phase 1: Frank-Wolfe
        n_fw = 5000 if n <= 500 else 3000
        for t in range(n_fw):
            if time.time() > deadline:

            z, k_star = frank_wolfe_step(h_curr, n)
            gamma = 2.0 / (t + 2.0)
            h_new = (1.0 - gamma) * h_curr + gamma * z
            h_new = np.clip(h_new, 0.0, 1.0)
            # Project to sum constraint

            diff = target - s

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] = np.clip(h[mask] + diff / np.sum(mask), 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    scale = n_old / n_new
    x_new = np.arange(n_new, dtype=float) * scale
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def gradient_active_lag(h, n, k_star):
    s = k_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:s+m])
        grad[s:] -= h[:m]
    else:
        t = -s
        m = n - t
        grad[t:] += (1.0 - h[:m])
        grad[:m] -= h[t:t+m]
    return grad

def pg_optimize(h, n, dx, max_iter=5000, alpha_init=1.0, tol=1e-12):
    """Projected gradient with backtracking line search."""
    target = n / 2.0
    c5_curr = compute_c5(h, dx)
    alpha = alpha_init / np.sqrt(n)
    no_improve = 0
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        g = gradient_active_lag(h, n, k_star)
        
        step_alpha = alpha
        improved = False
        for _ in range(15):
            h_try = project_target(h - step_alpha * g, n)
            c5_try = compute_c5(h_try, dx)
            if c5_try < c5_curr - tol:
                h = h_try
                c5_curr = c5_try
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve >= 50:
                alpha *= 0.8
                no_improve = 0
            if no_improve >= 500:
                break
        if it % 1000 == 999:
            alpha = max(alpha * 0.9, 1e-6 / np.sqrt(n))
    
    return h, c5_curr

def fw_optimize(h, n, dx, max_iter=5000):
    """Frank-Wolfe (conditional gradient) optimization."""
    target = n / 2.0
    c5_curr = compute_c5(h, dx)
    m = n // 2
    
    for t in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        g = gradient_active_lag(h, n, k_star)

        z = np.zeros(n)
        z[idx] = 1.0

        h_new = (1.0 - gamma) * h + gamma * z
        h_new = project_target(h_new, n)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < c5_curr:
            h = h_new
            c5_curr = c5_new
    
    return h, c5_curr

Also, I realize that the FW method with the subgradient of the active lag might not work well because the subgradient can change direction rapidly. A better FW approach would be to use the **exact gradient of the smooth max** or at least a **weighted average** of nearby active lags.

Let me also add a "multi-lag gradient" option where I average the gradients of the top-k lags (weighted by their correlation values). This gives a smoother direction.

Actually, let me think about this more carefully. The issue with using just the active lag gradient is that when two lags have nearly the same correlation, the gradient can flip between them, causing oscillation. A better approach is to use a "soft" gradient that accounts for all near-active lags.

For efficiency, I can limit this to the top 5-10 lags:
