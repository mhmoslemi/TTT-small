        s2 = np.sum(p[v_pos] * h[idx_pos[v_pos]])

import numpy as np
import time

# All helper functions at top level

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    p = exp_vals / np.sum(exp_vals)
    lags = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        idx_neg = i - lags
        idx_pos = i + lags
        v_neg = (idx_neg >= 0) & (idx_neg < n)
        v_pos = (idx_pos >= 0) & (idx_pos < n)
        s_neg = np.sum(p[v_neg] * h[idx_neg[v_neg]]) if np.any(v_neg) else 0.0
        s_pos = np.sum(p[v_pos] * h[idx_pos[v_pos]]) if np.any(v_pos) else 0.0
        grad[i] = dx * (1.0 - s_neg - s_pos)
    return grad

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def proj_grad_optimize(h0, n, T, lr_init, max_iter, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_f = softmax_obj(h, T)
    lr = lr_init
    stagnant = 0

            break
        g = softmax_grad(h, T)
        f_cur = softmax_obj(h, T)
        
        step = lr
        accepted = False

            f_trial = softmax_obj(h_trial, T)
            if f_trial < f_cur - 1e-12:
                h = h_trial
                accepted = True
                break
            step *= 0.5
            if step < 1e-10:

        if f_new < best_f - 1e-14:

            best_h = h.copy()
    
    return best_h

So the FFT-based gradient is:
```python
def softmax_grad_fast(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    p = exp_vals / np.sum(exp_vals)
    
    A = np.convolve(h, p, 'full')[n-1 : n-1+n]
    C = np.convolve(p, h, 'full')
    B = C[2*n-2 : n-2 : -1]
    
    grad = dx * (1.0 - A - B)
    return grad
```

This is O(n log n) per gradient! For n=2016, each gradient is ~50μs (one FFT). The correlate is O(n²) still...
