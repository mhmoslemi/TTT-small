def project_sum(h, n):
    """Project h to satisfy sum(h) = n/2 and 0<=h<=1."""
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0, 1)
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        adj = (target - s) / n
        h = np.clip(h + adj, 0, 1)

        else:

            pass
    
    return best_h, best_c5

def upsample(h, n_new, seed=42):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0, 1)
    return project_sum(h_new, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    
    # Start from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_c5 = compute_c5(h0, dx0)
    best_h = h0.copy()
    best_n = n0
    
    # Phase 1: 0-1 optimization at n0
    h_01 = make_01(h0, n0, seed)
    # Ensure sum constraint
    if np.sum(h_01) != n0/2:
        h_01 = project_sum(h_01, n0)
    c5_01 = compute_c5(h_01, dx0)
    
    if c5_01 < best_c5:
        best_c5 = c5_01
        best_h = h_01.copy()
    
    # SA at n0
    h_sa, c5_sa = sa_01(h_01, n0, t0 + budget_s * 0.3, seed)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
        best_n = n0
