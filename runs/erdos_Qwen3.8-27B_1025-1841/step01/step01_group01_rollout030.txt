```python
import numpy as np
import time

def compute_c5(h, n):
    """Compute C5 value for a vector h of length n."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, n):
    """Ensure sum(h) = n/2 and 0 <= h[i] <= 1."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    current = np.sum(h)
    if current > 0:
        h = h * (target / current)
        h = np.clip(h, 0.0, 1.0)
    # If still not matching, do a more careful repair
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need to increase sum; add to smallest entries
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-10:
                    break
        else:
            # Need to decrease sum
            idx = np.argsort(-h)
            for i in idx:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-10:
                    break
    return h

def binary_local_search(h_bin, n, max_rounds=5000, Trials_per_round=100):
    """Local search on binary vector with fixed weight n/2."""
    h = h_bin.copy().astype(float)
    current_obj = compute_c5(h, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        best_obj = current_obj
        best_move = None
        for _ in range(Trials_per_round):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_obj = compute_c5(h, n)
            if new_obj < best_obj:
                best_obj = new_obj
                best_move = (i, j)
            h[i], h[j] = 1.0, 0.0
        if best_move is not None:
            i, j = best_move
            h[i], h[j] = 0.0, 1.0
            current_obj = best_obj
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            improved = True

def simulated_annealing(h_bin, n, t_start=1.0, t_end=0.001, n_iter=50000):
    """Simulated annealing on binary vector."""
    h = h_bin.copy().astype(float)
    current_obj = compute_c5(h, n)
    best_h = h.copy()
    best_obj = current_obj
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / n_iter)
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_obj = compute_c5(h, n)
        delta = new_obj - current_obj
        if delta < 0 or np.random.rand() < np.exp(-delta / T):

                best_obj = new_obj
                best_h = h.copy()
        else:

    return best_h, best_obj
```
