def evaluate(h, n_points):
    dx = 2.0 / n_points
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    return np.max(corr) * dx

def project_onto_constraints(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    h = np.array(h, dtype=float).copy()
    # Clamp to [0,1] first
    h = np.clip(h, 0, 1)
    # Now adjust to meet sum constraint using bisection on lambda
    s = np.sum(h)
    if s == target_sum:
        return h
    # We need to find lambda such that sum(clamp(h - lambda, 0, 1)) = target_sum
    def f(lam):
        return np.sum(np.clip(h - lam, 0, 1)) - target_sum
    # f(0) = s - target_sum, f(1) = sum(clamp(h-1,0,1)) - target_sum = 0 - target_sum < 0
    # f(-1) = sum(clamp(h+1,0,1)) - target_sum = sum(1) - target_sum = N - target_sum > 0 (if target_sum < N)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h = np.clip(h - lam, 0, 1)
    # Final exact adjustment: distribute remaining mass
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute among elements not at bounds
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h
