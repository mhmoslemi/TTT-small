    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    max_val = np.max(corr)
    max_frac = np.where(corr >= max_val - 1e-10)[0]

        # For each m in 0..n-1:
        # contribution = (1-h[m-shift]) * [0<=m-shift<n] - h[m+shift] * [0<=m+shift<n]
        m = np.arange(n)

import numpy as np
from scipy.optimize import minimize
import time

_opt_params = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def c5_obj_grad(h):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    max_val = np.max(corr)
    max_idx = np.where(corr >= max_val - 1e-12)[0]
    m = np.arange(n)
    grad = np.zeros(n)
    for k in max_idx:
        shift = k - (n - 1)
        va = (m - shift >= 0) & (m - shift < n)
        vb = (m + shift >= 0) & (m + shift < n)
        c = np.zeros(n)
        c[va] = 1.0 - h[m[va] - shift]
        c[vb] -= h[m[vb] + shift]
        grad += c
    grad /= float(len(max_idx))
    return float(dx * max_val), dx * grad

def project_simplex(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _penalty_obj(x):
    n = _opt_params['n']
    target = n * 0.5
    f, g = c5_obj_grad(x)
    c = np.sum(x) - target
    mu = _opt_params['mu']
    lam = _opt_params['lam']
    return f + lam * c + 0.5 * mu * c * c

def _penalty_jac(x):
    n = _opt_params['n']
    target = n * 0.5
    f, g = c5_obj_grad(x)
    c = np.sum(x) - target
    mu = _opt_params['mu']
    lam = _opt_params['lam']
    return g + (lam + mu * c) * np.ones(n)

    h = project_simplex(h0, target)
    lam = 0.0
    mu = mu_start
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_f = compute_c5(h)

    for _ in range(max_outer):
        _opt_params['n'] = n
        _opt_params['mu'] = mu
        _opt_params['lam'] = lam
        res = minimize(_penalty_obj, h, jac=_penalty_jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = res.x
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        c = np.sum(h_new) - target
        lam = lam + mu * c
        if abs(c) > 1e-8:
            mu = min(mu * 2.0, 1e8)
        h = np.clip(h_new, 0.0, 1.0)
        if abs(c) < 1e-12:
            break

    h_final = project_simplex(best_h, target)
    return h_final, compute_c5(h_final)

def interp_h(h, n_old, n_new):

        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
