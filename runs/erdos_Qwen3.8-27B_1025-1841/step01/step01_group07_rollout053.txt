            best_h = bhn.copy()

        if bhn is not None and time.time() - t0 < budget_s - 40:
            hc, vc = continuous_refine(bhn, n, max_iter=1500)
            if hc is not None and vc < best_c5:
                best_c5 = vc
                best_h = hc.copy()

    if best_h is None:
        n = 46
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = eval_c5(best_h, n)

    return (best_h, best_c5, len(best_h))
