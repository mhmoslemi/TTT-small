    # Phase 1: Optimize at n0 with L-BFGS-B
    if time.time() < deadline:
        for T in [0.05, 0.02, 0.01]:
            if time.time() > deadline:
                break
            h_opt = lbbgss_optimize(best_h, n0, dx0, T, max_iter=200, mu=80.0)
            c5_try = compute_c5(h_opt, dx0)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_opt.copy()

    # Phase 2: Upsample and optimize at higher resolutions
    resolutions = []
    for mult in [2, 4]:
        nm = n0 * mult
        if nm <= 600:
            resolutions.append(nm)

    for n in resolutions:
        if time.time() > deadline:
            break
        dx = 2.0 / n
        h_curr = upsample(best_h, n)
        c5_curr = compute_c5(h_curr, dx)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

        # L-BFGS-B with decreasing temperature
        temps = [0.1, 0.05, 0.02, 0.01, 0.005]
        if n > 400:
            temps = [0.08, 0.04, 0.015, 0.008]
        for T in temps:
            if time.time() > deadline:
                break
            max_it = 250 if n <= 280 else 150
            mu = 50.0 if n <= 280 else 30.0
            h_curr = lbbgss_optimize(h_curr, n, dx, T, max_iter=max_it, mu=mu)
            c5_try = compute_c5(h_curr, dx)
            if c5_try < c5_curr:
                c5_curr = c5_try
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

        # Subgradient refinement
        if time.time() < deadline:
            n_sg = 1500 if n <= 280 else 800
            h_curr = subgradient_refine(h_curr, n, dx, max_iter=n_sg)
            c5_curr = compute_c5(h_curr, dx)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

        # Small perturbation search
        if time.time() < deadline and n <= 280:
            h_curr = small_perturbation_search(h_curr, n, dx, n_trials=100, rng=rng)
            c5_curr = compute_c5(h_curr, dx)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

    # Phase 3: Try alternative start (random binary) at n0
    if time.time() < deadline:
        n_alt = n0
        dx_alt = 2.0 / n_alt
        target = n_alt / 2
        h_rand = np.zeros(n_alt)
        idx = rng.choice(n_alt, size=int(target), replace=False)
        h_rand[idx] = 1.0
        h_rand = project_target(h_rand + rng.randn(n_alt) * 0.2, n_alt)
        for T in [0.05, 0.02]:
            if time.time() > deadline:
                break
            h_rand = lbbgss_optimize(h_rand, n_alt, dx_alt, T, max_iter=200, mu=80.0)
        c5_rand = compute_c5(h_rand, dx_alt)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()
            best_n = n_alt

    # Phase 4: Final polish - subgradient at best resolution
    if time.time() < deadline:
        dx_best = 2.0 / best_n
        h_polish = subgradient_refine(best_h, best_n, dx_best, max_iter=1000, alpha_scale=4.0)
        c5_polish = compute_c5(h_polish, dx_best)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish.copy()

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_target(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
