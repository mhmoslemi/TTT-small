import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return float(np.max(c) * dx)

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(c))
    k_eff = idx - (n - 1)
    grad = np.zeros(n)
    if k_eff >= 0:
        if k_eff == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            k = k_eff
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
    else:
        m = -k_eff
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad * dx

def project_constraint(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt * 0.5
            step = min(step, ((1.0 - h[mask]).min() if cnt > 0 else 0) * 0.9)
            h[mask] += step
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = (-diff) / cnt * 0.5
            step = min(step, (h[mask].min() if cnt > 0 else 0) * 0.9)
            h[mask] -= step
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if diff >= -1e-14:
                    break
    return h

def optimize_h(h0, time_limit):
    n = len(h0)
    t_start = time.time()
    h = project_constraint(h0.copy())
    best_val = compute_c5(h)
    best_local = h.copy()

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    class Timeout(Exception):
        pass

    def obj(x):
        if time.time() - t_start > time_limit:
            raise Timeout()
        return compute_c5(x)

    def grad(x):
        if time.time() - t_start > time_limit:
            raise Timeout()
        return c5_gradient(x)

    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 400, 'ftol': 1e-12})
        if not res.success and time.time() - t_start > time_limit:
            pass
        hx = project_constraint(res.x)
        vx = compute_c5(hx)
        if vx < best_val:
            best_local, best_val = hx, vx
    except Exception:
        pass

    if time.time() - t_start < time_limit:
        for _ in range(300):
            if time.time() - t_start > time_limit:
                break
            g = c5_gradient(best_local)
            lr = 0.05 / (n / 50.0 + 1)
            h_new = project_constraint(best_local - lr * g)
            v_new = compute_c5(h_new)
            if v_new < best_val - 1e-12:
                best_local, best_val = h_new, v_new
            else:
                lr *= 0.5
                h_new = project_constraint(best_local - lr * g)
                v_new = compute_c5(h_new)
                if v_new < best_val - 1e-14:
                    best_local, best_val = h_new, v_new
                else:
                    break
    return best_local

def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_constraint(h_dst)

    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    try:
