One more thought: the projected gradient method with subgradient will be slow to converge near the optimum because the max function makes the objective non-smooth. A better approach might be to use a smoothed version:

f_smooth(h) = (1/τ) * log(sum_k exp(τ * corr_k * dx))

This is a smooth approximation that converges to max as τ → ∞. We can optimize this with standard gradient methods.

The gradient of f_smooth is:
∂f_smooth/∂h[j] = (dx/τ) * sum_k exp(τ*corr_k*dx) * (∂corr_k/∂h[j]) / sum_k exp(τ*corr_k*dx)
= dx * sum_k w_k * (∂corr_k/∂h[j])

where w_k = exp(τ*corr_k*dx) / sum_k exp(τ*corr_k*dx) are softmax weights.

This is a weighted average of the individual gradients, which is smooth and well-defined. For large τ, it concentrates on the argmax lag.

This is a much better approach! Let me implement this.

```python
def c5_smooth(h, n_points, tau=100.0):
    n = n_points
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # log-sum-exp for numerical stability
    max_c = corr.max()
    log_sum = max_c + np.log(np.sum(np.exp(tau * (corr - max_c))))
    f = log_sum / tau
    return f

def c5_smooth_grad(h, n_points, tau=100.0):
    n = n_points
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = corr.max()
    weights = np.exp(tau * (corr - max_c))
    weights /= weights.sum()  # softmax weights
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        offset = k - (n - 1)
        j = np.arange(n)
        j1 = j + offset
        mask1 = (j1 >= 0) & (j1 < n)
        j2 = j - offset
        mask2 = (j2 >= 0) & (j2 < n)
        g = np.zeros(n)
        g[mask1] += (1.0 - h[j1[mask1]])
        g[mask2] -= h[j2[mask2]]
        grad += weights[k] * g
    
    return grad * dx
```

```python
def c5_smooth_grad(h, n, tau=100.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = corr.max()
    weights = np.exp(tau * (corr - max_c))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            # j+offset in [0,n) means j in [0, n-offset)
            # (1-h[j+offset]) contributes to grad[j] for j in [0, n-offset)
            grad[:n-offset] += wk * (1.0 - h[offset:])
            # j-offset in [0,n) means j in [offset, n)
            # -h[j-offset] contributes to grad[j] for j in [offset, n)
            grad[offset:] -= wk * h[:n-offset]
        else:
            o = -offset  # positive
            # j+offset in [0,n) means j in [o, n)
            grad[o:] += wk * (1.0 - h[:n-o])
            # j-offset in [0,n) means j in [0, n-o)
            grad[:n-o] -= wk * h[o:]
    
    return grad * dx
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = h.min() - target, h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - mid, 0.0, 1.0)

def c5_smooth_value_and_grad(h, n, tau):
    dx = 2.0 / n
    c = 1.0 - h
    corr = np.correlate(h, c, mode="full") * dx
    max_c = corr.max()
    shifted = tau * (corr - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    f = max_c + np.log(sum_exp) / tau
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            grad[:n - offset] += wk * (1.0 - h[offset:])
            grad[offset:] -= wk * h[:n - offset]
        else:
            o = -offset
            grad[o:] += wk * (1.0 - h[:n - o])
            grad[:n - o] -= wk * h[o:]
    grad *= dx
    return f, grad

def c5_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    tol = max(0.005 * max_val, 1e-8)
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        offset = k - (n - 1)
        if offset >= 0:
            grad[:n - offset] += (1.0 - h[offset:])
            grad[offset:] -= h[:n - offset]
        else:
            o = -offset
            grad[o:] += (1.0 - h[:n - o])
            grad[:n - o] -= h[o:]
    grad /= len(active)
    return grad * dx

def projected_gradient(h0, n, max_iters, tau_init, tau_final, t0, time_limit):
    h = project_feasible(h0.astype(float).copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stalled = 0
    alpha = 0.05
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        frac = it / max(max_iters - 1, 1)
        tau = tau_init + (tau_final - tau_init) * (frac ** 0.7)
        
        c_val = compute_c5(h, n)
        if c_val < best_c - 1e-13:
            best_c = c_val
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1
        if stalled > 800 and it > 500:
            break
        
        f_val, grad = c5_smooth_value_and_grad(h, n, tau)
        
        ok = False
        a = alpha
        for _ in range(30):
            h_new = project_feasible(h - a * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_val - 1e-14:
                h = h_new
                ok = True
                alpha = min(a * 1.3, 0.5)
                break
            a *= 0.5
        if not ok:
            alpha *= 0.6
            h = project_feasible(h - alpha * grad, n)
    
    return best_h, best_c

def slsqp_polish(h0, n):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        hp = np.clip(x, 0, 1)
        c = 1.0 - hp
        corr = np.correlate(hp, c, mode="full")
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    h = project_feasible(h0.astype(float).copy(), n)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       maxiter=150, options={'ftol': 1e-14, 'maxiter': 150})
        hf = project_feasible(res.x, n)
        return hf, compute_c5(hf, n)
    except Exception:
        return h, compute_c5(h, n)
