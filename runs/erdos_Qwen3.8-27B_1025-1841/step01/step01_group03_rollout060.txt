                xo = np.linspace(0, 2, mv.size, endpoint=False)
                xn = np.linspace(0, 2, n, endpoint=False)
                h0 = np.interp(xn, xo, mv)
    except Exception:
        pass
    if h0 is None:
        h0 = np.ones(n) * 0.5

    h0 = repair(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    # Phase 1: soft-max SLSQP with increasing alpha
    for alpha in [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]:
        if time.time() - t0 > budget_s * 0.55:
            break
        try:
            res = minimize(soft_obj, best_h, method='SLSQP', args=(n, alpha),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            ht = repair(res.x, n)
            c = compute_c5(ht)
            if c < best_c5:
                best_c5 = c
                best_h = ht.copy()
        except Exception:
            pass

    # Phase 2: exact max with SLSQP (multiple attempts)
    for attempt in range(4):
        if time.time() - t0 > budget_s * 0.68:
            break
        try:
            res = minimize(compute_c5, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            ht = repair(res.x, n)
            c = compute_c5(ht)
            if c < best_c5:
                best_c5 = c
                best_h = ht.copy()
        except Exception:
            pass

    # Phase 3: Random restarts
    for r in range(40):
        if time.time() - t0 > budget_s * 0.84:
            break
        scale = 0.2 * (0.85 ** r)
        hi = best_h + rng.normal(0, scale, n)
        hi = repair(hi, n)
        try:
            res = minimize(soft_obj, hi, method='SLSQP', args=(n, 200.0),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-13})
            ht = repair(res.x, n)
            c = compute_c5(ht)
            if c < best_c5:
                best_c5 = c
                best_h = ht.copy()
        except Exception:
            pass

    # Phase 4: Binary rounding heuristic
    h_bin = (best_h > 0.5).astype(float)
    h_bin = repair(h_bin, n)
    c_bin = compute_c5(h_bin)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin.copy()

    # Phase 5: Higher resolution
    for n2 in [160, 320, 640]:
        if time.time() - t0 > budget_s * 0.93:
            break
        _opt_n[0] = n2
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(xn, xo, best_h)
        h2 = repair(h2, n2)
        c5_2 = compute_c5(h2)
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            for alpha2 in [50.0, 200.0]:
                res = minimize(soft_obj, h2, method='SLSQP', args=(n2, alpha2),
                               bounds=bounds2, constraints=cons2,
                               options={'maxiter': 300, 'ftol': 1e-13})
                h2 = repair(res.x, n2)
                c5_2 = compute_c5(h2)
        except Exception:
            pass
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            n = n2

    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
