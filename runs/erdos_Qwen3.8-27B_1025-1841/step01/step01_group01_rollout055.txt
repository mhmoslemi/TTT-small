from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h, dx):
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return dx * np.max(corr)

def repair_h(h, n):
    h = np.clip(h.astype(float), 0, 1)
    target = n / 2.0
    for _ in range(500):
        current = np.sum(h)
        if abs(current - target) < 1e-14:
            break
        if current < 1e-15:
            return np.full(n, 0.5)
        h = h * (target / current)
        h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        step_size = diff / n
        if diff > 0:
            for i in range(n):
                add = min(step_size + 1e-10, 1.0 - h[i])
                if add > 0:
                    h[i] += add
                    diff -= add
        else:
            for i in range(n):
                sub = min(-step_size - 1e-10, h[i])
                if sub > 0:
                    h[i] -= sub
                    diff += sub
    return np.clip(h, 0, 1)

def smooth_max_c5(h, n, tau):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    vals = dx * corr
    return tau * logsumexp(vals / tau)

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def repair_h(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        if s < 1e-15:
            return np.full(n, 0.5)
        h = np.clip(h * (target / s), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            for i in range(n):
                add = min(1.0 - h[i], diff)
                if add > 1e-15:
                    h[i] += add
                    diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            for i in range(n):
                sub = min(h[i], -diff)
                if sub > 1e-15:
                    h[i] -= sub
                    diff += sub
                if abs(diff) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, tau, pen):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = dx * corr
    base = tau * logsumexp(vals / tau)
    return base + pen * (np.sum(h) - n / 2.0) ** 2

def hard_obj(x, n):
    h = repair_h(x, n)
    return compute_c5(h, 2.0 / n)

def interp_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=np.float64)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    return np.interp(x_new, x_old, h_old)

def optimize_smooth(h0, n, time_limit):
    dx = 2.0 / n
    best_h = repair_h(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    for tau in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() - t0 > time_limit:
            break
        res = minimize(smooth_obj, best_h, args=(n, tau, 1000.0),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9})
        h_opt = repair_h(res.x, n)
        c5_opt = compute_c5(h_opt, dx)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    return best_h, best_c5

def optimize_hard(h0, n, time_limit):
    dx = 2.0 / n
    best_h = repair_h(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)
    t0 = time.time()
    res = minimize(hard_obj, best_h, args=(n,),
                  method='Nelder-Mead',
                  options={'maxiter': 8000, 'xatol': 1e-11, 'fatol': 1e-13, 'adaptive': True})
    if time.time() - t0 > time_limit:
        pass
    h_opt = repair_h(res.x, n)
    c5_opt = compute_c5(h_opt, dx)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt
    return best_h, best_c5

def optimize_resolution(h0, n, time_limit):
    t_smooth = int(time_limit * 0.65)
    t_hard = max(30, time_limit - t_smooth)
    best_h, best_c5 = optimize_smooth(h0, n, t_smooth)
    if t_hard > 30:
        h2, c52 = optimize_hard(best_h, n, t_hard)
        if c52 < best_c5:
            best_h = h2
            best_c5 = c52
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=np.float64)
        n_init = len(h_init)
    except (NameError, TypeError):
        n_init = 45
        h_init = np.full(n_init, 0.5)

    h_init = repair_h(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, 2.0 / n_init)
    best_n = n_init

    # Phase 1: Optimize at initial resolution
    elapsed = time.time() - t_start
    time_left = budget_s - elapsed
    t1 = max(150, int(0.28 * time_left))
    h_opt, c5_opt = optimize_resolution(h_init, n_init, t1)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
        best_n = n_init

    # Phase 2: Higher resolutions
    for mult in [2, 4]:
        elapsed = time.time() - t_start
        time_left = budget_s - elapsed
        if time_left < 180:
            break
        n_new = n_init * mult
        h_interp = interp_h(best_h, best_n, n_new)
        h_interp = repair_h(h_interp, n_new)
        t_phase = max(100, int(0.28 * time_left))

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_new

    # Phase 3: Structured restarts
    elapsed = time.time() - t_start
    time_left = budget_s - elapsed
    restart_budget = max(0, int(time_left * 0.4))
    n_restart = best_n
    max_rs = 10
    t_each = max(20, restart_budget // max(rs, 1)) if max(rs) > 0 else 20
    for attempt in range(max_rs):
        elapsed = time.time() - t_start
        time_left = budget_s - elapsed
        if time_left < 60:
            break
        t_att = min(t_each, max(30, int(time_left * 0.08)))
        if t_att < 20:
            break
        if attempt == 0:
            h_rand = np.zeros(n_restart)
            h_rand[::2] = 1.0
            h_rand = repair_h(h_rand, n_restart)
        elif attempt == 1:
            h_rand = rng.permutation(np.array([1.0]*n_restart//2 + [0.0]*(n_restart - n_restart//2)))
            h_rand = repair_h(h_rand, n_restart)
        else:
            h_rand = rng.uniform(0.0, 1.0, n_restart)
            h_rand = repair_h(h_rand, n_restart)
        h_opt, c5_opt = optimize_resolution(h_rand, n_restart, t_att)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_restart

    # Phase 4: Final refinement
    elapsed = time.time() - t_start
    time_left = budget_s - elapsed
    if time_left > 60:
        h_opt, c5_opt = optimize_resolution(best_h, best_n, int(time_left - 10))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = best_n

    best_h = repair_h(best_h, best_n)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h.tolist(), final_c5, best_n)
