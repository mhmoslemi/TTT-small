import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-14:
            if diff > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    mc = np.max(corr)
    exp_c = np.exp(T * (corr - mc))
    S = np.sum(exp_c)
    w = exp_c / S
    f = (mc + np.log(S) / T) * dx
    
    # Gradient: df/dh[j] = dx * sum_s w_s * (g[j+s] - h[j-s]) [with bounds]
    grad = np.zeros(n)
    for s_idx in range(2 * n - 1):
        ws = w[s_idx]
        if ws < 1e-16:
            continue
        s = s_idx - (n - 1)  # actual shift
        if s >= 0:
            # g[j+s] for j in [0, n-1-s]
            grad[:n-s] += ws * g[s:]
            # -h[j-s] for j in [s, n)
            grad[s:] -= ws * h[:n-s]
        else:
            s2 = -s
            # g[j-s2] for j in [s2, n)
            grad[s2:] += ws * g[:n-s2]
            # -h[j+s2] for j in [0, n-1-s2)
            grad[:n-s2] -= ws * h[s2:]
    grad *= dx
    return f, grad

def lbgfs_optimize(h, T, maxiter, time_limit):
    n = len(h)
    start = time.time()
    
    def obj(x):
        f, _ = smooth_obj_grad(x, T)
        return f
    
    def grad(x):
        _, g = smooth_obj_grad(x, T)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
    return res.x

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=new_x[-1])  # Not periodic, use linear
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_descent(h, n_max_sweeps=50, time_limit=None):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_max_sweeps):
        if time_limit and time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            if time_limit and time.time() > time_limit:
                break
            old = h[j]
            candidates = [0.0, 1.0, 0.5,
                         max(0, old - 0.1), min(1, old + 0.1),
                         max(0, old - 0.05), min(1, old + 0.05)]
            best_trial = None
            best_val = best_c5
            for c in candidates:
                if abs(c - old) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                val = compute_c5(trial)
                if val < best_val - 1e-12:
                    best_val = val
                    best_trial = trial
            if best_trial is not None:
                h = best_trial
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):

            else:

            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, T, mu, n):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode='full')
    mc = np.max(corr)
    exp_c = np.exp(T * (corr - mc))
    S = np.sum(exp_c)
    f = (mc + np.log(S) / T) * dx
    f += mu * (np.sum(x) - n / 2.0) ** 2
    return f


def _smooth_grad(x, T, mu, n):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode='full')
    mc = np.max(corr)
    exp_c = np.exp(T * (corr - mc))
    S = np.sum(exp_c)
    w = exp_c / S
    grad = np.zeros(n)

            grad[:n - s2] += wm * g[s2:]
            grad[s2:] -= wm * x[:n - s2]

def _lbgfs_run(x0, T, mu, n, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj, x0, jac=lambda x: _smooth_grad(x, T, mu, n),
                   method='L-BFGS-B', bounds=bounds,
                   args=(), options={'maxiter': maxiter, 'gtol': 1e-10})
    return res.x


def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)


def random_binary_h(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def golden_ratio_h(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n) * (2.0 / n)
    h = ((x * phi) % 1.0 < 0.5).astype(float)
    return make_feasible(h, n)


def random_smooth_h(n, rng):
    h = rng.rand(n)
    return make_feasible(h, n)


def coord_descent(h, n_sweeps_max, time_limit):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps_max):
        if time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() > time_limit:
                break
            old = h[j]
            if old < 0.05:
                cands = [1.0, 0.5]
            elif old > 0.95:
                cands = [0.0, 0.5]
            else:
                cands = [0.0, 1.0, 0.5, max(0.0, old - 0.15), min(1.0, old + 0.15)]
            best_trial = None
            best_val = best_c5
            for c in cands:
                if abs(c - old) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                val = compute_c5(trial)
                if val < best_val - 1e-13:
                    best_val = val
                    best_trial = trial
            if best_trial is not None:
                h = best_trial
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5

    temps = [200.0, 100.0, 50.0, 30.0, 20.0, 12.0, 8.0]

    return best, best_c5
