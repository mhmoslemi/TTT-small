def compute_smooth_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights
    max_corr = np.max(corr)
    logits = T * (corr - max_corr)
    weights = np.exp(logits)
    weights /= np.sum(weights)
    
    # Compute gradient as weighted sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        L = k - (n - 1)
        # grad_L[j] = (1-h[j-L])*I[0<=j-L<n] - h[j+L]*I[0<=j+L<n]
        if L >= 0:

            grad[L:] += weights[k] * (1.0 - h[:n-L])
            # -h[j+L] valid for j in [0, n-1-L]
            grad[:n-L] += weights[k] * (-h[L:])
        else:

            # -h[j+L] = -h[j-|L|] valid for j in [|L|, n-1]

            grad[:n-abs_L] += weights[k] * (1.0 - h[abs_L:])
            grad[abs_L:] += weights[k] * (-h[:n-abs_L])
    
    return grad * dx, np.max(corr) * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_grad(h, T):
    """Compute gradient of smoothed max correlation."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # Softmax weights (shifted for numerical stability)
    max_c = np.max(corr)
    logits = T * (corr - max_c)
    weights = np.exp(logits)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
        wsum = 1.0
    weights /= wsum
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(len(corr)):

        if L >= 0:

                 alpha_start=0.5, alpha_end=0.01, momentum=0.9,
                 time_limit=None, start_time=None):

    n = len(h)
    h_curr = h.copy()
    v = np.zeros(n)
    
    for it in range(n_max_iter):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        
        # Current temperature
        frac = it / max(n_max_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        grad = compute_grad(h_curr, T)
        
        # Momentum update
        v = momentum * v - alpha * grad
        h_new = h_curr + v
        h_new = make_feasible(h_new, n)
        
        # Check improvement on true objective
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h_curr)
        
        if c5_new <= c5_old + 1e-12:
            h_curr = h_new
        else:
            # Rejected step, reduce momentum
            v *= 0.5
    
    return h_curr

    best = h.copy()

        if time_limit is not None and time.time() - start_time > time_limit:

            # Try a few candidates

            if 0 < old_val < 1:

            candidates = [c for c in candidates if 0 <= c <= 1 and c != old_val]

        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    max_c = np.max(corr)
    logits = T * (corr - max_c)
    weights = np.exp(logits)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
        wsum = 1.0
    weights /= wsum
    active = np.where(weights > 1e-7)[0]
    grad = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        w = weights[k]
        if L >= 0:
            if L < n:
                grad[L:] += w * omh[:n - L]
                grad[:n - L] -= w * h[L:]
        else:
            aL = -L
            if aL < n:
                grad[:n - aL] += w * omh[aL:]
                grad[aL:] -= w * h[:n - aL]
    return grad * dx

def anneal_grad_desc(h, max_iter, T_start, T_end, a_start, a_end, mom,
                     time_limit, start_time):
    n = len(h)
    h_c = h.copy()
    v = np.zeros(n)
    best = h_c.copy()
    best_c5 = compute_c5(h_c)
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        frac = it / max(max_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        a = a_start * (a_end / a_start) ** frac
        g = compute_grad(h_c, T)
        v = mom * v - a * g
        h_new = make_feasible(h_c + v, n)
        c5 = compute_c5(h_new)
        if c5 <= best_c5 + 1e-13:
            h_c = h_new
            if c5 < best_c5:
                best_c5 = c5
                best = h_c.copy()
        else:
            v *= 0.3
            h_try = make_feasible(h_c - 0.05 * a * g, n)
            c5t = compute_c5(h_try)
            if c5t < best_c5:
                best_c5 = c5t
                best = h_try.copy()
                h_c = h_try
    return best

def random_binary_h(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def block_binary_h(n, rng):
    bs = rng.randint(3, 12)
    nb = n // bs
    blocks = (rng.rand(nb) < 0.5).astype(float)
    h = np.repeat(blocks, bs)[:n]
    return make_feasible(h, n)

def slsqp_polish(h, maxiter, T, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def cons_fun(hv):
        return np.sum(hv) - n / 2.0
    def cons_jac(hv):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return make_feasible(res.x, n)
    except Exception:
        return h

def coord_polish(h, max_rounds, time_limit, start_time):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:
                break
            old = best[j]
            cands = [0.0, 1.0]
            if 0 < old < 1:
                cands.extend([max(0, old - 0.05), min(1, old + 0.05),
                              max(0, old - 0.02), min(1, old + 0.02)])
            for cv in cands:
                if abs(cv - old) < 1e-10:
                    continue
                trial = best.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best = trial
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

        h1 = upsample_h(best_h, 160)
        h1 = anneal_grad_desc(h1, 400, 20, 100, 0.4, 0.02, 0.8,

        c = compute_c5(h1)
        if c < best_c5:

    if rem() > 80:
        h2 = upsample_h(best_h, 320)
        h2 = anneal_grad_desc(h2, 800, 30, 150, 0.25, 0.008, 0.85,
                              rem() - 20, start_time)
        c = compute_c5(h2)
        if c < best_c5:
            best_c5, best_h = c, h2

    # Phase 3: SLSQP polish at n=320
    if rem() > 60:
        h3 = slsqp_polish(best_h, 300, 60, rem() - 15, start_time)
        c = compute_c5(h3)
        if c < best_c5:
            best_c5, best_h = c, h3

    # Phase 4: Multiple restarts at n=320
    for r in range(10):
        if rem() < 40:
            break
        n = 320
        if r == 0:
            hs = upsample_h(best_h, n) + rng.randn(n) * 0.03
        elif r == 1:
            hs = random_binary_h(n, rng)
        elif r == 2:
            hs = block_binary_h(n, rng)
        elif r == 3:
            hs = upsample_h(best_h, n) * 0.7 + 0.3 * random_binary_h(n, rng)
        elif r == 4:
            hs = upsample_h(best_h, n) + rng.randn(n) * 0.1
        elif r == 5:
            hs = random_binary_h(n, rng)
            for _ in range(rng.randint(10, 30)):
                hs[rng.randint(n)] = 1.0 - hs[rng.randint(n)]
        elif r == 6:
            freqs = rng.randint(1, 8, size=3)
            x = np.linspace(0, 2, n)
            hs = np.zeros(n)
            for f in freqs:
                hs += np.sin(2 * np.pi * f * x / 2.0 + rng.uniform(0, 6.28))
            hs = 0.5 + 0.4 * hs / (np.max(np.abs(hs)) + 1e-10)
            hs = np.clip(hs, 0, 1)
        elif r == 7:
            hs = upsample_h(best_h, n)
            idx = rng.randint(0, n, size=40)
            hs[idx] = rng.rand(40)
        else:
            hs = upsample_h(best_h, n) + rng.randn(n) * 0.06
        hs = make_feasible(hs, n)
        tl = min(rem() - 15, 45)
        if tl < 10:

        h_r = anneal_grad_desc(hs, 400, 25, 120, 0.2, 0.008, 0.85, tl, start_time)
        c = compute_c5(h_r)
        if c < best_c5:
            best_c5, best_h = c, h_r

    # Phase 5: Upsample to 640 and optimize
    if rem() > 150:
        h4 = upsample_h(best_h, 640)
        h4 = anneal_grad_desc(h4, 1500, 40, 250, 0.12, 0.003, 0.9,
                              rem() - 40, start_time)
        c = compute_c5(h4)
        if c < best_c5:
            best_c5, best_h = c, h4

    # Phase 6: SLSQP polish on best
    if rem() > 50:
        n = len(best_h)
        T_sl = 80.0 if n <= 320 else 120.0
        h6 = slsqp_polish(best_h, 200, T_sl, rem() - 10, start_time)
        c = compute_c5(h6)
        if c < best_c5:
            best_c5, best_h = c, h6

    # Phase 7: Try n=480
    if rem() > 120:
        h7 = upsample_h(best_h, 480)
        h7 = anneal_grad_desc(h7, 1000, 35, 180, 0.15, 0.005, 0.88,
                              rem() - 30, start_time)
        c = compute_c5(h7)
        if c < best_c5:
            best_c5, best_h = c, h7

    # Phase 8: Coordinate polish
    if rem() > 30:
        best_h, best_c5 = coord_polish(best_h, 4, rem() - 5, start_time)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
