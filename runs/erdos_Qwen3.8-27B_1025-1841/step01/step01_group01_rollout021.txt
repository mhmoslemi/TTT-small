import time

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max()

def compute_c5_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_bounded(y, S):
    lo = -y.max() - 1.0
    hi = 1.0 - y.min() + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(y + mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def grad_c5_component(h, k):
    n = len(h)
    m = k - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        idx1 = np.arange(m, n)
        grad[idx1] += (1.0 - h[idx1 - m])
        idx2 = np.arange(0, n - m)
        grad[idx2] -= h[idx2 + m]
    else:
        p = -m
        idx1 = np.arange(0, n - p)
        grad[idx1] += (1.0 - h[idx1 + p])
        idx2 = np.arange(p, n)
        grad[idx2] -= h[idx2 - p]
    return grad
