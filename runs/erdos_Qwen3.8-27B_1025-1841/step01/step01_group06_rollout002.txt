        grad[lo:lo+m] += (1.0 - h[lo:])
        grad[:m] -= h[lo:]
    return grad

def nm_obj_func(x, dx):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
