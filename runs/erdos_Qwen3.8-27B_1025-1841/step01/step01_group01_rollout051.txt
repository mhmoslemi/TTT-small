    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h = h + (diff / tot) * room
        else:
            room = h
            tot = room.sum()
            if tot > 1e-15:
                h = h + (diff / tot) * room
    return np.clip(h, 0.0, 1.0)

def _smooth_max(arr, beta):
    m = np.max(arr)
    shifted = np.clip((arr - m) * beta, -60.0, 60.0)
    return float(m + np.log(np.exp(beta) * np.mean(np.exp(shifted))) / beta)

def obj_smooth(x, n, dx, beta):
    h = project_feasible(x, n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return _smooth_max(corr, beta)

def obj_hard(x, n, dx):
    h = project_feasible(x, n)
    return compute_c5(h, dx)

def binary_local_search(h_bin, n, dx, rng, t0, budget_s, max_iter=5000):
    best_c5 = compute_c5(h_bin, dx)
    for _ in range(max_iter):
        if time.time() - t0 > budget_s - 40:
            break
        ones_idx = np.where(h_bin == 1.0)[0]
        zeros_idx = np.where(h_bin == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return h_bin, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(90) * 0.5

    best_h = h0.copy()
    best_n = len(h0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    n_candidates = [best_n, 90, 120, 180, 240, 360]
    n_list = sorted(set(nc for nc in n_candidates if nc >= 30))

    for n in n_list:
        if time.time() - t0 > budget_s - 80:
            break

        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n

        if n == best_n:
            h_work = best_h.copy()
        elif n == len(h0):
            h_work = h0.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n)
            f_interp = interp1d(x_old, best_h, kind='linear')
            h_work = f_interp(x_new)
            h_work = project_feasible(h_work, n)

        for beta in [15, 40, 80, 160, 320]:
            if time.time() - t0 > budget_s - 80:
                break
            try:
                res = minimize(
                    partial(obj_smooth, n=n, dx=dx, beta=beta),
                    h_work, method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 400, 'maxfun': 2000, 'ftol': 1e-14}
                )
                h_work = project_feasible(res.x, n)
                c5_new = compute_c5(h_work, dx)
                if c5_new < best_c5:
                    best_h = h_work.copy()
                    best_n = n
                    best_c5 = c5_new
            except Exception:
                pass

        if time.time() - t0 < budget_s - 60:
            try:
                res = minimize(
                    partial(obj_hard, n=n, dx=dx),
                    h_work, method='Nelder-Mead',
                    options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-12}
                )
                h_work = project_feasible(res.x, n)
                c5_new = compute_c5(h_work, dx)
                if c5_new < best_c5:
                    best_h = h_work.copy()
                    best_n = n
                    best_c5 = c5_new
            except Exception:
                pass

        if time.time() - t0 < budget_s - 60:
            h_bin = (h_work > 0.5).astype(float)
            h_bin = project_feasible(h_bin, n)
            h_bin, c5_bin = binary_local_search(h_bin, n, dx, rng, t0, budget_s, max_iter=3000)
            if c5_bin < best_c5:
                best_h = h_bin.copy()
                best_n = n
                best_c5 = c5_bin

        if time.time() - t0 < budget_s - 120:
            h_rand = (rng.rand(n) > 0.5).astype(float)
            h_rand = project_feasible(h_rand, n)
            h_rand, c5_rand = binary_local_search(h_rand, n, dx, rng, t0, budget_s, max_iter=2000)
            if c5_rand < best_c5:
                best_h = h_rand.copy()
                best_n = n
                best_c5 = c5_rand

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h.tolist(), best_c5, best_n
