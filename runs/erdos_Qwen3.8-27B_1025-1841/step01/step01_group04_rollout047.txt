    for _ in range(100):

        d = target_sum - s
        if abs(d) < 1e-14:

import numpy as np
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k_full, n):
    """Gradient of c[k_full] w.r.t. h, where k_full is index in full corr array."""
    lag = k_full - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # c = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        # dc/dh[i+lag] += (1-h[i]) for i in 0..n-1-lag
        # dc/dh[i] += -h[i+lag] for i in 0..n-1-lag
        idx = np.arange(n - lag)
        g[idx + lag] += (1.0 - h[idx])
        g[idx] -= h[idx + lag]
    else:
        j = -lag
        # c = sum_{i=j}^{n-1} h[i-j]*(1-h[i])
        # dc/dh[i-j] += (1-h[i]) for i in j..n-1
        # dc/dh[i] += -h[i-j] for i in j..n-1
        idx = np.arange(j, n)
        g[idx - j] += (1.0 - h[idx])
        g[idx] -= h[idx - j]
    return g

def smooth_grad(h, n, T=10.0, top_m=5):
    """Compute softmax-weighted gradient of max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use top_m for efficiency
    top_idx = np.argsort(corr)[-top_m:][::-1]
    w = corr[top_idx]
    w = np.exp((w - w.max()) / T)
    w = w / w.sum()
    g = np.zeros(n)
    for k, wi in zip(top_idx, w):
        g += wi * grad_lag(h, k, n)
    return g, float(corr.max())

    return best_h, best_c

def coord_descent(h0, n, target, deadline, n_sweeps=100):
    h = h0.copy()
    best_c = evaluate_c5(h)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Randomize order
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            # Try a few values
            for new_val in [old - 0.1, old - 0.05, old + 0.05, old + 0.1, 0.0, 1.0]:
                if new_val < 0 or new_val > 1:

                h[j] = new_val

                c = corr.max()
                if c < best_c:

                    improved = True
                else:
                    h[j] = old
                    break
            # Re-project if needed
            s = np.sum(h)
            d = target - s

                mask = (h < 1.0) if d > 0 else (h > 0.0)
                if np.sum(mask) > 0:
                    h[mask] += d / np.sum(mask)
                    h = np.clip(h, 0.0, 1.0)
        
        if not improved and sweep > 10:
            break
    
    c = evaluate_c5(h)
    if c < best_c:
        best_c = c
    return h, best_c

def slsqp_polish(h0, n, target, deadline):
    """SLSQP optimization with exact objective."""
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return c.max()
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        k_max = np.argmax(corr)
        return grad_lag(x, k_max, n)
    
    def const(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': const,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, target)
            c_new = evaluate_c5(h_new)
            return h_new, c_new
    except Exception:
        pass
    return h0, evaluate_c5(h0)

    return np.interp(xt, xf, h)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k_full, n):
    lag = k_full - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        g[idx + lag] += (1.0 - h[idx])
        g[idx] -= h[idx + lag]
    else:
        j = -lag
        idx = np.arange(j, n)
        g[idx - j] += (1.0 - h[idx])
        g[idx] -= h[idx - j]
    return g

def subgrad_opt(h0, n, target, deadline, alpha0=0.05, cooling=0.99997, T0=5.0, top_m=5):
    h = h0.copy()
    alpha = alpha0
    T = T0
    best_c = evaluate_c5(h)
    best_h = h.copy()
    ncorr = 2 * n - 1
    lags = np.arange(ncorr) - (n - 1)
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(corr.max())
        top_idx = np.argsort(corr)[-top_m:][::-1]
        w = corr[top_idx]
        w = np.exp((w - w.max()) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(w)) / len(w)
        else:
            w = w / w_sum
        g = np.zeros(n)
        for k, wi in zip(top_idx, w):
            g += wi * grad_lag(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        alpha_cur = alpha
        h_new = h - alpha_cur * g
        h_new = project_feasible(h_new, target)
        c_new = evaluate_c5(h_new)
        if c_new < best_c + 0.01:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            alpha *= 0.9
        it += 1
        alpha *= cooling
        if it % 2000 == 0:
            T *= 0.85
    c_final = evaluate_c5(h)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def coord_descent_fast(h0, n, target, deadline, max_sweeps=500):
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c = float(corr.max())
    ncorr = 2 * n - 1
    lags = np.arange(ncorr) - (n - 1)
    deltas = [-0.15, -0.1, -0.05, -0.02, 0.02, 0.05, 0.1, 0.15, -0.3, 0.3]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            m1 = (j - lags >= 0) & (j - lags < n)
            m2 = (j + lags >= 0) & (j + lags < n)
            partial = np.zeros(ncorr)
            partial[m1] += (1.0 - h[j - lags[m1]])
            partial[m2] -= h[j + lags[m2]]
            best_nc = corr.max()
            best_dv = 0.0
            for dv in deltas:
                nv = old + dv
                if nv < -1e-9 or nv > 1.0 + 1e-9:
                    continue
                nv = float(np.clip(nv, 0.0, 1.0))
                nc = (corr + dv * partial).max()
                if nc < best_nc:
                    best_nc = nc
                    best_dv = dv
            if best_dv != 0.0:
                h[j] = float(np.clip(old + best_dv, 0.0, 1.0))
                corr += best_dv * partial
                if best_nc < best_c:
                    best_c = float(best_nc)
                    improved = True
        s = np.sum(h)
        d = target - s
        if abs(d) > 1e-9:
            mask = (h < 1.0) if d > 0 else (h > 0.0)
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += d / cnt
                h = np.clip(h, 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                bc = float(corr.max())
                if bc < best_c:
                    best_c = bc
        if not improved and sweep > 5:
            break
    return h, best_c

def slsqp_polish(h0, n, target, deadline):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(c.max())
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        k_max = int(np.argmax(corr))
        return grad_lag(x, k_max, n)
    def const_fun(x):
        return np.sum(x) - target
    def const_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': const_fun, 'jac': const_jac}],
                       options={'maxiter': 300, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, target)
            c_new = evaluate_c5(h_new)
            return h_new, c_new
    except Exception:
        pass
    return h0, evaluate_c5(h0)

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def softmax_obj(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(60, 0.5)

    n0 = len(h_start)
    target0 = n0 / 2.0
    best_h = project_feasible(h_start, target0)
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Aggressive subgradient (300s)
    dl1 = min(deadline, t0 + 300)
    h1, c1 = subgrad_opt(best_h, n0, target0, dl1, alpha0=0.08, cooling=0.99999, T0=8.0, top_m=8)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: Coordinate descent refinement (250s)
    dl2 = min(deadline, t0 + 550)
    h2, c2 = coord_descent_fast(best_h, n0, target0, dl2, max_sweeps=1000)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2

    # Phase 3: Fine subgradient (200s)
    dl3 = min(deadline, t0 + 750)
    h3, c3 = subgrad_opt(best_h, n0, target0, dl3, alpha0=0.01, cooling=0.999995, T0=2.0, top_m=3)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3

    # Phase 4: More coordinate passes (150s)
    dl4 = min(deadline, t0 + 900)
    h4, c4 = coord_descent_fast(best_h, n0, target0, dl4, max_sweeps=800)
    if c4 < best_c5:
        best_c5 = c4
        best_h = h4

    # Phase 5: SLSQP polish
    if time.time() < deadline - 120:
        dl5 = min(deadline, t0 + 1100)
        h5, c5 = slsqp_polish(best_h, n0, target0, dl5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5

    # Phase 6: Try higher resolution
    if time.time() < deadline - 80:
        n_hi = n0 * 2
        target_hi = n_hi / 2.0
        h_hi = interp_to(best_h, n0, n_hi)
        h_hi = project_feasible(h_hi, target_hi)
        dl6 = min(deadline, t0 + budget_s - 40)
        hh, ch = subgrad_opt(h_hi, n_hi, target_hi, dl6, alpha0=0.03, cooling=0.99999, T0=4.0, top_m=5)
        ch = evaluate_c5(hh)
        if ch < best_c5:
            best_c5 = ch
            best_h = hh

    # Final: one more coord pass at best resolution
    if time.time() < deadline - 5:
        n_f = len(best_h)
        t_f = n_f / 2.0
        hf, cf = coord_descent_fast(best_h, n_f, t_f, deadline, max_sweeps=200)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
