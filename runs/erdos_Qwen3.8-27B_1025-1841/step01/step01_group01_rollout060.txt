import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def soft_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    val = dx * (max_c + np.log(np.sum(exp_c)) / beta)
    # Weights for gradient
    w = exp_c / np.sum(exp_c)
    return val, w

def grad_soft_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    _, w = soft_c5(h, beta)
    # w has length 2n-1, indexed by m=0,...,2n-2
    # Shift s = m - (n-1), ranging from -(n-1) to (n-1)
    
    # c_s = sum_{valid n} h[n+s] * (1-h[n])
    # d(c_s)/d(h[j]) = (1-h[n]) where n+s=j, i.e., n=j-s: contributes (1-h[j-s]) if valid
    #                  + h[n+s] * (-1) where n=j: contributes -h[j+s] if valid
    # So d(c_s)/d(h[j]) = (1-h[j-s])_{j-s valid} - h[j+s]_{j+s valid}
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        s = m - (n-1)
        # For each j, contribution to grad[j] from weight w[m]:
        # dx * w[m] * [(1-h[j-s]) if 0<=j-s<n else 0] - [h[j+s] if 0<=j+s<n else 0]

        # Second term: for j in range(max(0,-s), min(n, n-s)), subtract h[j+s]
        
        j_min1 = max(0, s)
        j_max1 = min(n, n + s)
        if j_min1 < j_max1:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-14:
        return h
    # Bisection for lambda: find lambda s.t. sum(clip(h + lambda, 0, 1)) = target_sum
    # (if diff > 0, lambda > 0; if diff < 0, lambda < 0)
    if diff > 0:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(corr.max())
    ec = np.exp(beta * (corr - mc))
    sw = float(ec.sum())
    obj = dx * (mc + np.log(sw) / beta)
    w = ec / sw
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wm = w[m] * dx
        if wm < 1e-14:
            continue
        a = max(0, s)
        b = min(n, n + s)
        if a < b:
            grad[a:b] += wm * (1.0 - h[a - s:b - s])
        c = max(0, -s)
        d = min(n, n - s)
        if c < d:
            grad[c:d] -= wm * h[c + s:d + s]
    return obj, grad

def optimize(h, target_sum, time_limit, seed, beta_start=5.0, beta_end=500.0):
    t_start = time.time()
    best = h.copy()
    best_c5 = compute_c5(h)
    beta = beta_start
    it = 0
    total_iters = int(time_limit * 30000.0 / len(h))
    total_iters = max(total_iters, 500)
    
    while time.time() - t_start < time_limit:
        # Anneal beta
        frac = min(1.0, it / total_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        lr = 0.8 / (beta * 0.1 + 1.0)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        
        h = h_new
        it += 1
    
    return best, best_c5
