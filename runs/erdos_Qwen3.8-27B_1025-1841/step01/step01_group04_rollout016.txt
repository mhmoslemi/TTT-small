from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_grad_softmax(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = T * corr * dx
    m = np.max(x)
    exp_x = np.exp(x - m)
    sum_exp = np.sum(exp_x)
    w = exp_x / sum_exp
    obj = (m + np.log(sum_exp)) / T * dx
    
    g = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - n + 1
        wk = w[k]
        if wk < 1e-14:
            continue
        if s >= 0:
            if s < n:
                grad[:n-s] += wk * g[s:]
                grad[s:] -= wk * h[:n-s]
        else:
            t = -s
            if t < n:
                grad[t:] += wk * g[:n-t]
                grad[:n-t] -= wk * h[t:]
    grad *= dx
    return obj, grad

def obj_grad_exact_max(h):
    """Gradient at the exact max (subgradient)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    obj = corr[k_star] * dx
    
    s = k_star - n + 1
    g = 1.0 - h
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n-s] += g[s:]
            grad[s:] -= h[:n-s]
    else:
        t = -s
        if t < n:
            grad[t:] += g[:n-t]
            grad[:n-t] -= h[t:]
    grad *= dx
    return obj, grad

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:

        if d > 0:
            mask = h < 1.0

        step = d / cnt
        # Don't overshoot

            step = min(step, max_step) * 1.0  # just clip after

    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h)
    return project(h_new)

def proj_grad_optimize(h0, T, n_steps, deadline, lr_init=0.01):
    """Projected gradient with backtracking line search"""
    n = len(h0)
    h = h0.copy()
    best_obj = obj_grad_softmax(h, T)[0]
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        obj, grad = obj_grad_softmax(h, T)

        alpha = lr
        for _ in range(20):
            h_new = h - alpha * grad
            h_new = project(h_new)
            new_obj, _ = obj_grad_softmax(h_new, T)
            if new_obj < obj:
                h = h_new
                break
            alpha *= 0.5
        else:
            # If no improvement, reduce learning rate
            lr *= 0.5
            break
        
        # Adaptive learning rate
        if new_obj < best_obj:
            best_obj = new_obj
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
        
        if new_obj < obj - 1e-15:
            obj = new_obj
    
    return h

def slsqp_with_grad(h0, T, maxiter, deadline):

    def fun(x):

        return obj_grad_softmax(x, T)[1]

        if res.x is not None and np.all(np.isfinite(res.x)):

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def adj_grad(h, w):
    n = len(h)
    g = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            if s < n and n - s > 0:
                grad[s:] += wk * g[:n - s]
                grad[:n - s] -= wk * h[s:]
        else:
            t = -s
            if t < n and n - t > 0:
                grad[:n - t] += wk * g[t:]
                grad[t:] -= wk * h[:n - t]
    return grad

def obj_grad_softmax(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = T * corr * dx
    m = np.max(x)
    exp_x = np.exp(x - m)
    sum_exp = np.sum(exp_x)
    w = exp_x / sum_exp
    obj = (m + np.log(sum_exp)) / T * dx
    grad = adj_grad(h, w) * dx
    return obj, grad

        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new)

def slsqp_opt(h0, T, maxiter, deadline):
    n = len(h0)
    h0 = project(h0)

    def fun(x):
        return obj_grad_softmax(x, T)[0]

    def fprime(x):
        return obj_grad_softmax(x, T)[1]

    def c_eq(x):
        return np.sum(x) - n / 2.0

    def c_eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h0, jac=fprime, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x)
    except Exception:
        pass
    return h0

def proj_grad_opt(h0, T, n_steps, deadline, lr=0.005):
    n = len(h0)
    h = h0.copy()
    best_obj = obj_grad_softmax(h, T)[0]
    cur_lr = lr
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        obj, grad = obj_grad_softmax(h, T)
        found = False
        alpha = cur_lr
        for _ in range(25):
            h_new = project(h - alpha * grad)
            new_obj, _ = obj_grad_softmax(h_new, T)
            if new_obj < obj - 1e-16:
                h = h_new
                found = True
                break
            alpha *= 0.5
        if not found:
            cur_lr *= 0.5
            if cur_lr < 1e-7:
                break
        else:
            if new_obj < best_obj - 1e-10:
                cur_lr = min(cur_lr * 1.1, 0.05)
            best_obj = min(best_obj, new_obj)
    return h

def topk_subgrad_step(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-K:]
    w = np.zeros(2 * n - 1)
    w[top_k] = 1.0 / K
    grad = adj_grad(h, w) * dx
    return corr[top_k].max() * dx, grad

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(336, 0.5)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    n = len(h0)

    # Phase 1: Anneal at initial resolution
    T_sched = [15, 30, 60, 120, 250, 500, 1000]
    for T in T_sched:
        if time.time() > deadline - 400:
            break
        h0 = slsqp_opt(h0, T, maxiter=250, deadline=deadline)
        c5 = evaluate_c5(h0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()
        if time.time() < deadline - 350:
            h0 = proj_grad_opt(h0, T, 150, deadline, lr=0.004)
            c5 = evaluate_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()

    # Phase 2: Upsample to 2x and optimize
    if time.time() < deadline - 300:
        n2 = n * 2
        h0 = upsample(best_h, n2)
        n = n2
        for T in [20, 60, 150, 400, 800]:
            if time.time() > deadline - 250:
                break
            h0 = slsqp_opt(h0, T, maxiter=200, deadline=deadline)
            c5 = evaluate_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
            if time.time() < deadline - 200:
                h0 = proj_grad_opt(h0, T, 120, deadline, lr=0.003)
                c5 = evaluate_c5(h0)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h0.copy()

    # Phase 3: Upsample to 3x (from original) and optimize
    if time.time() < deadline - 200:
        n3 = len(h0) * 3 // 2
        h0 = upsample(best_h, n3)
        n = n3
        for T in [25, 70, 200, 500]:
            if time.time() > deadline - 150:
                break
            h0 = slsqp_opt(h0, T, maxiter=200, deadline=deadline)
            c5 = evaluate_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
            if time.time() < deadline - 120:
                h0 = proj_grad_opt(h0, T, 100, deadline, lr=0.003)
                c5 = evaluate_c5(h0)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h0.copy()

    # Phase 4: Top-K subgradient refinement
    if time.time() < deadline - 40:
        h_ref = best_h.copy()
        for K in [3, 5]:
            lr = 0.002
            for _ in range(300):
                if time.time() > deadline - 5:
                    break
                _, grad = topk_subgrad_step(h_ref, K)
                h_new = project(h_ref - lr * grad)
                c5_new = evaluate_c5(h_new)
                if c5_new < evaluate_c5(h_ref) - 1e-12:
                    h_ref = h_new
                else:
                    lr *= 0.8
                    if lr < 1e-6:
                        break
            c5 = evaluate_c5(h_ref)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()

    # Phase 5: Random restarts
    n_best = len(best_h)
    for attempt in range(15):
        if time.time() > deadline - 10:
            break
        sigma = 0.04
        h_pert = best_h + np.random.randn(n_best) * sigma
        h_pert = project(h_pert)
        h_pert = slsqp_opt(h_pert, 150, maxiter=100, deadline=deadline)
        c5 = evaluate_c5(h_pert)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert.copy()

    # Final validation
    best_h = project(best_h)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
