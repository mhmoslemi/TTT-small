import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project(h, n_points):
    """Project to satisfy constraints: 0 <= h <= 1, sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    target = n_points / 2.0
    current = h.sum()
    if current == 0:
        h = np.ones(n_points) * 0.5
    else:
        # Scale to meet integral constraint
        alpha = target / current
        h = h * alpha
        h = np.clip(h, 0, 1)
        # If still not matching, adjust
        if h.sum() < target:
            # Increase values where h < 1
            diff = target - h.sum()
            mask = h < 1.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0, 1)
        elif h.sum() > target:
            diff = h.sum() - target
            mask = h > 0.0
            if mask.sum() > 0:
                h[mask] -= diff / mask.sum()
                h = np.clip(h, 0, 1)
    return h

def subgradient(h, n_points):
    """Compute subgradient of C5(h) = max_k corr_k(h) * dx"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_lag = np.argmax(corr)  # 0 to 2n-2, actual lag = best_lag - (n-1)
    lag = best_lag - (n - 1)
    
    grad = np.zeros(n_points)
    # f_lag(h) = sum_i h[i] * (1 - h[i+lag])
    # d/dh[l] = (1 - h[l+lag]) if l+lag valid (from h[i] where i=l)
    #           - h[l-lag] if l-lag valid (from h[i+lag] where i+lag=l)
    
    if lag >= 0:
        # h[i] * (1-h[i+lag]): i ranges 0 to n-1-lag
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] -= h[i]
    else:
        # lag < 0, let m = -lag > 0
        # h[i] * (1-h[i-m]): i ranges m to n-1
        m = -lag
        for i in range(m, n):
            grad[i] += (1.0 - h[i - m])
            grad[i - m] -= h[i]
    
    return grad * dx

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-10:
        return np.ones(n) * 0.5
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.any():
                h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if mask.any():
                h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_subgradient(h, n):
    corr = compute_corr(h)
    best = int(np.argmax(corr))
    lag = best - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        m = -lag
        idx = np.arange(m, n)
        grad[idx] += (1.0 - h[idx - m])
        grad[idx - m] -= h[idx]
    return grad * dx, best

import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= (0.5 * n) / s
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - h.sum()
    if abs(d) > 1e-14:
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        if mask.any():
            h[mask] += d / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _sg(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    bi = int(np.argmax(c))
    lag = bi - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        g[idx] += 1.0 - h[idx + lag]
        g[idx + lag] -= h[idx]
    else:
        m = -lag
        idx = np.arange(m, n)
        g[idx] += 1.0 - h[idx - m]
        g[idx - m] -= h[idx]
    return g * (2.0 / n)

def _sg_multi(h, n, num_lags=5):
    c = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(c)[::-1][:num_lags]
    g = np.zeros(n)
    for bi in top:
        lag = int(bi) - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            g[idx] += 1.0 - h[idx + lag]
            g[idx + lag] -= h[idx]
        else:
            m = -lag
            idx = np.arange(m, n)
            g[idx] += 1.0 - h[idx - m]
            g[idx - m] -= h[idx]
    return g * (2.0 / n) / num_lags

def _subgradient_opt(h, n, t0, deadline, rs, lr_init=0.03):
    mom = np.zeros(n)
    lr = lr_init
    best = h.copy()
    best_c = _c5(best)
    stall = 0
    while time.time() < deadline:
        g = _sg_multi(h, n, 3)
        mom = 0.88 * mom + 0.12 * g
        h = _proj(h - lr * mom, n)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            h = best + rs.randn(n) * 0.008
            h = _proj(h, n)
            lr *= 0.6
            stall = 0
        lr = max(lr * 0.99997, 0.0008)
    return best, best_c

def _sa_binary(hb, n, t0, deadline, rs, T0=0.04):
    T = T0
    c5b = _c5(hb)
    best_b = hb.copy()
    best_c5 = c5b
    zeros_list = np.where(hb == 0.0)[0]
    it = 0
    while time.time() < deadline:
        it += 1
        ones = np.where(hb == 1.0)[0]
        if len(ones) == 0 or len(zeros_list) == 0:
            break
        i = int(ones[rs.randint(len(ones))])
        j = int(zeros_list[rs.randint(len(zeros_list))])
        hb[i], hb[j] = hb[j], hb[i]
        zeros_list = np.where(hb == 0.0)[0]
        c = _c5(hb)
        delta = c - c5b
        if delta < 0 or (T > 1e-8 and rs.rand() < np.exp(-delta / T)):
            c5b = c
            if c5b < best_c5:
                best_c5 = c5b
                best_b = hb.copy()
        else:
            hb[i], hb[j] = hb[j], hb[i]
            zeros_list = np.where(hb == 0.0)[0]
        T *= 0.99992
    return best_b, best_c5

def _make_binary(h, n, rs):
    hb = (h >= 0.5).astype(float)
    ns = int(hb.sum())
    tgt = n // 2
    if ns > tgt:
        for idx in rs.choice(np.where(hb == 1)[0], ns - tgt, replace=False):
            hb[idx] = 0.0
    elif ns < tgt:
        for idx in rs.choice(np.where(hb == 0)[0], tgt - ns, replace=False):
            hb[idx] = 1.0
    return hb

def run(seed=42, budget_s=1300, **kwargs):
