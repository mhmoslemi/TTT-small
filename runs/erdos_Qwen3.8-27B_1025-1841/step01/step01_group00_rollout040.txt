def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_to(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

    grad *= dx
    obj = dx * (M + np.log(S) / tau)
    return obj, grad

def nesterov_opt(h, n, tau, max_iters, time_limit, lr=0.1):

    v = np.zeros(n)
    beta = 0.9
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # Nesterov step
        h_y = project_to_feasible(h + beta * v, n)
        obj, grad = smooth_max_grad(h_y, tau)
        v_new = beta * v - lr * grad
        h = project_to_feasible(h + v_new, n)
        v = v_new
        
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    
    return best_h, best_c5

def binary_local_search(h, n, max_iters, time_limit, rng):
    """Local search on binary solution using swap moves."""
    # Threshold to binary
    binary = (h > 0.5).astype(float)
    # Adjust to satisfy constraint: need exactly n/2 ones
    num_ones = int(binary.sum())
    target = n // 2
    if num_ones > target:
        # Set some 1s to 0s
        one_indices = np.where(binary == 1)[0]
        to_remove = rng.choice(one_indices, num_ones - target, replace=False)
        binary[to_remove] = 0
    elif num_ones < target:
        zero_indices = np.where(binary == 0)[0]
        to_add = rng.choice(zero_indices, target - num_ones, replace=False)
        binary[to_add] = 1
    
    best_c5 = compute_c5(binary)
    best_h = binary.copy()
    start = time.time()
    temp = 0.01
    cooling = 0.9995
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # Pick a random swap
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        new_h = best_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c5 = compute_c5(new_h)
        
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            best_h = new_h
            best_c5 = new_c5
        
        temp *= cooling
    
    return best_h, best_c5

        j1 = min(n, n - shift)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 + shift:j1 + shift])
        j0b = max(0, shift)
        j1b = min(n, n + shift)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b - shift:j1b - shift]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 400:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-13:
                break
    return best_h, best_c5


def binary_local_search(h, n, max_moves, time_limit, rng):
    binary = (h > 0.5).astype(np.float64)
    num_ones = int(binary.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(binary == 1)[0]
        to_rm = rng.choice(one_idx, num_ones - target, replace=False)
        binary[to_rm] = 0.0
    elif num_ones < target:
        zero_idx = np.where(binary == 0)[0]
        to_add = rng.choice(zero_idx, target - num_ones, replace=False)
        binary[to_add] = 1.0
    best_c5 = compute_c5(binary)
    best_h = binary.copy()
    start = time.time()
    temp = 0.003
    cooling = 0.99995
    for it in range(max_moves):
        if time.time() - start > time_limit:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h)
        delta = new_c5 - best_c5
        if delta < 0 or (temp > 1e-10 and rng.random() < np.exp(-delta / temp)):
            best_h = new_h
            best_c5 = new_c5
        temp *= cooling
    return best_h, best_c5


def beatty_start(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        frac = (i * alpha) % 1.0
        if frac < 0.5:
            h[i] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    h_init = initial_h_values.copy()
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def tl():
        return max(1.0, budget_s - (time.time() - start_time) - 4.0)

    # Phase 1: FISTA smooth at n=1200 with multi-tau
    n1 = 1200
    if tl() > 120:
        h1 = project_to_feasible(interp_to(best_h, n1), n1)
        for tau in [50.0, 25.0, 12.0, 6.0]:
            if tl() < 30:
                break
            h1, c5_1 = fista_smooth(h1, n1, tau, max_iters=8000,
                                     time_limit=min(tl() * 0.28, 130), lr=0.08)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()
            h1 = best_h.copy()

    # Phase 2: FISTA smooth at n=2500
    n2 = 2500
    if tl() > 100:
        h2 = project_to_feasible(interp_to(best_h, n2), n2)
        for tau in [40.0, 18.0, 8.0]:
            if tl() < 30:
                break
            h2, c5_2 = fista_smooth(h2, n2, tau, max_iters=4000,
                                     time_limit=min(tl() * 0.3, 120), lr=0.04)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
            h2 = best_h.copy()

    # Phase 3: Subgradient polish
    if tl() > 60:
        n_cur = len(best_h)
        h3, c5_3 = subgrad_opt(best_h, n_cur, max_iters=200000,
                                time_limit=tl() * 0.25, lr=0.004)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Binary local search
    if tl() > 50:
        n_cur = len(best_h)
        h4, c5_4 = binary_local_search(best_h, n_cur, max_moves=500000,
                                        time_limit=tl() * 0.35, rng=rng)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Try Beatty-sequence alternative start at moderate n
    if tl() > 80:
        n_alt = 1200
        alpha = (1.0 + np.sqrt(5.0)) / 2.0
        h_beat = project_to_feasible(beatty_start(n_alt, alpha), n_alt)
        for tau in [20.0, 8.0]:
            if tl() < 25:
                break
            h_beat, c5_b = fista_smooth(h_beat, n_alt, tau, max_iters=4000,
                                         time_limit=min(tl() * 0.35, 80), lr=0.06)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_beat.copy()

    # Phase 6: Final subgradient polish
    if tl() > 25:
        n_fin = len(best_h)
        h6, c5_6 = subgrad_opt(best_h, n_fin, max_iters=200000,
                                time_limit=tl() * 0.45, lr=0.002)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final projection
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
