def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        t = (lo + hi) / 2.0
        s = np.sum(np.clip(x - t, 0.0, 1.0))
        if s > target:

        else:

    cm = np.max(corr)
    vals = np.exp(T * (corr - cm))
