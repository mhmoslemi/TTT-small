def project_to_simplex(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(v)
    # Bisection on threshold
    lo, hi = v.min() - target_sum, v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    return float(np.max(c) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    exponents = alpha * (vals - m)
    exponents = np.clip(exponents, -500, 0)
    return m + (1.0/alpha) * np.log(np.sum(np.exp(exponents)))

def make_objective(n, dx, alpha):
    def obj(h):
        c = np.correlate(h, 1-h, mode='full')
        return smooth_max(c * dx, alpha)
    return obj

def resample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.clip(np.interp(x_new, x_old, h_old), 0, 1)

def enforce_sum(h, target):
    """Project h to have sum=target, keeping values in [0,1]"""
    h = h.copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase sum: increase values that are < 1
        room = 1.0 - h
        total_room = room.sum()
        if total_room < 1e-12:
            return np.clip(h, 0, 1)
        h += diff * room / total_room
    else:
        # Need to decrease sum: decrease values that are > 0
        room = h.copy()
        total_room = room.sum()
        if total_room < 1e-12:
            return np.clip(h, 0, 1)
        h += diff * room / total_room
    return np.clip(h, 0, 1)

  - Swap h[i] and h[j]
  - Check if max(c) decreased
  - If so, accept the swap
- Repeat until no improvement

The key optimization: when we swap h[i]=1→0 and h[j]=0→1, how does c change?

c_old[k] = sum_m h[m] * (1-h[m+k-(n-1)])
c_new[k] = c_old[k] + delta_k

where delta_k accounts for the change at positions i and j.
