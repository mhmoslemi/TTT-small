    prob.solve(solver=cp.ECOS)

    mask2 = (j - s >= 0) & (j - s < n)
    grad[mask2] += (-h[j[mask2] - s])

    best_c5 = evaluate_c5(h)

                    break
            else:
                # No improvement, reduce tolerance and continue
                tol = max(0.01, tol * 0.5)
                if tol < 0.01:
                    break
    
    return best_h, best_c5

def soft_max_slsqp(h, n, T, maxiter, deadline):
    """SLSQP with soft-max objective"""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        return float(m + np.log(np.sum(np.exp((corr - m) * T))) / T)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        w = np.exp((corr - m) * T)
        w = w / np.sum(w) * T  # softmax weights * T
        # grad of log-sum-exp / T w.r.t. corr is w/T... 
        # Actually: d/dx [log(sum exp(T*corr_k))/T] = (1/T) * T * softmax(T*corr) = softmax(T*corr)
        # So d(obj)/dx = sum_k softmax_k * d(corr_k)/dx
        # d(obj)/dx = sum_k softmax_k * dx * [(1-h[j+s_k]) - h[j-s_k]]
        
        # This is expensive to compute analytically for each x.
        # Let's use finite differences or just let SLSQP do it numerically.
        return None  # Let SLSQP use numerical gradient
    
    def const_eq(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': const_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except:
        pass
    return h

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_single(h, n, s):
    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)
    m1 = (j + s >= 0) & (j + s < n)
    grad[m1] += (1.0 - h[j[m1] + s])
    m2 = (j - s >= 0) & (j - s < n)
    grad[m2] += (-h[j[m2] - s])
    return grad * dx

def solve_lp_subproblem(h, n, corr_vals, indices):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for k in indices:
        s = k - n + 1
        g_k = grad_single(h, n, s)
        c_k = corr_vals[k]
        constraints.append(c_k + g_k @ (h_var - h) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS)
    except Exception:
        return None
    if h_var.value is None:
        return None
    return np.asarray(h_var.value).flatten()

                corr_vals = all_corr(h)

def soft_max_slsqp(h, n, T, maxiter, deadline):

    target = n / 2.0

    def obj(x):

        m = np.max(corr)
        return float(m + np.log(np.sum(np.exp((corr - m) * T))) / T)
