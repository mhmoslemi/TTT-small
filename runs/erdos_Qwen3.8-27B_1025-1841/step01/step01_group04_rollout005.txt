import numpy as np
from scipy.optimize import minimize
import time

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smx_obj(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    mx = np.max(scaled)
    return float((mx + np.log(np.sum(np.exp(scaled - mx)))) / T * (2.0 / n))

def smx_weights(corr, T):
    scaled = corr * T
    mx = np.max(scaled)
    e = np.exp(scaled - mx)
    return e / np.sum(e)

def smx_grad(h, n, weights):
    dx = 2.0 / n
    nc = 2 * n - 1
    grad = np.zeros(n)
    omh = 1.0 - h
    for j in range(n):
        lo = max(0, n - 1 - j)
        hi = min(nc - 1, nc - 1 - j)
        if hi >= lo:
            idx = j + np.arange(lo, hi + 1) - (n - 1)
            t1 = np.dot(weights[lo:hi + 1], omh[idx])
        else:
            t1 = 0.0
        lo2 = max(0, j - n + 2)
        hi2 = min(nc - 1, j + n - 1)
        if hi2 >= lo2:
            idx2 = j - np.arange(lo2, hi2 + 1) + (n - 1)
            t2 = np.dot(weights[lo2:hi2 + 1], h[idx2])
        else:
            t2 = 0.0
        grad[j] = dx * (t1 - t2)
    return grad

def proj_grad_step(h, n, T, deadline, lr_init=0.5, nesterov=True):
    h = h.copy()
    h_prev = h.copy()
    lr = lr_init
    best_obj = smx_obj(h, n, T)
    stall = 0

    for step in range(20000):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        w = smx_weights(corr, T)
        obj = float((np.max(corr * T) + np.log(np.sum(w))) / T * (2.0 / n))

        if nesterov and step > 0:
            beta = (step - 1.0) / (step + 2.0)
            y = h + beta * (h - h_prev)
            y = project(y, n)
        else:
            y = h.copy()

        corr_y = np.correlate(y, 1.0 - y, mode='full')
        w_y = smx_weights(corr_y, T)
        grad = smx_grad(y, n, w_y)

        alpha = lr
        accepted = False
        for bt in range(20):
            h_new = project(y - alpha * grad, n)
            obj_new = smx_obj(h_new, n, T)
            if obj_new < obj - 1e-14:
                h_prev = h
                h = h_new
                accepted = True
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    stall = 0
                else:
                    stall += 1
                break
            alpha *= 0.4
        if not accepted:
            if stall > 50:
                lr *= 0.5
            continue

        lr = min(lr * 1.3, lr_init * 3.0)
        if stall > 200:
            lr *= 0.7
            stall = 100

    return h, best_obj

        return smx_obj(x, n, T)

    return h
