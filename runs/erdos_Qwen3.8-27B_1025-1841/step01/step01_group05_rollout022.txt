import numpy as np
import time

def evaluate_c5(h):
    N = len(h)
    dx = 2.0 / N
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h):
    N = len(h)
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    max_val = np.max(corr)
    # Active set: all m where corr[m] is within tol of max
    tol = 1e-6 * max_val
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(N)
    for m in active:
        shift = m - (N - 1)
        # dc[m]/dh[i] = (1 - h[i+shift]) * [valid] - h[i-shift] * [valid]
        # i+shift valid: 0 <= i+shift < N => -shift <= i < N-shift
        # i-shift valid: 0 <= i-shift < N => shift <= i < N+shift
        
        # First term: (1 - h[i+shift]) for i in [max(0,-shift), min(N, N-shift))
        i_lo1 = max(0, -shift)
        i_hi1 = min(N, N - shift)
        if i_lo1 < i_hi1:
            grad[i_lo1:i_hi1] += (1.0 - h[i_lo1 + shift:i_hi1 + shift])
        
        # Second term: -h[i-shift] for i in [max(0,shift), min(N, N+shift))
        i_lo2 = max(0, shift)
        i_hi2 = min(N, N + shift)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= h[i_lo2 - shift:i_hi2 - shift]
    
    grad /= len(active)
    return grad, max_val

def project_feasible(h, target_sum):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return projected

    h = np.clip(h, 0, 1)
    h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    lr = 0.1  # learning rate

        if time.time() - time_limit > 0:
            break
        
        grad, max_val = compute_subgradient(h)
        c5 = max_val * (2.0 / N)

            best_h = h.copy()
        
        # Gradient descent step

        c5_new = evaluate_c5(h_new)
        if c5_new < c5:

            # Reduce learning rate slightly
            lr *= 0.999
        else:
            # Reduce learning rate
            lr *= 0.9
            if lr < 1e-8:
                break
    
    return best_h, best_c5

    x_new = np.linspace(0, 2, N_new, endpoint=False) + 1.0 / N_new
    h_new = np.interp(x_new, x_old, h)

import numpy as np

def evaluate_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = 1e-8
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(N)
    for m in active:
        d = m - (N - 1)
        # First term: (1 - h[j+d]) for valid j+d
        lo1 = max(0, -d)
        hi1 = min(N, N - d)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
        # Second term: -h[j-d] for valid j-d
        lo2 = max(0, d)
        hi2 = min(N, N + d)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 - d:hi2 - d]
    
    grad /= len(active)
    return grad, max_val

            h_cur = perturb_init(best_h, N0, rng, 0.05 / (r + 1))
        h_opt, c5_opt = grad_optimize(h_cur, N0, 50000, dl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Try binary sequences at N0
    bin_dl = start_time + budget_s * 0.40
    if bin_dl > time.time():
        h_bin, c5_bin = try_binary(N0, rng, bin_dl)
        if h_bin is not None and c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
        # Also polish the best binary with subgradient
        if h_bin is not None:
            pol_dl = min(bin_dl + (budget_s * 0.05), hard_deadline)
            h_pol, c5_pol = grad_optimize(h_bin, N0, 20000, pol_dl)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
            # Also try polishing original
            h_pol2, c5_pol2 = grad_optimize(best_h, N0, 20000, min(pol_dl + (budget_s * 0.05), hard_deadline))
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2

    # Phase 3: Try higher resolutions
    remaining_frac = (hard_deadline - time.time()) / budget_s
    N = N0
    for level in range(4):
        N *= 2
        if N > 500 or remaining_frac < 0.05:
            break
        alloc = budget_s * 0.12
        h_interp = interp_to_res(best_h, N)
        h_interp = project_feasible(h_interp, N / 2.0)
        sub_dl = time.time() + alloc
        for r in range(3):
            if time.time() > sub_dl or time.time() > hard_deadline:
                break
            if r == 0:
                h_cur = h_interp.copy()
            else:
                h_base = interp_to_res(best_h, N)
                h_base = project_feasible(h_base, N / 2.0)
                h_cur = perturb_init(h_base, N, rng, 0.03 / (r + 1))
            h_opt, c5_opt = grad_optimize(h_cur, N, 50000, min(sub_dl, hard_deadline))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        remaining_frac = (hard_deadline - time.time()) / budget_s

    # Phase 4: Final polish at best resolution
    N_best = len(best_h)
    if time.time() < hard_deadline - 5:
        h_final, c5_final = grad_optimize(best_h, N_best, 50000, hard_deadline)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
