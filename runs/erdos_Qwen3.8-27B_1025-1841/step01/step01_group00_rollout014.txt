import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = float(g.min()) - 2.0
    hi = float(g.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(g - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = float(g.min()) - 2.0
    hi = float(g.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(g - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def compute_c5(h):
    return compute_c5_all(h).max()


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def coord_descent_solve(h, n, time_limit, start_time, inner_iters=40, proj_every=5):
    dx = 2.0 / n
    m = 2 * n - 1
    C5 = compute_c5_all(h)
    best_c5 = float(C5.max())
    best_h = h.copy()
    shifts = np.arange(-(n - 1), n)

    t1 = np.zeros(m)
    t2 = np.zeros(m)
    g = np.zeros(m)
    A = np.zeros(m)
    temp = np.zeros(m)

    sweep = 0
    while True:
        if time.time() - start_time > time_limit:
            break

        for j in range(n):
            old_hj = h[j]
            j_ms = j - shifts
            j_ps = j + shifts

            t1.fill(0.0)
            v1 = (j_ms >= 0) & (j_ms < n)
            t1[v1] = 1.0 - h[j_ms[v1]]

            t2.fill(0.0)
            v2 = (j_ps >= 0) & (j_ps < n)
            t2[v2] = h[j_ps[v2]]

            np.subtract(t1, t2, out=g)
            g *= dx

            np.multiply(old_hj, g, out=temp)
            np.subtract(C5, temp, out=A)

            lo = 0.0
            hi = 1.0
            for _ in range(inner_iters):
                mid = (lo + hi) * 0.5
                np.multiply(mid, g, out=temp)
                np.add(A, temp, out=t1)
                k_star = int(np.argmax(t1))
                gk = g[k_star]
                if gk < -1e-15:
                    lo = mid
                elif gk > 1e-15:
                    hi = mid
                else:
                    break

            new_hj = (lo + hi) * 0.5
            delta = new_hj - old_hj
            if abs(delta) > 1e-14:
                h[j] = new_hj
                np.multiply(delta, g, out=temp)
                np.add(C5, temp, out=C5)

        sweep += 1

        if sweep % proj_every == 0:
            h = project_to_feasible(h, n)
            C5 = compute_c5_all(h)

        cur = float(C5.max())
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()

        if best_c5 < 0.37900:
            break

    return best_h, best_c5


def smooth_grad_polish(h, n, tau, lr, start_time, time_limit):
    dx = 2.0 / n
    m = 2 * n - 1
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(20000):
        if time.time() - start_time > time_limit:
            break

        C5 = compute_c5_all(h)
        max_c = C5.max()
        shifted = tau * (C5 - max_c)
        shifted = np.clip(shifted, -400.0, 400.0)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()

        grad = np.zeros(n)
        for k in range(m):
            pk = p[k]
            if pk < 1e-14:
                continue
            s = k - (n - 1)
            j_a = max(0, -s)
            j_b = min(n, n - s)
            if j_a < j_b:
                idx = np.arange(j_a, j_b)
                grad[j_a:j_b] += pk * (1.0 - h[idx + s])
            j_c = max(0, s)
            j_d = min(n, n + s)
            if j_c < j_d:
                idx2 = np.arange(j_c, j_d)
                grad[j_c:j_d] -= pk * h[idx2 - s]

        grad *= dx
        h = project_to_feasible(h - lr * grad, n)
        cur = compute_c5(h)
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()

    return best_h, best_c5
