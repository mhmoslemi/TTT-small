import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    omy = 1.0 - h
    corr = np.correlate(h, omy, mode="full") * dx
    return np.max(corr)

def smooth_max_value(vals, T):
    v_max = np.max(vals)
    shifted = vals - v_max
    return v_max + (1.0 / T) * np.log(np.sum(np.exp(T * shifted)))

def obj_fn(h, n, dx, T, penalty_coef):
    omy = 1.0 - h
    corr = np.correlate(h, omy, mode="full") * dx
    smax = smooth_max_value(corr, T)
    sum_pen = penalty_coef * (np.sum(h) - n / 2.0) ** 2
    return smax + sum_pen

def make_initializations(n, dx, target_sum, parent, rng, n_inits=6):
    inits = []
    if parent is not None:
        n_parent = len(parent)
        xp = np.linspace(0, 2, n_parent, endpoint=False)
        xn = np.linspace(0, 2, n, endpoint=False)
        h_up = np.interp(xn, xp, parent, left=parent[0], right=parent[-1])
        h_up = np.clip(h_up, 0, 1)
        h_up = h_up + (target_sum - np.sum(h_up)) / n
        h_up = np.clip(h_up, 0, 1)
        inits.append(h_up.copy())
    for _ in range(3):
        h = np.zeros(n)
        ones = rng.choice(n, size=int(target_sum), replace=False)
        h[ones] = 1.0
        inits.append(h.copy())
    h = np.zeros(n)
    bs = int(n * 0.25)
    h[:bs] = 1.0
    h[2*bs:3*bs] = 1.0
    s = np.sum(h)
    if s < target_sum:
        zeros = np.where(h == 0)[0]
        extra = int(target_sum - s)
        chosen = rng.choice(zeros, size=min(extra, len(zeros)), replace=False)
        h[chosen] = 1.0
    elif s > target_sum:
        ones = np.where(h == 1)[0]
        extra = int(s - target_sum)
        chosen = rng.choice(ones, size=min(extra, len(ones)), replace=False)
        h[chosen] = 0.0
    inits.append(h.copy())
    # Alternating pattern
    h_alt = np.zeros(n)
    h_alt[::4] = 1.0
    h_alt[1::4] = 1.0
    s = np.sum(h_alt)
    if s < target_sum:
        zeros = np.where(h_alt == 0)[0]
        extra = int(target_sum - s)
        if extra < len(zeros):
            chosen = rng.choice(zeros, size=extra, replace=False)
            h_alt[chosen] = 1.0
    inits.append(h_alt.copy())
    return inits

def optimize_one(h0, n, dx, T_schedule, penalty_coef, time_limit, start_time):
    current_h = h0.copy()
    best_c5 = compute_c5(current_h, dx)
    best_h = current_h.copy()
    for T in T_schedule:
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        remaining_frac = max(0, 1.0 - elapsed / time_limit)
        max_iter = max(30, int(150 * min(1.0, remaining_frac * 2)))
        try:
            res = minimize(
                lambda x: obj_fn(x, n, dx, T, penalty_coef),
                current_h,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            current_h = np.clip(res.x, 0, 1)
        except Exception:
            pass
        c5 = compute_c5(current_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
    return best_h, best_c5
