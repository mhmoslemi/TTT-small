from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_simplex_constraint(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1"""
    h = h.copy()
    diff = np.sum(h) - n / 2.0
    # Simple: subtract uniformly, then clip
    h -= diff / n
    h = np.clip(h, 0, 1)
    # If still not exact, do another pass
    diff = np.sum(h) - n / 2.0
    idx_free = (h > 0) & (h < 1)
    if np.sum(idx_free) > 0:

        h[idx_free] -= adjust
        h = np.clip(h, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _constraint_fun(x):
    return np.sum(x) - _global_n / 2.0

def project_constraint(h, n):
    h = h.copy()
    for _ in range(50):
        diff = np.sum(h) - n / 2.0
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) == 0:
            free = np.ones(n, dtype=bool)
        h[free] -= diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_start, n, n_iter, seed_val):
    rng = np.random.RandomState(seed_val)
    h = h_start.copy()
    c5 = compute_c5(h)
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-6:
            continue
        h_new = h.copy()
        h_new[i], h_new[j] = h[j], h[i]
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-12:
            h = h_new
            c5 = c5_new
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - elapsed()

    # Get parent
    try:
        initial = np.array(initial_h_values, dtype=float)
        n_init = len(initial)
        best_h = initial.copy()
        best_c5 = compute_c5(initial)
        best_n = n_init
    except (NameError, AttributeError):
        initial = None
        n_init = 0

    # n values to try
    n_list = []
    if best_n > 0:
        n_list.append(best_n)
    for nt in [30, 40, 44, 48, 50, 52, 56, 60, 64, 80, 100, 120]:
        if nt not in n_list and nt >= 8:
            n_list.append(nt)
    # Sort: put best_n first
    n_list.sort(key=lambda x: (x != best_n, x))

    for n in n_list:
        if tleft() < 45:
            break
        _global_n = n

        # Build initial points
        h0_list = []
        if n == n_init and initial is not None:
            h0_list.append(initial.copy())
        else:
            # Interpolated from parent
            if initial is not None:
                h_interp = np.interp(np.linspace(0, n_init - 1, n), range(n_init), initial)
                h_interp = project_constraint(h_interp, n)
                h0_list.append(h_interp)
            # Block
            hb = np.zeros(n)
            hb[:n // 2] = 1.0
            if n % 2 == 1:
                hb[n // 2] = 0.5
            h0_list.append(project_constraint(hb, n))
            # Golden-ratio spacing
            if n % 2 == 0:
                positions = sorted((int(i * n / (n // 2)) % n for i in range(n // 2)))
                hg = np.zeros(n)
                for p in positions:
                    hg[p] = 1.0
                h0_list.append(hg)

        # SLSQP optimization for each initial point
        for ix, h0 in enumerate(h0_list):
            if tleft() < 30:
                break

            max_iter = max(80, int(min(400, tleft() * 2)))

                    best_h = h_opt.copy()
                    best_n = n

        h0 = project_constraint(h0, n_p)

                compute_c5, h0, method='SLSQP',
                bounds=[(0.0, 1.0)] * n_p,
                constraints=[{'type': 'eq', 'fun': _constraint_fun}],
                options={'maxiter': 200, 'ftol': 1e-14}

                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_p
        except Exception:
            pass

    return best_h, best_c5, best_n
