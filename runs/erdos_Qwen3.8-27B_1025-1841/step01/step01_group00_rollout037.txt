    """Project g onto {h: 0<=h<=1, sum(h)=n/2} using bisection."""

        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            c5_try = compute_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
            stall = 0
        else:
            stall += 1

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)

        if stall > 120:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        h = h_new

    return best_h, best_c5


def subgrad_optimize(h, n, max_iters, time_limit, lr=0.005, top_m=3):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idxs = np.argsort(corr)[-top_m:][::-1]
        grad = np.zeros(n)
        for k in top_idxs:
            sh = k - n + 1
            if sh >= 0:
                grad[:n - sh] += (1.0 - h[sh:])
                grad[sh:] -= h[:n - sh]
            else:
                s2 = -sh
                grad[s2:] += (1.0 - h[:n - s2])
                grad[:n - s2] -= h[s2:]
        grad *= dx / top_m

        h_new = project_to_feasible(h - lr * grad, n)

        if stall > 400:
            lr *= 0.6
            stall = 0
            if lr < 1e-13:
                break
        h = h_new

    return best_h, best_c5


def binary_nudge(h, n):
    """Push values toward 0 or 1 while maintaining sum constraint."""
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    # Identify values close to boundaries
    mid_mask = (h > 0.1) & (h < 0.9)
    if not np.any(mid_mask):
        return h
    # Try pushing each mid value to nearest boundary
    c5_cur = compute_c5(h)
    for _ in range(3):
        improved = False
        for i in np.where(mid_mask)[0]:
            for val in [0.0, 1.0]:
                delta = val - h[i]
                if abs(delta) < 0.1:
                    continue
                # Find a coordinate to compensate
                if delta > 0:
                    # Need to decrease another coordinate
                    others = np.where((h > delta + 1e-6) & (np.arange(n) != i))[0]
                else:
                    others = np.where((h < 1.0 + delta - 1e-6) & (np.arange(n) != i))[0]
                if len(others) == 0:
                    continue
                j = others[np.argmin(np.abs(h[others] - h[i]))]
                h_try = h.copy()
                h_try[i] = val
                h_try[j] = h[j] - delta
                if h_try[j] < 0 or h_try[j] > 1:
                    continue
                h_try = project_to_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-8:
                    h = h_try
                    c5_cur = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Interpolate to n=1800, smooth optimization (tau=8)
    n1 = 1800
    if remaining() > 120:
        h1 = project_to_feasible(interp_periodic(best_h, n1), n1)
        tl = remaining() * 0.18
        h1, c5_1 = adam_optimize(h1, n1, tau=8, max_iters=15000, time_limit=tl, lr=0.25)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: tau=30
    if remaining() > 100:
        h2 = project_to_feasible(best_h.copy(), n1)
        tl = remaining() * 0.15
        h2, c5_2 = adam_optimize(h2, n1, tau=30, max_iters=12000, time_limit=tl, lr=0.12)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: tau=100
    if remaining() > 80:
        h3 = project_to_feasible(best_h.copy(), n1)
        tl = remaining() * 0.15
        h3, c5_3 = adam_optimize(h3, n1, tau=100, max_iters=10000, time_limit=tl, lr=0.06)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: subgradient polish (top-5 active shifts)
    if remaining() > 60:
        n4 = len(best_h)
        h4, c5_4 = subgrad_optimize(best_h, n4, max_iters=200000,
                                    time_limit=remaining() * 0.18, lr=0.004, top_m=5)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: try n=2400
    n5 = 2400
    if remaining() > 100:
        h5 = project_to_feasible(interp_periodic(best_h, n5), n5)
        tl = remaining() * 0.15
        h5, c5_5 = adam_optimize(h5, n5, tau=50, max_iters=8000, time_limit=tl * 0.6, lr=0.1)
        if remaining() > 40:
            h5, c5_5b = adam_optimize(h5, n5, tau=150, max_iters=6000,
                                      time_limit=tl * 0.4, lr=0.05)
            if c5_5b < c5_5:
                c5_5 = c5_5b
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: subgradient on best at current resolution
    if remaining() > 40:
        n6 = len(best_h)
        h6, c5_6 = subgrad_optimize(best_h, n6, max_iters=200000,
                                    time_limit=remaining() * 0.2, lr=0.003, top_m=3)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: shake and restart
    if remaining() > 60:
        n7 = len(best_h)
        for trial in range(3):
            if remaining() < 25:
                break
            h7 = best_h.copy()
            h7 += np.random.randn(n7) * 0.02
            h7 = project_to_feasible(h7, n7)
            h7, c5_7 = adam_optimize(h7, n7, tau=50, max_iters=5000,
                                     time_limit=remaining() * 0.15, lr=0.1)
            if c5_7 < best_c5:
                best_c5 = c5_7
                best_h = h7.copy()

    # Phase 8: final subgradient polish
    if remaining() > 20:
        n8 = len(best_h)
        h8, c5_8 = subgrad_optimize(best_h, n8, max_iters=200000,
                                    time_limit=remaining() * 0.35, lr=0.001, top_m=3)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final: binary nudge attempt
    if remaining() > 15:
        h9 = binary_nudge(best_h, len(best_h))
        h9 = project_to_feasible(h9, len(h9))
        c5_9 = compute_c5(h9)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
