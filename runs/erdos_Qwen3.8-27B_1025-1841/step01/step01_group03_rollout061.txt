def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(h - mid, 0.0, 1.0)
        if np.sum(z) > target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp((vals - m) * tau))) / tau

def gradient_smooth_max(h, tau):

        if wl > 1e-12:
            grad[:n-L] += wl * (1.0 - h[L:])
            grad[L:] += wl * (-h[:-L])
    return dx * grad

        g = gradient_smooth_max(y, tau)
        lr = 1.0 / L_est
        h_new = project_feasible(y - lr * g, target_sum)

        h = h_new
        
        if it % 100 == 0:
            c5 = compute_c5(h)

                best_h_local = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_local:
        best_local = c5
        best_h_local = h.copy()
    return best_h_local, best_local

def make_binary_starts(n, rng, count=3):
    starts = []
    for _ in range(count):
        idx = list(range(n))
        rng.shuffle(idx)
        b = np.zeros(n)
        b[idx[:n//2]] = 1.0
        starts.append(b)
    # Thue-Morse-like
    tm = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c += v & 1
            v >>= 1
        tm[i] = 1.0 if c % 2 == 0 else 0.0
    if sum(tm) > n * 0.7:
        tm = 1.0 - tm
    # Adjust to exact n/2 ones
    ones = np.where(tm == 1.0)[0]
    if len(ones) > n//2:
        tm[ones[n//2:]] = 0.0
    elif len(ones) < n//2:
        zeros = np.where(tm == 0.0)[0]
        tm[zeros[:n//2 - len(ones)]] = 1.0
    starts.append(tm)
    return starts
