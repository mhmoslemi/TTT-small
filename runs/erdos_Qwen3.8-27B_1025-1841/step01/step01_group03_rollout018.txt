    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(y)
    lo = -np.max(y) - 1.0
    hi = -np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < S:
            lo = mid
        else:
            hi = mid

            break

def compute_lag_gradient(h, n, lag):
    """Compute gradient of correlation at given lag."""
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] += (-h[lag:])
    else:
        M = -lag
        if M < n:
            grad[:n - M] += (1.0 - h[M:])
            grad[M:] += (-h[:n - M])
    return grad

def smooth_max_and_grad(h, n, beta=200):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    shifted = vals - vals.max()
    exps = np.exp(beta * shifted)
    total = np.sum(exps)
    weights = exps / total
    obj = vals.max() + np.log(total) / beta
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    thresh = 1e-8
    active = np.where(weights > thresh)[0]
    for k in active:
        lag = k - (n - 1)
        grad += weights[k] * compute_lag_gradient(h, n, lag)
    grad *= dx
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(y, S):
    n = len(y)
    lo = -np.max(y) - 1.0
    hi = -np.min(y) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s < S:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def lag_gradient(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:] += 1.0 - h[:m]
            grad[:m] += -h[lag:]
    else:
        M = -lag
        m = n - M
        if m > 0:
            grad[:m] += 1.0 - h[M:]
            grad[M:] += -h[:m]
    return grad

def smooth_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr * dx
    vshift = v - v.max()
    e = np.exp(beta * vshift)
    w = e / e.sum()
    obj = v.max() + np.log(e.sum()) / beta
    grad = np.zeros(n)
    # Only significant weights
    idx = np.where(w > 1e-8)[0]
    for k in idx:
        lag = k - (n - 1)
        grad += w[k] * lag_gradient(h, n, lag)
    grad *= dx
    return obj, grad

def fista_optimize(h0, n, time_limit, beta=300):
    S = n / 2.0
    t0 = time.time()
    x = project_onto_constraint(h0, S)
    y = x.copy()
    x_prev = x.copy()
    t = 1.0
    alpha = 2.0 / n  # step size, needs tuning
    best = x.copy()
    best_c5 = compute_c5(x, n)
    
    it = 0
    while time.time() - t0 < time_limit:
        _, grad = smooth_obj_grad(y, n, beta)
        
        x_new = project_onto_constraint(y - alpha * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + (t - 1.0) / t_new * (x_new - x_prev)
        y = project_onto_constraint(y_new, S)
        
        c5 = compute_c5(x_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best = x_new.copy()
        
        x_prev = x
        x = x_new
        t = t_new
        it += 1
        
        # Adapt alpha
        if it % 100 == 0:
            alpha *= 0.98
    
    return best, best_c5
