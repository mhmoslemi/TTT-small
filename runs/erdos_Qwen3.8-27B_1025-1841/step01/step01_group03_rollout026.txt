def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)


def smooth_obj_grad(h, alpha, n_pts):
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    exp_val = np.exp(alpha * (corr - max_c))
    S = np.sum(exp_val)
    w = exp_val / S
    obj = max_c + np.log(S) / alpha
    grad = np.zeros(n_pts)
    for m in range(2 * n_pts - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n_pts - 1)
        wm = w[m] * dx
        if k >= 0:
            if k < n_pts:
                grad[k:] += wm * (1.0 - h[:n_pts - k])
            if k > 0:
                grad[:n_pts - k] -= wm * h[k:]
        else:

            if absk < n_pts:
