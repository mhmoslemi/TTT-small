This is related to the "crossing number" or "discrepancy" of a binary sequence. For a random binary sequence, the max overlap at any lag is roughly n/4 (by concentration). But we can do better with structured sequences.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    h += (target - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    if new_n == len(h):
        return h.copy()
    return make_feasible(np.interp(np.linspace(0, 2, new_n), np.linspace(0, 2, len(h)), h), new_n)

def softmax_obj_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))
    sum_exp = exp_c.sum()
    w = exp_c / sum_exp
    obj = (max_c + np.log(sum_exp) / T) * dx + mu * (h.sum() - n / 2.0) ** 2
    
    grad = np.zeros(n)
    # Gradient computation
    # grad[j] = sum_tau w[tau+n-1]*(1-h[j+tau]) - sum_tau w[tau+n-1]*h[j-tau] + 2*mu*(sum-h_n/2)
    for tau in range(-(n-1), n):
        wi = w[tau + n - 1]
        if wi < 1e-12:
            continue
        if tau >= 0:
            grad[:n-tau] += wi * (1.0 - h[tau:])
            grad[tau:] -= wi * h[:n-tau]
        else:
            grad[-tau:] += wi * (1.0 - h[:n+tau])
            grad[:n+tau] -= wi * h[-tau:]
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return obj, grad

def run_lbfgsb(h, T, mu, maxiter, start_time, budget_s):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    elapsed = time.time() - start_time
    if elapsed > budget_s - 30:
        return h
    res = minimize(softmax_obj_grad, h, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-6, 'maxls': 50})
    if res.success or res.fun < softmax_obj_grad(h, T, mu)[0]:
        return make_feasible(res.x, n)
    return h

def coord_descent(h, start_time, budget_s, max_rounds=10):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 20:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 20:
                break
            old = h[j]
            for cand in [0.0, 1.0, 0.5, old+0.05, old-0.05, old+0.1, old-0.1]:
                if cand < -0.01 or cand > 1.01:
                    continue
                trial = h.copy()
                trial[j] = np.clip(cand, 0, 1)
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = trial
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def bilinear_search(h, start_time, budget_s):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    h += (target - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    if new_n == len(h):
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h))
    new_x = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def softmax_obj_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))
    sum_exp = exp_c.sum()
    W = exp_c / sum_exp
    obj = (max_c + np.log(sum_exp) / T) * dx + mu * (h.sum() - n / 2.0) ** 2
    
    # Gradient using correlation
    W_rev = W[::-1]
    term1 = np.correlate(1.0 - h, W, mode='full')[n-1:3*n-2]  # positions n-1 to 3n-3 → n values
    term2 = np.correlate(h, W_rev, mode='full')[n-1:3*n-2]
    constraint = h.sum() - n / 2.0
    grad = term1 - term2 + 2.0 * mu * constraint
    return obj, grad
