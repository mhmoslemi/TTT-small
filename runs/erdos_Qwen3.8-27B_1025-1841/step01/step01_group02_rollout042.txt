```python
def compute_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    obj = (m + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    weights = np.exp(T * shifted)
    weights = weights / np.sum(weights)
    
    # q has length 2n-1
    n1 = n - 1
    conv_part = np.convolve(1.0 - h, weights, mode='full')
    corr_part = np.correlate(h, weights, mode='full')
    
    # grad[j] = dx * (conv_part[n1+j] - corr_part[n1+j])
    grad = dx * (conv_part[n1:n1+n] - corr_part[n1:n1+n])
    
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    obj = (m + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    weights = np.exp(T * shifted)
    wsum = np.sum(weights)
    weights = weights / wsum
    
    n1 = n - 1
    conv_part = np.convolve(1.0 - h, weights, mode='full')
    corr_part = np.correlate(h, weights, mode='full')
    
    grad = dx * (conv_part[n1:n1+n] - corr_part[n1:n1+n])
    
    return obj, grad

        h0 = make_feasible(h0, n_init)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Resolution ladder
    resolutions = [320, 640, 1024]
    
    for n_pts in resolutions:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 30
        if remaining < 60:

        else:
```

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    obj = (m + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    weights = np.exp(T * shifted)
    weights /= np.sum(weights)
    n1 = n - 1
    conv_part = np.convolve(1.0 - h, weights, mode='full')
    corr_part = np.correlate(h, weights, mode='full')
    grad = dx * (conv_part[n1:n1+n] - corr_part[n1:n1+n])
    return obj, grad

def coordinate_polish(h, n_pts, max_time):
    """Polish h using coordinate descent."""
    start = time.time()
    best_c5 = compute_c5(h)
    converged = False
    while not converged and time.time() - start < max_time:
        converged = True
        for j in range(n_pts):
            if time.time() - start > max_time:
                return h, best_c5
            old = h[j]
            # Try a set of candidates
            candidates = [0.0, 1.0, 0.5, old]
            for dv in [0.01, 0.02, 0.05, 0.1, 0.2]:
                candidates.append(old + dv)
                candidates.append(old - dv)
            best_new = old
            best_c = best_c5
            for cv in candidates:
                if cv < -0.001 or cv > 1.001:
                    continue
                cv = np.clip(cv, 0.0, 1.0)
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_new = cv
            if best_new != old:
                h[j] = best_new
                h = make_feasible(h, n_pts)
                best_c5 = best_c
                converged = False
    return h, best_c5
