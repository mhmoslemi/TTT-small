import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h, target_sum):
    h = np.array(h, dtype=np.float64).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(5000):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 1e-14
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] -= step
            np.clip(h, 0.0, 1.0, out=h)
        else:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            step = (-diff) / np.sum(mask)
            h[mask] += step
            np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_func(x, n_points):
    return compute_c5(x)

def sum_con_fun(x, n_points):
    return np.sum(x) - n_points / 2.0

def sum_con_jac(x, n_points):
    return np.ones(n_points)

def interp_parent(parent, n_points):
    n_p = len(parent)
    x_old = np.linspace(0.0, 2.0, n_p)
    x_new = np.linspace(0.0, 2.0, n_points)
    h_new = np.interp(x_new, x_old, parent)
    return project_sum(h_new, n_points / 2.0)

def shift_parent(parent, n_points, shift_idx):
    h = interp_parent(parent, n_points)
    h = np.roll(h, shift_idx)
    return project_sum(h, n_points / 2.0)

def random_init(n_points, target_sum, rng):
    h = rng.rand(n_points)
    return project_sum(h, target_sum)

def smooth_random_init(n_points, target_sum, rng):
    h = rng.rand(n_points)
    for _ in range(10):
        h = (h[:-1] + h[1:]) / 2.0
        h = np.concatenate([h, [h[-1]]])
    return project_sum(h, target_sum)

def periodic_init(n_points, target_sum, period, duty, rng):
    phases = np.arange(n_points) % period
    h = (phases < period * duty).astype(np.float64)
    h += rng.uniform(-0.1, 0.1, n_points)
    return project_sum(h, target_sum)

def gaussian_init(n_points, target_sum, rng, n_bumps, sigma):
    h = np.zeros(n_points)
    centers = rng.randint(0, n_points, n_bumps)
    for c in centers:
        for i in range(n_points):
            d = abs(i - c)
            d = min(d, n_points - d)
            h[i] += np.exp(-d * d / (2.0 * sigma * sigma))
    h = (h - h.min()) / (h.max() - h.min() + 1e-12)
    return project_sum(h, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t0 = time.time()

    best_h = None
    best_c5 = 0.38085
    best_n = 0

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        if parent is not None and len(parent) > 0:
            c5_p = compute_c5(parent)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = parent.copy()
                best_n = len(parent)
    except Exception:
        pass

    rng = np.random.RandomState(seed)

    resolutions = [48, 64, 96, 128, 160, 192, 256, 320, 384]

    for n_points in resolutions:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.70:
            break

        target_sum = n_points / 2.0
        bounds = [(0.0, 1.0)] * n_points
        constr = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n_points,)}]

        inits = []

        if parent is not None:
            inits.append(interp_parent(parent, n_points))
            for sh in [1, 2, -1, 3]:
                inits.append(shift_parent(parent, n_points, sh))

        n_rand = 6 if n_points <= 96 else 4
        for _ in range(n_rand):
            inits.append(random_init(n_points, target_sum, rng))
        for _ in range(3):
            inits.append(smooth_random_init(n_points, target_sum, rng))

        for period in [3, 4, 5, 6, 7, 8, 12]:
            for duty in [0.33, 0.5, 0.67]:
                inits.append(periodic_init(n_points, target_sum, period, duty, rng))

        for nb in [2, 3, 4]:
            for sg in [3.0, 5.0, 8.0]:
                inits.append(gaussian_init(n_points, target_sum, rng, nb, sg))

        maxiter = 400 if n_points <= 96 else 250
        ftol = 1e-10

        for idx, h0 in enumerate(inits):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.85:
                break

            h0_clip = np.clip(h0, 0.0, 1.0)
            h0_clip = project_sum(h0_clip, target_sum)

            try:
                res = minimize(
                    fun=obj_func,
                    x0=h0_clip,
                    args=(n_points,),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constr,
                    options={'maxiter': maxiter, 'ftol': ftol}
                )
                h_sol = np.clip(res.x, 0.0, 1.0)
                h_sol = project_sum(h_sol, target_sum)
                c5_val = compute_c5(h_sol)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_sol.copy()
                    best_n = n_points
            except Exception:
                pass

    # Refinement phase: perturb best and re-optimize
    elapsed = time.time() - t0
    if best_h is not None and elapsed < budget_s * 0.88:
        n_best = len(best_h)
        target_sum_b = n_best / 2.0
        bounds_b = [(0.0, 1.0)] * n_best
        constr_b = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n_best,)}]

        for pert_scale in [0.15, 0.1, 0.05, 0.03]:
            if time.time() - t0 > budget_s * 0.95:
                break
            for trial in range(4):
                if time.time() - t0 > budget_s * 0.97:
                    break
                h_pert = best_h + rng.normal(0.0, pert_scale, n_best)
                h_pert = project_sum(h_pert, target_sum_b)
                try:
                    res = minimize(
                        fun=obj_func,
                        x0=h_pert,
                        args=(n_best,),
                        method='SLSQP',
                        bounds=bounds_b,
                        constraints=constr_b,
                        options={'maxiter': 300, 'ftol': 1e-11}
                    )
                    h_sol = np.clip(res.x, 0.0, 1.0)
                    h_sol = project_sum(h_sol, target_sum_b)
                    c5_val = compute_c5(h_sol)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_sol.copy()
                        best_n = n_best
                except Exception:
                    pass

    # Final check: also try a higher resolution from best
    elapsed = time.time() - t0
    if best_h is not None and elapsed < budget_s * 0.90:
        for n_high in [256, 384, 512]:
            if time.time() - t0 > budget_s * 0.96:
                break
            h_hi = interp_parent(best_h, n_high)
            bounds_hi = [(0.0, 1.0)] * n_high
            constr_hi = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n_high,)}]
            try:
                res = minimize(
                    fun=obj_func,
                    x0=h_hi,
                    args=(n_high,),
                    method='SLSQP',
                    bounds=bounds_hi,
                    constraints=constr_hi,
                    options={'maxiter': 300, 'ftol': 1e-10}
                )
                h_sol = np.clip(res.x, 0.0, 1.0)
                h_sol = project_sum(h_sol, n_high / 2.0)
                c5_val = compute_c5(h_sol)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_sol.copy()
                    best_n = n_high
            except Exception:
                pass

    if best_h is None:
        n_fb = 64
        best_h = np.full(n_fb, 0.5)
        best_h = project_sum(best_h, n_fb / 2.0)
        best_n = n_fb
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
