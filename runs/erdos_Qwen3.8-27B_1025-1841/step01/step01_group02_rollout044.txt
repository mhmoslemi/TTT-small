import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            masks = h < 1.0 if diff > 0 else h > 0.0
            cnt = np.sum(masks)
            if cnt > 0:
                h[masks] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_descend(h, max_sweeps=500, time_limit=None, start_time=None, seed=0):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(corr))
    rng = np.random.RandomState(seed)
    k_shifts = np.arange(2 * n - 1) - (n - 1)

    for sweep in range(max_sweeps):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        order = rng.permutation(n)
        for j in order:
            v = h[j]
            D = np.zeros(2 * n - 1)
            idx1 = j + k_shifts
            mask1 = (idx1 >= 0) & (idx1 < n)
            if np.any(mask1):
                D[mask1] += (1.0 - h[idx1[mask1]])
            idx2 = j - k_shifts
            mask2 = (idx2 >= 0) & (idx2 < n)
            if np.any(mask2):
                D[mask2] -= h[idx2[mask2]]

            best_v = v
            best_max = np.max(corr)
            for cand in (0.0, 1.0):
                if cand == v:
                    continue
                new_c = corr + (cand - v) * D
                c_val = np.max(new_c)
                if c_val < best_max - 1e-14:
                    best_max = c_val
                    best_v = cand

            if 0.0 < v < 1.0:
                lo, hi = 0.0, 1.0
                mid = v
                for _ in range(25):
                    mid = (lo + hi) * 0.5
                    nc = corr + (mid - v) * D
                    c_mid = np.max(nc)
                    m_star = int(np.argmax(nc))
                    g = D[m_star]
                    if g < -1e-12:
                        lo = mid
                    elif g > 1e-12:
                        hi = mid
                    else:
                        break
                    if hi - lo < 1e-13:
                        break
                nc = corr + (mid - v) * D
                c_mid = np.max(nc)
                if c_mid < best_max - 1e-14:
                    best_max = c_mid
                    best_v = mid

            if abs(best_v - v) > 1e-14:
                h[j] = best_v
                corr += (best_v - v) * D
                if best_max < best_c:
                    best_c = best_max
                    improved = True

        h = make_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c = float(np.max(corr))
        if cur_c < best_c:
            best_c = cur_c

def _smooth_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(h, n):
    return np.sum(h) - n / 2.0

def _con_jac(n):
    return np.ones(n)

def slsqp_refine(h, n, budget_left, T=60.0):
    if budget_left < 60:
        return h
    global _GD_N, _GD_T
    _GD_N = n
    _GD_T = T
    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: _con_jac(n)}]
    try:
        res = minimize(_oplan_fun, h.copy(), method='SLSQP',
