```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_obj(h, T):
    dx = 2.0 / len(h)
    corr = compute_corr(h)
    scaled = corr * T
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    scaled = corr * T
    m = np.max(scaled)
    p = np.exp(scaled - m)
    psum = p.sum()
    if psum < 1e-300:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p /= psum
    
    grad = np.zeros(n)
    omh = 1.0 - h
    
    for l_idx in range(2*n-1):
        if p[l_idx] < 1e-18:
            continue
        l = l_idx - (n-1)
        # First term: (1-h[j+l]) for j where 0<=j+l<n
        j_lo = max(0, -l)
        j_hi = min(n-1, n-1-l)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += p[l_idx] * omh[j_lo+l : j_hi+l+1]
        # Second term: -h[j-l] for j where 0<=j-l<n
        j_lo2 = max(0, l)
        j_hi2 = min(n-1, n-1+l)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= p[l_idx] * h[j_lo2-l : j_hi2-l+1]
    
    grad *= dx
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
```

```python
    return np.correlate(h, 1.0 - h, mode='full')

    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    scaled = corr * T
    m = np.max(scaled)
    p = np.exp(scaled - m)
    psum = p.sum()
    if psum < 1e-300:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p /= psum
    grad = np.zeros(n)
    omh = 1.0 - h

        w = p[l_idx]
        if w < 1e-18:

        l = l_idx - (n-1)

        j_hi = min(n-1, n-1-l)

        j_hi2 = min(n-1, n-1+l)
```

```python
def soft_max_obj(h, T):
    dx = 2.0 / len(h)
    corr = compute_corr(h)
    scaled = corr * T
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    scaled = corr * T
    m = np.max(scaled)
    p = np.exp(scaled - m)
    psum = p.sum()
    if psum < 1e-300:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p /= psum
    grad = np.zeros(n)
    omh = 1.0 - h
    for l_idx in range(2 * n - 1):
        w = p[l_idx]
        if w < 1e-18:
            continue
        l = l_idx - (n - 1)
        j_lo = max(0, -l)
        j_hi = min(n - 1, n - 1 - l)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * omh[j_lo + l:j_hi + l + 1]
        j_lo2 = max(0, l)
        j_hi2 = min(n - 1, n - 1 + l)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= w * h[j_lo2 - l:j_hi2 - l + 1]
    grad *= dx
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def proj_grad_momentum(h0, n, T, deadline, max_iter=3000, lr_init=0.015, mu=0.85):
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    f_prev = soft_max_obj(h, T)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = soft_max_grad(h, T)
        g_proj = g - g.mean()
        v = mu * v - lr * g_proj
        h_new = project_feasible(h + v, n)
        f_new = soft_max_obj(h_new, T)
        if f_new > f_prev + 1e-12:
            v = 0.5 * v
            h_new = project_feasible(h + v, n)
            f_new = soft_max_obj(h_new, T)
            if f_new > f_prev + 1e-10:
                h_new = project_feasible(h - 0.5 * lr * g_proj, n)
                f_new = soft_max_obj(h_new, T)
                lr *= 0.7
        h = h_new
        f_cur = soft_max_obj(h, T)
        if f_cur < f_prev - 1e-15:
            stall = 0
        else:
            stall += 1
        if stall > 300:
            lr = lr_init * 1.3
            v = 0.0
            stall = 0
        f_prev = f_cur
        c5_cur = evaluate_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

def binary_local_search(h0, n, deadline):
    target = n // 2
    h_bin = (h0 >= 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    best_c5 = evaluate_c5(h_bin)
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    rng = np.random.RandomState(42)
    for it in range(100000):
        if time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = evaluate_c5(h_bin)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_bin.copy()
            ones = np.where(best_h == 1.0)[0]
            zeros = np.where(best_h == 0.0)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return best_h, best_c5

def slsqp_polish(h0, n, deadline):
    target = n / 2.0
    T = 150.0
    def f(x):
        return soft_max_obj(x, T)
    def fgrad(x):
        return soft_max_grad(x, T)
    def c_eq(x):
        return x.sum() - target
    def c_eq_jac(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, jac=fgrad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                       options={'ftol': 1e-15, 'maxiter': 400})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(448, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n_cur = len(best_h)

    # Phase 1: Optimize at start resolution with decreasing T
    for T in [40.0, 20.0, 10.0, 5.0]:
        if time.time() > deadline - 120:
            break
        h_opt, c5_opt = proj_grad_momentum(best_h, n_cur, T, deadline - 120, max_iter=4000, lr_init=0.012)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Upsample and optimize at higher resolution
    n_up = min(n_cur * 2, 896)
    if time.time() < deadline - 150 and n_up > n_cur:
        h_up = interp_to(best_h, n_cur, n_up)
        h_up = project_feasible(h_up, n_up)
        for T in [25.0, 12.0, 6.0]:
            if time.time() > deadline - 30:
                break
            h_opt, c5_opt = proj_grad_momentum(h_up, n_up, T, deadline - 30, max_iter=3000, lr_init=0.01)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n_cur = n_up

    # Phase 3: Multiple random restarts
    n_best = len(best_h)
    for attempt in range(60):
        if time.time() > deadline - 25:
            break
        scale = 0.1 * np.exp(-attempt / 20.0) + 0.03
        h_pert = best_h + np.random.randn(n_best) * scale
        h_pert = project_feasible(h_pert, n_best)
        T_r = 12.0 if attempt < 30 else 6.0
        h_opt, c5_opt = proj_grad_momentum(h_pert, n_best, T_r, deadline - 25, max_iter=1500, lr_init=0.02, mu=0.8)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_best = len(best_h)

    # Phase 4: SLSQP polish
    if time.time() < deadline - 20:
        h_pol = slsqp_polish(best_h, n_best, deadline - 10)
        c5_pol = evaluate_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 5: Binary local search
    if time.time() < deadline - 5:
        h_bin, c5_bin = binary_local_search(best_h, n_best, deadline)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Phase 6: Final SLSQP on best
    if time.time() < deadline - 3:
        n_fin = len(best_h)
        h_fin = slsqp_polish(best_h, n_fin, deadline)
        c5_fin = evaluate_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
