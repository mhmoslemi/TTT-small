import numpy as np
import time
from scipy.optimize import minimize

_target_sum = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    h = np.clip(x, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(2000):
        diff = h.sum() - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] -= diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
        else:
            mask = h < 1
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += (-diff) / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def constraint_eq(x):
    return x.sum() - _target_sum

def interpolate_vec(old_h, old_n, new_n):
    if old_n == new_n:
        return old_h.copy()
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

def make_block_start(n, period_frac=4, rng=None):
    x0 = np.zeros(n)
    period = max(2, n // period_frac)
    for i in range(n):
        frac = (i % period) / period
        x0[i] = 1.0 if frac < 0.45 else 0.1
    return project_feasible(x0, n)

def make_random_start(n, rng):
    x0 = rng.random(n)
    return project_feasible(x0, n)

def make_uniform_start(n):
    return np.full(n, 0.5)

    _target_sum = n * 0.5

    res = minimize(
        compute_c5, x0, method='SLSQP',
        bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': tol}
    )
    return project_feasible(res.x, n)

        c5 = compute_c5(perturbed)

            best_c5 = c5
