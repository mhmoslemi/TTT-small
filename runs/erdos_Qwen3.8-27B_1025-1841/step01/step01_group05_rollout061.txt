def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    
    s = np.sum(h)

        return np.clip(h, 0, 1)
    
    if s > target:
        # Need to reduce: find λ > 0 such that sum(clip(h - λ, 0, 1)) = target
        # clip(h - λ, 0, 1) = max(0, min(1, h-λ))
        # For λ > 0: entries with h_i ≤ λ become 0
        #                entries with λ < h_i ≤ 1+λ stay h_i - λ... wait no
        #                entries with h_i > λ: become h_i - λ (if h_i - λ ≤ 1, i.e., h_i ≤ 1+λ, always true for h_i ≤ 1)
        #                entries with h_i ≤ λ: become 0

        # We need this to equal target

import numpy as np
from scipy.optimize import minimize
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    
    if s > target:
        lo, hi = 0.0, max(h) + 1e-10
        for _ in range(80):
            mid = (lo + hi) / 2
            if np.sum(np.maximum(h - mid, 0)) > target:
                lo = mid
            else:
                hi = mid
        h = np.maximum(h - (lo + hi) / 2, 0.0)
    else:
        lo, hi = 0.0, 1.0 - min(h) + 1e-10
        for _ in range(80):
            mid = (lo + hi) / 2
            if np.sum(np.minimum(h + mid, 1.0)) < target:
                lo = mid
            else:
                hi = mid
        h = np.minimum(h + (lo + hi) / 2, 1.0)
    
    return np.clip(h, 0.0, 1.0)

def compute_lag_gradient(h, lag, n):
    grad = np.zeros(n)
    if lag > 0:
        idx1 = np.arange(lag, n)
        grad[idx1] += 1.0 - h[idx1 - lag]
        idx2 = np.arange(0, n - lag)
        grad[idx2] -= h[idx2 + lag]
    elif lag < 0:
        m = -lag
        idx1 = np.arange(0, n - m)
        grad[idx1] += 1.0 - h[idx1 + m]
        idx2 = np.arange(m, n)
        grad[idx2] -= h[idx2 - m]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def interpolate_up(h, n_old, n_new):
    """Scale up h from n_old to n_new points by repetition/interpolation"""
    ratio = n_new / n_old
    if ratio == 1:
        return h.copy()
    # For integer ratio, repeat each value
    if abs(ratio - round(ratio)) < 1e-10:
        r = int(round(ratio))
        h_new = np.repeat(h, r)
    else:
        # Linear interpolation
        x_old = np.arange(n_old) / n_old * 2.0
        x_new = np.arange(n_new) / n_new * 2.0
        h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def optimize_shrink(h, n, rng, time_budget):
    """Main optimization loop using subgradient method"""
    t0 = time.time()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = c5_exact(best_h)
    
    lr = 0.1
    min_lr = 1e-6
    no_improve = 0
    patience = 3000
    best_patience = patience
    
    while time.time() - t0 < time_budget:
        # Compute all correlations
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        
        # Subgradient of the max
        grad = compute_lag_gradient(h, lag, n)
        
        # Step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        new_c5 = c5_exact(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr = max(lr * 0.95, min_lr)
            if no_improve > patience:

                h = best_h + rng.normal(0, 0.02, n)
                h = project_feasible(h, n)
                lr = 0.05
                no_improve = 0
                patience = max(500, patience // 2)
        
        # Time check
        if (time.time() - t0) > time_budget:

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

For the Erdős problem, the optimal h is known to have a specific structure: it's related to covering systems. For the minimum overlap constant, the optimal function has a "staircase" or "threshold" structure.

One key insight: if h(x) = 1 for x ∈ [0, a] and h(x) = 0 for x ∈ (a, 2], then:
- sum(h) = a/2 * ... wait, let me think in discrete terms.
- h = [1, 1, ..., 1, 0, 0, ..., 0] with n/2 ones and n/2 zeros
- Then correlation at lag k: counts how many 1s in h align with 0s in h shifted by k

For a step function (one block of 1s, one block of 0s):
- At lag 0: overlap = (number of 1s) * (number of 0s that a 1 overlaps with) = 0 (since h[i](1-h[i]) = 0 always)
- At lag k > 0: the overlap is the number of positions where h[i]=1 and h[i+k]=0
  If 1s are at positions 0..n/2-1 and 0s at n/2..n-1:
  For lag k: positions i where i < n/2 and i+k ≥ n/2, i.e., n/2-k ≤ i < n/2
  Count = min(k, n/2) positions
  So corr_k = min(k, n/2) for k > 0

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        lo, hi = 0.0, h.max()

            else:
                hi = mid

                lo = mid
            else:
                hi = mid

def lag_gradient(h, lag, n):
    g = np.zeros(n)
    if lag > 0:
        g[lag:] += 1.0 - h[:n-lag]  # h[j-lag] for j in [lag, n-1]
        g[:n-lag] -= h[lag:]  # h[j+lag] for j in [0, n-1-lag]
    elif lag < 0:
        m = -lag
        g[:n-m] += 1.0 - h[m:]  # h[j+m] for j in [0, n-1-m]
        g[m:] -= h[:n-m]  # h[j-m] for j in [m, n-1]
    else:

    return g

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        lo, hi = 0.0, float(h.max())
        for _ in range(64):
            mid = (lo + hi) * 0.5
            if np.maximum(h - mid, 0).sum() > target:
                lo = mid
            else:
                hi = mid
        return np.maximum(h - (lo + hi) * 0.5, 0.0)
    else:
        lo, hi = 0.0, 1.0 - float(h.min())
        for _ in range(64):
            mid = (lo + hi) * 0.5
            if np.minimum(h + mid, 1.0).sum() < target:
                lo = mid
            else:
                hi = mid
        return np.minimum(h + (lo + hi) * 0.5, 1.0)

def lag_grad(h, lag, n):
    g = np.zeros(n)
    if lag > 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    elif lag < 0:
        m = -lag
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def interp_up(h, no, nn):
    if no == nn:
        return h.copy()
    return project_feasible(np.interp(np.linspace(0, 1, nn), np.linspace(0, 1, no), h), nn)
