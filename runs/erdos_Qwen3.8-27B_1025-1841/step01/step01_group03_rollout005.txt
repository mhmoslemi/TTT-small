def project_feasible(p, n):
    """Project p onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    # Binary search for lambda
    lo, hi = p.min() - 1, p.max() + 1
    target = n / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(p - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(p - (lo + hi) / 2, 0, 1)
    # Fine adjustment
    diff = target - h.sum()
    # Distribute the difference to elements not at boundaries
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def compute_c5(h):
    """Compute C5 for a given h vector"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return (corr.max()) * dx, corr * dx

    shifted = beta * vals - (beta * vals).max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    val = (vals.max() + np.log(weights.sum()) / beta)
    return val, weights

def project_feasible(p, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(p - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(p - lam, 0, 1)
    # Correct any rounding
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)

def obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()

    exp_shifted = np.exp(shifted)

        if w < 1e-12:

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5, n_final
