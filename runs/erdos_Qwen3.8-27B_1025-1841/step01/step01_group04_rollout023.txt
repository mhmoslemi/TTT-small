def subgradient_step(h, n, lr, deadline=None, max_active=5):
    """One step of subgradient method using the top few active constraints."""
    if deadline is not None and time.time() > deadline:
        return h
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f_vals = corr * dx
    # Get the top max_active lags
    top_idx = np.argsort(f_vals)[-max_active:]
    # Average their gradients
    grad = np.zeros(n)
    for m in top_idx:
        k = n - 1 - m
        # g_k[j] = (1-h[j+k]) if 0<=j+k<n, minus h[j-k] if 0<=j-k<n
        j_start = max(0, -k)
        j_end = min(n, n - k)
        if j_start < j_end:
            grad[j_start:j_end] += (1.0 - h[j_start+k:j_end+k])
        j_start2 = max(0, k)
        j_end2 = min(n, n + k)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= h[j_start2-k:j_end2-k]
    grad /= len(top_idx)
    h_new = h - lr * grad
    h_new = project(h_new, n)
    return h_new

def slsqp_opt(h0, n, T, maxiter=400):
    target = n / 2.0
    def f(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        scaled = corr * T
        m = np.max(scaled)
        return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

        f_vals = corr * dx

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    xt = np.linspace(0, 2, n_to, endpoint=False)

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

        pass

def compute_subgradient(h, n, top_k=10):
    """Compute subgradient of max_k f_k(h) using top-k active constraints."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f_vals = corr * dx
    # Get top-k lags
    idx = np.argsort(f_vals)[-top_k:]
    grad = np.zeros(n)
    for m in idx:
        k = n - 1 - m
        j_start = max(0, -k)
        j_end = min(n, n - k)
        if j_start < j_end:
            grad[j_start:j_end] += (1.0 - h[j_start+k:j_end+k])
        j_start2 = max(0, k)
        j_end2 = min(n, n + k)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= h[j_start2-k:j_end2-k]
    grad /= len(idx)
    return grad, np.max(f_vals)

    h = h0.copy()
    v = np.zeros(n)  # momentum

    best_c5 = evaluate_c5(h0)
