```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def random_binary_h(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    return h

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exps = np.exp(T * shifted)
    S = np.sum(exps)
    rho = exps / S
    obj = (cmax + np.log(S) / T) * dx
    
    # Gradient
    N = 1
    while N < 4 * n:
        N *= 2
    
    h_pad = np.zeros(N)
    h_pad[:n] = h
    rho_pad = np.zeros(N)
    rho_pad[:2*n-1] = rho
    rho_r_pad = np.zeros(N)
    rho_r_pad[:2*n-1] = rho[::-1]
    
    H = np.fft.rfft(h_pad)
    R = np.fft.rfft(rho_pad)
    R_r = np.fft.rfft(rho_r_pad)
    
    conv = np.fft.irfft(R * H, N)
    conv_r = np.fft.irfft(R_r * H, N)
    
    C1 = conv[n-1 : 2*n-1]
    C2 = conv_r[n-1 : 2*n-1]
    
    grad = dx * (1.0 - C1 - C2)
    return obj, grad

def projected_gradient_step(h, grad, step_size, n):

def smooth_gradient(h, T):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exps = np.exp(T * shifted)
    rho = exps / np.sum(exps)
    
    # For each j, compute:
    # grad[j] = dx * sum over k where indices are valid of rho[k] * ((1-h[j+k-N+1]) - h[j-k+N-1])
    
    # Vectorized: create index matrix
    # j ranges 0..N-1, k ranges 0..2N-2
    # index1 = j + k - N + 1 (must be in [0, N-1])
    # index2 = j - k + N - 1 (must be in [0, N-1])
    
    j_idx = np.arange(N)[:, None]  # (N, 1)
    k_idx = np.arange(2*N-1)[None, :]  # (1, 2N-1)
    
    idx1 = j_idx + k_idx - N + 1  # (N, 2N-1)
    idx2 = j_idx - k_idx + N - 1  # (N, 2N-1)
    
    valid1 = (idx1 >= 0) & (idx1 < N)
    valid2 = (idx2 >= 0) & (idx2 < N)
    
    # h at idx1 (with 0 where invalid)
    h1 = np.where(valid1, h[np.clip(idx1, 0, N-1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, N-1)], 0.0)
    
    # grad[j] = dx * sum_k rho[k] * (valid1 - h1 - h2)
    # Note: (1-h[idx1])_valid = valid1 - h1 (since h1=0 where invalid)
    # Also: -h[idx2]_valid = -h2 (since h2=0 where invalid)
    
    grad = dx * (rho[None, :] * (valid1.astype(float) - h1 - h2)).sum(axis=1)
    
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def random_binary_h(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    return h

def smooth_obj_and_grad(h, T):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exps = np.exp(T * shifted)
    S = np.sum(exps)
    rho = exps / S
    obj = (cmax + np.log(S) / T) * dx

    j_idx = np.arange(N)[:, None]
    k_idx = np.arange(2 * N - 1)[None, :]
    idx1 = j_idx + k_idx - N + 1
    idx2 = j_idx - k_idx + N + 1 - 1
    valid1 = (idx1 >= 0) & (idx1 < N)
    valid2 = (idx2 >= 0) & (idx2 < N)
    h1 = np.where(valid1, h[np.clip(idx1, 0, N - 1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, N - 1)], 0.0)
    grad = dx * (rho[None, :] * (valid1.astype(float) - h1 - h2)).sum(axis=1)
    return obj, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_descent(h, start_time, budget_s, T_start, T_end, max_iters, min_time_left=40):
    n = len(h)
    T = T_start
    step = 1.0
    min_step = 0.005
    h = h.copy()
    
    for it in range(max_iters):
        elapsed = time.time() - start_time
        if budget_s - elapsed < min_time_left:
            break
        
        obj, grad = smooth_obj_and_grad(h, T)
        
        alpha = step
        f0 = obj
        improved = False
        for ls in range(12):
            h_new = h - alpha * grad
            h_new = project_feasible(h_new, n)
            obj_new, _ = smooth_obj_and_grad(h_new, T)
            directional = alpha * np.sum(grad * (h - h_new))
            if obj_new < f0 - 1e-4 * max(directional, 1e-8):
                h = h_new
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            step *= 0.6
            if step < min_step:
                break
        else:
            step = min(step * 1.02, 2.0)
        
        if it % 30 == 29:
            T = min(T * 1.25, T_end)
    
    return h, T

def coord_descent_polish(h, start_time, budget_s, max_sweeps=20):
    n = len(h)
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > budget_s - 20:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 20:
                break
            old_val = h[j]
            candidates = [0.0, 1.0, 0.5, old_val + 0.04, old_val - 0.04,
                          old_val + 0.02, old_val - 0.02, old_val + 0.01, old_val - 0.01]
            best_cand = old_val
            best_c = best_c5
            for cand in candidates:
                if cand < 0 or cand > 1:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-10:
                    best_c = c
                    best_cand = cand
            if best_cand != old_val:
                h[j] = best_cand
                h = project_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    
    return h, best_c5

    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_c5 = float('inf')
    best_h = None
    
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

        best_c5 = compute_c5(best_h)
    
    # Phase 1: Gradient descent at starting resolution
    h_iter, _ = gradient_descent(best_h, start_time, budget_s, T_start=40, T_end=250, max_iters=300, min_time_left=300)
    c = compute_c5(h_iter)
    if c < best_c5:
        best_c5 = c
        best_h = h_iter.copy()
    
    # Phase 2: Higher T refinement at same resolution
    if time.time() - start_time < budget_s - 200:
        h_iter2, _ = gradient_descent(best_h, start_time, budget_s, T_start=150, T_end=400, max_iters=200, min_time_left=200)
        c = compute_c5(h_iter2)
        if c < best_c5:
            best_c5 = c
            best_h = h_iter2.copy()
    
    # Phase 3: Try a few random restarts at same resolution
    for r in range(4):
        if time.time() - start_time > budget_s - 180:
            break
        h_rand = best_h + rng.randn(n0) * 0.08
        h_rand = project_feasible(h_rand, n0)
        h_iter3, _ = gradient_descent(h_rand, start_time, budget_s, T_start=50, T_end=200, max_iters=100, min_time_left=180)
        c = compute_c5(h_iter3)
        if c < best_c5:
            best_c5 = c
            best_h = h_iter3.copy()
    
    # Phase 4: Upsample to higher resolution and refine
    for target_n in [480, 640]:
        if time.time() - start_time > budget_s - 120:
            break
        h_high = upsample_h(best_h, target_n)
        h_high, _ = gradient_descent(h_high, start_time, budget_s, T_start=60, T_end=300, max_iters=150, min_time_left=120)
        c = compute_c5(h_high)
        if c < best_c5:
            best_c5 = c
            best_h = h_high.copy()
    
    # Phase 5: Coordinate descent polish
    best_h, best_c5 = coord_descent_polish(best_h, start_time, budget_s, max_sweeps=25)
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    return best_h, best_c5, len(best_h)
