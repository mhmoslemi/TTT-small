def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_and_gradient(h):
    """Compute C5 and subgradient (from active lag)"""
    n = len(h)
    dx = 2.0 / n

    idx = np.argmax(corr)
    # idx corresponds to lag = idx - (n-1)
    lag = idx - (n - 1)

    grad = np.zeros(n)

        j1 = m - lag
        if 0 <= j1 < n:

        j2 = m + lag
