```python
def compute_c5_gradient(h, dx):

    cor = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(cor)
    s = k_star - (n - 1)
    
    grad = np.zeros(n)

        # df_s/dh[j] = (1-h[j-s]) if j-s in [0,n) else 0
        #              - h[j+s] if j+s in [0,n) else 0

            if j + t < n:
```

The bisection works because:
- As λ → -∞: clip(h-λ, 0, 1) → 1 for all i, sum → n
- As λ → +∞: clip(h-λ, 0, 1) → 0 for all i, sum → 0
- We want sum = target_sum = n/2

The function g(λ) = sum(clip(h-λ, 0, 1)) is monotonically decreasing in λ.
So if g(λ) > target_sum, we need to increase λ (set lo = λ).
If g(λ) < target_sum, we need to decrease λ (set hi = λ).

Let me also think about using a smoothed max for better gradients. The issue with using just the argmax is that the gradient can be discontinuous when the argmax changes. A smoothed version:

C₅_smooth(h) = (1/T) * log(sum_k exp(T * f_k(h)))

where f_k(h) = correlate(h, 1-h, 'full')[k] * dx

The gradient of this is:
∇C₅_smooth = (1/T) * sum_k exp(T*f_k) / sum_k exp(T*f_k) * ∇f_k

This is a weighted average of the individual gradients, which is smoother.

Let me implement this:

```python
def compute_c5_smooth(h, dx, T=10.0):
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    # Shift to avoid overflow
    cor_shifted = cor - np.max(cor)
    weights = np.exp(T * cor_shifted)
    log_sum = np.log(np.sum(weights))
    c5 = np.max(cor) + log_sum / T
    return c5, weights / np.sum(weights)

def compute_c5_smooth_gradient(h, dx, T=10.0):
    n = len(h)
    cor_raw = np.correlate(h, 1.0 - h, mode='full')
    cor = cor_raw * dx
    cor_shifted = cor - np.max(cor)
    weights = np.exp(T * cor_shifted)
    weights /= np.sum(weights)
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k in range(len(cor_raw)):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
        else:
            t = -s
            g[:n-t] += (1.0 - h[t:])
            g[t:] -= h[:n-t]
        grad += weights[k] * g * dx
    
    return grad
```

```python
import numpy as np
import time

def compute_c5(h, dx):
    cor = np.correlate(h, 1.0 - h, mode='full')
    return np.max(cor) * dx

def project_onto_set(h, target_sum):
    """Project onto {x: 0<=x_i<=1, sum(x_i)=target_sum}"""
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def upsample(h_old, n_new):
    x_old = np.arange(len(h_old))
    x_new = np.linspace(0, len(h_old)-1, n_new)
    return np.interp(x_new, x_old, h_old)

def grad_descent(h, n, max_iters, time_limit, verbose=False):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_onto_set(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    lr = 0.5
    momentum = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    
    start = time.time()
    patience = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute gradient (using argmax)
        cor = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(cor)
        s = k_star - (n - 1)
        
        grad = np.zeros(n)
        if s >= 0:
            if n - s > 0:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] -= h[s:]
        else:
            tshift = -s
            if n - tshift > 0:
                grad[:n-tshift] += (1.0 - h[tshift:])
                grad[tshift:] -= h[:n-tshift]
        grad *= dx
        
        # Adam update
        t += 1
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * grad**2
        m_hat = m_t / (1 - beta1**t)
        v_hat = v_t / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project
        h_new = project_onto_set(h_new, target_sum)
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:
                # Simplex-free step: try different learning rate
                lr *= 0.5
                if lr < 0.01:
                    # Random restart in neighborhood
                    h = project_onto_set(h + 0.05 * np.random.randn(n), target_sum)
                    lr = 0.5
                    patience = 0
                continue
        
        h = h_new
        if verbose and it % 1000 == 0:
            print(f"  iter {it}, C5={best_c5:.6f}, lr={lr:.4f}")
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # Get initial values
    if 'initial_h_values' in dir() or True:
        try:
            h_init = np.array(initial_h_values, dtype=float)
        except:
            h_init = None
    
    if h_init is None:

        n = 200
        h_init = np.ones(n) * 0.5
        # Make it sum to n/2
        h_init = project_onto_set(h_init, n/2.0)
    
    # Try multiple resolutions
    resolutions = [100, 200, 300, 500]
    
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.85:
            break
        
        h_curr = upsample(h_init, n) if len(h_init) != n else h_init.copy()
        
        # Also try the current best upsampled
        if best_h is not None and len(best_h) != n:
            h_curr2 = upsample(best_h, n)
            # Use the better one
            c5_1 = compute_c5(project_onto_set(h_curr, n/2.0), 2.0/n)
            c5_2 = compute_c5(project_onto_set(h_curr2, n/2.0), 2.0/n)
            if c5_2 < c5_1:
                h_curr = h_curr2
```
