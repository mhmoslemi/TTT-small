       lo = g.min() - 2.0
       hi = g.max() + 2.0
       for _ in range(60):

           s = np.sum(np.clip(g - mid, 0.0, 1.0))
           if s > target:
               lo = mid
           else:
               hi = mid
       return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

   def interp_periodic(h, target_n):
       n = len(h)
       if n == target_n:
           return h.copy()
       x_old = np.arange(n) * (2.0 / n)
       x_new = np.arange(target_n) * (2.0 / target_n)
       return np.interp(x_new, x_old, h, period=2.0)

   def compute_corr(h):
       return np.correlate(h, 1.0 - h, mode='full')

               stall += 1
           
           # Top-K subgradient
           k_max = np.argmax(corr)
           max_val = corr[k_max]
           threshold = max_val - 0.01 * max_val
           top_ids = np.where(corr >= threshold)[0]
           if len(top_ids) > top_k:
               top_ids = top_ids[np.argsort(corr[top_ids])[-top_k:]]
           
           grad = np.zeros(n)
           for k_idx in top_ids:
               sh = k_idx - n + 1
               j0 = max(0, sh)
               j1 = min(n, sh + n)
               if j0 < j1:
                   grad[j0:j1] += (1.0 - h[np.arange(j0, j1) - sh])
               j0b = max(0, -sh)
               j1b = min(n, n - sh)
               if j0b < j1b:
                   grad[j0b:j1b] -= h[np.arange(j0b, j1b) + sh]
           grad /= len(top_ids)
           grad *= dx
           
           v = beta * v - lr * grad
           h_new = project_to_feasible(h + v, n)
           
           if stall > 200:
               lr *= 0.5
               v *= 0.5
               stall = 0
               if lr < 1e-14:
                   break
           
           h = h_new
       
       return best_h, best_c5
