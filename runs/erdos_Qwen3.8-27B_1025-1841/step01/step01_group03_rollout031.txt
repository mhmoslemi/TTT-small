def compute_c5(h):

def project_bounded_simplex(v, target_sum):
    n = len(v)
    lo, hi = -1.0, 2.0

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:

def smooth_grad(h, alpha):
    """Compute gradient of smooth approximation of max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    Z = np.sum(exp_c)
    w = exp_c / Z  # softmax weights
    
    grad = np.zeros(n)
    for p in range(n):
        for j in range(2 * n - 1):

            k2 = p - j + (n - 1)
            if 0 <= k2 < n:

import numpy as np

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_bounded_simplex(v, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    Z = np.sum(exp_c)
    w = exp_c / Z
    p_idx = np.arange(n)[:, None]
    j_idx = np.arange(2 * n - 1)[None, :]
    k1 = p_idx + j_idx - (n - 1)
    valid1 = (k1 >= 0) & (k1 < n)
    k2 = p_idx - j_idx + (n - 1)
    valid2 = (k2 >= 0) & (k2 < n)
    h_clipped1 = h[np.clip(k1, 0, n - 1)]
    h_clipped2 = h[np.clip(k2, 0, n - 1)]
    w_row = w[None, :]
    term1 = np.sum(w_row * np.where(valid1, 1.0 - h_clipped1, 0.0), axis=1)
    term2 = np.sum(w_row * np.where(valid2, h_clipped2, 0.0), axis=1)
    return (term1 - term2) / alpha

    current_c5 = compute_c5(h)
    best_c5 = current_c5
