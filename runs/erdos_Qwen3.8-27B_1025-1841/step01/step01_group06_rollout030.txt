import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            h[:] = target / n
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_weights(corr, temp):
    m = np.max(corr)
    if temp < 1e-13:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        return m, w
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -50.0, None)
    e = np.exp(shifted)
    se = np.sum(e)
    return m + temp * np.log(se), e / se

def upsample(h, n_new):
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = np.arange(n_old, dtype=float) * dx_old
    x_new = np.arange(n_new, dtype=float) * dx_new
    h_new = np.interp(x_new, x_old, h)
    return project_to_constraint(np.clip(h_new, 0.0, 1.0), n_new)

def obj_and_grad(x, dx, temp, n, penalty):
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    _, weights = get_weights(corr, temp)
    grad_sm = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-11:
            continue
        k = s - (n - 1)
        if k >= 0:
            m = n - k
            grad_sm[:m] += w * (1.0 - hx[k:])
            grad_sm[k:] -= w * hx[:m]
        else:
            ka = -k
            m = n - ka
            grad_sm[ka:] += w * (1.0 - hx[:m])
            grad_sm[:m] -= w * hx[ka:]
    s_sum = np.sum(x) - n / 2.0
    grad = grad_sm * dx + 2.0 * penalty * s_sum
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    val, _ = get_weights(corr, temp)
    val = val * dx + penalty * s_sum * s_sum
    return val, grad
