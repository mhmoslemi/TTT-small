import math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute correction
    if diff > 0:
        mask = h < 1.0
        idx = np.where(mask)[0]
        if len(idx) == 0:
            return h
        per = diff / len(idx)
        h[idx] = np.minimum(h[idx] + per, 1.0)
    else:
        mask = h > 0.0
        idx = np.where(mask)[0]
        if len(idx) == 0:
            return h
        per = diff / len(idx)
        h[idx] = np.maximum(h[idx] + per, 0.0)
    # Fine-tune
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h[-1] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new)

    temp = 0.02

    no_improve = 0

        if time.time() > t_deadline:
            break
        
        # Pick two random indices
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Random direction and magnitude
        delta = rng.uniform(0.02, 0.15) * (1 if rng.random() < 0.5 else -1)
        
        h_new_i = h[i] + delta
        h_new_j = h[j] - delta
        if h_new_i < 0 or h_new_i > 1 or h_new_j < 0 or h_new_j > 1:

        # Apply move
        h[i] += delta
        h[j] -= delta
        one_minus_h[i] -= delta
        one_minus_h[j] += delta
        
        c5_new = float(np.max(np.correlate(h, one_minus_h, mode='full')) * dx)

```python
import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.float64(np.sum(h))
    if abs(diff) < 1e-14:
        return h
    # Spread correction to unfixed elements
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    per = diff / len(idx)
    h[idx] += per
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff2 = target - np.float64(np.sum(h))
    if abs(diff2) > 1e-14:
        # Find a coordinate that can absorb the residual
        if diff2 > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2[0]] += diff2
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    h_new = np.interp(x_new, x_old, h, period=0)  # no wrapping
    return project_feasible(h_new)

def slsqp_optimize(h0, n_iter_max=500, tol=1e-8):
    """Optimize using SLSQP."""
    n = len(h0)
    def objective(h):
        return evaluate_c5(h)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': n_iter_max, 'ftol': tol})
    
    h_opt = np.array(res.x)
    h_opt = project_feasible(h_opt)
    return h_opt, evaluate_c5(h_opt)

def gradient_local_search(h, n_iter, rng, t_deadline):
    """Local search using gradient-guided 2-swaps."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        
        # Compute correlations to find active lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        active_idx = int(np.argmax(corr))
        lag = active_idx - (n - 1)
        
        # Compute subgradient for active lag
        grad = np.zeros(n)
        # For lag k: ∂/∂h[i] of Σ_j h[j](1-h[j+k]) = (1-h[i+k]) - h[i-k] (valid indices)
        for i in range(n):
            g = 0.0
            if 0 <= i + lag < n:
                g += (1.0 - h[i + lag])
            if 0 <= i - lag < n:
                g -= h[i - lag]
            grad[i] = g
        grad *= dx
        
        # Find best pair: high grad at source, low grad at destination
        # We want to decrease active correlation: move mass from high-grad to low-grad
        grad_centered = grad - np.mean(grad)
        
        # Pick source from top 20% of grad, destination from bottom 20%
        sorted_idx = np.argsort(grad_centered)
        top = sorted_idx[-n//5:]
        bot = sorted_idx[:n//5]
        
        i = top[rng.randint(len(top))]
        j = bot[rng.randint(len(bot))]
        
        if i == j:
            continue
        
        # Magnitude proportional to gradient difference
        delta_mag = min(0.1, 0.5 * abs(grad_centered[i] - grad_centered[j]))
        delta = delta_mag * (1.0 if grad_centered[i] > grad_centered[j] else -1.0)
        
        h_new_i = h[i] + delta
        h_new_j = h[j] - delta
        if h_new_i < 0 or h_new_i > 1 or h_new_j < 0 or h_new_j > 1:
            # Try random swap as fallback
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(0.02, 0.1) * (1 if rng.random() < 0.5 else -1)
            h_new_i = h[i] + delta
            h_new_j = h[j] - delta
            if h_new_i < 0 or h_new_i > 1 or h_new_j < 0 or h_new_j > 1:
                continue
        
        h[i] += delta
        h[j] -= delta
        c5_new = evaluate_c5(h)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        elif c5_new < c5:
            pass  # accept improvements over current
        else:
            # Reject (revert)
            h[i] -= delta
            h[j] += delta
    
    return best_h, best_c5

    n = len(h)
    h = h.copy()

    c5 = float(np.max(np.correlate(h, one_m_h, mode='full')) * dx)

        j = rng.integers(n)
        if i == j:

        h_new_j = h[j] - delta

        one_m_h[j] = 1.0 - h_new_j
        
        c5_new = float(np.max(np.correlate(h, one_m_h, mode='full')) * dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            temp = max(temp * 0.97, 0.005)
        elif rng.random() < math.exp(-(c5_new - c5) / temp):
            c5 = c5_new
        else:
            h[i] = h_new_i - delta
            h[j] = h_new_j + delta
            one_m_h[i] = 1.0 - h[i]
            one_m_h[j] = 1.0 - h[j]
        
        temp = max(temp * 0.9995, 0.003)
    
    return best_h, best_c5
