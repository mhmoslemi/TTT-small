def project_feasible(h, n):
    # Clip to [0, 1]
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum to n/2
    target = n / 2.0
    current_sum = np.sum(h)
    if current_sum > target:
        # Need to reduce
        excess = current_sum - target
        mask = h > 0
        h[mask] -= excess / mask.sum()
        h = np.clip(h, 0.0, 1.0)
        # Fine adjustment
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                mask = h < 1
            else:
                mask = h > 0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0, 1)
    elif current_sum < target:
        deficit = target - current_sum
        mask = h < 1
        h[mask] += deficit / mask.sum()
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)

            else:
                mask = h > 0
            if mask.sum() > 0:

                h = np.clip(h, 0, 1)

    """Upscale h from n_old points to n_new points using linear interpolation."""

def compute_c5_grad(h):
    """Compute C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    c5 = max_val * dx
    
    # Find active lags
    tol = 1e-8
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        # d(c_k)/d(h[j]) = (1 - h[j-shift]) - h[j+shift] (with bounds)
        # Vectorize: for each j, j-shift and j+shift
        js = np.arange(n) - shift  # j - shift for all j
        jp = np.arange(n) + shift  # j + shift for all j
        
        h_js = np.zeros(n)
        h_jp = np.zeros(n)
        valid_s = (js >= 0) & (js < n)

        h_js[valid_s] = h[js[valid_s]]
        h_jp[valid_p] = h[jp[valid_p]]
        
        grad += (1.0 - h_js) - h_jp
    
    grad *= dx / len(active)
    return c5, grad

    best_h = _proj(best_h, best_n)
    best_c5 = _c5(best_h)

    return best_h, best_c5, best_n
