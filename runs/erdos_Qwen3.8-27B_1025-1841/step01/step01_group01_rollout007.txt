def grad_active(h, dx, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if abs(w) < 1e-14:
            continue
        ell = int(lags[i])
        if ell >= 0:
            if ell < n:
                grad[ell:] += w * omh[:n - ell]
                grad[:n - ell] += w * (-h[ell:])
        else:
            ll = -ell
            if ll < n:
                grad[:n - ll] += w * omh[ll:]
                grad[ll:] += w * (-h[:n - ll])
    return dx * grad

def optimize_adam(h, n, dx, deadline, tau0=50.0, lr0=0.03, max_iter=5000):
    h = h.copy()
    m_len = 2 * n - 1
    lags_arr = np.arange(m_len) - (n - 1)
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    b1, b2, eps_a = 0.85, 0.999, 1e-7
    tc = 0
    tau = tau0
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = np.correlate(h, 1.0 - h, mode="full") * dx
        c_max = c_all.max()
        shifted = tau * (c_all - c_max)
        exp_v = np.exp(np.clip(shifted, -50, 0))
        w = exp_v / exp_v.sum()
        thresh = 3.0 / m_len
        act = w > thresh
        if act.sum() < 1:
            top_k = max(1, m_len // 6)
            order = np.argpartition(c_all, -top_k)
            act = np.zeros(m_len, dtype=bool)
            act[order[-top_k:]] = True
        a_idx = np.where(act)[0]
        a_w = w[a_idx]
        a_w = a_w / a_w.sum()
        g = grad_active(h, dx, lags_arr[a_idx], a_w)
        tc += 1
        m_a = b1 * m_a + (1 - b1) * g
        v_a = b2 * v_a + (1 - b2) * g * g
        m_hat = m_a / (1 - b1 ** tc)
        v_hat = v_a / (1 - b2 ** tc)
        lr = lr0 / (1 + 0.005 * tc)
        h = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps_a), n)
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5 - 1e-11:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
            tau *= 0.985
        else:
            stall += 1
            tau *= 1.015
        tau = float(np.clip(tau, 3.0, 3000.0))
        if stall > 300:
            break
    return best_h, best_c5

def coord_descent(h, n, dx, deadline, max_sweeps=150):
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    d_step = [0.03, -0.03, 0.07, -0.07, 0.015, -0.015]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        imp = False
        for j in range(n):
            if time.time() > deadline:
                break
            for d in d_step:
                h_t = h.copy()
                h_t[j] += d
                h_t = project_feasible(h_t, n)
                c5_t = compute_c5(h_t, dx)
                if c5_t < best_c5 - 1e-13:
                    h = h_t
                    best_c5 = c5_t
                    imp = True
                    break
        if not imp:
            for d in [0.005, -0.005, 0.002, -0.002]:
                imp = False
                for j in range(n):
                    if time.time() > deadline:
                        break
                    h_t = h.copy()
                    h_t[j] += d
                    h_t = project_feasible(h_t, n)
                    c5_t = compute_c5(h_t, dx)
                    if c5_t < best_c5 - 1e-13:
                        h = h_t
                        best_c5 = c5_t
                        imp = True
                if not imp:
                    break
            if not imp:
                break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl_main = t0 + budget_s * 0.82
    dl_fin = t0 + budget_s * 0.97

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    else:
        h_init = np.full(45, 0.5)
        n_init = 45

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_init)
    if 'evaluate_erdos_solution' in globals():
        try:
            best_c5 = evaluate_erdos_solution(best_h, len(best_h))
        except Exception:
            best_c5 = compute_c5(best_h, 2.0 / n_init)

    res_list = sorted(set([n_init, 50, 60, 75, 90, 100, 120]))

    for n in res_list:
        if time.time() > dl_main:
            break
        dx = 2.0 / n
        starts = []

        if n == n_init:
            starts.append(h_init.copy())
        else:
            xo = np.linspace(0, 2, n_init, endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            starts.append(project_feasible(np.interp(xn, xo, h_init), n))

        nh = len(best_h)
        if n != nh:
            xo2 = np.linspace(0, 2, nh, endpoint=False)
            xn2 = np.linspace(0, 2, n, endpoint=False)
            starts.append(project_feasible(np.interp(xn2, xo2, best_h), n))

        t = np.linspace(0, 2, n, endpoint=False)
        for _ in range(2):
            cf = rng.uniform(-0.35, 0.35, 5)
            hs = 0.5 + sum(cf[k] * np.sin((k + 1) * np.pi * t) for k in range(len(cf)))
            starts.append(project_feasible(hs, n))

        hb = (rng.rand(n) > 0.5).astype(float)
        starts.append(project_feasible(hb, n))

        for s in starts:
            if time.time() > dl_main:
                break
            ho, co = optimize_adam(s, n, dx, dl_main, tau0=40.0, lr0=0.025, max_iter=4000)
            hc, cc = coord_descent(ho, n, dx, dl_main, max_sweeps=120)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

    n_b = len(best_h)
    dx_b = 2.0 / n_b
    hf, cf1 = optimize_adam(best_h, n_b, dx_b, dl_fin, tau0=100.0, lr0=0.01, max_iter=6000)
    hf, cf2 = coord_descent(hf, n_b, dx_b, dl_fin, max_sweeps=200)
    if cf2 < best_c5:
        best_c5 = cf2
        best_h = hf

    c5_chk = compute_c5(best_h, 2.0 / len(best_h))
    if 'evaluate_erdos_solution' in globals():
        try:
            ev = evaluate_erdos_solution(best_h, len(best_h))
            if ev < c5_chk:
                c5_chk = ev
        except Exception:
            pass
    return (best_h, c5_chk, len(best_h))
