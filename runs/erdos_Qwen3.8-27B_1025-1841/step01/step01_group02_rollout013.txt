    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_s = np.exp(T * (corr - max_c))
    Z = np.sum(exp_s)
    w = exp_s / Z
    val = (max_c + np.log(Z) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            p = -lag
            grad[p:] += wk * (1.0 - h[:n - p])
            grad[:n - p] -= wk * h[p:]
    grad *= dx
    return val, grad

            break

                break

            # c[k] = A_k + t * B_k where:
            # A_k = sum_{i≠j, valid} h[i]*(1-h[i+lag]) + terms where h[j] is in (1-h[.]) part
            # B_k = terms where h[j] is in h[i] part

            # c[k] = sum_{i valid} h[i]*(1-h[i+lag])

            #   h[j] in h[i] when i=j: contributes (1-h[j+lag]) * t
            #   h[j] in (1-h[i+lag]) when i+lag=j, i=j-lag: contributes h[j-lag] * (-t)

            # A_k = c[k]_current - h[j]*B_k  (since c[k] = A_k + h[j]*B_k)

            # c[k] with h[j]=t:
            # = (original c[k] with h[j]) - h[j]*B_k + t*B_k
            # = (c[k] - h[j]*B_k) + t*B_k
            # So A_k = c[k] - h[j]*B_k
            
            hj = h[j]
            # Compute B for all lags
            # B_k for lag: (1-h[j+lag])*I(0<=j+lag<n) - h[j-lag]*I(0<=j-lag<n)
            # Vectorized:
            # For all lags from -(n-1) to (n-1):
            lags = np.arange(-(n-1), n)
            # (1-h[j+lag]) valid when 0<=j+lag<n
            idx1 = j + lags  # index into h for j+lag
            valid1 = (idx1 >= 0) & (idx1 < n)
            # h[j-lag] valid when 0<=j-lag<n
            idx2 = j - lags
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            B = np.zeros(2*n-1)
            B[valid1] += (1.0 - h[idx1[valid1]])
            B[valid2] -= h[idx2[valid2]]
            
            # Current correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            A = corr - hj * B
            
            # g(t) = max_k (A_k + t*B_k)
            # Evaluate on grid
            vals = A[None, :] + grid[:, None] * B[None, :]  # (n_grid, 2n-1)
            g = np.max(vals, axis=1)  # (n_grid,)
            best_idx = np.argmin(g)
            t_opt = grid[best_idx]
            
            if abs(t_opt - hj) > 1e-8:
                h[j] = t_opt
                # Adjust sum: redistribute difference
                diff = hj - t_opt
                if abs(diff) > 1e-12:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    h_other = h[mask]
                    # Adjust others to compensate
                    s = np.sum(h)
                    target = n / 2.0
                    adj = (target - s) / (n - 1)
                    h[mask] += adj
                    h = np.clip(h, 0.0, 1.0)
                    # Re-fix sum
                    h = project_sum(h, n)
                
                c_new = compute_c5(h)
                if c_new < best_c5 - 1e-12:
                    best_c5 = c_new
                    improved = True
        
        if not improved:
            break
    
    return h, best_c5

import numpy as np

_cur_dx = 1.0
_cur_n = 640

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return project_sum(h.copy(), new_n)
    old_x = np.arange(old_n, dtype=float) / old_n * 2.0
    new_x = np.arange(new_n, dtype=float) / new_n * 2.0
    h_new = np.interp(new_x, old_x, h)
    return project_sum(h_new, new_n)

def _penalty_obj(x):
    n = _cur_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    exp_s = np.exp(_cur_T * (corr - max_c))
    Z = np.sum(exp_s)
    val = (max_c + np.log(Z) / _cur_T) * dx
    s = np.sum(x)
    target = n / 2.0
    penalty = 500.0 * (s - target) ** 2
    val += penalty
    grad = np.zeros(n)
    w = exp_s / Z
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - x[lag:])
            grad[lag:] -= wk * x[:n - lag]
        else:
            p = -lag
            grad[p:] += wk * (1.0 - x[:n - p])
            grad[:n - p] -= wk * x[p:]

    grad += 2.0 * 500.0 * (s - target)

    _cur_n = n

    res = minimize(_penalty_obj, h, method='L-BFGS-B',
                   bounds=bounds, jac=True,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})

def coord_descent(h, n, n_grid=100, max_passes=50, time_limit=None, t0=None):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    grid = np.linspace(0.0, 1.0, n_grid)
    lags = np.arange(-(n - 1), n)
