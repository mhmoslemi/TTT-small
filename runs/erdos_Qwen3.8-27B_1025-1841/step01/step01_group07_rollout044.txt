                break
            scale = 0.08 * (1.0 + 0.5 * rep)
            h_p = h_curr + rng.randn(n) * scale
            h_p = fix_sum(h_p, target)

            try:
                r4 = minimize(compute_c5, h_p, args=(dx,), method='SLSQP',
                              bounds=bounds, constraints=cons_eq,
                              options={'maxiter': 120, 'ftol': 1e-10})
                h_p = fix_sum(r4.x, target)
            except Exception:
                h_p = fix_sum(h_p, target)

            c5_p = compute_c5(h_p, dx)
            if c5_p < c5_curr:
                h_curr = h_p
                c5_curr = c5_p
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
                best_n = n

    # Final polish on best
    if best_h is not None and remaining() > 40:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons_eq = [{'type': 'eq', 'fun': lambda h, t=target: h.sum() - t}]

        # Multi-tau smoothing
        for tau in [15.0, 10.0, 5.0]:
            if remaining() < 20:
                break
            try:
                rf = minimize(smooth_max, best_h, args=(dx, tau), method='SLSQP',
                              bounds=bounds, constraints=cons_eq,
                              options={'maxiter': 150, 'ftol': 1e-11})
                h_f = fix_sum(rf.x, target)
                c5_f = compute_c5(h_f, dx)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f
            except Exception:
                pass

        # Final SLSQP on exact objective
        if remaining() > 20:
            try:
                rf2 = minimize(compute_c5, best_h, args=(dx,), method='SLSQP',
                               bounds=bounds, constraints=cons_eq,
                               options={'maxiter': 400, 'ftol': 1e-13})
                h_f = fix_sum(rf2.x, target)
                c5_f = compute_c5(h_f, dx)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f
            except Exception:
                pass

        # Coordinate-descent style polish: nudge pairs to reduce max lag
        if remaining() > 30:
            comp = 1.0 - best_h
            corr = np.correlate(best_h, comp, mode='full')
            for _ in range(3):
                if remaining() < 10:
                    break
                max_idx = int(np.argmax(corr))
                # Try small swaps/adjustments
                for i in range(min(20, n)):
                    for j in range(i + 1, min(20, n)):
                        h_t = best_h.copy()
                        h_t[i] = np.clip(h_t[i] + 0.02, 0, 1)
                        h_t[j] = np.clip(h_t[j] - 0.02, 0, 1)
                        h_t = fix_sum(h_t, target)
                        c5_t = compute_c5(h_t, dx)
                        if c5_t < best_c5:
                            best_c5 = c5_t
                            best_h = h_t

    if best_h is None:
        best_h = np.ones(47) * 0.5
        best_n = 47
        best_c5 = compute_c5(best_h, 2.0 / 47)

    return best_h, best_c5, best_n
