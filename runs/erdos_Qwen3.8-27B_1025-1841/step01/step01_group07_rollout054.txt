import numpy as np
from scipy.optimize import minimize
import time

_CFG = {}

def _c5(h):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * _CFG['dx'])

def _proj(h, t):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        d = t - float(np.sum(h))
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-16
        else:
            m = h > 1e-16
        cnt = int(np.sum(m))
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    return _c5(x)

def _eq_fun(x):
    return float(np.sum(x) - _CFG['tgt'])

def _eq_jac(x):
    return np.ones(len(x))

def _rand_init(n, rng):
    return _proj(rng.rand(n), n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    nb = rng.randint(2, 5)
    for _ in range(nb):
        s = rng.randint(0, n)
        w = rng.randint(1, max(2, n // 4))
        h[s:s + w] = 1.0
    s = float(np.sum(h))
    if s > 1e-10:
        h = h * (n / 2.0) / s
    else:
        h = 0.5 * np.ones(n)
    return _proj(h, n / 2.0)

def _sin_init(n, rng):
    freq = rng.randint(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    h = 0.5 + rng.uniform(0.15, 0.45) * np.sin(2 * np.pi * freq * x / n + phase)
    return _proj(h, n / 2.0)

def _try_slsqp(h0, n, bounds, cons):
    tgt = _CFG['tgt']
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, maxiter=400, maxfev=2000,
                       options={'ftol': 1e-13})
        h_new = _proj(res.x, tgt)
        c = _c5(h_new)
        return h_new, c
    except Exception:
        return None, None

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    grids = []
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        grids.append(n0)
    for s in [47, 64, 94, 128]:
        if s not in grids:
            grids.append(s)
    grids = [g for g in grids if g >= 16]

    for n in grids:
        elapsed = time.time() - t0
        if budget_s - elapsed < 25:
            break

        _CFG['dx'] = 2.0 / n
        _CFG['tgt'] = n / 2.0
        tgt = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        if n == grids[0] and 'initial_h_values' in globals():
            h0 = np.asarray(initial_h_values, dtype=float).copy()
        else:
            h0 = np.ones(n) * 0.5

        h_best = _proj(h0, tgt)
        c5_best = _c5(h_best)

        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            best_n = n

        # Phase 1: SLSQP from initial
        h_new, c_new = _try_slsqp(h_best.copy(), n, bounds, cons)
        if h_new is not None and c_new < c5_best:
            c5_best, h_best = c_new, h_new

        # Phase 2: Multi-restart
        n_rest = max(8, int((budget_s - (time.time() - t0)) / 15))

            elif r % 7 == 1:
                h_cur = _rand_init(n, rng)

                h_cur = _block_init(n, rng)
