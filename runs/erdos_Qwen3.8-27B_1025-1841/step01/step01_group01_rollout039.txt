def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
