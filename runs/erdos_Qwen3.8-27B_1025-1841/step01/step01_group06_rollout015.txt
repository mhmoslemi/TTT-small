```python
        #      - h[j-L] if 0<=j-L<n, else 0
```

```python
def min_max_1d(a, b):
    # Minimize f(t) = max_i (a[i]*t + b[i]) over t in [0,1]
    # f is convex
    
    def f(t):
        return np.max(a*t + b)
    
    f0 = f(0.0)
    f1 = f(1.0)
    
    # Check if optimum is at boundary
    # Subgradient at 0: set of a[i] where a[i]*0 + b[i] = f(0)
    active0 = np.where(a*0 + b >= f0 - 1e-12)[0]
    if len(active0) > 0 and np.all(a[active0] >= 0):
        return 0.0
    
    active1 = np.where(a*1 + b >= f1 - 1e-12)[0]
    if len(active1) > 0 and np.all(a[active1] <= 0):
        return 1.0
    
    # Bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(30):
        mid = (lo + hi) / 2
        vals = a*mid + b
        f_mid = np.max(vals)
        active = np.where(vals >= f_mid - 1e-12)[0]
        # Subgradient is [min(a[active]), max(a[active])]
        sg_min = np.min(a[active])
        sg_max = np.max(a[active])
        if sg_min > 0:  # function increasing, go right
            lo = mid
        elif sg_max < 0:  # function decreasing, go left
            hi = mid
        else:  # 0 in subgradient
            break
    
    return (lo + hi) / 2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.max(np.abs(h_new - h)) < 1e-15:
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-13:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            step = rem / len(idx)
            for i in idx:
                h[i] += step
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_sum(h_new, n_new)

def min_max_1d(a, b, cur):
    if np.max(a) - np.min(a) < 1e-14:
        return cur
    f0 = np.max(b)
    f1 = np.max(a + b)
    active0 = b >= f0 - 1e-10
    if np.all(a[active0] >= -1e-10):
        return 0.0
    active1 = (a + b) >= f1 - 1e-10
    if np.all(a[active1] <= 1e-10):
        return 1.0
    lo, hi = 0.0, 1.0

        sg_max = np.max(a[active])
        if sg_min > 1e-10:
            lo = mid
        elif sg_max < -1e-10:
            hi = mid
        else:
            break
    return float(np.clip(0.5 * (lo + hi), 0.0, 1.0))

def coord_sweep(h, n):
    m = 2 * n - 1
    nj = n - 1
    s_vals = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
    for j in range(n):
        a = np.zeros(m, dtype=np.float64)
        v1 = (j + s_vals >= 0) & (j + s_vals < n)
        a[v1] += (1.0 - h[j + s_vals[v1]])
        v2 = (j - s_vals >= 0) & (j - s_vals < n)
        a[v2] -= h[j - s_vals[v2]]
        b = corr - a * h[j]
        t_star = min_max_1d(a, b, h[j])
        delta = t_star - h[j]
        if abs(delta) > 1e-15:
            corr += a * delta
            h[j] = t_star
    return h

def coord_sweep_reverse(h, n):
    m = 2 * n - 1

    s_vals = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)

def subgradient_refine(h, n, n_iters=500):
    m = 2 * n - 1
    nj = n - 1
    dx = 2.0 / n
    alpha = 0.05
    best = compute_c5(h, dx)
    h_temp = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h_temp, 1.0 - h_temp, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - nj
        grad = np.zeros(n)
        if s >= 0:
            mcnt = n - s
            grad[s:s+mcnt] += (1.0 - h_temp[:mcnt])
            grad[:mcnt] -= h_temp[s:s+mcnt]
        else:
            lo = -s
            mcnt = n - lo
            grad[lo:lo+mcnt] += (1.0 - h_temp[lo:])
            grad[:mcnt] -= h_temp[lo:]
        grad *= dx
        h_new = h_temp - alpha * grad
        h_new = project_sum(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best - 1e-12:
            h_temp = h_new
            best = c5_new
        else:
            alpha *= 0.95
            if alpha < 0.001:
                break
        if it % 100 == 99:
            alpha *= 0.8
    return h_temp, best
```
