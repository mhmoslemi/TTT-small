from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, n, T):
    dx = 2.0 / n
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip((corr - max_c) * T, -100, 100)
    s = np.sum(np.exp(shifted))
    return max_c + np.log(s) / T

def soft_max_grad(h, n, T):
    dx = 2.0 / n
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip((corr - max_c) * T, -100, 100)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    weights = exp_s / s
    first = np.convolve(weights, 1.0 - h, mode='full')[n-1:2*n-1]
    second = np.correlate(h, weights, mode='full')[n-1:2*n-1]
    return dx * (first - second)

    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-12:
        h = h * target / s
        h = np.clip(h, 0, 1)

    corr = np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    h_c = np.clip(h, 0, 1)
    weights = get_weights(h_c, n, T)
    first = np.convolve(weights, 1.0 - h_c, mode='full')[n-1:2*n-1]
    second = np.correlate(h_c, weights, mode='full')[n-1:2*n-1]
    return dx * (first - second)

def project(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(10):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        h *= target / s
        h = np.clip(h, 0, 1)
    return h

def scale_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def optimize_gd(h, n, T, lr_init, max_iter, time_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = lr_init
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad = compute_gradient(h, n, T)
        h_new = project(h - lr * grad, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Backtracking
        if c5_new > best_c5 + 1e-4:
            lr *= 0.5
        else:
            lr = min(lr * 1.05, 0.3)
        h = h_new
    return best_h, best_c5

def optimize_sa(h, n, n_steps, temp_start, temp_end, seed, time_limit):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    t0 = time.time()
    c_temp = np.log(max(temp_start,1e-8)/max(temp_end,1e-8)) / max(n_steps,1)
    for i in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        temp = temp_start * np.exp(-c_temp * i)
        step_size = min(0.5, temp * 0.3)
        h_new = project(h + rng.normal(0, step_size, n), n)
        c5_new = compute_c5(h_new, n)
        if c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(temp, 1e-10)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    else:
        n = 84
        h = np.zeros(n)
        # V-shape initial: h[i] = 1 - |2*(i+0.5)/n - 1|
        x = (np.arange(n) + 0.5) * 2.0 / n
        h = 1.0 - np.abs(x - 1.0)
        h = project(h, n)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    # Also try V-shape at n if not already
    if n >= 42:
        x = (np.arange(n) + 0.5) * 2.0 / n
        h_v = 1.0 - np.abs(x - 1.0)
        h_v = project(h_v, n)
        c5_v = compute_c5(h_v, n)
        if c5_v < best_c5:
            best_h = h_v
            best_c5 = c5_v
    
    # Phase 1: GD
    elapsed = time.time() - start_time
    t_limit = min(200, (budget_s - elapsed) * 0.2)
    h1, c5_1 = optimize_gd(best_h, n, T=30, lr_init=0.05, max_iter=200, time_limit=t_limit)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1
    
    # Phase 2: SA
    elapsed = time.time() - start_time
    t_limit = min(200, (budget_s - elapsed) * 0.2)
    n_sa = int(t_limit * 30)
    h2, c5_2 = optimize_sa(best_h, n, n_sa, temp_start=0.5, temp_end=0.01, seed=seed, time_limit=t_limit)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2
    
    # Phase 3: More GD with different T
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 100
    if remaining > 50:
        for T_try in [50, 100, 15]:
            if time.time() - start_time > budget_s - 100:
                break
            h3, c5_3 = optimize_gd(best_h, n, T=T_try, lr_init=0.03, max_iter=100, time_limit=min(100, remaining*0.3))
            if c5_3 < best_c5:
                best_h, best_c5 = h3, c5_3
    
    # Phase 4: Scale up
    for factor in [2, 4, 8]:
        n_new = n * factor
        if n_new > 512 or time.time() - start_time > budget_s - 120:
            break
        h_scaled = scale_h(best_h, len(best_h), n_new)
        c5_scaled = compute_c5(h_scaled, n_new)
        
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 60
        if remaining < 30:
            if c5_scaled < best_c5:
                best_h, best_c5 = h_scaled, c5_scaled
            break
        
        t_gd = min(remaining * 0.5, 200)
        h4, c5_4 = optimize_gd(h_scaled, n_new, T=40, lr_init=0.03, max_iter=150, time_limit=t_gd)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4
        elif c5_scaled < best_c5:
            best_h, best_c5 = h_scaled, c5_scaled
        
        # SA at new resolution
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 40
        if remaining > 30:
            t_sa = min(remaining * 0.4, 150)
            n_sa_new = int(t_sa * 20)
            h5, c5_5 = optimize_sa(best_h, len(best_h), n_sa_new, temp_start=0.3, temp_end=0.005, seed=seed+len(best_h), time_limit=t_sa)
            if c5_5 < best_c5:
                best_h, best_c5 = h5, c5_5
            
            # Final GD
            if time.time() - start_time < budget_s - 30:
                h6, c5_6 = optimize_gd(best_h, len(best_h), T=50, lr_init=0.02
