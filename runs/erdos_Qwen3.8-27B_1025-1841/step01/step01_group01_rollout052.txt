import numpy as np
import time
from scipy.optimize import minimize

_global_n = 0
_global_beta = 50.0
_global_penalty = 10000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def softmax_c5(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    return m + np.log(np.sum(e)) / beta

def penalized_obj(x):
    n = _global_n
    h = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    base = softmax_c5(h, n, _global_beta)
    return base + _global_penalty * (np.sum(h) - target) ** 2
