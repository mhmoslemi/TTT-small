import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_random_h(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def ensure_constraint(h, n, seed):
    h = h.copy()
    n_ones = int(h.sum() + 0.5)
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        to_flip = rng.choice(ones, n_ones - target, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        to_flip = rng.choice(zeros, target - n_ones, replace=False)
        h[to_flip] = 1.0
    return h

def greedy_search(h, time_limit, seed):
    n = len(h)
    c5 = compute_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_improve = 1e-14
        best_i, best_j = -1, -1

        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                h[j] = 0.0
                h[i] = 1.0
                improve = c5 - new_c5
                if improve > best_improve:
                    best_improve = improve
                    best_i, best_j = i, j

        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        c5 = compute_c5(h)

    return h, c5

def sa_search(h, time_limit, seed, c5_init):
    n = len(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    T = 0.004
    T_min = 0.0002
    cooling = 0.99995
    c5 = c5_init
    best_c5 = c5
    best_h = h.copy()

    while time.time() - start < time_limit and T > T_min:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - c5

        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[j] = 0.0
            h[i] = 1.0

        T *= cooling

    return best_h, best_c5
