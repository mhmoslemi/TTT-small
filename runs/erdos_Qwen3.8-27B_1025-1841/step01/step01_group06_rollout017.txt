For mode="full", the output has length len(a) + len(v) - 1 = 2n - 1.
The index k in the output corresponds to the "lag" of c = k - (len(v) - 1) = k - (n-1).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        adj = diff / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.max(np.abs(h_new - h)) < 1e-15:
            h = h_new
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-13:
        if rem > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            h = h.copy()
            h[mask] += rem / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_constraint(h_new, n_new)

def smooth_grad(h, n, temp):
    corr = compute_corr(h)
    m = np.max(corr)
    exp_c = np.exp((corr - m) / temp)
    w = exp_c / np.sum(exp_c)
    lags = np.arange(2 * n - 1) - (n - 1)
    i_arr = np.arange(n)[:, None]
    l_arr = lags[None, :]
    idx1 = i_arr - l_arr
    v1 = (idx1 >= 0) & (idx1 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2 = i_arr + l_arr
    v2 = (idx2 >= 0) & (idx2 < n)
    idx2c = np.clip(idx2, 0, n - 1)
    part1 = np.sum(w[None, :] * (1.0 - h[idx1c]) * v1, axis=1)
    part2 = np.sum(w[None, :] * h[idx2c] * v2, axis=1)
    return part1 - part2

def subgrad_one(h, n, k):
    lag = k - (n - 1)
    grad = np.zeros(n)
    i = np.arange(n)
    j1 = i - lag
    v1 = (j1 >= 0) & (j1 < n)
    grad[v1] += 1.0 - h[j1[v1]]
    j2 = i + lag
    v2 = (j2 >= 0) & (j2 < n)
    grad[v2] -= h[j2[v2]]
    return grad

def smooth_descent(h, n, dx, t_end, max_iter=5000):
    c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    temps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for temp in temps:
        if time.time() > t_end:
            break
        alpha = 1.5 / n
        stuck = 0
        for it in range(max_iter // len(temps)):
            if time.time() > t_end:
                break
            grad = smooth_grad(h, n, temp)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            step = alpha * grad * min(1.0, 3.0 / gn)
            h_new = project_constraint(h - step, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5 - 1e-13:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                stuck = 0
                alpha = min(alpha * 1.04, 5.0 / n)
            else:
                stuck += 1
                alpha *= 0.85
                if stuck > 400:
                    break
    return best_h, best_c5

def subgrad_descent(h, n, dx, t_end, max_iter=8000):
    c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    alpha = 2.0 / n
    stuck = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        k_star = int(np.argmax(corr))
        grad = subgrad_one(h, n, k_star)
        h_new = project_constraint(h - alpha * grad, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5 - 1e-13:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stuck = 0
            alpha = min(alpha * 1.01, 6.0 / n)
        else:
            stuck += 1
            alpha *= 0.93
            if stuck > 600:
                break
    return best_h, best_c5

def slsqp_polish(h, n, dx, t_end):
    temp = 0.003
    target = n / 2.0

    def obj_fn(x):
        hx = np.clip(x, 0.0, 1.0)
        c = compute_corr(hx)
        mx = np.max(c)
        return (mx + temp * np.log(np.sum(np.exp((c - mx) / temp)))) * dx

    def grad_fn(x):
        hx = np.clip(x, 0.0, 1.0)
        return smooth_grad(hx, n, temp) * dx

    try:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': 250, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            ht = project_constraint(res.x, n)
            ct = compute_c5(ht, dx)
            return ht, ct
    except Exception:
        pass
    return h, compute_c5(h, dx)
