```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_manifold(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    m = np.max(g)
    return float(m + np.log(np.sum(np.exp(T * (g - m)))) / T)

def compute_softmax_p(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    m = np.max(g)
    exp_g = np.exp(T * (g - m))
    return exp_g / np.sum(exp_g)

def compute_grad(h, p, dx):
    n = len(h)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ps = p[s + n - 1]
        if ps < 1e-16:
            continue
        lo = max(0, -s)
        hi = min(n-1, n-1-s)
        grad[lo:hi+1] += ps * one_minus_h[lo+s:hi+s+1]
        if s > 0:
            grad[s:n] -= ps * h[:n-s]
        else:
            if n + s > 0:
                grad[:n+s] -= ps * h[-s:]
    return grad * dx

def slsqp_with_grad(h0, n, T, maxiter=200):
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        return soft_max_obj(x, T)
    
    def grad(x):
        p = compute_softmax_p(x, T)
        return compute_grad(x, p, dx)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=cons,
                       options={'ftol': 1e-12, 'maxiter': maxiter})
        if np.all(np.isfinite(res.x)):
            return project_to_manifold(res.x, n)
    except Exception:
        pass
    return h0.copy()

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)
    
    best_h = project_to_manifold(h_start, len(h_start))
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Adaptive temperature at starting resolution
    n0 = len(best_h)
    T_schedule = [20, 40, 80, 160, 320]
    h_work = best_h.copy()
    
    for T in T_schedule:
        if time.time() > deadline - 60:
            break
        h_new = slsqp_with_grad(h_work, n0, T, maxiter=300)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_work = h_new
    
    # Phase 2: Multi-resolution refinement
    for n_target in [n0 * 2, n0 * 3]:
        if time.time() > deadline - 60:
            break
        if n_target > 2000:
            continue
        h_up = interp_to(best_h, len(best_h), n_target)
        h_up = project_to_manifold(h_up, n_target)
        for T in [40, 80, 200]:
            if time.time() > deadline - 30:
                break
            h_new = slsqp_with_grad(h_up, n_target, T, maxiter=200)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_up = h_new
    
    # Phase 3: Random restarts at best resolution
    n_best = len(best_h)
    for attempt in range(20):
        if time.time() > deadline - 20:
            break
        h_pert = best_h + np.random.randn(n_best) * 0.1
        h_pert = project_to_manifold(h_pert, n_best)
        T = 100 + np.random.rand() * 100
        h_new = slsqp_with_grad(h_pert, n_best, T, maxiter=150)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    
    # Phase 4: Final polishing
    if time.time() < deadline - 10:
        h_new = slsqp_with_grad(best_h, n_best, 500, maxiter=200)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
