```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def objective_cont(h, dx, n):
    """Continuous objective with penalty for constraint violation."""
    h_clipped = np.clip(h, 0, 1)
    c5 = compute_c5(h_clipped, dx)
    # Penalty for sum constraint
    penalty = 10.0 * (np.sum(h) - n/2)**2
    # Penalty for box constraint
    penalty += 10.0 * (np.sum(np.maximum(0, -h))**2 + np.sum(np.maximum(0, h-1))**2)
    return c5 + penalty

def local_search_binary(h, dx, n, max_iters=100000, time_limit=60):
    """Stochastic local search on binary sequences."""
    h = h.copy()
    binary_h = (h > 0.5).astype(float)
    c5 = compute_c5(binary_h, dx)
    
    one_positions = np.where(binary_h == 1)[0]
    zero_positions = np.where(binary_h == 0)[0]
    
    start = time.time()
    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        i = np.random.choice(one_positions)
        j = np.random.choice(zero_positions)
        
        # Swap
        binary_h[i] = 0
        binary_h[j] = 1
        new_c5 = compute_c5(binary_h, dx)
        
        if new_c5 <= c5:
            c5 = new_c5
            one_positions = np.where(binary_h == 1)[0]
            zero_positions = np.where(binary_h == 0)[0]
        else:
            # Revert
            binary_h[i] = 1
            binary_h[j] = 0
        it += 1
    
    return binary_h, c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def enforce_binary(h, n):
    h = (h > 0.5).astype(float)
    ones = int(h.sum())
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, ones - target, replace=False)] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, target - ones, replace=False)] = 1
    return h

def local_search(h_init, dx, n, time_limit, rng):
    h = enforce_binary(h_init, n)
    c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    temp = 0.002
    patience = 0
    
    while time.time() - t0 < time_limit:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        i = oi[rng.randint(len(oi))]
        j = zi[rng.randint(len(zi))]
        h[i], h[j] = 0.0, 1.0
        nc5 = compute_c5(h, dx)
        d = nc5 - c5
        if d < 0 or rng.rand() < np.exp(-d / max(temp, 1e-8)):
            c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            h[i], h[j] = 1.0, 0.0
        temp *= 0.9997
        if patience > 8000:
            patience = 0
            temp = 0.002  # restart temperature
    
    return best_h, best_c5

def continuous_refine(h_init, n, dx, time_limit, rng):
    """Refine using smooth optimization."""
    h = h_init.astype(float).copy()
    h = np.clip(h, 0.01, 0.99)  # Avoid boundaries
    # Project to sum constraint
    h = h / h.sum() * (n / 2.0)
    h = np.clip(h, 0.0, 1.0)
    
    beta = 50.0
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        # Smooth max
        m = np.max(corr)
        return np.sum(corr * np.exp(beta * (corr - m))) / len(corr) * 0 + m + np.sum(np.logaddexp.reduce(beta * (corr - m))) / (len(corr) * beta)
        # Actually simpler: just use the max plus a penalty
        # Or use: m + (1/beta) * log(sum(exp(beta*(corr-m))))
    
    # Simpler: just use the raw max (L-BFGS-B can handle non-smooth)
    def obj2(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        penalty = 100.0 * (np.sum(x) - n/2.0)**2
        return np.max(corr) + penalty
    
    res = minimize(obj2, h, method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': 200, 'maxfun': 2000})
    
    h_res = np.clip(res.x, 0, 1)
```
