import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_soft_and_rho(corr, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    m = np.max(shifted)
    exps = np.exp(shifted - m)
    S = np.sum(exps)
    rho = exps / S
    soft = (mc + m) + np.log(S) / T  # = mc + m/T + log(S)/T ... wait
    # Actually: (1/T)*log(sum exp(T*corr)) = (1/T)*log(exp(T*mc)*exp(m)*S) = mc + m/T + log(S)/T
    # Hmm: T*corr - T*mc = shifted. m = max(shifted). 
    # sum exp(T*corr) = exp(T*mc) * sum exp(shifted) = exp(T*mc) * exp(m) * S
    # (1/T)*log(sum exp(T*corr)) = mc + m/T + log(S)/T
    soft = mc + m / T + np.log(S) / T
    return soft, rho

def compute_gradient(v, T, n, dx, corr):
    offset = n - 1
    _, rho = compute_soft_and_rho(corr, T)
    grad = np.zeros(n)
    omv = 1.0 - v
    for k in range(2 * n - 1):
        r = rho[k]
        if r < 1e-18:
            continue
        l = k - offset
        if l >= 0:
            if l < n:
                grad[:n-l] += r * omv[l:]
                grad[l:] -= r * v[:n-l]
        else:
            m = -l
            if m < n:
                grad[m:] += r * omv[:n-m]
                grad[:n-m] -= r * v[m:]
    grad *= dx
    return grad

def project_onto_simplex(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    proj = np.clip(v - tau, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0 - 1e-15
        else:
            mask = proj > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            adj = diff / cnt
            proj[mask] = np.clip(proj[mask] + adj, 0.0, 1.0)
    return proj

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    return project_onto_simplex(h, n * 0.5)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n) * (2.0 / old_n)
    new_x = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def pgd_optimize(h, n, T_start, T_end, max_iter, deadline):
    dx = 2.0 / n
    target = n * 0.5
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    momentum = np.zeros(n)
    beta = 0.9
    stall = 0
    max_stall = 100
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        frac = it / max(1, max_iter - 1)
        T = T_start + (T_end - T_start) * frac
        
        corr = compute_corr(h)
        grad = compute_gradient(h, T, n, dx, corr)
        
        momentum = beta * momentum + grad
        direction = -momentum
        
        # Line search
        soft_old, _ = compute_soft_and_rho(corr, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Initial step size based on gradient magnitude
        alpha = min(1.0, 0.1 / (grad_norm + 1e-8))
        
        accepted = False
        for _ in range(30):
            h_trial = project_onto_simplex(h + alpha * direction, target)
            if not np.allclose(h_trial, h, atol=1e-14):
                corr_t = compute_corr(h_trial)
                soft_t, _ = compute_soft_and_rho(corr_t, T)
                if soft_t < soft_old - 1e-10:
                    accepted = True
                    break
            alpha *= 0.5
        if not accepted:
            stall += 1
            if stall > max_stall:
                break
            continue
        
        h = h_trial
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def slow_pgd(h, n, deadline, base_T=100):

def coord_polish(h, n, deadline):
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    for rnd in range(200):
        if time.time() > deadline - 5:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline - 5:
                return h, best_c5
            old = h[j]
            cands = np.array([0.0, 1.0, 0.5])
            if old > 0.02:
                cands = np.append(cands, old - 0.01)
            if old < 0.98:
                cands = np.append(cands, old + 0.01)
            if old > 0.05:
                cands = np.append(cands, old - 0.03)
            if old < 0.95:
                cands = np.append(cands, old + 0.03)
            cands = cands[(cands >= 0) & (cands <= 1)]
            best_cv = old
            best_c = best_c5
            for cv in cands:
                if abs(cv - old) < 1e-9:

                improved = True
            else:
                h[j] = old
                h = make_feasible(h, n)
        if not improved:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_soft_and_rho(corr, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    m = np.max(shifted)
    exps = np.exp(shifted - m)
    S = np.sum(exps)
    rho = exps / S
    soft = mc + m / T + np.log(S) / T
    return soft, rho

def compute_gradient(v, T, n, dx, corr):
    offset = n - 1
    _, rho = compute_soft_and_rho(corr, T)
    grad = np.zeros(n)
    omv = 1.0 - v
    for k in range(2 * n - 1):
        r = rho[k]
        if r < 1e-18:
            continue
        l = k - offset
        if l >= 0:
            if l < n:
                grad[:n - l] += r * omv[l:]
                grad[l:] -= r * v[:n - l]
        else:
            m = -l
            if m < n:
                grad[m:] += r * omv[:n - m]
                grad[:n - m] -= r * v[m:]
    grad *= dx
    return grad

def project_onto_simplex(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    proj = np.clip(v - tau, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0 - 1e-15
        else:
            mask = proj > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] = np.clip(proj[mask] + diff / cnt, 0.0, 1.0)
    return proj

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    return project_onto_simplex(h, n * 0.5)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n) * (2.0 / old_n)
    new_x = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def pgd_optimize(h, n, T_start, T_end, max_iter, deadline):
    dx = 2.0 / n
    target = n * 0.5
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    momentum = np.zeros(n)
    beta = 0.9
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = it / max(1, max_iter - 1)
        T = T_start + (T_end - T_start) * frac
        corr = compute_corr(h)
        grad = compute_gradient(h, T, n, dx, corr)
        momentum = beta * momentum + grad
        direction = -momentum
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        soft_old, _ = compute_soft_and_rho(corr, T)
        alpha = min(2.0 / T, 2.0 / (gnorm * n * 0.5 + 1e-8))
        if alpha > 5.0:
            alpha = 5.0
        accepted = False
        for _ in range(35):
            h_trial = project_onto_simplex(h + alpha * direction, target)
            if not np.allclose(h_trial, h, atol=1e-13):
                corr_t = compute_corr(h_trial)
                soft_t, _ = compute_soft_and_rho(corr_t, T)
                if soft_t < soft_old - 1e-11:
                    accepted = True
                    break
            alpha *= 0.5
        if not accepted:
            stall += 1
            if stall > 150:
                break
            continue
        h = h_trial
        c5_now = compute_c5(h)

            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5
