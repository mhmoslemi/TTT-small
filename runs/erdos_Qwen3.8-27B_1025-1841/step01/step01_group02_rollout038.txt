```python
def lbfgs_objective(h, n, t, penalty_coeff):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(t * shifted))) / t
    pen = penalty_coeff * (np.sum(h) - n/2.0)**2
    return soft * dx + pen

def lbfgs_gradient(h, n, t, penalty_coeff):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(t * shifted)
    weights /= np.sum(weights)
    # dc_k/dh[m] computation
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        for m in range(n):
            val = 0.0
            # h[m] appears as first factor: 1-h[m+k-n+1] if valid
            j1 = m + k - n + 1
            if 0 <= j1 < n:
                val += (1.0 - h[j1])
            # h[m] appears in (1-h[...]): -h[m-k+n-1] if valid
            j2 = m - k + n - 1
            if 0 <= j2 < n:
                val -= h[j2]
            grad[m] += weights[k] * val
    grad *= dx
    # Add penalty gradient
    grad += 2.0 * penalty_coeff * (np.sum(h) - n/2.0)
    return grad
```

For L ≥ 0:

- Second term: p+L ranges from 0 to n-1, so p ranges from 0 to n-1-L. h[p+L] = h[L], h[L+1], ..., h[n-1]

For L < 0, let L = -M where M > 0:
- First term: p-L = p+M ranges from 0 to n-1, so p ranges from 0 to n-1-M. (1-h[p+M]) = 1-h[M], 1-h[M+1], ..., 1-h[n-1]
- Second term: p+L = p-M ranges from 0 to n-1, so p ranges from M to n-1. h[p-M] = h[0], h[1], ..., h[n-1-M]

```python
def _soft_max_grad_vectorized(h):
    n = _G_N
    t = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = corr - mc
    exp_t = np.exp(t * shifted)
    sum_exp = exp_t.sum()
    weights = exp_t / sum_exp
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        L = k - (n - 1)
        if L >= 0:
            # First term: for p = L..n-1, add (1-h[p-L]) * wk
            grad[L:] += wk * (1.0 - h[:n-L])
            # Second term: for p = 0..n-1-L, subtract h[p+L] * wk
            grad[:n-L] -= wk * h[L:]
        else:
            M = -L
            # First term: for p = 0..n-1-M, add (1-h[p+M]) * wk
            grad[:n-M] += wk * (1.0 - h[M:])
            # Second term: for p = M..n-1, subtract h[p-M] * wk
            grad[M:] -= wk * h[:n-M]
    
    grad *= dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 40.0
_G_PENALTY = 1e5

def _soft_max_obj(h):
    n = _G_N
    t = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(t * shifted))) / t) * dx

def _soft_max_grad_vec(h):
    n = _G_N
    t = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = corr - mc
    exp_t = np.exp(t * shifted)
    sum_exp = exp_t.sum()
    weights = exp_t / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        L = k - (n - 1)
        if L >= 0:
            m = n - L
            grad[L:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[L:]
        else:
            M = -L
            m = n - M
            grad[:m] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:m]
    grad *= dx
    return grad

def _pen_obj(h):
    return _soft_max_obj(h) + _G_PENALTY * (np.sum(h) - _G_N / 2.0) ** 2

def _pen_grad(h):
    g = _soft_max_grad_vec(h)
    g += 2.0 * _G_PENALTY * (np.sum(h) - _G_N / 2.0)
    return g

def _con_fun(h):
    return np.sum(h) - _G_N / 2.0

def _con_jac(h):
    return np.ones(_G_N)

def _make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float) / old_n
    new_x = np.arange(new_n, dtype=float) / new_n
    h_new = np.interp(new_x, old_x, h)
    return _make_feasible(h_new, new_n)

def _ternary_c5(h, j):
    n = len(h)
    dx = 2.0 / n
    lo, hi = 0.0, 1.0
    target = _G_N / 2.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        t1 = h.copy()
        t1[j] = m1
        t1 = _make_feasible(t1, n)
        c1 = _compute_c5(t1)
        t2 = h.copy()
        t2[j] = m2
        t2 = _make_feasible(t2, n)
        c2 = _compute_c5(t2)
        if c1 < c2:
            hi = m2
        else:
            lo = m1
    best_v = (lo + hi) / 2.0
    trial = h.copy()
    trial[j] = best_v
    trial = _make_feasible(trial, n)
    return trial, _compute_c5(trial)

    h0 = None

    def elapsed():
        return time.time() - start_time

        return budget_s - elapsed()
```
