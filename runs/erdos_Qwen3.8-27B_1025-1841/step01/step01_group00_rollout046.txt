import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    thresh = 1e-9
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < thresh:
            continue
        sh = k - n + 1
        if sh >= 0:
            if sh < n:
                grad[sh:n] += pk * (1.0 - h[0:n - sh])
                grad[0:n - sh] -= pk * h[sh:n]
        else:
            s2 = -sh
            if s2 < n:
                grad[0:n - s2] += pk * (1.0 - h[s2:n])
                grad[s2:n] -= pk * h[0:n - s2]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def adam_optimize(h, n, tau, max_iters, time_limit, lr, beta1, beta2, restart_every):

    best_h = h.copy()

    m = np.zeros(n)
    v2 = np.zeros(n)
    eps = 1e-8
    t = 0
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > restart_every:
            h = best_h + 0.02 * np.random.randn(n)
            h = project_to_feasible(h, n)
            m = np.zeros(n)
            v2 = np.zeros(n)
            t = 0
            stall = 0
            lr = max(lr * 0.7, 0.003)
            continue
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v2_hat) + eps)
        h = project_to_feasible(h_new, n)
    return best_h, best_c5


def subgrad_optimize(h, n, max_iters, time_limit, lr):
    best_c5 = compute_c5(h)
    best_h = h.copy()

    beta = 0.9
    stall = 0

        if time.time() - start > time_limit:

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        if sh >= 0:
            if sh < n:
                grad[sh:n] += (1.0 - h[0:n - sh])
                grad[0:n - sh] -= h[sh:n]
        else:

                grad[s2:n] -= h[0:n - s2]
