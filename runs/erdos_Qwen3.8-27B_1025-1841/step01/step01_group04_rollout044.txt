    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_obj(x, n, T):
    dx = 2.0 / n
    corr = compute_corr(x)
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad(x, n, T):
    dx = 2.0 / n
    corr = compute_corr(x)
    scaled = T * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    
    w_rev = w[::-1]
    # first_sum[k] = sum_i (1-x[i]) * w[n-1+i-k]

def project_feasible(h, n):

        if d > 0:

        else:

        cnt = np.sum(mask)
        if cnt == 0:

        h[mask] += d / cnt
