import time

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_max(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
    else:
        a = -lag
        if n - a > 0:
            grad[:n-a] += (1.0 - h[a:])
            grad[a:] -= h[:n-a]
    return grad, k_star

def _smooth_obj_grad(h, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    n = len(h)
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / eps)
    Z = np.sum(exp_c)
    obj = max_c + eps * np.log(Z)
    w = exp_c / Z
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += w[k] * (1.0 - h[:n-lag])
                grad[:n-lag] -= w[k] * h[lag:]
        else:
            a = -lag
            if n - a > 0:
                grad[:n-a] += w[k] * (1.0 - h[a:])
                grad[a:] -= w[k] * h[:n-a]
    return obj, grad

def _project(y, target_sum):

        x = np.clip(y - mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid

def _obj_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_single_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:] += (1.0 - h[:m])
            g[:m] -= h[lag:]
    else:
        a = -lag
        m = n - a
        if m > 0:
            g[:m] += (1.0 - h[a:])
            g[a:] -= h[:m]
    return g

def _smooth_obj_grad(h, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    n = len(h)
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / eps)
    Z = np.sum(exp_c)
    obj = max_c + eps * np.log(Z)
    w = exp_c / Z
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-9:
            continue
        lag = k - (n - 1)
        gk = _grad_single_lag(h, lag, n)
        grad += w[k] * gk
    return obj, grad
