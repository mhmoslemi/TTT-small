import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    return np.max(r) * dx

def smooth_max(r, beta):
    m = np.max(r)
    return m + np.log(np.sum(np.exp(r - m))) / beta

def smooth_max_gradient(r, beta, h, n):
    """Compute gradient of smooth max of correlation values."""
    dx = 2.0 / n
    r_max = np.max(r)
    w = np.exp(beta * (r - r_max))
    w /= np.sum(w)
    # w has length 2n-1
    # grad_j = dx * [sum_{i=0}^{j} w[j-i] * (1-h[i]) - sum_{i=j}^{n-1} w[i-j] * h[i]]
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_{i=0}^{j} w[j-i] * (1-h[i])
        term1 = np.dot(w[:j+1][::-1], (1.0 - h[:j+1]))
        # Second term: sum_{i=j}^{n-1} w[i-j] * h[i]
        term2 = np.dot(w[:n-j], h[j:])
        grad[j] = dx * (term1 - term2)
    return grad

def project_onto_constraint(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}."""
    target = n / 2
    # Find t such that sum(clip(h+t, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def projected_gradient_descent(h, n, beta, lr, max_iter, time_limit):
    """Run projected gradient descent with smooth max objective."""
    start = time.time()
    h = h.copy().astype(float)
    h = project_onto_constraint(h, n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    prev_obj = smooth_max(np.correlate(h, 1.0-h, mode='full') * (2.0/n), beta)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        r = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        obj = smooth_max(r, beta)
        grad = smooth_max_gradient(r, beta, h, n)
        
        # Backtracking line search
        step = lr
        h_new = h
        for _ in range(30):
            h_new = h - step * grad
            h_new = project_onto_constraint(h_new, n)
            r_new = np.correlate(h_new, 1.0 - h_new, mode='full') * (2.0 / n)
            obj_new = smooth_max(r_new, beta)
            if obj_new < obj - 1e-4 * step * np.dot(grad, h_new - h):
                break
            step *= 0.5
        
        h = h_new
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Convergence check
        if abs(obj - prev_obj) < 1e-8:
            break
        prev_obj = obj
    
    return best_h, best_c5

def make_binary_init(n, seed):
    """Create a random binary-like initial guess."""
    rng = np.random.RandomState(seed)
    h = rng.randint(0, 2, size=n).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        to_flip = rng.choice(ones_idx, size=n_ones - target, replace=False)
        h[to_flip] = 0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zeros_idx, size=target - n_ones, replace=False)
        h[to_flip] = 1
    return h

def make_block_init(n, n_blocks, seed):
    """Create a block-structured initial guess."""
    rng = np.random.RandomState(seed)
    block_size = n // (2 * n_blocks)
    h = np.zeros(n)
    for i in range(0, n, 2 * block_size):
        end = min(i + block_size, n)
        h[i:end] = 1.0
    # Adjust

    target = n // 2
    if n_ones != target:
        # Flip some
        if n_ones > target:

            to_flip = rng.choice(ones_idx, size=n_ones - target, replace=False)
            h[to_flip] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            to_flip = rng.choice(zeros_idx, size=target - n_ones, replace=False)
            h[to_flip] = 1
    return h

            h_gold[positions] = 1.0

Wait, actually the analytical computation in a loop is also O(n²) but with lower constant. Let me just do it in a loop:

```python
def compute_gradient(h, n, beta):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    r_max = np.max(r)
    w = np.exp(beta * (r - r_max))
    w /= np.sum(w)
    # w has length 2n-1, w[m] corresponds to lag m-(n-1)
    
    grad = np.zeros(n)
    for p in range(n):
        s = 0.0
        # Sum over all lags ℓ = -(n-1) to (n-1)
        for lag in range(-(n-1), n):
            m = lag + (n-1)  # index into w
            wm = w[m]
            if wm < 1e-15:
                continue
            # Contribution from this lag
            # d(r_lag)/d(h[p]) = dx * [(1-h[p-lag])*I(0≤p-lag<n) - h[p+lag]*I(0≤p+lag<n)]
            contrib = 0.0
            if 0 <= p - lag < n:
                contrib += (1.0 - h[p - lag])
            if 0 <= p + lag < n:
                contrib -= h[p + lag]
            s += wm * contrib
        grad[p] = dx * s
    return grad
```

This is O(n²) per gradient. For n=200, that's 40000 iterations of the inner loop, each doing a few ops. Very fast.
