import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_target(np.clip(h_new, 0.0, 1.0), n_new)

def smooth_max_and_grad(h, dx, temp):
    n = len(h)
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode="full")
    cm = np.max(corr)
    cs = corr - cm
    ev = np.exp(cs / temp)
    ws = ev.sum()
    w = ev / ws
    obj = (cm + temp * np.log(ws)) * dx
    grad = np.zeros(n)
    lags = np.arange(len(corr)) - (n - 1)
    for i in range(len(corr)):
        wi = w[i]
        if wi < 1e-11:
            continue
        k = lags[i]
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:] += wi * (1.0 - hc[:m])
                grad[:m] -= wi * hc[k:]
        else:
            m = -k
            if m < n:
                grad[:n - m] += wi * (1.0 - hc[m:])
                grad[m:] -= wi * hc[:n - m]
    grad *= dx
    return obj, grad

def pgd_anneal(h, dx, temps, iters_per_temp, time_limit):
    n = len(h)
    alpha = 5.0 / n
    h_work = h.copy()
    for temp in temps:
        for it in range(iters_per_temp):
            if time.time() > time_limit:
                return h_work
            obj, grad = smooth_max_and_grad(h_work, dx, temp)
            t = alpha
            improved = False
            for _ in range(20):
                h_trial = project_target(h_work - t * grad, n)
                obj_trial, _ = smooth_max_and_grad(h_trial, dx, temp)
                if obj_trial < obj:
                    h_work = h_trial
                    improved = True
                    break
                t *= 0.5
            if improved:
                alpha = min(t * 1.03, 2.0)
            else:
                alpha *= 0.6
    return h_work

    h = rng.uniform(0.3, 0.7, n)
