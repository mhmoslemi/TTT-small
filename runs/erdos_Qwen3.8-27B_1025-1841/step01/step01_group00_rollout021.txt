```python
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def smooth_max_obj_grad(h, tau, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corrs = len(corr)
    
    if top_k is not None and top_k < n_corrs:
        # Use top-k only
        idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        idx = np.arange(n_corrs)
    
    corr_sk = corr[idx]
    max_c = corr_sk.max()
    shifted = np.clip(tau * (corr_sk - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    grad = np.zeros(n)
    for ii in range(len(idx)):
        pk = p[ii]
        if pk < 1e-14:
            continue
        k = idx[ii]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            indices = np.arange(j0, j1)
            grad[indices] += pk * (1.0 - h[indices - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            indices = np.arange(j0b, j1b)
            grad[indices] -= pk * h[indices + sh]
    grad *= dx
    return obj, grad, c5_val(corr, dx)

def c5_val(corr, dx):
    return corr.max() * dx
```

Hmm wait, I realize the gradient computation with the loop over k is the bottleneck. For n=1800 and top_k=100, that's 100 iterations each doing O(n) work = 180,000 operations. That should be fine in numpy.

But for the full sum (top_k = 2n-1 = 3599), it's 3599 * 1800 ≈ 6.5M operations, which might be slow in a Python loop.

Let me optimize the gradient computation by restructuring:

For each shift sh, the contribution to grad is:
grad[j] += pk * (1 - h[j-sh]) for valid j (where j-sh ∈ [0,n))
grad[j] -= pk * h[j+sh] for valid j (where j+sh ∈ [0,n))

I can precompute for each shift the indices and use vectorized numpy operations. But the loop over shifts is still there.

Alternative: Use the fact that the gradient is a sum of shifted correlations:
grad = dx * (sum_k p_k * shift(1-h, sh_k) - sum_k p_k * shift(h, sh_k))

where shift(f, s)[j] = f[j-s] if j-s ∈ [0,n), 0 otherwise.

This can be computed as:
grad = dx * (cross_correlate(p_shifted, 1-h) - cross_correlate(p_shifted, h))

where p_shifted is a sequence of length 2n-1 with p_k at position k.

Actually, this is getting complicated. Let me just use the loop and accept the cost. For n=900 and top_k=50, it's fast enough.
