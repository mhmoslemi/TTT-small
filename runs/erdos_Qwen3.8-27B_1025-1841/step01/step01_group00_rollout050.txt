    grad *= dx
    return obj, grad


def nesterov_smooth(h, tau, max_iters, time_limit, lr_init=0.15):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    obj0, _ = smooth_obj(h, tau)
    best_obj = obj0
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj(h, tau)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 120:
            lr *= 0.6
            v *= 0.4
            stall = 0
            if lr < 1e-10:
                break
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_obj


def subgrad_weighted(h, k_top=5, temp=0.01, max_iters=20000, time_limit=60, lr_init=0.006):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    _, corr = compute_c5(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 400:
            lr *= 0.6
            v *= 0.4
            stall = 0
            if lr < 1e-13:
                break
        top_k = np.argsort(corr)[-k_top:]
        cmax = corr.max()
        w = np.exp(-(cmax - corr[top_k]) / (temp * dx + 1e-15))
        w = w / w.sum()
        grad = np.zeros(n)
        for ii in range(len(top_k)):
            k = top_k[ii]
            pk = w[ii]
            sh = k - n + 1
            i0 = max(0, -sh)
            i1 = min(n, n - sh)
            if i0 < i1:
                grad[i0:i1] += pk * (1.0 - h[i0 + sh:i1 + sh])
                grad[i0 + sh:i1 + sh] -= pk * h[i0:i1]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def perturb(h, sigma):
    n = len(h)
    noise = np.random.randn(n) * sigma
    return project_to_feasible(h + noise, n)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n = len(h_init)
    best_h = h_init.copy()
    best_c5, _ = compute_c5(best_h)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase 1: Multi-tau smooth optimization
    for tau, lr in [(25.0, 0.18), (12.0, 0.12), (6.0, 0.08)]:
        if remaining() < 50:
            break
        h_cur = best_h.copy()
        tl = min(remaining() * 0.15, 150)
        h_cur, _ = nesterov_smooth(h_cur, tau, max_iters=20000, time_limit=tl, lr_init=lr)
        c5_cur, _ = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 2: Weighted subgradient
    if remaining() > 50:
        h_cur = best_h.copy()
        h_cur, c5_cur = subgrad_weighted(h_cur, k_top=8, temp=0.02, max_iters=50000, time_limit=remaining() * 0.18, lr_init=0.005)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 3: Coarser resolution exploration then refine
    if remaining() > 150:
        n_coarse = n // 3
        h_coarse = interp_periodic(best_h, n_coarse)
        h_coarse = project_to_feasible(h_coarse, n_coarse)
        h_coarse, _ = nesterov_smooth(h_coarse, 10.0, max_iters=8000, time_limit=min(remaining() * 0.12, 100), lr_init=0.2)
        h_refined = interp_periodic(h_coarse, n)
        h_refined = project_to_feasible(h_refined, n)
        h_refined, _ = nesterov_smooth(h_refined, 15.0, max_iters=5000, time_limit=min(remaining() * 0.08, 60), lr_init=0.08)
        c5_r, _ = compute_c5(h_refined)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_refined.copy()

    # Phase 4: High-tau smooth (near-exact max tracking)
    if remaining() > 80:
        h_cur = best_h.copy()
        for tau, lr in [(60.0, 0.06), (100.0, 0.04)]:
            if remaining() < 30:
                break
            h_cur, _ = nesterov_smooth(h_cur, tau, max_iters=10000, time_limit=min(remaining() * 0.1, 80), lr_init=lr)
            c5_cur, _ = compute_c5(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()

    # Phase 5: Perturbation restarts
    if remaining() > 100:
        for sig in [0.015, 0.03, 0.008]:
            if remaining() < 40:
                break
            h_pert = perturb(best_h, sig)
            h_pert, _ = nesterov_smooth(h_pert, 15.0, max_iters=6000, time_limit=min(remaining() * 0.1, 60), lr_init=0.1)
            h_pert, c5_pert = subgrad_weighted(h_pert, k_top=5, temp=0.01, max_iters=15000, time_limit=min(remaining() * 0.08, 50), lr_init=0.004)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    # Phase 6: Final fine subgradient
    if remaining() > 30:
        h_final = best_h.copy()
        h_final, c5_final = subgrad_weighted(h_final, k_top=4, temp=0.005, max_iters=60000, time_limit=remaining() * 0.35, lr_init=0.002)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: One more smooth pass with very high tau
    if remaining() > 40:
        h_cur = best_h.copy()
        h_cur, _ = nesterov_smooth(h_cur, 200.0, max_iters=8000, time_limit=min(remaining() * 0.2, 60), lr_init=0.03)
        c5_cur, _ = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    best_h = project_to_feasible(best_h, n)
    final_c5, _ = compute_c5(best_h)
    return best_h, final_c5, n
