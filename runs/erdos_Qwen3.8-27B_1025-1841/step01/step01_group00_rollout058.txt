import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_to(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0, 2.0, n, endpoint=False)
    x_new = np.linspace(0, 2.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, h)
