import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def subgradient_max(h, n, top_k=5, window_frac=0.1):

    active_all = np.unique(np.concatenate([active, top_idx]))

    lag_offset = n - 1
    grad = np.zeros(n)

        if lag >= 0:

            # h[a] appears as second arg (1-h[a]) for a in [lag, n-1]

            # h[a] appears as second arg: a-lag = a+lag_neg in [0,n) means a in [0, n-lag_neg-1]
            if lag_neg > 0:
                grad[0:n-lag_neg] += w * (-h[lag_neg:n])
    
    grad *= dx
    return grad, max_val

        return h0
    return res_x

def subgrad_descent(h0, n, n_iter, deadline, top_k=5, window_frac=0.05):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    lr = 0.01
    patience = 0
    max_patience = 50
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        grad, cur_max = subgradient_max(h, n, top_k=top_k, window_frac=window_frac)
        # Take step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5 - 1e-10:
            h = h_new
            best_c5 = c5_new
            patience = 0
            # Adaptive: increase lr slightly if improving
            lr = min(lr * 1.05, 0.05)
        else:
            patience += 1
            lr *= 0.85
            if patience > max_patience:
                # Try a random nudge
                h_pert = h + np.random.randn(n) * 0.02
                h_pert = project_feasible(h_pert, n)
                c5_pert = evaluate_c5(h_pert)
                if c5_pert < best_c5:
                    h = h_pert
                    best_c5 = c5_pert
                    patience = 0
                    lr = 0.02
                else:
                    break
    
    return h, best_c5

def local_search(h0, n, n_trials, deadline):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    rng = np.random.RandomState(12345)
    
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        # Random perturbation
        scale = 0.03 * max(1, 1 - trial / n_trials)
        h_pert = h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n)
        c5_pert = evaluate_c5(h_pert)
        if c5_pert < best_c5 - 1e-9:
            h = h_pert
            best_c5 = c5_pert
            # Follow up with subgradient
            h2, c52 = subgrad_descent(h, n, 50, min(deadline, time.time() + 5), top_k=3)
            if c52 < best_c5:
                h = h2
                best_c5 = c52
                
    return h, best_c5

def coordinate_refine(h0, n, deadline, n_rounds=5):
    """Try adjusting pairs of coordinates to reduce max overlap."""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try a random subset of indices
        idxs = np.random.choice(n, size=min(200, n), replace=False)
        for i in idxs:
            if time.time() > deadline:
                break
            for delta in [0.01, -0.01, 0.005, -0.005]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                # Need to compensate: adjust another index
                j = (i + 1) % n
                if h[j] - delta < 0 or h[j] - delta > 1:
                    continue
                h[i] += delta
                h[j] -= delta
                c5_new = evaluate_c5(h)
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
        if not improved and rnd > 1:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(42, 0.5)
    
    n_start = len(h_start)
    best_h = project_feasible(h_start, n_start)
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: SLSQP with softmax at multiple T values
    T_list = [100.0, 300.0, 800.0, 2000.0]
    for T in T_list:
        if time.time() > deadline - 60:
            break
        h_new = slsqp_opt(best_h, n_start, T, maxiter=200, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 2: Subgradient descent on true max (main phase)
    time_left = deadline - time.time()
    n_iter_subgrad = min(2000, max(200, int(time_left * 0.5)))
    best_h, best_c5 = subgrad_descent(best_h, n_start, n_iter_subgrad, deadline, top_k=5, window_frac=0.05)
    
    # Phase 3: Local search with random perturbations
    time_left = deadline - time.time()
    if time_left > 30:
        n_trials = min(200, max(20, int(time_left * 0.4)))
        best_h, best_c5 = local_search(best_h, n_start, n_trials, deadline)
    
    # Phase 4: Try different resolutions
    for n_try in [n_start * 2, n_start // 2]:
        if time.time() > deadline - 30:
            break
        if n_try < 50 or n_try > 2000:
            continue
        h_interp = interp_to(best_h, n_start, n_try)
        h_interp = project_feasible(h_interp, n_try)
        # Quick SLSQP
        h_try = slsqp_opt(h_interp, n_try, 500.0, maxiter=100, deadline=deadline)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            # Interpolate back to original resolution for consistency
            best_h = project_feasible(h_try, n_try)
            best_c5 = c5_try
            n_start = n_try
    
    # Phase 5: Final subgradient refinement
    time_left = deadline - time.time()
    if time_left > 20:
        n_final = len(best_h)
        n_iter_final = min(1000, max(100, int(time_left * 0.6)))
        best_h, best_c5 = subgrad_descent(best_h, n_final, n_iter_final, deadline, top_k=3, window_frac=0.03)
    
    # Phase 6: Coordinate refinement
    time_left = deadline - time.time()
    if time_left > 15:
        n_final = len(best_h)
        n_rounds = min(10, max(3, int(time_left / 5)))
        best_h, best_c5 = coordinate_refine(best_h, n_final, deadline, n_rounds=n_rounds)
    
    best_c5 = evaluate_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Final projection
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
