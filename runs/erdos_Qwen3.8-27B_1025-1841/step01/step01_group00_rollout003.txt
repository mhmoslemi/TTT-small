import numpy as np
import time

def project_to_feasible(g, n):
    """Project g to [0,1]^n with sum = n/2."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad(h, tau, top_k=None):
    """Compute smooth approximation of max(corr) and its gradient.
    If top_k is set, use weighted top-k objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    
    if top_k is not None and top_k < n_corr:
        # Weighted top-k: weight proportional to rank
        idx = np.argsort(corr)[::-1][:top_k]
        weights = np.zeros(n_corr)
        weights[idx] = np.arange(top_k, 0, -1, dtype=float)
        weights /= weights.sum()
        obj = np.dot(weights, corr) * dx
        # Gradient: sum_k w_k * grad_corr_k
        grad = np.zeros(n)
        for k in idx:
            wk = weights[k]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += wk * (1.0 - h[np.arange(j0, j1) - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= wk * h[np.arange(j0b, j1b) + sh]
        grad *= dx
        return obj, grad
    else:
        # Log-sum-exp
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        obj = dx * (max_c + np.log(s) / tau)
        grad = np.zeros(n)
        for k in range(n_corr):
            pk = p[k]
            if pk < 1e-15:
                continue
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
        grad *= dx
        return obj, grad

def optimize_adaptive(h, n, start_tau, end_tau, max_iters, time_limit, lr=None):
    """Optimize with adaptive tau (annealing) and Nesterov momentum."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, start_tau)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.3
    
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    n_iter = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        n_iter += 1
        # Anneal tau
        frac = min(1.0, it / max(1, max_iters * 0.8))
        tau = start_tau + (end_tau - start_tau) * frac
        
        # Determine if to use top-k
        if frac > 0.7:
            top_k = max(1, int(10 * (frac - 0.7) / 0.3) + 1)
        else:
            top_k = None
        
        obj, grad = smooth_obj_grad(h, tau, top_k=top_k)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Nesterov step
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        # Backtracking if objective increased significantly
        obj_new, _ = smooth_obj_grad(h_new, tau, top_k=top_k)
        if obj_new > obj + 0.01:
            v *= 0.5
            h_new = project_to_feasible(h + v, n)
        
        if stall > 100:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-9:
                break
        
        h = h_new
    
    return best_h, best_obj

        if c5_cur < best_c5 - 1e-14:

            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[::-1][:5]

            j0 = max(0, sh)
            j1 = min(n, sh + n)

            j0b = max(0, -sh)
            j1b = min(n, n - sh)

            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-12:

    return best_h, best_c5

def perturb_escape(h, n, seed, strength=0.02):
    """Add structured perturbation to escape local minima."""
    rng = np.random.RandomState(seed)
    # Add smooth random perturbation
    freq = rng.randint(1, 5)

    pert = strength * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
    # Also add small random noise
    pert += strength * 0.5 * (rng.randn(n) - 0.5)
    return project_to_feasible(h + pert, n)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):

        s = np.sum(np.clip(g - mid, 0.0, 1.0))

def interp_periodic(h, target_n):
