from scipy.optimize import minimize

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def smooth_obj(h, T):
    corr = compute_corr(h)
    Tcorr = T * corr
    Tcorr -= np.max(Tcorr)
    s = np.exp(Tcorr).sum()
    return np.max(corr) + np.log(s) / T

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    Tcorr = T * corr
    Tcorr -= np.max(Tcorr)
    exp_Tcorr = np.exp(Tcorr)
    w = exp_Tcorr / exp_Tcorr.sum()
    
    w_cumsum = np.concatenate(([0.0], np.cumsum(w)))
    sliding_sum = w_cumsum[n:2*n] - w_cumsum[0:n]
    
    conv_hw = np.convolve(h, w, mode='full')
    w_tail = w[n-1:]
    corr2 = np.correlate(h, w_tail, mode='full')
    
    grad = dx * (sliding_sum - conv_hw[n-1:n-1+n] - corr2[n-1:n-1+n])
    return grad

def obj_wrap(h, T):
    return smooth_obj(h, T)

def grad_wrap(h, T):
    return smooth_grad(h, T)

def slsqp_optimize(h, n, T, maxiter=300, ftol=1e-12):
    def obj(hv):
        return smooth_obj(hv, T)
    def grad(hv):
        return smooth_grad(hv, T)
    def constr(hv):
        return np.sum(hv) - n / 2.0
    def constr_jac(hv):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constr, 'jac': constr_jac}]
    
    res = minimize(obj, h, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def proj_grad_optimize(h, n, T, n_iter=2000, lr0=0.01, momentum=0.9):
    h = h.copy()
    v = np.zeros(n)
    lr = lr0
    prev_obj = smooth_obj(h, T)
    
    for it in range(n_iter):
        g = smooth_grad(h, T)
        v = momentum * v + g
        step = lr / (1 + 1e-8 + np.max(np.abs(v))) * 0.5
        h_new = h - step * v
        h_new = project_feasible(h_new, n)
        
        new_obj = smooth_obj(h_new, T)
        if not np.isfinite(new_obj):
            h_new = h.copy()
            new_obj = prev_obj
            lr *= 0.5
            v = 0.5 * v
        elif new_obj > prev_obj + 1e-10:
            lr *= 0.7
            v = 0.5 * v
        else:
            lr = min(lr * 1.05, 0.5)
        
        h = h_new
        prev_obj = new_obj
    
    return h
