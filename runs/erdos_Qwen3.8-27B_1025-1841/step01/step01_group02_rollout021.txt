def compute_gradient(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    shifted = corr - corr.max()
    exp_shifted = np.exp(T * shifted)
    rho = exp_shifted / exp_shifted.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        s = k - (n - 1)  # shift
        if abs(rho[k]) < 1e-15:
            continue
        if s == 0:
            grad += rho[k] * (1.0 - 2.0 * h)
        elif s > 0:
            # ∂c_k/∂h[j] = (1-h[j+s]) for j=0..n-1-s, minus h[j-s] for j=s..n-1
            grad[:n-s] += rho[k] * (1.0 - h[s:])
            grad[s:] -= rho[k] * h[:n-s]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}"""
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def compute_gradient(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    exp_shifted = np.exp(T * (corr - max_c))
    Z = exp_shifted.sum()
    rho = exp_shifted / Z
    
    grad = np.zeros(n)
    # s = 0 (k = n-1)
    grad += rho[n-1] * (1.0 - 2.0 * h)
    
    for s in range(1, n):
        # s > 0: k = n-1+s
        if rho[n-1+s] > 1e-10:
            grad[s:] += rho[n-1+s] * (1.0 - h[:n-s])
            grad[:n-s] -= rho[n-1+s] * h[s:]
        # s < 0: k = n-1-s
        if rho[n-1-s] > 1e-10:
            grad[s:] += rho[n-1-s] * (1.0 - h[:n-s])
            grad[:n-s] -= rho[n-1-s] * h[s:]
    
    return grad

def pgd_optimize(h, T, max_iter, time_fn):
    """Projected gradient descent with log-sum-exp smoothing"""
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    alpha = 4.0 / (T * n)  # initial step size
    for it in range(max_iter):
        if time_fn() :  # time up
            break
        
        h = project_feasible(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        grad = compute_gradient(h, T)
        
        # Backtracking line search
        for _ in range(20):
            h_new = project_feasible(h - alpha * grad, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 + 1e-12:
                h = h_new
                alpha = min(alpha * 1.05, 8.0 / (T * n))
                break
            alpha *= 0.5
        else:
            break
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def gradient_lse(h, T):
    """Compute gradient of log-sum-exp smoothed objective"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    exp_shifted = np.exp(T * (corr - max_c))
    Z = exp_shifted.sum()
    rho = exp_shifted / Z
    
    grad = np.zeros(n)
    grad += rho[n-1] * (1.0 - 2.0 * h)
    for s in range(1, n):
        r = rho[n-1+s]
        if r > 1e-14:
            grad[s:] += r * (1.0 - h[:n-s])
            grad[:n-s] -= r * h[s:]
        r = rho[n-1-s]
        if r > 1e-14:
            grad[s:] += r * (1.0 - h[:n-s])
            grad[:n-s] -= r * h[s:]
    return grad * dx

def pgd(h, T, max_iter, deadline):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 6.0 / (T * n)
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        grad = gradient_lse(h, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        improved = False
        for _ in range(20):
            h_new = project_feasible(h - alpha * grad, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-13:
                h = h_new
                best_c5 = c5
                best_h = h.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            alpha = min(alpha * 1.2, 12.0 / (T * n))
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5

def to_binary(h, n, rng=None):

def binary_2opt(h, n_sweeps=50, deadline=None, rng=None):
    """2-opt local search on binary vectors"""
    if deadline is None:
        deadline = np.inf
    if rng is None:
        rng = np.random.RandomState(0)
    
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = list(zeros)
        rng.shuffle(z_list)
        for i in z_list:
            if time.time() > deadline:
                break
            o_list = list(ones)
            rng.shuffle(o_list)
            for j in o_list:
                # Swap: i becomes 1, j becomes 0

                h[j] = 0.0

                if new_c5 < c5 - 1e-13:

                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0
            if time.time() > deadline:
                break
        if not improved:
