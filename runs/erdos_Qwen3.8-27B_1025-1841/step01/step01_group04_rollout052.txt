def compute_gradient(h, T, n):
    """Compute gradient of smoothed log-sum-exp objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    p = exp_vals / np.exp_vals.sum()  # softmax weights, length 2n-1
    
    # Gradient: sum_k p[k] * dcross[k]/dh[j] * dx
    # dcross[k]/dh[j] = (1-h)[j+k-(n-1)] - h[j-(k-(n-1))]
    # Let shift = k - (n-1), ranges from -(n-1) to (n-1)
    # = (1-h)[j+shift] - h[j-shift] (with out of bounds = 0)
    
    grad = np.zeros(n)

        if p[k] < 1e-15:
            continue
        shift = k - (n-1)
        weight = p[k] * dx
        # (1-h)[j+shift] for j+shift in [0, n-1]
        if shift >= 0:
            # j+shift in [0,n-1] means j in [0, n-1-shift]
            grad[:n-shift] += weight * (1.0 - h[shift:])
        else:

            grad[-shift:] += weight * (1.0 - h[:n+shift])

        if shift >= 0:
            # j-shift in [0,n-1] means j in [shift, n-1]
            grad[shift:] -= weight * h[:n-shift]
        else:
            # j-shift in [0,n-1] means j in [0, n-1+shift]
            grad[:n+shift] -= weight * h[-shift:]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_softmax_weights(corr, T, dx):
    scaled = T * corr * dx
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    return exps / np.sum(exps)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_test)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_gradient(h, T, dx, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    p = compute_softmax_weights(corr, T, dx)
    
    # grad[j] = dx * sum_k p[k] * ((1-h)[j+k-(n-1)] - h[j-(k-(n-1))])
    # Let shift = k - (n-1), ranges -(n-1) to (n-1)
    # grad[j] = dx * sum_{shift} p[shift+(n-1)] * ((1-h)[j+shift] - h[j-shift])
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        # (1-h)[j+shift]: valid when 0 <= j+shift <= n-1
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
        else:
            grad[-shift:] += w * (1.0 - h[:n + shift])
        # -h[j-shift]: valid when 0 <= j-shift <= n-1
        if shift >= 0:
            grad[shift:] -= w * h[:n - shift]
        else:
            grad[:n + shift] -= w * h[-shift:]
    return grad

def proj_grad_step(h, T, dx, n, max_iter=2000, deadline=None):
    h = h.copy()
    lr = 0.5
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = compute_gradient(h, T, dx, n)
        # Backtracking line search
        f0 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        step = lr
        improved = False
        for _ in range(30):
            h_try = project_feasible(h - step * g, n)
            f1 = np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * dx
            if f1 < f0 - 1e-12:
                h = h_try
                improved = True
                break
            step *= 0.5
        if not improved:
            break
        # Occasionally increase step size
        if improved:
            lr = min(step * 1.5, 2.0)
    
    return h

def coord_descent(h, n, max_sweeps=50, deadline=None):
    """Coordinate descent for the max objective."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    target = n / 2.0
    best_obj = evaluate_c5(h)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        changed = False
        for j in np.random.permutation(n):
            # Try to improve h[j]
            # Fix all other coordinates, vary h[j]
            # The sum constraint means we need to keep sum = target
            # Strategy: vary h[j] and h[j2] (paired) to maintain sum
            # Simpler: just try a few values of h[j] and project
            
            old_val = h[j]
            # Try values: 0, 1, and current ± small steps
            candidates = [0.0, 1.0, old_val, max(0, old_val - 0.1), min(1, old_val + 0.1)]
            best_c = h[j].copy()  # keep old
            best_f = None
            
            for v in candidates:
                if abs(v - old_val) < 1e-6:
                    continue
                h[j] = v
                h = project_feasible(h, n)
                f = evaluate_c5(h)
                if best_f is None or f < best_f:
                    best_f = f
                    best_c = h.copy()
            h = best_c
            if best_f < best_obj - 1e-10:
                best_obj = best_f
                changed = True
        
        if not changed:
            break
    
    return h

def evaluate_c5(h):

def project_feasible(h, n):
    """Euclidean projection onto {x: 0≤x≤1, sum(x)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    # Fix rounding

    if abs(diff) > 1e-10:

        if cnt > 0:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def softobj_and_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    p = exps / exps.sum()
    obj = (m + np.log(exps.sum())) / T * dx
    grad = np.zeros(n)
    thresh = 1e-12
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < thresh:
            continue
        s = n - 1 - k
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            ss = -s
            grad[ss:] += w * (1.0 - h[:n - ss])
            grad[:n - ss] -= w * h[ss:]
    return obj, grad

def pgd_optimize(h, n, T, max_iter, deadline):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    lr = 0.8
    best_local = evaluate_c5(h)
    prev_obj = None
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, grad = softobj_and_grad(h, n, T)
        if prev_obj is not None and abs(obj - prev_obj) < 1e-14:
            stall += 1
        else:
            stall = 0
        if stall > 200:
            break
        prev_obj = obj
        step = lr
        accepted = False
        for _ in range(40):
            h_try = project_feasible(h - step * grad, n)
            if not np.all(h_try == h):
                obj_try, _ = softobj_and_grad(h_try, n, T)
                if obj_try < obj - 1e-15:
                    h = h_try
                    accepted = True
                    break
            step *= 0.5
        if not accepted:
            break
        lr = min(step * 1.3, 3.0)
        c5_now = evaluate_c5(h)
        if c5_now < best_local:
            best_local = c5_now
    return h, best_local

def coord_descent_polish(h, n, max_sweeps=200, deadline=None):
    h = project_feasible(h.copy(), n)
    best_obj = evaluate_c5(h)
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if deadline is not None and time.time() > deadline:
                break
            for v in (0.0, 1.0):
                if abs(v - h[j]) < 1e-4:
                    continue
                h_save = h.copy()
                h[j] = v
                h = project_feasible(h, n)
                f = evaluate_c5(h)
                if f < best_obj - 1e-12:
                    best_obj = f
                    improved = True
                else:
                    h = h_save
        if not improved:
            break
    return h, best_obj

def binary_round(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def swap_polish(h, n, max_rounds=50, deadline=None):
    h = project_feasible(h.copy(), n)
    best_obj = evaluate_c5(h)
    rng = np.random.RandomState(77)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_try = min(len(ones) * len(zeros), 2000)
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_save = h.copy()
            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, n)
            f = evaluate_c5(h)
            if f < best_obj - 1e-12:
                best_obj = f
                improved = True
            else:
                h = h_save
        if not improved:
            break
    return h, best_obj

def interp_to(h_from, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h_from)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(336, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Multi-T projected gradient
    T_phases = [(80, 3000), (40, 2000), (15, 2000), (8, 1500), (4, 1500), (2, 1000)]
    for T, max_it in T_phases:
        if time.time() > deadline - 120:
            break
        h_work, local_best = pgd_optimize(best_h, n, T, max_it, time.time() + (deadline - time.time()) * 0.7)
        c5 = evaluate_c5(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()

    # Phase 2: Coordinate descent polish
    if time.time() < deadline - 60:
        h_cd, obj_cd = coord_descent_polish(best_h, n, max_sweeps=300, deadline=deadline - 30)
        if obj_cd < best_c5:
            best_c5 = obj_cd
            best_h = h_cd.copy()

    # Phase 3: Binary rounding + swap
    h_bin = binary_round(best_h, n)
    c5_bin = evaluate_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    if time.time() < deadline - 20:
        h_sw, obj_sw = swap_polish(best_h, n, max_rounds=100, deadline=deadline - 10)
        if obj_sw < best_c5:
            best_c5 = obj_sw
            best_h = h_sw.copy()

    # Phase 4: Random perturbations + refine
    for attempt in range(40):
        if time.time() > deadline - 20:
            break
        rng = np.random.RandomState(seed + 500 + attempt)
        h_pert = best_h + rng.randn(n) * 0.05
        h_pert = project_feasible(h_pert, n)
        hp, lp = pgd_optimize(h_pert, n, 30, 800, min(deadline - 10, time.time() + 8))
        c5p = evaluate_c5(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 5: Try different resolution
    for n_alt in [504, 672]:
        if time.time() > deadline - 40:
            break
        h_alt = interp_to(best_h, n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)
        h_alt, _ = pgd_optimize(h_alt, n_alt, 40, 1000, min(deadline - 10, time.time() + 60))
        c5_alt = evaluate_c5(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            n = n_alt

    # Final coordinate descent at best resolution
    if time.time() < deadline - 5:
        h_fin, obj_fin = coord_descent_polish(best_h, len(best_h), max_sweeps=100, deadline=deadline)
        if obj_fin < best_c5:
            best_c5 = obj_fin
            best_h = h_fin.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
