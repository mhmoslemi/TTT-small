    return h

    """Upsample parent to length n and convert to binary with n/2 ones"""
    np_ = len(parent)
    h = np.zeros(n, dtype=np.float64)
    for i in range(n):
        pos = i * np_ / n
        idx = int(pos)
        if idx >= np_:
            idx = np_ - 1
        frac = pos - idx
        if idx < np_ - 1:
            h[i] = (1 - frac) * parent[idx] + frac * parent[idx + 1]

import numpy as np
import time

def _eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_binary(n, rng):
    h = np.zeros(n, dtype=np.float64)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h

def _parent_to_binary(parent, n, rng):
    np_ = len(parent)
    h = np.zeros(n, dtype=np.float64)
    for i in range(n):
        pos = i * float(np_) / n
        idx = min(int(pos), np_ - 1)
        frac = pos - idx
        if idx < np_ - 1:
            h[i] = (1.0 - frac) * parent[idx] + frac * parent[idx + 1]
        else:
            h[i] = parent[idx]
    h = (h >= 0.5).astype(np.float64)
    cnt = int(h.sum())
    target = n // 2
    if cnt > target:
        pos_ones = np.where(h == 1)[0]
        order = np.argsort(h[pos_ones])
        for k in range(cnt - target):
            h[pos_ones[order[k]]] = 0.0
    elif cnt < target:
        pos_zeros = np.where(h == 0)[0]
        order = np.argsort(-h[pos_zeros])
        for k in range(target - cnt):
            h[pos_zeros[order[k]]] = 1.0
    return h

def _upsample_best(h_prev, n_prev, n_new, rng):
    h = np.zeros(n_new, dtype=np.float64)
    for i in range(n_new):
        pos = i * float(n_prev) / n_new
        idx = min(int(pos), n_prev - 1)
        frac = pos - idx
        if idx < n_prev - 1:
            h[i] = (1.0 - frac) * h_prev[idx] + frac * h_prev[idx + 1]
        else:
            h[i] = h_prev[idx]
    h = (h >= 0.5).astype(np.float64)
    cnt = int(h.sum())
    target = n_new // 2
    if cnt > target:
        pos_ones = np.where(h == 1)[0]
        order = np.argsort(h[pos_ones])
        for k in range(cnt - target):
            h[pos_ones[order[k]]] = 0.0

        order = np.argsort(-h[pos_zeros])
        for k in range(target - cnt):

    return h

    h = h0.copy()
    cur = _eval_c5(h, n)
    best_h = h.copy()
    best = cur
    T = 0.03
    Tm = 0.0003
    cool = 0.99985
    it = 0
    stagnation = 0

    while time.time() < deadline:

        if it > 3000 and stagnation > 5000 and it % 3000 == 0:

                    break

                q = zeros[rng.integers(len(zeros))]
                h[p] = 0.0
                h[q] = 1.0
            cur = _eval_c5(h, n)
            T = max(T, 0.025)

        q = zeros[rng.integers(len(zeros))]
