```python
def compute_grad(h, w, dx):
    n = len(h)
    # k_p = p - (n-1) for p = 0, ..., 2n-2
    k = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # j goes from 0 to n-1
    j_idx = np.arange(n)  # shape (n,)
    
    # j - k_p: shape (n, 2n-1)
    jmk = j_idx[:, None] - k[None, :]  # shape (n, 2n-1)
    # 0 <= j-k_p < n
    valid1 = (jmk >= 0) & (jmk < n)  # shape (n, 2n-1)
    
    # j + k_p: shape (n, 2n-1)
    jpk = j_idx[:, None] + k[None, :]
    valid2 = (jpk >= 0) & (jpk < n)
    
    # (1-h[j-k_p]) for valid entries
    # h[jmk] where valid1
    term1 = np.where(valid1, 1.0 - h[np.clip(jmk, 0, n-1)], 0.0)
    term2 = np.where(valid2, h[np.clip(jpk, 0, n-1)], 0.0)
    
    # grad[j] = dx^2 * sum_p w[p] * (term1[j,p] - term2[j,p])
    grad = dx * dx * (term1 - term2) @ w
    return grad
```

```python
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj_grad(h, T):
    """Compute soft-max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp  # softmax weights
    obj = (m + np.log(sum_exp)) / T * dx
    
    # Gradient via vectorized computation
    k = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    j_idx = np.arange(n)  # shape (n,)
    
    jmk = j_idx[:, None] - k[None, :]  # (n, 2n-1)
    valid1 = (jmk >= 0) & (jmk < n)
    jpk = j_idx[:, None] + k[None, :]
    valid2 = (jpk >= 0) & (jpk < n)
    
    # term1[j,p] = (1-h[j-k_p]) if valid else 0
    idx1 = np.clip(jmk, 0, n-1)
    term1 = np.where(valid1, 1.0 - h[idx1], 0.0)
    
    # term2[j,p] = h[j+k_p] if valid else 0
    idx2 = np.clip(jpk, 0, n-1)
    term2 = np.where(valid2, h[idx2], 0.0)
    
    grad = (dx * dx) * ((term1 - term2) @ w)
    return obj, grad

def slsqp_analytic(h0, n, T, maxiter, deadline):
    """SLSQP optimization with analytic gradient."""
    target = n / 2.0
    
    def obj_func(h):
        val, _ = soft_max_obj_grad(h, T)
        return val
    
    def grad_func(h):
        _, g = soft_max_obj_grad(h, T)
        return g
    
    def eq_constraint(h):
        return np.sum(h) - target
    
    def eq_jac(h):
        return np.ones(n)
    
    try:
        res = minimize(obj_func, h0, jac=grad_func, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() < deadline:
            if res.x is not None and np.all(np.isfinite(res.x)):
                return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def lbfgs_penalty(h0, n, T, deadline, penalty=100.0, maxiter=500):
    """L-BFGS-B with penalty for constraint."""
    target = n / 2.0
    
    def obj_func(h):
        val, g = soft_max_obj_grad(h, T)
        penalty_term = penalty * (np.sum(h) - target) ** 2
        penalty_grad = 2 * penalty * (np.sum(h) - target) * np.ones(n)
        return val + penalty_term, g + penalty_grad
    
    def obj_func_only(h):
        val, _ = soft_max_obj_grad(h, T)
        return val + penalty * (np.sum(h) - target) ** 2
    
    def grad_func(h):
        _, g = soft_max_obj_grad(h, T)
        return g + 2 * penalty * (np.sum(h) - target) * np.ones(n)
    
    try:
        res = minimize(obj_func_only, h0, jac=grad_func, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0
```
