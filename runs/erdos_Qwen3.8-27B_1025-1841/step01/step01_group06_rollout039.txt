def project_target(h, n):

        if len(idx) > 0:

            h = np.clip(h, 0.0, 1.0)

def upsample(h, n_new):

    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)

    return project_target(np.clip(h_new, 0.0, 1.0), n_new)

        else:
            sp = -s
            grad[sp:] += w[k] * (1.0 - h[:n-sp])
