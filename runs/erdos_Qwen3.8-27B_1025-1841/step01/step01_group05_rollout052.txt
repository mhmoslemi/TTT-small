import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def interp_to_n(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new points."""
    if n_old == n_new:
        return h_old.astype(float).copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    s = np.sum(h_new)
    if s > 1e-10:
        h_new *= target / s
    h_new = np.clip(h_new, 0.0, 1.0)
    # Fix sum after clipping
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-10:
        h_new += diff / n_new
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def to_binary_feasible(h, n, rng):
    """Make h binary with exactly n//2 ones."""
    h = (h > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        for k in range(ones - target):
            h[idx[k]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        for k in range(target - ones):
            h[idx[k]] = 1
    return h

def local_search(h, n, max_iter, rng, anneal=True):
    """Binary local search with optional simulated annealing."""
    dx = 2.0 / n
    h = to_binary_feasible(h.copy(), n, rng)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    temp = 0.001 if anneal else 0.0
    cool = 0.99995
    
    # Maintain lists
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for it in range(max_iter):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        accept = new_c5 < cur_c5
        if not accept and temp > 1e-10:
            delta = cur_c5 - new_c5
            if delta > -1e-12 and rng.random() < np.exp(delta / temp):
                accept = True
        
        if accept:
            cur_c5 = new_c5
            ones.remove(i)  # Wait, i is now 0
            ones.append(j)  # j is now 1
            zeros.remove(j)
            zeros.append(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        temp *= cool
    
    return best_h, best_c5

def slsqp_optimize(h0, n, maxiter=200, ftol=1e-8):
    """SLSQP optimization for general h."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(h):
        return compute_c5(h)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target}]
    bounds = [(0, 1)] * n
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
    # Clip and fix
    h = np.clip(res.x, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0, 1)
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def interp_to_n(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.astype(float).copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    s = np.sum(h_new)
    if s > 1e-10:
        h_new *= target / s
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-10:
        h_new += diff / n_new
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def to_binary(h, n, rng):
    h = (h > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:ones - target]] = 0.0
    elif ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - ones]] = 1.0
    return h

def local_search_sa(h, n, max_iter, rng, T0=0.003, cool=0.99995):
    h = to_binary(h.copy(), n, rng)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur = best_c5
    T = T0
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_iter):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h)
        if new_c < cur or (T > 1e-12 and rng.random() < np.exp((cur - new_c) / T)):
            cur = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c5:
                best_c5 = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= cool
    return best_h, best_c5

def slsqp_opt(h0, n, maxiter=150):
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h = np.clip(res.x, 0.0, 1.0)
        d = target - np.sum(h)
        if abs(d) > 1e-10:
            h += d / n
            h = np.clip(h, 0.0, 1.0)
        return h, compute_c5(h)
    except Exception:
        return h0, compute_c5(h0)
