    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)

def nesterov_subgradient(h, n, dx, max_iter, t_deadline, rng):
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    momentum = np.zeros(n)
    beta = 0.9
    alpha = 0.008
    no_improve = 0
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        if rng.random() < 0.05:
            k_star = rng.randint(0, 2 * n - 1)
        grad = subgradient_at(h, n, k_star)
        momentum = beta * momentum - alpha * grad
        h = project_onto_set(h + momentum, n)
        if it % 100 == 99:
            val = compute_c5(h, dx)
            if val < best_val - 1e-12:
                best_h = h.copy()
                best_val = val
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    alpha *= 0.85
                    momentum *= 0.5
                    no_improve = 0
        if it % 20000 == 19999:
            alpha *= 0.9
    final_val = compute_c5(h, dx)
    if final_val < best_val:
        best_h = h.copy()
        best_val = final_val
    return best_h, best_val

def try_binary_threshold(h, n):
    dx = 2.0 / n
    for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
        hb = np.where(h > thresh, 1.0, 0.0)
        if abs(np.sum(hb) - n / 2.0) < 1.0:
            s = np.sum(hb)
            if s > n / 2.0:
                idx = np.where(hb == 1.0)[0]
                order = h[idx]
                flip = int(s - n / 2.0)
                if flip > 0 and flip <= len(order):
                    hb[idx[np.argsort(order)[:flip]]] = 0.0
            else:
                idx = np.where(hb == 0.0)[0]
                order = h[idx]
                flip = int(n / 2.0 - s)
                if flip > 0 and flip <= len(order):
                    hb[idx[np.argsort(order)[-flip:]]] = 1.0
            if abs(np.sum(hb) - n / 2.0) < 1e-10:
                c5 = compute_c5(hb, dx)
                return hb, c5
    return None, None

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.93
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_c5 = compute_c5(h0, dx0)
    best_h = h0.copy()
    best_n = n0

    h_curr = h0.copy()
    for T in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if time.time() > deadline:
            break
        h_curr = l_bfgs_smooth_optimize(h_curr, n0, dx0, T, lam=1000,
                                         max_iter=400, t_deadline=deadline)
        h_curr = project_onto_set(h_curr, n0)
        c5 = compute_c5(h_curr, dx0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()

    for mult in [2, 4]:
        n = n0 * mult
        if n > 560 or time.time() > deadline:
            break
        dx = 2.0 / n
        h_curr = upsample(best_h, n)
        for T in [0.05, 0.02, 0.01]:
            if time.time() > deadline:
                break
            h_curr = l_bfgs_smooth_optimize(h_curr, n, dx, T, lam=1000,
                                             max_iter=250, t_deadline=deadline)
            h_curr = project_onto_set(h_curr, n)
        h_curr, c5_sg = nesterov_subgradient(h_curr, n, dx,
                                              max_iter=200000, t_deadline=deadline, rng=rng)
        h_curr = project_onto_set(h_curr, n)
        c5 = compute_c5(h_curr, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()
            best_n = n

    if best_n > n0 and time.time() < t0 + budget_s * 0.88:
        h_polish = upsample(best_h, n0)
        for T in [0.005, 0.002]:
            if time.time() > deadline:
                break
            h_polish = l_bfgs_smooth_optimize(h_polish, n0, dx0, T, lam=5000,
                                               max_iter=300, t_deadline=deadline)
            h_polish = project_onto_set(h_polish, n0)
        c5 = compute_c5(h_polish, dx0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_polish
            best_n = n0

    # Try binary threshold
    hb, c5b = try_binary_threshold(best_h, best_n)
    if hb is not None and c5b < best_c5:
        best_h = hb
        best_c5 = c5b

    # Try SLSQP polish at n0 if time remains
    if time.time() < t0 + budget_s * 0.85:
        n_s = n0
        dx_s = 2.0 / n_s
        target_s = n_s / 2.0
        h_s = upsample(best_h, n_s) if best_n != n_s else best_h.copy()
        T_s = 0.005
        def slsqp_obj(x):
            hx = np.clip(x, 0.0, 1.0)
            corr = np.correlate(hx, 1.0 - hx, mode="full")
            m = np.max(corr)
            cs = (corr - m) / T_s
            return (m + T_s * np.log(np.sum(np.exp(cs)))) * dx_s
        def slsqp_jac(x):
            hx = np.clip(x, 0.0, 1.0)
            corr = np.correlate(hx, 1.0 - hx, mode="full")
            m = np.max(corr)
            cs = (corr - m) / T_s
            w = np.exp(cs)
            wn = w / np.sum(w)
            W = np.zeros(n_s)
            W[0] = wn[n_s - 1]
            if n_s > 1:
                W[1:] = wn[n_s:] + wn[n_s - 2::-1]
            g = W[0] * (1.0 - 2.0 * hx)
            for s in range(1, n_s):
                if W[s] < 1e-15:
                    continue
                g[s:] += W[s] * (1.0 - hx[:n_s - s])
                g[:n_s - s] -= W[s] * hx[s:]
            return g * dx_s
        try:
            res = minimize(slsqp_obj, h_s, jac=slsqp_jac, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_s,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_s,
                                        'jac': lambda x: np.ones(n_s)}],
                          options={'maxiter': 500, 'ftol': 1e-12})
            h_test = project_onto_set(res.x, n_s)
            c5_t = compute_c5(h_test, dx_s)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_test
                best_n = n_s
        except Exception:
            pass

    best_h = project_onto_set(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
