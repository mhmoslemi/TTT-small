  Contribution: -p[m]*h[j+m-n+1], let v = j+m-n+1, so m = v-n+1-j, v ∈ [0, n-1]

  So second term = -sum_{v=n-1-j}^{n-1} p[v-n+1-j]*h[v]

    T = 0.005
    T_min = 1e-7
    cooling = 0.9995
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            # Update lists
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
            # lists unchanged
        
        if it % 100 == 0:
            T = max(T_min, T * cooling)
    
    return best_h, best_c5

def slsqp_polish(h, T, maxiter, time_limit, start_time):
    """Polish with SLSQP using exact gradient."""
    n = len(h)
    
    def obj(h):
        c, _ = softmax_obj_and_grad(h, T)
        return c
    
    def grad(h):
        _, g = softmax_obj_and_grad(h, T)
        return g
    
    def cons_fun(h):
        return np.sum(h) - n / 2.0
    
    def cons_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or compute_c5(res.x) < compute_c5(h):
            return res.x, compute_c5(res.x)
    except Exception:
        pass
    return h, compute_c5(h)

def coordinate_polish(h, time_limit, start_time):
    """Coordinate descent polish."""
    n = len(h)
    best_c5 = compute_c5(h)
    for sweep in range(20):
        if time.time() - start_time > time_limit - 5:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 5:
                break
            old = h[j]
            # Try a few values
            for candidate in [0.0, 1.0, 0.5, old*0.5, old*1.5, old+0.05, old-0.05]:
                candidate = max(0.0, min(1.0, candidate))
                trial = h.copy()
                trial[j] = candidate
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = trial
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5
