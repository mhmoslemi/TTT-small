def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-12:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
    return np.clip(h, 0.0, 1.0)

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old) + 1.0 / n_old
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_sum(h_new, n_new / 2.0)

def pair_exchange_iterations(h, corr, n, rng, n_iters):
    dx = 2.0 / n

    best_val = float(np.max(corr)) * dx
    improved = 0
    for it in range(n_iters):
        j = rng.randint(n)
        l = rng.randint(n)
        if l == j:
            continue
        A = np.zeros(n_c)
        m1 = (shifts >= -j) & (shifts < n - j)
        A[m1] += (1.0 - h[j + shifts[m1]])
        m2 = (shifts > j - n) & (shifts <= j)
        A[m2] -= h[j - shifts[m2]]
        m3 = (shifts >= -l) & (shifts < n - l)
        A[m3] -= (1.0 - h[l + shifts[m3]])
        m4 = (shifts > l - n) & (shifts <= l)
        A[m4] += h[l - shifts[m4]]
        lo = max(-h[j], h[l] - 1.0)
        hi = min(1.0 - h[j], h[l])
        if hi - lo < 1e-14:
            continue
