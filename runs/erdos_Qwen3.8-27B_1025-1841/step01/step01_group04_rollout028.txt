    h = np.array(h, dtype=float).copy()

        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)

        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)

def soft_max_value_and_grad(h, n, T):

    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    scaled = corr * T
    mx = np.max(scaled)
    lse = mx + np.log(np.sum(np.exp(scaled - mx)))
    p = np.exp(scaled - lse)

            mp = -m
            if mp < n:
                grad[:n - mp] += pk * omh[mp:]
                grad[mp:] -= pk * h[:n - mp]

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def soft_max_val_grad(h, n, T):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    scaled = corr * T
    mx = np.max(scaled)
    lse = mx + np.log(np.sum(np.exp(scaled - mx)))
    p = np.exp(scaled - lse)
    val = lse / T * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        if pk < 1e-12:

            if m < n:

                grad[:n - m] -= pk * h[m:]
        else:
            mp = -m
            if mp < n:
                grad[:n - mp] += pk * omh[mp:]
                grad[mp:] -= pk * h[:n - mp]
