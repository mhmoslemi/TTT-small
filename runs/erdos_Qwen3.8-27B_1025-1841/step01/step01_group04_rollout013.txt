    h = np.array(h, dtype=float).copy()

import numpy as np
from scipy.optimize import minimize
import time

# Globals for SLSQP callbacks
_g_T = 100.0
_g_n = 336

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def soft_max_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = T * corr
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    obj = (m + np.log(sum_exp)) / T * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        a = k - (n - 1)
        i_s1 = max(0, -a)
        i_e1 = min(n, n - a)
        if i_e1 > i_s1:
            grad[i_s1:i_e1] += p[k] * (1.0 - h[i_s1 + a:i_e1 + a])
        i_s2 = max(0, a)
        i_e2 = min(n, n + a)
        if i_e2 > i_s2:
            grad[i_s2:i_e2] -= p[k] * h[i_s2 - a:i_e2 - a]
    grad *= dx
    return obj, grad

def _obj_f(x):
    o, _ = soft_max_obj_and_grad(x, _g_T)
    return o

def _grad_f(x):
    _, g = soft_max_obj_and_grad(x, _g_T)
    return g

def _con_f(x):
    return np.sum(x) - _g_n / 2.0

def slsqp_opt(h0, n, T, maxiter=500, deadline=None):
    global _g_T, _g_n
    _g_T = T
    _g_n = n
    try:
        res = minimize(_obj_f, h0, jac=_grad_f, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con_f}],
                       options={'ftol': 1e-12, 'maxiter': maxiter, 'disp': False})
        if deadline is not None and time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def fourier_perturb(h, n, rng, scale=0.05):
    """Add Fourier mode perturbations to escape local minima."""
    modes = rng.choice(n // 2, size=min(5, n // 4), replace=False)
    h_pert = h.copy()
    for m in modes:
        amp = scale * rng.randn()
        h_pert += amp * np.cos(2 * np.pi * m * np.arange(n) / n)
        h_pert += amp * np.sin(2 * np.pi * m * np.arange(n) / n)
    return project(h_pert, n)

def random_feasible(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 20

    global _g_T, _g_n

    _g_n = 336

        h_start = np.array(initial_h_values, dtype=float).copy()
    else:

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Also try at the parent resolution first
    n_parent = len(h_start)
    
    # Phase 1: Optimize at parent resolution with increasing T
    T_schedule = [20.0, 50.0, 100.0, 200.0, 400.0]
    h_work = h_start.copy()
    for T in T_schedule:
        if time.time() > deadline - 100:
            break
        h_new = slsqp_opt(h_work, n_parent, T, maxiter=300, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_work = h_new

    # Phase 2: Upscale to higher resolution and optimize
    n_high = max(2 * n_parent, 672)  # e.g., 672
    h_high = interp_to(best_h, n_parent, n_high)
    h_high = project(h_high, n_high)
    
    T_schedule_high = [30.0, 60.0, 100.0, 200.0, 400.0, 800.0]
    for T in T_schedule_high:
        if time.time() > deadline - 60:
            break
        h_new = slsqp_opt(h_high, n_high, T, maxiter=400, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_high = h_new

    # Phase 3: Try even higher resolution if time permits
    n_very_high = min(4 * n_parent, 1344)
    if n_very_high > n_high and time.time() < deadline - 120:
        h_vh = interp_to(best_h, len(best_h), n_very_high)
        h_vh = project(h_vh, n_very_high)
        for T in [50.0, 100.0, 200.0, 400.0]:
            if time.time() > deadline - 40:
                break
            h_new = slsqp_opt(h_vh, n_very_high, T, maxiter=300, deadline=deadline)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_vh = h_new

    # Phase 4: Structured restarts at best resolution
    n_best = len(best_h)
    rng = np.random.RandomState(seed + 777)
    for attempt in range(50):
        if time.time() > deadline - 30:
            break
        h_pert = fourier_perturb(best_h, n_best, rng, scale=0.06)
        h_new = slsqp_opt(h_pert, n_best, 200.0, maxiter=200, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-6:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Random restarts

        h_rand = random_feasible(n_best, seed + 5000 + attempt)
        h_new = slsqp_opt(h_rand, n_best, 100.0, maxiter=200, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:

        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def soft_max_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = T * corr
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    obj = (m + np.log(sum_exp)) / T * dx
    
    grad = np.zeros(n)
    active = p > 1e-12
    for k in np.where(active)[0]:
