term1[j] = ∑_m K1[m] * h[j-m] = (K1 ⊛ h)[j] (convolution)
term2[j] = ∑_m K1[m] * h[j+m] = ∑_m K1[-m] * h[j-m] = (K1_rev ⊛ h)[j] (convolution with reversed kernel)

Using FFT:
- Compute FFT of h (zero-padded to size ≥ 3n)
- Compute FFT of K1
- term1 = ifft(FFT(K1) * FFT(h)) (take first n elements)
- Compute FFT of K1_rev
- term2 = ifft(FFT(K1_rev) * FFT(h)) (take first n elements)

This is O(n log n) per gradient step!

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n * 0.5
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def objective(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = float(np.max(scaled))
    return (m + float(np.log(np.sum(np.exp(scaled - m))))) / T

def gradient(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = float(np.max(scaled))
    exp_s = np.exp(scaled - m)
    Z = float(np.sum(exp_s))
    p = exp_s / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        shift = k - (n - 1)
        w = pk * dx
        if shift >= 0:
            grad[shift:] += w * (1.0 - h[:n - shift])
            grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
    return grad

def pgd(h0, n, T, deadline, max_iter, lr_init):
    h = h0.copy()
    lr = lr_init
    best_h = h0.copy()
    best_obj = objective(h, n, T)
    stalled = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = gradient(h, n, T)
        # Backtracking
        for bt in range(50):
            h_new = project_feasible(h - lr * g, n)
            obj_new = objective(h_new, n, T)
            if obj_new < best_obj - 1e-14 * max(1.0, abs(best_obj)):
                h = h_new
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_h = h.copy()
                    stalled = 0
                else:
                    stalled += 1
                break
            lr *= 0.5
        else:
            break
        lr = min(lr * 1.3, lr_init)
        if stalled > 200:
            break
    return best_h

def interp_up(h, n_to):
    n_from = len(h)
    if n_from >= n_to:
        return h[:n_to].copy()  # or just return h
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)
```

```python
def gradient_fft(h, n, p):
    """Compute gradient using FFT.
    h: array of length n
    p: probability vector of length 2n-1 (p[k] for shift k-(n-1))
    """
    dx = 2.0 / n
    N = 1
    while N < 3 * n:
        N <<= 1
    
    # K1[m] for m=0..n-1 corresponds to shifts 0..n-1 (p[n-1..2n-2])
    # K1[m] for m=n..2n-2 corresponds to shifts -(n-1)..-1 (p[0..n-2])
    # In the FFT representation, we place K1 centered at 0:
    # K1_fft_array[0] = K1[0] (shift 0)
    # K1_fft_array[1..n-1] = K1[1..n-1] (shifts 1..n-1)
    # K1_fft_array[N-(n-1)..N-1] = K1[n..2n-2] (shifts -(n-1)..-1)
    
    K1 = np.zeros(N)
    K1[0:n] = p[n-1:2*n-1]  # shifts 0..n-1
    K1[N-(n-1):N] = p[0:n-1]  # shifts -(n-1)..-1
    
    # K2[m] = K1[-m] (for the correlation term)
    # K2_fft_array[0] = K2[0] = K1[0]
    # K2_fft_array[1..n-1] = K2[1..n-1] = K1[-1..-(n-1)] = K1[n..2n-2] = p[0..n-2]
    # K2_fft_array[N-(n-1)..N-1] = K2[2-n..-1]... 
    
    # Actually, K2[m] = K1[-m]. In the FFT array:
    # K1 is at positions [0, 1, ..., n-1, N-(n-1), ..., N-1]
    # K2[m] = K1[-m], so K2 at position x in FFT = K1 at position (-x mod N)
    # So K2_fft_array[x] = K1_fft_array[(-x) % N]
    
    K2 = np.zeros(N)
    for i in range(n):  # This is O(n), fine
        K2[i] = K1[(-i) % N]
    
    # Simpler: K2 is just the circular reversal of K1 (with K1[0] fixed)
    K2[0] = K1[0]
    K2[1:] = K1[:0:-1]  # Hmm, not quite right for circular
    
    # Let me just do it explicitly:
    K2 = np.zeros(N)
    K2[0] = K1[0]
    # For i = 1..n-1: K2[i] = K1[N-i] (which is K1 at shift -(n-1)+i... )
    # K2[1] = K1[-1] = K1[N-1] = p[n-2] (shift -1)
    # K2[2] = K1[-2] = K1[N-2] = p[n-3] (shift -2)
    # ...
    # K2[n-1] = K1[-(n-1)] = K1[N-(n-1)] = p[0] (shift -(n-1))
    K2[1:n] = K1[N-1:N-1-(n-1):-1]  # K1[N-1], K1[N-2], ..., K1[N-(n-1)]
    # For i = N-(n-1)..N-1: K2[i] = K1[N-i]
    # K2[N-(n-1)] = K1[n-1] = p[2n-2] (shift n-1)
    # K2[N-(n-2)] = K1[n-2] = p[2n-3]
    # ...
    # K2[N-1] = K1[1] = p[n] (shift 1)
    K2[N-(n-1):N] = K1[n-1:0:-1]  # K1[n-1], K1[n-2], ..., K1[1]
    
    H = np.fft.rfft(h, N)
    K1_F = np.fft.rfft(K1, N)
    K2_F = np.fft.rfft(K2, N)
    
    conv1 = np.fft.irfft(K1_F * H, N)[:n]
    conv2 = np.fft.irfft(K2_F * H, N)[:n]
    
    grad = dx * (1.0 - conv1 - conv2)
    return grad
```
