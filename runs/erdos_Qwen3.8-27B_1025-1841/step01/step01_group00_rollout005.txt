def project_to_feasible(g, n):
    """Project g onto {h : 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Use bisection on threshold
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

                stall = 0
                if lr < 1e-12:
                    break
        h = h_new
    
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return project_to_feasible(np.interp(x_new, x_old, h, period=2.0), n_new)

def subgrad_polish(h, n, time_limit, lr_init=0.003):
    start = time.time()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.7
    lr = lr_init
    stall = 0
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, 'full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                if lr < 1e-12:
                    break
                stall = 0
        
        k_star = int(np.argmax(corr))
        sv = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sv)
        j1 = min(n, sv + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 + sv:j1 + sv])
        j0b = max(0, -sv)
        j1b = min(n, n - sv)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + sv:j1b + sv]
        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5
