```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(x, n, T, lam):
    dx = 2.0 / n
    omx = 1.0 - x
    corr = np.correlate(x, omx, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    p = exps / exps.sum()
    
    smax = (m + np.log(exps.sum())) / T
    s = x.sum()
    obj = smax * dx + lam * (s - n/2.0)**2
    
    # Gradient using convolution
    r1 = np.convolve(omx, p, 'full')[n-1:2*n-1]
    r2 = np.convolve(x, p[::-1], 'full')[n-1:2*n-1]
    grad = dx * (r1 - r2)
    if lam > 0:
        grad += 2.0 * lam * (s - n/2.0)
    
    return obj, grad

def optimize_lbfgsb(h0, n, T, lam, maxiter=300):
    res = minimize(lambda x: smooth_obj_grad(x, n, T, lam),
                   h0, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                           'maxls': 50, 'maxcor': 30})
    return res.x if res.x is not None else h0

def upsample(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(40, 0.5)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Determine resolutions to try
    n_start = len(h_init)
    resolutions = []
    if n_start <= 100:
        resolutions = [n_start, 200, 400, 800, 1200]
    elif n_start <= 400:
        resolutions = [n_start, 2*n_start, 4*n_start]
    else:
        resolutions = [n_start, 2*n_start]
    
    # Filter out resolutions beyond what's feasible
    max_n = 2000
    
    for n in resolutions:
        if n > max_n:
            continue
        if time.time() > deadline - 100:
            break
        
        # Get working h at this resolution
        if n == len(best_h):
            h_work = best_h.copy()
        elif n > len(best_h):
            h_work = upsample(best_h, len(best_h), n)
            h_work = project_onto_constraint(h_work, n)
        else:
            continue
        
        # Optimize with increasing T
        for T in [50, 200, 500, 1000, 2000]:
            if time.time() > deadline - 100:
                break
            lam = 1e6
            h_new = optimize_lbfgsb(h_work, n, T, lam, maxiter=200)
            h_new = project_onto_constraint(h_new, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_work = h_new
    
    # Random restarts at best resolution
    n_best = len(best_h)
    n_restarts = 20
    for i in range(n_restarts):
        if time.time() > deadline - 30:
            break
        # Different types of perturbations
        if i % 3 == 0:
            h_pert = best_h + rng.randn(n_best) * 0.05
        elif i % 3 == 1:
            # Binary-like restart
            h_pert = np.zeros(n_best)
            idx = rng.choice(n_best, n_best//2, replace=False)
            h_pert[idx] = 1.0
            h_pert += 0.01 * rng.randn(n_best)
        else:
            # Structured perturbation
            h_pert = best_h.copy()
            block = max(1, n_best // 20)
            for b in range(0, n_best - block, block):
                h_pert[b:b+block] = 1.0 - h_pert[b:b+block]
        
        h_pert = project_onto_constraint(h_pert, n_best)
        for T in [200, 1000]:
            if time.time() > deadline - 30:
                break
            h_new = optimize_lbfgsb(h_pert, n_best, T, 1e6, maxiter=150)
            h_new = project_onto_constraint(h_new, n_best)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_pert = h_new
    
    # Final: try even higher resolution if time permits
    if time.time() < deadline - 60 and n_best < 1500:
        n_hi = min(1500, n_best * 2)
        h_hi = upsample(best_h, n_best, n_hi)
        h_hi = project_onto_constraint(h_hi, n_hi)
        for T in [500, 2000]:
            if time.time() > deadline - 20:
                break
            h_new = optimize_lbfgsb(h_hi, n_hi, T, 1e6, maxiter=150)
            h_new = project_onto_constraint(h_new, n_hi)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_hi = h_new
    
    return (best_h, best_c5, len(best_h))
```

Let me also add a coordinate descent polishing step, which can be very effective for this kind of problem:

For each coordinate i, fix all other h[j] and find the optimal h[i]:
- The objective as a function of h[i] is: max_k (a_k * h[i] + b_k) * dx + lam*(sum - target)²
- The first term is convex piecewise-linear in h[i]
- The minimum of a convex piecewise-linear function on [0,1] is at a breakpoint or boundary

But with the penalty, it's no longer piecewise-linear. Let me skip this for now and rely on L-BFGS-B.

Actually, let me add a simpler polishing: after L-BFGS-B, do a few rounds of "evaluate all coordinates, try small perturbations, keep if improved". This is essentially coordinate descent with perturbation.

Let me also think about whether I should try cvxpy or other solvers. The problem is non-convex (bilinear), so convex solvers won't directly help. But I could:
- Linearize the max: C5 >= R[k] * dx for all k, minimize C5 subject to this. But the constraints are still non-convex (bilinear in h).
- Use iterative linearization: fix h, compute R, add constraint R[k] <= C5, optimize h... this is like Dinkelbach's algorithm or bisection.

Actually, here's another idea: **bisection on C5**. For a given target C, check if there exists h such that:
- 0 <= h[i] <= 1
