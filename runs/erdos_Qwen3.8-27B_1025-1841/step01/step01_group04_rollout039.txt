import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / T)

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    m = np.max(corr)
    weights = np.exp(corr - m) / np.sum(np.exp(corr - m))
    # Gradient of soft_max: sum_k weights[k] * grad(corr_k)
    # This is expensive; use subgradient approximation
    # For now, just return the subgradient of the max
    tau_star = np.argmax(corr_raw)
    lag = tau_star - (n - 1)
    abs_lag = abs(lag)
    if abs_lag == 0:
        sg = 2.0 * h
    else:
        if lag > 0:
            h_plus = np.concatenate((h[lag:], np.zeros(lag)))
            h_minus = np.concatenate((np.zeros(lag), h[:n-lag]))
        else:
            h_plus = np.concatenate((np.zeros(abs_lag), h[:n-abs_lag]))
            h_minus = np.concatenate((h[abs_lag:], np.zeros(abs_lag)))
        sg = -(h_plus + h_minus)  # negative gradient direction (we want to decrease)
    return sg * dx

def subgrad_descent(h_init, n_steps, alpha0, deadline):

    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)

        if time.time() > deadline:
            break
        alpha = alpha0 / (1.0 + step / 100.0)

        abs_lag = abs(lag)

            direction = 1.0 - 2.0 * h  # gradient of f_0

            else:

                h_minus = np.concatenate((h[abs_lag:], np.zeros(abs_lag)))
            direction = 1.0 - h_plus - h_minus  # gradient

**Algorithm: Adaptive Subgradient Descent with Momentum**

1. Start from initial_h_values, upsample to target resolution
2. Nesterov-accelerated subgradient descent:
   - α_t = α_0 / sqrt(t+1)
   - β_t = (1 + (t+1)/2) / (1 + (t+3)/2) ≈ 0.5 + 0.5*t/(t+2) → approaches 0.5...

3. At each step:
   a. y = h + β*(h - h_prev)  (momentum extrapolation)
   b. Compute corr(y), find argmax lag τ*
   c. Compute subgradient: sg[j] = (1-y[j+τ*]) - (1-y[j-τ*]) = y[j-τ*] - y[j+τ*]
   d. h_new = y - α_t * sg
   e. Project h_new to [0,1] with sum = n/2
   f. h_prev = h, h = h_new

4. Track the best solution (lowest C5) seen so far.

The key advantage: each step is O(n) for the correlation (actually O(n²) in the worst case for np.correlate, but it's in C so it's fast) + O(n) for the gradient + O(n*60) for projection.

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    if abs(np.sum(h) - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(80):

        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:

def subgrad_nesterov(h0, n_steps, alpha0, beta, deadline):
    """Nesterov-accelerated subgradient descent"""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(n_steps):
        if time.time() > deadline:
            break
        alpha = alpha0 / np.sqrt(t + 1.0)
        
        # Momentum extrapolation
        if t == 0:
            y = h
        else:
            y = h + beta * (h - h_prev)
            y = project_box_sum(y, target)
        
        # Compute correlation and find active lag
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_val = np.max(corr) * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = y.copy()
        
        tau_star = int(np.argmax(corr))
        lag = tau_star - (n - 1)
        abs_lag = abs(lag)
        
        # Compute subgradient at y
        if abs_lag == 0:
            sg = (1.0 - 2.0 * y)  # ∇f_0
        else:
            if lag > 0:
                y_plus = np.concatenate((y[lag:], np.zeros(lag)))
                y_minus = np.concatenate((np.zeros(lag), y[:n-lag]))
            else:
                y_plus = np.concatenate((np.zeros(abs_lag), y[:n-abs_lag]))
                y_minus = np.concatenate((y[abs_lag:], np.zeros(abs_lag)))

        h_new = project_box_sum(h_new, target)
        h_prev = h
        h = h_new
    
    # Final evaluation
    c5_final = evaluate_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    
    return best_h, best_c5

def slsqp_refine(h0, T, maxiter, deadline):

    target = n / 2.0

        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(corr - m))) / T

                xm = np.concatenate((x[abs_lag:], np.zeros(abs_lag)))

        pass

                y_m = np.concatenate((y[abs_lag:], np.zeros(abs_lag)))
            sg = y_m - y_p  # = (1-y_p) - (1-y_m)
        
        h_new = y - step_alpha * sg
        h_new = project_box_sum(h_new, target)
        h_prev = h
        h = h_new
    
    c5_final = evaluate_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def coordinate_sweep(h, n_sweeps, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full').copy()
        
        for j in range(n):
            if time.time() > deadline:
                break
            
            # Compute d: change in corr per unit change in h[j]
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                tau = k - (n - 1)
                jp = j + tau
                jm = j - tau
                d[k] = ((1.0 - h[jp]) if 0 <= jp < n else 0.0) - ((h[jm]) if 0 <= jm < n else 0.0)
            
            s_lo = -h[j]
            s_hi = 1.0 - h[j]
            for _ in range(30):
                s_mid = (s_lo + s_hi) * 0.5
                val = corr + s_mid * d
                max_idx = int(np.argmax(val))
                if d[max_idx] > 1e-14:
                    s_hi = s_mid
                elif d[max_idx] < -1e-14:
                    s_lo = s_mid
                else:
                    break
            
            s_opt = (s_lo + s_hi) * 0.5
            h[j] += s_opt
            corr += s_opt * d
            one_minus_h[j] = 1.0 - h[j]
        
        h = project_box_sum(h, target)
    
    return h

def interp_to(h, n_to):
    n_from = len(h)
    if n_from == n_to:
        return h.copy()
    xf = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    xt = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    return np.interp(xt, xf, h)
