def evaluate_c5(h, dx):
    """Compute C5 = max of correlate(h, 1-h, 'full') * dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum, 0 <= h <= 1"""
    # Simple: scale and clip
    # But this might not preserve the sum exactly
    # Use Wasserstein projection (iterative)
    n = len(h)

    # Adjust to meet sum constraint
    for _ in range(100):
        diff = h_proj.sum() - target_sum
        if abs(diff) < 1e-10:
            break
        # Add/subtract uniformly from values that can be adjusted
        mask_up = h_proj < 1
        mask_dn = h_proj > 0
        if diff > 0 and mask_dn.sum() > 0:
            h_proj[mask_dn] -= diff / mask_dn.sum()
        elif diff < 0 and mask_up.sum() > 0:
            h_proj[mask_up] += (-diff) / mask_up.sum()
        h_proj = np.clip(h_proj, 0, 1)

    return best_h, final_c5, n
