import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_and_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, int(np.argmax(corr))

def grad_corr_at(h, k):
    """Gradient of corr[k] w.r.t. h, where k is index in full correlation output."""
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # c[k] = sum_{j=0}^{n-1-lag} h[j+lag]*(1-h[j])
        # dc/dh[i] = (1-h[i-lag])*1{lag<=i} - h[i+lag]*1{i<=n-1-lag}
        for i in range(lag, n):
            grad[i] += (1.0 - h[i - lag])
        for i in range(0, n - lag):
            grad[i] -= h[i + lag]
    else:
        l = -lag
        # c[k] = sum_{m=0}^{n-1-l} h[m]*(1-h[m+l])
        # dc/dh[i] = (1-h[i+l])*1{i<=n-1-l} - h[i-l]*1{i>=l}
        for i in range(0, n - l):
            grad[i] += (1.0 - h[i + l])
        for i in range(l, n):
            grad[i] -= h[i - l]
    return grad

def project_feasible(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} using bisection."""
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

            g[i] += (1.0 - h[i + l])

        for i in range(shift, n):
            g[i] += (1.0 - h[i - shift])
        for i in range(0, n - shift):
            g[i] -= h[i + shift]
    else:
        l = -shift
        for i in range(0, n - l):
            g[i] += (1.0 - h[i + l])
        for i in range(l, n):
            g[i] -= h[i - l]
    return g

def pgd_optimize(h, dx, max_iter=50000, time_limit=30.0):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h.copy(), target)
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    alpha = 1.0
    t_start = time.time()
    stale = 0
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        g = grad_max_corr(h)
        h_new = project_feasible(h - alpha * g * dx, target)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5 - 1e-11:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            alpha = min(alpha * 1.02, 100.0)
            stale = 0
        else:
            h = h_new
            alpha *= 0.82
            stale += 1
            if stale > 800:
                h = project_feasible(h + 0.02 * np.random.randn(n), target)
                alpha = 1.0
                stale = 0
    return best_h, best_c5

def binary_local_search(h, dx, max_rounds=200, time_limit=30.0):
    n = len(h)
    target = n / 2.0
    h = (h >= 0.5).astype(float)
    ones_count = int(np.sum(h))
    if ones_count > target:
        idx = np.where(h == 1)[0]
        for j in idx[ones_count - int(target):]:
            h[j] = 0.0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        for j in idx[:int(target) - ones_count]:
            h[j] = 1.0
    best_c5 = evaluate_c5(h, dx)
    t_start = time.time()
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        for i in range(len(one_idx)):
            for j in range(len(zero_idx)):
                h[one_idx[i]] = 0.0
                h[zero_idx[j]] = 1.0
                c5_trial = evaluate_c5(h, dx)
                if c5_trial < best_c5 - 1e-12:
                    best_c5 = c5_trial
                    improved = True
                else:
                    h[one_idx[i]] = 1.0
                    h[zero_idx[j]] = 0.0
                    continue
                h[one_idx[i]] = 1.0
                h[zero_idx[j]] = 0.0
                one_idx = np.where(h == 1)[0]
                zero_idx = np.where(h == 0)[0]
                break
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def interp_up(h_coarse, n_fine):
    n_c = len(h_coarse)
    x_c = (np.arange(n_c) + 0.5) * (2.0 / n_c)
    x_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    h_f = np.interp(x_f, x_c, h_coarse)
    return project_feasible(h_f, n_fine / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        h0 = np.full(42, 0.5)

    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = project_feasible(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx0)
    best_n = n0

    def elapsed():
        return time.time() - t0

    if elapsed() < budget_s - 60:
        try:
            def obj_slsqp(h_v):
                return evaluate_c5(np.clip(h_v, 0, 1), dx0)
            cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n0 / 2.0,
                     'jac': lambda h: np.ones(n0)}]
            bounds = [(0.0, 1.0)] * n0
            res = minimize(obj_slsqp, h0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-12})
            h_s = project_feasible(np.clip(res.x, 0, 1), n0 / 2.0)
            c5_s = evaluate_c5(h_s, dx0)
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s
        except:
            pass

    if elapsed() < budget_s - 40:
        tl = min(120.0, (budget_s - elapsed()) * 0.35)
        h_p, c5_p = pgd_optimize(best_h, dx0, max_iter=80000, time_limit=tl)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    for r in range(6):
        if elapsed() > budget_s - 250:
            break
        h_r = project_feasible(best_h + 0.03 * np.random.randn(n0), n0 / 2.0)
        h_r, c5_r = pgd_optimize(h_r, dx0, max_iter=40000, time_limit=25.0)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
            best_n = n0

    if elapsed() < budget_s - 50:
        h_b, c5_b = binary_local_search(best_h, dx0, max_rounds=300, time_limit=30.0)
        if c5_b < best_c5:
            best_h, best_c5, best_n = h_b, c5_b, n0
            h_p2, c5_p2 = pgd_optimize(best_h, dx0, max_iter=30000, time_limit=20.0)
            if c5_p2 < best_c5:
                best_h, best_c5 = h_p2, c5_p2

    for n_f in [84, 168, 336, 672, 1260]:
        if elapsed() > budget_s - 80:
            break
        rem = budget_s - elapsed()
        if rem < 60:
            break
        h_f = interp_up(best_h, n_f)
        dx_f = 2.0 / n_f
        alloc = min(rem * 0.45, 250.0)
        if alloc < 15:
            break
        h_f, c5_f = pgd_optimize(h_f, dx_f, max_iter=40000, time_limit=alloc)
        if c5_f < best_c5:
            best_h, best_c5, best_n = h_f, c5_f, n_f

    for n_f in [84, 168, 336]:
        if elapsed() > budget_s - 40:
            break
        rem = budget_s - elapsed()
        if rem < 30:
            break
        h_f = interp_up(best_h, n_f)
        dx_f = 2.0 / n_f
        h_b2, c5_b2 = binary_local_search(h_f, dx_f, max_rounds=100, time_limit=min(20, rem * 0.4))
        if c5_b2 < best_c5:
            best_h, best_c5, best_n = h_b2, c5_b2, n_f

    dx = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = evaluate_c5(best_h, dx)

    return (best_h, final_c5, best_n)
