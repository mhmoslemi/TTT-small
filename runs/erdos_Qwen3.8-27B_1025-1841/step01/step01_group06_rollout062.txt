from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    """Project h to {0≤h≤1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Distribute the adjustment
        adj = diff / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-14):
            break
        h = h_new
    # Fine correction
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-12:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            per = rem / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_to_feasible(h.copy(), n_new)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)  # periodic interp
    return project_to_feasible(np.clip(h_new, 0, 1), n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    rem = target - np.sum(h)
    if abs(rem) > 1e-14:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h.copy(), n_new)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_new)

def smooth_max_val(corr, temp):
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def smooth_grad_full(h, temp, n):
    corr = compute_corr(h)
    m = np.max(corr)
    p = np.exp((corr - m) / temp)
    p /= p.sum()
    grad = np.zeros(n)
    w0 = p[n - 1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)
    for L in range(1, n):
        wp = p[n - 1 + L]
        if wp > 1e-12:
            grad[:n - L] -= wp * h[L:]
            grad[L:] += wp * (1.0 - h[:n - L])
        wn = p[n - 1 - L]
        if wn > 1e-12:
            grad[:n - L] += wn * (1.0 - h[L:])
            grad[L:] -= wn * h[:n - L]
    return grad

def p_norm_smooth(h, p, n):
    corr = compute_corr(h)
    S = float(len(corr))
    return np.sum(corr**p) / S

def p_norm_grad(h, p, n):
    corr = compute_corr(h)
    S = float(len(corr))
    val = (np.sum(corr**p) / S) ** (1.0 / p)
    if val < 1e-15:
        return np.zeros(n)
    w = (corr ** (p - 1)) / (S * val ** (p - 1))
    w_sum = np.sum(w)
    if w_sum < 1e-15:
        return np.zeros(n)
    w /= w_sum
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)
    for L in range(1, n):
        wp = w[n - 1 + L]
        if wp > 1e-12:
            grad[:n - L] -= wp * h[L:]

        wn = w[n - 1 - L]
        if wn > 1e-12:
            grad[:n - L] += wn * (1.0 - h[L:])
            grad[L:] -= wn * h[:n - L]
    return grad * val
