    # This is getting too complicated. Let me just use the direct O(n²) approach but vectorized with numpy.
    
    # Actually, for n up to 2000, the O(n²) approach with a Python loop over 2n-1 lags
    # where each lag does O(n) work via numpy should be fast enough.
    # Let me estimate: for n=1500, 2n-1 = 2999 lags, each doing ~1500 ops in numpy.
    # Python loop overhead: ~1μs per iteration. numpy op: ~5μs for small arrays.
    # Total: 2999 * 10μs ≈ 30ms per gradient. 10000 iters = 300s. Fine!
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            j1 = n - sh
            if j1 > 0:
                grad[:j1] += pk * (1.0 - h[sh:n])
                grad[sh:n] -= pk * h[:j1]
        else:
            s = -sh
            j1 = n - s
            if j1 > 0:
                grad[s:n] += pk * (1.0 - h[:j1])
                grad[:j1] -= pk * h[s:n]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad
```

**Backtracking line search**:
- Start with a large lr.
- If the objective increases, halve lr and retry.
- This ensures monotone decrease (approximately, since we also project).

def optimize_smooth_with_line_search(h, n, tau, max_iters, time_limit, lr_init=1.0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    obj0, _ = smooth_obj_grad_fast(h, tau)
    best_obj = obj0
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad_fast(h, tau)
        
        # Backtracking line search
        step = lr
        for _ in range(20):
            h_trial = project_to_feasible(h - step * grad, n)
            obj_trial, _ = smooth_obj_grad_fast(h_trial, tau)
            if obj_trial < obj - 1e-16:
                break
            step *= 0.5
        
        h = h_trial
        obj = obj_trial
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
        
        # Adaptive lr
        if it % 100 == 0:
            # Check progress
            pass
    
    return best_h, best_obj

Let me also think about **what makes this problem hard**. The main challenge is that the max function creates a non-smooth objective with potentially many "active" constraints. Near the optimum, many correlations are close to the maximum, and we need to reduce all of them simultaneously.

The smooth approximation (log-sum-exp) handles this by assigning weight to all correlations, not just the maximum. The tau parameter controls how "peaky" the weights are:
- Large tau: only the max gets significant weight → behaves like subgradient of max
- Small tau: all correlations get roughly equal weight → behaves like minimizing the average

For our problem, I think an intermediate tau is best: large enough to focus on the max, but small enough to also reduce the near-max correlations.

I can build up a list of "active" correlations (those within epsilon of the max) and use their average gradient instead of just the max gradient. ```python
def optimize_active_set(h, n, max_iters, time_limit, eps=0.005, initial_lr=None):
    """Subgradient method using active set."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr else 0.01
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        
        # Find active set: correlations within eps of max
        max_corr = corr.max()
        active = np.where(corr >= max_corr - eps / dx)[0]
        
        # Average gradient over active set
        grad = np.zeros(n)
        for k in active:
            sh = k - (n - 1)
            if sh >= 0:
                j1 = n - sh
                if j1 > 0:
                    grad[:j1] += (1.0 - h[sh:n])
                    grad[sh:n] -= h[:j1]
            else:
                s = -sh
                j1 = n - s
                if j1 > 0:
                    grad[s:n] += (1.0 - h[:j1])
                    grad[:j1] -= h[s:n]
        grad /= len(active)
        grad *= dx
        
        h = project_to_feasible(h - lr * grad, n)
        lr *= 0.999  # slow decay
    
    return best_h, best_c5
```

This active-set approach is a compromise between the true max (subgradient) and the smooth max (all correlations). It should converge faster because it uses more gradient information.

Let me also consider: should I try n=900 (same as parent) but with better optimization? Or n=1200? Higher resolution gives more degrees of freedom but also more correlations to control. The C₅ value scales as 1/n in a certain sense (the dx factor), so the correlation values are proportional to n, but the C₅ = max(corr) * dx = max(corr) * 2/n should be resolution-independent (if the function is the same, just sampled differently).

Wait, that's an important point! If I have a function h defined on [0,2] and I sample it at higher resolution, the C₅ value should be approximately the same (the integral is better approximated). So higher resolution doesn't directly help reduce C₅; it just gives a better approximation of the true integral.

However, higher resolution gives more degrees of freedom (n variables vs. fewer), which can help the optimizer find a better solution. The trade-off is:
- More variables → easier to satisfy constraints → potentially lower C₅
- But also more correlations to control → harder optimization

In practice, for this problem, I expect that going from n=900 to n=1200 or n=1500 will give a small improvement (maybe 0.0001-0.001) due to the extra degrees of freedom.

Let me also think about **structural insights**. The optimal h should have some structure. For the Erdős problem, I recall that the optimal function has a "waist" - it's high in the middle and low at the edges (or vice versa). This creates a correlation profile where the peak is at a non-obvious shift.

Actually, for the specific problem with h: [0,2] → [0,1] and ∫h = 1, the function h = 1/2 + c*cos(2πx) might be a good starting point. Let's check:
h(x) = 1/2 + c*cos(2πx/2) = 1/2 + c*cos(πx)
∫₀² h dx = 1/2*2 + c*∫₀² cos(πx)dx = 1 + 0 = 1 ✓

Overlap at shift k:
∫ h(x)(1-h(x+k))dx = ∫ (1/2 + c*cos(πx))(1/2 - c*cos(π(x+k)))dx
= 1/4 - c/2 * ∫cos(π(x+k))dx + c/2 * ∫cos(πx)dx - c² * ∫cos(πx)cos(π(x+k))dx
= 1/4 - c² * ∫cos(πx)cos(πx+πk)dx
= 1/4 - c²/2 * ∫[cos(2πx+πk) + cos(πk)]dx (using product-to-sum)
= 1/4 - c²/2 * [0 + 2*cos(πk)] (the integral of cos(2πx+πk) over [0,2] is 0)
= 1/4 - c² * cos(πk)

So the overlap varies between 1/4 - c² (at k=1, odd) and 1/4 + c² (at k=0,2, even).

The maximum is 1/4 + c². To minimize this, set c=0, giving C₅ = 1/4 = 0.25. But that's the constant function h=1/2, which we showed gives C₅ = 0.5 due to the boundary effects of linear correlation!

Ah, the issue is that the linear correlation at large shifts has less overlap, so the integral is smaller there. The max is at shift 0 (or near it), where the full overlap gives 1/2 for h=1/2.

So the single cosine mode isn't enough. We need higher frequency components.

Let me try h(x) = 1/2 + c*cos(2πnx) for higher n:
∫₀² cos(2πnx)cos(2πn(x+k))dx = ∫ cos(2πnx)cos(2πnx + 2πnk)dx
= 1/2 ∫ [cos(4πnx + 2πnk) + cos(2πnk)]dx
= 1/2 * 0 + 1/2 * 2 * cos(2πnk) = cos(2πnk)

So overlap = 1/4 - c² * cos(2πnk). The max over k is 1/4 + c² (at k=0). Same issue.

The problem is that for any single frequency, the overlap at k=0 is always 1/4 + c² (or 1/4 - c² depending on the frequency).

For a binary function at k=0: overlap = 0 (as we computed). So binary is much better at k=0.

For a binary function, the overlap at shift k is the count of (1,0) pairs at distance k, times dx. The question is: can we arrange n/2 ones to minimize the maximum (1,0) count over all shifts?

This is essentially the **discrepancy problem** or **sequential discrepancy**. The optimal discrepancy for a sequence of n/2 ones and n/2 zeros is O(√n) (by Spencer's theorem or the balanced sequence construction).

If the max (1,0) count is about √n * n/4... hmm, that doesn't seem right either.

Actually, for a "balanced" binary sequence (like a van der Corput or Thue-Morse sequence), the correlation at shift k is approximately (n-k)/4 (the "random" value) with fluctuations of order √(n-k). So the max correlation is about (n/4) + √n, and C₅ ≈ (n/4 + √n) * 2/n ≈ 1/2 + 2√n/n ≈ 1/2.

That's still about 0.5, much worse than 0.38. So binary functions aren't the answer.

The optimal function must be non-binary! The key is that a non-binary function can have a correlation at shift 0 that's close to 0 (by being close to binary) while having lower correlations at other shifts (by being "smooth" in a way that anti-correlates h(x) and h(x+k)).

This is a tension: binary is good for shift 0 but bad for other shifts; smooth (like 1/2) is bad for shift 0 but "uniform" for other shifts. The optimum is somewhere in between.

I think the optimal function looks like a "smoother" binary function - close to 0 or 1 in most places, but with transitions that are carefully positioned to minimize the cross-correlation at all shifts.

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def grad_single_corr(h, k):
    n = len(h)
    grad = np.zeros(n)
    sh = k - (n - 1)
    if sh >= 0:
        j1 = n - sh
        if j1 > 0:
            grad[:j1] += (1.0 - h[sh:n])
            grad[sh:n] -= h[:j1]
    else:
        s = -sh
        j1 = n - s
        if j1 > 0:
            grad[s:n] += (1.0 - h[:j1])
            grad[:j1] -= h[s:n]
    return grad

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        g = grad_single_corr(h, k)
        grad += pk * g
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def optimize_smooth(h, n, tau, max_iters, time_limit, lr_init=0.3):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 100:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-9:
                break
        elif stall == 0 and it % 200 == 0:
            lr *= 0.98

        h = h_new

    return best_h, best_obj

def optimize_active(h, n, max_iters, time_limit, eps_frac=0.02, lr_init=0.005):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        max_corr = corr.max()
        c5 = max_corr * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        thresh = max_corr - eps_frac * dx * n
        active = np.where(corr >= thresh)[0]

        grad = np.zeros(n)
        for k in active:
            grad += grad_single_corr(h, k)
        grad /= len(active)
        grad *= dx

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 150:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        elif stall == 0 and it % 300 == 0:
            lr *= 0.995

        h = h_new

    return best_h, best_c5

def optimize_topk(h, n, max_iters, time_limit, topk=10, lr_init=0.003):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += grad_single_corr(h, k)
        grad /= topk
        grad *= dx

        h_new = project_to_feasible(h - lr * grad, n)

        if stall > 200:
            lr *= 0.5
            stall = 0
            if lr < 1e-12:
                break

        h = h_new

    return best_h, best_c5
