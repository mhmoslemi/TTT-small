    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def exact_subgradient(h, n, k_star):
    s = k_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[s:s+m] += (1.0 - h[:m])
        grad[:m] -= h[s:s+m]
    else:
        t = -s
        m = n - t
        grad[:m] += (1.0 - h[t:t+m])
        grad[t:t+m] -= h[:m]
    return grad

def smoothed_subgradient(h, n, temp, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp((corr - m) / temp)
    total = np.sum(weights)
    if total < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= total
    grad = np.zeros(n)
    sig = np.where(weights > 1e-8)[0]
    for k_idx in sig:
        w = weights[k_idx]
        s = k_idx - (n - 1)
        if s >= 0:
            m2 = n - s
            grad[s:s+m2] += w * (1.0 - h[:m2])
            grad[:m2] -= w * h[s:s+m2]
        else:
            t = -s
            m2 = n - t
            grad[:m2] += w * (1.0 - h[t:t+m2])
            grad[t:t+m2] -= w * h[:m2]
    return grad * dx

def optimization_phase1(h, n, dx, n_iter, start_temp, end_temp, base_lr, deadline):
    """Smoothed projected gradient with Nesterov acceleration."""
    c5 = compute_c5(h, dx)
    x = h.copy()
    y = h.copy()
    mu = 1.0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t_it = 1.0 / (1.0 + 0.5 * np.sqrt(1.0 + 4.0 * mu**2))
        mu = (1.0 + t_it) / 2.0
        
        temp = start_temp * (end_temp / start_temp) ** (it / max(1, n_iter))
        grad = smoothed_subgradient(y, n, temp, dx)
        
        x_new = y - base_lr * grad
        x_new = project_target(x_new, n)
        y_new = x_new + (1.0 - 1.0/t_it) * (x_new - y)
        y_new = project_target(y_new, n)
        x = x_new
        y = y_new
        
        if it % 200 == 199:
            c5_test = compute_c5(y, dx)
            if c5_test < c5:
                c5 = c5_test
                x = y.copy()
    
    c5_final = compute_c5(y, dx)
    return y, c5_final

def optimization_phase2(h, n, dx, n_iter, deadline, lower_bound=0.25):
    """Exact subgradient with Polyak step size and averaging."""
    c5 = compute_c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    running_avg = h.copy()
    best_avg_c5 = compute_c5(running_avg, dx)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Polyak step size
        gap = c5 - lower_bound
        if gap < 1e-10:
            gap = 1e-10
        alpha = gap / (np.linalg.norm(exact_subgradient(h, n, k_star))**2 + 1e-10)
        alpha = min(alpha, 0.1)  # cap step size
        
        grad = exact_subgradient(h, n, k_star)
        h_new = h - alpha * grad
        h_new = project_target(h_new, n)
        
        # Update running average
        running_avg = (it * running_avg + h_new) / (it + 1)
        running_avg = project_target(running_avg, n)
        avg_c5 = compute_c5(running_avg, dx)
        if avg_c5 < best_avg_c5:
            best_avg_c5 = avg_c5
            best_h = running_avg.copy()
            best_c5 = avg_c5
        
        h = h_new
    
    return best_h, best_c5

def coordinate_descent(h, n, dx, n_sweeps, deadline):
    """Final coordinate descent refinement."""
    c5 = compute_c5(h, dx)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            h_old = h[j]
            # Try to optimize h[j] while maintaining the constraint
            # We'll pair h[j] with another coordinate to maintain sum
            # Find the best partner
            # Actually, simpler: just try a small perturbation
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_test = h.copy()
                h_test[j] = np.clip(h[j] + delta, 0, 1)
                # Adjust another coordinate to maintain sum
                target = n / 2.0
                new_sum = np.sum(h_test)
                diff = target - new_sum
                if abs(diff) < 1e-10:
                    pass
                else:
                    # Find a good partner to adjust
                    partner = (j + 1) % n
                    h_test[partner] = np.clip(h[partner] + diff, 0, 1)
                    # If still off, distribute among a few
                    remaining = target - np.sum(h_test)
                    if abs(remaining) > 1e-10:
                        for k in range(n):
                            if k != j and k != partner:
                                h_test[k] = np.clip(h_test[k] + remaining / (n-2), 0, 1)
                        break
                if np.sum(h_test) != target:
                    h_test = project_target(h_test, n)
                c5_test = compute_c5(h_test, dx)
                if c5_test < c5 - 1e-10:
                    h = h_test
                    c5 = c5_test
                    improved = True
                    break
        if not improved and sweep > 3:
            break
    
    return h, c5
