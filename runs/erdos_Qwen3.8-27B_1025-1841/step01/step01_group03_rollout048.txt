    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        if diff > 0:
            step = min(step, (1.0 - h[mask]).min())
        else:
            step = max(step, -h[mask].min())
        h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    return h

    return np.correlate(h, 1.0 - h, mode='full')

    corr = get_corr(hc)
    vals = corr * dx
    m = np.max(vals)
    return float(m + np.log(np.exp(beta * (vals - m)).sum()) / beta)


def smooth_grad(h, dx, beta):
    n = len(h)
    hc = np.clip(h, 0.0, 1.0)
    corr = get_corr(hc)
    vals = corr * dx
    m = np.max(vals)
    exps = np.exp(beta * (vals - m))
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += w * (1.0 - hc[lag:])
            if n - lag > 0:
                grad[lag:] -= w * hc[:n - lag]
        else:
            nl = -lag
            grad[nl:] += w * (1.0 - hc[:n - nl])
            grad[:n - nl] -= w * hc[nl:]
