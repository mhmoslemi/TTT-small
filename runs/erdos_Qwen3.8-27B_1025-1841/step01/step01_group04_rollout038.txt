    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / 2.0)
    h = np.clip(h, 0.0, 1.0)
    h = project(h, n)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    start_n = len(best_h)

    # Phase 1: Multi-temperature projected gradient at starting resolution
    n = len(best_h)
    for T in [80.0, 40.0, 20.0, 10.0, 5.0]:
        if time.time() > deadline - 180:
            break
        h_new, c5_new = proj_grad_momentum(best_h, n, T, max_steps=8000, deadline=deadline - 180, lr0=0.4, momentum=0.92)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Hard max polish
    if time.time() < deadline - 150:
        h_new, c5_new = proj_grad_momentum(best_h, n, 1.0, max_steps=5000, deadline=deadline - 150, lr0=0.15, momentum=0.95, use_hard_max=True)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: SLSQP refinement
    if time.time() < deadline - 120:
        h_new = slsqp_refine(best_h, n, 30.0, maxiter=500, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Higher resolution attempt
    if time.time() < deadline - 100:
        n2 = 2 * n
        if n2 <= 2048:
            h_hi = interp_to(best_h, n, n2)
            h_hi = project(h_hi, n2)
            h_new, c5_new = proj_grad_momentum(h_hi, n2, 30.0, max_steps=5000, deadline=deadline - 80, lr0=0.3, momentum=0.92)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                n = n2
            if time.time() < deadline - 60:
                h_new, c5_new = proj_grad_momentum(best_h, n, 10.0, max_steps=3000, deadline=deadline - 60, lr0=0.2, momentum=0.93)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 5: Structural restarts
    n = len(best_h)
    for freq in [2, 3, 4, 5, 6, 8]:
        if time.time() > deadline - 40:
            break
        h_struct = make_structural_init(n, freq)
        h_new, c5_new = proj_grad_momentum(h_struct, n, 20.0, max_steps=3000, deadline=deadline - 40, lr0=0.4, momentum=0.92)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Random perturbation restarts
    for attempt in range(40):
        if time.time() > deadline - 10:
            break
        rng = np.random.RandomState(seed + attempt * 7)
        h_pert = best_h + rng.randn(n) * 0.04
        h_pert = project(h_pert, n)
        h_new, c5_new = proj_grad_momentum(h_pert, n, 15.0, max_steps=2000, deadline=deadline - 5, lr0=0.3, momentum=0.92)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 7: Final hard-max polish
    if time.time() < deadline - 15:
        h_new, c5_new = proj_grad_momentum(best_h, n, 1.0, max_steps=3000, deadline=deadline - 5, lr0=0.12, momentum=0.95, use_hard_max=True)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
