        diff = target - s

        if nfree == 0:

        h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(lags, temp):
    """Smooth approximation of max using log-sum-exp"""
    m = np.max(lags)
    # Shift for numerical stability
    shifted = lags - m
    logsum = np.log(np.sum(np.exp(shifted / temp)))
    return m + temp * logsum

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        nf = np.sum(free)
        if nf == 0:
            break
        h[free] += diff / nf
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return project_target(np.interp(x_new, x_old, h), n_new)

Hmm, but the centered gradient trick is a bit ad-hoc. Let me think about this differently.

The issue is: we want to minimize f(h) = smooth_max(corr(h)) subject to:
- 0 ≤ h[i] ≤ 1
- ∑ h[i] = n/2

The Lagrangian is: L = f(h) - λ(∑h - n/2) + box terms

The gradient step should be: h → h - α(∇f(h) - λ*1)
where λ is chosen so that the step preserves the sum constraint.

If we center the gradient: g_c = g - mean(g), then mean(g_c) = 0, so the step in direction g_c doesn't change the sum (to first order). This is a standard trick.

But after projection (clipping to [0,1] and adjusting sum), the sum is exactly preserved anyway. So the centering is just to make the unconstrained step direction "sum-preserving", which helps the projection step.

Actually, I think a cleaner approach is:
1. Compute gradient g
2. Take step: h_new = h - α * g
3. Project: h_new = project_target(h_new, n)  [clip to [0,1] then adjust sum]
4. Backtracking: if objective decreased, accept; else reduce α

The project_target handles both box and sum constraints. The centering is optional and might help convergence.

Let me also think about whether to use the centered gradient. If we don't center:
- The step h - α*g will generally change the sum
- The projection will fix the sum, but by redistributing the excess
- This redistribution might not be in the right direction

If we center:
- The step preserves the sum to first order
- The projection only needs to handle clipping
- This is cleaner

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
        else:
            free = h > 1e-14
        nf = np.sum(free)
        if nf == 0:
            break
        h[free] += diff / nf
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    return project_target(np.clip(np.interp(x_new, x_old, h), 0, 1), n_new)

def smooth_max_val(c, T):
    cm = np.max(c)
    return cm + T * np.log(np.sum(np.exp((c - cm) / T)))

def smooth_max_weights(c, T):
    cm = np.max(c)
    ec = np.exp((c - cm) / T)
    return ec / np.sum(ec)

def grad_smooth_max(h, n, T):
    c = np.correlate(h, 1.0 - h, mode="full")
    p = smooth_max_weights(c, T)
    q = np.ones(n) - h
    hf = h[::-1]
    corr1 = np.correlate(p, q, mode="full")
    corr2 = np.correlate(p, hf, mode="full")
    g = np.zeros(n)
    for j in range(n):
        g[j] = corr1[n - 1 - j] - corr2[j]
    return g

def optimize_pg(h, n, t0, deadline, max_iters=2000, T_start=0.4, T_end=0.005):
    dx = 2.0 / n
    h = project_target(h.copy(), n)
    c5_curr = compute_c5(h, dx)
    n_temps = 10
    T_sched = np.geomspace(T_start, T_end, n_temps)
    for Ti, T in enumerate(T_sched):
        iters_per_T = max(100, max_iters // n_temps)
        alpha = 4.0 / n
        for it in range(iters_per_T):
            if time.time() > deadline:
                break
            c = np.correlate(h, 1.0 - h, mode="full")
            fval = smooth_max_val(c, T)
            g = grad_smooth_max(h, n, T)
            gc = g - np.mean(g)
            gnorm = np.linalg.norm(gc)
            if gnorm < 1e-10:
                break
            accepted = False
            for a_try in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.125, alpha * 0.0625]:
                h_try = project_target(h - a_try * gc, n)
                c_try = np.correlate(h_try, 1.0 - h_try, mode="full")
                f_try = smooth_max_val(c_try, T)
                if f_try <= fval + 1e-12:
                    h = h_try
                    accepted = True
                    break
            if not accepted:
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            else:
                alpha = min(alpha * 1.05, 8.0 / n)
            if it % 200 == 199:
                alpha *= 0.7
        # Check improvement
        c5_check = compute_c5(h, dx)
        if c5_check < c5_curr:
            c5_curr = c5_check
    return h, c5_curr
