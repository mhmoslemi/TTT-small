```python
def project_h(h, n):
    """Project h onto {0 <= h_i <= 1, sum(h_i) = n/2}."""
    target = n / 2.0
    # Simple approach: iterative
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = np.sum(h)
    diff = target - s
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
```

```python
def project_exact(h, n):
    """Exact projection onto {0 <= h_i <= 1, sum(h_i) = n/2} using water-filling."""
    target = n / 2.0
    # We want to find tau such that sum(clip(h_i - tau, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0.0, 1.0)
```

- Swap h[i] and h[j] (preserves sum)
- Flip a pair: set h[i] = h[i]+δ, h[j] = h[j]-δ (preserves sum)
- Single coordinate: change h[i] and adjust all others proportionally

The swap move is particularly interesting because it preserves the sum constraint exactly.

```python
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    scaled = T * corr * dx
```
