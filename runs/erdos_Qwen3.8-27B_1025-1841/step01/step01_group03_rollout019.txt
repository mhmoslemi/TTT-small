```python
def corr_after_swap(c, h, p, q, n):
    """Update full correlation c after swapping h[p] (1→0) and h[q] (0→1). Returns new c and new h."""
    new_c = c.copy()
    # Lag index: c[j] corresponds to lag ℓ = j - (n-1), j = 0..2n-2
    # For each lag ℓ, compute Δc
    # Δc[ℓ] = -(1-h[p-ℓ]) + (1-h[q-ℓ]) + h[p+ℓ] - h[q+ℓ]
    # Valid only if indices are in [0, n-1]
    
    for ell in range(-(n-1), n):
        j = ell + (n-1)  # index into c
        delta = 0.0
        # First factor change at position p: was 1, now 0
        # Removes (1-h[p-ell]) if p-ell valid
        idx = p - ell
        if 0 <= idx < n:
            delta -= (1.0 - h[idx])
        # First factor change at position q: was 0, now 1
        # Adds (1-h[q-ell]) if q-ell valid
        idx = q - ell
        if 0 <= idx < n:
            delta += (1.0 - h[idx])
        # Second factor: at position i where i-ell = p, i = p+ell
        # Was h[i]*(1-h[p]) = h[i]*0 = 0, now h[i]*(1-0) = h[i]
        idx = p + ell
        if 0 <= idx < n:
            delta += h[idx]
        # Second factor: at position i where i-ell = q, i = q+ell
        # Was h[i]*(1-h[q]) = h[i]*1 = h[i], now h[i]*(1-1) = 0
        idx = q + ell
        if 0 <= idx < n:
            delta -= h[idx]
        new_c[j] += delta
    
    # Update h
    h = h.copy()
    h[p] = 0.0
    h[q] = 1.0
    return new_c, h
```
