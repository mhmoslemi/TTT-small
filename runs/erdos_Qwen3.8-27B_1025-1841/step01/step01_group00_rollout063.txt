def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def fft_correlate(a, b):
    na, nb = len(a), len(b)
    n_out = na + nb - 1
    N = 1
    while N < n_out:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * np.conj(B), N)[:n_out]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = fft_correlate(h, 1.0 - h)
    return c.max() * dx, c

def nesterov_optimize(h, n, tau, max_iters, time_limit, lr=0.1):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_obj, _ = smooth_obj_grad_fft(h, tau)
    best_h = h.copy()
    v = np.zeros(n)
    start = time.time()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad_fft(h, tau)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:

        if stall > 200:

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def fft_correlate(a, b):
    na = len(a)
    nb = len(b)
    N = 1
    while N < na + nb - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return np.concatenate([c[N - na + 1:], c[:na]])


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = fft_correlate(h, 1.0 - h)
    return c.max() * dx, c


def smooth_obj_grad_fft(h, tau):
    n = len(h)
    dx = 2.0 / n
    N_corr = 1
    while N_corr < 2 * n - 1:
        N_corr <<= 1
    N_grad = 1
    while N_grad < 3 * n - 1:
        N_grad <<= 1
    A = np.fft.rfft(h, N_corr)
    B = np.fft.rfft(1.0 - h, N_corr)
    corr = np.fft.irfft(A * np.conj(B), N_corr)[:2 * n - 1]
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    S = exp_c.sum()
    p = exp_c / S
    obj = dx * (max_c + np.log(S) / tau)
    w = p
    v = np.roll(w, -(n - 1))
    N = N_grad
    omh = np.zeros(N)
    omh[:n] = 1.0 - h
    hh = np.zeros(N)
    hh[:n] = h
    ww = np.zeros(N)
    ww[:2 * n - 1] = w
    vv = np.zeros(N)
    vv[:2 * n - 1] = v
    conv1 = np.fft.irfft(np.fft.rfft(omh, N) * np.fft.rfft(ww, N), N)[:n]
    conv2 = np.fft.irfft(np.fft.rfft(hh, N) * np.fft.rfft(vv, N), N)[:n]
    grad = dx * (conv1 - conv2)
    return obj, grad


def nesterov_optimize(h, n, tau, max_iters, time_limit, lr=0.1, momentum=0.9):
    h = project_to_feasible(h.copy(), n)
    best_obj, _ = smooth_obj_grad_fft(h, tau)
    best_h = h.copy()
    v = np.zeros(n)
    h_prev = h.copy()
    start = time.time()
    stall = 0
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad_fft(h, tau)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break
        y = h + momentum * (h - h_prev)
        h_prev = h.copy()
        y -= lr * grad
        h = project_to_feasible(y, n)
        v = h - h_prev
    return best_h, best_obj


def subgrad_optimize(h, n, max_iters, time_limit, lr=0.01):

    h = project_to_feasible(h.copy(), n)
    c5, corr = compute_c5(h)

    np.random.seed(seed)
    h_init = initial_h_values.copy()
    n_init = len(h_init)

    def remaining():
        return max(1.0, budget_s - (time.time() - start_time))

    best_c5, _ = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Smooth gradient at initial resolution
    if remaining() > 120:
        h1 = h_init.copy()
        tl = remaining() * 0.18
        taus = [(15.0, 0.3), (50.0, 0.15), (150.0, 0.05)]
        for tau, lr_init in taus:
            if remaining() < 30:
                break
            h1, _ = nesterov_optimize(h1, n_init, tau, 50000, tl / 3.0, lr=lr_init)
            c5, _ = compute_c5(h1)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h1.copy()

    # Phase 2: Double resolution
    n2 = n_init * 2
    if remaining() > 120:
        h2 = project_to_feasible(interp_periodic(best_h, n2), n2)
        tl = remaining() * 0.2
        taus2 = [(20.0, 0.15), (60.0, 0.08), (200.0, 0.03)]
        for tau, lr_init in taus2:
            if remaining() < 30:
                break
            h2, _ = nesterov_optimize(h2, n2, tau, 40000, tl / 3.0, lr=lr_init)
            c5, _ = compute_c5(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 3: Quadruple resolution
    n3 = n_init * 4
    if remaining() > 100:
        h3 = project_to_feasible(interp_periodic(best_h, n3), n3)
        tl = remaining() * 0.25
        taus3 = [(25.0, 0.1), (80.0, 0.05), (300.0, 0.02)]
        for tau, lr_init in taus3:
            if remaining() < 30:
                break
            h3, _ = nesterov_optimize(h3, n3, tau, 30000, tl / 3.0, lr=lr_init)
            c5, _ = compute_c5(h3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3.copy()

    # Phase 4: Subgradient refinement
    if remaining() > 60:
        h4 = best_h.copy()
        h4, c5_4 = subgrad_optimize(h4, len(h4), 200000, remaining() * 0.2, lr=0.005)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Perturbation restart
    if remaining() > 80:
        n_alt = len(best_h)
        h5 = best_h + np.random.randn(n_alt) * 0.015
        h5 = project_to_feasible(h5, n_alt)
        h5, _ = nesterov_optimize(h5, n_alt, 80.0, 30000, remaining() * 0.15, lr=0.04)
        c5_5, _ = compute_c5(h5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Second perturbation
    if remaining() > 50:
        n_alt2 = len(best_h)
        h6 = best_h + np.random.randn(n_alt2) * 0.008
        h6 = project_to_feasible(h6, n_alt2)
        h6, _ = nesterov_optimize(h6, n_alt2, 200.0, 20000, remaining() * 0.12, lr=0.02)
        c5_6, _ = compute_c5(h6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Final subgradient polish
    if remaining() > 20:
        hf = best_h.copy()
        hf, c5_f = subgrad_optimize(hf, len(hf), 100000, remaining() * 0.35, lr=0.002)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5, _ = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
