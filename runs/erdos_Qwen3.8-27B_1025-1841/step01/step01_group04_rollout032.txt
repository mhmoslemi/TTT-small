def adam_step(params, grad, m, v, t, lr, beta1, beta2, eps):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    params = params - lr * m_hat / (np.sqrt(v_hat) + eps)

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    n = len(h)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    W = exp_scaled / np.sum(exp_scaled)
    obj = (m + np.log(np.sum(exp_scaled))) / T * dx
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = W[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        # First term: (1-h[j-k]) for j-k in [0, n-1]
        lo1 = max(0, k)
        hi1 = min(n - 1, k + n - 1)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            grad[js] += w * (1.0 - h[js - k])
        # Second term: -h[j+k] for j+k in [0, n-1]
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            js2 = np.arange(lo2, hi2 + 1)
            grad[js2] -= w * h[js2 + k]
    grad *= dx
    return obj, grad

def adam_optimize(h, T, n_steps, lr, deadline):

    v_adam = np.zeros(n)

        obj, grad = smooth_obj_and_grad(h_cur, T)
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad * grad)
        m_hat = m_adam / (1.0 - beta1 ** t_adam)

        h_new = h_cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_constraint(h_new, target)
        h_cur = h_new

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    W = exp_scaled / np.sum(exp_scaled)
    obj = (m + np.log(np.sum(exp_scaled))) / T * dx
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = W[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n - 1, k + n - 1)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            grad[js] += w * (1.0 - h[js - k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            js2 = np.arange(lo2, hi2 + 1)
            grad[js2] -= w * h[js2 + k]
    grad *= dx
    return obj, grad

def adam_optimize(h, T, n_steps, lr, deadline):
    n = len(h)
    target = n / 2.0
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    h_cur = h.copy()
    for it in range(n_steps):
        if time.time() > deadline:
            break
        obj, grad = smooth_obj_and_grad(h_cur, T)
        t_adam += 1
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad * grad)
        m_hat = m_adam / (1.0 - beta1 ** t_adam)
        v_hat = v_adam / (1.0 - beta2 ** t_adam)
        h_new = h_cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_constraint(h_new, target)
        h_cur = h_new
    return h_cur

def dual_symmetry(h, n):
    return 1.0 - h[(np.arange(n) + n // 2) % n]

    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()

        h_start = np.full(336, 0.5)

    n = len(h_start)
    target = n / 2.0
    best_h = project_to_constraint(h_start, target)
    best_c5 = evaluate_c5(best_h)

    def try_update(h_candidate):

        c5_val = evaluate_c5(h_candidate)

        hd = dual_symmetry(h_candidate, n)
