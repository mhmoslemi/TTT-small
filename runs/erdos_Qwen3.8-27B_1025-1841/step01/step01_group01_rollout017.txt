            h_r, c5_r = optimize_at_n(h_rand, best_n, [500.0, 5000.0], max_iter=200)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h, best_n)

    if final_c5 > 0.5:
        final_h = np.full(best_n, 0.5)
        final_c5 = 0.5

    return final_h.tolist(), float(final_c5), best_n
