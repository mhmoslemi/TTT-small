    m is the lag index in the 'full' output (0 to 2n-2).

        max_val = np.max(corr)

`c_k = sum_n a_{n+k} * conj(v_n)`

`c_k = sum_n h_{n+k} * (1-h_n)` for valid n, k

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_fast(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) < 1e-12:
        return h
    if d > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_offset(h, offset):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, offset)
    hi1 = min(n, offset + n)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[np.arange(lo1, hi1) - offset])
    lo2 = max(0, -offset)
    hi2 = min(n, n - offset)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[np.arange(lo2, hi2) + offset]
    return grad

def subgrad_phase(h, n, n_steps, lr_start, lr_end, top_k, tw_start, tw_end, deadline, momentum):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    eval_every = max(1000, n_steps // 50)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        t_frac = float(step) / max(1, n_steps)
        lr = lr_start * (lr_end / lr_start) ** t_frac
        tw = tw_start + (tw_end - tw_start) * t_frac

        corr = compute_corr(h)
        max_val = corr.max()
        if top_k >= len(corr):
            top_idx = np.array(range(len(corr)))
            w = np.ones(len(corr)) / len(corr)
        else:
            top_idx = np.argsort(corr)[::-1][:top_k]
            w = np.exp((corr[top_idx] - max_val) * tw)
            w /= w.sum()

        grad = np.zeros(n)
        for j in range(top_k):
            m = int(top_idx[j])
            offset = m - (n - 1)
            grad += w[j] * grad_for_offset(h, offset)

        v = momentum * v + grad
        h = h - lr * v
        h = project_fast(h, n)

        if (step + 1) % eval_every == 0 or step == n_steps - 1:
            val = evaluate_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val

def slsqp_polish(h, n, deadline):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return float((m + np.log(np.sum(np.exp(corr - m)))) * dx)
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-10, 'maxiter': 120, 'disp': False})
        if time.time() > deadline:
            return h
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h

def coord_descent_pass(h, n, deadline, n_passes=50):
    h = h.copy()
    best_val = evaluate_c5(h)
    for p in range(n_passes):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        max_val = corr.max()
        worst_lag = int(np.argmax(corr))
        offset = worst_lag - (n - 1)
        grad_w = grad_for_offset(h, offset)
        order = np.argsort(np.abs(grad_w))[::-1][:30]
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            g = grad_w[i]
            if abs(g) < 1e-8:
                continue
            if g > 0:
                trial = h.copy()
                trial[i] = max(0.0, h[i] - min(0.1, h[i]))
            else:
                trial = h.copy()
                trial[i] = min(1.0, h[i] + min(0.1, 1.0 - h[i]))
            trial = project_fast(trial, n)
            c_new = evaluate_c5(trial)
            if c_new < best_val - 1e-10:
                h = trial
                best_val = c_new
                improved = True
        if not improved:
            break
    return h, best_val
