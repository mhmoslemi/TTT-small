import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 for a binary (or continuous) vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, corr * dx

def evaluate_erdos(h):
    """Evaluate C5 for given h."""
    c5, _ = compute_c5(h)
    return c5

def random_binary(n, seed=42):
    """Generate random binary vector with n/2 ones."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1
    return h

def hill_climb(h, max_iter=10000, time_limit=100):
    """Local search by swapping pairs of 1s and 0s."""
    n = len(h)
    h = h.copy()
    current_c5 = evaluate_erdos(h)
    np_ones = np.where(h == 1)[0]
    np_zeros = np.where(h == 0)[0]
    start_time = time.time()
    
    improved = True
    while improved and time.time() - start_time < time_limit:
        improved = False
        best_c5 = current_c5
        best_swap = None
        for i in range(len(np_ones)):
            for j in range(len(np_zeros)):
                a, b = np_ones[i], np_zeros[j]
                h[a] = 0
                h[b] = 1
                c5 = evaluate_erdos(h)
                h[a] = 1
                h[b] = 0
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_swap = (a, b)
        if best_swap is not None:
            a, b = best_swap
            h[a] = 0
            h[b] = 1
            current_c5 = best_c5
            np_ones = np.where(h == 1)[0]
            np_zeros = np.where(h == 0)[0]
            improved = True
    return h, current_c5

def simulated_annealing(h, n_iter=50000, T_start=1.0, T_end=0.01, time_limit=200):
    """Simulated annealing for the binary swap problem."""
    n = len(h)
    h = h.copy()
    current_c5 = evaluate_erdos(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(0)
    start_time = time.time()
    
    for it in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        T = T_start * (T_end / T_start) ** (it / n_iter)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = rng.choice(ones)
        b = rng.choice(zeros)
        h[a] = 0
        h[b] = 1
        new_c5 = evaluate_erdos(h)
        delta = new_c5 - current_c5

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = 1
            h[b] = 0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def hill_climb_best(h, time_limit=50):
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    start = time.time()
    improved = True
    while improved and time.time() - start < time_limit:
        improved = False
        best_c5 = current_c5
        best_move = None
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for a in ones:
            for b in zeros:
                h[a] = 0.0
                h[b] = 1.0
                c5 = compute_c5(h)
                h[a] = 1.0
                h[b] = 0.0
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best_move = (a, b)
        if best_move is not None:
            a, b = best_move
            h[a] = 0.0
            h[b] = 1.0
            current_c5 = best_c5
            improved = True
    return h, current_c5

def simulated_anneal(h, time_limit=100, seed=0):
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    n_iter = 0
    while time.time() - start < time_limit:
        n_iter += 1
        T = max(0.01, 1.0 * np.exp(-n_iter / 5000))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        a = rng.choice(ones)
        b = rng.choice(zeros)
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
    return best_h, best_c5

    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
