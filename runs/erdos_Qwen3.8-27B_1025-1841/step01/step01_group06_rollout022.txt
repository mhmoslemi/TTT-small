import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-15):
            h = h_new
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-12:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def smooth_max_obj(x, dx, temp):
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -500, 500)
    logsum = np.log(np.sum(np.exp(shifted)))
    return (m + temp * logsum) * dx

def smooth_max_grad(x, dx, temp):
    n = len(x)
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -500, 500)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    grad = np.zeros(n)
    sig = np.where(weights > 1e-12)[0]
    for idx in sig:
        w = weights[idx] * dx
        s = idx - (n - 1)
        if s >= 0:
            m_ = n - s
            grad[s:s+m_] += w * (1.0 - hx[:m_])
            grad[:m_] += w * (-hx[s:s+m_])
        else:
            m_ = -s
            grad[m_:] += w * (1.0 - hx[:n-m_])
            grad[:n-m_] += w * (-hx[m_:])
    return grad

def binary_push(h, n):
    target = int(n / 2.0)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    s = int(np.sum(h_bin))
    if s == target:
        return h_bin
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    if s > target:
        for idx in order:
            if h_bin[idx] == 1.0:
                h_bin[idx] = 0.0
                s -= 1
                if s == target:
                    break
    else:
        for idx in order:
            if h_bin[idx] == 0.0:
                h_bin[idx] = 1.0
                s += 1
                if s == target:
                    break
    return h_bin

def do_l_bfgs(h, n, dx, temps, max_iters, time_limit):
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr, dx)
    bnds = [(0.0, 1.0)] * n
    for temp in temps:
        if time.time() > time_limit:
            break
        try:
            res = minimize(smooth_max_obj, h_curr, jac=smooth_max_grad,
                          args=(dx, temp), method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': max_iters, 'ftol': 1e-15,
                                   'gtol': 1e-10, 'maxls': 30})
            h_new = project_target(res.x, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5_curr - 1e-14:
                h_curr = h_new
                c5_curr = c5_new
        except Exception:
            pass
    return h_curr, c5_curr

    deadline = t0 + budget_s * 0.93
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)

        dx = 2.0 / n

        else:

            best_n = n

        # Temperature schedule based on resolution
