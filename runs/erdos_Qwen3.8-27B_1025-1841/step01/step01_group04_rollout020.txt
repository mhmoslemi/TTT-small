    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def slbfgs_opt(h0, n, T, deadline, maxiter=500):
    """L-BFGS-B optimization with soft-max objective."""
    target = n / 2.0
    
    def objective_and_grad(x):
        # Soft-max objective
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        scaled = corr * T
        m = np.max(scaled)
        obj = (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx
        
        # Numerical gradient (central differences)
        eps = 1e-7
        grad = np.zeros(n)
        for i in range(n):
            xp = x.copy()
            xm = x.copy()
            xp[i] = min(1.0, x[i] + eps)
            xm[i] = max(0.0, x[i] - eps)
            denom = xp[i] - xm[i]
            if denom < 1e-14:
                grad[i] = 0.0
            else:
                cp = np.correlate(xp, 1.0 - xp, mode='full')
                cm = np.correlate(xm, 1.0 - xm, mode='full')
                sp = cp * T
                sm_ = cm * T
                mp = np.max(sp)
                mm = np.max(sm_)
                fp = (mp + np.log(np.sum(np.exp(sp - mp)))) / T * dx
                fm = (mm + np.log(np.sum(np.exp(sm_ - mm)))) / T * dx
                grad[i] = (fp - fm) / denom
        
        # Add penalty for sum constraint
        sum_dev = np.sum(x) - target
        obj += 1e6 * sum_dev ** 2
        grad += 2e6 * sum_dev
        
        return obj, grad
    
    try:
        res = minimize(objective_and_grad, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 2000})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0
