def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum, 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the adjustment
    if diff > 0:
        # Need to increase sum: add to elements that are < 1
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room < diff:
            return np.clip(h + diff / len(h), 0, 1)
        h = h + diff * room / total_room
    else:
        # Need to decrease sum: subtract from elements that are > 0
        room = h.copy()
        total_room = np.sum(room)

            return np.clip(h + diff / len(h), 0, 1)

import numpy as np
import time
from scipy.optimize import minimize, differential_evolution
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
    else:
        room = h.copy()
    tr = np.sum(room)
    if tr < 1e-12:
        return h
    h = h + diff * room / tr
    return np.clip(h, 0.0, 1.0)

def make_h_interp(params, k, n):
    widths = list(params[:k - 1])
    widths.append(1.0 - sum(widths))
    gaps = list(params[k - 1:2 * k - 1])
    gaps.append(1.0 - sum(gaps))
    if any(w <= 1e-6 for w in widths) or any(g <= 0 for g in gaps):
        return None
    x = np.arange(n) * (2.0 / n)
    h = np.zeros(n)
    pos = 0.0
    for j in range(k):
        a = pos + gaps[j]
        b = a + widths[j]
        h[x >= a - 1e-9] = 1.0
        h[x >= b - 1e-9] = 0.0
        pos = b
    return h

def interval_obj(params, k, n):
    h = make_h_interp(params, k, n)
    if h is None:
        return 1.0
    c5 = compute_c5(h)
    pen = 0.5 * abs(np.sum(h) - n / 2.0) / n
    return c5 + pen

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(c)
    sh = alpha * (c - cm)
    log_s = np.log(np.sum(np.exp(sh)))
    return dx * (cm + log_s / alpha)
