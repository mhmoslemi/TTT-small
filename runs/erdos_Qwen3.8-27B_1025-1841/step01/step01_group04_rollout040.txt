def project_sum(h, n):
    """Project h to satisfy sum(h) = n/2, 0 <= h <= 1"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_1d(output, h, i, n, n_eval=8):
    """Solve 1D subproblem for coordinate i using coordinate descent."""
    m = 2 * n - 1
    s = np.arange(m) - (n - 1)
    valid1 = (i - s >= 0) & (i - s < n)
    valid2 = (i + s >= 0) & (i + s < n)
    
    slope = np.zeros(m)
    slope[valid1] = 1.0 - h[i - s[valid1]]
    slope[valid2] -= h[i + s[valid2]]
    
    intercept = output - h[i] * slope
    
    # Minimize f(t) = max(intercept + slope * t) over t in [0, 1]
    # Evaluate at several points
    ts = np.linspace(0, 1, n_eval)
    vals = intercept[None, :] + slope[None, :] * ts[:, None]
    f_vals = np.max(vals, axis=1)
    
    best_idx = np.argmin(f_vals)
    t_best = ts[best_idx]
    
    # Refine with bisection around best
    lo = max(0.0, ts[max(0, best_idx-1)])
    hi = min(1.0, ts[min(n_eval-1, best_idx+1)])
    
    # Bisection on subgradient
    for _ in range(20):
        mid = (lo + hi) / 2
        # Subgradient at mid: slopes of active constraints
        active = intercept + slope * mid
        max_val = np.max(active)

            hi = mid
        elif np.max(sl) < 0:
            lo = mid
        else:
            break
    
    t_opt = (lo + hi) / 2
    return t_opt

    output = np.correlate(h, 1.0 - h, mode='full')

        idx = np.random.permutation(n)
        for i in idx:
            t_opt = coord_descent_1d(output, h, i, n, n_eval)
            # Update output incrementally
            s = np.arange(m) - (n - 1)
            delta = t_opt - h[i]
            if abs(delta) > 1e-15:
                valid1 = (i - s >= 0) & (i - s < n)
                valid2 = (i + s >= 0) & (i + s < n)
                doutput = np.zeros(m)
                doutput[valid1] = delta * (1.0 - h[i - s[valid1]])
                doutput[valid2] += delta * (-h[i + s[valid2]])
                output += doutput
                h[i] = t_opt
    
    return h, output

def slsqp_opt(h0, n, T, maxiter=300, deadline=None):
    """SLSQP optimization with softmax smooth max."""
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        scaled = corr * T
        mx = np.max(scaled)
        return float((mx + np.log(np.sum(np.exp(scaled - mx)))) / T * dx)
    
    def c_eq(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0

def active_set_slsqp(h0, n, top_k=10, T=50.0, maxiter=200, deadline=None):

    dx = 2.0 / n

def solve_1d_minimax(a, b, n_iter=10):
    """Solve min_t max_k (a_k + b_k * t) for t in [0,1]."""
    # Initial candidates
    t0 = 0.0
    t1 = 1.0
    f0 = np.max(a)
    f1 = np.max(a + b)
    
    best_t = t0 if f0 <= f1 else t1
    best_f = min(f0, f1)
    
    # Iterative: find the two active lines and their intersection
    k_upper = np.argmax(a)  # active at t=0
    k_lower = np.argmax(a + b)  # active at t=1
    
    for _ in range(n_iter):
        if k_upper == k_lower:
            break
        if b[k_upper] == b[k_lower]:
            break
        t_cross = (a[k_lower] - a[k_upper]) / (b[k_upper] - b[k_lower])
        if t_cross < 0 or t_cross > 1:
            break
        # Check if t_cross is feasible
        vals = a + b * t_cross
        max_val = np.max(vals)
        # Find all lines that achieve the max at t_cross
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        if len(active) >= 2:
            # Check if the intersection is valid
            slopes_at_cross = b[active]
            if np.min(slopes_at_cross) <= 0 <= np.max(slopes_at_cross):
                new_f = max_val
                if new_f < best_f - 1e-14:
                    best_f = new_f
                    best_t = t_cross
                break
        else:
            # Only one active line, update
            k = active[0]
            if b[k] > 0:
                # Function increasing, min is to the left
                k_upper = k
            else:
                k_lower = k
    
    return best_t

                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_cd2.copy()
                    h_hi = h_cd2.copy()
        # SLSQP at higher res
        if time.time() < deadline - 40:
            for T in [200.0, 600.0]:
                if time.time() > deadline - 20:
                    break
                h_new = slsqp_opt(best_h.copy(), len(best_h), T, maxiter=150, deadline=deadline)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            # Final CD
            for _ in range(10):
                if time.time() > deadline - 5:
                    break
                h_cd = coord_descent_step(best_h, len(best_h), rng, deadline)
                c5_cd = evaluate_c5(h_cd)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                else:
                    break

    # Phase 5: Final polish at best resolution
    n_final = len(best_h)
    for _ in range(50):
        if time.time() > deadline - 3:
            break
        h_cd = coord_descent_step(best_h, n_final, rng, deadline)
        c5_cd = evaluate_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        else:
            break

    best_h = project_sum(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
