import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_gradient(h, T):
    """Compute gradient of (1/T)*log(sum(exp(T*corr)))*dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    corr_shifted = corr - corr.max()  # for numerical stability
    w = np.exp(T * corr_shifted)
    w /= w.sum()
    # Gradient via correlate
    first = np.correlate(w, 1.0 - h, mode='valid')
    second = np.correlate(w[::-1], h, mode='valid')[::-1]
    grad = dx * (first - second)
    return grad, corr.max() * dx

def projected_grad_descent(h, T_start, T_end, n_iter, lr_start, lr_end, time_limit):
    """Projected gradient descent with annealing temperature."""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    t0 = time.time()
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        # Anneal temperature

def coordinate_descent_polish(h, n_max_iter=20):

    n = len(h)

from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    shifted = corr - cmax
    w = np.exp(T * shifted)
    w /= w.sum()
    a = 1.0 - h
    b = h
    grad = np.zeros(n)
    for i in range(n):
        kmax1 = n - i
        if kmax1 > 0:
            grad[i] += np.dot(w[:kmax1], a[i:i+kmax1])
        k2lo = max(0, i - n + 1)
        klen = i - k2lo + 1
        if klen > 0:
            grad[i] -= np.dot(w[k2lo:k2lo+klen], b[i-(klen-1):i+1])
    grad *= dx
    return grad

    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    t0 = time.time()
    n = n_pts
    prev_c5 = best_c5

        if time.time() - t0 > time_limit:

        T = T_start + (T_end - T_start) * frac
        lr = lr_start * (lr_end / lr_start) ** frac
        grad = smooth_grad(h_cur, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Adaptive step with Armijo-like check

            step = lr * trial_scale

            h_new = make_feasible(h_new, n)
            c5_try = compute_c5(h_new)
            if c5_try < best_c5 + 1e-14:

                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
                break

            stall += 3

def _sl_obj(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    shifted = corr - cmax
    soft = cmax + np.log(np.sum(np.exp(_SL_T * shifted))) / _SL_T
    return soft * dx

def _sl_eq(h):
    return np.sum(h) - _SL_N / 2.0

def _sl_jac(h):
    return np.ones(_SL_N)

def slsqp_polish(h, T, maxiter=300):

    _SL_N = len(h)
    _SL_T = T
    bounds = [(0.0, 1.0)] * _SL_N

    try:
        res = minimize(_sl_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x
    except Exception:
