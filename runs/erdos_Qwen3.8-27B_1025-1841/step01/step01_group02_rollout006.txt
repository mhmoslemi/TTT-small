    exponents = T * (corr - mc)
    exponents = np.clip(exponents, -500, 0)
    ew = np.exp(exponents)
    ws = ew.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = ew / ws
    obj = (mc + np.log(max(ws, 1e-300)) / T) * dx
    one_minus_h = 1.0 - h
    part1 = (w[idx] * one_minus_h[None, :]).sum(axis=1)
    part2 = (w[idx.T] * h[None, :]).sum(axis=1)
    grad = dx * (part1 - part2)
    return obj, grad

def proj_grad_optimize(h, n, idx, start_time, budget_s, reserve_s=50):
    deadline = start_time + budget_s - reserve_s
    T_schedule = [8.0, 15.0, 25.0, 40.0, 60.0, 90.0, 130.0, 200.0, 320.0, 500.0]
    obj, grad = softmax_obj_grad(h, T_schedule[0], idx)
    best_local_c5 = compute_c5(h)
    best_local_h = h.copy()
    for pi, T in enumerate(T_schedule):
        if time.time() > deadline:
            break
        alpha = 2.0 / (1.0 + T * 0.04)
        stagnant = 0
        while time.time() < deadline:
            h_cand = make_feasible(h - alpha * grad, n)
            obj_cand, grad_cand = softmax_obj_grad(h_cand, T, idx)
            if obj_cand < obj - 1e-16:
                h = h_cand
                obj = obj_cand
                grad = grad_cand
                stagnant = 0
                if alpha < 10.0:
                    alpha *= 1.02
            else:
                alpha *= 0.5
                h_cand = make_feasible(h - alpha * grad, n)
                obj_cand, grad_cand = softmax_obj_grad(h_cand, T, idx)
                if obj_cand < obj - 1e-16:
                    h = h_cand
                    obj = obj_cand
                    grad = grad_cand
                    stagnant = 0
                else:
                    stalegrad = alpha < 0.01
                    if stalegrad:
                        break
            c5 = compute_c5(h)
            if c5 < best_local_c5:
                best_local_c5 = c5
                best_local_h = h.copy()
        obj, grad = softmax_obj_grad(h, T, idx)
    return best_local_h, best_local_c5

def coord_descent_polish(h, n, start_time, budget_s, reserve_s=15, max_rounds=25):
    deadline = start_time + budget_s - reserve_s
    best_c5 = compute_c5(h)
    deltas_coarse = [0.03, -0.03, 0.06, -0.06, 0.015, -0.015]
    deltas_fine = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        deltas = deltas_coarse if rnd < 10 else deltas_fine
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            for d in deltas:
                c = old + d
                if c < 0 or c > 1:
                    continue
                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    h[:] = trial
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        h0 = (rng.rand(320) < 0.5).astype(float)
        h0 = make_feasible(h0, 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    for n in [640, 800, 1000]:
        if time.time() - start_time > budget_s - 120:
            break
        idx = build_idx(n)
        h_work = upsample_h(best_h, n)
        h_opt, c5_opt = proj_grad_optimize(h_work, n, idx, start_time, budget_s, reserve_s=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_pol, c5_pol = coord_descent_polish(h_opt, n, start_time, budget_s, reserve_s=40, max_rounds=12)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

        if time.time() - start_time < budget_s - 100:
            h_rand = rng.rand(n)
            h_rand = make_feasible(h_rand, n)
            h_opt2, c5_opt2 = proj_grad_optimize(h_rand, n, idx, start_time, budget_s, reserve_s=60)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
            h_pol2, c5_pol2 = coord_descent_polish(h_opt2, n, start_time, budget_s, reserve_s=30, max_rounds=8)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()

    n_f = len(best_h)
    best_h, best_c5 = coord_descent_polish(best_h, n_f, start_time, budget_s, reserve_s=10, max_rounds=30)
    best_h = make_feasible(best_h, n_f)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_f
