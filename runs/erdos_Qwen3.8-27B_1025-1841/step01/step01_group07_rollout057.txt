import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_constraint(h, target):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = int(m.sum())
        if c == 0:
            break
        h[m] += diff / c
        h = np.clip(h, 0.0, 1.0)
    return h


class ObjExact:
    def __init__(self, n):
        self.n = n
        self.tgt = n / 2.0
        self.dx = 2.0 / n

    def __call__(self, x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode="full")
        return float(c.max()) * self.dx + 300.0 * (x.sum() - self.tgt) ** 2


class ObjSoft:
    def __init__(self, n, tau=0.004):
        self.n = n
        self.tau = tau
        self.tgt = n / 2.0
        self.dx = 2.0 / n

    def __call__(self, x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode="full")
        m = float(c.max())
        sm = m + self.tau * float(np.log(np.exp((c - m) / self.tau).sum()))
        return sm * self.dx + 300.0 * (x.sum() - self.tgt) ** 2


def interp_n(h, n):
    ns = len(h)
    if ns == n:
        return h.copy()
    xs = (np.arange(ns) + 0.5) / ns
    xd = (np.arange(n) + 0.5) / n
    return np.interp(xd, xs, h)

    step = 0.05

        h_new = h.copy()

        h_new = project_to_constraint(h_new, tgt)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    return h, c5_cur
