import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_to_n(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        sh = k - (n - 1)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            g[j0:j1] += p[k] * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            g[j0b:j1b] -= p[k] * h[idx + sh]
    g *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, g

def topk_subgrad(h, k=10):
    """Subgradient using top-k worst lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = np.argsort(corr)[-k:]
    g = np.zeros(n)
    for k_idx in topk:
        sh = k_idx - (n - 1)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            g[j0:j1] += (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            g[j0b:j1b] -= h[idx + sh]
    g *= dx / k
    c5 = corr.max() * dx
    return c5, g

    if rem() > 120:
        h1 = interp_to_n(best_h, n1)
        h1 = project_to_feasible(h1, n1)
        tl = min(rem() * 0.25, 350.0)
        h1 = optimize_smooth(h1, n1, tau=80, time_limit=tl, lr=0.15)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: n=5000, smooth max tau=100
    n2 = 5000
    if rem() > 120:
        h2 = interp_to_n(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(rem() * 0.25, 350.0)
        h2 = optimize_smooth(h2, n2, tau=100, time_limit=tl, lr=0.1)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Top-k subgradient at best resolution
    if rem() > 60:
        nf = len(best_h)
        hf = best_h.copy()
        hf, c5_f = optimize_topk(hf, nf, time_limit=min(rem() * 0.2, 200.0), lr=0.003, k_top=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()

    # Phase 4: Binary local search
    if rem() > 40:
        nb = len(best_h)
        hb = best_h.copy()
        hb, c5_b = binary_local_search(hb, nb, time_limit=min(rem() * 0.25, 250.0))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = hb.copy()

    # Phase 5: Second smooth pass from binary solution at same n
    if rem() > 40:
        ns = len(best_h)
        hs = best_h.copy()
        hs = project_to_feasible(hs, ns)
        hs = optimize_smooth(hs, ns, tau=120, time_limit=min(rem() * 0.2, 150.0), lr=0.05)
        c5_s = compute_c5(hs)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = hs.copy()

    # Phase 6: Final top-k polish
    if rem() > 20:
        nn = len(best_h)
        hn = best_h.copy()
        hn, c5_n = optimize_topk(hn, nn, time_limit=min(rem() * 0.3, 100.0), lr=0.002, k_top=5)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = hn.copy()

    # Final binary search squeeze
    if rem() > 15:
        nf2 = len(best_h)
        hf2 = best_h.copy()
        hf2, c5_f2 = binary_local_search(hf2, nf2, time_limit=min(rem() * 0.4, 80.0))
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = hf2.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
