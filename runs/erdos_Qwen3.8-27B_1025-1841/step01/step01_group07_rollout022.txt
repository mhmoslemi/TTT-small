def coordinate_descent(h, dx, max_iters=200):
    """Simple coordinate descent: adjust one coordinate at a time, maintaining sum."""
    n = len(h)
    target = np.sum(h)
    c5 = compute_c5(h, dx)
    
    for _ in range(max_iters):
        improved = False
        for i in range(n):
            # Try adjusting h[i] by trying different values
            # To maintain sum, we adjust another coordinate
            # Simple approach: try h[i] + delta, and distribute -delta to others
            orig = h[i]
            for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
                new_val = orig + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                # Redistribute to maintain sum
                diff = -delta
                if delta > 0:
                    mask = h > 0  # can decrease
                else:
                    mask = h < 1  # can increase
                mask[i] = False
                if not np.any(mask):
                    h[i] = orig
                    continue
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
                new_c5 = compute_c5(h, dx)
                if new_c5 < c5 - 1e-12:
                    c5 = new_c5
                    improved = True
                    break
                else:
                    h[i] = orig
        if not improved:
            break
    return h, c5

import numpy as np
from scipy.optimize import minimize

_params = {}

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def c5_and_grad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * dx
    k_star = np.argmax(corr)
    s = n - 1 - k_star
    j = np.arange(n)
    grad = np.ones(n) * dx
    jm = j - s
    mask = (jm >= 0) & (jm < n)
    grad[mask] -= h[jm[mask]]
    jp = j + s
    mask = (jp >= 0) & (jp < n)
    grad[mask] -= h[jp[mask]]
    return c5, grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    return compute_c5(h, _params['dx'])

def _obj_grad(h):
    return c5_and_grad(h, _params['dx'])[1]

    return np.sum(h) - _params['target']

    return np.ones(len(h))
