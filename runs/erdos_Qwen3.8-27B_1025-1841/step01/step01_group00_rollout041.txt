    # Precompute S[k] = dot(h0[:n-k], h0[k:])
    S = np.zeros(n)
    for k in range(n):
        S[k] = np.dot(h0[:n-k], h0[k:])
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1
    ]
    
    for k in range(n):
        # Gradient of S_k at h0
        g = np.zeros(n)
        if k == 0:
            g = 2.0 * h0
        else:
            g[:n-k] += h0[k:]
            g[k:] += h0[:n-k]
        
        dot_g_h0 = float(np.dot(g, h0))
        rhs_const = dx * (dot_g_h0 - S[k])
        
        if k == 0:
            # A_0 = B_0 = sum(h) = n/2 (fixed by constraint)
            # t >= dx * (n/2 - S_0(h))
            # Linearized: t >= dx * (n/2 - S[0] - g·(h-h0))
            # t + dx * g·h >= dx * (n/2 - S[0] + dot_g_h0)
            rhs = dx * (n / 2.0 - S[0] + dot_g_h0)
            constraints.append(t_var + dx * (g @ h_var) >= rhs)
        else:
            # Constraint A: t >= dx * (A_k(h) - S_k(h))
            # A_k(h) = sum_{i=0}^{n-1-k} h[i]
            # Linearized: t + dx*(g - eA)·h >= rhs_const
            # where eA[i] = 1 if i < n-k
            eA = np.zeros(n)
            eA[:n-k] = 1.0
            constraints.append(t_var + dx * ((g - eA) @ h_var) >= rhs_const)
            
            # Constraint B: t >= dx * (B_k(h) - S_k(h))
            # B_k(h) = sum_{i=k}^{n-1} h[i]
            eB = np.zeros(n)
            eB[k:] = 1.0
            constraints.append(t_var + dx * ((g - eB) @ h_var) >= rhs_const)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=cp.GLOP)
    
    h_val = np.array(h_var.value).flatten()
    t_val = float(t_var.value)
    
    if h_val is None or np.any(np.isnan(h_val)):
        return None, np.inf
    
    # Damped step
    if alpha < 1.0:
        h_val = (1.0 - alpha) * h0 + alpha * h_val
    
    return h_val, t_val

def smooth_gradient_step(h, n, tau, lr):
    """One smooth gradient step for polishing."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    
    # Softmax weights
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-16:
            continue
        sh = k - n + 1
        # h[i] * (1-h[i+sh]) contributes to correlation k
        # d/dh[j] of corr[k] involves h[j] and h[j-sh]
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += p[k] * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= p[k] * h[idx + sh]
    grad *= dx
    
    return grad

        h3, c5_3 = smooth_refine(best_h, n3, time_limit=min(120, remaining() * 0.2), initial_lr=0.04)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: SLP at 4x resolution
    n4 = 4 * n_init
    if remaining() > 90:
        h4 = project_to_feasible(interp_to_n(best_h, n4), n4)
        h4, c5_4 = slp_refine(h4, n4, max_iters=15, time_limit=min(200, remaining() * 0.2))
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Smooth polish
    if remaining() > 30:
        n5 = len(best_h)
        h5, c5_5 = smooth_refine(best_h, n5, time_limit=min(150, remaining() * 0.25), initial_lr=0.02)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: One more SLP round at final resolution
    if remaining() > 40:
        n6 = len(best_h)
        h6, c5_6 = slp_refine(best_h, n6, max_iters=10, time_limit=min(100, remaining() * 0.25))
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Final smooth polish
    if remaining() > 15:
        n7 = len(best_h)
        h7, c5_7 = smooth_refine(best_h, n7, time_limit=remaining() * 0.4, initial_lr=0.01)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5_direct(best_h)

    return best_h, final_c5, len(best_h)
