def fft_correlation(h, g, n):
    """Compute full linear cross-correlation of h and g (both length n) using FFT."""
    N = 1  # find next power of 2 >= 2n-1
    while N < 2*n - 1:
        N *= 2
    h_pad = np.zeros(N)
    g_pad = np.zeros(N)
    h_pad[:n] = h
    g_pad[:n] = g
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(g_pad)
    corr = np.fft.irfft(H * np.conj(G))
    return corr[:2*n-1]

    # Compute correlation via FFT
    N = 1
    while N < 2*n - 1:
        N *= 2
    h_pad = np.zeros(N)
    h_pad[:n] = h
    g_pad = np.zeros(N)
    g_pad[:n] = 1.0 - h
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(g_pad)
    corr = np.fft.irfft(H * np.conj(G))[:2*n-1]
    
    # Compute smooth max
    c_scaled = corr * dx
    max_c = c_scaled.max()
    shifted = tau * (c_scaled - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    # Compute gradient
    # first_term: cross-correlation of (1-h) and p
    N2 = 1
    while N2 < n + 2*n - 1:  # 3n - 2
        N2 *= 2
    a_pad = np.zeros(N2)

    b_pad = np.zeros(N2)
    b_pad[:2*n-1] = p
    A = np.fft.rfft(a_pad)
    B = np.fft.rfft(b_pad)
    cc = np.fft.irfft(np.conj(A) * B)
    first_term = cc[n-1::-1]  # length n
    
    # second_term: h * (prefix_p[n:2n] - prefix_p[0:2n:2])
    prefix_p = np.zeros(2*n)
    prefix_p[1:] = np.cumsum(p)
    second_term = h * (prefix_p[n:2*n] - prefix_p[0:2*n:2])
    
    grad = dx * (first_term - second_term)
    
    return obj, grad, corr

import numpy as np
import time

def fft_full_correlation(a, b):
    """Compute full linear cross-correlation of a and b using FFT."""
    n = len(a)
    m = len(b)
    N = 1
    while N < n + m - 1:
        N *= 2
    a_pad = np.zeros(N)
    b_pad = np.zeros(N)
    a_pad[:n] = a
    b_pad[:m] = b
    A = np.fft.rfft(a_pad)
    B = np.fft.rfft(b_pad)
    corr = np.fft.irfft(A * np.conj(B))
    return corr[:n+m-1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_full_correlation(h, 1.0 - h)
    return corr.max() * dx

def smooth_max_grad(h, tau, dx):
    """Compute smooth max objective and gradient."""
    n = len(h)
    # Forward: compute correlation
    corr = fft_full_correlation(h, 1.0 - h)
    
    # Smooth max
    c_scaled = corr * dx
    max_c = c_scaled.max()
    shifted = np.clip(tau * (c_scaled - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s

    # = (1/tau)*log(s) + max_c. Yes, correct.
    
    # Gradient
    # first_term: cc of (1-h) with p
    N2 = 1
    while N2 < 3*n - 2:
        N2 *= 2
    a_pad = np.zeros(N2)
    a_pad[:n] = 1.0 - h

    A = np.fft.rfft(a_pad)

    second_term = h * (prefix_p[n:2*n] - prefix_p[0:n])
    
    grad = dx * (first_term - second_term)
    
    return obj, grad

def project_to_feasible(g, n):
    """Project g to the feasible set: 0<=h<=1, sum(h)=n/2."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    """Interpolate h to target_n points with periodic boundary."""
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    # Extend h periodically for interpolation
    h_ext = np.concatenate([h, h, h])
    x_ext = np.concatenate([np.arange(n)*2.0/n - 2.0, np.arange(n)*2.0/n, np.arange(n)*2.0/n + 2.0])
    return np.interp(x_new, x_ext, h_ext)

def adam_smooth_max(h, n, tau, max_iters, time_limit, lr=0.5, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimization with smooth max objective."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    _, grad = smooth_max_grad(h, tau, dx)
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        t_count += 1
        _, grad = smooth_max_grad(h, tau, dx)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t_count)
        v_hat = v / (1 - beta2**t_count)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = project_to_feasible(h_new, n)
        
        # Check improvement every 50 iterations
        if (it + 1) % 50 == 0:
            c5_check = compute_c5(h_new)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def subgrad_optimize(h, n, max_iters, time_limit, lr=0.01):
    """Subgradient descent on the actual max."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

        corr = fft_full_correlation(h, 1.0 - h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
        
        k_star = np.argmax(corr)
        s = k_star - n + 1
        
        # Gradient of c[k_star] w.r.t. h
        grad = np.zeros(n)
        j0 = max(0, -s)
        j1 = min(n, n - s)
        if j0 < j1:
            indices = np.arange(j0, j1)
            grad[indices] += (1.0 - h[indices + s])
        
        j0b = max(0, s)
        j1b = min(n, n + s)
        if j0b < j1b:
            indices = np.arange(j0b, j1b)
            grad[indices] -= h[indices - s]
        
        grad *= dx
        h = project_to_feasible(h - lr * grad, n)
    
    return best_h, best_c5

    N = 1
    while N < n + m - 1:
        N *= 2
    a_pad = np.zeros(N)
    b_pad = np.zeros(N)
    a_pad[:n] = a
    b_pad[:m] = b
    A = np.fft.rfft(a_pad)
    B = np.fft.rfft(b_pad)
    corr = np.fft.irfft(A * np.conj(B))
    return corr[:n + m - 1]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_full_correlation(h, 1.0 - h)
    return corr.max() * dx


def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = fft_full_correlation(h, 1.0 - h)
    c_scaled = corr * dx
    max_c = c_scaled.max()
    shifted = np.clip(tau * (c_scaled - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = max_c + np.log(s) / tau
    N2 = 1
    while N2 < 3 * n - 2:
        N2 *= 2
    a_pad = np.zeros(N2)
    a_pad[:n] = 1.0 - h
    b_pad = np.zeros(N2)
    b_pad[:2 * n - 1] = p
    A = np.fft.rfft(a_pad)
    B = np.fft.rfft(b_pad)
    cc = np.fft.irfft(np.conj(A) * B)
    first_term = cc[n - 1::-1]
    prefix_p = np.zeros(2 * n)
    prefix_p[1:] = np.cumsum(p)
    second_term = h * (prefix_p[n:2 * n] - prefix_p[0:n])
    grad = dx * (first_term - second_term)
    return obj, grad


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_to_res(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, target_n)
    return np.interp(x_new, x_old, h)


def adam_optimize(h, n, tau, max_iters, time_limit, lr, beta1, beta2, epsilon):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    _, grad = smooth_max_grad(h, tau)
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    start = time.time()
    stall_count = 0
    best_stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t_count += 1
        _, grad = smooth_max_grad(h, tau)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = project_to_feasible(h_new, n)
        if (it + 1) % 100 == 0:
            c5_check = compute_c5(h_new)
            if c5_check < best_c5 - 1e-14:
                best_c5 = c5_check
                best_h = h_new.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > best_stall:
                    best_stall = stall_count
        h = h_new
    return best_h, best_c5


def subgrad_optimize(h, n, max_iters, time_limit, lr):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = fft_full_correlation(h, 1.0 - h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
        k_star = np.argmax(corr)
        s = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, -s)
        j1 = min(n, n - s)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += (1.0 - h[idx + s])
        j0b = max(0, s)
        j1b = min(n, n + s)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= h[idx - s]
        grad *= dx
        h = project_to_feasible(h - lr * grad, n)
    return best_h, best_c5


def multi_tau_optimize(h, n, tau_list, max_iters_each, time_limit, lr):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for tau in tau_list:
        if time.time() - start > time_limit:
            break
        remaining_tl = time_limit - (time.time() - start)
        if remaining_tl < 5:
            break
        h_cur, c5_cur = adam_optimize(best_h, n, tau, max_iters_each, remaining_tl, lr, 0.9, 0.999, 1e-8)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def _remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: moderate resolution n=2000
    if _remaining() > 120:
        n1 = 2000
        h1 = project_to_feasible(interp_to_res(best_h, n1), n1)
        tl = min(_remaining() * 0.2, 250)
        h1, c5_1 = multi_tau_optimize(h1, n1, [15.0, 25.0, 40.0], 3000, tl, 0.8)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: n=5000
    if _remaining() > 100:
        n2 = 5000
        h2 = project_to_feasible(interp_to_res(best_h, n2), n2)
        tl = min(_remaining() * 0.25, 300)
        h2, c5_2 = multi_tau_optimize(h2, n2, [20.0, 35.0], 2500, tl, 0.5)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: n=10000
    if _remaining() > 100:
        n3 = 10000
        h3 = project_to_feasible(interp_to_res(best_h, n3), n3)
        tl = min(_remaining() * 0.25, 300)
        h3, c5_3 = adam_optimize(h3, n3, tau=30.0, max_iters=5000, time_limit=tl * 0.6, lr=0.3, beta1=0.9, beta2=0.999, epsilon=1e-8)
        if _remaining() > 50:
            h3, c5_3b = adam_optimize(h3, n3, tau=50.0, max_iters=3000, time_limit=_remaining() * 0.2, lr=0.2, beta1=0.9, beta2=0.999, epsilon=1e-8)
            if c5_3b < c5_3:
                c5_3 = c5_3b
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: subgradient at current resolution
    if _remaining() > 30:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = subgrad_optimize(h_f, n_f, max_iters=200000, time_limit=_remaining() * 0.15, lr=0.15)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: random perturbations to escape local optima
    if _remaining() > 60:
        n_p = len(best_h)
        for attempt in range(8):
            if _remaining() < 25:
                break
            noise_scale = 0.03 + 0.02 * attempt
            noise = np.random.randn(n_p) * noise_scale
            h_noise = project_to_feasible(best_h + noise, n_p)
            h_noise, c5_noise = adam_optimize(h_noise, n_p, tau=25.0, max_iters=2000, time_limit=_remaining() * 0.08, lr=0.4, beta1=0.9, beta2=0.999, epsilon=1e-8)
            if c5_noise < best_c5:
                best_c5 = c5_noise
                best_h = h_noise.copy()

    # Phase 6: try higher resolution if time allows
    if _remaining() > 80:
        n4 = 20000
        if n4 > len(best_h):
            h4 = project_to_feasible(interp_to_res(best_h, n4), n4)
            tl = min(_remaining() * 0.3, 200)
            h4, c5_4 = adam_optimize(h4, n4, tau=25.0, max_iters=3000, time_limit=tl, lr=0.2, beta1=0.9, beta2=0.999, epsilon=1e-8)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()

    # Phase 7: final subgradient polish
    if _remaining() > 15:
        n_final = len(best_h)
        h_final = best_h.copy()
        h_final, c5_final = subgrad_optimize(h_final, n_final, max_iters=100000, time_limit=_remaining() * 0.3, lr=0.08)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
