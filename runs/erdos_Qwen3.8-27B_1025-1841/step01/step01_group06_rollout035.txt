    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    c5_curr = float(np.max(corr) * dx)

    no_improve = 0
    patience = check_interval * 60

        if time.time() > deadline:
            break
        grad = get_subgradient(h, n, k_star)
        h_new = project_target(h - alpha * grad, n)
        
        if (it + 1) % check_interval == 0:
            corr = np.correlate(h_new, 1.0 - h_new, mode="full")
            new_k = int(np.argmax(corr))
            c5_new = float(np.max(corr) * dx)
            if new_k != k_star:
                k_star = new_k
        else:

            c5_curr = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % (check_interval * 5) == 0:
                alpha *= 0.85
            if no_improve > patience:
                break
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    return h, float(np.max(corr) * dx)

    c5_curr = compute_c5(h, dx)
    target = n / 2.0

        penalty = 100.0

            corr = np.correlate(hx, 1.0 - hx, mode="full")
            m = np.max(corr)
            logsum = np.log(np.sum(np.exp((corr - m) / temp)))
            base = (m + temp * logsum) * dx
            sum_dev = np.sum(hx) - target
            return base + penalty * sum_dev * sum_dev

        try:

                c5_new = compute_c5(h_new, dx)
                if c5_new < c5_curr:
                    h = h_new
                    c5_curr = c5_new
        except Exception:
            pass
    return h, c5_curr

def reduced_polish(h, n, dx, deadline):

def get_subgradient(h, n, k_star):
    s = k_star - (n - 1)

        m = n - s
        grad[:m] += (1.0 - h[s:s+m])
        grad[s:s+m] -= h[:m]
    else:
        lo = -s
        m = n - lo
        grad[lo:] += (1.0 - h[:m])
        grad[:m] -= h[lo:]
    return grad
