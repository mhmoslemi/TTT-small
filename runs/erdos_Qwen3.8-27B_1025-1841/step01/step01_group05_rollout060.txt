def project_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def c5_val(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def sg(h, beta, gam):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    mxc = np.max(c)
    e = np.exp(beta * (c - mxc))
    p = e / e.sum()
    omh = 1.0 - h
    g = np.zeros(N)

import numpy as np
import time

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def _grad(h, beta, gam):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    mxc = c.max()
    w = np.exp(beta * (c - mxc))
    w /= w.sum()
    om = 1.0 - h
    g = np.zeros(N)
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-7:
            continue
        o = k - N + 1
        if o >= 0:
            g[o:] += wk * om[:N - o]
            g[:N - o] -= wk * h[o:]
        else:
            a = -o
            g[:N - a] += wk * om[a:]
            g[a:] -= wk * h[:N - a]
    g *= dx
    if gam:
        g += gam * (1.0 - 2.0 * h)
    return g

def _opt(h, N, t_end, beta_init=30.0):
    tgt = N * 0.5
    bh = h.copy()
    bc = _c5(h)
    lr = 0.1
    st = 0
    beta = beta_init
    while time.time() < t_end:
        gr = _grad(h, beta, 0.0)
        if np.abs(gr).max() < 1e-14:
            break
        a = lr
        ok = False
        for _ in range(40):
            ht = _proj(h - a * gr, tgt)
            ct = _c5(ht)
            if ct < bc - 1e-15:
                h[:] = ht
                bh[:] = ht
                bc = ct
                ok = True
                break
            a *= 0.5
        if ok:
            st = 0
            lr = min(lr * 1.05, 0.5)
        else:
            st += 1
            lr *= 0.85
            if 15 <= st % 30 <= 25:
                gr2 = _grad(h, beta, 0.05)
                a2 = max(lr, 0.01) * 0.5
                ht2 = _proj(h - a2 * gr2, tgt)
                ct2 = _c5(ht2)
                if ct2 < bc - 1e-15:
                    h[:] = ht2
                    bh[:] = ht2
                    bc = ct2
                    st = max(0, st - 15)
            if st % 200 == 100:
                beta = max(10.0, beta * 0.5)
            elif st % 200 == 0 and st > 0:
                beta = min(80.0, beta * 1.5)
            if st > 600 or lr < 1e-13:
                break
    return bh, bc
