import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def smooth_max_c5(h, dx, beta):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_sum(h, target):
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0, 1)

def resample(h_old, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old) + 1.0 / n_old
    x_new = np.arange(n_new) * (2.0 / n_new) + 1.0 / n_new
    return np.interp(x_new, x_old, h_old)

def optimize_n(n, parent, parent_n, seed, time_left):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_local = None
    best_local_c5 = float('inf')
    t0 = time.time()
    n_starts = 3 if time_left > 200 else 2

    starts = []
    if parent is not None:
        if n == parent_n:
            starts.append(parent.copy())
        else:
            starts.append(resample(parent, parent_n, n))
        h_pert = starts[0] + 0.05 * np.random.randn(n)
        starts.append(h_pert)
    h_rand = np.random.rand(n)
    starts.append(h_rand)

        if time.time() - t0 > time_left - 5:

            h0 = project_sum(res.x, target)

        if c5 < best_local_c5:
            best_local_c5 = c5
            best_local = h0.copy()
