    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def cd_optimize(h, max_sweeps=5000, time_limit=float('inf'), n_try=200, verbose=False):
    """Coordinate descent with exact 1D subproblem solving."""
    n = len(h)
    h = np.array(h, dtype=float).copy()
    best_c5 = compute_c5(h)
    
    ts = np.linspace(0.0, 1.0, n_try)
    
    for sweep in range(max_sweeps):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            
            # Compute a and b: corr[k] = a[k]*t + b[k]
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = old_val
            
            a = corr1 - corr0  # shape (2n-1,)
            b = corr0          # shape (2n-1,)

                hi = min(1.0, ts[best_idx + 1])

                    vals_fine = np.max(b[None, :] + a[None, :] * ts_fine[:, None], axis=1)
                    best_idx_f = np.argmin(vals_fine)
                    best_t = ts_fine[best_idx_f]

                    improved = True

    n = len(h)
    dx = 2.0 / n
    h = np.array(h, dtype=float).copy()
    best_c5 = compute_c5(h)
    h_best = h.copy()

        max_val = np.max(corr)

                if 0 <= idx2 < n:
                    g -= h[idx2]
                grad[i] += g

        h_new = np.clip(h_new, 0.0, 1.0)
        # Project to sum constraint
        diff = n / 2.0 - np.sum(h_new)
        h_new = h_new + diff / n

            h = h_new

                h = h_new2

def compute_subgradient(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_lags:
        m = k - n + 1  # shift
        for i in range(n):
            g = 0.0
            idx1 = i + m
            if 0 <= idx1 < n:
                g += 1.0 - h[idx1]

            idx2 = i - m
            if 0 <= idx2 < n:

        mask2 = (idx2 >= 0) & (idx2 < n)
