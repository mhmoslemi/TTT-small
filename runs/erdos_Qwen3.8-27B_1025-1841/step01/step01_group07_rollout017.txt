    sum_exp = np.sum(exp_shifted)

Let me think about the problem from a different angle. The key constraint is that for EVERY lag k, the integral ∫h(x)(1-h(x+k))dx should be small. This means h(x) and h(x+k) should be "aligned" - where h is high, h shifted should also be high.

This is related to the concept of "autocorrelation" - we want the autocorrelation of h to be as "flat" as possible (like a Barker sequence or Zadoff-Chu sequence in signal processing).

            h_s = best_h + 0.02 * rng.randn(n)
            h_s = np.clip(h_s, 0, 1)
            h_s = project_sum(h_s, ts)
            beta = 40
            lam = 200.0
        elif it == 2:
            h_s = 0.5 + 0.35 * np.cos(2.0 * np.pi * 3.0 * x / 2.0 + rng.rand() * 6.28)
            h_s = np.clip(h_s, 0, 1)
            h_s = project_sum(h_s, ts)
            beta = 40
            lam = 200.0
        elif it == 3:
            h_s = 0.5 + 0.4 * np.cos(2.0 * np.pi * 5.0 * x / 2.0) + 0.1 * np.cos(2.0 * np.pi * 7.0 * x / 2.0)
            h_s = np.clip(h_s, 0, 1)
            h_s = project_sum(h_s, ts)
            beta = 40
            lam = 200.0
        elif it == 4:
            h_s = np.zeros(n)
            bs = 12
            for i in range(0, n - bs // 2, bs):
                h_s[i:i + bs // 2] = 1.0
            h_s = project_sum(h_s, ts)
            beta = 40
            lam = 200.0
        elif it == 5:
            h_s = 0.5 + 0.45 * np.sin(2.0 * np.pi * 4.0 * x / 2.0 + rng.rand() * 3.14)
            h_s = np.clip(h_s, 0, 1)
            h_s = project_sum(h_s, ts)
            beta = 40
            lam = 200.0
        else:
            h_s = best_h + 0.005 * rng.randn(n)
            h_s = np.clip(h_s, 0, 1)
            h_s = project_sum(h_s, ts)
            beta = 100
            lam = 2000.0

        _opt['beta'] = beta
        _opt['lam'] = lam

        maxiter = 150 if beta > 80 else 250
        res = minimize(
            _obj, h_s, jac=_jac, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'maxfun': 6000, 'ftol': 1e-14}
        )

        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = project_sum(h_new, ts)
        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final polish with high beta
    _opt['beta'] = 300
    _opt['lam'] = 10000.0
    res = minimize(
        _obj, best_h, jac=_jac, method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 300, 'maxfun': 8000, 'ftol': 1e-16}
    )
    h_f = np.clip(res.x, 0.0, 1.0)
    h_f = project_sum(h_f, ts)
    c5_f = compute_c5(h_f, dx)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f

    # Second polish
    if time.time() - t0 < budget_s * 0.95:
        _opt['beta'] = 500
        _opt['lam'] = 20000.0
        res2 = minimize(
            _obj, best_h, jac=_jac, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'maxfun': 5000, 'ftol': 1e-16}
        )
        h_f2 = np.clip(res2.x, 0.0, 1.0)
        h_f2 = project_sum(h_f2, ts)
        c5_f2 = compute_c5(h_f2, dx)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, ts)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n)
