import time

_g_target = 0.0
_g_dx = 0.0
_g_temp = 0.0

def _constraint_fun(x):
    return np.sum(x) - _g_target

def _constraint_jac(x):
    return np.ones(len(x))

def _smax_obj(x):
    hx = np.clip(x, 0.0, 1.0)
    c = np.correlate(hx, 1.0 - hx, mode="full")
    m = np.max(c)
    return (m + _g_temp * np.log(np.sum(np.exp((c - m) / _g_temp)))) * _g_dx

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)

        s = -shift
        m = n - s
        if m <= 0:

import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0
_g_dx = 0.0
_g_temp = 0.0

def _constraint_fun(x):
    return np.sum(x) - _g_target

def _constraint_jac(x):
    return np.ones(len(x))

def _smax_obj(x):
    hx = np.clip(x, 0.0, 1.0)
    c = np.correlate(hx, 1.0 - hx, mode="full")
    m = np.max(c)
    return (m + _g_temp * np.log(np.sum(np.exp((c - m) / _g_temp)))) * _g_dx

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(150):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_target(np.clip(np.interp(x_new, x_old, h), 0.0, 1.0), n_new)

        if m <= 0:

        grad[shift:] -= h[:m]
