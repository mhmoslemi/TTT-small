def subgrad_step(h, n):
    one_h = 1.0 - h
    c = np.correlate(h, one_h, mode="full")
    k_star = np.argmax(c)
    m = k_star - (n - 1)
    grad = np.ones(n)
    if m >= 0:
        if m < n:
            grad[m:] -= h[:n-m]
            grad[:n-m] -= h[m:]
    else:
        p = -m
        if p < n:
            grad[:n-p] -= h[p:]
            grad[p:] -= h[:n-p]
    return grad

def subgradient_descent(h0, n, max_iter, lr0=0.1, tol=1e-10):
    dx = 2.0 / n
    h = project_feasible(h0.copy().astype(float), n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    lr = lr0
    for it in range(max_iter):
        grad = subgrad_step(h, n)
        h_new = h - lr * dx * grad
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)

            best_h = h_new.copy()
            h = h_new
            # Could increase lr slightly
        else:
            h = best_h  # revert
            lr *= 0.95
        if lr < 1e-8:

    return best_h, best_val

def slsqp_refine(h0, n, max_iter=300):
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode="full")
        return np.max(c) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                   'jac': lambda h: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(obj, project_feasible(h0.copy(), n), method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h_opt = project_feasible(res.x, n)
        val = compute_c5(h_opt, n)
        return h_opt, val
    except:
        return h0, compute_c5(h0, n)

    for it in range(max_iter):
        c = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(c)
        m = k_star - (n - 1)
        grad = np.ones(n)
        if m >= 0:
            if m < n:
                grad[m:] -= h[:n - m]
                grad[:n - m] -= h[m:]
        else:
            p = -m
            if p < n:
                grad[:n - p] -= h[p:]
                grad[p:] -= h[:n - p]
        h_new = project_feasible(h - lr * dx * grad, n)
        val = compute_c5(h_new, n)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 1.0)
        else:
            stall += 1
            lr *= 0.92
            if stall > 500:
                break
    return best_h, best_val

def interpolate_to(h, old_n, new_n):
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, new_n)

def make_random_start(n, rng):
    h = rng.rand(n)
    return project_feasible(h, n)

def make_structured_starts(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    starts = []
    starts.append(project_feasible(np.full(n, 0.5), n))
    starts.append(project_feasible(np.exp(-3*(x-1)**2), n))
    starts.append(project_feasible(0.5 + 0.4*np.sin(2*np.pi*x), n))
    starts.append(project_feasible(0.5 + 0.3*np.sin(4*np.pi*x), n))
    starts.append(project_feasible(np.ones(n) * 0.5 + 0.3*(x - 1.0), n))
    for _ in range(3):
        h = project_feasible(np.exp(-0.5*(x - rng.uniform(0.3, 1.7))**2 / rng.uniform(0.05, 0.5)), n)
        starts.append(h)
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    elapsed = lambda: time.time() - t0
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_start = len(h_init)
    except Exception:
        n_start = 45
        h_init = np.full(n_start, 0.5)
    
    best_h = project_feasible(h_init.copy(), n_start)
    best_val = compute_c5(best_h, best_h and len(best_h) or n_start)
    best_n = n_start
    
    rng = np.random.RandomState(seed)
    
    # Phase 1: BCD at starting resolution
    if elapsed() < budget_s * 0.25:
        h_b, v_b = block_coordinate_descent(best_h, best_n, max_sweeps=300)
        if v_b < best_val:
            best_h, best_val = h_b, v_b
    
    # Phase 2: SLSQP from best
    if elapsed() < budget_s * 0.4:
        h_s, v_s = slsqp_refine(best_h, best_n, max_iter=500)
        if v_s < best_val:
            best_h, best_val = h_s, v_s
        h_b, v_b = block_coordinate_descent(best_h, best_n, max_sweeps=200)
        if v_b < best_val:
            best_h, best_val = h_b, v_b
    
    # Phase 3: Multi-start
    for start_idx in range(30):
        if elapsed() > budget_s * 0.6:
            break
        if start_idx < 5:
            h_rand = make_structured_starts(best_n, rng)[start_idx % 9]
        else:
            h_rand = make_random_start(best_n, rng)
        
        h_b, v_b = block_coordinate_descent(h_rand, best_n, max_sweeps=100)
        if v_b < best_val:
            best_h, best_val = h_b, v_b
            if elapsed() < budget_s * 0.55:
                h_s, v_s = slsqp_refine(best_h, best_n, max_iter=300)
                if v_s < best_val:
                    best_h, best_val = h_s, v_s
    
    # Phase 4: Subgradient polish
    if elapsed() < budget_s * 0.7:
        h_sg, v_sg = subgradient_descent(best_h, best_n, max_iter=50000, lr0=0.5)
        if v_sg < best_val:
            best_h, best_val = h_sg, v_sg
        h_b, v_b = block_coordinate_descent(best_h, best_n, max_sweeps=300)
        if v_b < best_val:
            best_h, best_val = h_b, v_b
    
    # Phase 5: Higher resolution
    for mult in [2, 3]:
        if elapsed() > budget_s * 0.85:
            break
        new_n = best_n * mult
        if new_n > 500:
            break
        h_interp = interpolate_to(best_h, best_n, new_n)
        h_b, v_b = block_coordinate_descent(h_interp, new_n, max_sweeps=150)
        if v_b < best_val:
            best_h, best_val, best_n = h_b, v_b, new_n
            if elapsed() < budget_s * 0.9:
                h_s, v_s = slsqp_refine(best_h, best_n, max_iter=300)
                if v_s < best_val:
                    best_h, best_val = h_s, v_s
                h_b2, v_b2 = block_coordinate_descent(best_h, best_n, max_sweeps=200)
                if v_b2 < best_val:
                    best_h, best_val = h_b2, v_b2
        # Also try random starts at new resolution
        for _ in range(5):
            if elapsed() > budget_s * 0.85:
                break
            h_rand = make_random_start(new_n, rng)
            h_b, v_b = block_coordinate_descent(h_rand, new_n, max_sweeps=80)
            if v_b < best_val:
                best_h, best_val, best_n = h_b, v_b, new_n
    
    # Final polish
    if elapsed() < budget_s * 0.97:
        h_b, v_b = block_coordinate_descent(best_h, best_n, max_sweeps=500)
        if v_b < best_val:
            best_h, best_val = h_b, v_b
        h_s, v_s = slsqp_refine(best_h, best_n, max_iter=300)
        if v_s < best_val:
            best_h, best_val = h_s, v_s
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    
    return (best_h.tolist(), best_val, best_n)
