import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj_and_grad(h, n, dx, T, penalty=1000.0):
    """Compute soft-max objective and its gradient."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = T * dx * corr
    m = np.max(s)
    exp_s = np.exp(s - m)
    w = exp_s / np.sum(exp_s)
    val = (m + np.log(np.sum(exp_s))) / T * dx
    
    # FFT-based gradient
    N = 1
    while N < 4 * n:
        N *= 2
    W = fft(w, N)
    Wr = fft(w[::-1], N)
    H = fft(h, N)
    H1 = fft(1.0 - h, N)
    conv1 = ifft(W * H1).real
    conv2 = ifft(Wr * H).real
    grad = dx * dx * (conv1[n-1:n-1+n] - conv2[n-1:n-1+n])
    
    # Add penalty gradient
    sum_diff = np.sum(h) - n / 2.0
    val += penalty * sum_diff ** 2
    grad += 2.0 * penalty * sum_diff * np.ones(n)
    
    return val, grad

def project_onto_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

    """Run L-BFGS-B optimization at a given resolution."""
    dx = 2.0 / n
    target = n / 2.0
    
    h = project_onto_feasible(h0.copy(), n)
    
    best_h = h.copy()
    best_val = None
    
    def obj(x):
        return soft_max_obj_and_grad(x, n, dx, T, penalty)[0]
    
    def grad(x):
        return soft_max_obj_and_grad(x, n, dx, T, penalty)[1]
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_onto_feasible(res.x, n)
            c5_new = evaluate_c5(h_new)
            if best_val is None or c5_new < best_val:
                best_val = c5_new
                best_h = h_new
    except Exception:
        pass
    
    return best_h, (best_val if best_val is not None else evaluate_c5(best_h))

    resolutions = sorted(set([start_n, 2*start_n, 4*start_n]), reverse=False)
    # Cap at reasonable size

    T_schedule = [50.0, 100.0, 200.0, 500.0, 1000.0]

    h_current = h_start.copy()

        if time.time() > deadline - 30:
            break
        
        if n == current_n:
            h_work = h_current.copy()
        else:
            h_work = upscale(h_current, current_n, n)
            h_work = project_onto_feasible(h_work, n)

            if c5 < best_c5:

    conv2 = ifft(Wr * H).real
    return dx * dx * (conv1[n - 1:n - 1 + n] - conv2[n - 1:n - 1 + n])
