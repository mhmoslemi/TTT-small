from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, temperature=10.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Smooth max
    max_val = np.max(corr)
    shifted = corr - max_val
    result = max_val + np.log(np.sum(np.exp(temperature * shifted))) / temperature

            g += w[k] * _grad_shift(x, shifts[k])
    g *= dx
    diff = x.sum() - target
    return g + 200.0 * diff

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
    return np.clip(h, 0.0, 1.0)

def _cont_opt(h0, target, temps, max_iter, t_start, t_limit):
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    n = len(h)
    for t in temps:
        if time.time() - t_start > t_limit:
            break
        res = minimize(_obj, h, args=(t, target),
                       jac=lambda x: _grad(x, t, target),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'gtol': 1e-9})
        h = _project(res.x, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def _bin_search(h_init, rng, n_iter, t_start, t_limit, T0=0.4):
    n = len(h_init)
    h = h_init.copy().astype(np.float64)
    c5 = compute_c5(h)
    best = h.copy()
    best_c5 = c5
    T = T0
    cool = 0.99985
    for i in range(n_iter):
        if time.time() - t_start > t_limit:
            break
        T = max(0.003, T * cool)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        nc5 = compute_c5(h)
        if nc5 <= c5 or (rng.rand() < np.exp(-(nc5 - c5) / T)):
            c5 = nc5
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
    return best, best_c5

def _fix_bin_sum(h, target, rng):
    h = h.copy()
    cur = int(h.sum())
    if cur < int(target):
        idx = np.where(h == 0)[0].copy()
        rng.shuffle(idx)
        for i in range(int(target) - cur):
            h[idx[i]] = 1.0
    elif cur > int(target):
        idx = np.where(h == 1)[0].copy()
        rng.shuffle(idx)
        for i in range(cur - int(target)):
            h[idx[i]] = 0.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    t_limit = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    n_par = len(initial_h_values)
    h_par = np.array(initial_h_values, dtype=np.float64).copy()
    tgt_par = n_par / 2.0
    h_par = _project(h_par, tgt_par)

    best_h = h_par.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_par

    # Phase 1: continuous optimization on parent
    h1, c5_1 = _cont_opt(h_par, tgt_par, [20, 60, 150, 400], 500, t0, t_limit)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: binary search from continuous result
    h_bin = _fix_bin_sum((best_h > 0.5).astype(np.float64), tgt_par, rng)
    h2, c5_2 = _bin_search(h_bin, rng, 500000, t0, t_limit)
    if c5_2 < best_c5:
        best_h, best_c5, best_n = h2, c5_2, n_par

    # Phase 3: try other resolutions
    for n_try in [40, 64, 100, 128, 160, 256, 400]:
        if time.time() > t_limit - 20:
            break
        if n_try == best_n:
            continue
        x_old = np.arange(best_n) / best_n
        x_new = np.arange(n_try) / n_try
        h_try = np.interp(x_new, x_old, best_h)
        h_try = _project(h_try, n_try / 2.0)

        h3, c5_3 = _cont_opt(h_try, n_try / 2.0, [30, 100, 300], 400, t0, t_limit - 10)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n_try
            continue

        # binary at this resolution
        h_b = _fix_bin_sum((best_h > 0.5).astype(np.float64), n_try / 2.0, rng)
        h4, c5_4 = _bin_search(h_b, rng, 300000, t0, t_limit - 5)
        if c5_4 < best_c5:
            best_h, best_c5, best_n = h4, c5_4, n_try

    # Phase 4: final polish
    if time.time() < t_limit - 10:
        h5, c5_5 = _cont_opt(best_h, best_n / 2.0, [50, 100, 200, 500, 1000], 600, t0, t_limit)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    return best_h, best_c5, len(best_h)
