import numpy as np
import time

def correlate_full_fft(a, b):
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    R = np.fft.irfft(A * np.conj(B), N)
    result = np.empty(2*n-1)
    result[n-1:] = R[0:n]
    result[:n-1] = R[N-n+1:N]
    return result

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate_full_fft(h, 1.0 - h)
    return corr.max() * dx

def smooth_objective_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    
    # Correlation via FFT
    N = 1
    while N < 2*n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(b, N)
    R = np.fft.irfft(H * np.conj(B), N)
    corr = np.empty(2*n-1)
    corr[n-1:] = R[0:n]
    corr[:n-1] = R[N-n+1:N]
    
    # Softmax
    cmax = corr.max()
    shifted = np.clip(tau * (corr - cmax), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    obj = dx * (cmax + np.log(s) / tau)
    
    # Gradient via FFT convolutions
    # S1[j] = sum_m p_lag[m] * h[j+m] (zero-padded)
    # S2[j] = sum_m p_lag[m] * h[j-m] (zero-padded)
    # W[j] = sum of p[k] where n-1-j <= k <= n-1+j
    
    # For S1: convolution of reversed-p with h
    # For S2: convolution of p with h
    N2 = 1
    while N2 < 3*n - 2:
        N2 <<= 1
    
    P_arr = np.zeros(N2)
    P_arr[:2*n-1] = p
    
    P_rev = np.zeros(N2)
    P_rev[:2*n-1] = p[::-1]
    
    H_arr = np.zeros(N2)
    H_arr[:n] = h
    
    conv_s2 = np.fft.irfft(np.fft.rfft(P_arr) * np.fft.rfft(H_arr), N2)
    conv_s1 = np.fft.irfft(np.fft.rfft(P_rev) * np.fft.rfft(H_arr), N2)
    
    S2 = conv_s2[n-1:n-1+n]
    S1 = conv_s1[n-1:n-1+n]
    
    # Window sum W[j] = sum_{k=n-1-j}^{n-1+j} p[k] (clipped to [0, 2n-2])
    cumsum_p = np.concatenate([[0], np.cumsum(p)])
    W = np.empty(n)
    for j in range(n):
        lo_idx = max(0, n-1-j)
        hi_idx = min(2*n-2, n-1+j) + 1
        W[j] = cumsum_p[hi_idx] - cumsum_p[lo_idx]
    
    grad = dx * (W - S1 - S2)
    
    return obj, grad, corr

    best_obj = np.inf
    best_h = h.copy()

        v = np.zeros(n)

            c5_cur = corr.max() * (2.0/n)

                best_h = h.copy()
                stall = 0

            h = project_to_feasible(h + v, n)

                v *= 0.3
                stall = 0
                if lr < 1e-10:

    return best_h

        if stall > 500:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-12:

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)
