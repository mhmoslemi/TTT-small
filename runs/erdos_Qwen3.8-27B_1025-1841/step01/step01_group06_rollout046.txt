```python
def smooth_max_gradient(h, n, T, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_terms = np.exp((corr - m_val) / T)
    w = exp_terms / np.sum(exp_terms)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += w[k] * (1.0 - h[s:s+m])
            grad[s:] -= w[k] * h[:m]
        else:
            t = -s
            m = n - t
            grad[t:] += w[k] * (1.0 - h[:m])
            grad[:m] -= w[k] * h[t:]
    
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
```

```python
def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(600):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = h < 1.0 - 1e-13
        else:
            free = h > 1e-13
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-11:
        if s < target:
            free = h < 1.0 - 1e-9
        else:
            free = h > 1e-9
        if np.any(free):
            h[free] += (target - s) / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_to_constraint(h_new, n_new)

def smooth_max_obj_grad(h, dx, T, n, target, gamma):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = (corr - m_val) / T
    exp_terms = np.exp(shifted)
    w = exp_terms / np.sum(exp_terms)
    soft_max = (m_val + T * np.log(np.sum(exp_terms))) * dx
    sum_h = np.sum(h)
    penalty = gamma * (sum_h - target) ** 2
    obj = soft_max + penalty
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            m = n - s
            grad[:m] += wk * (1.0 - h[s:s+m])
            grad[s:] -= wk * h[:m]
        else:
            t = -s
            m = n - t
            grad[t:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[t:]
    grad *= dx
    grad += 2.0 * gamma * (sum_h - target)
    return obj, grad

def subgradient_step(h, n, k_star):
    s = k_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:s+m])
        grad[s:] -= h[:m]
    else:
        t = -s
        m = n - t
        grad[t:] += (1.0 - h[:m])
        grad[:m] -= h[t:]
    return grad
```
