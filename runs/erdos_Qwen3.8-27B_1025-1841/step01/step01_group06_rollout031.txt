    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_active_gradient(h, n):
    """Compute gradient of the max correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    # corr index k_star corresponds to lag = k_star - (n-1)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        # c[lag] = sum_{i=0}^{m-1} h[i] * (1 - h[i+lag])
        # d/dh[i] for i in [0, m-1]: (1 - h[i+lag])
        # d/dh[i+lag] for i in [0, m-1]: -h[i]
        grad[:m] += (1.0 - h[lag:lag+m])
        grad[lag:lag+m] -= h[:m]
    else:
        m = n + lag  # lag is negative, so m = n - |lag|
        # c[lag] = sum_{i=-lag}^{n-1} h[i] * (1 - h[i+lag])
        # where i+lag ranges from 0 to n-1+lag = m-1
        start = -lag
        grad[start:] += (1.0 - h[:m])
        h[start:] -= 0  # no-op, just for clarity
        grad[:m] -= h[start:]
    return grad, k_star, corr

def bb_step_size(h_list, g_list):
    """Barziland-Borwein step size."""
    if len(h_list) < 2:
        return 1.0
    y = h_list[-1] - h_list[-2]
    s = g_list[-1] - g_list[-2]

    if ss < 1e-14 or abs(ys) < 1e-14:

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
