```python
def compute_grad_topk(h, p, k_indices, k_count, tau):
    """Compute gradient using only top-k correlation values."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in k_indices:
        sh = k - n + 1
        pk = p[k]
        if pk < 1e-16:
            continue
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0-sh:j1-sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b-sh:j1b-sh]
    return grad * dx
```
