```python
    corr = np.correlate(h, onemh, mode="full")
    return float(np.max(corr) * dx)

    target = n / 2.0
    h = np.clip(x, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break

        if np.any(free):
            adj = diff / np.sum(free)
            h[free] = np.clip(h[free] + adj, 0.0, 1.0)
        else:
            h = np.clip(h + diff / n, 0.0, 1.0)
    return h

    c5 = compute_c5(h, n)
    pen = penalty * (np.sum(h) - n / 2.0) ** 2
```

```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h, n):
    dx = 2.0 / n
    onemh = 1.0 - h
    corr = np.correlate(h, onemh, mode="full")
    return float(np.max(corr) * dx)

def project_h(x, n, max_iter=200):
    target = n / 2.0
    h = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree == 0:
            break
        h[free] = np.clip(h[free] + diff / nfree, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init_raw = np.array(initial_h_values, dtype=float)
    else:
        h_init_raw = np.full(45, 0.5)
    n = len(h_init_raw)
    
    best_h = project_h(h_init_raw.copy(), n)
    best_c5 = compute_c5(best_h, n)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Phase 1: L-BFGS-B on native resolution
    bounds = [(0.0, 1.0)] * n
    
    while time_left() > budget_s * 0.15:
        h_work = best_h.copy()
        # Add small perturbation for diversity
        if rng.random() < 0.3:
            h_work = project_h(h_work + rng.normal(0, 0.01, n), n)
        
        def obj(x):
            h = project_h(x, n)
            return compute_c5(h, n)
        
        res = minimize(obj, h_work, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = project_h(res.x, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 2: Try higher resolutions
    for mult in [2, 4, 8, 16]:
        n_new = n * mult
        if n_new > 10000 or time_left() < 60:
            break
        
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_new)
        f = interp1d(x_old, best_h, kind='cubic', bounds_error=False, fill_value='extrapolate')
        h_interp = project_h(np.clip(f(x_new), 0, 1), n_new)
        c5_interp = compute_c5(h_interp, n_new)
        
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
        
        # Optimize at new resolution
        bounds_new = [(0.0, 1.0)] * n_new
        def obj_new(x):
            h = project_h(x, n_new)
            return compute_c5(h, n_new)
        
        res = minimize(obj_new, h_interp, method='L-BFGS-B', bounds=bounds_new,
                       options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = project_h(res.x, n_new)
        c5_new = compute_c5(h_new, n_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            n = n_new  # Update for subsequent phases
    
    # Phase 3: Final polish
    while time_left() > 30:
        def obj_final(x):
            h = project_h(x, len(best_h))
            return compute_c5(h, len(best_h))
        
        res = minimize(obj_final, best_h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*len(best_h),
                       options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-14})
        h_new = project_h(res.x, len(best_h))
        c5_new = compute_c5(h_new, len(best_h))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            break
    
    return list(best_h), float(best_c5), len(best_h)
```

```python
        if nfree == 0:
```

def coord_optimize(h, n, max_sweeps=1000):
    """Coordinate descent with golden section for each coordinate."""
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    improved = True
    sweeps = 0
    while improved and sweeps < max_sweeps:
        improved = False
        sweeps += 1
        for i in range(n):
            old = h[i]
            # Golden section on [0, 1]
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            h[i] = c
            fc = compute_c5(h, n)
            h[i] = d

                    fd = fc

                    c = d

            h[i] = 1.0

            # else keep best_t
