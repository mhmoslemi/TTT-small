        _OPT_T = 500.0
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 6: Final coordinate descent pass
    n_pts = len(best_h)
    for round_num in range(20):
        if remaining() < 5:
            break
        improved = False
        for j in range(n_pts):
            if remaining() < 3:
                break
            old = best_h[j]
            best_c = best_c5
            best_trial = old
            for cand in [0.0, 1.0, max(0.0, old-0.01), max(0.0, old-0.005),
                         min(1.0, old+0.01), min(1.0, old+0.005)]:
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                best_h = best_h.copy()
                best_h[j] = best_trial
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
