import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary_with_sum(n, target_sum, rng):
    """Create a random binary array of length n with given sum."""
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:target_sum]] = 1.0
    return h

def swap_move(h, i, j, rng):
    """Swap h[i] and h[j] (one is 1, other is 0)."""
    h[i], h[j] = h[j], h[i]
    return h

def local_search(h, rng, max_iters, time_limit=None, start_time=None):
    """Greedy local search with swaps."""
    n = len(h)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iters):
        if time_limit and time.time() - start_time > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try random swap
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        pos_i = ones[i]
        pos_j = zeros[j]
        
        h[pos_i] = 0.0
        h[pos_j] = 1.0
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            # Revert
            h[pos_i] = 1.0
            h[pos_j] = 0.0
    
    return best_h, best_c5

def simulated_annealing(h, rng, max_iters, start_temp, end_temp, time_limit=None, start_time=None):
    """Simulated annealing with swap moves."""
    n = len(h)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    current_h = h.copy()
    
    ones = np.where(current_h == 1.0)[0]
    zeros = np.where(current_h == 0.0)[0]
    
    for it in range(max_iters):
        if time_limit and start_time and time.time() - start_time > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        temp = start_temp * (end_temp / start_temp) ** (it / max_iters)
        
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        pos_i = ones[i]
        pos_j = zeros[j]
        
        current_h[pos_i] = 0.0
        current_h[pos_j] = 1.0
        new_c5 = compute_c5(current_h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
            ones = np.where(current_h == 1.0)[0]
            zeros = np.where(current_h == 0.0)[0]
        else:
            current_h[pos_i] = 1.0
            current_h[pos_j] = 0.0
    
    return best_h, best_c5

where N is the number of valid i.

If the sequence is "balanced" (sum s[i] ≈ 0 over any window), and the autocorrelation sum s[i]*s[i+m] is small, then this would be small.

This connects to the theory of low-autocorrelation sequences! The problem is essentially asking for a ±1 sequence of length n (with sum 0, i.e., equally many +1 and -1) that minimizes max_m |autocorrelation at lag m| (in a bounded sense).

Wait, let me be more careful. For m ≥ 0:
c[m] = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])

In terms of s:
c[m] = sum_{i=0}^{n-1-m} (1+s[i])/2 * (1-s[i+m])/2

The last term is the (one-sided) autocorrelation of s at lag m.

If s is balanced (sum s[i] = 0 for the full sequence), then sum_{i=0}^{n-1-m} s[i] ≈ -sum_{i=n-m}^{n-1} s[i] (small if the sequence is "well-mixed").

Similarly for sum_{i=0}^{n-1-m} s[i+m] = sum_{j=m}^{n-1} s[j] ≈ -sum_{j=0}^{m-1} s[j].

So c[m] ≈ (1/4) [(n-m) + autocorr_s(m)] where autocorr_s(m) = sum_{i=0}^{n-1-m} s[i]*s[i+m].

To minimize c[m], we want autocorr_s(m) to be as negative as possible (close to -(n-m)). But autocorr_s(m) is bounded below by -(n-m) (when s[i] = -s[i+m] for all valid i).

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_pattern(n, block_ones, block_zeros, rng):
    """Create a periodic pattern with block_ones 1s followed by block_zeros 0s."""
    period = block_ones + block_zeros
    if n % period != 0:
        return None
    total_ones_per_period = block_ones
    total_ones = (n // period) * total_ones_per_period
    if total_ones != n // 2:
        return None
    h = np.zeros(n)
    for start in range(0, n, period):
        h[start:start + block_ones] = 1.0
    return h

def random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    return h

def local_search(h, rng, time_limit, start_time):
    """First-improvement local search with random swaps."""
    n = len(h)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    stall = 0
    max_stall = 1000
    
    while time.time() - start_time < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        n_trials = 100
        for _ in range(n_trials):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            if new_c5 < current_c5:
                current_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i], h[j] = h[j], h[i]
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
    
    return best_h, best_c5

def simulated_annealing(h, rng, time_limit, start_time, t_start=0.1, t_end=0.001):
    n = len(h)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    cur_h = h.copy()
    
    elapsed_0 = time.time() - start_time
    total_time = time_limit - elapsed_0
    if total_time <= 0:
        return best_h, best_c5
    
    t = t_start
    while time.time() - start_time < time_limit:
        # Update temperature
        frac = (time.time() - start_time - elapsed_0) / total_time
        t = t_start * (t_end / t_start) ** min(frac, 1.0)
        
        ones = np.where(cur_h == 1.0)[0]
        zeros = np.where(cur_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_periodic(n, ones_per_period, period):
    if n % period != 0:
        return None
    if ones_per_period * (n // period) != n // 2:
        return None

    for start in range(0, n, period):
        h[start:start + ones_per_period] = 1.0
    return h
