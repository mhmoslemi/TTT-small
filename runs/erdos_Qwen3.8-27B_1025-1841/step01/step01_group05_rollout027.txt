def evaluate_c5(h):

    return np.max(corr)

    h = np.clip(h, 0, 1)
    target_sum = len(h) / 2.0
    current_sum = np.sum(h)

        return h

    diff = target_sum - current_sum
    h = h + diff / len(h)
