import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

_obj_cfg = {}

def _smooth_objective(x):
    n = _obj_cfg['n']
    dx = _obj_cfg['dx']
    alpha = _obj_cfg['alpha']
    target = n / 2.0
    hh = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hh, 1.0 - hh, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    lse = (1.0 / alpha) * np.log(np.sum(np.exp(shifted))) + m
    penalty = _obj_cfg['penalty'] * (np.sum(hh) - target) ** 2
    return dx * lse + penalty


def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _interpolate_to(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    f = interp1d(x_old, h_old, kind='linear', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return _repair(h_new, n_new)

    t_start = time.time()
    h = h_init.copy()
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        _obj_cfg['n'] = n
        _obj_cfg['dx'] = dx
        _obj_cfg['alpha'] = alpha
        _obj_cfg['penalty'] = 50.0
        res = minimize(_smooth_objective, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14, 'gtol': 1e-12})
        h = np.clip(res.x, 0.0, 1.0)
        h = _repair(h, n)
    return h

    h[0::2] = 1.0
    return _repair(h, n)

def _make_constant(n):

    t_start = time.time()
    h = h.copy()
    for _ in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            # Try a few candidate values
            best_val = old_val
            best_obj = _compute_c5(h, dx)
            for cand in [0.0, 0.25, 0.5, 0.75, 1.0]:

                h[i] = cand
                obj = _compute_c5(h, dx)
                if obj < best_obj:
                    best_obj = obj
                    best_val = cand
            h[i] = best_val
            if best_val != old_val:
                improved = True
        if not improved:
            break
    return _repair(h, n)

import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

_obj_cfg = {}


def _smooth_objective(x):
    n = _obj_cfg['n']
    dx = _obj_cfg['dx']
    alpha = _obj_cfg['alpha']
    target = n / 2.0
    hh = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hh, 1.0 - hh, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    lse = (1.0 / alpha) * np.log(np.sum(np.exp(shifted))) + m
    penalty = _obj_cfg['penalty'] * (np.sum(hh) - target) ** 2
    return dx * lse + penalty


def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _interpolate_to(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    f = interp1d(x_old, h_old, kind='linear', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return _repair(h_new, n_new)


def _optimize_lbfgsb(h_init, n, dx, alphas, maxiter_per_alpha, time_limit):
    t_start = time.time()
    h = h_init.copy()
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        _obj_cfg['n'] = n
        _obj_cfg['dx'] = dx
        _obj_cfg['alpha'] = alpha
        _obj_cfg['penalty'] = 50.0
        try:
            res = minimize(_smooth_objective, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h = np.clip(res.x, 0.0, 1.0)
        except Exception:
            pass
        h = _repair(h, n)
    return h


def _make_alternating(n):
    h = np.zeros(n)
    h[0::2] = 1.0
    return _repair(h, n)


def _make_block(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return _repair(h, n)


def _make_block_sym(n):
    h = np.zeros(n)
    a = n // 4
    h[:a] = 1.0
    h[n // 2:n // 2 + a] = 1.0
    return _repair(h, n)


def _coordinate_descent(h, n, dx, max_sweeps, time_limit, rng):
    t_start = time.time()
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        perm = rng.permutation(n)
        for i in perm:
            old_val = h[i]
            best_val = old_val
            best_obj = _compute_c5(h, dx)
            for frac in [0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0]:
                if abs(frac - old_val) < 1e-10:
                    continue
                h[i] = frac
                obj = _compute_c5(h, dx)
                if obj < best_obj:
                    best_obj = obj
                    best_val = frac
            h[i] = best_val
    return _repair(h, n)


def _simulated_anneal(h, n, dx, n_steps, t_start, t_end, rng, time_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    for step in range(n_steps):
        if time.time() - time_limit:
            break
        frac = (t_start - step * (t_start - t_end) / n_steps)
        i = rng.randint(n)
        delta = rng.uniform(-0.5, 0.5)
        h_old = h[i]
        h[i] = np.clip(h_old + delta, 0.0, 1.0)
        h = _repair(h, n)
        c5_new = _compute_c5(h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        elif c5_new < _compute_c5(best_h, dx):
            pass
        else:
            if rng.random() > np.exp(-((c5_new - best_c5) / (frac + 1e-10))):
                h[i] = h_old
                h = _repair(h, n)
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = 1.0
    best_h = None
    best_n = 0

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - 0.5 - elapsed()

    n_list = [45, 90, 180, 360]
    inits_per_n = {}

    parent_h = None
    try:
        if initial_h_values is not None:
            parent_h = np.array(initial_h_values, dtype=float)
            inits_per_n[len(parent_h)] = parent_h.copy()
    except NameError:
        pass

    alphas_schedule = [20, 50, 100, 200, 500, 1000]
    alphas_fast = [50, 200, 500]

    for n in n_list:
        if remaining() < 40:
            break
        dx = 2.0 / n
        target = n / 2.0

        init_list = []
        if n in inits_per_n:
            init_list.append(inits_per_n[n])
        init_list.append(_make_alternating(n))
        init_list.append(_make_block(n))
        init_list.append(_make_block_sym(n))
        for _ in range(3):
            h_rand = rng.uniform(0, 1, n)
            init_list.append(_repair(h_rand, n))
        if parent_h is not None and n != len(parent_h):
            init_list.append(_interpolate_to(parent_h, len(parent_h), n))

        for h_init in init_list:
            if remaining() < 30:
                break
            alphas_use = alphas_schedule if len(h_init) <= 90 else alphas_fast
            h_opt = _optimize_lbfgsb(h_init, n, dx, alphas_use, 150, remaining() * 0.4)
            c5_val = _compute_c5(h_opt, dx)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n
                inits_per_n[n] = h_opt.copy()

        if remaining() < 60:
            break

        h_cd = inits_per_n.get(n, _make_alternating(n))
        cd_time = min(remaining() * 0.3, 120.0)
        max_cd_sweeps = 3 if n <= 90 else 1
        h_cd = _coordinate_descent(h_cd, n, dx, max_cd_sweeps, cd_time, rng)
        c5_cd = _compute_c5(h_cd, dx)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n

    if remaining() < 20 or best_h is None:
        if best_h is None:
            best_h = _make_alternating(45)
            best_n = 45
            best_c5 = _compute_c5(best_h, 2.0 / 45)
        return (best_h, best_c5, best_n)

    if remaining() > 30:
        _sa_time = remaining() * 0.7
        sa_h = _simulated_anneal(best_h, best_n, 2.0 / best_n,
                                 2000, 0.01, 0.001, rng, _sa_time)
        if sa_h is not None:
            sa_h = _repair(sa_h, best_n)
            c5_sa = _compute_c5(sa_h, 2.0 / best_n)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = sa_h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
