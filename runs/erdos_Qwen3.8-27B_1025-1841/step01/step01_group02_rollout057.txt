import numpy as np
from scipy.optimize import minimize
import time

_OPT_T = 50.0
_OPT_N = 200
_OPT_DX = 0.01
_OPT_PEN = 500.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def random_binary_h(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    n_ones = n // 2
    diff = n_ones - int(np.round(np.sum(h)))
    if diff > 0:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx, min(diff, len(idx)), replace=False)] = 1.0
    elif diff < 0:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx, min(-diff, len(idx)), replace=False)] = 0.0
    return make_feasible(h, n)

def soft_obj(h):
    n = _OPT_N
    dx = _OPT_DX
    T = _OPT_T
    pen = _OPT_PEN
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    ls = np.log(np.sum(np.exp(shifted)))
    obj = (mc + ls / T) * dx
    s = np.sum(h) - n / 2.0
    obj += pen * s * s
    return obj

def soft_grad(h):
    n = _OPT_N
    dx = _OPT_DX
    T = _OPT_T
    pen = _OPT_PEN
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    h_pad = np.zeros(3 * n - 2)
    h_pad[n-1:2*n-1] = h
    A = np.correlate(h_pad, p, 'full')[2*n-2 : 3*n-2]
    B = np.correlate(h_pad, p[::-1], 'full')[2*n-2 : 3*n-2]
    grad = dx * (1.0 - A - B)
    s = np.sum(h) - n / 2.0
    grad += 2.0 * pen * s
    return grad

def con_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def con_jac(h):
    return np.ones(_OPT_N)

def run(seed=42, budget_s=1300, **kwargs):
    global _OPT_T, _OPT_N, _OPT_DX, _OPT_PEN
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = random_binary_h(200, rng)
    h0 = make_feasible(h0, len(h0))
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    phases = [
        (160, [10, 30, 80], 4, 300.0, 1200),
        (320, [30, 80, 200], 4, 800.0, 1500),
        (480, [60, 150, 350], 3, 1500.0, 1800),
        (640, [100, 300, 500], 3, 3000.0, 2000),
        (800, [200, 400], 2, 5000.0, 2500),
    ]

    for n_pts, temps, n_restarts, pen_lbm, maxiter in phases:
        if budget_s - (time.time() - start_time) < 80:
            break
        start_pts = [upsample_h(best_h, n_pts), random_binary_h(n_pts, rng),
                     make_feasible(upsample_h(best_h, n_pts) + rng.randn(n_pts)*0.04, n_pts),
                     make_feasible(rng.rand(n_pts), n_pts)]
        for T in temps:

                _OPT_N = n_pts

                _OPT_T = T
                _OPT_PEN = pen_lbm

                                   bounds=bounds, options={'maxiter': maxiter,

                    h_opt = make_feasible(res.x, n_pts)

                        best_c5 = c5
                        best_h = h_opt.copy()

                except Exception:

                           bounds=bounds, constraints=cons,

            if c5 < best_c5:
                best_c5 = c5

        except Exception:
            pass

    # Higher temp SLSQP to sharpen
    if budget_s - (time.time() - start_time) > 60:
        n_pts = len(best_h)
        _OPT_N = n_pts
        _OPT_DX = 2.0 / n_pts
        _OPT_T = 800.0

            h_opt = make_feasible(res.x, n_pts)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        except Exception:
            pass

    # Coordinate descent polish
    n = len(best_h)
    dx = 2.0 / n
    improved = True
    pi = 0
    while improved and pi < 40 and (budget_s - (time.time() - start_time)) > 10:
        improved = False
        pi += 1
        for j in range(n):
            if (budget_s - (time.time() - start_time)) < 5:
                break
            ov = best_h[j]
            bc = best_c5
            bt = None
            for cand in [0.0, 1.0, 0.5, max(0.0, ov-0.05), max(0.0, ov-0.02),
                         max(0.0, ov-0.005), min(1.0, ov+0.005), min(1.0, ov+0.02),
                         min(1.0, ov+0.05)]:

                tr[j] = cand
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < bc - 1e-12:
                    bc = c
                    bt = tr
            if bt is not None:
                best_c5 = bc
                best_h = bt
                improved = True

    # Final
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    return best_h, best_c5, len(best_h)
