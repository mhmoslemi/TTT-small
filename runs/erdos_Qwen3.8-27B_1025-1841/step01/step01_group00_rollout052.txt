    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(len(corr)):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - (n - 1)
        i0 = max(0, sh)
        i1 = min(n, n + sh)
        if i0 < i1:
            grad[i0:i1] += pk * (1.0 - h[i0 - sh:i1 - sh])
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] -= pk * h[j0 + sh:j1 + sh]
    grad *= dx
    obj = dx * (mc + np.log(s) / tau)
    return obj, grad


def opt_smooth(h, n, tau, time_limit, lr_init=0.35):
    h = project_to_feasible(h.copy(), n)
    obj, grad = smooth_obj_grad(h, tau)
    best_obj, best_h = obj, h.copy()
    t_start = time.time()
    lr = lr_init
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    stall = 0
    while time.time() - t_start < time_limit:
        t += 1
        m1 = b1 * m1 + (1.0 - b1) * grad
        m2 = b2 * m2 + (1.0 - b2) * grad * grad
        step = lr * m1 / (1.0 - b1 ** t) / (np.sqrt(m2 / (1.0 - b2 ** t)) + eps)
        h = project_to_feasible(h - step, n)
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-13:
            best_obj, best_h = obj, h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 150:
            lr *= 0.7
            stall = 0
            if lr < 1e-6:
                break
    return best_h, best_obj


def opt_subgrad(h, n, time_limit, lr_init=0.003):
    h = project_to_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    v = np.zeros(n)
    beta = 0.92
    lr = lr_init
    stall = 0
    while time.time() - t_start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        ks = np.argmax(corr)
        c5 = corr[ks] * dx
        if c5 < best_c5 - 1e-13:
            best_c5, best_h = c5, h.copy()
            stall = 0
        else:
            stall += 1
        sh = ks - (n - 1)
        g = np.zeros(n)
        i0 = max(0, sh)
        i1 = min(n, n + sh)
        if i0 < i1:
            g[i0:i1] = 1.0 - h[i0 - sh:i1 - sh]
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            g[j0:j1] -= h[j0 + sh:j1 + sh]
        g *= dx
        v = beta * v - lr * g
        h = project_to_feasible(h + v, n)
        lr *= 0.9997
        if stall > 600:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break
    return best_h, best_c5


def opt_subgrad_nesterov(h, n, time_limit, lr_init=0.005):
    h = project_to_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    v = np.zeros(n)
    beta = 0.9
    lr = lr_init
    stall = 0
    y = h.copy()
    while time.time() - t_start < time_limit:
        corr = np.correlate(y, 1.0 - y, mode='full')
        ks = np.argmax(corr)
        sh = ks - (n - 1)
        g = np.zeros(n)
        i0 = max(0, sh)
        i1 = min(n, n + sh)
        if i0 < i1:
            g[i0:i1] = 1.0 - y[i0 - sh:i1 - sh]
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            g[j0:j1] -= y[j0 + sh:j1 + sh]
        g *= dx
        v_new = beta * v
        h_new = project_to_feasible(y + v_new - lr * g, n)
        v = v_new - lr * g
        y = h_new
        c5 = compute_c5(y)
        if c5 < best_c5 - 1e-13:
            best_c5, best_h = c5, y.copy()
            stall = 0
        else:
            stall += 1
        # Slowly decay lr
        lr = max(lr * 0.9998, 1e-7)
        if stall > 400:
            lr *= 0.5
            v *= 0.3
            stall = 0
    return best_h, best_c5


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    h = initial_h_values.copy()
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Smooth optimization with increasing tau (focus on max)
    taus = [12, 25, 50, 100, 200, 400]
    for tau in taus:
        if rem() < 70:
            break
        tl = min(rem() * 0.15, 180)
        h_opt, _ = opt_smooth(best_h, n, tau, tl, lr_init=0.3)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Phase 2: Subgradient polish (momentum)
    if rem() > 60:
        h_opt, c5 = opt_subgrad(best_h, n, rem() * 0.2, lr_init=0.002)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Phase 3: Nesterov subgradient
    if rem() > 60:
        h_opt, c5 = opt_subgrad_nesterov(best_h, n, rem() * 0.2, lr_init=0.004)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Phase 4: Lower resolution exploration for diversity
    if rem() > 120:
        n2 = max(200, n // 3)
        h2 = project_to_feasible(interp_periodic(best_h, n2), n2)
        for tau in [30, 100]:
            if rem() < 50:
                break
            h2, _ = opt_smooth(h2, n2, tau, rem() * 0.12, lr_init=0.4)
        if rem() > 40:
            h2, _ = opt_subgrad(h2, n2, rem() * 0.1, lr_init=0.005)
        h_back = project_to_feasible(interp_periodic(h2, n), n)
        c5_back = compute_c5(h_back)
        if c5_back < best_c5:
            best_c5, best_h = c5_back, h_back.copy()

    # Phase 5: Second round smooth at higher tau
    if rem() > 80:
        tl = min(rem() * 0.15, 150)
        h_opt, _ = opt_smooth(best_h, n, 300, tl, lr_init=0.15)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Phase 6: Final subgradient micro-polish
    if rem() > 30:
        h_opt, c5 = opt_subgrad(best_h, n, rem() * 0.35, lr_init=0.0008)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Phase 7: Tiny step Nesterov polish
    if rem() > 20:
        h_opt, c5 = opt_subgrad_nesterov(best_h, n, rem() * 0.5, lr_init=0.001)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    best_h = project_to_feasible(best_h, n)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
