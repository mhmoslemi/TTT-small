    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    lo, hi = -1.0, 1.0

            lo = mid
        else:

        else:

        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)

    """Circular cross-correlation using FFT."""
    A = np.fft.rfft(a, n)
    B = np.fft.rfft(b, n)
    return np.fft.irfft(A * np.conj(B), n)

    B = np.fft.irfft(np.fft.rfft(w_flip) * np.fft.rfft(h), n)

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def circ_corr_fft(a, b, n):
    A = np.fft.rfft(a, n)
    B = np.fft.rfft(b, n)
    return np.fft.irfft(A * np.conj(B), n)

def get_obj_grad(h, T, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = circ_corr_fft(h, g, n)
    scaled = T * dx * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    Z = np.sum(exp_s)
    w = exp_s / Z
    obj = dx / T * (m + np.log(Z))
    
    # Gradient
    w_flip = np.zeros(n)
    w_flip[0] = w[0]
    if n > 1:
        w_flip[1:] = w[:0:-1]
    
    A = np.fft.irfft(np.fft.rfft(w) * np.fft.rfft(g), n)
    B = np.fft.irfft(np.fft.rfft(w_flip) * np.fft.rfft(h), n)
    grad = dx * (A - B)
    return obj, grad

    lr = 0.5

                    yy = np.dot(y, y)

                    else:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def circ_corr_fft(a, b, n):
    A = np.fft.rfft(a, n)
    B = np.fft.rfft(b, n)
    return np.fft.irfft(A * np.conj(B), n)

def get_obj_grad(h, T, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = circ_corr_fft(h, g, n)
    scaled = T * dx * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    Z = np.sum(exp_s)
    w = exp_s / Z
    obj = dx / T * (m + np.log(Z))
    w_flip = np.zeros(n)
    w_flip[0] = w[0]
    if n > 1:
        w_flip[1:] = w[:0:-1]
    A = np.fft.irfft(np.fft.rfft(w) * np.fft.rfft(g), n)
    B = np.fft.irfft(np.fft.rfft(w_flip) * np.fft.rfft(h), n)
    grad = dx * (A - B)
    return obj, grad

def interp_to(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def grad_descent(h_init, n, target, T_list, iters_per_T, deadline):
    h = project_to_constraint(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    prev_h = h.copy()
    prev_grad = None
    lr = 0.5
    dx = 2.0 / n

    for T in T_list:
        for it in range(iters_per_T):
            if time.time() > deadline:
                return best_h, best_c5
            obj, grad = get_obj_grad(h, T, n)

            if prev_grad is not None:
                s = h - prev_h
                y = grad - prev_grad
                ys = np.dot(s, y)
                ss = np.dot(s, s)
                yy = np.dot(y, y)
                if ys > 1e-14 and ss > 1e-14 and yy > 1e-14:
                    lam1 = ss / ys
                    lam2 = np.dot(s, y) / yy
                    lr_new = np.clip(0.5 * (lam1 + lam2), 1e-8, 3.0)
                    lr = 0.7 * lr + 0.3 * lr_new

            prev_h = h.copy()
            prev_grad = grad.copy()

            alpha = lr
            h_new = project_to_constraint(h - alpha * grad, target)
            obj_new, _ = get_obj_grad(h_new, T, n)
            if obj_new > obj:
                for _ in range(15):
                    alpha *= 0.5
                    h_new = project_to_constraint(h - alpha * grad, target)
                    obj_new, _ = get_obj_grad(h_new, T, n)
                    if obj_new <= obj + 1e-10:
                        break

            h = h_new

            if it % 40 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    c5 = evaluate_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def coord_descent_polish(h, n, target, n_passes, deadline):
    h = project_to_constraint(h.copy(), target)
    best = evaluate_c5(h)
    order = np.random.permutation(n)
    for _ in range(n_passes):
        if time.time() > deadline:
            break
        for i in order:
            if time.time() > deadline:
                break
            for delta in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                if delta < 0 and h[i] + delta < 0:
                    continue
                if delta > 0 and h[i] + delta > 1:
                    continue
                h_old = h[i]
                h[i] += delta
                h_p = project_to_constraint(h, target)
                c5_p = evaluate_c5(h_p)
                if c5_p < best - 1e-12:
                    best = c5_p
                    h = h_p
                else:
                    h = project_to_constraint(h_p, target)
                    h[i] = h_old
    return h, best

def make_periodic_init(n, period, duty, target):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return project_to_constraint(h, target)

def make_cos_init(n, freq, phase, target):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.45 * np.cos(2 * np.pi * freq * x + phase)
    return project_to_constraint(h, target)
