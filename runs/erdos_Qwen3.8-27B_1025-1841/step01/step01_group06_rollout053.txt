            break
            h = h_new

    m = np.max(corr)
    logsum = np.log(np.sum(np.exp((corr - m) / temp)))
    return (m + temp * logsum) * dx

    temp = 0.01

        if it % 200 == 199:
            temp *= 0.9
            alpha *= 0.85
        
        g = smooth_max_grad(h_curr, n, temp, dx)
        # Projected gradient step
        h_new = h_curr - alpha * g
        h_new = project_target(h_new, n)

        else:

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(800):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        adj = (target - s) / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-15):
            h = h_new
            break
        h = h_new
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-14:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.arange(n_new, dtype=float)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def smooth_max_val(h, n, temp, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    logsum = np.log(np.sum(np.exp((corr - m) / temp)))
    return (m + temp * logsum) * dx

def smooth_max_grad(h, n, temp, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    p = np.exp((corr - m) / temp)
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        return np.zeros(n)
    p /= p_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        tau = k - (n - 1)
        if tau >= 0:
            rem = n - tau
            if rem > 0:
                grad[tau:] += p[k] * (1.0 - h[:rem])
                grad[:rem] -= p[k] * h[tau:]
        else:
            s = -tau
            rem = n - s
            if rem > 0:
                grad[:rem] += p[k] * (1.0 - h[s:])
                grad[s:] -= p[k] * h[:rem]
    return grad * dx

def pgd_optimize(h, n, dx, max_iter=5000, t0=time.time(), t_limit=200):
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr, dx)
    best_h = h_curr.copy()
    temp = 0.008
    alpha = 0.08 / n
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > t_limit:
            break
        if it % 300 == 299:
            temp *= 0.92
            alpha *= 0.8
        g = smooth_max_grad(h_curr, n, temp, dx)
        gnrm = np.linalg.norm(g)
        if gnrm < 1e-14:
            break
        h_new = h_curr - alpha * g
        h_new = project_target(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new
            stall = 0
            alpha = min(alpha * 1.03, 0.8 / n)
        else:
            stall += 1
            if stall > 100:
                alpha *= 0.6
                stall = 0
    return best_h, best_c5

def slsqp_optimize(h, n, dx, temp, maxiter=300, t0=time.time(), t_limit=60):
    target = n / 2.0
    def obj(x):
        return smooth_max_val(np.clip(x, 0, 1), n, temp, dx)
    def obj_grad(x):
        return smooth_max_grad(np.clip(x, 0, 1), n, temp, dx)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if np.isfinite(res.fun):
            return project_target(res.x, n)
    except Exception:
        pass
    return h

    rng = np.random.RandomState(seed)

        dx = 2.0 / n

            c5_new = compute_c5(h_new, dx)

                c5_curr = c5_new

        # PGD with adaptive step
        t_pg = time.time()
        pg_limit = 150 if n <= 280 else 200
        h_pg, c5_pg = pgd_optimize(h_curr, n, dx, max_iter=8000, t0=t_pg, t_limit=pg_limit)
