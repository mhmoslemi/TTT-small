                break

        h = np.clip(h, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_smoothed(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    return max_c + np.log(np.sum(np.exp(beta * (corr - max_c)))) / beta

def c5_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    weights = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-10:
            continue
        s = m - (n - 1)

        j1s = max(0, s)
        j1e = min(n, n + s)  # exclusive end, since min(n-1, n-1+s)+1
