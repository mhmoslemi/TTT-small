def compute_weighted_grad(h, weights):
    """Compute gradient of weighted sum of f(k) values."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # s from 0 to n-1 (positive shifts)
    for s in range(n):
        w = weights[n - 1 + s]
        if abs(w) < 1e-14:
            continue
        if s == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        else:
            grad[s:] += w * dx * (1.0 - h[:n - s])  # (1-h[j-s]) for j>=s
            grad[:n - s] -= w * dx * h[s:]  # -h[j+s] for j<n-s

        if abs(w) < 1e-14:
            continue
        grad[:n - sp] += w * dx * (1.0 - h[sp:])  # (1-h[j+sp])
        grad[sp:] -= w * dx * h[:n - sp]  # -h[j-sp]
    return grad

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_weights(f_vals, T):
    shifted = f_vals - np.max(f_vals)
    exp_vals = np.exp(T * shifted)
    total = np.sum(exp_vals)
    if total < 1e-300 or not np.isfinite(total):
        return np.ones(len(f_vals)) / len(f_vals)
    return exp_vals / total

def compute_weighted_grad(h, weights_full):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(n):
        w = weights_full[n - 1 + s]
        if abs(w) < 1e-14:
            continue
        if s == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        else:
            grad[s:] += w * dx * (1.0 - h[:n - s])
            grad[:n - s] -= w * dx * h[s:]
    for sp in range(1, n):
        w = weights_full[n - 1 - sp]
        if abs(w) < 1e-14:
            continue
        grad[:n - sp] += w * dx * (1.0 - h[sp:])
        grad[sp:] -= w * dx * h[:n - sp]
    return grad

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def gradient_descent(h, n_iters, T_start, T_end, lr_start, lr_end, momentum=0.85):
    n = len(h)
    v = np.zeros(n)
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        T = T_start + (T_end - T_start) * frac
        lr = lr_start * ((lr_end / lr_start) ** frac)

            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c5

def _slsqp_objective(h, n_opt, t_opt):
    dx = 2.0 / n_opt
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(t_opt * shifted))) / t_opt
    return soft * dx

def _slsqp_obj(h, n_opt, t_opt):
    return _slsqp_objective(h, n_opt, t_opt)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        h0 = (rng.rand(320) < 0.5).astype(float)
        h0 = project_feasible(h0, 320)

    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    elapsed = time.time() - start_time

    # Phase 1: Gradient descent at original resolution
    if budget_s - elapsed > 60:
        h_gd, c_gd = gradient_descent(best_h, 800, 25.0, 120.0, 0.4, 0.05, momentum=0.85)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd
        elif c_gd < best_c5 + 0.005:
            best_h = h_gd

    elapsed = time.time() - start_time

    # Phase 2: SLSQP polish at current resolution
    if budget_s - elapsed > 80 and len(best_h) <= 500:
        n_pts = len(best_h)
        for T_slsqp in [60.0, 150.0]:
            if budget_s - (time.time() - start_time) < 40:
                break
            bounds = [(0.0, 1.0)] * n_pts
            cons = [{'type': 'eq', 'fun': lambda x, n=n_pts: np.sum(x) - n / 2.0,
                     'jac': lambda x, n=n_pts: np.ones(n)}]
            try:
                res = minimize(_slsqp_obj, best_h, args=(n_pts, T_slsqp), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 600, 'ftol': 1e-12})
                c_try = compute_c5(res.x)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = project_feasible(res.x, n_pts)
            except Exception:
                pass

    elapsed = time.time() - start_time

    # Phase 3: Perturbation + re-optimize cycles
    if budget_s - elapsed > 200:
        for pert_size in [0.03, 0.08, 0.15]:
            if budget_s - (time.time() - start_time) < 80:
                break
            h_pert = best_h + rng.randn(len(best_h)) * pert_size
            h_pert = project_feasible(h_pert, len(best_h))
            h_new, c_new = gradient_descent(h_pert, 500, 30.0, 150.0, 0.3, 0.03, momentum=0.85)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    elapsed = time.time() - start_time

    # Phase 4: Upsample to finer resolution and refine
    if budget_s - elapsed > 150:
        n_fine = 640
        h_fine = upsample_h(best_h, n_fine)
        h_fine, c_fine = gradient_descent(h_fine, 600, 30.0, 120.0, 0.25, 0.03, momentum=0.85)
        if c_fine < best_c5:
            best_c5 = c_fine
            best_h = h_fine

        if budget_s - (time.time() - start_time) > 60:
            n_pts_f = len(best_h)
            if n_pts_f <= 700:
                for T_slsqp in [80.0]:
                    bounds = [(0.0, 1.0)] * n_pts_f
                    cons = [{'type': 'eq', 'fun': lambda x, n=n_pts_f: np.sum(x) - n / 2.0,
                             'jac': lambda x, n=n_pts_f: np.ones(n)}]
                    try:
                        res = minimize(_slsqp_obj, best_h, args=(n_pts_f, T_slsqp), method='SLSQP',
                                       bounds=bounds, constraints=cons,
                                       options={'maxiter': 400, 'ftol': 1e-12})
                        c_try = compute_c5(res.x)
                        if c_try < best_c5:
                            best_c5 = c_try
                            best_h = project_feasible(res.x, n_pts_f)
                    except Exception:
                        pass

    elapsed = time.time() - start_time

    # Phase 5: Coordinate descent polish
    if budget_s - elapsed > 15:
        n = len(best_h)
        improved = True
        rounds = 0
        while improved and rounds < 25:
            improved = False
            rounds += 1
            if time.time() - start_time > budget_s - 8:
                break
            for j in range(n):
                if time.time() - start_time > budget_s - 5:
                    break
                ov = best_h[j]
                cands = [0.0, 1.0, 0.5, ov + 0.02, ov - 0.02,
                         ov + 0.01, ov - 0.01, ov + 0.05, ov - 0.05]
                for cand in cands:
                    if cand < 0.0 or cand > 1.0:
                        continue
                    trial = best_h.copy()
                    trial[j] = cand
                    trial = project_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-12:
                        best_c5 = c
                        best_h = trial.copy()
                        improved = True
                        break

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
