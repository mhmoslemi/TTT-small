def project(h, n):

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def grad_active(h, n, top_k=30):
    """Compute gradient of sum of top_k active correlation values (times dx)."""
    dx = 2.0 / n
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for m in top_idx:
        offset = m - (n - 1)
        g = np.zeros(n)
        if offset >= 0:
            if n - offset > 0:
                g[:n-offset] -= h[offset:]
            if offset < n:
                g[offset:] += one_mh[:n-offset]
        else:
            p = -offset
            if p < n:
                g[p:] += one_mh[:n-p]
            if n - p > 0:
                g[:n-p] -= h[p:]
        grad += g
    return grad * dx

def subgradient_h(h, n, top_k, lr0, n_steps, deadline, verbose=False):
    """Projected subgradient descent on active constraints."""
    h = h.copy()
    best = evaluate_c5(h)
    best_h = h.copy()
    lr = lr0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = grad_active(h, n, top_k)
        
        # Adaptive step size: reduce if no improvement
        h_new = project(h - lr * g, n)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best:
            best = c5_new
            best_h = h_new.copy()
            h = h_new
            # Maybe increase lr slightly

            if lr < 1e-6:

                lr = lr0
                h = project(h + np.random.randn(n) * 0.01, n)

def softmax_obj(h, n, T, top_k=50):
    """Smooth approximation: softmax of top_k correlations."""
    dx = 2.0 / n
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full') * T
    # Use all for smoothness, or just top_k
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(corr - m)))) / T
    return float(val * dx)

def slsqp_smooth(h0, n, T, maxiter, deadline):
    target = n / 2.0
    def f(x):
        return softmax_obj(x, n, T)
    try:

                       bounds=[(0.0, 1.0)] * n,

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def grad_active(h, n, top_k=40):
    dx = 2.0 / n
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in top_idx:
        offset = m - (n - 1)
        if offset >= 0:
            lo = n - offset
            if lo > 0:
                grad[:lo] -= h[offset:]
                grad[offset:] += one_mh[:lo]
        else:
            p = -offset
            if p < n:
                lo = n - p
                grad[p:] += one_mh[:lo]
                grad[:lo] -= h[p:]
    return grad * dx

def subgradient_opt(h, n, top_k, lr0, n_steps, deadline):
    h = h.copy().astype(float)
    best = evaluate_c5(h)
    best_h = h.copy()
    lr = lr0
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = grad_active(h, n, top_k)

        h = best_local_h

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * T
    m = np.max(corr)
    return float((m + np.log(np.sum(np.exp(corr - m)))) / T * dx)
