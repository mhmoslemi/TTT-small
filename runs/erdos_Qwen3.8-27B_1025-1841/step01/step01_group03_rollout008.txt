```python
import numpy as np
import time

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (corr * dx).max()

def swap_move(h, i, j):
    """Swap: set h[i]=0, h[j]=1 (i currently 1, j currently 0)."""
    h[i] = 0
    h[j] = 1

def local_search_binary(h, rng, max_time):
    """Simulated annealing for binary h with fixed sum."""
    n = len(h)
    t_end = time.time() + max_time
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    T = 0.01  # initial temperature
    T_min = 1e-6
    cooling = 0.9995
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    while time.time() < t_end:
        # Pick a random 1 and a random 0 to swap
        ia = rng.integers(len(ones_list))
        ib = rng.integers(len(zeros_list))
        a = ones_list[ia]
        b = zeros_list[ib]
        
        # Make the swap
        h[a] = 0
        h[b] = 1
        
        # Evaluate
        new_c5 = compute_c5(h)
        
        # Accept?
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / T):
            # Accept
            ones_list[ia] = b
            zeros_list[ib] = a
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[a] = 1
            h[b] = 0
        
        T *= cooling
        
        # If temperature too low, restart
        if T < T_min:
            T = 0.01
            # Maybe perturb
            for _ in range(n // 10):
                ia = rng.integers(len(ones_list))
                ib = rng.integers(len(zeros_list))
                a = ones_list[ia]
                b = zeros_list[ib]
                h[a] = 0
                h[b] = 1
                ones_list[ia] = b
                zeros_list[ib] = a
            current_c5 = compute_c5(h)
    
    return best_h, best_c5
```

Let me use a different approach. The total "disagreement" over all lags:

For a 0-1 sequence, each pair (i,j) with i≠j contributes to exactly one lag k=j-i, and contributes to the disagreement if h[i]≠h[j]. Total disagreements = #{(i,j): i<j, h[i]≠h[j]} * 2 (each pair counted twice, once in each direction).

```python
    t0 = time.time()
    t_hard_end = t0 + budget_s - 20

    best_c5 = float('inf')
    best_h = None
    best_n = 80

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        pc5 = compute_c5(parent)
        best_c5 = pc5
        best_h = parent.copy()
        best_n = len(parent)
    except Exception:
        pass

    configs = [
        (80, 0.12), (100, 0.10), (120, 0.12), (160, 0.14),
        (200, 0.14), (240, 0.14), (320, 0.14), (400, 0.10)
    ]

    for n, frac in configs:
        remaining = t_hard_end - time.time()
        if remaining < 40:
            break
        alloc = min(frac * budget_s, remaining - 30)
        if alloc < 25:
            continue
        t_limit = time.time() + alloc

        n_runs = max(2, int(alloc / 90))
        per_run = (alloc - 10) / n_runs

        for r in range(n_runs):
            if time.time() > t_limit - 5:
                break
            if parent is not None and r == 0:
                init = parent
            else:
                init = None
            h0 = make_binary(rng, n, init)
            c5_0 = compute_c5(h0)
            if c5_0 < best_c5:
                best_c5 = c5_0
                best_h = h0.copy()
                best_n = n

            sa_time = max(5, per_run * 0.6)
            h_sa, c5_sa = sa_optimize(h0, rng, time.time() + sa_time, T0=0.02, cooldown=0.99996)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n

            if time.time() < t_limit - 8:
                hc_time = min(t_limit - time.time() - 3, per_run * 0.3)
                h_hc, c5_hc = hill_climb(h_sa, rng, time.time() + hc_time)
                if c5_hc < best_c5:
                    best_c5 = c5_hc
                    best_h = h_hc.copy()
                    best_n = n

    if best_h is None:
        n = 80
        best_h = make_binary(rng, n, parent)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final polish on the best solution found
    if time.time() < t_hard_end - 5 and best_h is not None:
        remaining = t_hard_end - time.time() - 3
        if remaining > 10:
            h_fin, c5_fin = hill_climb(best_h, rng, time.time() + min(remaining, 60), max_rounds=50)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin

    return best_h, best_c5, best_n
```
