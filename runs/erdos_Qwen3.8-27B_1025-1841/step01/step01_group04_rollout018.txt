            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj_and_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    obj = (m + np.log(w_sum)) / T
    w = w / w_sum  # normalized weights, sum to 1
    
    # Gradient
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        m_lag = lag_idx - (n - 1)
        wi = w[lag_idx]
        if wi < 1e-14:
            continue
        # term1: (1-x[j+m]) for valid j
        if m_lag >= 0:
            lo1, hi1 = 0, n - m_lag
        else:
            lo1, hi1 = -m_lag, n
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - x[lo1 + m_lag:hi1 + m_lag])
        # term2: -x[j-m] for valid j
        if m_lag >= 0:
            lo2, hi2 = m_lag, n
        else:
            lo2, hi2 = 0, n + m_lag
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * x[lo2 - m_lag:hi2 - m_lag]
    grad *= dx
    return obj, grad

def slsqp_with_grad(h0, n, T, maxiter=200, deadline=None):
    def f(x):
        obj, _ = soft_max_obj_and_grad(x, n, T)
        return obj
    def g(x):
        _, grad = soft_max_obj_and_grad(x, n, T)
        return grad
    def c_eq(x):
        return np.sum(x) - n / 2.0
    def c_eq_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                       options={'ftol': 1e-16, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def binary_local_search(h, n, max_rounds=50, deadline=None):
    # Round to binary
    hb = (h > 0.5).astype(float)
    # Fix sum
    target = n // 2
    ones = np.sum(hb == 1)
    if ones > target:
        # Remove excess ones (those closest to 0.5 in original h)
        idx = np.where(hb == 1)[0]
        idx_sorted = idx[np.argsort(h[idx])]  # closest to 0 first
        for i in range(int(ones) - target):
            hb[idx_sorted[i]] = 0
    elif ones < target:
        idx = np.where(hb == 0)[0]
        idx_sorted = idx[np.argsort(-h[idx])]  # closest to 1 first
        for i in range(target - int(ones)):
            hb[idx_sorted[i]] = 1
    
    best_corr = np.max(np.correlate(hb, 1-hb, mode='full'))
    
    for _round in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Try single flips
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            old_val = hb[i]
            new_val = 1.0 - old_val
            hb[i] = new_val
            corr = np.max(np.correlate(hb, 1-hb, mode='full'))
            if corr < best_corr - 1e-12:
                best_corr = corr
                improved = True
            else:
                hb[i] = old_val
        if not improved:
            break
    return hb, best_corr * (2.0/n)

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

Wait, I need to check the budget. With n=336:
- SLSQP with analytical gradient: each function+gradient eval is O(n²) ≈ 112K ops. SLSQP might do 300 iterations, each needing a few function/gradient evals. So ~1M ops per SLSQP call. Fast!
- Binary local search: each round tries n=336 flips, each requiring a full correlation O(n²) = 112K. So 336*112K ≈ 38M per round. 80 rounds = 3B. This is too slow!

I need to optimize the binary local search. Instead of recomputing the full correlation after each flip, I can update it incrementally.

When we flip bit at position i (from 0 to 1 or 1 to 0), how does the correlation change?
c[m] = sum_j h[j]*(1-h[j+m]) changes.

If h[i] changes from 0 to 1:
- For the term where j=i: h[i]*(1-h[i+m]) was 0, now is (1-h[i+m]). Change: +(1-h[i+m])
- For the term where j+m=i, i.e., j=i-m: h[i-m]*(1-h[i]) was h[i-m]*1, now is h[i-m]*0. Change: -h[i-m]

So Δc[m] = (1-h[i+m]) - h[i-m] for each lag m (with boundary conditions).

This means I can update the correlation in O(n) per flip instead of O(n²)!

def binary_flip_search(hb, n, max_rounds=200, deadline=None):
    target = n // 2
    assert int(np.sum(hb)) == target
    
    corr = np.correlate(hb, 1-hb, mode='full')
    best_max = np.max(corr)
    
    for round_num in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            # Try flipping bit i
            old = hb[i]
            delta = 1.0 - 2.0 * old  # +1 if 0->1, -1 if 1->0
            
            # Update correlation incrementally
            # For lag m: Δc[m] = delta * (1-hb[i+m]) - delta * hb[i-m] (with bounds)
            # m ranges from -(n-1) to (n-1), index = m + (n-1)
            
            # term1: delta * (1-hb[i+m]) for 0<=i+m<n => m in [-i, n-1-i]
            m_lo1 = max(-(n-1), -i)
            m_hi1 = min(n-1, n-1-i)
            if m_hi1 >= m_lo1:
                corr[m_lo1+(n-1):m_hi1+(n-1)+1] += delta * (1.0 - hb[i+m_lo1:i+m_hi1+1])
            
            # term2: -delta * hb[i-m] for 0<=i-m<n => m in [i-(n-1), i]
            m_lo2 = max(-(n-1), i-(n-1))
            m_hi2 = min(n-1, i)
            if m_hi2 >= m_lo2:
                corr[m_lo2+(n-1):m_hi2+(n-1)+1] -= delta * hb[i-m_lo2:i-m_hi2+1]
            
            new_max = np.max(corr)
            if new_max < best_max - 1e-12:
                hb[i] = 1.0 - old
                best_max = new_max
                improved = True
            else:
                # Undo
                if m_hi1 >= m_lo1:
                    corr[m_lo1+(n-1):m_hi1+(n-1)+1] -= delta * (1.0 - hb[i+m_lo1:i+m_hi1+1])
                if m_hi2 >= m_lo2:
                    corr[m_lo2+(n-1):m_hi2+(n-1)+1] += delta * hb[i-m_lo2:i-m_hi2+1]
        
        if not improved:
            break
    
    return hb, best_max * (2.0/n)
