            t_i = max(2.0, min(t_per_init, budget_s - (time.time() - start_time) - 60))
            if t_i < 1:
                break
            h_opt, c5_opt = optimize(init_h, n, t_i, rng)
            if c5_opt < best_local_c5:
                best_local_c5 = c5_opt
                best_local_h = h_opt

        if best_local_h is not None and best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h
            best_n = n
            # Extra polish on best
            remaining2 = budget_s - (time.time() - start_time)
            if remaining2 > 40:
                h_opt, c5_opt = optimize(best_h, best_n, min(60, remaining2 - 30), rng)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    if best_h is None:
        n = 64
        best_h = random_binary(n, rng)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
