import numpy as np

def _c5(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _make_binary(h, target, rng):
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    while s > target:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx)] = 0
        s -= 1
    while s < target:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx)] = 1
        s += 1
    return h
