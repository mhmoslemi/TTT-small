    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    """Project h to be feasible: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target = n / 2.0
    diff = target - current_sum
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            # Need to add mass; add to positions with room
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 0:
                h += diff * room / total_room
        else:
            # Need to remove mass
            space = h
            total_space = np.sum(space)
            if total_space > 0:
                h += diff * space / total_space
    h = np.clip(h, 0, 1)
    # Final adjustment
    h[np.argmax(h)] += target - np.sum(h)
    h = np.clip(h, 0, 1)
    return h
