    """Return (c5, gradient) using subgradient at argmax shift"""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    m_star = np.argmax(corr)
    c5_val = corr[m_star] * dx
    n = len(h)
    shift = m_star - (n - 1)
    
    grad = np.zeros(n)
    # grad[j] = dx * [(1-h[j+shift]) if 0<=j+shift<n else 0 - (h[j-shift] if 0<=j-shift<n else 0)]
    j_range = np.arange(n)
    jp = j_range + shift
    jm = j_range - shift
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    
    grad[valid_p] += (1.0 - h[jp[valid_p]])
    grad[valid_m] -= h[jm[valid_m]]
    grad *= dx
    return c5_val, grad
