import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time


def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = fftconvolve(h, (1.0 - h)[::-1], mode="full")
    return dx * np.max(corr)


def soft_max(arr, temp):
    scaled = arr / temp
    m = np.max(scaled)
    return temp * (m + np.log(np.sum(np.exp(scaled - m))))


def project_to_constraint(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def optimize_from_start(h_start, n, dx, target_sum, temp, maxiter, time_limit):

        if time.time() - t_start > time_limit:

        h = np.clip(x, 0.0, 1.0)

        h_new = project_to_constraint(res.x, target_sum)
        return h_new
    except (TimeoutError, Exception):
        return None
