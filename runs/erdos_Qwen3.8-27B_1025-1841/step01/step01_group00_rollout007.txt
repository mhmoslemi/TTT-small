import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 3.0
    hi = g.max() + 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = tau * corr * dx
    a_max = a.max()
    exp_a = np.exp(a - a_max)
    s = exp_a.sum()
    p = exp_a / s
    obj = (a_max + np.log(s)) / tau

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        j0a = max(k, 0)
        j1a = min(n + k, n)
        if j0a < j1a:
            idx = np.arange(j0a, j1a)
            grad[idx] += pk * (1.0 - h[idx - k])
        j0b = max(0, -k)
        j1b = min(n, n - k)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + k]
    grad *= dx
    return obj, grad


def adam_smooth(h, n, tau, max_iters, time_limit, lr):
    h = project_to_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    v_s = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    stall = 0
    start = time.time()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t_count += 1
        obj, grad = smooth_obj_and_grad(h, tau)
        m = beta1 * m + (1.0 - beta1) * grad
        v_s = beta2 * v_s + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v_s / (1.0 - beta2 ** t_count)
        h_raw = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_raw, n)

        if it % 50 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 40 and lr > 1e-6:
                lr *= 0.7
                stall = 0

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def subgrad_optimize(h, n, max_iters, time_limit, lr_init):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.8
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 1:
            top_k = np.argsort(corr)[-3:]
            sh_target = int(top_k[0]) - (n - 1)
        else:
            sh_target = int(np.argmax(corr)) - (n - 1)

        grad = np.zeros(n)
        j0 = max(0, sh_target)
        j1 = min(n, n + sh_target)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += (1.0 - h[idx - sh_target])
        j0b = max(0, -sh_target)
        j1b = min(n, n - sh_target)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= h[idx + sh_target]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

        if stall > 500:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-15:
                break
    return best_h, best_c5

        n_flip = rng.randint(1, 5)
        idxs = rng.choice(n, size=n_flip, replace=False)
        for idx in idxs:
            h_try[idx] = 1.0 - h_try[idx]
        h_try = project_to_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_start = initial_h_values.copy()
    best_c5 = compute_c5(h_start)
    best_h = h_start.copy()
    best_n = len(h_start)

    # Phase 1: n=2000, multi-tau Adam
    if remaining() > 120:
        n1 = 2000
        h1 = interp_periodic(h_start, n1)
        h1 = project_to_feasible(h1, n1)
        tau_schedule = [8, 20, 50, 100, 200]
        t_budget = min(remaining() * 0.35, 400.0)
        for tau in tau_schedule:
            if remaining() < 30:
                break
            iters = int(t_budget / len(tau_schedule) / max(0.02, (2 * n1 - 1) * 3e-6))
            iters = max(iters, 500)
            h1, c5_1 = adam_smooth(h1, n1, tau, iters, remaining() * 0.2, lr=0.003)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()
                best_n = n1

    # Phase 2: n=4000, fewer tau stages
    if remaining() > 120:
        n2 = 4000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tau_schedule2 = [10, 40, 100]
        for tau in tau_schedule2:
            if remaining() < 30:
                break
            iters = max(500, int(min(remaining() * 0.15, 60.0) / ((2 * n2 - 1) * 5e-6)))
            h2, c5_2 = adam_smooth(h2, n2, tau, iters, remaining() * 0.15, lr=0.002)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

    # Phase 3: Subgradient refinement
    if remaining() > 60:
        h3 = best_h.copy()
        h3, c5_3 = subgrad_optimize(h3, len(h3), 200000, remaining() * 0.25, lr_init=0.005)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Local perturbation search
    if remaining() > 60:
        h4, c5_4 = local_perturb(best_h, len(best_h), rng, n_trials=500)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Second subgrad pass with smaller lr
    if remaining() > 30:
        h5 = best_h.copy()
        h5, c5_5 = subgrad_optimize(h5, len(h5), 200000, remaining() * 0.3, lr_init=0.002)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Final tiny subgrad polish
    if remaining() > 15:
        h6 = best_h.copy()
        h6, c5_6 = subgrad_optimize(h6, len(h6), 100000, remaining() * 0.3, lr_init=0.0005)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final feasibility and evaluation
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    final_n = len(best_h)

    # Also check the original resolution as fallback
    if final_n != best_n:
        h_check = project_to_feasible(best_h, len(best_h))
        c5_check = compute_c5(h_check)
        if c5_check < final_c5:
            final_c5 = c5_check
            best_h = h_check

    return best_h, final_c5, len(best_h)
