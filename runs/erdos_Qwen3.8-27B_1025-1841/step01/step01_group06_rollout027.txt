def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if len(free) == 0:
            break
        step = diff / len(free)
        h[free] = np.clip(h[free] + step, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_onto_constraint(h.copy(), n_new)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_constraint(h_new, n_new)

def subgrad_at_lag(h, n, s):
    d = abs(s)
    if d >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    m = n - d
    if s >= 0:
        grad[:m] += (1.0 - h[d:])
        grad[d:] -= h[:m]
    else:
        grad[d:] += (1.0 - h[:m])
        grad[:m] -= h[d:]
    return grad

def smooth_max_val(x, dx, temp):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx

def smooth_max_grad(x, n, dx, temp):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

        if w < 1e-10:
            continue
        s = k - (n - 1)
        grad += w * subgrad_at_lag(h, n, s)
    return grad * dx

    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    smooth = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    viol = np.sum(h) - n / 2.0
    return smooth * dx + penalty * viol * viol

def penalized_grad(x, n, dx, temp, penalty):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp((corr - m) / temp)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if w < 1e-10:

        s = k - (n - 1)
        grad += w * subgrad_at_lag(h, n, s)
    grad *= dx
    viol = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * viol
    return grad
