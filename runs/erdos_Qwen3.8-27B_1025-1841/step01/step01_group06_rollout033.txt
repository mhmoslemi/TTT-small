            grad[t:n] += w * g[0:n-t]
        
        # Second term: -h[j-s] for j where 0≤j<n and 0≤j-s<n

        else:

            t = -s
            grad[0:n-t] -= w * h[t:n]
    
    return grad * dx  # scale by dx since we're optimizing the integral

def project_target(h, n):

                best_h = h_c.copy()
                best_n = n_c

        # Targeted local search
        if time.time() < deadline:
            h_tl, c5_tl = targeted_local_search(h_c, n_c, dx_c, 30000, rng, deadline)
            if c5_tl < c5_c:
                h_c = h_tl
                c5_c = c5_tl
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                best_n = n_c

    # Phase D: Final refinement on best solution
    remaining = deadline - time.time()
    if remaining > 30:
        n_f = best_n
        dx_f = 2.0 / n_f
        h_f = best_h.copy()
        c5_f = compute_c5(h_f, dx_f)

        # Brief gradient polish
        if n_f <= 560:
            for it in range(2000):
                if time.time() > deadline:
                    break
                grad = grad_smooth_max(h_f, n_f, dx_f, 0.01)
                h_new = h_f - 0.005 * grad
                h_new = project_target(h_new, n_f)
                c5_new = compute_c5(h_new, dx_f)
                h_f = h_new
                if c5_new < c5_f:
                    c5_f = c5_new

        # Brief targeted search
        if time.time() < deadline:
            h_tl, c5_tl = targeted_local_search(h_f, n_f, dx_f, 20000, rng, deadline)
            if c5_tl < c5_f:
                h_f = h_tl
                c5_f = c5_tl

        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n_f

    best_h = project_target(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
