import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def project_feasible(h, n):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2} using bisection on lambda."""
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(h - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
    return np.clip(x, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

defsmooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    max_s = scaled.max()
    return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / T

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    max_s = scaled.max()
    exps = np.exp(scaled - max_s)
    p = exps / exps.sum()
    
    # FFT-based gradient
    size = 1
    while size < 4 * n:
        size *= 2
    P = np.fft.rfft(p, size)
    H = np.fft.rfft(h, size)
    p_rev = p[::-1]
    PR = np.fft.rfft(p_rev, size)
    
    conv_A = np.fft.irfft(P * H, size)
    conv_B = np.fft.irfft(PR * H, size)
    
    offset = n - 1
    A = conv_A[offset:offset + n]
    B = conv_B[offset:offset + n]
    
    # grad of F = (1/T)*log(sum(exp(T*dx*corr))) w.r.t. h[j]
    # = dx * (1 - A[j] - B[j])
    obj = (max_s + np.log(exps.sum())) / T
    grad = dx * (1.0 - A - B)
    return obj, grad

def pgd_optimize(h, T_init, n_iter, budget_s, start_time, lr_init=5.0):
    """Projected gradient descent with adaptive parameters."""
    n = len(h)
    h = h.copy()
    T = T_init
    lr = lr_init
    best_local_c5 = compute_c5(h)
    
    for it in range(n_iter):
        if time.time() - start_time > budget_s - 20:
            break
        
        obj, grad = compute_grad(h, T)
        
        # Backtracking line search
        improved = False

            alpha = lr * factor
            h_trial = project_feasible(h - alpha * grad, n)
            obj_trial = smooth_obj(h_trial, T)
            if obj_trial < obj - 1e-12:

                improved = True
                break

            if lr < 0.1:

        lr *= 0.9995
        
        if it % 50 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_local_c5:
                best_local_c5 = c5_now
    
    return h, best_local_c5

def coord_descent_polish(h, max_sweeps=100):
    n = len(h)
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            best_val = old
            best_c = best_c5
            candidates = [0.0, 1.0, 0.5,
                         old + 0.01, old - 0.01,
                         old + 0.005, old - 0.005,
                         old + 0.02, old - 0.02,
                         old + 0.05, old - 0.05]
            for c in candidates:

                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                c_c5 = compute_c5(trial)
                if c_c5 < best_c - 1e-12:
                    best_c = c_c5
                    best_val = c
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    
    return h, best_c5

def compute_grad(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    max_s = scaled.max()

    p = exps / exps.sum()
    obj = (max_s + np.log(exps.sum())) / T
    
    # FFT size
    size = 1
    while size < 4 * n:
        size *= 2
    
    # A[j] = (p * h)[j+n-1]
    P = np.fft.rfft(p, size)
    H = np.fft.rfft(h, size)
    conv_ph = np.fft.irfft(P * H, size)
    A = conv_ph[n-1 : 2*n-1]  # positions n-1 to 2n-2

    q = p[::-1]

    S1 = np.array([p_ext[j:j+n].sum() for j in range(n)])

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(h - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
    return np.clip(x, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    max_s = scaled.max()
    return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / T

def compute_grad_normalized(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    max_s = scaled.max()
    exps = np.exp(scaled - max_s)
    p = exps / exps.sum()
    obj = (max_s + np.log(exps.sum())) / T

    size = 1
    while size < 4 * n:
        size *= 2

    P = np.fft.rfft(p, size)
    H = np.fft.rfft(h, size)
    conv_ph = np.fft.irfft(P * H, size)
    A = conv_ph[n - 1:2 * n - 1]

    q = p[::-1]
    Q = np.fft.rfft(q, size)
    conv_qh = np.fft.irfft(Q * H, size)
    B = conv_qh[n - 1:2 * n - 1]

    p_ext = np.zeros(3 * n - 2)
    p_ext[:2 * n - 1] = p
    p_cum = np.concatenate([[0.0], np.cumsum(p_ext)])
    S1 = p_cum[n:n + n] - p_cum[:n]

    grad_norm = S1 - A - B
    return obj, grad_norm, p

def pgd_optimize(h, T_init, n_iter, budget_s, start_time, lr_scale=1.0):
    n = len(h)
    h = h.copy()
    T = T_init
    best_local_c5 = compute_c5(h)
    lr = 0.05 * lr_scale

    for it in range(n_iter):
        if time.time() - start_time > budget_s - 25:
            break

        obj, grad_norm, p = compute_grad_normalized(h, T)
        step_dir = -grad_norm

        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.04]:
            alpha = lr * factor
            h_trial = project_feasible(h + alpha * step_dir, n)
            obj_trial = smooth_obj_val(h_trial, T)
            if obj_trial < obj - 1e-14:
                h = h_trial
                improved = True
                break

        if not improved:
            lr *= 0.8
            if lr < 0.001:
                lr = 0.05 * lr_scale
                T *= 1.01

        T *= 1.001
        lr *= 0.9998

        if it % 40 == 0 or it == n_iter - 1:
            c5_now = compute_c5(h)
            if c5_now < best_local_c5:
                best_local_c5 = c5_now

    return h, best_local_c5

def coord_descent_polish(h, max_sweeps=60, time_limit=None, start_time=None):
    n = len(h)
    best_c5 = compute_c5(h)
    h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - start_time > time_limit - 15:
            break
        improved = False
        for j in range(n):
            if time_limit is not None and time.time() - start_time > time_limit - 15:
                break
            old = h[j]
            best_val = old
            best_c = best_c5
            cands = [0.0, 1.0, 0.5,
                     old + 0.01, old - 0.01,
                     old + 0.02, old - 0.02,
                     old + 0.005, old - 0.005]
            for c in cands:
                if c < -0.001 or c > 1.001:
                    continue
                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                c_c5 = compute_c5(trial)
                if c_c5 < best_c - 1e-13:
                    best_c = c_c5
                    best_val = c
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
                no_improve = 0
            else:
                no_improve += 1
        if not improved:
            break
    return h, best_c5

def random_binary_h(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    n_ones = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n_ones:
        to_flip = rng.choice(ones, len(ones) - n_ones, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < n_ones:
        to_flip = rng.choice(zeros, n_ones - len(ones), replace=False)
        h[to_flip] = 1.0
    return h
