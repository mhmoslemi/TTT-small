    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    H = np.fft.rfft(h, fft_size)

import numpy as np
import time


def project_feasible(g, n):
    target = n / 2.0
    lo, hi = g.min() - 2.0, g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    n = len(h)
    if n <= 6000:
        return np.correlate(h, 1.0 - h, mode='full')
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    H = np.fft.rfft(h, fft_size)
    F = np.fft.rfft(1.0 - h, fft_size)
    return np.fft.irfft(H * np.conj(F), fft_size)[:size]


def compute_c5(h):
    return compute_corr(h).max() * (2.0 / len(h))


def smooth_obj_grad(h, tau, top_m=8):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s

    if top_m < len(p):
        idx = np.argsort(-exp_c)[:top_m]
        p = np.zeros_like(p)
        p[idx] = exp_c[idx] / exp_c[idx].sum()

    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        pk = p[k]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr


def optimize_smooth(h, n, tau, time_limit, initial_lr=None, top_m=8):
    h = project_feasible(h, n).copy()
    best_obj, _, _ = smooth_obj_grad(h, tau, top_m)
    best_h = h.copy()
    start = time.time()
    lr = 0.3 if initial_lr is None else initial_lr
    v = np.zeros(n)
    beta = 0.9
    stall = 0

    for it in range(100000):
        if time.time() - start > time_limit:
            break
        obj, grad, _ = smooth_obj_grad(h, tau, top_m)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        v = beta * v - lr * grad
        h_new = project_feasible(h + v, n)

        if stall > 60:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break

        h = h_new

    return best_h, best_obj


def optimize_subgrad(h, n, time_limit, initial_lr=None):
    dx = 2.0 / n
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.005 if initial_lr is None else initial_lr
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[np.arange(j0b, j1b) + sh]
        grad *= dx

        v = beta * v - lr * grad
        h_new = project_feasible(h + v, n)

        if stall > 300:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break

        h = h_new

    return best_h, best_c5


def optimize_multitau(h, n, time_limit, tau_list=None, top_m=8):
    if tau_list is None:
        tau_list = [10.0, 25.0, 60.0, 150.0]
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    frac = len(tau_list)
    for i, tau in enumerate(tau_list):
        remaining = time_limit - (time.time() - start)
        if remaining < 20:
            break
        tl = remaining / frac
        h_cur, _ = optimize_smooth(h, n, tau, time_limit=tl, initial_lr=0.25, top_m=top_m)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        h = h_cur
    return best_h, best_c5
