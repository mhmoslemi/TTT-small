- Specifically, result[k] changes only if i or j is in the range of the sum for lag k
- For position i: it appears in result[k] for k where n=i (term h[i+k](1-h[i])) or n+k=i (term h[i](1-h[n]))

- So changing h[i] affects O(n) lags
- Changing both h[i] and h[j] affects O(n) lags total

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial solution
    n = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Simulated annealing
    T = 0.01  # Initial temperature
    T_min = 1e-6
    cool_rate = 0.999
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() - start_time < budget_s * 0.7:
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Make swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        delta = new_c5 - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            # Accept
            best_c5 = new_c5 if new_c5 < best_c5 else best_c5
            # Update tracking
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Reject
            h[i] = 1
            h[j] = 0
        
        T *= cool_rate
    
    # Return best
