    dx = 2.0 / n
    R = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(R)

    active = np.where(R >= max_val - 1e-10)[0]
    s_vals = active - (n - 1)  # shifts
    grad = np.zeros(n)
    for s in s_vals:
        for j in range(n):
            g = 0.0
            if 0 <= j - s < n:
                g += (1 - h[j - s])
            if 0 <= j + s < n:
                g -= h[j + s]
            grad[j] += g
    grad /= len(s_vals)
    return max_val * dx, grad * dx
