    corr = np.correlate(h, 1 - h, mode="full")

    n = len(h)

    corr = np.correlate(h, 1 - h, mode="full")

    max_val = np.max(vals)
    exp_vals = np.exp(vals - max_val)
    p = exp_vals / np.sum(exp_vals)

    # Penalty for sum constraint
    sum_h = np.sum(h)
    target = n / 2.0
    penalty = lam * (sum_h - target) ** 2
    obj += penalty
    
    # Gradient of smooth max

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max_obj_and_grad(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = alpha * dx * corr
    max_val = np.max(vals)
    exp_vals = np.exp(vals - max_val)
    p = exp_vals / np.sum(exp_vals)
    
    obj = (max_val + np.log(np.sum(exp_vals))) / alpha
    
    sum_h = np.sum(h)
    target = n / 2.0
    penalty = lam * (sum_h - target) ** 2
    obj += penalty
    
    # Vectorized gradient
    row = np.arange(n)[:, None]
    col = np.arange(n)[None, :]
    idx_matrix = col - row + n - 1
    A = np.where((idx_matrix >= 0) & (idx_matrix <= 2*n-2), p[idx_matrix], 0.0)
    term1 = A @ (1.0 - h)
    term2 = A.T @ h
    grad_sm = dx * (term1 - term2)
    
    grad_pen = 2.0 * lam * (sum_h - target)
    grad = grad_sm + grad_pen
    
    return obj, grad
