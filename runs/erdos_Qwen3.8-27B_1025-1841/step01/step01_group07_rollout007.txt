import time

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_sum(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1}"""
    h = h.copy()
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-12:
            break
        # Need to reduce sum by diff (if diff>0) or increase by -diff
        # Subtract diff/n from all, then clip
        h = h - diff / len(h)
        h = np.clip(h, 0, 1)
    # Final adjustment: distribute remaining error to non-saturated coords
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h

def optimize_coordinate(h, i, n):
    """Optimize h[i] given all other coordinates fixed, minimizing max correlation"""
    # Compute for each lag s (shift = s - (n-1)), the correlation as function of t = h[i]
    # c_s(t) = intercept_s + slope_s * t
    shifts = np.arange(-(n-1), n)  # actual shifts
    n_lags = len(shifts)
    
    # Base correlation with h[i] = 0
    h0 = h.copy()
    h0[i] = 0.0
    base_corr = np.correlate(h0, 1.0 - h0, mode='full')  # length 2n-1
    
    # For each lag, compute the slope (coefficient of t)
    # For shift s: 
    #   index i appears as first element if i+s is valid (0 <= i+s < n)
    #   index i appears as second element if i-s is valid (0 <= i-s < n)
    #   contribution when i is first: t * (1 - h[i+s])
    #   contribution when i is second: h[i-s] * (1 - t) = h[i-s] - h[i-s]*t
    #   total: t*(1-h[i+s]) + h[i-s] - h[i-s]*t = h[i-s] + t*(1-h[i+s]-h[i-s])
    #   (each term only if the index is valid)
    
    slopes = np.zeros(n_lags)
    intercepts = base_corr.copy()
    
    for k_idx, s in enumerate(shifts):
        # As first element: (i, i+s)
        j1 = i + s
        if 0 <= j1 < n:

            slopes[k_idx] += (1 - h[j1])

        j2 = i - s

            intercepts[k_idx] += h[j2]

    candidates = {0.0, 1.0}

            h[free] -= diff / nf

    return h

def compute_slopes(h, i):
    n = len(h)
    n_lags = 2 * n - 1
    slopes = np.empty(n_lags)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        s = 0.0
        j1 = i - k
        if 0 <= j1 < n:
            s += 1.0 - h[j1]
        j2 = i + k
        if 0 <= j2 < n:
            s -= h[j2]
        slopes[idx] = s
    return slopes

def find_optimal_t(base, slopes):
    n_lags = len(base)
    db = base[:, None] - base[None, :]
    ds = slopes[None, :] - slopes[:, None]
    mask = np.abs(ds) > 1e-14
    safe_ds = np.where(mask, ds, 1.0)
    t_mat = np.where(mask, db / safe_ds, np.nan)
    valid = (~np.isnan(t_mat)) & (t_mat >= 0) & (t_mat <= 1)
    candidates = t_mat[valid]
    if len(candidates) == 0:
        candidates = np.array([0.0, 1.0])
    else:
        candidates = np.concatenate([candidates, [0.0, 1.0]])
    vals = base[None, :] + slopes[None, :] * candidates[:, None]
    max_vals = np.max(vals, axis=1)
    best_idx = int(np.argmin(max_vals))
    return float(candidates[best_idx])

def coordinate_descent(h, max_sweeps=200, tol=1e-11):
    n = len(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(n):
            old = h[i]
            h_base = h.copy()
            h_base[i] = 0.0
            base = np.correlate(h_base, 1.0 - h_base, mode='full')
            slopes = compute_slopes(h_base, i)
            t_opt = find_optimal_t(base, slopes)
            h[i] = t_opt
            delta = abs(t_opt - old)
            if delta > max_change:
                max_change = delta
        h = project_onto_sum(h, n / 2.0)
        if max_change < tol:
            break
    return h

def slsqp_polish(h, maxiter=300):
    n = len(h)
    def obj(x):
        return evaluate_c5(x)
    def obj_grad(x):
        g = np.zeros(n)
        eps = 1e-7
        f0 = evaluate_c5(x)
        for i in range(n):
            x[i] += eps
            fp = evaluate_c5(x)
            x[i] -= 2.0 * eps
            fm = evaluate_c5(x)
            x[i] += eps
            g[i] = (fp - fm) / (2.0 * eps)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        x = np.clip(res.x, 0.0, 1.0)
        x = project_onto_sum(x, n / 2.0)
        return x
    except Exception:
        return h

def make_structural_init(n):
    h = np.empty(n)
    h[:n // 2] = 1.0
    h[n // 2:] = 0.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return project_onto_sum(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    # Obtain initial point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = make_structural_init(47)

    n = len(h0)
    h0 = project_onto_sum(h0, n / 2.0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Coordinate descent from initial
    h1 = coordinate_descent(h0, max_sweeps=300, tol=1e-12)
    c1 = evaluate_c5(h1)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: SLSQP polish
    if time.time() - t_start < budget_s * 0.3:
        h2 = slsqp_polish(best_h, maxiter=500)
        c2 = evaluate_c5(h2)
        if c2 < best_c5:
            h2 = coordinate_descent(h2, max_sweeps=100)
            c2 = evaluate_c5(h2)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2

    # Phase 3: Random restarts
    n_restarts = 30
    for r in range(n_restarts):
        if time.time() - t_start > budget_s - 60:
            break
        pert_size = 0.15 * (1.0 - r / float(n_restarts)) + 0.03
        h_p = best_h + rng.randn(len(best_h)) * pert_size
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_onto_sum(h_p, len(best_h) / 2.0)
        h_opt = coordinate_descent(h_p, max_sweeps=80, tol=1e-10)
        c_opt = evaluate_c5(h_opt)
        if c_opt < best_c5:
            best_h, best_c5 = h_opt, c_opt

    # Phase 4: Try different n values
    if time.time() - t_start < budget_s - 90:
        for n_try in [48, 50, 52, 56, 60, 64, 72, 80]:
            if time.time() - t_start > budget_s - 60:
                break
            h_t = make_structural_init(n_try)
            # Also try 0.5 flat
            h_flat = np.full(n_try, 0.5)
            for h_init in [h_t, h_flat]:
                if time.time() - t_start > budget_s - 45:
                    break
                h_opt = coordinate_descent(h_init, max_sweeps=150, tol=1e-10)
                c_opt = evaluate_c5(h_opt)
                if c_opt < best_c5:
                    best_h, best_c5 = h_opt, c_opt

    # Phase 5: Final polish
    if time.time() - t_start < budget_s - 30:
        h_final = coordinate_descent(best_h, max_sweeps=300, tol=1e-13)
        c_final = evaluate_c5(h_final)
        if c_final < best_c5:
            best_h, best_c5 = h_final, c_final
        h_final2 = slsqp_polish(best_h, maxiter=200)
        c_final2 = evaluate_c5(h_final2)
        if c_final2 < best_c5:
            h_final2 = coordinate_descent(h_final2, max_sweeps=100)
            c_final2 = evaluate_c5(h_final2)
            if c_final2 < best_c5:
                best_h, best_c5 = h_final2, c_final2

    return best_h, best_c5, len(best_h)
