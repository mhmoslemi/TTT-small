def find_optimal_hj(h, j, n):
    n_corr = 2*n - 1
    s_range = np.arange(-(n-1), n)  # shifts
    
    # For each shift s, compute A_s and B_s
    # result[s + n - 1] = sum_i h[i] * (1-h[i+s]) (valid i)
    # h[j] appears when: i=j (need 0≤j+s<n) or i=j-s (need 0≤j-s<n)
    
    A = np.zeros(n_corr)
    B = np.zeros(n_corr)
    
    for idx, s in enumerate(s_range):
        # Term when i = j: h[j] * (1-h[j+s])
        if 0 <= j+s < n:
            A[idx] += (1.0 - h[j+s])
            # B[idx] += 0 (no constant term from this)
        else:
            A[idx] += 0.0
        
        # Term when i = j-s: h[j-s] * (1-h[j])
        if 0 <= j-s < n:
            A[idx] += (-1.0)
            B[idx] += h[j-s]
        else:
            A[idx] += 0.0
    
    # g(t) = max of (A[k]*t + B[k]) for k=0..n_corr-1
    # Minimize over t in [0,1]
    
    # Evaluate at endpoints
    g0 = np.max(B)
    g1 = np.max(A + B)
    
    # Find candidate interior points
    # The minimum of max of lines is where the "active" set changes
    # Simple approach: check all pairwise intersections that fall in [0,1]
    # But that's O(n_corr^2) which is fine for n_corr ~ 100
    
    best_t = 0 if g0 <= g1 else 1
    best_val = min(g0, g1)
    
    # Check pairwise intersections
    for i in range(n_corr):
        for k in range(i+1, n_corr):
            if abs(A[i] - A[k]) < 1e-12:
                continue
            t = (B[k] - B[i]) / (A[i] - A[k])
            if 0 < t < 1:
                # Check if both lines i and k are on the upper envelope at t
                val_i = A[i]*t + B[i]
                # Check if this is actually the max at t
                vals = A*t + B

                    if val_i < best_val:

                        best_t = t

    valid_js = (js >= 0) & (js < n)

    jms = j - s_range  # j-s for each s

    B = np.zeros(n_corr)
    
    A[valid_js] = 1.0 - h[js[valid_js]]
    A[valid_jms] -= 1.0
    B[valid_jms] = h[jms[valid_jms]]
    
    # Binary search
    g0 = np.max(B)
    g1 = np.max(A + B)
    lo = min(g0, g1) - 1.0
    hi = max(g0, g1) + 1.0
    
    pos_mask = A > 1e-12
    neg_mask = A < -1e-12
    zero_mask = ~(pos_mask | neg_mask)
    
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        
        # Feasibility check
        if zero_mask.any() and B[zero_mask].max() > mid:
            lo = mid
            continue
        
        lb = 0.0
        ub = 1.0
        
        if neg_mask.any():
            lb = max(lb, np.max((mid - B[neg_mask]) / A[neg_mask]))
            if lb > 1.0:
                lo = mid
                continue
        
        if pos_mask.any():
            ub = min(ub, np.min((mid - B[pos_mask]) / A[pos_mask]))
            if ub < 0.0:
                lo = mid
                continue
        
        if lb <= ub:
            hi = mid
        else:
            lo = mid
    
    v_star = hi
    
    # Find t
    lb = 0.0
    ub = 1.0
    if neg_mask.any():
        lb = max(lb, np.max((v_star - B[neg_mask]) / A[neg_mask]))
    if pos_mask.any():
        ub = min(ub, np.min((v_star - B[pos_mask]) / A[pos_mask]))
    
    t_star = 0.5 * (lb + ub)
    return float(np.clip(t_star, 0.0, 1.0))
