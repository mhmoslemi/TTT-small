    max_val = np.max(corr)

    return grad

def subgradient_descent(h0, n, deadline, lr_init=0.1, lr_decay=0.999):
    """Subgradient descent with projection."""
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    momentum = np.zeros(n)
    mu = 0.9  # momentum parameter
    
    step = 0
    while time.time() < deadline:

            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr *= lr_decay

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = corr >= max_val - 1e-10
    n_active = np.sum(active)
    if n_active == 0:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        d = k - (n - 1)
        if d >= 0:
            grad[d:] += (1.0 - h[:n - d])
            if n - d > 0:
                grad[:n - d] -= h[d:]
        else:
            p = -d
            grad[:n - p] += (1.0 - h[p:])
            if n - p > 0:
                grad[p:] -= h[:n - p]
    grad *= dx
    return grad

def subgradient_descent(h0, n, deadline, lr_init=0.5):
    h = h0.copy()
    momentum = np.zeros(n)
    mu = 0.85
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step = 0
    no_improve = 0
    patience = 200
    
    while time.time() < deadline and no_improve < patience:
        step += 1
        if time.time() > deadline:
            break
        g = compute_subgradient(h, n)
        momentum = mu * momentum + (1 - mu) * g
        lr = lr_init / (1.0 + 0.01 * step)
        h_new = h - lr * momentum
        h_new = project_to_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
    
    return best_h, best_c5

def sharpen(h, n, tau=0.1):
    """Push values toward 0 or 1, preserving the mean."""
    mean = np.mean(h)
    # sigmoid sharpening
    h_sharp = 1.0 / (1.0 + np.exp(-h / tau))
    # Rescale to maintain mean
    h_sharp = (h_sharp - h_sharp.mean()) / (h_sharp.std() + 1e-12)
    h_sharp = h_sharp * np.std(h) + mean
    h_sharp = np.clip(h_sharp, 0, 1)
    return project_to_feasible(h_sharp, n)

def binary_round(h, n, rng):
    """Round h to binary, preserving exactly n/2 ones."""
    target = n // 2
    # Sort by value, keep top n/2
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[indices[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, deadline, max_iters=10000):
    """Local search on binary solutions: try swapping pairs."""
    h = h_bin.copy()
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        c5_new = evaluate_c5(h)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            # Update sets
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Revert
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def sim_annealing_binary(h_bin, n, deadline, T_init=1.0, T_end=0.01):
    """Simulated annealing on binary solutions."""
    h = h_bin.copy()
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    rng = np.random.RandomState(123)
    n_steps = 50000
    for it in range(n_steps):
        if time.time() > deadline:
            break
        T = T_init * (T_end / T_init) ** (it / n_steps)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        c5_new = evaluate_c5(h)
        delta = c5_new - c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def interpolate_to_n(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    if n_to % n_from == 0:
        # Upsampling by repetition
        factor = n_to // n_from
        h_new = np.repeat(h, factor)
        return h_new
    elif n_from % n_to == 0:
        # Downsampling by averaging
        factor = n_from // n_to
        h_new = h.reshape(n_to, factor).mean(axis=1)
        return project_to_feasible(h_new, n_to)
    else:
        x_from = np.linspace(0, 2, n_from, endpoint=False)
        x_to = np.linspace(0, 2, n_to, endpoint=False)
        h_new = np.interp(x_to, x_from, h, period=2.0)
        return project_to_feasible(h_new, n_to)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(42, 0.5)
    
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Subgradient descent at original resolution
    n_orig = len(h_start)
    h1, c5_1 = subgradient_descent(h_start, n_orig, min(t0 + 200, deadline - 100), lr_init=0.3)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    # Phase 2: Try higher resolutions
    if time.time() < deadline - 150:
        n_high = n_orig * 2  # Double resolution
        h_high = interpolate_to_n(best_h, n_orig, n_high)
        h2, c5_2 = subgradient_descent(h_high, n_high, min(t0 + 400, deadline - 100), lr_init=0.2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
        # Downsample back for comparison
        h2_low = interpolate_to_n(h2, n_high, n_orig)
        c5_2_low = evaluate_c5(h2_low)
        if c5_2_low < best_c5:
            best_c5 = c5_2_low
            best_h = h2_low
    
    # Phase 3: Sharpening + subgradient
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        h_sharp = sharpen(best_h, n_cur, tau=0.05)
        h3, c5_3 = subgradient_descent(h_sharp, n_cur, min(t0 + 600, deadline - 80), lr_init=0.1)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
    
    # Phase 4: Binary local search
    if time.time() < deadline - 50:
        n_cur = len(best_h)
        h_bin = binary_round(best_h, n_cur, np.random.RandomState(seed))
        h4, c5_4 = binary_local_search(h_bin, n_cur, min(t0 + 900, deadline - 30), max_iters=20000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
    
    # Phase 5: Simulated annealing for binary
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        h_bin = binary_round(best_h, n_cur, np.random.RandomState(seed))
        h5, c5_5 = sim_annealing_binary(h_bin, n_cur, deadline, T_init=0.5, T_end=0.01)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5
    
    # Final: try SLSQP polish
    if time.time() < deadline - 5 and len(best_h) <= 500:
        n_cur = len(best_h)
        target = n_cur / 2.0
        T_soft = 50.0
        def soft_obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            scaled = corr * T_soft
            m = np.max(scaled)
            return float((m + np.log(np.sum(np.exp(scaled - m)))) / T_soft * (2.0/n_cur))
        
        try:
            res = minimize(soft_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_cur,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                          options={'ftol': 1e-12, 'maxiter': 200})
            h6 = project_to_feasible(res.x, n_cur)
            c5_6 = evaluate_c5(h6)
            if c5_6 < best_c5:
                best_c5 = c5_6
                best_h = h6
        except Exception:
            pass
    
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = corr >= max_val - 1e-12
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        d = k - (n - 1)
        if d >= 0:
            grad[d:] += (1.0 - h[:n - d])
            grad[:n - d] -= h[d:]
        else:
            p = -d
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
    grad *= dx
    return grad

def subgrad_step(h, n):

            g[:n-d] -= h[d:]

For arrays a (length n) and v (length n):
np.correlate(a, v, mode='full') has length 2n-1.
The [k]-th element (0-indexed) is:
  sum_{m} a[m] * v[m + (n-1-k)]
  (where the product is included only when 0 ≤ m + (n-1-k) < n)

So with a=h and v=1-h:
  c[k] = sum_{m: 0≤m+n-1-k<n} h[m] * (1-h[m+n-1-k])
        = sum_{m: k-n+1≤m≤k} h[m] * (1-h[m-(k-n+1)])

Let d = k-(n-1). Then m ranges from max(0, d) to min(n-1, n-1+d), and:
  c[k] = sum_{m=max(0,d)}^{min(n-1, n-1+d)} h[m] * (1-h[m-d])

For d ≥ 0: m from d to n-1: c[k] = sum_{m=d}^{n-1} h[m](1-h[m-d]) ✓
For d < 0: let p = -d > 0. m from 0 to n-1-p: c[k] = sum_{m=0}^{n-1-p} h[m](1-h[m+p]) ✓

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_vec(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    act = np.where(corr >= mx - 1e-12)[0]
    g = np.zeros(n)
    for k in act:
        d = k - (n - 1)
        if d >= 0:
            g[d:] += 1.0 - h[:n - d]
            g[:n - d] -= h[d:]
        else:
            p = -d
            g[:n - p] += 1.0 - h[p:]
            g[p:] -= h[:n - p]
    return g * dx

def subgrad_opt(h0, n, deadline, lr=0.1, mu=0.9, stall_limit=800):
    h = h0.copy()
    v = np.zeros(n)
    best = h.copy()
    best_c5 = evaluate_c5(h)
    t = 0
    stall = 0
    while time.time() < deadline:
        t += 1
        g = subgrad_vec(h, n)
        v = mu * v + (1.0 - mu) * g
        step = lr / (1.0 + 0.003 * t) ** 0.7
        h = project(h - step * v, n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5, best = c5, h.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                break
    return best, best_c5

def interp_up(h, nf, nt):
    if nt % nf == 0:
        return np.repeat(h, nt // nf)
    xf = np.linspace(0, 2, nf, endpoint=False)
    xt = np.linspace(0, 2, nt, endpoint=False)
    return np.interp(xt, xf, h)

def to_binary(h, n):
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_search(h0, n, deadline, seed=42, n_iter=800000):
    h = to_binary(h0, n).copy()
    c5 = evaluate_c5(h)
    best_h, best_c5 = h.copy(), c5
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    rng = np.random.RandomState(seed)
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5n = evaluate_c5(h)
        if c5n < best_c5 - 1e-15:
            if c5n < c5:
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                c5 = c5n
            best_c5, best_h = c5n, h.copy()
            stall = 0
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
            if stall > 5000:

                    ones.remove(ii)
                    zeros.remove(jj)
                    ones.append(jj)

                stall = 0

def slsqp_polish(h0, n, deadline):
    dx = 2.0 / n
    T = 80.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * T
        m = c.max()
        return float((m + np.log(np.sum(np.exp(c - m)))) / T * dx)
    def con(x):
        return x.sum() - n / 2.0

                     bounds=[(0.0, 1.0)] * n,

                     options={'ftol': 1e-14, 'maxiter': 300})
        hp = project(r.x, n)
        return hp, evaluate_c5(hp)
    except Exception:
        return h0, evaluate_c5(h0)

    h = np.zeros(n)

    for i in range(n // 2):

    h[idx] = 1.0
    return h
