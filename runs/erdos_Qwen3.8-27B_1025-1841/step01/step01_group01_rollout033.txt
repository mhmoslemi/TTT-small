```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    initial = np.array(initial_h_values, dtype=float)
    n_init = len(initial)
    
    best_h = initial.copy()
    best_c5 = compute_c5(best_h, n_init)
    best_n = n_init
    
    def time_left():
        return budget_s - 30 - (time.time() - t0)
    
    def make_feasible(h, n):
        """Repair h to satisfy 0<=h<=1 and sum(h)=n/2"""
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        for _ in range(1000):
            diff = target - np.sum(h)
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                room = 1.0 - h
            else:
                room = h
            total_room = np.sum(room)
            if total_room < 1e-12:
                break
            h = h + (diff / total_room) * room
            h = np.clip(h, 0.0, 1.0)
        return h
    
    def interpolate_to(h_src, n_src, n_dst):
        x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
        x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
        h_dst = np.interp(x_dst, x_src, h_src)
        return make_feasible(h_dst, n_dst)
    
    resolutions = [n_init, 50, 100, 200, 400]
    
    for n in resolutions:
        if time_left() < 60:
            break
        
        if n == n_init:
            h = initial.copy()
        else:
            h = interpolate_to(best_h, best_n, n)
        
        # Optimize at this resolution
        dx = 2.0 / n
        target_sum = n / 2.0
        
        for attempt in range(3):
            if time_left() < 30:
                break
            
            # Add small perturbation for different attempts
            if attempt > 0:
                h = h + np.random.randn(n) * 0.01
                h = make_feasible(h, n)
            
            beta = 30.0 + attempt * 20.0
            
            def obj(hv):
                corr = np.correlate(hv, 1.0 - hv, mode="full")
                vals = corr * dx
                return float(logsumexp(beta * vals) / beta)
            
            res = minimize(
                obj, h, method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                             'jac': lambda x: np.ones(n)}],
                options={'maxiter': 300, 'ftol': 1e-9}
            )
            
            if res.x is not None:
                h_opt = make_feasible(res.x, n)
                c5_val = compute_c5(h_opt, n)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
                    best_n = n
                h = h_opt
    
    return best_h, best_c5, best_n
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        total_room = np.sum(room)
        if total_room < 1e-15:
            break
        h = h + (diff / total_room) * room
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_src, n_src, n_dst):
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return make_feasible(h_dst, n_dst)

def smooth_max_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    return float(logsumexp(beta * vals) / beta), vals

def smooth_max_grad(h, n, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*corr*dx))) w.r.t. h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    # Softmax weights
    m = np.max(vals)
    e = np.exp(beta * (vals - m))
    s = np.sum(e)
    p = e / s  # shape (2n-1,)
    
    # grad[j] = dx * sum_k p[k] * [(1-h[j-d_k]) * 1_{0<=j-d_k<n} - h[j+d_k] * 1_{0<=j+d_k<n}]
    # where d_k = k - (n-1)
    
    grad = np.zeros(n)
    # d ranges from -(n-1) to (n-1)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        pk = p[k] * dx
        if d >= 0:
            # (1-h[j-d]) for j in [d, n-1]
            js = np.arange(d, n)
            grad[js] += pk * (1.0 - h[js - d])
            # -h[j+d] for j in [0, n-1-d]
            js2 = np.arange(0, n - d)
            grad[js2] -= pk * h[js2 + d]
        else:
            sd = -d  # sd > 0
            # (1-h[j-d]) = (1-h[j+sd]) for j in [0, n-1-sd]
            js = np.arange(0, n - sd)
            grad[js] += pk * (1.0 - h[js + sd])
            # -h[j+d] = -h[j-sd] for j in [sd, n-1]
            js2 = np.arange(sd, n)
            grad[js2] -= pk * h[js2 - sd]
    
    return grad

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    initial = np.array(initial_h_values, dtype=float)
    n_init = len(initial)
    
    best_h = initial.copy()
    best_c5 = compute_c5(best_h, n_init)
    best_n = n_init
    
    def time_ok(min_remaining=30.0):
        return (time.time() - t0) + min_remaining < budget_s
    
    def optimize(h, n, betas=[20, 50, 100, 200, 400]):
        h = make_feasible(h, n)
        target_sum = n / 2.0
        best_local = compute_c5(h, n)
        
        for beta in betas:
            if not time_ok():
                break
            
            def obj_and_grad(hv):
                if not time_ok():
                    return 1e10, np.zeros(n)
                val, _ = smooth_max_obj(hv, n, beta)
                grad = smooth_max_grad(hv, n, beta)
                return val, grad
            
            res = minimize(
                obj_and_grad, h, jac=True, method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                             'jac': lambda x: np.ones(n)}],
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            
            if res.x is not None:
                h_candidate = make_feasible(res.x, n)
                c5_val = compute_c5(h_candidate, n)
                if c5_val < best_local:
                    best_local = c5_val
                    h = h_candidate
        
        return h, best_local
    
    # Try different resolutions
    resolutions = [n_init, 50, 100, 200, 400]
    
    for n in resolutions:
        if not time_ok(60):
            break
        
        if n == n_init:
            h = initial.copy()
        else:
            h = interpolate_h(best_h, best_n, n)
        
        # Multiple attempts with different perturbations
        for attempt in range(3):
            if not time_ok(30):
                break
            
            if attempt > 0:
                h = h + np.random.randn(n) * (0.02 / (attempt + 1))
                h = make_feasible(h, n)
            
            h_opt, c5_opt = optimize(h, n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
    
    return best_h, best_c5, best_n
```
