            valid_zeros = [x for x in valid if h[x] == 0]
            
            if len(valid_ones) > 0 and len(valid_zeros) > 0:
                i = np.random.choice(valid_ones)
                j = np.random.choice(valid_zeros)
                h[i] = 0
                h[j] = 1
                new_max = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if new_max < current_max:
                    current_max = new_max
                    new_c5 = new_max * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    improved = True
            else:
                # Fallback to random swap
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                new_max = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if new_max < current_max:
                    current_max = new_max
                    improved = True
                    new_c5 = new_max * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    h[i] = 1
                    h[j] = 0
        
        if not improved:
            k = np.random.randint(5, 12)
            for _ in range(k):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
                    h[i] = 0
                    h[j] = 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    have_initial = 'initial_h_values' in globals() and initial_h_values is not None
    if have_initial:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 80
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1
        np.random.shuffle(h0)
    
    h = make_binary(h0, n0)
    c5 = evaluate_c5(h, n0)
    best_c5 = c5
    best_h = h.copy()
    best_n = n0
    
    # Phase 1: General local search (40% of budget)
    alloc = (budget_s - (time.time() - t0)) * 0.4
    h, c5 = local_search(h, n0, time_limit=alloc, t_start=t0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n0
    
    # Phase 2: Targeted search on bottleneck lags (30% of budget)
    elapsed = time.time() - t0
    alloc2 = (budget_s - elapsed) * 0.45
    h, c5 = targeted_search(best_h, n0, time_limit=alloc2, t_start=t0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n0
    
    # Phase 3: Random restarts from perturbed best (15% of budget)
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 30
    if remaining > 20:
        n_restarts = max(1, int(remaining // 40))
        for s in range(n_restarts):
            if time.time() - t0 > budget_s - 40:
                break
            h_try = best_h.copy()
            n_flip = np.random.randint(5, 20)
            for _ in range(n_flip):
                ones = np.where(h_try == 1)[0]
                zeros = np.where(h_try == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
                    h_try[i] = 0
                    h_try[j] = 1
            tl = min(40, budget_s - (time.time() - t0) - 35)
            if tl <= 2:
                break
            h_try, c5_try = local_search(h_try, n0, time_limit=tl, t_start=t0)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n0
    
    # Phase 4: Try higher resolutions (remaining time)
    elapsed = time.time() - t0
    for n_new in [160, 240, 320, 480, 640]:
        if time.time() - t0 > budget_s - 40:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin = make_binary(h_interp, n_new)
        tl = min(100, budget_s - (time.time() - t0) - 35)
        if tl <= 5:
            break
        h_new, c5_new = local_search(h_bin, n_new, time_limit=tl, t_start=t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_new
    
    # Final safety check
    if time.time() - t0 > budget_s - 5 and best_h is not None:
        pass  # already have best
    
    if best_h is None:
        best_h = h
        best_c5 = c5
        best_n = n0
    
    return (best_h, best_c5, best_n)
