def compute_c5(h):
    """Compute C5 bound."""
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def project_to_feasible(g, n):
    """Project g to feasible set [0,1]^n with sum = n/2."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    # Sort by value, keep top n/2 as 1
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:n//2]] = 1.0
    return b

def binary_local_search(h, n, time_limit, seed=42):
    """Simulated annealing with bit flips."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h, n).copy()
    corr = compute_corr(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    
    start = time.time()
    T = 0.01  # initial temperature
    T_min = 1e-6
    cool_rate = 0.999
    
    while time.time() - start < time_limit:
        # Pick a random position to flip
        i = rng.randint(n)
        old_val = h[i]
        new_val = 1.0 - old_val
        delta = new_val - old_val  # +1 or -1
        
        # Compute change in correlation
        # For each shift m (0 to 2n-2), the change is:
        # delta * (1 - h[i + m - n + 1]) - delta * h[i - m + n - 1]
        # where out-of-range indices contribute 0
        m = np.arange(2 * n - 1)
        # Index 1: i + m - n + 1, valid when 0 <= i + m - n + 1 < n
        idx1 = i + m - n + 1
        mask1 = (idx1 >= 0) & (idx1 < n)
        # Index 2: i - m + n - 1, valid when 0 <= i - m + n - 1 < n
        idx2 = i - m + n - 1
        mask2 = (idx2 >= 0) & (idx2 < n)
        
        h1 = np.zeros(2 * n - 1)
        h1[mask1] = h[idx1[mask1]]
        h2 = np.zeros(2 * n - 1)
        h2[mask2] = h[idx2[mask2]]
        
        delta_corr = delta * (1.0 - h1) - delta * h2
        
        new_corr = corr + delta_corr
        new_c5 = new_corr.max() * dx
        
        # Accept with probability based on temperature
        if new_c5 < best_c5:
            # Always accept if better than best
            h[i] = new_val
            corr = new_corr
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        elif rng.rand() < np.exp(-(new_c5 - best_c5) / T):
            h[i] = new_val
            corr = new_corr
        
        # Cool
        T = max(T_min, T * cool_rate)
        
        # Reset temperature periodically
        if T <= T_min * 2:
            T = 0.01
    
    return best_h, best_c5

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):

    return compute_corr(h).max() * (2.0 / n)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:n // 2]] = 1.0
    return b

def corr_delta_for_flip(h, i, n):
    """Compute the change in correlation when h[i] is flipped (binary)."""
    delta = 1.0 - 2.0 * h[i]
    k = np.arange(-(n - 1), n)
    idx_plus = i + k
    idx_minus = i - k
    valid_plus = (idx_plus >= 0) & (idx_plus < n)
    valid_minus = (idx_minus >= 0) & (idx_minus < n)
    h_plus = np.zeros(2 * n - 1)
    h_minus = np.zeros(2 * n - 1)
    h_plus[valid_plus] = h[idx_plus[valid_plus]]
    h_minus[valid_minus] = h[idx_minus[valid_minus]]
    delta_corr = delta * (1.0 - h_plus) - delta * h_minus
    delta_corr[n - 1] = 0.0  # k=0 always 0 for binary
    return delta_corr

def binary_sa(h_init, n, time_limit, seed=42, T0=0.003, Tmin=1e-8, cool=0.9997):

    best_h = h.copy()

    T = T0
    restart_count = 0

            if restart_count > 100:

def binary_greedy(h_init, n, time_limit, seed=42):
    """First-improvement greedy: keep flipping if it improves."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = compute_corr(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    
    while time.time() - start < time_limit:
        # Try a batch of random flips, keep the best
        batch_size = min(n, 200)
        indices = rng.choice(n, batch_size, replace=False)
        best_delta = 0
        best_idx = -1
        best_new_corr = None

            delta_corr = corr_delta_for_flip(h, i, n)

            c5_val = new_corr.max()

def continuous_optimize(h, n, time_limit, tau=20.0, lr=0.1, max_iters=100000):
    """Smooth max + projected gradient."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        
        # Gradient of smooth max approximation
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-16:
                continue
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
        grad *= dx
        
        obj = dx * (max_c + np.log(s) / tau)
        if obj < best_c5 - 1e-14:
            best_c5 = max_c * dx
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        
        if stall > 200:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
    
    return best_h, compute_c5(best_h)

def interp_periodic(h, target_n):
    n = len(h)

    return np.interp(x_new, x_old, h, period=2.0)
