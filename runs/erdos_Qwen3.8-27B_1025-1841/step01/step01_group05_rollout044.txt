def compute_gradient(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # i goes from 0 to 2n-2, k_i = i - (n-1) from -(n-1) to (n-1)
    
    corr_shifted = corr - corr.max()
    w = np.exp(corr_shifted / tau)
    w /= w.sum()
    
    # Build gradient
    # grad[j] = (dx/tau) * sum_i w[i] * ((1-h[j-k_i]) * [0<=j-k_i<n] - h[j+k_i] * [0<=j+k_i<n])
    # where k_i = i - (n-1)
    
    ks = np.arange(2*n-1) - (n-1)  # shape (2n-1), values -(n-1) to (n-1)
    js = np.arange(n)  # shape (n)
    
    # j - k_i: shape (n, 2n-1)
    jmk = js[:, None] - ks[None, :]  # (n, 2n-1)
    jpk = js[:, None] + ks[None, :]  # (n, 2n-1)
    
    mask_a = (jmk >= 0) & (jmk < n)  # (n, 2n-1)
    mask_b = (jpk >= 0) & (jpk < n)  # (n, 2n-1)
    
    # term_a[j, i] = 1 - h[jmk[j,i]] if valid
    h_a = np.zeros_like(jmk, dtype=float)
    h_a[mask_a] = h[jmk[mask_a]]
    term_a = np.where(mask_a, 1.0 - h_a, 0.0)
    
    h_b = np.zeros_like(jpk, dtype=float)
    h_b[mask_b] = h[jpk[mask_b]]

    grad = (dx / tau) * ((term_a - term_b) @ w)
