def project_feasible(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}"""
    # Bisection on lambda: h = clip(v - lambda, 0, 1), find lambda s.t. sum = target_sum
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0, 1)
    return h

    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr) * dx)

def softmax_objective(h, n, eps):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    g_max = np.max(g)
    logits = (g - g_max) / eps
    logits = np.clip(logits, -50, 50)

    sum_exp = np.sum(exp_logits)
    obj = g_max + eps * np.log(sum_exp)
    weights = exp_logits / sum_exp
    return obj, weights, g
