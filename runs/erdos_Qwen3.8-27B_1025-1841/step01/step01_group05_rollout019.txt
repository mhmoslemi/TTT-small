import numpy as np
import time

def eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 0 and abs(s - n/2.0) > 1e-10:
        h = h * (n / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    m = k_max - (n - 1)
    grad = np.zeros(n)
    # I[0 <= j-m < n] * (1-h[j-m])
    valid1 = (np.arange(n) - m) >= 0
    valid1 &= (np.arange(n) - m) < n
    grad[valid1] += (1.0 - h[np.arange(n)[valid1] - m])
    # -I[0 <= j+m < n] * h[j+m]
    valid2 = (np.arange(n) + m) >= 0
    valid2 &= (np.arange(n) + m) < n
    grad[valid2] -= h[np.arange(n)[valid2] + m]
    return grad * dx

    best_h = h.copy()

    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = h.sum()
    if s > 1e-15:

    return np.clip(h, 0.0, 1.0)

def compute_subgrad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    m = k_max - (n - 1)
    j = np.arange(n)
    jm = j - m
    jp = j + m
    grad = np.zeros(n)
    v1 = (jm >= 0) & (jm < n)
    v2 = (jp >= 0) & (jp < n)
    if np.any(v1):

    if np.any(v2):
