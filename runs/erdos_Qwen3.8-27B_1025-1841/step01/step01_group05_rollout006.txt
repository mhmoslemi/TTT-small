        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _optimize(h_init, n, t_deadline, seed):
    rng = np.random.RandomState(seed)
    h = _proj(h_init, n)
    best_h = h.copy()
    best_c5 = _eval_c5(h, n)
    eps = 40.0
    lr = 0.04
    min_lr = 1e-6
    stuck = 0

    while time.time() < t_deadline:
        grad = _grad_smooth(h, n, eps)
        grad -= grad.mean()
        improved = False
        alpha = lr
        for _ in range(50):
            h_try = _proj(h - alpha * grad, n)
            c5_try = _eval_c5(h_try, n)
            if c5_try < best_c5 - 1e-12:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            stuck = 0
            lr = max(alpha * 1.2, min_lr)
        else:
            stuck += 1
            if stuck % 15 == 0:
                scale = 0.12 if stuck < 60 else (0.06 if stuck < 120 else 0.03)
                h = _proj(best_h + rng.normal(0, scale, n), n)
            if stuck % 40 == 0:
                # flip some random entries
                k_flip = rng.randint(3, 10)
                idxs = rng.choice(n, size=k_flip, replace=False)
                h[idxs] = 1.0 - h[idxs]
                h = _proj(h, n)
            lr = max(min_lr, lr * 0.8)
            if eps < 120 and stuck > 50:
                eps *= 1.4
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(42)
        h0[:21] = 1.0

    n = len(h0)
    best_h = _proj(h0, n)
    best_c5 = _eval_c5(best_h, n)

    # Phase 1: Optimize from initial (60% budget)
    t_phase1 = t0 + budget_s * 0.55
    h1, c5_1 = _optimize(h0, n, t_phase1, seed)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Multiple perturbations and re-optimizations (30% budget)
    rng = np.random.RandomState(seed + 77)
    t_phase2_end = t0 + budget_s * 0.90
    attempt = 0
    while time.time() < t_phase2_end:
        attempt += 1
        scale = 0.06 + 0.06 * rng.random()
        h_pert = _proj(best_h + rng.normal(0, scale, n), n)
        sub_deadline = min(time.time() + 12, t_phase2_end)
        h_opt, c5_opt = _optimize(h_pert, n, sub_deadline, seed + attempt * 31)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Fine-tune from best (10% budget)
    h_fin, c5_fin = _optimize(best_h, n, deadline, seed + 9999)
    if c5_fin < best_c5:
        best_h, best_c5 = h_fin, c5_fin

    # Final evaluation
    final_c5 = _eval_c5(best_h, n)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
