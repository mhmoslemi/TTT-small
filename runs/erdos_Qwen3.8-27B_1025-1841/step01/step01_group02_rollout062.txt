def compute_obj_grad(h, T, penalty=100.0):

    dx = 2.0 / n

    obj = smooth_logsumexp * dx
    
    # Penalty for sum constraint
    s = np.sum(h) - n / 2.0
    obj += penalty * s * s
    
    # Gradient of smooth max
    grad = np.zeros(n)

One more thought: the coordinate descent can be made much more efficient. Instead of trying 8 values per coordinate (each requiring a full C5 evaluation which is O(n^2)), I can:
1. Precompute which c_k values change when h[j] changes
2. Only recompute the relevant c_k values
3. Find the maximum among those

When h[j] changes, c_k changes only for k where the correlation involves position j. For the full correlation of h with 1-h, position j appears in c_k for k where:
- j is a valid first index: j in [0, n-1] → always valid
- j+s_k is a valid second index: 0 ≤ j+s_k < n → s_k in [-j, n-1-j] → k in [n-1-j, n-1+j]

So c_k for k in [n-1-j, n-1+j] changes when h[j] changes. That's a range of 2j+1 to 2(n-1-j)+1 values, at most 2n-1.

The change in c_k is:
- c_k includes term h[j]*(1-h[j+s_k]): changes by Δh[j]*(1-h[j+s_k])
- c_k includes term h[j-s_k]*(1-h[j]): if j-s_k is valid, changes by -h[j-s_k]*Δh[j]

So Δc_k = Δh[j] * [(1-h[j+s_k]) - h[j-s_k]] (for valid terms)

This is O(n) per coordinate evaluation (instead of O(n^2)), so coordinate descent becomes O(n) per candidate value, O(n*8) per coordinate, O(n^2*8) per full pass, O(n^2*8*passes) total.

from scipy.optimize import minimize
import time

_g_T = 30.0
_g_pen = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _obj(h):
    T = _g_T
    pen = _g_pen
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    ec = np.exp(T * (c - cm))
    lse = cm + np.log(np.sum(ec)) / T
    s = np.sum(h) - n * 0.5
    return lse * dx + pen * s * s

def _jac(h):
    T = _g_T
    pen = _g_pen
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    ec = np.exp(T * (c - cm))
    w = ec / np.sum(ec)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        sk = k - (n - 1)
        if sk >= 0:
            lo, hi = 0, n - sk - 1
        else:
            lo, hi = -sk, n - 1
        if hi < lo:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx] += wk * (1.0 - h[idx + sk])
        g[idx + sk] -= wk * h[idx]
    g *= dx
    s = np.sum(h) - n * 0.5
    g += 2.0 * pen * s
    return g

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n * 0.5
    h += (target - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        m = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = m.sum()
        if cnt:
            h[m] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    if nn == len(h):
        return h.copy()
    return make_feasible(np.interp(np.linspace(0.0, 2.0, nn), np.linspace(0.0, 2.0, len(h)), h), nn)
