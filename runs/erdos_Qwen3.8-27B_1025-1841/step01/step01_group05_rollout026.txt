The projection: given a point y, project to {h: 0<=h[i]<=1, sum(h)=n/2}.
This is a quadratic programming problem: minimize ||y-h||^2 subject to 0<=h[i]<=1 and sum(h)=n/2.
This can be solved by bisection on the Lagrange multiplier for the sum constraint:
h[i] = clip(y[i] - lambda, 0, 1), choose lambda so sum(h) = n/2.

```python
import numpy as np
import time

def upsample(h_old, n_new, n_old=None):
    """Upsample h from n_old to n_new using linear interpolation."""
    if n_old is None:
        n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False) + 0.5/n_old
    x_new = np.linspace(0, 1, n_new, endpoint=False) + 0.5/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def project_feasible(y, n):
    """Project y onto {h: 0<=h[i]<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        lam = (lo + hi) / 2
        h = np.clip(y - lam, 0, 1)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if abs(s - target) < 1e-12:
            break
    h = np.clip(y - lam, 0, 1)
    # Small adjustment for exactness
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute evenly among non-boundary elements
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def grad_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n-s] += (1 - h[s:])
            g[s:] -= h[:n-s]
    else:
        ss = -s
        if ss < n:
            g[ss:] += (1 - h[:n-ss])
            g[:n-ss] -= h[ss:]
    return g

def subgradient(h, dx, top_frac=0.1):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    c5 = max_val * dx
    
    # Select top fraction of lags

    grad = np.zeros(n)
    for s in active_s:
        grad += grad_for_shift(h, s)
    grad /= len(active_s)
    grad *= dx
    return c5, grad

def projected_subgradient_descent(h, n, dx, steps=5000, lr_init=0.01, lr_decay=0.999):
    """Projected subgradient descent."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr_init
    for step in range(steps):
        c5, grad = subgradient(h, dx, top_frac=0.05)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= lr_decay
        if lr < 1e-6:
            break
    return best_h, best_c5

def simulated_annealing(h, n, dx, steps=5000, T_init=0.05, T_final=0.001):
    """Simulated annealing with projection."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    T = T_init
    cool = (T_final / T_init) ** (1.0 / steps)
    
    current_c5 = best_c5
    for step in range(steps):
        # Random perturbation
        delta = np.random.randn(n) * 0.1
        h_new = h + delta
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / T):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        T *= cool
    
    return best_h, best_c5
```

```python
import numpy as np
import time

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old, period=None)
    return h_new

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    lam = (lo + hi) / 2.0
    for _ in range(80):
        lam = (lo + hi) / 2.0
        h = np.clip(y - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if abs(s - target) < 1e-13:
            break
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def grad_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    else:
        ss = -s
        if ss < n:
            g[ss:] += (1.0 - h[:n - ss])
            g[:n - ss] -= h[ss:]
    return g

def subgradient(h, top_frac=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    c5 = max_val * dx
    threshold = max_val * (1.0 - top_frac)

    grad = np.zeros(n)
    for s in active_s:
        grad += grad_for_shift(h, int(s))
    grad /= max(1, len(active_s))
    grad *= dx
    return c5, grad

        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= lr_decay
        if lr < 1e-8:
            break
    return best_h, best_c5

def simulated_annealing(h, n, steps, T_init, T_final, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    T = T_init
    cool = (T_final / T_init) ** (1.0 / max(1, steps))
    for step in range(steps):

            break
        scale = 0.05 + 0.1 * (T / T_init)

        c5_new = compute_c5(h_new)
        if c5_new < current_c5 or np.random.random() < np.exp(max(0, -(c5_new - current_c5) / max(T, 1e-10))):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        T *= cool
    return best_h, best_c5
```
