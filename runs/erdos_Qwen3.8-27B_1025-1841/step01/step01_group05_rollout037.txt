import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    c5 = float(corr[k_star] * dx)
    shift = k_star - (n - 1)
    
    grad = np.zeros(n)
    # For each m, grad[m] = (1-h[m-shift]) if 0<=m-shift<n, minus h[m+shift] if 0<=m+shift<n
    # All multiplied by dx
    
    # Term 1: (1-h[m-shift]) for valid m-shift

    grad[valid1] += (1.0 - h[np.arange(n)[valid1] - shift])

        if diff > 0:

def c5_gradient(h):
    """Compute C5 and its subgradient at the maximizing lag."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    v2 = idx + shift
    valid2 = (v2 >= 0) & (v2 < n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    c5 = float(corr[k_star] * dx)
    shift = k_star - (n - 1)
    idx = np.arange(n)
    grad = np.zeros(n)
    v1 = idx - shift
    valid1 = (v1 >= 0) & (v1 < n)
    grad[valid1] += (1.0 - h[v1[valid1]])
    v2 = idx + shift
    valid2 = (v2 >= 0) & (v2 < n)
    grad[valid2] -= h[v2[valid2]]
    grad *= dx
    return c5, grad
