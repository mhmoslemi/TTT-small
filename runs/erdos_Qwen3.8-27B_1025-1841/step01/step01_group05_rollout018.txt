import numpy as np
from scipy.optimize import minimize, linear_sum_assignment
import time

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    excess = np.sum(h) - target
    if excess > 0:
        # Need to reduce: take from largest elements
        idx = np.argsort(h)[::-1]
        for i in idx:
            if excess <= 1e-15:
                break
            reduce = min(h[i], excess)
            h[i] -= reduce
            excess -= reduce
    elif excess < 0:
        # Need to increase: add to smallest elements
        deficit = -excess
        idx = np.argsort(h)
        for i in idx:
            if deficit <= 1e-15:
                break
            add = min(1.0 - h[i], deficit)
            h[i] += add
            deficit -= add
    # Final exact fix (floating point)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Adjust one element
        if diff > 0:
            idx = np.argmin(h)
            h[idx] += min(diff, 1.0 - h[idx])
        else:
            idx = np.argmax(h)
            h[idx] += max(diff, -h[idx])
    return h

def compute_c5(h, n):
    """Compute C5 = max_k corr(h, 1-h, k) * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective(h_flat, n):
    """Objective for optimizer: project then compute c5"""
    h = h_flat.reshape(-1)
    h = project_to_feasible(h, n)
    return compute_c5(h, n)

def interpolate_to_resolution(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new points using linear interpolation"""
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0/n_old  # cell centers
    # Actually, let's use simple index-based interpolation
    x_new = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0/n_new
    h_interp = np.interp(x_new, x_old, h_old, period=2.0)  # periodic? No, not necessarily

    h_interp = np.interp(x_new, x_old, h_old)
    h_interp = project_to_feasible(h_interp, n_new)
    return h_interp

def project_to_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)

    s = h.sum()

            if excess <= 1e-13:

            h[i] -= r
            excess -= r
    elif diff > 1e-13:  # too little, need to increase

        idx = np.argsort(h)
        for i in idx:
            if deficit <= 1e-13:
                break

            h[i] += a
            deficit -= a

import numpy as np
from scipy.optimize import minimize
import time

_nm_state = {}

def nm_callback(x):
    if time.time() - _nm_state['t0'] > _nm_state['tmax']:
        return True
    return False

def project_to_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = 0.5 * n
    s = h.sum()
    diff = target - s
    if diff < -1e-13:
        excess = -diff
        idx = np.argsort(h)[::-1]
        for i in idx:
            if excess <= 1e-13:
                break
            r = min(h[i], excess)
            h[i] -= r
            excess -= r
    elif diff > 1e-13:
        deficit = diff
        idx = np.argsort(h)
        for i in idx:
            if deficit <= 1e-13:
                break
            a = min(1.0 - h[i], deficit)
            h[i] += a
            deficit -= a
    residual = target - h.sum()
    if abs(residual) > 1e-15:
        if residual > 0:
            i = int(np.argmin(h))
            h[i] += min(residual, 1.0 - h[i])
        else:
            i = int(np.argmax(h))
            h[i] += max(residual, -h[i])
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def objective_func(x, n):
    h = project_to_feasible(x, n)
    return compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    fo = 2.0 / n_old
    fn = 2.0 / n_new
    xo = (np.arange(n_old) + 0.5) * fo
    xn = (np.arange(n_new) + 0.5) * fn
    h_new = np.interp(xn, xo, h_old)
    return project_to_feasible(h_new, n_new)

def nm_optimize(h0, n, t_limit):
    _nm_state['t0'] = time.time()
    _nm_state['tmax'] = t_limit
    maxiter = max(200, int(t_limit * 400000.0 / max(n, 5)))
    res = minimize(objective_func, h0.copy(), args=(n,),
                   method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11},
                   callback=nm_callback)
    return project_to_feasible(res.x, n)

def grad_optimize(h0, n, t_limit, rng):
    h = h0.copy()
    c5 = compute_c5(h, n)
    dx = 2.0 / n
    t0 = time.time()
    alpha = 0.1
    stall = 0

    while time.time() - t0 < t_limit:
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)

        grad = np.zeros(n)
        k = k_star
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:] += (1.0 - h[:m])
                grad[:m] += -h[k:]
        else:
            m = -k
            if m < n:
                grad[m:] += (1.0 - h[:n-m])
                grad[:n-m] += -h[m:]
        grad *= dx
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        improved = False
        for _ in range(30):
            h_try = project_to_feasible(h - alpha * grad, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < c5 - 1e-14:
                h = h_try
                c5 = c5_try
                alpha = min(alpha * 1.2, 1.0)
                stall = 0
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stall += 1
            if stall >= 30 or alpha < 1e-8:
                h = project_to_feasible(h + rng.uniform(-0.1, 0.1, n), n)
                c5 = compute_c5(h, n)
                alpha = 0.1
                stall = 0
    return h, c5

def coord_polish(h, n, t_limit, rng):
    t0 = time.time()
    h = h.copy()
    c5 = compute_c5(h, n)
    while time.time() - t0 < t_limit:
        idx = int(rng.randint(n))
        saved = h.copy()
        for d in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.002, -0.002]:
            h_try = saved.copy()
            v = saved[idx] + d

                continue
            h_try[idx] = v
            h_try = project_to_feasible(h_try, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < c5 - 1e-14:
                h = h_try
                c5 = c5_try
                break
    return h, c5
