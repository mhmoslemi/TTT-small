def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h, n):
    b = (h > 0.5).copy().astype(float)
    n_ones = int(b.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(b == 1)[0]
        order = np.argsort(h[idx])
        b[idx[order[:n_ones - target]]] = 0
    elif n_ones < target:
        idx = np.where(b == 0)[0]
        order = np.argsort(-h[idx])
        b[idx[order[:target - n_ones]]] = 1
    return b

def sa_search(h, n, rng, time_budget, T_init=0.002):
    t_start = time.time()
    c5 = evaluate_c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    T = T_init
    patience = 0
    
    while time.time() - t_start < time_budget:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        zi = zeros[rng.randint(len(zeros))]
        oi = ones[rng.randint(len(ones))]
        h[zi] = 1.0
        h[oi] = 0.0
        new_c5 = evaluate_c5(h, n)
        delta = new_c5 - c5
        if delta <= 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            c5 = new_c5
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
            patience += 1
        else:
            h[zi] = 0.0
            h[oi] = 1.0
        if patience > 5000:
            T = max(1e-7, T * 0.3)
            patience = 0
        T = max(1e-7, T * 0.99998)
    
    return best_h, best_c5
