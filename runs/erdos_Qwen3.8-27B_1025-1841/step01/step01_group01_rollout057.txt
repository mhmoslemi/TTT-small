import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-9)[0]
    else:
        idx = np.where(h > 1e-9)[0]
    if len(idx) > 0:
        per = diff / len(idx)
        h[idx] += per
    else:
        h += diff / n
    return np.clip(h, 0.0, 1.0)

def interp_to(h_src, n_dst):
    if h_src is None or len(h_src) == 0:
        return np.full(n_dst, 0.5)
    if len(h_src) == n_dst:
        return h_src.copy()
    xs = np.linspace(0.0, 2.0, len(h_src))
    xd = np.linspace(0.0, 2.0, n_dst)
    hd = np.interp(xd, xs, h_src)
    return np.clip(hd, 0.0, 1.0)

def cosine_init(n):
    idx = np.arange(n)
    return np.cos(np.pi * idx / n) ** 2

def optimize_at_n(n, h_start, max_iter, t0, budget_s):
    if time.time() - t0 > budget_s - 30:
        return None, None
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h0 = fix_sum(h_start.copy(), n)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        h_res = np.clip(res.x, 0.0, 1.0)
        if abs(np.sum(h_res) - target) < 0.1:
            h_res = fix_sum(h_res, n)
            return h_res, compute_c5(h_res)
    except Exception:
        pass
    try:
        res = minimize(compute_c5, h0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        h_res = np.clip(res.x, 0.0, 1.0)
        h_res = fix_sum(h_res, n)
        if abs(np.sum(h_res) - target) < 0.1:
            return h_res, compute_c5(h_res)
    except Exception:
        pass
    return None, None

def try_pbm(n, h_start, t0, budget_s, max_iter):
    if time.time() - t0 > budget_s - 30:
        return None, None
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h0 = fix_sum(h_start.copy(), n)
    penalty = 10.0
    for _ in range(3):
        if time.time() - t0 > budget_s - 30:
            break
        def obj_p(x):
            return compute_c5(x) + penalty * (np.sum(x) - target) ** 2
        try:
            res = minimize(obj_p, h0, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            h_res = np.clip(res.x, 0.0, 1.0)
            h_res = fix_sum(h_res, n)
        except Exception:
            return None, None
        h0 = h_res
        penalty *= 10.0
    if abs(np.sum(h0) - target) < 0.05:
        return h0, compute_c5(h0)
    return None, None

    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Try initial construction
    try:

        if len(h_arr) > 0:
            best_h = h_arr
            best_n = len(h_arr)
            best_h = fix_sum(best_h, best_n)
            best_c5 = compute_c5(best_h)
    except (NameError, TypeError, AttributeError):
        pass

    # Cosine construction at several n
    for n_cos in [50, 90, 100, 150, 200]:
        if time.time() - t0 > budget_s - 60:
            break
        h_cos = cosine_init(n_cos)
        c5_cos = compute_c5(h_cos)
        if c5_cos < best_c5:
            best_c5 = c5_cos
            best_h = h_cos.copy()
            best_n = n_cos

    # Optimize at best resolution
    if best_h is not None:
        if time.time() - t0 < budget_s - 120:
            h_opt, c5_opt = optimize_at_n(best_n, best_h, 2000, t0, budget_s)
            if h_opt is not None and c5_opt < best_c5:

                best_h = h_opt

        if time.time() - t0 < budget_s - 120:

            if h_opt is not None and c5_opt < best_c5:

                best_h = h_opt

    # Random restarts
    n_cur = best_n if best_n > 0 else 90
    n_restarts = 15
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 180:
            break
        scale = 0.05 + 0.03 * i
        h_rand = best_h + rng.randn(n_cur) * scale
        h_rand = np.clip(h_rand, 0.0, 1.0)
        h_rand = fix_sum(h_rand, n_cur)

            best_h = h_opt

    for n_new in [80, 100, 120, 160, 200, 280, 360]:
        if time.time() - t0 > budget_s - 60:

        if n_new == best_n:

        h_start = interp_to(best_h, n_new)
        h_opt, c5_opt = optimize_at_n(n_new, h_start, 500, t0, budget_s)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_new

    # Final refinement
    if time.time() - t0 < budget_s - 40:
        h_opt, c5_opt = optimize_at_n(best_n, best_h, 3000, t0, budget_s)
        if h_opt is not None and c5_opt < best_c5:

            best_h = h_opt

            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = cosine_init(100)
        best_n = 100
    best_h = fix_sum(np.clip(best_h, 0.0, 1.0), best_n)
    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), best_c5, best_n)
