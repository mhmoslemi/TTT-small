import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_active(h, top_k=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top k active lags
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[np.argsort(corr[top_indices][::-1])]  # sort by value desc
    weights = corr[top_indices]
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_indices):
        s = idx - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[0:n-s] += w * (1 - h[s:n])
                grad[s:n] += w * (-h[0:n-s])
        else:
            ss = -s
            if n - ss > 0:
                grad[ss:n] += w * (1 - h[0:n-ss])
                grad[0:n-ss] += w * (-h[ss:n])
    grad *= dx
    return grad

def project_feasible(h, target):
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def optimize_level(h, time_limit, start_time):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.05
    patience = 0
    max_patience = 200
    
    while time.time() - start_time < time_limit:
        grad = gradient_active(h, top_k=3)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
            lr = min(lr * 1.05, 0.5)
        else:
            patience += 1
            lr *= 0.95
            h_new = h  # revert
        
        if patience > max_patience:
            # Add random perturbation
            pert = np.random.randn(n) * 0.01
            h_new = h + pert
            h_new = project_feasible(h_new, target)
            patience = 0
            lr = 0.1
        
        h = h_new
    
    return best_h, best_c5

def interpolate_up(h, factor=2):
    n = len(h)
    h_new = np.zeros(n * factor)
    for i in range(factor):
        h_new[i::factor] = h
    return h_new

    t1 = time.time()
    best_h, best_c5 = optimize_level(h, t1 + time_per_level - start_time, start_time)

    """Compute gradient of smoothed C5, using top-m active shifts."""
    n = len(h)
    dx = 2.0 / n

                grad[s:] -= w * h[:n-s]

            ss = -s

                grad[:n-ss] -= w * h[ss:]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def c5_grad(h, T=30.0, top_m=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    top_idx = np.argsort(f)[-top_m:]
    top_f = f[top_idx]
    exp_top = np.exp(T * (top_f - top_f.max()))
    weights = exp_top / exp_top.sum()
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        s = int(idx) - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n-s]
        else:
            ss = -s
            if n - ss > 0:
                grad[ss:] += w * (1.0 - h[:n-ss])
                grad[:n-ss] -= w * h[ss:]
    return grad * dx

def optimize_subgradient(h, time_limit_abs, T_init=30.0):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_size = 0.05
    T = T_init
    no_improve = 0

    while time.time() < time_limit_abs:
        grad = c5_grad(h, T=T, top_m=5)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        direction = -grad / gnorm
        h_trial = h + step_size * direction
        h_trial = project_feasible(h_trial, target)
        c5_trial = compute_c5(h_trial)

        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()

            no_improve = 0
            step_size = min(step_size * 1.02, 0.15)
        else:
            no_improve += 1
            step_size *= 0.96
            if no_improve > 400:
                h = best_h + np.random.randn(n) * 0.03
                h = project_feasible(h, target)
                no_improve = 0
                step_size = 0.06
                T = max(T * 0.7, 10.0)
            if step_size < 0.001:

                h = project_feasible(h, target)
                no_improve = 0
                step_size = 0.05
                T = max(T * 0.8, 10.0)

    return best_h, best_c5

def interp_up(h):
    n = len(h)
    h_new = np.empty(2 * n)
    h_new[0::2] = h
    h_new[1::2] = h
    return h_new

def run(seed=42, budget_s=1300, **kwargs):

    t0 = time.time()
    deadline = t0 + budget_s - 10
