Let me simplify the gradient. Actually, for the smooth max objective:
f(x) = (1/T) * log(Σ_k exp(T * g_k(x))) * (2/n)

The gradient is:
∇f = (2/(n*T)) * Σ_k p_k * ∇g_k(x)  where p_k = exp(T*g_k) / Σ_j exp(T*g_j)

And ∇g_k[j] can be computed as:
For lag k (where k ranges from -(n-1) to n-1):
- g_k = Σ_{i} x_{i+k} * (1 - x_i)  (sum over valid i)
- ∂g_k/∂x_j = Σ_i [δ_{j,i+k} * (1-x_i) + δ_{j,i} * (-x_{i+k})]

For j in [0, n-1]:
- First term: valid when i = j-k and 0 ≤ j-k < n → contributes (1-x_{j-k}) if 0 ≤ j-k < n
- Second term: valid when i = j and 0 ≤ j+k < n → contributes -x_{j+k} if 0 ≤ j+k < n

So: ∂g_k/∂x_j = (1-x_{j-k}) * [0≤j-k<n] + (-x_{j+k}) * [0≤j+k<n]

This is O(1) per (k,j) pair, so O(n²) total for all k and j.

Let me verify: for k=0:
∂g_0/∂x_j = (1-x_j) + (-x_j) = 1 - 2x_j
g_0 = Σ_i x_i(1-x_i), ∂g_0/∂x_j = 1-2x_j ✓

For k=1 (n≥2):
g_1 = Σ_{i=0}^{n-2} x_{i+1}(1-x_i)

∂g_1/∂x_{n-1} = (1-x_{n-2}) (only first term, i=n-2, j=n-1, j-1=n-2 valid, j+k=n invalid)

```python
def smooth_max_grad(x, n, T=10.0):
    """Compute value and gradient of smooth max of correlation"""
    c = np.correlate(x, 1.0 - x, mode='full')  # length 2n-1
    cT = c * T
    mx = cT.max()
    exp_c = np.exp(cT - mx)
    p = exp_c / exp_c.sum()  # softmax weights, length 2n-1
    
    # Value
    val = (mx + np.log(exp_c.sum())) / T * 2.0 / n  # = mx/T + log(sum)/T all * 2/n
    # Actually: (1/T)*log(sum(exp(cT))) * 2/n = (1/T)*(mx + log(sum(exp(c-mx))))*2/n
    
    # Gradient
    # For each lag index k_idx (0 to 2n-2), actual lag k = k_idx - (n-1)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if p[k_idx] < 1e-14:
            continue
        k = k_idx - (n - 1)
        w = p[k_idx]
        for j in range(n):
            g = 0.0
            if 0 <= j - k < n:
                g += (1.0 - x[j - k])
            if 0 <= j + k < n:
                g += (-x[j + k])
            grad[j] += w * g
    grad *= 2.0 / n
    return val, grad
```

        x = project(x, n)
        c = np.correlate(x, 1.0 - x, mode='full')
        cT = c * T
        mx = cT.max()
        ec = np.exp(cT - mx)
        s = ec.sum()
        val = (mx + np.log(s)) / T * (2.0 / n)
        p = ec / s
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            w = p[k_idx]
            if w < 1e-14:
                continue
            k = k_idx - (n - 1)
            a = max(0, k)
            b = min(n, k + n)
            if a < b:
                js = np.arange(a, b)
                grad[a:b] += w * (1.0 - x[js - k])
            a2 = max(0, -k)
            b2 = min(n, n - k)
            if a2 < b2:
                js2 = np.arange(a2, b2)
                grad[a2:b2] -= w * x[js2 + k]
        grad *= 2.0 / n
        return val, grad

    def thue_morse(n):
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
        return h

    def optimize_n(n, h0, T, maxiter):
        h0 = project(h0, n)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(lambda x: sm_obj(x, T, n)[0], h0,
                          jac=lambda x: sm_obj(x, T, n)[1],
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            return project(res.x, n)
        except:
            return h0

    def coord_round(h, n, max_sweeps=50):
        hc = project(h, n).copy()
        cur = c5_val(hc)
        for _ in range(max_sweeps):
            if time.time() - t0 > budget_s - 20:
                break
            any_improve = False
            for i in range(n):
                for nv in (0.0, 1.0):
                    if abs(hc[i] - nv) < 1e-9:
                        continue
                    saved = hc.copy()
                    hc[i] = nv
                    hc_try = project(hc, n)
                    v = c5_val(hc_try)
                    if v < cur - 1e-12:
                        hc = hc_try
                        cur = v
                        any_improve = True
                        break
                    hc = saved
            if not any_improve:
                break
        return hc

    n_list = [44, 48, 56, 64, 72, 80, 96, 100]

    for n in n_list:
        if time.time() - t0 > budget_s - 80:
            break
        n_trials = 3 if n <= 80 else 2
        for trial in range(n_trials):
            if time.time() - t0 > budget_s - 40:
                break

            if trial == 0 and h_init is not None:
                ni = len(h_init)
                x_new = (np.arange(n) + 0.5) * 2.0 / n
                x_old = (np.arange(ni) + 0.5) * 2.0 / ni
                h0 = np.interp(x_new, x_old, h_init)
            elif trial == 1:
                h0 = thue_morse(n)
            else:
                h0 = (rng.rand(n) > 0.5).astype(float)

            h0 = project(h0, n)
            T = 8.0
            h_opt = optimize_n(n, h0, T, 300)
            update(h_opt)

            # Try a second SLSQP with different T from the first result
            if time.time() - t0 < budget_s - 40:
                h_opt2 = optimize_n(n, h_opt, 15.0, 200)
                update(h_opt2)
                h_opt = h_opt2

            # Coordinate rounding
            h_rnd = coord_round(h_opt, n)
            update(h_rnd)

    # Final sweep: try to improve best solution with more rounding
    if best_h is not None and time.time() - t0 < budget_s - 10:
        h_final = coord_round(best_h, best_n, max_sweeps=200)
        update(h_final)

    return (best_h, best_c5, best_n)
