def gradient_step(h, ts, step_size=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    max_val = vals.max()
    # Active set: those within 10% of max
    active = vals > max_val * 0.9
    # Compute gradient of each active f_j
    # f_j = dx * corr[j], lag = j - (n-1)
    grad = np.zeros(n)
    for j in np.where(active)[0]:
        lag = j - (n - 1)
        # corr[j] = sum over valid i of h[i] * (1 - h[i-lag])
        # d/dh[p]: 
        # if p is the "first" index: contributes (1-h[p-lag])
        # if p is the "second" index: contributes -h[p+lag]
        for p in range(n):

            if 0 <= q < n:
                grad[p] += (1.0 - h[q])
            q2 = p + lag
            if 0 <= q2 < n:
                grad[p] -= h[q2]
    grad *= dx
    # Normalize
    gn = np.linalg.norm(grad)
    if gn > 0:
        grad /= gn
    # Step in negative gradient direction (to reduce max)
    h_new = h - step_size * grad
    h_new = project(h_new, ts)
    return h_new

    h0 = np.array(igv, dtype=float) if igv is not None else np.full(42, 0.5)
    n0 = len(h0)

    bh = project(h0, n0 / 2.0).copy()
    bc = compute_c5(bh)

    # Phase 1: SA at base resolution with multiple starts
    rng = np.random.RandomState(seed)
    starts = [bh.copy()]
    for _ in range(4):
        hs = project(rng.random(n0), n0 / 2.0)
        starts.append(hs)

    trial_time = max(10, min(120, left() / (len(starts) + 0.5)))
    for si, hs in enumerate(starts):
        if left() < 15:
            break
        ni = int(min(60000, trial_time * 600))
        hi, ci = sa(hs, ni, 0.08, 0.0003, seed + 7 * si)
        if ci < bc:
            bh, bc = hi, ci

    # Phase 2: Gradient descent at base resolution
    if left() > 20:
        gh, gc = gradient_descend(bh, bh.size / 2.0, n_iter=800)
        if gc < bc:
            bh, bc = gh, gc

    # Phase 3: SA + gradient at base
    if left() > 30:
        ni = int(min(40000, left() * 300))
        hi, ci = sa(bh, ni, 0.04, 0.0002, seed + 999)
        if left() > 15:
            hi, ci = gradient_descend(hi, hi.size / 2.0, n_iter=400)
        if ci < bc:
            bh, bc = hi, ci

    # Phase 4: Upsample and optimize
    for mult in [2, 4, 8]:
        rn = n0 * mult
        if left() < 50:
            break
        hu = interp_resample(bh, rn)
        # SA at new resolution
        ni = int(min(30000, left() * 200))
        hi, ci = sa(hu, ni, 0.03, 0.0002, seed + 5000 * mult)
        # Gradient at new resolution
        if left() > 20:
            hi, ci = gradient_descend(hi, rn / 2.0, n_iter=400)
        if ci < bc:
            bh, bc = hi, ci
            n0 = rn  # update for next upsample

    # Phase 5: Final polish
    if left() > 15:
        gh, gc = gradient_descend(bh, bh.size / 2.0, n_iter=500, lr0=0.002, lr1=0.00005)
        if gc < bc:
            bh, bc = gh, gc

    if left() > 10:
        ni = int(min(20000, left() * 400))
        hi, ci = sa(bh, ni, 0.01, 0.0001, seed + 77777)
        if ci < bc:
            bh, bc = hi, ci

    bc = compute_c5(bh)
    return bh, bc, len(bh)
