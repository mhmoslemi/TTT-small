```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_target(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        adj = diff / n
        h_new = np.clip(h + adj, 0.0, 1.0)
        if np.allclose(h_new, h, atol=1e-15):
            h = h_new
            break
        h = h_new
    # Final correction
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-12:
        if rem > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            h[idx] += rem / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_target(h_new, n_new)

def coord_descent(h, n, max_sweeps=1000, tol=1e-10, time_limit=None, t0=None):
    dx = 2.0 / n
    h = project_target(h, n).copy()
    best_c5 = compute_c5(h, dx)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > t0 + time_limit:
            break
        improved = False
        for pos in range(n):
            old = h[pos]
            # Find optimal t in [0,1] for h[pos]
            # Use ternary search on convex f(t) = max_k c[k](t)
            def f1d(t):
                h[pos] = t
                corr = compute_corr(h)
                return np.max(corr)
            
            # Ternary search
            a, b = 0.0, 1.0
            for _ in range(40):
                m1 = a + (b - a) / 3
                m2 = b - (b - a) / 3
                if f1d(m1) < f1d(m2):
                    b = m2
                else:
                    a = m1
            t_opt = (a + b) / 2
            h[pos] = t_opt
            h = project_target(h, n)
            if abs(t_opt - old) > 1e-10:
                improved = True
        
        c5_curr = compute_c5(h, dx)
        if c5_curr < best_c5:
            best_c5 = c5_curr
        if not improved and sweep > 5:
            break
    
    return h, best_c5
```

The breakpoints are at t = (c0[j] - c0[i]) / (c1[i] - c1[j]) for pairs (i,j) where c1[i] ≠ c1[j]. There are O(n²) such pairs, but the actual number of breakpoints of the max function is O(n) (each line can be on the upper envelope for at most one interval).

1. Evaluate f at t=0 and t=1
2. If f is increasing at 0 (all active slopes positive) and decreasing at 1 (all active slopes negative), the minimum is in between
3. Use binary search: at each t, compute the argmax. If the slope at t (c1[argmax]) is ≥ 0, move left; else move right.
