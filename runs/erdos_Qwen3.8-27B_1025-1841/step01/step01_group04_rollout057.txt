import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def soft_max_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    w = exps / np.sum(exps)
    k_range = np.arange(2 * n - 1)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for j in range(n):
        idx1 = j - k_range + n - 1
        m1 = (idx1 >= 0) & (idx1 < n)
        term1[j] = np.dot(w[m1], 1.0 - x[idx1[m1]])
        idx2 = j + k_range - n + 1
        m2 = (idx2 >= 0) & (idx2 < n)
        term2[j] = np.dot(w[m2], x[idx2[m2]])
    return dx * (term1 - term2)

def _f(x):
    n, T, lam = _state['n'], _state['T'], _state['lam']
    val = soft_max_val(x, T)
    s = np.sum(x) - 0.5 * n
    return val + lam * s * s

def _g(x):
    n, T, lam = _state['n'], _state['T'], _state['lam']
    g = soft_max_grad(x, T)
    g += 2.0 * lam * (np.sum(x) - 0.5 * n)
    return g

def project(h, n):
    target = 0.5 * n
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def optimize_lbfgs(h0, n, T, lam, maxiter):
    _state['n'] = n
    _state['T'] = T
    _state['lam'] = lam
    try:
        res = minimize(_f, h0, jac=_g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'ftol': 1e-15, 'maxiter': maxiter, 'gtol': 1e-7})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def polish_max(h, n, n_steps, deadline):
    h = h.copy()
    dx = 2.0 / n
    k_range = np.arange(2 * n - 1)
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        w = np.zeros(2 * n - 1)
        w[k_max] = 1.0
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        for j in range(n):
            idx1 = j - k_range + n - 1
            m1 = (idx1 >= 0) & (idx1 < n)
            term1[j] = np.dot(w[m1], 1.0 - h[idx1[m1]])
            idx2 = j + k_range - n + 1
            m2 = (idx2 >= 0) & (idx2 < n)
            term2[j] = np.dot(w[m2], h[idx2[m2]])
        grad = dx * (term1 - term2)
        lr = 0.05
        improved = False
        for _ls in range(15):
            h_new = project(h - lr * grad, n)
            if evaluate_c5(h_new) < evaluate_c5(h):
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(42, 0.5)

    best_h = project(h_start, len(h_start))
    best_c5 = evaluate_c5(best_h)

    n_start = len(h_start)
    resolutions = []
    for factor in [1, 2, 3, 5, 8, 12]:
        n_try = n_start * factor
        if n_try not in resolutions:
            resolutions.append(n_try)
    resolutions.sort()

    T_schedule = [15.0, 40.0, 100.0, 250.0, 500.0, 1000.0]

    for n in resolutions:
        if time.time() > deadline - 80:
            break
        if n > 5000:
            break

        if n == len(best_h):
            h_work = best_h.copy()
        else:
            h_work = interp_to(best_h, len(best_h), n)
            h_work = project(h_work, n)

        for T in T_schedule:
            if time.time() > deadline - 50:
                break
            remaining = deadline - time.time()
            if remaining < 20:
                break
            maxiter = min(1000, max(100, int(remaining * 1.5)))
            lam = 1e4 if n <= 500 else 1e5

            h_new = optimize_lbfgs(h_work, n, T, lam, maxiter)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_work = h_new

        if time.time() < deadline - 40 and n >= n_start:
            h_pol = polish_max(h_work, n, 40, deadline)
            c5_pol = evaluate_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
            h_work = h_pol

    n_best = len(best_h)
    remaining = deadline - time.time()
    n_restarts = max(5, min(int(remaining / 25), 40))

    for i in range(n_restarts):
        if time.time() > deadline - 15:
            break
        scale = 0.03 + 0.12 * (i % 4)
        h_pert = best_h + np.random.randn(n_best) * scale
        h_pert = project(h_pert, n_best)
        T_r = 80.0 + 120.0 * (i % 5)
        maxiter_r = min(600, max(80, int((deadline - time.time()) * 0.4)))
        h_new = optimize_lbfgs(h_pert, n_best, T_r, 1e4, maxiter_r)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if time.time() < deadline - 60:
            h_pol = polish_max(h_new, n_best, 20, deadline)
            c5_pol = evaluate_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if time.time() < deadline - 20:
        for i in range(5):
            if time.time() > deadline - 10:
                break
            h_rand = np.random.rand(n_best)
            h_rand = project(h_rand, n_best)
            h_new = optimize_lbfgs(h_rand, n_best, 50.0, 1e4, 200)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    if time.time() < deadline - 5:
        best_h = polish_max(best_h, n_best, 30, deadline)
        best_c5 = evaluate_c5(best_h)

    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
