       current_sum = np.sum(h_clipped)

       best_targets = targets

           if abs(current_sum - targets) < tol:
               repaired = h_clipped
           else:
               rep_scale = best_targets / current_sum
               h_scaled = h_clipped * rep_scale
               h_scaled_clipped = np.clip(h_scaled, 0, 1)
               h_repaired = h_scaled_clipped
               sum_adjusted = np.sum(h_repaired)
               if abs(sum_adjusted - best_targets) < tol:
                   repaired = h_repaired
               else:
                   # Uniform fallback
