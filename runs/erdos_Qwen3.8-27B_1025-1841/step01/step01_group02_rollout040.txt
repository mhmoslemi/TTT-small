import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def active_lags(h, top_m=10):
    """Return the top_m lags with highest overlap."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    values = corr * dx
    idx = np.argsort(values)[::-1][:top_m]
    return idx, values[idx]

def subgradient_step(h, lag_k, n, step_size):
    """Compute subgradient for lag k and take a step."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag_k == 0:
        for i in range(n):
            grad[i] = dx * (1 - 2*h[i])
    else:
        k = abs(lag_k)
        # g_k = dx * sum_{i=0}^{n-k-1} h_i (1-h_{i+k})
        # d/dh_j: if j < n-k: dx*(1-h_{j+k}); if j >= k: dx*(-h_{j-k})
        for i in range(n - k):
            grad[i] += dx * (1 - h[i+k])
            grad[i+k] -= dx * h[i]
    # Project gradient step
    h_new = h - step_size * grad
    h_new = make_feasible(h_new, n)
    return h_new

def coordinate_refine(h, n, n_rounds=3):
    """For each coordinate, try a range of values and pick the best."""
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = h[j]
            best_val = old_val
            best_c = best_c5
            # Try a grid of values around current
            candidates = np.linspace(0, 1, 21)
            # Also try values near current
            candidates = np.unique(np.concatenate([candidates, 
                                                    np.clip(old_val + np.linspace(-0.1, 0.1, 11), 0, 1)]))
            for c in candidates:
                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_c - 1e-12:
                    best_c = c5
                    best_val = c
            if best_val != old_val:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

2. Finding the breakpoints in [0,1] (O(n²) worst case, but typically much less)

                              bounds=[(0.0, 1.0)] * N,
                              constraints=[{'type': 'eq',
                                            'fun': lambda x: _con_fun(x, N),
                                            'jac': lambda x: _con_jac(x, N)}],
                              options={'maxiter': maxiter, 'ftol': 1e-10})
                c = compute_c5(res.x)
                if c < best_c5:
                    best_c5 = c
                    best_h = res.x.copy()
                    h = res.x.copy()
            except Exception:
                pass

        # Main coordinate descent
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 15:
            if n_pts <= 400:
                n_r = min(20, int(remaining * 0.4))
            elif n_pts <= 700:
                n_r = min(12, int(remaining * 0.3))
            else:
                n_r = min(8, int(remaining * 0.2))
            h, c = coord_descent(h, n_pts, n_rounds=max(n_r, 4), grid_size=41)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

        # Perturbation + re-optimize
        elapsed = time.time() - start_time
        if budget_s - elapsed > 30:
            dx = 2.0 / n_pts
            for attempt in range(6):
                elapsed = time.time() - start_time
                if budget_s - elapsed < 25:
                    break
                if attempt == 0:
                    h_p = best_h + rng.randn(n_pts) * 0.03
                elif attempt == 1:
                    h_p = best_h + rng.randn(n_pts) * 0.07
                elif attempt == 2:
                    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                    k_star = np.argmax(corr)
                    lag = k_star - (n_pts - 1)
                    k = abs(lag)
                    grad = np.zeros(n_pts)
                    if k == 0:
                        grad = dx * (1.0 - 2.0 * best_h)
                    else:
                        for i in range(n_pts - k):
                            grad[i] += dx * (1.0 - best_h[i + k])
                            grad[i + k] -= dx * best_h[i]
                    h_p = best_h - 0.008 * grad
                elif attempt == 3:
                    h_bin = (best_h > 0.5).astype(float)
                    h_bin = make_feasible(h_bin, n_pts)
                    h_p = 0.6 * best_h + 0.4 * h_bin
                elif attempt == 4:
                    h_p = best_h.copy()
                    bs = rng.randint(0, n_pts // 3)
                    bl = rng.randint(4, max(5, n_pts // 5))
                    h_p[bs:bs + bl] = 1.0 - h_p[bs:bs + bl]
                else:
                    # Score-based: add noise weighted by local overlap
                    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                    h_p = best_h + rng.randn(n_pts) * 0.05 * (1.0 - best_h) * best_h

                h_p = make_feasible(h_p, n_pts)
                h_p, c_p = coord_descent(h_p, n_pts, n_rounds=4, grid_size=31)
                if c_p < best_c5:
                    best_c5 = c_p
                    best_h = h_p.copy()

        # Fine coordinate descent
        elapsed = time.time() - start_time
        if budget_s - elapsed > 15:
            h, c = coord_descent(best_h, n_pts, n_rounds=4, grid_size=81)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

    # Final ultra-fine polish
    n_pts = len(best_h)
    elapsed = time.time() - start_time
    if budget_s - elapsed > 10:
        h, c = coord_descent(best_h, n_pts, n_rounds=3, grid_size=121)
        if c < best_c5:
            best_c5 = c
            best_h = h

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
