import numpy as np
from scipy.optimize import minimize
import time

def project_onto_feasible(x, target):
    """Project x onto {0<=x_i<=1, sum(x_i)=target}."""
    x = np.array(x, dtype=float).copy()
    n = len(x)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x_proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(x_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    x_proj = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - np.sum(x_proj)
    if abs(diff) < 1e-10:
        return x_proj
    # Distribute remainder
    if diff > 0:
        mask = x_proj < 1.0
    else:
        mask = x_proj > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        x_proj[mask] += diff / cnt
    x_proj = np.clip(x_proj, 0.0, 1.0)
    return x_proj

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    """Project x onto {0<=x_i<=1, sum(x_i)=target}."""
    x = np.asarray(x, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        xp = np.clip(x - mid, 0.0, 1.0)
        s = xp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    xp = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - xp.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = xp < 1.0 - 1e-15
        else:
            mask = xp > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            xp = xp.copy()
            xp[mask] += diff / cnt
            xp = np.clip(xp, 0.0, 1.0)
    return xp

def corr_full(h):
    return np.correlate(h, 1.0-h, mode='full')

def obj_value(h, T=None):
    """Compute the actual C5 or smoothed objective."""
    n = len(h)
    dx = 2.0/n
    corr = corr_full(h)
    if T is None:
        return float(np.max(corr * dx))
    else:
        scaled = corr * dx * T
        scaled -= scaled.max()
        e = np.exp(scaled)
        return float((scaled.max() + np.log(e.sum())) / T * dx)

def grad_smooth(h, T):
    """Analytical gradient of softmax objective."""
    n = len(h)
    dx = 2.0/n
    corr = corr_full(h)
    scaled = corr * dx * T
    scaled -= scaled.max()
    e = np.exp(scaled)
    p = e / e.sum()
    
    grad = np.zeros(n)
    nm1 = n - 1
    for j in range(n):
        # First term
        k_lo = max(0, nm1 - j)
        k_hi = min(2*n-2, 2*n-2-j)
        if k_lo <= k_hi:
            m = j + np.arange(k_lo, k_hi+1) - nm1
            t1 = np.dot(p[k_lo:k_hi+1], 1.0 - h[m])
        else:
            t1 = 0.0
        # Second term
        k_lo2 = max(0, j)
        k_hi2 = min(2*n-2, j+nm1)
        if k_lo2 <= k_hi2:
            m2 = j - np.arange(k_lo2, k_hi2+1) + nm1
            t2 = np.dot(p[k_lo2:k_hi2+1], h[m2])
        else:
            t2 = 0.0
        grad[j] = dx * (t1 - t2)
    return grad

def project_grad_desc(h0, n, T, max_iter, deadline, step0=0.01):
    """Projected gradient descent with backtracking."""
    target = n / 2.0
    h = h0.copy()
    best_c5 = obj_value(h)
    h_best = h.copy()
    
    step = step0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_smooth(h, T)
        # Backtracking line search
        for _ in range(20):
            h_new = h - step * g
            h_new = project_feasible(h_new, target)
            f_new = obj_value(h_new, T)
            if f_new < obj_value(h, T):
                break
            step *= 0.5
        else:
            break
        h = h_new
        # Update best
        c5 = obj_value(h)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()
        step = min(step * 1.05, step0)  # slowly increase
    
    return h_best, best_c5

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):

    t0 = time.time()
    deadline = t0 + budget_s - 10

    else:
        h_start = np.full(400, 0.5)
    
    n_start = len(h_start)
    h_start = project_feasible(h_start, n_start/2.0)
    
    best_h = h_start.copy()
    best_c5 = obj_value(best_h)
    best_n = n_start
    
    # Multi-resolution strategy
    resolutions = sorted(set([n_start, 504, 672]))
    # Filter: only go up from start
    resolutions = [r for r in resolutions if r >= n_start]
    
    T_schedule = [10, 30, 100, 300, 500, 800]

        if time.time() > deadline - 30:
            break
        
        if n == best_n:
            h_work = best_h.copy()
        else:
            h_work = interp_to(best_h, best_n, n)
            h_work = project_feasible(h_work, n/2.0)
        
        for T in T_schedule:
            if time.time() > deadline - 30:
                break

            remaining = deadline - time.time()
            max_it = min(5000, max(200, int(remaining * 50)))  # ~50 iters/sec estimate
            if n > 500:
                max_it = min(max_it, 3000)
            
            h_new, c5_new = project_grad_desc(h_work, n, T, max_it, deadline, step0=0.005)
            c5_check = obj_value(h_new)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h_new.copy()
                best_n = n
            h_work = h_new
        
        # After PGD at this resolution, try SLSQP polish
        if time.time() < deadline - 60 and n <= 504:
            h_polish = h_work.copy()
            def slsqp_obj(x):
                return obj_value(x, T=300)
            def slsqp_eq(x):
                return np.sum(x) - n/2.0
            try:
                res = minimize(slsqp_obj, h_polish, method='SLSQP',

                             options={'maxiter':200, 'ftol':1e-12})
                if res.success or (res.x is not None and np.all(np.isfinite(res.x))):

                    c5_s = obj_value(h_s)
                    if c5_s < best_c5:
                        best_c5 = c5_s
                        best_h = h_s.copy()
                        best_n = n

    # Random restarts at best resolution
    n_best = best_n
    n_restarts = 10
    for r in range(n_restarts):
        if time.time() > deadline - 20:

        h_pert = best_h + np.random.randn(n_best) * 0.05
        h_pert = project_feasible(h_pert, n_best/2.0)

                best_h = h_new.copy()
            h_pert = h_new

                best_h = h_new.copy()
    
    # Final evaluation
    best_c5 = obj_value(best_h)
    return (best_h, best_c5, len(best_h))

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    x = np.asarray(x, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        xp = np.clip(x - mid, 0.0, 1.0)
        s = xp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    xp = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - xp.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = xp < 1.0 - 1e-15
        else:
            mask = xp > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            xp = xp.copy()
            xp[mask] += diff / cnt
            xp = np.clip(xp, 0.0, 1.0)
    return xp

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def obj_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(corr_full(h)) * dx)

def obj_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    s = c * T * dx
    s -= s.max()
    e = np.exp(s)
    return float((s.max() + np.log(e.sum())) / T * dx)

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    s = c * T * dx
    s -= s.max()
    e = np.exp(s)
    p = e / e.sum()
    nm1 = n - 1
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(0, nm1 - j)
        k_hi = min(2 * n - 2, 2 * n - 2 - j)
        t1 = 0.0
        if k_lo <= k_hi:
            m = j + np.arange(k_lo, k_hi + 1) - nm1
            t1 = np.dot(p[k_lo:k_hi + 1], 1.0 - h[m])
        k_lo2 = max(0, j)
        k_hi2 = min(2 * n - 2, j + nm1)
        t2 = 0.0
        if k_lo2 <= k_hi2:
            m2 = j - np.arange(k_lo2, k_hi2 + 1) + nm1
            t2 = np.dot(p[k_lo2:k_hi2 + 1], h[m2])
        grad[j] = dx * (t1 - t2)
    return grad

def pgd_optimize(h0, T, max_iter, deadline, step0=0.005, lr_decay=0.999):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    f_cur = obj_smooth(h, T)
    best_f = f_cur
    best_c5 = obj_exact(h)
    h_best = h.copy()
    step = step0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_smooth(h, T)
        for _ in range(30):
            h_new = h - step * g
            h_new = project_feasible(h_new, target)
            f_new = obj_smooth(h_new, T)
            if f_new < f_cur:
                break
            step *= 0.5
        else:
            break
        h = h_new
        f_cur = obj_smooth(h, T)
        if f_cur < best_f:
            best_f = f_cur
            c5 = obj_exact(h)
            if c5 < best_c5:
                best_c5 = c5
                h_best = h.copy()
        step = min(step * 1.01, step0 * lr_decay)
    return h_best, best_c5

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def slsqp_polish(h0, n, deadline):
    target = n / 2.0
    def f(x):
        return obj_smooth(x, 500.0)
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'maxiter': 300, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, target)
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1300, **kwargs):

    t0 = time.time()

        for T in T_schedule:

            iters_per_sec = 800 if n <= 400 else (400 if n <= 550 else 200)

            step0 = 0.005 if T < 100 else (0.002 if T < 500 else 0.0008)
            h_new, c5_new = pgd_optimize(h_work, T, max_it, deadline, step0=step0)

                best_n = n

    # Random restarts
