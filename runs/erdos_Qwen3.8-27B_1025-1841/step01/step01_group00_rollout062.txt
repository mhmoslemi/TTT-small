    """Project to {h: 0<=h<=1, sum(h)=n/2} using bisection on threshold."""

    lo = g.min() - 2.0

        if s > target:
            lo = mid
        else:

def fast_optimize(h, n, time_limit, tau=30.0, top_k=5, lr=0.3):
    """Optimize h using top-k softmax smooth max with projected gradient."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    start = time.time()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    
    while time.time() - start < time_limit:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Top-k softmax for smooth max
        top_k_actual = min(top_k, len(corr))
        idx = np.argpartition(corr, -top_k_actual)[-top_k_actual:]
        top_vals = corr[idx]
        max_val = top_vals.max()
        weights = np.exp(tau * (top_vals - max_val))
        weights /= weights.sum()
        
        # Gradient w.r.t. h
        grad = np.zeros(n)
        for wi, ki in zip(weights, idx):
            s = ki - (n - 1)
            # ∂/∂h[j] of Σ_i h[i](1-h[i+s]) for valid i
            # = (1-h[j+s]) - h[j-s] (with boundary handling)
            j = np.arange(n)
            # Term 1: (1-h[j+s]) for valid j where j+s in [0,n)
            js = j + s
            valid1 = (js >= 0) & (js < n)
            grad[valid1] += wi * (1.0 - h[js[valid1]])
            # Term 2: -h[j-s] for valid j where j-s in [0,n)
            js2 = j - s
            valid2 = (js2 >= 0) & (js2 < n)
            grad[valid2] -= wi * h[js2[valid2]]
        
        grad *= dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h = project_to_feasible(h_new, n)
    
    return best_h, best_c5

def compute_grad_shift(h, s):

    grad = np.zeros(n)
    # Term 1: I[0 <= j+s < n] * (1 - h[j+s])
    # Valid j: max(0, -s) <= j < min(n, n-s)
    j_lo = max(0, -s)
    j_hi = min(n, n - s)
    if j_lo < j_hi:

    # Term 2: -I[0 <= j-s < n] * h[j-s]
    # Valid j: max(0, s) <= j < min(n, n+s)
    j_lo2 = max(0, s)
    j_hi2 = min(n, n + s)
    if j_lo2 < j_hi2:
        grad[j_lo2:j_hi2] -= h[j_lo2 - s : j_hi2 - s]
    return grad

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def compute_grad_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    j_lo = max(0, -s)
    j_hi = min(n, n - s)
    if j_lo < j_hi:
        grad[j_lo:j_hi] += (1.0 - h[j_lo + s:j_hi + s])
    j_lo2 = max(0, s)
    j_hi2 = min(n, n + s)
    if j_lo2 < j_hi2:
        grad[j_lo2:j_hi2] -= h[j_lo2 - s:j_hi2 - s]
    return grad

def optimize(h_init, n, time_limit, tau=20.0, top_k=3, lr_init=0.3):
    dx = 2.0 / n
    h = project_to_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    t = 0
    lr = lr_init
    stall_count = 0
    
    while time.time() - start < time_limit:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        
        if c5_cur < best_c5 - 1e-11:
            best_c5 = c5_cur
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Compute top-k indices
        k_actual = min(top_k, 2*n-1)
        top_idx = np.argpartition(corr, -k_actual)[-k_actual:]
        top_vals = corr[top_idx]
        max_val = top_vals.max()
        
        if t > 100:
            # Increase tau over time to focus more on the max
            tau_eff = min(tau * (1 + t / 500.0), tau * 3)
        else:
            tau_eff = tau
        
        weights = np.exp(tau_eff * (top_vals - max_val))
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            s = ki - (n - 1)
            grad += wi * compute_grad_shift(h, s)
        grad *= dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        
        # Learning rate schedule
        if stall_count > 200:
            lr *= 0.7
            m *= 0.5
            v *= 0.5
            stall_count = 0
            if lr < 1e-10:
                break
    
    return best_h, best_c5

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def compute_grad_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    j_lo1 = max(0, s)
    j_hi1 = min(n, s + n)
    if j_lo1 < j_hi1:
        grad[j_lo1:j_hi1] += (1.0 - h[j_lo1 - s:j_hi1 - s])
    j_lo2 = max(0, -s)
    j_hi2 = min(n, n - s)
    if j_lo2 < j_hi2:
        grad[j_lo2:j_hi2] -= h[j_lo2 + s:j_hi2 + s]
    return grad


def optimize_smooth(h_init, n, time_limit, tau=20.0, top_k=3, lr_init=0.3):
    dx = 2.0 / n
    h = project_to_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    t = 0
    lr = lr_init
    stall = 0
    while time.time() - start < time_limit:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_act = min(top_k, 2 * n - 1)
        top_idx = np.argpartition(corr, -k_act)[-k_act:]
        top_vals = corr[top_idx]
        max_val = top_vals.max()
        tau_eff = tau * min(1.0 + 0.3 * (t / max(1, int(time_limit / 100.0))), 3.0)
        exp_arr = np.exp(tau_eff * (top_vals - max_val))
        wsum = exp_arr.sum()
        weights = exp_arr / wsum if wsum > 1e-30 else np.ones(k_act) / k_act
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            if wi > 1e-12:
                s = ki - (n - 1)
                grad += wi * compute_grad_shift(h, s)
        grad *= dx
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        if stall > 300:
            lr *= 0.6
            m *= 0.3
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
    return best_h, best_c5

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Optimize at parent resolution with top-3 smooth max
    if remaining() > 120:
        tl = min(280.0, remaining() * 0.22)
        h1, c5_1 = optimize_smooth(h_init, n_init, tl, tau=20, top_k=3, lr_init=0.3)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Up-resolve to 2x and optimize
    n2 = n_init * 2
    if remaining() > 120:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(280.0, remaining() * 0.22)
        h2, c5_2 = optimize_smooth(h2, n2, tl, tau=15, top_k=3, lr_init=0.2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Top-2 focus optimization (narrower softmax)
    n_cur = len(best_h)
    if remaining() > 80:
        tl = min(180.0, remaining() * 0.15)
        h3, c5_3 = optimize_topk_focus(best_h, n_cur, tl, tau=30, top_k=2, lr_init=0.12)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Subgradient on the active shift
    if remaining() > 60:
        n_cur = len(best_h)
        tl = min(150.0, remaining() * 0.12)
        h4, c5_4 = optimize_subgrad(best_h, n_cur, tl, lr_init=0.008)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Try 4x resolution
    n4 = n_init * 4
    if remaining() > 100:
        h5 = interp_periodic(best_h, n4)
        h5 = project_to_feasible(h5, n4)
        tl = min(150.0, remaining() * 0.12)
        h5, c5_5 = optimize_smooth(h5, n4, tl, tau=15, top_k=3, lr_init=0.12)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if remaining() > 40:
            tl2 = min(80.0, remaining() * 0.06)
            h5b, c5_5b = optimize_subgrad(h5, n4, tl2, lr_init=0.005)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: Random restarts with small perturbations
    n_final = len(best_h)
    if remaining() > 60:
        for trial in range(4):
            if remaining() < 30:
                break
            noise_scale = 0.03 + 0.02 * trial
            noise = np.random.randn(n_final) * noise_scale
            h_trial = project_to_feasible(best_h + noise, n_final)
            tl = min(40.0, remaining() * 0.06)
            ht, ct = optimize_smooth(h_trial, n_final, tl, tau=25, top_k=2, lr_init=0.1)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()

    # Phase 7: Final subgrad polish
    if remaining() > 20:
        n_final = len(best_h)
        tl = min(50.0, remaining() * 0.05)
        h_fin, c5_fin = optimize_subgrad(best_h, n_final, tl, lr_init=0.003)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
