import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * dx, c

def project_feasible(x, n, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h < 1) if diff > 0 else (h > 0)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0, 1)
    return h

def subgradient_step(h, c, n, dx, step_size):
    """One subgradient step for minimizing max_k c[k]"""
    max_val = np.max(c)
    # Get subgradient (using all argmax lags for stability)
    tol = 1e-10
    active = np.where(np.abs(c - max_val) < tol)[0]
    
    g = np.zeros(n)
    for k in active:
        period = k - (n - 1)
        # ∂c[k]/∂h[j] = (1-h[j-period]) if valid - h[j+period] if valid
        # j-period valid: 0 <= j-period < n AND j-period in [max(0,-period), min(n, n-period))
        # j valid: 0 <= j < n AND j in [max(0,-period), min(n, n-period))
        
        n_lo = max(0, -period)
        n_hi = min(n, n - period)  # exclusive
        
        # For each j, the contribution is:
        # (1-h[j-period]) * I(j-period in [n_lo, n_hi)) - h[j+period] * I(j in [n_lo, n_hi))
        
        # First term: j-period in [n_lo, n_hi) means j in [n_lo+period, n_hi+period)
        j_lo1 = n_lo + period
        j_hi1 = n_hi + period
        # Clip to [0, n)
        j_lo1 = max(0, j_lo1)
        j_hi1 = min(n, j_hi1)
        for j in range(j_lo1, j_hi1):
            g[j] += (1 - h[j - period])
        
        # Second term: j in [n_lo, n_hi)
        j_lo2 = max(0, n_lo)
        j_hi2 = min(n, n_hi)
        for j in range(j_lo2, j_hi2):
            g[j] -= h[j + period]
    
    g /= len(active)
    return g * dx

                t2 = np.where((jp2 >= 0) & (jp2 < n), h[np.clip(jp2, 0, n - 1)], 0.0)
                t3 = np.where((jm1 >= 0) & (jm1 < n), h[np.clip(jm1, 0, n - 1)], 0.0)
                t4 = np.where((jm2 >= 0) & (jm2 < n), h[np.clip(jm2, 0, n - 1)], 0.0)
                a = t1 + t3 - t2 - t4
                max_d = min(h[j1], 1.0 - h[j2])
                min_d = -min(h[j2], 1.0 - h[j1])
                if max_d < 1e-12 and min_d > -1e-12:
                    continue
                span = max_d - min_d
                if span < 1e-12:
                    continue
                deltas = min_d + fracs * span
                new_vals = np.max(c_vals[None, :] + deltas[:, None] * a[None, :], axis=1) * dx
                idx_best = np.argmin(new_vals)
                best_val = new_vals[idx_best]
                if best_val < c5 - 1e-14:
                    delta = deltas[idx_best]
                    h[j1] -= delta
                    h[j2] += delta
                    np.clip(h, 0.0, 1.0, out=h)
                    c5 = best_val
                    improved = True
                    c_vals = c_vals + delta * a
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_init

    # Phase 1: Local search on initial resolution
    limit1 = start_time + budget_s * 0.30
    h1, c5_1 = local_search(h_init.copy(), n_init, limit1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n_init

    # Phase 2: Multi-start with perturbation on same resolution
    for attempt in range(5):
        if time.time() - start_time > budget_s * 0.50:
            break
        h_pert = h1.copy()
        n_pert = rng.integers(3, 10)
        idx = rng.choice(n_init, size=n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.15, 0.15, n_pert)
        h_pert = project_feasible(h_pert, n_init / 2.0)
        limit2 = start_time + budget_s * 0.50
        h2, c5_2 = local_search(h_pert, n_init, limit2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n_init

    # Phase 3: Try higher resolutions
    for n_new in [90, 120, 180]:
        if time.time() - start_time > budget_s * 0.85:
            break
        x_old = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h, period=2.0)
        h_new = project_feasible(h_new, n_new / 2.0)
        limit3 = start_time + budget_s * 0.85
        h3, c5_3 = local_search(h_new, n_new, limit3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n_new

    # Phase 4: Final local search refinement
    if time.time() - start_time < budget_s * 0.95:
        best_h, best_c5 = local_search(best_h, best_n, start_time + budget_s * 0.95)

    # Ensure sum constraint is met as precisely as possible
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-15:
        mask = (best_h < 1.0) if diff > 0 else (best_h > 0.0)
        cnt = np.sum(mask)
        if cnt > 0:
            best_h[mask] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
