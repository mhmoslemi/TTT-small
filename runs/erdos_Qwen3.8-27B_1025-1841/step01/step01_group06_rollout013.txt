        h2 = cd_phase(h2, n2, 2.0/n2, 100000, t0 + budget_s * 0.42, hard_frac=0.7)
        h2 = project_target(h2, n2)
        c5 = compute_c5(h2, 2.0/n2)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h2.copy(), n2

    # Phase 3: CD at 4*n0
    n4 = 4 * n0
    if n4 <= 840:
        h4 = upsample(best_h, n4)
        h4 = cd_phase(h4, n4, 2.0/n4, 50000, t0 + budget_s * 0.65, hard_frac=0.6)
        h4 = project_target(h4, n4)
        c5 = compute_c5(h4, 2.0/n4)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h4.copy(), n4

    # Phase 4: Back to n2 for refinement with soft CD then hard CD
    if n2 <= 560 and time.time() < t0 + budget_s * 0.80:
        h2r = upsample(best_h, n2)
        h2r = cd_phase(h2r, n2, 2.0/n2, 80000, t0 + budget_s * 0.80, hard_frac=0.4, soft_temp=0.05)
        h2r = project_target(h2r, n2)
        c5 = compute_c5(h2r, 2.0/n2)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h2r.copy(), n2

    # Phase 5: L-BFGS-B polish at best resolution
    if time.time() < deadline - 20:
        n_pol = best_n
        dx_pol = 2.0 / n_pol
        h_pol = best_h + np.random.RandomState(seed).randn(n_pol) * 0.02
        h_pol = project_target(h_pol, n_pol)
        h_pol = lbfgs_polish(h_pol, n_pol, dx_pol, n_pol / 2.0, deadline)
        c5 = compute_c5(h_pol, dx_pol)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_pol.copy(), n_pol

    # Phase 6: Final CD + polish at best resolution
    if time.time() < deadline - 10:
        n_fin = best_n
        dx_fin = 2.0 / n_fin
        h_fin = best_h.copy()
        h_fin = cd_phase(h_fin, n_fin, dx_fin, 30000, deadline - 15, hard_frac=0.8)
        h_fin = project_target(h_fin, n_fin)
        h_fin = lbfgs_polish(h_fin, n_fin, dx_fin, n_fin / 2.0, deadline)
        h_fin = project_target(h_fin, n_fin)
        c5 = compute_c5(h_fin, dx_fin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fin.copy(), n_fin

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_target(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
