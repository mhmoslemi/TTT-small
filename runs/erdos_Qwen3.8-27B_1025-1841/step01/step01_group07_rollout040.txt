import numpy as np
from scipy.optimize import minimize

def evaluate_c5(h, n):
    """Compute C5 for given h values."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    # Adjust to make sum = n/2
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            # Need to increase sum
            # Add to elements that can be increased
            mask = h < 1.0
            if np.sum(mask) == 0:
                return h
            add_per = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + add_per, 1.0)
            # If still not at target, iterate
            for _ in range(10):
                diff = target - np.sum(h)
                if abs(diff) < 1e-10:
                    break
                if diff > 0:
                    mask = h < 1.0
                    if np.sum(mask) == 0:
                        break
                    add_per = diff / np.sum(mask)
                    h[mask] = np.minimum(h[mask] + add_per, 1.0)
                else:
                    mask = h > 0.0
                    if np.sum(mask) == 0:
                        break
                    sub_per = (-diff) / np.sum(mask)
                    h[mask] = np.maximum(h[mask] - sub_per, 0.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    import time
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Try to use initial_h_values if available
    try:
        parent = np.array(initial_h_values, dtype=float)
        parent_n = len(parent)
    except:
        parent = None
        parent_n = 0
    
    # Define resolutions to try
    resolutions = [64, 128, 256, 512]
    
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.8:
            break
        
        # Create initial guess
        if parent is not None:
            # Interpolate parent to new resolution
            x_parent = np.linspace(0, 2, parent_n)
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_parent, parent)
        else:
            h = np.full(n, 0.5)
        
        h = project_h(h, n)
        
        # Evaluate initial
        c5 = evaluate_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        
        # Simulated annealing
        T = 0.1  # Initial temperature
        cooling = 0.9995
        step_size = 0.1
        
        for iteration in range(200000):
            if time.time() - start_time > budget_s * 0.9:
                break
            
            # Generate a move
            move_type = rng.randint(3)
            h_new = h.copy()
            
            if move_type == 0:
                # Single perturbation
                i = rng.randint(n)
                delta = rng.uniform(-step_size, step_size)
                h_new[i] = h[i] + delta
                # Find j to compensate
                j = rng.randint(n)
                while j == i:
                    j = rng.randint(n)
                h_new[j] = h[j] - delta
            elif move_type == 1:
                # Swap mass between two positions
                i = rng.randint(n)
                j = rng.randint(n)
                while j == i:
                    j = rng.randint(n)
                amt = rng.uniform(-step_size, step_size)
                h_new[i] = h[i] + amt
                h_new[j] = h[j] - amt
            else:

                i = rng.randint(n - 1)
                length = rng.randint(1, 4)
                delta = rng.uniform(-step_size, step_size)
                h_new[i:i+length] += delta
                # Compensate elsewhere
                j = rng.randint(n - length)

                h_new[j:j+length] -= delta

            c5_new = evaluate_c5(h_new, n)
            
            # Accept/reject
            if c5_new < c5 or rng.uniform() < np.exp(-(c5_new - c5) / max(T, 1e-10)):
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n

def subgradient_descent(h, n, num_iter, step_size):
    for it in range(num_iter):
        corr = np.correlate(h, 1-h, mode="full")
        k_max = np.argmax(corr)
        # Subgradient of corr[k_max] w.r.t. h
        # corr[k_max] = sum_i h[i]*(1-h[i+lag]) where lag = k_max - (n-1)
        lag = k_max - (n-1)
        # d(corr)/d(h[j]) = (1-h[j+lag]) for j where j+lag is valid
        #                          + h[j-lag]*(... ) for j where j-lag is valid
        # Actually: corr[k] = sum_i h[i] - sum_i h[i]*h[i+lag]
        # d(corr)/d(h[j]) = 1 - h[j+lag] (if j+lag valid) - h[j-lag] (if j-lag valid)
        
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j + lag < n:
                grad[j] += 1 - h[j+lag]
            if 0 <= j - lag < n:
                grad[j] -= h[j-lag]
        
        # Subgradient step
        h_new = h - step_size * grad
        h_new = project_h(h_new, n)
        
        # Check improvement
        c5_old = np.max(corr) * 2/n
        c5_new = evaluate_c5(h_new, n)
        if c5_new < c5_old:
            h = h_new
        # else: reduce step size
    
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0

            if cnt == 0:

            h = np.minimum(h, 1.0)

            if cnt == 0:

                i = rng.randint(n - block_size)
                j = rng.randint(n - block_size)
                if abs(i - j) < block_size:
                    continue
                delta = rng.uniform(-step / 2, step / 2)
                h_new[i:i+block_size] += delta
                h_new[j:j+block_size] -= delta
            else:
                # Four-point exchange
                idx = rng.choice(n, size=4, replace=False)
                deltas = rng.uniform(-step/2, step/2, size=4)
                deltas -= deltas.mean()  # preserve sum
                h_new[idx] += deltas
            
            h_new = project_h(h_new, n)
            c5_new = evaluate_c5(h_new)
            evals += 1
            
            if c5_new < c5 or (T > 1e-6 and rng.random() < np.exp(-(c5_new - c5) / T)):
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        T *= cooling
        step = max(0.01, step * 0.99)
    
    return best_h, best_c5

def coordinate_descent(h, n, max_time, rng):
    """Coordinate descent refinement."""
    h = h.copy().astype(np.float64)
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()
    start = time.time()
    
    step_sizes = [0.05, 0.02, 0.01, 0.005]
    
    while time.time() - start < max_time:
        improved = False
        for i in range(n):
            if time.time() - start > max_time:
                break
            for ss in step_sizes:
                for sign in [1, -1]:
                    delta = sign * ss
                    h_new = h.copy()
                    h_new[i] += delta
                    # Compensate at a random other position
                    j = rng.randint(n)
                    while j == i:
                        j = rng.randint(n)
                    h_new[j] -= delta
                    h_new = project_h(h_new, n)
                    c5_new = evaluate_c5(h_new)

                        h = h_new
                        c5 = c5_new
                        improved = True
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                        break
                if improved:
                    break
    return best_h, best_c5

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def project_h(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h[mask] = np.minimum(h[mask], 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h[mask] = np.maximum(h[mask], 0.0)
    return h

def make_sin2(n):
    x = np.linspace(0.0, 2.0, n)
    return project_h(np.sin(np.pi * x / 2.0) ** 2, n)

def make_tri(n):
    x = np.linspace(0.0, 2.0, n)
    return project_h(1.0 - np.abs(x - 1.0), n)

def make_cos_bump(n):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 * (1.0 + np.cos(np.pi * (x - 1.0)))
    return project_h(h, n)

def interp_to(h_src, n_dst):
    x_s = np.linspace(0.0, 2.0, len(h_src))
    x_d = np.linspace(0.0, 2.0, n_dst)
    return project_h(np.interp(x_d, x_s, h_src), n_dst)

def sa_optimize(h, n, max_time, rng):
    h = h.astype(np.float64).copy()
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()
    T = 0.04
    cooling = 0.99985
    step = 0.18
    start = time.time()

    while time.time() - start < max_time and T > 1e-7:
        h_new = h.copy()
        move = rng.randint(4)
        if move == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            while j == i:
                j = rng.randint(n)
            d = rng.uniform(-step, step)
            h_new[i] += d
            h_new[j] -= d
        elif move == 1:
            i = rng.randint(n)
            j = rng.randint(n)
            while j == i:
                j = rng.randint(n)
            k = rng.randint(n)
            while k == i or k == j:
                k = rng.randint(n)
            d1 = rng.uniform(-step, step)
            d2 = rng.uniform(-step, step)
            d3 = -(d1 + d2)
            h_new[i] += d1
            h_new[j] += d2
            h_new[k] += d3
        elif move == 2:
            bs = rng.randint(1, max(2, n // 12))
            i = rng.randint(max(1, n - bs))
            j = rng.randint(max(1, n - bs))
            if abs(i - j) < bs:
                T *= cooling
                continue
            d = rng.uniform(-step * 0.5, step * 0.5)
            h_new[i:i+bs] += d

        else:

        n_cand = len(candidates)
        t_per = alloc / n_cand

        for ci in range(n_cand):
            if time.time() - t0 > budget_s - 15:
                break
            h0 = candidates[ci].copy()
            sa_t = t_per * 0.75
            cd_t = t_per * 0.25
            h_sa, c5_sa = sa_optimize(h0, n, sa_t, rng)
            h_cd, c5_cd = local_refine(h_sa, n, cd_t, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

    if best_h is None:
        best_n = 64
        best_h = make_sin2(best_n)
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_n
