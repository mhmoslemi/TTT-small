        w = W[s]
        if w < 1e-12:
            continue
        m = n - s
        grad[s:] += w * (1.0 - h[:m])
        grad[:m] -= w * h[s:s+m]
    
    return grad, c_max

def subgradient_optimize(h_init, n, dx, max_iter=5000, time_limit=None, t_start=None):
    """Subgradient with momentum and adaptive step size."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    n_active = n
    alpha = 4.0 / np.sqrt(n)  # initial step size
    momentum = 0.9
    v = np.zeros(n)
    
    no_improve = 0
    stall_count = 0
    
    for it in range(max_iter):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        
        grad, c_max = compute_weighted_grad(h, n)
        
        # Momentum update
        v = momentum * v - alpha * grad
        h_new = h + v
        h_new = project_target(h_new, n)
        
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            no_improve = 0
        else:
            h = h_new  # keep going even without best improvement (momentum might help)
            no_improve += 1
        
        if no_improve > 100:
            alpha *= 0.7
            momentum = min(momentum * 0.99, 0.5)
            no_improve = 0
            v = np.zeros(n)  # reset momentum
        
        # Periodic annealing
        if it % 2000 == 1999:
            alpha *= 0.5
    
    return best_h, best_c5

def binary_round(h, n):
    """Round h to binary while maintaining sum constraint."""
    target = n / 2
    # Sort by value descending, take top n/2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:int(target)]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, dx, max_sweeps=50, time_limit=None, t_start=None):
    """Local search on binary solutions: try all pairwise swaps."""
    h = h_bin.copy()
    best_c5 = compute_c5(h, dx)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    improved = True
    sweep = 0
    while improved and sweep < max_sweeps:
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        improved = False
        sweep += 1
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        for a in ones:
            if time_limit and t_start and time.time() - t_start > time_limit:
                break
            for b in zeros:
                # Try swap: h[a] 1→0, h[b] 0→1
                h[a] = 0.0
                h[b] = 1.0
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    improved = True
                else:
                    h[a] = 1.0
                    h[b] = 0.0
    
    return h, best_c5
