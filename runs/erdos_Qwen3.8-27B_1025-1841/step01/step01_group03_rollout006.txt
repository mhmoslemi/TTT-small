def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection on additive shift."""
    lo = target - h.sum() - 1.0
    hi = target - h.sum() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) / 2.0
    return np.clip(h + t, 0.0, 1.0)

    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    while len(ones_idx) > target:
        i = ones_idx[rng.integers(len(ones_idx))]
        h[i] = 0
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
    while len(ones_idx) < target:
        j = zeros_idx[rng.integers(len(zeros_idx))]
        h[j] = 1
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
    
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    
    for it in range(n_iterations):
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        # Swap
        h[i], h[j] = h[j], h[i]
        new_c5 = evaluate_c5(h)
        if new_c5 <= current_c5 or rng.random() < np.exp((current_c5 - new_c5) / temperature):
            current_c5 = new_c5
            if new_c5 < best_c5:

        else:
            h[i], h[j] = h[j], h[i]  # revert
        temperature *= 0.9999
        if temperature < 0.001:
            temperature = 0.001
    
    return h, best_c5

            h[i] = 0.0

            h[i] = 1.0

            b = c0

            # = minimize max_k (a[k]*x + b[k]) over x in [0,1]
            # Ternary search (function is convex in x)
            lo, hi = 0.0, 1.0
            for _ in range(50):

                m2 = hi - (hi - lo) / 3.0

                f2 = np.max(a * m2 + b)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            h[i] = (lo + hi) / 2.0
        
        # Project to maintain sum constraint
        h = project_sum(h, target)

        h_cd3 = coord_descent(best_h, n_sweeps=500, rng=rng)
        c5 = evaluate_c5(h_cd3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd3.copy()
    
    # Phase 5: Try doubled resolution
    if time_left() > 120:
        n2 = n * 2
        h_up = np.repeat(best_h, 2)
        h_up = project_sum(h_up, n2 / 2.0)
        h_cd_up = coord_descent(h_up, n_sweeps=100, rng=rng)
        c5_up = evaluate_c5(h_cd_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_cd_up.copy()
            # Polish at higher res
            if time_left() > 60:
                h_cd_up2 = coord_descent(best_h, n_sweeps=200, rng=rng)
                c5_up2 = evaluate_c5(h_cd_up2)
                if c5_up2 < best_c5:
                    best_c5 = c5_up2
                    best_h = h_cd_up2.copy()
    
    # Phase 6: Final CD polish
    if time_left() > 5:
        h_final = coord_descent(best_h, n_sweeps=200, rng=rng)
        c5 = evaluate_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
    
    # Ensure constraints
    nf = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, nf / 2.0)
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, nf
