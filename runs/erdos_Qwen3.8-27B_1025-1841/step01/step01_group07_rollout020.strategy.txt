def scale_parent(parent, n_new):
    # Piecewise linear scaling from old samples to new grid
    n_old = len(parent)
    total_interval = 2.0
    dx_old = total_interval / n_old
    dx_new = total_interval / n_new
    x_old = np.linspace(0, total_interval, n_old)
    x_new = np.linspace(0, total_interval, n_new)
    h_samp = np.interp(x_new, x_old, parent)
    h_clip = np.clip(h_samp, 0, 1)
    sum_total = np.sum(h_clip)
    # Rescale to satisfy integral constraint
    scale_factor = (n_new/2) / sum_total if sum_total > 0 else 1.0
    h_scaled = h_clip * scale_factor
    # Final scaling check
    final_sum = np.sum(h_scaled) / n_new
    if abs(final_sum - 0.5) > 1e-3:
        # Fallback to uniform clipping distribution if rescaling fails
        h_uniform = np.ones_like(h_scaled) * (n_new/2 / n_new)
        logger.info("Rescaling failed, using uniform")
        return h_uniform
    return h_scaled

## 3.3 Repair function (constraint restoration)

```python
def repair(x, n_points=100):
    h = x["h"].copy()
    def clip_to_range(vec):
        return np.clip(vec, 0, 1).astype(float)
    def enforce_integral(vec, target=n_points/2):
        if np.abs(np.sum(vec) - target) < 1e-5:
            return vec
        excess = (target - np.sum(vec)) / len(vec)
        return [val + excess for val in vec]
    # Box constraints
    h_box = clip_to_range(h)
    # Integral sum
    h_fixed = enforce_integral(h_box)
    return h_fixed
```
