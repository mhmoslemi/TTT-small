    beta = 0.9

    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += 1.0 - h[s:]
        g[s:] += -h[:n-s]
    else:
        sp = -s
        g[sp:] += 1.0 - h[:n-sp]
        g[:n-sp] += -h[sp:]
    return g

def proj_feasible(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        m = (lo + hi) * 0.5
        p = np.clip(h - m, 0, 1)
        if p.sum() > target:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mx = np.max(corr)
    shifted = np.clip(beta * (corr - mx), -500, 0)
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-10:
            continue
        s = m - (n - 1)
        grad += w[m] * grad_shift(h, s)
    grad *= dx
    return grad

def optimize_pg(h0, tlimit, beta=60.0):
    n = len(h0)
    target = n / 2.0
    h = proj_feasible(h0.copy(), target)
    best_h = h.copy()
    best_v = compute_c5(h)
    t0 = time.time()
    y = h.copy()
    beta_t = 1.0
    step_base = 1.0

    for it in range(100000):
        if time.time() - t0 > tlimit:
            break
        bt = (1.0 + np.sqrt(1.0 + 4.0 * beta_t * beta_t)) * 0.5
        alpha = (beta_t - 1.0) / bt
        y = h + alpha * (h - y)
        y = np.clip(y, 0, 1)

        grad = smooth_grad(y, beta)
        step = step_base / np.sqrt(it + 1.0)

        h_new = proj_feasible(y - step * grad, target)
        v = compute_c5(h_new)
        if v < best_v:
            best_v = v
            best_h = h_new.copy()
        if v > compute_c5(y) + 1e-6:
            step_base *= 0.7
        else:
            step_base *= 1.01

        h = h_new
        beta_t = bt

    return best_h, best_v

def coord_refine(h0, tlimit):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_v = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(777)

    while time.time() - t0 < tlimit:
        improved = False
        for i in range(n):
            if time.time() - t0 > tlimit:
                break
            for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                h_try = h.copy()
                h_try[i] += delta
                h_try = proj_feasible(h_try, target)
                v = compute_c5(h_try)
                if v < best_v - 1e-10:
                    best_v = v
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
        if not improved:
            # Try random perturbations
            for _ in range(50):
                i = rng.randint(n)
                h_try = h.copy()
                h_try[i] += rng.uniform(-0.2, 0.2)
                h_try = proj_feasible(h_try, target)
                v = compute_c5(h_try)
                if v < best_v - 1e-10:
                    best_v = v
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
            if not improved:
                break
    return best_h, best_v

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return np.interp(x_new, x_old, h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (bin(i).count('1') % 2 == 0) else 0.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    n0 = 45
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        n0 = len(h_parent)
    else:
        h_parent = np.ones(n0) * 0.5

    h_parent = proj_feasible(h_parent, n0 / 2.0)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    candidates = [(h_parent, n0, "parent")]

    for n_tm in [64, 128, 256]:
        candidates.append((thue_morse(n_tm), n_tm, "tm"))

    for factor in [2, 3, 4]:
        n_up = n0 * factor
        h_up = proj_feasible(upsample(h_parent, n_up), n_up / 2.0)
        candidates.append((h_up, n_up, "up"))

    for n_rb in [80, 160]:
        for bs in [2, 3, 4, 5, 7]:
            h_rb = np.zeros(n_rb)
            n_blocks = n_rb // bs
            for b in range(n_blocks):
                if rng.random() < 0.5:
                    h_rb[b*bs:(b+1)*bs] = 1.0
            h_rb = proj_feasible(h_rb, n_rb / 2.0)
            candidates.append((h_rb, n_rb, "rb"))

    n_cand = len(candidates)
    time_per = min(90.0, (budget_s - 180.0) / max(n_cand, 1))

    for idx, (h_c, n_c, tag) in enumerate(candidates):
        if time.time() - t_start > budget_s - 90:
            break
        rem = min(time_per, budget_s - 90 - (time.time() - t_start))
        if rem < 8:
            break
        h_opt, c5_opt = optimize_pg(h_c, rem, beta=60.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Coordinate refinement on best
    if time.time() - t_start < budget_s - 40:
        rem = budget_s - 40 - (time.time() - t_start)
        h_opt, c5_opt = coord_refine(best_h, min(rem * 0.4, 120))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Second pass: further gradient optimization on best
    if time.time() - t_start < budget_s - 30:
        rem = budget_s - 30 - (time.time() - t_start)
        h_opt, c5_opt = optimize_pg(best_h, min(rem * 0.5, 150), beta=100.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Try upscaling best and re-optimizing
    if time.time() - t_start < budget_s - 30:
        rem = budget_s - 30 - (time.time() - t_start)
        n_best = len(best_h)
        for n_try in [n_best * 2, n_best * 3]:
            if time.time() - t_start > budget_s - 20:
                break
            rem2 = budget_s - 20 - (time.time() - t_start)
            if rem2 < 15:
                break
            h_up = proj_feasible(upsample(best_h, n_try), n_try / 2.0)
            h_opt, c5_opt = optimize_pg(h_up, min(rem2 * 0.6, 120), beta=50.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Final coordinate refinement
    if time.time() - t_start < budget_s - 10:
        rem = budget_s - 10 - (time.time() - t_start)
        h_opt, c5_opt = coord_refine(best_h, rem)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    return best_h, best_c5, len(best_h)
