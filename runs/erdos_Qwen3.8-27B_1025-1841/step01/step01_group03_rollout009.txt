    # Let shift = k - (n-1), range: -(n-1), ..., (n-1)
    # Δc[shift] = -(1-h[i+shift]) + h[i-shift] + (1-h[j+shift]) - h[j-shift]
    # where h[x] = 0 if x < 0 or x >= n
    
    max_val = 0
    for k in range(2*n-1):
        shift = k - (n-1)
        delta = 0
        # -(1-h[i+shift])
        ia = i + shift
        if 0 <= ia < n:
            delta -= (1 - h[ia])
        # +h[i-shift]
        ib = i - shift
        if 0 <= ib < n:
            delta += h[ib]
        # +(1-h[j+shift])
        ja = j + shift
        if 0 <= ja < n:
            delta += (1 - h[ja])
        # -h[j-shift]
        jb = j - shift
        if 0 <= jb < n:
            delta -= h[jb]
        
        new_val = corr[k] + delta
        if new_val > max_val:
            max_val = new_val
    
    return max_val

def compute_c5(h, dx):
    corr = np.correlate(h.astype(np.float64), (1-h).astype(np.float64), mode='full')
    return np.max(corr) * dx

def get_corr(h, n):
    return np.correlate(h.astype(np.float64), (1-h).astype(np.float64), mode='full')

def delta_max_after_swap(h, i, j, n, corr):
    """Compute max of new correlation after swapping h[i](1→0) with h[j](0→1)."""
    max_val = 0.0
    for shift in range(-(n-1), n):
        d = 0.0
        a = i + shift
        if 0 <= a < n:
            d -= (1 - h[a])
        b = i - shift
        if 0 <= b < n:
            d += h[b]
        c = j + shift
        if 0 <= c < n:
            d += (1 - h[c])
        e = j - shift
        if 0 <= e < n:
            d -= h[e]
        val = corr[shift + n - 1] + d
        if val > max_val:
            max_val = val
    return max_val

def local_search(h, dx, max_passes=500, time_limit=None, t0=None):
    n = len(h)
    h = h.copy().astype(np.float64)
    corr = get_corr(h, n)
    current_max = corr.max()
    current_c5 = current_max * dx
    
    if time_limit is None:
        time_limit = 30
    if t0 is None:
        t0 = time.time()
    
    for p in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for i in ones:
            for j in zeros:
                new_max = delta_max_after_swap(h, i, j, n, corr)
                if new_max < current_max:
                    # Accept swap
                    h[i], h[j] = h[j], h[i]
                    corr = get_corr(h, n)
                    current_max = corr.max()
                    current_c5 = current_max * dx
                    improved = True
                    ones = list(np.where(h == 1)[0])
                    zeros = list(np.where(h == 0)[0])
                    break
            if improved:
                break

            break

    n = len(h)
    h = h.copy().astype(np.float64)
    corr = get_corr(h, n)
    current_max = corr.max()
    current_c5 = current_max * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    if time_limit is None:
        time_limit = 60
    if t0 is None:
        t0 = time.time()
    
    rng = np.random.RandomState(42)
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        T = start_T * (end_T / start_T) ** (it / n_iter)

        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        new_max = delta_max_after_swap(h, i, j, n, corr)
        delta = new_max - current_max

            corr = get_corr(h, n)
