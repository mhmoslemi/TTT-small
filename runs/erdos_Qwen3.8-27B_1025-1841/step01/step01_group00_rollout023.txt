import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - sh])
            grad[idx] -= pk * h[idx + sh]
    grad *= dx
    return obj, grad

def optimize_smooth(h, n, tau, max_iters, time_limit, initial_lr=None):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj = smooth_obj_grad(h, tau)[0]
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr is not None else 0.3
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            stall = 0
        else:
            stall += 1
        
        # Check actual C5 every 50 iters
        if it % 50 == 0 or stall > 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > 100:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        
        h = h_new
    
    return best_h, best_c5

def optimize_subgrad_multi(h, n, max_iters, time_limit, initial_lr=0.01, top_k_frac=0.05):
