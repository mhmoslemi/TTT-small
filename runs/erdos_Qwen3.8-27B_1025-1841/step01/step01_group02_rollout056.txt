def soft_max_grad_vec(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    w = np.exp(T * shifted)
    w /= w.sum()
    
    # grad[j] = sum_k w[k] * ((1-h[j+k-n+1]) - h[j-k+n+1]) [bounds checked]
    # Vectorize: for each offset m, compute contribution to all j
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        k = m  # the correlation index
        # Contribution to grad[j] from k=m:
        # (1-h[j+m-n+1]) if 0<=j+m-n+1<n → j in [n-1-m, n-1-m+n] = [n-1-m, 2n-1-m]
        # minus h[j-m+n-1] if 0<=j-m+n-1<n → j in [m-n+1, m+n-1]
        
        # First term: j from max(0, n-1-m) to min(n-1, 2n-1-m)
        j_start1 = max(0, n - 1 - m)
        j_end1 = min(n, 2*n - 1 - m)  # exclusive

            h_idx = idx + m - n + 1
            grad[idx] += w[m] * (1.0 - h[h_idx])
        
        # Second term: j from max(0, m-n+1) to min(n-1, m+n-1)
        j_start2 = max(0, m - n + 1)
        j_end2 = min(n, m + n - 1)  # exclusive... wait
        # j-m+n-1 in [0,n-1] → j in [m-n+1, m+n-1]
        j_start2 = max(0, m - n + 1)

The true objective is C₅(h) = max_k f_k(h) where f_k(h) = corr(k) * dx.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _sm_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    val = (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    s = np.sum(h) - n / 2.0
    return val + lam * s * s

def _sm_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    w = np.exp(T * shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        j_lo1 = max(0, n - 1 - m)
        j_hi1 = min(n, 2 * n - 1 - m)
        if j_lo1 < j_hi1:
            idx = np.arange(j_lo1, j_hi1)
            h_idx = idx + m - n + 1
            grad[idx] += w[m] * (1.0 - h[h_idx])
        j_lo2 = max(0, m - n + 1)
        j_hi2 = min(n, m + n)
        if j_lo2 < j_hi2:
            idx = np.arange(j_lo2, j_hi2)
            h_idx = idx - m + n - 1
            grad[idx] -= w[m] * h[h_idx]
    s = np.sum(h) - n / 2.0
    grad += 2.0 * lam * s
    return grad * dx

def _topm_obj(h, m, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_idx = np.argsort(corr)[::-1]
    top = corr[sorted_idx[:m]]
    mc = top[0]
    soft_top = mc + np.log(np.sum(np.exp(T * (top - mc)))) / T
    rest_max = corr[sorted_idx[m:]].max() if m < len(corr) else 0.0
    return (0.7 * soft_top + 0.3 * rest_max) * dx

def _cd_polish(h, n, max_rounds, time_limit):
    dx = 2.0 / n
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        for j in range(n):

            old = h[j]

            best_local_v = old
            for dv in [0.0, 1.0, old + 0.15, old - 0.15, old + 0.07, old - 0.07,

                v = max(0.0, min(1.0, dv))
                if abs(v - old) < 1e-12:

                trial = h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_local_c - 1e-12:
                    best_local_c = c
                    best_local_v = v
                    improved = True
            if best_local_v != old:
                h[j] = best_local_v
                h = make_feasible(h, n)
                best_c = best_local_c
        if not improved:
            break
    return h, best_c
