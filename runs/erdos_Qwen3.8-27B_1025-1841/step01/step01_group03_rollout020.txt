def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n / 2.0
    if abs(s - target) < 1e-10:
        return h
    # Adjust to meet the constraint
    diff = target - s
    if diff > 0:
        # Need to increase some values
        mask = h < 1.0
        n_avail = np.sum(mask)
        if n_avail == 0:
            return h
        adj = diff / n_avail
        h[mask] = np.minimum(h[mask] + adj, 1.0)
    else:
        # Need to decrease some values
        mask = h > 0.0
        n_avail = np.sum(mask)
        if n_avail == 0:
            return h
        adj = diff / n_avail
        h[mask] = np.maximum(h[mask] + adj, 0.0)
    # If still not exactly right, do a second pass
    s2 = np.sum(h)
    diff2 = target - s2
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        n_avail = np.sum(mask)

            adj = diff2 / n_avail
