```python
def binary_improve(h, deadline):
    n = len(h)
    h = np.array(h, dtype=float)
    # Round to binary
    h = (h > 0.5).astype(float)
    # Fix to have exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > n//2:
        ones_idx = np.where(h==1)[0]
        # Remove random excess ones
        excess = np.random.choice(ones_idx, n_ones - n//2, replace=False)
        h[excess] = 0
    elif n_ones < n//2:
        zeros_idx = np.where(h==0)[0]
        deficit = np.random.choice(zeros_idx, n//2 - n_ones, replace=False)
        h[deficit] = 1
    
    def get_corr(h):
        return np.correlate(h, 1-h, mode='full')
    
    corr = get_corr(h)
    max_val = np.max(corr)
    
    while time.time() < deadline:
        improved = False
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        
        # Try a subset of random swaps
        n_tries = min(len(ones)*len(zeros), 500)
        for _ in range(n_tries):
            a = ones[np.random.randint(len(ones))]
            b = zeros[np.random.randint(len(zeros))]
            if a == b:
                continue
            h[a], h[b] = h[b], h[a]
            new_corr = get_corr(h)
            new_max = np.max(new_corr)
            if new_max < max_val:
                max_val = new_max
                corr = new_corr
                improved = True
                ones = np.where(h==1)[0]
                zeros = np.where(h==0)[0]
            else:
                h[a], h[b] = h[b], h[a]  # revert
        
        if not improved:
            # Try perturbation: random swap anyway
            if len(ones) > 0 and len(zeros) > 0:
                a = ones[np.random.randint(len(ones))]
                b = zeros[np.random.randint(len(zeros))]
                h[a], h[b] = h[b], h[a]
    
    return h
```

                    g[j] += w * (1 - 2 * x[j])
                else:
                    g[j] += w * (1 - x[j + ts])
            if ts != 0 and 0 <= j - ts < n:
                g[j] += w * (-x[j - ts])
    return g

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def slsqp_opt(h0, n, T, maxiter=300, deadline=None):
    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T)
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-12, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def round_to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(np.float64)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    while n_ones > target:
        ones = np.where(h_bin == 1)[0]
        h_bin[ones[rng.randint(len(ones))]] = 0
        n_ones -= 1
    while n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        h_bin[zeros[rng.randint(len(zeros))]] = 1
        n_ones += 1
    return h_bin

def binary_local_search(h, deadline, rng, max_iterations=20000):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    stuck = 0
    sa_temp = 2.0

    for it in range(max_iterations):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = 150

        for _ in range(n_tries):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a], h[b] = h[b], h[a]
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_max = np.max(new_corr)
            if new_max < max_val:
                max_val = new_max
                corr = new_corr
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stuck = 0
            else:
                delta = new_max - max_val
                accept = delta < 0.5 and rng.rand() < np.exp(-delta / max(sa_temp, 0.5))
                if not accept:
                    h[a], h[b] = h[b], h[a]

        # Targeted swaps at worst lags
        if not improved:
            worst_indices = np.argsort(corr)[-5:]
            for wi in worst_indices:
                if time.time() > deadline:
                    break
                t_shift = int(wi) - (n - 1)
                for a in ones[rng.randint(len(ones), size=min(20, len(ones)))]:
                    b_fwd = a + t_shift
                    if 0 <= b_fwd < n and h[b_fwd] == 0:
                        h[a], h[b_fwd] = h[b_fwd], h[a]
                        new_corr = np.correlate(h, 1 - h, mode='full')
                        new_max = np.max(new_corr)
                        if new_max < max_val:
                            max_val = new_max
                            corr = new_corr
                            improved = True
                            ones = np.where(h == 1)[0]
                            zeros = np.where(h == 0)[0]
                            stuck = 0
                        else:
                            h[a], h[b_fwd] = h[b_fwd], h[a]
                    b_bwd = a - t_shift
                    if t_shift != 0 and 0 <= b_bwd < n and h[b_bwd] == 0:
                        h[a], h[b_bwd] = h[b_bwd], h[a]
                        new_corr = np.correlate(h, 1 - h, mode='full')
                        new_max = np.max(new_corr)
                        if new_max < max_val:
                            max_val = new_max
                            corr = new_corr
                            improved = True
                            ones = np.where(h == 1)[0]
                            zeros = np.where(h == 0)[0]
                            stuck = 0
                        else:
                            h[a], h[b_bwd] = h[b_bwd], h[a]

        if not improved:
            stuck += 1
            sa_temp = max(0.5, sa_temp * 0.95)
            if stuck > 100:
                for _ in range(3):
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    h[a], h[b] = h[b], h[a]
                corr = np.correlate(h, 1 - h, mode='full')
                max_val = np.max(corr)
                stuck = 0
                sa_temp = 2.0

    return h, float(max_val * 2.0 / n)

def projected_gradient(h0, n, deadline, n_steps=5000):
    h = h0.copy()
    lr = 0.05
    momentum = 0.9
    v = np.zeros(n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = int(np.argmax(corr))
        t_shift = max_idx - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            if 0 <= j + t_shift < n:
                if t_shift == 0:
                    g[j] = (1 - 2 * h[j])
                else:
                    g[j] = (1 - h[j + t_shift])
            else:
                g[j] = 0.0
            if t_shift != 0 and 0 <= j - t_shift < n:
                g[j] += (-h[j - t_shift])
        v = momentum * v - lr * g
        h = h + v
        h = project(h, n)
        if step % 100 == 99:
            lr = max(0.005, lr * 0.95)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n = len(h_start)

    # Phase 1: Quick SLSQP to refine continuous solution
    for T in [80.0, 40.0]:
        if time.time() > deadline - 500:
            break
        h_new = slsqp_opt(best_h, n, T, maxiter=200, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Binary local search (main differentiator)
    if time.time() < deadline - 200:
        h_bin = round_to_binary(best_h, n, rng)
        c5_bin = evaluate_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        
        h_bl, c5_bl = binary_local_search(h_bin, deadline - 180, rng, max_iterations=15000)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl

    # Phase 3: More SLSQP from best solution with smaller T
    if time.time() < deadline - 100:
        for T in [20.0, 10.0]:
            if time.time() > deadline - 80:
                break
            h_new = slsqp_opt(best_h, n, T, maxiter=300, deadline=deadline)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Projected gradient refinement
    if time.time() < deadline - 60:
        h_pg = projected_gradient(best_h, n, deadline - 50, n_steps=3000)
        c5_pg = evaluate_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Phase 5: Try binary local search again from new best
    if time.time() < deadline - 30:
        h_bin2 = round_to_binary(best_h, n, rng)
        c5_b2 = evaluate_c5(h_bin2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_bin2
        h_bl2, c5_bl2 = binary_local_search(h_bin2, deadline - 10, rng, max_iterations=5000)
        if c5_bl2 < best_c5:
            best_c5 = c5_bl2
            best_h = h_bl2

    # Phase 6: Random restarts at different resolutions
    for attempt in range(10):
        if time.time() > deadline - 10:
            break
        n_try = [280, 336, 420, 560][attempt % 4]
        if n_try == n:
            h_pert = best_h + np.random.randn(n) * 0.05
        else:
            h_pert = interp_to(best_h, n, n_try)
        h_pert = project(h_pert, n_try)
        h_new = slsqp_opt(h_pert, n_try, 30.0, maxiter=150, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final binary polish
    if time.time() < deadline - 5:
        nf = len(best_h)
        h_final_bin = round_to_binary(best_h, nf, rng)
        h_final_bin, c5_fb = binary_local_search(h_final_bin, deadline, rng, max_iterations=3000)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_final_bin

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
