def gradient_softmax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    weights = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-10:
            continue
        # c[k] = sum_i h[i + k - (n-1)] * (1-h[i])
        # ∂c[k]/∂h[j] = (1-h[j-k+n-1])*1{0≤j-k+n-1<n} - h[j+k-n+1]*1{0≤j+k-n+1<n}
        i1 = np.arange(n) - k + (n - 1)
        i2 = np.arange(n) + k - (n - 1)
        m1 = (i1 >= 0) & (i1 < n)
        m2 = (i2 >= 0) & (i2 < n)
        safe_i1 = np.clip(i1, 0, n-1)
        safe_i2 = np.clip(i2, 0, n-1)
        grad += w * ((1.0 - h[safe_i1]) * m1 - h[safe_i2] * m2)
    
    return grad * dx

def sa_binary(n, rng, max_iter=200000, time_limit=60):
    """Simulated annealing for binary sequence."""
    target = n // 2
    h = np.zeros(n)
    ones = rng.choice(n, target, replace=False)
    h[ones] = 1.0
    dx = 2.0 / n
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, 'full')) * dx
    
    T = 0.05
    cooling = 0.99995
    start = time.time()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = np.max(np.correlate(h, 1-h, 'full')) * dx
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cooling
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_to_constraint(h, target_sum):
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def gradient_softmax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    weights = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-10:
            continue
        i1 = np.arange(n) - k + (n - 1)
        i2 = np.arange(n) + k - (n - 1)
        m1 = (i1 >= 0) & (i1 < n)
        m2 = (i2 >= 0) & (i2 < n)
        safe_i1 = np.clip(i1, 0, n - 1)
        safe_i2 = np.clip(i2, 0, n - 1)
        grad += w * ((1.0 - h[safe_i1]) * m1 - h[safe_i2] * m2)
    return grad * dx

def quadratic_residue_seq(p):
    """Generate quadratic residue sequence of length p (prime)."""
    s = np.zeros(p)
    for a in range(1, p):
        for b in range(1, a):
            if (b * b) % p == a:
                s[a] = 1.0
                break
    return s

    """Convert continuous h to binary with exactly n//2 ones."""

        order = np.argsort(h)
        for idx in order:

                break

        order = np.argsort(-h)

                break

    return h_bin

    """Pair-flip local search on binary sequence."""
    n = len(h)
    dx = 2.0 / n
    best_c5 = np.max(np.correlate(h, 1.0 - h, 'full')) * dx
    start = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        if not ones or not zeros:
            break
        
        # Shuffle pairs
        pairs = [(i, j) for i in ones for j in zeros]
        rng.shuffle(pairs)
        
        for i, j in pairs:
            if time.time() - start > time_limit:
                break
            h[i], h[j] = 0.0, 1.0
            c = np.correlate(h, 1.0 - h, 'full')
            new_max = np.max(c) * dx
            if new_max < best_c5 - 1e-12:
                best_c5 = new_max
                improved = True
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
                pairs = [(a, b) for a in ones for b in zeros]
                rng.shuffle(pairs)
            else:
                h[i], h[j] = 1.0, 0.0
        
        if not improved:
            break
    
    return h, best_c5

def sa_binary_from(h_start, rng, max_iter=200000, time_limit=30.0):
    """Simulated annealing starting from h_start."""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1.0 - h, 'full')) * dx
    T = 0.03
    cooling = 0.99997
    start = time.time()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c = np.correlate(h, 1.0 - h, 'full')
        new_c5 = np.max(c) * dx
        delta = new_c5 - best_c5
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cooling
    
    return best_h, best_c5
