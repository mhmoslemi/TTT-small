            h_ls_r, c5_ls_r = binary_local_search(h_bin_r, dx, 200, rng, kick_size=1)
            if c5_ls_r < best_c5:
                best_c5 = c5_ls_r
                best_h = h_ls_r.copy()
                best_n = n_try

    # Final polish: PGD on best solution
    if best_h is not None and time.time() < deadline - 5:
        n_b = best_n
        dx_b = 2.0 / n_b
        tgt_b = float(np.sum(best_h))
        h_pol, c5_pol = pgd_optimize(best_h.copy(), dx_b, tgt_b, 1000, deadline)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_b

    return best_h, best_c5, best_n
