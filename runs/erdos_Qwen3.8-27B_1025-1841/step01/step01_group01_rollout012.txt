def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair(h, n):
    """Ensure sum(h) = n/2 and 0 <= h <= 1"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    # Adjust by scaling
    s = np.sum(h)
    if s > 0:
        h = h * (target / s)
        h = np.clip(h, 0, 1)
    # Fine adjustment
    diff = target - np.sum(h)
    if diff > 0:
        # Add to elements with most room
        room = 1 - h
        idx = np.argsort(-room)
        for i in idx:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff <= 1e-10:
                break
    elif diff < 0:
        idx = np.argsort(h)
        for i in idx:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-10:
                break
    return h

def local_search_binary(h, dx, max_iters=100000, rng=None):
    """Swap-based local search for binary sequences"""
    if rng is None:
        rng = np.random.RandomState()
    n = len(h)
    c5 = compute_c5(h, dx)
    
    # Precompute correlation
    corr = np.correlate(h, 1 - h, mode="full")
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    iters = 0
    while iters < max_iters:
        iters += 1
        # Pick a random 1 and 0 to swap
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))

        pos_j = zeros[j]

        h[pos_j] = 1

            h[pos_j] = 0

def compute_corr(h):
    """Compute full linear cross-correlation of h and (1-h) using FFT"""
    n = len(h)

        N <<= 1
    H = np.fft.rfft(h, N)

def gradient_refine(h, dx, rng, max_iters=5000, lr=0.01):
    """Projected gradient descent allowing fractional values"""
    n = len(h)
    h = h.copy().astype(float)
    c5 = compute_c5(h, dx)
    best_c5 = c5
    best_h = h.copy()

        corr = compute_corr(h)
        k_max = np.argmax(corr)
        m = k_max - (n - 1)  # offset
        
        # Gradient of c[m] w.r.t. h[i]
        # c[m] = sum_i h[i]*(1-h[i+m]) for valid i
        grad = np.zeros(n)
        for i in range(n):

            if 0 <= i + m < n:

            if 0 <= i - m < n:
                g -= h[i - m]
            grad[i] = g

        # Project: clip to [0,1] and fix sum
        h = np.clip(h, 0, 1)
        target = n / 2.0
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            # Distribute adjustment
            h = h + diff / n
            h = np.clip(h, 0, 1)
            # Fine fix
            diff = target - np.sum(h)
            if diff > 0:
                room = 1.0 - h
                for i in np.argsort(-room)[:int(abs(diff))+1]:
                    add = min(diff, room[i])
                    h[i] += add
                    diff -= add
                    if diff <= 0: break

                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
                    if diff >= 0: break

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

import numpy as np
import time

def compute_corr(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    O = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(O), N)
    return corr[:2*n-1]

def compute_c5(h, dx):
    return float(np.max(compute_corr(h))) * dx

def repair_binary(h, rng):
    n = len(h)
    target = n // 2
    s = int(round(np.sum(h)))
    if s > target:
        ones = np.where(h > 0.5)[0].tolist()
        rng.shuffle(ones)
        for idx in ones[:s - target]:
            h[idx] = 0.0
    elif s < target:
        zeros = np.where(h <= 0.5)[0].tolist()
        rng.shuffle(zeros)
        for idx in zeros[:target - s]:
            h[idx] = 1.0
    return h

def fast_anneal(h, dx, rng, iters, T0=0.0004):
    n = len(h)
    c5 = float(np.max(compute_corr(h))) * dx
    best_c5 = c5
    best_h = h.copy()
    ones_l = list(np.where(h == 1)[0])
    zeros_l = list(np.where(h == 0)[0])
    cool = (1.0 / T0 * 1e-8) ** (1.0 / max(iters, 1))
    for it in range(iters):
        T = T0 * (cool ** it)
        ai = rng.randint(len(ones_l))
        bi = rng.randint(len(zeros_l))
        a = ones_l[ai]
        b = zeros_l[bi]
        h[a] = 0.0
        h[b] = 1.0
        ones_l[ai] = b
        zeros_l[bi] = a
        nc5 = float(np.max(compute_corr(h))) * dx
        d = nc5 - c5
        if d <= 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            c5 = nc5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
            ones_l[ai] = a
            zeros_l[bi] = b
    return best_h, best_c5

def hard_improve(h, dx, rng, iters):
    n = len(h)
    c5 = float(np.max(compute_corr(h))) * dx
    ones_l = list(np.where(h == 1)[0])
    zeros_l = list(np.where(h == 0)[0])
    stall = 0
    for _ in range(iters):
        if stall > iters * 0.3:
            break
        ai = rng.randint(len(ones_l))
        bi = rng.randint(len(zeros_l))
        a = ones_l[ai]
        b = zeros_l[bi]
        h[a] = 0.0
        h[b] = 1.0
        nc5 = float(np.max(compute_corr(h))) * dx
        if nc5 < c5:
            c5 = nc5
            ones_l[ai] = b
            zeros_l[bi] = a
            stall = 0
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
    return h, c5

def two_swap_improve(h, dx, rng, iters):
    n = len(h)
    c5 = float(np.max(compute_corr(h))) * dx
    best_c5 = c5
    best_h = h.copy()
    for _ in range(iters):
        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]
        a1, a2 = rng.choice(ones_arr, 2, replace=False)
        b1, b2 = rng.choice(zeros_arr, 2, replace=False)
        pts = [a1, a2, b1, b2]
        old_vals = h[pts].copy()
        h[a1] = 0.0
        h[a2] = 0.0
        h[b1] = 1.0
        h[b2] = 1.0
        nc5 = float(np.max(compute_corr(h))) * dx
        if nc5 < c5:
            c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[pts] = old_vals
    return best_h, best_c5

def grad_refine(h, dx, max_iters=3000, lr0=0.05):
    n = len(h)
    h = h.copy().astype(float)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr0
    for it in range(max_iters):
        corr = compute_corr(h)
        k_max = int(np.argmax(corr))
        m = k_max - (n - 1)
        grad = np.zeros(n)
        if m >= 0:
            if m > 0:
                grad[:n-m] += (1.0 - h[m:])
            grad += (1.0 - h)
            if m > 0:
                grad[m:] -= h[:-m]
            else:
                grad -= h
        else:
            p = -m
            if p > 0:
                grad[p:] += (1.0 - h[:n-p])
            grad += (1.0 - h)
            if p > 0:
                grad[:n-p] -= h[p:]
            else:
                grad -= h
        grad *= dx
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if diff > 0:
                idx = np.argsort(1.0 - h)
                for i in idx:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                    if diff <= 0:
                        break
            else:
                idx = np.argsort(h)
                for i in idx:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
                    if diff >= 0:
                        break
        nc5 = compute_c5(h, dx)
        if nc5 < best_c5:
            best_c5 = nc5
            best_h = h.copy()
        lr *= 0.999
    return best_h, best_c5
