    offset = max_idx - (n - 1)
    grad = np.zeros(n)
    j_min = max(0, -offset)
    j_max = min(n, n - offset)

        i1 = j_range + offset
        i2 = j_range
        grad[i1] += (1 - h[j_range])
        grad[i2] -= h[i1]

import numpy as np

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(corr, n):
    return float(np.max(corr) * 2.0 / n)

def _subgrad(h, n, corr, frac=0.92):
    """Subgradient of max(corr) w.r.t. h, averaged over active components."""
    dx = 2.0 / n
    mv = float(np.max(corr))
    if mv <= 0:
        return np.zeros(n)
    active = np.where(corr >= frac * mv)[0]
    grad = np.zeros(n)
    for k in active:
        off = int(k) - (n - 1)
        jlo = max(0, -off)
        jhi = min(n, n - off)
        if jlo < jhi:
            jr = np.arange(jlo, jhi)
            i1 = jr + off
            grad[i1] += 1.0 - h[jr]
            grad[jr] -= h[i1]
    return grad * dx / len(active)

def _proj(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
        d2 = t - h.sum()
        if abs(d2) > 1e-8:
            m = (h > 1e-7) & (h < 1.0 - 1e-7)
            if m.any():
                h[m] += d2 / m.sum()
                np.clip(h[m], 0.0, 1.0, out=h[m])
    return h

def _interp(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    return np.interp(np.linspace(0, n_old - 1, n_new), np.arange(n_old), h_old)
