    cands = [n_parent, 32, 40, 44, 48, 50, 56, 64, 72, 80, 96, 112, 128, 160]
    resolutions = sorted(set(c for c in cands if 16 <= c <= 256))

    # Allocate time: more to parent and nearby resolutions
    total_budget = budget_s - 20
    weights = []
    for r in resolutions:
        if r == n_parent:
            weights.append(4.0)
        elif abs(r - n_parent) <= 8:
            weights.append(2.5)
        elif abs(r - n_parent) <= 20:
            weights.append(1.5)
        else:
            weights.append(0.8)
    wsum = sum(weights)
    time_alloc = [total_budget * w / wsum for w in weights]

    best_global_c5 = float('inf')
    best_global_h = None
    best_global_n = 0

    x_parent = np.linspace(0, 2, n_parent, endpoint=False)

    for idx, n in enumerate(resolutions):
        remaining = budget_s - 12 - (time.time() - t0)
        if remaining < 4:
            break
        t_alloc = min(time_alloc[idx], remaining)
        if t_alloc < 2:
            continue

        x_new = np.linspace(0, 2, n, endpoint=False)
        h_init = make_feasible(np.interp(x_new, x_parent, h_parent), n)

        h_opt, c5_opt = optimize(h_init, n, t_alloc, rng)

        if c5_opt < best_global_c5:
            best_global_c5 = c5_opt
            best_global_h = h_opt
            best_global_n = n

        # For even n, also try binary start
        if n % 2 == 0 and remaining > 30 and c5_opt > 0.375:
            remaining2 = budget_s - 12 - (time.time() - t0)
            if remaining2 > 10:
                h_bin = np.zeros(n)
                h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
                t_bin = min(remaining2 * 0.4, t_alloc)
                h_bin_opt, c5_bin = optimize(h_bin, n, t_bin, rng)
                if c5_bin < best_global_c5:
                    best_global_c5 = c5_bin
                    best_global_h = h_bin_opt
                    best_global_n = n

    if best_global_h is None:
        best_global_h = make_feasible(h_parent.copy(), n_parent)
        best_global_n = n_parent
        best_global_c5 = eval_c5(best_global_h, best_global_n)

    return best_global_h, best_global_c5, best_global_n
