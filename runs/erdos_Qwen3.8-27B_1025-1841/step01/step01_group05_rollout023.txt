import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def upsample(h, n_to):
    n_from = len(h)
    if n_from == n_to:
        return h.copy()
    # Linear interpolation style upsampling
    x_old = np.arange(n_from)
    x_new = np.arange(n_to)
    return np.interp(x_new, x_old, h).copy()

def make_feasible(h, n):
    """Ensure 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-12:
        # Distribute correction
        if s < target:
            # Need to add (target - s) across all elements
            capacity = 1.0 - h
            total_cap = capacity.sum()
            if total_cap > 0:
                h += (target - s) * capacity / total_cap
                h = np.clip(h, 0.0, 1.0)
        else:
            # Need to subtract (s - target)
            total = h.sum()
            if total > 0:
                h -= (s - target) * h / total
                h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    s = h.sum()
    if abs(s - target) > 1e-15:
        # Tiny adjustment to the element closest to 0.5
        diff = target - s
        idxs = np.argsort(np.abs(h - 0.5))
        for idx in idxs:
            room = min(1.0 - h[idx], h[idx])  # Can go either way
            if diff > 0:
                room = 1.0 - h[idx]
            else:
                room = h[idx]
            adj = min(abs(diff), room)
            h[idx] += np.sign(diff) * adj
            diff -= np.sign(diff) * adj
            if abs(diff) < 1e-15:
                break
    return h

C5 = max_k ∫ h(x)(1-h(x+k)) dx

For a binary function (h ∈ {0,1}), if A = {x: h(x)=1}, then:
∫ h(x)(1-h(x+k)) dx = |A| - |A ∩ (A-k)| = 1 - |A ∩ (A-k)|

So C5 = 1 - min_k |A ∩ (A-k)|

We want to maximize min_k |A ∩ (A-k)|, i.e., make the autocorrelation as "flat" as possible (all lags have similar overlap).

For a "perfect" difference set construction, all non-zero lags would have the same correlation value. This is related to Bose-Chowla difference sets and related combinatorial designs.

But we're working with a continuous/discrete domain [0,2] discretized into n points, and the correlation is linear (not circular), so boundary effects matter.

    return h_best, c5_final, len(h_best)
