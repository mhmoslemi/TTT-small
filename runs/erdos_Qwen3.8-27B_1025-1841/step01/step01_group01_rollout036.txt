def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    if alpha < 1:
        return m
    shifted = np.clip(alpha * (vals - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        val = smooth_max(corr, alpha)
        val += penalty * (np.sum(x) - target_sum) ** 2
        return val
    return obj

        return np.sum(np.clip(h - lam, 0, 1))

        if f(mid) > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    if alpha < 1.0:
        return m
    shifted = np.clip(alpha * (vals - m), -500.0, 500.0)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def make_obj(x, dx, alpha, penalty, target_sum):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    val = smooth_max(corr, alpha)
    val += penalty * (np.sum(x) - target_sum) ** 2
    return val

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def interpolate_h(h, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    old_c = (np.arange(n_old) + 0.5) * dx_old
    new_c = (np.arange(n_new) + 0.5) * dx_new
    f = interp1d(old_c, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    return np.clip(f(new_c), 0.0, 1.0)

def optimize_res(h_init, n, dx, time_limit, rng):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    penalty = 50.0
    t0 = time.time()
    alphas = [5.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        max_iter = 300
        def obj_func(x):
            return make_obj(x, dx, alpha, penalty, target_sum)
        try:
            res = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
        except Exception:
            break
        h = project_feasible(res.x, target_sum)
        val = compute_c5(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        if time.time() - t0 > time_limit:
            break
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    else:
        n0 = 45
        h0 = np.ones(n0) * 0.5

    best_h = project_feasible(h0, n0 / 2.0)
    best_n = n0
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    # Phase 1: Optimize at initial resolution
    t1 = min(350, budget_s * 0.22)
    best_h, best_c5 = optimize_res(best_h, best_n, 2.0 / best_n, t1, rng)

    # Random restarts at same resolution
    t_rem = budget_s - (time.time() - t_start)
    n_rest = min(4, max(1, int(t_rem / 200)))
    for _ in range(n_rest):
        if time.time() - t_start > budget_s - 150:
            break
        h_r = project_feasible(best_h + rng.randn(best_n) * 0.15, best_n / 2.0)
        tl = min(100, (budget_s - (time.time() - t_start)) * 0.12)
        h_c, c5_c = optimize_res(h_r, best_n, 2.0 / best_n, tl, rng)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 2: Higher resolutions
    for target_n in [90, 180, 360, 720]:
        if time.time() - t_start > budget_s - 120:
            break
        h_interp = interpolate_h(best_h, best_n, target_n)
        h_interp = project_feasible(h_interp, target_n / 2.0)
        dx_new = 2.0 / target_n
        c5_interp = compute_c5(h_interp, dx_new)
        tl = min(280, (budget_s - (time.time() - t_start)) * 0.35)
        h_opt, c5_opt = optimize_res(h_interp, target_n, dx_new, tl, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = target_n
        elif c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = target_n

    # Final verification
    final_dx = 2.0 / best_n
    final_c5 = compute_c5(best_h, final_dx)

    return (best_h, final_c5, best_n)
