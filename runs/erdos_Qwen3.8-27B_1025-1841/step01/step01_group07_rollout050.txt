from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_c5(h, n_points, temperature):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft max
    max_corr = np.max(corr)
    shifted = (corr - max_corr) / temperature
    vals = np.exp(shifted)
    soft_max = max_corr + temperature * np.log(np.sum(vals))
    return float(soft_max * dx)

def penalized_objective(h, n_points, penalty, temperature):
    c5 = soft_c5(h, n_points, temperature)
    sum_pen = penalty * (h.sum() - n_points / 2.0) ** 2

    diff = target_sum - h.sum()

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def soft_objective(h, n_points, temperature, penalty):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = (corr - m) / temperature
    soft_max = m + temperature * np.log(np.sum(np.exp(shifted)))
    return float(soft_max * dx + penalty * (h.sum() - n_points / 2.0) ** 2)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to_resolution(parent, n_new):
    n_old = len(parent)
    if n_old == n_new:
        return parent.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, parent, left=0.0, right=0.0)
