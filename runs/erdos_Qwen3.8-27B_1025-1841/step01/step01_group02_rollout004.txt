import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def _cd_binary_pass(h, n, rng=None):
    h = h.copy()
    improved = False
    order = np.arange(n)
    if rng is not None:
        rng.shuffle(order)
    max_corr = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    for j in order:
        old_j = h[j]
        best_val = old_j
        best_mc = max_corr
        for nv in (0.0, 1.0):
            if abs(nv - old_j) < 1e-9:
                continue
            h[j] = nv
            mc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if mc < best_mc - 1e-12:
                best_mc = mc
                best_val = nv
        h[j] = best_val
        if best_val != old_j:
            improved = True
            max_corr = best_mc
    h = project_feasible(h, n)
    return h, improved

def _cd_swap_pass(h, n, rng, n_swaps=200):
    h = h.copy()
    max_corr = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    improved = False
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, improved
    for _ in range(n_swaps):
        i = rng.choice(zeros)
        j = rng.choice(ones)
        h[i] = 1.0
        h[j] = 0.0
        mc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        if mc < max_corr - 1e-12:
            max_corr = mc
            improved = True
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[i] = 0.0
            h[j] = 1.0
    return h, improved

def _nonbinary_refine(h, n, n_trials=200, deadline=None):
    h = h.copy()
    max_corr = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    for _ in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        s = np.argmax(np.correlate(h, 1.0 - h, mode='full')) - (n - 1)
        j = int(np.random.randint(n))
        old = h[j]
        if old < 0.3:
            cands = [old + 0.1, old + 0.15, 0.5]
        elif old > 0.7:
            cands = [old - 0.1, old - 0.15, 0.5]
        else:
            cands = [0.0, 1.0, old + 0.1, old - 0.1]
        best_v = old
        best_mc = max_corr
        for cv in cands:
            cv = np.clip(cv, 0.0, 1.0)
            if abs(cv - old) < 1e-6:
                continue
            h[j] = cv
            mc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if mc < best_mc - 1e-12:
                best_mc = mc
                best_v = cv
        h[j] = best_v
        if best_v != old:
            max_corr = best_mc
    h = project_feasible(h, n)
    return h

def _slsqp_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = corr - mc
    return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx

def _slsqp_eq(h, n):
    return np.sum(h) - n / 2.0

def _slsqp_jac(h, n):
    return np.ones(n)

def _slsqp_optimize(h, n, T, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _slsqp_eq(x, n),
             'jac': lambda x: _slsqp_jac(x, n)}]
    try:
        res = minimize(_slsqp_obj, h.copy(), args=(n, T),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h
