    return soft_max_obj(x, alpha, target) + pen * (np.sum(x) - target) ** 2

def sum_eq_fun(x):
    return np.sum(x) - x.shape[0] * 0.5

def sum_eq_jac(x):
    return np.ones(x.shape[0])

def project_feasible(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            tr = np.sum(room[mask])
            if tr > 0:
                h[mask] += diff * room[mask] / tr
        else:
            mask = h > 1e-10
            tot = np.sum(h[mask])
            if tot > 0:
                h[mask] += diff * h[mask] / tot
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    init = None
    init_n = 42
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
        init_n = len(init)

    def remain():
        return budget_s - (time.time() - t0)

    if init is not None:
        init = project_feasible(init, init_n * 0.5)
        best_c5 = compute_c5(init, init_n)
        best_h = init.copy()
        best_n = init_n

    n_list = sorted(set([init_n if init is not None else 42] + [56, 70, 84, 100, 120, 140, 180, 220]))

    for n in n_list:
        if remain() < 15:
            break
        target = n * 0.5

        if n == init_n and init is not None:
            h0 = init.copy()
        elif best_h is not None:
            h0 = interp_to_n(best_h, n)
        else:
            h0 = np.full(n, 0.5)
        h0 = project_feasible(h0, target)

        if compute_c5(h0, n) < best_c5:
            best_c5 = compute_c5(h0, n)
            best_h = h0.copy()
            best_n = n

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]

        # SLSQP at multiple alpha values
        alphas = [20.0, 50.0, 100.0, 200.0]
        if n > 100:
            alphas = [30.0, 80.0]

        for alpha in alphas:
            if remain() < 10:
                break
            try:
                if n <= 100:
                    res = minimize(soft_max_obj, h0, args=(alpha, target),
                                  method='SLSQP', bounds=bounds, constraints=cons,
                                  options={'maxiter': 300, 'ftol': 1e-12})
                else:
                    res = minimize(soft_max_obj_pen, h0, args=(alpha, target, 2000.0),
                                  method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': 500, 'ftol': 1e-14})
                    res.x = project_feasible(res.x, target)
                if res.x is not None:
                    h_opt = project_feasible(res.x, target)
                    c5_opt = compute_c5(h_opt, n)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        best_n = n
                        h0 = h_opt.copy()
            except Exception:
                pass

        # Perturbation refinement
        if remain() > 8 and best_n == n:
            for trial in range(30):
                if remain() < 5:
                    break
                sigma = 0.08 * max(0.1, 1.0 - trial / 30.0)
                h_p = project_feasible(best_h + sigma * rng.randn(n), target)
                c5_p = compute_c5(h_p, n)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                    best_n = n
            # Second round with smaller sigma
            for trial in range(20):
                if remain() < 5:
                    break
                sigma = 0.02 * max(0.1, 1.0 - trial / 20.0)
                h_p = project_feasible(best_h + sigma * rng.randn(n), target)
                c5_p = compute_c5(h_p, n)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                    best_n = n

    # Final polish on best solution
    if best_h is not None and remain() > 5:
        n = best_n
        target = n * 0.5
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
        for alpha in [150.0, 300.0]:
            if remain() < 4:
                break
            try:
                if n <= 100:
                    res = minimize(soft_max_obj, best_h, args=(alpha, target),
                                  method='SLSQP', bounds=bounds, constraints=cons,
                                  options={'maxiter': 200, 'ftol': 1e-13})
                else:
                    res = minimize(soft_max_obj_pen, best_h, args=(alpha, target, 5000.0),
                                  method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': 400, 'ftol': 1e-15})
                    res.x = project_feasible(res.x, target)
                if res.x is not None:
                    h_f = project_feasible(res.x, target)
                    c5_f = compute_c5(h_f, n)
                    if c5_f < best_c5:
                        best_c5 = c5_f
                        best_h = h_f.copy()
                        best_n = n
            except Exception:
                pass

    if best_h is None:
        best_h = np.full(42, 0.5)
        best_n = 42
        best_c5 = compute_c5(best_h, best_n)

    return (best_h, best_c5, best_n)
