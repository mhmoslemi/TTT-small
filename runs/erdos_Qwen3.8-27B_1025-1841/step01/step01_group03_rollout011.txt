import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def _fun_and_jac(h, dx, temp, pw):
    n = len(h)
    omh = 1.0 - h
    c = np.correlate(h, omh, mode="full") * dx
    nc = 2 * n - 1
    sc = c / temp
    ms = np.max(sc)
    e = np.exp(sc - ms)
    se = np.sum(e)
    w = e / se
    fval = temp * (ms + np.log(se))
    grad = np.zeros(n)
    for m in range(nc):
        k = m - (n - 1)
        wm = w[m] * dx
        if abs(wm) < 1e-16:
            continue
        lo = max(0, k)
        hi = min(n, n + k)
        if lo < hi:
            grad[lo:hi] += wm * (1.0 - h[lo - k:hi - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2 + k:hi2 + k]
    sv = np.sum(h) - n * 0.5
    fval += pw * sv * sv
    grad += 2.0 * pw * sv
    return fval, grad

def _project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def _interp(h, new_n):
    if len(h) == new_n:
        return h.copy()
    ox = np.linspace(0.0, 2.0, len(h), endpoint=False)
    nx = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(nx, ox, h)

def _rand_init(n, rng):
    h = 0.5 + 0.35 * rng.randn(n)
    return _project(h, n * 0.5)

def _sine_init(n, rng):
    f = rng.randint(1, 6)
    ph = rng.uniform(0, 6.28)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
    return _project(h, n * 0.5)

def _optimize_phase(h0, n, dx, temps, pw, time_limit):
    t_start = time.time()
    best_h = h0.copy()
    best_c5 = _compute_c5(h0, dx)
    cur = h0.copy()
    for temp in temps:
        if time.time() - t_start > time_limit:
            break
        res = minimize(
            _fun_and_jac, cur, method='L-BFGS-B', jac=True,
            bounds=[(0.0, 1.0)] * n, args=(dx, temp, pw),
            options={'maxiter': 800, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 300000, 'disp': False}
        )
        h_new = _project(res.x, n * 0.5)
        c5_new = _compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        cur = best_h.copy()
    return best_h, best_c5

        hp, cp = _optimize_phase(best_h, n_best, dx_best, [0.002, 0.001], pw, tlimit)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()

    # Final validation
    best_h = _project(best_h, n_best * 0.5)
    best_c5 = _compute_c5(best_h, 2.0 / n_best)

    return best_h, best_c5, n_best
