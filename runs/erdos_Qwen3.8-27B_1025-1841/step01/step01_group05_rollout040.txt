def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    current = np.sum(h)
    if current == 0:
        return np.full(n, 0.5)
    # Simple scaling with clipping
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        current = np.sum(h)
        if abs(current - target) < 1e-10:
            break
        diff = target - current
        # Distribute the difference uniformly among non-bound variables
        free = (h > 0) & (h < 1)
        if np.sum(free) == 0:
            break
        h[free] += diff / np.sum(free)
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    current = np.sum(h)
    if current != target:
        diff = target - current
        free = (h > 0) & (h < 1)
        if np.sum(free) > 0:

def softmax_gradient(h, n, alpha):
    """Compute gradient of softmax-smoothed C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))

            continue

        m = np.arange(n)

        idx2 = m - t
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        g[valid1] += w[k] * (1.0 - h[idx1[valid1]])
        g[valid2] -= w[k] * h[idx2[valid2]]
    return g * dx, w, corr

For the exact max, the subgradient at h is the gradient of the maximizing shift (or a convex combination of gradients of all maximizing shifts).

Let me use a hybrid: use the top-m shifts with softmax weights (not the full softmax over all shifts, which would be too diffuse).

def optimize_h(h, n, time_budget=100.0):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    eta = 2.0 / n
    momentum = 0.9
    v = np.zeros(n)
    
    while time.time() - start < time_budget - 5:
        # Compute correlations
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_curr = np.max(corr)
        
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
        
        # Select top-m shifts with softmax weights
        m = min(10, 2*n-1)
        top_m = np.argsort(corr)[-m:]
        top_vals = corr[top_m]
        top_max = np.max(top_vals)
        weights = np.exp(50 * (top_vals - top_max))
        weights /= np.sum(weights)

        g = np.zeros(n)
        for j, k in enumerate(top_m):

            v2 = (idx2 >= 0) & (idx2 < n)

        v = momentum * v - eta * g
        h_new = h + v
        h_new = project_feasible(h_new, n)

Actually, for the quantization: given a continuous h, the best binary approximation with sum = n/2 is to take the top n/2 entries as 1 and the rest as 0. This is because for binary h, c_k = dx*(n/2 - |S∩(S+k)|), and we want to minimize the max, which is equivalent to maximizing the min autocorrelation.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.clip(h - tau, 0.0, 1.0)
    # Ensure exact sum
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Spread the difference among all elements proportionally
        # to their "room"
        if diff > 0:
            room = 1.0 - result
            total = room.sum()
            if total > 1e-12:
                result += room * (diff / total)
        else:
            room = result
            total = room.sum()
            if total > 1e-12:
                result -= room * (-diff / total)
        result = np.clip(result, 0.0, 1.0)
    return result

def weighted_subgradient(h, n, top_m=10, temp=50.0):
    """Compute weighted subgradient using top-m shifts."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = min(top_m, 2*n-1)
    top_m_idx = np.argsort(corr)[-m:]
    top_vals = corr[top_m_idx]
    top_max = np.max(top_vals)
    weights = np.exp(temp * (top_vals - top_max))
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for j in range(m):
        wk = weights[j]
        if wk < 1e-12:
            continue
        k = top_m_idx[j]
        t = k - (n-1)
        m_arr = np.arange(n)
        idx1 = m_arr + t
        idx2 = m_arr - t
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        g[v1] += wk * (1.0 - h[idx1[v1]])
        g[v2] -= wk * h[idx2[v2]]
    g *= dx
    return g, corr

def optimize_h(h, n, time_budget=100.0, num_restarts=3):
    """Optimize h at resolution n."""
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for restart in range(num_restarts):
        if time.time() - start > time_budget - 10:
            break
        
        if restart > 0:
            # Perturb
            h_cur = best_h + 0.1 * np.random.randn(n)
            h_cur = project_feasible(h_cur, n)
        else:
            h_cur = h.copy()
        
        eta = 3.0 / n
        momentum_coeff = 0.85
        v = np.zeros(n)
        stagnant = 0
        
        for it in range(200000):
            if time.time() - start > time_budget - 5:
                break
            
            g, corr = weighted_subgradient(h_cur, n, top_m=min(10, 2*n-1))
            c5_curr = np.max(corr)
            
            # Momentum step
            v = momentum_coeff * v - eta * g
            h_new = h_cur + v
            h_new = project_feasible(h_new, n)
            
            c5_new = compute_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                stagnant = 0
            elif c5_new < c5_curr:
                stagnant = 0
            else:
                stagnant += 1
            
            h_cur = h_new
            
            # Adapt step size
            if stagnant > 50:
                eta *= 0.8
                v *= 0.5
                if stagnant > 500:
                    break
            elif it % 200 == 0:
                eta = min(eta * 1.02, 20.0 / n)
    
    return best_h, best_c5

def interpolate_to(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new and project."""
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_quantize(h, n):
    """Quantize h to nearest binary with correct sum."""
    target = n // 2
    # Take top target entries
    threshold_idx = np.argsort(h)[-target:]
    result = np.zeros(n)
    result[threshold_idx] = 1.0
    return result

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(42, 0.5)
    
    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: Optimize at initial resolution
    time_per_phase = lambda: max(30, (budget_s - 60) - (time.time() - start_time))
    
    h_opt, c5_opt = optimize_h(h0, n0, time_budget=min(200, time_per_phase()))
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()
    
    # Also try binary quantization
    h_bin = binary_quantize(best_h, n0)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin.copy()
    
    # Phase 2: Multi-resolution refinement
    target_resolutions = [84, 120, 168, 240, 336, 480, 672]
    current_n = len(best_h)
    
    for target_n in target_resolutions:
        if target_n <= current_n:
            continue
        elapsed = time.time() - start_time
        remaining = budget_s - 60 - elapsed
        if remaining < 40:
            break
        
        # Interpolate
        h_fine = interpolate_to(best_h, current_n, target_n)
        current_n = target_n
        
        # Optimize
        h_fine, c5_fine = optimize_h(h_fine, target_n, time_budget=remaining * 0.7)
        if c5_fine < best_c5:
            best_c5, best_h = c5_fine, h_fine.copy()
        
        # Try binary quantization at this resolution
        h_bin = binary_quantize(best_h, target_n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin.copy()
        
        elapsed = time.time() - start_time
        remaining = budget_s - 60 - elapsed
        if remaining < 40:
            break
    
    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
