import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += d / np.sum(mask)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += d / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_top_k(h, n, idx, w):
    """
    Gradient of sum w_i * overlap[idx_i]
    idx: indices into full correlation array (length 2n-1)
    w: weights
    """
    grad = np.zeros(n)
    for i in range(len(idx)):
        j = idx[i]
        wi = w[i]
        s = j - (n - 1)  # shift
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            # f_s = sum_{m=0}^{n-1-s} h[m+s]*(1-h[m])
            # d/dh[j] = (1-h[j-s]) for j in [s, n-1]
            #         - h[j+s] for j in [0, n-1-s]
            if s < n:
                # First part: j in [s, n-1]
                grad[s:] += wi * (1.0 - h[:n-s])
                # Second part: j in [0, n-1-s]
                grad[:n-s] += wi * (-h[s:])
        else:
            # s < 0, let sp = -s
            sp = -s
            # f_{-sp} = sum_{m=0}^{n-1-sp} h[m]*(1-h[m+sp])
            # d/dh[j] = (1-h[j+sp]) for j in [0, n-1-sp]
            #         - h[j-sp] for j in [sp, n-1]
            if sp < n:
                grad[:n-sp] += wi * (1.0 - h[sp:])
                grad[sp:] += wi * (-h[:n-sp])
    dx = 2.0 / n
    return grad * dx

def top_k_descent(h, n, deadline, K=10, lr_init=0.01, momentum=0.9, n_steps=5000, lr_decay=0.999):
    h = h.copy()
    vel = np.zeros(n)
    lr = lr_init
    best_local = h.copy()
    best_c5_local = compute_overlaps(h).max()
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        ov = compute_overlaps(h)
        c5_val = ov.max()
        if c5_val < best_c5_local - 1e-15:
            best_c5_local = c5_val
            best_local = h.copy()
        
        # Top-K with softmax weighting
        top_idx = np.argsort(ov)[-K:]
        top_vals = ov[top_idx]
        T_w = 30.0
        exp_v = np.exp(T_w * (top_vals - top_vals.max()))
        w = exp_v / exp_v.sum()
        
        g = grad_top_k(h, n, top_idx, w)
        vel = momentum * vel - lr * g
        h = h + vel
        h = project(h, n)
        
        lr *= lr_decay
        if lr < 1e-8:
            break
    return best_local

def interp_upsample(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    h_new = np.interp(xt, xf, h)
    return h_new

    n_from = len(h)
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def downsample(h, n_to):
    n_from = len(h)
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def binarize_preserve(h, n):
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def soft_binarize(h, n, sharpness=10.0):
    mid = np.median(h)
    h_soft = 1.0 / (1.0 + np.exp(-sharpness * (h - mid)))
    return project(h_soft, n)

def cyclic_gd(h, n, deadline, K_list, lr0=0.01, mom=0.9, max_steps=20000):
    h = h.copy()
    vel = np.zeros(n)
    lr = lr0
    best = h.copy()
    best_c5 = compute_overlaps(best).max()
    ki = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        K = K_list[ki % len(K_list)]
        ov = compute_overlaps(h)
        c5 = ov.max()
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best = h.copy()
        top_pos = np.argsort(ov)[-K:]
        tv = ov[top_pos]
        w = np.exp(40.0 * (tv - tv.max()))
        w /= w.sum()
        g = grad_top_k(h, n, top_pos, w)
        vel = mom * vel - lr * g
        h = h + vel
        h = project(h, n)
        lr *= 0.9999
        if step % 500 == 0:
            ki += 1
        if lr < 1e-12:
            break
    return best

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(336, 0.5)

    n0 = len(h0)
    best_h = project(h0, n0)
    best_c5 = compute_overlaps(best_h).max()

    # Phase 1: Refine at original resolution with multiple K values
    remain = deadline - time.time()
    if remain > 60:
        for K in [5, 10, 20, 50]:
            if time.time() > deadline - 60:
                break
            alloc = min(30.0, (deadline - time.time()) * 0.25)
            h_new = gd_top_k(best_h, n0, deadline - alloc + time.time(),
                             K=K, lr0=0.015, mom=0.92, max_steps=15000)
            c5 = compute_overlaps(h_new).max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
            # Also try cyclic
            if time.time() < deadline - 60:
                h_cyc = cyclic_gd(best_h, n0, deadline - 40, [5, 10, 20],
                                   lr0=0.01, mom=0.9, max_steps=10000)
                c5_c = compute_overlaps(h_cyc).max()
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_cyc

    # Phase 2: Upsample to n=672
    if time.time() < deadline - 200:
        n_up = 2 * n0
        h_up = upsample(best_h, n_up)
        h_up = project(h_up, n_up)
        for K in [10, 20]:
            if time.time() > deadline - 100:
                break
            h_new = gd_top_k(h_up, n_up, deadline - 60,
                             K=K, lr0=0.008, mom=0.92, max_steps=12000)
            c5 = compute_overlaps(h_new).max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
            h_up = h_new
        h_up = cyclic_gd(h_up, n_up, deadline - 30, [8, 15, 25],
                         lr0=0.006, mom=0.9, max_steps=8000)
        c5 = compute_overlaps(h_up).max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_up

    # Phase 3: Binarization strategies
    n_best = len(best_h)
    if time.time() < deadline - 150:
        # Try various binarization thresholds
        for attempt in range(15):
            if time.time() > deadline - 30:
                break
            noise = np.random.randn(n_best) * (0.03 + 0.02 * attempt)
            h_pert = best_h + noise
            h_bin = binarize_preserve(h_pert, n_best)
            h_new = gd_top_k(h_bin, n_best, min(deadline - 15, time.time() + 8),
                             K=10, lr0=0.012, mom=0.9, max_steps=4000)
            c5 = compute_overlaps(h_new).max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

        # Soft binarization
        for sharp in [5.0, 10.0, 20.0]:
            if time.time() > deadline - 30:
                break
            h_soft = soft_binarize(best_h, n_best, sharp)
            h_new = gd_top_k(h_soft, n_best, min(deadline - 15, time.time() + 8),
                             K=10, lr0=0.01, mom=0.9, max_steps=3000)
            c5 = compute_overlaps(h_new).max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase 4: Random restarts (larger perturbations to find new basins)
    n_best = len(best_h)
    n_restarts = 30
    if time.time() < deadline - 40:
        per_restart = max(3.0, (deadline - time.time() - 20) / n_restarts)
        for attempt in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.05 + 0.15 * (attempt / n_restarts)
            h_pert = best_h + np.random.randn(n_best) * scale
            h_pert = project(h_pert, n_best)
            h_new = gd_top_k(h_pert, n_best, min(deadline - 10, time.time() + per_restart * 0.8),
                             K=10, lr0=0.02, mom=0.9, max_steps=5000)
            c5 = compute_overlaps(h_new).max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase 5: Final fine-tuning with very small steps
    if time.time() < deadline - 15:
        h_final = gd_top_k(best_h, n_best, deadline - 5,
                           K=5, lr0=0.003, mom=0.85, max_steps=10000)
        c5 = compute_overlaps(h_final).max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final

    best_h = project(best_h, len(best_h))
    best_c5 = compute_overlaps(best_h).max()
    return (best_h, best_c5, len(best_h))
