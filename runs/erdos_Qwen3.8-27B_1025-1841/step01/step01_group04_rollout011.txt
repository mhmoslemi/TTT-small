```python
def compute_gradient(h, n, T, dx):
    v = 1.0 - h
    corr = np.correlate(h, v, 'full')  # length 2n-1
    scaled = T * dx * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # corr_full[k] = sum_i h[i]*(1-h[n-1-k+i])
    # d corr_full[k] / d h[t] = (1-h[t-(n-1-k)])_valid - h[t+(n-1-k)]_valid
    for k in range(2*n-1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        m_k = n-1-k
        # (1-h[t-m_k]) for 0<=t-m_k<n, i.e., m_k<=t<m_k+n
        lo1 = max(0, m_k)
        hi1 = min(n, m_k + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1 - h[lo1-m_k:hi1-m_k])
        # -h[t+m_k] for 0<=t+m_k<n, i.e., -m_k<=t<n-m_k
        lo2 = max(0, -m_k)
        hi2 = min(n, n - m_k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2+m_k:hi2+m_k]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_value(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def analytic_gradient(x, n, T):
    dx = 2.0 / n
    v = 1.0 - x
    corr = np.correlate(x, v, mode='full')  # length 2n-1
    scaled = T * dx * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    w = w / w_sum  # simplex weights
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = dx * w[k]
        if wk < 1e-15:
            continue
        m_k = n - 1 - k
        # Term 1: (1-x[t-m_k]) for valid t
        # t-m_k in [0, n-1] => t in [m_k, m_k+n-1], intersect [0, n-1]
        lo1 = max(0, m_k)
        hi1 = min(n - 1, m_k + n - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * (1.0 - x[idx - m_k])
        # Term 2: -x[t+m_k] for valid t
        # t+m_k in [0, n-1] => t in [-m_k, n-1-m_k], intersect [0, n-1]
        lo2 = max(0, -m_k)
        hi2 = min(n - 1, n - 1 - m_k)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] -= wk * x[idx + m_k]
    return grad

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_with_grad(h0, n, T, maxiter=1000, deadline=None):
    def f(x):
        return soft_max_value(x, n, T)
    def g(x):
        return analytic_gradient(x, n, T)
    def c_eq(x):
        return np.sum(x) - n / 2.0
    def c_jac(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_jac}],
                       options={'ftol': 1e-15, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def fista_optimize(h0, n, T, n_iter, deadline, lr_init=None):
    h = h0.copy()
    y = h0.copy()
    t_mom = 1.0
    dx = 2.0 / n
    
    if lr_init is None:
        lr_init = 1.0 / (4.0 * dx * max(1, T * 0.1))
    lr = lr_init
    
    best_h = h0.copy()
    best_f = soft_max_value(h0, n, T)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        grad = analytic_gradient(y, n, T)
        y = y - lr * grad
        h_new = project(y, n)
        
        f_new = soft_max_value(h_new, n, T)  # expensive but we need it for best tracking
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = h_new + (t_mom - 1.0) / t_new * (h_new - h)
        h = h_new
        t_mom = t_new
    
    return best_h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)
```

```python
def analytic_gradient(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = dx * w[k]
        if wk < 1e-16:
            continue
        p = n - 1 - k
        # First term: (1-x[t+p]) for t in [max(0,-p), min(n-1, n-1-p)]
        lo1 = max(0, -p)
        hi1 = min(n - 1, n - 1 - p)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * (1.0 - x[lo1+p:hi1+1+p])
        # Second term: -x[t-p] for t in [max(0,p), n-1]
        lo2 = max(0, p)
        hi2 = n - 1
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wk * x[lo2-p:hi2+1-p]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_value(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def analytic_gradient(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = dx * w[k]
        if wk < 1e-16:
            continue
        p = n - 1 - k
        lo1 = max(0, -p)
        hi1 = min(n - 1, n - 1 - p)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wk * (1.0 - x[lo1 + p:hi1 + 1 + p])
        lo2 = max(0, p)
        hi2 = n - 1
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= wk * x[lo2 - p:hi2 + 1 - p]
    return grad

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_with_grad(h0, n, T, maxiter=500, deadline=None):
    def f(x):
        return soft_max_value(x, n, T)
    def g(x):
        return analytic_gradient(x, n, T)
    def c_eq(x):
        return np.sum(x) - n / 2.0
    def c_jac(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_jac}],
                       options={'ftol': 1e-15, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def penalty_lbfgsb(h0, n, T, deadline):
    h = h0.copy()
    target = n / 2.0
    dx = 2.0 / n
    for mu in [1.0, 10.0, 100.0, 1000.0, 10000.0, 100000.0]:
        if time.time() > deadline - 3:
            break
        def f_pen(x):
            val = soft_max_value(x, n, T)
            s = np.sum(x) - target
            return val + mu * s * s / 2.0
        def g_pen(x):
            g = analytic_gradient(x, n, T)
            s = np.sum(x) - target
            g += mu * s
            return g
        try:
            res = minimize(f_pen, h, jac=g_pen, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h = project(res.x, n)
        except Exception:
            pass
    return h

def fista_optimize(h0, n, T, n_iter, deadline, lr=1.0):
    h = h0.copy()
    y = h0.copy()
    t_mom = 1.0
    best_h = h0.copy()
    best_f = soft_max_value(h0, n, T)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        grad = analytic_gradient(y, n, T)
        y = y - lr * grad
        h_new = project(y, n)
        f_new = soft_max_value(h_new, n, T)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = h_new + (t_mom - 1.0) / t_new * (h_new - h)
        h = h_new
        t_mom = t_new
    return best_h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n0 = len(h_start)

    # Phase 1: Penalty L-BFGS-B at parent resolution with decreasing T
    for T in [80.0, 40.0, 20.0, 10.0]:
        if time.time() > deadline - 60:
            break
        h_new = penalty_lbfgsb(best_h, n0, T, deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: SLSQP polish at low T
    for T in [10.0, 5.0, 3.0]:
        if time.time() > deadline - 30:
            break
        h_new = slsqp_with_grad(best_h, n0, T, maxiter=400, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: FISTA with moderate step
    if time.time() < deadline - 40:
        h_new = fista_optimize(best_h, n0, T=15.0, n_iter=8000, deadline=deadline, lr=0.8)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Try nearby resolutions
    for n_try in [332, 340, 344, 348, 352, 356, 360]:
        if time.time() > deadline - 30:
            break
        h_try = interp_to(best_h, n0, n_try)
        h_try = project(h_try, n_try)
        h_try = penalty_lbfgsb(h_try, n_try, 20.0, deadline)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Random perturbations
    n_best = len(best_h)
    for attempt in range(30):
        if time.time() > deadline - 15:
            break
        scale = 0.04 + 0.04 * (attempt % 3)
        h_pert = best_h + np.random.randn(n_best) * scale
        h_pert = project(h_pert, n_best)
        h_new = penalty_lbfgsb(h_pert, n_best, 15.0, deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if time.time() < deadline - 20:
            h_new2 = slsqp_with_grad(best_h, n_best, 5.0, maxiter=150, deadline=deadline)
            c5_new2 = evaluate_c5(h_new2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()

    # Phase 6: Upscale to n=672
    if time.time() < deadline - 40:
        h_up = interp_to(best_h, n_best, 672)
        h_up = project(h_up, 672)
        h_up = penalty_lbfgsb(h_up, 672, 20.0, deadline)
        c5_up = evaluate_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()

    # Final SLSQP polish
    if time.time() < deadline - 10:
        h_new = slsqp_with_grad(best_h, len(best_h), 5.0, maxiter=300, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
