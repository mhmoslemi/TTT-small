def project(h, n):

                best_h = h_hi.copy()
            if time.time() < deadline - 40:
                h_hi2 = slsqp_opt(h_hi, n_try, 30.0, maxiter=200, deadline=deadline)
                c_hi2 = evaluate_c5(h_hi2)
                if c_hi2 < best_c5:
                    best_c5 = c_hi2
                    best_h = h_hi2.copy()

    # Phase 6: L-BFGS polish on best solution
    if time.time() < deadline - 30:
        h_lb = lbfgs_opt(best_h, len(best_h), 50.0, maxiter=300, deadline=deadline)
        c_lb = evaluate_c5(h_lb)
        if c_lb < best_c5:
            best_c5 = c_lb
            best_h = h_lb.copy()

    # Final polish with very low T gradient descent
    if time.time() < deadline - 5:
        h_final, c_final = proj_grad_descent(best_h, len(best_h), 10.0, max_iter=500, lr_init=0.005, deadline=deadline)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()

    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
