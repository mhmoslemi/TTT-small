Δcorr[sh] = (1-h[i+sh])*[0≤i+sh<n] - h[i-sh]*[0≤i-sh<n] - (1-h[j+sh])*[0≤j+sh<n] + h[j-sh]*[0≤j-sh<n]

For the local search, the key operation is: given the current correlation vector, find a pair flip that reduces the maximum. The maximum is at shift k*. We want to reduce corr[k*].

The change in corr[k*] for flipping (i, j) is:

where sh* = k* - (n-1).

To reduce corr[k*], we want Δcorr[k*] < 0.

The term (1-h[i+sh*]) is 1 if h[i+sh*]=0 and 0 if h[i+sh*]=1.
The term h[i-sh*] is 1 if h[i-sh*]=1 and 0 otherwise.

So Δcorr[k*] = [h[i+sh*]==0] - [h[i-sh*]==1] - [h[j+sh*]==0] + [h[j-sh*]==1]

For the first flip (i: 0→1):
- Contribution: [h[i+sh*]==0] - [h[i-sh*]==1]
- This is +1 if h[i+sh*]=0 (increases corr), -1 if h[i-sh*]=1 (decreases corr)
- Net: -1 if h[i+sh*]=1 AND h[i-sh*]=1 (decreases corr by 1)
  +1 if h[i+sh*]=0 AND h[i-sh*]=0 (increases corr by 1)
  0 otherwise

For the second flip (j: 1→0):
- Contribution: -[h[j+sh*]==0] + [h[j-sh*]==1]
- This is -1 if h[j+sh*]=0 (decreases corr), +1 if h[j-sh*]=1 (increases corr)
- Net: -1 if h[j+sh*]=0 AND h[j-sh*]=0 (decreases corr by 1)
  +1 if h[j+sh*]=1 AND h[j-sh*]=1 (increases corr by 1)
  0 otherwise

So to decrease corr[k*], we want:
- For i: h[i+sh*]=1 and h[i-sh*]=1 (i is a 0 that has 1s at both shifted positions)
- For j: h[j+sh*]=0 and h[j-sh*]=0 (j is a 1 that has 0s at both shifted positions)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[k] += pk * (1.0 - h[j0:j1 - sh] if sh >= 0 else h[j0:j1 - sh])
    # Recompute gradient properly
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, max_iters, time_limit, lr_init=0.3):

    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n)
    u = np.zeros(n)
    t = 0
    stall = 0

    for it in range(max_iters):

            gain_i = 0

            for j in one_list:

                gain_j = 0

        if best_i >= 0 and best_j >= 0:

            j1 = min(n_final, sh + n_final)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h_final[idx - sh])
            j0b = max(0, -sh)
            j1b = min(n_final, n_final - sh)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h_final[idx + sh]
            grad *= dx
            v = 0.85 * v - lr * grad
            h_final = project_to_feasible(h_final + v, n_final)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: One more binary local search attempt with different seed
    if remaining() > 60:
        n_try = len(best_h)
        h_try = (best_h > 0.5).astype(np.float64)
        h_try, c5_try = guided_local_search(h_try, n_try, max_iters=500000, time_limit=remaining() * 0.4, seed=seed + 100)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
