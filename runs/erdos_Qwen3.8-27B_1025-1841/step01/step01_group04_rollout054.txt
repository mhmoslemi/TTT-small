import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    h_new = np.interp(xt, xf, h)
    return project(h_new, n_to / 2.0)

def softmax_obj_grad(h, T, mu=50.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp
    obj = (m + np.log(sum_exp)) / T * dx + mu * (np.sum(h) - target) ** 2
    
    # Gradient
    d_arr = p[n - 1:]  # length n
    t1_corr = np.correlate(d_arr, 1.0 - h, mode='full')
    t1 = t1_corr[n-1:n - 1 - n:-1]  # t1[j] = t1_corr[n-1-j]
    t2_conv = np.convolve(h, d_arr, mode='full')
    t2 = t2_conv[:n]
    
    grad_sm = dx * (t1 - t2)
    grad_pen = 2.0 * mu * (np.sum(h) - target)
    grad = grad_sm + grad_pen
    
    return obj, grad

def lgbfsq_opt(h0, T, mu, maxiter=500, deadline=None):
    n = len(h0)
    def fun(x):
        return softmax_obj_grad(x, T, mu)[0]
    def grad(x):
        return softmax_obj_grad(x, T, mu)[1]
    try:
        res = minimize(fun, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxls': 50, 'maxfun': maxiter * 3})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n / 2.0)
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1300, **kwargs):

    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        h_start = np.full(336, 0.5)
    
    n = len(h_start)
    h_start = project(h_start, n / 2.0)
    
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Multi-T optimization with L-BFGS-B
    T_schedule = [30, 60, 120, 250, 500, 1000]
    for T in T_schedule:
        if time.time() > deadline - 30:
            break
        h_new = lgbfsq_opt(best_h, T, 50.0, maxiter=300, deadline=deadline)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    
    return (best_h, best_c5, n)

import time

def evaluate_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

        d = target - s
        if abs(d) < 1e-14:

        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt

    return h

    xt = np.linspace(0, 2, n_to, endpoint=False)
    return project(np.interp(xt, xf, h), n_to / 2.0)

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp

    t2 = np.convolve(h, d_arr, mode='full')[:n]

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return project(np.interp(xt, xf, h), n_to / 2.0)

def softmax_obj_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp
    obj = (m + np.log(sum_exp)) / T * dx + mu * (np.sum(h) - target) ** 2
    d_arr = p[n - 1:]
    t1 = np.correlate(d_arr, 1.0 - h, mode='full')[n - 1:n - 1 - n:-1]
    t2 = np.convolve(h, d_arr, mode='full')[:n]
    grad = dx * (t1 - t2) + 2.0 * mu * (np.sum(h) - target)
    return obj, grad

def opt_lbgfb(h0, T, mu, maxiter=400, deadline=None):
    n = len(h0)
    def fun(x):
        return softmax_obj_grad(x, T, mu)[0]
    def jac(x):
        return softmax_obj_grad(x, T, mu)[1]
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 50, 'maxfun': maxiter * 5})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n / 2.0)
    except Exception:
        pass
    return h0
