    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max() - corr.min()

    w = expo / expo.sum()
    c5 = float(corr.max() * dx)

def smooth_c5_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    expo = np.exp(tau * (corr - corr.max()))
    w = expo / expo.sum()
    c5 = float(corr.max() * dx)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += (1.0 - 2.0 * h) * wi
            else:
                grad[lag:] += (1.0 - h[:n - lag]) * wi
                grad[:n - lag] -= h[lag:] * wi
        else:
            L = -lag
            grad[:n - L] += (1.0 - h[L:]) * wi
            grad[L:] -= h[:n - L] * wi
    return c5, grad * dx

def projected_descent(h0, target_sum, max_iter, tau, time_limit):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    h = project_simplex_box(h, target_sum)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    t_start = time.time()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        c5, grad = smooth_c5_grad(h, dx, tau)
        v = momentum * v + (1 - momentum) * grad
        h_new = h - lr * v
        h_proj = project_simplex_box(h_new, target_sum)
        c5_new = eval_c5(h_proj)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_proj.copy()
        
        if c5_new <= c5 + 1e-10:
            h = h_proj
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
        
        # Anneal tau
        if it % 200 == 0 and it > 0:
            tau *= 0.95
    
    return best_h, best_c5

def slsqp_polish(h0, target_sum, time_limit):
    n = len(h0)
    x0 = h0.copy()
    
    def obj(x):
        return eval_c5(x)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                   'jac': lambda x: np.ones(n)}]
    
    t_start = time.time()
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success or res.fun < obj(x0):
            h_res = np.clip(res.x, 0, 1)
            h_res = project_simplex_box(h_res, target_sum)
            return h_res, eval_c5(h_res)
    except:
        pass
    return x0, eval_c5(x0)
