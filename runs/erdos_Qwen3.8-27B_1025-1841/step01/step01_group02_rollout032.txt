Actually, for a 1D minimax of linear functions, there's a simpler approach:

- Its minimum is where f'(t) = 0 (if it exists in the interior) or at a boundary
- f'(t) = α_k where k is the argmax (or a combination at a breakpoint)
- So the minimum is where the α_k of the active constraints crosses 0

- If the entire subgradient is positive (min > 0): go left
- Otherwise (0 is in the subgradient): stop

    shifted = corr - mc
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    return (mc + np.log(np.sum(exp_s)) / T) * dx

def _grad_fn(h):
    T = _CUR_T
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    omh = 1.0 - h
    for offset in range(2 * n - 1):
        k = offset - (n - 1)
        w = weights[offset]
        if w < 1e-14:
            continue
        js = max(0, k)
        je = min(n, n + k)
        if je > js:
            grad[js:je] += w * omh[js - k:je - k]
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if je2 > js2:
            grad[js2:je2] -= w * h[js2 + k:je2 + k]
    return grad * dx

def _cons_fun(h):
    return np.sum(h) - len(h) / 2.0

def _cons_jac(h):
    return np.ones(len(h))

def _timeup(margin=10):
    return (time.time() - _START) > _BUDGET - margin

def coord_pass(h, max_passes=10):
    n = len(h)
    best_c5 = compute_c5(h)
    for p in range(max_passes):
        if _timeup(15):
            break
        improved = False
        for j in range(n):
            if _timeup(15):
                break
            hz = h.copy()
            hz[j] = 0.0
            ho = h.copy()
            ho[j] = 1.0
            c0 = np.correlate(hz, 1.0 - hz, mode='full')
            c1 = np.correlate(ho, 1.0 - ho, mode='full')
            alphas = c1 - c0
            betas = c0
            lo_t, hi_t = 0.0, 1.0
            for _ in range(50):
                mid_t = 0.5 * (lo_t + hi_t)
                vals = alphas * mid_t + betas
                mx = np.max(vals)
                active = vals >= mx - 1e-9
                if np.max(alphas[active]) < -1e-9:
                    lo_t = mid_t
                elif np.min(alphas[active]) > 1e-9:
                    hi_t = mid_t
                else:
                    break
            t_opt = 0.5 * (lo_t + hi_t)
            if abs(t_opt - h[j]) > 1e-7:
                h[j] = t_opt
                improved = True
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return h, best_c5

def slsqp_opt(h, T, maxiter=500):
    global _CUR_T
    _CUR_T = T
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h

def subgrad_step(h, n_steps=200):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    step_sz = 0.05
    for _ in range(n_steps):
        if _timeup(15):
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        js = max(0, k_star)
        je = min(n, n + k_star)
        if je > js:
            grad[js:je] += (1.0 - h[js - k_star:je - k_star])
        js2 = max(0, -k_star)
        je2 = min(n, n - k_star)
        if je2 > js2:
            grad[js2:je2] -= h[js2 + k_star:je2 + k_star]
        grad *= dx
        h_new = make_feasible(h - step_sz * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            h = h_new
        else:
            step_sz *= 0.9
            if step_sz < 1e-6:
                break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _BUDGET, _START
    _BUDGET = budget_s
    _START = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is not None:
        h0 = make_feasible(h0, len(h0))
        best_c5 = compute_c5(h0)
        best_h = h0.copy()
    else:
        n_init = 320
        h0 = np.zeros(n_init)
        h0[:n_init // 2] = 1.0
        best_c5 = compute_c5(h0)
        best_h = h0.copy()

    # Phase 1: SLSQP with analytical gradient at current resolution
    for T in [30, 20, 15, 10, 7, 5, 3, 2]:
        if _timeup(40):
            break
        h_opt = slsqp_opt(best_h, T, maxiter=400)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        best_h, best_c5 = coord_pass(best_h, max_passes=3)
        if _timeup(40):
            break

    # Phase 2: Subgradient polishing
    best_h, best_c5 = subgrad_step(best_h, n_steps=300)
    best_h, best_c5 = coord_pass(best_h, max_passes=5)

    # Phase 3: Higher resolution
    for n_new in [480, 640, 960]:
        if _timeup(100):
            break
        h_high = upsample_h(best_h, n_new)
        for T in [15, 10, 5, 3]:
            if _timeup(40):
                break
            h_opt = slsqp_opt(h_high, T, maxiter=300)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            best_h, best_c5 = coord_pass(best_h, max_passes=3)
            if _timeup(50):
                break

    # Phase 4: Perturbation and re-optimization to escape local minima
    if not _timeup(120):
        n = len(best_h)
        for attempt in range(3):
            if _timeup(40):
                break
            h_pert = best_h + rng.randn(n) * 0.03
            h_pert = make_feasible(h_pert, n)
            for T in [5, 3]:
                if _timeup(20):
                    break
                h_opt = slsqp_opt(h_pert, T, maxiter=300)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
            best_h, best_c5 = coord_pass(best_h, max_passes=5)

    # Phase 5: Final intensive coordinate descent
    best_h, best_c5 = coord_pass(best_h, max_passes=50)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
