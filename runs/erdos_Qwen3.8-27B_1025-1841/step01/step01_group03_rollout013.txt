def get_all_overlaps(h, n):

            j = int(rng.choice(zeros))

def continuous_refine(h, n, time_limit=20.0):
    """Refine a binary solution in continuous [0,1]^n with sum constraint."""
    start = time.time()
    
    def objective(x):
        if time.time() - start > time_limit:
            return np.inf
        # Project: ensure sum(x) = n/2 approximately
        return compute_c5(x, n)
    
    def objective_and_grad(x):
        if time.time() - start > time_limit:
            return np.inf, np.zeros(n)
        val = compute_c5(x, n)
        # Subgradient: find the max k
        corr = np.correlate(x, 1 - x, mode="full")
        k_max = np.argmax(corr)
        # Gradient of corr[k_max] w.r.t. x
        lag = k_max - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            for idx in range(n - lag):
                grad[idx] += (1 - x[idx + lag])
                grad[idx + lag] += x[idx]
        else:
            l = -lag
            for idx in range(n - l):
                grad[idx] += x[idx + l]
                grad[idx + l] += (1 - x[idx])
        return val * (2.0 / n), grad * (2.0 / n)
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0, 1)] * n
    
    res = minimize(objective_and_grad, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 200, 'ftol': 1e-6})
    
    if res.success or res.fun < compute_c5(h, n):
        # Project to feasible
        x = res.x
        x = np.clip(x, 0, 1)
        # Fix sum
        s = np.sum(x)
        if abs(s - n/2) > 1e-6:
            # Simple projection
            x = x + (n/2 - s) / n
            x = np.clip(x, 0, 1)
        return x, compute_c5(x, n)
    
    return h, compute_c5(h, n)

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")

    h = h.copy()
