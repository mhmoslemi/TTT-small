def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def best_t(c0, cf):
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = c0 + mid * cf
        mv = vals.max()
        mask = vals >= mv - 1e-9 * max(1.0, abs(mv))
        if not mask.any():
            mask = np.zeros(len(vals), dtype=bool)
            mask[vals.argmax()] = True
        if cf[mask].min() > 1e-12:
            hi = mid
        elif cf[mask].max() < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descend(h, max_sweeps=200, time_limit=None):
    n = len(h)
    dx = 2.0 / n
    best = eval_c5(h)

            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = old
            t = best_t(c0, c1 - c0)
            h[i] = t
            nc = np.correlate(h, 1.0 - h, mode='full').max() * dx
            if nc < best - 1e-14:
                best = nc
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return h, best

def fix_sum(h, target):
    h = h.astype(np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) < 1e-13:
        return h
    if d > 0:
        for i in np.argsort(1.0 - h):
            a = min(1.0 - h[i], d)
            h[i] += a
            d -= a
            if d < 1e-14:
                break
    else:
        for i in np.argsort(h):
            a = min(h[i], -d)
            h[i] -= a
            d += a
            if d > -1e-14:
                break
    return np.clip(h, 0.0, 1.0)
