def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # Water-filling correction
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)

def project_feasible(v, n):
    """Project v onto {0≤h≤1, sum(h)=n/2} via bisection on lambda."""
    target = n / 2.0
    lo, hi = -10.0, 10.0
    def sum_at(lam):
        return np.sum(np.clip(v - lam, 0.0, 1.0))
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if sum_at(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_optimize(h, n, total_iters, time_limit, start_time, tau_start=100, tau_end=5000):

        if time.time() - start_time > time_limit:
            break
        # Adaptive tau
        progress = it / max(total_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = tau * corr * dx
    z = z - z.max()
    w = np.exp(z)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            j1 = np.arange(k, n)
            grad[j1] += w[m] * (1.0 - h[j1 - k])
            j2 = np.arange(0, n - k)
            grad[j2] -= w[m] * h[j2 + k]
        else:
            kp = -k
            j1 = np.arange(0, n - kp)
            grad[j1] += w[m] * (1.0 - h[j1 + kp])
            j2 = np.arange(kp, n)
            grad[j2] -= w[m] * h[j2 - kp]
    return grad * dx

def subgradient_optimize(h, n, iters, time_limit, start_time, tau_start=20.0, tau_end=3000.0, alpha0=0.05):
    best_h = h.copy()
    best_c = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.7
    for it in range(iters):
        if time.time() - start_time > time_limit:
            break
        progress = it / max(iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        g = smooth_max_grad(h, n, tau)
        alpha = alpha0 / (1.0 + 0.01 * it)
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = h - alpha * momentum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def coord_descent_binary(h, n, max_rounds=50, time_limit=600, start_time=None):
    dx = 2.0 / n
    best_c = compute_c5(h)
    improved_any = True
    rnd = 0
    while improved_any and rnd < max_rounds:
        improved_any = False
        rnd += 1
        if start_time is not None and time.time() - start_time > time_limit - 10:
            break

def coord_descent_fine(h, n, max_rounds=20, time_limit=300, start_time=None):

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

            pattern.extend([0.0] * block)

    h = make_feasible(h, n)
    return h
