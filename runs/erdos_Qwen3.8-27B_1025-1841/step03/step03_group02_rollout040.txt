def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial
    try:
        h = np.array(initial_h_values, dtype=float)
    except:
        h = np.full(960, 0.5)
    
    n = len(h)
    h = make_feasible(h, n)
    
    # Convert to binary
    b = to_binary(h, n)
    
    best_c5 = compute_c5(b)
    best_b = b.copy()
    
    # Main local search loop
    T = 0.005  # initial temperature
    T_min = 0.0001
    curr_c5 = best_c5
    curr_b = b.copy()
    
    eval_count = 0
    while True:
        remaining = budget_s - (time.time() - start)
        if remaining < 5:
            break
        
        # Propose a swap
        # Strategy: pick i from ones, j from zeros
        ones = np.where(curr_b == 1)[0]
        zeros = np.where(curr_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        curr_b[i] = 0
        curr_b[j] = 1
        new_c5 = compute_c5(curr_b)
        eval_count += 1
        
        delta = new_c5 - curr_c5
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = curr_b.copy()
        else:
            curr_b[i] = 1
            curr_b[j] = 0
        
        # Anneal
        T = max(T * 0.9999, T_min)
        
        # Periodic restart from best
        if eval_count % 100000 == 0:
            curr_b = best_b.copy()
            curr_c5 = best_c5
    
    return best_b, best_c5, n

2. Main loop (budget ≈ 1200s):

Actually, I realize I can be smarter: instead of evaluating all candidates fully, I can use the O(n) update for the correlation at the worst lag only. The change in C(k*) from flipping h[fp] from 0 to 1 is:
- If k* > 0: C(k*) = sum_j h[j+k*](1-h)[j]. Flipping h[fp] 0→1:
  - If fp is in the "j" position (i.e., j = fp): C(k*) changes by -h[fp+k*] (the (1-h)[fp] term goes from 1 to 0, so we lose h[fp+k*])
    Wait: C(k*) includes term h[fp+k*] * (1-h)[fp]. Before: (1-h)[fp] = 1. After: 0. So change = -h[fp+k*].
  - If fp is in the "j+k*" position (i.e., j+k* = fp, j = fp-k*): C(k*) changes by +h[fp] * (1-h)[j]... wait, h[fp] goes from 0 to 1, and (1-h)[j] = (1-h)[fp-k*]. So change = (1-h)[fp-k*].
  
  Total change in C(k*): ΔC(k*) = (1-h)[fp-k*] - h[fp+k*] (if both in range)
  where fp-k* >= 0 for the first term, fp+k* < n for the second term.

This is O(1)! So I can quickly estimate the change in the worst lag without full correlation.

Then I only need to do a full correlation check for the winning candidate. This reduces the cost per iteration to:
- O(1) for estimating ΔC(k*) for each candidate: ~0 μs
- O(n²) for the one full correlation: ~0.4 ms
- Total: ~0.5 ms per iteration

That gives us ~2.4 million iterations in 1200s!

    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n = len(h_init)
    h_init = make_feasible(h_init, n)
    b = to_binary(h_init, n)
    
    best_b = b.copy()
    best_c5 = compute_c5(b)
    curr_b = b.copy()
    curr_c5 = best_c5
    corr = np.correlate(curr_b, 1.0 - curr_b, mode='full')
    
    T = 0.005
    T_min = 1e-5
    stagnation = 0
    it = 0
    
    while True:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        
        it += 1
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        C_max = corr[max_idx]
        
        if C_max == 0:
            break
        
        # Generate candidates: positions to flip 0→1 that reduce C(k*)
        ones = np.where(curr_b == 1)[0]
        zeros = np.where(curr_b == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if k_star > 0:
            # C(k*) = sum_{j=0}^{n-1-k*} b[j+k*](1-b)[j]
            # Bad pairs: j where b[j+k*]=1, b[j]=0
            # Flipping b[j] 0→1 reduces C(k*) by b[j+k*] = 1
            valid_j = np.arange(0, n - k_star)
            bad_j = valid_j[(curr_b[valid_j + k_star] == 1) & (curr_b[valid_j] == 0)]
            
            if len(bad_j) > 0:
                # Try up to 5 random bad positions
                n_try = min(5, len(bad_j))
                chosen = rng.choice(bad_j, size=n_try, replace=False)
                best_gain = 0
                best_fp = None
                for fp in chosen:
                    # fp is j, flip curr_b[fp] 0→1
                    # Reduction in C(k*): the term h[fp+k*]*(1-h)[fp] goes from 1 to 0
                    gain = 1.0  # always 1 for binary
                    if gain > best_gain:
                        best_gain = gain
                        best_fp = fp
                # All have same gain, just pick one
                best_fp = chosen[0]
            else:
                best_fp = zeros[rng.randint(len(zeros))]
        elif k_star < 0:
            abs_k = -k_star

            valid_m = np.arange(0, n - abs_k)
            bad_m = valid_m[(curr_b[valid_m] == 1) & (curr_b[valid_m + abs_k] == 0)]
