    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:

        else:

    if n is None:
        n = len(h)
    s = abs(int(shift))
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

    n = len(h)

    mc = corr.max()

- h[i] appears in out[n-1+s] when j=i and i+s is valid: contribution δ*(1-h[i+s])
- h[i] appears in out[n-1+s] when j+s=i (j=i-s) and i-s is valid: contribution -h[i-s]*δ

So Δc(s) = δ * [(1-h[i+s]) - h[i-s]] where out-of-range terms are 0.

For a swap (δ_i at position i, -δ_i at position j):
Δc(s) = δ_i * [(1-h[i+s]) - h[i-s]] - δ_i * [(1-h[j+s]) - h[j-s]]
       = δ_i * [(h[j+s] - h[i+s]) + (h[i-s] - h[j-s])]

I can precompute A_i(s) = (1-h[i+s]) - h[i-s] for all i and s. But that's an n×(2n-1) matrix, which for n=960 is about 1.8M values. That's fine to store.

Then for a swap (i, j, δ):
Δc(s) = δ * (A_i(s) - A_j(s)) for all s.

The new max is max_s (c(s) + δ*(A_i(s) - A_j(s))).

I can precompute c(s) and A_i(s) - A_j(s) for the specific pair, then compute the new max in O(n).

But A_i(s) depends on h which changes with each accepted swap. So I need to update A after each swap. Updating A for position i (change h[i] by δ) affects:
- A_i(s) for all s: A_i(s) = (1-h[i+s]) - h[i-s], changing h[i] doesn't directly change A_i(s) unless i+s = i or i-s = i, i.e., s=0.
  Wait, A_i(s) = (1-h[i+s]) - h[i-s]. Changing h[i] affects this only when i+s = i (s=0) or i-s = i (s=0). So only A_i(0) changes.
  
  But wait, changing h[i] also affects A_k(s) for other k! A_k(s) = (1-h[k+s]) - h[k-s]. This changes when k+s = i or k-s = i.
  - k+s = i → k = i-s
  - k-s = i → k = i+s
  
  So changing h[i] by δ affects A_{i-s}(s) and A_{i+s}(s) for all s. That's O(n) entries.

This is getting complex. Let me just recompute the full correlation after each accepted swap, and for candidate swaps, use the incremental formula.

For the binary case with n points and n/2 ones:
- The overlap at shift 0 is: n/2 - n/2 = 0... wait no.
  c(0) = ∑ h[i](1-h[i]) = ∑ h[i] - ∑ h[i]². For binary: = n/2 - n/2 = 0.
  Wait, that's because h[i](1-h[i]) = 0 for binary h.
  
  Oh! For binary h, c(0) = 0! That's a key insight.
  For non-binary h, c(0) = ∑ h[i] - ∑ h[i]² = n/2 - ∑h[i]².
  This is maximized when h[i] = 1/2 for all i (giving c(0) = n/4), and minimized (to 0) when h is binary.

So the shift-0 overlap is 0 for binary solutions. This means binary solutions automatically have zero overlap at shift 0, which is good.

For shift s ≠ 0: c(s) = ∑ h[i](1-h[i+s]). For binary h, this counts the number of "1 followed by 0" at offset s.

So the problem for binary h is: minimize max_s (number of "1 at position i, 0 at position i+s").

This is a combinatorial optimization problem. For the Thue-Morse sequence, the autocorrelation has a specific structure.

Let me check: for Thue-Morse of length n=2^m, the number of positions i where t[i]=1 and t[i+s]=0 is:
- For s=0: 0 (as computed)
- For other s: depends on the structure

The Thue-Morse sequence has the property that for any s, the number of 1s in the first n-s positions that are followed by 0 at offset s is at most (n-s)/2 + O(√n). This gives a relatively flat autocorrelation.

But can we do better than Thue-Morse? The problem is to arrange n/2 ones and n/2 zeros in a circle (or line) to minimize the maximum "1-0" count at any shift.

For a line (not circle), the constraint is slightly different because the overlap region shrinks for large shifts.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def fista_focal(h, n_iter, top_k, lr0, lr1, check_interval=50):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    dx = 2.0 / n
    bc = compute_c5(x)
    bh = x.copy()
    
    # Initialize top shifts
    corr = np.correlate(z, 1.0 - z, mode='full')
    idx_top = np.argsort(corr)[::-1][:top_k]
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        lr = lr0 * (lr1 / lr0) ** f
        
        # Update top shifts periodically
        if (it + 1) % 20 == 0:
            corr = np.correlate(z, 1.0 - z, mode='full')
            idx_top = np.argsort(corr)[::-1][:top_k]
        
        # Compute weighted gradient over top shifts
        g = np.zeros(n)
        for k in idx_top:
            s = k - (n - 1)
            ss = abs(s)
            if ss >= n:
                continue
            if s >= 0:
                g[ss:] += 1.0 - z[:n-ss]
                g[:n-ss] -= z[ss:]
            else:
                g[:n-ss] += 1.0 - z[ss:]
                g[ss:] -= z[:n-ss]
        g *= dx / len(idx_top)
        
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        x = xn
        tk = tn
        
        if (it + 1) % check_interval == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def swap_descent(h, n_trials, rng):
    n = len(h)
    x = h.astype(float).copy()
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = float(corr.max())
    
    for _ in range(n_trials):
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        oi, oj = x[i], x[j]
        found = False
        for delta in [0.5, 0.25, 0.1, 0.05]:
            for sign in [1, -1]:
                di = sign * delta
                ni = oi + di
                nj = oj - di
                if ni < -1e-9 or ni > 1+1e-9 or nj < -1e-9 or nj > 1+1e-9:
                    continue
                ni = np.clip(ni, 0, 1)
                nj = np.clip(nj, 0, 1)
                if abs(ni - oi) < 1e-10 and abs(nj - oj) < 1e-10:
                    continue
                x[i], x[j] = ni, nj
                new_mc = float(np.max(np.correlate(x, 1.0 - x, mode='full')))
                if new_mc < mc - 1e-10:
                    mc = new_mc
                    found = True
                    break
                x[i], x[j] = oi, oj
            if found:
                break
    return x, float(mc * 2.0 / n)

Let me use the incremental approach for the check, and only recompute fully after accepting:

For a swap (i→i+di, j→j-di), the change in c(s) for each shift s:
Δc(s) = di * [(1-h[i+s]) - h[i-s]] - di * [(1-h[j+s]) - h[j-s]]

But computing this for all s is O(n). And checking if max(c + Δc) < mc is also O(n). So each trial is O(n) instead of O(n²).

For n=960, O(n) = 960 ops per trial. 10000 trials * 960 = 9.6M ops. Very fast!

But I need to handle the boundary conditions correctly. Let me define:
For shift s (ranging from -(n-1) to n-1):
c(s) = ∑_{j: max(0,-s) ≤ j < min(n, n-s)} h[j] * (1-h[j+s])

When I change h[i] by δ:
- h[i] appears as h[j] when j=i, contributing δ*(1-h[i+s]) if 0≤i<n and 0≤i+s<n
- h[i] appears as h[j+s] when j=i-s, contributing -h[i-s]*δ if 0≤i-s<n and 0≤i<n

So Δc(s) from changing h[i] by δ:
= δ * (1-h[i+s]) * [0≤i<n and 0≤i+s<n] - δ * h[i-s] * [0≤i-s<n and 0≤i<n]
= δ * [(1-h[i+s]) * I_valid(i, i+s) - h[i-s] * I_valid(i-s, i)]

Where I_valid(a, b) = 1 if 0≤a<n and 0≤b<n.

Since i is always in [0,n-1], I_valid(i, i+s) = 1 if 0≤i+s<n, and I_valid(i-s, i) = 1 if 0≤i-s<n.

So Δc(s) = δ * [(1-h[i+s]) * [0≤i+s<n] - h[i-s] * [0≤i-s<n]]

For a swap (i gets +δ, j gets -δ):
Δc(s) = δ * [(1-h[i+s])*[valid] - h[i-s]*[valid] - (1-h[j+s])*[valid] + h[j-s]*[valid]]
       = δ * [(h[j+s]-h[i+s])*[valid_i][valid_j_for_first] ... ]

This is getting complicated with the indicators. Let me just precompute arrays.

Define for position p and shift s:
A(p, s) = (1-h[p+s]) if 0≤p+s<n, else 0
B(p, s) = h[p-s] if 0≤p-s<n, else 0

Then Δc(s) from changing h[p] by δ = δ * (A(p,s) - B(p,s))

For swap (i: +δ, j: -δ):
Δc(s) = δ * (A(i,s) - B(i,s) - A(j,s) + B(j,s))

I can precompute A and B as n × (2n-1) arrays (indexed by position and shift+1 to handle negative shifts).

But updating A and B after each accepted swap is O(n) (only the row for i and j change).

Actually wait, A(p,s) depends on h[p+s], and B(p,s) depends on h[p-s]. So changing h[k] affects:
- A(p,s) when p+s=k, i.e., A(k-s, s) for all s → all of row k-s... no, it's column s of row k-s.
  
Hmm, this is a 2D structure. Let me think differently.

Actually, the simplest efficient approach: 
- Precompute corr = np.correlate(h, 1-h, 'full') once
- For a proposed swap, compute Δc(s) for all s in O(n) using the formula
- Check if max(corr + Δc) < mc
- If yes, update h and recompute corr (O(n²) but only for accepted swaps)

The key insight is that we only pay O(n²) for accepted swaps, and O(n) for rejected ones. If most swaps are rejected (which they should be in later stages), this is much faster.

def swap_descent_fast(h, n_trials, rng, mc_init=None):
    n = len(h)
    x = h.astype(float).copy()
    corr = np.correlate(x, 1.0 - x, mode='full')
    if mc_init is not None:
        mc = mc_init
    else:
        mc = float(corr.max())
    
    # Precompute helpers for O(n) incremental check
    # For position p, the "influence vector" over shifts:
    # V_p(s) = (1-x[p+s])*[0≤p+s<n] - x[p-s]*[0≤p-s<n]
    # We compute this on the fly for specific pairs
    
    for _ in range(n_trials):
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        oi, oj = x[i], x[j]
        for delta in [0.5, 0.25, 0.1, 0.05, 0.02]:
            for sign in [1, -1]:
                di = sign * delta
                ni = oi + di
                nj = oj - di
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                ni = np.clip(ni, 0, 1)
                nj = np.clip(nj, 0, 1)
                
                # Compute Δc(s) for all shifts s
                # s ranges from -(n-1) to n-1, indexed as 0 to 2n-2 in corr
                # Δc(s) = di * V_i(s) - di * V_j(s)
                # where V_p(s) = (1-x[p+s])*[0≤p+s<n] - x[p-s]*[0≤p-s<n]
                
                # For V_i(s): nonzero when 0≤i+s<n (first term) or 0≤i-s<n (second term)
                # i+s in [0,n-1] → s in [-i, n-1-i]
                # i-s in [0,n-1] → s in [i-(n-1), i]
                # Combined: s in [max(-(n-1), -i, i-(n-1)), min(n-1, n-1-i, i)]
                # = s in [-(n-1), n-1] ∩ [-i, n-1-i] ∪ [i-(n-1), i]
                # Hmm, let me just compute it vectorially
                
                # V_i(s) for s = -(n-1), ..., n-1:
                # First part: (1-x[i+s]) for s where 0≤i+s<n → s in [-i, n-1-i]
                # Second part: -x[i-s] for s where 0≤i-s<n → s in [i-(n-1), i]
                
                # Since s ranges from -(n-1) to n-1, and i in [0,n-1]:
                # -i ≥ -(n-1) and n-1-i ≤ n-1, so first part range is always valid
                # i-(n-1) ≥ -(n-1) and i ≤ n-1, so second part range is always valid
                
                vi = np.zeros(2*n-1)
                si_lo = -i  # s such that i+s=0
                si_hi = n-1-i  # s such that i+s=n-1
                # Map s to index: idx = s + (n-1)
                vi[si_lo + n - 1 : si_hi + n] = 1.0 - x[i + np.arange(si_lo, si_hi+1)]
                
                sj_lo = i - (n-1)  # s such that i-s=n-1
                sj_hi = i  # s such that i-s=0
                vi[sj_lo + n - 1 : sj_hi + n] -= x[i - np.arange(sj_lo, sj_hi+1)]
                
                # Similarly for V_j
                vj = np.zeros(2*n-1)
                sj_lo2 = -j
                sj_hi2 = n-1-j
                vj[sj_lo2 + n - 1 : sj_hi2 + n] = 1.0 - x[j + np.arange(sj_lo2, sj_hi2+1)]
                sj_lo3 = j - (n-1)
                sj_hi3 = j
                vj[sj_lo3 + n - 1 : sj_hi3 + n] -= x[j - np.arange(sj_lo3, sj_hi3+1)]
                
                dc = di * (vi - vj)
                new_mc = float(np.max(corr + dc))
                
                if new_mc < mc - 1e-12:
                    x[i], x[j] = ni, nj
                    corr = np.correlate(x, 1.0 - x, mode='full')
                    mc = float(corr.max())
                    break
            else:
                continue
            break
    
    return x, float(mc * 2.0 / n)

def swap_improve(h, n_trials, rng):
    """Try random swaps to reduce max overlap."""
    n = len(h)
    x = h.astype(float).copy()
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = float(corr.max())
    dx = 2.0 / n
    
    accepted = 0
    for _ in range(n_trials):
        i, j = rng.randint(0, n, 2)
        if i == j:

        oi, oj = x[i], x[j]
        for delta in [0.5, 0.2, 0.1, 0.05]:

                nj = oj - di

                ni = min(max(ni, 0.0), 1.0)

                    corr = np.correlate(x, 1.0 - x, mode='full')

                    break

        return time.time() - t0
    def remaining():
        return budget_s - elapsed()

        print(f"S2: C5={bc:.6f} acc={acc} t={elapsed():.0f}", flush=True)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def fista_top(h, n_iter, top_k, lr0, lr1, check_iv=25):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    dx = 2.0 / n
    bc = compute_c5(x)
    bh = x.copy()
    corr_z = np.correlate(z, 1.0 - z, mode='full')
    idx_top = np.argsort(corr_z)[::-1][:top_k]
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        lr = lr0 * (lr1 / lr0) ** f
        if (it + 1) % 12 == 0:
            corr_z = np.correlate(z, 1.0 - z, mode='full')
            idx_top = np.argsort(corr_z)[::-1][:top_k]
        g = np.zeros(n)
        for k in idx_top:
            s = k - (n - 1)
            ss = abs(s)
            if ss >= n:

            if s >= 0:

            else:

                continue
            if s >= 0:

                i_zero = ss + iprod

def slsqp_smooth(h, T=5.0, mi=400):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    def _obj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = c.max()
        sm = mc + np.log(np.sum(np.exp(T * (c - mc)))) / T
        return sm * dx
    def _con(hv):
        return np.sum(hv) - target
    def _con_jac(hv):
        return np.ones(n)
    try:
        r = minimize(_obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con, 'jac': _con_jac}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = project_c(tm_seq(512), 512)

    n0 = len(h0)
    bc = compute_c5(h0)
    bh = h0.copy()
    print(f"I: n={n0} C5={bc:.6f}", flush=True)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Focal FISTA on initial
    if rem() > 250:
        h1, c1 = fista_top(bh, 8000, 30, 0.12, 0.004, 25)
        if c1 < bc:
            bc, bh = c1, h1.copy()
        print(f"F1: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 2: Fast swap improvement
    if rem() > 80:
        nt = min(50000, int(rem() * 1500))
        h2, c2, acc = swap_improve_fast(bh, nt, rng)
        if c2 < bc:
            bc, bh = c2, h2.copy()
        print(f"S1: C5={bc:.6f} acc={acc} t={time.time()-t0:.0f}", flush=True)

    # Phase 3: Tighter FISTA (fewer top shifts)
    if rem() > 200:
        h3, c3 = fista_top(bh, 7000, 15, 0.08, 0.002, 20)
        if c3 < bc:
            bc, bh = c3, h3.copy()
        print(f"F2: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 4: More swaps
    if rem() > 60:
        nt = min(50000, int(rem() * 1500))
        h4, c4, acc = swap_improve_fast(bh, nt, rng)
        if c4 < bc:
            bc, bh = c4, h4.copy()
        print(f"S2: C5={bc:.6f} acc={acc} t={time.time()-t0:.0f}", flush=True)

    # Phase 5: Guided swaps targeting worst shifts
    if rem() > 60:
        h5, c5, acc = guided_swap(bh, 300, rng)
        if c5 < bc:
            bc, bh = c5, h5.copy()
        print(f"G1: C5={bc:.6f} acc={acc} t={time.time()-t0:.0f}", flush=True)

    # Phase 6: SLSQP polish at current res if small enough
    if len(bh) <= 512 and rem() > 40:
        bs = slsqp_smooth(bh, 4.0, 350)
        cs = compute_c5(bs)
        if cs < bc:
            bc, bh = cs, bs
        print(f"SL: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 7: Explore lower resolutions with Thue-Morse and random starts
    for n_try in [256, 200, 160, 128]:
        if rem() < 50:
            break
        starts = [project_c(tm_seq(n_try), n_try), project_c(rng.rand(n_try), n_try)]
        best_low = None
        best_low_c = float('inf')
        for hs in starts:
            if rem() < 20:
                break
            ho, co = fista_top(hs, 4000, 20, 0.1, 0.004, 25)
            if rem() > 15:
                ho, co2, _ = swap_improve_fast(ho, 10000, rng)
                if co2 < co:
                    co = co2
            if co < best_low_c:
                best_low_c, best_low = co, ho.copy()
        if best_low is not None and best_low_c < bc:
            bc, bh = best_low_c, best_low
            n0 = n_try
            print(f"  LR n={n_try}: C5={bc:.6f}", flush=True)
        print(f"  LR n={n_try}: best={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 8: Upscale if we found better at lower res
    if rem() > 120 and len(bh) < 960:
        target_n = 960
        hu = upsample(bh, target_n)
        cu = compute_c5(hu)
        if rem() > 80:
            hu, cu2 = fista_top(hu, 5000, 25, 0.08, 0.003, 25)
            if cu2 < cu:
                cu = cu2
        if rem() > 30:
            hu, cu3, _ = swap_improve_fast(hu, 15000, rng)
            if cu3 < cu:
                cu = cu3
        if cu < bc:
            bc, bh = cu, hu.copy()
        print(f"UP960: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 9: Final ultra-fine FISTA + swaps
    if rem() > 30:
        h9, c9 = fista_top(bh, 4000, 10, 0.03, 0.001, 20)
        if c9 < bc:
            bc, bh = c9, h9.copy()
    if rem() > 15:
        bh, c10, _ = swap_improve_fast(bh, min(20000, int(rem() * 2000)), rng)
        if c10 < bc:
            bc = c10

    # Final projection and evaluate
    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
