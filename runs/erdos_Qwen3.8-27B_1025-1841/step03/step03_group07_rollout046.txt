import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)


def smooth_grad(h, n, tau, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    max_c = top_vals.max()
    shifted = np.clip(tau * (top_vals - max_c), -50.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)

    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        pk = p[ii]
        k = top_idx[ii]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    return obj, grad

    if rem > 40:
        n_u = len(best_h)
        tl = min(35, rem * 0.12)
        h1, c5 = adam_opt(best_h, n_u, tau=5.0, time_limit=tl, lr=5e-5, top_k=3, binary_str=0.05)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 8: Binary snap + local search
    rem = get_remaining(st, budget_s)
    if rem > 50:
        n_u = len(best_h)
        h_b, c5_b = binary_local(best_h, n_u, time_limit=min(40, rem * 0.5), n_trials=300)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b.copy()

    # Phase 9: Final Adam polish from best
    rem = get_remaining(st, budget_s)
    if rem > 25:
        n_u = len(best_h)
        tl = min(22, rem * 0.7)
        h1, c5 = adam_opt(best_h, n_u, tau=8.0, time_limit=tl, lr=4e-5, top_k=3, binary_str=0.03)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Final projection
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
