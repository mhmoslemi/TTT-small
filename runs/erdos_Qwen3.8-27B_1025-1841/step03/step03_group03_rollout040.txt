def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-14:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def golden_section_corr(affected, d, m_out, lo, hi, max_iter=50):
    gr = 0.6180339887498949
    a, b = lo, hi
    c = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = max(m_out, np.max(affected + c * d))
    fd = max(m_out, np.max(affected + d_pt * d))
    for _ in range(max_iter):
        if fc < fd:
            b = d_pt
            d_pt = c
            fd = fc
            c = b - gr * (b - a)
            fc = max(m_out, np.max(affected + c * d))
        else:
            a = c
            c = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = max(m_out, np.max(affected + d_pt * d))
    mid = (a + b) / 2.0

    return mid, fmid

    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(t * shifted))) / t) * dx
