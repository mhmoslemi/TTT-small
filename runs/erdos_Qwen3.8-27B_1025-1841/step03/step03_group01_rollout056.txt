def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # actual overlap values

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_feasible(h, n):

    target = n / 2.0

        else:

        if cnt == 0:

    """Solve linearized QP at h_star.
    Returns new h that minimizes the linearized max overlap.
    """
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    if max_t is not None:
        constraints.append(t_var <= max_t)
    
    # For each top lag, add linearized constraint
    for lag_idx in top_lags:
        tau = lag_idx - (n - 1)
        # Build coefficient vector for the linearized overlap
        coeff = np.zeros(n)
        if tau >= 0:

            # Linearized: sum_{i=0}^{n-1-tau} [(1-h*[i+tau])*h[i] - h*[i]*h[i+tau]]
            for i in range(n - tau):
                coeff[i] += (1.0 - h_star[i + tau])
                coeff[i + tau] -= h_star[i]
        else:
            # tau < 0
            for i in range(-tau, n):
                coeff[i] += (1.0 - h_star[i + tau])
                coeff[i + tau] -= h_star[i]
        
        constraints.append(cp.sum(coeff * h_var) * dx <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000)
    except Exception:
        try:
            prob.solve(solver=cp.CVXOPT)
        except Exception:
            return None
    
    if h_var.value is None:
        return None
    
    h_new = np.clip(h_var.value, 0.0, 1.0)
    return make_feasible(h_new, n)
