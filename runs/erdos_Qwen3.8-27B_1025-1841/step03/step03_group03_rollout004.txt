def block_coord_descent(h, n, block_size=4, max_rounds=30, time_limit=None, start_time=None):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def subgrad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = np.argmax(corr)
    lag = k_idx - (n - 1)
    p = abs(lag)
    if p < n:
        grad = np.zeros(n)
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] += -h[:n-p]
    else:
        grad = np.zeros(n)
    return float(corr[k_idx] * dx), grad * dx

    h = h_flat
    val, grad = subgrad_c5(h)
    s = np.sum(h) - n / 2.0
    val += lam * s * s
    grad = grad + 2.0 * lam * s
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n * 0.5
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = m.sum()
        if c == 0:
            break
        h[m] = np.clip(h[m] + d / c, 0.0, 1.0)
    return h

def subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ki = np.argmax(corr)
    lag = ki - (n - 1)
    p = abs(lag)
    g = np.zeros(n)
    if p < n:
        g[:n-p] += 1.0 - h[p:]
        g[p:] += -h[:n-p]
    return float(corr[ki] * dx), g * dx

def fg(x, n, lam):
    h = x
    v, g = subgrad(h)
    s = h.sum() - 0.5 * n
    return v + lam * s * s, g + 2.0 * lam * s

def lbfgs(h, n, iters, lam=100.0):
    b = [(0.0, 1.0)] * n
    r = minimize(fg, h, jac=True, args=(n, lam), method='L-BFGS-B',
                 bounds=b, options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-9})
    return make_feasible(np.clip(r.x, 0, 1), n)
