    if time.time() - start_time < budget_s - 300:
        h_new, c_new = stochastic_subgrad(best_h, n1, n_iter=300,
                                           time_limit=budget_s, start_time=start_time, rng=rng)
        if c_new < best_c - 1e-11:
            best_c = c_new
            best_h = h_new

    # Phase 3: Coordinate descent
    best_h, best_c = coord_descent(best_h, n1, max_rounds=4,
                                    time_limit=budget_s, start_time=start_time)

    # Phase 4: Binary projection + refinement
    if time.time() - start_time < budget_s - 120:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n1)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
        best_h, best_c = coord_descent(best_h, n1, max_rounds=3,
                                        time_limit=budget_s, start_time=start_time)

    # Phase 5: Higher resolution (4096) if time permits
    if time.time() - start_time < budget_s - 150:
        n2 = 4096
        h2 = upsample_h(best_h, n2)
        for t in [20, 80, 200]:
            if time.time() - start_time > budget_s - 80:
                break
            h_new, c_new = projected_gradient_step(h2, n2, t, n_iter=100,
                                                     time_limit=budget_s, start_time=start_time)
            if c_new < best_c - 1e-11:
                best_c = c_new
                best_h = h_new
                h2 = h_new
            else:
                h2 = h_new
        best_h, best_c = coord_descent(best_h, n2, max_rounds=2,
                                        time_limit=budget_s, start_time=start_time)

    # Phase 6: Try SLSQP at moderate resolution for local polish
    if time.time() - start_time < budget_s - 80:
        n3 = 640
        h3 = upsample_h(best_h, n3)
        def obj_func(x):
            xx = project_feasible(x, n3)
            return compute_c5(xx)
        bounds = [(0.0, 1.0)] * n3
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n3 / 2.0,
                 'jac': lambda x: np.ones(n3)}]
        try:
            res = minimize(obj_func, h3, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
            h3f = project_feasible(res.x, n3)
            c3f = compute_c5(h3f)
            if c3f < best_c - 1e-11:
                best_c = c3f
                best_h = upsample_h(h3f, len(best_h))
        except Exception:
            pass

    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
