    Returns array of length 2n-1 (indexed by full correlation index)."""

    for d in range(-(n-1), n):

        idx = n - 1 + d
        val = 0.0

        val = 0.0
        if 0 <= j + d < n:
            val -= h[j + d]
        if 0 <= j - d < n:
            val += (1.0 - h[j - d])
        delta[idx] += val
    
    return delta
