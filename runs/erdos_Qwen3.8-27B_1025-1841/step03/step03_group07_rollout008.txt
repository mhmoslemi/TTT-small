    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)[:2*n-1]

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 3.0
    hi = g.max() + 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr


def soft_max_grad(h, n, tau, top_k=8):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if top_k < nc:
        top_idx = np.argpartition(corr, nc - top_k)[nc - top_k:]
    else:
        top_idx = np.arange(nc)
    top_vals = corr[top_idx]
    m = top_vals.max()
    sh = (top_vals - m) * tau
    sh -= sh.max()
    exp_s = np.exp(sh)
    s = exp_s.sum()
    p = exp_s / s
    obj = dx * (m + np.log(s) / tau)
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        w = p[ii] * dx
        k = top_idx[ii]
        L = k - n + 1
        j_lo = max(0, -L)
        j_hi = min(n - 1, n - 1 - L)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo + L:j_hi + 1 + L])
        j2_lo = max(0, -L)
        j2_hi = min(n - 1, n - 1 - L)
        if j2_lo <= j2_hi:
            jp = j2_lo + L
            grad[j2_lo + L:j2_hi + 1 + L] -= w * h[j2_lo:j2_hi + 1]
    return obj, grad


def nesterov_adapt(h, n, tau_start, tau_end, time_limit, top_k=10, lr_scale=0.35):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    h_prev = h.copy()
    tau = tau_start
    t = 1.0
    stall = 0
    lr = lr_scale / tau_start

    while time.time() - t0 < time_limit:
        rem = time_limit - (time.time() - t0)
        frac = 1.0 - rem / time_limit
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_scale / max(tau, 1.0)

        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.95)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        obj, grad = soft_max_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)

        if int(t) % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-16:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 4000:
                lr *= 0.7
                stall = 0
                if lr < 1e-14:
                    break

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)

        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def subgrad_refine(h, n, time_limit, lr=0.001, top_k=4):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    vx = np.zeros(n)
    beta = 0.85
    stall = 0

    while time.time() - t0 < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * (2.0 / n)
        if c5_now < best_c5 - 1e-16:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            lr *= 0.72
            vx *= 0.4
            stall = 0
            if lr < 1e-15:
                break

        nc = len(corr)
        if top_k < nc:
            top_idx = np.argpartition(corr, nc - top_k)[nc - top_k:]
        else:
            top_idx = np.arange(nc)
        top_vals = corr[top_idx]
        w_exp = np.exp(top_vals - top_vals[0])
        w_exp /= w_exp.sum()

        dx = 2.0 / n
        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = w_exp[ii] * dx
            k = top_idx[ii]

            j2_hi = min(n - 1, n - 1 - L)

        h = project_to_feasible(h + vx, n)

        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def local_swap_search(h, n, time_limit, n_trials=200000):

    best_c5 = compute_c5(h)

        # Want this negative: (1-h[j+L*]) - (1-h[j'+L*]) < h[j-L*] - h[j'-L*]

            if not candidate_mask.any():

            jp = rng.choice(np.where(candidate_mask)[0])

def interp_to_n(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def structured_init(n, variant=0):

    if variant == 0:
        h = np.zeros(n)
        h[x < 1.0] = 1.0
    elif variant == 1:
        h = 0.5 + 0.4 * np.sin(np.pi * x)

    else:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
