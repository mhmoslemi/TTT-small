def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -80, 80)
    val = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    return val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def binarize(h, n, rng):
    hb = (h > 0.5).astype(float)

def softmax_objective(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    shifted = np.clip(T * (corr - mc), -80, 80)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def lbgfb_opt(h_start, n, T, mu, max_iter=300):
    def f(h):
        v = softmax_objective(h, n, T)
        viol = np.sum(h) - n / 2.0
        return v + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)
