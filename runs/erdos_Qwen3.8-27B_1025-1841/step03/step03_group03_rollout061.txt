def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def fast_coord_descent(h, n, max_rounds=40, time_limit=None, start_time=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c = np.max(corr) * dx
    best_h = h.copy()
    cands = np.array([0.0, 0.15, 0.25, 0.4, 0.5, 0.6, 0.75, 0.85, 1.0])

    for rnd in range(max_rounds):
        if time_limit and start_time and time.time() - start_time > time_limit - 30:
            break
        improved = False
        for j in range(n):
            old = best_h[j]
            best_val = old
            best_mc = np.max(corr)
            best_corr = corr

            for cv in cands:
                if abs(cv - old) < 1e-10:
                    continue
                delta = cv - old
                m_lo1 = n - 1 - j
                m_hi1 = 2 * n - 2 - j
                m_lo2 = max(0, j - n + 1)
                m_hi2 = min(2 * n - 2, j + n - 1)
                m_lo = m_lo1 if m_lo1 < m_lo2 else m_lo2
                m_hi = m_hi1 if m_hi1 > m_hi2 else m_hi2
                m_range = np.arange(m_lo, m_hi + 1)
                shifts = m_range - n + 1
                jp = j + shifts
                jm = j - shifts
                mask1 = (jp >= 0) & (jp < n)
                mask2 = (jm >= 0) & (jm < n)
                term1 = np.where(mask1, delta * (1.0 - best_h[np.clip(jp, 0, n-1)]), 0.0)
                term2 = np.where(mask2, delta * best_h[np.clip(jm, 0, n-1)], 0.0)
                new_seg = corr[m_lo:m_hi+1] + term1 - term2
                mc = max(np.max(new_seg),
                         (np.max(corr[:m_lo]) if m_lo > 0 else -1),
                         (np.max(corr[m_hi+1:]) if m_hi < 2*n-2 else -1))
                if mc < best_mc - 1e-13:
                    best_mc = mc
                    best_val = cv
                    best_corr = corr.copy()
                    best_corr[m_lo:m_hi+1] += term1 - term2

            if best_val != old:
                best_h[j] = best_val
                corr = best_corr
                improved = True
                nc = np.max(corr) * dx
                if nc < best_c:
                    best_c = nc
        if not improved:
            break
    return make_feasible(best_h, n), best_c

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        n_init = 640
        h0 = (rng.rand(n_init) < 0.5).astype(float)
        h0 = make_feasible(h0, n_init)

    n1 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradient, increasing t
    for t_val, mi in [(20, 400), (50, 500), (150, 500), (400, 600), (1000, 500)]:
        if time.time() - t0 > budget_s - 400:
            break
        h_new = slsqp_opt(best_h, n1, t_val, maxiter=mi)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-9:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Coordinate descent at n1
    if time.time() - t0 < budget_s - 250:
        h_cd, c_cd = fast_coord_descent(best_h, n1, max_rounds=60,
                                        time_limit=budget_s, start_time=t0)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 3: Upsample and optimize at higher resolution
    n2 = 1280
    if time.time() - t0 < budget_s - 250:
        h2 = upsample_h(best_h, n2)
        for t_val, mi in [(40, 400), (120, 500), (400, 500)]:
            if time.time() - t0 > budget_s - 180:
                break
            h_new = slsqp_opt(h2, n2, t_val, maxiter=mi)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-9:
                best_c5 = c_new
                best_h = h_new
                h2 = h_new
            else:
                h2 = h_new
        if time.time() - t0 < budget_s - 130:
            h_cd2, c_cd2 = fast_coord_descent(h2, n2, max_rounds=30,
                                              time_limit=budget_s, start_time=t0)
            if c_cd2 < best_c5:
                best_c5 = c_cd2
                best_h = h_cd2

    # Phase 4: Random restart at n1
    if time.time() - t0 < budget_s - 120:
        h_rand = rng.rand(n1)
        h_rand = make_feasible(h_rand, n1)
        h_new = slsqp_opt(h_rand, n1, 80, maxiter=400)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-9:
            best_c5 = c_new
            best_h = h_new
            h_cd3, c_cd3 = fast_coord_descent(best_h, n1, max_rounds=30,
                                              time_limit=budget_s, start_time=t0)
            if c_cd3 < best_c5:
                best_c5 = c_cd3
                best_h = h_cd3

    # Phase 5: Another structured restart (block pattern)
    if time.time() - t0 < budget_s - 100:
        n_blk = 640
        block_size = 8
        n_blocks = n_blk // block_size
        pattern = rng.permutation(n_blocks)
        vals = np.zeros(n_blk)
        for bi in range(n_blocks):
            if bi < n_blocks // 2:
                vals[bi*block_size:(bi+1)*block_size] = 1.0
        h_blk = vals.copy()
        h_blk = make_feasible(h_blk, n_blk)
        h_new = slsqp_opt(h_blk, n_blk, 60, maxiter=500)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-9:
            best_c5 = c_new
            best_h = h_new
            h_cd4, c_cd4 = fast_coord_descent(best_h, n_blk, max_rounds=40,
                                              time_limit=budget_s, start_time=t0)
            if c_cd4 < best_c5:
                best_c5 = c_cd4
                best_h = h_cd4

    # Phase 6: Final refinement - coordinate descent on best solution
    if time.time() - t0 < budget_s - 40:
        n_fin = len(best_h)
        h_fin, c_fin = fast_coord_descent(best_h, n_fin, max_rounds=50,
                                          time_limit=budget_s, start_time=t0)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin

    # Phase 7: Binary threshold attempt
    if time.time() - t0 < budget_s - 20:
        n_bt = len(best_h)
        h_bt = (best_h > 0.5).astype(float)
        h_bt = make_feasible(h_bt, n_bt)
        c_bt = compute_c5(h_bt)
        if c_bt < best_c5 - 1e-9:
            best_c5 = c_bt
            best_h = h_bt
            h_cd5, c_cd5 = fast_coord_descent(best_h, n_bt, max_rounds=20,
                                              time_limit=budget_s, start_time=t0)
            if c_cd5 < best_c5:
                best_c5 = c_cd5
                best_h = h_cd5

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
