def project(h, n):
    # Project onto {h: 0<=h<=1, sum(h)=n/2}
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 1.0  # lambda range
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any numerical issues
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        # Distribute among non-bound elements
        mask = (proj > 1e-8) & (proj < 1 - 1e-8)
        if np.sum(mask) > 0:
            proj[mask] += diff / np.sum(mask)
            proj = np.clip(proj, 0, 1)
    return proj

def soft_gradient(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, 'full')  # length 2n-1
    # Softmax weights
    tc = t * corr * dx
    tc -= np.max(tc)  # for numerical stability
    w = np.exp(tc)
    w /= np.sum(w)
    
    # Compute gradient
    # grad[j] = dx * sum_k w[k] * [(1-h[j+s_k])*I[0<=j+s_k<n] - h[j-s_k]*I[0<=j-s_k<n]]
    # where s_k = k - (n-1)
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        wk = w[k]
        if wk < 1e-15:
            continue
        if s >= 0:
            # (1-h[j+s]) for j=0..n-1-s
            grad[:n-s] += wk * (1-h[s:])
            # -h[j-s] for j=s..n-1
            grad[s:] -= wk * h[:n-s]
        else:
            sp = -s
            # (1-h[j-sp]) for j=sp..n-1
            grad[sp:] += wk * (1-h[:n-sp])
            # -h[j+sp] for j=0..n-1-sp
            grad[:n-sp] -= wk * h[sp:]
    grad *= dx
    return grad, np.max(corr) * dx

def soft_gradient(h, n, t):
    """Compute gradient of soft-max objective (1/t)*log(sum exp(t*corr*dx))."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # Softmax weights (numerically stable)
    tc = t * corr * dx
    tc -= np.max(tc)
    w = np.exp(tc)

    # d(corr_k)/d(h_j) = (1-h[j+s_k])*I[0<=j+s_k<n] - h[j-s_k]*I[0<=j-s_k<n]

    dx = 2.0 / n
    best_c = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 8:
                break
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            if old <= 1e-10:
                cands = [0.0, 0.2, 0.5]
            elif old >= 1.0 - 1e-10:
                cands = [1.0, 0.8, 0.5]
            else:
                cands = [0.0, 1.0, max(0.0, old - 0.08), min(1.0, old + 0.08),
                         max(0.0, old - 0.03), min(1.0, old + 0.03)]
            best_mc = best_c / dx
            best_trial = None
            for cv in cands:
                if abs(cv - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                mc = np.max(np.correlate(trial, 1.0 - trial, mode='full'))
                if mc < best_mc:
                    best_mc = mc
                    best_trial = trial.copy()
            if best_trial is not None:
                h = best_trial
                best_c = best_mc * dx
                improved = True
        if not improved:
            break
    return h, best_c

def multi_temp_gradient(h, n, temps, max_iter, time_limit, start_time):
    h = h.copy().astype(float)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    it_per_temp = max_iter // len(temps)
    for t in temps:
        if time.time() - start_time > time_limit - 10:
            break
        alpha = min(5.0 / (n * t), 0.1)
        for it in range(it_per_temp):
            if time.time() - start_time > time_limit - 10:
                break
            grad, c_val = soft_gradient(h, n, t)
            if c_val < best_c - 1e-12:
                best_c = c_val
                best_h = h.copy()
            improved = False
            for _ in range(10):
                h_new = project(h - alpha * grad, n)
                c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
                if c_new < c_val - 1e-14:
                    h = h_new
                    alpha = min(alpha * 1.1, 0.15)
                    improved = True
                    break
                alpha *= 0.6
            if not improved:
                alpha *= 0.4
                if alpha < 1e-9:
                    break
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        n_init = 640
        h0 = (rng.rand(n_init) < 0.5).astype(float)
        h0 = make_feasible(h0, n_init)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    def elapsed():
        return time.time() - start_time
    
    # Phase 1: Multi-temp projected gradient at parent resolution
    n1 = len(best_h)
    if elapsed() < budget_s - 400:
        h_new, c_new = multi_temp_gradient(best_h, n1, [15, 40, 100, 300],
                                           800, budget_s, start_time)
        if c_new < best_c5 - 1e-10:
            best_c5 = c_new
            best_h = h_new
    
    # Phase 2: Coordinate descent refinement
    if elapsed() < budget_s - 250:
        h_cd, c_cd = fast_coord_descent(best_h, n1, max_rounds=25,
                                         time_limit=budget_s, start_time=start_time, rng=rng)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd
    
    # Phase 3: Upsample to 1280
    if elapsed() < budget_s - 200:
        n3 = 1280
        h3 = upsample_h(best_h, n3)
        h3, c3 = multi_temp_gradient(h3, n3, [20, 60, 200], 600, budget_s, start_time)
        if c3 < best_c5 - 1e-10:
            best_c5 = c3
            best_h = h3
        if elapsed() < budget_s - 120:
            h3, c3 = fast_coord_descent(best_h, min(n3, len(best_h)), max_rounds=8,
                                         time_limit=budget_s, start_time=start_time, rng=rng)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3
    
    # Phase 4: Upsample to 2048
    if elapsed() < budget_s - 150:
        n4 = 2048
        h4 = upsample_h(best_h, n4)
        h4, c4 = multi_temp_gradient(h4, n4, [30, 100, 300], 400, budget_s, start_time)
        if c4 < best_c5 - 1e-10:
            best_c5 = c4
            best_h = h4
        if elapsed() < budget_s - 60:
            h4, c4 = fast_coord_descent(best_h, len(best_h), max_rounds=4,
                                         time_limit=budget_s, start_time=start_time, rng=rng)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4
    
    # Phase 5: Random restarts at n=640
    if elapsed() < budget_s - 100:
        n_r = 640
        for restart in range(3):
            if elapsed() > budget_s - 50:
                break
            hr = rng.rand(n_r)
            hr = make_feasible(hr, n_r)
            hr, cr = multi_temp_gradient(hr, n_r, [20, 80, 250], 300, budget_s, start_time)
            if cr < best_c5 - 1e-10:
                best_c5 = cr
                best_h = hr
    
    # Phase 6: Final coordinate descent pass
    if elapsed() < budget_s - 20:
        h_f, c_f = fast_coord_descent(best_h, len(best_h), max_rounds=10,
                                       time_limit=budget_s, start_time=start_time, rng=rng)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    # Binary rounding attempt
    if elapsed() < budget_s - 15:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, len(best_h))
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5 - 1e-10:
            best_c5 = c_bin
            best_h = h_bin
            h_f2, c_f2 = fast_coord_descent(best_h, len(best_h), max_rounds=5,
                                             time_limit=budget_s, start_time=start_time, rng=rng)
            if c_f2 < best_c5:
                best_c5 = c_f2
                best_h = h_f2

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
