import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Spread the remaining diff
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        remaining = diff
        for idx in order:
            if diff > 0:
                add = min(remaining, 1.0 - h[idx])
            else:
                add = max(remaining, h[idx])
            h[idx] += add
            remaining -= add
            if abs(remaining) < 1e-14:
                break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def coord_descent(h, n, max_sweeps=5, tolerance=1e-10):
    """Exact coordinate descent with 1D minimax per coordinate."""
    c = np.array(np.correlate(h, 1.0 - h, mode='full'), dtype=float)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    best_c5 = float(np.max(c) * dx)
    offsets = np.arange(n_lags) - (n - 1)
    
    for sweep in range(max_sweeps):
        sweep_improvement = 0.0
        for j in range(n):
            # Compute B_k for coordinate j
            # B_k = 1 - h_clamped[j - offset_k] - h_clamped[j + offset_k]
            idx1 = j - offsets  # length n_lags
            idx2 = j + offsets  # length n_lags
            B = np.ones(n_lags)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            B[v1] -= h[idx1[v1]]
            B[v2] -= h[idx2[v2]]
            
            # A_k = c[k] - h[j] * B_k
            A = c - h[j] * B
            
            # 1D minimax: find t in [0,1] minimizing max_k(A_k + t*B_k)
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = A + mid * B
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-12
                b_act = B[active]
                if np.max(b_act) < -1e-12:
                    lo = mid
                elif np.min(b_act) > 1e-12:
                    hi = mid
                else:
                    break
            t_opt = 0.5 * (lo + hi)
            
            # Check if this improves C5
            new_c = c + (t_opt - h[j]) * B
            new_max = np.max(new_c) * dx
            if new_max < best_c5 - tolerance:
                h[j] = t_opt
                c = new_c
                best_c5 = new_max
                sweep_improvement += 1
            elif new_max < best_c5:
                # Small improvement, still accept
                h[j] = t_opt
                c = new_c
                best_c5 = new_max
                sweep_improvement += 1
        
        if sweep_improvement == 0:
            break
    
    return h, best_c5

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break

            break

                    t_opt = 0.0
                    lo = hi = 0.0
                else:
                    lo, hi = 0.0, 1.0
            else:

                    t_opt = 1.0
                    lo = hi = 1.0
                else:
                    lo, hi = 0.0, 1.0

                    vals = A + mid * B
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-10
                    b_act = B[active]
                    if len(b_act) == 0:
                        t_opt = mid
                        break
                    if np.max(b_act) <= 1e-10:
                        # All active B <= 0: f decreasing, go right
                        lo = mid
                    elif np.min(b_act) >= -1e-10:
                        # All active B >= 0: f increasing, go left
                        hi = mid
                    else:
                        # 0 in subdifferential
                        t_opt = mid
                        break
                else:
                    t_opt = 0.5 * (lo + hi)
            
            if abs(t_opt - h[j]) < 1e-12:
                continue
            
            new_c = c + (t_opt - h[j]) * B
            new_max = float(np.max(new_c) * dx)
            if new_max < best_c5 - 1e-12:
                h[j] = t_opt
                c = new_c
                best_c5 = new_max

            new_best = float(np.max(c) * dx)
            if new_best < best_c5:
                best_c5 = new_best
            else:
                best_c5 = new_best
        
        if not any_improve:
            break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _minimax_1d(A, B):
    lo, hi = 0.0, 1.0
    f_lo = np.max(A)
    f_hi = np.max(A + B)
    if f_lo <= f_hi:
        act = A >= f_lo - 1e-12
        if np.max(B[act]) <= 1e-12:
            return 0.0
    else:
        act = (A + B) >= f_hi - 1e-12
        if np.min(B[act]) >= -1e-12:
            return 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = A + mid * B
        fm = np.max(vals)
        act = vals >= fm - 1e-10
        ba = B[act]
        if len(ba) == 0:
            return mid
        if np.max(ba) <= 1e-10:
            lo = mid
        elif np.min(ba) >= -1e-10:
            hi = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def coord_descent(h, n, max_sweeps=5, time_limit=120.0, t0=None):
    h = h.copy()
    c = np.array(np.correlate(h, 1.0 - h, mode='full'), dtype=float)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    best_c5 = float(np.max(c) * dx)
    offsets = np.arange(n_lags) - (n - 1)
    if t0 is None:
        t0 = time.time()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        any_imp = False
        for j in range(n):
            idx1 = j - offsets
            idx2 = j + offsets
            B = np.ones(n_lags)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            B[v1] -= h[idx1[v1]]
            B[v2] -= h[idx2[v2]]
            A = c - h[j] * B
            t_opt = _minimax_1d(A, B)
            if abs(t_opt - h[j]) < 1e-13:
                continue
            new_c = c + (t_opt - h[j]) * B
            new_max = float(np.max(new_c) * dx)
            if new_max < best_c5:
                h[j] = t_opt
                c = new_c
                best_c5 = new_max
                any_imp = True
        h = make_feasible(h, n)
        c = np.array(np.correlate(h, 1.0 - h, mode='full'), dtype=float)
        best_c5 = float(np.max(c) * dx)
        if not any_imp:
            break
    return h, best_c5

def _slsqp_smooth(h, n, T, max_iter, time_limit):
    t0 = time.time()
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}],
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

    def time_left():

    return best_h, final_c5, len(best_h)
