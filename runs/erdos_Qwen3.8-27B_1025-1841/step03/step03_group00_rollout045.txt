def lbgfb_softmax(h_start, n, T, mu, max_iter):
    dx = 2.0 / n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = float(np.max(corr))
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80.0, 80.0)
        soft = mc + np.log(np.sum(np.exp(shifted))) / T
        viol = np.sum(h) - n / 2.0
        return (soft + mu * viol * viol) * dx

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def slsqp_polish(h_start, n, T, max_iter):
    dx = 2.0 / n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = float(np.max(corr))
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80.0, 80.0)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

    def con_eq(h):
        return np.sum(h) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': lambda h: np.ones(n)}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, n_rounds, n_tries_per_round, sa_temp=0):
    h = h_bin.copy()
    current = compute_c5(h)
    best_h = h.copy()
    best_c = current
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(n_rounds):
        improved = False
        for _ in range(n_tries_per_round):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi = ones[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            delta = c - current
            accept = delta < 0
            if not accept and sa_temp > 0:
                if rng.random() < np.exp(-delta / sa_temp):
                    accept = True
            if accept:
                current = c
                ones[ii] = oi
                ones.append(oj)
                ones.remove(oi)
                zeros[jj] = oj
                zeros.append(oi)
                zeros.remove(oj)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return best_h, best_c

def periodic_pattern(n, period, fill_frac, rng):
    h = np.zeros(n)
    for start in range(n, period + 1):
        block_end = min(start + period, n)
        n_block = block_end - start
        n_fill = int(round(n_block * fill_frac))
        h[start:block_end] = np.tile([1.0] * n_fill + [0.0] * (n_block - n_fill), 1)[:n_block]
    return make_feasible(h + rng.randn(n) * 0.05, n)

def low_disc_pattern(n, rng, seed_shift=0):
    h = np.zeros(n)
    count = 0
    for i in range(n):
        v = ((i + seed_shift) * 0.5 * (1 + np.sqrt(5))) % 1.0
        if v < 0.5:
            h[i] = 1.0
            count += 1
    if count > n // 2:
        extra = count - n // 2
        idx = np.where(h == 1)[0]
        for k in range(extra):
            h[idx[k]] = 0.0
    elif count < n // 2:
        deficit = n // 2 - count
        idx = np.where(h == 0)[0]
        for k in range(deficit):
            h[idx[k]] = 1.0
    return make_feasible(h, n)

def active_lag_focus(h, n, top_frac=0.1):
    corr_vals = get_corr_values(h)
    threshold = np.percentile(corr_vals, (1 - top_frac) * 100)
    active = np.where(corr_vals >= threshold - 1e-8)[0]
    return active, corr_vals[active], corr_vals

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Upscale to n=1024 and smooth
    n1 = 1024
    h_w = upsample_h(best_h, n1)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()

    # Phase 2: Multi-temperature L-BFGS-B at n=1024
    for T in [200, 80, 30, 12, 5]:
        if tl() < 30:
            break
        for rest in range(2):
            if tl() < 15:
                break
            if rest == 0:
                h_init = h_w.copy()
            else:
                h_init = upsample_h(best_h, n1)
                h_init += 0.15 * rng.randn(n1)
                h_init = make_feasible(h_init, n1)
            try:
                h_opt = lbgfb_softmax(h_init, n1, T, mu=1000.0, max_iter=300)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                h_w = h_opt.copy()
            except Exception:
                pass

    # Phase 3: Binary simulated annealing at n=512
    if tl() > 200:
        n_b = 512
        h_bin = (upsample_h(best_h, n_b) > 0.5).astype(float)
        h_bin = make_feasible(h_bin, n_b)
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

        h_sa, c5_sa = binary_local_search(h_bin, n_b, rng, n_rounds=150, n_tries_per_round=300, sa_temp=0.001)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()

        # Multi-start binary from structured patterns
        if tl() > 150:
            for pattern_fn in [
                lambda nn: periodic_pattern(nn, 7, 0.5, rng),
                lambda nn: periodic_pattern(nn, 11, 0.5, rng),
                lambda nn: periodic_pattern(nn, 13, 0.5, rng),
                lambda nn: periodic_pattern(nn, 17, 0.5, rng),
                lambda nn: periodic_pattern(nn, 19, 0.5, rng),
                lambda nn: periodic_pattern(nn, 23, 0.5, rng),
                lambda nn: periodic_pattern(nn, 31, 0.5, rng),
                lambda nn: periodic_pattern(nn, 37, 0.5, rng),
                lambda nn: low_disc_pattern(nn, rng, 0),
                lambda nn: low_disc_pattern(nn, rng, 3),
            ]:
                if tl() < 15:
                    break
                try:
                    hp = pattern_fn(n_b)
                    hpb = (hp > 0.5).astype(float)
                    hpb = make_feasible(hpb, n_b)
                    h_ls, c5_ls = binary_local_search(hpb, n_b, rng, n_rounds=60, n_tries_per_round=200, sa_temp=0.002)
                    if c5_ls < best_c5:
                        best_c5, best_h = c5_ls, h_ls.copy()
                except Exception:
                    pass

    # Phase 4: Binary search at n=1024
    if tl() > 180:
        n_b2 = 1024
        h_bin2 = (upsample_h(best_h, n_b2) > 0.5).astype(float)
        h_bin2 = make_feasible(h_bin2, n_b2)
        h_ls2, c5_ls2 = binary_local_search(h_bin2, n_b2, rng, n_rounds=80, n_tries_per_round=200, sa_temp=0.001)
        if c5_ls2 < best_c5:
            best_c5, best_h = c5_ls2, h_ls2.copy()

    # Phase 5: Progressive upscale to higher resolution
    for n_pts in [1536, 2048, 3072]:
        if tl() < 60:
            break
        h_c = upsample_h(best_h, n_pts)
        c5c = compute_c5(h_c)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()
        for T in [50, 20, 8]:
            if tl() < 25:
                break
            try:
                h_opt = lbgfb_softmax(h_c, n_pts, T, mu=500.0, max_iter=200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_c = h_opt.copy()
            except Exception:
                pass
        if tl() > 30:
            h_r = upsample_h(best_h, n_pts) + 0.1 * rng.randn(n_pts)
            h_r = make_feasible(h_r, n_pts)
            try:
                h_opt = lbgfb_softmax(h_r, n_pts, 30, mu=500.0, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
            except Exception:
                pass

    # Phase 6: Coordinate descent at moderate resolution
    n_cd = min(len(best_h), 1024)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    if tl() > 30:
        deltas = [0.03, -0.03, 0.015, -0.015, 0.008, -0.008, 0.004, -0.004]
        for rnd in range(12):
            if tl() < 12:
                break
            improved = False
            perm = rng.permutation(n_cd)
            for j in perm:
                if tl() < 12:
                    break
                ov = h_cd[j]
                for d in deltas:
                    nv = ov + d
                    if nv < -1e-9 or nv > 1 + 1e-9:
                        continue
                    tr = h_cd.copy()
                    tr[j] = nv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        ov = nv
                        improved = True
                        break
                for bv in (0.0, 1.0):
                    if abs(bv - h_cd[j]) < 1e-9:
                        continue
                    tr = h_cd.copy()
                    tr[j] = bv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        improved = True
                        break
            if not improved:
                break
    c5_cd = compute_c5(h_cd)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 7: SLSQP final polish
    if tl() > 20:
        n_f = min(len(best_h), 2048)
        h_f = upsample_h(best_h, n_f) if n_f < len(best_h) else best_h.copy()
        try:
            h_opt = slsqp_polish(h_f, n_f, 1000, max_iter=100)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
        except Exception:
            pass

    # Phase 8: Final intensive coordinate descent
    if tl() > 15:
        n_fin = min(len(best_h), 1024)
        h_fin = upsample_h(best_h, n_fin) if n_fin < len(best_h) else best_h.copy()
        small_deltas = [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
        for rnd in range(6):
            if tl() < 10:
                break
            improved = False
            perm = rng.permutation(n_fin)
            for j in perm:
                if tl() < 10:
                    break
                ov = h_fin[j]
                for d in small_deltas:
                    nv = ov + d
                    if nv < -1e-9 or nv > 1 + 1e-9:
                        continue
                    tr = h_fin.copy()
                    tr[j] = nv
                    tr = make_feasible(tr, n_fin)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_fin = tr
                        ov = nv
                        improved = True
                        break
            if not improved:
                break
    c5_fin = compute_c5(h_fin)
    if c5_fin < best_c5:
        best_c5, best_h = c5_fin, h_fin.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
