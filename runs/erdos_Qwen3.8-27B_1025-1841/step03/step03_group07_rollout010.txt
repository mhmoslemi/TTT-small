def project_to_feasible(g, n):
    target = n / 2.0

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def max_subgrad_opt(h, n, time_limit, lr_init=0.0015):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    lr = lr_init

    for it in range(500000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        c5_now = max_c * dx

        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-16:
                break

        tol = max_c * 0.002 + 1e-8
        top_idx = np.where(corr >= max_c - tol)[0]

        grad = np.zeros(n)
        for k in top_idx:
            sh = k - n + 1
            i0 = max(0, -sh)
            i1 = min(n, n - sh)
            if i0 < i1:
                grad[i0:i1] += (1.0 - h[i0+sh:i1+sh])
                j0 = max(0, i0+sh)
                j1 = min(n, i1+sh)
                if j0 < j1:
                    grad[j0:j1] -= h[j0-sh:j1-sh]
        grad *= dx / len(top_idx)

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def binary_local_search(h, n, time_limit, n_trials=8000, seed=42):
    start = time.time()
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:n_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - n_ones]:
            h_bin[zero_idx[idx]] = 1

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    rng = np.random.RandomState(seed)
    temp = 0.005

    for trial in range(n_trials):
        if time.time() - start > time_limit:
            break

        one_idx = np.where(h_bin == 1)[0]
        zero_idx = np.where(h_bin == 0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)

        delta = c5_new - best_c5
        if delta < 0 or (delta < temp and rng.rand() < np.exp(-delta / max(temp, 1e-9))):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

        if trial % 500 == 0:
            temp *= 0.95

    if compute_c5(h_bin) < best_c5:
        best_c5 = compute_c5(h_bin)
        best_h = h_bin.copy()
    return best_h, best_c5


def nesterov_fine(h, n, time_limit, lr, top_k=8):
    """Very low tau Nesterov for final polish."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    stall = 0

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        mu = min((t - 1.0) / (t + 2.0), 0.8)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = corr.max()

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        shifted = np.clip(5.0 * (top_vals - max_c), -100.0, 0.0)
        exp_c = np.exp(shifted)
        s_exp = exp_c.sum()
        p = exp_c / s_exp

        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = p[ii]
            k = int(top_idx[ii])
            sh = k - n + 1
            i0 = max(0, -sh)
            i1 = min(n, n - sh)
            if i0 < i1:
                grad[i0:i1] += pk * (1.0 - h[i0+sh:i1+sh])
                j0 = max(0, i0+sh)
                j1 = min(n, i1+sh)
                if j0 < j1:
                    grad[j0:j1] -= pk * h[j0-sh:j1-sh]
        grad *= dx

        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)

        if it % 60 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 60
            if stall > 5000:
                lr *= 0.4
                stall = 0
                if lr < 1e-18:
                    break

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n0 = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Work at n=4000
    n = 4000
    h_work = interp_periodic(h_init, n)
    h_work = project_to_feasible(h_work, n)

    # Phase 1: Nesterov tau=80 (broad convergence)
    rem = get_remaining(start_time, budget_s)
    if rem > 140:
        tl = min(110, rem * 0.18)
        h_work, c5 = nesterov_opt(h_work, n, tau=80.0, time_limit=tl, lr=0.15, top_k=15)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 2: Nesterov tau=25 (medium)
    rem = get_remaining(start_time, budget_s)
    if rem > 110:
        tl = min(90, rem * 0.15)
        h_work, c5 = nesterov_opt(h_work, n, tau=25.0, time_limit=tl, lr=0.05, top_k=12)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 3: Nesterov tau=8 (sharper)
    rem = get_remaining(start_time, budget_s)
    if rem > 90:
        tl = min(75, rem * 0.14)
        h_work, c5 = nesterov_opt(h_work, n, tau=8.0, time_limit=tl, lr=0.018, top_k=10)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 4: Max-subgradient with momentum
    rem = get_remaining(start_time, budget_s)
    if rem > 70:
        tl = min(60, rem * 0.14)
        h_work, c5 = max_subgrad_opt(best_h, n, time_limit=tl, lr_init=0.0012)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 5: Binary local search
    rem = get_remaining(start_time, budget_s)
    if rem > 55:
        tl = min(45, rem * 0.2)
        h_bin, c5 = binary_local_search(best_h, n, time_limit=tl, n_trials=12000, seed=seed)
        if c5 < best_c5:
            best_c5, best_h = c5, h_bin.copy()

    # Phase 6: Fine Nesterov polish (tau=5)
    rem = get_remaining(start_time, budget_s)
    if rem > 45:
        tl = min(40, rem * 0.2)
        h_work, c5 = nesterov_fine(best_h, n, time_limit=tl, lr=0.008, top_k=8)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 7: Another binary local search from improved point
    rem = get_remaining(start_time, budget_s)
    if rem > 35:
        tl = min(30, rem * 0.2)
        h_bin, c5 = binary_local_search(best_h, n, time_limit=tl, n_trials=8000, seed=seed+1)
        if c5 < best_c5:
            best_c5, best_h = c5, h_bin.copy()

    # Phase 8: Final fine-tune Nesterov
    rem = get_remaining(start_time, budget_s)
    if rem > 25:
        tl = min(20, rem * 0.3)
        h_work, c5 = nesterov_fine(best_h, n, time_limit=tl, lr=0.003, top_k=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 9: Try multi-start perturbation if time allows
    rem = get_remaining(start_time, budget_s)
    if rem > 50:
        for pert_seed in [seed + 100, seed + 200]:
            rem = get_remaining(start_time, budget_s)
            if rem < 30:
                break
            rng_p = np.random.RandomState(pert_seed)
            h_p = best_h.copy() + rng_p.randn(n) * 0.008
            h_p = project_to_feasible(h_p, n)
            tl = min(25, rem * 0.3)
            h_p, c5_p = nesterov_opt(h_p, n, tau=15.0, time_limit=tl, lr=0.03, top_k=10)
            if c5_p < best_c5:
                best_c5, best_h = c5_p, h_p.copy()

    # Phase 10: Final max-subgrad polish
    rem = get_remaining(start_time, budget_s)
    if rem > 15:
        tl = min(12, rem * 0.5)
        h_work, c5 = max_subgrad_opt(best_h, n, time_limit=tl, lr_init=0.0005)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Final projection and evaluation
    best_h = project_to_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
