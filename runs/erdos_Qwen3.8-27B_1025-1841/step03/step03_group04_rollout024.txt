    obj = _obj_topk if use_topk else _obj_softmax
    if top_idx is not None and not use_topk:
        obj = _obj_topk_shift
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return make_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, steps, max_passes=3):
    best_c5 = compute_c5(h)
    for _ in range(max_passes):
        improved = False
        for j in range(n):
            old_val = h[j]
            best_val = old_val
            best_c = best_c5
            for step in steps:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * step
                    if 0.0 <= v <= 1.0:
                        trial = h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def binary_sa(h, n, start_temp, end_temp, n_iter, rng):
    """Simulated annealing with swap moves on binary representation."""
    h_bin = (h > 0.5).astype(float)
    h_bin = make_feasible(h_bin, n)
    ones_idx = np.where(h_bin > 0.5)[0]
    zeros_idx = np.where(h_bin < 0.5)[0]
    best = h_bin.copy()
    best_c = compute_c5(h_bin)
    cur_c = best_c
    for i in range(n_iter):
        T = start_temp * (end_temp / start_temp) ** (i / max(n_iter - 1, 1))
        swap = rng.random() < 0.3
        if swap and len(ones_idx) > 0 and len(zeros_idx) > 0:
            oi = rng.randint(len(ones_idx))
            zi = rng.randint(len(zeros_idx))
            a = ones_idx[oi]
            b = zeros_idx[zi]
            trial = h_bin.copy()
            trial[a] = 0.0
            trial[b] = 1.0
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            delta = c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                h_bin = trial
                cur_c = c
                ones_idx[oi] = b
                zeros_idx[zi] = a
                if cur_c < best_c - 1e-15:
                    best_c = cur_c
                    best = h_bin.copy()
        else:
            idx = rng.randint(n)
            if h_bin[idx] > 0.5:
                candidates = zeros_idx
            else:
                candidates = ones_idx
            if len(candidates) == 0:
                continue
            target = candidates[rng.randint(len(candidates))]
            trial = h_bin.copy()
            trial[idx] = 1.0 - trial[idx]
            trial[target] = 1.0 - trial[target]
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            delta = c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                h_bin = trial
                cur_c = c
                if h_bin[idx] > 0.5:
                    zeros_idx = zeros_idx[zeros_idx != target]
                    ones_idx = np.append(ones_idx, target)
                    ones_idx = ones_idx[ones_idx != idx]
                    zeros_idx = np.append(zeros_idx, idx)
                else:
                    ones_idx = ones_idx[ones_idx != target]
                    zeros_idx = np.append(zeros_idx, target)
                    zeros_idx = zeros_idx[zeros_idx != idx]
                    ones_idx = np.append(ones_idx, idx)
                if cur_c < best_c - 1e-15:
                    best_c = cur_c
                    best = h_bin.copy()
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_fine = len(h_init)
    h_init = make_feasible(h_init, n_fine)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Quick SLSQP refinement of parent (match parent quality quickly)
    for T in [100, 400]:
        if remaining() < 60:
            break
        h_new = slsqp_refine(best_h, T, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Top-k SLSQP (different objective, different landscape)
    if remaining() > 100:
        _G_TOPK = 10
        h_new = slsqp_refine(best_h, T=300, maxiter=250, use_topk=True)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: Coordinate descent to tighten
    if remaining() > 80:
        best_h, best_c5 = coord_descent(best_h, n_fine, [0.001, 0.002, 0.005, 0.01, 0.02], max_passes=2)

    # Phase 4: DE on coarse grid for global exploration
    if remaining() > 350:
        n_coarse = 96
        dx_c = 2.0 / n_coarse
        target_c = n_coarse / 2.0

        def coarse_obj(x):
            x = np.clip(x, 0.0, 1.0)
            corr = np.correlate(x, 1.0 - x, mode='full')
            return float(np.max(corr) * dx_c)

        de_budget = min(remaining() - 100, 300)
        max_evals = int(de_budget * 80)
        max_evals = max(max_evals, 2000)
        max_evals = min(max_evals, 12000)

        try:
            x0 = upsample(best_h, n_coarse)
            initial_pop = [x0]
            for _ in range(39):
                initial_pop.append(rng.rand(n_coarse))
            de_res = differential_evolution(coarse_obj,
                                            [(0.0, 1.0)] * n_coarse,
                                            maxiter=200, max_evals=max_evals,
                                            pop_size=40, mutation=(0.5, 1.2),
                                            recombination=0.7, tol=1e-12,
                                            seed=seed, polish=False,
                                            init='sobol')
            if de_res is not None and de_res.x is not None:
                h_coarse = make_feasible(de_res.x, n_coarse)
                c5_coarse = coarse_obj(h_coarse)
                h_up = upsample(h_coarse, n_fine)
                c5_up = compute_c5(h_up)
                if c5_up < best_c5 + 0.002:
                    for T in [200, 500]:
                        if remaining() < 80:
                            break
                        h_new = slsqp_refine(h_up, T, maxiter=250)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new
                    if remaining() > 60:
                        best_h, best_c5 = coord_descent(best_h, n_fine, [0.002, 0.005, 0.01], max_passes=2)
        except Exception:
            pass

    # Phase 5: Binary SA to escape local opt
    if remaining() > 250:
        sa_budget = min(remaining() - 50, 200)
        n_sa = int(sa_budget * 8000)
        n_sa = max(n_sa, 50000)
        n_sa = min(n_sa, 500000)
        h_sa, c5_sa = binary_sa(best_h, n_fine, 0.002, 0.0001, n_sa, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            # Refine SA result with SLSQP
            if remaining() > 80:
                h_new = slsqp_refine(best_h, T=300, maxiter=250)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            if remaining() > 50:
                best_h, best_c5 = coord_descent(best_h, n_fine, [0.001, 0.002, 0.005], max_passes=2)

    # Phase 6: Shift-targeted SLSQP
    if remaining() > 80:
        n_pts = len(best_h)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        mc = np.max(corr)
        threshold = mc - 0.005 * mc
        top_idx = np.where(corr >= threshold)[0]
        if len(top_idx) > 12:
            top_vals = corr[top_idx]
            sort_order = np.argsort(-top_vals)
            top_idx = top_idx[sort_order[:12]]
        elif len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(corr))])
        for T in [300, 600]:
            if remaining() < 50:
                break
            h_new = slsqp_refine(best_h, T, maxiter=250, top_idx=top_idx)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 7: Random restarts
    if remaining() > 150:
        for restart in range(4):
            if remaining() < 60:
                break
            scale = 0.01 + 0.015 * restart
            h_try = make_feasible(best_h + rng.randn(n_fine) * scale, n_fine)
            h_new = slsqp_refine(h_try, T=300, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                if remaining() > 40:
                    best_h, best_c5 = coord_descent(best_h, n_fine, [0.002, 0.005, 0.01], max_passes=1)

    # Phase 8: Fine coordinate descent
    if remaining() > 40:
        best_h, best_c5 = coord_descent(best_h, n_fine, [0.0005, 0.001, 0.002, 0.005], max_passes=2)

    # Phase 9: Binary snap
    if remaining() > 15:
        n_pts = len(best_h)
        for j in range(n_pts):
            if remaining() < 3:
                break
            if 0.02 < best_h[j] < 0.98:
                for val in [0.0, 1.0]:
                    trial = best_h.copy()
                    trial[j] = val
                    trial = make_feasible(trial, n_pts)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        best_h = trial

    # Phase 10: Final ultra-fine CD
    if remaining() > 10:
        best_h, best_c5 = coord_descent(best_h, n_fine, [0.0003, 0.0005, 0.001], max_passes=1)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
