def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    shifted = corr - mc
    return mc + np.log(np.sum(np.exp(T * shifted))) / T + 1e-8 * np.sum((h - 0.5)**2)

def soft_grad(h, n, T):
    # Numerical gradient (cheap for moderate n)
    g = np.zeros(n)
    eps = 1e-7
    for i in range(n):
        h2 = h.copy()
        h2[i] += eps
        g[i] = (soft_obj(h2, n, T) - soft_obj(h, n, T)) / eps
    return g

                                 start_time=start_time)
        best_c5 = compute_c5(best_h)
        
        # Re-run coord descent after swaps
        if remaining() > 40:
            best_h = true_coord_descent(best_h, n_sweeps=400, seed=seed+300,
                                        time_limit=remaining()-15, start_time=start_time)
            best_c5 = compute_c5(best_h)
    
    # Phase 6: Final fine coordinate descent
    if remaining() > 20:
        best_h = true_coord_descent(best_h, n_sweeps=500, seed=seed+400,
                                    time_limit=remaining()-5, start_time=start_time,
                                    tol=1e-16)
        best_c5 = compute_c5(best_h)
    
    # Phase 7: Try lower resolution and see if it helps (sometimes lower n gives lower C5)
    if remaining() > 30:
        for n_try in [320, 480]:
            h_try = upsample_h(best_h, n_try)
            tl_try = min(remaining() - 10, 40)
            h_try = true_coord_descent(h_try, n_sweeps=400, seed=seed+500+n_try,
                                       time_limit=tl_try, start_time=start_time)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                best_c5 = c_try
                best_h = h_try
    
    # Final projection
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
