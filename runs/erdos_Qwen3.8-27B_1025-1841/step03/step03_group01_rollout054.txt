    vals = top_k_vals[:_SLSQP_K]
    T = 50.0
    m = vals.max()
    return (m + np.log(np.sum(np.exp(T*(vals - m))))/T) * (2.0/_SLSQP_N)

def _slsqp_cons(h):
    return np.sum(h) - _SLSQP_N / 2.0

                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-16})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if 0 <= lag < n:
        cnt = n - lag
        grad[lag:lag+cnt] += (1.0 - h[:cnt])
        grad[:cnt] -= h[lag:lag+cnt]
    elif -n < lag < 0:
        k2 = -lag
        cnt = n - k2
        grad[k2:k2+cnt] += (1.0 - h[:cnt])
        grad[:cnt] -= h[k2:k2+cnt]
    return grad

def subgrad_optimize(h, n, max_iter=200, top_k=3, lr_init=0.008, cool=0.008):
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    center = n - 1
    for it in range(max_iter):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        lags = top_idx - center
        grad = np.zeros(n)
        for lag in lags:
            grad += grad_corr(h, int(lag))
        grad /= top_k
        lr = lr_init / (1.0 + cool * it)
        h_new = h - lr * grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        corr = get_corr(h)
    return best_h, best_c5

_SLSQP_N = 0
_SLSQP_K = 3

def _slsqp_obj(h):
    corr = get_corr(h)
    vals = np.sort(corr)[::-1]
    vals = vals[:_SLSQP_K]
    T = 80.0
    m = vals.max()
    return (m + np.log(np.sum(np.exp(T * (vals - m)))) / T) * (2.0 / _SLSQP_N)

def _slsqp_cons_eq(h):
    return np.sum(h) - _SLSQP_N / 2.0

def _slsqp_cons_jac(h):
    return np.ones(_SLSQP_N)

    _SLSQP_N = n
    _SLSQP_K = top_k
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_eq, 'jac': _slsqp_cons_jac}]
    res = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)
