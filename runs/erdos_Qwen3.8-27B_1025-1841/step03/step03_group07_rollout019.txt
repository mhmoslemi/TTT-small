def block_flip_search(h, n, time_limit, start_time, seed=42):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    cur_c5 = best_c5
    n_reject = 0
    
    for it in range(200000):
        if time.time() - start_time > time_limit:
            break
        
        # Random block complement
        blen = rng.randint(3, min(40, n // 5))
        start = rng.randint(0, n - blen)
        
        h_trial = h.copy()
        h_trial[start:start+blen] = 1.0 - h_trial[start:start+blen]
        h_trial = project_to_feasible(h_trial, n)
        
        c5_t = compute_c5(h_trial)
        
        if c5_t < cur_c5:
            h = h_trial
            cur_c5 = c5_t
            n_reject = 0
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()
        else:
            n_reject += 1
            # Geometric cooling
            temp = 0.05 * (0.999 ** it) + 0.001
            if c5_t - cur_c5 < temp * rng.exponential(1.0):
                h = h_trial
                cur_c5 = c5_t
    
    return best_h, best_c5

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def soft_obj_grad(h, n, tau, top_k=20):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    max_val = top_vals.max()
    shifted = np.clip(tau * (top_vals - max_val), -50.0, 0.0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    p = exp_s / s
    obj = (max_val + np.log(s) / tau) * dx
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        pk = p[ii]
        k = int(top_idx[ii])
        sh = k - n + 1
        a1 = max(0, sh)
        b1 = min(n, n + sh)
        if a1 < b1:
            grad[a1:b1] += pk * (1.0 - h[a1 - sh:b1 - sh])
        a2 = max(0, -sh)
        b2 = min(n, n - sh)
        if a2 < b2:
            grad[a2:b2] -= pk * h[a2 + sh:b2 + sh]
    grad *= dx
    return obj, grad


def nesterov_opt(h, n, tau0, tau1, time_limit, start_time, lr=None, top_k=20):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    h_prev = h.copy()
    t = 1.0
    if lr is None:
        lr = 0.5 / max(tau0, 10.0)
    stall = 0
    check_interval = 40

    for it in range(10000000):
        if time.time() - start_time > time_limit:
            break
        frac = min(1.0, it / 50000.0)
        tau = tau0 * (tau1 / tau0) ** frac
        mu = min((t - 1.0) / (t + 2.0), 0.85)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj, grad = soft_obj_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        if it % check_interval == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-12:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += check_interval
            if stall > 4000:
                lr *= 0.6
                stall = 0
                if lr < 1e-12:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def block_flip_search(h, n, time_limit, start_time, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    cur_c5 = best_c5
    n_reject = 0
    step = 0

    for it in range(500000):
        if time.time() - start_time > time_limit:
            break
        blen = rng.randint(2, min(35, n // 6))
        start = rng.randint(0, n - blen)
        h_trial = h.copy()
        h_trial[start:start + blen] = 1.0 - h_trial[start:start + blen]
        h_trial = project_to_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < cur_c5:
            h = h_trial
            cur_c5 = c5_t
            n_reject = 0
            step = 0
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()
        else:
            n_reject += 1
            step += 1
            if step > 2000:
                step = 0
                n_reject = 0
            temp = 0.03 + 0.02 / (1.0 + 0.0001 * n_reject)
            if rng.random() < np.exp(-(c5_t - cur_c5) / temp):
                h = h_trial
                cur_c5 = c5_t
    return best_h, best_c5


def subgrad_polish(h, n, time_limit, start_time, lr=0.0008, top_k=5):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.75
    stall = 0

    for it in range(10000000):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-15:
                break
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        max_v = top_vals.max()
        w = np.clip(top_vals - max_v, -50.0, 0.0)
        w = np.exp(w * 5.0)
        w /= w.sum()
        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = w[ii]
            k = int(top_idx[ii])
            sh = k - n + 1
            a1 = max(0, sh)
            b1 = min(n, n + sh)
            if a1 < b1:
                grad[a1:b1] += pk * (1.0 - h[a1 - sh:b1 - sh])
            a2 = max(0, -sh)
            b2 = min(n, n - sh)
            if a2 < b2:
                grad[a2:b2] -= pk * h[a2 + sh:b2 + sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))
