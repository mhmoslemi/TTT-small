```python
def try_swap(h, a, b, corr, active_shifts, M):
    """Check if swapping h[a] and h[b] reduces the max correlation.
    active_shifts: indices in corr array where corr is near max
    M: current max of corr
    Returns new max if improved, None otherwise"""
    n = len(h)
    da = h[b] - h[a]  # change at position a
    db = h[a] - h[b]  # change at position b (=-da)
    
    new_max = M
    for si in active_shifts:
        # si is index in corr array, shift = si - (n-1)
        s = si - (n - 1)
        # Δcorr[si] = da * (1-h[a+s]) - da*h[a-s] + db*(1-h[b+s]) - db*h[b-s]
        # = da * [(1-h[a+s]) - h[a-s]] + db * [(1-h[b+s]) - h[b-s]]
        # where h[j] = 0 if j < 0 or j >= n
        delta = 0.0
        # da contribution at position a
        if a + s >= 0 and a + s < n:
            delta += da * (1.0 - h[a+s])
        if a - s >= 0 and a - s < n:
            delta -= da * h[a-s]
        # db contribution at position b
        if b + s >= 0 and b + s < n:
            delta += db * (1.0 - h[b+s])
        if b - s >= 0 and b - s < n:
            delta -= db * h[b-s]
        new_val = corr[si] + delta
        if new_val > new_max:
            new_max = new_val
        if new_max >= M:
            break  # Can't be an improvement
    if new_max < M:
        return new_max
    return None
```

```python
    s = abs(shift)
    if s >= n:

    g = np.zeros(n)
    if shift >= 0:

    else:

    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.3 * f)
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)

        tot = np.sum(exps)
        if tot < 1e-300:

            g = _subgrad_one(z, gi)
        else:

            top_k = min(80, len(w))

            g = np.zeros(n)
            dx = 2.0 / n

                if wi < 1e-12:

            g *= dx
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:

    return bh, bc
```

A better approach: **Coordinate-wise optimization of the active shifts**

For each active shift s, the correlation is:

This is a quadratic form in h. To reduce it, we want to make h[i] and h[i+s] "aligned" (both high or both low).

1. The sum constraint
2. Multiple shifts share the same h values
3. The bounds [0,1]

So a coupled optimization is needed. But we can do **cyclic coordinate descent**:
- Fix all but two coordinates h[a], h[b]
- Optimize over h[a], h[b] subject to h[a]+h[b] = const (preserving sum)
- This is a 1D problem: h[a] = c - h[b], minimize max over active shifts

corr_s = ∑ h[i]*(1-h[i+s]). If only h[a] and h[b] vary:
- Terms involving h[a]: h[a]*(1-h[a+s]) and h[a-s]*(1-h[a]) [if valid]
- Terms involving h[b]: h[b]*(1-h[b+s]) and h[b-s]*(1-h[b]) [if valid]

First order terms: h[a]*(1-h[a+s]) = h[a] - h[a]*h[a+s]
If a+s ≠ b and a-s ≠ b: this is linear in h[a] (h[a+s] is fixed)

```python
def local_search(h, n_trials, rng, T_start, T_end):
    n = len(h)
    h = h.astype(float).copy()
    curr_c = compute_c5(h)
    best_c = curr_c
    best_h = h.copy()
    step_sizes = np.array([0.02, 0.05, 0.1, 0.15, 0.3, 0.5])
    active_cache = None
    active_timer = 0
    for trial in range(n_trials):
        active_timer += 1
        if active_cache is None or active_timer % 3000 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            M = np.max(corr)
            active_cache = np.where(corr > M - 0.03 * M)[0]
        progress = trial / max(n_trials - 1, 1)
        T = T_start * (T_end / T_start) ** progress
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        mt = rng.randint(4)
        h_new = h.copy()
        if mt == 0:
            h_new[a], h_new[b] = h[b], h[a]
        elif mt == 1:
            d = step_sizes[rng.randint(len(step_sizes))]
            if rng.rand() < 0.5:
                h_new[a] += d
                h_new[b] -= d
            else:
                h_new[a] -= d
                h_new[b] += d
            if h_new[a] < 0 or h_new[a] > 1 or h_new[b] < 0 or h_new[b] > 1:
                continue
        elif mt == 2:
            if len(active_cache) > 0:
                si = active_cache[rng.randint(len(active_cache))]
                s = si - (n - 1)
                s = abs(s)
                if s == 0 or s >= n:
                    continue
                a2 = rng.randint(n - s)
                h_new2 = h.copy()
                h_new2[a2], h_new2[a2 + s] = h[a2 + s], h[a2]
                new_c2 = compute_c5(h_new2)
                delta2 = new_c2 - curr_c
                if delta2 < 0 or rng.rand() < np.exp(-delta2 / max(T, 1e-10)):
                    h = h_new2
                    curr_c = new_c2
                    if curr_c < best_c:
                        best_c = curr_c
                        best_h = h.copy()
                    continue
        else:
            d = step_sizes[rng.randint(len(step_sizes))]
            h_new[a] = np.clip(h[a] + d * (rng.rand() - 0.5) * 2.0, 0, 1)
            h_new[b] = h[b] + (h[a] - h_new[a])
            if h_new[b] < 0 or h_new[b] > 1:
                continue
        new_c = compute_c5(h_new)
        delta = new_c - curr_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            curr_c = new_c
            if curr_c < best_c:
                best_c = curr_c
                best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, T, max_iter):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(c)
        return (mc + np.sum(np.exp(np.clip(T * (c - mc), -500, 0))) / T) * dx
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(c)
        exps = np.exp(np.clip(T * (c - mc), -500, 0))
        w = exps / np.sum(exps)
        g = np.zeros(n)
        for i in range(len(c)):
            if w[i] < 1e-10:
                continue
            shift = i - (n - 1)
            g += w[i] * _subgrad_one(x, shift)
        return g * dx
    def con(x):
        return np.sum(x) - n / 2.0
    def con_j(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con, 'jac': con_j}],
                     options={'maxiter': max_iter, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            cx = compute_c5(hx)
            if cx < compute_c5(h):
                return hx, cx
    except Exception:
        pass
    return h, compute_c5(h)

def upsample_fn(h, nn):
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def remaining():
        return budget_s - elapsed()
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = project_c(rng.rand(960), 960)
    n = len(h)
    bc = compute_c5(h)
    bh = h.copy()
    print(f"Init: n={n} C5={bc:.6f}", flush=True)
    # Phase 1: FISTA
    if remaining() > 250:
        h2, c2 = fista_opt(bh, 4000, 25.0, 0.05, 0.09)
        if c2 < bc:
            bc, bh = c2, h2.copy()
        print(f"F1: C5={bc:.6f} t={elapsed():.0f}s", flush=True)
    # Phase 2: Local search (main)
    if remaining() > 500:
        h3, c3 = local_search(bh, 100000, rng, 0.005, 0.00005)
        if c3 < bc:
            bc, bh = c3, h3.copy()
        print(f"LS1: C5={bc:.6f} t={elapsed():.0f}s", flush=True)
    # Phase 3: SLSQP
    if remaining() > 150:
        for TV in [8.0, 3.0, 1.5]:
            if remaining() < 30:
                break
            h4, c4 = slsqp_smooth(bh, TV, 350)
            if c4 < bc:
                bc, bh = c4, h4
        print(f"SLSQP: C5={bc:.6f} t={elapsed():.0f}s", flush=True)
    # Phase 4: Second local search
    if remaining() > 250:
        rng2 = np.random.RandomState(seed + 100)
        h5, c5 = local_search(bh, 80000, rng2, 0.003, 0.00002)
        if c5 < bc:
            bc, bh = c5, h5.copy()
        print(f"LS2: C5={bc:.6f} t={elapsed():.0f}s", flush=True)
    # Phase 5: Try higher resolution
    if remaining() > 200:
        n2 = int(n * 1.25)
        if n2 != n and remaining() > 180:
            hu = upsample_fn(bh, n2)
            cu = compute_c5(hu)
            if remaining() > 120:
                hf, cf = fista_opt(hu, 2500, 15.0, 0.05, 0.06)
                if cf < cu:
                    hu, cu = hf, cf
            if remaining() > 80:
                hl, cl = local_search(hu, 30000, np.random.RandomState(seed + 200), 0.003, 0.00005)
                if cl < cu:
                    hu, cu = hl, cl
            cf = compute_c5(hu)
            if cf < bc:
                bc, bh = cf, hu.copy()
            print(f"RES{n2}: C5={bc:.6f} t={elapsed():.0f}s", flush=True)
    # Phase 6: Final local search
    if remaining() > 80:
        ns = int(min(remaining() * 120, 50000))
        ns = max(ns, 10000)
        rng3 = np.random.RandomState(seed + 300)
        h6, c6 = local_search(bh, ns, rng3, 0.002, 0.00001)
        if c6 < bc:
            bc, bh = c6, h6.copy()
    # Phase 7: Final SLSQP
    if remaining() > 20:
        bh2, c7 = slsqp_smooth(bh, 1.0, 200)
        if c7 < bc:
            bc, bh = c7, bh2
    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={elapsed():.0f}s", flush=True)
    return bh, bc, len(bh)
```
