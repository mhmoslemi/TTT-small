    n = len(h)
    dx = 2.0 / n

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def slp_iterate(h_star, n, solver='ECOS', max_it_in_slp=1):
    """One SLP step: build LP from linearization at h_star, solve it."""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    
    # Build constraint matrix A (n_shifts x n) and constant vector b
    # For output index m (0 to 2n-2), shift k = m - (n-1)
    # result[m] = sum_{valid n} h[n+k] * (1-h[n])
    # Linearized: sum_j A[m,j]*h[j] + b[m]
    
    A = np.zeros((n_shifts, n))
    b = np.zeros(n_shifts)
    
    for m in range(n_shifts):
        k = m - (n - 1)
        if k >= 0:
            # n ranges from 0 to n-1-k
            # h[n+k]*(1-h[n])
            valid = np.arange(0, n - k)
            # Linearized: (1-h*[n])*h[n+k] - h*[n+k]*h[n] + h*[n+k]*(1-h*[n])
            # Coeff of h[j]:
            #   (1-h*[j-k]) for j in [k, n-1] (from first term, n=j-k)
            #   -h*[j+k] for j in [0, n-1-k] (from second term, n=j)
            A[m, k:n] += (1.0 - h_star[0:n-k])
            A[m, 0:n-k] += -h_star[k:n]
            # Constant: h*[n+k]*(1-h*[n]) for n=0..n-1-k
            # But we need f(x*) - sum_j A[m,j]*x*[j]
            f_star = np.sum(h_star[k:n] * (1.0 - h_star[0:n-k]))
            # sum_j A[m,j]*h*[j]:
            # sum_{j=k}^{n-1} (1-h*[j-k])*h*[j] + sum_{j=0}^{n-1-k} (-h*[j+k])*h*[j]
            s1 = np.sum((1.0 - h_star[0:n-k]) * h_star[k:n])
            s2 = np.sum(-h_star[k:n] * h_star[0:n-k])
            b[m] = f_star - (s1 + s2)
        else:
            kk = -k  # positive
            # n ranges from kk to n-1
            # h[n+k]*(1-h[n]) = h[n-kk]*(1-h[n])
            valid = np.arange(kk, n)
            # Coeff of h[j]:
            #   (1-h*[j+kk]) for j in [0, n-1-kk] (from first term, n=j+kk)
            #   -h*[j-kk] for j in [kk, n-1] (from second term, n=j)
            A[m, 0:n-kk] += (1.0 - h_star[kk:n])
            A[m, kk:n] += -h_star[0:n-kk]
            f_star = np.sum(h_star[0:n-kk] * (1.0 - h_star[kk:n]))
            s1 = np.sum((1.0 - h_star[kk:n]) * h_star[0:n-kk])
            s2 = np.sum(-h_star[0:n-kk] * h_star[kk:n])
            b[m] = f_star - (s1 + s2)
    
    # Now solve the LP
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        A @ x + b <= t / dx,
        x >= 0,
        x <= 1,
        cp.sum(x) == n / 2.0
    ]
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        return h_star.copy()
    
    if x.value is None:
        return h_star.copy()
    
    return make_feasible(np.asarray(x.value, dtype=float), n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def build_slp_matrix(h_star, n):
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    A = np.zeros((n_shifts, n))
    b = np.zeros(n_shifts)
    for m in range(n_shifts):
        k = m - (n - 1)
        if k >= 0:
            A[m, k:n] += (1.0 - h_star[0:n-k])
            A[m, 0:n-k] += -h_star[k:n]
            f_star = np.sum(h_star[k:n] * (1.0 - h_star[0:n-k]))
            s = np.sum((1.0 - h_star[0:n-k]) * h_star[k:n]) + np.sum(-h_star[k:n] * h_star[0:n-k])
            b[m] = f_star - s
        else:
            kk = -k
            A[m, 0:n-kk] += (1.0 - h_star[kk:n])
            A[m, kk:n] += -h_star[0:n-kk]
            f_star = np.sum(h_star[0:n-kk] * (1.0 - h_star[kk:n]))
            s = np.sum((1.0 - h_star[kk:n]) * h_star[0:n-kk]) + np.sum(-h_star[0:n-kk] * h_star[kk:n])
            b[m] = f_star - s
    return A, b, dx

def slp_solve(h_star, n, delta=0.3, solver='ECOS'):
    A, b, dx = build_slp_matrix(h_star, n)
    x = cp.Variable(n)
    t = cp.Variable()
    lo = np.maximum(0.0, h_star - delta)
    hi = np.minimum(1.0, h_star + delta)
    constraints = [
        A @ x + b <= t / dx,
        x >= lo,
        x <= hi,
        cp.sum(x) == n / 2.0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver, max_iters=200000)
    except Exception:
        try:
            prob.solve(solver='SCS', max_iters=200000)
        except Exception:
            return h_star.copy()
    if x.value is None:
        return h_star.copy()
    x_val = np.clip(np.asarray(x.value, dtype=float), 0.0, 1.0)
    return make_feasible(x_val, n)

            if n_cd < len(best_h):
                best_h = upsample_h(h_cd, len(best_h))
            else:
                best_h = h_cd.copy()

    # Phase 6: Final SLP polish
    if time_left() > 25:
        try:
            h_new = slp_solve(best_h, len(best_h), delta=0.02, solver='ECOS')
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
