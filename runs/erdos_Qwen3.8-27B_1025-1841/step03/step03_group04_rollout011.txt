    # Phase 1: Binary snap + local binary pair-flip search
    h_bin = (h_init > 0.5).astype(float)
    h_bin = make_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    if remaining() > 120:
        best_h, best_c5 = _binary_local_search(best_h, n, rng, 15)

    # Phase 2: SLSQP with progressive temperature at base resolution
    for T_val in [80.0, 200.0, 500.0]:
        if remaining() < 60:
            break
        x = _do_slsqp(best_h, T_val, None, 350, n)
        c5_res = compute_c5(x)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = make_feasible(x, n)

    # Phase 3: Shift-targeted SLSQP (focus on top correlation lags)
    if remaining() > 100:
        for _ in range(3):
            if remaining() < 50:
                break
            corr = get_corr(best_h)
            mc = np.max(corr)
            thr = mc - 0.008 * mc
            top_idx = np.where(corr >= thr)[0]
            if len(top_idx) > 15:
                tv = corr[top_idx]
                so = np.argsort(-tv)
                top_idx = top_idx[so[:15]]
            elif len(top_idx) == 0:
                top_idx = np.array([int(np.argmax(corr))])
            for T_val in [300.0, 600.0]:
                if remaining() < 40:
                    break
                x = _do_slsqp(best_h, T_val, top_idx, 250, n)
                c5_res = compute_c5(x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x, n)

    # Phase 4: Block-based local search
    if remaining() > 100:
        n_pts = len(best_h)
        for block_size in [4, 8, 16]:
            if remaining() < 40:
                break
            num_blocks = n_pts // block_size
            for _pass in range(2):
                if remaining() < 25:
                    break
                improved = False
                for b in range(num_blocks):
                    if remaining() < 10:
                        break
                    s_i = b * block_size
                    e_i = s_i + block_size
                    for _it in range(15):
                        pert = rng.randn(block_size) * 0.015
                        trial_block = np.clip(best_h[s_i:e_i] + pert, 0, 1)
                        trial = best_h.copy()
                        trial[s_i:e_i] = trial_block
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c5 - 1e-15:
                            best_h = trial
                            best_c5 = c
                            improved = True
                if not improved:
                    break

    # Phase 5: Fine coordinate descent
    if remaining() > 80:
        n_pts = len(best_h)
        for _pass in range(4):
            if remaining() < 20:
                break
            for j in range(n_pts):
                if remaining() < 5:
                    break
                old_val = best_h[j]
                best_val = old_val
                best_c = best_c5
                for step in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    for sign in [-1.0, 1.0]:
                        v = old_val + sign * step
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n_pts)
                            c = compute_c5(trial)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_val = v
                if best_val != old_val:
                    best_h[j] = best_val
                    best_h = make_feasible(best_h, n_pts)
                    best_c5 = best_c

    # Phase 6: Binary relabeling with pair flips (more aggressive)
    if remaining() > 60:
        h_b2 = (best_h > 0.5).astype(float)
        h_b2 = make_feasible(h_b2, n)
        h_b2, c5_b2 = _binary_local_search(h_b2, n, rng, 20)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
        # Also try threshold at different values
        for thresh in [0.4, 0.6]:
            h_t = (best_h > thresh).astype(float)
            h_t = make_feasible(h_t, n)
            h_t, c5_t = _binary_local_search(h_t, n, rng, 10)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 7: Random restarts with SLSQP
    for _restart in range(6):
        if remaining() < 50:
            break
        n_pts = len(best_h)
        scales = [0.03, 0.06, 0.015, 0.08, 0.02, 0.04]
        sc = scales[_restart % len(scales)]
        h_try = make_feasible(best_h + rng.randn(n_pts) * sc, n_pts)
        x = _do_slsqp(h_try, 250, None, 200, n_pts)
        c5_res = compute_c5(x)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = make_feasible(x, n_pts)

    # Phase 8: Upsample to 2x if time allows
    if remaining() > 350:
        n_up = n * 2
        if n_up <= 1920:
            h_up = upsample_h(best_h, n_up)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5 + 0.003:
                x = _do_slsqp(h_up, 150, None, 200, n_up)
                c5_res = compute_c5(x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x, n_up)
                if remaining() > 150:
                    corr_u = get_corr(best_h)
                    mc_u = np.max(corr_u)
                    thr_u = mc_u - 0.01 * mc_u
                    top_u = np.where(corr_u >= thr_u)[0]
                    if len(top_u) > 20:
                        tv_u = corr_u[top_u]
                        so_u = np.argsort(-tv_u)
                        top_u = top_u[so_u[:20]]
                    elif len(top_u) == 0:
                        top_u = np.array([int(np.argmax(corr_u))])
                    x = _do_slsqp(best_h, 400, top_u, 200, len(best_h))
                    c5_res = compute_c5(x)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = make_feasible(x, len(best_h))

    # Phase 9: Final ultra-fine coordinate descent
    if remaining() > 20:
        n_pts = len(best_h)
        for _rd in range(2):
            for j in range(n_pts):
                if remaining() < 3:
                    break
                old_val = best_h[j]
                best_val = old_val
                best_c = best_c5
                for step in [0.0005, 0.001, 0.002, 0.004]:
                    for sign in [-1.0, 1.0]:
                        v = old_val + sign * step
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n_pts)
                            c = compute_c5(trial)
                            if c < best_c - 1e-16:
                                best_c = c
                                best_val = v
                if best_val != old_val:
                    best_h[j] = best_val
                    best_h = make_feasible(best_h, n_pts)
                    best_c5 = best_c

    # Phase 10: Final binary check
    if remaining() > 5:
        h_bf = (best_h > 0.5).astype(float)
        h_bf = make_feasible(h_bf, len(best_h))
        c5_bf = compute_c5(h_bf)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
