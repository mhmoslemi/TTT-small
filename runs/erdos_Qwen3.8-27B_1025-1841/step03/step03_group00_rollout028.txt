import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = np.sum(mask)
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    h_ext = np.concatenate([h, h, h])

    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -100, 100)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def compute_gradient(h, n, T, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-30:
        weights = np.zeros(2*n-1)
        weights[np.argmax(corr)] = 1.0
    else:
        weights = weights / sw
    lh = 1.0 - h
    first = np.zeros(n)
    second = np.zeros(n)
    for p in range(2 * n - 1):
        wv = weights[p]
        if wv < 1e-12:
            continue
        k = p - (n - 1)
        if k >= 0:
            ln = n - k
            if ln > 0:
                first[k:k+ln] += wv * lh[:ln]
                second[:ln] += wv * h[k:k+ln]
        else:
            kk = -k
            ln = n - kk
            if ln > 0:
                first[:ln] += wv * lh[kk:kk+ln]
                second[kk:kk+ln] += wv * h[:ln]
    grad = dx * (first - second)
    viol = np.sum(h) - n / 2.0
    grad += 2.0 * mu * viol
    return grad

def lbgfb_optimize(h_start, n, T, mu, max_iter=200):
