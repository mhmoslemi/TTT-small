    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _obj(x):
    dx = 2.0 / _G_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = _G_T * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    sm = m + log_sum / _G_T
    viol = np.sum(x) - _G_n / 2.0
    return sm * dx + _G_lam * viol * viol

def _grad(x):
    n = _G_n
    dx = 2.0 / n
    T = _G_T
    lam = _G_lam
    
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    # Compute gradient using vectorized approach
    # S1[j] = sum_k w[k]*(1-x[j-shift[k]]) for valid (0<=j-shift[k]<n)
    # S2[j] = sum_k w[k]*x[j+shift[k]] for valid (0<=j+shift[k]<n)
    
    k_arr = np.arange(2 * n - 1)
    shifts = k_arr - (n - 1)
    
    J = np.arange(n)
    diff1 = J[:, None] - shifts[None, :]
    diff2 = J[:, None] + shifts[None, :]
    
    valid1 = (diff1 >= 0) & (diff1 < n)
    valid2 = (diff2 >= 0) & (diff2 < n)
    
    d1 = np.clip(diff1, 0, n - 1)
    d2 = np.clip(diff2, 0, n - 1)
    
    W = w[None, :]
    
    S1 = (valid1 * W * (1.0 - x[d1])).sum(axis=1)
    S2 = (valid2 * W * x[d2]).sum(axis=1)
    
    grad = dx * (S1 - S2)
    viol = np.sum(x) - n / 2.0
    grad += 2.0 * lam * viol
    
    return grad

def lfbfgs(h_start, n, T, lam, max_iter):
    global _G_n, _G_T, _G_lam
    _G_n = n
    _G_T = T
    _G_lam = lam
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h_start, jac=_grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-8})
    return make_feasible(res.x, n)

def subgradient_step(h, n, n_iters, step0):
    """Polish using projected subgradient on the non-smooth max."""
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h)
    step = step0
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        
        # Subgradient of the max shift's correlation
        grad = np.zeros(n)
        if shift >= 0:
            valid = n - shift
            grad[:valid] += (1.0 - h[shift:shift + valid])
            grad[shift:] += (-h[:valid])
        else:
            valid = -shift
            grad[:valid] += (-h[valid:])
            grad[valid:] += (1.0 - h[:valid])
        
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        
        if c_new < best_c:
            h = h_new
            best_c = c_new
            step = min(step * 1.01, 1.0)
        else:
            step *= 0.95
            if step < 0.001:
                break
    
    return h, best_c
