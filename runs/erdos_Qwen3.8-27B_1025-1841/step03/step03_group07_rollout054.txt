    s = exp_c.sum()
    p = exp_c / s

        k = top_idx[i_idx]

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def soft_obj_grad(h, n, tau, top_k=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    shifted = np.clip(tau * (top_vals - max_c), -50.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = p[i_idx]
        k = top_idx[i_idx]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def adam_optimize(h, n, tau, time_limit, lr=0.01, top_k=10, beta1=0.9, beta2=0.999):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    stall = 0
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        obj, grad = soft_obj_grad(h, n, tau, top_k)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
        t += 1
        if it % 25 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += 25
            if stall > 2000:
                lr *= 0.5
                m *= 0.3
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def nesterov_optimize(h, n, tau, time_limit, lr=0.05, top_k=10):

    best_h = h.copy()

            best_h = h.copy()

        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)

            pk = w[i_idx]
            k = top_idx[i_idx]
            sh = k - n + 1
            j0 = max(0, sh)

                idx = np.arange(j0, j1)

                idx = np.arange(j0b, j1b)

        h_trial = project_to_feasible(h - step * grad, n)
