    # Phase 4: Targeted local search (novel: focus on worst lags)
    if remaining() > 60:
        best_h, best_c5 = targeted_local_search(best_h, len(best_h), 50, rng)
    
    # Phase 5: Coordinate descent
    if remaining() > 60:
        cands = ['zero', 'one', 'center', 'up1', 'down1', 'up2', 'down2']
        best_h, best_c5 = coord_descent(best_h, len(best_h), 30, cands)
    
    # Phase 6: Swap-based search (novel)
    if remaining() > 90:
        n_pts = len(best_h)
        n_swaps = min(50000, int(remaining() * 200))
        best_h, best_c5 = swap_search(best_h, n_pts, n_swaps, rng)
    
    # Phase 7: Final targeted search + coordinate descent
    if remaining() > 30:
        best_h, best_c5 = targeted_local_search(best_h, len(best_h), 30, rng)
    
    if remaining() > 20:
        cands = ['zero', 'one', 'up1', 'down1', 'up2', 'down2']
        best_h, best_c5 = coord_descent(best_h, len(best_h), 20, cands)
    
    # Phase 8: Final SLSQP polish
    if remaining() > 30:
        n_pts = len(best_h)
        _G_N = n_pts
        _G_T = 1000.0
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_raw_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-15})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass
    
    # Final: binary snap + verify
    if remaining() > 10:
        best_h, best_c5 = binary_snap(best_h, len(best_h))
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
