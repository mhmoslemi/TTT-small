def upsample(h, new_n):

- L-BFGS-B with maxiter=200: ~200 gradients = ~38s

That's much better! And L-BFGS-B handles bounds (0≤h≤1) natively.

The constraint sum(h) = n/2 can be handled by:

Let me use the penalty approach. With a large enough λ, it will effectively enforce the constraint.

def obj_penalty(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    soft = logsumexp(T * corr) / T
    penalty = lam * (np.sum(h) - n/2.0)**2

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def binary_sa(h, n, iters, T0, T1, rng):
    b = np.where(h >= 0.5, 1.0, 0.0).copy()
    tgt = n // 2
    while int(b.sum()) > tgt:
        idx = np.where(b == 1)[0]
        b[idx[rng.randint(len(idx))]] = 0
    while int(b.sum()) < tgt:
        idx = np.where(b == 0)[0]
        b[idx[rng.randint(len(idx))]] = 1
    cur = compute_c5(b)
    best = cur
    bb = b.copy()
    for i in range(iters):
        T = T0 * (T1 / T0) ** (i / max(iters, 1))
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        si = ones[rng.randint(len(ones))]
        sj = zeros[rng.randint(len(zeros))]
        b[si] = 0
        b[sj] = 1
        nc = compute_c5(b)
        if nc <= cur or rng.random() < np.exp(-(nc - cur) / max(T, 1e-14)):
            cur = nc
            if nc < best:
                best = nc
                bb = b.copy()
        else:
            b[si] = 1
            b[sj] = 0
    return bb, best

def binary_sa_block(h, n, iters, T0, T1, rng):
    b = np.where(h >= 0.5, 1.0, 0.0).copy()
    tgt = n // 2
    while int(b.sum()) > tgt:
        idx = np.where(b == 1)[0]
        b[idx[rng.randint(len(idx))]] = 0
    while int(b.sum()) < tgt:
        idx = np.where(b == 0)[0]
        b[idx[rng.randint(len(idx))]] = 1
    cur = compute_c5(b)
    best = cur
    bb = b.copy()
    for i in range(iters):
        T = T0 * (T1 / T0) ** (i / max(iters, 1))
        bs = rng.randint(1, 5)
        start = rng.randint(0, n - bs)
        seg_ones = np.where(b[start:start+bs] == 1)[0]
        seg_zeros = np.where(b[start:start+bs] == 0)[0]
        if len(seg_ones) == 0 or len(seg_zeros) == 0:
            continue
        sw = min(len(seg_ones), len(seg_zeros))
        co = rng.choice(seg_ones, size=sw, replace=False)
        cz = rng.choice(seg_zeros, size=sw, replace=False)
        b[start+co] = 0
        b[start+cz] = 1
        nc = compute_c5(b)
        if nc <= cur or rng.random() < np.exp(-(nc - cur) / max(T, 1e-14)):
            cur = nc
            if nc < best:
                best = nc
                bb = b.copy()
        else:
            b[start+co] = 1
            b[start+cz] = 0
    return bb, best

def lbfgsb_run(h, n, T, lam, maxiter, ftol, time_limit):
    start = time.time()

    def objective(x):
        if time.time() - start > time_limit:
            return compute_c5(x) + 1e6
        c = corr_dx(x)
        soft = logsumexp(T * c) / T
        pen = lam * (np.sum(x) - n / 2.0) ** 2
        return soft + pen

    bounds = [(0.0, 1.0)] * n
    res = minimize(objective, h.copy(), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
    return res.x

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_fine(h, n, max_rounds, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        for j in range(n):
            old = h[j]
            for cand in [0.0, 1.0, max(0.0, old - 0.01), min(1.0, old + 0.01)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    h = trial
                    old = cand
                    break
    return h, best_c5
