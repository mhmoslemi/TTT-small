def _fg(x):
    return smooth_obj_and_grad(x, _CUR_T, _CUR_LAM)

def upsample_to(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def lbgfs_optimize(h_start, T, lam, maxiter):
    global _CUR_T, _CUR_LAM
    _CUR_T = T
    _CUR_LAM = lam
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_fg, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return make_feasible(res.x, n)
    except Exception:
        return h_start

def coordinate_descent(h, n, n_rounds, time_limit):
    start_t = time.time()
    best_c = compute_c5(h)
    for _ in range(n_rounds):
        if time.time() - start_t > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_t > time_limit:
                break
            old = h[j]
            best_val = old
            best_c_local = best_c
            for cand in (0.0, 1.0, 0.25, 0.5, 0.75,
                         max(0.0, old - 0.04), max(0.0, old - 0.01),
                         max(0.0, old - 0.003),
                         min(1.0, old + 0.003), min(1.0, old + 0.01),
                         min(1.0, old + 0.04)):
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_val = cand
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_c_local
                improved = True
        if not improved:
            break
    return h, best_c

def pairwise_search(h, n, time_limit):
    start_t = time.time()
    best_c = compute_c5(h)
    diff = np.abs(h - 0.5)
    idx_order = np.argsort(diff)
    for i in range(min(60, len(idx_order))):
        if time.time() - start_t > time_limit:
            break
        j1 = idx_order[i]
        old1 = h[j1]
        for target1 in (0.0, 1.0):
            if abs(target1 - old1) < 1e-6:
                continue
            trial = h.copy()
            trial[j1] = target1
            trial = make_feasible(trial, n)
            c1 = compute_c5(trial)
            if c1 < best_c - 1e-14:
                h = trial
                best_c = c1
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _CUR_T, _CUR_LAM
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_parent = len(h_init)
    h_init = make_feasible(h_init, n_parent)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    configs = [
        (n_parent, 40.0, 1e-6, 400),
        (n_parent, 100.0, 1e-8, 350),
        (n_parent, 250.0, 1e-10, 300),
        (2 * n_parent, 40.0, 1e-6, 250),
        (2 * n_parent, 100.0, 1e-8, 200),
        (2 * n_parent, 250.0, 1e-10, 200),
        (2 * n_parent, 500.0, 1e-12, 150),
    ]

    for cfg_n, T, lam, maxiter in configs:
        if remaining() < 50:
            break
        h_base = upsample_to(best_h, cfg_n)
        n_starts = 3 if cfg_n <= n_parent else 2
        for r in range(n_starts):
            if remaining() < 40:
                break
            if r == 0:
                h_start = h_base.copy()
            elif r == 1:
                h_start = h_base + rng.randn(cfg_n) * 0.015
                h_start = make_feasible(h_start, cfg_n)
            else:
                h_start = h_base + rng.randn(cfg_n) * 0.04
                h_start = make_feasible(h_start, cfg_n)
            h_res = lbgfs_optimize(h_start, T, lam, maxiter)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()

    n_best = len(best_h)
    for T_focus in (400.0, 800.0, 1500.0):
        if remaining() < 30:
            break
        h_res = lbgfs_optimize(best_h, T_focus, 1e-12, 200)
        c5_res = compute_c5(h_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            n_best = len(best_h)

    n_pts = len(best_h)
    best_h, best_c5 = coordinate_descent(best_h, n_pts, 25, remaining() - 20)

    if remaining() > 15:
        best_h, best_c5 = pairwise_search(best_h, len(best_h), remaining() - 10)

    if remaining() > 25:
        for attempt in range(6):
            if remaining() < 18:
                break
            h_pert = best_h.copy()
            n_pert = max(3, len(h_pert) // 80)
            idx_pert = rng.choice(len(h_pert), size=n_pert, replace=False)
            for idx in idx_pert:
                h_pert[idx] += rng.uniform(-0.08, 0.08)
            h_pert = make_feasible(h_pert, len(h_pert))
            h_res = lbgfs_optimize(h_pert, 600.0, 1e-11, 80)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                n_pts = len(best_h)

    if remaining() > 10:
        best_h, best_c5 = coordinate_descent(best_h, len(best_h), 15, remaining() - 5)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
