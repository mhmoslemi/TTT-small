    """Project x onto {y: 0 <= y <= 1, sum(y) = target_sum}"""
    # Bisection on lambda: y_i = clip(x_i - lambda, 0, 1)
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0, 1)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    # Adjust
    if diff > 0:
        mask = (y < 1 - 1e-12) & (y > 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0, 1)

            y = np.clip(y, 0, 1)

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    if new_n == old_n:

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

        #   contribution to S1: +1
        #   contribution to S2: -h[j+offset] (first term: j is i, j+offset is i+offset)
        # For each j in [i_min+offset, i_max+offset):
        #   contribution to S2: -h[j-offset] (second term: j-offset is i, j is i+offset)

        if offset >= 0:

            grad[i_min:i_max] -= h[i_min+offset:i_max+offset]
        else:

            # offset is negative, so j+offset < j
            lo = i_min
            hi = i_max
            grad[lo:hi] -= h[lo+offset:hi+offset]

        lo2 = i_min + offset
        hi2 = i_max + offset

        lo2 = max(0, lo2)
        hi2 = min(n, hi2)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2-offset:hi2-offset]
    
    grad /= len(active)
    return grad * dx, max_val * dx

```python
def project_feasible(x, n):
    """Project x onto {y: 0 <= y <= 1, sum(y) = n/2}."""
    target = n / 2.0
    # Bisection on lambda
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any small discrepancy
    diff = target - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y
```

    corr = np.correlate(h, 1.0 - h, mode='full')

    for k in active:
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n-1, n-1-offset)
        if i_hi < i_lo:
            continue
        # Term 1: (1-h[i+offset]) for i in [i_lo, i_hi]
        lo1, hi1 = i_lo, i_hi + 1
        grad[lo1:hi1] += (1.0 - h[lo1+offset:hi1+offset])
        # Term 2: -h[i-offset] for i in [i_lo, i_hi]
        lo2, hi2 = i_lo + offset, i_hi + offset + 1
        # Clamp
        lo2c = max(0, lo2)
        hi2c = min(n, hi2)
        if hi2c > lo2c:
            grad[lo2c:hi2c] -= h[lo2c-offset:hi2c-offset]
    
    grad /= len(active)
    grad *= dx
    
    # Penalty gradient
    viol = np.sum(h) - n / 2.0
    grad += 2.0 * mu * viol * np.ones(n)
    
    obj = max_val * dx + mu * viol * viol
    return obj, grad

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def obj_and_grad(h, n, mu=1000.0, T=1000.0):
    """Compute objective and gradient. Uses smoothed max for differentiability."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if T > 1e6:
        # Use exact max (subgradient)
        max_val = corr.max()
        active = np.where(corr >= max_val - 1e-12)[0]
        grad_corr = np.zeros(n)
        for k in active:
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_hi >= i_lo:
                lo1, hi1 = i_lo, i_hi + 1
                grad_corr[lo1:hi1] += (1.0 - h[lo1 + offset:hi1 + offset])
                lo2, hi2 = i_lo + offset, i_hi + offset + 1
                lo2c, hi2c = max(0, lo2), min(n, hi2)
                if hi2c > lo2c:
                    grad_corr[lo2c:hi2c] -= h[lo2c - offset:hi2c - offset]
        grad_corr /= len(active)
        obj = max_val * dx
    else:
        # Smoothed max via log-sum-exp
        max_val = corr.max()
        shifted = T * (corr - max_val)
        shifted = np.clip(shifted, -80, 80)
        weights = np.exp(shifted)
        wsum = weights.sum()
        obj = (max_val + np.log(wsum) / T) * dx
        
        # Gradient of smoothed max
        # d/dh[j] of log-sum-exp(T*(corr - max_val)) / T
        # = (1/T) * sum_k weights[k] * T * d(corr[k])/dh[j] / wsum
        # = sum_k (weights[k]/wsum) * d(corr[k])/dh[j]
        # So it's a weighted average of the per-k gradients
        grad_corr = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] / wsum < 1e-8:
                continue
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_hi >= i_lo:
                lo1, hi1 = i_lo, i_hi + 1
                grad_corr[lo1:hi1] += (weights[k] / wsum) * (1.0 - h[lo1 + offset:hi1 + offset])
                lo2, hi2 = i_lo + offset, i_hi + offset + 1
                lo2c, hi2c = max(0, lo2), min(n, hi2)
                if hi2c > lo2c:
                    grad_corr[lo2c:hi2c] -= (weights[k] / wsum) * h[lo2c - offset:hi2c - offset]
        grad_corr *= dx
    
    # Add penalty
    viol = np.sum(h) - n / 2.0
    grad = grad_corr + 2.0 * mu * viol * np.ones(n)
    obj = obj + mu * viol * viol
    
    return obj, grad

            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_hi >= i_lo:
                lo1, hi1 = i_lo, i_hi + 1
                grad_corr[lo1:hi1] += pk * (1.0 - h[lo1 + offset:hi1 + offset])
                lo2, hi2 = i_lo + offset, i_hi + offset + 1
                lo2c, hi2c = max(0, lo2), min(n, hi2)
                if hi2c > lo2c:
                    grad_corr[lo2c:hi2c] -= pk * h[lo2c - offset:hi2c - offset]
        grad_corr *= dx
    viol = np.sum(h) - n / 2.0
    grad = grad_corr + 2.0 * mu * viol
    obj_val = obj + mu * viol * viol
    return obj_val, grad

def _fun(h, n, mu, T):
    return obj_and_grad(h, n, mu, T)[0]

def _jac(h, n, mu, T):
    return obj_and_grad(h, n, mu, T)[1]

def lbgfb_with_grad(h_start, n, T, mu, max_iter=2000):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_fun, h_start, args=(n, mu, T), jac=_jac,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

def _slsqp_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -80, 80)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def _slsqp_con(h, n, T):
    return np.sum(h) - n / 2.0

def _slsqp_con_jac(h, n, T):
    return np.ones(n)

def slsqp_with_grad(h_start, n, T, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    res = minimize(_slsqp_obj, h_start, args=(n, T), constraints=cons,
                   method='SLSQP', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def check_time(start_time, budget_s):
    return budget_s - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: L-BFGS-B with exact gradient at n=1024
    n1 = 1024
    h1 = upsample_h(best_h, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    temp_schedule = [30, 100, 500, 3000]
    for T in temp_schedule:
        if check_time(start_time, budget_s) < 50:
            break
        h_opt = lbgfb_with_grad(h1, n1, T, mu=1000, max_iter=3000)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
        h1 = h_opt.copy()

    # Random restarts at n=1024
    for restart in range(4):
        if check_time(start_time, budget_s) < 50:
            break
        h_r = h1 + 0.08 * rng.randn(n1)
        h_r = make_feasible(h_r, n1)
        for T in [50, 500]:
            if check_time(start_time, budget_s) < 30:
                break
            h_opt = lbgfb_with_grad(h_r, n1, T, mu=1000, max_iter=2000)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            h_r = h_opt.copy()

    # Phase 2: Higher resolution n=2048
    if check_time(start_time, budget_s) > 120:
        n2 = 2048
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
        for T in [50, 500, 5000]:
            if check_time(start_time, budget_s) < 50:
                break
            h_opt = lbgfb_with_grad(h2, n2, T, mu=1000, max_iter=2500)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            h2 = h_opt.copy()

    # Phase 3: Try n=1280 as alternative
    if check_time(start_time, budget_s) > 80:
        n3 = 1280
        h3 = upsample_h(best_h, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
        for T in [100, 2000]:
            if check_time(start_time, budget_s) < 40:
                break
            h_opt = lbgfb_with_grad(h3, n3, T, mu=1000, max_iter=2000)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            h3 = h_opt.copy()

    # Phase 4: SLSQP polish at moderate resolution
    if check_time(start_time, budget_s) > 50:
        n_s = min(len(best_h), 512)
        h_s = upsample_h(best_h, n_s) if n_s < len(best_h) else best_h.copy()
        for T in [300, 3000]:
            if check_time(start_time, budget_s) < 25:
                break
            h_opt = slsqp_with_grad(h_s, n_s, T, max_iter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            h_s = h_opt.copy()

    # Phase 5: Coordinate descent polish at best resolution
    n_cd = len(best_h)
    h_cd = best_h.copy()
    if check_time(start_time, budget_s) > 20:
        deltas = [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
        stride = max(1, n_cd // 400)
        for sweep in range(4):
            if check_time(start_time, budget_s) < 10:
                break
            improved = False
            for j in range(0, n_cd, stride):
                if check_time(start_time, budget_s) < 10:
                    break
                ov = h_cd[j]
                for d in deltas:
                    nv = ov + d
                    if nv < 0 or nv > 1:
                        continue
                    tr = h_cd.copy()
                    tr[j] = nv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        ov = nv
                        improved = True
                        break
            if not improved:
                break

    c5_cd = compute_c5(h_cd)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Final SLSQP squeeze
    if check_time(start_time, budget_s) > 20:
        n_f = min(len(best_h), 512)
        h_f = upsample_h(best_h, n_f) if n_f < len(best_h) else best_h.copy()
        h_opt = slsqp_with_grad(h_f, n_f, 5000, max_iter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
