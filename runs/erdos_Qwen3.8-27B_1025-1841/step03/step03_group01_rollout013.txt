def make_feasible(h, n):
    # Ensure 0 <= h <= 1 and sum(h) = n/2
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0

        if abs(diff) < 1e-13:

        else:
            mask = h > 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbfgs_smooth_max(h_start, n, T, mu, max_iter):
    dx = 2.0 / n
    target = n / 2.0

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -400, 400)
        lse = np.log(np.sum(np.exp(shifted)))
        penalty = mu * (np.sum(h) - target) ** 2
        return (mc + lse / T) * dx + penalty

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16,
                                'gtol': 1e-14, 'maxls': 30, 'maxfun': max_iter * (n + 10)})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start.copy(), n)

def slsqp_smooth_max(h_start, n, T, max_iter):
    dx = 2.0 / n
    target = n / 2.0

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -400, 400)
        lse = np.log(np.sum(np.exp(shifted)))
        return (mc + lse / T) * dx

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start.copy(), n)

def swap_search(h_start, n, time_budget, rng, base_steps=None):
    h_curr = h_start.copy()
    best_c5 = compute_c5(h_curr)
    if base_steps is None:
        base_steps = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    ss_idx = 0
    stall = 0
    t0 = time.time()
    while time.time() - t0 < time_budget:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            stall += 1
            if stall > 3000:
                ss_idx = min(ss_idx + 1, len(base_steps) - 1)
                stall = 0
            continue
        delta = base_steps[ss_idx] * rng.choice([-1.0, 1.0])
        hi = h_curr[i] + delta
        hj = h_curr[j] - delta
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            stall += 1
            if stall > 3000:
                ss_idx = min(ss_idx + 1, len(base_steps) - 1)
                stall = 0

        h_curr[i] = hi
        h_curr[j] = hj
        c = compute_c5(h_curr)
        if c < best_c5 - 1e-15:

        else:

            h_curr[j] = hj + delta
            stall += 1
