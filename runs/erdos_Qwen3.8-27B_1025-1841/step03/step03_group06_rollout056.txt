def binary_snap(h, n):
    """Snap to binary while maintaining sum = n/2."""
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Need to flip (n_ones - target) positions from 1 to 0
        # Choose positions closest to 0.5 (i.e., most "uncertain")
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])  # closest to 0.5 first
        for i in range(n_ones - target):
            h_bin[ones_idx[order[i]]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])  # closest to 0.5 first
        for i in range(target - n_ones):
            h_bin[zeros_idx[order[i]]] = 1.0
    return h_bin

import numpy as np
from scipy.optimize import minimize
import time

_PARAMS = {}

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_snap(h, n):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    if n % 2 != 0:
        target_ones = (n - 1) // 2
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - target_ones):
            h_bin[ones_idx[order[i]]] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(target_ones - n_ones):
            h_bin[zeros_idx[order[i]]] = 1.0
    return h_bin

    n = _PARAMS['n']
    T = _PARAMS['T']
    pen = _PARAMS['pen']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return soft * dx + pen * (np.sum(x) - n / 2.0) ** 2

def _obj_slsqp(x):
    n = _PARAMS['n']
    T = _PARAMS['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    n = _PARAMS['n']

    return np.ones(_PARAMS['n'])
