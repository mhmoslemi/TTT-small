        idx = zeros[np.argmax(h[zeros])]

def binary_sa(h, n, max_iters, T0, T1, rng, time_limit=None):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    t_start = time.time()
    for it in range(max_iters):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        move = rng.randint(3)
        if move == 0:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = 0, 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i], b[j] = 1, 0
        elif move == 1:
            bs = rng.randint(2, 7)
            s1 = rng.randint(0, n - bs)
            s2 = rng.randint(0, n - bs)
            if s2 < s1: s1, s2 = s2, s1
            seg1 = b[s1:s1+bs]
            seg2 = b[s2:s2+bs]
            o1 = np.where(seg1 == 1)[0]
            z2 = np.where(seg2 == 0)[0]
            if len(o1) == 0 or len(z2) == 0: continue
            cnt = min(len(o1), len(z2))
            c1 = rng.choice(o1, size=cnt, replace=False)
            c2 = rng.choice(z2, size=cnt, replace=False)
            b[s1+c1] = 0
            b[s2+c2] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[s1+c1] = 1
                b[s2+c2] = 0
        else:
            # Rotation: shift a block
            bs = rng.randint(2, 5)
            s = rng.randint(0, n - bs)
            shift = rng.choice([-2, -1, 1, 2])
            ns = s + shift
            if ns < 0 or ns + bs > n: continue
            orig = b[s:s+bs].copy()
            ones_in = np.where(orig == 1)[0]
            if len(ones_in) == 0 or len(ones_in) == bs: continue
            # Try the move
            b[s:s+bs] = 0
            placed = 0
            for p in ones_in:
                tgt = ns + p
                if 0 <= tgt < n and b[tgt] == 0:
                    b[tgt] = 1
                    placed += 1
            if placed != len(ones_in):
                # Can't place all, revert
                b[s:s+bs] = orig
                continue
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[s:s+bs] = 0
                for p in ones_in:
                    b[ns + p] = 0
                b[s:s+bs] = orig
    return best_b, best_c5

def make_feasible(h, n):

def slsqp_soft(h, n, T, maxiter, ftol):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -700, 0)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

            best_h = b_alt2.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100:
            h_try, c5_try = proj_grad_descent(best_h, len(best_h), 150.0, 3000, 0.25, start_time, remaining - 5)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 8: Coordinate descent on best solution
    n_pts = len(best_h)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 20, start_time, remaining)

    # Phase 9: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_pts = len(best_h)
        try:
            x_res = slsqp_soft(best_h, n_pts, 600.0, 300, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 10: Final projected GD
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        h_try, c5_try = proj_grad_descent(best_h, n_pts, 1000.0, 2000, 0.1, start_time, remaining - 2)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
