import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    H = fft(h, N)
    G = fft(1.0 - h, N)
    corr = np.real(ifft(H * np.conj(G)))
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:

        mc = np.max(conv)
        sh = conv - mc
        return (mc + np.log(np.sum(np.exp(np.clip(T * sh, -500, 500))) / T)) * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
