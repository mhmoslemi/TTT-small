    """Update corr in-place after flipping h[j]. Returns new h[j]."""
    # h[j] changes from old to 1-old, so delta = 1 - 2*old
    old = h[j]
    delta = 1.0 - 2.0 * old
    
    # corr[m] = sum_i h[i] * (1-h[i-m+n-1]) for valid i and i-m+n-1
    # Change in corr[m]:
    #   Term 1: h[j] * (1-h[j-m+n-1]) → changes by delta * (1-h[j-m+n-1])

    # Term 2 valid when 0 ≤ j+m-n+1 ≤ n-1, i.e., n-1-j ≤ m ≤ n-1+j

        shifted = np.clip(shifted, -70, 70)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(hv) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13})
    return make_feasible(res.x, n)

def slsqp_refine(h_start, n, T, max_iter=100):
    dx = 2.0 / n
    def obj(hv):
        corr = compute_corr(hv)
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -70, 70)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    def con(hv):
        return np.sum(hv) - n / 2.0
    def jac(hv):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)
    n0 = len(h0)
    h0 = make_feasible(h0, n0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Binary SA at current resolution
    h_bin = to_binary(best_h, n0)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin.copy()

    n_sa = n0
    h_work = best_h.copy()
    sa_iters = min(500000, int(80000 * n0 / 640))
    h_res, c5_res, _ = sa_pair_swap(h_work, n_sa, rng, sa_iters, T0=1.5, T_end=0.01)
    if c5_res < best_c5:
        best_c5, best_h = c5_res, h_res.copy()

    # Phase 2: Multi-resolution L-BFGS-B
    for n_pts in [1024, 1536, 2048]:
        if tl() < 60:
            break
        h_c = upsample_h(best_h, n_pts)
        c5_c = compute_c5(h_c)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c.copy()
        for T in [50, 200, 800]:
            if tl() < 25:
                break
            try:
                h_opt = lbgfb_optimize(h_c, n_pts, T, mu=300.0, max_iter=180)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                h_c = h_opt.copy()
            except Exception:
                pass

    # Phase 3: Binary SA at higher resolution
    for n_pts in [1280, 1920]:
        if tl() < 80:
            break
        h_hi = upsample_h(best_h, n_pts)
        h_hi_bin = to_binary(h_hi, n_pts)
        c5_hi = compute_c5(h_hi_bin)
        if c5_hi < best_c5:
            best_c5, best_h = c5_hi, h_hi_bin.copy()
        sa_it = min(300000, int(40000 * n_pts / 640))
        h_r, c5_r, _ = sa_pair_swap(h_hi_bin, n_pts, rng, sa_it, T0=1.0, T_end=0.05)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()

    # Phase 4: Targeted local search (sweep)
    n_ls = min(len(best_h), 1280)
    h_ls = upsample_h(best_h, n_ls) if n_ls < len(best_h) else best_h.copy()
    h_ls_b = to_binary(h_ls, n_ls)
    if tl() > 40:
        for sweep in range(6):
            if tl() < 15:
                break
            improved = False
            ones = np.where(h_ls_b > 0.5)[0].tolist()
            zeros = np.where(h_ls_b <= 0.5)[0].tolist()
            c5_cur = compute_c5(h_ls_b)
            for _ in range(3000):
                if tl() < 15:
                    break
                oi = rng.randint(len(ones))
                oz = rng.randint(len(zeros))
                i, j = int(ones[oi]), int(zeros[oz])
                new_h = h_ls_b.copy()
                new_h[i], new_h[j] = 0.0, 1.0
                c_new = compute_c5(new_h)
                if c_new < c5_cur - 1e-14:
                    h_ls_b = new_h
                    c5_cur = c_new
                    ones[oi], zeros[oz] = j, i
                    improved = True
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h_ls_b.copy()
            if not improved:
                break

    # Phase 5: SLSQP polish
    if tl() > 25:
        n_s = min(len(best_h), 1024)
        h_s = upsample_h(best_h, n_s) if n_s < len(best_h) else best_h.copy()
        for T in [500, 3000]:
            if tl() < 12:
                break
            try:
                h_opt = slsqp_refine(h_s, n_s, T, max_iter=80)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                h_s = h_opt.copy()
            except Exception:
                pass

    # Phase 6: Final coordinate descent
    if tl() > 20:
        n_cd = min(len(best_h), 1024)
        h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
        for rnd in range(4):
            if tl() < 10:
                break
            any_imp = False
            deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
            for j in range(0, n_cd, 2):
                if tl() < 10:
                    break
                ov = h_cd[j]
                for d in deltas:
                    nv = ov + d
                    if nv < 0 or nv > 1:
                        continue
                    tr = h_cd.copy()
                    tr[j] = nv
                    diff = n_cd / 2.0 - np.sum(tr)
                    if abs(diff) < 1e-12:
                        break
                    mask = (tr > 1e-12) & (tr < 1.0 - 1e-12)
                    if diff > 0:
                        mask = tr < 1.0 - 1e-12
                    else:
                        mask = tr > 1e-12
                    cnt = np.sum(mask)
                    if cnt == 0:
                        continue
                    tr[mask] = np.clip(tr[mask] - diff / cnt, 0.0, 1.0)
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-13:
                        best_c5, best_h = c, tr.copy()
                        h_cd = tr
                        ov = nv
                        any_imp = True
                        break
            if not any_imp:
                break

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
