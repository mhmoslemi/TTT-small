    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = np.where(h < 1.0 - 1e-14)[0]
        else:
            mask = np.where(h > 1e-14)[0]
        if len(mask) == 0:
            break
        step = diff / len(mask)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def smooth_obj(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -50, 50)
    soft = (mc + np.log(np.mean(np.exp(shifted))) / T) * dx
    viol = np.sum(h) - n / 2.0
    return soft + mu * viol * viol

                   args=(T, mu), options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

            j = zeros[rng.randint(len(zeros))]

                h[j] = 0

    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def correlate_fft(a, b):
    """Compute np.correlate(a, b, mode='full') using FFT"""
    na, nb = len(a), len(b)
    size = na + nb - 1
    # Pad to next power of 2 for speed
    fsize = 1

def correlate_via_fft(a, b):
    na, nb = len(a), len(b)
    size = na + nb - 1
    fsize = 1 << (size - 1).bit_length()  # next power of 2
