        x_new = project_feasible(y - alpha * g, n)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * tnest * tnest)) / 2.0
        y = x_new + ((tnest - 1.0) / t_new) * (x_new - x)
        y = project_feasible(y, n)
        x = x_new
        tnest = t_new

        if it % 200 == 199:
            c5_chk = compute_c5(x_new)
            if c5_chk >= best_c5:
                alpha *= 0.85
            else:
                alpha = min(alpha * 1.08, 0.4)

        if it % 50 == 49:
            c5_now = compute_c5(x_new)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = x_new.copy()

    # Phase 2: SLSQP targeted on top shifts
    if budget_s - (time.time() - t0) > 120:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        _G_TI = get_top_idx(corr, 0.004, 25)
        _G_T_SLE = 600.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
        try:
            res = minimize(obj_sle, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 500, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 3: Coordinate descent
    for cd_pass in range(8):
        if budget_s - (time.time() - t0) < 50:
            break
        improved = False
        for j in range(n):
            if budget_s - (time.time() - t0) < 10:
                break
            ov = best_h[j]
            bv = ov
            bc = best_c5
            for step in [0.001, 0.003, 0.008, 0.02, 0.05]:
                for sgn in [-1.0, 1.0]:
                    v = ov + sgn * step
                    if 0.0 <= v <= 1.0:
                        tr = best_h.copy()
                        tr[j] = v
                        tr = make_feasible(tr, n)
                        c = compute_c5(tr)
                        if c < bc - 1e-15:
                            bc = c
                            bv = v
            if bv != ov:
                best_h[j] = bv
                best_h = make_feasible(best_h, n)
                best_c5 = bc
                improved = True
        if not improved:
            break

    # Phase 4: Second SLSQP with updated top
    if budget_s - (time.time() - t0) > 100:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        _G_TI = get_top_idx(corr, 0.003, 15)
        _G_T_SLE = 900.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
        try:
            res = minimize(obj_sle, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 400, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 5: Fine coordinate descent
    for cd_pass in range(5):
        if budget_s - (time.time() - t0) < 25:
            break
        improved = False
        for j in range(n):
            if budget_s - (time.time() - t0) < 5:
                break
            ov = best_h[j]
            bv = ov
            bc = best_c5
            for step in [0.0003, 0.0008, 0.002, 0.005]:
                for sgn in [-1.0, 1.0]:
                    v = ov + sgn * step
                    if 0.0 <= v <= 1.0:
                        tr = best_h.copy()
                        tr[j] = v
                        tr = make_feasible(tr, n)
                        c = compute_c5(tr)
                        if c < bc - 1e-15:
                            bc = c
                            bv = v
            if bv != ov:
                best_h[j] = bv
                best_h = make_feasible(best_h, n)
                best_c5 = bc
                improved = True
        if not improved:
            break

    # Phase 6: Random restarts with SLSQP
    for rst in range(8):
        if budget_s - (time.time() - t0) < 60:
            break
        scale = 0.006 if rst < 4 else 0.015
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        corr = np.correlate(h_try, 1.0 - h_try, mode='full')
        _G_TI = get_top_idx(corr, 0.005, 25)
        _G_T_SLE = 500.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
        try:
            res = minimize(obj_sle, h_try, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 7: Final FISTA burst
    if budget_s - (time.time() - t0) > 40:
        x2 = best_h.copy()
        y2 = best_h.copy()
        tn2 = 1.0
        a2 = 0.05
        T2 = 200.0
        for it2 in range(10000):
            if budget_s - (time.time() - t0) < 10:
                break
            corr2 = np.correlate(y2, 1.0 - y2, mode='full')
            ti2 = get_top_idx(corr2, 0.004, 30)
            ct2 = corr2[ti2]
            g2 = subgrad_weighted(y2, ti2, ct2, T2, n)
            gn2 = np.linalg.norm(g2)
            if gn2 < 1e-12:
                break
            x2n = project_feasible(y2 - a2 * g2, n)
            tn2n = (1.0 + np.sqrt(1.0 + 4.0 * tn2 * tn2)) / 2.0
            y2 = x2n + ((tn2 - 1.0) / tn2n) * (x2n - x2)
            y2 = project_feasible(y2, n)
            x2 = x2n
            tn2 = tn2n
            if it2 % 100 == 99:
                c5_2 = compute_c5(x2n)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = x2n.copy()

    # Phase 8: Final micro coordinate descent
    if budget_s - (time.time() - t0) > 15:
        for j in range(n):
            if budget_s - (time.time() - t0) < 3:
                break
            ov = best_h[j]
            bv = ov
            bc = best_c5
            for step in [0.0001, 0.0003, 0.001]:
                for sgn in [-1.0, 1.0]:
                    v = ov + sgn * step
                    if 0.0 <= v <= 1.0:
                        tr = best_h.copy()
                        tr[j] = v
                        tr = make_feasible(tr, n)
                        c = compute_c5(tr)
                        if c < bc - 1e-16:
                            bc = c
                            bv = v
            if bv != ov:
                best_h[j] = bv
                best_h = make_feasible(best_h, n)
                best_c5 = bc

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
