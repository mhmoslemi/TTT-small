def project_to_feasible(g, n):

    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def grad_smooth_max(h, tau, top_k=30):
    """Gradient of the smooth approximation to max(corr)."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    L = len(corr)
    if top_k < L:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(L)
    top_vals = corr[top_idx]
    w = np.exp(tau * (top_vals - top_vals.max()))
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        m = top_idx[idx]
        s = m - n + 1

        # (1-h[j-s]) * 1_{valid} - h[j+s] * 1_{valid}

        j0b = max(0, -s)
        j1b = min(n, n - s)

    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    if lr is None:
        lr = 0.5 / tau
    
    h_prev = h.copy()
    t = 1.0
    stall_count = 0
    
    while time.time() - start < time_limit:
        it_mu = (t - 1.0) / (t + 2.0)
        it_mu = min(it_mu, momentum_cap)
        y = h + it_mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        grad = grad_smooth_max(y, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        # Check improvement every so often
        stall_count += 1
        if stall_count % 50 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall_count = 0
            else:
                # Reduce learning rate if stuck
                if stall_count > 2000:
                    lr *= 0.5
                    stall_count = 0
                    if lr < 1e-14:
                        break
        
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def coordinate_descent(h0, time_limit, sweep_limit=500):
    """Coordinate descent: for each j, find optimal h[j]."""
    n = len(h0)
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    corr = compute_corr(h).copy()
    best_c5 = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    
    j_order = np.arange(n)
    sweeps = 0
    
    while time.time() - start < time_limit and sweeps < sweep_limit:
        np.random.shuffle(j_order)
        sweeps += 1
        improved = False
        
        for j in j_order:
            old_val = h[j]
            
            # Compute slopes s[k] for all k where h[j] appears
            # h[j] appears in corr[m] for m in [n-1-j, n-1+j] (shift s = m-n+1 in [-j, j])
            # slope = (1-h[j-s]) - h[j+s] (with bounds checking)
            
            # Valid shifts: s such that 0 <= j-s < n and 0 <= j+s < n (for the two terms)
            # First term (1-h[j-s]): valid when 0 <= j-s < n, i.e., -j < s <= j... 
            #   Actually: 0 <= j-s < n → j-n < s <= j
            # Second term -h[j+s]: valid when 0 <= j+s < n → -j < s < n-j
            # Combined: s ranges from max(-j, j-n+1) to min(j, n-j-1)
            #   = s ∈ [max(1-j, j-n+1), min(j, n-j-1)]
            #   For 0 <= j < n: max(1-j, j-n+1) = 1-j if j < n/2, else j-n+1
            #   min(j, n-j-1) = j if j < n/2, else n-j-1
            
            # Simpler: just iterate over all valid s
            s_min = max(1 - j, j - n + 1)
            s_max = min(j, n - j - 1)
            
            if s_min > s_max:
                # h[j] doesn't appear in any correlation (isolated point)
                continue
            
            # For each s in [s_min, s_max], the index in corr is m = s + n - 1
            # slope[s] = (1-h[j-s]) * 1_{0<=j-s<n} - h[j+s] * 1_{0<=j+s<n}
            
            # Actually, let me recompute. In corr[m] = Σ_i h[i](1-h[i+s]) where s=m-n+1:
            # h[j] appears when:
            #   i = j: contribution h[j](1-h[j+s]), valid if 0<=j+s<n → s > -j, s < n-j
            #   i+s = j, i.e., i = j-s: contribution h[j-s](1-h[j]), valid if 0<=j-s<n → s < j, s > j-n
            #     i.e., j-n < s < j
            # So combined valid s: max(-j+1, j-n+1) <= s <= min(j-1, n-j-1)
            # Wait, I need to be more careful with the inequalities.
            # i=j valid: 0<=j<n (always) and 0<=j+s<n → -j <= s < n-j → s ∈ [-j, n-j-1]
            # i=j-s valid: 0<=j-s<n → j-n < s <= j → s ∈ [j-n+1, j]
            #   and 0<=j<n (always for the "shifted" index i+s=j which is in [0,n))
            # Combined: s ∈ [max(-j, j-n+1), min(n-j-1, j)]
            
            s_lo = max(-j, j - n + 1)
            s_hi = min(n - j - 1, j)
            
            if s_lo > s_hi:
                continue
            
            # Compute slopes for all valid s
            # s_values = arange(s_lo, s_hi+1)
            # For each s: slope = (1-h[j+s])*1_{s>=-j} - h[j-s]*1_{s<=j}
            #   But since s_lo >= -j and s_hi <= j, both terms are always present!
            #   Wait: s_lo = max(-j, j-n+1) >= -j, so j+s >= j+s_lo >= j-j = 0. ✓
            #   s_hi = min(n-j-1, j) <= j, so j-s >= j-s_hi >= j-j = 0. ✓
            #   Also j+s <= j+s_hi <= j+n-j-1 = n-1. ✓
            #   And j-s <= j-s_lo <= j-(j-n+1) = n-1. ✓
            # So both indices are always valid! Great.
            
            # slopes: for s in [s_lo, s_hi]
            # slope(s) = (1 - h[j+s]) - h[j-s]
            
            # For a candidate t (=new h[j]):
            #   corr[m] changes by (t - old_val) * slope(s)
            #   New corr[m] = corr[m] + (t - old_val) * slope(s)
            #   New max = max(corr) considering the changes
            
            # To find optimal t: minimize max_k (corr[m_k] + (t-old_val)*slope(s_k))
            #   where m_k = s_k + n - 1 for s_k in [s_lo, s_hi]
            #   and also all other corr values (which don't change)
            
            # The unchanged max is: max(corr[all other indices])
            other_max = corr.max()  # This includes the ones that change, so it's an upper bound
            # Actually, we need the max of unchanged entries:
            # corr indices that change: [s_lo+n-1, s_hi+n-1]
            # corr indices that don't change: [0, s_lo+n-1-1] ∪ [s_hi+n-1+1, 2n-2]
            m_lo = s_lo + n - 1
            m_hi = s_hi + n - 1
            
            if m_lo > 0:
                other_max_left = corr[:m_lo].max()
            else:
                other_max_left = -np.inf
            if m_hi < 2*n - 2:
                other_max_right = corr[m_hi+1:].max()
            else:
                other_max_right = -np.inf
            other_max = max(other_max_left, other_max_right)
            
            changed_corr = corr[m_lo:m_hi+1].copy()
            # slopes for s in [s_lo, s_hi]:
            # slope(s) = (1-h[j+s]) - h[j-s]
            s_arr = np.arange(s_lo, s_hi + 1)
            slopes = (1.0 - h[j + s_arr]) - h[j - s_arr]
            
            # f(t) = max(other_max, max_k (changed_corr[k] + (t - old_val) * slopes[k]))
            # Let u = t - old_val, so we minimize over u ∈ [-(old_val), (1-old_val)]:
            # f(u) = max(other_max, max_k (changed_corr[k] + u * slopes[k]))
            
            # This is: max(other_max, max_k (changed_corr[k] + u * slopes[k]))
            # The inner max is a convex piecewise linear function of u.
            # The outer max with other_max just clips from below.
            
            # To find the minimum: the function g(u) = max_k (changed_corr[k] + u*slopes[k])
            # is convex. f(u) = max(other_max, g(u)) is also convex.
            # Minimum of f on [u_lo, u_hi] where u_lo = -old_val, u_hi = 1-old_val.
            
            # Simple approach: evaluate at a few points and refine.
            # Or: the minimum of g(u) is where the subgradient contains 0.
            # The subgradient of g at u is [min{slopes[k] : k active}, max{slopes[k] : k active}]
            # where active = {k : changed_corr[k] + u*slopes[k] = g(u)}
            
            # Practical: try u = 0 (current), u that makes the current argmax line cross other_max,
            # and the "balanced" point.
            
            # Let's use a simple approach: try several candidates.
            u_cands = [0.0, -old_val, 1.0 - old_val]
            
            # Add the point where g(u) = other_max (if it exists)
            # g(u) <= other_max means all changed_corr[k] + u*slopes[k] <= other_max
            # For slopes[k] > 0: u <= (other_max - changed_corr[k]) / slopes[k]
            # For slopes[k] < 0: u >= (other_max - changed_corr[k]) / slopes[k]
            # For slopes[k] = 0: need changed_corr[k] <= other_max (else infeasible)
            
            # The minimum of f is at:
            # - u = u_lo (left boundary)
            # - u = u_hi (right boundary)  
            # - u where g(u) is minimized (if below other_max)
            # - u where g(u) = other_max (transition point)
            
            # For the minimum of g(u): it's at the point where the "upper envelope" of lines
            # achieves its minimum. Since the lines are max of linear functions, the minimum
            # is at a breakpoint or at the boundary.
            
            # Quick method: evaluate g at u=0 and at the "optimal" u for the top few lines.
            # Find the current max line:
            cur_max_idx = np.argmax(changed_corr)
            cur_max_val = changed_corr[cur_max_idx]
            
            if cur_max_val > other_max:
                # g(0) > other_max, so f(0) = g(0)
                # Try reducing: if slopes[cur_max_idx] < 0, increase u
                # if slopes[cur_max_idx] > 0, decrease u
                pass
            
            # Simplest: just try a few values of u
            # u that minimizes the top line: doesn't help since it's linear
            # u where top line = second line: (c2 - c1)/(s1 - s2) if s1 != s2
            # u where top line = other_max: (other_max - c1)/s1 if s1 != 0
            
            # Let me just try: u=0, u at boundary, and a few "balanced" points
            top2_idx = np.argsort(changed_corr + 0*1)[-3:]  # top 3
            for i in range(len(top2_idx)):
                for jj in range(len(top2_idx)):
                    if i == jj:
                        continue
                    si = slopes[top2_idx[i]]
                    sj = slopes[top2_idx[jj]]
                    if abs(si - sj) > 1e-12:
                        u_cross = (changed_corr[top2_idx[jj]] - changed_corr[top2_idx[i]]) / (si - sj)
                        u_cands.append(u_cross)
            
            # Also add u where top line = other_max
            for idx in range(min(3, len(changed_corr))):
                if abs(slopes[idx]) > 1e-12 and other_max > -np.inf:
                    u_cross_om = (other_max - changed_corr[idx]) / slopes[idx]
                    u_cands.append(u_cross_om)
            
            # Evaluate f at all candidates (clipped to [u_lo, u_hi])
            best_u = 0.0
            best_f = np.inf
            for u in u_cands:
                u = np.clip(u, -old_val, 1.0 - old_val)
                new_corr_vals = changed_corr + u * slopes
                f_val = max(other_max, new_corr_vals.max()) if len(new_corr_vals) > 0 else other_max
                if f_val < best_f:
                    best_f = f_val
                    best_u = u
            
            new_val = old_val + best_u
            if abs(best_u) > 1e-12:
                h[j] = new_val
                # Update corr
                delta = best_u
                corr[m_lo:m_hi+1] += delta * slopes
                if corr.max() * dx < best_c5:
                    best_c5 = corr.max() * dx
                    best_h = h.copy()
                    improved = True
        
        # Check time and whether we're making progress
        if not improved and time.time() - start > time_limit * 0.5:
            break
    
    return best_h, best_c5

def stochastic_refine(h0, time_limit, step_sizes=None):
    """Stochastic local search: randomly perturb coordinates and accept improvements."""
    n = len(h0)
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    iter_count = 0
    no_improve = 0
    
    while time.time() - start < time_limit:
        # Pick a random batch of coordinates
        batch_size = min(n, 200)
        indices = np.random.choice(n, batch_size, replace=False)
        
        # Try perturbing them
        for j in indices:
            step = np.random.choice(step_sizes)
            direction = np.random.choice([-1, 1])
            new_val = h[j] + direction * step
            if new_val < 0 or new_val > 1:
                new_val = np.clip(new_val, 0, 1)
                if new_val == h[j]:
                    continue
            
            # Try this change
            old = h[j]
            h[j] = new_val
            c5_new = compute_c5(h)
            
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                h[j] = old  # revert
                no_improve += 1
        
        iter_count += batch_size
        if no_improve > batch_size * 5:
            # Try smaller steps
            step_sizes = [s * 0.5 for s in step_sizes]
            no_improve = 0
        
        if time.time() - start > time_limit:
            break
    
    return best_h, best_c5
