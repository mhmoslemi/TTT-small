        return np.sum(x) - n / 2.0

def swap_local_search(h, n, max_passes=20, n_neighbors=50, time_limit=60):
    best = h.copy()

    start = time.time()

        if time.time() - start > time_limit:
            break
        ones = np.where(best >= 0.999)[0]
        zeros = np.where(best <= 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            # Not binary enough for swap search
            break
        improved = False
        for i in ones:
            if time.time() - start > time_limit:
                break
            # Try swapping with nearby zeros
            dists = np.abs(zeros - i)
            nearest_idx = np.argsort(dists)[:n_neighbors]
            for ni in nearest_idx:
                j = zeros[ni]
                trial = best.copy()

                trial[j] = 1.0
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best = trial
                    best_c = c
                    improved = True
                    break
            if improved and p % 3 == 0:
                # Don't restart, just continue
                pass
        if not improved:
            break
    return best, best_c

def slsqp_softmax(h, n, T, maxiter, ftol):
    """SLSQP with log-sum-exp soft-max of max correlation."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def constraint(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < obj(h) * 1.001:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h

def coordinate_descent(h, n, n_rounds=10, time_limit=60):
    """Fine coordinate descent with multiple candidate values."""
    best = h.copy()
    best_c = compute_c5(best)
    start = time.time()
    
    for r in range(n_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            old = best[j]
            cands = [0.0, 1.0]
            for d in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]:
                v = old + d
                if 0.0 <= v <= 1.0:
                    cands.append(v)
            best_c_j = best_c
            best_val = old
            for val in cands:
                if abs(val - old) < 1e-10:
                    continue
                trial = best.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_j - 1e-13:
                    best_c_j = c
                    best_val = val
            if best_val != old:
                best[j] = best_val
                best = make_feasible(best, n)
                best_c = best_c_j
                improved = True
        if not improved:
            break
    return best, best_c

    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(320, 0.5)
    
    h_init = make_feasible(h_init, len(h_init))
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Phase 1: Downsample and SLSQP at coarse resolution
    if remaining() > 200:
        h_coarse = downsample_h(h_init, 240)
        h_coarse = slsqp_softmax(h_coarse, 240, T=50, maxiter=300, ftol=1e-12)
        best_h = upsample_h(h_coarse, len(h_init))
        best_c5 = compute_c5(best_h)
    
    # Phase 2: Binary snap at medium resolution
    if remaining() > 150:
        h_med = downsample_h(best_h, 480)
        h_med, c_med = binary_snap(h_med, 480, max_sweeps=8)
        if c_med < best_c5 * 480 / len(h_med):  # normalized comparison
            # Actually just compare directly since dx changes
            pass
        # Compare in C5 terms
        c_med_actual = compute_c5(h_med)
        h_med_up = upsample_h(h_med, len(best_h))
        c_med_up = compute_c5(h_med_up)
        if c_med_up < best_c5:
            best_h = h_med_up
            best_c5 = c_med_up
    
    # Phase 3: SA at medium resolution
    if remaining() > 200:
        h_sa = downsample_h(best_h, 480)
        h_sa, c_sa = sa_binary(h_sa, 480, n_iter=150000, T_start=0.008, T_end=0.0005, seed=seed)
