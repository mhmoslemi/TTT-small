def slsqp_smooth(h, n, T, maxiter, ftol, start_time, budget_s):
    dx = 2.0 / n
    
    def obj(x):
        corr = fast_correlate(x, 1.0 - x)
        mc = np.max(corr)
        # Log-sum-exp for smooth max
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n

                   bounds=bounds,

Actually, let me reconsider the whole approach. The key insight is:

The problem is equivalent to finding a binary sequence of length n with n/2 ones that minimizes the maximum "disagreement" with its shifts. This is related to the concept of "balanced sequences" or "low-discrepancy sequences."

A different angle: think of it as a scheduling problem. We have n time slots and n/2 "active" slots. For each lag k, we count how many active slots at time t have inactive slots at time t+k. We want to minimize the maximum over all lags.

This is very similar to the problem of designing a "balanced" binary sequence with minimal peak autocorrelation (or rather, minimal peak cross-correlation with its complement).

Let me try yet another approach: **Greedy construction with local search**.

1. Start with a random binary sequence with n/2 ones.
2. For each position, determine if flipping it (and compensating elsewhere) reduces the max correlation.
3. Repeat until no improvement is possible.

Actually, this is essentially coordinate descent on the binary variables, which is what the previous code does in its "coord_descent" phase but for continuous variables.

Let me try a different strategy: **Iterated Local Search (ILS)**.
1. Solve the binary problem to local optimality (using SA or direct local search).
2. Perturb the solution (flip k random bits, maintaining the sum constraint).
3. Re-solve to local optimality.
4. Keep the best solution found.

This is a well-known metaheuristic that can escape local optima.

import numpy as np

import time

def fast_correlate(a, b):
    N = len(a) + len(b) - 1
    Np = 1
    while Np < N:
        Np *= 2
    fa = np.fft.rfft(a, Np)
    fb = np.fft.rfft(b[::-1], Np)
    return np.fft.irfft(fa * fb, Np)[:N]

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    correlation = fast_correlate(h, 1.0 - h)
    return float(np.max(correlation)) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_single(b, n, max_iters, T0, T1, rng, start_time, budget_s):

    curr_c5 = compute_c5_fast(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 30:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        
        while True:
            i = rng.randint(n)
            if b[i] == 1:
                break
        while True:
            j = rng.randint(n)
            if b[j] == 0:
                break
        
        b[i] = 0.0
        b[j] = 1.0
        new_c5 = compute_c5_fast(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    
    return best_b, best_c5

def local_search_binary(b, n, rng, start_time, budget_s, max_rounds=50):
    """Direct local search: try all single swaps, keep best."""
    b = b.copy()
    curr_c5 = compute_c5_fast(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    ones = set(np.where(b == 1)[0])
    zeros = set(np.where(b == 0)[0])
    
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 30:
            break
        improved = False
        best_delta = 0
        best_swap = None
        
        # Try a random subset of swaps (all would be O(n^2))
        ones_list = list(ones)
        zeros_list = list(zeros)
        n_tries = min(2000, len(ones_list) * len(zeros_list))
        
        for _ in range(n_tries):
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            b[i] = 0.0
            b[j] = 1.0
            new_c5 = compute_c5_fast(b)
            delta = curr_c5 - new_c5
            if delta > best_delta:
                best_delta = delta
                best_swap = (i, j)
            b[i] = 1.0
            b[j] = 0.0
        
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c5 -= best_delta
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            improved = True
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_b = b.copy()
        
        if not improved:
            break
    
    return best_b, best_c5

def slsqp_smooth(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    
    def obj(x):
        corr = fast_correlate(x, 1.0 - x)
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

import numpy as np
from scipy.optimize import minimize
import time

def fast_correlate(a, b):
    N = len(a) + len(b) - 1
    Np = 1
    while Np < N:
        Np *= 2
    fa = np.fft.rfft(a, Np)
    fb = np.fft.rfft(b[::-1], Np)
    return np.fft.irfft(fa * fb, Np)[:N]

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    correlation = fast_correlate(h, 1.0 - h)
    return float(np.max(correlation)) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_single(b, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = b.copy()
    curr_c5 = compute_c5_fast(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 30:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        while True:
            i = rng.randint(n)
            if b[i] == 1:
                break
        while True:
            j = rng.randint(n)
            if b[j] == 0:
                break
        b[i] = 0.0
        b[j] = 1.0
        new_c5 = compute_c5_fast(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c5

def local_search_binary(b, n, rng, start_time, budget_s, max_rounds=80, tries_per_round=3000):
    b = b.copy()
    curr_c5 = compute_c5_fast(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 30:
            break
        improved = False
        best_delta = 0.0
        best_swap = None
        ones_list = np.where(b == 1)[0]
        zeros_list = np.where(b == 0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        n_tries = min(tries_per_round, len(ones_list) * len(zeros_list))
        for t in range(n_tries):
            if (t % 500 == 0) and (time.time() - start_time > budget_s - 30):
                break
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            b[i] = 0.0
            b[j] = 1.0
            new_c5 = compute_c5_fast(b)
            delta = curr_c5 - new_c5
            if delta > best_delta:
                best_delta = delta
                best_swap = (i, j)
            b[i] = 1.0
            b[j] = 0.0
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c5 -= best_delta
            improved = True
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_b = b.copy()
        if not improved:
            break
    return best_b, best_c5

def slsqp_smooth(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = fast_correlate(x, 1.0 - x)
        mc = float(np.max(corr))
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x
