    dx = 2.0 / n
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    
    while time.time() - start < time_budget:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_curr = corr.max() * dx
        
        if c5_curr < best_c5 - 1e-15:
            best_c5 = c5_curr
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

            # Random perturbation to escape
            idx = np.random.randint(0, n, 50)
            h[idx] += np.random.randn(50) * 0.01
            h = project_to_feasible(h, n)
            stall = 0
            if stall > 5000:
                break
        
        # Find top M shifts
        top_k = np.argpartition(corr, -top_m)[-top_m:]
        top_vals = corr[top_k]
        
        # Compute combined gradient
        G = np.zeros(n)
        for k_idx in top_k:
            w = 1.0  # equal weight; could use softmax
            G += w * shift_grad(h, k_idx, n)

        i_max = np.argmax(G)  # best to decrease h[i]
        j_min = np.argmin(G)  # best to increase h[j]

Hmm, but this still calls np.correlate many times. Let me think about whether there's a way to avoid this.

For a paired move (h[i]-δ, h[j]+δ), the change in correlation at shift k is:
Δcorr[k] = -δ * g_k[i] + δ * g_k[j] = δ * (g_k[j] - g_k[i])

where g_k is the shift gradient at k. So if I precompute g_k for the top M shifts, I can estimate the effect on those M shifts without a full correlate. The risk is that a non-top shift becomes the new max, but for small δ this is unlikely.

So the algorithm becomes:
1. Precompute g_k for top M shifts (M * O(n))
2. For the paired move (i_dec, i_inc, step):
   - For each of the M shifts: estimated_new_corr[k] = corr[k] + step * (g_k[i_inc] - g_k[i_dec])
   - If max of estimated_new_corr < c5_curr - margin: accept
3. Every N accepted moves, do full recompute to verify

This reduces the cost per trial from O(n²) to O(M*n) ≈ O(n) for small M.

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def shift_grad(h, k_idx, n):
    dx = 2.0 / n
    sh = k_idx - n + 1
    grad = np.zeros(n)
    if sh >= 0:
        m = n - sh
        grad[sh:n] += (1.0 - h[0:m])
        grad[0:m] -= h[sh:n]
    else:
        s = -sh
        m = n - s
        grad[0:m] += (1.0 - h[s:n])
        grad[s:n] -= h[0:m]
    return grad * dx

def soft_obj_grad(h, n, tau, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        g = shift_grad(h, top_idx[ii], n)
        grad += p[ii] * g
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr
