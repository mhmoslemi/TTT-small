    """Compute all R_k values"""
    dx = 2.0/n
    one_minus_h = 1.0 - h

def compute_grad_smoothed(h, n, T):
    """Compute gradient of (1/T)*log(sum exp(T*R_k)) approximately"""
    # R values
    R = np.array([R_k(h) for k in range(-(n-1), n)])  # O(n²)
    # Softmax weights
    max_R = np.max(R)
    weights = np.exp(T*(R - max_R))
    weights /= weights.sum()
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if w < 1e-10: continue
        lag = k - (n-1)
        if lag == 0:
            grad += w * (1 - 2*h)
        elif lag > 0:
            for j in range(n - lag):
                grad[j] += w * (1 - h[j+lag])
                grad[j+lag] -= w * h[j]
        else:
            lag = -lag
            for j in range(lag, n):
                grad[j] += w * (1 - h[j-lag])
                grad[j-lag] -= w * h[j]
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if len(free) > 0:
                h[free[0]] = np.clip(h[free[0]] + diff, 0.0, 1.0)
        else:
            free = np.where(h > 0.0)[0]
            if len(free) > 0:

    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    target = n // 2
    idx = np.argsort(h)[::-1]

    hb[idx[:target]] = 1.0

    if rng is None:

    zeros = np.where(h == 0.0)[0]

    while time.time() - start < max_time:
        R = np.correlate(h, 1-h, mode='full')
        max_idx = R.argmax()
        max_val = R[max_idx]
        lag = max_idx - (n-1)
        
        improved = False
        # Try up to 5000 random swaps
        for _ in range(5000):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            
            h[i] = 0.0
            h[j] = 1.0
            new_R = np.correlate(h, 1-h, mode='full')
            new_max = new_R.max()
            
            if new_max < max_val - 1e-15:
                # Accept
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                max_val = new_max
                stall = 0
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if not improved:
            stall += 1
            if stall > 5:
                break
        # Update best
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5

def coord_descent(h, n, max_passes=100, n_grid=50, max_time=300):
    """Continuous coordinate descent with grid search per coordinate"""
    start = time.time()
    dx = 2.0/n
    idx0 = n - 1  # index of lag 0
    
    for p in range(max_passes):
        if time.time() - start > max_time:
            break
        improved = 0
        for i in range(n):
            if time.time() - start > max_time:
                break
            # Compute R at h[i]=0 and h[i]=1
            old = h[i]
            h[i] = 0.0
            R0 = np.correlate(h, 1-h, mode='full').astype(float)
            h[i] = 1.0
            R1 = np.correlate(h, 1-h, mode='full').astype(float)
            
            # For l≠0: linear interpolation
            # For l=0: R_0(x) = R0[idx0] + x*(1-x)
            x_vals = np.linspace(0, 1, n_grid)
            R_lin = R0[np.newaxis,:] + np.outer(x_vals, R1 - R0)
            R_lin[:, idx0] = R0[idx0] + x_vals * (1.0 - x_vals)
            
            max_R = R_lin.max(axis=1)
            best_i = np.argmin(max_R)
            new_val = x_vals[best_i]
            
            if abs(new_val - old) > 1e-6:
                h[i] = new_val
                improved += 1
            else:
                h[i] = old
        
        h = make_feasible(h, n)
        if improved == 0:
            break
    
    return h

def smooth_gradient_step(h, n, T, lr, momentum=None):
    """One step of gradient descent on smoothed max objective"""
    dx = 2.0/n
    R = np.correlate(h, 1-h, mode='full') * dx
    
    # Softmax weights
    max_R = R.max()
    exps = np.exp(T * (R - max_R))
    weights = exps / exps.sum()
    
    # Only use top K weights for efficiency
    K = min(20, len(R))
    top_K = np.argsort(weights)[::-1][:K]
    w_K = weights[top_K]
    w_K = w_K / w_K.sum()  # renormalize
    
    # Compute gradient
    grad = np.zeros(n)
    for k_idx in top_K:
        w = w_K[k_idx]
        lag = k_idx - (n-1)
        if lag == 0:
            grad += w * (1.0 - 2.0*h)
        elif lag > 0:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
        else:
            lp = -lag
            grad[lp:] += w * (1.0 - h[:n-lp])
            grad[:n-lp] -= w * h[lp:]
    
    return grad

def gradient_descent(h, n, max_time=200, T_init=50.0, T_final=500.0):
    """Smoothed gradient descent with annealing"""
    start = time.time()
    dx = 2.0/n
    n_iters = 5000
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(n_iters):
        if time.time() - start > max_time:
            break
        # Anneal temperature
        frac = it / n_iters
        T = T_init * (T_final/T_init)**frac
        
        grad = smooth_gradient_step(h, n, T, 1.0)
        
        # Nesterov momentum
        momentum = beta * momentum - 0.01 * grad
        h = h + momentum
        
        # Project
        h = make_feasible(h, n)
    
    return h

    return make_feasible(h_new, new_n)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_swap_search(h, n, max_time=120, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    start = time.time()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    stall = 0
    temp = 0.01
    while time.time() - start < max_time:
        R = np.correlate(h, 1.0 - h, mode='full')
        max_idx = R.argmax()
        max_val = R[max_idx]
        lag = int(max_idx) - (n - 1)
        improved = False
        # Targeted swaps: use score based on worst lag
        if lag > 0:
            scores = np.zeros(n)
            scores[:n-lag] += (1.0 - h[lag:])
            scores[lag:] += h[:n-lag]
        elif lag < 0:
            lp = -lag
            scores[lp:] += (1.0 - h[:n-lp])
            scores[:n-lp] += h[lp:]
        else:
            scores = 2.0 * h - 1.0
        # Try targeted then random swaps
        try_count = 0
        for _ in range(8000):
            if time.time() - start > max_time:
                break
            if try_count < 2000 and len(ones) > 0 and len(zeros) > 0:
                # Targeted: pick 1 with high score, 0 with low score
                one_scores = scores[ones]
                zero_scores = scores[zeros]
                top_ones = ones[np.argsort(one_scores)[::-1][:20]]
                bot_zeros = zeros[np.argsort(zero_scores)[:20]]
                i = top_ones[rng.randint(len(top_ones))]
                j = bot_zeros[rng.randint(len(bot_zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            try_count += 1
            h[i] = 0.0
            h[j] = 1.0
            new_R = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(new_R.max())
            if new_max < max_val - 1e-14:
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                max_val = new_max
                stall = 0
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            stall += 1
            temp *= 0.8
            if stall > 8:
                break
            # Perturb: accept random worse moves
            for _ in range(100 * stall):
                if len(ones) < 1 or len(zeros) < 1:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_max = float(np.correlate(h, 1.0 - h, mode='full').max())
                if new_max < max_val + temp * n:
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    max_val = new_max
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def coord_descent(h, n, max_passes=80, n_grid=50, max_time=300):
    start = time.time()
    idx0 = n - 1
    x_vals = np.linspace(0.0, 1.0, n_grid)
    for p in range(max_passes):
        if time.time() - start > max_time:
            break
        improved = 0
        for i in range(n):
            if time.time() - start > max_time:
                break
            old = h[i]
            h[i] = 0.0
            R0 = np.correlate(h, 1.0 - h, mode='full').astype(float)
            h[i] = 1.0
            R1 = np.correlate(h, 1.0 - h, mode='full').astype(float)
            h[i] = old
            R_lin = R0[np.newaxis, :] + np.outer(x_vals, R1 - R0)
            R_lin[:, idx0] = R0[idx0] + x_vals * (1.0 - x_vals)
            max_R = R_lin.max(axis=1)
            best_idx = int(np.argmin(max_R))
            new_val = x_vals[best_idx]
            if abs(new_val - old) > 1e-8:
                h[i] = new_val
                improved += 1
        h = make_feasible(h, n)
        if improved == 0:
            break
    return h

def smooth_grad_step(h, n, T):
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full') * dx
    max_R = R.max()
    exps = np.exp(T * (R - max_R))
    s = exps.sum()
    if s < 1e-300:
        weights = np.zeros_like(R)
        weights[R.argmax()] = 1.0
    else:
        weights = exps / s
    K = min(30, len(R))
    top_K_idx = np.argsort(weights)[::-1][:K]
    w_K = weights[top_K_idx]
    w_K = w_K / w_K.sum()
    grad = np.zeros(n)
    for w, k_idx in zip(w_K, top_K_idx):
        lag = int(k_idx) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
        else:
            lp = -lag
            grad[lp:] += w * (1.0 - h[:n-lp])
            grad[:n-lp] -= w * h[lp:]
    return grad

def gradient_descent(h, n, max_time=200):
    start = time.time()
    n_iters = 8000
    momentum = np.zeros(n)
    beta = 0.85
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        if time.time() - start > max_time:
            break
        frac = it / n_iters
        T = 30.0 * (3000.0 / 30.0) ** frac
        lr = 0.02 / (1.0 + 0.01 * it)
        grad = smooth_grad_step(h, n, T)
        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h_new = make_feasible(h_new, n)
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Initialize from parent or fallback
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.tile([1.0, 0.0], 320)
    n = len(h)
    h = make_feasible(h, n)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Binary targeted local search
    if time_left() > 130:
        hb = to_binary(h, n)
        c5b_init = compute_c5(hb)
        if c5b_init < best_c5:
            best_c5 = c5b_init
            best_h = hb.copy()
        ht = to_binary(max(best_h, hb, axis=0), n)
        ht, c5t = binary_swap_search(ht, n, max_time=min(120, time_left() - 10), rng=rng)
        if c5t < best_c5:
            best_c5 = c5t
            best_h = ht.copy()

    # Phase 2: Continuous coordinate descent
    if time_left() > 220:
        h_cd = best_h.copy()
        # Blend in some continuous values for a better starting point
        h_cd = 0.7 * h_cd + 0.3 * (0.5 + 0.1 * np.sin(np.pi * np.arange(n) / n * 3.0))
        h_cd = make_feasible(h_cd, n)
        h_cd = coord_descent(h_cd, n, max_passes=60, n_grid=50, max_time=min(260, time_left() - 20))
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        # Also try CD from binary
        if time_left() > 100:
            hb2 = to_binary(best_h, n)
            h_cd2 = hb2.copy()
            h_cd2 = coord_descent(h_cd2, n, max_passes=40, n_grid=40, max_time=min(90, time_left() - 20))
            c5_cd2 = compute_c5(h_cd2)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    # Phase 3: Smoothed gradient descent
    if time_left() > 120:
        h_gd = best_h.copy()
        h_gd, c5_gd = gradient_descent(h_gd, n, max_time=min(180, time_left() - 20))
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()
        # Second GD round from best
        if time_left() > 80:
            h_gd2 = best_h.copy()
            h_gd2, c5_gd2 = gradient_descent(h_gd2, n, max_time=min(70, time_left() - 15))
            if c5_gd2 < best_c5:
                best_c5 = c5_gd2
                best_h = h_gd2.copy()

    # Phase 4: Coordinate descent again for final polish
    if time_left() > 60:
        h_pol = best_h.copy()
        h_pol = coord_descent(h_pol, n, max_passes=30, n_grid=60, max_time=min(50, time_left() - 10))
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Try higher resolution
    if time_left() > 80:
        n2 = min(2 * n, 1280)
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        else:
            h2 = best_h.copy()  # stay at lower res if upsampled is worse
        if time_left() > 40:
            h2 = coord_descent(h2, n2, max_passes=15, n_grid=30, max_time=min(35, time_left() - 10))
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Final: one more binary search from best
    if time_left() > 50:
        hb_final = to_binary(best_h, len(best_h))
        hf = len(hb_final)
        hb_final, c5bf = binary_swap_search(hb_final, hf, max_time=min(40, time_left() - 10), rng=rng)
        if c5bf < best_c5:
            best_c5 = c5bf
            best_h = hb_final.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
