import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lse_objective(h, n, T, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -100.0, 100.0)
    soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    viol = np.sum(h) - n / 2.0
    return soft + mu * viol * viol

def lbgfb_run(h_start, n, T, mu, max_iter):
    def f(h):
        return lse_objective(h, n, T, mu)
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

            try:

                c5_tp = compute_c5(h_tp)
                if c5_tp < best_c5:
                    best_c5 = c5_tp
                    best_h = h_tp.copy()
            except Exception:
                break

    # Phase 5: Upsample to n2=1280 and optimize
    n2 = 1280
    if remaining() > 200:
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        temps2 = [50.0, 200.0, 800.0, 3000.0, 10000.0]
        for T in temps2:
            if remaining() < 120:
                break
            for rest in range(2):
                if remaining() < 110:
                    break
                if rest == 0:
                    h_init = best_h.copy() if len(best_h) == n2 else upsample_h(best_h, n2)
                else:
                    h_init = make_feasible(upsample_h(best_h, n2) + 0.08 * rng.randn(n2), n2)
                try:
                    h_opt = lbgfb_run(h_init, n2, T, 500.0, 400)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                except Exception:
                    pass

    # Phase 6: Binary LS at n2
    if remaining() > 100:
        h_bin2 = (best_h > 0.5).astype(float).copy()
        n_ones = int(np.sum(h_bin2 > 0.5))
        n_target = n2 // 2
        if n_ones > n_target:
            one_idx = np.where(h_bin2 > 0.5)[0]
            perm = rng.permutation(len(one_idx))
            for k in range(n_ones - n_target):
                h_bin2[int(one_idx[perm[k]])] = 0.0
        elif n_ones < n_target:
            zero_idx = np.where(h_bin2 <= 0.5)[0]
            perm = rng.permutation(len(zero_idx))
            for k in range(n_target - n_ones):
                h_bin2[int(zero_idx[perm[k]])] = 1.0
        h_bl2, c5_bl2 = binary_ls(h_bin2, n2, rng, 60, 2000)
        if c5_bl2 < best_c5:
            best_c5 = c5_bl2
            best_h = h_bl2.copy()

    # Phase 7: Coordinate descent
    if remaining() > 40:
        n_cd = min(len(best_h), 1280)
        h_cd = best_h.copy() if n_cd == len(best_h) else upsample_h(best_h, n_cd)
        h_cd, c5_cd = coord_descent(h_cd, n_cd, rng, max_rounds=4)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: SLSQP final polish
    if remaining() > 25:
        n_f = len(best_h)
        try:
            h_fp = slsqp_polish(best_h, n_f, 5000.0, max_iter=80)
            c5_fp = compute_c5(h_fp)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()
        except Exception:
            pass

    # Final feasibility and evaluation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
