def fft_correlate(a, b):
    """Compute full correlation using FFT, matching np.correlate(a, b, 'full')"""
    n = len(a)
    m = len(b)
    size = n + m - 1
    # Pad to next power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    corr = np.fft.irfft(A * B.conj(), fft_size)[:size]
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_optimize(h_start, n, T, mu, max_iter=300, ftol=1e-18):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -500, 500)
        # Numerically stable log-sum-exp
        log_sum = mc + np.log(np.sum(np.exp(shifted))) / T
        soft = log_sum * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-14,
                            'maxls': 50, 'maxfun': 500 * n})

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h) if diff > 0 else np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def corr_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g[:] += (1.0 - h)
            g[:] += -h
        else:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] += -h[k:]
    else:
        kp = -k
        g[:n - kp] += (1.0 - h[kp:])
        g[kp:] += -h[:n - kp]
    return g

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -400, 400)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40})
    return make_feasible(res.x, n)

def subgradient_multi(h, n, max_iter=500, lr_start=0.04, n_active=3):
    h = h.copy()
    current = compute_c5(h)
    lr = lr_start
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        g_total = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            g_total += corr_gradient(h, k)
        g_total -= np.mean(g_total)
        g_norm = np.linalg.norm(g_total)
        if g_norm < 1e-14:
            break
        h_new = make_feasible(h - lr * g_total / g_norm * 0.1, n)
        c = compute_c5(h_new)
        if c < current - 1e-16:
            h = h_new
            current = c
            lr = min(lr * 1.03, 2.0)
        else:
            lr *= 0.6
        if lr < 1e-10:
            break
    return h, current

def periodic_optimize(h_init, n, period, T, max_iter=100):
    n_rep = n // period
    if n_rep * period != n:
        return None
    pattern = h_init[:period].copy()
    dx = 2.0 / n
    target_sum = period / 2.0
    mu_p = 1000.0
    def obj(pat):
        h_full = np.tile(pat, n_rep)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -300, 300)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        return soft + mu_p * (np.sum(pat) - target_sum) ** 2
    bounds = [(0.0, 1.0)] * period
    try:
        res = minimize(obj, pattern, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_full = np.tile(res.x, n_rep)
        return make_feasible(h_full, n)
    except Exception:
        return None

def binary_pair_search(h, n, rng, max_rounds=150, tries_per_round=1500):
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        for i in range(n_ones - target):
            h[idx[i]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        for i in range(target - n_ones):
            h[idx[i]] = 1.0
    current = compute_c5(h)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        ntries = min(tries_per_round, len(ones) * len(zeros))
        for _ in range(ntries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi = ones[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def fine_local_search(h, n, rng, max_rounds=50):
    h = h.copy()
    current = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        top3 = np.argsort(corr)[-3:]
        g_acc = np.zeros(n)
        for idx in top3:
            k = int(idx) - (n - 1)
            g_acc += corr_gradient(h, k)
        g_acc -= np.mean(g_acc)
        top_idx = np.argsort(np.abs(g_acc))[-80:]
        rng.shuffle(top_idx)
        for j in top_idx:
            for delta in [0.015, -0.015, 0.008, -0.008, 0.003, -0.003]:
                h_try = h.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                if abs(delta) < 1e-12:
                    continue
                comp = int(np.argmin(h_try)) if delta > 0 else int(np.argmax(h_try))
                if comp == j:

                c = compute_c5(h_try)
                if c < current - 1e-15:
                    h = h_try
                    current = c
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, current
