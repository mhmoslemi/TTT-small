        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def get_rem(start, budget):
    return max(0.0, budget - (time.time() - start))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Coarse global at n=400
    n_c = 400
    h_c = interp_periodic(h_init, n_c)
    h_c = project_to_feasible(h_c, n_c)
    rem = get_rem(start_time, budget_s)
    if rem > 180:
        h_c, c5 = smooth_optimize(h_c, n_c, tau=80.0, time_limit=min(60, rem * 0.12), top_k=20, lr=0.06)
        h_cu = interp_periodic(h_c, n_init)
        h_cu = project_to_feasible(h_cu, n_init)
        if compute_c5(h_cu) < best_c5:
            best_c5 = compute_c5(h_cu)
            best_h = h_cu.copy()

    # Phase 2: Medium at n=1500
    n_m = 1500
    h_m = interp_periodic(best_h, n_m)
    h_m = project_to_feasible(h_m, n_m)
    rem = get_rem(start_time, budget_s)
    if rem > 200:
        h_m, c5 = smooth_optimize(h_m, n_m, tau=150.0, time_limit=min(90, rem * 0.18), top_k=15, lr=0.08)
        h_mu = interp_periodic(h_m, n_init)
        h_mu = project_to_feasible(h_mu, n_init)
        cm = compute_c5(h_mu)
        if cm < best_c5:
            best_c5 = cm
            best_h = h_mu.copy()

    # Phase 3: Fine at native resolution with tau annealing
    h_f = project_to_feasible(best_h.copy(), n_init)
    for tau, tk, lr_v in [(250, 14, 0.10), (100, 10, 0.05), (40, 8, 0.025)]:
        rem = get_rem(start_time, budget_s)
        if rem < 130:
            break
        tl = min(100, rem * 0.17)
        h_f, c5 = smooth_optimize(h_f, n_init, tau=tau, time_limit=tl, top_k=tk, lr=lr_v)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f.copy()

    # Phase 4: Multi-start with structured perturbations
    rem = get_rem(start_time, budget_s)
    rng = np.random.RandomState(seed + 1)
    for trial in range(5):
        rem = get_rem(start_time, budget_s)
        if rem < 80:
            break
        if trial < 2:
            h_p = best_h.copy() + rng.randn(n_init) * 0.008
        elif trial < 4:
            h_p = best_h.copy()
            idx = rng.choice(n_init, size=n_init // 10, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        else:
            sh = rng.randint(1, 5)
            h_p = np.roll(best_h, sh)
        h_p = project_to_feasible(h_p, n_init)
        tl = min(55, rem * 0.17)
        h_p, c5_p = smooth_optimize(h_p, n_init, tau=50.0, time_limit=tl, top_k=10, lr=0.04)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 5: Low-tau Nesterov refinement
    rem = get_rem(start_time, budget_s)
    if rem > 70:
        tl = min(55, rem * 0.3)
        h_f, c5 = smooth_optimize(best_h, n_init, tau=12.0, time_limit=tl, top_k=6, lr=0.012)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f.copy()

    # Phase 6: Subgradient polish (moderate)
    rem = get_rem(start_time, budget_s)
    if rem > 50:
        tl = min(40, rem * 0.35)
        h_s, c5_s = subgrad_optimize(best_h, n_init, time_limit=tl, top_k=5, lr=0.0015)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 7: Very tight final subgradient
    rem = get_rem(start_time, budget_s)
    if rem > 20:
        tl = min(18, rem * 0.6)
        h_s, c5_s = subgrad_optimize(best_h, n_init, time_limit=tl, top_k=3, lr=0.0003)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 8: Ultra-fine polish
    rem = get_rem(start_time, budget_s)
    if rem > 10:
        tl = min(8, rem * 0.7)
        h_s, c5_s = subgrad_optimize(best_h, n_init, time_limit=tl, top_k=2, lr=0.00008)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    best_h = project_to_feasible(best_h, n_init)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, n_init
