                best_h, best_c5 = pair_swap_optimize(best_h, len(best_h), best_c5, rng, 40)
            # Coord descent at higher res
            if rem() > 30:
                best_h, best_c5, _ = coord_descent(best_h, len(best_h), best_c5,
                                                    steps=[0.001, 0.003, 0.01])

    # Phase 6: Final ultra-fine coordinate descent
    n_f = len(best_h)
    for p in range(3):
        if rem() < 8:
            break
        for j in range(n_f):
            if rem() < 3:
                break
            old = best_h[j]
            best_v = old
            best_c = best_c5
            for step in [0.0003, 0.0007, 0.0015, 0.004]:
                for s in [-1.0, 1.0]:
                    v = old + s * step
                    if 0.0 <= v <= 1.0 and abs(v - old) > 1e-10:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n_f)
                        c = compute_c5(trial)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_v = v
            for v in [0.0, 1.0]:
                if abs(v - old) < 1e-8:
                    continue
                trial = best_h.copy()
                trial[j] = v
                trial = make_feasible(trial, n_f)
                c = compute_c5(trial)
                if c < best_c - 1e-16:
                    best_c = c
                    best_v = v
            if best_v != old:
                best_h[j] = best_v
                best_h = make_feasible(best_h, n_f)
                best_c5 = best_c

    # Phase 7: Last-chance SLSQP with very high T
    if rem() > 50:
        _G_N = len(best_h)
        _G_T = 5000.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * len(best_h)
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj_full, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-18})
            c5r = compute_c5(res.x)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = make_feasible(res.x, len(best_h))
        except Exception:
            pass

    # Phase 8: Random perturb + quick optimize
    for rt in range(4):
        if rem() < 45:
            break
        n_f = len(best_h)
        scale = 0.01 + 0.01 * rt
        h_try = make_feasible(best_h + rng.randn(n_f) * scale, n_f)
        _G_N = n_f
        _G_T = 500.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n_f
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj_full, h_try, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-17})
            c5r = compute_c5(res.x)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = make_feasible(res.x, n_f)
                # Quick coord descent after improvement
                best_h, best_c5, _ = coord_descent(best_h, n_f, best_c5,
                                                    steps=[0.001, 0.003, 0.01])
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
