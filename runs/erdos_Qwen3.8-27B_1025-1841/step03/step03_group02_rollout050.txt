        mc = np.max(corr)

        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

    N = len(x)
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    logits = T * (corr - mc)
    logits -= np.max(logits)  # numerical stability
    w = np.exp(logits)
    w /= w.sum()
    
    grad = np.zeros(N)
    for m in range(2*N - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        # First term: (1-h[N-1-m+j]) for j in [max(0, m-N+1), min(N-1, m)]
        j_lo1 = max(0, m - N + 1)
        j_hi1 = min(N - 1, m)
        if j_lo1 <= j_hi1:
            idx = j_lo1 + np.arange(j_hi1 - j_lo1 + 1)
            h_idx = N - 1 - m + idx
            grad[idx] += wm * (1 - x[h_idx])
        
        # Second term: -x[m-N+1+j] for j in [max(0, N-1-m), min(N-1, 2*N-2-m)]
        j_lo2 = max(0, N - 1 - m)
        j_hi2 = min(N - 1, 2*N - 2 - m)
        if j_lo2 <= j_hi2:
            idx = j_lo2 + np.arange(j_hi2 - j_lo2 + 1)
            h_idx = m - N + 1 + idx
            grad[idx] -= wm * x[h_idx]
    
    return grad * dx

def slsqp_with_grad(h, n, T, maxiter):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T*(corr-mc))))/T) * dx
    
    def grad(x):
        # Analytical gradient as derived above
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        logits = T * (corr - mc)
        logits -= np.max(logits)
        w = np.exp(logits)
        w /= w.sum()
        
        g = np.zeros(n)
        for m in range(2*n-1):
            wm = w[m]
            if wm < 1e-15:
                continue
            # First term: (1-x[N-1-m+j]) for j in [max(0,m-N+1), min(N-1,m)]
            j1_lo = max(0, m - n + 1)
            j1_hi = min(n - 1, m)
            if j1_lo <= j1_hi:
                idx = np.arange(j1_lo, j1_hi + 1)
                hidx = n - 1 - m + idx
                g[idx] += wm * (1.0 - x[hidx])
            
            # Second term: -x[m-N+1+j] for j in [max(0,N-1-m), min(N-1,2N-2-m)]
            j2_lo = max(0, n - 1 - m)
            j2_hi = min(n - 1, 2*n - 2 - m)
            if j2_lo <= j2_hi:
                idx = np.arange(j2_lo, j2_hi + 1)
                hidx = m - n + 1 + idx
                g[idx] -= wm * x[hidx]
        
        return g * dx
    
    def con_fun(x):
        return np.sum(x) - n/2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    res = minimize(obj, h.copy(), method='SLSQP', jac=grad,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x
