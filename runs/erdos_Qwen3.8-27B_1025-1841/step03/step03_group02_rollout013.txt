        h = h_new
        it += 1
        if it % 500 == 0 and verbose:
            print(f"  PG iter {it}, T={T:.1f}, c5={best_c5:.6f}")
    return best_h, best_c5

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def anti_peak_sa(b, n, max_iters, T0, T1, start_time, budget_s, rng):
    dx = 2.0 / n
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 60:
            break
        frac = it / max(1, max_iters)
        T = T0 * (T1 / T0) ** frac
        
        corr = np.correlate(b, 1.0 - b, mode='full')
        peak_idx = np.argmax(corr)
        k_peak = peak_idx - (n - 1)
        
        # Find contributors to peak
        if k_peak >= 0:
            i_range = range(0, n - k_peak)
        else:
            i_range = range(-k_peak, n)
        
        contrib = []
        b_arr = b
        if k_peak >= 0:
            for i in range(0, n - k_peak):
                if b_arr[i] == 1 and b_arr[i + k_peak] == 0:
                    contrib.append(i)
        else:
            for i in range(-k_peak, n):
                if b_arr[i] == 1 and b_arr[i - k_peak] == 0:
                    contrib.append(i)
        
        if contrib and rng.random() < 0.7:
            i = contrib[rng.randint(len(contrib))]
            if k_peak >= 0:
                j = i + k_peak
            else:
                j = i - k_peak
            if b[i] == 1 and b[j] == 0:
                b[i] = 0
                b[j] = 1
            else:
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
                i2 = ones[rng.randint(len(ones))]
                j2 = zeros[rng.randint(len(zeros))]
                b[i2] = 0
                b[j2] = 1
                i, j = i2, j2
        else:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
        
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def block_sa(b, n, max_iters, T0, T1, start_time, budget_s, rng):
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 60:
            break
        frac = it / max(1, max_iters)
        T = T0 * (T1 / T0) ** frac
        bs = rng.randint(2, 7)
        start = rng.randint(0, n - bs)
        seg = b[start:start+bs]
        ones = np.where(seg == 1)[0]
        zeros = np.where(seg == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        sc = min(len(ones), len(zeros))
        co = rng.choice(ones, size=sc, replace=False)
        cz = rng.choice(zeros, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best_c5

def coord_descent(h, n, max_rounds, start_time, budget_s, coarse=True):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    steps = [0.0, 1.0, 0.05, 0.01, -0.05, -0.01] if coarse else [0.0, 1.0, 0.005, -0.005]
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        if time.time() - start_time > budget_s - 20:
            break
        for j in range(n):
            if time.time() - start_time > budget_s - 15:
                return h, best_c5
            old = h[j]
            best_c = best_c5
            best_val = old
            for s in steps:
                cand = old + s
                if cand < 0 or cand > 1 or abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = project_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_sum(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum, 'jac': slsqp_sum_jac}]
    def obj(x):
        return slsqp_obj(x, n, T)
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    # Phase 1: Projected gradient (low T for broad search)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        h1, c5_1 = proj_grad_descent(h_init, n0, 15.0, 80.0, 6000, start_time, budget_s)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # Phase 2: Anti-peak SA on binarized solution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        b0 = to_binary(best_h, n0)
        b1, c5_b1 = anti_peak_sa(b0, n0, 30000, 0.008, 0.00005, start_time, budget_s, rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = b1.copy()
    
    # Phase 3: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        b2, c5_b2 = block_sa(to_binary(best_h, n0), n0, 20000, 0.006, 0.00005, start_time, budget_s, rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = b2.copy()
    
    # Phase 4: Continuous polish (higher T for peak focus)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        h2, c5_2 = proj_grad_descent(best_h, n0, 50.0, 300.0, 5000, start_time, budget_s)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # Phase 5: Another SA round from improved continuous
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        b3 = to_binary(best_h, n0)
        b4, c5_b4 = anti_peak_sa(b3, n0, 25000, 0.005, 0.00003, start_time, budget_s, rng)
        if c5_b4 < best_c5:
            best_c5 = c5_b4
            best_h = b4.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 150:
            h3, c5_3 = proj_grad_descent(best_h, n0, 100.0, 500.0, 4000, start_time, budget_s)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
    
    # Phase 6: Try higher resolution n=1920
    remaining = budget_s - (time.time() - start_time)
    if remaining > 600:
        n1 = 1920
        h_up = upsample_h(best_h, n1)
        h_up, c5_up = proj_grad_descent(h_up, n1, 20.0, 150.0, 4000, start_time, budget_s)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 200:
            b_up = to_binary(best_h, n1)
            b_up, c5_bu = anti_peak_sa(b_up, n1, 15000, 0.005, 0.00003, start_time, budget_s, rng)
            if c5_bu < best_c5:
                best_c5 = c5_bu
                best_h = b_up.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 150:
                h_up2, c5_u2 = proj_grad_descent(best_h, n1, 100.0, 400.0, 3000, start_time, budget_s)
                if c5_u2 < best_c5:
                    best_c5 = c5_u2
                    best_h = h_up2.copy()
    
    # Phase 7: SLSQP polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        n_pts = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_pts, 200.0, 300, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass
    
    # Phase 8: Fine coordinate descent
    n_pts = len(best_h)
    best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 20, start_time, budget_s, coarse=False)
    
    # Phase 9: Final SLSQP with very high T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_pts = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_pts, 500.0, 200, 1e-16)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass
    
    # Phase 10: Final micro coordinate descent
    n_pts = len(best_h)
    best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 10, start_time, budget_s, coarse=False)
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
