import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista(h, n_iter, T0, T1, lr, check=30, momentum=True):

    bh = x.copy()

    if c < bc:

Let me also think about the problem structure more carefully.

The correlation at lag k (where corr index is k + n - 1) is:
- For k ≥ 0: corr[k+n-1] = Σ_{i=0}^{n-1-k} h[i+k] * (1-h[i])
- For k < 0: corr[k+n-1] = Σ_{i=0}^{n-1+k} h[i] * (1-h[i-k])

The C5 = dx * max_k corr[k+n-1]

At the optimum, we want to minimize the maximum over all lags. This is a min-max problem.

Key insight: The "active" lags are those where corr is close to the max. At the optimum, we want to spread the "mass" of correlations more evenly.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    c = full_corr(h)
    mc = float(np.max(c))
    exps = np.exp(T * (c - mc))
    tot = float(np.sum(exps))
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(c)) - (n - 1)) * dx
    w = exps / tot
    g = np.zeros(n)
    for i in np.argsort(w)[::-1][:top_k]:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def smooth_val(h, T):
    n = len(h)
    dx = 2.0 / n
    c = full_corr(h)
    mc = float(np.max(c))
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def fista(h, n_iter, T0, T1, lr, check=30):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.5 * f)
        g = smooth_grad(z, T)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

_lb_keys = {'T': 0, 'n': 0, 'lam': 0}

def _lb_f(h):
    T = _lb_keys['T']; n = _lb_keys['n']; lam = _lb_keys['lam']
    c = full_corr(h)
    mc = float(np.max(c))
    val = (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * (2.0 / n)
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def _lb_g(h):
    T = _lb_keys['T']; n = _lb_keys['n']; lam = _lb_keys['lam']
    g = smooth_grad(h, T)
    g = g + 2.0 * lam * (np.sum(h) - n / 2.0)
    return g

def lbfgs_smooth(h, T=5.0, lam=1000.0, maxiter=300):
    n = len(h)
    _lb_keys['T'] = T; _lb_keys['n'] = n; _lb_keys['lam'] = lam
    try:
        r = minimize(_lb_f, h, jac=_lb_g, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                              'maxls': 20})
        hx = project_c(r.x, n)
        if compute_c5(hx) < compute_c5(h):
            return hx
    except Exception:
        pass
    return h

_sq_keys = {'T': 0, 'n': 0}

def _sq_f(h):
    return smooth_val(h, _sq_keys['T'])

def _sq_g(h):
    return smooth_grad(h, _sq_keys['T'])

def _sq_con_f(h):
    return np.sum(h) - _sq_keys['n'] / 2.0

def _sq_con_g(h):
    return np.ones(_sq_keys['n'])

def slsqp_smooth(h, T=3.0, maxiter=400):
    n = len(h)
    _sq_keys['T'] = T; _sq_keys['n'] = n
    try:
        r = minimize(_sq_f, h, jac=_sq_g, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sq_con_f, 'jac': _sq_con_g}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hx = project_c(r.x, n)
        if compute_c5(hx) < compute_c5(h):
            return hx
    except Exception:
        pass
    return h

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])

    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

        if c < bc:
            bc = c
        else:
            bh[i] = 1.0; bh[j] = 0.0
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    c = full_corr(h)
    mc = float(np.max(c))
    exps = np.exp(T * (c - mc))
    tot = float(np.sum(exps))
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(c)) - (n - 1)) * dx
    w = exps / tot
    g = np.zeros(n)
    for i in np.argsort(w)[::-1][:top_k]:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def smooth_val(h, T):
    n = len(h)
    dx = 2.0 / n
    c = full_corr(h)
    mc = float(np.max(c))
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def fista(h, n_iter, T0, T1, lr, check=30):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.5 * f)
        g = smooth_grad(z, T)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

_lb_st = {}

def _lb_f(h):
    T = _lb_st['T']; n = _lb_st['n']; lam = _lb_st['lam']
    c = full_corr(h)
    mc = float(np.max(c))
    val = (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * (2.0 / n)
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def _lb_g(h):
    T = _lb_st['T']; n = _lb_st['n']; lam = _lb_st['lam']
    g = smooth_grad(h, T)
    return g + 2.0 * lam * (np.sum(h) - n / 2.0)

def lbfgs_smooth(h, T=5.0, lam=1000.0, maxiter=300):
    n = len(h)
    _lb_st['T'] = T; _lb_st['n'] = n; _lb_st['lam'] = lam
    try:
        r = minimize(_lb_f, h, jac=_lb_g, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 20})
        hx = project_c(r.x, n)
        if compute_c5(hx) < compute_c5(h):
            return hx
    except Exception:
        pass
    return h

_sq_st = {}

def _sq_f(h):
    return smooth_val(h, _sq_st['T'])

def _sq_g(h):
    return smooth_grad(h, _sq_st['T'])

def _sq_cnf(h):
    return np.sum(h) - _sq_st['n'] / 2.0

def _sq_cng(h):

def slsqp_smooth(h, T=3.0, maxiter=400):
    n = len(h)
    _sq_st['T'] = T; _sq_st['n'] = n
    try:

                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sq_cnf, 'jac': _sq_cng}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hx = project_c(r.x, n)
        if compute_c5(hx) < compute_c5(h):
            return hx
    except Exception:
        pass
    return h
