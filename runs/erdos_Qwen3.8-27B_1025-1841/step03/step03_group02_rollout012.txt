    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = T * (corr - mc)

    w = exp_s / exp_s.sum()
    obj = (mc + np.log(exp_s.sum()) / T) * dx
    
    grad = np.zeros(n)
    shifts = np.arange(len(corr)) - (n - 1)

        wk = w[k]

        best_h = h.copy()
    return best_h, best_c5

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while b.sum() > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while b.sum() < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = to_binary(h, n)
    dx = 2.0 / n
    curr_corr = np.correlate(b, 1.0 - b, mode='full')
    curr_max = float(np.max(curr_corr) * dx)
    best_c5 = curr_max
    best_b = b.copy()
    for it in range(max_iters):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 15:
            break
        frac = float(it) / max(max_iters - 1, 1)
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_corr = np.correlate(b, 1.0 - b, mode='full')
        new_max = float(np.max(new_corr) * dx)
        if new_max <= curr_max or (T > 1e-12 and rng.random() < np.exp(-(new_max - curr_max) / T)):
            curr_max = new_max
            curr_corr = new_corr
            if new_max < best_c5:
                best_c5 = new_max
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def binary_sa_multiblock(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = to_binary(h, n)
    dx = 2.0 / n
    curr_max = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
    best_c5 = curr_max
    best_b = b.copy()
    for it in range(max_iters):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 15:
            break
        frac = float(it) / max(max_iters - 1, 1)
        T = T0 * (T1 / T0) ** frac
        r = rng.random()
        if r < 0.6:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
            new_max = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
            if not (new_max <= curr_max or (T > 1e-12 and rng.random() < np.exp(-(new_max - curr_max) / T))):
                b[i] = 1
                b[j] = 0
                continue
        else:
            bs = rng.randint(2, 8)
            start = rng.randint(0, n - bs)
            end = start + bs
            seg = b[start:end].copy()
            ones_in = np.where(seg == 1)[0]
            zeros_in = np.where(seg == 0)[0]
            if len(ones_in) == 0 or len(zeros_in) == 0:
                continue
            k_swap = min(len(ones_in), len(zeros_in), rng.randint(1, min(4, len(ones_in), len(zeros_in)) + 1))
            ci = rng.choice(ones_in, size=k_swap, replace=False)
            cj = rng.choice(zeros_in, size=k_swap, replace=False)
            b[start + ci] = 0
            b[start + cj] = 1
            new_max = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
            if not (new_max <= curr_max or (T > 1e-12 and rng.random() < np.exp(-(new_max - curr_max) / T))):
                b[start + ci] = 1
                b[start + cj] = 0
                continue
        if new_max < best_c5:
            best_c5 = new_max
            best_b = b.copy()
        curr_max = new_max
    return best_b, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Binary SA for structural improvement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        b, c5b = binary_sa(h_init, n, 30000, 0.01, 0.00001, rng, start_time, budget_s)
        if c5b < best_c5:
            best_c5, best_h = c5b, b.copy()

    # Phase 2: Projected gradient continuous optimization
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        hp, c5p = projected_gradient(best_h.copy(), n, 20.0, 150.0, 4000, start_time, budget_s)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 3: Multi-block binary SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        b2, c5b2 = binary_sa_multiblock(best_h, n, 35000, 0.006, 0.00001, rng, start_time, budget_s)
        if c5b2 < best_c5:
            best_c5, best_h = c5b2, b2.copy()

    # Phase 4: Second projected gradient with higher T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        hp2, c5p2 = projected_gradient(best_h.copy(), n, 60.0, 400.0, 3000, start_time, budget_s)
        if c5p2 < best_c5:
            best_c5, best_h = c5p2, hp2.copy()

    # Phase 5: Final binary SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        b3, c5b3 = binary_sa(best_h, n, 30000, 0.003, 0.000005, rng, start_time, budget_s)
        if c5b3 < best_c5:
            best_c5, best_h = c5b3, b3.copy()

    # Phase 6: Final projected gradient polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        hp3, c5p3 = projected_gradient(best_h.copy(), n, 150.0, 1000.0, 2000, start_time, budget_s)
        if c5p3 < best_c5:
            best_c5, best_h = c5p3, hp3.copy()

    # Phase 7: ULTRA binary SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        b4, c5b4 = binary_sa(best_h, n, 20000, 0.0015, 0.000003, rng, start_time, budget_s)
        if c5b4 < best_c5:
            best_c5, best_h = c5b4, b4.copy()

    # Phase 8: Final check and feasibility
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
