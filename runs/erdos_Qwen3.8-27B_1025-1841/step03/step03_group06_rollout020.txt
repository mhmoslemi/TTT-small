import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_DZ(h, n):
    DZ = np.zeros((n, 2*n-1))
    for i in range(n):
        DZ[i, n-1-i : 2*n-1-i] = -h
        DZ[i, i : i+n] += 1.0 - h[::-1]
    return DZ

def _smooth_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0/n)

def _con_fun(x):
    return np.sum(x) - _GLOBAL_N / 2.0

def _con_jac(x):
    return np.ones(_GLOBAL_N)

    best_h = h_bin.copy()
