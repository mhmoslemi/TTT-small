import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def smooth_max_obj(h, n, T, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -80, 80)
    soft = (m + np.log(np.sum(np.exp(shifted))) / T) * dx
    viol = np.sum(h) - n / 2.0
    return soft + mu * viol * viol

def smooth_max_grad(h, n, T, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -80, 80)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp  # softmax weights over the correlation values
    
    # corr has length 2n-1, indexed 0 to 2n-2
    # corr[n-1+k] corresponds to shift k (in our convention)
    # c_k = sum_{i=max(0,-k)}^{min(n-1, n-1-k)} h[min(n-1, i+k)] * (1-h[i])
    # Actually, let me use the definition:
    # np.correlate(h, 1-h, 'full')[j] = sum_i h[i] * (1-h[i-(j-(n-1))])
    # Let k = j - (n-1), then:
    # corr[j] = sum_i h[i] * (1-h[i-k]) for valid i
    
    # So corr[n-1+k] = sum_i h[i] * (1-h[i-k]) = sum_i h[i+k] * (1-h[i])
    # (reindexing with i -> i+k)
    
    # ∂corr[n-1+k]/∂h[j]:
    # corr[n-1+k] = sum_{i: 0<=i<n, 0<=i+k<n} h[i+k](1-h[i])
    # = sum_{i: max(0,-k)<=i<min(n, n-k)} h[i+k](1-h[i])
    
    # ∂/∂h[j] of h[i+k]: nonzero when i+k = j, i.e., i = j-k
    #   contributes (1-h[i]) = (1-h[j-k]) if 0<=j-k<n and 0<=i<n (i.e., 0<=j-k<n)
    #   and also need i+k = j in range, which is automatic if j in [0,n-1]
    #   condition: 0 <= j-k < n AND 0 <= j-k < n (i in range) AND 0 <= j < n (j in range)
    #   Simplified: 0 <= j-k < n
    # ∂/∂h[j] of (1-h[i]): nonzero when i = j
    #   contributes -h[i+k] = -h[j+k] if 0<=j<n and 0<=j+k<n
    #   condition: 0 <= j+k < n
    
    # So ∂corr[n-1+k]/∂h[j] = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]
    
    grad = np.zeros(n)
    # For each shift k (from -(n-1) to n-1):
    for k in range(-(n-1), n):
        w = weights[k + n - 1] * dx
        if w < 1e-12:
            continue
        # ∂c_k/∂h[j] = (1-h[j-k]) for j in [max(0,k), n-1] (when k>=0, j-k >= 0 means j >= k)
        #              - h[j+k] for j in [0, min(n-1, n-1-k)] (when k>=0, j+k < n means j < n-k)
        if k >= 0:
            # (1-h[j-k]) for j in [k, n-1]
            if k < n:
                grad[k:] += w * (1.0 - h[:n-k])
            # -h[j+k] for j in [0, n-k-1]
            if k < n:
                grad[:n-k] -= w * h[k:]
        else:
            kk = -k
            # (1-h[j-k]) = (1-h[j+kk]) for j in [0, n-kk-1]
            if kk < n:
                grad[:n-kk] += w * (1.0 - h[kk:])
            # -h[j+k] = -h[j-kk] for j in [kk, n-1]
            if kk < n:
                grad[kk:] -= w * h[:n-kk]
    
    # Add constraint gradient
    viol = np.sum(h) - n / 2.0
    grad += 2.0 * mu * viol
    
    return grad

def adam_optimize(h_start, n, T, mu, lr, max_iter, time_limit):
    h = h_start.copy().astype(float)
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    
    for t in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        f = smooth_max_obj(h, n, T, mu)
        g = smooth_max_grad(h, n, T, mu)
        m_t = beta1 * m_t + (1 - beta1) * g
        v_t = beta2 * v_t + (1 - beta2) * g ** 2
        m_hat = m_t / (1 - beta1 ** (t + 1))
        v_hat = v_t / (1 - beta2 ** (t + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr_new = lr * 1.0  # keep lr
        else:
            lr_new = lr * 0.8  # reduce lr
        h = h_new
        lr = lr_new
        if lr < 1e-8:
            break
    
    return best_h, best_c5

def lbgfb_optimize(h_start, n, T, mu, max_iter, time_limit):
    t0 = time.time()
    def f(h):
        return smooth_max_obj(h, n, T, mu)
    def g(h):
        if time.time() - t0 > time_limit:
            return None
        return smooth_max_grad(h, n, T, mu)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h_start, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

def binary_local_search(h_bin, n, rng, max_rounds):
    h = h_bin.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(3000, len(ones) * len(zeros))
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi = ones[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones[ii] = oj  # replace
                zeros[jj] = oi  # replace
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        # Count trailing ones in binary representation of i
        x = i
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        tm[i] = cnt % 2
    return tm

def van_der_corput(n, base=2):
    vdc = np.zeros(n)
    for i in range(n):
        f = 0.0
        denom = 1.0
        val = i + 1  # use i+1 to avoid 0
        while val > 0:
            denom *= base
            f += (val % base) / denom
            val //= base
        vdc[i] = f
    return vdc
