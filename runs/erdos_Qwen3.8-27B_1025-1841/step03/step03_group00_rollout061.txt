For lag k (0-indexed in the full correlation, where k = n-1 corresponds to "no shift"), the correlation value is:
corr[k] = sum_i h[i] * (1-h[i + k - (n-1)])

To reduce corr[k], I want to either:
- Change h[i] from 1 to 0 where (1-h[i+k-(n-1)]) = 1 (i.e., h[i+k-(n-1)] = 0)
- Change h[i+k-(n-1)] from 0 to 1 (which makes (1-h) = 0)

So targeting specific lags for reduction is possible.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -100, 100)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def binary_sa(h_bin, n, rng, t0_temp, t1_temp, deadline, max_rounds=300, n_tries=3000):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    t = t0_temp
    cool = (t1_temp / t0_temp) ** (1.0 / max(max_rounds, 1))
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        t *= cool
        imp = False
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            nc = compute_c5(h)
            d = nc - cur
            if d < 0 or (t > 1e-12 and rng.random() < np.exp(-d / t)):
                cur = nc
                ones[ii] = oj
                zeros[jj] = oi
                imp = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not imp and rnd > 80:
            break
    return h, cur

def lbgfb_run(h_start, n, T, mu, max_iter):
    def f(h):
        return soft_max_obj(h, n, T) + mu * (np.sum(h) - n / 2.0) ** 2
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def coord_desc(h_in, n, deadline, max_rounds=80):
    h = h_in.copy()
    cur = compute_c5(h)
    deltas = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        imp = False
        for j in range(n):
            if time.time() > deadline:
                break
            sav = h[j]
            for d in deltas:
                nv = sav + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[j] = max(0.0, min(1.0, nv))
                diff = n / 2.0 - np.sum(h)
                if abs(diff) > 1e-13:
                    k = np.argmin(h) if diff > 0 else np.argmax(h)
                    h[k] = np.clip(h[k] + diff, 0.0, 1.0)
                c = compute_c5(h)
                if c < cur - 1e-13:
                    cur = c
                    imp = True
                    break
                h[j] = sav
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                h[j] = bv
                diff = n / 2.0 - np.sum(h)
                if abs(diff) > 1e-13:
                    k = np.argmin(h) if diff > 0 else np.argmax(h)
                    h[k] = np.clip(h[k] + diff, 0.0, 1.0)
                c = compute_c5(h)
                if c < cur - 1e-13:
                    cur = c
                    imp = True
                    break
                h[j] = sav
        if not imp:
            break
    h = make_feasible(h, n)
    return h, compute_c5(h)

    return hb
