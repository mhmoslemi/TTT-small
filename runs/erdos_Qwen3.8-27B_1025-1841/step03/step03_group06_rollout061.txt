    n = len(h)

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:

        if cnt == 0:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def swap_delta(h, j, k, n):
    """Compute Δc for all shifts when swapping h[j] (1→0) and h[k] (0→1)."""
    # Δc_shift = h[j-shift] + h[j+shift] - h[k-shift] - h[k+shift]
    # for valid indices
    shifts = np.arange(-(n-1), n)  # -n+1 to n-1
    # j-shift must be in [0, n-1] and j+shift must be in [0, n-1]
    # k-shift must be in [0, n-1] and k+shift must be in [0, n-1]
    
    delta_c = np.zeros(2*n - 1)
    for idx, s in enumerate(shifts):
        val = 0.0
        if 0 <= j - s < n:
            val += h[j - s]
        if 0 <= j + s < n:
            val += h[j + s]
        if 0 <= k - s < n:
            val -= h[k - s]
        if 0 <= k + s < n:
            val -= h[k - s]
        delta_c[idx] = val
    return delta_c

```python
def swap_delta(h, j, k, n):
    shifts = np.arange(-(n-1), n)
    js = j - shifts  # positions in h for j-shift
    jp = j + shifts  # positions in h for j+shift
    ks = k - shifts
    kp = k + shifts
    
    # Mask valid indices
    v_jm = (js >= 0) & (js < n)
    v_jp = (jp >= 0) & (jp < n)
    v_km = (ks >= 0) & (ks < n)
    v_kp = (kp >= 0) & (kp < n)
    
    delta_c = np.zeros(2*n - 1)
    delta_c += np.where(v_jm, h[js[v_jm]], 0)
    delta_c += np.where(v_jp, h[jp[v_jp]], 0)
    delta_c -= np.where(v_km, h[ks[v_km]], 0)
    delta_c -= np.where(v_kp, h[kp[v_kp]], 0)
    return delta_c
```

def simulated_annealing_binary(h_init, n, n_iters, T_start, T_end, time_limit, start_time):
    h = h_init.astype(float).copy()
    # Make it binary
    # Sort and threshold to get exactly n/2 ones
    sorted_h = np.sort(h)
    thr = sorted_h[n//2]
    h = (h >= thr).astype(float)
    # Fix if not exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > n//2:
        # Randomly set some 1s to 0
        one_idx = np.where(h == 1)[0]
        to_flip = np.random.choice(one_idx, n_ones - n//2, replace=False)
        h[to_flip] = 0
    elif n_ones < n//2:
        zero_idx = np.where(h == 0)[0]
        to_flip = np.random.choice(zero_idx, n//2 - n_ones, replace=False)
        h[to_flip] = 1
    
    corr = full_corr(h)
    max_c = np.max(corr)
    best_h = h.copy()
    best_max_c = max_c
    
    one_set = np.where(h == 1)[0]
    zero_set = np.where(h == 0)[0]
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        # Temperature
        T = T_start * (T_end / T_start) ** (it / max(n_iters, 1))
        
        # Pick random swap
        j = one_set[np.random.randint(len(one_set))]
        k = zero_set[np.random.randint(len(zero_set))]

        delta_c = swap_delta(h, j, k, n)

        delta_max = new_max_c - max_c

            h[k] = 1
            corr = new_corr
            max_c = new_max_c

    js_p = j + shifts

    ks_p = k + shifts

    m4 = (ks_p >= 0) & (ks_p < n)

    delta[m2] += h[js_p[m2]]
    delta[m3] -= h[ks_m[m3]]
    delta[m4] -= h[ks_p[m4]]

        # Let me define w_centered = w (length 2n-1, centered at n-1)
        # sum_s w[n-1+s] * (1-x[j-s]) = sum_s w[s'] * (1-x[j-(s'-(n-1))]) where s' = n-1+s
        # = sum_s' w[s'] * (1-x[j-n+1+s'])
        # Let t = s'-(n-1) = s: 1-x[j-t]... this is circular
        
        # I think the cleanest way is:
        # g1[j] = sum_{valid s} w[n-1+s] * (1-x[j-s])
        # This is the cross-correlation of (1-x) with w at lag j
        # = np.correlate(1-x, w, 'full')[j + (len(w)-1 - len(1-x) + 1)]
        # Hmm, the indexing is tricky.
        
        # Let me just do it numerically for safety:
        g = np.zeros(n)
        shifts_arr = np.arange(-(n-1), n)
        for s in shifts_arr:
            wi = w[n-1+s]
            if wi < 1e-15:
                continue
            js_m = j_arr - s  # j - s
            js_p = j_arr + s  # j + s
            m1 = (js_m >= 0) & (js_m < n)
            m2 = (js_p >= 0) & (js_p < n)
            g[m1] += wi * (1.0 - x[js_m[m1]])
            g[m2] -= wi * x[js_p[m2]]
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-13})
    return res.x

OK the gradient computation is getting too complex and error-prone. Let me skip the analytic gradient and just use SLSQP with numerical gradients (which will be slow but correct), or better yet, skip SLSQP entirely and use only the binary search + coordinate descent.

The binary search should get us most of the way. The coordinate descent handles the continuous fine-tuning. SLSQP was the previous approach's main tool, and it might not add much beyond what the binary search + coordinate descent achieves.

Let me simplify:
1. Binary local search (main optimization)
2. Coordinate descent (continuous fine-tuning)
3. Optional: one SLSQP run at the end (with numerical gradients, short maxiter)

Actually, for n=960, SLSQP with numerical gradients would need 960 function evaluations per gradient. Each function eval is ~2ms. So one gradient is ~2s. With 100 iterations, that's 200s. Too slow.

For n=480: 480 * 1ms = 0.5s per gradient, 100 iterations = 50s. Feasible.

Let me skip SLSQP and focus on:
1. Binary local search at multiple resolutions
2. Coordinate descent
3. "Discrete refinement": try values in {0, 0.1, 0.2, ..., 0.9, 1.0} for each position

This should be effective enough.
