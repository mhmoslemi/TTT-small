Then the problem is: minimize over x in [0,1]: max_k (corr[k] + s_k*(x - h[j]))

Let me define d_k = s_k, and c_k = corr[k] - s_k * h[j]. Then we minimize max_k (c_k + d_k * x) over x in [0,1].

This is the classic "minimize the maximum of linear functions" problem, which can be solved by finding the Chebyshev center of the set of lines.

A practical approach: 
1. Compute g(0) = max_k c_k and the set of k that achieve it
2. Compute g(1) = max_k (c_k + d_k) and the set of k that achieve it
3. The subgradient at x is {d_k : k ∈ argmax(c_k + d_k*x)}
4. Use bisection: if the subgradient at x is all positive, move left; all negative, move right; else x is optimal (or close enough)

Actually, for a 1D convex piecewise-linear function, the minimum is at a point where 0 is in the subdifferential, which means there exist k1 with d_{k1} < 0 and k2 with d_{k2} > 0 both achieving the max (or d_k = 0 for all max-achieving k).

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_polytope(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if abs(np.sum(h) - n/2.0) < 1e-10:
        return h
    return project_polytope(h, n / 2.0)

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def exact_coord_descent(h, n, max_rounds=10, time_limit=None, start_time=None, grid_size=200):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(corr) * dx)
    x_grid = np.linspace(0.0, 1.0, grid_size)
    n_corr = 2 * n - 1
    
    for rnd in range(max_rounds):
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit - 30:
                break
        improved = False
        for j in range(n):
            if start_time is not None and time_limit is not None:
                if time.time() - start_time > time_limit - 30:
                    return h, best_c
            
            # Compute s vector (length 2n-1)
            s = np.zeros(n_corr)
            # Part 1: (1-h[j+k]) for k where 0<=j+k<n
            k_lo = max(-(n-1), -j)
            k_hi = min(n-1, n-1-j)
            if k_lo <= k_hi:
                ks = np.arange(k_lo, k_hi+1)
                s[ks + n - 1] += (1.0 - h[j + ks])
            # Part 2: -h[j-k] for k where 0<=j-k<n
            k_lo2 = max(-(n-1), j-n+1)
            k_hi2 = min(j, n-1)
            if k_lo2 <= k_hi2:
                ks2 = np.arange(k_lo2, k_hi2+1)
                s[ks2 + n - 1] -= h[j - ks2]
            
            # Find optimal x
            base = corr - s * h[j]
            # Vectorized: (n_corr, grid_size) matrix
            vals = base[:, None] + s[:, None] * x_grid[None, :]
            max_vals = vals.max(axis=0)
            best_idx = np.argmin(max_vals)
            best_x = x_grid[best_idx]
            
            if abs(best_x - h[j]) > 1e-8:
                h[j] = best_x
                h = project_polytope(h, n/2.0)

                new_c = float(np.max(corr) * dx)
                if new_c < best_c:

                    improved = True

def subgradient_optimize(h, n, n_iters=500, time_limit=None, start_time=None, alpha0=0.05):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.85
    patience = 0
    best_iter = 0
    
    for it in range(n_iters):
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit - 30:
                break
        
        alpha = alpha0 / (1.0 + 0.05 * it)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        k = k_star - (n - 1)
        
        # Compute gradient (vectorized)
        grad = np.zeros(n)
        if k >= 0:
            j1 = slice(0, n - k)
            grad[j1] += (1.0 - h[k:])  # (1-h[j+k]) for j=0..n-k-1
            j2 = slice(k, n)
            grad[j2] -= h[:-k] if k > 0 else h  # h[j-k] for j=k..n-1
        else:
            kk = -k
            j1 = slice(kk, n)
            grad[j1] += (1.0 - h[:n-kk])  # (1-h[j+k]) for j=-k..n-1
            j2 = slice(0, n - kk)
            grad[j2] -= h[kk:]  # h[j-k] for j=0..n-kk-1
        grad *= dx
        
        # Nesterov momentum
        momentum = beta * momentum - alpha * grad
        h_new = project_polytope(h + momentum, n/2.0)
        
        c_new = compute_c5(h_new)

            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h = h_new
        
        if patience > 100:
            break
    
    return best_h, best_c

C(0) = sum_i h[i]*(1-h[i]) = sum_i h[i] - sum_i h[i]^2

This is quadratic in each h[i]! So C(0) is NOT linear in h[j].

For k≠0: C(k) = sum_i h[i]*(1-h[i+k]). The terms involving h[j] are:
- h[j]*(1-h[j+k]) (linear in h[j])
- h[j-k]*(1-h[j]) (linear in h[j])

So for k≠0, C(k) IS linear in h[j]. Only C(0) is quadratic.

This means my "exact coordinate descent" is exact for all lags except k=0. The k=0 lag contributes a quadratic term.

To handle this properly, for coordinate j, the objective as a function of x = h[j] is:
g(x) = max_k (C_k(x))

where:
- C_0(x) = [sum_{i≠j} h[i](1-h[i])] + x(1-x) = const_0 + x - x^2
- C_k(x) = base_k + s_k * x for k≠0

So g(x) = max(C_0(x), max_{k≠0} (base_k + s_k * x))
