def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    d = 1.0 - h
    N = 2 * n
    H = rfft(h, N)
    D = rfft(d[::-1], N)
    corr = irfft(H * D, N)[:2*n-1]
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            mask = np.ones(n, dtype=bool)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    excess = int(np.sum(b)) - target
    if excess > 0:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        for i in range(min(excess, len(ones))):
            b[ones[order[i]]] = 0
    elif excess < 0:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(min(-excess, len(zeros))):
            b[zeros[order[i]]] = 1
    return b

        j = zeros_list[rng.randint(len(zeros_list))]
        b[i] = 0; b[j] = 1
        new_c5 = compute_c5_fft(b)

            curr_c5 = new_c5
            ones_list.remove(i); zeros_list.remove(j)
            ones_list.append(j); zeros_list.append(i)
            if new_c5 < best_c5:
                best_c5 = new_c5; best_b = b.copy()
        else:
            b[i] = 1; b[j] = 0
    return best_b, best_c5

```python
import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    d = 1.0 - h
    N = 2 * n
    H = rfft(h, N)
    D = rfft(d[::-1], N)
    corr = irfft(H * D, N)[:2 * n - 1]
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            mask = np.ones(n, dtype=bool)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    excess = int(np.sum(b)) - target
    if excess > 0:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        for i in range(min(excess, len(ones))):
            b[ones[order[i]]] = 0
    elif excess < 0:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(min(-excess, len(zeros))):
            b[zeros[order[i]]] = 1
    return b


def binary_sa_point(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = to_binary(h, n).copy()
    curr_c5 = compute_c5_fft(b)
    best_c5 = curr_c5
    best_b = b.copy()
    ones_list = list(np.where(b == 1)[0])
    zeros_list = list(np.where(b == 0)[0])
    for it in range(max_iters):
        if (it & 511) == 0 and (budget_s - (time.time() - start_time) < 10):
            break
        f = float(it) / max_iters
        T = T0 * (T1 / T0) ** f
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        b[i] = 0
        b[j] = 1
        nc = compute_c5_fft(b)
        if nc <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(nc - curr_c5) / T)):
            curr_c5 = nc
            ki = ones_list.index(i)
            ones_list[ki] = ones_list[-1]
            ones_list.pop()
            kj = zeros_list.index(j)
            zeros_list[kj] = zeros_list[-1]
