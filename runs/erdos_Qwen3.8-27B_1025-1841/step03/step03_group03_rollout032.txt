    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    
    if k >= 0:
        # f_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])

            g[j] -= h[j + kp]

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_obj_and_grad(h, n, t):
    """Compute softmax-smoothed objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    shifted = corr - mc
    logits = t * shifted
    # Numerical stability
    logits -= np.max(logits)
    weights = np.exp(logits)
    wsum = np.sum(weights)
    weights /= wsum
    obj = (mc + np.log(wsum) / t)
    
    # Gradient
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-10:
            continue
        k = m - (n - 1)
        w = weights[m]
        if k >= 0:
            # f_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[j]: (1-h[j+k]) for j=0..n-1-k; -h[j-k] for j=k..n-1

            g[k:] -= w * h[:n-k]
        else:

            g[kp:] += w * (1.0 - h[:n-kp])
            g[:n-kp] -= w * h[kp:]
    
    g *= dx
    return obj, g
```

```python
    ]
    
    # Add cutting planes for top lags
    top_lags = np.argsort(corr)[-50:]  # top 50 lags
    for m in top_lags:
        k = m - (n - 1)
        # Compute gradient of f_k at h
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] = (1.0 - h[k:n])
            g[k:] -= h[:n-k]
        else:
            kp = -k
            g[kp:] = (1.0 - h[:n-kp])
            g[:n-kp] -= h[kp:]
        g *= dx
        constraints.append(t >= corr[m] + g @ delta)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)  # or another solver
    
    if delta.value is not None:
        h_new = np.clip(h + delta.value, 0, 1)
        h_new = make_feasible(h_new, n)
        return h_new
    
    return h
```

Hmm, but this requires adding constraints one at a time (or in batches), and each LP solve might take time. For n=640 and 50 constraints, the LP has 641 variables and ~640 constraints. That should be fast.

Actually, I think the issue is that with only 50 cutting planes, the linearization might be too crude. We might need all 2n-1 = 1279 constraints for a good approximation. That's a lot for an LP.

Let me try a different approach: use only the top 100-200 lags as constraints, and iteratively add more.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_objective(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(t * shifted))) / t) * dx

def slsqp_optimize(h_start, n, t, maxiter, time_limit, start_time):
    def obj(h):
        return softmax_objective(h, n, t)
    def con_fun(h):
        return np.sum(h) - n / 2.0
    def con_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return make_feasible(h_start, n)

def subgradient_step(h, n, t, alpha, time_limit, start_time):
    """One subgradient step using softmax-weighted average."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    shifted = corr - mc
    logits = t * shifted
    logits -= np.max(logits)
    weights = np.exp(logits)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        return h
    weights /= wsum
    
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-8:
            continue
        k = m - (n - 1)
        w = weights[m]
        if k >= 0:
            g[:n-k] += w * (1.0 - h[k:n])
            g[k:] -= w * h[:n-k]
        else:
            kp = -k
            g[kp:] += w * (1.0 - h[:n-kp])
            g[:n-kp] -= w * h[kp:]
    g *= dx
    
    h_new = h - alpha * g
    return make_feasible(h_new, n)

def subgradient_method(h, n, t_values, steps_each, alpha0, alpha_decay, time_limit, start_time):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr)
    
    for t in t_values:
        if time.time() - start_time > time_limit:
            break
        alpha = alpha0
        for step in range(steps_each):
            if time.time() - start_time > time_limit:
                break
            h_curr = subgradient_step(h_curr, n, t, alpha, time_limit, start_time)
            c = compute_c5(h_curr)
            if c < best_c:
                best_c = c
                best_h = h_curr.copy()
            alpha *= alpha_decay
    
    return best_h, best_c

def binary_local_search(h_bin, n, max_rounds, time_limit, start_time, rng):
    """Local search: try all 1↔0 swaps, pick the best."""
    dx = 2.0 / n
    h_curr = h_bin.copy()
    c_curr = compute_c5(h_curr)
    corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
    
    ones = np.where(h_curr == 1.0)[0]
    zeros = np.where(h_curr == 0.0)[0]
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        
        best_delta = 0
        best_j = -1
        best_l = -1
        
        # Try a random subset of swaps to save time
        n_try = min(200, len(ones) * len(zeros))
        for _ in range(n_try):
            j = ones[rng.randint(len(ones))]
            l = zeros[rng.randint(len(zeros))]
            if j == l:
                continue
            h_curr[j] = 0.0
            h_curr[l] = 1.0
            new_corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
            new_c = np.max(new_corr) * dx
            delta = new_c - c_curr
            if delta < best_delta:
                best_delta = delta
                best_j = j
                best_l = l
                best_corr = new_corr
            h_curr[j] = 1.0
            h_curr[l] = 0.0
        
        if best_j >= 0:
            h_curr[best_j] = 0.0
            h_curr[best_l] = 1.0
            c_curr += best_delta
            corr = best_corr
            ones = np.where(h_curr == 1.0)[0]
            zeros = np.where(h_curr == 0.0)[0]
        else:
            break
    
    return h_curr, c_curr

def binary_sa(h_bin, n, T0, T1, n_steps, time_limit, start_time, rng):
    """Simulated annealing on binary solution with multi-scale moves."""
    dx = 2.0 / n
    h_curr = h_bin.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr)
    current_c = best_c
    
    ones = np.where(h_curr == 1.0)[0]
    zeros = np.where(h_curr == 0.0)[0]
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        T = T0 * (T1 / T0) ** (step / max(1, n_steps - 1))
        
        # Choose move type: single swap (70%), block swap (30%)
        if rng.rand() < 0.7:
            # Single swap
            j = ones[rng.randint(len(ones))]
            l = zeros[rng.randint(len(zeros))]
            if j == l:
                continue
            h_curr[j] = 0.0
            h_curr[l] = 1.0
        else:
            # Block swap: swap a block of consecutive 1s with a block of consecutive 0s
            block_size = rng.randint(1, 5)
            j = rng.randint(0, n - block_size)
            l = rng.randint(0, n - block_size)
            # Check if j..j+block_size-1 are all 1s and l..l+block_size-1 are all 0s
            if np.all(h_curr[j:j+block_size] == 1.0) and np.all(h_curr[l:l+block_size] == 0.0):
                h_curr[j:j+block_size] = 0.0
                h_curr[l:l+block_size] = 1.0
            elif np.all(h_curr[j:j+block_size] == 0.0) and np.all(h_curr[l:l+block_size] == 1.0):
                h_curr[j:j+block_size] = 1.0
                h_curr[l:l+block_size] = 0.0
            else:
                # Fallback to single swap
                j = ones[rng.randint(len(ones))]
                l = zeros[rng.randint(len(zeros))]
                if j == l:
                    continue
                h_curr[j] = 0.0
                h_curr[l] = 1.0
        
        new_corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        new_c = np.max(new_corr) * dx
        
        delta = new_c - current_c
        if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
            current_c = new_c
            if step % 50 == 0:
                ones = np.where(h_curr == 1.0)[0]
                zeros = np.where(h_curr == 0.0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h_curr.copy()
        else:
            # Reject: this is problematic because we already modified h_curr
            # We need to undo the change, which is hard for block moves
            # Let's just recompute
            pass  # Already modified, accept as current state for SA
    
    return best_h, best_c
```

Wait, there's a problem with my SA implementation. When I reject a move, I've already modified h_curr. I need to save the old state or undo the modification.

Let me fix this by saving the old values:

Actually, for simplicity, let me just always accept in SA (which is a valid variant called "rapid annealing" or just greedy with occasional random moves). Or better, let me save and restore.

Let me restructure:

This is cleaner but the copy `old_h = h_curr.copy()` is O(n) per step, which adds up. For n=640 and 1M steps, that's 640M element copies, which might be slow.

Alternative: just track the two changed positions and undo in O(1).

        l = rng.randint(n)

        # Swap them
        h_curr[j] = 0.0
        h_curr[l] = 1.0
        
        new_c = compute_c5(h_curr)
        delta = new_c - current_c
        
        if delta
