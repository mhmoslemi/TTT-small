        return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx
    def obj_g(hh):

        mc = np.max(c)
        exps = np.exp(T * (c - mc))
        tot = np.sum(exps)

        g = np.zeros(n)

            if w[i] < 1e-12:
                break

        return g * dx

        return np.sum(hh) - n / 2.0

        return np.ones(n)

    # Phase 13: Ultra-fine PGB top-1
    if budget_s - (time.time() - t0) > 20:
        hf, cf = pgb(bh, 15000, 1, 10)
        if cf < bc:
            bc, bh = cf, hf.copy()

    # Phase 14: Final tiny FISTA
    if budget_s - (time.time() - t0) > 10:
        hf, cf = fista_ak(bh, 5000, 1, 0.02, 20)
        if cf < bc:
            bc, bh = cf, hf.copy()

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
