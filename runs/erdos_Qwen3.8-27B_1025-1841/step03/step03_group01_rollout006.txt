            j1 = np.arange(0, n - p)
            grad[j1] += w * (1 - h[j1 + p])
            j2 = np.arange(p, n)
            grad[j2] -= w * h[j2]
    
    return grad * (2.0 / n)  # multiply by dx for the objective

            if abs(compute_c5(h) - best_c5) < tol * 10:
                break

    h = np.clip(z, 0.0, 1.0)
    target = n / 2.0
    # Bisection for tau
    lo, hi = -1.0, 1.0
    for _ in range(50):
        tau = (lo + hi) / 2
        h_test = np.clip(z - tau, 0.0, 1.0)
        s = np.sum(h_test)
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(z - (lo+hi)/2, 0.0, 1.0)
    # Fine adjust
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = (h < 1.0) & (h > 0.0)
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)

                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
            else:
                h[np.argmax(h)] += diff
