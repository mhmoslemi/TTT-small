            h_start = best_h + rng.randn(n_pts) * 0.008
            h_start = make_feasible(h_start, n_pts)
        try:
            res = minimize(_obj_func, h_start, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13})
            if np.all(np.isfinite(res.x)):
                c5_r = compute_c5(res.x)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 2: Upsample and optimize at higher resolution
    if remaining() > 350:
        n_high = 1280
        h_high = upsample_h(best_h, n_high)
        best_hh = h_high.copy()
        best_ch = compute_c5(h_high)
        _G_N = n_high
        _G_DX = 2.0 / n_high
        bounds_h = [(0.0, 1.0)] * n_high
        cons_h = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        for T, maxiter in [(50, 400), (180, 350)]:
            if remaining() < 120:
                break
            _G_T = T
            h_start = best_hh.copy()
            try:
                res = minimize(_obj_func, h_start, method='SLSQP',
                               bounds=bounds_h, constraints=cons_h,
                               options={'maxiter': maxiter, 'ftol': 1e-13})
                if np.all(np.isfinite(res.x)):
                    c5_r = compute_c5(res.x)
                    if c5_r < best_ch:
                        best_ch = c5_r
                        best_hh = make_feasible(res.x, n_high)
            except Exception:
                pass
        if best_ch < best_c5:
            best_c5 = best_ch
            best_h = best_hh
            n_pts = n_high

    # Phase 3: Try n=1600 if time allows
    if remaining() > 250 and n_pts < 1600:
        n_hi2 = 1600
        h_hi2 = upsample_h(best_h, n_hi2)
        best_hh2 = h_hi2.copy()
        best_ch2 = compute_c5(h_hi2)
        _G_N = n_hi2
        _G_DX = 2.0 / n_hi2
        bounds_2 = [(0.0, 1.0)] * n_hi2
        cons_2 = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        for T, maxiter in [(60, 300)]:
            if remaining() < 100:
                break
            _G_T = T
            try:
                res = minimize(_obj_func, best_hh2, method='SLSQP',
                               bounds=bounds_2, constraints=cons_2,
                               options={'maxiter': maxiter, 'ftol': 1e-13})
                if np.all(np.isfinite(res.x)):
                    c5_r = compute_c5(res.x)
                    if c5_r < best_ch2:
                        best_ch2 = c5_r
                        best_hh2 = make_feasible(res.x, n_hi2)
            except Exception:
                pass
        if best_ch2 < best_c5:
            best_c5 = best_ch2
            best_h = best_hh2
            n_pts = n_hi2

    # Phase 4: Binary snap + local search
    if remaining() > 70:
        n_pts = len(best_h)
        b_h = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(b_h))
        target_ones = n_pts // 2
        if n_ones > target_ones:
            idx1 = np.where(b_h > 0.5)[0]
            for k in range(n_ones - target_ones):
                b_h[idx1[k]] = 0.0
        elif n_ones < target_ones:
            idx0 = np.where(b_h < 0.5)[0]
            for k in range(target_ones - n_ones):
                b_h[idx0[k]] = 1.0
        b_h = make_feasible(b_h, n_pts)
        c_b = compute_c5(b_h)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = b_h.copy()
        if remaining() > 50:
            best_h, best_c5 = binary_local_search(best_h, len(best_h), rng,
                                                   max_rounds=30,
                                                   time_check=remaining() - 20,
                                                   start_t=start_time)

    # Phase 5: Coordinate descent on non-binary coords
    if remaining() > 40:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descend(best_h, n_pts,
                                        time_check=remaining() - 10,
                                        start_t=start_time, step=1, n_cands=4)

    # Phase 6: Final SLSQP polish at very high T
    if remaining() > 70:
        n_pts = len(best_h)
        _G_N = n_pts
        _G_T = 800.0
        _G_DX = 2.0 / n_pts
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_func, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            if np.all(np.isfinite(res.x)):
                c5_r = compute_c5(res.x)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 7: Final fine coordinate descent
    if remaining() > 20:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descend(best_h, n_pts,
                                        time_check=remaining() - 3,
                                        start_t=start_time, step=2, n_cands=5)

    # Phase 8: One more binary local search if solution is near-binary
    if remaining() > 30 and np.sum((best_h > 0.1) & (best_h < 0.9)) < len(best_h) * 0.1:
        n_pts = len(best_h)
        best_h, best_c5 = binary_local_search(best_h, n_pts, rng,
                                               max_rounds=15,
                                               time_check=remaining() - 5,
                                               start_t=start_time)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
