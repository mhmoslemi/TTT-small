    shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = p[i_idx]
        k = top_idx[i_idx]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def nesterov_opt(h, n, tau, time_limit, lr, top_k=8):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj, grad = soft_obj_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        if it % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 5000:
                lr *= 0.55
                stall = 0
                if lr < 1e-16:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def subgrad_opt(h, n, time_limit, lr=0.001, top_k=5):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.75
    stall = 0
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            lr *= 0.65
            v *= 0.4
            stall = 0
            if lr < 1e-16:
                break
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals.max())
        w /= w.sum()
        grad = np.zeros(n)
        for i_idx in range(len(top_idx)):
            pk = w[i_idx]
            k = top_idx[i_idx]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def top_k_direct_reduce(h, n, time_limit, top_k=3, lr=0.008):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.6
    stall = 0
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-16:
                break
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        grad = np.zeros(n)
        for k in top_idx:
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += (1.0 - h[np.arange(j0, j1) - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= h[np.arange(j0b, j1b) + sh]
        grad *= dx
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Upsample to n=8000 for more degrees of freedom
    n1 = 8000
    h1 = interp_periodic(h_init, n1)
    h1 = project_to_feasible(h1, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: Nesterov tau=500, broad search
    rem = get_remaining(start_time, budget_s)
    if rem > 200:
        tl = min(170, rem * 0.22)
        h1, c5 = nesterov_opt(h1, n1, tau=500.0, time_limit=tl, lr=0.25, top_k=20)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Nesterov tau=150
    rem = get_remaining(start_time, budget_s)
    if rem > 180:
        tl = min(150, rem * 0.19)
        h1, c5 = nesterov_opt(h1, n1, tau=150.0, time_limit=tl, lr=0.12, top_k=12)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 3: Nesterov tau=50
    rem = get_remaining(start_time, budget_s)
    if rem > 160:
        tl = min(130, rem * 0.18)
        h1, c5 = nesterov_opt(h1, n1, tau=50.0, time_limit=tl, lr=0.06, top_k=8)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 4: Nesterov tau=15
    rem = get_remaining(start_time, budget_s)
    if rem > 130:
        tl = min(110, rem * 0.17)
        h1, c5 = nesterov_opt(h1, n1, tau=15.0, time_limit=tl, lr=0.025, top_k=6)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 5: Top-k direct reduction (normalized gradient)
    rem = get_remaining(start_time, budget_s)
    if rem > 100:
        tl = min(90, rem * 0.15)
        h1, c5 = top_k_direct_reduce(h1, n1, time_limit=tl, top_k=3, lr=0.005)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 6: Subgradient refinement
    rem = get_remaining(start_time, budget_s)
    if rem > 80:
        tl = min(70, rem * 0.14)
        h1, c5 = subgrad_opt(h1, n1, time_limit=tl, lr=0.0008, top_k=4)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 7: Very low tau Nesterov for precision
    rem = get_remaining(start_time, budget_s)
    if rem > 70:
        tl = min(60, rem * 0.15)
        h1, c5 = nesterov_opt(h1, n1, tau=4.0, time_limit=tl, lr=0.008, top_k=4)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Multi-start perturbations
    rem = get_remaining(start_time, budget_s)
    for scale in [0.012, 0.006, 0.02, 0.004, 0.015]:
        rem = get_remaining(start_time, budget_s)
        if rem < 55:
            break
        h_p = best_h.copy() + np.random.randn(len(best_h)) * scale
        h_p = project_to_feasible(h_p, len(best_h))
        tl = min(45, rem * 0.2)
        h_p, c5_p = nesterov_opt(h_p, len(h_p), tau=40.0, time_limit=tl, lr=0.07, top_k=8)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Final subgradient polish
    rem = get_remaining(start_time, budget_s)
    if rem > 25:
        tl = min(22, rem * 0.7)
        h1, c5 = subgrad_opt(best_h, len(best_h), time_limit=tl, lr=0.0003, top_k=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Final top-k direct
    rem = get_remaining(start_time, budget_s)
    if rem > 20:
        tl = min(18, rem * 0.8)
        h1, c5 = top_k_direct_reduce(best_h, len(best_h), time_limit=tl, top_k=2, lr=0.003)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
