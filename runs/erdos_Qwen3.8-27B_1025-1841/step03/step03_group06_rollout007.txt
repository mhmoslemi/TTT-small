    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_round(h):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    return make_feasible(h_bin, len(h))

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SLSQP at parent resolution with temperature sweep
    temps = [30.0, 80.0, 200.0, 500.0]
    for T in temps:
        if elapsed() > budget_s - 180:
            break
        _g_n = len(best_h)
        _g_T = T
        bounds = [(0.0, 1.0)] * _g_n
        cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
        maxiter = 300 if _g_n <= 640 else 200
        try:
            res = minimize(_soft_obj, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            res_h = make_feasible(res.x, _g_n)
            c5_new = compute_c5(res_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = res_h
        except Exception:
            pass

    # Phase 2: Try upsampling to 2x for finer resolution
    n_hi = len(best_h) * 2
    if n_hi <= 1920 and elapsed() < budget_s - 250:
        h_hi = upsample_h(best_h, n_hi)
        _g_n = n_hi
        _g_T = 200.0
        bounds_hi = [(0.0, 1.0)] * n_hi
        cons_hi = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
        try:
            res = minimize(_soft_obj, h_hi, method='SLSQP',
                           bounds=bounds_hi, constraints=cons_hi,
                           options={'maxiter': 80, 'ftol': 1e-13})
            res_h = make_feasible(res.x, n_hi)
            c5_new = compute_c5(res_h)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = res_h
        except Exception:
            pass

    # Phase 3: Binary round
    n_cur = len(best_h)
    h_br = binary_round(best_h)
    c5_br = compute_c5(h_br)
    if c5_br < best_c5:
        best_c5 = c5_br
        best_h = h_br

    # Phase 4: Aggressive coordinate descent (binary-focused)
    for rnd in range(35):
        if elapsed() > budget_s - 120:
            break
        improved = False
        for j in range(n_cur):
            if elapsed() > budget_s - 100:
                break
            old = best_h[j]
            best_c_local = best_c5
            best_val = old
            cands = [0.0, 1.0]
            if 0.01 < old < 0.99:
                cands.append(round(old, 2))
                cands.append(0.5)
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_cur)
                c = compute_c5(trial)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_val = cand
            if best_val != old:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_cur)
                best_c5 = best_c_local
                improved = True
        if not improved:
            break

    # Phase 5: Block moves (flip groups of adjacent bits)
    if elapsed() < budget_s - 80:
        for bs in [2, 3, 4, 6]:
            if elapsed() > budget_s - 60:
                break
            n_blocks = n_cur // bs
            for bi in range(n_blocks):
                if elapsed() > budget_s - 50:
                    break
                s = bi * bs
                e = s + bs
                # Flip block
                trial = best_h.copy()
                trial[s:e] = 1.0 - trial[s:e]
                trial = make_feasible(trial, n_cur)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial
                # Set to 0
                trial0 = best_h.copy()
                trial0[s:e] = 0.0
                trial0 = make_feasible(trial0, n_cur)
                c0 = compute_c5(trial0)
                if c0 < best_c5 - 1e-14:
                    best_c5 = c0
                    best_h = trial0
                # Set to 1
                trial1 = best_h.copy()
                trial1[s:e] = 1.0
                trial1 = make_feasible(trial1, n_cur)
                c1 = compute_c5(trial1)
                if c1 < best_c5 - 1e-14:
                    best_c5 = c1
                    best_h = trial1

    # Phase 6: Simulated annealing with binary transitions
    if elapsed() < budget_s - 80:
        h_sa = best_h.copy()
        c5_sa = best_c5
        T_sa = 0.005
        T_sa_min = 5e-5
        n_sa = 25000
        for it in range(n_sa):
            if elapsed() > budget_s - 40:
                break
            T_sa = max(T_sa * 0.9996, T_sa_min)
            n_flips = rng.randint(1, 4)
            idxs = rng.randint(0, n_cur, size=n_flips)
            trial = h_sa.copy()
            for ix in idxs:
                trial[ix] = 1.0 - trial[ix]
            trial = make_feasible(trial, n_cur)
            c_new = compute_c5(trial)
            if c_new < c5_sa:
                h_sa = trial
                c5_sa = c_new
                if c5_sa < best_c5 - 1e-14:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
            else:
                delta = c_new - c5_sa
                if delta < 0.005:
                    accept = np.exp(-delta / max(T_sa, 1e-10))
                    if rng.random() < accept:
                        h_sa = trial
                        c5_sa = c_new

    # Phase 7: Perturbation + SLSQP re-optimization (iterative improvement)
    for pidx in range(4):
        if elapsed() > budget_s - 60:
            break
        h_pert = best_h + rng.randn(len(best_h)) * 0.02
        h_pert = make_feasible(h_pert, n_cur)
        _g_n = n_cur
        _g_T = 800.0
        bounds_p = [(0.0, 1.0)] * n_cur
        cons_p = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
        try:
            res = minimize(_soft_obj, h_pert, method='SLSQP',
                           bounds=bounds_p, constraints=cons_p,
                           options={'maxiter': 100, 'ftol': 1e-14})
            res_h = make_feasible(res.x, n_cur)
            c5_new = compute_c5(res_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = res_h
        except Exception:
            pass

    # Phase 8: Final coordinate descent polish
    for rnd in range(25):
        if elapsed() > budget_s - 15:
            break
        improved = False
        for j in range(n_cur):
            if elapsed() > budget_s - 10:
                break
            old = best_h[j]
            best_c_local = best_c5
            best_val = old
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_cur)
                c = compute_c5(trial)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_val = cand
            if best_val != old:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_cur)
                best_c5 = best_c_local
                improved = True
        if not improved:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
