def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def fft_corr(a, b):

    N = 1
    while N < n + m - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

            candidates_j = list(false_idx[:100])

        if not candidates_i or not candidates_j:
            candidates_i = list(true_idx[:100])
            candidates_j = list(false_idx[:100])

        i = candidates_i[np.random.randint(len(candidates_i))]
        j = candidates_j[np.random.randint(len(candidates_j))]

        S[i] = False
        S[j] = True
        h_test = S.astype(float)
        c5_test = compute_c5(h_test)

        if c5_test < best_c5 - 1e-15:
            best_c5 = c5_test
            best_S = S.copy()
            h_bin = h_test
            no_improve = 0
        else:
            S[i] = True
            S[j] = False
            no_improve += 1

    h_out = best_S.astype(float)
    return h_out, best_c5


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    n = 4000
    h1 = interp_periodic(h_init, n)
    h1 = project_to_feasible(h1, n)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: Nesterov high tau, broad top_k
    rem = get_remaining(start_time, budget_s)
    if rem > 250:
        tl = min(220, rem * 0.20)
        h1, c5 = nesterov_fft(h1, n, tau=150.0, time_limit=tl, lr0=0.15, top_k_max=30, top_k_min=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Nesterov medium tau
    rem = get_remaining(start_time, budget_s)
    if rem > 200:
        tl = min(180, rem * 0.18)
        h1, c5 = nesterov_fft(best_h, n, tau=50.0, time_limit=tl, lr0=0.06, top_k_max=20, top_k_min=4)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 3: Nesterov low tau
    rem = get_remaining(start_time, budget_s)
    if rem > 150:
        tl = min(130, rem * 0.16)
        h1, c5 = nesterov_fft(best_h, n, tau=15.0, time_limit=tl, lr0=0.02, top_k_max=12, top_k_min=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 4: Very low tau fine-tune
    rem = get_remaining(start_time, budget_s)
    if rem > 100:
        tl = min(80, rem * 0.12)
        h1, c5 = nesterov_fft(best_h, n, tau=5.0, time_limit=tl, lr0=0.008, top_k_max=8, top_k_min=2)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 5: Guided binary local search
    rem = get_remaining(start_time, budget_s)
    if rem > 80:
        tl = min(70, rem * 0.12)
        h1, c5 = guided_local_search(best_h, n, time_limit=tl, max_no_improve=30000)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 6: Random binary local search
    rem = get_remaining(start_time, budget_s)
    if rem > 60:
        tl = min(50, rem * 0.10)
        h1, c5 = binary_local_search(best_h, n, time_limit=tl, max_no_improve=20000)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 7: Perturbation + re-optimize
    for pert in [0.01, 0.005, 0.02, 0.003]:
        rem = get_remaining(start_time, budget_s)
        if rem < 60:
            break
        h_p = best_h.copy() + np.random.randn(n) * pert
        h_p = project_to_feasible(h_p, n)
        tl = min(45, rem * 0.15)
        h_p, c5_p = nesterov_fft(h_p, n, tau=30.0, time_limit=tl, lr0=0.08, top_k_max=15, top_k_min=3)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()
            # Quick binary refinement
            rem = get_remaining(start_time, budget_s)
            if rem > 40:
                tl2 = min(30, rem * 0.12)
                h2, c5_2 = guided_local_search(best_h, n, time_limit=tl2, max_no_improve=15000)
                if c5_2 < best_c5:
                    best_c5, best_h = c5_2, h2.copy()

    # Phase 8: Final guided local search
    rem = get_remaining(start_time, budget_s)
    if rem > 30:
        tl = min(25, rem * 0.4)
        h1, c5 = guided_local_search(best_h, n, time_limit=tl, max_no_improve=50000)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Final projection
    best_h = project_to_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
