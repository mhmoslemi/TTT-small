def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)[-int(np.ceil(abs(diff)))]
            for i in idx:
                h[i] = min(1.0, h[i] + diff / len(idx))
        else:
            idx = np.argsort(h)[int(np.ceil(abs(diff))):]
            for i in idx:
                h[i] = max(0.0, h[i] + diff / len(idx))
    return h

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def log_sum_exp_objective(h, T, n):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    OH = np.fft.rfft(one_minus_h, N)

    mc = np.max(corr)

def compute_c5_fft(h):

    n = len(h)
    dx = 2.0 / n

    N = 1 << (2 * n - 2).bit_length()
    H = np.fft.rfft(h, N)

    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12 * n:

        else:

    N = 1 << (2 * n - 2).bit_length()
    H = np.fft.rfft(h, N)
