    """Soft-max objective with penalty for constraint violation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -50, 50)
    soft = mc + np.log(np.sum(np.exp(shifted))) / T
    viol = np.sum(h) - n / 2.0
    return (soft + lam * viol * viol) * dx

def lbgfb_optimize(h_init, n, T, lam, max_iter, time_limit):
    """L-BFGS-B optimization of soft-max objective."""
    def f(h):
        return soft_max_obj(h, n, T, lam)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h_init, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                               'maxfun': 5000})
        h_opt = make_feasible(res.x, n)
    except Exception:
        h_opt = make_feasible(h_init, n)
    return h_opt

def binary_sa(h_init, n, rng, time_limit, T0=1.0, T1=0.01, max_iter=50000):
    """Simulated annealing on binary solutions."""
    h = h_init.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    n_iter = 0
    for i in range(max_iter):
        if time.time() > time_limit:
            break
        T = T0 * (T1 / T0) ** (i / max_iter)
        
        # Random swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        a, b = ones[ii], zeros[jj]
        
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / (T * dx * 10)):
            # Accept
            current_c5 = new_c5
            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, b)
            zeros = np.append(zeros, a)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Reject
            h[a] = 1.0
            h[b] = 0.0
    
    return best_h, best_c5

def greedy_binary_search(h_init, n, max_rounds=200):
    """Greedy binary local search: always take the best improving swap."""
    h = h_init.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for rnd in range(max_rounds):
        best_delta = 0
        best_pair = None
        
        # Sample a subset of swaps (can't check all N^2/4)
        n_samples = min(5000, len(ones) * len(zeros))
        for _ in range(n_samples):
            ii = np.random.randint(len(ones))
            jj = np.random.randint(len(zeros))
            a, b = ones[ii], zeros[jj]
            
            h[a] = 0.0
            h[b] = 1.0
            c = compute_c5(h)
            h[a] = 1.0
            h[b] = 0.0
            
            delta = current_c5 - c
            if delta > best_delta:
                best_delta = delta
                best_pair = (ii, jj, a, b)
        
        if best_pair is None:
            break
        
        ii, jj, a, b = best_pair
        h[a] = 0.0
        h[b] = 1.0
        current_c5 -= best_delta
        ones = np.delete(ones, ii)
        zeros = np.delete(zeros, jj)
        ones = np.append(ones, b)
        zeros = np.append(zeros, a)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5

def make_feasible(h, n):

                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current - 1e-15:
                    h = tr
                    current = c
                    improved = True
                    break
            # Try binary
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                tr = h.copy()
                tr[j] = bv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current - 1e-15:
                    h = tr
                    current = c
                    improved = True
        if not improved:
            break
    
    if current < best:
        best = current
        best_h = h.copy()
    return best_h, best
