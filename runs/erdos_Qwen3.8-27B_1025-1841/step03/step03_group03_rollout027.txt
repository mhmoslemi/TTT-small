                best_h = avg_h.copy()
    return best_h, best_c

def fine_coord_descent(h, n, max_rounds=40, time_limit=None, start_time=None):
    h = h.copy()
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time_limit is not None and time.time() - start_time > time_limit:
                return h, best_c
            old = h[j]
            for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
                for d in [-1, 1]:
                    nv = old + d * step
                    if nv < 0.0 or nv > 1.0:
                        continue
                    h[j] = nv
                    h = make_feasible(h, n)
                    c_new = compute_c5(h)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        old = nv
                    else:
                        h[j] = old
                        h = make_feasible(h, n)
            for tv in [0.0, 1.0]:
                if abs(h[j] - tv) > 1e-4:
                    h[j] = tv
                    h = make_feasible(h, n)
                    c_new = compute_c5(h)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[j] = old
                        h = make_feasible(h, n)
        if not improved:
            break
    return h, best_c

def binary_swap_search(h, n, max_iter=8000, time_limit=None, start_time=None):
    h = (h > 0.5).astype(float)
    h = make_feasible(h, n)
    best_c = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(7)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iter):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def block_coord_descent(h, n, block_size=4, max_rounds=20, time_limit=None, start_time=None):
    h = h.copy()
    best_c = compute_c5(h)
    from scipy.optimize import minimize
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        improved = False
        for start in range(0, n - block_size + 1, block_size):
            if time_limit is not None and time.time() - start_time > time_limit:
                return h, best_c
            idx = list(range(start, start + block_size))
            h0_block = h[idx].copy()
            h_rest_sum = h.sum() - h0_block.sum()
            
            def block_obj(x):
                trial = h.copy()
                trial[idx] = np.clip(x, 0.0, 1.0)
                remaining = n / 2.0 - h_rest_sum - trial[idx].sum()
                if remaining < -1e-12 or remaining > 1e-6 * (n - len(idx)):
                    return compute_c5(trial) + 100.0 * remaining**2
                return compute_c5(trial)
            
            res = minimize(block_obj, h0_block, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-10})
            if res.success and res.fun < best_c - 1e-12:
                h[idx] = np.clip(res.x, 0.0, 1.0)
                h = make_feasible(h, n)
                best_c = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        pass
    
    if h0 is None:
        n_init = 640
        h0 = np.zeros(n_init)
        h0[:n_init//2] = 1.0
    else:
        n_init = len(h0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Projected gradient descent at initial resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        h_new, c_new = pgd_optimize(best_h, n_init, max_iter=4000,
                                     time_limit=budget_s - 200, start_time=start_time)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new
    
    # Phase 2: Fine coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        h_new, c_new = fine_coord_descent(best_h, n_init, max_rounds=30,
                                           time_limit=budget_s - 150, start_time=start_time)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new
    
    # Phase 3: Block coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        h_new, c_new = block_coord_descent(best_h, n_init, block_size=4, max_rounds=10,
                                            time_limit=budget_s - 120, start_time=start_time)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new
    
    # Phase 4: Binary solution + swap search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        h_bin, c_bin = binary_swap_search(best_h, n_init, max_iter=5000,
                                           time_limit=budget_s - 100, start_time=start_time)
        if c_bin < best_c5 - 1e-12:
            best_c5 = c_bin
            best_h = h_bin
    
    # Phase 5: Try upsampled resolutions
    for target_n in [896, 1024, 1280]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 80:
            break
        h_up = upsample_h(best_h, target_n)
        if time.time() - start_time > budget_s - 60:
            break
        h_new, c_new = pgd_optimize(h_up, target_n, max_iter=2000,
                                     time_limit=budget_s - 60, start_time=start_time)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            h_new, c_new = fine_coord_descent(best_h, target_n, max_rounds=15,
                                               time_limit=budget_s - 40, start_time=start_time)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new
    
    # Phase 6: Final fine-tuning at best resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_final = len(best_h)
        h_new, c_new = fine_coord_descent(best_h, n_final, max_rounds=20,
                                           time_limit=budget_s - 10, start_time=start_time)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
