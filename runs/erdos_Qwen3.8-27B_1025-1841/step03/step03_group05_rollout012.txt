def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_max(h, n):
    """Exact subgradient of max_k result[k] (unweighted)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    kmax = int(np.argmax(corr))
    shift = kmax - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        s = shift
        if s < n:
            g[s:] += 1.0 - h[:n-s]
            g[:n-s] -= h[s:]
    else:
        s = -shift
        if s < n:
            g[:n-s] += 1.0 - h[s:]
            g[s:] -= h[:n-s]
    return g * dx, kmax

def smooth_subgrad(h, n, T, top_k=100):
    """Smoothed subgradient using log-sum-exp weights"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        kmax = int(np.argmax(corr))
        shift = kmax - (n - 1)
        g = np.zeros(n)
        if shift >= 0 and shift < n:
            g[shift:] += 1.0 - h[:n-shift]
            g[:n-shift] -= h[shift:]
        elif shift < 0 and -shift < n:
            g[:-shift] += 1.0 - h[-shift:]
            g[-shift:] -= h[:-shift]
        return g * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        shift = i - (n - 1)
        if shift >= 0 and shift < n:
            g[shift:] += wi * (1.0 - h[:n-shift])
            g[:n-shift] -= wi * h[shift:]
        elif shift < 0 and -shift < n:
            g[:n+shift] += wi * (1.0 - h[-shift:])
            g[-shift:] -= wi * h[:-shift]
    return g * dx

def lbfgs_smooth(h, T, n, maxiter=300):
    """L-BFGS-B on smooth objective"""
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T*(corr-mc))))/T) * (2.0/n)
    
    def grad(x):
        return smooth_subgrad(x, n, T, top_k=min(200, 2*n-1))
    
    def constraint(x):
        return np.sum(x) - n/2.0
    
    def constraint_jac(x):
        return np.ones(n)
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':constraint,'jac':constraint_jac}],
                   options={'maxiter':maxiter, 'ftol':1e-16, 'gtol':1e-10})
    hx = project_c(res.x, n)
    return hx, compute_c5(hx)

def coord_desc(h, n, n_iter=200, dx=2.0):
    """Coordinate descent with projection"""
    best = h.copy()
    best_c = compute_c5(best)
    for it in range(n_iter):
        improved = False
        for j in range(n):
            # Try h[j] -> 0 and h[j] -> 1
            for new_val in [0.0, 1.0]:
                if abs(h[j] - new_val) < 1e-12:
                    continue
                h[j] = new_val
                h_p = project_c(h, n)
                c = compute_c5(h_p)
                if c < best_c - 1e-14:
                    best = h_p.copy()
                    best_c = c
                    h = h_p.copy()
                    improved = True
                else:
                    h = h_p  # keep projected version
                    # Actually, this might change other coordinates...
                    # Let me restore h[j] and re-project properly
                    h = best.copy()
                    h[j] = new_val
                    h_p = project_c(h, n)
                    c = compute_c5(h_p)
                    if c < best_c - 1e-14:
                        best = h_p.copy()
                        best_c = c
                        h = h_p.copy()
                        improved = True
                    else:
                        h = best.copy()
        if not improved:
            break
    return best, best_c
