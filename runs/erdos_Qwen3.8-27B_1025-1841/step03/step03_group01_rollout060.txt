    best_h = h0.copy()

    # Phase 1: Momentum subgradient at n0
    sg_configs = [
        (0.1, 0.7, 0.005, 6000),
        (0.15, 0.8, 0.01, 5000),
        (0.05, 0.9, 0.002, 8000),
        (0.2, 0.65, 0.015, 5000),
        (0.08, 0.85, 0.004, 6000),
        (0.25, 0.5, 0.02, 4000),
        (0.12, 0.75, 0.008, 5000),
    ]
    for cfg in sg_configs:
        if tl() < 30:
            break
        h_sg, c_sg = momentum_subgrad(best_h, n0, cfg[3], cfg[0], cfg[1], cfg[2], t0, time.time() - t0 + tl() - 5)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()

    # Phase 2: Top-k smoothed subgradient
    for k_top in [3, 5]:
        if tl() < 30:
            break
        for alpha0, beta, gamma in [(0.12, 0.8, 0.008), (0.06, 0.9, 0.003)]:
            if tl() < 20:
                break
            h_sg, c_sg = topk_subgrad(best_h, n0, 4000, alpha0, beta, gamma, k_top, t0, time.time() - t0 + tl() - 5)
            if c_sg < best_c5:
                best_c5 = c_sg
                best_h = h_sg.copy()

    # Phase 3: Random restart subgradient
    for _r in range(4):
        if tl() < 25:
            break
        h_pert = best_h + rng.normal(0, 0.03, n0)
        h_pert = make_feasible(h_pert, n0)
        h_sg, c_sg = momentum_subgrad(h_pert, n0, 3000, 0.1, 0.75, 0.006, t0, time.time() - t0 + tl() - 5)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()

    # Phase 4: SLSQP polish at n0
    if tl() > 60:
        for T in [50.0, 100.0, 300.0]:
            if tl() < 30:
                break
            h_opt = slsqp_optimize(best_h, n0, T, 200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 5: Increase resolution and continue
    for n_new in [2 * n0, 4 * n0]:
        if tl() < 60:
            break
        h_up = upsample_h(best_h, n_new)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()

        # Subgradient at new resolution
        sg_configs2 = [(0.08, 0.8, 0.005, 3000), (0.12, 0.7, 0.01, 2500)]
        for cfg in sg_configs2:
            if tl() < 30:
                break
            h_sg, c_sg = momentum_subgrad(best_h, n_new, cfg[3], cfg[0], cfg[1], cfg[2], t0, time.time() - t0 + tl() - 5)
            if c_sg < best_c5:
                best_c5 = c_sg
                best_h = h_sg.copy()

        # SLSQP at new resolution
        if tl() > 40:
            h_opt = slsqp_optimize(best_h, n_new, 200.0, 150)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 6: Coordinate descent polish
    if tl() > 30:
        n_cd = min(len(best_h), 1280)
        if n_cd < len(best_h):
            h_cd_start = upsample_h(best_h, n_cd)
        else:
            h_cd_start = best_h.copy()
        h_cd, c_cd = coord_descent(h_cd_start, n_cd, t0, time.time() - t0 + tl() - 10, max_rounds=4)
        if n_cd < len(best_h):
            h_full = upsample_h(h_cd, len(best_h))
            c_full = compute_c5(h_full)
            if c_full < best_c5:
                best_c5 = c_full
                best_h = h_full.copy()
        else:
            if c_cd < best_c5:
                best_c5 = c_cd
                best_h = h_cd.copy()

    # Phase 7: Final SLSQP pass
    if tl() > 25:
        h_opt = slsqp_optimize(best_h, len(best_h), 500.0, 100)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
