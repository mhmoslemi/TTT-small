        if diff > 0:

        else:
            mask = h > 0.0

        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

        n_pts = len(best_h)
        iters8 = min(40000, int((remaining - 60) * 1000))
        b8, c5_8 = binary_sa_adaptive(best_h, n_pts, max(iters8, 10000),
                                       0.003, 0.000001, rng)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = b8.copy()

    # Phase 9: Final continuous polish (very sharp T)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        n_pts = len(best_h)
        T_final = [1000.0, 5000.0, 20000.0]
        best_h, best_c5 = continuous_refine(best_h, n_pts, T_final, maxiter=200,
                                             start_time=start_time, budget_s=budget_s)

    # Phase 10: One more binary + continuous cycle if time
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        n_pts = len(best_h)
        iters10 = min(30000, int((remaining - 60) * 1000))
        b10, c5_10 = binary_sa_adaptive(best_h, n_pts, max(iters10, 5000),
                                         0.002, 0.000001, rng)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = b10.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 50:
            best_h, best_c5 = continuous_refine(best_h, len(best_h),
                                                 [2000.0, 10000.0], maxiter=150,
                                                 start_time=start_time, budget_s=budget_s)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
