def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_sum(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection on lambda"""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(proj)
    if abs(diff) < 1e-12:
        return proj
    mask = (proj > 0) & (proj < 1)
    if np.sum(mask) > 0:
        proj[mask] += diff / np.sum(mask)
        proj = np.clip(proj, 0.0, 1.0)
    return proj

    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def proj_grad_opt(h, n, temps, iters_per_T, start_time, budget):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for T in temps:
        h_curr = best_h.copy()
        for it in range(iters_per_T):
            if time.time() - start_time > budget - 30:
                return best_h, best_c5
            obj, grad = smooth_obj_grad(h_curr, T)
            # Fixed step with decay
            alpha = 2.0 / (T + 1.0) * (1.0 - 0.5 * it / iters_per_T)
            h_new = project_feasible(h_curr - alpha * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_curr = h_new
            elif c5_new < compute_c5(h_curr) + 1e-12:
                h_curr = h_new
            else:
                # Reduce step
                alpha *= 0.5
                h_new = project_feasible(h_curr - alpha * grad, n)
                h_curr = h_new
    return best_h, best_c5

def sa_binary(h, n, n_iters, T0, T1, rng, start_time, budget):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while b.sum() > target:
        idx = np.where(b == 1)[0]
        b[idx[rng.randint(len(idx))]] = 0
    while b.sum() < target:
        idx = np.where(b == 0)[0]
        b[idx[rng.randint(len(idx))]] = 1
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    for it in range(n_iters):
        if time.time() - start_time > budget - 10:
            break
        T = T0 * (T1/T0) ** (it / n_iters)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0; b[j] = 1
        new_c = compute_c5(b)
        if new_c <= curr or (T > 1e-14 and rng.random() < np.exp(-(new_c-curr)/T)):
            curr = new_c
            if new_c < best:
                best = new_c
                best_b = b.copy()
        else:
            b[i] = 1; b[j] = 0
    return best_b, best
