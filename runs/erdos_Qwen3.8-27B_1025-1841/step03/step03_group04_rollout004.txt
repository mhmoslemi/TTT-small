import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0
_G_IDX = None

def _softmax_obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_IDX is not None:
        corr = corr[_G_IDX]
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc = np.minimum(Tc, 0.0)
    lse = np.log(np.sum(np.exp(Tc)))
    return (mc + lse / T) * dx

def _eq_fun(x):
    return np.sum(x) - _G_N / 2.0

def _eq_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def get_top_idx(h, count=5, tol_frac=0.003):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    thr = mc - tol_frac * mc
    idx = np.where(corr >= thr)[0]
    if len(idx) > count:
        tv = corr[idx]
        idx = idx[np.argsort(-tv)[:count]]
    elif len(idx) == 0:
        idx = np.array([int(np.argmax(corr))])
    return idx

def slsqp_solve(h, n, T, top_idx, maxiter, ftol):
    global _G_N, _G_T, _G_IDX
    _G_N = n
    _G_T = T
    _G_IDX = top_idx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_softmax_obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h

def coord_descent(h, n, best_c5, steps, n_passes, time_limit, start_time):
    for _ in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:
                return h, best_c5
            old = h[j]
            best_v = old
            best_c = best_c5
            for s in steps:
                for sg in (-1.0, 1.0):
                    v = old + sg * s
                    if 0.0 <= v <= 1.0 and abs(v - old) > 1e-12:
                        trial = h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_v = v
            for v in (0.0, 1.0):
                if abs(v - old) > 1e-6:
                    trial = h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_v = v
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def downsample(h, new_n):
    old_n = len(h)
    if new_n >= old_n:
        return make_feasible(h.copy(), new_n) if new_n == old_n else make_feasible(
            np.interp(np.linspace(0, 2, new_n), np.linspace(0, 2, old_n), h), new_n)
    idx = np.linspace(0, old_n - 1, new_n).astype(int)
    return make_feasible(h[idx], new_n)

def upsample(h, new_n):
    old_n = len(h)
    hx = np.linspace(0, 2, old_n)
    nx = np.linspace(0, 2, new_n)
    h_new = np.interp(nx, hx, h)
    return make_feasible(h_new, new_n)
