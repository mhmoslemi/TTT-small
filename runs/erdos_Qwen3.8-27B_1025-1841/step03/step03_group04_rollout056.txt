Let me also add a "targeted flip" phase for the binary case:
- Find the worst shift k*
- For that shift, corr[k*] = sum_i h[i]*(1-h[i+k*])
- To reduce this, flip h[i] from 1 to 0 where h[i+k*] is close to 1 (reduces the term)
- Or flip h[i] from 0 to 1 where h[i+k*] is close to 0 (this increases the term, bad)
- Actually: if h[i]=1 and h[i+k*]=1, the term h[i]*(1-h[i+k*])=0. Flipping h[i] to 0 makes it 0*(1-1)=0. No change.
- If h[i]=1 and h[i+k*]=0, the term is 1*1=1. Flipping h[i] to 0 makes it 0. Reduces by 1!
- If h[i]=0 and h[i+k*]=1, the term is 0*0=0. Flipping h[i] to 1 makes it 1*0=0. No change.
- If h[i]=0 and h[i+k*]=0, the term is 0*1=0. Flipping h[i] to 1 makes it 1*1=1. Increases by 1! Bad.

So to reduce corr[k*]: flip 1→0 at positions i where h[i+k*]=0 (in the binary case).
To maintain sum: simultaneously flip 0→1 at a position j.
The flip 0→1 at j changes corr by:
- At shift k*: if h[j+k*]=0, increases corr[k*] by 1. If h[j+k*]=1, no change.
- At other shifts: various changes.

So the net change at shift k* from flipping (i:1→0) and (j:0→1):
- If h[i+k*]=0 and h[j+k*]=1: net change = -1 + 0 = -1. 
- If h[i+k*]=0 and h[j+k*]=0: net change = -1 + 1 = 0.
- We want the first case.

This gives a systematic way to find good flips.

                # d/dh[j] = (1-h[j+k]) [if j is the h[i] slot] + (-h[j-k]) [if h[j] = h[i+k] slot]
                # For shift k>0: valid i in [0, n-1-k]
                #   h[j] as h[i]: contribution (1-h[j+k]) if j in [0,n-1-k]
                #   h[j] as h[i+k]: contribution -h[i] where i=j-k, if j-k in [0,n-1-k]
                pass
            # Use simpler approach: finite difference on the top shift
            alpha = 0.002
            best_h_trial = best_h.copy()
            # Nudge: reduce at positions contributing to the max shift
            # corr[ck] involves pairs (i, i+k) for valid i
            if k > 0:
                for i in range(n - k):
                    if best_h[i] > 0.1 and (1.0 - best_h[i + k]) > 0.5:
                        best_h_trial[i] -= alpha
            elif k < 0:
                kk = -k
                for i in range(kk, n):
                    if best_h[i] > 0.1 and (1.0 - best_h[i - kk]) > 0.5:
                        best_h_trial[i] -= alpha
            best_h_trial = make_feasible(best_h_trial, n)
            c5_try = compute_c5(best_h_trial)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = best_h_trial
            else:
                alpha = alpha * 0.7

    # Phase 5: Binary local search with smart flips
    if budget_s - elapsed() > 150:
        h_bin = (best_h > 0.5).astype(float)
        deficit = n / 2.0 - np.sum(h_bin)
        if deficit > 0:
            zi = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zi])[:int(round(deficit))]
            h_bin[order] = 1.0
        elif deficit < 0:
            oi = np.where(h_bin == 1)[0]
            order = np.argsort(best_h[oi])[:int(round(-deficit))]
            h_bin[order] = 0.0
        c5_bin = compute_c5(h_bin)

        for itr in range(20):
            if budget_s - elapsed() < 30:
                break
            improved = False
            n_try = 400
            for _ in range(n_try):
                i1 = rng.randint(n)
                i2 = rng.randint(n)
                if h_bin[i1] == h_bin[i2] or i1 == i2:
                    continue
                h_bin[i1], h_bin[i2] = h_bin[i2], h_bin[i1]
                c5_try = compute_c5(h_bin)
                if c5_try < c5_bin - 1e-14:
                    c5_bin = c5_try
                    improved = True
                else:
                    h_bin[i1], h_bin[i2] = h_bin[i2], h_bin[i1]
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
            if not improved:
                break

    # Phase 6: Coordinate descent (multi-pass)
    if budget_s - elapsed() > 60:
        for cd_pass in range(4):
            if budget_s - elapsed() < 20:
                break
            for j in range(n):
                if budget_s - elapsed() < 10:
                    break
                old = best_h[j]
                best_c = best_c5
                best_v = old
                cands = [0.0, 1.0, 0.1, 0.3, 0.5, 0.7, 0.9]
                for v in cands:
                    if abs(v - old) < 1e-12:
                        continue
                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_v = v
                for s in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    for sgn in [-1, 1]:
                        v = old + sgn * s
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n)
                            c = compute_c5(trial)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_v = v
                if best_v != old:
                    best_h[j] = best_v
                    best_h = make_feasible(best_h, n)
                    best_c5 = best_c

    # Phase 7: Final high-T SLSQP polish
    if budget_s - elapsed() > 50:
        for T in [800, 1200]:
            if budget_s - elapsed() < 40:
                break
            _G_N = n
            _G_T = T
            _G_LAMBDA = 0.0
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res = minimize(_obj_full, best_h, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
                h_new = make_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 8: Random restarts
    for rstr in range(5):
        if budget_s - elapsed() < 50:
            break
        scale = 0.01 + 0.005 * rstr
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        _G_N = n
        _G_T = 400.0
        _G_LAMBDA = 500.0
        bounds = [(0.0, 1.0)] * n
        try:
            res = fmin_l_bfgs_b(_obj_full, h_try, pgtol=1e-9, maxiter=200,
                               bounds=bounds, iprint=0)
            h_new = make_feasible(res[0], n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Phase 9: Upsample to 2n if time allows
    if budget_s - elapsed() > 250:
        n_up = 2 * n
        h_up = upsample_h(best_h, n_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 + 0.003:
            _G_N = n_up
            _G_T = 150.0
            _G_LAMBDA = 500.0
            bounds = [(0.0, 1.0)] * n_up
            try:
                res = fmin_l_bfgs_b(_obj_full, h_up, pgtol=1e-9, maxiter=300,
                                   bounds=bounds, iprint=0)
                h_new = make_feasible(res[0], n_up)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
            if budget_s - elapsed() > 100:
                _G_N = len(best_h)
                _G_T = 400.0
                corr2 = get_corr(best_h)
                mc2 = np.max(corr2)
                thr2 = mc2 - 0.005 * mc2
                ti2 = np.where(corr2 >= thr2)[0]
                if len(ti2) > 12:
                    so2 = np.argsort(-corr2[ti2])
                    ti2 = ti2[so2[:12]]
                elif len(ti2) == 0:
                    ti2 = np.array([int(np.argmax(corr2))])
                _G_TTOP = ti2
                bounds2 = [(0.0, 1.0)] * len(best_h)
                cons2 = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
                try:
                    res2 = minimize(_obj_top, best_h, method='SLSQP', bounds=bounds2,
                                   constraints=cons2, options={'maxiter': 200, 'ftol': 1e-15})
                    h_new2 = make_feasible(res2.x, len(best_h))
                    c5_new2 = compute_c5(h_new2)
                    if c5_new2 < best_c5:
                        best_c5 = c5_new2
                        best_h = h_new2
                except Exception:
                    pass

    # Final: binary snap near-extreme values
    if budget_s - elapsed() > 15:
        for j in range(len(best_h)):
            if budget_s - elapsed() < 5:
                break
            for val in [0.0, 1.0]:
                if abs(best_h[j] - val) > 0.05:
                    continue
                trial = best_h.copy()
                trial[j] = val
                trial = make_feasible(trial, len(best_h))
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best_h = trial

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
