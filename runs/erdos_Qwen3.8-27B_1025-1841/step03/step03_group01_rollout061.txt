from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)[-int(abs(diff)):]
        else:
            idx = np.argsort(h)[:int(abs(diff)):]
        h[idx] = np.clip(h[idx] + np.sign(diff), 0.0, 1.0)
    return h

def upsample_h(h, new_n):

    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection"""
    target = n / 2.0
    # Bisection on lambda

        s = np.sum(proj)
        if s > target:
            hi = mid
        else:

        t_count += 1

        c5 = np.max(corr)

        weights = np.exp(weights / 0.05)  # concentration

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(h)[-max(1, int(np.ceil(abs(diff)))):]
        else:
            idx = np.argsort(h)[:max(1, int(np.ceil(abs(diff))))]
        h[idx] = np.clip(h[idx] + np.sign(diff), 0.0, 1.0)
    return h

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Numerical stability
    max_corr = np.max(corr)
    exp_corr = np.exp(T * (corr - max_corr))
    sum_exp = np.sum(exp_corr)
    obj = (max_corr + np.log(sum_exp) / T) * dx
    probs = exp_corr / sum_exp  # softmax weights, sums to 1
    
    # Gradient: dx * T * sum_k probs[k] * dcorr_k/dh[j]
    # For lag k (k_idx in corr array, actual lag = k_idx - (n-1)):
    #   if lag >= 0:
    #     dcorr/dh[j] = (1-h[j-lag]) for lag<=j<n
    #                  - h[j+lag] for j<n-lag
    #   if lag < 0, l=-lag:
    #     dcorr/dh[j] = (1-h[j+l]) for j<n-l
    #                  - h[j-l] for l<=j<n
    
    g = np.zeros(n)
    j_arr = np.arange(n)
    for k_idx in np.where(probs > 1e-15)[0]:
        p = probs[k_idx] * T * dx
        lag = k_idx - (n - 1)
        if lag >= 0:
            m1 = j_arr >= lag
            m2 = j_arr < n - lag
            g[m1] += p * (1.0 - h[j_arr[m1] - lag])
            g[m2] -= p * h[j_arr[m2] + lag]
        else:
            l = -lag
            m1 = j_arr < n - l
            m2 = j_arr >= l
            g[m1] += p * (1.0 - h[j_arr[m1] + l])
            g[m2] -= p * h[j_arr[m2] - l]

        nswap = min(nswap, len(ones), len(zeros))
        sf = rng.choice(ones, nswap, replace=False)
        st = rng.choice(zeros, nswap, replace=False)
        h[sf] = 0.0
        h[st] = 1.0
        nc = np.correlate(h, 1.0 - h, mode='full') * dx
        nc5 = float(np.max(nc))
        delta = nc5 - c5
        if delta <= 0 or (T > 1e-8 and rng.rand() < np.exp(-delta / T)):
            c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[sf] = 1.0
            h[st] = 0.0
        if step % 2000 == 0:
            T *= 0.97
        if step > 0 and step % restart_every == 0:
            h = best_h.copy()
            nj = int(rng.randint(3, 9))
            nj = min(nj, len(ones), len(zeros))
            jf = rng.choice(ones, nj, replace=False)
            jt = rng.choice(zeros, nj, replace=False)
            h[jf] = 0.0
            h[jt] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
            T = max(T, 0.003)
    return best_h, best_c5

def greedy_local_search(h, n, n_trials, rng):
    dx = 2.0 / n
    h = to_binary_fixed(h, n, rng)
    c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(n_trials):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc5 = compute_c5(h)
        if nc5 <= c5:
            c5 = nc5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i], h[j] = 1.0, 0.0
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile(np.array([1.0, 0.5, 0.0, 0.5]), 64)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Continuous optimization at n=2048
    n1 = 2048
    h1 = upsample_h(h0, n1)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    if tl() > 200:
        h1, _ = proj_grad_descent(h1, n1, T=30.0, max_iters=3000, time_limit=min(120, tl() - 60))
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()

        if tl() > 100:
            h1, _ = proj_grad_descent(h1, n1, T=100.0, max_iters=2000, time_limit=min(80, tl() - 40))
            c1 = compute_c5(h1)
            if c1 < best_c5:
                best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary SA at n=2048
    n2 = 2048
    h2 = upsample_h(best_h, n2)
    h2b = to_binary_fixed(h2, n2, rng)
    c2 = compute_c5(h2b)
    if c2 < best_c5:
        best_c5, best_h = c2, h2b.copy()

    if tl() > 100:
        h2b, c2 = sa_multiswap(h2b, n2, max_steps=400000, time_limit=min(350, tl() - 80),
                               rng=rng, T_init=0.004, max_kswap=3)
        if c2 < best_c5:
            best_c5, best_h = c2, h2b.copy()

    # Phase 3: Try at n=4096
    n3 = 4096
    if tl() > 120:
        h3 = upsample_h(best_h, n3)
        h3b = to_binary_fixed(h3, n3, rng)
        c3 = compute_c5(h3b)
        if c3 < best_c5:
            best_c5, best_h = c3, h3b.copy()

        if tl() > 80:
            h3b, c3 = sa_multiswap(h3b, n3, max_steps=200000, time_limit=min(250, tl() - 50),
                                   rng=rng, T_init=0.003, max_kswap=3)
            if c3 < best_c5:
                best_c5, best_h = c3, h3b.copy()

    # Phase 4: Greedy polish at whatever resolution we're at
    if tl() > 30:
        nn = len(best_h)
        h_best, c_best = greedy_local_search(best_h, nn, n_trials=50000, rng=rng)
        if c_best < best_c5:
            best_c5, best_h = c_best, h_best

    # Phase 5: Another SA round from best (different T, more multi-swaps)
    if tl() > 60:
        nn = len(best_h)
        h_t, c_t = sa_multiswap(best_h, nn, max_steps=150000, time_limit=min(100, tl() - 20),
                                rng=rng, T_init=0.002, max_kswap=5, restart_every=5000)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    # Final continuous polish
    if tl() > 40:
        nn = len(best_h)
        h_f, _ = proj_grad_descent(best_h, nn, T=500.0, max_iters=1000, time_limit=min(40, tl() - 10))
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, float(best_c5), len(best_h)
