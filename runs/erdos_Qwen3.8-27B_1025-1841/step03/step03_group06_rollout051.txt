def compute_gradient(h, w, n):

    lags = np.arange(2*n-1) - (n-1)
    j_idx = np.arange(n)
    idx1 = j_idx[:, None] - lags[None, :]  # (n, 2n-1)
    idx2 = j_idx[:, None] + lags[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    h1 = np.where(valid1, h[idx1], 0.0)  # (n, 2n-1)
    h2 = np.where(valid2, h[idx2], 0.0)
    
    term1 = (w[None, :] * (1 - h1) * valid1).sum(axis=1)
    term2 = (w[None, :] * h2 * valid2).sum(axis=1)
    
    return dx * (term1 - term2)
