    """Project h_raw onto {0 <= h <= 1, sum(h) = n/2}"""
    if target is None:
        target = n / 2.0
    # Binary search on the Lagrange multiplier
    lo, hi = -1.0, 2.0  # range for the shift
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h_raw - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid  # need to shift more
        else:
            hi = mid
    h_proj = np.clip(h_raw - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any rounding
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

import numpy as np
from scipy.optimize import minimize
import time

G_N = 0
G_T = 100.0
G_TOP_IDX = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def project_h(h_raw, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h_raw - mid, 0.0, 1.0)
        if np.sum(h_p) > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h_raw - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_p < 1.0
        else:
            mask = h_p > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_p = h_p.copy()
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def obj_smooth(x):
    n = G_N
    T = G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if G_TOP_IDX is not None:
        corr = corr[G_TOP_IDX]
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_fun(x):
    return np.sum(x) - G_N / 2.0

def con_jac(x):
    return np.ones(G_N)

def game_iterate(h, n, n_outer, n_inner, eta_start=10.0, eta_end=1.0, start_time=None, budget_s=0):
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    p = np.ones(n_shifts)
    p /= p.sum()
    h_curr = h.copy()
    best_local_c5 = compute_c5(h_curr)
    best_local_h = h_curr.copy()
    eta_range = np.linspace(eta_start, eta_end, max(n_outer, 1))
    for outer in range(n_outer):
        if start_time is not None and budget_s - (time.time() - start_time) < 30:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        T_p = eta_range[outer] / n_shifts
        p_new = np.exp(corr * dx / T_p)
        p_new /= p_new.sum()
        p = 0.5 * p + 0.5 * p_new
        p /= p.sum()
        full_corr = np.correlate(h_curr, p, mode='full')
        Mh = full_corr[3 * n - 3 - np.arange(n)]
        for inner in range(n_inner):
            if start_time is not None and budget_s - (time.time() - start_time) < 15:
                break
            alpha = 2.0 / (n + float(inner) + 1.0)
            h_raw = h_curr + alpha * (2.0 * Mh / dx)
            h_new = project_h(h_raw, n)
            if np.sum(np.abs(h_new - h_curr)) < 1e-14:
                h_curr = h_new
                break
            h_curr = h_new
        c_check = compute_c5(h_curr)
        if c_check < best_local_c5:
            best_local_c5 = c_check
            best_local_h = h_curr.copy()
    return best_local_h, best_local_c5
