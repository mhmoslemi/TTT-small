```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def subgradient_step(h, n, alpha, n_iter=100):

    best_h = h.copy()
    best_c = compute_c5(h)

        k_star = int(np.argmax(corr))
```

```python
def smooth_subgradient_optimize(h, n, t, max_iters, time_limit, start_time):
    """Minimize smooth max of correlations using projected gradient."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    # Initial step size
    alpha = 0.01
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit - 30:
            break
        
        # Compute smoothed objective gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        
        # Soft max weights
        mc = np.max(corr)
        shifted = corr - mc
        weights = np.exp(t * shifted) / np.sum(np.exp(t * shifted))
        
        # Gradient: weighted sum of individual gradients
        grad = np.zeros(n)
        # Only use top ~20 lags for efficiency
        top_k = np.argsort(weights)[-min(20, len(weights)):]
        for m in top_k:
            w = weights[m]
            if w < 1e-6:
                continue
            lag = m - (n - 1)
            if lag > 0:
                grad[:n-lag] += w * (1.0 - h[lag:])
                grad[lag:] += w * (-h[:n-lag])
            elif lag < 0:
                grad[-lag:] += w * (1.0 - h[:n+lag])
                grad[:n+lag] += w * (-h[-lag:])
            else:
                grad[:] += w * (1.0 - 2.0 * h)
        
        grad *= dx  # Scale by dx
        
        # Step
        h_new = h - alpha * grad
        h_new = make_feasible(h_new, n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
```

```python
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute uniformly, but respect bounds
        h_new = h + diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix residual with targeted adjustment
        residual = target - np.sum(h_new)
        if abs(residual) > 1e-14:
            if residual > 0:
                idx = np.where(h_new < 1.0)[0]
            else:
                idx = np.where(h_new > 0.0)[0]
            if len(idx) > 0:
                per = residual / len(idx)
                h_new[idx] = np.clip(h_new[idx] + per, 0.0, 1.0)
        return h_new
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def subgrad_opt(h, n, temperature, max_iters, time_limit, start_time):
    """Smooth subgradient optimization."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / n
    alpha = 0.005
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit - 30:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        exp_vals = np.exp(temperature * (corr - mc))
        weights = exp_vals / np.sum(exp_vals)
        
        # Compute gradient (top active lags only)
        threshold = 1e-3
        top_indices = np.where(weights > threshold)[0]
        grad = np.zeros(n)
        for m in top_indices:
            w = weights[m]
            lag = m - (n - 1)
            if lag > 0:
                grad[:n-lag] += w * (1.0 - h[lag:])
                grad[lag:] += -w * h[:n-lag]
            elif lag < 0:
                grad[-lag:] += w * (1.0 - h[:n+lag])
                grad[:n+lag] += -w * h[-lag:]
            else:
                grad += w * (1.0 - 2.0 * h)
        grad *= dx
        
        # Step with backtracking
        improved = False
        for _ in range(8):
            h_trial = h - alpha * grad
            h_trial = make_feasible(h_trial, n)
            c_trial = compute_c5(h_trial)
            if c_trial < compute_c5(h):
                h = h_trial
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            alpha = max(alpha * 0.5, 1e-6)
            if alpha < 1e-7:
                break
        else:
            alpha = min(alpha * 1.02, 0.02)
    
    return best_h, best_c

def coord_descent(h, n, max_rounds, time_limit, start_time, step_sizes=None):
    """Fine coordinate descent."""
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    best_c = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit - 20:
            break
        
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 20:
                return h, best_c
            
            old = h[j]
            best_val = old
            best_obj = best_c
            
            # Try perturbations
            for step in step_sizes:
                for direction in [-1, 1]:
                    trial_val = old + direction * step
                    if trial_val < -0.001 or trial_val > 1.001:
                        continue
                    trial_val = np.clip(trial_val, 0.0, 1.0)
                    h[j] = trial_val
                    h_temp = make_feasible(h, n)
                    c = compute_c5(h_temp)
                    if c < best_obj - 1e-13:
                        best_obj = c
                        best_val = trial_val
            
            # Also try 0 and 1
            for tv in [0.0, 1.0]:
                if abs(tv - old) < 1e-6:
                    continue
                h[j] = tv
                h_temp = make_feasible(h, n)
                c = compute_c5(h_temp)
                if c < best_obj - 1e-13:
                    best_obj = c
                    best_val = tv
            
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        
        if not improved:
            break
    
    return h, best_c

def block_search(h, n, time_limit, start_time, rng, max_blocks=500):
    """Try block perturbations."""
    best_h = h.copy()
    best_c = compute_c5(h)

        c = compute_c5(trial)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        residual = target - np.sum(h)
        if abs(residual) > 1e-14:
            if residual > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                per = residual / len(idx)
                h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def subgrad_opt(h, n, temperature, max_iters, time_limit, start_time):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / n
    alpha = 0.003
    stall_count = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit - 30:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        exp_vals = np.exp(temperature * (corr - mc))
        total_exp = np.sum(exp_vals)
        if total_exp < 1e-300:
            break
        weights = exp_vals / total_exp

        top_idx = np.where(weights > 5e-4)[0]
        if len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(weights))])

        grad = np.zeros(n)
        for m in top_idx:
            w = weights[m]
            lag = m - (n - 1)
            if lag > 0:
                grad[:n - lag] += w * (1.0 - h[lag:])
                grad[lag:] += -w * h[:n - lag]
            elif lag < 0:
                grad[-lag:] += w * (1.0 - h[:n + lag])
                grad[:n + lag] += -w * h[-lag:]
            else:
                grad += w * (1.0 - 2.0 * h)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        cur_c = compute_c5(h)
        improved = False
        for _ in range(10):
            h_trial = h - alpha * grad
            h_trial = make_feasible(h_trial, n)
            c_trial = compute_c5(h_trial)
            if c_trial < cur_c - 1e-14:
                h = h_trial
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = h.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                improved = True
                break
            alpha *= 0.5

        if not improved:
            alpha = max(alpha, 1e-8)
            stall_count += 1
            if stall_count > 200:
                break
            alpha *= 0.7
        else:
            alpha = min(alpha * 1.03, 0.015)

    return best_h, best_c

def coord_descent(h, n, max_rounds, time_limit, start_time, steps=None):
    if steps is None:
        steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h)

    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit - 15:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 15:
                return h, best_c
            old = h[j]
            best_val = old
            best_obj = best_c
            for step in steps:
                for d in (-1, 1):
                    tv = old + d * step
                    if tv < -1e-9 or tv > 1 + 1e-9:
                        continue
                    tv = np.clip(tv, 0.0, 1.0)
                    h[j] = tv
                    ht = make_feasible(h, n)
                    c = compute_c5(ht)
                    if c < best_obj - 1e-13:
                        best_obj = c
                        best_val = tv
            if abs(old) > 1e-6:
                h[j] = 0.0
                ht = make_feasible(h, n)
                c = compute_c5(ht)
                if c < best_obj - 1e-13:
                    best_obj = c
                    best_val = 0.0
            if abs(old - 1.0) > 1e-6:
                h[j] = 1.0
                ht = make_feasible(h, n)
                c = compute_c5(ht)
                if c < best_obj - 1e-13:
                    best_obj = c
                    best_val = 1.0
            h[j] = best_val
            h = make_feasible(h, n)
            if best_val != old:
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def block_search(h, n, time_limit, start_time, rng, max_iter=800):
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iter):
        if time.time() - start_time > time_limit - 15:
            break
        blen = rng.randint(2, max(3, n // 40))
        start = rng.randint(0, n - blen)
        end = start + blen
        trial = best_h.copy()
        delta = rng.uniform(-0.4, 0.4)
        trial[start:end] = np.clip(trial[start:end] + delta, 0.0, 1.0)
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c - 1e-13:
            best_c = c
            best_h = trial
            # Continue with smaller perturbations
            for _ in range(5):
                if time.time() - start_time > time_limit - 15:
                    break
                blen2 = rng.randint(2, blen + 1)
                s2 = max(0, start - rng.randint(0, 3))
                e2 = min(n, s2 + blen2)
                trial2 = best_h.copy()
                trial2[s2:e2] = np.clip(trial2[s2:e2] + rng.uniform(-0.1, 0.1), 0.0, 1.0)
                trial2 = make_feasible(trial2, n)
                c2 = compute_c5(trial2)
                if c2 < best_c - 1e-13:
                    best_c = c2
                    best_h = trial2
    return best_h, best_c

def random_restart_opt(n, seed, budget_s, start_time, temperature=50):
    rng = np.random.RandomState(seed)
    h = 0.5 + 0.4 * rng.randn(n)
    h = make_feasible(h, n)
    h, c = subgrad_opt(h, n, temperature, 300, budget_s, start_time)
    return h, c
```
