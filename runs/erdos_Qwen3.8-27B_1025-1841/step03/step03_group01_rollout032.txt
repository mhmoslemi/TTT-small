    """Try to reduce overlap at shift d by moving mass."""
    dx = 2.0 / n
    # Overlap at lag d: sum_i h[i] * (1-h[i+d]) for valid i
    if d >= 0:
        valid_i = range(0, n - d)
    else:
        valid_i = range(-d, n)
    
    # Find the worst pairs
    scores = []
    for i in valid_i:
        j = i + d
        # Contribution: h[i] * (1-h[j])
        # To reduce: decrease h[i] or increase h[j]
        # Moving mass from i to j: h[i] -= eps, h[j] += eps
        # Change in overlap: -eps*(1-h[j]) - h[i]*eps (approximately, if we move eps from i to j)
        # Actually: new h[i] = h[i]-eps, new h[j] = h[j]+eps
        # New contribution: (h[i]-eps)*(1-h[j]-eps) ≈ h[i]*(1-h[j]) - eps*(1-h[j]) - eps*h[i]
        # Change ≈ -eps*(1-h[j] + h[i])
        # This reduces the overlap at shift d by eps*(h[i]+1-h[j])
        # But it also affects other shifts!
        scores.append((i, j, h[i] * (1 - h[j])))

    scores.sort(key=lambda x: -x[2])
    
    # Move mass from worst pairs
    h_new = h.copy()
    for i, j, score in scores[:10]:  # top 10 worst pairs
        if h_new[i] > amount and h_new[j] < 1.0 - amount:
            h_new[i] -= amount
            h_new[j] += amount
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_simplex_box(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def subgradient_step(h, n, worst_d, eta):
    """One subgradient descent step at the worst shift."""
    grad = np.zeros(n)
    if worst_d >= 0:
        d = worst_d
        idx1 = np.arange(0, n - d)
        if len(idx1) > 0:
            grad[idx1] += (1 - h[idx1 + d])
            idx2 = np.arange(d, n)
            grad[idx2] -= h[idx2 - d]
    else:
        dd = -worst_d
        idx1 = np.arange(dd, n)
        if len(idx1) > 0:
            grad[idx1] += (1 - h[idx1 - dd])
            idx2 = np.arange(0, n - dd)
            grad[idx2] -= h[idx2 + dd]
    
    h_new = h - eta * grad
    h_new = project_onto_simplex_box(h_new, n)
    return h_new

def subgradient_descent(h, n, n_steps, eta0, decay):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    
    for t in range(n_steps):
        overlaps = compute_all_overlaps(h_curr)
        worst_idx = np.argmax(overlaps)
        worst_d = worst_idx - (n - 1)
        
        eta = eta0 / (1.0 + decay * t)
        h_curr = subgradient_step(h_curr, n, worst_d, eta)
        
        if t % 100 == 0:
            c5 = compute_c5(h_curr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()
    
    c5 = compute_c5(h_curr)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_curr.copy()
    
    return best_h, best_c5

def pair_swap_search(h, n, time_limit, rng):
    """Local search using pair swaps (preserves sum constraint)."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    start = time.time()
    
    while time.time() - start < time_limit:
        overlaps = compute_all_overlaps(h_curr)
        worst_idx = np.argmax(overlaps)
        worst_d = worst_idx - (n - 1)
        
        # Find top contributing pairs at the worst shift

            if d == 0:
