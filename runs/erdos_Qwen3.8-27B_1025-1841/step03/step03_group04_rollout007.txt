        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_full, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 2: Efficient coordinate descent (many passes)
    for p in range(25):
        if remaining() < 50:
            break
        best_h, c5_cd, n_imp = coord_descent_pass(best_h, n, remaining, min_remain=25)
        best_h = make_feasible(best_h, n)
        c5_cd = compute_c5(best_h)
        if c5_cd < best_c5:
            best_c5 = c5_cd
        if n_imp == 0:
            break

    # Phase 3: Shift-targeted SLSQP interleaved with CD
    for round_i in range(5):
        if remaining() < 60:
            break
        corr = get_corr(best_h)
        mc = float(np.max(corr))
        thr = mc - 0.005 * mc
        top_idx = np.where(corr >= thr)[0]
        if len(top_idx) > 20:
            tv = corr[top_idx]
            so = np.argsort(-tv)
            top_idx = top_idx[so[:20]]
        elif len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(corr))])
        _G_N = n
        _G_T = 500.0
        _G_TOP_IDX = top_idx
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_top, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass
        if remaining() > 40:
            best_h, _, _ = coord_descent_pass(best_h, n, remaining, min_remain=20)
            best_h = make_feasible(best_h, n)
            c5_cd = compute_c5(best_h)
            if c5_cd < best_c5:
                best_c5 = c5_cd

    # Phase 4: More CD passes
    for p in range(25):
        if remaining() < 35:
            break
        best_h, c5_cd, n_imp = coord_descent_pass(best_h, n, remaining, min_remain=18)
        best_h = make_feasible(best_h, n)
        c5_cd = compute_c5(best_h)
        if c5_cd < best_c5:
            best_c5 = c5_cd
        if n_imp == 0:
            break

    # Phase 5: Random restarts
    for restart in range(6):
        if remaining() < 70:
            break
        scale = 0.008 if restart < 3 else 0.02
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        _G_N = n
        _G_T = 300.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_full, h_try, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-15})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
            if remaining() > 30:
                best_h, _, _ = coord_descent_pass(best_h, n, remaining, min_remain=15)
                best_h = make_feasible(best_h, n)
                c5_cd = compute_c5(best_h)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
        except Exception:
            pass

    # Phase 6: Upsample to 1920
    if remaining() > 350:
        n_up = 1920
        h_up = upsample_h(best_h, n_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 + 0.003:
            _G_N = n_up
            _G_T = 200.0
            _G_TOP_IDX = None
            bounds_up = [(0.0, 1.0)] * n_up
            cons_up = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res_up = minimize(_obj_full, h_up, method='SLSQP',
                                  bounds=bounds_up, constraints=cons_up,
                                  options={'maxiter': 400, 'ftol': 1e-15})
                c5_r = compute_c5(res_up.x)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(res_up.x, n_up)
            except Exception:
                pass
            if remaining() > 120:
                for p in range(8):
                    if remaining() < 60:
                        break
                    best_h, _, n_imp = coord_descent_pass(best_h, n_up, remaining, min_remain=35)
                    best_h = make_feasible(best_h, n_up)
                    c5_cd = compute_c5(best_h)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                    if n_imp == 0:
                        break

    # Phase 7: Final SLSQP polish
    if remaining() > 60:
        n_final = len(best_h)
        _G_N = n_final
        _G_T = 600.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n_final
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_full, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n_final)
        except Exception:
            pass

    # Phase 8: Final CD polish
    for p in range(15):
        if remaining() < 15:
            break
        best_h, _, n_imp = coord_descent_pass(best_h, len(best_h), remaining, min_remain=8)
        best_h = make_feasible(best_h, len(best_h))
        c5_cd = compute_c5(best_h)
        if c5_cd < best_c5:
            best_c5 = c5_cd
        if n_imp == 0:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
