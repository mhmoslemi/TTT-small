import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def sa_optimize(h_start, n, time_budget, rng, t_init=0.004, t_min=1e-6):
    h = h_start.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start_t = time.time()
    n_iters = 0
    stagnation = 0
    best_iter = 0
    
    while True:
        elapsed = time.time() - start_t
        if elapsed >= time_budget:
            break
        time_frac = max(1.0 - elapsed / time_budget, 0.0)
        t = max(t_init * (time_frac ** 1.8), t_min)
        
        # Adaptive: increase t if stuck

            t = max(t * 3.0, t_init * 0.3)
            stagnation = 0
        
        h_old = h.copy()
        move_type = rng.randint(5)
        
        if move_type == 0:
            i = rng.randint(n)
            h[i] = np.clip(h[i] + rng.normal(0, 0.12), 0.0, 1.0)
        elif move_type == 1:
            L = rng.randint(2, 6)
            s = rng.randint(n)
            idxs = (np.arange(s, s + L)) % n

        else:
            # Pair swap (preserves sum)
            i, j = rng.choice(n, size=2, replace=False)
            tmp = h[i]
            h[i] = h[j]
            h[j] = tmp
        
        h = make_feasible(h, n)

def slsqp_optimize(h_start, n, T, max_iter):
    dx = 2.0 / n
    def obj_fn(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        log_sum = np.log(np.sum(np.exp(T * shifted)))
        return (mc + log_sum / T) * dx
    
    def cons_fun(h):
        return np.sum(h) - n / 2.0
    
    def cons_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_fn, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start.copy(), n)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_t = time.time()
    
    deltas = [0.02, -0.02, 0.01, -0.01, 0.05, -0.05, 0.005, -0.005]
    
    while (time.time() - start_t) < time_budget:
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if (time.time() - start_t) >= time_budget:
                break
            old_val = h[idx]
            for d in deltas:
                cand = old_val + d
                if cand < -0.01 or cand > 1.01:
                    continue
                trial = h.copy()
                trial[idx] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial.copy()
                    h = trial
                    old_val = cand
                    improved = True
            # Try binary
            for bv in (0.0, 1.0):
                if abs(bv - h[idx]) < 1e-9:
                    continue
                trial = h.copy()
                trial[idx] = bv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial.copy()
                    h = trial
                    improved = True
        if not improved:
            break
    
    return best_h, best_c5
