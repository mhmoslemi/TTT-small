import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_onto_box_sum(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_at(h, shift):
    """Subgradient of c(shift) w.r.t. h, where shift can be negative."""
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_opt(h, n_iter, T0, T1, lr0, lr1, top_k, check_every):
    """FISTA with softmax subgradient for min-max problem."""
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    t_k = 1.0
    best_c, best_h = compute_c5(x)[0], x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr0 + (lr1 - lr0) * f
        
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        
        # Softmax weights
        shifted = corr - mc
        max_shifted = np.max(shifted)
        exps = np.exp(T * (shifted - max_shifted))
        tot = np.sum(exps)
        w = exps / tot
        
        # Top-k subgradients
        idx = np.argsort(w)[::-1][:top_k]
        g = np.zeros(n)
        for i in idx:
            if w[i] < 1e-14:
                break
            g += w[i] * subgrad_at(z, i - (n - 1))
        
        # FISTA step
        x_new = project_onto_box_sum(z - lr * g, n)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x_new + (t_k - 1.0) / t_next * (x_new - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = x_new
        t_k = t_next
        
        if (it + 1) % check_every == 0:
            c, _ = compute_c5(x)
            if c < best_c:
                best_c, best_h = c, x.copy()
    
    c, _ = compute_c5(x)
    if c < best_c:
        best_c, best_h = c, x.copy()
    return best_h, best_c

    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def fista(h, nit, T0, T1, lr0, lr1, topk, chk):

    x = h.astype(float).copy()
    z = x.copy()

    bc, bh = compute_c5(x), x.copy()
    for it in range(nit):

_S = {}  # global state for SLSQP callbacks

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_cs(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = np.clip(h - mid, 0.0, 1.0)
        if np.sum(vals) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _sg(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def fista(h, nit, T0, T1, lr0, lr1, topk, chk):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(nit):
        fr = it / max(nit - 1, 1)
        T = T0 * (T1 / T0) ** fr
        lr = lr0 + (lr1 - lr0) * fr
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        sh = corr - mc
        ms = np.max(sh)
        ex = np.exp(T * (sh - ms))
        tot = np.sum(ex)
        w = ex / tot
        idx = np.argsort(w)[::-1][:topk]
        g = np.zeros(n)
        for i in idx:
            wi = w[i]
            if wi < 1e-14:
                break
            g += wi * _sg(z, i - (n-1))
        xn = project_cs(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)

        x = xn

        if (it + 1) % chk == 0:

                bc = c
                bh = x.copy()
    c = compute_c5(x)
    if c < bc:
        bc = c
        bh = x.copy()
    return bh, bc

    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    sh = c - mc
    ms = np.max(sh)
    return (mc + ms + np.log(np.sum(np.exp(T * (sh - ms)))) / T) * dx

def _scon(h):
    return np.sum(h) - _S['n'] / 2.0

def _sjac(h):
    return np.ones(_S['n'])

def slsqp(h, T, mi):
    n = len(h)
    _S['T'] = T
    _S['n'] = n
    try:
        r = minimize(_sobj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_cs(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:

    return t[:n].astype(float)

    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_cs(np.interp(nx, ox, h), nn)
