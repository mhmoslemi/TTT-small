    valid_i_second = (m >= -(n-1)) & (m <= n-1-i)
    # Validity: 0 <= j-m <= n-1

    im[im < 0] = 0  # will be masked out

    ip = i + m  # index into b for "second factor" contribution from i

    h_init = make_feasible(h_init, n0)
    b_initial = to_binary(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Multi-replica SA at n0
    t_p1 = budget_s * 0.30
    est_time = 0.000012 * (n0 / 960.0) ** 2
    max_it_p1 = int(t_p1 / est_time)
    max_it_p1 = max(max_it_p1, 50000)
    b1, e1 = multi_replica_sa(b_initial, n0, max_it_p1, [0.2, 0.1, 0.05, 0.025], rng,
                              time_limit=budget_s * 0.30, start_time=start_time)
    c5_1 = e1 * (2.0 / n0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = b1.copy()

    # Phase 2: Upscale to higher resolution and SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        n1 = n0 * 2
        h_up = upsample_h(best_h, n1)
        b_up = to_binary(h_up, n1)
        t_p2 = remaining * 0.45
        est_time_1 = 0.000012 * (n1 / 960.0) ** 2
        max_it_p2 = int(t_p2 / est_time_1)
        max_it_p2 = max(max_it_p2, 20000)
        rng2 = np.random.RandomState(seed + 100)
        b2, e2 = multi_replica_sa(b_up, n1, max_it_p2, [0.3, 0.15, 0.07, 0.035], rng2,
                                  time_limit=budget_s * 0.20, start_time=start_time)
        c5_2 = e2 * (2.0 / n1)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = b2.copy()

    # Phase 3: Guided local search on best solution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        n_cur = len(best_h)
        b_cur = to_binary(best_h, n_cur)
        t_p3 = remaining * 0.30
        est_time_3 = 0.000012 * (n_cur / 960.0) ** 2 * 10
        max_it_p3 = int(t_p3 / est_time_3)
        max_it_p3 = max(max_it_p3, 2000)
        rng3 = np.random.RandomState(seed + 200)
        b3, e3 = guided_local_search(b_cur, n_cur, max_it_p3, rng3)
        c5_3 = e3 * (2.0 / n_cur)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = b3.copy()

    # Phase 4: Another SA round from perturbed best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        n_cur = len(best_h)
        b_pert = to_binary(best_h, n_cur)
        for _ in range(50):
            ones = np.where(b_pert == 1)[0]
            zeros = np.where(b_pert == 0)[0]
            b_pert[ones[rng.randint(len(ones))]] = 0
            b_pert[zeros[rng.randint(len(zeros))]] = 1
        t_p4 = remaining * 0.50
        est_time_4 = 0.000012 * (n_cur / 960.0) ** 2
        max_it_p4 = int(t_p4 / est_time_4)
        max_it_p4 = max(max_it_p4, 50000)
        rng4 = np.random.RandomState(seed + 300)
        b4, e4 = multi_replica_sa(b_pert, n_cur, max_it_p4, [0.1, 0.05, 0.025, 0.012], rng4,
                                  time_limit=budget_s * 0.20, start_time=start_time)
        c5_4 = e4 * (2.0 / n_cur)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = b4.copy()

    # Phase 5: Final guided local search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        n_cur = len(best_h)
        b_final = to_binary(best_h, n_cur)
        t_p5 = remaining * 0.60
        est_time_5 = 0.000012 * (n_cur / 960.0) ** 2 * 10
        max_it_p5 = int(t_p5 / est_time_5)
        max_it_p5 = max(max_it_p5, 1000)
        rng5 = np.random.RandomState(seed + 400)
        b5, e5 = guided_local_search(b_final, n_cur, max_it_p5, rng5)
        c5_5 = e5 * (2.0 / n_cur)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = b5.copy()

    # Final verification
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
