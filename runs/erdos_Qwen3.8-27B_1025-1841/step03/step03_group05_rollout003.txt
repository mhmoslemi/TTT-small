def solve_1d_max(a, b):
    """Solve min_{t in [0,1]} max_s (a[s] + b[s]*t)"""
    m = len(a)
    if m == 0:
        return 0.0, 0.0
    
    # Filter: remove dominated lines
    # A line i is dominated if for all t in [0,1], a_i + b_i*t <= a_j + b_j*t for some j
    # This is complex; skip for now since m is small
    
    # Build upper envelope
    # Sort by slope, then by intercept (for same slope, keep max intercept)
    order = np.argsort(b)
    # Remove duplicates (same slope, keep max intercept)
    unique = [order[0]]
    for idx in order[1:]:
        if b[idx] > b[unique[-1]] + 1e-15:
            unique.append(idx)
        elif a[idx] > a[unique[-1]] + 1e-15:
            unique[-1] = idx  # same slope, higher intercept
    
    # Build upper envelope (convex hull of lines in dual)
    # For lines sorted by slope, the upper envelope is found by:
    # envelope = [first line]
    # for each new line: while envelope top has breakpoint > new breakpoint, pop
    envelope = [unique[0]]
    for idx in unique[1:]:
        while len(envelope) >= 2:
            i1 = envelope[-2]
            i2 = envelope[-1]
            # Breakpoint between i1 and i2
            # a_i1 + b_i1*t = a_i2 + b_i2*t → t = (a_i2-a_i1)/(b_i1-b_i2)
            if b[i2] == b[i1]:
                break
            bp12 = (a[i2] - a[i1]) / (b[i1] - b[i2])
            # Breakpoint between i2 and idx
            if b[idx] == b[i2]:
                break
            bp2n = (a[idx] - a[i2]) / (b[i2] - b[idx])
            if bp2n >= bp12:  # new breakpoint is to the right, i2 is not needed
                envelope.pop()
            else:
                break
        envelope.append(idx)
    
    # Now find the minimum of the envelope on [0,1]
    # The envelope is a sequence of lines, each valid on an interval
    # Find where the slope of the envelope crosses 0
    best_t = 0.0
    best_f = np.inf
    for i, idx in enumerate(envelope):
        # Determine valid interval for this line
        if i == 0:
            t_start = 0.0
        else:
            prev = envelope[i-1]
            if b[idx] == b[prev]:
                continue
            t_start = (a[idx] - a[prev]) / (b[prev] - b[idx])
        if i == len(envelope) - 1:
            t_end = 1.0
        else:
            nxt = envelope[i+1]
            if b[nxt] == b[idx]:
                continue
            t_end = (a[nxt] - a[idx]) / (b[idx] - b[nxt])
        
        # Clamp to [0,1]
        t_start = max(t_start, 0.0)
        t_end = min(t_end, 1.0)
        if t_start > t_end:
            continue
        
        # On [t_start, t_end], f(t) = a[idx] + b[idx]*t
        # Minimum at endpoint where derivative pushes towards
        if b[idx] > 0:
            t_opt = t_start
        elif b[idx] < 0:
            t_opt = t_end
        else:
            t_opt = t_start  # constant, either endpoint
        
        f_opt = a[idx] + b[idx] * t_opt
        if f_opt < best_f:
            best_f = f_opt
            best_t = t_opt
    
    return best_t, best_f

def coordinate_descent(h, n_sweeps=1000, time_limit=300):
    n = len(h)
    h = h.astype(float).copy()
    t0 = time.time()
    
    # Compute initial correlations
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = 0
        for i in range(n):
            old = h[i]
            # Determine affected shifts
            # s range: [-i, n-1-i] ∪ [i-n+1, i] = effectively [-min(i,n-1-i) ... ]
            # Actually let's just compute b_s for all s and mask
            s_vals = np.arange(-(n-1), n)  # all possible shifts
            # For each s, compute the coefficient b_s (change in corr_s per unit change in h[i])
            b_s = np.zeros(2*n-1)
            # Term 1: h[i] as h[n] with n=i, shift s
            # Valid if: i in [max(0,-s), min(n-1, n-1-s)] and 0 <= i+s < n
            s_arr = s_vals
            valid_n = (np.maximum(0, -s_arr) <= i) & (i <= np.minimum(n-1, n-1-s_arr))
            valid_is = (i + s_arr >= 0) & (i + s_arr < n)
            mask1 = valid_n & valid_is
            b_s[mask1] += -h[i + s_arr[mask1]]
            
            # Term 2: h[i] as h[n+s] with n=i-s, shift s
            # Valid if: 0 <= i-s < n and i-s in [max(0,-s), min(n-1, n-1-s)]
            valid_is2 = (i - s_arr >= 0) & (i - s_arr < n)
            valid_n2 = (np.maximum(0, -s_arr) <= i - s_arr) & (i - s_arr <= np.minimum(n-1, n-1-s_arr))
            mask2 = valid_is2 & valid_n2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any rounding
    diff = target - np.sum(result)
    if abs(diff) < 1e-10:
        return result
    # Distribute the difference
    idx = np.nonzero((result > 0) & (result < 1))[0]
    if len(idx) == 0:
        idx = np.arange(n)
    result[idx] += diff / len(idx)
    return np.clip(result, 0.0, 1.0)

def solve_1d(a, b):
    """Solve min_{t in [0,1]} max_s (a[s] + b[s]*t). Returns optimal t."""
    # f'_+(t) = max active slope, non-decreasing in t
    def max_slope(t):
        v = a + b * t
        fm = np.max(v)
        active = v >= fm - 1e-9
        return np.max(b[active])
    
    def min_slope(t):
        v = a + b * t
        fm = np.max(v)
        active = v >= fm - 1e-9
        return np.min(b[active])
    
    # Check if min is at 0
    if max_slope(0.0) >= -1e-8:
        return 0.0
    # Check if min is at 1
    if min_slope(1.0) <= 1e-8:
        return 1.0
    
    # Binary search for where max_slope crosses 0
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if max_slope(mid) > 1e-12:
            hi = mid
        else:
            lo = mid
    
    # The optimal t is at the kink. Return the midpoint.
    # But let's refine: the kink is where two specific lines cross.
    t_star = (lo + hi) * 0.5
    
    # Try to find the exact kink
    v = a + b * t_star
    fm = np.max(v)
    active = v >= fm - 1e-7
    if len(np.where(active)[0]) >= 2:
        slopes_active = b[active]
        # Find the pair with min and max slope
        min_s_idx = np.argmin(slopes_active)
        max_s_idx = np.argmax(slopes_active)
        active_idx = np.where(active)[0]
        i1 = active_idx[min_s_idx]
        i2 = active_idx[max_s_idx]
        if abs(b[i2] - b[i1]) > 1e-15:
            t_kink = (a[i2] - a[i1]) / (b[i1] - b[i2])
            if 0.0 <= t_kink <= 1.0:
                # Verify this is indeed the minimum
                f_kink = np.max(a + b * t_kink)
                f_star = np.max(a + b * t_star)
                if f_kink <= f_star + 1e-12:
                    return t_kink
    
    return t_star

def coord_descent(h, max_sweeps=5000, time_limit=100.0, verbose=False):
    n = len(h)
    h = h.astype(float).copy()
    t_start = time.time()
    two_n_minus_1 = 2 * n - 1
    s_vals = np.arange(-(n-1), n)  # shift values
    
    # Precompute validity masks (depend on i, so can't fully precompute)
    # But we can precompute the boundary conditions
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = 0
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        for i in range(n):
            old = h[i]
            
            # Compute b_s: coefficient of (t - old) in each corr_s
            # Term 1: h[i] appears as h[n] (n=i) → contributes -h[i+s] to b_s
            #   Valid if: max(0,-s) <= i <= min(n-1, n-1-s) AND 0 <= i+s < n
            s_arr = s_vals
            # For term 1: i+s in [0,n) and i in valid range for shift s
            i_plus_s = i + s_arr
            cond1 = (i_plus_s >= 0) & (i_plus_s < n)
            # valid range for shift s: max(0,-s) to min(n-1, n-1-s)
            lo_valid = np.maximum(0, -s_arr)
            hi_valid = np.minimum(n-1, n-1-s_arr)
            cond1 &= (lo_valid <= i) & (i <= hi_valid)
            
            # Term 2: h[i] appears as h[n+s] (n=i-s) → contributes (1-h[i-s]) to b_s
            i_minus_s = i - s_arr
            cond2 = (i_minus_s >= 0) & (i_minus_s < n)
            cond2 &= (lo_valid <= i_minus_s) & (i_minus_s <= hi_valid)
            
            b_s = np.zeros(two_n_minus_1)
            b_s[cond1] = -h[i_plus_s[cond1]]
            b_s[cond2] += (1.0 - h[i_minus_s[cond2]])
            
            a_s = corr - old * b_s
            
            t_new = solve_1d(a_s, b_s)
            
            if abs(t_new - old) > 1e-12:
                delta = t_new - old
                h[i] = t_new
                corr += delta * b_s
                improved += 1
        # End coordinate loop
        
        if improved == 0 and sweep > 0:
            break
        if verbose and sweep % 100 == 0:
            c = compute_c5(h)
            print(f"  CD sweep {sweep}: C5={c:.6f}", flush=True)
    
    return h

def upsample_h(h, nn):

        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    hu = np.interp(nx, ox, h)
    return project_c(hu, nn)
