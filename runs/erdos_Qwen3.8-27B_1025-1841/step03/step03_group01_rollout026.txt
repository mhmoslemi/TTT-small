from scipy.fft import fft, ifft, next_fast_len
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_to_set(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2} using bisection"""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(projected)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = projected < 1.0 - 1e-12
        else:
            mask = projected > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            projected[mask] += diff / cnt
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def circular_corr_f(h):
    """Compute circular correlation f_k = dx * sum_i h[i] * (1-h[(i+k)%n])"""
    n = len(h)
    dx = 2.0 / n
    N = next_fast_len(2 * n)
    H = fft(h, N)
    O = fft(1.0 - h, N)
    # Circular cross-correlation: corr[k] = sum_i h[i] * conj(1-h[(i+k)%n])
    # = sum_i h[i] * (1-h[-i+k])... 
    # Actually, let me use the standard formula.
    # The circular cross-correlation of a and b is:
    # (a ⋆ b)[k] = sum_i a[(i+k)%n] * conj(b[i])
    # In FFT: ifft(conj(fft(a)) * fft(b))
    # So: ifft(conj(H) * O) gives sum_i h[(i+k)%n] * (1-h[i])
    # That's not quite what I want.
    # I want: sum_i h[i] * (1-h[(i+k)%n])
    # = sum_i h[i] * conj(1-h[(i+k)%n])  [since real]
    # Let a = h, b = 1-h
    # a ⋆ b [k] = sum_i a[(i+k)%n] * b[i] = sum_i h[(i+k)%n] * (1-h[i])
    # That's the reverse of what I want.
    # b ⋆ a [k] = sum_i b[(i+k)%n] * a[i] = sum_i (1-h[(i+k)%n]) * h[i]  ✓
    # In FFT: ifft(conj(fft(b)) * fft(a)) = ifft(conj(O) * H)
    corr = np.real(ifft(np.conj(O) * H, N))[:n]
    return corr * dx

def compute_gradient(h, T):
    """Compute gradient of smoothed max using circular correlation"""
    n = len(h)
    dx = 2.0 / n
    f_k = circular_corr_f(h)
    # Softmax weights
    f_shifted = f_k - np.max(f_k)
    w = np.exp(T * f_shifted)
    w = w / np.sum(w)
    # Gradient: dx * (1 - (w*h)_j - (w_rev*h)_j)
    N = next_fast_len(2 * n)
    W = fft(w, N)
    H = fft(h, N)
    conv_wh = np.real(ifft(W * H, N))[:n]       # (w*h)_j = sum_k w_k h_{(j-k)%n}
    conv_wh_rev = np.real(ifft(np.conj(W) * H, N))[:n]  # (w_rev*h)_j = sum_k w_k h_{(j+k)%n}
    grad = dx * (1.0 - conv_wh - conv_wh_rev)
    return grad

def fista_optimize(h0, n, T_schedule, max_iter_per_T, time_limit):
    """FISTA with adaptive temperature schedule"""
    start_time = time.time()
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_f = 0.0
    
    for T in T_schedule:
        if time.time() - start_time > time_limit:
            break
        # Reset FISTA state
        y = h.copy()
        t_k = 1.0
        step = 1.0 / T  # Initial step size (1/L where L ~ T for the smoothed function)
        
        for it in range(max_iter_per_T):
            if time.time() - start_time > time_limit:
                break
            grad = compute_gradient(y, T)
            # Gradient step
            y_new = y - step * grad
            y_new = project_to_set(y_new, n)
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = y_new + ((t_k - 1.0) / t_new) * (y_new - h)
            y = project_to_set(y, n)
            h = y_new
            t_k = t_new
            # Adaptive step size (backtracking)
            # For now, use fixed step
            # f_val = compute_c5(h)  # Too expensive to do every iter
            if it % 100 == 0:
                step *= 0.5  # Conservative decay
                if step < 1e-10:
                    break
    
    return h

def circular_cross_correlate(a, b):
    """Circular cross-correlation: (a ⋆ b)[k] = sum_i a[i] * b[(i+k) % n]"""
    n = len(a)
    N = next_fast_len(2 * n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
