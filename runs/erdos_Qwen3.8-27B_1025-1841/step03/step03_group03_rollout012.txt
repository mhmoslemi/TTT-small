def optimize_one_coord(h, j):
    n = len(h)
    shift_offsets = np.arange(2*n-1) - (n-1)  # shifts from -(n-1) to (n-1)
    
    # For each shift s, slope = 1 - h[j+s]_valid - h[j-s]_valid
    js = j + shift_offsets
    jm = j - shift_offsets
    h_plus = np.where((js >= 0) & (js < n), h[js], 0.0)
    h_minus = np.where((jm >= 0) & (jm < n), h[jm], 0.0)
    slopes = 1.0 - h_plus - h_minus
    
    # Current correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    bases = corr - h[j] * slopes
    
    # Find t in [0,1] minimizing max_k (bases[k] + t*slopes[k])
    # Candidates: 0, 1, and intersections
    candidates = [0.0, 1.0]
    
    # Add intersections of top lines
    vals_at_0 = bases
    vals_at_1 = bases + slopes
    top_k0 = np.argsort(-vals_at_0)[:5]
    top_k1 = np.argsort(-vals_at_1)[:5]
    
    for a in top_k0:
        for b in top_k1:
            if a == b:
                continue
            denom = slopes[a] - slopes[b]
            if abs(denom) > 1e-12:
                t = (bases[b] - bases[a]) / denom
                if -0.01 < t < 1.01:
                    candidates.append(t)
    
    # Also add midpoints between top candidates
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        t = np.clip(t, 0.0, 1.0)
        val = np.max(bases + t * slopes)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t

def make_feasible(h, n):

    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
            # Final fix
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-15:
                h[0] += diff2
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):

    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_slopes_bases(h, j):
    """Compute slopes and bases for correlation when varying h[j]."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - (n - 1)
    
    js = j + shifts  # j + shift
    jm = j - shifts  # j - shift
    
    h_plus = np.zeros(n_corr)
    mask_plus = (js >= 0) & (js < n)
    h_plus[mask_plus] = h[js[mask_plus]]
    
    h_minus = np.zeros(n_corr)
    mask_minus = (jm >= 0) & (jm < n)
    h_minus[mask_minus] = h[jm[mask_minus]]
    
    slopes = 1.0 - h_plus - h_minus
    bases = corr - h[j] * slopes
    
    return bases, slopes

def optimize_one_coord(h, j):
    """Find optimal h[j] in [0,1] given other coords fixed."""
    n = len(h)
    bases, slopes = compute_slopes_bases(h, j)
    
    # Evaluate at boundaries
    val_0 = np.max(bases)
    val_1 = np.max(bases + slopes)
    
    # Find candidate interior points
    candidates = [0.0, 1.0]
    
    # Top active at t=0 and t=1
    top0 = np.argsort(-bases)[:8]
    top1 = np.argsort(-(bases + slopes))[:8]
    
    for a in top0:
        for b in top1:
            if a == b:
                continue
            denom = slopes[a] - slopes[b]
            if abs(denom) > 1e-14:
                t = (bases[b] - bases[a]) / denom
                if 0.0 < t < 1.0:
                    candidates.append(t)
    
    # Also add midpoints of top candidates
    best_t = 0.0 if val_0 <= val_1 else 1.0
    best_val = min(val_0, val_1)
    
    for t in candidates:
        t = float(np.clip(t, 0.0, 1.0))
        val = np.max(bases + t * slopes)
        if val < best_val - 1e-15:
            best_val = val
            best_t = t
    
    return best_t

def exact_coord_descent(h, max_rounds=100, time_check=None, start_time=None, budget=1300):
    n = len(h)
    best_c = compute_c5(h)
    
    for rnd in range(max_rounds):

            new_val = optimize_one_coord(h, j)
            if abs(new_val - h[j]) > 1e-12:
                h[j] = new_val
                improved = True
        if not improved:
            break
        c = compute_c5(h)
        if c < best_c:
            best_c = c
    
    return h, best_c

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def optimize_one_coord(h, j):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - (n - 1)
    js = j + shifts
    jm = j - shifts
    h_plus = np.where((js >= 0) & (js < n), h[np.clip(js, 0, n - 1)], 0.0)
    h_minus = np.where((jm >= 0) & (jm < n), h[np.clip(jm, 0, n - 1)], 0.0)
    slopes = 1.0 - h_plus - h_minus
    bases = corr - h[j] * slopes
    val_0 = np.max(bases)
    val_1 = np.max(bases + slopes)
    candidates = [0.0, 1.0]
    top0 = np.argsort(-bases)[:12]
    top1 = np.argsort(-(bases + slopes))[:12]
    for a in top0:
        for b in top1:
            if a == b:
                continue
            denom = slopes[a] - slopes[b]
            if abs(denom) > 1e-14:
                t = (bases[b] - bases[a]) / denom
                if 0.0 < t < 1.0:
                    candidates.append(float(t))
    best_t = 0.0 if val_0 <= val_1 else 1.0
    best_val = min(val_0, val_1)
    for t in candidates:
        val = np.max(bases + t * slopes)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t

def exact_cd(h, max_rounds=50, start_time=None, budget=1300):
    n = len(h)
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if start_time is not None and time.time() - start_time > budget - 40:
            break
        improved = False
        for j in range(n):
            new_val = optimize_one_coord(h, j)
            if abs(new_val - h[j]) > 1e-11:
                h[j] = new_val
                improved = True
        h = make_feasible(h, n)
        if not improved:
            break
        c = compute_c5(h)
        if c < best_c:
            best_c = c
    return h, best_c

def binary_cd(h, max_rounds=30, start_time=None, budget=1300):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = make_feasible(h, n)
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if start_time is not None and time.time() - start_time > budget - 40:
            break
        improved = False
        for j in range(n):
            for val in [0.0, 1.0]:
                if val == h[j]:
                    continue
                trial = h.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-12:
                    h = trial
                    best_c = c
                    improved = True
                    break
        if not improved:
            break
    return h, best_c
