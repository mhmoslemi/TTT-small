def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

Let me now think about the SMO-style approach for the coordinate descent phase:

In SMO (like in SVM), we update two coordinates at a time to maintain the equality constraint. For our problem:
- Change h[i] by +δ and h[j] by -δ (maintains sum)
- Or change h[i] by +δ and h[j] by +ε (if we're adjusting the sum, but we shouldn't need to)

The effect on the correlation:
For each lag m:
Δc_m = δ * [dc_m/dh[i] + dc_m/dh[j]_neg]
where dc_m/dh[i] is as before, and the j term is -δ * dc_m/dh[j]

So Δc_m = δ * (dc_m/dh[i] - dc_m/dh[j])

We want to choose (i, j, δ) that minimizes the max correlation.

This is complex to do exactly, but a greedy approach:
1. Find the current max lag m*
2. For that lag, find the best (i, j) pair to reduce c_m*
3. Make the update

Or even simpler: for the current max lag m*, the "pressure" on each element j is dc_m*/dh[j]. We want to increase elements with negative pressure and decrease elements with positive pressure.

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 960
_G_T = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(val)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _Gfun(x):
    N = _G_N
    T = _G_T
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _Gconj(x):
    return np.sum(x) - _G_N / 2.0

def _Gconj_jac(x):
    return np.ones(_G_N)

def smooth_obj_grad(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exp_vals = np.exp(T * (corr - mc))
    total = np.sum(exp_vals)
    obj = (mc + np.log(total) / T) * dx
    p = exp_vals / total
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        w = p[k]
        if w < 1e-15:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m > 0:
            grad[:n - m] += w * (1.0 - h[m:n])
            grad[m:n] -= w * h[:n - m]
        else:
            r = -m
            grad[r:n] += w * (1.0 - h[:n - r])
            grad[:n - r] -= w * h[r:n]
    grad *= dx
    return obj, grad
