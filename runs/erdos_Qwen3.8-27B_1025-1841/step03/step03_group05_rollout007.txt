def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g
