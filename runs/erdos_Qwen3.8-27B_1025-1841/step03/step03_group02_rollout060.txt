```python
def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            # Find any element
            if diff > 0:
                h[h.argmin()] += diff
            else:
                h[h.argmax()] += diff
            break
        cnt = np.sum(mask)
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_exact_binary(h, n):

    b = (h >= 0.5).astype(np.float64)
    target = n // 2
    s = int(np.sum(b))
    if s > target:
        # Remove (s - target) ones with lowest h values
        ones_idx = np.where(b == 1)[0]
        order = np.argsort(h[ones_idx])
        for k in range(s - target):
            b[ones_idx[order[k]]] = 0
    elif s < target:
        zeros_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for k in range(target - s):
            b[zeros_idx[order[k]]] = 1
    return b
```

The key insight: for the max overlap at shift k*, we want to reduce it. The overlap at shift k (for k>0) is:

- ov(k*) might increase by 1 if b[j+k*]=0 (from the second flip, since now b[j]=1 and b[j+k*]=0)
- Net change in ov(k*): -1 + (1 if b[j+k*]=0 else 0) = -1 or 0

```python
        # corr[k] = sum_i x[i] * (1-x[i+k]) for i valid
        # d(corr[k_max])/dx[j] = (1-x[j-k_max]) if j-k_max is valid (from x[j] term)
        #                       + (-x[j+k_max]) if j+k_max is valid (from (1-x[j]) term)
        #                       + (1-x[j-k_max])...

        # c[k] = sum_i x[i] * (1 - x[i + k - (n-1)])

        # Valid: 0 ≤ i < n and 0 ≤ i+m < n
        # i.e., max(0, -m) ≤ i ≤ min(n-1, n-1-m)
        
        # d(c[k])/dx[j]:
        # Term 1: x[j] appears when i=j and j+m is valid: contributes (1-x[j+m])
        # Term 2: (1-x[j]) appears when i+m=j, i.e., i=j-m and j-m is valid: contributes -x[j-m]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_N_SL = 0
_T_SL = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    if s > target:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        for k in range(s - target):
            b[ones[order[k]]] = 0
    elif s < target:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for k in range(target - s):
            b[zeros[order[k]]] = 1
    return b

def targeted_sa(b, n, max_iters, T0, T1, rng, start_time, budget_s):
    dx = 2.0 / n
    corr = np.correlate(b, 1.0 - b, mode='full')
    curr_c5 = float(np.max(corr) * dx)
    best_c5 = curr_c5
    best_b = b.copy()
    ones = np.where(b == 1)[0].copy()
    zeros = np.where(b == 0)[0].copy()
    
    it = 0
    no_improve = 0
    while it < max_iters:
        if time.time() - start_time > budget_s - 30:
            break
        it += 1
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        
        # Every 200 iters, refresh ones/zeros
        if it % 200 == 0:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
        
        # Choose move
        if rng.random() < 0.7:
            # Targeted: find worst lag
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            if k_star > 0 and len(ones) > 0:
                # Find i in ones where i+k_star is valid and b[i+k_star]==0
                valid = (ones + k_star < n)
                if np.any(valid):
                    candidates = ones[valid]
                    # Further filter: b[i+k_star]==0
                    m = (b[candidates + k_star] == 0)
                    if np.any(m):
                        i = candidates[m[rng.randint(np.sum(m))]]
                    else:
                        i = candidates[rng.randint(len(candidates))]
                else:
                    i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            elif k_star < 0:
                kk = -k_star
                # overlap at -kk: for i where b[i+kk]=1 and b[i]=0
                # Flip b[i+kk] 1→0 and b[i] 0→1
                valid_zeros = zeros[zeros + kk < n]
                if len(valid_zeros) > 0:
                    m = (b[valid_zeros + kk] == 1)
                    if np.any(m):
                        i_idx = valid_zeros[m[rng.randint(np.sum(m))]]
                    else:
                        i_idx = valid_zeros[rng.randint(len(valid_zeros))]
                    # We flip b[i_idx+kk] 1→0 and set b[i_idx] 0→1
                    i = i_idx + kk  # the 1→0 position
                    j = i_idx       # the 0→1 position
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Perform swap
        b[i] = 0.0
        b[j] = 1.0
        
        new_corr = np.correlate(b, 1.0 - b, mode='full')
        new_c5 = float(np.max(new_corr) * dx)
        
        if new_c5 <= curr_c5 or (T > 1e-15 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            corr = new_corr
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            b[i] = 1.0
            b[j] = 0.0
            no_improve += 1
        
        # Restart mechanism
        if no_improve > 5000 and T > 0.001:
            # Accept a worse solution to escape
            b = best_b.copy()
            corr = np.correlate(b, 1.0 - b, mode='full')
            curr_c5 = float(np.max(corr) * dx)
            no_improve = 0
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
    
    return best_b, best_c5

def slsqp_obj(x):
    n = _N_SL
    T = _T_SL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _N_SL, _T_SL
    _N_SL = n
    _T_SL = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    res = minimize(slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 20:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 20:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            # Try 0, 1, and small perturbations
            for cand in (0.0, 1.0):
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    if s > target:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        for k in range(s - target):
            b[ones[order[k]]] = 0
    elif s < target:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for k in range(target - s):
            b[zeros[order[k]]] = 1
    return b

def targeted_sa(b, n, max_iters, T0, T1, rng, start_time, budget_s, targeted_frac=0.7):

def slsqp_obj(x):
    n = _N_SL
    T = _T_SL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _N_SL, _T_SL
    _N_SL = n
    _T_SL = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    res = minimize(slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x
```
