    res = minimize(objective, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

def systematic_binary_search(h_bin, n, rng, max_sweeps=60):
    h = h_bin.copy()
    curr, _ = _fft_corr_max(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for sweep in range(max_sweeps):
        improved = False
        # Try all single flips
        for oi in ones[:]:
            h[oi] = 0.0
            c, _ = _fft_corr_max(h)
            if c < curr - 1e-14:
                curr = c
                ones.remove(oi)
                zeros.append(oi)
                improved = True
            else:
                h[oi] = 1.0
        for oj in zeros[:]:
            h[oj] = 1.0
            c, _ = _fft_corr_max(h)
            if c < curr - 1e-14:
                curr = c
                zeros.remove(oj)
                ones.append(oj)
                improved = True
            else:
                h[oj] = 0.0
        # Try pair flips
        n_pairs = min(3000, len(ones) * len(zeros))
        for _ in range(n_pairs):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c, _ = _fft_corr_max(h)
            if c < curr - 1e-14:
                curr = c
                if ii < len(ones):
                    ones[ii] = oj
                if jj < len(zeros):
                    zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, curr

def simulated_annealing_binary(h_bin, n, rng, max_iters=5000, T_start=0.3, T_end=0.01):
    h = h_bin.copy()
    curr, _ = _fft_corr_max(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    best_h = h.copy()
    best_c = curr
    alpha = (T_end / T_start) ** (1.0 / max_iters)
    for it in range(max_iters):
        T = T_start * (alpha ** it)
        if rng.rand() < 0.5:
            i = rng.randint(len(ones))
            oi = ones[i]
            h[oi] = 0.0
            c, _ = _fft_corr_max(h)
            if c < curr or rng.rand() < np.exp(-(c - curr) / (T * 0.1)):
                curr = c
                ones = np.delete(ones, i)
                zeros = np.append(zeros, oi)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[oi] = 1.0
        else:
            j = rng.randint(len(zeros))
            oj = zeros[j]
            h[oj] = 1.0
            c, _ = _fft_corr_max(h)
            if c < curr or rng.rand() < np.exp(-(c - curr) / (T * 0.1)):
                curr = c
                zeros = np.delete(zeros, j)
                ones = np.append(ones, oj)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[oj] = 0.0
    return best_h, best_c

def coordinate_descent(h_start, n, max_rounds=12):
    h = h_start.copy()
    curr, _ = _fft_corr_max(h)
    dx = 2.0 / n
    deltas = [0.05, -0.05, 0.025, -0.025, 0.01, -0.01, 0.005, -0.005]
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n)
                c, _ = _fft_corr_max(tr)
                if c < curr - 1e-14:
                    curr = c
                    h = tr
                    ov = nv
                    improved = True
                    break
            if h[j] > 0.01 and h[j] < 0.99:
                for bv in (0.0, 1.0):
                    tr = h.copy()
                    tr[j] = bv
                    tr = make_feasible(tr, n)
                    c, _ = _fft_corr_max(tr)
                    if c < curr - 1e-14:
                        curr = c
                        h = tr
                        improved = True
        if not improved:
            break
    return h, curr

def slsqp_polish(h_start, n, T, max_iter=100):
    dx = 2.0 / n
    n_full = 2 * n - 1
    size = 1 << (n_full - 1).bit_length()

    def obj(h):
        omh = 1.0 - h
        H = np.fft.rfft(h, size)
        K = np.fft.rfft(omh, size)
        corr = np.fft.irfft(np.conj(H) * K, size)[:n_full]
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -70, 70)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

    def con(h):
        return np.sum(h) - n / 2.0

    def jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
    return make_feasible(res.x, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    n0 = len(h0)

    # Phase 1: Smooth optimization at parent resolution
    for T in [30, 80, 200, 500, 1200, 3000]:
        if tl() < 30:
            break
        h_init = upsample(best_h, n0)
        if T > 200:
            h_init = h_init + 0.1 * rng.randn(n0)
            h_init = make_feasible(h_init, n0)
        try:
            h_opt = lbgfb_optimize(h_init, n0, T, mu=800.0, max_iter=400)
            c, _ = _fft_corr_max(h_opt)
            c = compute_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 2: Systematic binary local search
    if tl() > 80:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.round(np.sum(h_bin)))
        n_target = n0 // 2
        if n_ones > n_target:
            one_idx = np.where(h_bin == 1)[0]
            rng.shuffle(one_idx)
            for k in range(n_ones - n_target):
                h_bin[one_idx[k]] = 0.0
        elif n_ones < n_target:
            zero_idx = np.where(h_bin == 0)[0]
            rng.shuffle(zero_idx)
            for k in range(n_target - n_ones):
                h_bin[zero_idx[k]] = 1.0
        c5b, _ = _fft_corr_max(h_bin)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_bin.copy()

        h_bs, c5_bs = systematic_binary_search(h_bin, n0, rng, max_sweeps=50)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()

    # Phase 3: Simulated annealing for binary
    if tl() > 120:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.round(np.sum(h_bin)))
        n_target = n0 // 2
        if n_ones != n_target:
            if n_ones > n_target:
                one_idx = np.where(h_bin == 1)[0]
                for k in range(n_ones - n_target):
                    h_bin[one_idx[k]] = 0.0
            else:
                zero_idx = np.where(h_bin == 0)[0]
                for k in range(n_target - n_ones):
                    h_bin[zero_idx[k]] = 1.0
        for sa_round in range(3):
            if tl() < 60:
                break
            h_sa, c5_sa = simulated_annealing_binary(h_bin, n0, rng,
                                                     max_iters=3000, T_start=0.4, T_end=0.02)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h_bin = h_sa.copy()

        # Additional systematic search after SA
        if tl() > 40:
            h_bs2, c5_bs2 = systematic_binary_search(h_bin, n0, rng, max_sweeps=40)
            if c5_bs2 < best_c5:
                best_c5 = c5_bs2
                best_h = h_bs2.copy()

    # Phase 4: Upsample to higher resolution and refine
    for n_up in [2 * n0, 3 * n0]:
        if tl() < 80:
            break
        h_c = upsample(best_h, n_up)
        c5c, _ = _fft_corr_max(h_c)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h_c.copy()

        for T in [50, 200, 800, 2000]:
            if tl() < 30:
                break
            try:
                h_opt = lbgfb_optimize(h_c, n_up, T, mu=500.0, max_iter=300)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_c = h_opt.copy()
            except Exception:
                pass

    # Phase 5: Coordinate descent polish at best resolution
    n_cd = len(best_h)
    h_cd = best_h.copy()
    if tl() > 60:
        h_cd, c5_cd = coordinate_descent(h_cd, n_cd, max_rounds=10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Final SLSQP squeeze
    if tl() > 30:
        n_f = min(len(best_h), 1280)
        h_f = upsample(best_h, n_f) if n_f < len(best_h) else best_h.copy()
        for T in [500, 3000, 8000]:
            if tl() < 15:
                break
            try:
                h_opt = slsqp_polish(h_f, n_f, T, max_iter=80)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_f = h_opt.copy()
            except Exception:
                pass

    # Final coordinate descent
    if tl() > 20:
        h_f2, c5_f2 = coordinate_descent(best_h, len(best_h), max_rounds=6)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
