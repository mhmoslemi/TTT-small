from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-14:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista(h, n_iter, T0, T1, lr, check, top_k=100):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.3 * f)
        g = sm_grad(z, T, top_k)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def slsqp(h, T=5.0, mi=600):
    n = len(h)
    dx = 2.0 / n
    
    def sobj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx
    
    def sobj_grad(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        exps = np.exp(T * (c - mc))
        w = exps / np.sum(exps)
        g = np.zeros(n)
        top = 50
        idx = np.argsort(w)[::-1][:top]
        for i in idx:
            if w[i] < 1e-14:
                break
            g += w[i] * _subgrad_one(hv, i - (n-1))
        return g * dx
    
    def scon(hv):
        return np.sum(hv) - n / 2.0
    
    def sjac(hv):
        return np.ones(n)
    
    try:
        r = minimize(sobj, h, method='SLSQP',
                     jac=sobj_grad,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': scon, 'jac': sjac}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            return hx
    except Exception:
        pass
    return h

        g[:n - s] -= h[s:]

                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def slsqp_optimize(h, T=5.0, maxiter=400):
    n = len(h)
    dx = 2.0 / n

    def obj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

    def obj_grad(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        exps = np.exp(T * (c - mc))
        w = exps / np.sum(exps)
        g = np.zeros(n)
        top = 60
        idx = np.argsort(w)[::-1][:top]
        for i in idx:
            if w[i] < 1e-14:
                break
            g += w[i] * _subgrad_one(hv, i - (n - 1))
        return g * dx

    def con_fun(hv):
        return np.sum(hv) - n / 2.0

    def con_jac(hv):
        return np.ones(n)

    try:
        r = minimize(obj, h, method='SLSQP', jac=obj_grad,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        if r.x is not None:
            return project_c(r.x, n)
    except Exception:
        pass
    return h

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].astype(float)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = project_c(rng.rand(960), 960)

    n0 = len(h0)
    bh = project_c(h0, n0)
    bc = compute_c5(bh)
    print(f"I: n={n0} C5={bc:.6f}", flush=True)

    # Phase 1: FISTA continuation with multiple temperature stages
    stages = [
        (60.0, 2.0, 0.15, 6000, 100),
        (15.0, 0.5, 0.08, 6000, 80),
        (6.0, 0.1, 0.04, 5000, 60),
        (3.0, 0.03, 0.015, 5000, 40),
        (1.5, 0.01, 0.006, 4000, 25),
    ]

    for si, (T0v, T1v, lr_v, nit, tk) in enumerate(stages):
        if budget_s - (time.time() - t0) < 50:
            break
        h_res, c_res = fista(bh, nit, T0v, T1v, lr_v, 60, tk)
        if c_res < bc:
            bc, bh = c_res, h_res.copy()
        print(f"S{si}: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 2: Aggressive SLSQP at current resolution
    for T in [12.0, 6.0, 3.0, 1.5, 0.8, 0.4]:
        if budget_s - (time.time() - t0) < 25:
            break
        hx = slsqp_optimize(bh, T, 500)
        cx = compute_c5(hx)
        if cx < bc:
            bc, bh = cx, hx.copy()
        print(f"SLQ2 T={T}: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 3: FISTA refinement
    if budget_s - (time.time() - t0) > 50:
        h_res, c_res = fista(bh, 5000, 4.0, 0.02, 0.012, 40, 30)
        if c_res < bc:
            bc, bh = c_res, h_res.copy()
    if budget_s - (time.time() - t0) > 25:
        h_res, c_res = fista(bh, 3000, 1.0, 0.005, 0.004, 30, 20)
        if c_res < bc:
            bc, bh = c_res, h_res.copy()
    print(f"F3: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 4: SLSQP again after FISTA
    for T in [2.0, 1.0, 0.5, 0.3]:
        if budget_s - (time.time() - t0) < 20:
            break
        hx = slsqp_optimize(bh, T, 400)
        cx = compute_c5(hx)
        if cx < bc:
            bc, bh = cx, hx.copy()
    print(f"SLQ4: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 5: Diverse perturbation restarts with SLSQP polish
    np_ = len(bh)
    n_tries = 25
    for trial in range(n_tries):
        if budget_s - (time.time() - t0) < 30:
            break
        # Different perturbation strategies
        if trial % 3 == 0:
            # Small random perturbation
            mask = rng.rand(np_) < 0.15
            hp = bh + rng.randn(np_) * 0.03 * mask
        elif trial % 3 == 1:
            # Block perturbation
            blk = np_ // 4
            start = int(rng.rand() * (np_ - blk))
            hp = bh.copy()
            hp[start:start+blk] += rng.randn(blk) * 0.08
        else:
            # Thue-Morse blend
            tm = tm_seq(np_)
            alpha = 0.05 + 0.03 * (trial % 5)
            hp = (1 - alpha) * bh + alpha * tm
        hp = project_c(hp, np_)
        hr, cr = fista(hp, 2000, 8.0, 0.05, 0.04, 40, 50)
        if cr < bc:
            if budget_s - (time.time() - t0) > 15:
                hr2 = slsqp_optimize(hr, 1.0, 250)
                cr2 = compute_c5(hr2)
                if cr2 < bc:
                    bc, bh = cr2, hr2.copy()
                else:
                    bc, bh = cr, hr.copy()
            else:
                bc, bh = cr, hr.copy()
            print(f"  P{trial}: C5={bc:.6f}", flush=True)
    print(f"P5: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 6: Try at reduced resolution (different basin)
    n_reduced = max(256, np_ // 2)
    if n_reduced < np_ and budget_s - (time.time() - t0) > 120:
        h_red = upsample(bh, n_reduced)
        for T in [5.0, 1.5, 0.5]:
            if budget_s - (time.time() - t0) < 20:
                break
            hr = slsqp_optimize(h_red, T, 400)
            cr = compute_c5(hr)
            if cr < compute_c5(h_red):
                h_red = hr
        # Upsample back
        h_back = upsample(h_red, np_)
        hb, cb = fista(h_back, 3000, 5.0, 0.03, 0.02, 40, 40)
        if cb < bc:
            hb2 = slsqp_optimize(hb, 0.5, 300)
            cb2 = compute_c5(hb2)
            if cb2 < bc:
                bc, bh = cb2, hb2.copy()
            else:
                bc, bh = cb, hb.copy()
            print(f"  RED: C5={bc:.6f}", flush=True)
    print(f"P6: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 7: Final aggressive polish
    if budget_s - (time.time() - t0) > 20:
        bh = slsqp_optimize(bh, 0.5, 400)
        bc = compute_c5(bh)
    if budget_s - (time.time() - t0) > 10:
        hf, cf = fista(bh, 3000, 2.0, 0.005, 0.004, 30, 20)
        if cf < bc:
            bc, bh = cf, hf.copy()
    if budget_s - (time.time() - t0) > 10:
        bh = slsqp_optimize(bh, 0.3, 300)
        bc = compute_c5(bh)

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
